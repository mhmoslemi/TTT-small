"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- FISTA with log-sum-exp smoothing (continuation on T)
- Adaptive peak-shaving coordinate descent
- Multi-length, multi-start exploration
- Dual (Hedge) methods for weight adaptation
"""

import numpy as np
from scipy.signal import fftconvolve
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_objective(a, T):
    """Compute log-sum-exp smoothed max of conv(a,a)."""
    c = fast_convolve(a)
    c_max = c.max()
    # For numerical stability
    shifted = T * (c - c_max)
    # Clip for stability
    shifted = np.clip(shifted, -500, 500)
    log_sum = c_max + np.log(np.sum(np.exp(shifted))) / T
    return log_sum

def smooth_gradient(a, T):
    """
    Compute gradient of the log-sum-exp smoothed max of conv(a,a).
    grad[i] = 2 * sum_{k} p[k] * a[k-i] * 1[0 <= k-i < n]
    which equals 2 * np.correlate(p, a, mode='valid')
    """
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.ones(len(c)) / len(c)
    else:
        p = p / p_sum
    # grad[i] = 2 * sum_m a[m] * p[i+m] = 2 * correlate(p, a, 'valid')[i]
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return grad

def fista_smoothed(a0, time_limit, T_init=1.0, T_max=50.0, T_mult=1.3,
                    max_iter_per_T=50, verbose=False):
    """
    FISTA with log-sum-exp smoothing and continuation on T.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    T = T_init
    
    while time.monotonic() - t0 < time_limit - 0.1:
        # Run FISTA for current T
        x = a.copy()
        x_prev = a.copy()
        t_val = 1.0
        lr = None  # Will be set by backtracking
        
        for it in range(max_iter_per_T):
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            
            # FISTA momentum
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            if t_val > 1:
                momentum = (t_val - 1.0) / t_new
            else:
                momentum = 0.0
            y = x + momentum * (x - x_prev)
            
            # Compute gradient at y
            grad = smooth_gradient(y, T)
            
            # Backtracking line search
            if lr is None:
                lr = 0.01
            else:
                lr *= 2.0
            
            accepted = False
            for _ in range(20):
                x_new = project_simplex(y - lr * grad)
                # Check descent condition
                obj_new = smooth_objective(x_new, T)
                obj_y = smooth_objective(y, T)
                grad_lip = np.sum(grad * (x_new - y)) + 0.5 * (1.0 / lr) * np.sum((x_new - y) ** 2)
                if obj_new <= obj_y + grad_lip:
                    accepted = True
                    break
                lr *= 0.5
            
            if not accepted:
                lr = max(lr, 1e-10)
                x_new = project_simplex(y - lr * grad)
            
            x_prev = x
            x = x_new
            t_val = t_new
            
            # Check improvement on the true objective
            true_obj = fast_convolve(x).max()
            if true_obj < best_obj - 1e-14:
                best_obj = true_obj
                best_a = x.copy()
        
        # Update a for next T
        a = x.copy()
        
        # Increase T
        T = T_mult * T
        if T > T_max:
            T = T_max
            break
        
        if verbose:
            print(f"    FISTA T={T:.2f}, obj={best_obj:.10f}")
    
    return best_a, best_obj

def peak_shave(a0, time_limit, max_peaks=5):
    """
    Peak shaving: iteratively reduce the top peaks of conv(a,a)
    by redistributing mass.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    step = 0.003
    
    while time.monotonic() - t0 < time_limit - 0.05:
        conv = fast_convolve(a)
        # Get top peaks
        peak_indices = np.argsort(conv)[::-1][:max_peaks]
        
        improved = False
        for k in peak_indices:
            k = int(k)
            # Find contributing pairs
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            if js > je:
                continue
            
            # Compute contribution of each j
            contributions = np.array([a[j] * a[k - j] for j in range(js, je + 1)])
            
            if contributions.max() < 1e-15:
                continue
            
            # Top contributor
            top_j = js + int(np.argmax(contributions))
            top_val = contributions.max()
            
            # Try to move mass from top_j to a location that doesn't contribute to this peak
            candidates = []
            for j_target in range(n):
                if j_target == top_j or j_target == k - top_j:
                    continue
                # Check if moving to j_target would increase other peaks
                # Simple heuristic: prefer locations far from all peaks
                dist_to_peaks = min(abs(j_target - pk) for pk in peak_indices) + min(abs(n - 1 - j_target - (pk - (n-1))) for pk in peak_indices)
                candidates.append((dist_to_peaks, j_target))
            
            candidates.sort(reverse=True)
            
            for (_, j_target) in candidates[:5]:
                for frac in [step, step * 0.5, step * 2.0]:
                    delta = a[top_j] * frac
                    if delta < 1e-12:
                        continue
                    candidate = a.copy()
                    candidate[top_j] -= delta
                    candidate[j_target] += delta
                    if candidate[top_j] < -1e-12:
                        continue
                    candidate = np.maximum(candidate, 0)
                    csum = candidate.sum()
                    if csum < 1e-15:
                        continue
                    candidate = candidate / csum
                    val = fast_convolve(candidate).max()
                    if val < best_obj - 1e-15:
                        a = candidate
                        best_obj = val
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if improved:
                break
        
        if not improved:
            step *= 0.95
            if step < 1e-7:
                break
            # Occasional perturbation
            if int((time.monotonic() - t0) * 10) % 50 == 0:
                candidate = project_simplex(a + rng.standard_normal(n) * 0.002)
                val = fast_convolve(candidate).max()
                if val < best_obj:
                    a = candidate
                    best_a = a.copy()
                    best_obj = val
    
    return best_a, best_obj

def hedge_optimize(a0, time_limit, T_init=5.0, lr_init=0.05, verbose=False):
    """
    Hedge (exponentiated gradient) method for the min-max problem.
    Alternates between:
    - Updating dual weights p (Hedge update on peaks)
    - Solving the primal subproblem (quadratic minimization)
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    # Initialize dual weights uniformly
    conv = fast_convolve(a)
    K = len(conv)
    p = np.ones(K) / K
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    T = T_init
    
    while time.monotonic() - t0 < time_limit - 0.05:
        # Update p (Hedge)
        conv = fast_convolve(a)
        p_new = p * np.exp(T * conv / p.sum() if p.sum() > 0 else T * conv)
        # Actually, proper Hedge: p_k *= exp(T * loss_k)
        # where loss_k = conv(a)[k]
        p_new = p * np.exp(T * conv)
        p_sum = p_new.sum()
        if p_sum > 0 and np.isfinite(p_sum):
            p = p_new / p_sum
        else:
            # Reset
            c_max = conv.max()
            p = np.exp(T * (conv - c_max))
            p = p / p.sum()
        
        # Primal step: minimize sum_k p_k * conv(a)[k] subject to simplex
        # This is min a^T (sum_k p_k B_k) a s.t. a >= 0, sum(a) = 1
        # The gradient of the weighted objective is:
        # grad[i] = 2 * sum_k p_k * a[k-i] * 1[0 <= k-i < n]
        # = 2 * correlate(p, a, 'valid')[i]
        # 
        # For the QP, we can use gradient descent with projection
        # Or use the fact that for the unconstrained problem, the optimum
        # satisfies 2 * Q * a = nu * 1 where Q = sum_k p_k B_k
        
        # Use a few steps of projected gradient descent on the QP
        for _ in range(3):
            grad = 2.0 * np.correlate(p, a, mode='valid')
            # The "Lipschitz constant" of the QP objective is about 2 * max eigenvalue of Q
            # Approximate with 2 * sum(p) * n = 2n (rough upper bound)
            lr_qp = lr_init / max(1.0, n * 0.01)
            a_new = project_simplex(a - lr_qp * grad)
            # Check if it improved the true objective
            val_new = fast_convolve(a_new).max()
            if val_new < fast_convolve(a).max() + 1e-10:
                a = a_new
        
        # Track best
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
        
        # Adapt T
        if (int((time.monotonic() - t0) * 100) % 10 == 0):
            T *= 1.1
    
    return best_a, best_obj

def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s

def generate_starts(n, parent, rng):
    """Generate diverse initializations."""
    starts = []
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    
    # Parent (resampled)
    if parent is not None:
        starts.append(resample_sequence(parent, n))
    
    # Uniform
    starts.append(np.ones(n) / n)
    
    # Cosine taper
    cos_t = np.cos(np.pi * x / (n - 1))
    w = 0.5 * (1 + cos_t)
    w = np.maximum(w, 0.01)
    starts.append(w / w.sum())
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - mid) / (0.15 * n)) ** 2)
    starts.append(g / g.sum())
    
    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    starts.append(edge / edge.sum())
    
    # Broad gaussian
    g2 = np.exp(-0.5 * ((x - mid) / (0.4 * n)) ** 2)
    starts.append(g2 / g2.sum())
    
    # Two peaks
    g1 = np.exp(-0.5 * ((x - 0.3 * (n-1)) / (0.08 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.7 * (n-1)) / (0.08 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())
    
    # S-curve
    s_curve = 1.0 / (1.0 + np.exp(-10 * (x / (n-1) - 0.5)))
    starts.append(s_curve / s_curve.sum())
    
    # Random Dirichlet (2 variants)
    starts.append(rng.dirichlet(np.ones(n) * 0.5))
    starts.append(rng.dirichlet(np.ones(n) * 0.2))
    
    return starts

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- FISTA with log-sum-exp smoothing and continuation on T
- Hedge (exponentiated gradient) dual method
- Adaptive peak-shaving coordinate descent
- Multi-length, multi-start exploration
- Polyak step-size subgradient as final polish
"""

import numpy as np
from scipy.signal import fftconvolve
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_value_and_weights(a, T):
    """
    Compute the log-sum-exp smoothed value and the softmax weights.
    Returns (smoothed_value, weights_p).
    """
    c = fast_convolve(a)
    c_max = c.max()
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    log_sum_exp = c_max + np.log(exp_shifted.sum()) / T
    p = exp_shifted / exp_shifted.sum()
    return log_sum_exp, p


def smooth_gradient_from_weights(a, p):
    """
    Compute gradient given pre-computed weights p.
    grad[i] = 2 * sum_m a[m] * p[i+m] = 2 * correlate(p, a, 'valid')[i]
    """
    return 2.0 * np.correlate(p, a, mode='valid')


def fista_smoothed(a0, time_limit, T_init=2.0, T_max=60.0, T_mult=1.35,
                   max_iter_per_T=100, verbose=False):
    """
    FISTA with log-sum-exp smoothing and geometric continuation on T.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    T = T_init

    while time.monotonic() - t0 < time_limit - 0.1:
        # FISTA inner loop for current T
        x = a.copy()
        x_prev = a.copy()
        t_val = 1.0
        lr = 1.0 / (n * T)  # Initial step size estimate

        for it in range(max_iter_per_T):
            if time.monotonic() - t0 > time_limit - 0.1:
                break

            # FISTA momentum
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            if t_val > 1.0:
                momentum = (t_val - 1.0) / t_new
            else:
                momentum = 0.0
            y = x + momentum * (x - x_prev)
            y = np.maximum(y, 0.0)  # Keep in non-negative orthant for stability

            # Compute gradient at y
            val_y, p_y = smooth_value_and_weights(y, T)
            grad_y = smooth_gradient_from_weights(y, p_y)

            # Backtracking line search
            accepted = False
            for _bt in range(15):
                x_new = project_simplex(y - lr * grad_y)
                val_new, _ = smooth_value_and_weights(x_new, T)
                # Descent lemma condition
                rhs = val_y + np.dot(grad_y, x_new - y) + (0.5 / lr) * np.dot(x_new - y, x_new - y)
                if val_new <= rhs + 1e-12:
                    accepted = True
                    break
                lr *= 0.5

            if not accepted:
                x_new = project_simplex(y - lr * grad_y)

            x_prev = x
            x = x_new
            t_val = t_new

            # Update lr for next iteration (increase if we had to backtrack)
            if accepted and _bt > 3:
                lr = min(lr * 1.5, 1.0)
            elif accepted:
                lr = min(lr * 1.1, 1.0)

            # Track best on true objective
            true_obj = fast_convolve(x).max()
            if true_obj < best_obj - 1e-14:
                best_obj = true_obj
                best_a = x.copy()

        # Move to next T
        a = x.copy()
        T = T_mult * T
        if T > T_max:
            T = T_max
            # Do a few more iterations at max T
            x = a.copy()
            x_prev = a.copy()
            t_val = 1.0
            for it in range(50):
                if time.monotonic() - t0 > time_limit - 0.1:
                    break
                t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
                momentum = (t_val - 1.0) / t_new if t_val > 1.0 else 0.0
                y = x + momentum * (x - x_prev)
                y = np.maximum(y, 0.0)
                val_y, p_y = smooth_value_and_weights(y, T)
                grad_y = smooth_gradient_from_weights(y, p_y)
                x_new = project_simplex(y - lr * grad_y)
                x_prev = x
                x = x_new
                t_val = t_new
                true_obj = fast_convolve(x).max()
                if true_obj < best_obj - 1e-14:
                    best_obj = true_obj
                    best_a = x.copy()
            break

        if verbose and (time.monotonic() - t0) % 5 < 0.1:
            print(f"    FISTA T={T:.1f}, obj={best_obj:.10f}")

    return best_a, best_obj


def hedge_primal(a0, time_limit, T_init=5.0, inner_steps=5, verbose=False):
    """
    Hedge method: alternate between dual weight updates and primal QP steps.
    The primal step minimizes the weighted sum of convolution terms.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    T = T_init
    lr = 1.0 / (n * 0.5)
    stagnation = 0

    it = 0
    while time.monotonic() - t0 < time_limit - 0.05:
        it += 1
        if it % 200 == 0 and verbose:
            print(f"    Hedge it={it}, obj={best_obj:.10f}")

        # Update dual weights (Hedge / multiplicative weights)
        conv = fast_convolve(a)
        conv_shifted = conv - conv.max()
        p = np.exp(T * conv_shifted)
        p_sum = p.sum()
        if p_sum < 1e-300 or not np.isfinite(p_sum):
            p = np.ones(len(conv)) / len(conv)
        else:
            p = p / p_sum

        # Primal step: minimize sum_k p_k * conv(a, a)_k via projected gradient
        for _ in range(inner_steps):
            grad = smooth_gradient_from_weights(a, p)
            a_new = project_simplex(a - lr * grad)
            # Simple acceptance: accept if true objective doesn't worsen too much
            obj_new = fast_convolve(a_new).max()
            obj_cur = fast_convolve(a).max()
            if obj_new <= obj_cur + 1e-12:
                a = a_new
            else:
                # Try half step
                a_half = project_simplex(a - (lr / 2) * grad)
                obj_half = fast_convolve(a_half).max()
                if obj_half <= obj_cur + 1e-12:
                    a = a_half
                    lr *= 0.7
                else:
                    lr *= 0.5
                    break

        # Track best
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Adapt parameters
        if stagnation > 50:
            T *= 1.05
            lr *= 0.95
        if stagnation > 200:
            # Perturb
            rng = np.random.default_rng(it * 7919)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init * 2
            lr = 1.0 / (n * 0.5)
            stagnation = 0

    return best_a, best_obj


def polyak_subgradient(a0, time_limit, top_k=10):
    """
    Subgradient method with Polyak step size.
    Uses the top-k peaks for the subgradient.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    M_ub = best_obj  # Upper bound on optimal value

    t0 = time.monotonic()
    accum_grad = np.zeros(n)
    accum_count = 0
    avg_window = 20

    while time.monotonic() - t0 < time_limit - 0.05:
        conv = fast_convolve(a)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            M_ub = min(M_ub, cur_obj)

        # Compute subgradient: weighted average of top-k peak gradients
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peak_indices]
        pv_max = pv.max()
        w = np.exp((pv - pv_max) * 5.0)
        w = w / w.sum()

        grad = np.zeros(n)
        for i in range(len(peak_indices)):
            if w[i] < 1e-15:
                continue
            k = int(peak_indices[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += w[i] * 2.0 * a[k - idx_arr]

        # Polyak step size
        gap = cur_obj - M_ub * 0.999  # Small safety margin
        if gap < 1e-14:
            gap = 1e-6
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break
        lr = gap / grad_norm_sq
        lr = min(lr, 0.1)  # Cap step size

        a_new = project_simplex(a - lr * grad)
        a = a_new

    return best_a, best_obj


def peak_shave(a0, time_limit, max_peaks=5, initial_step=0.004):
    """
    Peak shaving: directly target top peaks by redistributing mass.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    step = initial_step
    no_improve = 0

    while time.monotonic() - t0 < time_limit - 0.05:
        conv = fast_convolve(a)
        peak_indices = np.argsort(conv)[::-1][:max_peaks]

        improved = False
        for ki in range(len(peak_indices)):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue

            # Find top contributing pair
            best_contrib = 0.0
            best_j = js
            for j in range(js, je + 1):
                c = a[j] * a[k - j]
                if c > best_contrib:
                    best_contrib = c
                    best_j = j

            if best_contrib < 1e-12:
                continue

            # Try moving mass from best_j to various targets
            # Prefer targets that are "far" from all peaks
            peak_set = set(int(pk) for pk in peak_indices)
            candidates = []
            for jt in range(0, n, max(1, n // 200)):  # Sample candidate targets
                if jt == best_j or jt == k - best_j:
                    continue
                # Distance to closest peak (in the relevant sense)
                min_dist = min(abs(jt - pk) for pk in peak_indices)
                candidates.append((min_dist, jt))

            candidates.sort(reverse=True)

            for (_, j_target) in candidates[:8]:
                for frac in [step, step * 0.5, step * 1.5, step * 0.25]:
                    delta = a[best_j] * frac
                    if delta < 1e-13:
                        continue
                    candidate = a.copy()
                    candidate[best_j] -= delta
                    candidate[j_target] += delta
                    if candidate[best_j] < -1e-12:
                        continue
                    candidate = np.maximum(candidate, 0.0)
                    csum = candidate.sum()
                    if csum < 1e-15:
                        continue
                    candidate = candidate / csum
                    val = fast_convolve(candidate).max()
                    if val < best_obj - 1e-14:
                        a = candidate
                        best_obj = val
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    break

            if improved:
                break

        if not improved:
            no_improve += 1
            step *= 0.94
            if step < 1e-8:
                break
            # Periodic perturbation to escape local minima
            if no_improve % 50 == 0:
                candidate = project_simplex(a + rng.standard_normal(n) * 0.002)
                val = fast_convolve(candidate).max()
                if val < best_obj:
                    a = candidate
                    best_a = a.copy()
                    best_obj = val
                    no_improve = 0
            elif no_improve > 500:
                break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s
