"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Log-sum-exp smoothed gradient with mirror map (Cartesian in theta-space)
- Annealed temperature for adaptive peak focusing
- Peak-directed coordinate descent for local polish
- SCA with LP oracle for final refinement

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.special import softmax
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(a, p, fft_size):
    """
    Compute gradient of log-sum-exp(conv(a,a)) weighted by p.
    grad[i] = 2 * sum_k p[k] * a[k-i] (cross-correlation of p and a)
    """
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(np.conj(P) * A, fft_size)
    n = len(a)
    return 2.0 * corr[:n]


def smooth_grad_descent(a0, time_limit, tau_init=200.0, tau_final=5000.0,
                        lr_init=0.05, verbose=False):
    """
    Mirror descent in theta-space using log-sum-exp smoothed gradient.
    a = softmax(theta), optimize theta directly (no projection needed).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-30)
    a = a / a.sum()
    # Initialize theta
    theta = np.log(a + 1e-30)
    
    fft_size = 1 << (3 * n - 1).bit_length()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    tau = tau_init
    lr = lr_init
    momentum = np.zeros(n)
    beta = 0.85
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    max_iters = int(time_limit * 500)  # rough cap
    
    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        cur_max = conv.max()
        
        # Compute softmax weights (temperature-controlled)
        shifted = tau * (conv - cur_max)
        p = np.exp(shifted)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(2 * n - 1)
            p[int(np.argmax(conv))] = 1.0
        else:
            p = p / p_sum
        
        # Compute gradient of the smoothed objective
        grad = compute_smooth_grad(a, p, fft_size)
        
        # Mirror map gradient: in theta-space, grad_theta = a * (grad - dot(a, grad))
        dot_ag = np.dot(a, grad)
        grad_theta = a * (grad - dot_ag)
        
        # Momentum
        momentum = beta * momentum + (1.0 - beta) * grad_theta
        
        # Try several step sizes (backtracking)
        best_alpha = 0.0
        best_val = cur_max
        for alpha in [1.0, 0.5, 0.25, 0.1]:
            theta_new = theta - lr * alpha * momentum
            # Softmax with clipping for numerical stability
            theta_clipped = np.clip(theta_new, -50, 50)
            a_new = np.exp(theta_clipped - theta_clipped.max())
            s = a_new.sum()
            if s < 1e-30:
                continue
            a_new = a_new / s
            val = fast_convolve(a_new).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            theta = theta - lr * best_alpha * momentum
            theta_clipped = np.clip(theta, -50, 50)
            a = np.exp(theta_clipped - theta_clipped.max())
            a = a / a.sum()
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if no_improve > 30:
            lr *= 0.8
            no_improve = 0
        if lr < 0.001:
            lr = lr_init * 0.3
            no_improve = 0
        
        # Anneal temperature
        progress = min(1.0, it / max_iters)
        target_tau = tau_init * (tau_final / tau_init) ** progress
        tau = 0.95 * tau + 0.05 * target_tau
        
        # Stuck detection: perturb theta
        if no_improve > 100:
            rng = np.random.default_rng((it * 7919) % (2**31 - 1))
            theta = theta + rng.standard_normal(n) * 0.01
            theta_clipped = np.clip(theta, -50, 50)
            a = np.exp(theta_clipped - theta_clipped.max())
            a = a / a.sum()
            lr = lr_init
            no_improve = 0
            tau = tau_init
        
        if verbose and it % 500 == 0:
            print(f"    SGD it={it}, obj={best_obj:.10f}, tau={tau:.0f}, lr={lr:.5f}")
    
    return best_a, best_obj


def peak_coordinate_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent targeting the maximum peak.
    For each coordinate, find the best adjustment to reduce the max convolution value.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        peak_val = conv[k_peak]
        
        improved_this_pass = False
        
        # Try adjusting top coordinates
        # Sort by contribution to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = np.zeros(n)
        for j in range(js, je + 1):
            contrib[j] = a[j] * a[k_peak - j]
        
        # Try top 50 contributing coordinates
        top_coords = np.argsort(contrib)[::-1][:min(50, n)]
        
        for i in top_coords:
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            
            # Try reducing a[i] and redistributing
            ai = a[i]
            if ai < 1e-8:
                continue
            
            # Try several fractions to reduce
            for frac in [0.05, 0.1, 0.2, 0.3]:
                delta = frac * ai
                # Where to add the mass: try outside the peak window
                # and at positions that don't contribute to the peak
                outside = np.array([j for j in range(n) if j < js or j > je])
                if len(outside) == 0:
                    outside = np.array([j for j in range(n) if j != i])
                
                if len(outside) == 0:
                    continue
                
                # Distribute to outside positions
                a_mod = a.copy()
                a_mod[i] -= delta
                # Add proportionally to existing values at outside positions
                ext = a_mod[outside]
                ext_sum = ext.sum()
                if ext_sum < 1e-15:
                    a_mod[outside] += delta / len(outside)
                else:
                    a_mod[outside] += delta * ext / ext_sum
                
                a_mod = np.maximum(a_mod, 0)
                s = a_mod.sum()
                if s < 1e-15:
                    continue
                a_mod = a_mod / s
                
                new_obj = fast_convolve(a_mod).max()
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a_mod.copy()
                    a = a_mod
                    improved_this_pass = True
                    break  # Next coordinate
        
        if not improved_this_pass:
            no_improve += 1
            if no_improve > 10:
                # Perturb and restart
                rng = np.random.default_rng((it * 1299709) % (2**31 - 1))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                no_improve = 0
        else:
            no_improve = 0
    
    return best_a, best_obj


def balanced_subgradient(a0, time_limit, top_k=20, verbose=False):
    """
    Subgradient method using top-k peaks with proportional weighting.
    Uses a proper line search and restart strategy.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = 0.03
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        
        # Top-k peaks with smooth weighting
        k_use = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_use)[-k_use:]
        pv = conv[peaks]
        # Exponential weighting (sharper = focus on max)
        pw = np.exp(10.0 * (pv - pv.max()))
        pw = pw / pw.sum()
        
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            w = pw[idx]
            if w < 1e-12:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            # grad[j] += w * 2 * a[k-j]
            js_je = np.arange(js, je + 1)
            grad[js_je] += w * 2.0 * a[k - js_je]
        
        # Center the gradient (tangent to simplex)
        grad = grad - grad.mean()
        
        # Line search (geometric)
        best_alpha = 0.0
        best_val = conv.max()
        alpha = 1.0
        for _ in range(8):
            a_new = a - lr * alpha * grad
            a_new = project_simplex(a_new)
            val = fast_convolve(a_new).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
            alpha *= 0.5
        
        if best_alpha > 0:
            a = a - lr * best_alpha * grad
            a = project_simplex(a)
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Adaptive lr
        if no_improve > 20:
            lr *= 0.85
        if lr < 0.001:
            lr = 0.03
            no_improve = 0
        
        if no_improve > 80:
            rng = np.random.default_rng((it * 48271) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            lr = 0.03
            no_improve = 0
    
    return best_a, best_obj


def build_lp_matrix(n):
    """Precompute sparsity pattern for LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_oracle(f, M, n, row_idx=None, col_idx=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    if row_idx is None or col_idx is None:
        row_idx, col_idx = build_lp_matrix(n)
    
    c = -np.ones(n)
    k_arr = row_idx
    i_arr = col_idx
    vals = f[k_arr - i_arr]
    
    A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def sca_optimize(f0, time_limit, M_init=None, verbose=False):
    """SCA optimization with adaptive M and LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    row_idx, col_idx = build_lp_matrix(n)
    
    conv = fast_convolve(f)
    M = conv.max() if M_init is None else M_init
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        g_raw = solve_lp_oracle(f, M, n, row_idx, col_idx)
        
        if g_raw is None:
            M *= 1.03
            no_improve += 1
            if no_improve > 50:
                M = fast_convolve(f).max() * 1.01
                no_improve = 0
            continue
        
        g = g_raw / g_raw.sum()
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
                f = project_simplex(f + rng.standard_normal(n) * 0.003)
                M = fast_convolve(f).max() * 1.01
                no_improve = 0
            continue
        
        # Line search over blends
        best_val = fast_convolve(f).max()
        best_ft = f.copy()
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_ft = ft.copy()
        
        if best_val < best_obj - 1e-14:
            f = best_ft
            best_obj = best_val
            best_f = f.copy()
            M = best_val * 0.998
            no_improve = 0
        else:
            no_improve += 1
            M *= 0.997
        
        if no_improve > 60:
            rng = np.random.default_rng((it * 31831 + 7) % (2**31 - 1))
            scale = 0.002 + 0.003 * min(1.0, (no_improve - 60) / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max() * 1.01
            no_improve = 0
    
    return best_f, best_obj


def multi_peak_smoothing(a0, time_limit, verbose=False):
    """
    Iteratively equalize peaks by redistributing mass from high-peak
    contributors to low-peak positions.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        mean_conv = conv.mean()
        
        # Find peaks above mean
        above = np.where(conv > mean_conv * 1.05)[0]
        if len(above) == 0:
            no_improve += 1
        else:
            # For each above-mean peak, try to reduce it
            improved = False
            for k in above[:10]:  # Top 10 above-mean peaks
                if time.monotonic() - t0 > time_limit - 0.5:
                    break
                
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                
                # Contributors to this peak
                contrib = np.zeros(n)
                for j in range(js, je + 1):
                    contrib[j] = a[j] * a[k - j]
                
                total_c = contrib.sum()
                if total_c < 1e-15:
                    continue
                
                # Try to reduce the top contributor
                top_i = int(np.argmax(contrib))
                if a[top_i] < 1e-8:
                    continue
                
                # Reduce top_i by a small amount
                for frac in [0.02, 0.05, 0.08]:
                    delta = frac * a[top_i]
                    a_mod = a.copy()
                    a_mod[top_i] -= delta
                    
                    # Distribute to positions that DON'T contribute to this peak
                    outside_mask = np.ones(n, dtype=bool)
                    for kk in range(max(0, k - n + 1), min(n, k + 1)):
                        outside_mask[kk - (k - max(0, k - n + 1)) + max(0, k - n + 1)]  # complex
                    # Simpler: positions j where k-j is out of [0, n-1]
                    outside_idx = []
                    for j in range(n):
                        if j < js or j > je:
                            outside_idx.append(j)
                    if not outside_idx:
                        continue
                    
                    ext = a_mod[outside_idx]
                    ext_sum = ext.sum()
                    if ext_sum < 1e-15:
                        a_mod[outside_idx] += delta / len(outside_idx)
                    else:
                        a_mod[outside_idx] += delta * ext / ext_sum
                    
                    a_mod = np.maximum(a_mod, 0)
                    a_mod = a_mod / a_mod.sum()
                    new_obj = fast_convolve(a_mod).max()
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a_mod.copy()
                        a = a_mod
                        improved = True
                        break
            
            if improved:
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve > 20:
            rng = np.random.default_rng((it * 9973) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0
    
    return best_a, best_obj


def structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    t = x / max(n - 1, 1)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Two Gaussian bumps
    for c1, c2, w in [(0.25, 0.75, 0.08), (0.3, 0.7, 0.1), (0.2, 0.8, 0.06)]:
        g = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
        inits.append(g / g.sum())
    
    # Sinusoidal modulation
    for freq, amp in [(2, 0.3), (3, 0.4), (4, 0.5), (1, 0.2)]:
        s = 1.0 + amp * np.sin(2 * np.pi * freq * t)
        s = np.maximum(s, 0.01)
        inits.append(s / s.sum())
    
    # Edge-heavy
    edge_w = 1.0 / (np.minimum(x, n - 1 - x).astype(float) + 1.0) ** 1.2
    inits.append(edge_w / edge_w.sum())
    
    # Dirichlet
    for alpha_val in [0.3, 1.0, 3.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha_val))
    
    # Cosine window
    cw = 0.5 * (1 + np.cos(2 * np.pi * t)) + 0.05
    inits.append(cw / cw.sum())
    
    # "BCE-like" pattern (from additive combinatorics)
    if n % 2 == 0:
        half = n // 2
        pattern = np.zeros(n)
        pattern[::2] = 1.0
        pattern[1::2] = 0.3
        inits.append(pattern / pattern.sum())
    
    return inits

            a_opt, _ = smooth_grad_descent(best_seq, time_limit=t_alloc,
                                           tau_init=200.0, tau_final=5000.0,
                                           lr_init=0.05, verbose=True)
            try_seq(a_opt, "sgd_main")
            
            # Second round with different params
            remaining = deadline - time.monotonic() - 400
            if remaining > 30:
                t_alloc2 = min(remaining, 130)
                print(f"  SGD round2 t={t_alloc2:.0f}s")
                a_opt2, _ = smooth_grad_descent(a_opt, time_limit=t_alloc2,
                                                tau_init=1000.0, tau_final=20000.0,
                                                lr_init=0.02, verbose=True)
                try_seq(a_opt2, "sgd_r2")
    
    print(f"  P1 done: best={best_score:.8f}, t={time.monotonic()-t1:.0f}s")
    
    # === Phase 2: Balanced subgradient (250s) ===
    print("\n=== Phase 2: Balanced Subgradient ===")
    t2 = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        for k in [10, 20, 40]:
            remaining = deadline - time.monotonic() - 300
            if remaining < 20:
                break
            t_alloc = min(remaining, 80)
            print(f"  Subgrad k={k}, t={t_alloc:.0f}s")
            a_opt, _ = balanced_subgradient(best_seq, time_limit=t_alloc, top_k=k)
            try_seq(a_opt, f"balsg_k{k}")
    
    print(f"  P2 done: best={best_score:.8f}, t={time.monotonic()-t2:.0f}s")
    
    # === Phase 3: SCA refinement (300s) ===
    print("\n=== Phase 3: SCA Refinement ===")
    t3 = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 300
        if remaining > 30:
            t_alloc = min(remaining, 200)
            print(f"  SCA t={t_alloc:.0f}s")
            a_opt, _ = sca_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "sca")
            
            remaining = deadline - time.monotonic() - 150
            if remaining > 20:
                t_alloc2 = min(remaining, 100)
                a_opt2, _ = sca_optimize(a_opt, time_limit=t_alloc2)
                try_seq(a_opt2, "sca2")
    
    print(f"  P3 done: best={best_score:.8f}, t={time.monotonic()-t3:.0f}s")
    
    # === Phase 4: Peak coordinate descent (150s) ===
    print("\n=== Phase 4: Peak Coordinate Descent ===")
    t4 = time.monotonic()
    
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 150
        if remaining > 20:
            t_alloc = min(remaining, 100)
            print(f"  PCD t={t_alloc:.0f}s")
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "pcd")
            
            remaining = deadline - time.monotonic() - 80
            if remaining > 15:
                t_alloc2 = min(remaining, 60)
                a_opt2, _ = multi_peak_smoothing(a_opt, time_limit=t_alloc2)
                try_seq(a_opt2, "mps")
    
    print(f"  P4 done: best={best_score:.8f}, t={time.monotonic()-t4:.0f}s")
    
    # === Phase 5: Diversified restarts (200s) ===
    print("\n=== Phase 5: Diversified Restarts ===")
    t5 = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        scales = [0.002, 0.005, 0.01, 0.02, 0.04]
        for scale in scales:
            remaining = deadline - time.monotonic() - 50
            if remaining < 20:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(30, remaining - 30)
            if t_alloc < 5:
                break
            print(f"  Restart scale={scale}, t={t_alloc:.0f}s")
            a_opt, _ = smooth_grad_descent(f_pert, time_limit=t_alloc,
                                           tau_init=500.0, tau_final=10000.0,
                                           lr_init=0.03)
            try_seq(a_opt, f"restart_{scale}")
    
    print(f"  P5 done: best={best_score:.8f}, t={time.monotonic()-t5:.0f}s")
    
    # === Phase 6: Final polish (remaining - 20s) ===
    print("\n=== Phase 6: Final Polish ===")
    t6 = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        remaining = deadline - time.monotonic() - 15
        if remaining > 10:
            t_alloc = min(remaining, 30)
            print(f"  Final SGD t={t_alloc:.0f}s")
            a_opt, _ = smooth_grad_descent(best_seq, time_limit=t_alloc,
                                           tau_init=5000.0, tau_final=50000.0,
                                           lr_init=0.01)
            try_seq(a_opt, "final_sgd")
        
        remaining = deadline - time.monotonic() - 10
        if remaining > 5:
            a_opt, _ = balanced_subgradient(best_seq, time_limit=min(remaining, 15), top_k=5)
            try_seq(a_opt, "final_bsg")
        
        remaining = deadline - time.monotonic() - 5
        if remaining > 3:
            a_opt, _ = multi_peak_smoothing(best_seq, time_limit=min(remaining, 8))
            try_seq(a_opt, "final_mps")
    
    print(f"  P6 done: best={best_score:.8f}, t={time.monotonic()-t6:.0f}s")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final check
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
