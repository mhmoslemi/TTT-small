    n = len(a)

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def pnorm_grad_fft(a, p):
    """Compute gradient of (1/p)*sum_k c[k]^p where c=conv(a,a). O(n log n)."""
    n = len(a)
    c = fast_convolve(a)
    c_safe = np.maximum(c, 1e-30)
    d = c_safe ** (p - 1)
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    D = np.fft.rfft(d, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(np.conj(D) * A, fft_size)
    return cross[:n]
