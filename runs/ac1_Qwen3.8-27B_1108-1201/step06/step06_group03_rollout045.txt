- Log-sum-exp gradient descent with adaptive temperature (fast smooth optimization)
- Frank-Wolfe with LP oracle for big moves when stuck
- Multi-peak targeted coordinate descent for final polishing

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: sum=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1)) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (f*g)[k] = sum_j A[k,j]*g[j]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})

    if g is None or g.sum() < 1e-15:
        return f, M, False
    
    g = g / g.sum()
    
    if np.max(np.abs(g - f)) < 1e-10:
        return f, M, False
    
    # Line search
    best_alpha = 0.0
    best_val = M + 1e-14
    
    # Coarse: 20 points
    for alpha in np.linspace(0.02, 1.0, 20):
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:

        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    # Fine: 10 points around best
    if best_alpha > 0.01:
        lo = max(0.0, best_alpha - 0.02)
        hi = min(1.0, best_alpha + 0.02)
        for alpha in np.linspace(lo, hi, 10):
            ft = (1.0 - alpha) * f + alpha * g

        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        return f_new, best_val, True
    
    return f, M, False

                            best_a = a.copy()
                            improved_this_round = True
                            break
                    if improved_this_round:

                if improved_this_round:

            if improved_this_round:
                break
        
        if not improved_this_round:
            step_scale *= 0.92
            if step_scale < 1e-7:
                break

            if it % 300 == 0:

                candidate = project_simplex(a + rng.standard_normal(n) * max(step_scale * 3, 0.001))
                val = fast_convolve(candidate).max()
                if val < best_obj:

                    best_obj = val
                    best_a = a.copy()
                    step_scale = 0.002
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_starts(n, parent, rng):
    """Generate diverse initializations."""
    starts = []
    x = np.arange(n, dtype=np.float64)
    
    # Parent (resampled)
    if parent is not None:
        starts.append(resample_sequence(parent, n))
    
    # Uniform
    starts.append(np.ones(n) / n)
    
    # Cosine taper (emphasize center)
    cos_t = np.cos(np.pi * (x / (n - 1) - 0.5))
    w = 0.5 * (1 + cos_t)
    w = np.maximum(w, 0.01)
    starts.append(w / w.sum())
    
    # Inverse cosine (emphasize edges)
    w = 0.5 * (1 - cos_t)
    w = np.maximum(w, 0.01)
    starts.append(w / w.sum())
    
    # Gaussian (center)
    g = np.exp(-0.5 * ((x - (n - 1) / 2) / (0.15 * n)) ** 2)
    starts.append(g / g.sum())
    
    # Two Gaussians
    g1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.06 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.06 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())
    
    # Quartic (flatter top)
    t = 2.0 * x / (n - 1) - 1.0
    w = (1 - t**4)
    w = np.maximum(w, 0.01)
    starts.append(w / w.sum())
    
    # Random Dirichlet (concentrated)
    starts.append(rng.dirichlet(np.ones(n) * 0.2))
    
    # Random Dirichlet (uniform-ish)
    starts.append(rng.dirichlet(np.ones(n) * 1.0))
    
    # "Bump" - concentrated in middle third
    mask = np.zeros(n)
    mask[n//4:3*n//4] = 1.0
    starts.append(mask / mask.sum())
    
    return starts
