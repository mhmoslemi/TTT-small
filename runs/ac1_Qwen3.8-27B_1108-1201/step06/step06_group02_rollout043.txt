"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses Frank-Wolfe with LP oracle, smoothed gradient descent, and targeted peak reduction.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute gradient: 2 * (p[j+m] * a[m]) summed over m."""
    n = len(a)
    result = np.zeros(n)
    for j in range(n):
        ms = np.arange(n)
        valid = (j + ms < len(p))
        result[j] = np.dot(p[j + ms[valid]], a[ms[valid]])
    return 2.0 * result


def compute_gradient(a, T):
    """Smoothed gradient using softmax-weighted peaks."""
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()
    return hankel_mv(p, a), conv

def fast_convolve(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute vector j -> sum_m p[j+m]*a[m] (valid indices only)."""
    n = len(a)
    result = np.zeros(n)
    for j in range(n):
        ms = np.arange(n)
        valid = (j + ms < len(p))
        result[j] = np.dot(p[j + ms[valid]], a[ms[valid]])
    return 2.0 * result


def compute_smoothed_gradient(a, T):
    """Compute smoothed subgradient via softmax-weighted peaks."""
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()
    grad = hankel_mv(p, a)
    return grad, conv

    """Solve LP: max sum(b) s.t. conv(f,b)[k] <= M, b >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:

    except Exception:

    return None

        conv = fast_convolve(a)
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 2.0
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]

        v = momentum * v - lr * sg
        a = project_simplex(a + v)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9998
        momentum = min(0.98, momentum + 0.00005)

        if no_improve > 150:
            noise = 0.001 + 0.004 * min(1.0, (no_improve - 150) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init * 0.7
            no_improve = 0

    return best_a, best_obj


def try_and_update(a, state, label):
    """Evaluate a candidate and update global best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.10f} (n={})".format(label, score, len(a)))


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid optimization:
    Phase 1: Quick multi-n exploration with FW-LP (~150s)
    Phase 2: Deep FW-LP on best sequence (~600s)
    Phase 3: Fourier + gradient methods (~500s)
    Phase 4: Targeted peak reduction + final polish (~350s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)
    state = [float('inf'), None]

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.10f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        ps = float(evaluate_sequence(parent.tolist()))
        state = [ps, parent.copy()]
        print("Using default uniform parent: n=768, score={:.10f}".format(ps))

    n_parent = len(parent)

    # === Phase 1: Quick Multi-n Exploration ===
    print("\n=== Phase 1: Quick Multi-n Exploration ===")
    t_phase = time.monotonic()

    # Evaluate uniform sequence at parent length
    uniform = np.ones(n_parent) / n_parent
    try_and_update(uniform, state, "uniform_n{}".format(n_parent))

    # Try a few different lengths
    candidate_ns = sorted(set([n_parent, 512, 1024, 1280]))
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 2048]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 10:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss > 1e-10:
                start = start / ss
            else:
                start = np.ones(n_try) / n_try

        t_alloc = min(remaining, 35)
        print("  FW n={}, t={:.0f}s".format(n_try, t_alloc))
        a_opt, _ = fw_lp_optimize(start, time_limit=t_alloc)
        try_and_update(a_opt, state, "fw_n{}".format(n_try))

        remaining = deadline - time.monotonic() - 1150
        if remaining > 8:
            a_opt2, _ = smoothed_grad_descent(a_opt, time_limit=min(remaining, 12),
                                              T_init=0.001, lr_init=0.03)
            try_and_update(a_opt2, state, "sg_n{}".format(n_try))

    print("  Phase 1 done: best={:.10f} ({:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 2: Deep FW-LP ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])
        remaining = deadline - time.monotonic() - 800
        if remaining > 20:
            t_alloc = min(remaining, 500)
            print("  Deep FW n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = fw_lp_optimize(state[1], time_limit=t_alloc, seed=111)
            try_and_update(a_opt, state, "deep_fw")

            # Perturbed restarts
            for scale in [0.004, 0.01, 0.02]:
                remaining = deadline - time.monotonic() - 600
                if remaining < 10:
                    break
                f_pert = project_simplex(state[1] + rng.standard_normal(n_best) * scale)
                t_alloc = min(remaining, 50)
                a_opt2, _ = fw_lp_optimize(f_pert, time_limit=t_alloc,
                                           seed=222 + int(scale * 1000))
                try_and_update(a_opt2, state, "fw_pert")

    print("  Phase 2 done: best={:.10f} ({:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 3: Fourier + Gradient ===
    print("\n=== Phase 3: Fourier + Gradient Methods ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Fourier refinement with moderate modes
        remaining = deadline - time.monotonic() - 350
        if remaining > 15:
            t_alloc = min(remaining, 150)
            print("  Fourier(12), t={:.0f}s".format(t_alloc))
            a_opt, _ = fourier_refine(state[1], time_limit=t_alloc, n_fourier=12)
            try_and_update(a_opt, state, "fourier_12")

        # Fourier with more modes
        remaining = deadline - time.monotonic() - 250
        if remaining > 12:
            t_alloc = min(remaining, 80)
            print("  Fourier(20), t={:.0f}s".format(t_alloc))
            a_opt, _ = fourier_refine(state[1], time_limit=t_alloc, n_fourier=20)
            try_and_update(a_opt, state, "fourier_20")

        # Smoothed gradient with different temps
        for T_init, lr_init in [(0.0003, 0.04), (0.001, 0.025)]:
            remaining = deadline - time.monotonic() - 150
            if remaining < 10:
                break
            t_alloc = min(remaining, 70)
            print("  SmoothedGrad T={:.4f} lr={:.3f}".format(T_init, lr_init))
            a_opt, _ = smoothed_grad_descent(state[1], time_limit=t_alloc,
                                             T_init=T_init, lr_init=lr_init)
            try_and_update(a_opt, state, "smooth_T{}".format(int(T_init * 10000)))

        # Multi-peak subgradient
        for top_k, lr, mom in [(10, 0.003, 0.94), (30, 0.002, 0.92)]:
            remaining = deadline - time.monotonic() - 80
            if remaining < 8:
                break
            t_alloc = min(remaining, 50)
            print("  MultiPeakSG k={}".format(top_k))
            a_opt, _ = multi_peak_subgradient(state[1], time_limit=t_alloc,
                                              top_k=top_k, lr_init=lr, momentum=mom)
            try_and_update(a_opt, state, "mpsg_{}".format(top_k))

    print("  Phase 3 done: best={:.10f} ({:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 4: Targeted Peak Reduction + Final Polish ===
    print("\n=== Phase 4: Final Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Targeted peak reduction
        remaining = deadline - time.monotonic() - 80
        if remaining > 8:
            t_alloc = min(remaining, 80)
            print("  TargetedPeakReduction, t={:.0f}s".format(t_alloc))
            a_opt, _ = targeted_peak_reduction(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "targeted")

        # Final FW pass
        remaining = deadline - time.monotonic() - 40
        if remaining > 8:
            t_alloc = min(remaining, 40)
            print("  FinalFW, t={:.0f}s".format(t_alloc))
            a_opt, _ = fw_lp_optimize(state[1], time_limit=t_alloc, seed=999)
            try_and_update(a_opt, state, "final_fw")

        # Fine smoothed gradient
        remaining = deadline - time.monotonic() - 20
        if remaining > 5:
            t_alloc = min(remaining, 20)
            print("  FineSmooth, t={:.0f}s".format(t_alloc))
            a_opt, _ = smoothed_grad_descent(state[1], time_limit=t_alloc,
                                             T_init=0.0001, lr_init=0.005)
            try_and_update(a_opt, state, "fine_smooth")

        # Tiny perturbations
        remaining = deadline - time.monotonic() - 5
        if remaining > 3:
            for _ in range(3):
                remaining = deadline - time.monotonic() - 2
                if remaining < 3:
                    break
                noise = rng.standard_normal(len(state[1])) * 0.0003
                f_pert = project_simplex(state[1] + noise)
                a_opt, _ = smoothed_grad_descent(f_pert, time_limit=min(remaining, 4),
                                                 T_init=0.0002, lr_init=0.003)
                try_and_update(a_opt, state, "micro_perturb")

    print("  Phase 4 done: best={:.10f} ({:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(768) / 768
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL: score={:.10f}, n={}".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
