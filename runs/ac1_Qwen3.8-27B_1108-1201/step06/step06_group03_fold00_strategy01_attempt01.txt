def fast_convolve(a):
    """Efficient computation of full convolution via FFT with zero-padding"""
    n = len(a)
    size = 2*n - 1
    # Pad if necessary
    pad_size = max(0, size - len(a) - n)

def simplex_project(v, tol=1e-10):
    """Compute Euclidean projection onto non-negative simplex"""
    n = len(v)

            rho = i
            break
    if rho == 0:

    theta = (cssv[rho] / rho - 1.0) / (rho + 1)
