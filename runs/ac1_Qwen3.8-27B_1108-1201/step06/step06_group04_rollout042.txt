def project_simplex(v):
    n=len(v)
    if n==1: return np.array([max(v[0],0.0)])
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    cond=u*np.arange(1,n+1) > (cssv-1)
    rho=np.nonzero(cond)[0][-1]
    theta=(cssv[rho]-1.0)/(rho+1.0)
    w=np.maximum(v-theta,0.0)
    # Due to float, sum may not exactly 1; divide
    s=w.sum()
    if s>0: w/=s
    else: w=np.ones(n)/n
    return w
```

```python
"""
Search for non-negative step-function heights minimizing
    2*n*max(conv(a,a))/(sum(a))^2.

The program is a hybrid local-search / convex-smoothing solver:
- starts from the provided parent construction and from several shaped random inits,
- upsamples to several lengths and keeps only the most promising ones,
- polishes with L-BFGS-B on a softmax parameterization and a log-sum-exp surrogate,
- uses peak-weighted projected gradient steps in probability-simplex space,
- occasionally uses a linear-programming direction based on the SOTA LP idea,
- finishes with small-threshold zeroing and final verification.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize as _minimize
    HAVE_MINIMIZE = True
except Exception:
    _minimize = None
    HAVE_MINIMIZE = False

try:
    from scipy.optimize import linprog as _linprog
    import scipy.sparse as _sp
    HAVE_LP = True
except Exception:
    _linprog = None
    _sp = None
    HAVE_LP = False


class SearchTimeout(Exception):
    pass


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:size]


def normalize_sequence(a):
    a = np.asarray(a, dtype=float).ravel()
    if a.size == 0:
        return np.ones(1, dtype=float)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(a.size, dtype=float) / float(a.size)
    return a / s


def project_simplex(v):
    v = np.asarray(v, dtype=float).ravel()
    n = v.size
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        w = np.ones(n, dtype=float) / float(n)
    else:
        rho = int(np.nonzero(cond)[0][-1])
        theta = (cssv[rho] - 1.0) / float(rho + 1)
        w = np.maximum(v - theta, 0.0)
        s = float(w.sum())
        if s < 1e-15:
            w = np.ones(n, dtype=float) / float(n)
        else:
            w = w / s
    return w


def hankel_conv(b, a):
    """
    Returns h_i = sum_{j=0}^{len(a)-1} b[i+j] * a_j for i=0..len(a)-1.
    This is the gradient support for weighted sums of linear-convolution coefficients.
    """
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    B = np.fft.rfft(b, N)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(B * np.conj(A), N)[:n]
    return np.asarray(out, dtype=float)


def softmax_from_y(y):
    y = np.asarray(y, dtype=float).ravel()
    y = np.where(np.isfinite(y), y, 0.0)
    y = np.clip(y, -50.0, 50.0)
    y = y - float(y.max())
    e = np.exp(y)
    s = float(e.sum())
    if s < 1e-15:
        return np.ones(y.size, dtype=float) / float(y.size)
    return e / s


def score_from_sequence(a):
    a = normalize_sequence(a)
    n = len(a)
    c = fast_convolve(a)
    return float(2.0 * float(n) * float(c.max()))


def resample_sequence(parent, n):
    p = normalize_sequence(parent)
    m = len(p)
    if m == n:
        return p.copy()
    if m == 1:
        return np.full(n, float(p[0]), dtype=float)
    xp = np.linspace(0.0, 1.0, m)
    x = np.linspace(0.0, 1.0, n)
    y = np.interp(x, xp, p, left=float(p[0]), right=float(p[-1]))
    return normalize_sequence(y)


def unique_preserve(xs):
    out = []
    for x in xs:
        if x not in out:
            out.append(x)
    return out


def sample_list(lst, k, rng):
    if len(lst) <= k:
        return list(lst)
    idx = rng.choice(len(lst), size=k, replace=False)
    return [lst[int(i)] for i in idx]


def make_shaped_inits(n, rng=None):
    inits = []
    x = np.linspace(0.0, 1.0, n)
    xm = x - 0.5

    for c in (-1.5, -1.0, -0.5, 0.5, 1.0, 1.5):
        y = 1.0 + c * xm
        y = np.maximum(y, 1e-4)
        inits.append(normalize_sequence(y))

    for mu in (0.25, 0.35, 0.5, 0.65, 0.75):
        for sig in (0.12, 0.25, 0.45):
            y = np.exp(-0.5 * ((x - mu) / sig) ** 2)
            inits.append(normalize_sequence(y))

    for c in (-1.0, -0.5, 0.5, 1.0):
        y = 1.0 + c * np.sin(np.pi * x)
        y = np.maximum(y, 1e-4)
        inits.append(normalize_sequence(y))

        y = 1.0 + c * np.cos(2.0 * np.pi * x)
        y = np.maximum(y, 1e-4)
        inits.append(normalize_sequence(y))

    if rng is not None:
        inits.append(normalize_sequence(rng.dirichlet(np.ones(n, dtype=float) * 0.5)))
        inits.append(normalize_sequence(rng.dirichlet(np.ones(n, dtype=float) * 0.2)))

    return inits


class SoftMaxProblem:
    def __init__(self, T, n, time_limit):
        self.T = float(max(1e-30, T))
        self.n = int(n)
        self.time_limit = float(time_limit)
        self.t0 = time.monotonic()
        self.scale = float(self.n * self.n)
        self.best_a = None
        self.bestM = float("inf")

    def _check(self):
        if time.monotonic() - self.t0 > self.time_limit + 5.0:
            raise SearchTimeout()

    def _conv_for_y(self, y):
        y = np.asarray(y, dtype=float).ravel()
        y = np.where(np.isfinite(y), y, 0.0)
        y = np.clip(y, -50.0, 50.0)
        y = y - float(y.max())
        e = np.exp(y)
        s = float(e.sum())
        if s < 1e-15:
            a = np.ones(self.n, dtype=float) / float(self.n)
        else:
            a = e / s
        c = fast_convolve(a)
        return a, c

    def update_best(self, a, c):
        cm = float(c.max())
        if cm < self.bestM:
            self.bestM = cm
            self.best_a = np.asarray(a, dtype=float).copy()

    def fun(self, y):
        self._check()
        a, c = self._conv_for_y(y)
        cm = float(c.max())
        self.update_best(a, c)
        scaled = (c - cm) / self.T
        p = np.exp(scaled)
        s = float(p.sum())
        if not np.isfinite(s) or s <= 0.0:
            phi = cm
        else:
            phi = cm + self.T * float(np.log(s))
        return float(self.scale * phi)

    def jac(self, y):
        self._check()
        a, c = self._conv_for_y(y)
        cm = float(c.max())
        self.update_best(a, c)

        scaled = (c - cm) / self.T
        p = np.exp(scaled)
        s = float(p.sum())
        if not np.isfinite(s) or s <= 0.0:
            p = np.zeros_like(c)
            p[int(np.argmax(c))] = 1.0
        else:
            p = p / s

        grad_a = 2.0 * hankel_conv(p, a)
        S = float(np.dot(a, grad_a))
        gy = grad_a * a - a * S
        return self.scale * gy


def run_lbfgs(a_init, T, maxiter, time_limit, noise_scale=0.0, rng=None):
    a0 = normalize_sequence(a_init)
    n = len(a0)
    M0 = float(fast_convolve(a0).max())
    prob = SoftMaxProblem(T, n, time_limit)
    prob.best_a = a0.copy()
    prob.bestM = M0

    if not HAVE_MINIMIZE or time_limit < 2.0:
        return prob.best_a, prob.bestM

    y0 = np.log(np.maximum(a0, 1e-300))
    y0 = np.where(np.isfinite(y0), y0, 0.0)
    y0 = np.clip(y0, -50.0, 50.0)
    y0 = y0 - float(y0.mean())
    y0 = np.clip(y0, -50.0, 50.0)

    if noise_scale > 0.0 and rng is not None:
        y0 = y0 + rng.standard_normal(n) * float(noise_scale)
        y0 = np.clip(y0, -50.0, 50.0)

    bounds = [(-50.0, 50.0)] * n
    options = {
        "maxiter": int(max(10, int(maxiter))),
        "maxls": 20,
        "ftol": 1e-12,
        "gtol": 1e-6,
    }

    res = None
    try:
        res = _minimize(prob.fun, y0, jac=prob.jac, method="L-BFGS-B",
                        bounds=bounds, options=options)
    except SearchTimeout:
        res = None
    except Exception:
        res = None

    a = prob.best_a
    if res is not None and hasattr(res, "x") and np.all(np.isfinite(res.x)):
        a2 = softmax_from_y(res.x)
        M2 = float(fast_convolve(a2).max())
        if M2 < prob.bestM:
            prob.bestM = M2
            prob.best_a = a2.copy()
            a = a2

    return np.asarray(a, dtype=float), float(prob.bestM)


def make_weight_vector(c, mode, t_factor, n, rng):
    m = len(c)
    if m == 0:
        return np.zeros(0, dtype=float)

    c = np.asarray(c, dtype=float)
    if mode == 0 or mode == 3:
        T = max(1e-30, float(t_factor) / max(1, n))
        scaled = c / T
        scaled -= float(scaled.max())
        w = np.exp(scaled)
        s = float(w.sum())
        if not np.isfinite(s) or s <= 0.0:
            return np.ones(m, dtype=float) / float(m)
        return w / s

    if mode == 5:
        K = min(m, 8)
        if K <= 0:
            return np.ones(m, dtype=float) / float(m)
        if K >= m:
            idx = np.arange(m)
        else:
            idx = np.argpartition(c, -K)[-K:]
        width = max(1.0, float(n) / 20.0)
        x = np.arange(m, dtype=float)
        w = np.zeros(m, dtype=float)
        for k in idx:
            w = w + np.exp(-((x - float(k)) / width) ** 2)
        s = float(w.sum())
        if not np.isfinite(s) or s <= 0.0:
            return np.ones(m, dtype=float) / float(m)
        return w / s

    if mode == 1:
        K = max(4, min(m, 16))
    elif mode == 2:
        K = max(8, min(m, 64))
    elif mode == 4:
        K = max(4, min(m, 32))
    else:
        K = max(4, min(m, 16))

    if K <= 0:
        return np.ones(m, dtype=float) / float(m)
    if K >= m:
        idx = np.arange(m)
    else:
        idx = np.argpartition(c, -K)[-K:]

    w = np.zeros(m, dtype=float)
    if mode == 4 and rng is not None:
        vals = rng.dirichlet(np.ones(len(idx), dtype=float))
        w[idx] = vals
    else:
        vals = np.asarray(c[idx], dtype=float)
        vals = vals - float(vals.min()) + 1e-300
        s = float(vals.sum())
        if s <= 0.0:
            return np.ones(m, dtype=float) / float(m)
        w[idx] = vals / s
    return w


def direction_search(a_init, time_limit, seed=0):
    a = normalize_sequence(a_init)
    n = len(a)
    best = a.copy()
    bestM = float(fast_convolve(best).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    v = np.zeros(n, dtype=float)
    beta = 0.5
    alpha = 1.0
    t_factor = 0.3
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < max(1.0, float(time_limit)) - 0.5:
        c = fast_convolve(a)
        M = float(c.max())
        if M < bestM:
            bestM = M
            best = a.copy()

        mode = it % 6
        w = make_weight_vector(c, mode, t_factor, n, rng)
        h = hankel_conv(w, a)
        hs = float(np.max(np.abs(h)))

        if hs < 1e-18:
            no_improve += 1
            alpha = max(0.05, alpha * 0.8)
        else:
            dirv = -h / hs
            v = beta * v + dirv
            vmax = float(np.max(np.abs(v)))
            if vmax <= 0.0:
                v = dirv
            else:
                v = v / vmax

            base = alpha / float(n)
            cand_best = a
            cand_bestM = M

            for mult in (0.25, 0.5, 1.0, 2.0):
                cand = project_simplex(a + base * mult * v)
                cm = float(fast_convolve(cand).max())
                if cm < cand_bestM:
                    cand_bestM = cm
                    cand_best = cand

            if cand_bestM < M - 1e-15:
                a = cand_best
                if cand_bestM < bestM:
                    bestM = cand_bestM
                    best = a.copy()
                no_improve = 0
                alpha = min(5.0, alpha * 1.03)
                t_factor = max(0.05, t_factor * 0.95)
            else:
                no_improve += 1
                alpha = max(0.05, alpha * 0.8)
                t_factor = min(2.0, t_factor * 1.1)

        it += 1

        if no_improve > 60:
            scale = (0.05 + 0.01 * min(50, no_improve)) / float(n)
            a = project_simplex(best + rng.standard_normal(n) * scale)
            v = np.zeros(n, dtype=float)
            alpha = 1.0
            t_factor = 0.2
            no_improve = 0

    c = fast_convolve(best)
    bestM = float(c.max())
    return best, bestM


def lp_blend(a, time_limit=30.0):
    a = normalize_sequence(a)
    n = len(a)
    c = fast_convolve(a)
    M = float(c.max())
    best = a.copy()
    bestM = M

    if not HAVE_LP or n > 1200 or time_limit < 3.0 or M <= 0.0 or not np.isfinite(M):
        return best, bestM

    M_lp = M * 1.00000001 + 1e-300
    scale = 1.0 / (float(n) * M_lp)

    diags = []
    for d in range(n):
        val = float(a[d]) * scale
        if val == 0.0 or not np.isfinite(val):
            diags.append(None)
        else:
            diags.append(np.full(n, val, dtype=float))

    A = None
    try:
        A = _sp.diags(diags, offsets=list(range(n)), shape=(2 * n - 1, n), format="csc")
        c_obj = -np.ones(n, dtype=float)
        b_ub = np.ones(2 * n - 1, dtype=float)
        bounds = [(0.0, None)] * n
        opts = {
            "presolve": True,
            "time_limit": int(max(1, float(time_limit) * 0.9)),
        }
        res = _linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs", options=opts)
    except Exception:
        res = None

    if res is not None and hasattr(res, "x") and np.all(np.isfinite(res.x)):
        r = np.maximum(np.asarray(res.x, dtype=float), 0.0)
        s = float(r.sum())
        if s > 1e-12:
            h = r / s
            for alpha in (0.1, 0.25, 0.5, 0.75, 1.0):
                cand = project_simplex((1.0 - alpha) * a + alpha * h)
                cm = float(fast_convolve(cand).max())
                if cm < bestM:
                    bestM = cm
                    best = cand.copy()

    return best, float(bestM)


def try_zero_mask(a):
    a = normalize_sequence(a)
    n = len(a)
    best = a.copy()
    bestM = float(fast_convolve(best).max())

    for tau in (1e-12, 1e-10, 1e-8, 1e-6,
                1e-10 / max(1, n), 1e-8 / max(1, n), 1e-6 / max(1, n),
                1e-4 / max(1, n), 1e-3 / max(1, n)):
        cand = a.copy()
        cand[cand < tau] = 0.0
        s = float(cand.sum())
        if s > 1e-15:
            cand = cand / s
            m = float(fast_convolve(cand).max())
            if m < bestM:
                bestM = m
                best = cand.copy()

    for frac in (0.001, 0.005, 0.01, 0.02, 0.05):
        k = int(float(n) * frac)
        if 0 < k < n:
            thresh = float(np.partition(a, k)[k])
            cand = a.copy()
            cand[cand < thresh] = 0.0
            s = float(cand.sum())
            if s > 1e-15:
                cand = cand / s
                m = float(fast_convolve(cand).max())
                if m < bestM:
                    bestM = m
                    best = cand.copy()

    return best, float(bestM)


def optimize_at_length(a_init, budget_s, seed=0, allow_lp=True):
    a = normalize_sequence(a_init)
    n = len(a)
    best = a.copy()
    bestM = float(fast_convolve(best).max())
    t0 = time.monotonic()
    budget_s = float(max(5.0, float(budget_s)))
    rng_local = np.random.default_rng(seed)

    if HAVE_MINIMIZE and budget_s - (time.monotonic() - t0) > 35.0:
        t1 = min(70.0, 0.22 * budget_s)
        a1, M1 = run_lbfgs(a, 0.35 / float(n), 80, t1, 0.0, None)
        if M1 < bestM:
            bestM = float(M1)
            best = normalize_sequence(a1)

    if HAVE_MINIMIZE and budget_s - (time.monotonic() - t0) > 30.0:
        t2 = min(50.0, 0.15 * budget_s)
        a2, M2 = run_lbfgs(best, 0.12 / float(n), 70, t2, 0.1, rng_local)
        if M2 < bestM:
            bestM = float(M2)
            best = normalize_sequence(a2)

    if budget_s - (time.monotonic() - t0) > 40.0:
        t3 = min(180.0, 0.35 * budget_s)
        a3, M3 = direction_search(best, t3, seed + 101)
        if M3 < bestM:
            bestM = float(M3)
            best = normalize_sequence(a3)

    if allow_lp and n <= 1200 and budget_s - (time.monotonic() - t0) > 25.0:
        t4 = min(45.0, 0.10 * budget_s)
        a4, M4 = lp_blend(best, t4)
        if M4 < bestM:
            bestM = float(M4)
            best = normalize_sequence(a4)

        if budget_s - (time.monotonic() - t0) > 15.0:
            t5 = min(45.0, 0.10 * budget_s)
            a5, M5 = direction_search(best, t5, seed + 202)
            if M5 < bestM:
                bestM = float(M5)
                best = normalize_sequence(a5)

    if budget_s - (time.monotonic() - t0) > 4.0:
        a6, M6 = try_zero_mask(best)
        if M6 < bestM:
            bestM = float(M6)
            best = normalize_sequence(a6)

    return best, float(bestM)


def propose_candidate(seed=42, budget_s=1800):
    start = time.monotonic()
    deadline = start + max(120.0, float(budget_s) - 90.0)
    rng = np.random.default_rng(seed)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=float).ravel()
        if p.size > 0 and np.all(np.isfinite(p)):
            p = np.maximum(p, 0.0)
            s = float(p.sum())
            if s > 1e-12:
                parent = p / s
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(512, dtype=float) / 512.0
    parent = normalize_sequence(parent)

    best = parent.copy()
    best_n = len(best)
    bestM = float(fast_convolve(best).max())
    best_score = float(2.0 * float(best_n) * bestM)
    print(f"Initial parent: n={best_n}, score={best_score:.10f}")

    # Quick shaped inits at the parent length.
    if time.monotonic() < deadline:
        shaped = make_shaped_inits(best_n, np.random.default_rng(seed + 7))
        for cand in sample_list(shaped, 12, rng):
            cand = normalize_sequence(cand)
            m = float(fast_convolve(cand).max())
            sc = float(2.0 * float(len(cand)) * m)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = cand.copy()
                best_n = len(cand)
                bestM = m
                print(f"Shaped init improved: n={best_n}, score={best_score:.10f}")

    # Build a candidate length list and quickly evaluate parent resamples.
    base_ns = [
        best_n,
        int(0.5 * best_n),
        int(0.75 * best_n),
        best_n + 256,
        best_n + 512,
        best_n + 1024,
        int(2.0 * best_n),
        int(3.0 * best_n),
        1024,
        1536,
        2048,
        2560,
    ]
    ns = []
    for n in base_ns:
        n = int(n)
        if 128 <= n <= 2560:
            ns.append(n)
    ns = unique_preserve(ns)
    ns.sort()
    if len(ns) > 12:
        ns = ns[1::2] + [ns[-1]]
        ns = unique_preserve(ns)

    quick = []
    for n in ns:
        a = resample_sequence(best, n)
        m = float(fast_convolve(a).max())
        sc = float(2.0 * float(n) * m)
        quick.append((sc, n))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = a.copy()
            best_n = n
            bestM = m
            print(f"Resample improved: n={n}, score={best_score:.10f}")

    selected = [best_n]
    selected.extend([n for _, n in sorted(quick)][:3])
    selected.append(ns[-1])
    selected = unique_preserve(selected)

    # Main optimization across selected lengths.
    for idx, n in enumerate(selected):
        remaining = deadline - time.monotonic()
        if remaining < 70.0:
            break

        init = resample_sequence(best, n)
        initM = float(fast_convolve(init).max())

        shaped = make_shaped_inits(n, np.random.default_rng(seed + 1000 + idx))
        for cand in sample_list(shaped, 10, rng):
            cand = normalize_sequence(cand)
            m = float(fast_convolve(cand).max())
            if m < initM:
                initM = m
                init = cand.copy()

        sc = float(2.0 * float(n) * initM)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = init.copy()
            best_n = n
            bestM = float(initM)
            print(f"Init for n={n}: score={best_score:.10f}")

        if n == best_n:
            alloc = min(550.0, 0.35 * remaining)
        elif n > best_n:
            alloc = min(350.0, 0.25 * remaining)
        else:
            alloc = min(180.0, 0.15 * remaining)

        alloc = max(40.0, alloc)
        if alloc > remaining - 60.0:
            alloc = max(30.0, remaining - 60.0)

        print(f"Optimizing n={n}, allocated {alloc:.0f}s, current best {best_score:.10f}")
        a_opt, M_opt = optimize_at_length(init, alloc, seed + 2000 * (idx + 1),
                                          allow_lp=(n <= 1200))
        sc = float(2.0 * float(len(a_opt)) * float(M_opt))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = a_opt.copy()
            best_n = len(a_opt)
            bestM = float(M_opt)
            print(f"Improved after optimizing n={len(a_opt)}: score={best_score:.10f}")

    # Final polish on the current best.
    remaining = deadline - time.monotonic()
    if remaining > 60.0:
        print("Final polish on best length")
        a_opt, M_opt = optimize_at_length(best, min(250.0, 0.5 * remaining),
                                          seed + 999, allow_lp=(best_n <= 1200))
        sc = float(2.0 * float(len(a_opt)) * float(M_opt))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = a_opt.copy()
            best_n = len(a_opt)
            bestM = float(M_opt)
            print(f"Final polish improved: score={best_score:.10f}")

    # Try a few larger lengths from the current best if time remains.
    attempts = [
        best_n + 384,
        best_n + 640,
        best_n + 1024,
        max(best_n + 512, 1536),
        2048,
        2560,
    ]
    attempts = unique_preserve([n for n in attempts if best_n < n <= 2560])

    for n in attempts:
        remaining = deadline - time.monotonic()
        if remaining < 120.0:
            break

        init = resample_sequence(best, n)
        initM = float(fast_convolve(init).max())
        sc = float(2.0 * float(n) * initM)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = init.copy()
            best_n = n
            bestM = float(initM)
            print(f"Larger resample improved: n={n}, score={best_score:.10f}")

        alloc = min(220.0, 0.3 * remaining)
        alloc = max(30.0, alloc)
        if alloc > remaining - 40.0:
            alloc = max(20.0, remaining - 40.0)

        print(f"Trying larger n={n}, allocated {alloc:.0f}s")
        a_opt, M_opt = optimize_at_length(init, alloc, seed + 5000 + n, allow_lp=(n <= 1200))
        sc = float(2.0 * float(len(a_opt)) * float(M_opt))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = a_opt.copy()
            best_n = len(a_opt)
            bestM = float(M_opt)
            print(f"Larger n improved: n={len(a_opt)}, score={best_score:.10f}")

    # Tiny final second-order / subgradient attempts.
    remaining = deadline - time.monotonic()
    if remaining > 30.0 and HAVE_MINIMIZE:
        a_opt, M_opt = run_lbfgs(best, 0.05 / float(best_n), 80, min(60.0, 0.5 * remaining),
                                 0.0, rng)
        sc = float(2.0 * float(len(a_opt)) * float(M_opt))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = a_opt.copy()
            best_n = len(a_opt)
            bestM = float(M_opt)
            print(f"Tiny L-BFGS improved: score={best_score:.10f}")

    remaining = deadline - time.monotonic()
    if remaining > 30.0:
        a_opt, M_opt = direction_search(best, min(60.0, 0.5 * remaining), seed + 777)
        sc = float(2.0 * float(len(a_opt)) * float(M_opt))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = a_opt.copy()
            best_n = len(a_opt)
            bestM = float(M_opt)
            print(f"Direction polish improved: score={best_score:.10f}")

    best, bestM = try_zero_mask(best)
    best_score = float(2.0 * float(len(best)) * bestM)
    best_n = len(best)
    print(f"After zeroing: n={best_n}, internal score={best_score:.10f}")

    out = [float(x) for x in best]
    try:
        official = float(evaluate_sequence(out))
        if np.isfinite(official):
            print(f"Official score: {official:.10f}")
            return out
    except Exception:
        pass

    # Fallback should not be needed.
    fallback = np.ones(384, dtype=float) / 384.0
    return [float(x) for x in fallback]
