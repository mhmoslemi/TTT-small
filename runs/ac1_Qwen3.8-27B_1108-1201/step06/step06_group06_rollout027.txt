"""
Hybrid optimization using smooth max approximation with FFT-accelerated gradients,
adaptive temperature annealing, and LP-based refinement for minimizing peak
autocorrelation of non-negative sequences.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_max_grad(a, T):
    """Compute gradient of T*log(sum exp(conv(a)/T)) w.r.t. a.
    Returns (gradient, soft_max_value)."""
    n = len(a)

    scaled = conv / T
    scaled -= scaled.max()

"""
Hybrid optimization for minimum peak autocorrelation using FFT-accelerated
soft-max gradient descent with adaptive temperature annealing, active-set
LP refinement, and diverse multi-start exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_max_grad(a, T):
    """Compute gradient of T*log(sum exp(conv(a)/T)) via FFT. Returns (grad, obj)."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()
    # grad[m] = 2 * sum_k p[k] * a[k-m] (with bounds)
    # = 2 * cross_correlation(p, a)[m]
    # cross_corr[m] = conv(p, a[::-1])[m + n - 1]
    fft_size = 1 << (2 * n - 2).bit_length()
    P = np.fft.rfft(p, fft_size)
    A_rev = np.fft.rfft(a[::-1], fft_size)
    cross = np.fft.irfft(P * A_rev, fft_size)[:2 * n - 1]
    grad = 2.0 * cross[n - 1:2 * n - 1]
    return grad, max_c


def active_set_lp_step(a, n_constraints=40):
    """Solve active-set LP for FW-style direction. Returns direction g or None."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    
    # Get active set: top constraints
    active_idx = np.argsort(conv)[::-1][:n_constraints]
    
    # LP: min t s.t. 2*sum_i a[k-i]*g[i] - t <= M - conv[k] for k in active
    #                         t >= 0, sum(g)=1, g>=0
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_rows = []
    b_vals = []
    for k in active_idx:
        row = np.zeros(n + 1)
        for i in range(n):
            ki = k - i
            if 0 <= ki < n:
                row[i] = 2.0 * a[ki]
        row[-1] = -1.0
        A_rows.append(row)
        b_vals.append(M - conv[k])
    
    # t >= 0
    row = np.zeros(n + 1)
    row[-1] = -1.0
    A_rows.append(row)
    b_vals.append(0.0)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            g = np.maximum(res.x[:n], 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None

def line_search_f(f, g, max_alpha=1.0):

        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = mid

        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 2.0]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
        
        a = best_candidate
        
        # Occasional LP step for larger jumps
        if it % use_lp_every == 0 and time.monotonic() - t0 < time_limit - 5:
            g = active_set_lp_step(a, n_constraints=30)
            if g is not None:
                alpha, obj = line_search_f(a, g)
                if alpha > 0.01 and obj < best_val - 1e-12:
                    a = np.maximum((1 - alpha) * a + alpha * g, 0)
                    a = a / a.sum()
                    best_val = obj
        
        obj = fast_convolve(a).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            # Slightly decrease T on improvement
            T = max(T * 0.999, T_end)
        else:
            no_improve += 1
        
        if no_improve > 20:
            T = min(T * 1.05, T_start)
            lr *= 0.95
            no_improve = 0
        if no_improve > 100:
            T = max(T * 0.5, T_end)
            lr = lr_start * 0.7
        if no_improve > 300:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_start * 0.5
            lr = lr_start
            no_improve = 0
            prev_grad = None
            prev_step = None
        
        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    grad_opt: it={it}, T={T:.6f}, obj={obj:.8f}, best={best_obj:.8f}, t={elapsed:.0f}s")
    
    return best_a, best_obj


def fw_lp_optimize(a0, time_limit):
    """Frank-Wolfe with full LP oracle (slower but more powerful)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        # Try active set LP first (faster)
        g = active_set_lp_step(a, n_constraints=50)
        if g is None:
            g = full_lp_step(a)
        if g is None:
            m_exp = min(m_exp * 1.5, 8.0)
            if m_exp >= 8.0:
                rng = np.random.default_rng(it * 31 + 7)
                g = project_simplex(a + rng.standard_normal(n) * 0.005)
                m_exp = 1.0
            else:
                continue
        
        if np.max(np.abs(g - a)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 7919 + 13)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
                m_exp = 1.0
                no_improve = 0
            continue
        
        alpha, best_val = line_search_f(a, g)
        
        if alpha > 1e-10:
            a = np.maximum((1.0 - alpha) * a + alpha * g, 0.0)
            a = a / a.sum()
            obj = fast_convolve(a).max()
            m_exp = 1.0
            
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.2, 8.0)
        
        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    fw_lp: it={it}, obj={best_obj:.8f}, t={elapsed:.0f}s")
    
    return best_a, best_obj


def local_search(a0, time_limit, seed=42):
    """Local search with diverse perturbation strategies."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        improved = False
        
        # Multi-scale random perturbations
        for scale in [0.0002, 0.0005, 0.001, 0.003, 0.008]:
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(a + pert)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break
        
        # Coordinate moves (transfer mass between two positions)
        if not improved and it % 3 == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                for frac in [0.05, 0.1, 0.2, 0.4]:
                    candidate = a.copy()
                    amount = candidate[i] * frac
                    candidate[i] -= amount
                    candidate[j] += amount
                    candidate = np.maximum(candidate, 0)
                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break
        
        # Sinusoidal structured perturbation
        if not improved and it % 20 == 0:
            freq = rng.uniform(0.5, 10)
            phase = rng.uniform(0, 2 * np.pi)
            x = np.arange(n)
            pattern = np.sin(2 * np.pi * freq * x / n + phase) * 0.003
            candidate = project_simplex(a + pattern)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
        
        # Smoothed perturbation (Gaussian kernel)
        if not improved and it % 10 == 0:
            center = int(rng.integers(0, n))
            width = rng.uniform(5, 50)
            x = np.arange(n)
            bump = np.exp(-0.5 * ((x - center) / width) ** 2) * 0.005
            candidate = project_simplex(a + bump)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 500:
                break
        else:
            no_improve = 0
    
    return best_a, best_obj


def try_seq(a, label, best_ref):
    """Evaluate sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_ref
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_ref[1]:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return (a_norm.copy(), score)
    return best_ref


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry: multi-phase hybrid search combining fast gradient optimization,
    active-set LP, full LP, and local search for minimum peak autocorrelation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_start = time.monotonic()
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        parent = np.ones(1024) / 1024
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Using uniform init: n={len(parent)}, score={best_score:.8f}")
    
    n_parent = len(parent)
    
    # Phase 1: Quick multi-n exploration with gradient method
    print("\n=== Phase 1: Multi-n Gradient Exploration ===")
    t_phase = time.monotonic()
    
    candidate_ns = [512, 768, 1024, 1280, 1536, 2048]
    if n_parent not in candidate_ns:
        candidate_ns.append(n_parent)
    candidate_ns.sort()
    
    for n_try in candidate_ns:
        remaining = deadline - time.monotonic()
        if remaining < 200:
            break
        
        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try
        
        # Quick gradient optimization
        t_alloc = min(remaining * 0.06, 30)
        a_opt, _ = smooth_gradient_optimize(start, time_limit=t_alloc,
                                             T_start=0.005, T_end=0.0001,
                                             lr_start=0.03, use_lp_every=30)
        best_seq, best_score = try_seq(a_opt, f"grad_n{n_try}", (best_seq, best_score))
        
        # Quick active-set LP
        remaining2 = deadline - time.monotonic()
        if remaining2 > 150:
            t_alloc2 = min(remaining2 * 0.03, 15)
            a_opt2, _ = fw_lp_optimize(a_opt, time_limit=t_alloc2)
            best_seq, best_score = try_seq(a_opt2, f"lp_n{n_try}", (best_seq, best_score))
    
    # Also try diverse structural initializations
    n_best = len(best_seq)
    structures = []
    x = np.arange(n_best, dtype=np.float64)
    
    # Exponential decay
    for rate in [0.003, 0.008, 0.015]:
        exp_d = np.exp(-rate * x)
        structures.append(exp_d / exp_d.sum())
    
    # Bilinear (peak in middle)
    bilin = np.minimum(x + 1, n_best - x)
    structures.append(bilin / bilin.sum())
    
    # Gaussian
    for sigma in [0.15, 0.25, 0.35]:
        gauss = np.exp(-0.5 * ((x - n_best // 2) / (sigma * n_best)) ** 2)
        structures.append(gauss / gauss.sum())
    
    # Two bumps
    g1 = np.exp(-0.5 * ((x - 0.25 * (n_best - 1)) / (0.1 * n_best)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75 * (n_best - 1)) / (0.1 * n_best)) ** 2)
    structures.append((g1 + g2) / (g1 + g2).sum())
    
    for si, struct in enumerate(structures):
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break
        # Blend with best
        for alpha in [0.3, 0.6]:
            blend = project_simplex((1 - alpha) * best_seq + alpha * struct)
            t_alloc = min(remaining * 0.04, 20)
            a_opt, _ = smooth_gradient_optimize(blend, time_limit=t_alloc,
                                                 T_start=0.01, T_end=0.0001,
                                                 lr_start=0.03)
            best_seq, best_score = try_seq(a_opt, f"struct{si}_a{alpha}", (best_seq, best_score))
    
    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")
    
    # Phase 2: Deep gradient optimization on best
    print("\n=== Phase 2: Deep Gradient Optimization ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    
    if remaining > 200:
        t_alloc = min(remaining * 0.35, 500)
        print(f"  Deep gradient, n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = smooth_gradient_optimize(best_seq, time_limit=t_alloc,
                                             T_start=0.003, T_end=0.00005,
                                             lr_start=0.02, use_lp_every=40)
        best_seq, best_score = try_seq(a_opt, "deep_grad", (best_seq, best_score))
    
    # FW-LP refinement
    remaining = deadline - time.monotonic()
    if remaining > 150:
        t_alloc = min(remaining * 0.2, 200)
        print(f"  FW-LP, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, "fw_lp_deep", (best_seq, best_score))
    
    # Perturbed restarts
    for scale in [0.002, 0.005, 0.01, 0.02]:
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * scale)
        t_alloc = min(remaining * 0.1, 80)
        a_opt, _ = smooth_gradient_optimize(f_pert, time_limit=t_alloc,
                                             T_start=0.002, T_end=0.00005,
                                             lr_start=0.02, use_lp_every=30)
        best_seq, best_score = try_seq(a_opt, f"pert_{scale}", (best_seq, best_score))
    
    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")
    
    # Phase 3: Different temperature regimes
    print("\n=== Phase 3: Temperature Regimes ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    configs = [
        (0.001, 0.00001, 0.01),
        (0.0005, 0.000005, 0.005),
        (0.01, 0.001, 0.04),
        (0.003, 0.00002, 0.015),
    ]
    
    for ci, (T_s, T_e, lr) in enumerate(configs):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        a_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
        t_alloc = min(remaining * 0.2, 120)
        a_opt, _ = smooth_gradient_optimize(a_pert, time_limit=t_alloc,
                                             T_start=T_s, T_end=T_e,
                                             lr_start=lr, use_lp_every=25)
        best_seq, best_score = try_seq(a_opt, f"temp{ci}_T{T_s}", (best_seq, best_score))
    
    # Active-set LP intensive
    remaining = deadline - time.monotonic()
    if remaining > 80:
        t_alloc = min(remaining * 0.25, 100)
        print(f"  Active-set LP intensive, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, "active_lp", (best_seq, best_score))
    
    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")
    
    # Phase 4: Alternative n values
    print("\n=== Phase 4: Alternative Sizes ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    
    for factor in [0.5, 1.5, 2.0]:
        n_try = int(len(best_seq) * factor)
        if n_try < 256 or n_try > 4096:
            continue
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        start = np.interp(np.linspace(0, 1, n_try),
                          np.linspace(0, 1, len(best_seq)), best_seq)
        start = np.maximum(start, 0)
        start = start / start.sum()
        t_alloc = min(remaining * 0.2, 80)
        a_opt, _ = smooth_gradient_optimize(start, time_limit=t_alloc,
                                             T_start=0.005, T_end=0.0001,
                                             lr_start=0.03, use_lp_every=30)
        best_seq, best_score = try_seq(a_opt, f"alt_n{n_try}", (best_seq, best_score))
    
    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")
    
    # Phase 5: Final polish
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    # Local search
    if remaining > 60:
        t_alloc = min(remaining * 0.3, 120)

        a_opt, _ = local_search(best_seq, time_limit=t_alloc, seed=12345)

        a_opt, _ = smooth_gradient_optimize(best_seq, time_limit=t_alloc,
                                             T_start=0.0001, T_end=0.000005,
                                             lr_start=0.005, use_lp_every=100)
        best_seq, best_score = try_seq(a_opt, "fine_grad", (best_seq, best_score))

    remaining = deadline - time.monotonic()
    if remaining > 30:
        for _ in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            noise = rng.standard_normal(n_best) * 0.0003
            f_pert = project_simplex(best_seq + noise)

    remaining = deadline - time.monotonic()

    if remaining > 15:

    # Fallback
    if best_seq is None:

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
