Now, let me think about the differential evolution approach more carefully.

Differential Evolution (DE):
- Population of size NP (e.g., 20-30)
- Each generation:
  1. For each individual i:
     - Select three distinct random indices r1, r2, r3 (different from i)
     - Create trial: v = x_r1 + F * (x_r2 - x_r3)
     - Project v onto simplex
     - If f(v) <= f(x_i), replace x_i with v (binomial crossover with rate CR)
  2. Keep the best individual (elitism)

Parameters:
- F (scale factor): 0.5-1.0, can be adaptive
- CR (crossover rate): 0.5-1.0
- NP (population size): 5*n to 10*n (but we're limited by time)

For our problem, since we're on the simplex, the "DE vector" v might not be on the simplex, so we project it. This is fine.

Let me also add a "L-SHADE" style adaptive F and CR:
- F is sampled from a Cauchy distribution centered at F_p (which adapts)
- CR is sampled from a normal distribution centered at CR_p

This is the approach from LSHADE (a well-known DE variant).

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive differential evolution
with smoothed gradient local search.

Approach:
1. Diverse initialization with structured patterns
2. LSHADE-style differential evolution for global exploration  
3. Nesterov-accelerated smoothed gradient for local refinement
4. Multi-scale polish for final improvement
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_grad(p, a):
    """Compute 2 * H(p) @ a where H(p)_{m,j} = p[j+m]."""
    n = len(a)
    # Build index matrix: H[m,j] = p[j+m]
    J = np.arange(n)[:, None]  # (n, 1)
    M = np.arange(n)[None, :]  # (1, n)
    idx = J + M  # (n, n), values in [0, 2n-2]
    H = p[idx]
    return 2.0 * (H @ a)


def objective_from_conv(conv):
    """Extract objective-related quantity from convolution."""
    return conv.max()


def get_p(conv, T):
    """Get softmax weights for peaks."""
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()
    return p


def nesterov_smoothed_grad(a0, time_limit, T_init=0.001, lr_init=0.01):
    """Nesterov-accelerated smoothed gradient descent on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    y = a.copy()
    v = np.zeros(n)
    t_nest = 1.0
    lr = lr_init
    T = T_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Nesterov lookahead
        alpha = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / alpha
        y = (1.0 + beta) * a - beta * y
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y / ys
        t_nest += 1.0
        
        # Compute gradient
        conv = fast_convolve(y)
        p = get_p(conv, T)
        grad = hankel_grad(p, y)
        
        # Backtracking line search
        best_candidate = a
        best_val = conv.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
        
        a = best_candidate
        obj = best_val
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive parameters
        if no_improve > 50:
            T *= 1.05
            lr *= 0.95
        if no_improve > 150:
            T *= 0.7
        if no_improve > 300:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            y = a.copy()
            v = np.zeros(n)
            t_nest = 1.0
            T = T_init
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj


def gradient_descent_simple(a0, time_limit, lr=0.01, num_peaks=10):
    """Simple gradient descent using top-k peak gradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        
        # Top-k peaks
        k = min(num_peaks, len(conv))
        peaks = np.argpartition(conv, -k)[-k:]
        peak_vals = conv[peaks]
        # Weight by value
        w = peak_vals / peak_vals.sum()
        
        # Gradient: sum of gradients at peak lags
        grad = np.zeros(n)
        for i, pk in enumerate(peaks):
            if w[i] < 1e-14:
                continue
            # d(conv[pk])/da[m] = a[pk-m] + a[2m-pk] (with bounds)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            for m in range(js, je + 1):
                partner1 = pk - m
                partner2 = 2 * m - pk
                contrib = 0.0
                if 0 <= partner1 < n:
                    contrib += a[partner1]
                if 0 <= partner2 < n:
                    contrib += a[partner2]
                grad[m] += 2.0 * w[i] * contrib
        
        # Line search
        best_candidate = a
        best_val = conv.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
