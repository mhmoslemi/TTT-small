"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses coordinate descent, simulated annealing, and frequency-domain methods.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_score(a):
    """Compute the objective value (before the 2n factor)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def coordinate_descent(a0, time_limit, step_decay=0.999):
    """Coordinate descent: optimize one coordinate at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_score = fast_convolve(a).max()
    
    t0 = time.monotonic()
    step = 0.01
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_pos = int(np.argmax(conv))
        max_val = conv.max()
        
        # Find the top contributor to the peak
        # (a*a)[max_pos] = sum_i a[i] * a[max_pos - i]
        js = max(0, max_pos - n + 1)
        je = min(n - 1, max_pos)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[max_pos - idx]
        
        # Find the position to reduce (largest contributor)
        i_reduce = idx[np.argmax(contribs)]
        
        # Find a position to increase (one that doesn't contribute much)
        # Prefer positions that contribute little to the peak
        if len(idx) < n:
            # There are positions outside the affected range
            outside = np.setdiff1d(np.arange(n), idx)
            i_increase = outside[np.argmin(a[outside])]
        else:
            # All positions contribute; pick the one with smallest contribution
            i_increase = idx[np.argmin(contribs)]
        
        # Try transferring mass
        improved = False
        for frac in [0.01, 0.005, 0.002, 0.001, 0.05, 0.02]:
            delta = a[i_reduce] * frac
            if delta < 1e-10:

            candidate = a.copy()
            candidate[i_reduce] -= delta
            candidate[i_increase] += delta
            candidate = np.maximum(candidate, 0.0)
            cs = candidate.sum()

            new_conv = fast_convolve(candidate)

            if new_max < best_score - 1e-15:

                best_score = new_max
                best_a = a.copy()
                no_improve = 0
                improved = True

                no_improve = 0

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses targeted peak-shaving local search, simulated annealing with FFT-based

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score_fast(a):
    """Compute 2n * max(conv) / sum^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def directed_local_search(a0, time_limit, rng_seed=42):
    """
    Aggressive directed local search:
    - At each step, find the peak in the convolution
    - Identify top contributors to the peak
    - Transfer mass from contributors to "quiet" positions
    - Accept if the peak (or overall score) decreases
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    max_val = conv.max()
    best_a = a.copy()
    best_score = max_val

    rng = np.random.default_rng(rng_seed)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        max_pos = int(np.argmax(conv))
        max_val = conv[max_pos]

        # Compute contributions to the peak
        js = max(0, max_pos - n + 1)
        je = min(n - 1, max_pos)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[max_pos - idx]
        total_contrib = contribs.sum()

        if total_contrib < 1e-15:
            break

        # Find top contributors
        top_k = min(5, len(idx))
        top_order = np.argsort(contribs)[::-1][:top_k]
        top_positions = idx[top_order]

        # Find "quiet" positions (low contribution to all peaks)
        # Use the convolution values as a proxy: positions where
        # adding mass won't create new peaks
        # Simple heuristic: positions with low a[j] that are far from peak
        if len(idx) < n:
            outside = np.setdiff1d(np.arange(n), idx)
            # Pick from outside positions with lowest a
            quiet_candidates = outside[np.argsort(a[outside])[:min(10, len(outside))]]
        else:
            quiet_candidates = idx[np.argsort(contribs)[:min(10, len(idx))]]

        improved = False
        for i_red in top_positions:
            if a[i_red] < 1e-10:
                continue

                if quiet_pos == i_red:
                    continue

                for frac in [0.05, 0.02, 0.01, 0.005, 0.001]:

                        continue

                    candidate[i_red] -= delta
                    candidate[quiet_pos] += delta
                    candidate = np.maximum(candidate, 0.0)
                    cs = candidate.sum()
                    if cs < 1e-15:
                        continue
                    candidate = candidate / cs
                    new_conv = fast_convolve(candidate)
                    new_max = new_conv.max()

                    if new_max < best_score - 1e-14:
                        a = candidate
                        best_score = new_max
                        best_a = a.copy()
                        no_improve = 0
                        improved = True
                        break

            if no_improve % 20 == 0:

                scale = 0.002 + 0.008 * min(1.0, no_improve / 200.0)
                candidate = project_simplex(best_a + rng.standard_normal(n) * scale)
                new_conv = fast_convolve(candidate)
                new_max = new_conv.max()
                if new_max < best_score - 1e-14:

                    best_score = new_max

def simulated_annealing(a0, time_limit, T_init=0.0005, T_end=0.00001, rng_seed=123):
    """
    Simulated annealing with targeted moves.
    Move: pick two positions, transfer mass between them.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    current_cost = conv.max()
    best_a = a.copy()
    best_cost = current_cost

    rng = np.random.default_rng(rng_seed)
    t0 = time.monotonic()
    n_total_steps = max(1000, int(time_limit * 2000))  # ~2000 steps/sec target
    step = 0

    best_eval = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        progress = step / n_total_steps
        T = T_init * (T_end / T_init) ** progress

        # Choose move type
        r = rng.random()
        if r < 0.6:
            # Targeted: reduce mass at a peak contributor
            conv = fast_convolve(a)
            max_pos = int(np.argmax(conv))
            js = max(0, max_pos - n + 1)
            je = min(n - 1, max_pos)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[max_pos - idx]
            # Pick a top contributor
            top_frac = min(0.3, 0.05 + 0.3 * progress)
            rank = int(np.argmax(contribs > contribs.max() * (1 - top_frac) * rng.random() + contribs * rng.random() * top_frac))
            i_red = idx[rank]
            # Pick a quiet position
            j_inc = rng.integers(0, n)
            while j_inc == i_red:
                j_inc = rng.integers(0, n)
        else:
            # Random pair
            i_red = rng.integers(0, n)
            j_inc = rng.integers(0, n)
            while j_inc == i_red:
                j_inc = rng.integers(0, n)

        if a[i_red] < 1e-12:
            continue

        # Transfer amount (adaptive)
        transfer_scale = (0.01 + 0.04 * progress) * a[i_red]
        delta = transfer_scale * rng.uniform(0.3, 1.0)
        delta = min(delta, a[i_red] * 0.8)

        candidate = a.copy()
        candidate[i_red] -= delta
        candidate[j_inc] += delta
        candidate = np.maximum(candidate, 0.0)
        cs = candidate.sum()
        if cs < 1e-15:
            continue
        candidate = candidate / cs

        new_conv = fast_convolve(candidate)
        new_cost = new_conv.max()

        delta_cost = new_cost - current_cost
        if delta_cost < 0 or rng.random() < np.exp(-delta_cost / T):
            a = candidate
            current_cost = new_cost
            if new_cost < best_cost:
                best_cost = new_cost
                best_a = a.copy()

    return best_a, 2.0 * n * best_cost

def frequency_domain_optimize(a0, time_limit, n_modes=20):
    """
    Optimize in a truncated frequency domain representation.
    Parametrize a(x) = 1/n + sum_{m=1}^{M} (c_m cos(2πmx/n) + s_m sin(2πmx/n))
    and optimize the coefficients to minimize max convolution.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    M = min(n_modes, n // 2 - 1)

    x = np.arange(n, dtype=np.float64)

    # Get initial Fourier coefficients
    a_centered = a - 1.0 / n
    init_coeffs = np.zeros(2 * M)
    for m in range(1, M + 1):
        init_coeffs[2 * (m - 1)] = 2.0 * np.sum(a_centered * np.cos(2.0 * np.pi * m * x / n))
        init_coeffs[2 * (m - 1) + 1] = 2.0 * np.sum(a_centered * np.sin(2.0 * np.pi * m * x / n))

    def fourier_to_seq(coeffs):
        base = np.ones(n) / n
        for m in range(1, M + 1):
            base += (coeffs[2 * (m - 1)] * np.cos(2.0 * np.pi * m * x / n) +
                     coeffs[2 * (m - 1) + 1] * np.sin(2.0 * np.pi * m * x / n)) / n
        base = np.maximum(base, 0.0)
        ss = base.sum()
        if ss < 1e-15:
            return np.ones(n) / n
        return base / ss

    def objective(coeffs):
        seq = fourier_to_seq(coeffs)
        conv = fast_convolve(seq)
        return conv.max()

    best_a = a.copy()
    best_cost = objective(init_coeffs)

    rng = np.random.default_rng(777)
    t0 = time.monotonic()
    n_restarts = max(3, int(time_limit / 20))

    for r in range(n_restarts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        if r == 0:
            x0 = init_coeffs.copy()
        else:
            scale = 0.003 * (1 + 0.2 * r)
            x0 = init_coeffs + rng.standard_normal(2 * M) * scale

        try:
            res = minimize(objective, x0, method='Nelder-Mead',
                           options={'maxiter': 5000, 'xatol': 1e-12,
                                    'fatol': 1e-14, 'maxfev': 30000})
            if res.fun < best_cost:
                best_cost = res.fun
                best_a = fourier_to_seq(res.x)
        except Exception:
            pass

    return best_a, 2.0 * n * best_cost


def multi_peak_smoothing(a0, time_limit, T=0.001, lr=0.02):
    """
    Smoothed gradient descent focusing on multiple peaks simultaneously.
    Uses softmax-weighted subgradient to reduce all significant peaks.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_cost = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    v = np.zeros(n)
    mom = 0.9

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        # Softmax weighting over peaks
        scaled = conv / T
        scaled -= scaled.max()
        weights = np.exp(scaled)
        weights = weights / weights.sum()

        # Compute subgradient: for each non-zero weight position k,
        # the gradient w.r.t. a[j] is 2 * weight[k] * a[k-j] (for valid k-j)
        grad = np.zeros(n)
        # Only use top-weighted positions for efficiency
        top_positions = np.argsort(weights)[::-1][:min(50, n)]
        for k in top_positions:
            w = weights[k]
            if w < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w * a[k - idx]

        # Momentum update
        v = mom * v - lr * grad
        a_new = project_simplex(a + v)

        new_cost = fast_convolve(a_new).max()
        if new_cost < best_cost - 1e-14:
            best_cost = new_cost
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Don't update a if it didn't help

        # Adapt learning rate
        if no_improve > 30:
            lr *= 0.95
            T *= 1.02
        if no_improve > 60:
            T *= 0.9
        if no_improve > 200:
            rng = np.random.default_rng(it * 31 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = 0.02
            T = 0.001
            no_improve = 0
        if no_improve > 1000:
            break

    return best_a, 2.0 * n * best_cost

        if remaining > 5:
            t_alloc = min(remaining, 20)
            print("  Ultra-fine, t={:.0f}s".format(t_alloc))
            a_opt, _ = pair_swap_search(state[1], time_limit=t_alloc, rng_seed=888)
            try_and_update(a_opt, state, "ultra_fine")

    print("  Phase 5 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(384) / 384
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
