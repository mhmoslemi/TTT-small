```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS on softmax parameterization with annealed log-sum-exp (primary)
2. Projected subgradient with momentum (polish)
3. Frank-Wolfe with LP oracle (diversity)
4. Multi-n exploration and perturbation restarts
5. Active-set Newton refinement
Key innovation: unconstrained optimization via softmax parameterization avoids
projection artifacts and allows L-BFGS to work efficiently.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """FFT-based full auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_obj_grad(theta, T):
    """Compute objective and gradient for L-BFGS via softmax parameterization."""
    n = len(theta)
    # Stable softmax
    t = theta - theta.max()
    a = np.exp(t)
    Z = a.sum()
    a = a / Z

    # Auto-convolution
    conv_a = fast_convolve(a)
    m = len(conv_a)  # 2n-1

    # Log-sum-exp (smooth max approximation)
    scaled = conv_a / T
    max_s = scaled.max()
    exp_s = np.exp(scaled - max_s)
    lse_val = T * (max_s + np.log(exp_s.sum() + 1e-300))

    # Softmax weights for gradient
    p = exp_s / (exp_s.sum() + 1e-300)

    # h_i = sum_j a_j * p_{i+j}  (computed via convolution trick)
    h = np.convolve(a[::-1], p, mode='full')[n - 1:2 * n - 1]

    # c_bar = sum_k p_k * conv_a_k
    c_bar = np.dot(p, conv_a)

    # Gradient: 2 * a_i * (h_i - c_bar)
    grad = 2.0 * a * (h - c_bar)

    return lse_val, grad


def lbfsg_anneal(a0, time_limit, T_start=None, T_end=1e-4, cooling=0.75):
    """L-BFGS with annealed log-sum-exp temperature."""
    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 < 1e-15:
        a0 = np.ones(len(a0)) / len(a0)
        s0 = 1.0
    a0 = a0 / s0
    n = len(a0)

    # Initial theta from a0
    theta = np.log(np.maximum(a0, 1e-30))
    # Center theta (softmax is shift-invariant)
    theta -= theta.mean()

    conv0 = fast_convolve(a0)
    if T_start is None:
        T_start = max(conv0.max() * 0.5, 0.01)

    T = T_start
    best_a = a0.copy()
    best_obj = conv0.max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        t_round = min(remaining, 60.0)
        result = minimize(
            softmax_obj_grad,
            x0=theta,
            args=(T,),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': 2000, 'iprint': 0, 'maxls': 50}
        )

        if result.success or result.fun < float('inf'):
            # Convert back from softmax parameterization
            t = result.x - result.x.max()
            a_new = np.exp(t)
            a_new = a_new / a_new.sum()
            conv_new = fast_convolve(a_new)
            obj_new = conv_new.max()

            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                print(f"    L-BFGS T={T:.6f}: obj={obj_new:.10f} (improved)")
            else:
                print(f"    L-BFGS T={T:.6f}: obj={obj_new:.10f} (no improve, best={best_obj:.10f})")

            theta = result.x
            theta -= theta.mean()

        # Cool down
        T *= cooling
        if T < T_end:
            T = T_end

        # Check if we've converged at the final temperature
        if T <= T_end and (time.monotonic() - t0) > time_limit * 0.5:
            # Do one more round with even smaller T
            T = max(T * 0.5, 1e-6)
            result = minimize(
                softmax_obj_grad,
                x0=theta,
                args=(T,),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-12,
                         'maxfun': 1000, 'iprint': 0}
            )
            if result.success or result.fun < float('inf'):
                t = result.x - result.x.max()
                a_new = np.exp(t)
                a_new = a_new / a_new.sum()
                conv_new = fast_convolve(a_new)
                obj_new = conv_new.max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    print(f"    L-BFGS final T={T:.8f}: obj={obj_new:.10f}")
            break

    return best_a, best_obj


def pgd_multi_peak(a0, time_limit, top_k=20, lr_init=0.02, mom_init=0.9):
    """Projected gradient descent with momentum on top-k peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = mom_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        obj = conv.max()

        # Subgradient from top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * a[k - idx_arr]

        # Momentum step with backtracking
        v_new = mom * v - lr * sg
        best_candidate = a
        best_val = obj
        for trial_lr in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            candidate = project_simplex(a + trial_lr * v_new)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                v = v_new
                break

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.92

        if no_improve > 30:
            lr *= 0.93
            mom = min(0.97, mom + 0.003)
        if no_improve > 100:
            top_k = min(60, top_k + 10)
        if no_improve > 300:
            rng = np.random.default_rng((it * 48271 + 17) % (2**31 - 1))
            noise = 0.003 + 0.005 * min(1.0, (no_improve - 300) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init
            mom = mom_init
            top_k = 20
            no_improve = 0

    return best_a, best_obj


def softmax_smoothed_lbfsg(a0, time_limit, T_init=0.001):
    """Use L-BFGS directly on the simplex (via penalty) with softmax-smoothed objective."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    def obj_grad(x):
        ax = np.maximum(x, 1e-15)
        conv = fast_convolve(ax)
        scaled = conv / T_init
        max_s = scaled.max()
        exp_s = np.exp(scaled - max_s)
        p = exp_s / exp_s.sum()
        # Gradient of sum_k p_k * conv_k wrt a
        # d conv_k / da_i = a_{k-i} + a_{k-i} (symmetry) = 2 * a_{k-i} (if valid)
        # Actually for linear conv: d conv_k / da_i = a_{k-i} + a_{k-i}... 
        # conv_k = sum_j a_j a_{k-j}
        # d conv_k / da_i = a_{k-i} (from j=i term) + a_i (from k-j=i term)
        # = a_{k-i} + a_i (when both valid)
        # But by symmetry of auto-conv: d conv_k / da_i = 2*a_{k-i} if 0<=k-i<n
        # Wait: conv_k = sum_j a_j * a_{k-j}
        # d/d a_i: appears when j=i (gives a_{k-i}) and when k-j=i i.e. j=k-i (gives a_i)
        # So d conv_k / d a_i = a_{k-i} + a_i if both i and k-i are in [0,n-1]
        # But for auto-convolution, a_{k-i} appears when k-i is valid
        # And a_i appears when k-i is valid (same condition)
        # Actually: conv_k = sum_j a_j * a_{k-j}, the term with a_i appears as:
        #   j=i: a_i * a_{k-i}  (if 0<=k-i<n)
        #   k-j=i => j=k-i: a_{k-i} * a_i  (if 0<=k-i<n)
        # So d conv_k / d a_i = a_{k-i} + a_{k-i} = 2*a_{k-i} if 0<=k-i<n, else a_{k-i} if k-i==i (i.e., k=2i)
        # Actually if k-i = i, we get a_i * a_i and the derivative is 2*a_i, which equals 2*a_{k-i} = 2*a_i. ✓
        # So: d conv_k / d a_i = 2*a_{k-i} if 0 <= k-i < n, else 0
        # (The "else 0" covers the case where k-i is out of range, meaning a_i doesn't appear in conv_k)
        # Wait, that's not right either. Let me redo:
        # conv_k = sum_{j: 0<=j<n, 0<=k-j<n} a_j * a_{k-j}
        # a_i appears in this sum when:
        #   (i) j=i: need 0<=i<n (always) and 0<=k-i<n
        #   (ii) k-j=i: need j=k-i, so 0<=k-i<n and 0<=i<n (always)
        # In case (i): contributes a_i * a_{k-i}, derivative wrt a_i is a_{k-i}
        # In case (ii): contributes a_{k-i} * a_i, derivative wrt a_i is a_{k-i}
        # But if i == k-i (i.e., k=2i), these are the same term, counted once.
        # So: d conv_k / d a_i = 2*a_{k-i} if 0<=k-i<n and k!=2i
        #                           = a_i if k=2i (i.e., k-i=i)
        # Actually when k=2i: case (i) and (ii) give the same product a_i*a_i, so it appears once.
        # d(a_i^2)/d(a_i) = 2*a_i. And 2*a_{k-i} = 2*a_i when k-i=i. So the formula 2*a_{k-i} works!
        # Because when k-i=i, the term a_k a_i = a_i a_i appears once but its derivative is 2*a_i = 2*a_{k-i}. ✓
        
        # So: grad_i = sum_k p_k * 2 * a_{k-i} (where a_{k-i}=0 if k-i out of range)
        # This is: grad = 2 * convolve(p, a, 'full') reversed... 
        
        # More precisely: grad_i = 2 * sum_k p_k * a_{k-i}
        # Let l = k-i: grad_i = 2 * sum_l p_{l+i} * a_l
        # This is the cross-correlation of p and a at lag i.
        
        # Compute via convolution: h_i = sum_l p_{l+i} * a_l
        # h = convolve(a[::-1], p, 'full')[n-1:2n-1]
        h = np.convolve(ax[::-1], p, mode='full')[n - 1:2 * n - 1]
        grad = 2.0 * h
        
        # Add simple penalty to keep a >= 0 and sum(a) ~ 1
        penalty = 100.0 * np.sum(np.maximum(-x, 0) ** 2) + 10.0 * (np.sum(x) - 1.0) ** 2
        pen_grad = -200.0 * np.minimum(x, 0) + 20.0 * (np.sum(x) - 1.0)
        
        val = T_init * (max_s + np.log(exp_s.sum())) + penalty
        return val, grad + pen_grad

    result = minimize(obj_grad, x0=a, method='L-BFGS-B', jac=True,
                      options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-8,
                               'maxfun': 1000, 'iprint': 0})
    if result.success or result.fun < float('inf'):
        a_new = np.maximum(result.x, 0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a
    else:
        a_new = a

    return a_new, fast_convolve(a_new).max()


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        g = solve_lp_oracle(f, M * m_exp)
        if g is None:
            m_exp = min(m_exp * 1.5, 8.0)
            no_improve += 1
            if no_improve > 20:
                break
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 20:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 40:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.003 + 0.004 * min(1.0, no_improve / 80.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def active_set_refine(a0, time_limit, tol=1e-12, max_iter=500):
    """Active-set refinement: optimize using only active constraints."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1
        if it % 50 == 0:
            print(f"    AS-refine it={it}, obj={best_obj:.10f}")

        conv = fast_convolve(a)
        M = conv.max()
        active_peaks = np.where(conv >= M * (1.0 - 0.005))[0]

        if len(active_peaks) <= 1:
            break

        # Build subgradient with uniform weights on active peaks
        pg = np.zeros(n)
        for k in active_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            pg[idx_arr] += a[k - idx_arr]
        pg /= len(active_peaks)

        # Line search
        best_candidate = a
        best_val = M
        for step in [0.002, 0.005, 0.01, 0.02, 0.05]:
            candidate = project_simplex(a - step * pg)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
        else:
            if it > 50:
                break

    return best_a, best_obj


def nelder_mead_polish(a0, time_limit):
    """Nelder-Mead on a small perturbation subspace."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    # Use a low-dimensional perturbation: a_perturbed = normalize(a + alpha * v)
    # where v is a fixed random direction and alpha is the variable
    rng = np.random.default_rng(42)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    for trial in range(20):
        if time.monotonic() - (time_limit) > 0:
            break
        v = rng.standard_normal(n)
        v = v / np.linalg.norm(v)

        def nm_obj(alpha):
            candidate = np.maximum(a + alpha * v, 0)
            s = candidate.sum()
            if s < 1e-15:
                return 1e10
            candidate = candidate / s
            return fast_convolve(candidate).max()

        # Bounded minimization of 1D function
        from scipy.optimize import minimize_scalar
        result = minimize_scalar(nm_obj, bounds=(-0.05, 0.05), method='bounded',
                                 options={'xatol': 1e-8})
        if result.fun < best_obj:
            candidate = np.maximum(a + result.x * v, 0)
            s = candidate.sum()
            if s > 1e-15:
                candidate = candidate / s
                best_obj = result.fun
                best_a = candidate.copy()

    return best_a, best_obj


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def try_and_update(a, best_state, label=""):
    """Evaluate sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_state
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    best_seq, best_score = best_state
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return (best_seq, best_score)


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: L-BFGS with softmax parameterization + annealing as primary,
    with PGD, FW-LP, and active-set methods for diversity and refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: L-BFGS annealing on parent length (primary method) ===
    print("\n=== Phase 1: L-BFGS Annealing (Primary) ===")
    t_start = time.monotonic()

    remaining = deadline - time.monotonic() - 900
    if remaining > 30:
        t_alloc = min(remaining, 350)
        print(f"  L-BFGS anneal n={n_parent}, t={t_alloc:.0f}s")
        a_opt, _ = lbfsg_anneal(best_seq, time_limit=t_alloc, T_start=None, T_end=1e-5, cooling=0.7)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "lbfsg_main")

    # Second L-BFGS run with different parameters
    remaining = deadline - time.monotonic() - 800
    if remaining > 20:
        t_alloc = min(remaining, 150)
        print(f"  L-BFGS anneal v2 n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = lbfsg_anneal(best_seq, time_limit=t_alloc, T_start=0.01, T_end=5e-5, cooling=0.8)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "lbfsg_v2")

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 2: Multi-n exploration with L-BFGS ===
    print("\n=== Phase 2: Multi-n L-BFGS ===")
    t_start = time.monotonic()

    n_best = len(best_seq)
    candidate_ns = [n_best]
    for n_try in [n_best // 2, n_best * 2, 384, 512, 640, 768, 896, 1024, 1280, 1536]:
        if n_try not in candidate_ns and 128 <= n_try <= 3072:
            candidate_ns.append(n_try)

    for n_try in sorted(candidate_ns):
        remaining = deadline - time.monotonic() - 500
        if remaining < 30:
            break

        if n_try == n_best:
            start = best_seq.copy()
        else:
            start = interp_sequence(best_seq, n_try)

        t_alloc = min(remaining, 60)
        print(f"  L-BFGS n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = lbfsg_anneal(start, time_limit=t_alloc, T_start=None, T_end=1e-4, cooling=0.75)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"lbfsg_n{n_try}")

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 3: PGD + FW-LP polish ===
    print("\n=== Phase 3: PGD & FW-LP Polish ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    # PGD
    remaining = deadline - time.monotonic() - 350
    if remaining > 20:
        t_alloc = min(remaining, 150)
        print(f"  PGD n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = pgd_multi_peak(best_seq, time_limit=t_alloc, top_k=20, lr_init=0.015, mom_init=0.92)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "pgd")

    # FW-LP
    remaining = deadline - time.monotonic() - 250
    if remaining > 20:
        t_alloc = min(remaining, 120)
        print(f"  FW-LP n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "fw_lp")

    # Another L-BFGS round after PGD/FW
    remaining = deadline - time.monotonic() - 200
    if remaining > 20:
        t_alloc = min(remaining, 100)
        print(f"  L-BFGS round 3 n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = lbfsg_anneal(best_seq, time_limit=t_alloc, T_start=0.005, T_end=3e-5, cooling=0.8)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "lbfsg_r3")

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 4: Active-set refinement ===
    print("\n=== Phase 4: Active-Set Refinement ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    remaining = deadline - time.monotonic() - 150
    if remaining > 15:
        t_alloc = min(remaining, 100)
        print(f"  Active-set n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = active_set_refine(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "active_set")

    # L-BFGS with very small T (near-exact max)
    remaining = deadline - time.monotonic() - 100
    if remaining > 15:
        t_alloc = min(remaining, 60)
        print(f"  L-BFGS fine T n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = lbfsg_anneal(best_seq, time_limit=t_alloc, T_start=0.001, T_end=1e-5, cooling=0.85)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "lbfsg_fine")

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Perturbation restarts and Nelder-Mead ===
    print("\n=== Phase 5: Perturbation & Nelder-Mead ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    for scale in [0.002, 0.005, 0.01]:
        remaining = deadline - time.monotonic() - 100
        if remaining < 15:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc = min(remaining, 50)
        a_opt, _ = lbfsg_anneal(f_pert, time_limit=t_alloc, T_start=0.01, T_end=5e-5, cooling=0.75)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"pert_{scale}")

    # Nelder-Mead polish
    remaining = deadline - time.monotonic() - 50
    if remaining > 10:
        t_alloc = min(remaining, 40)
        print(f"  Nelder-Mead n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = nelder_mead_polish(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "nm_polish")

    # PGD with different parameters
    remaining = deadline - time.monotonic() - 30
    if remaining > 10:
        t_alloc = min(remaining, 30)
        a_opt, _ = pgd_multi_peak(best_seq, time_limit=t_alloc, top_k=50, lr_init=0.005, mom_init=0.95)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "pgd_fine")

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 6: Final polish ===
    print("\n=== Phase 6: Final Polish ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    # Final L-BFGS with smallest T
    remaining = deadline - time.monotonic() - 10
    if remaining > 10:
        t_alloc = min(remaining, 30)
        a_opt, _ = lbfsg_anneal(best_seq, time_limit=t_alloc, T_start=0.001, T_end=1e-6, cooling=0.9)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "final_lbfsg")

    # Symmetry check
    sym_final = (best_seq + best_seq[::-1]) / 2.0
    sym_final = sym_final / sym_final.sum()
    (best_seq, best_score) = try_and_update(sym_final, (best_seq, best_score), "final_sym")

    # One more tiny perturbation
    remaining = deadline - time.monotonic() - 5
    if remaining > 5 and len(best_seq) > 0:
        noise = rng.standard_normal(len(best_seq)) * 0.001
        f_pert = project_simplex(best_seq + noise)
        (best_seq, best_score) = try_and_update(f_pert, (best_seq, best_score), "tiny_pert")

    print(f"  Phase 6 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
