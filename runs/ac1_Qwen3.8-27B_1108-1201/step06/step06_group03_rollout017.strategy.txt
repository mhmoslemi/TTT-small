            if s_val < 0.01:
                fallback_seq = np.ones(768) * 0.002
            fallback_seq = fallback_seq / s_val
            fallback_seq = np.clip(fallback_seq, 0, 1000, out=None)
            fallback_score = float(evaluate_sequence(fallback_seq.tolist()))
            best_score = fallback_score
            best_seq = fallback_seq
        except Exception as e:
            print(f"Final fallback error: {e}")
            # Return the parent sequence if available
            try:
                if parent_arr is not None:
                    fallback_seq = parent_arr / parent_arr.sum()
                    fallback_seq = np.clip(fallback_seq, 0, 1000, out=None)
                    fallback_score = evaluate_sequence(fallback_seq.tolist())
                    if np.isfinite(fallback_score):
                        best_score = fallback_score
                        best_seq = fallback_seq
            except:
                pass
    
    # Print final result if available
    if best_seq is not None and np.isfinite(best_score):
        print(f"FINALSCORE (after adjusted length probing):{best_score:.10f} (lower is better, target=1.5027)")
        return best_seq.tolist()
    else:
        print("No valid sequence found, returning default fallback.")
        # Return the uniform fallback if parent is unavailable
        return [0.002] * 2048 if len(best_seq) == 0 else [0.002] * 768
