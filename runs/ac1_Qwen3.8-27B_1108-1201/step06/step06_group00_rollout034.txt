Let me also think about the problem from a different angle. The objective is:

If we think of a as a probability distribution, then conv(a,a) is the distribution of the sum of two independent draws from a. The max is the peak of this sum distribution. We want to spread out this sum distribution as much as possible.

This is related to the theory of "extendible" distributions and "sumsets". The more "spread out" the distribution, the flatter the sum distribution.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated
LSE-smoothed gradient descent on the probability simplex, with
adaptive beta scheduling, frequency-domain diversification,
and peak-shaving local search.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_weights(conv, beta):
    """Softmax weights for LSE smoothing of the max."""
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_bc / total


def compute_gradient_fft(a, weights):
    """Compute gradient of LSE(beta * conv(a,a)) efficiently.
    
    grad[m] = sum_k w[k] * (a[k-m] + a[m-k])
    where a[j] = 0 for j outside [0, n-1], k ranges over [0, 2n-2].
    
    Split into:
      Part A[m] = sum_j w[j+m] * a[j]  (cross-correlation)
      Part B[m] = sum_j w[m-j] * a[j]  (convolution)
    """
    n = len(a)
    conv_len = 2 * n - 1
    
    # Need FFT size >= conv_len + n - 1 = 3n - 2 for full linear conv
    N = 1 << (3 * n - 2 - 1).bit_length()
    
    # Part B: linear convolution of weights (len 2n-1) and a (len n)
    # B[m] = sum_j weights[m-j] * a[j] for m = 0, ..., n-1
    W = np.fft.rfft(weights, N)
    A = np.fft.rfft(a, N)
    conv_wa = np.fft.irfft(W * A, N)
    part_B = conv_wa[:n]
    
    # Part A: A[m] = sum_j weights[j+m] * a[j]
    # Using w_rev[k] = weights[conv_len-1-k], we have:
    # A[m] = (a * w_rev)[conv_len-1-m]
    w_rev = weights[conv_len-1::-1]  # reversed
    W_rev = np.fft.rfft(w_rev, N)
    conv_a_wrev = np.fft.irfft(A * W_rev, N)
    # A[m] = conv_a_wrev[conv_len-1-m] for m=0..n-1
    # So A[0] = conv_a_wrev[conv_len-1], A[1] = conv_a_wrev[conv_len-2], ...
    # A = conv_a_wrev[conv_len-n:conv_len][::-1]
    part_A = conv_a_wrev[conv_len - n: conv_len][::-1]
    
    return part_A + part_B


def lse_objective(conv, beta):
    """Compute LSE(beta * conv)."""
    bc = beta * conv
    return np.log(np.exp(bc - bc.max()).sum()) / beta + bc.max() / beta


def optimize_gradient_descent(f0, time_limit, beta_start=20.0, beta_end=10000.0,
                                verbose=True):
    """Main optimization: LSE-smoothed gradient descent on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = 1.0 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    patience = 100
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        it += 1
        
        # Compute weights and gradient
        conv = fast_convolve(f)
        w = compute_lse_weights(conv, beta)
        grad = compute_gradient_fft(f, w)
        
        # Project gradient onto tangent space of simplex
        # The tangent space of {x: sum(x)=1, x>=0} at a point where x>0 
        # is {v: sum(v)=0}. We subtract the mean to project.
        grad_tangent = grad - grad.mean()
        
        gnorm = np.linalg.norm(grad_tangent)
        if gnorm < 1e-15:
            no_improve += 100
            break
        
        # Backtracking line search
        step = lr
        improved = False
        best_step_val = best_obj
        
        for _ in range(12):
            f_new = f - step * grad_tangent / gnorm
            f_new = project_simplex(f_new)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new.copy()
                improved = True
                no_improve = 0
                best_step_val = obj_new
                break
            
            step *= 0.5
        
        if not improved:
            no_improve += 1
            
            if no_improve >= 20:
                lr *= 0.9
                beta = min(beta * 1.3, beta_end)
            
            if no_improve >= patience:
                # Diversify: frequency domain perturbation
                rng = np.random.default_rng(it * 7919 + 13)
                F = np.fft.rfft(best_f)
                mag = np.abs(F)
                phase = np.angle(F)
                scale = 0.01 + 0.02 * min(1.0, no_improve / (2 * patience))
                mag_pert = np.maximum(mag * (1.0 + rng.standard_normal(len(mag)) * scale), 0)
                F_new = mag_pert * np.exp(1j * phase)
                f_new = np.fft.irfft(F_new, n)
                f_new = np.maximum(f_new, 0)
                f_new = project_simplex(f_new)
                conv_new = fast_convolve(f_new)
                if conv_new.max() < best_obj:
                    best_obj = conv_new.max()
                    best_f = f_new.copy()
                    f = f_new.copy()
                else:
                    f = best_f.copy()
                no_improve = 0
                lr = 1.0 / np.sqrt(n)
                # Don't reset beta, keep it high to focus on peaks
        
        # Slowly increase beta
        if beta < beta_end:
            beta *= 1.003
            # But decay it back if we're making progress
            if no_improve == 0:
                beta = max(beta * 0.99, beta_start)
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter={it}, obj={best_obj:.10f}, beta={beta:.1f}, lr={lr:.6f}, "
                  f"no_improve={no_improve}, t={elapsed:.0f}s")
    
    return best_f, best_obj


def peak_shaving(f, time_limit=30.0, verbose=False):
    """Local search: identify top peaks and try to reduce them."""
    f = np.maximum(f, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        return f, float('inf')
    f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    # Get top peaks
    num_peaks = min(30, len(conv))
    peak_indices = np.argpartition(conv, -num_peaks)[-num_peaks:]
    
    while time.monotonic() - t0 < time_limit:
        # Pick a random top peak
        k = int(peak_indices[np.random.randint(num_peaks)])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        
        if lo >= hi:
            continue
        
        # Find the pair (i, k-i) that contributes most to this peak
        i_range = np.arange(lo, hi + 1)
        contributions = f[i_range] * f[k - i_range]
        i_max = i_range[np.argmax(contributions)]
        i_other = k - i_max
        
        if i_other < 0 or i_other >= n or i_other == i_max:
            continue
        
        # Try moving small amounts of mass from i_max to other positions
        # to reduce this peak without increasing others
        f_curr = best_f.copy()
        
        for attempt in range(5):
            # Choose a destination
            # Prefer positions far from the peak
            dist = np.abs(np.arange(n) - (k // 2))
            dist[i_max] = 0
            dist[i_other] = 0
            dest = np.random.choice(n, p=dist / dist.sum())
            
            if dest == i_max or dest == i_other:
                continue
            
            # Move a small fraction of mass
            frac = 0.05 + 0.1 * np.random.random()
            delta = f_curr[i_max] * frac
            if delta < 1e-10:
                continue
            
            f_trial = f_curr.copy()
            f_trial[i_max] -= delta
            f_trial[dest] += delta
            f_trial = np.maximum(f_trial, 0)
            f_trial /= f_trial.sum()
            
            conv_trial = fast_convolve(f_trial)
            obj_trial = conv_trial.max()
            
            if obj_trial < best_obj - 1e-12:
                best_obj = obj_trial
                best_f = f_trial.copy()
                f_curr = f_trial.copy()
                # Update peak indices
                peak_indices = np.argpartition(conv_trial, -num_peaks)[-num_peaks:]
                break
    
    if verbose:
        print(f"    Peak shaving: obj={best_obj:.10f}")
    
    return best_f, best_obj


def freq_perturb(f, rng, strength=0.03):
    """Perturb f in the frequency domain."""
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated LSE-smoothed gradient descent on the probability simplex
2. Adaptive peak tracking with automatic beta scheduling
3. Frequency-domain and time-domain diversification
4. Peak-shaving local search with adaptive peak selection
5. Multi-length exploration via interpolation
6. Nesterov-accelerated projection for faster convergence

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_weights(conv, beta):
    """Softmax weights for LSE smoothing of the max."""
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_bc / total


def compute_gradient_fft(a, weights):
    """Compute gradient of LSE(beta * conv(a,a)) efficiently via FFT.
    
    grad[m] = sum_k w[k] * (a[k-m] + a[m-k])
    Split: Part A[m] = sum_j w[j+m]*a[j], Part B[m] = sum_j w[m-j]*a[j]
    """
    n = len(a)
    conv_len = 2 * n - 1
    N = 1 << (3 * n - 2 - 1).bit_length()
    
    W = np.fft.rfft(weights, N)
    A = np.fft.rfft(a, N)
    
    # Part B: (w * a)[m] for m = 0..n-1
    conv_wa = np.fft.irfft(W * A, N)
    part_B = conv_wa[:n]
    
    # Part A: (a * w_rev)[conv_len-1-m] for m = 0..n-1
    w_rev = weights[conv_len - 1::-1]
    W_rev = np.fft.rfft(w_rev, N)
    conv_a_wrev = np.fft.irfft(A * W_rev, N)
    part_A = conv_a_wrev[n - 1: 2 * n - 1][::-1]
    
    return part_A + part_B

    it = 0

def peak_shaving(f, time_limit=30.0, verbose=False):
    """Local search: reduce top peaks by redistributing mass."""
    f = np.maximum(f, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        return f, float('inf')
    f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    num_peaks = min(40, len(conv))
    peak_indices = np.argpartition(conv, -num_peaks)[-num_peaks:]
    
    iterations = 0
    while time.monotonic() - t0 < time_limit:
        iterations += 1
        # Weight peaks by their value (focus on the largest)
        peak_vals = conv[peak_indices]
        probs = peak_vals ** 4
        probs = probs / probs.sum()
        k = int(peak_indices[rng.choice(len(peak_indices), p=probs)])
        
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo >= hi:
            continue
        
        i_range = np.arange(lo, hi + 1)
        contributions = f[i_range] * f[k - i_range]
        i_max = i_range[np.argmax(contributions)]
        i_other = k - i_max
        
        if i_other < 0 or i_other >= n or i_other == i_max:
            continue
        
        f_curr = f.copy()
        found = False
        for _ in range(8):
            # Choose destination: prefer positions that are "quiet" in the convolution
            # (low contribution to nearby peaks)
            dist = np.abs(np.arange(n) - min(i_max, i_other)).astype(float)
            dist[i_max] = 0.001
            dist[i_other] = 0.001
            # Also penalize positions near other peaks
            for pk in peak_indices[:10]:
                dist[lo:pk+hi+1] += 0.5
            dist = dist + 0.01
            dist /= dist.sum()
            dest = rng.choice(n, p=dist)
            
            if dest == i_max or dest == i_other:
                continue
            
            frac = 0.03 + 0.12 * rng.random()
            delta = f_curr[i_max] * frac
            if delta < 1e-12:
                continue
            
            f_trial = f_curr.copy()
            f_trial[i_max] -= delta
            f_trial[dest] += delta
            f_trial = np.maximum(f_trial, 0)
            fs = f_trial.sum()
            if fs < 1e-15:
                continue
            f_trial /= fs
            
            conv_trial = fast_convolve(f_trial)
            obj_trial = conv_trial.max()
            
            if obj_trial < best_obj - 1e-12:
                best_obj = obj_trial
                best_f = f_trial.copy()
                f_curr = f_trial.copy()
                peak_indices = np.argpartition(conv_trial, -num_peaks)[-num_peaks:]
                found = True
                break
        
        if not found and iterations % 500 == 0:
            # Occasional larger perturbation
            f_curr = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv_check = fast_convolve(f_curr)
            if conv_check.max() < best_obj:
                best_obj = conv_check.max()
                best_f = f_curr.copy()
                peak_indices = np.argpartition(conv_check, -num_peaks)[-num_peaks:]
    
    if verbose:
        print(f"    Peak shaving: obj={best_obj:.10f} after {iterations} iters")
    
    return best_f, best_obj


def freq_perturb(f, rng, strength=0.03):
    """Perturb f in the frequency domain."""
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  >>> NEW BEST: {score:.10f} (n={len(a)})")
    return score


def coordinate_descent_pass(f, time_limit=30.0, verbose=False):
    """Direct coordinate descent: try adjusting each pair to reduce max conv."""
    f = np.maximum(f, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        return f, float('inf')
    f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(9999)
    
    # Focus on the top few peaks
    top_k = min(10, len(conv))
    peak_idx = np.argsort(conv)[-top_k:]
    
    while time.monotonic() - t0 < time_limit:
        k = int(peak_idx[rng.randint(top_k)])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi - lo < 2:
            continue
        
        # Find the pair (i, k-i) with max product
        i_vals = np.arange(lo, hi + 1)
        prods = f[i_vals] * f[k - i_vals]
        idx = np.argsort(prods)[-3:]  # top 3 contributing pairs
        for ii in idx:
            i1 = int(i_vals[ii])
            i2 = k - i1
            if i2 < 0 or i2 >= n or i2 == i1:
                continue
            
            # Try moving mass from i1 to a random quiet position
            for _ in range(5):
                # Pick a destination far from both i1 and i2
                dist = np.maximum(np.abs(np.arange(n) - i1), np.abs(np.arange(n) - i2)).astype(float)
                dist[i1] = 0
                dist[i2] = 0
                if dist.sum() < 1:
                    continue
                dest = rng.choice(n, p=dist / dist.sum())
                
                for frac in [0.02, 0.05, 0.1, 0.15]:
                    delta = f[i1] * frac
                    if delta < 1e-12:
                        break
                    f_trial = f.copy()
                    f_trial[i1] -= delta
                    f_trial[dest] += delta
                    f_trial = np.maximum(f_trial, 0)
                    fs = f_trial.sum()
                    if fs < 1e-15:
                        break
                    f_trial /= fs
                    conv_t = fast_convolve(f_trial)
                    if conv_t.max() < best_obj - 1e-13:
                        best_obj = conv_t.max()
                        best_f = f_trial.copy()
                        f = f_trial.copy()
                        conv = conv_t
                        peak_idx = np.argsort(conv)[-top_k:]
                        break
    
    if verbose:
        print(f"    Coord descent: obj={best_obj:.10f}")
    return best_f, best_obj

    5. Final high-beta polish
