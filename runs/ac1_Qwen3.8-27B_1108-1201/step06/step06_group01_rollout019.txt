                delta = a[best_j_src] * best_frac
                a[best_j_src] -= delta
                a[best_j_dst] += delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_obj = fast_convolve(a).max()
                best_a = a.copy()
                improved = True
                stall = 0

        if not improved:
            stall += 1

        # Perturbation to escape
        if stall > 15:
            rng = np.random.default_rng((it * 48271 + 13) % (2**31))
            conv = fast_convolve(best_a)
            peak_k = int(np.argmax(conv))
            noise = rng.standard_normal(n) * 0.002
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            for j in range(js, je + 1):
                partner = peak_k - j
                if 0 <= partner < n:
                    noise[j] -= 0.002 * best_a[partner]
            a = project_simplex(best_a + noise)
            stall = 0

        if stall > 150:
            break

    return best_a, best_obj


def spectral_perturbation_search(f0, time_limit):
    """Use eigenvectors of the Hessian of the active peak to guide perturbations."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Build Hessian of q_k for the active peak
        # H[j,m] = 2 if j+m == k_peak else 0
        # This is an anti-diagonal matrix
        H = np.zeros((n, n))
        for j in range(n):
            m = k_peak - j
            if 0 <= m < n:
                H[j, m] = 2.0

        # Get top eigenvectors (most "negative curvature" directions)
        # Since H is sparse (anti-diagonal), use sparse eig
        try:
            eigvals, eigvecs = np.linalg.eigh(H)
        except Exception:
            break

        # Use the most negative eigenvector as a perturbation direction
        # (moving along -eigvec of most negative eigenvalue increases q_k the least)
        # Actually we want to DECREASE q_k, so move along the eigenvector
        # with the most negative eigenvalue
        idx_neg = np.argmin(eigvals)
        direction = eigvecs[:, idx_neg]

        # Also try the positive direction
        for sign in [1.0, -1.0]:
            for scale in [0.001, 0.003, 0.005, 0.01, 0.02]:
                a_test = a + sign * scale * direction
                a_test = project_simplex(a_test)
                obj = fast_convolve(a_test).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a_test.copy()
                    a = a_test.copy()
                    break

        # Also try random directions in the span of top eigenvectors
        if it % 5 == 0:
            rng = np.random.default_rng((it * 7919 + 3) % (2**31))
            k_eig = min(5, n)
            top_vecs = eigvecs[:, :k_eig]
            coeffs = rng.standard_normal(k_eig)
            pert_dir = top_vecs @ coeffs
            for scale in [0.001, 0.003, 0.005]:
                a_test = project_simplex(a + scale * pert_dir)
                obj = fast_convolve(a_test).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a_test.copy()
                    a = a_test.copy()
                    break

    return best_a, best_obj


def resample_sequence(a, n_new):
    """Resample a sequence to a new length using interpolation."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def micro_polish(f0, time_limit):
    """Very fine structured perturbation for final improvement."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    scales = [0.0001, 0.0002, 0.0005, 0.001]
    si = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        scale = scales[min(si, len(scales) - 1)]
        rng = np.random.default_rng((it * 2654435761 + 7) % (2**32))

        conv = fast_convolve(a)
        # Get top 3 peaks
        k_top = min(3, len(conv))
        peaks = np.argpartition(conv, -k_top)[-k_top:]

        tried = 0
        while tried < 15 and time.monotonic() - t0 < time_limit - 0.5:
            tried += 1
            noise = rng.standard_normal(n) * scale
            # Structured bias against all top peaks
            for kp in peaks:
                kp = int(kp)
                js = max(0, kp - n + 1)
                je = min(n - 1, kp)
                for j in range(js, je + 1):
                    partner = kp - j
                    if 0 <= partner < n:
                        noise[j] -= scale * 2.0 * a[partner]
            candidate = project_simplex(a + noise)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                break

        if it % 50 == 0:
            si = min(si + 1, len(scales) - 1)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search combining FW-LP, Adam on
    smoothed objective, peak-equalization coordinate descent, spectral
    perturbation search, and micro-polish.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t0_total = time.monotonic()

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t0_total
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return score

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(384) / 384
        try_seq(parent, "uniform_init")

    n_parent = len(parent)

    # === Phase 1: Quick exploration of n values (200s) ===
    print("\n=== Phase 1: Quick n-Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent, n_parent - 32, n_parent + 32,
                               n_parent - 64, n_parent + 64]))
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 1500]

    for n_try in candidate_ns:
        if time.monotonic() > deadline - 1200:
            break
        if n_try == n_parent:
            start = parent.copy()
        else:
            start = resample_sequence(parent, n_try)

        t_alloc = min(35, deadline - time.monotonic() - 1200)
        if t_alloc < 8:
            break
        print(f"  Quick FW n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(start, time_limit=t_alloc, lp_time=5)
        try_seq(a_opt, f"quick_n{n_try}")

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW-LP (500s) ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        t_alloc = min(350, deadline - time.monotonic() - 700)
        if t_alloc > 20:
            print(f"  Deep FW n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, lp_time=10)
            try_seq(a_opt, "deep_fw")

        # Perturbed restarts from best
        rng = np.random.default_rng(seed + 1)
        for scale in [0.003, 0.008, 0.015]:
            t_alloc = min(50, deadline - time.monotonic() - 550)
            if t_alloc < 10:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc, lp_time=8)
            try_seq(a_opt, f"fw_pert{scale}")

    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Adam Smoothed Descent (350s) ===
    print("\n=== Phase 3: Adam Smoothed Descent ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        configs = [
            {'T_init': 0.0005, 'lr_init': 0.015},
            {'T_init': 0.002, 'lr_init': 0.01},
            {'T_init': 0.001, 'lr_init': 0.025},
            {'T_init': 0.0003, 'lr_init': 0.012},
        ]
        for ci, cfg in enumerate(configs):
            t_alloc = min(85, deadline - time.monotonic() - 400)
            if t_alloc < 10:
                break
            print(f"  Adam T={cfg['T_init']}, lr={cfg['lr_init']}, t={t_alloc:.0f}s")
            a_opt, _ = adam_smoothed_descent(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"adam_{ci}")

    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Peak-Equalization CD (250s) ===
    print("\n=== Phase 4: Peak-Equalization CD ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(200, deadline - time.monotonic() - 250)
        if t_alloc > 10:
            print(f"  Peak CD, t={t_alloc:.0f}s")
            a_opt, _ = peak_equalization_cd(best_seq, time_limit=t_alloc, n_targets=3)
            try_seq(a_opt, "peak_cd")

        t_alloc = min(50, deadline - time.monotonic() - 200)
        if t_alloc > 10:
            a_opt, _ = peak_equalization_cd(best_seq, time_limit=t_alloc, n_targets=5)
            try_seq(a_opt, "peak_cd5")

    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Spectral Perturbation (150s) ===
    print("\n=== Phase 5: Spectral Perturbation ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(120, deadline - time.monotonic() - 150)
        if t_alloc > 10:
            print(f"  Spectral search, t={t_alloc:.0f}s")
            a_opt, _ = spectral_perturbation_search(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "spectral")

    print(f"  Phase 5 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Final Micro-Polish (100s) ===
    print("\n=== Phase 6: Final Micro-Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Micro polish
        t_alloc = min(50, deadline - time.monotonic() - 50)
        if t_alloc > 5:
            print(f"  Micro polish, t={t_alloc:.0f}s")
            a_opt, _ = micro_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "micro")

        # Final FW pass
        t_alloc = min(30, deadline - time.monotonic() - 25)
        if t_alloc > 5:
            print(f"  Final FW, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, lp_time=5)
            try_seq(a_opt, "final_fw")

        # One more Adam pass with very low temperature
        t_alloc = min(20, deadline - time.monotonic() - 10)
        if t_alloc > 5:
            a_opt, _ = adam_smoothed_descent(best_seq, time_limit=t_alloc,
                                             T_init=0.0001, lr_init=0.005)
            try_seq(a_opt, "final_adam")

    print(f"  Phase 6 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Final verification
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, total time={time.monotonic()-t0_total:.0f}s)")
    return [float(x) for x in best_seq]
