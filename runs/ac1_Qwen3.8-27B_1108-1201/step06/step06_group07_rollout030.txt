```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Fast SCA with precomputed LP structure and HiGHS solver
- Smoothed max (log-sum-exp) optimization via L-BFGS for post-SCA polish
- Peak-equalization coordinate descent
- Multi-n exploration with spectral initialization
- Diversified restarts with momentum-based subgradient
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_pattern(n):
    """Precompute sparsity pattern for SCA subproblem LP."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
        rows.append(k)
        cols.append(n)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_sca_subproblem(f, n, row_idx, col_idx):
    """
    Solve SCA subproblem: min M s.t. (f*a)[k] <= M for all k, sum(a)=1, a>=0.
    Returns optimal a or None.
    """
    c = np.zeros(n + 1)
    c[n] = 1.0

    k_arr = row_idx
    i_arr = col_idx
    vals = np.empty(len(row_idx), dtype=np.float64)
    mask = (i_arr < n)
    vals[mask] = f[k_arr[mask] - i_arr[mask]]
    vals[~mask] = -1.0

    A_ub = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success and res.x is not None:
            a_opt = np.maximum(res.x[:n], 0.0)
            if a_opt.sum() > 1e-15:
                return a_opt / a_opt.sum()
    except Exception:
        pass
    return None


def sca_optimize(f0, time_limit, verbose=False):
    """SCA optimization with fast LP subproblem solver."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    row_idx, col_idx = build_lp_pattern(n)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        g = solve_sca_subproblem(f, n, row_idx, col_idx)

        if g is None:
            no_improve += 5
            if no_improve > 20:
                break
            continue

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Try blending to ensure improvement
        best_candidate = f.copy()
        best_val = best_obj

        for alpha in [0.3, 0.5, 0.7, 1.0]:
            cand = (1.0 - alpha) * f + alpha * g
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_candidate = cand.copy()

        if best_val < best_obj - 1e-13:
            f = best_candidate
            best_obj = best_val
            best_f = f.copy()
            no_improve = 0
        else:
            # Try smaller steps
            improved = False
            for alpha in [0.1, 0.05, 0.02]:
                cand = (1.0 - alpha) * f + alpha * g
                cand = np.maximum(cand, 0.0)
                cs = cand.sum()
                if cs < 1e-15:
                    continue
                cand = cand / cs
                val = fast_convolve(cand).max()
                if val < best_obj - 1e-14:
                    f = cand
                    best_obj = val
                    best_f = f.copy()
                    no_improve = 0
                    improved = True
                    break
            if not improved:
                no_improve += 1
                if no_improve > 50:
                    # Perturb and try again
                    rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                    best_obj = fast_convolve(f).max()
                    no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    SCA it={it}, obj={best_obj:.10f}, time={time.monotonic()-t0:.1f}s")

    return best_f, best_obj


def smoothed_max_objective(a, epsilon):
    """Compute smoothed max of auto-convolution and its gradient."""
    conv = fast_convolve(a)
    # Log-sum-exp for numerical stability
    c_max = conv.max()
    log_sum = c_max + np.log(np.sum(np.exp(epsilon * (conv - c_max))))
    obj = (c_max + np.log(np.sum(np.exp(epsilon * (conv - c_max)))) / epsilon) - 2.0 * np.log(np.sum(a) + 1e-15)

    # Actually, let's just use the raw smoothed max (we'll handle normalization separately)
    obj = c_max + np.log(np.sum(np.exp(epsilon * (conv - c_max)))) / epsilon
    return obj


def smoothed_max_grad(a, epsilon):
    """Gradient of smoothed max of auto-convolution."""
    n = len(a)
    conv = fast_convolve(a)
    c_max = conv.max()
    weights = np.exp(epsilon * (conv - c_max))
    weights /= weights.sum()

    # Gradient of (a*a)[k] w.r.t. a[j] = a[k-j] + a[j-k] (if valid)
    grad = np.zeros(n)
    for k in range(len(conv)):
        w = weights[k]
        if w < 1e-12:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            grad[i] += w * a[k - i]
    return grad


def l_bfgs_polish(f0, time_limit, epsilon_init=50.0):
    """Polish using L-BFGS on the smoothed max objective over the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    epsilon = epsilon_init
    epoch = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5.0:
            break

        epoch += 1

        def objective(x):
            x = np.maximum(x, 0)
            s = x.sum()
            if s < 1e-15:
                return 1e10
            x = x / s
            conv = fast_convolve(x)
            cm = conv.max()
            return cm + np.log(np.sum(np.exp(epsilon * (conv - cm)))) / epsilon

        def gradient(x):
            x = np.maximum(x, 0)
            s = x.sum()
            if s < 1e-15:
                return np.zeros(len(x))
            x = x / s
            conv = fast_convolve(x)
            cm = conv.max()
            weights = np.exp(epsilon * (conv - cm))
            weights /= weights.sum()

            grad = np.zeros(len(x))
            for k in range(len(conv)):
                w = weights[k]
                if w < 1e-12:
                    continue
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                for i in range(i_start, i_end + 1):
                    grad[i] += w * 2.0 * x[k - i]
            return grad

        # Use L-BFGS with penalty for leaving simplex
        def penalized_obj(x):
            base = objective(x)
            # Penalty for negative values
            neg = np.maximum(-x, 0)
            pen = 100.0 * neg.sum()
            return base + pen

        def penalized_grad(x):
            base_g = gradient(x)
            neg_mask = x < 0
            base_g = base_g.copy()
            base_g[neg_mask] += 100.0
            return base_g

        res = minimize(penalized_obj, f, jac=penalized_grad, method='L-BFGS-B',
                       options={'maxiter': 200, 'maxfun': 500,
                                'gtol': 1e-8, 'iprint': 0})

        if res.success or res.fun < 1e10:
            cand = np.maximum(res.x, 0)
            cs = cand.sum()
            if cs > 1e-15:
                cand = cand / cs
                val = fast_convolve(cand).max()
                if val < best_obj - 1e-12:
                    best_obj = val
                    best_f = cand.copy()

        # Also try projected version
        cand_proj = project_simplex(res.x if hasattr(res, 'x') and res.x is not None else f)
        val_proj = fast_convolve(cand_proj).max()
        if val_proj < best_obj - 1e-12:
            best_obj = val_proj
            best_f = cand_proj.copy()

        # Decrease epsilon (make smoothed max closer to true max)
        epsilon *= 1.5
        if epsilon > 500.0:
            epsilon = 500.0

        # Check time for next epoch
        if time.monotonic() - t0 > time_limit - 10:
            break

        # If not improving, try different epsilon
        if val_proj >= best_obj - 1e-12 and val_proj >= objective(project_simplex(best_f)) - 1e-12:
            epsilon = max(epsilon_init, epsilon * 0.7)

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate, find the best value that
    minimizes the peak of the convolution.
    """
    a = np.maximum(f0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    epoch = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break
        epoch += 1

        improved = False
        # Random order for diversity
        rng = np.random.default_rng((epoch * 48271 + 7) % (2**31 - 1))
        order = rng.permutation(n)

        for idx in order:
            if time.monotonic() - t0 > time_limit - 1.0:
                break

            j = int(order[idx])
            # Current value
            aj = a[j]

            # The convolution as a function of a[j] (keeping others fixed):
            # (a*a)[k] = a[j]*a[k-j] + a[k-j]*a[j] + (other terms not involving a[j])
            #           = 2*a[j]*a[k-j] + c_k  (for k where both j and k-j are in range)
            # Actually: (a*a)[k] includes a[j]*a[k-j] if 0<=k-j<n
            # and a[k-j]*a[j] if 0<=j-(k-j)<n, i.e., j <= k-j < j+n, i.e., 2j-n < k <= 2j
            # Wait, let me be more careful.
            # (a*a)[k] = sum_i a[i]*a[k-i]
            # The terms involving a[j]: i=j gives a[j]*a[k-j] (if 0<=k-j<n)
            #                                    i=k-j gives a[k-j]*a[j] (if 0<=k-j<n)
            # If j == k-j (i.e., k=2j), these are the same term: a[j]^2
            # Otherwise they're two separate terms.
            # So (a*a)[k] = 2*a[j]*a[k-j] + c_k for k != 2j (if k-j in range)
            #              = a[j]^2 + c_k for k = 2j
            #              = c_k otherwise

            # For binary search on a[j], the peak is a piecewise linear function of a[j].
            # We want to find a[j] that minimizes the peak, subject to sum(a)=1.
            # If we change a[j] from aj to aj + delta, we need to adjust another coordinate.
            # Simplest: just find the optimal a[j] given the constraint that the sum stays 1.
            # But this requires adjusting other coordinates too.
            #
            # Alternative: parameterize as a'[j] = a[j] + t, a'[j2] = a[j2] - t
            # for some other index j2. This is complex.
            #
            # Simpler approach: just try moving mass from j to a "good" position.
            # Find the position that minimizes the increase in peak.

            if aj < 1e-10:
                continue

            # Try reducing a[j] and adding to the best other position
            reduce_frac = 0.05
            reduce_amt = aj * reduce_frac

            # Find best target: try a few candidates
            best_target = -1
            best_target_val = best_obj

            # Try positions far from j (to avoid creating new peaks near k=2j)
            candidates = []
            for offset in [n // 4, n // 3, n // 2, 2 * n // 3, 3 * n // 4]:
                t1 = (j + offset) % n
                t2 = (j - offset) % n
                candidates.extend([t1, t2])
            # Also try the position with smallest current value
            min_idx = np.argmin(a)
            candidates.append(min_idx)

            candidates = [c for c in candidates if c != j]
            if not candidates:
                continue

            for tc in candidates[:8]:
                a_new = a.copy()
                a_new[j] -= reduce_amt
                a_new[tc] += reduce_amt
                if a_new[j] < 0:
                    continue
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                val = fast_convolve(a_new).max()
                if val < best_target_val - 1e-12:
                    best_target_val = val
                    best_target = tc

            if best_target >= 0 and best_target_val < best_obj - 1e-12:
                a[j] -= reduce_amt
                a[best_target] += reduce_amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_obj = fast_convolve(a).max()
                if best_obj < best_a.max() * 0 + np.infty:  # always true
                    best_a = a.copy()
                improved = True
                no_improve = 0
            else:
                no_improve += 1

        if no_improve > 3 * n:
            # Perturb
            rng = np.random.default_rng((epoch * 99991 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            best_obj = fast_convolve(a).max()
            no_improve = 0

        if not improved and no_improve > 5 * n:
            break

    return best_a, best_obj


def momentum_subgradient(f0, time_limit, top_k=10):
    """Momentum-based subgradient with peak weighting."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = 0.02
    momentum = 0.9

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)
        max_val = conv.max()

        # Weighted peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            # Temperature-based weighting
            temp = 0.01 * max_val
            weights = np.exp(pv / temp)
            weights /= weights.sum()

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            w = weights[idx]
            if w < 1e-12:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += w * a[k - j]

        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = project_simplex(a_new)

        # Line search
        best_cand = a
        best_val = max_val
        for alpha in [0.3, 0.5, 0.7, 1.0]:
            cand = (1 - alpha) * a + alpha * a_new
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_cand = cand

        a = best_cand
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            lr *= 0.8
            momentum *= 0.95
        if lr < 0.001:
            lr = 0.02
            momentum = 0.9
            no_improve = 0

        if no_improve > 200:
            rng = np.random.default_rng((it * 31831 + 1) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            lr = 0.02
            momentum = 0.9
            no_improve = 0

    return best_a, best_obj


def spectral_init(n, rng):
    """
    Generate a good initialization using spectral methods.
    We want the auto-convolution to be as flat as possible.
    Strategy: choose a random phase in frequency domain, take inverse FFT,
    then take absolute value and project to simplex.
    """
    # Method 1: Random phase (gives "white noise" like sequence)
    fft_size = 1 << (2 * n - 1).bit_length()
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    A = np.exp(1j * phases)
    A[0] = 1.0
    # Make it Hermitian-symmetric
    A_full = np.zeros(fft_size, dtype=np.complex128)
    A_full[:fft_size // 2 + 1] = A
    A_full[fft_size // 2 + 1:] = np.conj(A[1:fft_size // 2 + 1][::-1])
    a = np.abs(np.fft.ifft(A_full)[:n])
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n

    # Method 2: Smooth (low-pass) sequence
    x = np.arange(n) / (n - 1.0)
    smooth = np.exp(-2.0 * (x - 0.5) ** 2 / 0.3)
    smooth = smooth + 0.3 * np.ones(n)
    smooth = smooth / smooth.sum()

    # Method 3: Bimodal (helps reduce center peak)
    g1 = np.exp(-((x - 0.25) ** 2) / 0.02)
    g2 = np.exp(-((x - 0.75) ** 2) / 0.02)
    bimodal = g1 + g2
    bimodal = bimodal / bimodal.sum()

    # Method 4: Anti-symmetric perturbation of uniform
    anti = np.ones(n) / n + 0.1 * np.sin(2 * np.pi * x)
    anti = np.maximum(anti, 0.01)
    anti = anti / anti.sum()

    # Method 5: Cosine bell (Hann window)
    hann = 0.5 * (1 + np.cos(2 * np.pi * x))
    hann = hann / hann.sum()

    return [a, smooth, bimodal, anti, hann]


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: multi-phase hybrid optimization.
    Phase 1: Multi-n SCA
    Phase 2: L-BFGS polish on best
    Phase 3: Coordinate descent
    Phase 4: Momentum subgradient
    Phase 5: Diversified restarts
    Phase 6: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(512) / 512

    n_parent = len(parent)
    print(f"Starting with n_parent={n_parent}")

    # === Phase 1: Multi-n SCA (400s) ===
    print("\n=== Phase 1: Multi-n SCA ===")
    t1 = time.monotonic()

    ns_to_try = sorted(set([n_parent, 384, 512, 640, 768, 896, 1024, 1280]))
    ns_to_try = [n for n in ns_to_try if 256 <= n <= 1536]

    for n_try in ns_to_try:
        remaining_after = deadline - time.monotonic()
        if remaining_after < 350:
            break
        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            start = start / start.sum()

        t_alloc = min(40, remaining_after - 320)
        if t_alloc < 5:
            break
        print(f"  SCA n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = sca_optimize(start, time_limit=t_alloc)
        try_seq(a_opt, f"sca_n{n_try}")

    print(f"  P1 done: best={best_score:.8f}, t={time.monotonic()-t1:.0f}s")

    # === Phase 2: L-BFGS Polish (400s) ===
    print("\n=== Phase 2: L-BFGS Polish ===")
    t2 = time.monotonic()

    if best_seq is not None:
        for eps in [30.0, 100.0]:
            remaining = deadline - time.monotonic()
            if remaining < 200:
                break
            t_alloc = min(120, remaining - 180)
            if t_alloc < 10:
                break
            print(f"  L-BFGS eps={eps}, t={t_alloc:.0f}s")
            a_opt, _ = l_bfgs_polish(best_seq, time_limit=t_alloc, epsilon_init=eps)
            try_seq(a_opt, f"lbfgs_eps{eps}")

    print(f"  P2 done: best={best_score:.8f}, t={time.monotonic()-t2:.0f}s")

    # === Phase 3: Coordinate Descent (200s) ===
    print("\n=== Phase 3: Coordinate Descent ===")
    t3 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 150:
            t_alloc = min(100, remaining - 140)
            if t_alloc > 10:
                print(f"  CoordDesc t={t_alloc:.0f}s")
                a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_alloc)
                try_seq(a_opt, "coord_desc")

    print(f"  P3 done: best={best_score:.8f}, t={time.monotonic()-t3:.0f}s")

    # === Phase 4: Momentum Subgradient (200s) ===
    print("\n=== Phase 4: Momentum Subgradient ===")
    t4 = time.monotonic()

    if best_seq is not None:
        for k in [5, 10, 20]:
            remaining = deadline - time.monotonic()
            if remaining < 100:
                break
            t_alloc = min(60, remaining - 90)
            if t_alloc < 10:
                break
            print(f"  MomSG k={k}, t={t_alloc:.0f}s")
            a_opt, _ = momentum_subgradient(best_seq, time_limit=t_alloc, top_k=k)
            try_seq(a_opt, f"mom_sg_k{k}")

    print(f"  P4 done: best={best_score:.8f}, t={time.monotonic()-t4:.0f}s")

    # === Phase 5: Diversified Restarts (250s) ===
    print("\n=== Phase 5: Diversified Restarts ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Try spectral inits at the best n
        inits = spectral_init(n_best, rng)
        for i, init in enumerate(inits[:3]):
            remaining = deadline - time.monotonic()
            if remaining < 100:
                break
            t_alloc = min(40, remaining - 90)
            if t_alloc < 5:
                break
            print(f"  Spectral init {i}, t={t_alloc:.0f}s")
            a_opt, _ = sca_optimize(init, time_limit=t_alloc)
            try_seq(a_opt, f"spectral_{i}")

        # Perturbed restarts
        scales = [0.005, 0.01, 0.02, 0.04]
        for scale in scales:
            remaining = deadline - time.monotonic()
            if remaining < 80:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(30, remaining - 70)
            if t_alloc < 5:
                break
            print(f"  Restart scale={scale}, t={t_alloc:.0f}s")
            a_opt, _ = sca_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"restart_{scale}")

    # Try different n values with spectral init
    for n_try in [1024, 1280, 1536]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        inits = spectral_init(n_try, rng)
        best_init = None
        best_init_score = float('inf')
        for init in inits[:3]:
            sc = fast_convolve(init).max()
            if sc < best_init_score:
                best_init_score = sc
                best_init = init
        if best_init is not None:
            t_alloc = min(25, remaining - 50)
            if t_alloc < 5:
                break
            print(f"  Spectral n={n_try}, t={t_alloc:.0f}s")
            a_opt, _ = sca_optimize(best_init, time_limit=t_alloc)
            try_seq(a_opt, f"spectral_n{n_try}")

    print(f"  P5 done: best={best_score:.8f}, t={time.monotonic()-t5:.0f}s")

    # === Phase 6: Final Polish (remaining) ===
    print("\n=== Phase 6: Final Polish ===")
    t6 = time.monotonic()

    if best_seq is not None:
        # Tight L-BFGS
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t_alloc = min(60, remaining - 20)
            if t_alloc > 10:
                print(f"  Final L-BFGS t={t_alloc:.0f}s")
                a_opt, _ = l_bfgs_polish(best_seq, time_limit=t_alloc, epsilon_init=200.0)
                try_seq(a_opt, "final_lbfgs")

        # Final SCA
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t_alloc = min(40, remaining - 15)
            if t_alloc > 5:
                print(f"  Final SCA t={t_alloc:.0f}s")
                a_opt, _ = sca_optimize(best_seq, time_limit=t_alloc)
                try_seq(a_opt, "final_sca")

        # Final momentum SGD
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_alloc = min(20, remaining - 10)
            if t_alloc > 5:
                print(f"  Final MomSG t={t_alloc:.0f}s")
                a_opt, _ = momentum_subgradient(best_seq, time_limit=t_alloc, top_k=5)
                try_seq(a_opt, "final_momsg")

    print(f"  P6 done: best={best_score:.8f}, t={time.monotonic()-t6:.0f}s")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
