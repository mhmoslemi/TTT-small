"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- AFrank-Wolfe with LP oracle (convolution matrix)
- Multi-start with diverse asymmetric initializations
- Peak-targeted coordinate descent for polishing
- Adaptive length exploration

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, s=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    m = np.sort(v)[::-1]
    cssv = np.cumsum(m)
    cond = m * np.arange(1, n + 1) > (cssv - s)
    if not cond.any():
        return np.full(n, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(f):
    """Build the (2n-1) x n convolution matrix A where (f*g)[k] = A[k,:] @ g."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def solve_lp_oracle(f, M):
    """Solve LP: max 1^T g s.t. A(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success and res.x is not None:
            result = np.maximum(res.x, 0.0)
            if result.sum() > 1e-15:
                return result
    except Exception:
        pass
    return None

def fw_step(f, M, line_search_points=25):
    """One Frank-Wolfe step: solve LP oracle, line search, update."""
    n = len(f)
    g = solve_lp_oracle(f, M)
    if g is None:
        return None, None
    g = g / g.sum()
    
    if np.max(np.abs(g - f)) < 1e-12:
        return f, M
    
    # Line search
    best_alpha = 0.0
    best_val = np.inf
    for alpha in np.linspace(0.01, 1.0, line_search_points):
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    if best_alpha < 1e-10:
        return f, M
    
    ft = (1.0 - best_alpha) * f + best_alpha * g
    ft = np.maximum(ft, 0.0)
    return ft / ft.sum(), best_val

def resample(a, new_n):
    """Resample sequence to new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Frank-Wolfe with LP oracle (convolution matrix structure)
- Multi-start with diverse asymmetric initializations (ramps, exponentials, etc.)
- Peak-targeted coordinate descent for final polishing
- Adaptive length exploration and perturbation-based restarts
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, s=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    m = np.sort(v)[::-1]
    cssv = np.cumsum(m)
    cond = m * np.arange(1, n + 1) > (cssv - s)
    if not cond.any():
        return np.full(n, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build the (2n-1) x n matrix A where (f*g)[k] = A[k,:] @ g."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M):
    """Solve LP: max 1^T g s.t. A(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15})
        if res.success and res.x is not None:
            result = np.maximum(res.x, 0.0)
            if result.sum() > 1e-15:
                return result
    except Exception:
        pass
    return None


def fw_single_step(f, M, n_lasso=25):
    """One FW step: LP oracle + line search. Returns (new_f, new_M) or (None, None)."""
    n = len(f)
    g = solve_lp_oracle(f, M)
    if g is None:
        return None, None
    g = g / g.sum()

    if np.max(np.abs(g - f)) < 1e-13:
        return None, None

    best_alpha = 0.0
    best_val = np.inf
    for alpha in np.linspace(0.01, 1.0, n_lasso):
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha

    if best_alpha < 1e-12:
        return None, None

    ft = (1.0 - best_alpha) * f + best_alpha * g
    ft = np.maximum(ft, 0.0)
    st = ft.sum()
    if st < 1e-15:
        return None, None
    return ft / st, best_val

    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Linear ramps
    ramp_inc = (x + 1.0)
    inits.append(ramp_inc / ramp_inc.sum())
    ramp_dec = (n - x)
    inits.append(ramp_dec / ramp_dec.sum())

    # Quadratic
    q = (x - n / 2.0 + 0.5) ** 2 + 1.0
    inits.append(q / q.sum())

    # Power laws from left
    for p in [1.5, 2.0, 3.0]:
        pw = (x + 1.0) ** p
        inits.append(pw / pw.sum())

    # Power laws from right
    for p in [1.5, 2.0]:
        pw = (n - x) ** p
        inits.append(pw / pw.sum())

    # Cosine taper
    ct = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    ct = np.maximum(ct, 0.05)
    inits.append(ct / ct.sum())

    # Gaussian
    g = np.exp(-0.5 * ((x - n / 2.0) / (n / 5.0)) ** 2)
    inits.append(g / g.sum())

    # Exponential from left
    el = np.exp(-x / (n / 4.0))
    inits.append(el / el.sum())

    # Exponential from right
    er = np.exp(-(n - 1 - x) / (n / 4.0))
    inits.append(er / er.sum())

    # V-shape (low in middle)
    vs = np.exp(-np.abs(x - n / 2.0) / (n / 3.0))
    inits.append(vs / vs.sum())

    # Sinusoidal
    sw = np.abs(np.sin(np.pi * x / n)) + 0.2
    inits.append(sw / sw.sum())

    # Two peaks
    g1 = np.exp(-0.5 * ((x - 0.25 * n) / (n / 12.0)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75 * n) / (n / 12.0)) ** 2)
    inits.append((g1 + g2) / (g1 + g2).sum())

    # Random Dirichlet
    for conc in [0.5, 1.0, 2.0, 5.0]:
        inits.append(rng.dirichlet(np.ones(n) * conc))

    # Parent resampled
    if parent is not None and len(parent) != n:
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        pr = np.interp(x_new, x_old, parent)
        pr = np.maximum(pr, 0)
        ps = pr.sum()
        if ps > 1e-15:
            inits.append(pr / ps)
    elif parent is not None:
        inits.append(parent.copy())

    return inits


def resample_seq(a, new_n):
    """Resample sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Quick multi-start FW exploration (300s)
    Phase 2: Deep FW on best + perturbed restarts (650s)
    Phase 3: Coordinate descent polishing (450s)
    Phase 4: Final micro-optimization (remaining)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def record(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-15:
            return
        a_norm = a / s
        a_list = [float(x) for x in a_norm]
        score = float(evaluate_sequence(a_list))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            record(parent, "parent")
            print(f"Parent loaded: n={len(parent)}, score={best_score:.10f}")
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        record(parent, "uniform768")

    # ================================================================
    # Phase 1: Multi-start exploration (280s)
    # ================================================================
    print("\n=== Phase 1: Multi-Start Exploration ===")
    t1_start = time.monotonic()
    t1_budget = 280

    n_parent = len(parent)
    lengths = sorted(set([
        max(256, n_parent - 128), n_parent - 64, n_parent,
        n_parent + 64, n_parent + 128,
        512, 640, 896, 1024
    ]))
    lengths = [l for l in lengths if 256 <= l <= 1536]

    for n_try in lengths:
        if time.monotonic() - t1_start > t1_budget:
            break

        inits = generate_inits(n_try, parent, rng)

        # Quick screen: rank by initial objective
        scored = []
        for init in inits:
            obj = fast_convolve(init).max()
            scored.append((obj, init))
        scored.sort()

        # Try top 5-8 initializations with short FW
        n_tries = min(7, len(scored))
        for si in range(n_tries):
            if time.monotonic() - t1_start > t1_budget - 3:
                break
            t_alloc = min(t1_budget - (time.monotonic() - t1_start) - 2, 35)
            if t_alloc < 5:
                break
            a_opt, _ = fw_optimize(scored[si][1], time_limit=t_alloc)
            record(a_opt, f"p1_n{n_try}_{si}")

    print(f"Phase 1 done: best={best_score:.10f} ({time.monotonic() - t1_start:.0f}s)")

    # ================================================================
    # Phase 2: Deep FW (650s)
    # ================================================================
    print("\n=== Phase 2: Deep FW Optimization ===")
    t2_start = time.monotonic()
    t2_budget = 650

    if best_seq is not None:
        n_best = len(best_seq)

        # Main deep run
        t_alloc = min(t2_budget - (time.monotonic() - t2_start), 350)
        if t_alloc > 30:
            print(f"  Deep FW: n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            record(a_opt, "deep_fw")

        # Perturbed restarts
        for scale in [0.015, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            t_rem = t2_budget - (time.monotonic() - t2_start)
            if t_rem < 15:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(t_rem - 5, 70)
            if t_alloc < 10:
                break
            print(f"  Perturb scale={scale}, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(f_pert, time_limit=t_alloc)
            record(a_opt, f"pert_{scale}")

        # Length variations
        for factor in [0.8, 0.9, 1.1, 1.2, 1.4]:
            t_rem = t2_budget - (time.monotonic() - t2_start)
            if t_rem < 20:
                break
            new_n = int(round(n_best * factor))
            if new_n < 256 or new_n > 1600:
                continue
            if new_n == n_best:
                continue
            a_res = resample_seq(best_seq, new_n)
            t_alloc = min(t_rem - 10, 50)
            if t_alloc < 10:
                break
            print(f"  Resample n={new_n}, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(a_res, time_limit=t_alloc)
            record(a_opt, f"res_n{new_n}")

    print(f"Phase 2 done: best={best_score:.10f} ({time.monotonic() - t2_start:.0f}s)")

    # ================================================================
    # Phase 3: Coordinate Descent (400s)
    # ================================================================
    print("\n=== Phase 3: CD Polishing ===")
    t3_start = time.monotonic()
    t3_budget = 400

    if best_seq is not None:
        for rd in range(4):
            t_rem = t3_budget - (time.monotonic() - t3_start)
            if t_rem < 15:
                break
            t_alloc = min(t_rem - 10, 100)
            if t_alloc < 10:
                break
            print(f"  CD round {rd}, t={t_alloc:.0f}s")
            a_opt, _ = cd_polish(best_seq, time_limit=t_alloc)
            record(a_opt, f"cd_{rd}")

            # Short FW to escape CD local optima
            t_rem2 = t3_budget - (time.monotonic() - t3_start)
            if t_rem2 > 20:
                t_alloc2 = min(t_rem2 - 5, 40)
                if t_alloc2 > 5:
                    a_opt2, _ = fw_optimize(best_seq, time_limit=t_alloc2)
                    record(a_opt2, f"fw_cd_{rd}")

    print(f"Phase 3 done: best={best_score:.10f} ({time.monotonic() - t3_start:.0f}s)")

    # ================================================================
    # Phase 4: Final micro-optimization (remaining)
    # ================================================================
    print("\n=== Phase 4: Final Polish ===")
    t4_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        while time.monotonic() < deadline - 10:
            t_rem = deadline - time.monotonic()
            if t_rem < 10:
                break

            # CD pass
            t_alloc = min(t_rem - 5, 80)
            if t_alloc > 8:
                a_opt, _ = cd_polish(best_seq, time_limit=t_alloc)
                record(a_opt, "fin_cd")

            # Tiny perturb + FW
            t_rem2 = deadline - time.monotonic()
            if t_rem2 > 15:
                noise = rng.standard_normal(n_best) * 0.0004
                start = project_simplex(best_seq + noise)
                t_alloc2 = min(t_rem2 - 5, 40)
                if t_alloc2 > 5:
                    a_opt, _ = fw_optimize(start, time_limit=t_alloc2)
                    record(a_opt, "fin_fw")

    print(f"Phase 4 done: best={best_score:.10f} ({time.monotonic() - t4_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_list = [float(x) for x in best_seq]
    fs = float(evaluate_sequence(final_list))
    if np.isfinite(fs):
        best_score = fs
    print(f"\nFINAL: {best_score:.10f} (n={len(best_seq)})")
    return final_list
