        if not accepted:
            lr *= 0.9

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
            lr = min(lr * 1.02, 0.05)
        else:
            stagnant += 1

        if stagnant > 20:
            temp *= 1.08
            lr *= 0.97
        if stagnant > 80:
            temp *= 0.5
            lr *= 1.2
            mom = min(mom + 0.02, 0.98)
        if stagnant > 250:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            temp = temp_init
            lr = lr_init
            mom = mom_init
            stagnant = 0

    return best_a, best_obj


def peak_coordinate_descent(a0, time_limit, initial_step=0.003):
    """
    Coordinate descent targeting top peaks by moving mass from
    peak-contributing positions to safe positions.
    """
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    step = initial_step
    no_improve = 0
    it = 0

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        n_peaks = min(5, len(conv))
        peak_idx = np.argsort(conv)[::-1][:n_peaks]

        # Build set of "dangerous" positions (contributing to top peaks)
        dangerous = set()
        for k in peak_idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                if a[i] * a[k - i] > 0.01 * conv[k]:
                    dangerous.add(i)
                    dangerous.add(k - i)

        improved = False
        for k in peak_idx:
            k = int(k)
            pairs = get_peak_contributors(a, k, n)
            for (_, i1, i2) in pairs[:6]:
                if a[i1] < 1e-8 and a[i2] < 1e-8:
                    continue
                try_sources = [i1, i2]
                for src in try_sources:
                    if a[src] < 1e-8:
                        continue
                    # Find safe target positions
                    rng = np.random.default_rng((it * 104729 + k * 31 + src * 7) % (2**31))
                    attempts = 0
                    while attempts < 20:
                        tgt = int(rng.integers(0, n))
                        attempts += 1
                        if tgt == src or tgt == (k - src if 0 <= k - src < n else -1):
                            continue
                        delta = a[src] * step
                        if delta < 1e-12:
                            break
                        candidate = a.copy()
                        candidate[src] -= delta
                        candidate[tgt] += delta
                        if candidate[src] < -1e-12:
                            continue
                        candidate = np.maximum(candidate, 0.0)
                        s = candidate.sum()
                        if s < 1e-15:
                            continue
                        candidate = candidate / s
                        val = fast_convolve(candidate).max()
                        if val < best_obj - 1e-15:
                            a = candidate
                            best_obj = val
                            best_a = a.copy()
                            improved = True
                            break

                if improved:
                    break
            if improved:
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            step *= 0.94
            if step < 1e-7:
                break
            if no_improve % 100 == 0:
                rng = np.random.default_rng(it * 48271)
                candidate = project_simplex(a + rng.standard_normal(n) * step * 3)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
                    no_improve = 0
            if no_improve > 500:
                break

    return best_a, best_obj


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def make_starts(n, parent, rng):
    """Generate diverse initializations."""
    starts = []
    if parent is not None:
        starts.append(resample(parent, n))

    x = np.arange(n, dtype=np.float64)

    # Uniform
    starts.append(np.ones(n) / n)

    # Cosine taper
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    w = np.maximum(w, 0.01)
    starts.append(w / w.sum())

    # Sinusoidal modulation
    s1 = 1.0 + 0.3 * np.sin(2.0 * np.pi * 3.0 * x / n)
    s1 = np.maximum(s1, 0.01)
    starts.append(s1 / s1.sum())

    # Two-lobe
    s2 = 1.0 + 0.4 * np.cos(2.0 * np.pi * 2.0 * x / n)
    s2 = np.maximum(s2, 0.01)
    starts.append(s2 / s2.sum())

    # Edge-weighted
    d = np.minimum(x, n - 1 - x)
    e = 1.0 / (d + 2.0)
    starts.append(e / e.sum())

    # Random Dirichlet
    starts.append(rng.dirichlet(np.ones(n) * 0.5))
    starts.append(rng.dirichlet(np.ones(n) * 0.2))

    return starts


def evaluate_safe(a):
    """Safely evaluate a sequence, returning inf on error."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase optimization to minimize
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15 or len(a) < 2:
            return float('inf')
        a_norm = a / s
        score = evaluate_safe(a_norm)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")

    n_parent = len(parent)

    # === Phase 1: Quick multi-length exploration (250s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_phase = time.monotonic()

    lengths = sorted(set([n_parent, 512, 640, 768, 896, 1024, 1152, 1280, 1536]))
    lengths = [l for l in lengths if 128 <= l <= 2048]

    for n_try in lengths:
        remaining = deadline - time.monotonic() - 1300
        if remaining < 30:
            break

        starts = make_starts(n_try, parent, rng)
        for si, start in enumerate(starts[:5]):
            remaining = deadline - time.monotonic() - 1250
            if remaining < 8:
                break
            t_alloc = min(remaining, 15)
            a_opt, _ = subgradient_descent(start, time_limit=t_alloc)
            try_seq(a_opt, f"sg_n{n_try}_s{si}")

    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep subgradient on best (500s) ===
    print("\n=== Phase 2: Deep Subgradient ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        # Multiple configs
        configs = [
            {'temp_init': 0.001, 'top_k': 20, 'lr_init': 0.02, 'mom_init': 0.9},
            {'temp_init': 0.005, 'top_k': 50, 'lr_init': 0.01, 'mom_init': 0.95},
            {'temp_init': 0.01, 'top_k': 100, 'lr_init': 0.005, 'mom_init': 0.93},
            {'temp_init': 0.002, 'top_k': 30, 'lr_init': 0.015, 'mom_init': 0.92},
        ]

        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 700
            if remaining < 20:
                break
            t_alloc = min(remaining, 120)
            # Perturb slightly for diversity
            pert = rng.standard_normal(n_best) * (0.001 + 0.002 * ci)
            start = project_simplex(best_seq + pert)
            print(f"  SG config {ci} (T={cfg['temp_init']}, k={cfg['top_k']}), t={t_alloc:.0f}s")
            a_opt, _ = subgradient_descent(start, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"deep_sg_{ci}")

        # Also try at different lengths
        for n_alt in [n_best + 128, n_best - 128, n_best * 2]:
            if n_alt < 128 or n_alt > 2048:
                continue
            remaining = deadline - time.monotonic() - 600
            if remaining < 20:
                break
            a_res = resample(best_seq, n_alt)
            t_alloc = min(remaining, 100)
            print(f"  SG at n={n_alt}, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_descent(a_res, time_limit=t_alloc,
                                           temp_init=0.003, top_k=30,
                                           lr_init=0.012, mom_init=0.93)
            try_seq(a_opt, f"sg_n{n_alt}")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Peak coordinate descent (400s) ===
    print("\n=== Phase 3: Peak Coordinate Descent ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 400
        if remaining > 30:
            t_alloc = min(remaining, 250)
            print(f"  Peak CD, t={t_alloc:.0f}s")
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_cd_1")

        # Another round with smaller steps
        if best_seq is not None:
            remaining = deadline - time.monotonic() - 250
            if remaining > 20:
                t_alloc = min(remaining, 100)
                print(f"  Peak CD (fine), t={t_alloc:.0f}s")
                a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc,
                                                   initial_step=0.001)
                try_seq(a_opt, "peak_cd_fine")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Alternating refinement rounds (300s) ===
    print("\n=== Phase 4: Alternating Refinement ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        round_i = 0
        while time.monotonic() < deadline - 100:
            round_i += 1
            remaining = deadline - time.monotonic() - 100
            if remaining < 15:
                break

            if round_i % 2 == 1:
                # Subgradient round
                noise = rng.standard_normal(n_best) * 0.001
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 60)
                a_opt, _ = subgradient_descent(start, time_limit=t_alloc,
                                               temp_init=0.001, top_k=15,
                                               lr_init=0.008, mom_init=0.95)
                try_seq(a_opt, f"alt_sg_{round_i}")
            else:
                # Coordinate descent round
                t_alloc = min(remaining, 40)
                a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc,
                                                   initial_step=0.001)
                try_seq(a_opt, f"alt_cd_{round_i}")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final micro-polish (remaining) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        # Very fine coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 10:
            t_alloc = min(remaining, 60)
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc,
                                               initial_step=0.0005)
            try_seq(a_opt, "final_cd")

        # One more subgradient pass with very small steps
        remaining = deadline - time.monotonic()
        if remaining > 10:
            t_alloc = min(remaining, 30)
            a_opt, _ = subgradient_descent(best_seq, time_limit=t_alloc,
                                           temp_init=0.0005, top_k=10,
                                           lr_init=0.003, mom_init=0.97)
            try_seq(a_opt, "final_sg")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = evaluate_safe(best_seq)

    # Final verification
    fs = evaluate_safe(best_seq)
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
