"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(w, a):
    """Compute W @ a where W_{i,j} = w[i+j]. w has length 2n-1, a has length n."""
    return np.correlate(w, a, mode='valid')


def solve_qp_fista(w, a_init, n, time_limit=5.0, max_iters=5000):
    """
    Solve min a^T W a s.t. sum(a)=1, a>=0 using FISTA.
    W is Hankel: W_{i,j} = w[i+j].
    """
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Estimate Lipschitz constant
    # L = 2 * max row sum of W
    # Row i sum = sum_{j=0}^{n-1} w[i+j]
    row_sums = np.array([np.sum(w[i:i+n]) for i in range(n)])
    L = 2.0 * max(row_sums.max(), 1e-10)
    
    y = a.copy()
    tk = 1.0
    t_new = 1.0
    
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute gradient at y: 2*W*y
        grad = 2.0 * hankel_mv(w, y)
        
        # FISTA update
        a_new = project_simplex(y - grad / L)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tk * tk))
        
        # Update y for next iteration
        beta = (tk - 1.0) / t_new
        y = a_new + beta * (a_new - a)
        
        a = a_new
        tk = t_new
    
    return a


def solve_qp_active(w, a_init, n, max_iters=200, time_limit=5.0):
    """
    Solve min a^T W a s.t. sum(a)=1, a>=0 using active set method.
    W is Hankel: W_{i,j} = w[i+j] where w has length 2n-1.
    """
    t0 = time.monotonic()
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    active = set(np.where(a > 1e-10)[0])
    if not active:
        active = set(range(n))

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if len(active) == 0:
            return np.ones(n) / n
        if len(active) == 1:
            a = np.zeros(n)
            a[list(active)[0]] = 1.0
            return a

        act = sorted(active)
        k = len(act)

        # Build submatrix W_AA
        W_AA = np.array([[w[act[i] + act[j]] for j in range(k)] for i in range(k)])

        # Regularize
        W_AA_reg = W_AA + 1e-12 * np.eye(k)

        # Solve using Cholesky (W is PSD)
        try:
            L_chol = np.linalg.cholesky(W_AA_reg)
            x = np.linalg.solve(L_chol.T, np.linalg.solve(L_chol, np.ones(k)))
        except np.linalg.LinAlgError:
            try:
                x = np.linalg.lstsq(W_AA_reg, np.ones(k), rcond=None)[0]
            except Exception:
                x = np.ones(k) / k

        x_sum = x.sum()
        if abs(x_sum) < 1e-14:
            x = np.ones(k) / k
            x_sum = 1.0

        a_act = x / x_sum

        # Remove most negative component
        min_val = float('inf')
        min_idx = -1
        for i, idx in enumerate(act):
            if a_act[i] < 0:
                if a_act[i] < min_val:
                    min_val = a_act[i]
                    min_idx = idx
        if min_idx >= 0:
            active.discard(min_idx)
            a_new = np.zeros(n)
            for i, idx in enumerate(act):
                if idx != min_idx:
                    a_new[idx] = max(0.0, a_act[i])
            s2 = a_new.sum()
            if s2 > 1e-15:
                a = a_new / s2
            continue

        # Form full solution
        a = np.zeros(n)
        for i, idx in enumerate(act):
            a[idx] = max(0.0, a_act[i])

        # Compute gradient 2*W*a
        Wa = hankel_mv(w, a)
        grad = 2.0 * Wa

        # Lagrange multiplier mu
        mu = np.mean([grad[idx] for idx in act])

        # Check KKT for inactive
        inactive = [j for j in range(n) if j not in active]
        if inactive:
            min_viol = 0.0
            min_j = -1
            for j in inactive:
                viol = grad[j] - mu
                if viol < min_viol:
                    min_viol = viol
                    min_j = j
            if min_viol < -1e-7:
                active.add(min_j)
                continue

        # Converged
        return a

    return a


def spectral_shape(a0, time_limit=5.0, max_iters=200):
    """
    Gerchberg-Saxton style spectral shaping.
    Alternates between:
    - Time domain: project to non-negative simplex
    - Frequency domain: flatten the power spectrum
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current auto-correlation
        c = fast_convolve(a)
        cmax = c.max()
        
        # Create target: cap the peaks
        threshold = cmax * 0.8  # Try to reduce peaks by 20%
        c_target = np.minimum(c, threshold)
        
        # Ensure c_target is valid (auto-correlation of some sequence)
        # Go to frequency domain
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        C = np.fft.rfft(c_target, fft_size)
        
        # Current phase
        A = np.fft.rfft(a, fft_size)
        phase = np.angle(A)
        
        # Target amplitude: sqrt of power spectrum of c_target
        amp_target = np.sqrt(np.maximum(C.real, 0.0))
        # Hmm, this isn't quite right. Let me think...
        # c = ifft(|A|^2), so |A|^2 = fft(c)
        # We want |A|^2 = fft(c_target), so |A| = sqrt(fft(c_target))
        # But fft(c_target) might have negative components...
        
        # Actually, fft(c) = |A|^2 >= 0 always. But fft(c_target) might not be.
        # Let me use a different approach:
        # Instead of modifying c, modify |A| directly.
        
        # Alternative: compute |A|, flatten it, keep phase
        abs_A = np.abs(A)
        
        # Flatten: replace with geometric mean or similar
        # Simple approach: reduce the peaks in |A|
        abs_max = abs_A.max()
        if abs_max > 0:
            # Soft flattening
            alpha = 0.3
            abs_new = abs_A ** (1 - alpha) * (abs_max ** alpha)
            # This reduces dynamic range
        else:
            abs_new = abs_A.copy()
        
        # New A with flattened amplitude and same phase
        A_new = abs_new * np.exp(1j * phase)
        
        # Back to time domain
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        # Keep the better one
        if obj < cmax:
            a = a_new
    
    return best_a


def coordinate_descent(a0, time_limit=5.0, max_sweeps=100):
    """
    Coordinate descent with exact line search for min max_k conv(a,a) s.t. sum(a)=1, a>=0.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Try changing a[i] while keeping sum = 1
            # If we increase a[i] by delta, we must decrease others by delta/(n-1)
            # Or: set a[i] = a[i] + delta, and scale the rest to maintain sum=1
            
            current_ai = a[i]
            if current_ai < 1e-15:
                continue
            
            # Binary search for optimal a[i]
            # The objective as a function of a[i] (with others scaled) is:
            # Let a' = a with a[i] changed to v, and all other entries scaled by (1-v)/(1-a[i])
            # This is complex. Simpler: try a few candidate values.
            
            # Try reducing a[i] slightly and redistributing
            for frac in [0.5, 0.7, 0.85, 0.95]:
                new_ai = current_ai * frac
                delta = current_ai - new_ai
                if delta < 1e-15:
                    continue
                
                a_trial = a.copy()
                a_trial[i] = new_ai
                # Redistribute delta to other positions proportionally
                others = np.arange(n) != i
                a_trial[others] += delta * a[others] / a[others].sum()
                
                if a_trial.min() < 0:
                    continue
                
                obj = fast_convolve(a_trial).max()
                if obj < best_obj - 1e-14:
                    a = a_trial
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
            
            # Also try increasing a[i] (stealing from others)
            for frac in [1.2, 1.5, 2.0]:
                new_ai = current_ai * frac
                delta = new_ai - current_ai
                if delta < 1e-15:
                    continue
                
                a_trial = a.copy()
                a_trial[i] = new_ai
                others = np.arange(n) != i
                # Steal proportionally from others
                steal = delta * a[others] / a[others].sum()
                if (a[others] - steal).min() < 0:
                    continue
                a_trial[others] = a[others] - steal
                
                obj = fast_convolve(a_trial).max()
                if obj < best_obj - 1e-14:
                    a = a_trial
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            break
    
    return best_a, best_obj


def qp_primal_dual(a0, time_limit, T_init=0.01, use_topk=False, topk=20,
                   blend=0.5, use_fista=True, verbose=False):
    """
    Primal-dual iteration with adaptive method selection.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.3:
            break

        conv = fast_convolve(a)
        cmax = conv.max()

        # Compute weights
        if use_topk:
            k_use = min(topk, len(conv))
            peaks = np.argpartition(conv, -k_use)[-k_use:]
            w = np.zeros(len(conv))
            w[peaks] = 1.0 / k_use
        else:
            scaled = conv / T
            scaled -= scaled.max()
            w = np.exp(scaled)
            w = w / w.sum()

        # Solve QP
        qp_time = min(3.0, remaining * 0.3)
        if use_fista and n <= 2048:
            a_qp = solve_qp_fista(w, a, n, time_limit=qp_time)
        else:
            a_qp = solve_qp_active(w, a, n, max_iters=100, time_limit=qp_time)

        # Blend
        a_new = (1.0 - blend) * a + blend * a_qp
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a_qp
        else:
            a_new = a_new / s_new

        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adapt temperature and blend
        if no_improve > 10:
            T *= 1.2
            blend = max(0.2, blend * 0.9)
        if no_improve > 40:
            T *= 0.7  # Re-focus
        if no_improve > 100:
            rng = np.random.default_rng((it * 104729 + 7) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            T = T_init
            blend = 0.5
            no_improve = 0
            it = 0  # Reset counter for deterministic seed

        if verbose and it % 50 == 0:
            print(f"    qp_pd it={it}, obj={best_obj:.10f}, T={T:.6f}, no_imp={no_improve}")

    return best_a, best_obj


def nesterov_subgrad(a0, time_limit, lr_init=0.005, momentum_init=0.9):
    """Nesterov-accelerated subgradient method."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = momentum_init
    y = a.copy()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        y = a + mom * v
        y = project_simplex(y)

        conv_y = fast_convolve(y)
        k_peak = int(np.argmax(conv_y))
        
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * y[k_peak - idx]

        v_new = project_simplex(y - lr * grad) - y
        a = project_simplex(y + v_new)
        v = v_new

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9999
        mom = min(0.99, mom + 0.00003)

        if no_improve > 200:
            lr *= 0.7
            mom *= 0.9
        if no_improve > 500:
            rng = np.random.default_rng((it * 48271 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            lr = lr_init * 0.3
            mom = momentum_init
            no_improve = 0

    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid of:
- FISTA and active-set QP for primal-dual peak-avoidance iteration

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(w, a):
    """Compute W @ a where W_{i,j} = w[i+j]. w has length 2n-1, a has length n."""
    return np.correlate(w, a, mode='valid')


def solve_qp_fista(w, a_init, n, time_limit=5.0, max_iters=10000):
    """Solve min a^T W a s.t. sum(a)=1, a>=0 using FISTA. W is Hankel."""
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Estimate Lipschitz constant: L = 2 * max row sum of W
    row_sums = np.array([np.sum(w[i:i + n]) for i in range(n)])
    L = 2.0 * max(row_sums.max(), 1e-10)
    # Use a safety factor to avoid overshooting
    L_eff = L * 1.01

    a_prev = a.copy()
    y = a.copy()
    tk = 1.0

    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        grad = 2.0 * hankel_mv(w, y)
        a_new = project_simplex(y - grad / L_eff)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tk * tk))
        beta = (tk - 1.0) / t_new
        y = a_new + beta * (a_new - a_prev)

        a_prev = a
        a = a_new
        tk = t_new

    return a


def solve_qp_active(w, a_init, n, max_iters=300, time_limit=5.0):
    """Solve min a^T W a s.t. sum(a)=1, a>=0 using active set with Cholesky."""
    t0 = time.monotonic()
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    active = set(np.where(a > 1e-10)[0])
    if not active:
        active = set(range(n))

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if len(active) == 0:
            return np.ones(n) / n
        if len(active) == 1:
            a = np.zeros(n)
            a[list(active)[0]] = 1.0
            return a

        act = sorted(active)
        k = len(act)
        if k > 500:
            # For large active sets, build matrix efficiently
            act_arr = np.array(act)
            idx = act_arr[:, None] + act_arr[None, :]
            W_AA = w[idx]
        else:
            W_AA = np.array([[w[act[i] + act[j]] for j in range(k)] for i in range(k)])

        W_AA_reg = W_AA + 1e-12 * np.eye(k)

        try:
            L_chol = np.linalg.cholesky(W_AA_reg)
            tmp = np.linalg.solve(L_chol, np.ones(k))
            x = np.linalg.solve(L_chol.T, tmp)
        except np.linalg.LinAlgError:
            try:
                x = np.linalg.lstsq(W_AA_reg, np.ones(k), rcond=None)[0]
            except Exception:
                x = np.ones(k) / k

        x_sum = x.sum()
        if abs(x_sum) < 1e-14:
            x = np.ones(k) / k
            x_sum = 1.0

        a_act = x / x_sum

        min_val = float('inf')
        min_idx = -1
        for i, idx in enumerate(act):
            if a_act[i] < 0:
                if a_act[i] < min_val:
                    min_val = a_act[i]
                    min_idx = idx
        if min_idx >= 0:
            active.discard(min_idx)
            a_new = np.zeros(n)
            for i, idx in enumerate(act):
                if idx != min_idx:
                    a_new[idx] = max(0.0, a_act[i])
            s2 = a_new.sum()
            if s2 > 1e-15:
                a = a_new / s2
            continue

        a = np.zeros(n)
        for i, idx in enumerate(act):
            a[idx] = max(0.0, a_act[i])

        Wa = hankel_mv(w, a)
        grad = 2.0 * Wa
        mu = np.mean([grad[idx] for idx in act])

        inactive_mask = np.ones(n, dtype=bool)
        inactive_mask[act] = False
        inactive = np.where(inactive_mask)[0]
        if len(inactive) > 0:
            grads_inact = grad[inactive]
            viol = grads_inact - mu
            min_pos = np.argmin(viol)
            if viol[min_pos] < -1e-7:
                active.add(inactive[min_pos])
                continue

        return a

    return a


def spectral_shape(a0, time_limit=5.0, max_iters=500):
    """Gerchberg-Saxton spectral shaping: flatten |FFT(a)| while keeping phase."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    alpha = 0.4  # Flattening strength

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        abs_A = np.abs(A)
        phase = np.angle(A)

        # Compress dynamic range of |A|
        abs_max = abs_A.max()
        if abs_max < 1e-15:
            break
        # Normalize then apply power law
        abs_norm = abs_A / abs_max
        abs_new = abs_norm ** (1.0 - alpha)
        # Scale back to preserve total energy (approximately)
        scale = np.sqrt(np.sum(abs_A ** 2) / max(np.sum(abs_new ** 2), 1e-30))
        abs_new = abs_new * scale

        A_new = abs_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            # Try weaker flattening
            alpha *= 0.8
            continue
        a_new = a_new / s

        obj = fast_convolve(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce flattening
            alpha = max(0.05, alpha * 0.95)

    return best_a


def coordinate_descent(a0, time_limit=5.0, max_sweeps=200):
    """Coordinate descent with proportional redistribution."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    others_sum = a.sum() - a  # sum minus each element

    t0 = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 0.1:
                break

            current_ai = a[i]
            if current_ai < 1e-14:
                continue

            others_s = a.sum() - current_ai
            if others_s < 1e-14:
                continue

            # Try reducing a[i] and redistributing proportionally
            for frac in [0.6, 0.8, 0.9]:
                new_ai = current_ai * frac
                delta = current_ai - new_ai
                if delta < 1e-14:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_ai
                a_trial[np.arange(n) != i] *= (1.0 + delta / others_s)
                if a_trial.min() < 0:
                    continue
                obj = fast_convolve(a_trial).max()
                if obj < best_obj - 1e-15:
                    a = a_trial
                    best_obj = obj
                    best_a = a.copy()
                    others_sum = a.sum() - a
                    improved = True
                    break

            # Try increasing a[i] (steal from others proportionally)
            if current_ai > 1e-8:
                for frac in [1.3, 1.8, 2.5]:
                    new_ai = current_ai * frac
                    delta = new_ai - current_ai
                    if delta < 1e-14:
                        continue
                    steal_frac = delta / others_s
                    if steal_frac >= 0.99:
                        continue
                    a_trial = a.copy()
                    a_trial[i] = new_ai
                    mask = np.arange(n) != i
                    a_trial[mask] = a[mask] * (1.0 - steal_frac)
                    if a_trial.min() < -1e-12:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    obj = fast_convolve(a_trial).max()
                    if obj < best_obj - 1e-15:
                        a = a_trial
                        best_obj = obj
                        best_a = a.copy()
                        others_sum = a.sum() - a
                        improved = True
                        break

        if not improved and sweep > 0:
            break

    return best_a, best_obj


def qp_primal_dual(a0, time_limit, T_init=0.01, use_topk=False, topk=20,

        a = a / s

    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:

        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.3:
            break

        cmax = conv.max()

        if use_topk:
            k_use = min(topk, len(conv))

            w = np.zeros(len(conv))
            w[peaks] = 1.0 / k_use
        else:
            scaled = conv / T
            scaled -= scaled.max()
            w = np.exp(scaled)

        if obj < best_obj - 1e-14:

            no_improve = 0

def nesterov_subgrad(a0, time_limit, lr_init=0.005, momentum_init=0.9, seed_offset=0):

    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = momentum_init
    y = a.copy()

    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        y = project_simplex(y)

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * y[k_peak - idx]

        v_new = project_simplex(y - lr * grad) - y

            best_obj = obj

            no_improve = 0

def smoothed_logsumexp_grad(a0, time_limit, T_init=0.001, lr_init=0.02):
    """Smoothed gradient descent using log-sum-exp surrogate for max."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init

    t0 = time.monotonic()

        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient of smoothed max: sum_k p[k] * grad(c[k])
        # grad(c[k])[i] = 2 * (Wa)[i] where W depends on k
        # More precisely: d(c[k])/da[i] = a[k-i] + a[k-i+n] (boundary terms)
        # Simplified: grad_i = 2 * sum_k p[k] * (correlation of a with shifted a)
        # = 2 * (hankel_mv of p weighted)

        # Backtracking line search

        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()

                best_val = val
                best_candidate = candidate
                break

        a = best_candidate

        if a is best_val:  # Python quirk check - won't trigger

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj

            no_improve += 1

        if no_improve > 20:

            print(f"  Smoothed grad T=0.001 t={t_alloc:.0f}s")
            a_opt, _ = smoothed_logsumexp_grad(best_seq, time_limit=t_alloc,
                                                T_init=0.001, lr_init=0.02)
            a_opt = np.maximum(a_opt, 0)
            ss = a_opt.sum()
            if ss > 1e-10:
                sc = float(evaluate_sequence((a_opt / ss).tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = (a_opt / ss).copy()
                    print(f"  NEW BEST [smooth]: {sc:.10f}")

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")

    # ========== Phase 4: Coordinate Descent + Fine QP (~250s) ==========
    print("\n=== Phase 4: Coordinate Descent + Fine QP ===")
    t4 = time.monotonic()

    if best_seq is not None:
        # Coordinate descent
        remaining = deadline - time.monotonic() - 100
        if remaining > 20:
            t_alloc = min(remaining, 70)
            print(f"  Coordinate descent t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            a_opt = np.maximum(a_opt, 0)
            ss = a_opt.sum()
            if ss > 1e-10:
                sc = float(evaluate_sequence((a_opt / ss).tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = (a_opt / ss).copy()
                    print(f"  NEW BEST [coord]: {sc:.10f}")

        # Fine QP with low temperature
        remaining = deadline - time.monotonic() - 80
        if remaining > 15:
            t_alloc = min(remaining, 55)
            print(f"  Fine QP T=0.001 t={t_alloc:.0f}s")
            a_opt, _ = qp_primal_dual(best_seq, time_limit=t_alloc, T_init=0.001,
                                       use_topk=False, topk=5, blend=0.3, use_fista=True)
            a_opt = np.maximum(a_opt, 0)
            ss = a_opt.sum()
            if ss > 1e-10:
                sc = float(evaluate_sequence((a_opt / ss).tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = (a_opt / ss).copy()
                    print(f"  NEW BEST [fine_qp]: {sc:.10f}")

        # Another QP round
        remaining = deadline - time.monotonic() - 60
        if remaining > 15:
            t_alloc = min(remaining, 45)
            print(f"  QP round 2 t={t_alloc:.0f}s")
            a_opt, _ = qp_primal_dual(best_seq, time_limit=t_alloc, T_init=0.005,
                                       use_topk=True, topk=8, blend=0.4, use_fista=True,
                                       seed_offset=99)
            a_opt = np.maximum(a_opt, 0)
            ss = a_opt.sum()
            if ss > 1e-10:
                sc = float(evaluate_sequence((a_opt / ss).tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = (a_opt / ss).copy()
                    print(f"  NEW BEST [qp_r2]: {sc:.10f}")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # ========== Phase 5: Final Polish (~100s) ==========
    print("\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Tiny perturbations with quick refinement
        for pi in range(5):
            remaining = deadline - time.monotonic() - 10
            if remaining < 10:
                break
            noise = 0.0003 + 0.0004 * pi
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise)
            t_alloc = min(remaining, 15)
            a_opt, _ = qp_primal_dual(f_pert, time_limit=t_alloc, T_init=0.002,
                                       use_topk=False, topk=10, blend=0.5,
                                       use_fista=True, seed_offset=200 + pi)
            a_opt = np.maximum(a_opt, 0)
            ss = a_opt.sum()
            if ss > 1e-10:
                sc = float(evaluate_sequence((a_opt / ss).tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = (a_opt / ss).copy()
                    print(f"  NEW BEST [pert{pi}]: {sc:.10f}")

        # Final Nesterov
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            t_alloc = min(remaining, 20)
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=t_alloc, lr_init=0.001)
            a_opt = np.maximum(a_opt, 0)
            ss = a_opt.sum()
            if ss > 1e-10:
                sc = float(evaluate_sequence((a_opt / ss).tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = (a_opt / ss).copy()
                    print(f"  NEW BEST [final_nest]: {sc:.10f}")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence([float(x) for x in best_seq]))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
