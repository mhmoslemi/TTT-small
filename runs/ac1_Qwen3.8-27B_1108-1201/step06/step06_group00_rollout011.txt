        step = 1.0 / (2.0 * np.dot(grad, y) + beta * conv_y.max() + 1e-8)
        step = min(step, 1.0)

        y_lse = np.log(np.exp(beta * conv_y - beta * conv_y.max()).sum()) + conv_y.max()

        a_new = None
        for _ in range(20):
            a_candidate = project_simplex(y - step * grad)
            conv_c = fast_convolve(a_candidate)
            lse_c = np.log(np.exp(beta * conv_c - beta * conv_c.max()).sum()) + conv_c.max()
            if lse_c <= y_lse + 1e-12 * step * np.dot(grad, a_candidate - y):
                a_new = a_candidate
                break
            step *= 0.5

        if a_new is None:
            a_new = project_simplex(y - step * grad)

        # Optional: QP inner solve (every use_qp_every iterations)
        if use_qp_every > 0 and it % use_qp_every == 0:
            a_qp = a_new.copy()
            for _ in range(qp_iters):
                if time.monotonic() - t0 > time_limit - 2:
                    break
                conv_qp = fast_convolve(a_qp)
                w_qp = lse_weights(conv_qp, beta)
                grad_qp = 2.0 * matvec_C(a_qp, w_qp, n)
                # Simple gradient step with small fixed step
                lr_qp = 1.0 / (2.0 * beta * conv_qp.max() + beta + 1)
                a_qp_new = project_simplex(a_qp - lr_qp * grad_qp)
                if fast_convolve(a_qp_new).max() < conv_qp.max():
                    a_qp = a_qp_new
                else:
                    break
            a_new = a_qp

        # FISTA momentum update
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + ((t_k - 1.0) / t_next) * (a_new - a_prev)
        y = np.maximum(y, 0.0)
        y_s = y.sum()
        if y_s > 1e-15:
            y /= y_s
        else:
            y = a_new.copy()

        a_prev = a_new.copy()
        t_k = t_next

        # Track best
        conv_a = fast_convolve(a_new)
        if conv_a.max() < best_obj - 1e-14:
            best_obj = conv_a.max()
            best_f = a_new.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    FISTA it={it}, obj={best_obj:.8f}, beta={beta:.0f}")
        else:
            no_improve += 1

        # FISTA restart if stuck
        if no_improve > 200:
            y = best_f.copy()
            a_prev = best_f.copy()
            t_k = 1.0
            no_improve = 0

    return best_f, best_obj


def local_search(f0, time_limit, rng, n_peaks=15, verbose=False):
    """
    Aggressive local search: peak-aware coordinate moves.
    For each top peak, try moving mass from contributing positions to random targets.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    t0 = time.monotonic()
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    total_moves = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 5000:
            # Reset with small perturbation
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            conv = fast_convolve(f)
            best_obj = conv.max()
            best_f = f.copy()
            no_improve = 0
            continue

        # Get top peaks
        conv = fast_convolve(f)
        k_top = min(n_peaks, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]

        # Pick a random peak
        k = int(peak_idx[rng.integers(len(peak_idx))])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo >= hi:
            no_improve += 1
            continue

        # Find the contributing pair with max product
        i_range = np.arange(lo, hi + 1)
        products = f[i_range] * f[k - i_range]
        j_max = i_range[np.argmax(products)]
        j_other = k - j_max

        if j_other < 0 or j_other >= n or j_other == j_max:
            no_improve += 1
            continue

        # Try to move mass from j_max to a random position
        # The position should be far from both j_max and j_other to reduce the peak
        # But also shouldn't create a new peak
        n_attempts = 5
        improved = False

        for _ in range(n_attempts):
            j_dest = int(rng.integers(n))
            if j_dest == j_max or j_dest == j_other:
                continue
            # Also avoid positions that would create a peak at j_dest + j_other or j_dest + j_max
            # (rough check)
            delta = f[j_max] * (0.01 + 0.04 * rng.random())
            if delta < f[j_max] * 0.5:
                f_trial = f.copy()
                f_trial[j_max] -= delta
                f_trial[j_dest] += delta
                f_trial = np.maximum(f_trial, 0.0)
                f_trial /= f_trial.sum()
                conv_trial = fast_convolve(f_trial)
                obj_trial = conv_trial.max()
                total_moves += 1
                if obj_trial < best_obj - 1e-15:
                    best_obj = obj_trial
                    best_f = f_trial.copy()
                    f = f_trial.copy()
                    no_improve = 0
                    improved = True
                    break
            no_improve += 1

        if not improved:
            no_improve += 1

    return best_f, best_obj


def freq_perturb(f, rng, strength=0.03):
    """Perturb f in the frequency domain for diversification."""
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def try_sequence(a, best_tracker):
    """Evaluate and update best tracker."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.10f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Main optimization:
    Phase 1: Multi-length FISTA exploration
    Phase 2: Focused FISTA refinement with QP inner solves
    Phase 3: Aggressive local search
    Phase 4: Final perturbations and polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(seed)
    best_tracker = [float('inf'), None]

    print(f"Budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768.0
        print("Using uniform 768 as parent")
        try_sequence(parent, best_tracker)

    n_parent = len(parent)

    # === Phase 1: Multi-length FISTA exploration ===
    print("\n=== Phase 1: Multi-length FISTA exploration ===")
    t1_end = time.monotonic() + (deadline - time.monotonic()) * 0.3
    t1_start = time.monotonic()

    n_candidates = sorted(set([
        n_parent,
        max(256, n_parent - 16),
        min(2000, n_parent + 16),
        max(256, n_parent - 32),
        min(2000, n_parent + 32),
        max(256, int(n_parent * 0.85)),
        min(2000, int(n_parent * 1.15)),
    ]))
    n_candidates = [x for x in n_candidates if 200 <= x <= 2500]

    for nc in n_candidates:
        t_remaining = t1_end - time.monotonic()
        if t_remaining < 30:
            break
        t_alloc = min(120, t_remaining * 0.4)
        if t_alloc < 20:
            break

        if nc == n_parent:
            f_init = parent.copy()
        else:
            f_init = interpolate_to_length(parent, nc)

        print(f"  FISTA n={nc}, t={t_alloc:.0f}s")
        f_opt, obj = fista_optimize(f_init, time_limit=t_alloc,
                                     beta_start=10.0, beta_end=2000.0,
                                     use_qp_every=0, verbose=True)
        try_sequence(f_opt, best_tracker)

    # === Phase 2: Focused FISTA with QP inner solves ===
    print(f"\n=== Phase 2: FISTA + QP refinement ===")
    t2_end = time.monotonic() + (deadline - time.monotonic()) * 0.45

    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)

    # Run multiple FISTA passes with increasing complexity
    passes = [
        {"beta_start": 20, "beta_end": 3000, "use_qp_every": 0, "qp_iters": 0},
        {"beta_start": 50, "beta_end": 5000, "use_qp_every": 5, "qp_iters": 20},
        {"beta_start": 100, "beta_end": 8000, "use_qp_every": 3, "qp_iters": 30},
    ]

    for i, params in enumerate(passes):
        t_remaining = t2_end - time.monotonic()
        if t_remaining < 30:
            break
        t_alloc = min(180, t_remaining * 0.5)
        if t_alloc < 20:
            break

        current_best = best_tracker[1] if best_tracker[1] is not None else current_best
        n_cur = len(current_best)
        print(f"  Pass {i+1}: n={n_cur}, beta={params['beta_start']}->{params['beta_end']}, "
              f"qp_every={params['use_qp_every']}, t={t_alloc:.0f}s")

        f_opt, obj = fista_optimize(current_best, time_limit=t_alloc,
                                     beta_start=params["beta_start"],
                                     beta_end=params["beta_end"],
                                     use_qp_every=params["use_qp_every"],
                                     qp_iters=params["qp_iters"],
                                     verbose=True)
        try_sequence(f_opt, best_tracker)

    # === Phase 3: Aggressive local search ===
    print(f"\n=== Phase 3: Local search ===")
    t3_end = deadline - 60
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)

    t_alloc = t3_end - time.monotonic()
    if t_alloc > 20:
        print(f"  Local search: n={n_cur}, t={t_alloc:.0f}s")
        f_opt, obj = local_search(current_best, time_limit=t_alloc, rng=rng,
                                   n_peaks=20, verbose=True)
        try_sequence(f_opt, best_tracker)

        # Second local search pass if time allows
        t_remaining = t3_end - time.monotonic()
        if t_remaining > 20:
            current_best = best_tracker[1] if best_tracker[1] is not None else current_best
            rng2 = np.random.default_rng(seed + 12345)
            print(f"  Local search round 2: n={len(current_best)}, t={t_remaining:.0f}s")
            f_opt, obj = local_search(current_best, time_limit=t_remaining * 0.8,
                                       rng=rng2, n_peaks=10)
            try_sequence(f_opt, best_tracker)

    # === Phase 4: Final perturbations and polish ===
    print(f"\n=== Phase 4: Final polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)

    t_rem = deadline - time.monotonic()
    if t_rem > 30:
        # Brief FISTA with very high beta
        t_alloc = min(60, t_rem * 0.4)
        if t_alloc > 10:
            print(f"  High-beta FISTA: n={n_cur}, t={t_alloc:.0f}s")
            f_opt, _ = fista_optimize(current_best, time_limit=t_alloc,
                                       beta_start=500.0, beta_end=20000.0,
                                       use_qp_every=10, qp_iters=20)
            try_sequence(f_opt, best_tracker)

    # Quick perturbation attempts
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t_rem = deadline - time.monotonic()
    if t_rem > 10:
        print(f"  Final perturbations: t={t_rem:.0f}s")
        for attempt in range(30):
            if time.monotonic() > deadline - 5:
                break
            # Frequency perturbation
            f_pert = freq_perturb(current_best, rng, strength=0.005 + 0.002 * attempt)
            try_sequence(f_pert, best_tracker)

            # Small time-domain perturbation
            f_pert2 = project_simplex(current_best + rng.standard_normal(n_cur) * 0.001)
            try_sequence(f_pert2, best_tracker)

            # If we found something better, update
            if best_tracker[1] is not None and best_tracker[1] is not current_best:
                current_best = best_tracker[1]
                n_cur = len(current_best)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    print("WARNING: Using uniform fallback")
    fallback = np.ones(768) / 768.0
    return [float(x) for x in fallback]
