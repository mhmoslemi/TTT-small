import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_lp_sparsity(n):
    """Build sparse structure for LP constraints."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

def solve_sca_step(f, M, n, row_idx, col_idx):
    """Solve one SCA LP step."""
    k_arr = row_idx
    i_arr = col_idx
    vals = f[k_arr - i_arr]
    A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2*n-1, n))
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, 
                      method='highs', options={'presolve': True, 'time_limit': 10})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None

def sca_optimize_v2(f0, time_limit, verbose=False):
    """Improved SCA with line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    
    row_idx, col_idx = build_lp_sparsity(n)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        g = solve_sca_step(f, M, n, row_idx, col_idx)
        
        if g is None:
            M *= 1.02
            no_improve += 1
            if no_improve > 30:
                M = fast_convolve(f).max() * 1.01
                no_improve = 0
            continue
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                # Perturb
                rng = np.random.default_rng((it * 7919) % (2**31))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                M = fast_convolve(f).max() * 1.01
                no_improve = 0
            continue
        
        # Line search: find best alpha mixing f and g
        best_val = M
        best_alpha = 0.0
        best_ft = f.copy()
        
        # Binary search for best alpha
        lo, hi = 0.0, 1.0
        for _ in range(15):
            mid = (lo + hi) / 2
            ft = (1 - mid) * f + mid * g
            ft = np.maximum(ft, 0)
            ft = ft / ft.sum()
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
                best_ft = ft.copy()
                lo = mid
            else:
                hi = mid
        
        if best_val < best_obj - 1e-12:
            f = best_ft
            best_obj = best_val
            best_f = f.copy()
            M = best_val * 0.999
            no_improve = 0
        else:
            no_improve += 1
            M = max(M * 0.997, best_val * 0.999)
        
        if no_improve > 50:
            rng = np.random.default_rng((it * 31831) % (2**31))
            scale = 0.005 + 0.01 * min(1.0, no_improve / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max() * 1.01
            no_improve = 0
        
        if verbose and it % 50 == 0:
            print(f"    SCA it={it}, obj={best_obj:.10f}, M={M:.10f}, t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_sparsity(n):
    """Build sparse structure for SCA LP constraints."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_sca_lp(f, M, n, row_idx, col_idx, time_limit_lp=5.0):
    """Solve one SCA LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    k_arr = row_idx
    i_arr = col_idx
    vals = f[k_arr - i_arr]
    A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit_lp})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None


def sca_optimize(f0, time_limit, verbose=False):
    """Improved SCA with adaptive M and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    row_idx, col_idx = build_lp_sparsity(n)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)

                consecutive_fail = 0

        best_ft = f.copy()
        best_alpha = 0.0

            if st < 1e-15:
                hi = mid
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_ft = ft.copy()
                best_alpha = mid
                lo = mid
            else:
                hi = mid

            print(f"    SCA it={it}, obj={best_obj:.10f}, M={M:.10f}, t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj

        js = max(0, k_peak - n + 1)

            contrib[j] = a[j] * a[k_peak - j]

                if a[h] < 1e-10:

                        lo_t = mid_t
