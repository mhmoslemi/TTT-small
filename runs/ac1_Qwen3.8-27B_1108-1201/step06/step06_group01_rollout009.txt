def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    # Project onto {x >= 0, sum(x) = target_sum}
    # First project onto {x >= 0, sum(x) = 1} then scale
    v = v / target_sum  # temporarily normalize
    v = np.maximum(v, 0)
    v = _project_simplex_unit(v)
    return v * target_sum


def _project_simplex_unit(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if not len(rho):
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_oracle(f, M, n_lags=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0, sum(g) <= 1."""
    n = len(f)
    # Use only top n_lags constraints for speed
    conv_fn = fast_convolve(f)  # This is conv(f, f), but we need conv(f, g)
    # Actually we need (f*g)[k] = sum_i f[i]*g[k-i] for each k
    # This is a linear function of g
    
    c = -np.ones(n)  # maximize sum(g) = minimize -sum(g)
    
    if n_lags is None:
        n_lags = 2 * n - 1
    
    # Build constraint matrix: A_ub[g] = conv(f,g)
    # (f*g)[k] = sum_{i=0}^{n-1} f[i] * g[k-i] (with g[j]=0 for j outside [0,n-1])
    # For a given k, the coefficient of g[j] is f[k-j] if 0 <= k-j < n, else 0
    
    # For efficiency, only include the top constraints
    # But we don't know which will be tight, so include all (or a subset)
    
    # Build full constraint matrix
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] appears in conv(f,g)[k] for k = j, j+1, ..., j+n-1
        # with coefficient f[k-j]
        ks = np.arange(j, min(2*n-1, j+n))
        A_ub[ks, j] = f[ks - j]
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, 1.0)] * n  # g >= 0 and sum(g) <= 1 (handled by objective)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20, 'feasibility_tolerance': 1e-9})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                g = g / s
            return g
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.3:
            break

        g = solve_lp_oracle(f, M)
        if g is None:
            # Fallback: use uniform
            g = np.ones(n) / n
            no_improve += 1
            if no_improve > 20:
                break
            continue

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search: find best alpha in [0,1]
        best_alpha = 0.0
        best_val = M

        # Coarse search
        alphas = np.array([0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0])
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine search (golden section)
        if 0 < best_alpha < 1:
            # Narrow to neighborhood of best_alpha
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            gr = (np.sqrt(5) - 1) / 2
            for _ in range(15):
                m1 = hi - gr * (hi - lo)
                m2 = lo + gr * (hi - lo)
                ft1 = (1.0 - m1) * f + m1 * g
                ft1 = np.maximum(ft1, 0.0)
                ft1 = ft1 / ft1.sum()
                v1 = fast_convolve(ft1).max()
                
                ft2 = (1.0 - m2) * f + m2 * g
                ft2 = np.maximum(ft2, 0.0)
                ft2 = ft2 / ft2.sum()
                v2 = fast_convolve(ft2).max()
                
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
                if hi - lo < 1e-6:
                    break
            mid = (lo + hi) / 2
            ft_mid = (1.0 - mid) * f + mid * g
            ft_mid = np.maximum(ft_mid, 0.0)
            ft_mid = ft_mid / ft_mid.sum()
            vmid = fast_convolve(ft_mid).max()
            if vmid < best_val:
                best_val = vmid
                best_alpha = mid

        if best_alpha > 1e-12:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            # Perturbation
            rng = np.random.default_rng(it * 7919 + 13)
            conv = fast_convolve(f)
            peak_k = int(np.argmax(conv))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            
            noise = rng.standard_normal(n) * 0.003
            # Bias away from peak contributors
            for j in range(js, je + 1):
                partner = peak_k - j
                if 0 <= partner < n:
                    noise[j] -= 0.002 * f[partner]
            
            f = _project_simplex_unit(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def multiplicative_subgradient(f0, time_limit, eta_init=0.01):
    """Multiplicative weight update for the min-max convolution problem."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    eta = eta_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(f)
        M = conv.max()
        
        # Find top peaks
        n_peaks = min(20, len(conv))
        top_peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
        
        # Exponential weighting of peaks
        logits = (conv[top_peaks] - M) / (eta * max(M, 1e-10))
        weights = np.exp(logits)
        weights = weights / weights.sum()
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        for i, k in enumerate(top_peaks):
            if weights[i] < 1e-10:
                continue
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                partner = k - j
                if 0 <= partner < n:
                    grad[j] += 2.0 * weights[i] * f[partner]
        
        # Multiplicative update
        f_new = f * np.exp(-eta * grad)
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s
        
        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            f_new = f  # revert
        
        f = f_new
        eta *= 0.9995
        
        if no_improve > 80:
            rng = np.random.default_rng(it * 31 + 7)
            f = _project_simplex_unit(best_f + rng.standard_normal(n) * 0.002)
            eta = eta_init
            no_improve = 0

    return best_f, best_obj


def active_set_peel(f0, time_limit):
    """
    Active set method: identify the peak lag, then solve a sub-problem
    to reduce it while keeping other lags bounded.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(f)
        M = conv.max()
        peak_k = int(np.argmax(conv))
        
        # The peak is at lag peak_k. Variables that affect it:
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        
        # Try to reduce the peak by solving:
        # min (f*f)[peak_k] = sum_{j=js}^{je} f[j]*f[peak_k-j]
        # s.t. (f*f)[k] <= M for all k != peak_k (approximately)
        #      sum(f) = 1, f >= 0
        
        # Simple approach: move mass from high-contributing positions to low ones
        # Contribution of f[j] to the peak: f[peak_k - j]
        contrib = np.zeros(n)
        for j in range(js, je + 1):
            partner = peak_k - j
            contrib[j] = f[partner]
        
        # We want to reduce the peak. The peak = sum_j f[j]*f[peak_k-j]
        # If we decrease f[j] by delta and increase f[j'] by delta (where j' not in [js,je]),
        # the change in peak is approximately -delta*f[peak_k-j] (if j in [js,je])
        
        # Find the best "donor" (inside the peak region) and "receiver" (outside)
        # Donor: position j in [js,je] with high f[j]*f[peak_k-j]
        # Receiver: position j' outside [js,je] with low impact on other peaks
        
        donor_score = contrib * f[js:je+1]  # f[j]*f[peak_k-j] for j in [js,je]
        donor_local = np.argmax(donor_score)
        donor = js + donor_local
        
        # Receiver: try positions that don't contribute to the current peak
        # and have low values (so increasing them doesn't create new peaks)
        outside = np.setdiff1d(np.arange(n), np.arange(js, je+1))
        if len(outside) == 0:
            no_improve = 9999  # break
        else:
            # Receiver should have low current value and not contribute to other peaks
            recv_scores = f[outside]  # prefer low values
            # Also check how much increasing it would affect other peaks
            # (approximation: just use current value)
            receiver = outside[np.argmin(recv_scores)]
            
            # Transfer mass
            for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                delta = f[donor] * frac
                f_test = f.copy()
                f_test[donor] -= delta
                f_test[receiver] += delta
                f_test = np.maximum(f_test, 0.0)
                f_test = f_test / f_test.sum()
                c_new = fast_convolve(f_test)
                if c_new.max() < best_obj - 1e-14:
                    f = f_test
                    best_obj = c_new.max()
                    best_f = f.copy()
                    break
        
        # Also try: reduce the peak by making the sequence more "spread out"
        # at the peak region
        if it % 10 == 0:
            # Smooth the peak region
            half_width = max(1, (je - js) // 4)
            window = np.ones(2 * half_width + 1) / (2 * half_width + 1)
            f_smooth = f.copy()
            f_smooth[js:je+1] = np.convolve(f[js:je+1], window, mode='same')
            f_smooth = np.maximum(f_smooth, 0.0)
            f_smooth = f_smooth / f_smooth.sum()
            c_new = fast_convolve(f_smooth)
            if c_new.max() < best_obj - 1e-14:
                f = f_smooth
                best_obj = c_new.max()
                best_f = f.copy()

        # Periodic perturbation to escape local optima
        if it % 50 == 0:
            rng = np.random.default_rng(it * 104729 + 7)
            scale = 0.001 + 0.002 * (it % 100) / 100.0
            f = _project_simplex_unit(best_f + rng.standard_normal(n) * scale)

    return best_f, best_obj


def genetic_diversity(f0, time_limit, pop_size=8):
    """Population-based search with mutation and selection."""
    n = len(f0)
    f0 = np.maximum(f0.astype(np.float64), 0.0)
    f0 = f0 / f0.sum()
    
    # Initialize population
    pop = [f0.copy()]
    rng = np.random.default_rng(42)
    
    # Add diverse individuals
    for _ in range(pop_size - 1):
        # Different types of perturbation
        ptype = rng.integers(0, 4)
        if ptype == 0:
            # Gaussian noise
            ind = _project_simplex_unit(f0 + rng.standard_normal(n) * 0.05)
        elif ptype == 1:
            # Structured: flatten peaks
            conv = fast_convolve(f0)
            peak_k = int(np.argmax(conv))
            js = max(0, peak_k - n // 4)
            je = min(n - 1, peak_k + n // 4)
            ind = f0.copy()
            ind[js:je+1] *= 0.9
            ind = _project_simplex_unit(ind)
        elif ptype == 2:
            # Shift: circular shift
            shift = rng.integers(1, n // 4)
            ind = np.roll(f0, shift)
            # Fix the boundaries (reduce wrap-around artifacts)
            fade = int(n // 20)
            ind[:fade] *= np.linspace(0, 1, fade)
            ind[-fade:] *= np.linspace(1, 0, fade)
            ind = _project_simplex_unit(ind)
        else:
            # Crossover with f0
            mask = rng.random(n) > 0.5
            other = _project_simplex_unit(f0 + rng.standard_normal(n) * 0.1)
            ind = np.where(mask, f0, other)
            ind = _project_simplex_unit(ind)
        
        pop.append(ind)
    
    # Evaluate
    objs = [fast_convolve(p).max() for p in pop]
    best_idx = int(np.argmin(objs))
    best_f = pop[best_idx].copy()
    best_obj = objs[best_idx]
    
    t0 = time.monotonic()
    gen = 0
    
    while time.monotonic() - t0 < time_limit:
        gen += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        
        # Tournament selection
        sel_idx = rng.integers(0, pop_size, size=3)
        sel_objs = [objs[i] for i in sel_idx]
        parent_idx = sel_idx[int(np.argmin(sel_objs))]
        parent = pop[parent_idx].copy()
        
        # Mutation
        mut_type = rng.integers(0, 3)
        if mut_type == 0:
            # Gaussian
            scale = 0.01 + 0.02 * min(1.0, gen / 50.0)
            child = _project_simplex_unit(parent + rng.standard_normal(n) * scale)
        elif mut_type == 1:
            # Peak-targeted: reduce contribution to the worst peak
            conv = fast_convolve(parent)
            peak_k = int(np.argmax(conv))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            child = parent.copy()
            # Reduce values in the peak region
            reduction = rng.uniform(0.01, 0.05)
            for j in range(js, je + 1):
                child[j] *= (1 - reduction * rng.random())
            child = _project_simplex_unit(child)
        else:
            # Crossover with elite
            elite = pop[int(np.argmin(objs))]
            alpha = rng.uniform(0.3, 0.7)
            child = _project_simplex_unit(alpha * parent + (1 - alpha) * elite)
        
        # Evaluate child
        child_obj = fast_convolve(child).max()
        
        # Replace worst individual if child is better
        worst_idx = int(np.argmax(objs))
        if child_obj < objs[worst_idx]:
            pop[worst_idx] = child
            objs[worst_idx] = child_obj
        
        # Update best
        if child_obj < best_obj:
            best_obj = child_obj
            best_f = child.copy()
        
        # Every 20 generations, do a quick FW-LP on the best
        if gen % 20 == 0 and time.monotonic() - t0 < time_limit - 10:
            t_alloc = min(5.0, time_limit - (time.monotonic() - t0) - 5)
            if t_alloc > 1:
                opt, _ = fw_lp_optimize(best_f, time_limit=t_alloc)
                opt_obj = fast_convolve(opt).max()
                if opt_obj < best_obj:
                    best_obj = opt_obj
                    best_f = opt.copy()
                    # Update population
                    pop[int(np.argmax(objs))] = opt
                    objs[int(np.argmax(objs))] = opt_obj
    
    return best_f, best_obj


def parametric_initializations(n):
    """Generate diverse parametric initializations."""
    inits = []
    x = np.linspace(0, 1, n)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine (Hann window)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    inits.append(a / a.sum())
    
    # Hamming
    a = 0.54 + 0.46 * np.cos(2 * np.pi * x)
    inits.append(a / a.sum())
    
    # Blackman
    a = 0.42 + 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    a = np.maximum(a, 0)
    inits.append(a / a.sum())
    
    # Gaussian (center)
    mu, sigma = 0.5, 0.2
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    inits.append(a / a.sum())
    
    # Gaussian (wide)
    sigma = 0.4
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    inits.append(a / a.sum())
    
    # Sinc-like (main lobe only)
    t = (x - 0.5) * 2 * np.pi * 2  # 2 cycles
    a = np.zeros(n)
    mask = np.abs(t) < np.pi
    a[mask] = np.sinc(t[mask] / np.pi)
    a = np.maximum(a, 0)
    if a.sum() > 0:
        inits.append(a / a.sum())
    
    # Bimodal
    a = np.exp(-0.5 * ((x - 0.25) / 0.15) ** 2) + np.exp(-0.5 * ((x - 0.75) / 0.15) ** 2)
    inits.append(a / a.sum())
    
    # Triangular
    a = 1 - np.abs(2 * x - 1)
    inits.append(a / a.sum())
    
    # Super-Gaussian
    a = np.exp(-((x - 0.5) / 0.3) ** 4)
    inits.append(a / a.sum())
    
    # Power law
    a = 1 / (0.1 + np.abs(x - 0.5))
    inits.append(a / a.sum())
    
    return inits

            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try
