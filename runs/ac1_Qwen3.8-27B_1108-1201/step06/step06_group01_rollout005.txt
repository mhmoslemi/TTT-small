def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    # sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.success:
            g = np.maximum(res.x[:n], 0.0)
            t = res.x[n]
            gs = g.sum()
            if gs > 1e-15:
                g = g / gs
            return g, t
    except Exception:
        pass
    return None, None


def oracle_standard(f, M):
    """LP Oracle A (original): max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n + 1)  # maximize sum(g), t unused
    c[n] = 0.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = np.full(2 * n - 1, M)
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, n] = 0.0

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c[:n], A_ub=A_ub[:, :n], b_ub=b_ub, bounds=bounds[:n],
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success:
            g = np.maximum(res.x, 0.0)
            return g
    except Exception:
        pass
    return None


def fw_optimize_new(f0, time_limit, use_new_oracle=True):
    """Frank-Wolfe with improved oracle and adaptive strategy."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_scale = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.3:
            break

        if use_new_oracle:
            g, t_min = oracle_minmax(f)
            if g is None:
                # Fallback to standard oracle
                g = oracle_standard(f, M * 1.01)
                if g is None:
                    m_scale *= 1.1
                    if m_scale > 5:
                        m_scale = 1.0
                        g = np.ones(n) / n
                    continue
        else:
            g = oracle_standard(f, M * m_scale)
            if g is None:
                m_scale *= 1.1
                if m_scale > 5:
                    m_scale = 1.0
                    g = np.ones(n) / n
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.02)
            hi = min(1.0, best_alpha + 0.02)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-13:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_scale = 1.0

            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 30:
            # Structured perturbation targeting peaks
            rng = np.random.default_rng(it * 7919 + 3)
            conv_cur = fast_convolve(best_f)
            n_peaks = min(5, len(conv_cur))
            peak_lags = np.argpartition(conv_cur, -n_peaks)[-n_peaks:]

            noise = rng.standard_normal(n) * 0.003
            for k in peak_lags:
                k = int(k)
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                for j in range(js, je + 1):
                    partner = k - j
                    if 0 <= partner < n:
                        noise[j] -= 0.002 * best_f[partner]

            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
            if M < best_obj:
                best_obj = M
                best_f = f.copy()

    return best_f, best_obj


def spectral_iteration(f0, time_limit, n_iters=None):
    """
    Gerchberg-Saxton-like iteration: alternate between time and frequency domain
    to flatten the autocorrelation.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    if n_iters is None:
        n_iters = 10000

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # Forward FFT
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        A = np.fft.rfft(f, fft_size)

        # Current power spectrum
        power = np.abs(A) ** 2

        # Target: flat power spectrum (corresponds to delta autocorrelation)
        # But we want flat autocorrelation, which means flat power spectrum
        # Actually: flat autocorrelation <-> flat power spectrum
        # So we want |A|^2 to be flat, meaning |A| = constant

        # Replace magnitude with target, keep phase
        target_mag = np.sqrt(power.sum() / len(power))  # RMS magnitude
        # But we want to reduce peaks, so use a "soft" target
        # Cap the magnitude at some value
        p = power
        p_max = p.max()
        p_min = p.min()
        if p_max > 0:
            # Compression: raise low power, lower high power
            exponent = 0.7  # compression factor
            new_mag = p_max * (p / p_max) ** exponent
            # Renormalize to preserve sum
            new_power = new_mag ** 2
            target_power = target_mag ** 2
            scale = (target_power * len(power)) / new_power.sum()
            new_mag *= np.sqrt(scale)

        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)

        # Inverse FFT
        f_new = np.fft.irfft(A_new, fft_size)[:n]

        # Project to simplex
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()

        val = fast_convolve(f_new).max()
        if val < best_obj - 1e-15:
            best_obj = val
            best_f = f_new.copy()
            f = f_new

    return best_f, best_obj


def bottleneck_lp(f, time_limit):
    """
    When stuck, identify top peaks and solve a small LP to reduce them simultaneously.
    Variables: perturbations d[j] (can be positive or negative, bounded)
    Objective: minimize max_{k in top_peaks} (f*f + 2*f*d)[k]
    Subject to: f + d >= 0, sum(d) = 0, |d[j]| <= eps
    """
    f = f.astype(np.float64).copy()
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()

    # Get top peaks
    n_peaks = min(8, len(conv))
    peak_lags = np.argpartition(conv, -n_peaks)[-n_peaks:]
    peak_lags = peak_lags[np.argsort(conv[peak_lags])[::-1]]

    best_f = f.copy()
    best_obj = M

    t0 = time.monotonic()
    eps = 0.05

    for round_num in range(50):
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        # Re-identify peaks
        conv = fast_convolve(f)
        M = conv.max()
        peak_lags = np.argpartition(conv, -n_peaks)[-n_peaks:]
        peak_lags = peak_lags[np.argsort(conv[peak_lags])[::-1]]

        # Build LP: minimize t
        # s.t. for each peak k: 2*sum_j f[k-j]*d[j] + t >= 0 (i.e., c[k] + 2*(f*d)[k] <= M + t)
        #     f[j] + d[j] >= 0
        #     sum(d) = 0
        #     |d[j]| <= eps
        # Variables: d[0..n-1], t
        nv = n + 1
        c_vec = np.zeros(nv)
        c_vec[n] = 1.0  # minimize t

        A_ub = []
        b_ub = []

        # Peak constraints: (M - c[k] - 2*(f*d)[k]) + t >= 0
        # => 2*(f*d)[k] - t <= M - c[k]
        for k in peak_lags:
            k = int(k)
            row = np.zeros(nv)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                row[j] = 2.0 * f[k - j]
            row[n] = -1.0
            A_ub.append(row)
            b_ub.append(M - conv[k])

        # d[j] <= eps
        for j in range(n):
            row = np.zeros(nv)
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(eps)
        # -d[j] <= eps (i.e., d[j] >= -eps)
        for j in range(n):
            row = np.zeros(nv)
            row[j] = -1.0
            A_ub.append(row)
            b_ub.append(eps)
        # f[j] + d[j] >= 0 => -d[j] <= f[j]
        for j in range(n):
            row = np.zeros(nv)
            row[j] = -1.0
            A_ub.append(row)
            b_ub.append(f[j])

        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)

        A_eq = np.zeros((1, nv))
        A_eq[0, :n] = 1.0  # sum(d) = 0
        b_eq = np.array([0.0])

        bounds = [(-eps, eps)] * n + [(-M, M)]

        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 10})
            if res.success:
                d = res.x[:n]
                f_new = f + d
                f_new = np.maximum(f_new, 0.0)
                f_new = f_new / f_new.sum()
                val = fast_convolve(f_new).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_f = f_new.copy()
                    f = f_new
                    # Reduce eps as we make progress
                    eps *= 0.8
                else:
                    eps *= 0.7
                    if eps < 0.001:
                        break
            else:
                eps *= 0.5
                if eps < 0.001:
                    break
        except Exception:
            eps *= 0.5
            if eps < 0.001:
                break

    return best_f, best_obj


def micro_polish(a0, time_limit):
    """Fine perturbation-based polish targeting peak reduction."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    scales = [0.0002, 0.0005, 0.001, 0.002, 0.0005, 0.0002]
    si = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        scale = scales[si % len(scales)]
        rng = np.random.default_rng((it * 2654435761 + 1) % (2**32))

        tried = 0
        while tried < 15 and time.monotonic() - t0 < time_limit - 0.1:
            tried += 1
            conv = fast_convolve(a)
            n_peaks = min(3, len(conv))
            peak_lags = np.argpartition(conv, -n_peaks)[-n_peaks:]

            noise = rng.standard_normal(n) * scale
            for k in peak_lags:
                k = int(k)
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                for j in range(js, je + 1):
                    partner = k - j
                    if 0 <= partner < n:
                        noise[j] -= scale * 1.5 * a[partner]

            candidate = project_simplex(a + noise)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()

        if it % 200 == 0:
            si += 1

    return best_a, best_obj


def diverse_starts(n, rng):
    """Generate diverse initial sequences of length n."""
    starts = []

    # Uniform
    starts.append(np.ones(n) / n)

    # Cosine window
    x = np.linspace(0, np.pi, n)
    cw = np.sin(x) ** 2
    cw = np.maximum(cw, 0)
    starts.append(cw / cw.sum())

    # Triangular
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n)) + 1
    starts.append(tri / tri.sum())

    # Edge-concentrated (U-shaped)
    u_shape = np.ones(n)
    edge_decay = n * 0.3
    for j in range(n):
        dist_to_center = abs(j - (n - 1) / 2.0)
        u_shape[j] = 1.0 + 0.5 * (dist_to_center / edge_decay) ** 2
    starts.append(u_shape / u_shape.sum())

    # Two peaks
    two_peak = np.zeros(n)
    two_peak[:n // 4] = 2.0
    two_peak[3 * n // 4:] = 2.0
    two_peak[n // 4:3 * n // 4] = 0.5
    starts.append(two_peak / two_peak.sum())

    # Random
    starts.append(rng.dirichlet(np.ones(n)))

    # Exponential edges
    exp_edge = np.zeros(n)
    decay = 0.1
    for j in range(n):
        exp_edge[j] = np.exp(-decay * j) + np.exp(-decay * (n - 1 - j))
    starts.append(exp_edge / exp_edge.sum())

    return starts


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search with improved LP oracle,
    spectral iteration, and bottleneck LP for peak reduction.
    """
    rng = np.random.default_rng(seed)
    t0_total = time.monotonic()
    deadline = t0_total + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t0_total
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)

    # === Phase 1: Multi-n quick exploration (200s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent, max(256, n_parent - 96), n_parent + 96,
                                512, 640, 768, 896, 1024, 1280]))
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 1600]

    for n_try in candidate_ns:
        if time.monotonic() > deadline - 1400:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss > 1e-10:
                start = start / ss
            else:
                start = np.ones(n_try) / n_try

        t_alloc = min(25, deadline - time.monotonic() - 1400)
        if t_alloc < 5:
            break
        print(f"  FW n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = fw_optimize_new(start, time_limit=t_alloc, use_new_oracle=True)
        try_seq(a_opt, f"fw_n{n_try}")

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW with new oracle (500s) ===
    print("\n=== Phase 2: Deep FW (new oracle) ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        t_alloc = min(350, deadline - time.monotonic() - 900)
        if t_alloc > 20:
            print(f"  Deep FW n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize_new(best_seq, time_limit=t_alloc, use_new_oracle=True)
            try_seq(a_opt, "deep_fw_new")

        # Standard oracle pass
        t_alloc = min(80, deadline - time.monotonic() - 850)
        if t_alloc > 10:
            print(f"  FW std oracle n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize_new(best_seq, time_limit=t_alloc, use_new_oracle=False)
            try_seq(a_opt, "deep_fw_std")

    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Spectral iteration (200s) ===
    print("\n=== Phase 3: Spectral Iteration ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(150, deadline - time.monotonic() - 700)
        if t_alloc > 10:
            print(f"  Spectral n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, _ = spectral_iteration(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "spectral")

            # Follow up with FW
            t_alloc2 = min(50, deadline - time.monotonic() - 650)
            if t_alloc2 > 5:
                a_opt2, _ = fw_optimize_new(a_opt, time_limit=t_alloc2, use_new_oracle=True)
                try_seq(a_opt2, "spectral+fw")

    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Bottleneck LP (300s) ===
    print("\n=== Phase 4: Bottleneck LP ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(250, deadline - time.monotonic() - 400)
        if t_alloc > 10:
            print(f"  Bottleneck n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, _ = bottleneck_lp(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "bottleneck")

            # Another round from the result
            t_alloc2 = min(50, deadline - time.monotonic() - 350)
            if t_alloc2 > 5:
                a_opt2, _ = bottleneck_lp(a_opt, time_limit=t_alloc2)
                try_seq(a_opt2, "bottleneck2")

    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Perturbed restarts + FW (250s) ===
    print("\n=== Phase 5: Perturbed Restarts ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        perturb_scales = [0.003, 0.008, 0.02, 0.05]
        for scale in perturb_scales:
            t_alloc = min(60, deadline - time.monotonic() - 200)
            if t_alloc < 8:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            a_opt, _ = fw_optimize_new(f_pert, time_limit=t_alloc, use_new_oracle=True)
            try_seq(a_opt, f"pert_{scale}")

    print(f"  Phase 5 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Final polish (remaining time) ===
    print("\n=== Phase 6: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Micro polish
        t_alloc = min(80, deadline - time.monotonic() - 60)
        if t_alloc > 5:
            print(f"  Micro polish, t={t_alloc:.0f}s")
            a_opt, _ = micro_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "micro")

        # One more FW
        t_alloc = min(50, deadline - time.monotonic() - 30)
        if t_alloc > 5:
            print(f"  Final FW, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize_new(best_seq, time_limit=t_alloc, use_new_oracle=True)
            try_seq(a_opt, "final_fw")

        # Bottleneck on best
        t_alloc = min(40, deadline - time.monotonic() - 15)
        if t_alloc > 5:
            a_opt, _ = bottleneck_lp(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_bottleneck")

    print(f"  Phase 6 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, total time={time.monotonic()-t0_total:.0f}s)")
    return [float(x) for x in best_seq]
