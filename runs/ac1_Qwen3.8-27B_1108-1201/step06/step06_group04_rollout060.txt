Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

- Exploits PSD structure of Hankel matrices for efficient matvec

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_matvec(p, a):
    """
    Compute H @ a where H[i,j] = p[i+j] (p has length >= 2n-1, a has length n).
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    # H[i] = sum_j p[i+j] * a[j]
    # = convolve(p, a_rev, 'full')[n-1 : n-1+n]
    a_rev = a[::-1]
    size = len(p) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    Ar = np.fft.rfft(a_rev, fft_size)
    result = np.fft.irfft(P * Ar, fft_size)
    return result[n - 1: n - 1 + n]


def fista_qp_solve(p_hankel, n, a_init, max_iters=200, tol=1e-10):
    """
    Solve min (1/2) a^T H a s.t. sum(a)=1, a>=0 using FISTA.
    H is the Hankel matrix with H[i,j] = p_hankel[i+j].
    Returns the solution a.
    """
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    t_k = 1.0
    
    # Estimate Lipschitz constant using power iteration on H
    # H @ v can be computed via hankel_matvec
    # Use a few iterations to estimate lambda_max
    v = a.copy()
    v = v / np.linalg.norm(v)
    for _ in range(20):
        Hv = hankel_matvec(p_hankel, v)
        norm_Hv = np.linalg.norm(Hv)
        if norm_Hv < 1e-15:
            L = 1.0
            break
        v_new = Hv / norm_Hv
        L = v @ Hv / (v @ v) if v @ v > 1e-15 else 1.0
        v = v_new
    L = max(L * 1.01, 1e-8)
    lr = 1.0 / L

    for it in range(max_iters):
        # FISTA gradient step
        grad = hankel_matvec(p_hankel, y)
        z = project_simplex(y - lr * grad)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = z + ((t_k - 1.0) / t_new) * (z - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = z
        else:
            y = y / s_y  # Keep y on simplex for stability
        a = z
        t_k = t_new

        # Check convergence (change in a)
        if it > 10:
            if np.linalg.norm(a - (z)) < tol * max(1.0, np.linalg.norm(a)):
                break

    return a


def compute_dual_weights(conv, topk, tail_weight=0.01):
    """
    Compute dual weight distribution over peaks.
    Top-k peaks get high weight, rest get tail_weight (proportional).
    """
    n_peaks = len(conv)
    k = min(topk, n_peaks)
    
    # Get top-k indices
    top_idx = np.argpartition(conv, -k)[-k:]
    
    p = np.full(n_peaks, tail_weight)
    p[top_idx] = 1.0
    
    p_sum = p.sum()
    if p_sum < 1e-15:
        p = np.ones(n_peaks) / n_peaks
    else:
        p = p / p_sum
    
    return p


def softmax_dual_weights(conv, temperature):
    """Compute softmax weights over peaks."""
    scaled = conv / temperature
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-15:
        p = np.ones(len(conv)) / len(conv)
    else:
        p = p / p_sum
    return p

def multiplicative_weights(a0, time_limit, eta_init=0.1, eta_decay=0.999,
                           topk=5, use_soft=False, soft_T=0.01,
                           verbose=False):
    """
    Mirror descent (multiplicative weights) on the simplex.
    Subgradient of max_k conv(a,a)[k] at a is 2*a[k*-j] for the argmax peak k*.
    Update: a[j] proportional to a[j] * exp(-eta * grad[j])
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    eta = eta_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        
        if use_soft:
            # Soft subgradient: weighted average over peaks
            scaled = conv / soft_T
            scaled -= scaled.max()
            p = np.exp(scaled)
            p = p / p.sum()
            # g[j] = 2 * sum_k p[k] * a[k-j] (valid indices)
            # This is 2 * cross-correlation of p and a
            # g[j] = 2 * sum_k p[k] * a[k-j] where 0 <= k-j < n
            # = 2 * (p * a_rev_full)[n-1+j]... 
            # Let me just compute directly: for each j, g[j] = 2*sum_k p[k]*a[k-j]
            # This is the same as hankel_matvec(p, a) but with a specific indexing
            # hankel_matvec(p, a)[j] = sum_k p[j+k] * a[k]
            # We want: g[j] = 2*sum_k p[k] * a[k-j]
            # Let m = k-j: g[j] = 2*sum_m p[m+j] * a[m] = 2 * hankel_matvec(p, a)[j]
            g = 2.0 * hankel_matvec(p, a)
        else:
            k_star = int(np.argmax(conv))
            # g[j] = 2 * a[k_star - j] (when 0 <= k_star - j < n)
            g = np.zeros(n)
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            g[idx] = 2.0 * a[k_star - idx]

        # Multiplicative weight update
        updates = a * np.exp(-eta * g)
        s_up = updates.sum()
        if s_up < 1e-15:
            a = np.ones(n) / n
        else:
            a = updates / s_up

        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt
        if no_improve % 20 == 0:
            eta *= eta_decay
        if no_improve > 100:
            eta = eta_init * 0.5
            if no_improve > 300:
                rng = np.random.default_rng((it * 31 + 3) % (2**31))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                eta = eta_init
                no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    MW it={it}, obj={best_obj:.10f}, eta={eta:.6f}")

    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- FISTA-based QP primal-dual with adaptive dual weighting (top-k + tail)
- Exploits PSD structure of Hankel matrices for O(n log n) matvec via FFT
- Multiplicative weight (mirror descent) updates as alternative
- Subgradient-guided coordinate descent for final polish
- Multi-n exploration with asymmetric initializations
- Key insight: asymmetric sequences reduce center peak below 1/n
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_matvec(p, a):
    """
    Compute H @ a where H[i,j] = p[i+j] (p has length 2n-1, a has length n).
    Result[i] = sum_j p[i+j] * a[j] for i=0..n-1.
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    a_rev = a[::-1]
    size = len(p) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    Ar = np.fft.rfft(a_rev, fft_size)
    result = np.fft.irfft(P * Ar, fft_size)
    return result[n - 1: n - 1 + n]


def fista_qp_solve(p_hankel, n, a_init, max_iters=150):
    """
    Solve min (1/2) a^T H a s.t. sum(a)=1, a>=0 using FISTA.
    H is the Hankel matrix with H[i,j] = p_hankel[i+j].
    Warm-started from a_init. Returns the solution a.
    """
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    t_k = 1.0

    # Estimate Lipschitz constant using a few power iterations
    v = a.copy()
    v_norm = np.linalg.norm(v)
    if v_norm < 1e-15:
        L = 1.0
    else:
        v = v / v_norm
        L = 0.0
        for _ in range(15):
            Hv = hankel_matvec(p_hankel, v)
            v_dot_Hv = np.dot(v, Hv)
            Hv_norm = np.linalg.norm(Hv)
            if Hv_norm < 1e-15:
                L = 1.0
                break
            L = max(L, abs(v_dot_Hv))
            v = Hv / Hv_norm
    L = max(L * 1.05, 1e-10)
    lr = 1.0 / L

    a_prev = a.copy()
    for it in range(max_iters):
        grad = hankel_matvec(p_hankel, y)
        z = project_simplex(y - lr * grad)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = z + ((t_k - 1.0) / t_new) * (z - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = z.copy()
        else:
            y = y / s_y
        a_prev = a.copy()
        a = z
        t_k = t_new

        if it > 20 and np.linalg.norm(a - a_prev) < 1e-10:
            break

    return a


def compute_dual_weights_topk(conv, topk, tail_weight=0.01):
    """Top-k peaks get weight 1, rest get tail_weight. Normalized to sum 1."""
    n_peaks = len(conv)
    k = min(topk, n_peaks)
    top_idx = np.argpartition(conv, -k)[-k:]
    p = np.full(n_peaks, tail_weight)
    p[top_idx] = 1.0
    p_sum = p.sum()
    if p_sum < 1e-15:
        p = np.ones(n_peaks) / n_peaks
    else:
        p = p / p_sum
    return p


def compute_dual_weights_softmax(conv, temperature):
    """Softmax weights over peaks."""
    scaled = conv / temperature
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-15:
        p = np.ones(len(conv)) / len(conv)
    else:
        p = p / p_sum
    return p


def dual_primal_fista(a0, time_limit, topk=10, tail_weight=0.01,
                      blend=0.5, use_softmax=False, T_init=0.01,
                      fista_iters=80, verbose=False):
    """
    Primal-dual iteration with FISTA for the QP subproblem.
    Dual: top-k+tail or softmax weights on convolution peaks.
    Primal: FISTA on min a^T H a s.t. simplex.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    T = T_init
    current_topk = topk
    current_tailw = tail_weight

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.5:
            break

        conv = fast_convolve(a)

        if use_softmax:
            p = compute_dual_weights_softmax(conv, T)
        else:
            p = compute_dual_weights_topk(conv, current_topk, current_tailw)

        a_qp = fista_qp_solve(p, n, a, max_iters=fista_iters)

        a_new = (1.0 - blend) * a + blend * a_qp
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a_qp
        else:
            a_new = a_new / s_new

        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adaptive parameter tuning
        if no_improve > 10:
            current_topk = min(n, int(current_topk * 1.4) + 1)
            current_tailw = max(0.001, current_tailw * 0.7)
            if use_softmax:
                T *= 1.3
        if no_improve > 40:
            current_topk = max(3, int(current_topk * 0.6))
            current_tailw = min(0.2, current_tailw * 2.0)
            if use_softmax:
                T *= 0.6
        if no_improve > 120:
            rng = np.random.default_rng((it * 48271 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            T = T_init
            current_topk = topk
            current_tailw = tail_weight
            no_improve = 0

        if verbose and it % 30 == 0:
            print(f"    dPfista it={it}, obj={best_obj:.10f}, topk={current_topk}, tailw={current_tailw:.4f}")

    return best_a, best_obj


def multiplicative_weights_method(a0, time_limit, eta_init=0.05, eta_min=0.001,
                                  topk=3, use_soft=False, soft_T=0.005,
                                  verbose=False):
    """
    Mirror descent (multiplicative weights) on the simplex.
    Subgradient of max_k conv(a,a)[k] is 2*a[k*-j] for argmax peak k*.
    Update: a[j] proportional to a[j] * exp(-eta * grad[j])
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    eta = eta_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        if use_soft:
            # Soft subgradient using hankel_matvec
            scaled = conv / soft_T
            scaled -= scaled.max()
            p = np.exp(scaled)
            p_sum = p.sum()
            if p_sum > 1e-15:
                p = p / p_sum
            else:
                p = np.ones(len(conv)) / len(conv)
            g = 2.0 * hankel_matvec(p, a)
        else:
            k_star = int(np.argmax(conv))
            k = topk
            top_idx = np.argpartition(conv, -k)[-k:]
            g = np.zeros(n)
            for kk in top_idx:
                js = max(0, kk - n + 1)
                je = min(n - 1, kk)
                idx = np.arange(js, je + 1)
                g[idx] += a[kk - idx]
            g = 2.0 * g / len(top_idx)

        # Multiplicative weight update
        log_updates = np.log(np.maximum(a, 1e-30)) - eta * g
        log_updates -= log_updates.max()
        updates = np.exp(log_updates)
        s_up = updates.sum()
        if s_up < 1e-30:
            a = np.ones(n) / n
        else:
            a = updates / s_up
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a / s_a

        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt eta
        if no_improve % 30 == 0 and no_improve > 0:
            eta = max(eta_min, eta * 0.95)
        if no_improve > 200:
            eta = eta_init * 0.5
        if no_improve > 500:
            rng = np.random.default_rng((it * 31 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            eta = eta_init
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    MW it={it}, obj={best_obj:.10f}, eta={eta:.6f}")

    return best_a, best_obj


def local_search_cd(a0, time_limit, max_evals_per_iter=20, verbose=False):
    """
    Subgradient-guided coordinate descent local search.
    Moves mass from high-subgradient to low-subgradient positions.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(99999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(best_a)
        k_peak = int(np.argmax(conv))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = set(range(js, je + 1))
        non_contrib = [i for i in range(n) if i not in contrib]

        if not non_contrib:
            break

        # Subgradient for contributing positions
        g_contrib = {}
        for j in range(js, je + 1):
            partner = k_peak - j
            if 0 <= partner < n:
                g_contrib[j] = a[j] * a[partner]

        # Sort sources by contribution (descending)
        sorted_sources = sorted(g_contrib.keys(), key=lambda x: g_contrib[x], reverse=True)
        top_sources = sorted_sources[:5]

        # Targets: non-contributing positions, prefer those with low a values
        sorted_targets = sorted(non_contrib, key=lambda x: a[x])
        top_targets = sorted_targets[:min(15, len(sorted_targets))]

        improved = False
        for src in top_sources:
            if a[src] < 1e-10:
                continue
            for tgt in top_targets:
                for frac in [0.05, 0.1, 0.15, 0.25]:
                    shift = a[src] * frac
                    if shift < 1e-12:
                        continue
                    candidate = best_a.copy()
                    candidate[src] -= shift
                    if candidate[src] < -1e-12:
                        continue
                    candidate[tgt] += shift
                    candidate = np.maximum(candidate, 0)
                    s_c = candidate.sum()
                    if s_c < 1e-15:
                        continue
                    candidate = candidate / s_c
                    obj = fast_convolve(candidate).max()
                    if obj < best_obj - 1e-13:
                        best_obj = obj
                        best_a = candidate.copy()
                        a = candidate.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 50:
                # Random perturbation
                scale = 0.001 + 0.001 * (no_improve // 100)
                a = project_simplex(best_a + rng.standard_normal(n) * scale)
                no_improve = 0

            if no_improve > 2000:
                break

        if verbose and it % 100 == 0:
            print(f"    local_cd it={it}, obj={best_obj:.10f}")

    return best_a, best_obj


def random_restart_search(a0, time_limit, refine_fn, n_restarts=10, verbose=False):
    """
    Multi-start: perturb the best solution, refine, keep the best.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(77777)

    for r in range(n_restarts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        t_per = max(1.0, remaining / (n_restarts - r))

        # Generate perturbation
        scale = 0.002 + 0.003 * r / max(1, n_restarts - 1)
        perturbed = project_simplex(best_a + rng.standard_normal(n) * scale)

        # Refine
        a_opt, _ = refine_fn(perturbed, time_limit=min(t_per, remaining - 1))
        obj = fast_convolve(a_opt).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_opt.copy()
            if verbose:
                print(f"    restart {r}: improved to {best_obj:.10f}")

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations, emphasizing asymmetric sequences."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Exponential decay (asymmetric - mass on left)
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        e = np.exp(-alpha * x / n)
        inits.append(e / e.sum())

    # Exponential growth (asymmetric - mass on right)
    for alpha in [0.5, 1.0, 2.0]:
        e = np.exp(alpha * x / n)
        inits.append(e / e.sum())

    # Power law
    for beta in [0.5, 1.0, 1.5, 2.0]:
        p = (x + 1) ** (-beta)
        inits.append(p / p.sum())

    # Half-Gaussian (centered left of middle)
    for center in [0.2, 0.3, 0.4]:
        for sigma in [0.1, 0.15, 0.2, 0.3]:
            g = np.exp(-0.5 * ((x / n - center) / sigma) ** 2)
            inits.append(g / g.sum())

    # Beta-like (using approximation via power functions)
    for a_p, b_p in [(2, 5), (3, 8), (1, 3), (4, 10)]:
        b = (x + 1) ** (a_p - 1) * (n - x) ** (b_p - 1)
        b = np.maximum(b, 1e-30)
        inits.append(b / b.sum())

    # W-shaped (two peaks, asymmetric)
    for c1, c2, w in [(0.15, 0.6, 0.08), (0.2, 0.7, 0.1), (0.1, 0.5, 0.12)]:
        g = np.exp(-0.5 * ((x / n - c1) / w) ** 2) + 0.5 * np.exp(-0.5 * ((x / n - c2) / w) ** 2)
        inits.append(g / g.sum())

    # From parent (resampled)
    if parent is not None and len(parent) > 0:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 0.7))

    return inits


def quick_score(a):
    """Quick score computation (same as evaluate but without the function call overhead)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    conv = fast_convolve(a)
    return float(2 * n * conv.max() / (s * s))

                break
            t_alloc = min(remaining, max(30, int(700.0 / len(configs))))
            print(f"  FISTA-PD config {ci} (topk={cfg['topk']}, tailw={cfg['tail_weight']}), t={t_alloc:.0f}s")
            a_opt, _ = dual_primal_fista(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"fista_{ci}")

    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")

    # === Phase 3: Multiplicative weights (~300s) ===
    print("\n=== Phase 3: Multiplicative Weights ===")
    t3 = time.monotonic()

    if best_seq is not None:
        mw_configs = [
            {'eta_init': 0.03, 'topk': 3, 'use_soft': False},
            {'eta_init': 0.05, 'topk': 5, 'use_soft': False},
            {'eta_init': 0.02, 'topk': 5, 'use_soft': True, 'soft_T': 0.003},
            {'eta_init': 0.08, 'topk': 2, 'use_soft': False},
        ]

        for mi, mc in enumerate(mw_configs):
            remaining = deadline - time.monotonic() - 250
            if remaining < 15:
                break
            t_alloc = min(remaining, max(20, int(250.0 / len(mw_configs))))
            print(f"  MW config {mi} (eta={mc['eta_init']}), t={t_alloc:.0f}s")
            a_opt, _ = multiplicative_weights_method(best_seq, time_limit=t_alloc, **mc)
            try_seq(a_opt, f"mw_{mi}")

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")

    # === Phase 4: Local search polish (~300s) ===
    print("\n=== Phase 4: Local Search ===")
    t4 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 150
        if remaining > 30:
            t_alloc = min(remaining, 180)
            print(f"  Local search CD, t={t_alloc:.0f}s")
            a_opt, _ = local_search_cd(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "local_cd")

        # Quick FISTA after local search
        remaining = deadline - time.monotonic() - 100
        if remaining > 15 and best_seq is not None:
            t_alloc = min(remaining, 80)
            print(f"  FISTA polish, t={t_alloc:.0f}s")
            a_opt, _ = dual_primal_fista(best_seq, time_limit=t_alloc,
                                         topk=5, tail_weight=0.003,
                                         blend=0.3, fista_iters=150)
            try_seq(a_opt, "fista_polish")

        # More local search
        remaining = deadline - time.monotonic() - 60
        if remaining > 15 and best_seq is not None:
            t_alloc = min(remaining, 60)
            print(f"  Local search round 2, t={t_alloc:.0f}s")
            a_opt, _ = local_search_cd(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "local_cd2")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # === Phase 5: Multi-start refinement (~150s) ===
    print("\n=== Phase 5: Multi-start Refinement ===")
    t5 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 30
        if remaining > 30:
            # Define a quick refine function
            def _quick_refine(a_init, t_lim):
                return dual_primal_fista(a_init, time_limit=t_lim,
                                         topk=5, tail_weight=0.005,
                                         blend=0.4, fista_iters=60)

            print(f"  Multi-start, t={remaining:.0f}s")
            a_opt, _ = random_restart_search(best_seq, time_limit=remaining,
                                             refine_fn=_quick_refine,
                                             n_restarts=8, verbose=True)
            try_seq(a_opt, "multistart")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # === Phase 6: Final verification ===
    print("\n=== Phase 6: Final Verification ===")

    if best_seq is not None:
        # One final attempt with different n
        n_final = len(best_seq)
        for n_alt in [n_final - 32, n_final + 32]:
            if n_alt < 32:
                continue
            remaining = deadline - time.monotonic() - 5
            if remaining < 10:
                break
            parent_alt = np.interp(np.arange(n_alt),
                                   np.linspace(0, 1, n_final), best_seq)
            parent_alt = np.maximum(parent_alt, 0)
            ps = parent_alt.sum()
            if ps > 1e-10:
                parent_alt = parent_alt / ps
                a_opt, _ = dual_primal_fista(parent_alt, time_limit=min(remaining, 20),
                                             topk=8, tail_weight=0.01,
                                             blend=0.5, fista_iters=60)
                try_seq(a_opt, f"final_n{n_alt}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final score verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
