"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Fast O(n log n) smoothed gradient with Nesterov momentum
- Adaptive softmax temperature annealing
- LP oracle polishing for final convergence
- Multi-length exploration with spectral initialization
- Sparse structure exploitation via coordinate descent
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_gradient(w, y, T):
    """Compute gradient of (1/T)*log(sum exp(conv(y)/T)) at y, given weights w.
    
    grad_j = (2/T) * sum_k w_k * y_{k-j} for 0<=k-j<n
    
    Uses FFT for O(n log n) computation.
    """
    n = len(y)
    # (w cross-corr y)_j = sum_l w_{j+l} * y_l
    # This equals (w * y_rev)_{j + n - 1} where * is linear convolution
    y_rev = y[::-1]
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    Y = np.fft.rfft(y_rev, fft_size)
    result = np.fft.irfft(W * Y, fft_size)[:size]
    # Extract the cross-correlation: result[j + n - 1] for j = 0..n-1
    corr = result[n - 1:2 * n - 1]
    return (2.0 / T) * corr


def compute_objective(a):
    """Compute 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (f*g) = A @ g."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def smooth_gradient_optimize(a0, time_limit, T_init=0.002, lr_init=0.03,
                              mom=0.9, cooling=0.9997, verbose=False):
    """Fast smoothed gradient descent with Nesterov-style momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = compute_objective(a)
    v = np.zeros(n)

    t0 = time.monotonic()
    it = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 500 == 0:
            remaining = time_limit - (time.monotonic() - t0)
            if remaining < 0.1:
                break

        # Compute gradient
        conv = fast_convolve(a)
        # Softmax weights (numerically stable)
        conv_shifted = conv - conv.max()
        w = np.exp(conv_shifted / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(conv)) / len(conv)
        else:
            w = w / w_sum

        grad = softmax_gradient(w, a, T)

        # Gradient step with backtracking
        step = lr
        best_val = best_obj
        best_candidate = a
        for _bt in range(8):
            candidate = project_simplex(a - step * grad)
            val = compute_objective(candidate)
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                break
            step *= 0.5

        if best_candidate is not a and best_val < best_obj:
            v = mom * v + (best_candidate - a)
            a = best_candidate
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            v = mom * v
            stagnant += 1

        # Adaptive parameters
        if stagnant > 50:
            T *= 0.8
            lr *= 1.1
            v *= 0.5
            stagnant = 25
        if stagnant > 200:
            # Restart with perturbation
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            T = T_init
            lr = lr_init
            stagnant = 0

        # Anneal temperature
        T = max(T * cooling, 1e-5)
        # Decay learning rate slowly
        lr = max(lr_init * 0.9999**it, 0.001)

        if verbose and it % 500 == 0:
            print(f"    SG it={it}, obj={best_obj:.10f}, T={T:.6f}, lr={lr:.6f}, stag={stagnant}")

    return best_a, best_obj


def fw_optimize_fast(f0, time_limit, max_lp_calls=50):
    """Frank-Wolfe with LP oracle, but limited LP calls for speed."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    stale_n = 0
    again = 0
    lp_calls = 0
    m_relax = 1.0  # Start with no relaxation

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        # Solve LP (limited calls)
        if lp_calls < max_lp_calls:
            g = solve_lp_oracle(f, M * m_relax)
            lp_calls += 1
        else:
            g = None

        if g is None or g.sum() < 1e-15:
            m_relax = min(m_relax * 1.01, 1.1)
            stale_n += 1
            if stale_n > 50:
                # Fall back to gradient step
                conv = fast_convolve(f)
                conv_shifted = conv - conv.max()
                w = np.exp(conv_shifted / 0.001)
                w = w / w.sum()
                grad = softmax_gradient(w, f, 0.001)
                f = project_simplex(f - 0.01 * grad)
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.0
                stale_n = 0
            continue

        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-10:
            stale_n += 1
            if stale_n > 100:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in np.concatenate([np.linspace(0.01, 0.5, 10),
                                      np.linspace(0.5, 1.0, 5)]):
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-8:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                stale_n = 0
            else:
                stale_n += 1
        else:
            stale_n += 1

        if stale_n > 30:
            m_relax = min(m_relax * 1.02, 1.05)

    return best_f, best_obj


def peak_coordinate_refine(a0, time_limit):

def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_diverse_starts(n, parent, rng):
    """Generate diverse initializations."""
    starts = []
    if parent is not None:
        starts.append(resample_sequence(parent, n))

    # Uniform
    starts.append(np.ones(n) / n)

    # Cosine taper
    x = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 + np.cos(np.pi * x / max(n - 1, 1)))
    w = np.maximum(w, 0.01)
    starts.append(w / w.sum())

    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    starts.append(edge / edge.sum())

    # Two peaks
    g1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.1 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())

    # Random
    starts.append(rng.dirichlet(np.ones(n) * 0.5))

    return starts
