"""
Hybrid optimization for minimum peak autocorrelation step function.
Uses continuation-smoothed gradient with BB step size, Frank-Wolfe LP,
spectral flattening, multiplicative updates, L-BFGS refinement,
and multi-scale local search with diverse initializations.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_weights(conv, T):
    """Compute softmax weights for the convolution entries."""
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    s = p.sum()
    if s < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return p / s

def compute_gradient_fft(a, p):
    """
    Compute gradient of T*log(sum exp(c_k/T)) using FFT.
    grad[j] = 2 * sum_i a[i]*p[j+i] - a[j]*p[2j]
    """
    n = len(a)
    # rev_p[i] = p[2n-2-i]
    rev_p = p[::-1]  # length 2n-1
    # conv = a * rev_p (linear convolution, length n + 2n-1 - 1 = 3n-2)
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(rev_p, fft_size)
    conv = np.fft.irfft(A * B, fft_size)[:3 * n - 2]
    # h[j] = conv[2n-2-j] for j = 0..n-1
    h = conv[2 * n - 2 - np.arange(n)]
    grad = 2.0 * h
    # Subtract diagonal correction: a[j]*p[2j] where 2j < 2n-1
    diag_idx = np.arange(n)
    valid = 2 * diag_idx < 2 * n - 1
    grad[valid] -= a[valid] * p[2 * diag_idx[valid]]
    return grad

def build_lp_matrix(n):
    """Build sparse constraint matrix for LP oracle."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows), np.array(cols)

def solve_lp_oracle(f, M, n, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0. Returns normalized g or None."""
    c = -np.ones(n)
    vals = np.empty(len(row_idx))
    for idx in range(len(row_idx)):
        k = row_idx[idx]
        i = col_idx[idx]
        vals[idx] = f[k - i]
    A_mat = lil_matrix((2 * n - 1, n))
    A_mat[row_idx, col_idx] = vals
    A_mat = A_mat.tocsr()
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success and res.fun is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None

def continuation_smoothed(f0, time_limit, T_high=0.01, T_low=0.00005, n_steps=600):
    """
    Continuation method: smoothly anneal T from high to low.
    Uses projected gradient with Barzilai-Borwein step size.
    """
    a = np.maximum(f0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    # Logarithmic T schedule
    log_T_high = np.log(T_high)
    log_T_low = np.log(T_low)
    # Use a modified schedule: slow down at the end
    fracs = np.linspace(0, 1, n_steps)
    # Bias towards smaller T in the latter half
    fracs = fracs ** 1.5
    T_schedule = np.exp(log_T_high + fracs * (log_T_low - log_T_high))

    prev_a = a.copy()
    prev_g = None
    bb_alpha = 0.05  # initial step size

    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        T = T_schedule[it]

        conv = fast_convolve(a)
        p = softmax_weights(conv, T)
        g = compute_gradient_fft(a, p)

        # Barzilai-Borwein step size
        if prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            ss = np.dot(s_vec, s_vec)
            sy = np.dot(s_vec, y_vec)
            yy = np.dot(y_vec, y_vec)
            if ss > 1e-20 and abs(sy) > 1e-20:
                if it % 2 == 0:
                    bb_alpha = ss / abs(sy)
                else:
                    bb_alpha = sy / yy if yy > 1e-20 else bb_alpha
                bb_alpha = np.clip(bb_alpha, 1e-6, 0.5)
            else:
                bb_alpha = 0.03

        # Backtracking line search
        alpha = bb_alpha
        best_candidate = a
        best_val = conv.max()
        for _ in range(20):
            cand = project_simplex(a - alpha * g)
            cv = fast_convolve(cand)
            val = cv.max()
            if val < best_val - 1e-18:
                best_candidate = cand
                best_val = val
                break
            alpha *= 0.6
        a = best_candidate
        prev_a = a.copy()
        prev_g = g.copy()

        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()

    return best_a, best_obj

def multiplicative_grad(f0, time_limit, T_init=0.001, lr_init=0.02):
    """Multiplicative (exponential) gradient descent on smoothed objective."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    T = T_init
    lr = lr_init
    no_improve = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        conv = fast_convolve(a)
        p = softmax_weights(conv, T)
        g = compute_gradient_fft(a, p)
        # Multiplicative update
        a_new = a * np.exp(-lr * g)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            a = a_new
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 30:
            T *= 1.03
            lr *= 0.98
        if no_improve > 80:
            T *= 0.7
            lr = lr_init * 0.8
        if no_improve > 200:
            rng = np.random.default_rng((it * 104729 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            T = T_init
            lr = lr_init
            no_improve = 0
    return best_a, best_obj

def spectral_flatten(a, alpha=0.7):
    """Apply spectral flattening: modify DFT magnitude to be more flat."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    # Flatten: mag_new = mag^alpha / |mag^alpha|_norm
    if mag.max() > 1e-30:
        mag_flat = mag ** alpha
        a_new = np.fft.irfft(mag_flat * np.exp(1j * phase), fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            return a_new / s
    return a.copy()

def local_search(f0, time_limit, seed=42):
    """Multi-scale local search with spectral moves."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        improved = False
        # Random perturbations
        for scale in [0.0002, 0.0005, 0.001, 0.003, 0.008]:
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(a + pert)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break
        # Coordinate moves
        if not improved and it % 3 == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                for frac in [0.05, 0.1, 0.2, 0.4]:
                    candidate = a.copy()
                    amount = candidate[i] * frac
                    candidate[i] -= amount
                    candidate[j] += amount
                    candidate = np.maximum(candidate, 0)
                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break
        # Spectral move
        if not improved and it % 20 == 0:
            for alpha in [0.5, 0.7, 0.85]:
                candidate = spectral_flatten(a, alpha)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break
        # Pair of coordinate moves (2-opt)
        if not improved and it % 5 == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            k = int(rng.integers(0, n))
            if len(set([i, j, k])) == 3:
                for frac in [0.1, 0.2]:
                    candidate = a.copy()
                    amt = candidate[i] * frac
                    candidate[i] -= amt
                    candidate[j] += amt * 0.5
                    candidate[k] += amt * 0.5
                    candidate = np.maximum(candidate, 0)
                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break
        if not improved and it > 2000:
            break
    return best_a, best_obj

def fourier_random_init(n, seed=0):
    """Generate a random init with flat power spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1).bit_length()
    # Random phases, moderate magnitude variation
    mags = rng.chisquare(4, size=fft_size // 2 + 1)
    phases = rng.uniform(0, 2 * np.pi, size=fft_size // 2 + 1)
    A = mags * np.exp(1j * phases)
    # Make it Hermitian for real result
    A_full = np.zeros(fft_size, dtype=complex)
    half = fft_size // 2 + 1
    A_full[:half] = A
    A_full[half:] = np.conj(A[half-1:0:-1]) if fft_size // 2 > 0 else 0
    # Actually let's just do it properly
    a = np.fft.irfft(A, fft_size)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = rng.random(n)
        s = a.sum()
    return a / s

def lgbfgs_refine(f0, time_limit, T=0.0001):
    """Simple L-BFGS refinement on the smoothed objective."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    m_history = 8
    S = []  # list of s vectors
    Y = []  # list of y vectors
    prev_g = None
    prev_a = a.copy()
    gamma = 0.05  # initial step

    for it in range(5000):
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        conv = fast_convolve(a)
        p = softmax_weights(conv, T)
        g = compute_gradient_fft(a, p)

        # L-BFGS two-loop recursion
        d = g.copy()
        ns = min(m_history, len(S))
        alphas = np.zeros(ns)
        for i in range(ns - 1, -1, -1):
            s_i = S[i]
            y_i = Y[i]
            yy = np.dot(y_i, y_i)
            if yy > 1e-20:
                alphas[i] = np.dot(s_i, d) / yy
            d = d - alphas[i] * y_i

        # Scaling
        if ns > 0:
            s_last = S[ns - 1]
            y_last = Y[ns - 1]
            sy = np.dot(s_last, y_last)
            yy = np.dot(y_last, y_last)
            if yy > 1e-20 and sy > 1e-20:
                gamma_lb = sy / yy
            else:
                gamma_lb = 1.0 / max(np.linalg.norm(g), 1e-10)
        else:
            gamma_lb = 1.0 / max(np.linalg.norm(g), 1e-10)
        gamma_lb = np.clip(gamma_lb, 1e-8, 10.0)

        d = gamma_lb * d
        betas = np.zeros(ns)
        for i in range(ns):
            s_i = S[i]
            y_i = Y[i]
            yy = np.dot(y_i, y_i)
            if yy > 1e-20:
                betas[i] = np.dot(s_i, y_i) / yy
            d = d + (alphas[i] - betas[i]) * y_i

        d = -d  # descent direction

        # Line search with projection
        alpha = gamma_lb
        best_candidate = a
        best_val = conv.max()
        for _ in range(15):
            cand = project_simplex(a + alpha * d)
            cv = fast_convolve(cand)
            val = cv.max()
            if val < best_val - 1e-18:
                best_candidate = cand
                best_val = val
                break
            alpha *= 0.5
        a = best_candidate

        # Update history
        s_new = a - prev_a
        y_new = g - prev_g if prev_g is not None else g
        sy_check = np.dot(s_new, y_new)
        if sy_check > 1e-12:
            if len(S) >= m_history:
                S.pop(0)
                Y.pop(0)
            S.append(s_new)
            Y.append(y_new)
        prev_a = a.copy()
        prev_g = g.copy()

        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()

    return best_a, best_obj

def try_seq(a, label, best_ref):
    """Evaluate a sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_ref
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_ref
    if np.isfinite(score) and score < best_ref[1]:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return (a_norm.copy(), score)
    return best_ref

def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point: multi-phase hybrid search for minimum peak autocorrelation.
    Uses continuation smoothed gradient, FW-LP, multiplicative updates,
    L-BFGS, spectral flattening, and multi-scale local search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Using uniform init: score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Diverse initializations ===
    print("\n=== Phase 1: Diverse Initializations ===")
    t_phase = time.monotonic()

    inits = []
    # Parent
    inits.append(("parent", parent.copy()))
    # Fourier random at same n
    inits.append(("fourier_rand", fourier_random_init(n_parent, seed=seed)))
    # Random
    inits.append(("random", rng.random(n_parent) / rng.random(n_parent).sum()))
    # Structured: exponential decay
    x = np.arange(n_parent)
    for rate in [0.003, 0.01, 0.03]:
        exp_d = np.exp(-rate * x)
        inits.append((f"exp_{rate}", exp_d / exp_d.sum()))
    # Two-gaussian
    g1 = np.exp(-((x - 0.25 * n_parent) / (0.12 * n_parent)) ** 2)
    g2 = np.exp(-((x - 0.75 * n_parent) / (0.15 * n_parent)) ** 2)
    inits.append(("2gauss", (g1 + g2) / (g1 + g2).sum()))

    for name, init in inits:
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break
        t_alloc = min(remaining * 0.06, 30)
        a_opt, _ = fw_lp_optimize(init, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, f"init_{name}", (best_seq, best_score))

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Continuation smoothed gradient ===
    print("\n=== Phase 2: Continuation Smoothed Gradient ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    if remaining > 100:
        t_alloc = min(remaining * 0.35, 500)
        print(f"  Continuation T=0.01->0.00005, t={t_alloc:.0f}s")
        a_opt, _ = continuation_smoothed(best_seq, time_limit=t_alloc,
                                          T_high=0.01, T_low=0.00005, n_steps=800)
        best_seq, best_score = try_seq(a_opt, "continuation", (best_seq, best_score))

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Deep FW-LP ===
    print("\n=== Phase 3: Deep FW-LP ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    if remaining > 80:
        t_alloc = min(remaining * 0.25, 350)
        print(f"  Deep FW-LP, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, "deep_fw", (best_seq, best_score))

    # Perturbed FW-LP restarts
    for scale in [0.002, 0.005, 0.01, 0.02]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * scale)
        t_alloc2 = min(remaining * 0.1, 80)
        a_opt2, _ = fw_lp_optimize(f_pert, time_limit=t_alloc2)
        best_seq, best_score = try_seq(a_opt2, f"fw_pert_{scale}", (best_seq, best_score))

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Multiplicative gradient + L-BFGS ===
    print("\n=== Phase 4: Multiplicative + L-BFGS ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t_alloc = min(remaining * 0.2, 150)
        print(f"  Multiplicative grad, t={t_alloc:.0f}s")
        a_opt, _ = multiplicative_grad(best_seq, time_limit=t_alloc,
                                        T_init=0.001, lr_init=0.015)
        best_seq, best_score = try_seq(a_opt, "mult_grad", (best_seq, best_score))

    remaining = deadline - time.monotonic()
    if remaining > 50:
        t_alloc = min(remaining * 0.15, 100)
        print(f"  L-BFGS refine T=0.0001, t={t_alloc:.0f}s")
        a_opt, _ = lgbfgs_refine(best_seq, time_limit=t_alloc, T=0.0001)
        best_seq, best_score = try_seq(a_opt, "lbfgs", (best_seq, best_score))

    remaining = deadline - time.monotonic()
    if remaining > 40:
        t_alloc = min(remaining * 0.1, 60)
        print(f"  L-BFGS refine T=0.00003, t={t_alloc:.0f}s")
        a_opt, _ = lgbfgs_refine(best_seq, time_limit=t_alloc, T=0.00003)
        best_seq, best_score = try_seq(a_opt, "lbfgs_fine", (best_seq, best_score))

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Multi-n exploration ===
    print("\n=== Phase 5: Multi-n Exploration ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)

    for n_try in sorted(set([n_best // 2, n_best * 2, 512, 640, 896, 1024, 1280, 1536])):
        if n_try < 128 or n_try > 2500:
            continue
        if n_try == n_best:
            continue
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        start = np.interp(np.linspace(0, 1, n_try),
                          np.linspace(0, 1, n_best), best_seq)
        start = np.maximum(start, 0)
        ss = start.sum()
        start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try
        t_alloc = min(remaining * 0.15, 60)
        a_opt, _ = fw_lp_optimize(start, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, f"n{n_try}_fw", (best_seq, best_score))

        remaining = deadline - time.monotonic()
        if remaining > 40:
            t_alloc2 = min(remaining * 0.1, 30)
            a_opt2, _ = continuation_smoothed(a_opt, time_limit=t_alloc2,
                                               T_high=0.005, T_low=0.0001, n_steps=200)
            best_seq, best_score = try_seq(a_opt2, f"n{n_try}_cont", (best_seq, best_score))

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Final polish ===
    print("\n=== Phase 6: Final Polish ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Local search
    if remaining > 30:
        t_alloc = min(remaining * 0.25, 100)
        print(f"  Local search, t={t_alloc:.0f}s")
        a_opt, _ = local_search(best_seq, time_limit=t_alloc, seed=77777)
        best_seq, best_score = try_seq(a_opt, "local1", (best_seq, best_score))

    # Another local search with different seed
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t_alloc = min(remaining * 0.15, 50)
        a_opt, _ = local_search(best_seq, time_limit=t_alloc, seed=12321)
        best_seq, best_score = try_seq(a_opt, "local2", (best_seq, best_score))

    # Fine continuation
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t_alloc = min(remaining * 0.15, 40)
        a_opt, _ = continuation_smoothed(best_seq, time_limit=t_alloc,
                                          T_high=0.0003, T_low=0.00001, n_steps=300)
        best_seq, best_score = try_seq(a_opt, "fine_cont", (best_seq, best_score))

    # Tiny perturbations + quick optimization
    remaining = deadline - time.monotonic()
    for i in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        noise = rng.standard_normal(n_best) * 0.0003
        f_pert = project_simplex(best_seq + noise)
        a_opt, _ = lgbfgs_refine(f_pert, time_limit=min(remaining * 0.1, 8), T=0.00005)
        best_seq, best_score = try_seq(a_opt, f"micro_{i}", (best_seq, best_score))

    # Spectral flattening attempts
    remaining = deadline - time.monotonic()
    if remaining > 10:
        for alpha in [0.5, 0.6, 0.7, 0.8, 0.9]:
            remaining = deadline - time.monotonic()
            if remaining < 3:

            candidate = spectral_flatten(best_seq, alpha)
            a_opt, _ = local_search(candidate, time_limit=min(remaining * 0.3, 5), seed=alpha * 1000)
            best_seq, best_score = try_seq(a_opt, f"spectr_{alpha}", (best_seq, best_score))

    # Final verification
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
