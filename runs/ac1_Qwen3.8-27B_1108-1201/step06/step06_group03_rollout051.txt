import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def hankel_matvec(w, a):
    """Compute S @ a where S[i][j] = w[i+j] (Hankel matrix). O(n log n) via FFT."""
    n = len(a)
    w_flip = w[::-1]
    size = len(w_flip) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w_flip, fft_size)
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(W * A, fft_size)[:size]
    return conv[n - 1:2 * n - 1][::-1]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_active_weights(conv, K, temperature=0.01):
    """Compute weights for active peaks using top-K with softmax."""
    n_peaks = min(K, len(conv))
    peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
    pv = conv[peaks]
    pv_max = pv.max()
    if temperature <= 0:
        w = np.zeros(len(conv))
        w[peaks] = 1.0 / n_peaks
    else:
        w_peaks = np.exp((pv - pv_max) / temperature)
        w = np.zeros(len(conv))
        w[peaks] = w_peaks / w_peaks.sum()
    return w, peaks


def solve_qp_hankel(w_full, a_init, n, max_iters=200, lr=None):
    """
    Solve: minimize a^T S a subject to sum(a)=1, a>=0
    where S is Hankel with S[i][j] = w_full[i+j].
    Uses projected gradient with backtracking.
    """
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    if lr is None:
        # Estimate Lipschitz constant
        grad0 = hankel_matvec(w_full, a)
        lr = 0.5 / (np.max(np.abs(grad0)) + 1e-10)
        lr = min(lr, 2.0)

    best_a = a.copy()
    best_val = float(a @ hankel_matvec(w_full, a))

    for it in range(max_iters):
        grad = hankel_matvec(w_full, a)
        # Backtracking line search
        step = lr
        new_a = a
        new_val = best_val + 1
        for bt in range(20):
            cand = project_simplex(a - step * grad)
            val = float(cand @ hankel_matvec(w_full, cand))
            # Sufficient decrease
            predicted = float(best_val - step * (grad @ (a - cand)))
            if val < best_val - 0.1 * step * (grad @ (a - cand)):
                new_a = cand
                new_val = val
                break
            step *= 0.5

        if new_val < best_val - 1e-15:
            a = new_a
            best_val = new_val
            best_a = a.copy()
        elif it > 50:
            break

    return best_a, best_val

    Active-set QP optimization:
    - Maintain a set of K active peaks
    - Solve QP minimizing weighted sum of active peak values
    - Update active set based on progress
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    K = K_init

    t0 = time.monotonic()
    stagnant = 0
    it = 0
    temperature = 0.005 * best_obj

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # Compute active weights
        w, active_peaks = compute_active_weights(conv, K, temperature)
        w_full = np.zeros(2 * n - 1)
        w_full[active_peaks] = w[active_peaks]

        # Solve QP
        a_qp, qp_val = solve_qp_hankel(w_full, a, n, max_iters=100)

        # Evaluate QP solution
        conv_qp = fast_convolve(a_qp)
        qp_obj = conv_qp.max()

        if qp_obj < best_obj - 1e-14:
            a = a_qp
            best_obj = qp_obj
            best_a = a.copy()
            conv = conv_qp
            stagnant = 0
            # Gradually increase K
            K = min(K + 2, K_max)
            temperature *= 0.98
        else:
            # Blend with QP solution
            for alpha in [0.3, 0.5, 0.7]:
                a_blend = (1 - alpha) * a + alpha * a_qp
                a_blend = project_simplex(a_blend)
                conv_blend = fast_convolve(a_blend)
                if conv_blend.max() < best_obj - 1e-14:
                    a = a_blend
                    best_obj = conv_blend.max()
                    best_a = a.copy()
                    conv = conv_blend
                    break
            else:
                stagnant += 1
                # Adapt parameters
                if stagnant % 10 == 0:
                    temperature *= 1.1
                if stagnant > 30:
                    K = max(K_init, K // 2)
                    temperature = 0.005 * best_obj
                    # Perturb
                    rng = np.random.default_rng(it * 7919)
                    a = project_simplex(a + rng.standard_normal(n) * 0.003)
                    conv = fast_convolve(a)
                    stagnant = 0

        if verbose and it % 50 == 0:
            print(f"    QP-AS it={it}, obj={best_obj:.10f}, K={K}, stag={stagnant}")

    return best_a, best_obj


def penalty_smooth_optimize(a0, time_limit, mu_init=0.01, mu_decay=0.95, verbose=False):
    """
    Penalty-based smoothed optimization:
    minimize sum_k max(0, conv_k - t)^2 + t
    approximated by: minimize sum_k (conv_k - t)_+^2 + t
    Uses subgradient of the smoothed objective.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    mu = mu_init
    t0 = time.monotonic()
    it = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        t = conv.max()

        # Compute subgradient of the smoothed max
        # d/d a_j [sum_k max(0, conv_k - t)^2 + t] 
        # = 2 * sum_{k: conv_k > t} (conv_k - t) * d(conv_k)/da_j + dt/da
        # dt/da = (1/t) * sum_{k: conv_k = t} d(conv_k)/da (approximately)
        # For simplicity, use: grad = 2 * sum_{k in top} w_k * d(conv_k)/da

        top_k = min(30, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        threshold = pv.min() * 0.95
        weights = np.maximum(pv - threshold, 0)
        w_sum = weights.sum()
        if w_sum < 1e-15:
            weights = np.ones(len(peaks)) / len(peaks)
            w_sum = 1
        else:
            weights = weights / w_sum

        # Compute gradient: grad[j] = 2 * sum_i w_i * d(conv_i)/da_j
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if weights[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += weights[idx] * 2.0 * a[k - idx_arr]

        # Projected gradient step with backtracking
        lr = 0.1 / (np.max(np.abs(grad)) + 1e-10)
        lr = min(lr, 0.5)
        improved = False
        for frac in [1.0, 0.5, 0.25, 0.1]:
            candidate = project_simplex(a - frac * lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_obj - 1e-15:
                a = candidate
                best_obj = val
                best_a = a.copy()
                improved = True
                break

        if not improved:
            stagnant += 1
            mu *= mu_decay
            if stagnant > 50:
                rng = np.random.default_rng(it * 31)
                a = project_simplex(a + rng.standard_normal(n) * 0.002)
                stagnant = 0
                mu = mu_init * 0.5
        else:
            stagnant = 0

        if verbose and it % 100 == 0:
            print(f"    PenSmooth it={it}, obj={best_obj:.10f}, stag={stagnant}")

    return best_a, best_obj


def simulated_annealing(a0, time_limit, verbose=False):
    """
    Simulated annealing with targeted moves:
    - Transfer mass between pairs of coordinates
    - Focus moves on coordinates contributing to top peaks
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    T = 0.001 * best_obj
    T_min = 1e-7
    cool_rate = 0.9995

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        # Process in batches for efficiency
        end_time = min(time.monotonic() + 0.05, t0 + time_limit)
        while time.monotonic() < end_time:
            it += 1
            conv = fast_convolve(a)
            cur_obj = conv.max()

            # Pick a move
            if it % 50 == 0:
                # Large move: random coordinate
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
            else:
                # Targeted: contribute to top peak
                peak_idx = int(np.argmax(conv))
                js = max(0, peak_idx - n + 1)
                je = min(n - 1, peak_idx)
                if js <= je:
                    i = int(rng.integers(js, je + 1))
                else:
                    i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))

            if i == j:
                continue

            # Transfer mass
            delta = a[i] * rng.uniform(0.01, 0.1)
            if delta < 1e-14:
                continue
            if rng.random() < 0.5:
                candidate = a.copy()
                candidate[i] -= delta
                candidate[j] += delta
            else:
                delta = a[j] * rng.uniform(0.01, 0.1)
                if delta < 1e-14:
                    continue
                candidate = a.copy()
                candidate[j] -= delta
                candidate[i] += delta

            if candidate[i] < -1e-12 or candidate[j] < -1e-12:
                continue
            candidate = np.maximum(candidate, 0)
            cs = candidate.sum()
            candidate = candidate / cs

            new_obj = fast_convolve(candidate).max()
            delta_obj = new_obj - cur_obj

            if delta_obj < 0 or rng.random() < np.exp(-delta_obj / (T + 1e-15)):
                a = candidate
                if new_obj < best_obj - 1e-15:
                    best_obj = new_obj
                    best_a = a.copy()

            T = max(T * cool_rate, T_min)

        if verbose and it % 1000 == 0:
            print(f"    SA it={it}, obj={best_obj:.10f}, T={T:.2e}")

    return best_a, best_obj


def pairwise_coordinate_descent(a0, time_limit, verbose=False):
    """
    Pairwise coordinate transfer: for each peak, try to reduce it by
    transferring mass from contributing coordinates to random distant ones.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    step_fracs = [0.01, 0.005, 0.002, 0.001, 0.0005]
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        n_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        improved = False
        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue

            # Get contributing pairs sorted by contribution
            contribs = []
            for j in range(js, je + 1):
                c = a[j] * a[k - j]
                contribs.append((c, j, k - j))
            contribs.sort(reverse=True)

            for (c, j1, j2) in contribs[:8]:
                if c < 1e-10:
                    break
                for src in [j1, j2]:
                    if a[src] < 1e-8:
                        continue
                    for sf in step_fracs:
                        delta = a[src] * sf
                        # Try a few random targets
                        for _ in range(3):
                            tgt = int(rng.integers(0, n))
                            if tgt == src:
                                continue
                            candidate = a.copy()
                            candidate[src] -= delta
                            candidate[tgt] += delta
                            candidate = np.maximum(candidate, 0)
                            cs = candidate.sum()
                            candidate = candidate / cs
                            val = fast_convolve(candidate).max()
                            if val < best_obj - 1e-15:
                                a = candidate
                                best_obj = val
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
            if improved:
                break

        if not improved:
            # Perturb to escape
            if it % 100 == 0:
                scale = 0.001 * min(1.0, it / 1000)
                a = project_simplex(a + rng.standard_normal(n) * scale)
            elif it > 5000:
                break

        if verbose and it % 200 == 0:
            print(f"    PCD it={it}, obj={best_obj:.10f}")

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def spectral_init(n, k=0):
    """Generate spectral initializations using various frequency structures."""
    x = np.arange(n, dtype=np.float64)
    starts = []

    # Uniform
    starts.append(np.ones(n) / n)

    # Cosine taper
    w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    starts.append(np.maximum(w, 0.01) / np.maximum(w, 0.01).sum())

    # Sinusoidal modulation
    w = 1.0 + 0.3 * np.sin(2 * np.pi * k * x / n + phi_init(k))
    starts.append(np.maximum(w, 0.1) / np.maximum(w, 0.1).sum())

    # Multi-frequency
    w = 1.0 + 0.2 * np.sin(2 * np.pi * 3 * x / n) + 0.1 * np.cos(4 * np.pi * x / n)
    starts.append(np.maximum(w, 0.1) / np.maximum(w, 0.1).sum())

    # Edge-enhanced
    edge = 1.0 / (np.abs(x - n / 2) / (n / 2) + 0.3)
    starts.append(edge / edge.sum())

    # Zipper pattern
    w = np.where(x % 2 == 0, 1.0, 0.6)
    starts.append(w / w.sum())

    return starts


def phi_init(k):
    """Phase for spectral initialization."""
    return 0.5 * np.pi * k


def quick_projected_gradient(a0, time_limit):
    """Quick projected gradient for initial optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.05

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        # Subgradient of max
        peak_idx = int(np.argmax(conv))
        grad = np.zeros(n)
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx_arr = np.arange(js, je + 1)
        grad[idx_arr] = 2.0 * a[peak_idx - idx_arr]

        candidate = project_simplex(a - lr * grad)
        val = fast_convolve(candidate).max()
        if val < best_obj - 1e-15:
            a = candidate
            best_obj = val
            best_a = a.copy()
        else:
            lr *= 0.9
            if lr < 1e-6:
                break

    return best_a, best_obj


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle (kept from previous approach for reliability)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    stagnant = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.2:
            break

        g = solve_lp_oracle(f, M * 1.002)
        if g is None or g.sum() < 1e-15:
            g = solve_lp_oracle(f, M * 1.01)
        if g is None or g.sum() < 1e-15:
            stagnant += 1
            if stagnant > 15:
                rng = np.random.default_rng(it * 7919 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue

        g = g / g.sum()
        if np.max(np.abs(g - f)) < 1e-10:
            stagnant += 1
            continue

        # Line search
        best_alpha = 0.0
        best_val = M + 1e-14
        for alpha in np.linspace(0.01, 1.0, 20):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        if stagnant > 20:
            rng = np.random.default_rng(it * 31)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            M = fast_convolve(f).max()
            stagnant = 0

    return best_f, best_obj


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search for minimizing
    2*n*max(conv(a,a))/(sum(a))^2.

    Phases:
    1. Multi-n, multi-start quick exploration (180s)
    2. QP active-set deep optimization (450s)
    3. FW-LP cross-check (250s)
    4. Penalty smoothing + SA (300s)
    5. Final coordinate descent polish (remaining)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")

    n_parent = len(parent)

    # === Phase 1: Multi-n, multi-start exploration (180s) ===
    print("\n=== Phase 1: Multi-start Exploration ===")
    t_phase1 = time.monotonic()

    # Candidate lengths
    base_n = n_parent if 256 <= n_parent <= 2048 else 768
    lengths = sorted(set([
        base_n, max(256, base_n - 128), base_n + 128,
        384, 512, 640, 768, 896, 1024, 1152, 1280
    ]))
    lengths = [l for l in lengths if 256 <= l <= 1536]

    phase1_results = []
    for n_try in lengths:
        remaining = deadline - time.monotonic() - 1100
        if remaining < 30:
            break

        starts = spectral_init(n_try, k=n_try % 5 + 1)

        # Also add parent resampled
        if parent is not None:
            starts.append(resample_sequence(parent, n_try))

        # Random Dirichlet
        starts.append(rng.dirichlet(np.ones(n_try) * 0.5))

        for si, start in enumerate(starts[:5]):
            remaining = deadline - time.monotonic() - 1050
            if remaining < 8:
                break
            t_alloc = min(remaining, 12)
            a_opt, _ = quick_projected_gradient(start, time_limit=t_alloc)
            label = f"pg_n{n_try}_s{si}"
            sc = try_seq(a_opt, label)
            if np.isfinite(sc):
                phase1_results.append((sc, a_opt, n_try))

    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase1:.0f}s)")

    # Sort and keep top candidates
    phase1_results.sort(key=lambda x: x[0])

    # === Phase 2: QP active-set deep optimization (450s) ===
    print("\n=== Phase 2: QP Active-Set Optimization ===")
    t_phase2 = time.monotonic()

    # Optimize top few candidates from Phase 1
    candidates = [best_seq] if best_seq is not None else [parent]
    for (sc, a_opt, n_try) in phase1_results[:3]:
        if a_opt is not None and len(a_opt) != len(candidates[0]):
            candidates.append(a_opt)

    for ci, cand in enumerate(candidates[:3]):
        remaining = deadline - time.monotonic() - 700
        if remaining < 30:
            break
        t_alloc = min(remaining, 150)
        print(f"  QP-AS on candidate {ci} (n={len(cand)}), t={t_alloc:.0f}s")
        a_opt, _ = qp_active_set_optimize(cand, time_limit=t_alloc, K_init=30, K_max=150)
        try_seq(a_opt, f"qp_as_{ci}")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase2:.0f}s)")

    # === Phase 3: FW-LP cross-check (250s) ===
    print("\n=== Phase 3: FW-LP Optimization ===")
    t_phase3 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Main FW
        remaining = deadline - time.monotonic() - 450
        if remaining > 30:
            t_alloc = min(remaining, 200)
            print(f"  FW-LP main (n={n_best}), t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "fw_main")

        # Perturbed restarts
        for scale in [0.002, 0.005, 0.01]:
            remaining = deadline - time.monotonic() - 250
            if remaining < 20:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 50)
            a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"fw_pert_{scale}")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase3:.0f}s)")

    # === Phase 4: Penalty smoothing + SA (300s) ===
    print("\n=== Phase 4: Penalty + SA ===")
    t_phase4 = time.monotonic()

    if best_seq is not None:
        # Penalty smoothing
        remaining = deadline - time.monotonic() - 200
        if remaining > 20:
            t_alloc = min(remaining, 120)
            print(f"  Penalty smooth, t={t_alloc:.0f}s")
            a_opt, _ = penalty_smooth_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "penalty")

        # Simulated annealing
        remaining = deadline - time.monotonic() - 120
        if remaining > 20:
            t_alloc = min(remaining, 120)
            print(f"  SA, t={t_alloc:.0f}s")
            a_opt, _ = simulated_annealing(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "sa")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase4:.0f}s)")

    # === Phase 5: Final coordinate descent polish (remaining) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase5 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_alloc = min(remaining, 120)
            print(f"  Pairwise CD, t={t_alloc:.0f}s")
            a_opt, _ = pairwise_coordinate_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "final_pcd")

        # One more QP pass on the result
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_alloc = min(remaining, 40)
            a_opt, _ = qp_active_set_optimize(best_seq, time_limit=t_alloc, K_init=50, K_max=200)
            try_seq(a_opt, "final_qp")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase5:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
