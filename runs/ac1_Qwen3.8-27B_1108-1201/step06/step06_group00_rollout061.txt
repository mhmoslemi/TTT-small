    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

import numpy as np
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_value_and_grad(f, beta):
    """Compute LSE value and gradient for the auto-convolution."""
    n = len(f)
    R = fast_convolve(f)
    bc = beta * R
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    lse_val = (bc_max + np.log(exp_bc.sum())) / beta
    
    # Gradient: grad[j] = 2 * sum_k w[k] * f[k-j] for valid k-j
    # This is a cross-correlation
    grad = np.zeros(n)
    for j in range(n):

        k_hi = min(j + n - 1, 2 * n - 2)
        if k_lo <= k_hi:

            grad[j] = 2.0 * np.dot(w[k_lo:k_hi+1], f[m])
    return lse_val, grad, R

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():

    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(f, beta):
    """Compute LSE value and gradient using strided matrix for speed."""
    n = len(f)
    R = fast_convolve(f)  # length 2n-1
    bc = beta * R
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w_sum = exp_bc.sum()
    w = exp_bc / w_sum
    lse_val = (bc_max + np.log(w_sum)) / beta
    
    # grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * f[m]
    # Use sliding window view for efficient matrix-vector product
    from numpy.lib.stride_tricks import sliding_window_view
    W_mat = sliding_window_view(w, n)  # shape (n, n)
    grad = 2.0 * (W_mat @ f)
    return lse_val, grad, R

This is cleaner. Let me also add the coordinate descent:

For coordinate descent, I'll use the following approach:
1. Maintain the auto-correlation R explicitly
2. For each coordinate i, compute the optimal change using the current R
3. Update R incrementally

The incremental update: if f[i] changes by delta, then for each k in [i, i+n-1] (clipped to [0, 2n-2]):
R[k] += delta * (f[k-i] + f[k-i]) + (delta^2 if k == 2*i)
Wait, more carefully:
If f[i] → f[i] + delta:
R[k] = sum_j f[j]*f[k-j]
Terms involving f[i]:
- j=i: f[i]*f[k-i] → (f[i]+delta)*f[k-i] = f[i]*f[k-i] + delta*f[k-i]
- j=k-i: f[k-i]*f[i] → f[k-i]*(f[i]+delta) = f[k-i]*f[i] + delta*f[k-i]

If i ≠ k-i: total change = 2*delta*f[k-i]
If i = k-i (k=2i): the term is f[i]^2 → (f[i]+delta)^2, change = 2*delta*f[i] + delta^2

So: R[k] += 2*delta*f[k-i] + delta^2*I[k==2i] (where I is indicator)
And we also need: k-i must be in [0, n-1], and k must be in [0, 2n-2].

For the 1D subproblem at coordinate i:
We want to find delta (with -f[i] <= delta <= M-f[i] for some M to keep non-negative, and keeping sum=1) that minimizes:
max_k (R[k] + 2*delta*f[k-i] + delta^2*I[k==2i])

But we also need to maintain sum(f) = 1. If we change f[i] by delta, we need to compensate by changing some other coordinate. For simplicity, let's just change f[i] and then re-normalize. The re-normalization scales all f by 1/(1+delta) (approximately, for small delta). This changes the objective in a complicated way.

Actually, since the objective is scale-invariant (2*n*max_conv/(sum)^2), and we're working with sum=1, if I change f[i] by delta without compensating, the new sum is 1+delta, and the new objective is:
2*n*max(R_new) / (1+delta)^2

where R_new[k] = R[k] + 2*delta*f[k-i] + delta^2*I[k==2i] (approximately, since the change in f affects all terms involving f[i]).

Wait, this approximation is only valid if we don't change the other f[j]. But when we re-normalize, all f values change slightly. For small delta, the effect of re-normalization is second-order, so the approximation should be good for small steps.

Let me just use the simple approach: change f[i] by delta, re-normalize, compute the new max R, and use ternary search on delta.

        print(f"  CoordDesc n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = coordinate_descent(current, time_limit=t_alloc, n_passes=50, top_k=20)
        try_seq(f_opt, "CoordDesc")

    # Phase 3: Multi-start with perturbations
    print(f"\n=== Phase 3: Multi-start Exploration ===")
    t3_end = deadline - 120
    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_starts = 0
    while time.monotonic() < t3_end and n_starts < 12:
        n_starts += 1
        remaining = t3_end - time.monotonic()
        if remaining < 40:
            break
        t_sub = remaining / (12 - n_starts + 1) * 0.8
        if t_sub < 20:
            t_sub = 20
        if np.random.rand() < 0.5:
            f_init = freq_perturb(current, rng, strength=0.02 + 0.02 * n_starts)
        else:
            f_init = time_perturb(current, rng, strength=0.003 + 0.002 * n_starts)
        beta_short = [50, 100, 300, 800]
        f_opt, _ = lse_gradient_descent(f_init, time_limit=min(t_sub, 60), beta_list=beta_short, verbose=False)
        try_seq(f_opt, f"Multi-{n_starts}")
        # Quick coord descent on the result
        if time.monotonic() < t3_end - 20:
            f_opt2, _ = coordinate_descent(f_opt, time_limit=min(30, t3_end - time.monotonic() - 5),
                                           n_passes=10, top_k=15, verbose=False)
            try_seq(f_opt2, f"Multi-{n_starts}-cd")
        current = best_tracker[1] if best_tracker[1] is not None else current

    # Phase 4: Multi-length exploration
    print(f"\n=== Phase 4: Multi-length Exploration ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current)
    lengths_to_try = sorted(set([
        max(256, n_cur - 64), max(256, n_cur + 64),
        max(256, int(n_cur * 0.9)), max(256, int(n_cur * 1.1)),
        max(256, n_cur - 128), max(256, n_cur + 128),
    ]))
    for nc in lengths_to_try:
        if time.monotonic() > deadline - 60:
            break
        if nc == n_cur or nc > 2000 or nc < 128:
            continue
        t_rem = deadline - 60 - time.monotonic()
        if t_rem < 30:
            break
        f_interp = interpolate_to_length(current, nc)
        beta_short = [30, 80, 200, 500]
        f_opt, _ = lse_gradient_descent(f_interp, time_limit=min(60, t_rem * 0.4),
                                        beta_list=beta_short, verbose=False)
        try_seq(f_opt, f"Len-{nc}")
        current = best_tracker[1] if best_tracker[1] is not None else current

    # Phase 5: Final high-beta polish
    print(f"\n=== Phase 5: Final Polish ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current)
    t_rem = deadline - time.monotonic()
    if t_rem > 30:
        beta_final = [500, 1000, 2000, 5000, 10000]
        f_opt, _ = lse_gradient_descent(current, time_limit=min(80, t_rem * 0.6),
                                        beta_list=beta_final)
        try_seq(f_opt, "Final-LSE")

    # Final coordinate descent polish
    current = best_tracker[1] if best_tracker[1] is not None else current
    t_rem = deadline - time.monotonic()
    if t_rem > 20:
        f_opt, _ = coordinate_descent(current, time_limit=min(50, t_rem), n_passes=20, top_k=10)
        try_seq(f_opt, "Final-CD")

    # Final verification
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using uniform fallback")
    fallback = np.ones(768) / 768.0
    return [float(x) for x in fallback]
