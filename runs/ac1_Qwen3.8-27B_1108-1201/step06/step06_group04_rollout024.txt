import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient_pecak(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    idx = np.arange(js, je + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad

def frank_wolfe_peak(a0, time_limit, k_peaks=5, verbose=False):
    """Frank-Wolfe method with top-k peak weighting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    lr = 2.0 / 3.0  # Standard FW step size (1/t + 1 form simplifies to 2/3)
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        cmax = conv.max()
        
        # Weight top-k peaks
        k_use = min(k_peaks, len(conv))
        peaks = np.argpartition(conv, -k_use)[-k_use:]
        peak_vals = conv[peaks]
        
        # Normalize weights (proportional to excess over mean)
        mean_c = conv.mean()
        weights = np.maximum(peak_vals - mean_c, 0.01)
        weights = weights / weights.sum()
        
        # Combined gradient
        grad = np.zeros(n)
        for j in range(len(peaks)):
            grad += weights[j] * compute_gradient_peak(a, int(peaks[j]))
        
        # FW step: find s in argmin grad . x over simplex
        # This is the coordinate with minimum gradient (or uniform if all positive)
        s_idx = np.argmin(grad)
        s = np.zeros(n)
        s[s_idx] = 1.0
        
        # Step size: backtracking line search
        best_t = 0.0
        best_val = cmax
        for t_trial in [2.0/(it+2), 1.0/(it+1.0), 1.0/(2*(it+1))]:
            candidate = (1.0 - t_trial) * a + t_trial * s
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_t = t_trial
        
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * s
        
        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
            no_improve = 0
    
    return best_a, best_obj

    it = 0

            break

        sorted_idx = np.argsort(peak_vals)[::-1]
        weights = np.zeros(k_use)
        for rank, idx in enumerate(sorted_idx):
            weights[idx] = 1.0 / (1.0 + 0.5 * rank)
        weights = weights / weights.sum()
    else:
        weights = np.ones(k_use) / k_use

    grad = np.zeros(n)
    for j in range(len(peaks)):
        k = int(peaks[j])
        w = weights[j]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        grad[idx] += w * 2.0 * a[k - idx]
    return grad


def mirror_descent(a0, time_limit, eta_init=0.1, k_peaks=10, verbose=False):
    """Mirror descent (entropic) for min max conv(a,a) on simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    eta = eta_init
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break

        conv = fast_convolve(a)

        grad = compute_weighted_grad(a, conv, k_peaks, 'auto')

        # Mirror descent update
        log_ratio = -eta * grad
        log_ratio = np.clip(log_ratio, -40, 40)
        mult = np.exp(log_ratio)
        a_new = a * mult
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum

        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-11:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            if no_improve % 15 == 0:
                eta *= 0.85
            if no_improve > 150:
                rng = np.random.default_rng((it * 7919 + 13) % (2**31))
                scale = 0.002 + 0.002 * min(no_improve // 150, 5)
                a = project_simplex(best_a + rng.standard_normal(n) * scale)
                eta = eta_init * 0.5
                no_improve = 0
            if no_improve > 2000:
                break

        if verbose and it % 500 == 0:
            print(f"    MD it={it}, obj={best_obj:.10f}, eta={eta:.6f}, no_imp={no_improve}")

    return best_a, best_obj


def nesterov_subgrad(a0, time_limit, lr_init=0.01, momentum_init=0.9, k_peaks=5,
                     verbose=False):
    """Nesterov-accelerated subgradient method."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = momentum_init
    y = a.copy()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break

        y = a + mom * v
        y = np.maximum(y, 0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = a.copy()
        else:
            y = y / y_sum

        conv_y = fast_convolve(y)
        k_use = min(k_peaks, len(conv_y))
        grad = compute_weighted_grad(y, conv_y, k_use, 'auto')

        v_new = project_simplex(y - lr * grad) - y
        a = y + v_new
        a = project_simplex(a)
        v = v_new

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-11:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9997
        mom = min(0.995, mom + 0.00002)

        if no_improve > 100:
            lr *= 0.9
            mom *= 0.98
        if no_improve > 500:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = lr_init * 0.5
            mom = momentum_init
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    NS it={it}, obj={best_obj:.10f}, lr={lr:.6f}")

    return best_a, best_obj


def subgrad_backtrack(a0, time_limit, lr_init=1.0, k_peaks=3, verbose=False):
    """Projected subgradient with backtracking line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = lr_init

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break

        conv = fast_convolve(a)
        grad = compute_weighted_grad(a, conv, k_peaks, 'auto')

        # Backtracking line search
        current_lr = lr
        improved = False
        for _ in range(10):
            candidate = project_simplex(a - current_lr * grad)
            obj = fast_convolve(candidate).max()
            if obj < best_obj - 1e-12:
                a = candidate
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
                improved = True
                break
            current_lr *= 0.5

        if not improved:
            no_improve += 1
            lr *= 0.9
            if no_improve > 200:
                rng = np.random.default_rng((it * 31 + 7) % (2**31))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                lr = lr_init * 0.1
                no_improve = 0

    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False):
    """Cyclic coordinate descent with 1D search per coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        improved = False
        # Cycle through coordinates
        rng = np.random.default_rng((it * 104729) % (2**31))
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit - 0.05:
                break

            original = a[i]
            if original < 1e-12:
                continue

            # Try fractions of current value
            best_val = best_obj
            best_frac = 1.0
            for frac in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]:
                trial = original * frac
                if trial < 0:
                    continue
                a[i] = trial
                # Renormalize
                a_sum = a.sum()
                if a_sum < 1e-15:
                    a[i] = original
                    continue
                a_norm = a / a_sum
                obj = fast_convolve(a_norm).max()
                if obj < best_val - 1e-12:
                    best_val = obj
                    best_frac = frac
                a[i] = original

            if best_frac != 1.0:
                a[i] = original * best_frac
                a = a / a.sum()
                improved = True
                if best_val < best_obj - 1e-12:
                    best_obj = best_val
                    best_a = a.copy()

        if not improved and it > 50:
            break

    return best_a, best_obj


def qp_polish(a0, time_limit, k_peaks=5, verbose=False):
    """QP polish: solve min a^T W a where W comes from peak-weighted correlation."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        conv = fast_convolve(a)
        k_use = min(k_peaks, len(conv))
        peaks = np.argpartition(conv, -k_use)[-k_use:]
        peak_vals = conv[peaks]
        excess = np.maximum(peak_vals - conv.mean(), 1e-6)
        weights = excess / excess.sum()

        # Build weight vector w of length 2n-1
        w = np.zeros(2 * n - 1)
        for j in range(len(peaks)):
            w[int(peaks[j])] = weights[j]

        # Solve min a^T W a s.t. sum a = 1, a >= 0
        # Active set method
        active = set(np.where(a > 1e-10)[0])
        if not active:
            active = set(range(n))

        for _ in range(100):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            if len(active) <= 1:
                break

            act = sorted(active)
            k = len(act)
            if k > 500:
                # Too large for dense solve, subsample
                act = list(rng_perm((it, n))[:500]) if False else act[:500]
                k = len(act)

            W_AA = np.array([[w[act[i] + act[j]] for j in range(k)] for i in range(k)])
            W_AA_reg = W_AA + 1e-10 * np.eye(k)

            try:
                x = np.linalg.solve(W_AA_reg, np.ones(k))
            except np.linalg.LinAlgError:
                try:
                    x = np.linalg.lstsq(W_AA_reg, np.ones(k), rcond=None)[0]
                except Exception:
                    x = np.ones(k) / k

            x_sum = x.sum()
            if abs(x_sum) < 1e-14:
                x = np.ones(k) / k

            a_act = x / x_sum

            # Check negative
            min_val = a_act.min()
            if min_val < -1e-8:
                min_pos = np.argmin(a_act)
                active.discard(act[min_pos])
                continue

            # Form candidate
            a_new = np.zeros(n)
            for i2, idx in enumerate(act):
                a_new[idx] = max(0.0, a_act[i2])
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new = a_new / s

            # Check KKT for inactive set (sample if too large)
            Wa = np.zeros(n)
            for i2 in range(n):
                js = max(0, i2 + 1 - (2 * n - 2))
                je = min(n - 1, (2 * n - 2) - i2)
                if js <= je:
                    Wa[i2] = np.dot(w[i2 + js:i2 + je + 1], a_new[js:je + 1])

            mu = 2.0 * np.mean([Wa[idx] for idx in act])
            inactive = set(range(n)) - active
            min_viol = 0.0
            min_j = -1
            # Sample inactive for efficiency
            if len(inactive) > 200:
                sample = list(np.random.default_rng((it * 3 + _)).choice(sorted(inactive), 200))
            else:
                sample = list(inactive)
            for j in sample:
                viol = 2.0 * Wa[j] - mu
                if viol < min_viol:
                    min_viol = viol
                    min_j = j
            if min_j >= 0 and min_viol < -1e-6:
                active.add(min_j)
                continue

            # Converged
            obj = fast_convolve(a_new).max()
            if obj < best_obj - 1e-10:
                best_obj = obj
                best_a = a_new.copy()
            a = a_new
            break

    return best_a, best_obj


def interp_to_n(parent, n_target):
    """Interpolate parent sequence to target length."""
    n_src = len(parent)
    if n_target == n_src:
        return parent.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    result = np.interp(x_tgt, x_src, parent)
    result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return result / s


def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g / g.sum())
    # Two peaks
    g2 = (np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.08 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.08 * n)) ** 2))
    inits.append(g2 / g2.sum())
    # Sine
    s1 = np.maximum(0.5 + 0.5 * np.sin(2 * np.pi * x / n), 0.01)
    inits.append(s1 / s1.sum())
    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.3
    inits.append(alt / alt.sum())
    # Edge-heavy
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge / edge.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search for auto-convolution minimization.
    Strategy:
    1. Quick multi-n exploration with mirror descent
    2. Deep mirror descent + QP polish on best
    3. Nesterov subgradient for fine tuning
    4. Coordinate descent for final polish
    5. Verification
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        try_seq(parent, "uniform384")

    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")

    # === Phase 1: Multi-n exploration (~300s) ===
    print("\n=== Phase 1: Multi-n exploration ===")
    t1 = time.monotonic()

    candidate_ns = []
    # Include parent and neighbors
    for n_try in [n_parent, n_parent + 128, n_parent + 256,
                  max(256, n_parent - 128),
                  1024, 1280, 1536, 2048]:
        if n_try >= 64 and n_try not in candidate_ns:
            candidate_ns.append(n_try)
    candidate_ns = sorted(candidate_ns)

    time_per_n = min(60, (deadline - time.monotonic() - 800) / max(1, len(candidate_ns)))

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 800
        if remaining < 20:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = interp_to_n(parent, n_try)

        t_alloc = min(time_per_n, remaining * 0.9)
        print(f"  MD n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = mirror_descent(start, time_limit=t_alloc, eta_init=0.1, k_peaks=10)
        try_seq(a_opt, f"md_n{n_try}")

        # Quick QP polish
        remaining2 = deadline - time.monotonic() - 800
        if remaining2 > 15:
            a_opt2, _ = qp_polish(a_opt, time_limit=min(remaining2 * 0.5, 15), k_peaks=5)
            try_seq(a_opt2, f"qp_n{n_try}")

    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")

    # === Phase 2: Deep optimization on best (~700s) ===
    print("\n=== Phase 2: Deep optimization ===")
    t2 = time.monotonic()

    if best_seq is not None:
        # Mirror descent with various parameters
        configs = [
            {'eta_init': 0.05, 'k_peaks': 5},
            {'eta_init': 0.1, 'k_peaks': 10},
            {'eta_init': 0.2, 'k_peaks': 3},
            {'eta_init': 0.03, 'k_peaks': 15},
        ]

        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 400
            if remaining < 20:
                break
            t_alloc = min(remaining, 150)
            print(f"  MD deep cfg{ci} (eta={cfg['eta_init']}, k={cfg['k_peaks']}), t={t_alloc:.0f}s")
            a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"md_deep_{ci}")

        # QP polish
        remaining = deadline - time.monotonic() - 350
        if remaining > 20:
            t_alloc = min(remaining, 100)
            print(f"  QP polish k=5, t={t_alloc:.0f}s")
            a_opt, _ = qp_polish(best_seq, time_limit=t_alloc, k_peaks=5)
            try_seq(a_opt, "qp_deep_5")

            remaining = deadline - time.monotonic() - 350
            if remaining > 20:
                t_alloc = min(remaining, 80)
                print(f"  QP polish k=10, t={t_alloc:.0f}s")
                a_opt, _ = qp_polish(best_seq, time_limit=t_alloc, k_peaks=10)
                try_seq(a_opt, "qp_deep_10")

    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")

    # === Phase 3: Nesterov + subgradient (~400s) ===
    print("\n=== Phase 3: Nesterov & Subgradient ===")
    t3 = time.monotonic()

    if best_seq is not None:
        for lr_i in [0.005, 0.01, 0.02, 0.003]:
            remaining = deadline - time.monotonic() - 200
            if remaining < 15:
                break
            t_alloc = min(remaining, 100)
            print(f"  Nesterov lr={lr_i}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=t_alloc, lr_init=lr_i)
            try_seq(a_opt, f"nesterov_{lr_i}")

        # Subgradient with backtracking
        remaining = deadline - time.monotonic() - 150
        if remaining > 15:
            t_alloc = min(remaining, 80)
            print(f"  Subgrad backtrack, t={t_alloc:.0f}s")
            a_opt, _ = subgrad_backtrack(best_seq, time_limit=t_alloc, lr_init=0.5, k_peaks=3)
            try_seq(a_opt, "subgrad_bt")

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")

    # === Phase 4: Coordinate descent + fine polish (~250s) ===
    print("\n=== Phase 4: Coordinate Descent & Fine Polish ===")
    t4 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 100
        if remaining > 20:
            t_alloc = min(remaining, 100)
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_desc")

        # Fine mirror descent (very small eta)
        remaining = deadline - time.monotonic() - 80
        if remaining > 15:
            t_alloc = min(remaining, 60)
            print(f"  Fine MD eta=0.01, t={t_alloc:.0f}s")
            a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc, eta_init=0.01, k_peaks=3)
            try_seq(a_opt, "fine_md")

        # QP with k=3 (focus on dominant peak)
        remaining = deadline - time.monotonic() - 50
        if remaining > 15:
            t_alloc = min(remaining, 40)
            print(f"  QP k=3, t={t_alloc:.0f}s")
            a_opt, _ = qp_polish(best_seq, time_limit=t_alloc, k_peaks=3)
            try_seq(a_opt, "qp_k3")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # === Phase 5: Final perturbations (~100s) ===
    print("\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple tiny perturbations with quick refinement
        for pi in range(8):
            remaining = deadline - time.monotonic() - 5
            if remaining < 5:
                break
            noise_scales = [0.001, 0.0005, 0.002, 0.0003, 0.001, 0.003, 0.0007, 0.0015]
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scales[pi % len(noise_scales)])
            t_alloc = min(remaining, 10)
            a_opt, _ = mirror_descent(f_pert, time_limit=t_alloc, eta_init=0.02, k_peaks=5)
            try_seq(a_opt, f"perturb_{pi}")

        # One final Nesterov
        remaining = deadline - time.monotonic() - 10
        if remaining > 5:
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=min(remaining, 20), lr_init=0.002)
            try_seq(a_opt, "final_nest")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
