import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute 2n * max(conv) for normalized a (sum=1)."""
    a = a / a.sum()
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()

def lp_direction(a_star, num_active, delta=0.01):
    """
    Solve LP to find direction that reduces peak autocorrelation.
    a_star: current point (normalized, sum=1)
    num_active: number of top peaks to target
    delta: trust region radius
    """
    n = len(a_star)
    C = fast_convolve(a_star)
    
    # Identify active peaks
    top_idx = np.argsort(C)[::-1][:num_active]
    
    # Build LP:
    # min t
    # s.t. 2 * a_d^T (a_star shifted by k) - t <= -C[k] for k in top_idx
    #      sum a_d = 0
    #      -a_star <= a_d <= delta (element-wise upper bound on increase)
    #      ||a_d||_1 <= delta (trust region)
    
    # For efficiency, use the fact that the constraint matrix has structure
    # Row for peak k: [2*a_star[k-0], 2*a_star[k-1], ..., 2*a_star[k-(n-1)], -1, 0, ...]
    # (with boundary handling)
    
    m = num_active
    A = np.zeros((m, n + 1))
    b = np.zeros(m)
    
    for ii, k in enumerate(top_idx):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[ii, j] = 2.0 * a_star[idx]
        A[ii, n] = -1.0
        b[ii] = -C[k]
    
    # Equality: sum a_d = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Bounds
    lower = -a_star  # a_d[j] >= -a_star[j] (to keep a >= 0)
    upper = np.full(n, delta)  # limit increase
    bounds = list(zip(lower, upper)) + [(None, None)]
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t

        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-12})
        if res.success:
            a_d = res.x[:n]
            t_val = res.x[n]
            return a_d, t_val
    except Exception:
        pass
    
    return None, None

def line_search(a, direction, max_iter=30):
    """Find best step size along direction."""
    best_a = a.copy()
    best_val = score_normalized(a)
    
    # Try a range of step sizes
    alphas = [0.001, 0.003, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    
    for alpha in alphas:
        candidate = a + alpha * direction
        candidate = np.maximum(candidate, 0)
        s = candidate.sum()
        if s < 1e-15:
            continue
        candidate = candidate / s
        val = score_normalized(candidate)
        if val < best_val:
            best_val = val
            best_a = candidate.copy()
    
    return best_a, best_val

def optimize_lp(a_init, time_limit, num_active_list=None):

    if num_active_list is None:
        num_active_list = [5, 10, 20, 50, 100]
    
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = score_normalized(a)
    
    t0 = time.monotonic()
    no_improve = 0
    delta = 0.01
    
    while time.monotonic() - t0 < time_limit - 1.0:
        for num_active in num_active_list:
            if time.monotonic() - t0 > time_limit - 1.0:

            direction, t_val = lp_direction(a, num_active, delta)
            if direction is None:

            # Check direction is non-trivial
            if np.linalg.norm(direction) < 1e-12:

                continue
            
            # Line search
            a_new, new_score = line_search(a, direction)
            
            if new_score < best_score - 1e-12:
                a = a_new
                best_score = new_score
                best_a = a.copy()
                no_improve = 0
            elif new_score < score_normalized(a) - 1e-12:
                a = a_new
                no_improve = 0
            else:

                # Try smaller delta
                delta = max(delta * 0.8, 0.001)
        
        if no_improve > 20:

            a = project_simplex(best_a + rng.standard_normal(len(a)) * 0.002)

            delta = 0.01
        
        # Periodically try larger delta
        if no_improve == 10:
            delta = min(delta * 1.5, 0.05)
    
    return best_a, best_score

"""
LP-based peak reduction for minimum autocorrelation step function.
Uses active-set LP subproblems with trust regions, adaptive line search,
and multi-scale local optimization to minimize 2n*max(autocorrelation).
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_val(a):
    """Compute 2n * max(conv()) for normalized input."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    an = a / s
    conv = fast_convolve(an)
    return 2.0 * len(an) * conv.max()


def lp_solve(a_star, num_active, delta):
    """
    LP subproblem: find direction a_d that reduces peak autocorrelation.
    
    min t
    s.t. 2 * sum_j a_d[j] * a_star[k-j] - t <= -C[k]  for k in top num_active
         sum a_d = 0
         -a_star[j] <= a_d[j] <= delta
    """
    n = len(a_star)
    C = fast_convolve(a_star)
    
    # Top peaks
    top_idx = np.argsort(C)[::-1][:num_active]
    
    # Build constraint matrix
    m = num_active
    A = np.zeros((m, n + 1))
    b = np.zeros(m)
    
    for ii, k in enumerate(top_idx):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[ii, j] = 2.0 * a_star[idx]
        A[ii, n] = -1.0
        b[ii] = -C[k]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-a_star[j], delta) for j in range(n)] + [(None, None)]
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

    conv = fast_convolve(a)
    scaled = (conv - conv.max()) / T
    max_val = conv.max()
    return max_val + T * np.log(np.exp(scaled).sum()) / len(conv)

"""
LP-based peak reduction optimizer for minimum autocorrelation step function.
Combines active-set LP subproblems, FFT-based gradient descent on top peaks,
and multi-scale local search to minimize 2n*max(autocorrelation)/(sum)^2.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_val(a):
    """Compute 2n * max(conv) for normalized input (sum=1)."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    an = a / s
    conv = fast_convolve(an)
    return 2.0 * len(an) * conv.max()


def grad_top_peaks(a, top_k):
    """
    Compute gradient of sum_{top k peaks} C[k] with respect to a.
    Uses FFT-based cross-correlation.
    """
    n = len(a)
    C = fast_convolve(a)
    top_idx = np.argsort(C)[::-1][:top_k]
    
    # Weight: emphasize the peaks
    w = np.zeros(2 * n - 1)
    peak_vals = C[top_idx]
    # Use normalized weights (proportional to peak value)
    w[top_idx] = peak_vals
    w = w / w.sum()
    
    # Compute gradient using FFT
    fft_size = 1 << (2 * n - 2).bit_length()
    AW = np.fft.rfft(w, fft_size)
    AA = np.fft.rfft(a, fft_size)
    
    # g[i] = sum_k w[k] * a[k-i]
    G = np.fft.irfft(AW * np.conj(AA), fft_size)[:n]
    # h[i] = sum_k w[k] * a[i-k]  
    H = np.fft.irfft(AA * np.conj(AW), fft_size)[:n]
    
    return 2.0 * (G + H), w, C


def lp_direction(a_star, num_active, delta):
    """
    Solve LP to find direction reducing peak autocorrelation.
    Returns direction vector or None.
    """
    n = len(a_star)
    C = fast_convolve(a_star)
    top_idx = np.argsort(C)[::-1][:num_active]
    
    m = num_active
    # Build A_ub: m rows, n+1 columns
    # Row i (for peak k=top_idx[i]): 
    #   sum_j 2*a_star[k-j]*a_d[j] - t <= -C[k]
    
    A = np.zeros((m, n + 1))
    b = np.zeros(m)
    
    for ii in range(m):
        k = top_idx[ii]
        # Coefficient for a_d[j] is 2*a_star[k-j] if 0 <= k-j < n
        j_range = np.arange(max(0, k - n + 1), min(k, n - 1) + 1)
        A[ii, j_range] = 2.0 * a_star[k - j_range]
        A[ii, n] = -1.0
        b[ii] = -C[k]
    
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.zeros(1)
    
    bounds = [(-a_star[j], delta) for j in range(n)] + [(-1.0, 1.0)]
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None
