def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix_sparse(f, n):
    """Build sparse constraint matrix for LP oracle.
    A[k,j] = f[j] if 0 <= k-j < n, else 0.
    Shape: (2n-1) x n

    rows = []
    cols = []
    vals = []

        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.01 * min(1.0, (no_improve - 60) / 150.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

        if verbose and it % 50 == 0:
            print("    FW iter {}: obj={:.10f}, no_improve={}".format(it, best_obj, no_improve))

    return best_f, best_obj

def mirror_descent(a0, time_limit, T_init=0.01, lr_init=0.05):
    """Mirror descent (multiplicative weights) with subgradient of max conv."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = np.maximum(a, 1e-15)  # avoid log(0)
    a = a / a.sum()
    n = len(a)

    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)
        k_max = int(np.argmax(conv))

        # Subgradient: 2 * a reversed (for the active constraint)
        # grad[i] = 2 * a[k_max - i] for valid i
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        grad = np.zeros(n)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * a[k_max - idx]

        # Multiplicative update
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            # Try smaller step
            for frac in [0.5, 0.25, 0.1]:
                a_try = a * np.exp(-lr * frac * grad)
                a_try = np.maximum(a_try, 1e-15)
                a_try = a_try / a_try.sum()
                obj_try = fast_convolve(a_try).max()
                if obj_try < best_obj - 1e-15:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    no_improve = 0
                    a = a_try
                    break
            else:
                no_improve += 1

        lr *= 0.9995

        if no_improve > 30:
            lr = max(lr * 1.5, 0.01)
        if no_improve > 100:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = np.maximum(best_a + rng.standard_normal(n) * 0.003, 1e-15)
            a = a / a.sum()
            lr = lr_init
            no_improve = 0
        if no_improve > 300:
            break

    return best_a, best_obj


def smoothed_gradient_descent(a0, time_limit, T_init=0.001, lr_init=0.03):
    """Smoothed gradient descent using softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.3:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient: 2 * (M(p) @ a) where M(p)_{j,m} = p[j+m]
        grad = np.zeros(n)
        for j in range(n):
            # (M(p) @ a)[j] = sum_m p[j+m] * a[m]
            # = correlate(p, a)[j] with appropriate offset
            idx_m = np.arange(0, n)
            idx_p = j + idx_m
            valid = idx_p < len(p)
            grad[j] = np.dot(p[idx_p[valid]], a[idx_m[valid]])
        grad *= 2.0

        best_candidate = a
        best_val = best_obj

        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate

        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            T *= 1.05
            lr *= 0.97
        if no_improve > 40:
            T *= 0.8
        if no_improve > 150:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    """Coordinate descent: pairwise swaps to reduce max convolution."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stale = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Find the pair (i, j) that contributes most to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]

        # Find the best coordinate to move mass to
        # Try moving mass from j_max to coordinates that don't contribute to the peak
        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            break

        best_delta_found = False
        for delta in [0.001, 0.002, 0.005, 0.01, 0.02]:
            if a[j_max] < delta * 2:
                continue
            # Try each candidate destination
            candidates = outside[np.argsort(a[outside])[:min(20, len(outside))]]
            for j_dest in candidates:
                candidate = a.copy()
                candidate[j_max] -= delta
                candidate[j_dest] += delta
                candidate = np.maximum(candidate, 0.0)
                csum = candidate.sum()
                if csum < 1e-15:
                    continue
                candidate = candidate / csum
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    best_delta_found = True
                    break
            if best_delta_found:
                break

        if best_delta_found:
            stale = 0
        else:
            stale += 1
            if stale > 50:
                # Random perturbation
                candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    stale = 0
                elif stale > 500:
                    break

    return best_a, best_obj


def multi_peak_subgradient(a0, time_limit, top_k=20, lr_init=0.003, mom=0.93):
    """Multi-peak subgradient with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.3:
            break

        conv = fast_convolve(a)
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 2.0
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]

        v = mom * v - lr * sg
        a = project_simplex(a + v)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9997
        mom = min(0.97, mom + 0.00005)

        if no_improve > 200:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.002 + 0.004 * min(1.0, (no_improve - 200) / 400.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init * 0.6
            no_improve = 0

    return best_a, best_obj


def symmetrize(a):
    """Symmetrize a sequence and renormalize."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_sym / s


def try_and_update(a, state, label):
    """Evaluate a sequence and update state if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a)))


def generate_structured_inits(n, parent, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # From parent (interpolated)
    if parent is not None:
        px = np.interp(np.linspace(0, n - 1, n), np.linspace(0, len(parent) - 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Symmetrized parent
        inits.append(symmetrize(px))

    # Cosine bump
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    # Edge concentration
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    inits.append(edge_w / edge_w.sum())

    # Twin structure: pairs (i, n-1-i) equal
    half = n // 2
    twin = np.ones(half) / (n / 2.0)
    if n % 2 == 0:
        inits.append(np.concatenate([twin, twin[::-1]]) / n)
    else:
        inits.append(np.concatenate([twin, [1.0/n], twin[::-1]]) / n)

    # Periodic structures
    for period in [4, 6, 8, 12]:
        h = np.ones(n)
        for i in range(n):
            h[i] = 1.0 + 0.3 * np.cos(2.0 * np.pi * i / period)
        h = np.maximum(h, 0.01)
        inits.append(h / h.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    
    Phases:
    1. Multi-n exploration with quick FW-LP
    2. Deep FW-LP on best candidate
    3. Mirror descent + smoothed gradient
    4. Coordinate descent + multi-peak subgradient
    5. Final polish with symmetrization
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    state = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.8f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)
    print("Starting: best={:.8f}, n_parent={}".format(state[0], n_parent))

    # === Phase 1: Multi-n Exploration (~200s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent, 512, 768, 1024, 1280, 1536, 2048]))
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 4096]

    # Allocate more time to promising n values
    time_per_n = 25

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1400
        if remaining < 15:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, n_try - 1, n_try),
                              np.linspace(0, n_parent - 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try

        t_alloc = min(remaining, time_per_n)
        print("  FW-LP n={}, t={:.0f}s".format(n_try, t_alloc))
        a_opt, _ = fw_lp_optimize(start, time_limit=t_alloc)
        try_and_update(a_opt, state, "fw_n{}".format(n_try))

        # Also try symmetrized version
        a_sym = symmetrize(a_opt)
        try_and_update(a_sym, state, "fw_sym_n{}".format(n_try))

        # Quick mirror descent
        remaining = deadline - time.monotonic() - 1350
        if remaining > 8:
            a_md, _ = mirror_descent(a_opt, time_limit=min(remaining, 10))
            try_and_update(a_md, state, "md_n{}".format(n_try))

    print("  Phase 1 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 2: Deep FW-LP on best (~550s) ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])
        remaining = deadline - time.monotonic() - 850
        if remaining > 30:
            t_alloc = min(remaining, 400)
            print("  Deep FW-LP n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = fw_lp_optimize(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "deep_fw")

            # Symmetrized
            a_sym = symmetrize(a_opt)
            try_and_update(a_sym, state, "deep_fw_sym")
            if state[1] is not a_sym:
                # If symmetrization helped, optimize the symmetrized version
                remaining = deadline - time.monotonic() - 700
                if remaining > 20:
                    a_opt2, _ = fw_lp_optimize(state[1], time_limit=min(remaining, 80))
                    try_and_update(a_opt2, state, "deep_fw_sym2")

            # Perturbed restarts
            for scale in [0.003, 0.008, 0.015, 0.025]:
                remaining = deadline - time.monotonic() - 550
                if remaining < 10:
                    break
                f_pert = project_simplex(state[1] + rng.standard_normal(n_best) * scale)
                t_alloc = min(remaining, 50)
                a_opt2, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
                try_and_update(a_opt2, state, "fw_pert_{}".format(scale))

    print("  Phase 2 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 3: Mirror Descent + Smoothed Gradient (~350s) ===
    print("\n=== Phase 3: Mirror/Smoothed Methods ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Mirror descent with different settings
        for T_init, lr_init in [(0.01, 0.05), (0.005, 0.03), (0.02, 0.08)]:
            remaining = deadline - time.monotonic() - 200
            if remaining < 10:

            t_alloc = min(remaining, 80)
            print("  Mirror descent T={}, lr={}, t={:.0f}s".format(T_init, lr_init, t_alloc))
            a_opt, _ = mirror_descent(state[1], time_limit=t_alloc,
                                      T_init=T_init, lr_init=lr_init)
            try_and_update(a_opt, state, "md_{}/{}".format(T_init, lr_init))

        # Smoothed gradient descent
        for T_init, lr_init in [(0.0005, 0.04), (0.002, 0.03), (0.005, 0.02)]:
            remaining = deadline - time.monotonic() - 100
            if remaining < 8:
                break
            t_alloc = min(remaining, 60)
            print("  Smoothed grad T={}, lr={}, t={:.0f}s".format(T_init, lr_init, t_alloc))
            a_opt, _ = smoothed_gradient_descent(state[1], time_limit=t_alloc,
                                                 T_init=T_init, lr_init=lr_init)
            try_and_update(a_opt, state, "smooth_{}/{}".format(T_init, lr_init))

    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 4: Coordinate Descent + Multi-peak SG (~350s) ===
    print("\n=== Phase 4: Coordinate/Subgradient ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Coordinate descent
        remaining = deadline - time.monotonic() - 150
        if remaining > 10:
            t_alloc = min(remaining, 120)
            print("  Coordinate descent, t={:.0f}s".format(t_alloc))
            a_opt, _ = coordinate_descent(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "coord_desc")

        # Multi-peak subgradient
        for top_k, lr, mom in [(10, 0.003, 0.93), (30, 0.002, 0.92), (50, 0.001, 0.95), (100, 0.001, 0.94)]:
            remaining = deadline - time.monotonic() - 80
            if remaining < 8:
                break
            t_alloc = min(remaining, 50)
            print("  Multi-peak SG k={}, lr={}, t={:.0f}s".format(top_k, lr, t_alloc))
            a_opt, _ = multi_peak_subgradient(state[1], time_limit=t_alloc,
                                              top_k=top_k, lr_init=lr, mom=mom)
            try_and_update(a_opt, state, "mpsg_k{}".format(top_k))

    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 5: Final Polish (~250s) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Symmetrization + re-optimization
        a_sym = symmetrize(state[1])
        try_and_update(a_sym, state, "final_sym")

        # Final FW-LP
        remaining = deadline - time.monotonic() - 60
        if remaining > 10:
            t_alloc = min(remaining, 60)
            print("  Final FW-LP, t={:.0f}s".format(t_alloc))
            a_opt, _ = fw_lp_optimize(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "final_fw")

        # Fine smoothed gradient
        remaining = deadline - time.monotonic() - 30
        if remaining > 5:
            t_alloc = min(remaining, 25)
            a_opt, _ = smoothed_gradient_descent(state[1], time_limit=t_alloc,
                                                 T_init=0.0003, lr_init=0.01)
            try_and_update(a_opt, state, "final_smooth")

        # Tiny perturbations
        remaining = deadline - time.monotonic() - 15
        if remaining > 5:
            for _ in range(8):
                remaining = deadline - time.monotonic() - 5
                if remaining < 3:
                    break
                noise = rng.standard_normal(len(state[1])) * 0.0005
                f_pert = project_simplex(state[1] + noise)
                a_opt, _ = smoothed_gradient_descent(f_pert, time_limit=min(remaining, 4),
                                                     T_init=0.0002, lr_init=0.005)
                try_and_update(a_opt, state, "perturb_final")

    print("  Phase 5 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(768) / 768
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
