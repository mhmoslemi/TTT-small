```python
"""
Natural gradient (Newton) method with Hankel-structured Hessian for minimizing
peak auto-convolution of non-negative sequences. Uses CG for Hessian solve,
LP oracle for escape, and multi-n exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_matvec(p, v):
    """Compute H @ v where H[i,j] = p[i+j] for i,j in 0..n-1. p has length 2n-1."""
    n = len(v)
    # (H @ v)[i] = sum_{j=0}^{n-1} p[i+j] * v[j]
    # Use FFT: this is a cross-correlation
    v_rev = v[::-1]
    full_len = 2 * n - 1 + n - 1  # = 3n - 2
    fft_size = 1 << (full_len - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    V = np.fft.rfft(v_rev, fft_size)
    conv_full = np.fft.irfft(P * V, fft_size)
    return conv_full[n - 1:n - 1 + n]


def compute_gradient(a, p):
    """Compute gradient of sum_k p[k]*(a*a)[k] wrt a.
    grad[j] = sum_k p[k] * d(a*a)[k]/da[j]
    d(a*a)[k]/da[j] = 2*a[k-j] if k-j in [0,n-1] and k-j != j
                      = a[k/2] if j = k/2 (i.e., k = 2j)
    Simplified: grad[j] = 2 * sum_i a[i]*p[j+i] - sum_{k=2j} p[k]*a[j]
                 = 2 * (H_a @ a)[j] - (diagonal correction)
    where (H_a @ a)[j] = sum_i a[i]*p[j+i]
    Actually: grad[j] = sum_k p[k] * (a[k-j] + a[k-j] if k-j!=j else a[k-j])
    Let me just use the Hankel structure:
    grad = H(a,p) @ a + H(a,p) @ a - diag_correction
    Where H(a,p)[i,j] = p[i+j], so (H@ a)[i] = sum_j p[i+j]*a[j]
    And the correction is: for each j where 2j < 2n-1, subtract p[2j]*a[j]
    """
    n = len(a)
    # First term: 2 * (H @ a) where H[i,j] = p[i+j]
    Ha = hankel_matvec(p, a)
    # Correction: for diagonal entries where i=j and 2j < len(p)
    # The Hessian H_k[j,j] = 1 only if 2j = k, giving contribution p[2j]*a[j]
    # So grad[j] = 2*(H@a)[j] - p[2j]*a[j] if 2j < 2n-1
    correction = np.zeros(n)
    for j in range(n):
        k = 2 * j
        if k < len(p):
            correction[j] = p[k] * a[j]
    return 2.0 * Ha - correction


def natural_gradient_step(a, T, n_cg=100, reg=1e-10):
    """Compute natural gradient direction using CG on the Hankel Hessian.
    Returns (direction, grad, p, conv) where direction = H^{-1} @ grad.
    """
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()
    
    grad = compute_gradient(a, p)
    
    # Solve H @ d = grad using CG, where H[i,j] = p[i+j]
    # H is PSD (sum of PSD matrices), add small regularization for numerical stability
    
    def matvec(v):
        return hankel_matvec(p, v) + reg * v
    
    # Use CG (since H is symmetric PSD)
    d = np.zeros(n)
    r = grad.copy()
    p_vec = r.copy()
    rs_old = r @ r
    
    for i in range(n_cg):
        if np.sqrt(rs_old) < 1e-12:
            break
        Ap = matvec(p_vec)
        pAp = p_vec @ Ap
        if pAp < 1e-15:
            break
        alpha = rs_old / pAp
        d += alpha * p_vec
        r -= alpha * Ap
        rs_new = r @ r
        if np.sqrt(rs_new) < 1e-12:
            break
        beta = rs_new / rs_old
        p_vec = r + beta * p_vec
        rs_old = rs_new
    
    return d, grad, p, conv


def line_search_natural(a, d, conv_max, max_iter=30):
    """Backtracking line search for natural gradient step.
    Minimize max_conv(project_simplex(a - alpha * d))
    """
    n = len(a)
    best_alpha = 0.0
    best_val = conv_max
    
    # Try a range of alphas
    # Scale d to have unit norm for step size interpretation
    d_norm = np.linalg.norm(d)
    if d_norm < 1e-15:
        return 0.0, conv_max
    
    d_normalized = d / d_norm
    
    # Initial alpha based on gradient magnitude
    alpha_init = 0.1 / max(d_norm, 1e-6)
    
    alphas = [alpha_init * factor for factor in 
              [0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 1.0, 3.0, 10.0, 30.0, 100.0]]
    
    best_val = conv_max
    best_alpha = 0.0
    
    for alpha in alphas:
        candidate = a - alpha * d
        candidate = project_simplex(candidate)
        c = fast_convolve(candidate)
        val = c.max()
        if val < best_val - 1e-15:
            best_val = val
            best_alpha = alpha
    
    # Refine with ternary search around best_alpha
    if best_alpha > 0:
        lo = max(0.0, best_alpha * 0.5)
        hi = best_alpha * 2.0
        for _ in range(15):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            c1 = project_simplex(a - m1 * d)
            v1 = fast_convolve(c1).max()
            c2 = project_simplex(a - m2 * d)
            v2 = fast_convolve(c2).max()
            if v1 < v2:
                hi = m2
            else:
                lo = m1
        mid = (lo + hi) / 2.0
        c_mid = project_simplex(a - mid * d)
        v_mid = fast_convolve(c_mid).max()
        if v_mid < best_val:
            best_val = v_mid
            best_alpha = mid
    
    return best_alpha, best_val

    n = len(a)

    best_obj = conv.max()
    best_a = a.copy()
    
    T = T_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        d, grad, p, conv = natural_gradient_step(a, T, n_cg=80, reg=1e-10)
        conv_max = conv.max()
        
        alpha, new_val = line_search_natural(a, d, conv_max)
        
        if alpha > 1e-12:
            a = project_simplex(a - alpha * d)
            conv = fast_convolve(a)
            new_val = conv.max()
        
        if new_val < best_obj - 1e-14:
            best_obj = new_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            a = best_a.copy()
        
        # Adapt temperature
        if no_improve > 15:
            T *= 1.05
        if no_improve > 50:
            T = max(T * 0.8, 0.0001)
        if no_improve > 200:
            # Random restart
            rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            no_improve = 0
        
        # Print progress every 100 iterations
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NG iter {it}: obj={best_obj:.10f}, T={T:.6f}, no_improve={no_improve}, time={elapsed:.1f}s")
    
    return best_a, best_obj


def build_lp_oracle_data(n):
    """Precompute row/col indices for LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows), np.array(cols)


def solve_lp_oracle(f, M, n, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    vals = np.empty(len(row_idx))
    for idx in range(len(row_idx)):
        k = row_idx[idx]
        i = col_idx[idx]
        vals[idx] = f[k - i]
    
    A_mat = lil_matrix((2 * n - 1, n))
    A_mat[row_idx, col_idx] = vals
    A_mat = A_mat.tocsr()
    
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.fun is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def fw_lp_step(f, g, time_limit_inner=5.0):
    """Line search between f and g for Frank-Wolfe step."""
    t_start = time.monotonic()
    best_alpha = 0.0
    best_val = fast_convolve(f).max()
    
    alphas = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
    for alpha in alphas:
        if time.monotonic() - t_start > time_limit_inner:
            break
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    if 0 < best_alpha < 1:
        lo = max(0.0, best_alpha - 0.1)
        hi = min(1.0, best_alpha + 0.1)
        for _ in range(10):
            if time.monotonic() - t_start > time_limit_inner:
                break
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = np.maximum((1 - m1) * f + m1 * g, 0)
            f1 = f1 / f1.sum()
            v1 = fast_convolve(f1).max()
            f2 = np.maximum((1 - m2) * f + m2 * g, 0)
            f2 = f2 / f2.sum()
            v2 = fast_convolve(f2).max()
            if v1 < v2:
                hi = m2
            else:
                lo = m1
        mid = (lo + hi) / 2.0
        ft = np.maximum((1 - mid) * f + mid * g, 0)
        ft = ft / ft.sum()
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = mid
    
    return best_alpha, best_val


def lp_oracle_phase(f0, time_limit):
    """Frank-Wolfe with LP oracle for escape from flat regions."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    row_idx, col_idx = build_lp_oracle_data(n)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        
        g = solve_lp_oracle(f, M * m_exp, n, row_idx, col_idx)
        if g is None:
            m_exp = min(m_exp * 1.5, 5.0)
            if m_exp >= 5.0:
                rng = np.random.default_rng(it * 31 + 7)
                g = project_simplex(f + rng.standard_normal(n) * 0.005)
                m_exp = 1.0
            else:
                continue
        
        alpha, best_val = fw_lp_step(f, g)
        
        if alpha > 1e-10:
            f = (1.0 - alpha) * f + alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
        else:
            m_exp = min(m_exp * 1.2, 5.0)
            
        if it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}: obj={best_obj:.10f}, M_exp={m_exp:.2f}, time={elapsed:.1f}s")
    
    return best_f, best_obj


def try_seq(a, label, best_ref):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_ref
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_ref
    if np.isfinite(score) and score < best_ref[1]:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return (a_norm.copy(), score)
    return best_ref


def local_search_fine(a0, time_limit, seed=42):
    """Fine local search with small perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        improved = False
        
        # Multi-scale perturbations
        for scale in [0.0001, 0.0003, 0.001, 0.003]:
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(a + pert)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break
        
        # Coordinate moves
        if not improved and it % 3 == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j and a[i] > 1e-6:
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    candidate = a.copy()
                    amount = candidate[i] * frac
                    candidate[i] -= amount
                    candidate[j] += amount
                    candidate = np.maximum(candidate, 0)
                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved and it > 5000:
            break
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main optimization: Natural gradient (Newton) method with Hankel Hessian,
    LP oracle for escape, multi-n exploration, and local search polish.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    t_start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Using uniform init: score={best_score:.8f}")
    
    n_parent = len(parent)
    
    # === Phase 1: Quick multi-n exploration ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()
    
    candidate_ns = [512, 640, 768, 896, 1024, 1280, 1536]
    if n_parent not in candidate_ns:
        candidate_ns.append(n_parent)
    candidate_ns.sort()
    
    for n_try in candidate_ns:
        remaining = deadline - time.monotonic()
        if remaining < 200:
            break
        
        if n_try == n_parent:
            start = parent.copy()
        else:
            x_new = np.linspace(0, 1, n_try)
            x_old = np.linspace(0, 1, n_parent)
            start = np.interp(x_new, x_old, parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try
        
        # Quick natural gradient
        t_alloc = min(remaining * 0.04, 20)
        a_opt, _ = natural_gradient_optimize(start, time_limit=t_alloc, T_init=0.001)
        best_seq, best_score = try_seq(a_opt, f"ng_n{n_try}", (best_seq, best_score))
    
    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 2: Deep natural gradient on best n ===
    print("\n=== Phase 2: Deep Natural Gradient ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    
    if remaining > 100:
        t_alloc = min(remaining * 0.40, 500)
        print(f"  Deep NG n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = natural_gradient_optimize(best_seq, time_limit=t_alloc, T_init=0.0005)
        best_seq, best_score = try_seq(a_opt, "deep_ng", (best_seq, best_score))
    
    # Perturbed restarts
    for scale in [0.002, 0.005, 0.01, 0.02]:
        remaining = deadline - time.monotonic()
        if remaining < 100:

        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)

        best_seq, best_score = try_seq(a_opt, f"ng_pert_{scale}", (best_seq, best_score))
    
    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    print("\n=== Phase 4: Alternative n ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    
    n_best = len(best_seq)
    for n_try in [n_best + 32, n_best - 32, n_best * 2, n_best // 2]:
        if n_try < 128 or n_try > 3072:
            continue
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        x_old = np.linspace(0, 1, n_best)

    # Very fine natural gradient

        a_opt, _ = natural_gradient_optimize(best_seq, time_limit=t_alloc, T_init=0.0001)

    for i in range(5):

        f_pert = project_simplex(best_seq + noise)
        t_alloc = min(remaining * 0.15, 20)
        a_opt, _ = natural_gradient_optimize(f_pert, time_limit=t_alloc, T_init=0.0002)
        best_seq, best_score = try_seq(a_opt, f"micro_{i}", (best_seq, best_score))

    if remaining > 15:

        best_seq, best_score = try_seq(a_opt, "local2", (best_seq, best_score))

    # Fallback
    if best_seq is None:

    # Final verification
    try:

        if np.isfinite(fs) and fs < best_score:

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
```
