"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses adaptive smoothed gradient descent with analytic gradients,
Adam optimizer, multi-start with diverse initializations, and
targeted peak-equalization local search.
Key insight: the auto-convolution peak (at the middle lag for symmetric
sequences) can be reduced below the uniform baseline by finding sequences
whose auto-convolution is flatter than triangular.

import numpy as np
import time


def auto_convolve(a):
    """Compute full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the evaluation score for a non-negative sequence."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a = a / s
    n = len(a)
    c = auto_convolve(a)
    return float(2.0 * n * c.max() / 1.0)


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_max_and_grad(a, T):
    """
    Compute smoothed max of auto-convolution and its gradient w.r.t. a.
    Uses log-sum-exp for numerical stability.
    grad[m] = 2 * sum_k p[k] * a[k-m] where p = softmax(T * conv(a,a))
    """
    n = len(a)
    c = auto_convolve(a)
    # Log-sum-exp for numerical stability
    c_max = c.max()
    exp_c = np.exp(T * (c - c_max))
    p = exp_c / exp_c.sum()
    smoothed_val = c_max + np.log(exp_c.sum()) / T

    # Gradient: grad[m] = 2 * sum_{k} p[k] * a[k-m] (valid indices only)
    # This is cross-correlation: result[m] = sum_j a[j] * p[j+m]
    size = 1 << (4 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    P = np.fft.rfft(p, size)
    result = np.fft.irfft(A * np.conj(P), size)
    grad = 2.0 * result[:n].copy()

    return smoothed_val, grad, p, c

def peak_equalization(a0, time_limit, max_iter_per_peak=5):
    """
    Targeted local search: identify the peak lag and try to reduce it
    by moving mass from peak-contributing positions to non-peak positions.
    Uses exact 2-coordinate optimization via golden section search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = compute_score(a)

    t0 = time.monotonic()
    it = 0
    stall = 0

        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        c = auto_convolve(a)
        peak_idx = int(np.argmax(c[:n]))  # Due to symmetry, check first half
        peak_val = c[peak_idx]

        # Factors contributing to c[peak_idx]: a[j] * a[peak_idx - j] for valid j
        js_valid = np.arange(max(0, peak_idx - n + 1), min(n, peak_idx + 1))
        partners = peak_idx - js_valid

        # Find the pair with the largest product (biggest contributor)
        if len(js_valid) > 0:
            products = a[js_valid] * a[partners]
            top_idx = int(np.argmax(products))
            j1 = int(js_valid[top_idx])
            j2 = int(partners[top_idx])

            # Find positions that DON'T contribute to the peak
            # (i.e., positions j where peak_idx - j is out of [0, n-1])
            non_peak_mask = np.ones(n, dtype=bool)
            non_peak_mask[js_valid] = False
            non_peak_positions = np.where(non_peak_mask)[0]

            if len(non_peak_positions) == 0:
                break

            # Try moving mass from j1 (or j2) to a non-peak position
            best_gain = 0
            best_move = None

            for src in [j1, j2]:
                if a[src] < 1e-8:
                    continue

                    rng_s = np.random.default_rng(it * 31 + src)

                else:

                        a_test[dst] += delta
                        a_test = np.maximum(a_test, 0.0)
                        s_test = a_test.sum()
                        if s_test < 1e-15:

                        a_test = a_test / s_test
                        c_test = auto_convolve(a_test)
                        new_peak = c_test.max()
                        gain = peak_val - new_peak
                        if gain > best_gain:
                            best_gain = gain
                            best_move = (src, dst, delta)

            if best_move is not None and best_gain > 1e-10:
                src, dst, delta = best_move
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                score = compute_score(a)

                    best_score = score
                    best_a = a.copy()
                    stall = 0
                else:
                    a = best_a.copy()
                    stall += 1
            else:
                stall += 1
        else:
            break

        # Random restart if stalled
        if stall > 30:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            scale = 0.001 + 0.002 * min(1.0, stall / 60.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            stall = 0

        if stall > 200:
            break

    return best_a, best_score


def multi_scale_gradient(a0, time_limit):
    """
    Multi-scale optimization: start with coarse (large T) to get good structure,
    then refine with fine T to target the actual maximum.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    scales = [
        (10.0, 2.0, 0.02, 0.25),   # Coarse: high T, large lr, 25% time
        (2.0, 0.5, 0.01, 0.25),    # Medium
        (0.5, 0.1, 0.005, 0.25),   # Fine
        (0.1, 0.02, 0.002, 0.25),  # Very fine
    ]

    t_elapsed = 0.0
    for T_init, T_end, lr_init, frac in scales:
        t_alloc = time_limit * frac - 1.0
        if t_alloc < 2.0:
            break
        a, _ = adam_smoothed_optimize(a, time_limit=t_alloc, T_init=T_init,
                                       T_min=max(T_end, 0.01), lr_init=lr_init)
        t_elapsed += t_alloc

    return a, compute_score(a)


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search combining multi-scale gradient
    descent, peak equalization, and diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15 or len(a) == 0:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t0_total
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return score

    t0_total = time.monotonic()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = init_uniform(768)

    n_parent = len(parent)

    # === Phase 1: Multi-n, multi-init exploration ===
    print("\n=== Phase 1: Multi-n Multi-init Exploration ===")
    t_phase = time.monotonic()

    n_values = sorted(set([
        n_parent, max(256, n_parent - 128), min(1536, n_parent + 128),
        384, 512, 640, 768, 896, 1024, 1152, 1280
    ]))
    n_values = [n for n in n_values if 256 <= n <= 1400]

    initializers = [
        ("uniform", init_uniform),
        ("gauss", lambda n: init_gaussian(n, 0.3)),
        ("gauss_w", lambda n: init_gaussian(n, 0.5)),
        ("sine", lambda n: init_sine(n, 1)),
        ("sine2", lambda n: init_sine(n, 2)),
        ("vshape", init_v_shape),
        ("ushape", init_u_shape),
        ("bipart", init_bipartite),
        ("3bump", init_three_bumps),
    ]

    # Quick pass: try each (n, init) combo briefly
    for n_try in n_values:
        if time.monotonic() > deadline - 700:
            break
        for init_name, init_fn in initializers:
            if time.monotonic() > deadline - 650:
                break
            if time.monotonic() - t_phase > 250:
                break
            a_init = init_fn(n_try)
            if n_try == n_parent and init_name == "uniform":
                a_init = resample_sequence(parent, n_try)

            t_alloc = min(15.0, deadline - time.monotonic() - 600)
            if t_alloc < 3:
                break
            a_opt, sc = adam_smoothed_optimize(a_init, time_limit=t_alloc,
                                                T_init=3.0, T_min=0.3, lr_init=0.01)
            try_seq(a_opt, f"p1_{init_name}_n{n_try}")

    print(f"  Phase 1 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep multi-scale optimization on best candidates ===
    print("\n=== Phase 2: Deep Multi-scale Optimization ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Deep optimization on best
        t_alloc = min(300, deadline - time.monotonic() - 400)
        if t_alloc > 20:
            print(f"  Deep multi-scale n={n_best}, t={t_alloc:.0f}s")
            a_opt, sc = multi_scale_gradient(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "p2_deep")

        # Try resampling best to nearby n values
        for n_try in [n_best - 64, n_best + 64, n_best - 128, n_best + 128]:
            if n_try < 256 or n_try > 1400:
                continue
            if time.monotonic() > deadline - 350:
                break
            a_resample = resample_sequence(best_seq, n_try)
            t_alloc = min(60, deadline - time.monotonic() - 350)
            if t_alloc < 10:
                break
            a_opt, sc = multi_scale_gradient(a_resample, time_limit=t_alloc)
            try_seq(a_opt, f"p2_resample_n{n_try}")

    print(f"  Phase 2 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Peak equalization ===
    print("\n=== Phase 3: Peak Equalization ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(200, deadline - time.monotonic() - 300)
        if t_alloc > 10:
            print(f"  Peak equalization, t={t_alloc:.0f}s")
            a_opt, sc = peak_equalization(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "p3_peak")

    print(f"  Phase 3 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Alternating optimization + random restarts ===
    print("\n=== Phase 4: Alternating + Restarts ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Multiple restarts with perturbation
        for restart_i in range(8):
            if time.monotonic() > deadline - 200:
                break
            t_alloc = min(40, deadline - time.monotonic() - 200)
            if t_alloc < 5:
                break

            # Different perturbation strategies
            if restart_i % 4 == 0:
                # Gaussian noise
                a_init = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.005)
            elif restart_i % 4 == 1:
                # Structure-preserving: flip part of the sequence
                a_init = best_seq.copy()
                split = len(a_init) // 2
                a_init[split:] = a_init[split:][::-1] * 0.5 + 0.5 / len(a_init)
                a_init = project_simplex(a_init)
            elif restart_i % 4 == 2:
                # Blend with a different initialization
                a_alt = init_bipartite(len(best_seq))
                a_init = project_simplex(0.7 * best_seq + 0.3 * a_alt)
            else:
                # Boundary emphasis
                a_init = best_seq.copy()
                fade = np.linspace(1.0, 0.3, len(a_init))
                a_init = a_init * fade + a_init * fade[::-1]
                a_init = project_simplex(a_init)

            # Multi-scale on perturbed
            a_opt, sc = multi_scale_gradient(a_init, time_limit=t_alloc)
            tag = f"p4_r{restart_i}"
            try_seq(a_opt, tag)

            # Peak equalization follow-up
            if time.monotonic() > deadline - 200:
                break
            t_alloc2 = min(20, deadline - time.monotonic() - 200)
            if t_alloc2 > 5:
                a_opt2, sc2 = peak_equalization(a_opt, time_limit=t_alloc2)
                try_seq(a_opt2, tag + "_eq")

    print(f"  Phase 4 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final polish ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Fine-grained gradient with very low temperature
        t_alloc = min(80, deadline - time.monotonic() - 50)
        if t_alloc > 5:
            a_opt, sc = adam_smoothed_optimize(best_seq, time_limit=t_alloc,
                                                T_init=0.3, T_min=0.02, lr_init=0.003)
            try_seq(a_opt, "p5_fine")

        # Peak equalization
        t_alloc = min(60, deadline - time.monotonic() - 30)
        if t_alloc > 5:
            a_opt, sc = peak_equalization(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "p5_eq")

        # One more multi-scale from best
        t_alloc = min(40, deadline - time.monotonic() - 10)
        if t_alloc > 5:
            a_opt, sc = multi_scale_gradient(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "p5_ms")

    print(f"  Phase 5 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = init_uniform(768)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, total={time.monotonic()-t0_total:.0f}s)")
    return [float(x) for x in best_seq]
