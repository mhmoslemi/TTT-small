- There exists a subgradient direction d in the subdifferential of max at conv(a) such that:
  - For all i with a[i] > 0: `2 * (d * a)_i - lambda = 0` (where lambda is the Lagrange multiplier for the sum constraint)
  - For all i with a[i] = 0: `2 * (d * a)_i - lambda <= 0`

Here, `d` is a convex combination of the "peak indicators" (the unit vectors at the positions where conv achieves its max, weighted by the subdifferential weights).

This gives us a verification: at the optimum, for all active indices i (a[i] > 0), the value `2 * (d * a)_i` should be constant. We can use this to check convergence and to guide the optimization.

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(n):
    """Build sparse matrix for convolve(f, x) = C @ x."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

def convolve_row(f, k, n):
    """Get row k of the convolution matrix applied to f: (f * x)[k] = sum_i f[k-i]*x[i]."""
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    i = np.arange(i_start, i_end + 1)
    return f[k - i]

def lp_oracle(f, M, n, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    k_arr = row_idx
    i_arr = col_idx
    vals = f[k_arr - i_arr]
    A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def pnorm_descent(a0, time_limit, p=50, lr_init=0.01, verbose=False):
    """
    Minimize sum(conv(a,a)^p) using projected gradient descent.
    This approximates minimizing max(conv(a,a)) for large p.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    momentum = np.zeros(n)
    beta = 0.9  # momentum coefficient
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        conv = fast_convolve(a)
        # Compute gradient of sum(conv^p) w.r.t. a
        # d/da_i sum_k conv[k]^p = 2p * sum_k conv[k]^(p-1) * a[k-i] (for valid k-i)
        conv_p1 = conv ** (p - 1)
        # Gradient = 2p * correlate(conv_p1, a)
        # correlate(h, a)[i] = sum_k h[k] * a[k-i]
        # We can compute this via FFT
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        H = np.fft.rfft(conv_p1, fft_size)
        A = np.fft.rfft(a, fft_size)
        # correlate(h, a) = ifft(conj(H) * A) ... 
        # Actually, cross-correlation: corr(h,a)[i] = sum_k h[k+i] * a[k]
        # = ifft(conj(H) * A)[i] for circular correlation
        # But we need linear correlation, so we pad enough.
        # Let's use the definition: grad[i] = 2p * sum_{k} conv[k]^(p-1) * a[k-i]
        # = 2p * (conv_p1 * flip(a))[i + n - 1]  where * is linear convolution
        # Actually: sum_k h[k]*a[k-i] = (h * a_flipped)[i] where a_flipped[j] = a[n-1-j]
        # Let me just compute it directly for clarity (n=768, conv has 2n-1=1535 entries)
        # grad[i] = 2p * sum_{k-i >= 0, k-i < n} conv_p1[k] * a[k-i]
        # = 2p * sum_{j=0}^{n-1} conv_p1[j+i] * a[j]  (where j = k-i, valid when 0 <= j < n and 0 <= j+i < 2n-1)
        
        # Using FFT for efficiency:
        a_flip = a[::-1]
        fft_size2 = 1 << (n - 1 + n - 1 + 1).bit_length()  # enough for linear conv
        H_fft = np.fft.rfft(conv_p1, fft_size2)
        A_flip_fft = np.fft.rfft(a_flip, fft_size2)
        corr = np.fft.irfft(H_fft * A_flip_fft, fft_size2)[:2*n-1]
        # corr[i] = sum_j conv_p1[j] * a_flip[j-i] ... hmm this is getting confusing
        # Let me just do it directly with a loop over a subset for speed
        # Actually for n=768, a direct computation is:
        # grad[i] = 2p * sum_{j=max(0,i-(2n-2))}^{min(n-1, n-1-i+... )} 
        # This is complex. Let me use the correlation formula properly.
        
        # The gradient component for a_i is:
        # 2p * sum_{k=0}^{2n-2} conv[k]^(p-1) * a[k-i]  where a[j]=0 if j<0 or j>=n
        # = 2p * sum_{k=i}^{i+n-1} conv[k]^(p-1) * a[k-i]  (a[k-i] is nonzero only for k-i in [0,n-1])
        # = 2p * sum_{j=0}^{n-1} conv[i+j]^(p-1) * a[j]
        
        # This is the cross-correlation of conv_p1 and a, evaluated at lag i.
        # Using FFT: cross_corr(h, a)[i] = ifft(conj(FFT(h)) * FFT(a))[i] for circular
        # For linear, pad to size >= len(h) + len(a) - 1
        
        pad_size = (2 * n - 1) + n - 1  # = 3n - 2
        fft_s = 1 << (pad_size - 1).bit_length()
        Hf = np.fft.rfft(conv_p1, fft_s)
        Af = np.fft.rfft(a, fft_s)
        # Cross-correlation: ifft(conj(Hf) * Af)
        xcorr = np.fft.irfft(np.conj(Hf) * Af, fft_s)
        # xcorr[i] = sum_j h[j+i] * a[j] for circular correlation
        # For our linear case with sufficient padding:
        # xcorr[i] = sum_{j=0}^{n-1} conv_p1[i+j] * a[j] (for i in [0, n-1])
        # This is exactly what we want!
        grad = 2.0 * p * xcorr[:n]
        
        # Normalized gradient (projected)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-30:
            break
        
        # Momentum update
        momentum = beta * momentum + grad
        direction = -momentum
        
        # Line search
        best_alpha = 0.0
        best_val = conv.max()
        for alpha in [lr, lr*0.5, lr*0.25, lr*2]:
            cand = a + alpha * direction
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-15:
            a = a + best_alpha * direction
            a = np.maximum(a, 0)
            a = a / a.sum()
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
                lr = min(lr * 1.05, 0.1)
            else:
                no_improve += 1
                lr *= 0.8
        else:
            no_improve += 1
            lr *= 0.7
        
        if no_improve > 30:
            # Restart with perturbation
            rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            lr = lr_init
            no_improve = 0
            momentum = np.zeros(n)
        
        if verbose and it % 50 == 0:
            print(f"    PNorm it={it}, obj={best_obj:.10f}, lr={lr:.6f}")
    
    return best_a, best_obj

def coordinate_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent: optimize one element at a time with exact 1D line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    order = np.arange(n)
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        
        # Random or sequential order
        if it % 5 == 0:
            rng.shuffle(order)
        
        improved = False
        for i in order:
            if time.monotonic() - t0 > time_limit - 1.0:
                break
            
            # Try changing a[i] while keeping sum=1
            # Move mass from a[i] to a[j] (or from a[j] to a[i])
            # For efficiency, try a few j values
            # Actually, let's try: set a[i] = a[i] + delta, and adjust a[j] = a[j] - delta
            # for a specific j (e.g., the element that helps most)
            
            # Simpler: try scaling a[i] by factor and renormalizing
            # a_new = a; a_new[i] *= factor; a_new = a_new / a_new.sum()
            # This changes the shape.
            
            # Even simpler: move a small amount from i to j
            # Let's try: a[i] - delta, a[j] + delta for several (i,j) pairs
            
            # Actually, the cleanest 1D optimization:
            # Fix all a[k] for k != i, optimize a[i] (with sum constraint binding
            # through one other element, say the largest other element)
            
            # Let's try a different approach: for each i, find the best value of a[i]
            # assuming all other elements are fixed and the sum constraint is met by
            # scaling all other elements proportionally.
            
            # If we set a[i] = t, then sum of others = 1-t, so others are scaled by (1-t)/(1-a[i])
            # a_new[i] = t
            # a_new[j] = a[j] * (1-t)/(1-a[i]) for j != i
            
            if a[i] >= 0.999:
                continue  # Can't increase much more
            
            ai = a[i]
            # Try a few values of t
            candidates = []
            for t in [ai * 0.5, ai * 0.8, ai * 1.2, ai * 1.5, 0.0, 0.001, 0.01]:
                if t < 0 or t > 0.99:
                    continue
                if abs(1 - t) < 1e-10:
                    continue
                a_new = a.copy()
                a_new[i] = t
                a_new[1:] = a_new[1:] * (1 - t) / (1 - ai) if i == 0 else \
                           a_new * (1 - t) / (1 - ai)
                # Fix: scale all elements except i
                a_new = a.copy()
                a_new[i] = t
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a_new[mask] = a[mask] * (1 - t) / (1 - ai)
                a_new = np.maximum(a_new, 0)
                s = a_new.sum()
                if s < 1e-15:
                    continue
                a_new = a_new / s
                candidates.append(a_new)
            
            if not candidates:
                continue
            
            # Evaluate
            best_cand = a
            best_val = best_obj
            for a_c in candidates:
                val = fast_convolve(a_c).max()
                if val < best_val:
                    best_val = val
                    best_cand = a_c
            
            if best_val < best_obj - 1e-14:
                a = best_cand
                best_obj = best_val
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_obj

def spectral_flatten(a0, time_limit, verbose=False):
    """
    Iteratively modify the spectrum to flatten the convolution.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    alpha = 0.3
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        # Current spectrum
        A = np.fft.rfft(a, fft_size)
        
        # Current convolution spectrum
        C = A * A  # This is FFT of circular conv = linear conv (with our padding)
        c = np.fft.irfft(C, fft_size)[:size]
        max_c = c.max()
        
        # Target: flat spectrum for C, i.e., C should be (sum(c)/fft_size) * ones
        # But C = A^2, so we want A^2 to be constant magnitude
        # |A|^2 should be constant: |A| = sqrt(mean(|A|^2)) = sqrt(mean(c) * fft_size / fft_size)
        # Actually, by Parseval: sum(c^2) * fft_size = sum(|C|^2) = sum(|A|^4)
        # We want c to be flat: c[k] = sum(a)^2 / (2n-1) = 1/(2n-1)
        # In freq domain: C should have only DC component = 1 (if sum(a)=1 and size entries)
        # Actually: if c is constant = 1/(2n-1) for k=0..2n-2 and 0 otherwise,
        # then C = FFT(c) which is not just DC.
        
        # Simpler approach: we want |A| to be as constant as possible.
        # Target magnitude: |A_target| = sqrt(sum(|A|^2) / fft_size) for all frequencies
        target_mag = np.sqrt(np.sum(np.abs(A)**2) / fft_size)
        
        # New A: same phase, target magnitude
        phase = np.angle(A)
        A_new_mag = target_mag
        A_new = A_new_mag * np.exp(1j * phase)
        
        # Inverse FFT, take first n entries, project
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Blend
        a_blend = (1 - alpha) * a + alpha * a_new
        a_blend = np.maximum(a_blend, 0)
        a_blend = a_blend / a_blend.sum()
        
        val = fast_convolve(a_blend).max()
        if val < best_obj - 1e-14:
            best_obj = val
            best_a = a_blend.copy()
            a = a_blend
            alpha = min(alpha * 1.05, 0.8)
        else:
            alpha *= 0.9
            if alpha < 0.05:
                # Perturb
                rng = np.random.default_rng((it * 31831 + 7) % (2**31 - 1))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
                alpha = 0.3
        
        if verbose and it % 100 == 0:
            print(f"    Spectral it={it}, obj={best_obj:.10f}")
    
    return best_a, best_obj

def resize_sequence(a, new_n):
    """Resize a sequence to new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new
