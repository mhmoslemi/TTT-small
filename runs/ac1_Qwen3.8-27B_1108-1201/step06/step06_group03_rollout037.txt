def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral (Gerchberg-Saxton) initialization for fast good starts
2. Subgradient descent with peak-weighted shifted gradients (very fast, O(n log n) per iter)
3. Frank-Wolfe with LP oracle for medium-level refinement
4. Peak-targeted coordinate descent for final polishing
5. Multi-length exploration to find the best n
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_peak_gradient(a, k):
    """Compute gradient of (a*a)[k] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        grad[j] += 2.0 * a[k - j]
    return grad

def subgradient_descent(a0, time_limit, lr_init=0.01, top_k=5, cooldown=300):
    """Fast subgradient descent using peak-weighted shifted gradients."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    lr = lr_init
    t0 = time.monotonic()
    stagnant = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break
        
        conv = fast_convolve(a)
        peak_val = conv.max()
        
        # Get top-k peaks with softmax weighting
        if top_k < len(conv):
            peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peak_idx = np.arange(len(conv))
        
        pv = conv[peak_idx]
        T = max(peak_val * 0.01, 1e-10)  # Temperature
        w = np.exp((pv - pv.max()) / T)
        w = w / w.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for i in range(len(peak_idx)):
            if w[i] < 1e-12:
                continue
            grad += w[i] * compute_peak_gradient(a, int(peak_idx[i]))
        
        # Update with backtracking
        best_candidate = a
        best_val = peak_val
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            candidate = project_simplex(a - lr * frac * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
        
        a = best_candidate
        obj = fast_convolve(a).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Adaptive learning rate
        if stagnant > 50:
            lr *= 0.9
            if stagnant > 200:
                # Reset with perturbation
                rng = np.random.default_rng((it * 31 + 7) % (2**31))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                lr = lr_init
                stagnant = 0
    
    return best_a, best_obj

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build (2n-1) x n convolution matrix A where (f*g)[k] = sum_j A[k,j]*g[j]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        js = max(0, 0)
        A[max(0, 0):min(2*n-1, n), j] = f[max(0, 0):min(n, n)]
    # Correct construction:
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = max(0, 0)
        k_end = min(2 * n - 1, n)
        # A[k, j] = f[j] when 0 <= k-j < n, i.e., j <= k <= j+n-1
        ks = j
        ke = min(2 * n - 1, j + n)
        A[ks:ke, j] = f[j]
    return A


def solve_lp_oracle(f, M, timeout=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout or 30})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def spectral_init(n, seed=42, n_iters=300):
    """Gerchberg-Saxton iterative spectral flattening for initialization."""
    rng = np.random.default_rng(seed)
    a = rng.dirichlet(np.ones(n) * 0.5)
    target_mag = 1.0 / np.sqrt(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    for it in range(n_iters):
        A = np.fft.fft(a)
        phase = np.angle(A)
        # Blend towards flat spectrum
        blend = 0.5 + 0.5 * (it / n_iters)
        mag = np.abs(A)
        new_mag = blend * target_mag + (1 - blend) * mag
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real
        a = project_simplex(a_new)

        obj = fast_convolve(a).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        if (it + 1) % 100 == 0:
            a = project_simplex(a + rng.standard_normal(n) * 0.003)

    return best_a, best_obj


def subgradient_momentum(a0, time_limit, lr_init=0.02, top_k=10, temp_init=0.02):
    """
    Subgradient descent with Nesterov momentum and adaptive parameters.
    Uses the key insight: grad of (a*a)[k] at position m is 2*a[k-m].
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    lr = lr_init
    T = temp_init
    v = np.zeros(n)
    mom = 0.9
    t0 = time.monotonic()
    stagnant = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break

        conv = fast_convolve(a)
        peak_val = conv.max()

        # Top-k peaks with softmax
        if top_k < len(conv):
            peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peak_idx = np.arange(len(conv))

        pv = conv[peak_idx]
        T_eff = max(peak_val * T, 1e-15)
        w = np.exp((pv - pv.max()) / T_eff)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(peak_idx)) / len(peak_idx)
        else:
            w = w / w_sum

        # Compute weighted subgradient (vectorized)
        grad = np.zeros(n)
        for i in range(len(peak_idx)):
            wi = w[i]
            if wi < 1e-12:
                continue
            k = int(peak_idx[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * wi * a[k - idx]

        # Nesterov momentum + backtracking
        v = mom * v - lr * grad

        best_candidate = a
        best_val = peak_val
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            candidate = project_simplex(a + frac * v)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        # Adaptive parameters
        if stagnant > 30:
            T *= 1.05
            lr *= 0.95
        if stagnant > 100:
            T = max(0.001, T * 0.5)
            lr = min(0.05, lr * 1.5)
            mom = 0.95
        if stagnant > 300:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.002 + 0.003 * min(1.0, (stagnant - 300) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            lr = lr_init
            T = temp_init
            mom = 0.9
            stagnant = 0

    return best_a, best_obj


def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle and adaptive M."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    stagnant = 0
    it = 0
    m_relax = 1.001

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        g = solve_lp_oracle(f, M * m_relax)
        if g is None or g.sum() < 1e-15:
            g = solve_lp_oracle(f, M * 1.005)
        if g is None or g.sum() < 1e-15:
            stagnant += 1
            if stagnant > 30:
                rng = np.random.default_rng(it * 7919 + 3)
                scale = 0.002 + 0.003 * min(1.0, stagnant / 60.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.001
                stagnant = 0
            continue

        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-11:
            stagnant += 1
            if stagnant > 40:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue

        # Line search: coarse then fine
        best_alpha = 0.0
        best_val = M + 1e-14
        alphas_coarse = np.linspace(0.05, 1.0, 20)
        for alpha in alphas_coarse:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0.02:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            alphas_fine = np.linspace(lo, hi, 15)
            for alpha in alphas_fine:
                ft = (1.0 - alpha) * f + alpha * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha

        if best_alpha > 1e-9:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        if stagnant > 15:
            m_relax = min(m_relax * 1.03, 1.03)
        elif stagnant == 0:
            m_relax = max(1.001, m_relax * 0.95)

        if verbose and it % 50 == 0:
            print(f"    FW it={it}, obj={best_obj:.10f}, stagnant={stagnant}")

    return best_f, best_obj


def coordinate_descent(a0, time_limit):
    """
    Systematic coordinate descent: for each peak, try moving mass
    from contributing coordinates to distant locations.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    step_frac = 0.01
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        # Get top 3 peaks
        n_peaks = min(3, 2 * n - 1)
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        improved = False
        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue

            # Find the most contributing coordinate
            contribs = np.array([a[j] * a[k - j] for j in range(js, je + 1)])
            if contribs.max() < 1e-20:
                continue

            # Top contributing coordinates
            top_local = np.argsort(contribs)[::-1][:3]

            for ti in range(len(top_local)):
                j_src = js + int(top_local[ti])
                if a[j_src] < 1e-10:
                    continue

                # Try moving mass to several target locations
                # Target: positions that don't contribute much to current peaks
                # Heuristic: try positions far from all current peaks
                candidates = []
                for m in range(n):
                    if m == j_src:
                        continue
                    # Check if moving to m would help (rough estimate)
                    # New conv at peak positions would change by ~2*a[j_src]*a[m]
                    # We want m where a[k-m] is small for current peaks
                    score = 0.0
                    for pk in peak_indices:
                        pk_int = int(pk)
                        # Adding mass at m changes (a*a)[pk] by ~2*a[m_old]*a[pk-m]
                        # where m_old is the original mass (before move)
                        # After move, a[m] increases, so the change is:
                        # delta_conv[pk] = 2 * delta * a[pk - m] (approximately)
                        # We want this to be small
                        if 0 <= pk_int - m < n:
                            score += a[pk_int - m]
                        if 0 <= m - pk_int < n:
                            score += a[m - pk_int]
                    candidates.append((score, m))

                candidates.sort()

                for (_, m_target) in candidates[:5]:
                    delta = a[j_src] * step_frac
                    candidate = a.copy()
                    candidate[j_src] -= delta
                    candidate[m_target] += delta
                    if candidate[j_src] < -1e-14:
                        continue
                    candidate = np.maximum(candidate, 0)
                    cs = candidate.sum()
                    if cs < 1e-15:
                        continue
                    candidate = candidate / cs
                    val = fast_convolve(candidate).max()
                    if val < best_obj - 1e-15:
                        a = candidate
                        best_obj = val
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            step_frac *= 0.9
            if step_frac < 1e-7:
                break
            if it % 100 == 0:
                rng = np.random.default_rng(it * 48271)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
                    improved = True
                if it > 5000:
                    break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length via interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_starts(n, parent, rng):
    """Generate diverse initializations."""
    starts = []

    if parent is not None:
        starts.append(resample_sequence(parent, n))

    starts.append(np.ones(n) / n)

    # Cosine taper
    x = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 + np.cos(np.pi * x / (n - 1)))
    w = np.maximum(w, 0.01)
    starts.append(w / w.sum())

    # Gaussian
    g = np.exp(-0.5 * ((x - (n - 1) / 2) / (0.25 * n)) ** 2)
    starts.append(g / g.sum())

    # Edge-heavy
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    starts.append(edge / edge.sum())

    # Two peaks
    g1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.1 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())

    # Random
    for _ in range(2):
        starts.append(rng.dirichlet(np.ones(n) * 0.3))

    return starts

    2*n*max(conv(a,a))/(sum(a))^2.

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
