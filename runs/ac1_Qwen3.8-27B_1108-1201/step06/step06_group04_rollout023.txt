import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(w, a):
    """Compute W @ a where W_{i,j} = w[i+j]. w has length 2n-1, a has length n."""
    return np.correlate(w, a, mode='valid')

                         T_max=0.1, verbose=False):

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    best_lr = lr

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        conv = fast_convolve(a)
        # Compute softmax weights
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.ones(len(conv)) / len(conv)
        else:
            p = p / p_sum

        grad = 2.0 * hankel_mv(p, a)

        # Adaptive step size with backtracking
        improved = False
        for trial in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            candidate = project_simplex(a - trial * grad)
            c = fast_convolve(candidate)
            obj = c.max()
            if obj < best_obj - 1e-15:
                a = candidate
                best_obj = obj
                best_a = a.copy()
                best_lr = trial
                improved = True
                no_improve = 0
                break
        if not improved:
            no_improve += 1
            # Adapt temperature and learning rate
            if no_improve % 5 == 0:
                T *= 1.1
            if no_improve > 20:
                lr *= 0.9
                T = max(T_min, T * 0.8)
            if no_improve > 60:
                # Random restart near best
                rng = np.random.default_rng((it * 7919 + 13) % (2**31))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
                T = T_init
                lr = lr_init
                no_improve = 0

        if improved and no_improve == 0:
            lr = best_lr * 1.02  # Slightly increase if improving
            lr = min(lr, 0.2)

        if verbose and it % 100 == 0:
            print(f"      sg it={it}, obj={best_obj:.10f}, T={T:.6f}, lr={lr:.6f}")

    return best_a, best_obj


def mirror_descent_execute(a0, time_limit, lr_init=0.01, verbose=False):
    """Mirror descent (exponentiated gradient) on the simplex."""
    a = np.maximum(a0.astype(np.float64), 1e-30)
    a = a / a.sum()
    n = len(a)

    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Subgradient at the peak
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * a[k_peak - idx]

        # Mirror descent update: a_new[i] = a[i] * exp(-lr * grad[i])
        # This is equivalent to: a_new[i] = a[i] * exp(-lr * grad[i]) / Z
        logits = np.log(a + 1e-30) - lr * grad
        logits -= logits.max()
        exp_logits = np.exp(logits)
        a_new = exp_logits / exp_logits.sum()
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        c = fast_convolve(a_new)
        obj = c.max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            lr = min(lr * 1.01, 0.5)
        else:
            no_improve += 1
            if no_improve % 5 == 0:
                lr *= 0.95
            if no_improve > 50:
                rng = np.random.default_rng((it * 104729 + 7) % (2**31))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                lr = lr_init
                no_improve = 0

        a = a_new

        if verbose and it % 200 == 0:
            print(f"      md it={it}, obj={best_obj:.10f}, lr={lr:.6f}")

    return best_a, best_obj


def qp_solve_hankel(w, a_init, n, max_iters=300, time_limit=5.0):
    """
    Solve min a^T W a s.t. sum(a)=1, a>=0 where W is Hankel: W[i,j] = w[i+j].
    Active set method with Cholesky.
    """
    t0 = time.monotonic()
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    active = set(np.where(a > 1e-10)[0])
    if not active:
        active = set(range(n))

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if len(active) == 0:
            return np.ones(n) / n
        if len(active) == 1:
            a = np.zeros(n)
            a[list(active)[0]] = 1.0
            return a

        act = sorted(active)
        k = len(act)

        # Build submatrix
        act_arr = np.array(act)
        idx_sum = act_arr[:, None] + act_arr[None, :]
        W_AA = w[idx_sum]
        W_AA = 0.5 * (W_AA + W_AA.T)  # Ensure symmetry

        # Solve W_AA x = 1
        try:
            # Add small regularization
            W_AA_reg = W_AA + 1e-14 * np.eye(k)
            x = np.linalg.solve(W_AA_reg, np.ones(k))
        except np.linalg.LinAlgError:
            try:
                x = np.linalg.lstsq(W_AA + 1e-12 * np.eye(k), np.ones(k), rcond=None)[0]
            except Exception:
                x = np.ones(k) / k

        x_sum = x.sum()
        if abs(x_sum) < 1e-14:
            x = np.ones(k) / k
            x_sum = 1.0

        a_act = x / x_sum

        # Remove most negative
        min_val = a_act.min()
        if min_val < -1e-12:
            min_pos_in_act = int(np.argmin(a_act))
            active.discard(act[min_pos_in_act])
            a_new = np.zeros(n)
            for i, idx in enumerate(act):
                if i != min_pos_in_act:
                    a_new[idx] = max(0.0, a_act[i])
            s2 = a_new.sum()
            if s2 > 1e-15:
                a = a_new / s2
            continue

        # Form full solution
        a = np.zeros(n)
        for i, idx in enumerate(act):
            a[idx] = max(0.0, a_act[i])

        # Check KKT for inactive
        Wa = hankel_mv(w, a)
        grad = 2.0 * Wa
        mu = np.mean([grad[idx] for idx in act])

        inactive = set(range(n)) - active
        if inactive:
            min_viol = 0.0
            min_j = -1
            for j in sorted(inactive):
                viol = grad[j] - mu
                if viol < min_viol:
                    min_viol = viol
                    min_j = j
            if min_viol < -1e-10:
                active.add(min_j)
                continue

        return a

    return a


def primal_dual_execute(a0, time_limit, T_init=0.01, use_topk=False, topk=15,
                        blend=0.5, verbose=False):
    """Primal-dual QP iteration with adaptive weights."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    # Track peak values for adaptive weighting
    peak_history = []

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.3:
            break

        conv = fast_convolve(a)
        cmax = conv.max()

        # Adaptive weights
        if use_topk:
            k_use = min(topk, len(conv))
            peaks = np.argpartition(conv, -k_use)[-k_use:]
            w = np.zeros(len(conv))
            # Weight by peak height (higher peaks get more weight)
            peak_vals = conv[peaks]
            w[peaks] = peak_vals ** 2
            w_sum = w.sum()
            if w_sum > 0:
                w = w / w_sum
            else:
                w[peaks] = 1.0 / k_use
        else:
            scaled = conv / T
            scaled -= scaled.max()
            w = np.exp(scaled)
            w = w / w.sum()

        # Solve QP
        qp_time = min(2.0, remaining * 0.25)
        a_qp = qp_solve_hankel(w, a, n, max_iters=150, time_limit=qp_time)

        # Adaptive blending
        alpha = blend
        a_new = (1.0 - alpha) * a + alpha * a_qp
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a_qp
        else:
            a_new = a_new / s_new

        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adapt
        if no_improve > 8:
            T *= 1.2
        if no_improve > 25:
            T *= 0.7
            blend = min(0.8, blend * 1.1)
        if no_improve > 70:
            rng = np.random.default_rng((it * 31 + 41) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            T = T_init
            blend = 0.5
            no_improve = 0

        if verbose and it % 30 == 0:
            print(f"      pd it={it}, obj={best_obj:.10f}, T={T:.6f}")

    return best_a, best_obj


def coord_descent_execute(a0, time_limit, verbose=False):
    """Coordinate descent: try pairwise mass transfers to reduce peak."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Contributing positions to peak k_peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = list(range(js, je + 1))
        non_contrib = [j for j in range(n) if j not in set(contrib)]

        if not contrib or not non_contrib:
            break

        improved = False
        # Try moving mass from large contributors to non-contributors
        contrib_sorted = sorted(contrib, key=lambda i: a[i], reverse=True)
        nc_sorted = sorted(non_contrib, key=lambda i: a[i], reverse=True)

        for src in contrib_sorted[:5]:
            if a[src] < 1e-10:
                continue
            for dst in nc_sorted[:5]:
                for frac in [0.01, 0.03, 0.05, 0.1, 0.15]:
                    shift = a[src] * frac
                    if shift < 1e-12:
                        continue
                    a_try = a.copy()
                    a_try[src] -= shift
                    a_try[dst] += shift
                    if a_try[src] < 0:
                        continue
                    a_try = np.maximum(a_try, 0)
                    st = a_try.sum()
                    if st < 1e-15:
                        continue
                    a_try = a_try / st
                    c = fast_convolve(a_try)
                    if c.max() < best_obj - 1e-14:
                        a = a_try
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try random pairwise swaps
            rng = np.random.default_rng((it * 2749 + 3) % (2**31))
            for _ in range(10):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j or a[i] < 1e-10:
                    continue
                for frac in [0.05, 0.1, 0.2]:
                    shift = a[i] * frac
                    a_try = a.copy()
                    a_try[i] -= shift
                    a_try[j] += shift
                    a_try = np.maximum(a_try, 0)
                    st = a_try.sum()
                    if st < 1e-15:
                        continue
                    a_try = a_try / st
                    c = fast_convolve(a_try)
                    if c.max() < best_obj - 1e-14:
                        a = a_try
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and it > 500:
            break

        if verbose and it % 100 == 0:
            print(f"      cd it={it}, obj={best_obj:.10f}")

    return best_a, best_obj


def nesterov_subgrad_execute(a0, time_limit, lr_init=0.003, momentum_init=0.9, verbose=False):
    """Nesterov-accelerated subgradient for min max_k conv(a,a) on simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = momentum_init
    y = a.copy()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        # Nesterov lookahead
        y = a + mom * v

        conv_y = fast_convolve(y)
        k_peak = int(np.argmax(conv_y))

        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * y[k_peak - idx]

        v_new = project_simplex(y - lr * grad) - y
        a = y + v_new
        v = v_new
        a = project_simplex(a)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9997
        mom = min(0.995, mom + 0.00003)

        if no_improve > 80:
            lr *= 0.85
            mom *= 0.97
        if no_improve > 250:
            rng = np.random.default_rng((it * 48271 + 17) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            lr = lr_init * 0.5
            mom = momentum_init
            no_improve = 0

        if verbose and it % 300 == 0:
            print(f"      nesterov it={it}, obj={best_obj:.10f}")

    return best_a, best_obj


def peak_equalize_execute(a0, time_limit, verbose=False):
    """Try to equalize the top peaks by solving a small QP."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        # Get top peaks
        top_k = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:top_k]
        peak_values = conv[peak_indices]

        # Weight: emphasize the spread between peaks
        # Use weights proportional to how much each peak deviates above the mean
        mean_peak = peak_values.mean()
        deviations = np.maximum(peak_values - mean_peak, 0.0)
        w = np.zeros(len(conv))
        for i, pi in enumerate(peak_indices):
            w[pi] = deviations[i] + 0.1  # Small base weight
        w = w / w.sum()

        # Solve QP with these weights
        qp_time = min(1.5, (time_limit - (time.monotonic() - t0)) * 0.3)
        a_qp = qp_solve_hankel(w, a, n, max_iters=100, time_limit=qp_time)

        # Blend with current
        for alpha in [0.3, 0.5, 0.7, 0.9]:
            a_new = (1 - alpha) * a + alpha * a_qp
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            obj = fast_convolve(a_new).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
                break

        if not np.any(fast_convolve(a) < best_obj * 1.001):
            # Diminishing returns
            if it > 200:
                break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search for minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        try:
            score = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform768")

    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")

    # === Phase 1: Quick n-exploration (250s) ===
    print("\n=== Phase 1: Quick n-exploration ===")
    t1 = time.monotonic()

    candidate_ns = sorted(set([
        n_parent,
        max(128, n_parent - 96),
        n_parent + 96,

        remaining = deadline - time.monotonic() - 1200
        if remaining < 20:

                              np.linspace(0, 1, n_parent), parent)

        a_opt, _ = softened_grad_execute(start, time_limit=t_alloc, T_init=0.005, lr_init=0.03)
        try_seq(a_opt, f"sg_n{n_try}")

        # Quick mirror descent
        remaining = deadline - time.monotonic() - 1150
        if remaining > 10:
            t_alloc = min(remaining, 15)
            a_opt2, _ = mirror_descent_execute(a_opt, time_limit=t_alloc, lr_init=0.005)
            try_seq(a_opt2, f"md_n{n_try}")

    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")

    # === Phase 2: Deep optimization on best n (700s) ===
    print("\n=== Phase 2: Deep optimization ===")
    t2 = time.monotonic()

    if best_seq is not None:

        if remaining > 15:
            t_alloc = min(remaining, 80)
            print(f"  Peak equalize, t={t_alloc:.0f}s")
            a_opt, _ = peak_equalize_execute(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_eq")

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")

    # === Phase 4: Coordinate descent + fine tuning (250s) ===
    print("\n=== Phase 4: Coordinate Descent & Fine Tuning ===")
    t4 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 80
        if remaining > 10:
            t_alloc = min(remaining, 80)
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_opt, _ = coord_descent_execute(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord")

        # Fine smoothed gradient
        remaining = deadline - time.monotonic() - 50
        if remaining > 10:
            t_alloc = min(remaining, 50)
            print(f"  Fine SG, t={t_alloc:.0f}s")
            a_opt, _ = softened_grad_execute(best_seq, time_limit=t_alloc, T_init=0.0005, lr_init=0.01)
            try_seq(a_opt, "fine_sg")

        # Another round of PD with very low temperature
        remaining = deadline - time.monotonic() - 30
        if remaining > 10:
            t_alloc = min(remaining, 40)
            print(f"  Fine PD, t={t_alloc:.0f}s")
            a_opt, _ = primal_dual_execute(best_seq, time_limit=t_alloc, T_init=0.001,
                                           use_topk=True, topk=5, blend=0.3)
            try_seq(a_opt, "fine_pd")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # === Phase 5: Final polish (100s) ===
    print("\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple tiny perturbations with quick refinement
        for trial in range(8):
            remaining = deadline - time.monotonic() - 5
            if remaining < 3:
                break
            noise_scale = 0.0003 + 0.0003 * trial
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            t_alloc = min(remaining, 10)
            a_opt, _ = softened_grad_execute(f_pert, time_limit=t_alloc, T_init=0.0003, lr_init=0.005)
            try_seq(a_opt, f"perturb_{trial}")

        # Final Nesterov
        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            t_alloc = min(remaining, 20)
            a_opt, _ = nesterov_subgrad_execute(best_seq, time_limit=t_alloc, lr_init=0.001)
            try_seq(a_opt, "final_nesterov")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
