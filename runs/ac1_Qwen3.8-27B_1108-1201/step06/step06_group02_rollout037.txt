"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
New approaches:

- Top-K peak isolation with direct contribution-based reduction
- Adaptive n-reading via interpolation of best solution

- Fast LP oracle with greedy fallback

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return np.array([])
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_mv(p, a):
    """Compute sum_k p[k] * a[k-j] for all j (the 'Hankel' matrix-vector product)."""
    return np.correlate(p, a, mode='valid')

def solve_lp_oracle_fast(f, M, time_limit=5):
    """Fast LP oracle with greedy fallback."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix sparsely
    # conv(f,g)[k] = sum_j f[j]*g[k-j] <= M
    # For each k, the constraint involves g[j] for j in [max(0,k-n+1), min(n-1,k)]
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    # Greedy fallback: uniform direction
    return np.ones(n) / n

def multiplicative_optimize(a0, time_limit, T_init=0.001, lr_init=0.05, annealing_rate=0.9995):
    """
    Multiplicative (exponential) gradient descent.
    Update: a[j] <- a[j] * exp(-lr * grad[j]), then normalize.
    This preserves non-negativity and can converge faster.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        # Smoothed weights
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient
        grad = 2.0 * hankel_mv(p, a)

        # Multiplicative update
        # a_new[j] = a[j] * exp(-lr * grad[j])
        # To avoid underflow, center the gradient
        grad_centered = grad - grad.mean()
        scale = lr * np.abs(grad_centered).max()
        if scale > 1.0:
            grad_centered = grad_centered / scale * 1.0
        
        a_new = a * np.exp(-lr * grad_centered)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        obj = fast_convolve(a_new).max()
        
        # Accept or reject (simulated annealing)
        if obj < best_obj - 1e-14:
            a = a_new
            best_a = a.copy()
            best_obj = obj
            no_improve = 0
        elif obj < best_obj + T * 0.1 and np.random.random() < np.exp(-(obj - best_obj) / (T * 0.5)):
            a = a_new
            no_improve += 1
        else:
            no_improve += 1
            # Try smaller step
            if no_improve > 10:
                lr *= 0.9
                if lr < lr_init * 0.01:
                    # Restart
                    rng = np.random.default_rng(it * 31 + 7)
                    a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                    T = T_init
                    lr = lr_init
                    no_improve = 0
            continue

        # Anneal
        T *= annealing_rate
        if T < 1e-6:
            T = 1e-6

    return best_a, best_obj

def peak_isolation_reduce(a0, time_limit, top_k=5):
    """
    Identify top peaks in the convolution and directly reduce
    the elements that contribute most to each peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stale = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        
        # Get top K peaks
        if top_k >= len(conv):
            peak_idx = np.argsort(conv)[::-1]
        else:
            peak_idx = np.argpartition(conv, -top_k)[-top_k:]
            peak_idx = peak_idx[np.argsort(conv[peak_idx])[::-1]]

        improved = False
        
        for k in peak_idx:
            k = int(k)
            # Find elements contributing to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            affected = np.arange(js, je + 1)
            contributions = a[affected] * a[k - affected]
            
            # Find the element with maximum contribution
            j_max = affected[np.argmax(contributions)]
            
            # Find element with minimum contribution to any peak (to receive mass)
            # Compute total "pressure" on each element
            total_pressure = np.zeros(n)
            for kp in peak_idx:
                kp = int(kp)
                js_p = max(0, kp - n + 1)
                je_p = min(n - 1, kp)
                if js_p <= je_p:
                    aff_p = np.arange(js_p, je_p + 1)
                    total_pressure[aff_p] += a[aff_p] * a[kp - aff_p]
            
            # Don't move mass from elements that are already small
            min_val = a[j_max] * 0.5
            j_min = np.argmin(total_pressure)
            if a[j_min] < 1e-10:
                j_min = np.argsort(total_pressure)[3]  # 4th lowest pressure
            
            # Try to move mass
            for frac in [0.01, 0.02, 0.05, 0.1]:
                delta = a[j_max] * frac
                if delta < 1e-12:
                    break
                candidate = a.copy()
                candidate[j_max] -= delta
                candidate[j_min] += delta
                candidate = np.maximum(candidate, 0.0)
                cs = candidate.sum()
                if cs < 1e-15:
                    continue
                candidate = candidate / cs
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stale = 0
        else:
            stale += 1
            if stale > 50:
                # Perturb and restart
                rng = np.random.default_rng(it * 7919 + 13)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                stale = 0
                if stale > 200:
                    break

    return best_a, best_obj

def smoothed_grad_descent(a0, time_limit, T_init=0.001, lr_init=0.03):
    """Standard smoothed gradient descent with simplex projection."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = 2.0 * hankel_mv(p, a)

        best_candidate = a
        best_val = conv.max()

        for trial_lr in [lr, lr * 0.5, lr * 0.25]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            T *= 1.05
            lr *= 0.97
        if no_improve > 40:
            T *= 0.8
        if no_improve > 100:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.002 + 0.004 * min(1.0, (no_improve - 100) / 100.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
- Multiplicative (exponential) gradient updates for faster convergence and non-negativity
- Top-K peak isolation with direct contribution-based mass redistribution
- Adaptive n-exploration via interpolation and resampling
- Simulated annealing acceptance criterion for escaping local minima
- Mixed representation optimization (additive + multiplicative interleaved)
- Faster LP oracle with adaptive time limits and greedy fallback

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return np.array([])
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute the Hankel matrix-vector product: result[j] = sum_k p[k]*a[k-j]."""
    return np.correlate(p, a, mode='valid')


def solve_lp_oracle_fast(f, M, time_limit=8):
    """LP oracle: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

    Update: a[j] <- a[j] * exp(-lr * centered_grad[j]), then normalize.
    Optionally uses simulated annealing acceptance.
    """

    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    current_obj = best_obj

    t0 = time.monotonic()
    no_improve = 0
    it = 0

        if time.monotonic() - t0 > time_limit - 0.05:

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        grad = 2.0 * hankel_mv(p, a)

        # Alternate between additive and multiplicative
        if it % 3 == 0:
            # Additive step
            candidate = project_simplex(a - lr_add * grad)
        elif it % 3 == 1:
            # Multiplicative step
            grad_c = grad - grad.mean()
            max_abs = np.abs(grad_c).max()
            if max_abs > 1e-15:
                grad_c = grad_c / max_abs
            exponent = np.clip(-lr_mult * grad_c, -1.5, 1.5)
            candidate = a * np.exp(exponent)
            candidate = np.maximum(candidate, 1e-30)
            candidate = candidate / candidate.sum()
        else:
            # Blended step
            add_part = project_simplex(a - 0.5 * lr_add * grad)
            grad_c = grad - grad.mean()
            max_abs = np.abs(grad_c).max()
            if max_abs > 1e-15:
                grad_c = grad_c / max_abs
            mult_part = a * np.exp(np.clip(-0.5 * lr_mult * grad_c, -1.0, 1.0))
            mult_part = np.maximum(mult_part, 1e-30)
            mult_part = mult_part / mult_part.sum()
            candidate = 0.5 * add_part + 0.5 * mult_part
            candidate = project_simplex(candidate)

        obj = fast_convolve(candidate).max()
        if obj < best_obj - 1e-15:
            a = candidate
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 25:
            T *= 1.05
            lr_add *= 0.97
            lr_mult *= 0.97
        if no_improve > 80:
            noise = 0.002 + 0.004 * min(1.0, (no_improve - 80) / 100.0)
            a = project_simplex(best_a + local_rng.standard_normal(n) * noise)
            T = T_init
            no_improve = 0

    return best_a, best_obj


def generate_structural_inits(n, parent, local_rng):
    """Generate diverse structural initializations for exploration."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Interpolated parent
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g / g.sum())

    # Periodic structures
    for freq in [2, 3, 5]:
        for amp in [0.2, 0.4, 0.6]:
            h = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
            h = np.maximum(h, 0.01)
            inits.append(h / h.sum())

    # Random Dirichlet
    inits.append(local_rng.dirichlet(np.ones(n) * 0.5))
    inits.append(local_rng.dirichlet(np.ones(n) * 2.0))

    return inits


def try_and_update(a, state, label):
    """Evaluate a sequence and update state if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a)))


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid search:
    Phase 1: n-exploration with diverse initializations (250s)
    Phase 2: Deep FW-LP on best n (400s)
    Phase 3: Multiplicative + mixed optimization (350s)
    Phase 4: Peak isolation + gradient refinement (400s)
    Phase 5: Final polish with all methods (300s + buffer)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    state = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.8f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384

    n_parent = len(parent)

    # === Phase 1: n-Exploration with structural inits (~250s) ===
    print("\n=== Phase 1: n-Exploration ===")
    t_phase = time.monotonic()
    p1_budget = 0.14 * budget_s

    n_candidates = sorted(set([n_parent,
                                max(256, n_parent - 128), n_parent + 128,
                                max(256, n_parent - 256), n_parent + 256,
                                512, 640, 768, 896, 1024, 1152, 1280]))
    n_candidates = [n for n in n_candidates if 256 <= n <= 2048]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic() - (budget_s - p1_budget)
        if remaining < 15:
            break

        inits = generate_structural_inits(n_try, parent, rng)
        best_local = None
        best_local_obj = float('inf')

        for init in inits[:6]:
            remaining = deadline - time.monotonic() - (budget_s - p1_budget)
            if remaining < 5:
                break
            t_alloc = min(remaining, max(5, p1_budget // (len(n_candidates) * 6)))
            a_opt, obj = multiplicative_optimize(init, time_limit=t_alloc,
                                                  T_init=0.003, lr_init=0.05)
            if obj < best_local_obj:
                best_local_obj = obj
                best_local = a_opt

        if best_local is not None:
            try_and_update(best_local, state, "expl_n{}".format(n_try))

            remaining = deadline - time.monotonic() - (budget_s - p1_budget)
            if remaining > 8:
                a_opt2, _ = smoothed_grad_descent(best_local, time_limit=min(remaining, 10),
                                                   T_init=0.001, lr_init=0.03)
                try_and_update(a_opt2, state, "expl_sg_n{}".format(n_try))

    print("  Phase 1 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 2: Deep FW-LP (~400s) ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase = time.monotonic()
    p2_budget = 0.22 * budget_s

    if state[1] is not None:
        n_best = len(state[1])
        remaining = deadline - time.monotonic() - (budget_s - p2_budget - 600)
        if remaining > 20:
            t_alloc = min(remaining, p2_budget * 0.6)
            print("  FW-LP n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = fw_lp_optimize(state[1], time_limit=t_alloc, lp_time=12)
            try_and_update(a_opt, state, "deep_fw")

            # Perturbed restarts
            for scale in [0.002, 0.005, 0.01, 0.015]:
                remaining = deadline - time.monotonic() - (budget_s - p2_budget - 500)
                if remaining < 15:
                    break
                f_pert = project_simplex(state[1] + rng.standard_normal(n_best) * scale)
                t_alloc = min(remaining, p2_budget * 0.15)
                a_opt2, _ = fw_lp_optimize(f_pert, time_limit=t_alloc, lp_time=8)
                try_and_update(a_opt2, state, "fw_pert_{:.3f}".format(scale))

    print("  Phase 2 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 3: Multiplicative + Mixed optimization (~350s) ===
    print("\n=== Phase 3: Mult + Mixed ===")
    t_phase = time.monotonic()
    p3_budget = 0.20 * budget_s

    if state[1] is not None:
        # Multiple configurations
        configs = [
            ("mult", dict(T_init=0.0005, lr_init=0.06)),
            ("mult", dict(T_init=0.001, lr_init=0.04)),
            ("mult", dict(T_init=0.002, lr_init=0.03)),
            ("mixed", dict(T_init=0.001, lr_add=0.02, lr_mult=0.04)),
            ("mixed", dict(T_init=0.002, lr_add=0.015, lr_mult=0.03)),
        ]

        for method, params in configs:
            remaining = deadline - time.monotonic() - (budget_s - p3_budget - 450)
            if remaining < 15:
                break
            t_alloc = min(remaining, p3_budget // len(configs) + 10)
            if method == "mult":
                a_opt, _ = multiplicative_optimize(state[1], time_limit=t_alloc, **params)
            else:
                a_opt, _ = mixed_optimize(state[1], time_limit=t_alloc, **params)
            try_and_update(a_opt, state, "{}_{}".format(method, method))

    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 4: Peak isolation + gradient refinement (~400s) ===
    print("\n=== Phase 4: Peak Isolation + Gradient ===")
    t_phase = time.monotonic()
    p4_budget = 0.22 * budget_s

    if state[1] is not None:
        # Peak isolation with different K values
        for top_k in [3, 5, 8, 12]:
            remaining = deadline - time.monotonic() - (budget_s - p4_budget - 300)
            if remaining < 10:
                break
            t_alloc = min(remaining, p4_budget // 5 + 5)
            a_opt, _ = peak_isolation_reduce(state[1], time_limit=t_alloc, top_k=top_k)
            try_and_update(a_opt, state, "peak_k{}".format(top_k))

        # Smoothed gradient with various temperatures
        for T_init, lr_init in [(0.0003, 0.025), (0.0008, 0.02), (0.002, 0.015), (0.005, 0.01)]:
            remaining = deadline - time.monotonic() - (budget_s - p4_budget - 200)
            if remaining < 10:
                break
            t_alloc = min(remaining, p4_budget // 6)
            a_opt, _ = smoothed_grad_descent(state[1], time_limit=t_alloc,
                                             T_init=T_init, lr_init=lr_init)
            try_and_update(a_opt, state, "sg_T{}".format(T_init))

    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 5: Final polish (~300s + buffer) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Interleaved fine-tuning: alternate between methods
        n_best = len(state[1])
        polish_configs = [
            ("mult", dict(T_init=0.0002, lr_init=0.02)),
            ("mixed", dict(T_init=0.0005, lr_add=0.01, lr_mult=0.02)),
            ("sg", dict(T_init=0.0001, lr_init=0.008)),
            ("fw", None),
            ("peak", None),
        ]

        for method, params in polish_configs:
            remaining = deadline - time.monotonic() - 30
            if remaining < 8:
                break
            t_alloc = min(remaining, 60)
            if method == "mult":
                a_opt, _ = multiplicative_optimize(state[1], time_limit=t_alloc, **params)
            elif method == "mixed":
                a_opt, _ = mixed_optimize(state[1], time_limit=t_alloc, **params)
            elif method == "sg":
                a_opt, _ = smoothed_grad_descent(state[1], time_limit=t_alloc, **params)
            elif method == "fw":
                a_opt, _ = fw_lp_optimize(state[1], time_limit=t_alloc, lp_time=5)
            elif method == "peak":
                a_opt, _ = peak_isolation_reduce(state[1], time_limit=t_alloc, top_k=5)
            try_and_update(a_opt, state, "polish_{}".format(method))

        # Tiny perturbations with quick optimization
        remaining = deadline - time.monotonic() - 10
        if remaining > 5:
            for _ in range(4):
                remaining = deadline - time.monotonic() - 5
                if remaining < 3:
                    break
                noise = rng.standard_normal(n_best) * 0.0003
                f_pert = project_simplex(state[1] + noise)
                a_opt, _ = multiplicative_optimize(f_pert, time_limit=min(remaining, 5),
                                                   T_init=0.0001, lr_init=0.01,
                                                   use_anneal=False)
                try_and_update(a_opt, state, "perturb")

    print("  Phase 5 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(384) / 384
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
