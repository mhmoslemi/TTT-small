"""
Optimized step function search using smoothed gradient with Adam optimizer,
adaptive log-sum-exp temperature scheduling, peak equalization heuristics,
and multi-n exploration. Targets improvement over 1.504821 baseline.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad_smoothing(c, T, alpha=1.0):
    """Compute smoothed weights for gradient computation.
    
    Uses a power-softmax: w_k = (c_k/T)^alpha / sum((c_j/T)^alpha)
    This focuses more on peaks when alpha > 1.
    """
    c_clipped = np.maximum(c, 1e-30)
    scaled = c_clipped / (T + 1e-30)
    scaled = scaled ** alpha
    total = scaled.sum()
    if total < 1e-30:
        return np.ones(len(c)) / len(c)
    return scaled / total

def grad_from_weights(w, a):
    """Compute gradient of weighted sum of convolution values.
    
    grad_i = 2 * sum_k w_k * a_{k-i} for valid k-i
    This is 2 * correlate(w, a) (valid mode, with appropriate indexing)
    """
    n = len(a)
    # w has length 2n-1, a has length n
    # grad_i = 2 * sum_{k} w_k * a_{k-i} where 0 <= k-i < n and 0 <= k < 2n-1
    # = 2 * sum_m w_{i+m} * a_m where m = k-i, 0 <= m < n, 0 <= i+m < 2n-1
    # This is the cross-correlation of w and a
    # np.correlate(w, a, 'valid') gives sum_m w[i+m] * a[m] for i = 0..(len(w)-len(a))
    # len(w) = 2n-1, len(a) = n, so output length = 2n-1-n+1 = n. 
    return 2.0 * np.correlate(w, a, mode='valid')

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_weights_power(c, T, alpha):
    """Power-softmax weights: focuses more on peaks when alpha > 1."""
    c_safe = np.maximum(c, 1e-30)
    scaled = (c_safe / (T + 1e-30)) ** alpha
    total = scaled.sum()
    if total < 1e-30:
        return np.ones(len(c)) / len(c)
    return scaled / total

    """Gradient of sum_k w_k * c_k with respect to a."""
    return 2.0 * np.correlate(w, a, mode='valid')

            lr = lr_init * 0.7
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_obj


def nesterov_smoothed(a0, time_limit, T_init, T_final, lr_init, alpha_power):
    """Nesterov-accelerated smoothed gradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t = 0
    no_improve = 0

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit - 0.03:
            break

        # Nesterov lookahead
        a_look = a - v
        a_look = project_simplex(a_look)

        conv = fast_convolve(a_look)
        w = compute_weights_power(conv, T, alpha_power)
        grad = grad_weighted(w, a_look)

        v_new = 0.9 * v + lr * grad
        candidate = a - v_new
        candidate = project_simplex(candidate)

        best_candidate = a
        best_val = fast_convolve(a).max()

        for trial_scale in [1.0, 0.5, 0.25]:
            c = project_simplex(a - trial_scale * v_new)
            obj = fast_convolve(c).max()
            if obj < best_val - 1e-15:
                best_val = obj
                best_candidate = c

        a = best_candidate
        v = v_new * 0.9  # Dampen momentum
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        progress = min(1.0, t / max(1, int(time_limit * 30)))
        T = max(T_final, T_init * (T_final / T_init) ** progress)
        lr *= 0.9999

        if no_improve > 300:
            rng = np.random.default_rng((t * 31415 + 1) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_obj


def mirror_descent(a0, time_limit, lr_init=0.05, top_k=5):
    """Mirror descent (exponentiated gradient) focusing on top-k peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    t = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit - 0.03:
            break

        conv = fast_convolve(a)
        current_max = conv.max()

        # Get top-k peaks
        if top_k >= len(conv):
            top_idx = np.arange(len(conv))
        else:
            top_idx = np.argpartition(conv, -top_k)[-top_k:]

        # Average subgradient over top-k
        w = np.zeros(len(conv))
        w[top_idx] = 1.0 / min(top_k, len(conv))
        grad = grad_weighted(w, a)

        # Exponentiated gradient step
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - lr * grad
        a_new = np.exp(log_a_new - log_a_new.max())
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s

        c_new = fast_convolve(a_new)
        if c_new.max() < best_obj - 1e-15:
            best_obj = c_new.max()
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            # Still update a but keep best
            a = a_new

        lr *= 0.9995

        if no_improve > 200:
            rng = np.random.default_rng((t * 48271 + 1) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            no_improve = 0
            lr = lr_init * 0.5

    return best_a, best_obj


def peak_equalization(a0, time_limit, strategy='greedy'):
    """Reduce peaks by targeted mass transfers."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.03:
            break

        conv = fast_convolve(a)
        current_max = conv.max()

        # Find all peaks above threshold
        threshold = current_max - 1e-6
        peak_indices = np.where(conv >= threshold)[0]

        if len(peak_indices) == 0:
            break

        improved = False
        for k_peak in peak_indices:
            js = max(0, int(k_peak) - n + 1)
            je = min(n - 1, int(k_peak))
            affected = np.arange(js, je + 1)
            contributions = a[affected] * a[k_peak - affected]
            j_max = affected[np.argmax(contributions)]

            outside = np.setdiff1d(np.arange(n), affected)
            if len(outside) == 0:
                continue

            # Try multiple target positions
            candidates_j = outside[np.argsort(a[outside])[:5]]

            for j_target in candidates_j:
                for frac in [0.02, 0.01, 0.005, 0.002, 0.001]:
                    delta = a[j_max] * frac
                    if delta < 1e-15:
                        continue
                    candidate = a.copy()
                    candidate[j_max] -= delta
                    candidate[j_target] += delta
                    candidate = np.maximum(candidate, 0.0)
                    cs = candidate.sum()
                    if cs < 1e-15:
                        continue
                    candidate = candidate / cs
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50 and no_improve % 25 == 0:
                # Try random swaps
                rng = np.random.default_rng((it * 12345 + 7) % (2**31))
                for _ in range(10):
                    i1 = rng.integers(n)
                    i2 = rng.integers(n)
                    if i1 == i2:
                        continue
                    delta = a[i1] * 0.01
                    if delta < 1e-15:
                        continue
                    candidate = a.copy()
                    candidate[i1] -= delta
                    candidate[i2] += delta
                    candidate = np.maximum(candidate, 0.0)
                    candidate = candidate / candidate.sum()
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break

            if no_improve > 500:
                break

    return best_a, best_obj


def top_k_gradient(a0, time_limit, K, lr_init=0.01):
    """Gradient descent focusing uniformly on top-K peaks with Adam."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_improve = 0

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit - 0.03:
            break

        conv = fast_convolve(a)
        current_max = conv.max()

        if K >= len(conv):
            top_idx = np.arange(len(conv))
        else:
            top_idx = np.argpartition(conv, -K)[-K:]

        w = np.zeros(len(conv))
        w[top_idx] = 1.0 / len(top_idx)
        grad = grad_weighted(w, a)

        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad * grad
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)

        best_candidate = a
        best_val = current_max
        for trial_lr in [lr, lr * 0.5, lr * 0.25]:
            candidate = a - trial_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            candidate = project_simplex(candidate)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.99997

        if no_improve > 300:
            rng = np.random.default_rng((t * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.6

    return best_a, best_obj


def try_and_update(a, state):
    """Evaluate and update best state."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST: {:.10f} (n={})".format(score, len(a_norm)))


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid search:
    Phase 1: Multi-n exploration with Adam smoothed gradient
    Phase 2: Deep Adam + Nesterov on best n
    Phase 3: Mirror descent + Top-K gradient
    Phase 4: Peak equalization
    Phase 5: Final fine polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    state = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent: n={}, score={:.10f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)

    # === Phase 1: Multi-n Exploration (~250s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent, n_parent - 2, n_parent + 2,
                               512, 640, 700, 768, 800, 896, 1024]))
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 1200]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1100
        if remaining < 25:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                             np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try

        t_alloc = min(remaining, 30)
        a_opt, _ = adam_smoothed(start, time_limit=t_alloc,
                                  T_init=0.05, T_final=0.005,
                                  lr_init=0.01, alpha_power=2.0)
        try_and_update(a_opt, state)

    print("  Phase 1 best: {:.10f} ({:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 2: Deep Optimization (~550s) ===
    print("\n=== Phase 2: Deep Optimization ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Adam with various configs
        configs = [
            (0.03, 0.002, 0.008, 2.0),
            (0.01, 0.001, 0.012, 3.0),
            (0.05, 0.005, 0.006, 1.5),
        ]
        for T_i, T_f, lr_i, alpha in configs:
            remaining = deadline - time.monotonic() - 500
            if remaining < 50:
                break
            t_alloc = min(remaining, 150)
            print("  Adam T={:.4f}->{:.4f} lr={:.4f} a={:.1f} [{:.0f}s]".format(
                T_i, T_f, lr_i, alpha, t_alloc))
            a_opt, _ = adam_smoothed(state[1], time_limit=t_alloc,
                                      T_init=T_i, T_final=T_f,
                                      lr_init=lr_i, alpha_power=alpha)
            try_and_update(a_opt, state)

        # Nesterov
        remaining = deadline - time.monotonic() - 300
        if remaining > 40:
            t_alloc = min(remaining, 120)
            print("  Nesterov [{:.0f}s]".format(t_alloc))
            a_opt, _ = nesterov_smoothed(state[1], time_limit=t_alloc,
                                         T_init=0.02, T_final=0.002,
                                         lr_init=0.005, alpha_power=2.0)
            try_and_update(a_opt, state)

    print("  Phase 2 best: {:.10f} ({:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 3: Mirror Descent + Top-K (~350s) ===
    print("\n=== Phase 3: Mirror Descent + Top-K ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Mirror descent
        remaining = deadline - time.monotonic() - 300
        if remaining > 30:
            t_alloc = min(remaining, 80)
            print("  Mirror descent top5 [{:.0f}s]".format(t_alloc))
            a_opt, _ = mirror_descent(state[1], time_limit=t_alloc,
                                      lr_init=0.03, top_k=5)
            try_and_update(a_opt, state)

        remaining = deadline - time.monotonic() - 250
        if remaining > 30:
            t_alloc = min(remaining, 80)
            print("  Mirror descent top10 [{:.0f}s]".format(t_alloc))
            a_opt, _ = mirror_descent(state[1], time_limit=t_alloc,
                                      lr_init=0.02, top_k=10)
            try_and_update(a_opt, state)

        # Top-K gradient
        for K in [10, 30, 80]:
            remaining = deadline - time.monotonic() - 150
            if remaining < 30:
                break
            t_alloc = min(remaining, 60)
            print("  Top-K grad K={} [{:.0f}s]".format(K, t_alloc))
            a_opt, _ = top_k_gradient(state[1], time_limit=t_alloc, K=K, lr_init=0.008)
            try_and_update(a_opt, state)

    print("  Phase 3 best: {:.10f} ({:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 4: Peak Equalization (~300s) ===
    print("\n=== Phase 4: Peak Equalization ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        remaining = deadline - time.monotonic() - 120
        if remaining > 30:
            t_alloc = min(remaining, 180)
            print("  Peak equalization [{:.0f}s]".format(t_alloc))
            a_opt, _ = peak_equalization(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state)

        # Adam refinement after equalization
        remaining = deadline - time.monotonic() - 60
        if remaining > 20:
            t_alloc = min(remaining, 60)
            print("  Post-eq Adam [{:.0f}s]".format(t_alloc))
            a_opt, _ = adam_smoothed(state[1], time_limit=t_alloc,
                                      T_init=0.005, T_final=0.0005,
                                      lr_init=0.005, alpha_power=2.5)
            try_and_update(a_opt, state)

    print("  Phase 4 best: {:.10f} ({:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 5: Final Polish (~130s) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Very fine peak equalization
        remaining = deadline - time.monotonic() - 40
        if remaining > 20:
            t_alloc = min(remaining, 60)
            print("  Fine peak eq [{:.0f}s]".format(t_alloc))
            a_opt, _ = peak_equalization(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state)

        # Tiny perturbations with quick Adam
        remaining = deadline - time.monotonic() - 15
        for attempt in range(5):
            remaining = deadline - time.monotonic() - 8
            if remaining < 4:
                break
            noise_scale = 0.0002 + 0.0003 * attempt
            noise = rng.standard_normal(len(state[1])) * noise_scale
            f_pert = project_simplex(state[1] + noise)
            a_opt, _ = adam_smoothed(f_pert, time_limit=min(remaining, 8),
                                      T_init=0.002, T_final=0.0003,
                                      lr_init=0.003, alpha_power=2.0)
            try_and_update(a_opt, state)

    print("  Phase 5 best: {:.10f} ({:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = parent
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL: {:.10f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
