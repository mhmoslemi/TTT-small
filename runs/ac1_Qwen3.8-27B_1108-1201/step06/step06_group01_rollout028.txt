def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)

    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)

    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-15:

    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase hybrid approach:
- Fast FW-LP with full constraint oracle and adaptive M relaxation
- Smooth gradient descent with softmax-approximated max (fast local convergence)
- Nesterov-accelerated subgradient with adaptive top-k peak weighting
- Peak-targeted coordinate descent with golden-section refinement
- Multi-n sweep with structured initializations (Gaussian, raised cosine)
- Micro-polish with dual-informed perturbations

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_oracle(f, M, time_limit=0.8):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0. Returns g normalized to sum 1."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle, adaptive M, and smart restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    m_max = 4.0
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.2:
            break

        # Adaptive time limit per LP
        lp_tl = min(0.8, remaining * 0.01)
        g = solve_lp_oracle(f, M * m_exp, time_limit=lp_tl)

        if g is None:
            m_exp = min(m_exp * 1.15, m_max)
            if m_exp >= m_max:
                stall_count += 1
                if stall_count > 5:
                    # Restart with perturbation
                    rng = np.random.default_rng(it * 7919 + 3)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                    M = fast_convolve(f).max()
                    m_exp = 1.0
                    stall_count = 0
                    no_improve = 0
            continue

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                rng = np.random.default_rng(it * 104729 + 7)
                noise_scale = 0.002 + 0.002 * min(1.0, no_improve / 100.0)
                f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
                M = fast_convolve(f).max()
                m_exp = 1.0
                no_improve = 0
            elif no_improve % 5 == 0:
                m_exp = min(m_exp * 1.05, m_max)
            continue

        # Line search (coarse + fine)
        best_alpha = 0.0
        best_val = M
        alphas = [0.01, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5, 0.8, 1.0]
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.03)
            hi = min(1.0, best_alpha + 0.03)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                stall_count = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, m_max)
            stall_count += 1

    return best_f, best_obj


def smooth_gradient_descent(a0, time_limit, T_init=0.0008, lr_init=0.015):
    """
    Gradient descent on the softmax-smoothed objective.
    F_T(a) = T * log(sum_k exp(C_k(a)/T))
    grad_i = 2 * sum_j a[j] * p[i+j] where p_k = exp(C_k/T) / sum exp(C_k/T)
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        conv = fast_convolve(a)
        # Compute softmax weights
        shifted = (conv - conv.max()) / T
        p = np.exp(shifted)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(conv.shape)
            p[int(np.argmax(conv))] = 1.0
        else:
            p = p / p_sum

        # Gradient: grad[i] = 2 * sum_j a[j] * p[i+j]
        # = 2 * np.correlate(p, a, mode='valid')[i]
        grad = 2.0 * np.correlate(p, a, mode='valid')

        # Backtracking line search
        best_candidate = a
        best_val = conv.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05, lr * 0.02]:
            candidate = project_simplex(a - trial_lr * grad)
            c_new = fast_convolve(candidate)
            val = c_new.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameter updates
        if no_improve > 20:
            T *= 1.05
            lr *= 0.95
        if no_improve > 80:
            T *= 0.7
            lr *= 1.1
        if no_improve > 200:
            rng = np.random.default_rng((it * 48271 + 13) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def nesterov_subgradient(a0, time_limit, lr_init=0.004, max_k=40):
    """Nesterov-accelerated subgradient with adaptive top-k peak weighting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    y = a.copy()
    v = np.zeros(n)
    t_nest = 1.0
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        # Nesterov lookahead
        alpha = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / alpha
        y = (1.0 + beta) * a - beta * y
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y / ys
        t_nest += 1.0

        conv = fast_convolve(y)

        # Top-k peaks with polynomial decay weights
        k_eff = min(max_k, len(conv))
        peaks = np.argpartition(conv, -k_eff)[-k_eff:]
        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.8
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * y[k - idx_arr]

        # Update with momentum
        v = beta * v - lr * sg
        a_new = a + v
        a_new = project_simplex(a_new)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            a_new = a

        a = a_new
        lr *= (1.0 - 0.00015)

        # Adaptive restart
        if no_improve > 150:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.002 * min(3.0, np.sqrt(no_improve / 50.0))
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            y = a.copy()
            v = np.zeros(n)
            t_nest = 1.0
            no_improve = 0
            lr = lr_init

    return best_a, best_obj


def peak_coordinate_descent(a0, time_limit):
    """
    Targeted coordinate descent: for each top peak, try to reduce it
    by moving mass from high-contributing to low-contributing positions.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        top_k = min(5, len(conv))
        peak_lags = np.argpartition(conv, -top_k)[-top_k:]
        peak_lags = peak_lags[np.argsort(conv[peak_lags])[::-1]]

        improved = False

        for k_peak in peak_lags:
            if time.monotonic() - t0 > time_limit - 0.3:
                break

            js = max(0, int(k_peak) - n + 1)
            je = min(n - 1, int(k_peak))
            affected = np.arange(js, je + 1)

            if len(affected) < 2:
                continue

            # Sample affected positions
            if len(affected) > 25:
                rng_s = np.random.default_rng(it * 7 + k_peak)
                sampled = rng_s.choice(affected, size=25, replace=False)
            else:
                sampled = affected

            best_gain = 0
            best_move = None

            for ii in range(len(sampled)):
                for jj in range(ii + 1, len(sampled)):
                    j1 = sampled[ii]
                    j2 = sampled[jj]

                    contrib1 = a[k_peak - j1] if 0 <= k_peak - j1 < n else 0.0
                    contrib2 = a[k_peak - j2] if 0 <= k_peak - j2 < n else 0.0

                    if contrib1 >= contrib2 and a[j1] > 1e-8:
                        src, dst = j1, j2
                    elif contrib2 > contrib1 and a[j2] > 1e-8:
                        src, dst = j2, j1
                    else:
                        continue

                    for frac in [0.05, 0.15, 0.3, 0.5]:
                        delta = a[src] * frac
                        if delta < 1e-10:
                            continue
                        a_test = a.copy()
                        a_test[src] -= delta
                        a_test[dst] += delta
                        a_test = np.maximum(a_test, 0.0)
                        s = a_test.sum()
                        if s < 1e-15:
                            continue
                        a_test = a_test / s
                        c_new = fast_convolve(a_test)
                        gain = best_obj - c_new.max()
                        if gain > best_gain:
                            best_gain = gain
                            best_move = (src, dst, delta)

            if best_move is not None and best_gain > 1e-13:
                src, dst, delta = best_move
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                c_new = fast_convolve(a)
                best_obj = c_new.max()
                best_a = a.copy()
                improved = True
                stall = 0
            else:
                stall += 1

        if not improved:
            stall += 1

        if stall > 15:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise_scale = 0.0005 + 0.001 * min(1.0, stall / 40.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            stall = 0

        if stall > 150:
            break

    return best_a, best_obj


def micro_polish(a0, time_limit):
    """Very fine perturbation-based polish targeting peak reduction."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    scales = [0.00005, 0.0001, 0.0003, 0.001]
    si = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        scale = scales[si % len(scales)]
        rng = np.random.default_rng((it * 2654435761 + 1) % (2**32))

        tried = 0
        while tried < 8 and time.monotonic() - t0 < time_limit - 0.05:
            tried += 1
            conv = fast_convolve(a)
            peak_k = int(np.argmax(conv))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            noise = rng.standard_normal(n) * scale
            # Bias toward reducing peak contributors
            for j in range(js, je + 1):
                partner = peak_k - j
                if 0 <= partner < n:
                    noise[j] -= scale * 1.5 * a[partner]

            candidate = project_simplex(a + noise)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-16:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                break

        if it % 200 == 0:
            si += 1

    return best_a, best_obj


def gaussian_init(n, sigma_frac=0.35):
    """Gaussian-shaped initialization."""
    x = np.arange(n) / n
    center = 0.5
    sigma = sigma_frac / 2.0
    a = np.exp(-((x - center) ** 2) / (2 * sigma ** 2))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def raised_cosine_init(n):
    """Raised cosine window initialization."""
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / n))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a
