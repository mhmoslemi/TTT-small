import numpy as np
import time

def fast_autocorr(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_gradient(a, T):
    """Compute gradient of log-sum-exp smooth approximation of max(auto_corr)."""
    n = len(a)
    conv = fast_autocorr(a)
    scaled = T * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / (p.sum() + 1e-300)
    # Gradient: (grad)_j = sum_k p_k * a[k-j] for valid k-j
    # This is cross-correlation of p and a
    # p has length 2n-1, a has length n
    # (grad)_j = sum_{k: 0<=k-j<=n-1} p_k * a[k-j]
    #           = sum_{m=0}^{n-1} p_{j+m} * a[m]  (where j+m < 2n-1)
    # This is the 'valid' cross-correlation
    grad = np.correlate(p, a, mode='valid')
    return grad, conv, p

def hankel_mv(w, a):
    """Compute W @ a where W_{i,j} = w[i+j]. w has length >= 2n-1, a has length n."""
    return np.correlate(w, a, mode='valid')

def solve_qp_hankel(w, a_init, n, max_iters=500, time_limit=5.0):
    """
    Solve min a^T W a s.t. sum(a)=1, a>=0 using active set method.
    W is Hankel: W_{i,j} = w[i+j].
    """
    t0 = time.monotonic()
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    active = set(np.where(a > 1e-10)[0])
    if not active:
        active = set(range(n))

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if len(active) == 0:
            return np.ones(n) / n

        act = sorted(active)
        k = len(act)

        if k == 1:
            a = np.zeros(n)
            a[act[0]] = 1.0
            return a

        # Build KKT system: [W_A  1; 1^T  0] [a_A; mu] = [0; 1]
        W_AA = np.array([[w[act[i] + act[j]] for j in range(k)] for i in range(k)])
        W_AA += 1e-13 * np.eye(k)

        # KKT system
        KKT = np.zeros((k + 1, k + 1))
        KKT[:k, :k] = W_AA
        KKT[:k, k] = 1.0
        KKT[k, :k] = 1.0
        rhs = np.zeros(k + 1)
        rhs[k] = 1.0

        try:
            sol = np.linalg.solve(KKT, rhs)
        except np.linalg.LinAlgError:
            try:
                sol = np.linalg.lstsq(KKT, rhs, rcond=None)[0]
            except Exception:
                sol = np.zeros(k + 1)
                sol[:k] = 1.0 / k
                sol[k] = 0.0

        a_act = sol[:k]
        mu = sol[k]

        # Check for negative components
        min_idx = -1
        min_val = float('inf')
        for i in range(k):
            if a_act[i] < min_val:
                min_val = a_act[i]
                min_idx = i

        if min_val < -1e-10:
            # Remove the most negative component
            active.discard(act[min_idx])
            # Update a
            a = np.zeros(n)
            for i in range(k):
                if i != min_idx:
                    a[act[i]] = max(0.0, a_act[i])
            s2 = a.sum()
            if s2 > 1e-15:
                a = a / s2
            continue

        # All non-negative, check KKT for inactive
        a = np.zeros(n)
        for i in range(k):
            a[act[i]] = max(0.0, a_act[i])

        # Compute gradient 2*W*a
        Wa = hankel_mv(w, a)
        grad = 2.0 * Wa

        # Check KKT: for inactive j, grad[j] - mu >= 0
        inactive = set(range(n)) - active
        if inactive:
            worst_viol = 0.0
            worst_j = -1
            for j in sorted(inactive):
                viol = grad[j] - mu
                if viol < worst_viol:
                    worst_viol = viol
                    worst_j = j
            if worst_viol < -1e-8:
                active.add(worst_j)
                continue

        # Converged
        return a

    return a

        if no_improve > 50:
            T *= 0.7
        if no_improve > 150:
            rng = np.random.default_rng((it * 104729 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    QP-PD it={it}, obj={best_obj:.12f}, T={T:.6f}, no_imp={no_improve}")

    return best_a, best_obj

def coordinate_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent: for each pair (i,j), try to optimally redistribute
    mass between them to reduce the peak auto-correlation.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_autocorr(a).max()

    t0 = time.monotonic()
    it = 0
    improved_total = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        improved = False
        # Random subset of coordinate pairs for efficiency
        n_pairs = min(500, n * n // 4)
        seed = (it * 6151 + 17) % (2**31)
        rng = np.random.default_rng(seed)
        
        for _ in range(n_pairs):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            
            total = a[i] + a[j]
            if total < 1e-15:
                continue
            
            # Try different splits of total mass between i and j
            # The auto-corr is quadratic in (a[i], a[j]) for fixed other entries
            # So we can find the optimal split analytically
            # a*a at each lag k is a quadratic in a[i] and a[j]
            # (a*a)[k] = ... + a[i]*a[k-i] + a[j]*a[k-j] + a[i]*a[j]*[i+j==k or ...]
            # This is complicated. Let's just try a few splits.
            
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                a_try = a.copy()
                a_try[i] = total * frac
                a_try[j] = total * (1.0 - frac)
                obj = fast_autocorr(a_try).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        
        if not improved and it > 200:
            break

        if verbose and it % 50 == 0:
            print(f"    CD it={it}, obj={best_obj:.12f}")

    return best_a, best_obj

def random_perturb_refine(a0, time_limit, n_trials=100, verbose=False):
    """Try random perturbations and keep improvements."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        seed = (trial * 48271 + 7) % (2**31)
        rng = np.random.default_rng(seed)
        
        # Random perturbation
        scale = 0.001 + 0.01 / (1 + trial * 0.1)
        a_try = project_simplex(best_a + rng.standard_normal(n) * scale)
        obj = fast_autocorr(a_try).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_try.copy()
    
    return best_a, best_obj

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    
    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)
    
    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge / edge.sum())
    
    # Gaussian
    g = np.exp(-0.5 * ((x - mid) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Two peaks
    g2 = np.exp(-0.5 * ((x - 0.25 * (n-1)) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n-1)) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())
    
    # Cosine
    c1 = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    c1 = np.maximum(c1, 0.01)
    inits.append(c1 / c1.sum())
    
    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))
    
    # Palindromic random
    half = rng.dirichlet(np.ones(n // 2 + 1) * 0.5)
    pal = np.zeros(n)
    pal[:n//2+1] = half[:n//2+1]
    pal[n//2:] = half[:n//2][::-1]
    pal = np.maximum(pal, 0)
    if pal.sum() > 0:
        inits.append(pal / pal.sum())
    
    # Interpolate from parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())
    
    return inits
