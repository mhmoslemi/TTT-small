def peak_aware_gradient(a, top_k=20, temperature=0.001):
    """More efficient peak-aware subgradient computation."""
    n = len(a)
    conv = fast_convolve(a)
    
    # Get top peaks
    if top_k < len(conv):
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
    else:
        peak_idx = np.arange(len(conv))
    
    # Weight by excess over median
    peak_vals = conv[peak_idx]
    weight = peak_vals - np.median(conv)
    weight = np.maximum(weight, 0)
    if weight.sum() < 1e-15:
        weight = np.ones(len(peak_idx))
    weight /= weight.sum()
    
    # Compute subgradient
    s = np.zeros(n)
    for idx, k in enumerate(peak_idx):
        w = weight[idx]
        if w < 1e-10:
            continue
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            s[j_start:j_end+1] += w * 2.0 * a[k - j_end:k - j_start:-1]
    
    return s, conv

def nesterov_proximal_gradient(a0, time_limit, lr_init=0.05, mom_init=0.9,
                                  top_k=20, temp_init=0.001):
    """Nesterov-accelerated proximal gradient with peak-aware subgradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s_sum = a.sum()
    if s_sum < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s_sum
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    v = np.zeros(n)  # momentum
    lr = lr_init
    mom = mom_init
    temp = temp_init
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Nesterov lookahead
        y = a + mom * v
        
        # Compute subgradient at y
        sg, conv = peak_aware_gradient(y, top_k=top_k, temperature=temp)
        
        # Update
        v = mom * v - lr * sg
        a = project_simplex(a + v)
        
        obj = fast_convolve(a).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive parameters
        if no_improve > 30:
            temp *= 1.1
        if no_improve > 60:
            temp *= 0.7
            lr *= 0.8
        if no_improve > 100:
            # Restart with perturbation
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.002 + 0.005 * min(1.0, (no_improve - 100) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init * 0.7
            mom = mom_init
            temp = temp_init
            no_improve = 0
        
        # Slowly anneal
        if it % 100 == 0:
            top_k = min(len(conv), top_k + 2)
    
    return best_a, best_obj

def coordinate_swap_descent(a0, time_limit):
    """Coordinate descent via mass transfers between pairs."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        
        # Find the pair (i, j) with i+j=k_peak that maximizes a_i * a_j
        lo = max(0, k_peak - n + 1)
        hi = min(n - 1, k_peak)
        idx = np.arange(lo, hi + 1)
        pairs = a[idx] * a[k_peak - idx]
        j_max = idx[np.argmax(pairs)]
        
        # Find a "donor" position: minimize the increase in other peaks
        # Use the position with smallest a value outside the peak's support
        outside = np.setdiff1d(np.arange(n), idx)
        if len(outside) == 0:
            break
        j_min = outside[np.argmin(a[outside])]
        
        # Try several transfer amounts
        improved = False
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
            delta = a[j_max] * frac
            if delta < 1e-15:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min] += delta
            candidate = np.maximum(candidate, 0.0)
            cs = candidate.sum()
            if cs < 1e-15:
                continue
            candidate = candidate / cs
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try random swaps
            rng = np.random.default_rng(it * 12345)
            for _ in range(3):
                i1, i2 = rng.choice(n, size=2, replace=False)
                if a[i1] < 1e-10:
                    continue
                delta = a[i1] * 0.1
                candidate = a.copy()
                candidate[i1] -= delta
                candidate[i2] += delta
                candidate = np.maximum(candidate, 0.0)
                candidate = candidate / candidate.sum()
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break
            
            if not improved and it > 1000:
                break
    
    return best_a, best_obj

def lp_oracle_step(f, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(f,g)_k <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix efficiently
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j+n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def fw_with_lp(a0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_f = f.copy()
    best_obj = M
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        g = lp_oracle_step(f, M * m_exp, time_limit=min(remaining * 0.3, 10))
        if g is None:
            m_exp = min(m_exp * 1.5, 5.0)
            if m_exp > 4:
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 7919)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in np.linspace(0.01, 1.0, 15):
            ft = np.maximum((1 - alpha) * f + alpha * g, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = np.maximum((1 - best_alpha) * f + best_alpha * g, 0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.2, 5.0)
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 31)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, peak_idx, weights):
    """Compute weighted subgradient: sum_k w_k * grad b_k."""
    n = len(a)
    s = np.zeros(n)
    for idx in range(len(peak_idx)):
        w = weights[idx]
        if w < 1e-12:
            continue
        k = int(peak_idx[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            s[j_start:j_end + 1] += w * 2.0 * a[k - j_end:k - j_start:-1]
    return s


def peak_aware_subgradient(a, top_k=20, temperature=0.001):
    """Compute subgradient with adaptive peak weighting."""
    n = len(a)
    conv = fast_convolve(a)

    if top_k < len(conv):
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
    else:
        peak_idx = np.arange(len(conv))

    # Weight by exponential of excess over median
    med = np.median(conv)
    excess = conv[peak_idx] - med
    excess = np.maximum(excess, 1e-15)
    weights = np.exp(excess / temperature)
    weights /= weights.sum()

    s = compute_subgradient(a, peak_idx, weights)
    return s, conv, weights


def nesterov_optimize(a0, time_limit, lr_init=0.03, mom_init=0.85,

    best_obj = fast_convolve(a).max()

    v = np.zeros(n)
    lr = lr_init
    mom = mom_init
    temp = temp_init

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Nesterov lookahead
        y = a + mom * v

        a = project_simplex(a + v)

def soft_max_gradient(a, T=0.001):
    """Full softmax-weighted subgradient (all peaks, not just top-k)."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()

    # For efficiency, only use peaks with significant weight
    threshold = p.max() * 0.01
    active = np.where(p > threshold)[0]
    p_active = p[active]
    p_active = p_active / p_active.sum()

    s = compute_subgradient(a, active, p_active)
    return s, conv

def lp_oracle(f, M, time_limit=10):
    """LP: max sum(g) s.t. conv(f,g)_k <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def coordinate_descent(a0, time_limit):
    """Targeted coordinate swaps to reduce peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

        it += 1

            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        lo = max(0, k_peak - n + 1)
        hi = min(n - 1, k_peak)
        idx = np.arange(lo, hi + 1)
        if len(idx) == 0:
            break

        pairs = a[idx] * a[k_peak - idx]
        j_max = idx[np.argmax(pairs)]

        outside = np.setdiff1d(np.arange(n), idx)
        if len(outside) == 0:
            break
        j_min = outside[np.argmin(a[outside])]

        improved = False
        for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:

            candidate = a.copy()

            candidate[j_min] += delta
            candidate = np.maximum(candidate, 0.0)
            cs = candidate.sum()
            if cs < 1e-15:
                continue
            candidate = candidate / cs
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        if not improved:
            stuck += 1
            if stuck > 50:
                rng = np.random.default_rng(it * 12345)
                for _ in range(5):
                    i1, i2 = rng.choice(n, size=2, replace=False)

        else:

    return best_a, best_obj


def generate_shapes(n, parent, rng):
    """Generate diverse shape initializations."""
    shapes = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    shapes.append(np.ones(n) / n)

    # Gaussian centered
    for sigma_frac in [0.1, 0.15, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sigma_frac * n)) ** 2)
        shapes.append(g / g.sum())

    # Edge-weighted (decay from edges)
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    shapes.append(edge / edge.sum())

    # Triangular
    tri = 1.0 - 2.0 * np.abs(x - 0.5 * (n - 1)) / (n - 1)
    tri = np.maximum(tri, 0.01)
    shapes.append(tri / tri.sum())

    # Cosine bump
    cos_b = np.cos(np.pi * x / (n - 1)) ** 2 + 0.01
    shapes.append(cos_b / cos_b.sum())

    # Interpolated parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            shapes.append(px / ps)

    # Random Dirichlet
    for alpha in [0.3, 0.5, 1.0, 2.0]:
        shapes.append(rng.dirichlet(np.ones(n) * alpha))

    # Multi-frequency cosine
    for freq in [2, 3, 5]:
        h = 1.0 + 0.3 * np.cos(2.0 * np.pi * freq * x / n)
        h = np.maximum(h, 0.01)
        shapes.append(h / h.sum())

    return shapes

        for scale in [0.003, 0.007, 0.015, 0.03]:
            remaining = deadline - time.monotonic() - 500
            if remaining < 10:
                break
            f_pert = project_simplex(state[1] + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 40)
            a_opt, _ = nesterov_optimize(f_pert, time_limit=t_alloc,
                                         lr_init=0.02, mom_init=0.85,
                                         top_k=20, temp_init=0.001)
            try_and_update(a_opt, state, "ph2_pert_{}".format(scale))

    print("  Phase 2 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 3: LP-based Frank-Wolfe (~400s) ===
    print("\n=== Phase 3: FW-LP ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        remaining = deadline - time.monotonic() - 450
        if remaining > 30:
            t_alloc = min(remaining, 300)
            print("  FW-LP t={:.0f}s".format(t_alloc))
            a_opt, _ = fw_lp_optimize(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "ph3_fw")

        # Nesterov from FW result
        remaining = deadline - time.monotonic() - 250
        if remaining > 10:
            t_alloc = min(remaining, 60)
            a_opt, _ = nesterov_optimize(a_opt, time_limit=t_alloc,
                                         lr_init=0.01, mom_init=0.9,
                                         top_k=30, temp_init=0.0005)
            try_and_update(a_opt, state, "ph3_fw_nest")

        # FW from perturbed start
        remaining = deadline - time.monotonic() - 150
        if remaining > 20:
            f_pert = project_simplex(state[1] + rng.standard_normal(len(state[1])) * 0.005)
            t_alloc = min(remaining, 100)
            a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
            try_and_update(a_opt, state, "ph3_fw_pert")

    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 4: Coordinate descent (~200s) ===
    print("\n=== Phase 4: Coordinate Descent ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        remaining = deadline - time.monotonic() - 150
        if remaining > 10:
            t_alloc = min(remaining, 120)
            print("  Coord descent t={:.0f}s".format(t_alloc))
            a_opt, _ = coordinate_descent(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "ph4_coord")

            # Nesterov polish after coordinate descent
            remaining = deadline - time.monotonic() - 60
            if remaining > 8:
                t_alloc = min(remaining, 50)
                a_opt2, _ = nesterov_optimize(a_opt, time_limit=t_alloc,
                                              lr_init=0.008, mom_init=0.92,
                                              top_k=30, temp_init=0.0004)
                try_and_update(a_opt2, state, "ph4_coord_nest")

    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 5: Final polish (~150s) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Very fine Nesterov
        remaining = deadline - time.monotonic() - 40
        if remaining > 5:
            t_alloc = min(remaining, 60)
            print("  Fine Nesterov t={:.0f}s".format(t_alloc))
            a_opt, _ = nesterov_optimize(state[1], time_limit=t_alloc,
                                         lr_init=0.005, mom_init=0.95,
                                         top_k=50, temp_init=0.0002,
                                         restart_scale=0.002)
            try_and_update(a_opt, state, "ph5_fine")

        # Softmax with very low temperature
        remaining = deadline - time.monotonic() - 20
        if remaining > 5:
            t_alloc = min(remaining, 40)
            a_opt, _ = softmax_proximal(state[1], time_limit=t_alloc,
                                        T_init=0.0001, lr_init=0.02)
            try_and_update(a_opt, state, "ph5_soft")

        # Tiny perturbations with quick optimization
        remaining = deadline - time.monotonic() - 10
        if remaining > 5:
            for _ in range(8):
                remaining = deadline - time.monotonic() - 5
                if remaining < 3:
                    break
                noise = rng.standard_normal(len(state[1])) * 0.0008
                f_pert = project_simplex(state[1] + noise)
                t_alloc = min(remaining, 3)
                a_opt, _ = nesterov_optimize(f_pert, time_limit=t_alloc,
                                             lr_init=0.003, mom_init=0.9,
                                             top_k=20, temp_init=0.0003)
                try_and_update(a_opt, state, "ph5_tiny")

    print("  Phase 5 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Final comprehensive attempt with remaining time
    remaining = deadline - time.monotonic()
    if remaining > 20 and state[1] is not None:
        print("\n=== Bonus: Comprehensive Final Attempt ({:.0f}s) ===".format(remaining))
        # Combine best methods
        a_opt, _ = nesterov_optimize(state[1], time_limit=remaining * 0.4,
                                     lr_init=0.01, mom_init=0.9,
                                     top_k=30, temp_init=0.0005)
        try_and_update(a_opt, state, "bonus_nest")
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_opt, _ = fw_lp_optimize(state[1], time_limit=remaining * 0.5)
            try_and_update(a_opt, state, "bonus_fw")
        remaining = deadline - time.monotonic()
        if remaining > 5:
            a_opt, _ = coordinate_descent(state[1], time_limit=remaining * 0.8)
            try_and_update(a_opt, state, "bonus_coord")

    # Fallback
    if state[1] is None:
        state[1] = np.ones(384) / 384
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
