import time

def fft_size_for(n):
    """Smallest power of 2 >= 2n-1."""
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fs = fft_size_for(n)
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]

    Spectral flattening algorithm.
    Alternates between:
    - Frequency domain: flatten the power spectrum to reduce auto-convolution peaks
    - Time domain: enforce non-negativity and normalization
    """

    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    fs = fft_size_for(n)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    eta = eta_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Compute power spectrum
        A = np.fft.rfft(a, fs)
        S = np.abs(A)**2
        phase = A / np.maximum(np.abs(A), 1e-30)
        
        # Compute auto-convolution (circular = linear due to zero-padding)
        s_conv = np.fft.irfft(S, fs)[:2*n-1]

        c_max = s_conv[k_max]

        top_k = min(5, 2*n-1)

        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        freqs = np.arange(len(S))

        # = Re(IDFT(p_full rotated) evaluated at f)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Novel spectral (FFT-domain) flattening via error-reduction style iteration
- Primal-dual QP with Hankel structure (active set)
- Nesterov-accelerated subgradient on the max-conv object
- Symmetry-preserving updates
- Multi-n exploration with adaptive time allocation
- Coordinate descent with 1D minimax subproblems
"""

import numpy as np
import time


def fft_pow2(n):
    """Smallest power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fs = fft_pow2(size)
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(w, a):
    """Compute W @ a where W_{i,j} = w[i+j]. w length 2n-1, a length n."""
    return np.correlate(w, a, mode='valid')


def symmetrize(a):
    """Enforce a[i] = a[n-1-i]."""
    n = len(a)
    half = (n + 1) // 2
    a_sym = a.copy()
    for i in range(half):
        val = 0.5 * (a[i] + a[n - 1 - i])
        a_sym[i] = val
        a_sym[n - 1 - i] = val
    return a_sym


def spectral_flatten(a0, time_limit, eta_init=0.005, top_k=5, blend=0.8,
                     use_symmetry=True, verbose=False):
    """
    Spectral (FFT-domain) flattening: iteratively reduces auto-convolution peaks
    by adjusting the power spectrum, then enforcing time-domain non-negativity.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    fs = fft_pow2(2 * n - 1)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    eta = eta_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Compute power spectrum
        A = np.fft.rfft(a, fs)
        mag = np.abs(A)
        S = mag * mag
        phase = np.where(mag > 1e-30, A / mag, 1.0 + 0j)

        # Auto-convolution (circular = linear due to zero-padding)
        s_conv = np.fft.irfft(S, fs)[:2 * n - 1]

        c_max = s_conv.max()
        if c_max < 1e-30:
            break

        # Weight distribution over peaks (softmax)
        T = max(c_max * 0.05, 1e-10)
        scaled = s_conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-30:
            top_idx = int(np.argmax(s_conv))
            p = np.zeros(2 * n - 1)
            p[top_idx] = 1.0
        else:
            p = p / p_sum

        # Limit to top-k for focus
        if top_k < len(p):
            top_indices = np.argpartition(p, -top_k)[-top_k:]
            p_masked = np.zeros_like(p)
            p_masked[top_indices] = p[top_indices]
            p_masked /= p_masked.sum()
            p = p_masked

        # Frequency-domain gradient: sum_k p[k] * cos(2*pi*f*k/fs)
        # = Re(conj(FFT(p_padded)[f]))
        p_padded = np.zeros(2 * n - 1)
        p_padded[:] = p
        P = np.fft.rfft(p_padded, fs)
        grad_f = np.real(np.conj(P))

        # Update power spectrum
        S_new = S - eta * grad_f
        S_new = np.maximum(S_new, 0.0)

        # New amplitude
        amp_new = np.sqrt(S_new)
        A_new = phase * amp_new

        # Time-domain: enforce non-negativity
        a_new = np.fft.irfft(A_new, fs)[:n]

        if use_symmetry:
            a_new = symmetrize(a_new)

        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Blend for stability
        a_candidate = (1.0 - blend) * a + blend * a_new
        if use_symmetry:
            a_candidate = symmetrize(a_candidate)
        a_candidate = np.maximum(a_candidate, 0.0)
        s_c = a_candidate.sum()
        if s_c > 1e-15:
            a_candidate = a_candidate / s_c

        obj = fast_convolve(a_candidate).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_candidate.copy()
            no_improve = 0
            a = a_candidate
        else:
            a = a_candidate if obj <= best_obj + 1e-10 else a
            no_improve += 1

        # Adapt step size
        if no_improve > 5:
            eta *= 0.97
        if no_improve > 30:
            eta *= 0.9
        if no_improve > 100:
            # Larger step to escape
            eta = eta_init * 2.0
        if no_improve > 200:
            # Random restart
            rng = np.random.default_rng((it * 104729 + 7) % (2**31))
            a = np.maximum(best_a + rng.standard_normal(n) * 0.002, 0)
            if use_symmetry:
                a = symmetrize(a)
            a = a / a.sum()
            eta = eta_init
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    spectral it={it}, obj={best_obj:.10f}, eta={eta:.6f}, no_imp={no_improve}")

    return best_a, best_obj


def solve_qp_hankel(w, a_init, n, max_iters=300, time_limit=5.0):
    """
    Solve min a^T W a s.t. sum(a)=1, a>=0 using active set.
    W is Hankel: W_{i,j} = w[i+j].
    """
    t0 = time.monotonic()
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    active = set(np.where(a > 1e-10)[0])
    if not active:
        active = set(range(min(n, 200)))

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if len(active) == 0:
            a = np.zeros(n)
            a[0] = 1.0
            return a
        if len(active) == 1:
            a = np.zeros(n)
            a[list(active)[0]] = 1.0
            return a

        act = sorted(active)
        k = len(act)

        # For very large active sets, use gradient-based approach
        if k > 500:
            # Use a few gradient steps instead of solving the full system
            Wa = hankel_mv(w, a)
            grad = 2.0 * Wa
            mu = np.mean([grad[idx] for idx in act])
            # Simple projected gradient step
            a_act_new = a[act] - 0.1 * (grad[act] - mu)
            a_act_new = np.maximum(a_act_new, 0)
            s_act = a_act_new.sum()
            if s_act > 1e-15:
                a_act_new /= s_act
            else:
                a_act_new = np.ones(k) / k
            a = np.zeros(n)
            a[act] = a_act_new
            continue

        # Build submatrix W_A
        W_AA = np.array([[w[act[i] + act[j]] for j in range(k)] for i in range(k)])
        W_AA_reg = W_AA + 1e-10 * np.eye(k)

        try:
            x = np.linalg.solve(W_AA_reg, np.ones(k))
        except np.linalg.LinAlgError:
            try:
                x = np.linalg.lstsq(W_AA_reg, np.ones(k), rcond=None)[0]
            except Exception:
                x = np.ones(k) / k

        x_sum = x.sum()
        if abs(x_sum) < 1e-14:
            x = np.ones(k) / k
            x_sum = 1.0

        a_act = x / x_sum

        # Remove most negative component
        min_val = float('inf')
        min_idx = -1
        for i, idx in enumerate(act):
            if a_act[i] < min_val:
                min_val = a_act[i]
                min_idx = idx
        if min_val < -1e-12:
            active.discard(min_idx)
            a_new = np.zeros(n)
            for i, idx in enumerate(act):
                if idx != min_idx:
                    a_new[idx] = max(0.0, a_act[i])
            s2 = a_new.sum()
            if s2 > 1e-15:
                a = a_new / s2
            continue

        # Form full solution
        a = np.zeros(n)
        for i, idx in enumerate(act):
            a[idx] = max(0.0, a_act[i])
        s_a = a.sum()
        if s_a > 1e-15:
            a /= s_a

        # Check KKT for inactive
        Wa = hankel_mv(w, a)
        grad = 2.0 * Wa
        mu = np.mean([grad[idx] for idx in act])

        inactive = set(range(n)) - active
        if inactive:
            min_viol = 0.0
            min_j = -1
            for j in inactive:
                viol = grad[j] - mu
                if viol < min_viol:
                    min_viol = viol
                    min_j = j
            if min_viol < -1e-7:
                active.add(min_j)
                continue

        return a

    return a


def qp_primal_dual(a0, time_limit, T_init=0.01, use_topk=False, topk=15,
                   blend=0.5, use_symmetry=True, verbose=False):
    """
    Primal-dual QP iteration with Hankel weight matrices.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        conv = fast_convolve(a)
        cmax = conv.max()

        # Compute dual weights
        if use_topk:
            k_use = min(topk, len(conv))
            peaks = np.argpartition(conv, -k_use)[-k_use:]
            w = np.zeros(len(conv))
            w[peaks] = 1.0 / k_use
        else:
            scaled = conv / max(T, 1e-10)
            scaled -= scaled.max()
            w = np.exp(scaled)
            w = w / w.sum()

        # Solve QP
        qp_time = min(3.0, remaining * 0.25)
        a_qp = solve_qp_hankel(w, a, n, max_iters=150, time_limit=qp_time)

        if use_symmetry:
            a_qp = symmetrize(a_qp)

        # Blend
        a_new = (1.0 - blend) * a + blend * a_qp
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a_qp
        else:
            a_new = a_new / s_new

        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adapt temperature and blend

            T *= 1.2
        if no_improve > 50:
            T *= 0.7
            blend = max(0.2, blend * 0.8)
        if no_improve > 100:
            rng = np.random.default_rng((it * 48271 + 13) % (2**31))
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(best_a + noise)
            if use_symmetry:
                a = symmetrize(a)
                a = project_simplex(a)
            T = T_init
            blend = 0.5
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    qp_pd it={it}, obj={best_obj:.10f}, T={T:.6f}")

    return best_a, best_obj


def nesterov_subgrad(a0, time_limit, lr_init=0.005, mom_init=0.9,
                     use_symmetry=True, verbose=False):
    """
    Nesterov-accelerated subgradient for min max_k conv(a,a) s.t. a in simplex.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = momentum_init = mom_init
    y = a.copy()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        y = a + mom * v

        conv_y = fast_convolve(y)
        k_peak = int(np.argmax(conv_y))

        # Subgradient of conv_y[k_peak] w.r.t. a
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * y[k_peak - idx]

        v_new = project_simplex(y - lr * grad) - y
        a = y + v_new
        v = v_new
        a = project_simplex(a)

        if use_symmetry:
            a = symmetrize(a)
            a = project_simplex(a)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9997
        mom = min(0.995, mom + 0.00003)

        if no_improve > 150:
            lr *= 0.7
            mom *= 0.95
        if no_improve > 400:
            rng = np.random.default_rng((it * 314159 + 1) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            if use_symmetry:
                a = symmetrize(a)
                a = project_simplex(a)
            v = np.zeros(n)
            lr = lr_init * 0.5
            mom = momentum_init
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    nesterov it={it}, obj={best_obj:.10f}, lr={lr:.8f}")

    return best_a, best_obj


def smoothed_grad(a0, time_limit, T_init=0.001, lr_init=0.02,
                  use_symmetry=True, verbose=False):
    """Smoothed gradient descent with log-sum-exp surrogate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / max(T, 1e-10)
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = 2.0 * hankel_mv(p, a)

        # Backtracking line search
        best_candidate = a
        best_val = conv.max()
        trial_lr = lr
        for _ in range(5):
            candidate = project_simplex(a - trial_lr * grad)
            if use_symmetry:
                candidate = symmetrize(candidate)
                candidate = project_simplex(candidate)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
            trial_lr *= 0.5

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 20:
            T *= 1.15
            lr *= 0.95
        if no_improve > 80:
            T *= 0.5
            lr = lr_init * 0.5
        if no_improve > 250:
            rng = np.random.default_rng((it * 7919 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    smoothed it={it}, obj={best_obj:.10f}, T={T:.6f}")

    return best_a, best_obj


def coordinate_descent(a0, time_limit, use_symmetry=True, verbose=False):
    """
    Cyclic coordinate descent. For each coordinate, find the optimal value
    by minimizing the max of the affected convolution values.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stall = 0

    # Determine which coordinates to optimize
    if use_symmetry:
        coords = list(range((n + 1) // 2))
    else:
        coords = list(range(n))

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        it += 1
        improved = False

        for j in coords:
            if time.monotonic() - t0 > time_limit - 0.1:
                    break

            # We're changing a[j] (and a[n-1-j] if symmetric)
            # Try a range of values for a[j]
            current_val = a[j]
            if current_val < 1e-12:
                # Try small positive values
                candidates = [1e-4, 1e-3, 1e-3 * a.sum() / n * 10]
            else:
                # Try perturbations around current
                candidates = [current_val * f for f in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]]
                candidates = [max(c, 0.0) for c in candidates]

            best_candidate = a.copy()
            best_local = None

            for cv in candidates:
                trial = a.copy()
                # Adjust: move mass from j to maintain sum=1
                delta = cv - current_val
                if abs(delta) < 1e-15:
                    continue
                # Distribute delta across all other coordinates proportionally
                others = [i for i in range(n) if i != j and i != n - 1 - j]
                if not others:
                    continue
                # Subtract proportionally
                other_sum = a[others].sum()
                if other_sum < 1e-15:
                    continue
                for i in others:
                    trial[i] -= delta * a[i] / other_sum
                trial[j] = cv
                if use_symmetry:
                    trial[n - 1 - j] = cv

                trial = np.maximum(trial, 0)
                trial_sum = trial.sum()
                if trial_sum < 1e-15:
                    continue
                trial = trial / trial_sum

                obj = fast_convolve(trial).max()
                if best_local is None or obj < best_local:
                    best_local = obj
                    best_candidate = trial.copy()

            if best_local is not None and best_local < best_obj - 1e-14:
                a = best_candidate
                best_obj = best_local
                best_a = a.copy()
                improved = True

        if not improved:
            stall += 1
            if stall > 3:
                break
        else:
            stall = 0

        if verbose and it % 10 == 0:
            print(f"    coord it={it}, obj={best_obj:.10f}")

    return best_a, best_obj


def generate_diverse_inits(n, rng, num=6):
    """Generate diverse initializations for exploration."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    inits.append(edge / edge.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.12 * n)) ** 2)
    inits.append(g / g.sum())

    # Two peaks
    g2 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.08 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.08 * n)) ** 2)
    inits.append(g2 / g2.sum())

    # Cosine window
    c1 = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    c1 = np.maximum(c1, 0.01)
    inits.append(c1 / c1.sum())

    # Sinc-like
    t = (x - 0.5 * (n - 1)) / (0.1 * n)
    sinc = np.sinc(t)
    sinc = np.maximum(sinc, 0.0)
    if sinc.sum() > 1e-10:
        inits.append(sinc / sinc.sum())

    # Symmetric random
    r = rng.dirichlet(np.ones(n) * 0.5)
    r_sym = symmetrize(r)
    r_sym = np.maximum(r_sym, 0)
    if r_sym.sum() > 1e-10:
        inits.append(r_sym / r_sym.sum())

    return inits[:num]


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search combining spectral,
    primal-dual QP, Nesterov, and coordinate methods.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)}, t={time.monotonic():.0f})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")

    # === Phase 1: Quick multi-n exploration with spectral method (~400s) ===
    print("\n=== Phase 1: Multi-n spectral exploration ===")
    t1 = time.monotonic()

    candidate_ns = sorted(set([
        max(128, n_parent - 128),
        n_parent,
        n_parent + 128,
        n_parent + 256,
        max(256, n_parent // 2),
        n_parent * 2 if n_parent * 2 <= 3072 else n_parent,
        512, 768, 1024, 1280, 1536,
    ]))
    # Filter to reasonable range
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 2560]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1000
        if remaining < 30:
            break

        # Generate initializations
        if n_try == n_parent and parent is not None:
            inits = [parent.copy()]
            inits.extend(generate_diverse_inits(n_try, rng, num=3))
        else:
            # Interpolate parent or use fresh inits
            px = np.interp(np.linspace(0, 1, n_try),
                           np.linspace(0, 1, n_parent), parent)
            px = np.maximum(px, 0)
            if px.sum() > 1e-10:
                inits = [px / px.sum()]
            else:
                inits = [np.ones(n_try) / n_try]
            inits.extend(generate_diverse_inits(n_try, rng, num=2))

        t_alloc_total = min(remaining, 60)
        t_per_init = t_alloc_total / len(inits)

        for ii, init in enumerate(inits):
            if time.monotonic() > deadline - 1000:
                break
            t_alloc = min(t_per_init, 30)
            a_opt, _ = spectral_flatten(init, time_limit=t_alloc, eta_init=0.005,
                                        top_k=5, blend=0.7, use_symmetry=True)
            try_seq(a_opt, f"spec_n{n_try}_i{ii}")

        # Also try QP-PD on best found for this n
        if best_seq is not None and len(best_seq) == n_try:
            remaining = deadline - time.monotonic() - 950
            if remaining > 20:
                a_opt2, _ = qp_primal_dual(best_seq, time_limit=min(remaining, 25),
                                           T_init=0.01, use_topk=False, topk=15,
                                           blend=0.5, use_symmetry=True)
                try_seq(a_opt2, f"qp_n{n_try}")

    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")

    # === Phase 2: Deep spectral optimization (~500s) ===
    print("\n=== Phase 2: Deep spectral optimization ===")
    t2 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple spectral configs
        spec_configs = [
            {'eta_init': 0.003, 'top_k': 3, 'blend': 0.6},
            {'eta_init': 0.008, 'top_k': 5, 'blend': 0.8},
            {'eta_init': 0.005, 'top_k': 10, 'blend': 0.7},
            {'eta_init': 0.002, 'top_k': 5, 'blend': 0.5},
            {'eta_init': 0.01, 'top_k': 3, 'blend': 0.9},
        ]

        for ci, cfg in enumerate(spec_configs):
            remaining = deadline - time.monotonic() - 500
            if remaining < 20:
                break
            t_alloc = min(remaining, 100)
            print(f"  Spectral config {ci} (eta={cfg['eta_init']}, topk={cfg['top_k']}), t={t_alloc:.0f}s")
            a_opt, _ = spectral_flatten(best_seq, time_limit=t_alloc, **cfg, use_symmetry=True)
            try_seq(a_opt, f"spec_deep_{ci}")

    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")

    # === Phase 3: QP primal-dual deep (~300s) ===
    print("\n=== Phase 3: Deep QP Primal-Dual ===")
    t3 = time.monotonic()

    if best_seq is not None:
        qp_configs = [
            {'T_init': 0.005, 'use_topk': False, 'topk': 20, 'blend': 0.4},
            {'T_init': 0.01, 'use_topk': True, 'topk': 8, 'blend': 0.5},
            {'T_init': 0.003, 'use_topk': False, 'topk': 15, 'blend': 0.3},
            {'T_init': 0.02, 'use_topk': True, 'topk': 5, 'blend': 0.6},
        ]

        for ci, cfg in enumerate(qp_configs):
            remaining = deadline - time.monotonic() - 300
            if remaining < 20:
                break
            t_alloc = min(remaining, 75)
            print(f"  QP-PD config {ci}, t={t_alloc:.0f}s")
            a_opt, _ = qp_primal_dual(best_seq, time_limit=t_alloc, **cfg, use_symmetry=True)
            try_seq(a_opt, f"qp_deep_{ci}")

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")

    # === Phase 4: Nesterov + smoothed gradient (~250s) ===
    print("\n=== Phase 4: Nesterov & Smoothed Gradient ===")
    t4 = time.monotonic()

    if best_seq is not None:
        for lr_i in [0.003, 0.007, 0.002]:
            remaining = deadline - time.monotonic() - 200
            if remaining < 15:
                break
            t_alloc = min(remaining, 60)
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=t_alloc, lr_init=lr_i,
                                        use_symmetry=True)
            try_seq(a_opt, f"nesterov_lr{lr_i}")

        for T_i in [0.0005, 0.002, 0.0001]:
            remaining = deadline - time.monotonic() - 100
            if remaining < 15:
                break
            t_alloc = min(remaining, 50)
            a_opt, _ = smoothed_grad(best_seq, time_limit=t_alloc, T_init=T_i,
                                     lr_init=0.015, use_symmetry=True)
            try_seq(a_opt, f"smooth_T{T_i}")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # === Phase 5: Coordinate descent + alternating methods (~200s) ===
    print("\n=== Phase 5: Coordinate descent & alternation ===")
    t5 = time.monotonic()

    if best_seq is not None:
        # Coordinate descent
        remaining = deadline - time.monotonic() - 100
        if remaining > 20:
            t_alloc = min(remaining, 80)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc, use_symmetry=True)
            try_seq(a_opt, "coord")

        # Alternate spectral -> nesterov -> spectral
        for alt in range(4):
            remaining = deadline - time.monotonic() - 50
            if remaining < 10:
                break
            t_alloc1 = min(remaining * 0.4, 30)
            t_alloc2 = min(remaining * 0.3, 20)

            a_opt1, _ = spectral_flatten(best_seq, time_limit=t_alloc1,
                                         eta_init=0.003 + 0.002 * alt,
                                         top_k=3 + alt, blend=0.7, use_symmetry=True)
            try_seq(a_opt1, f"alt_spec_{alt}")

            a_opt2, _ = nesterov_subgrad(best_seq, time_limit=t_alloc2,
                                         lr_init=0.002, use_symmetry=True)
            try_seq(a_opt2, f"alt_nest_{alt}")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # === Phase 6: Final polish with tiny perturbations (~100s) ===
    print("\n=== Phase 6: Final polish ===")
    t6 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        for _ in range(8):
            remaining = deadline - time.monotonic() - 5
            if remaining < 5:
                break
            noise_scale = 0.0003 + 0.0005 * _
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            f_pert = symmetrize(f_pert)
            f_pert = project_simplex(f_pert)

            # Quick spectral polish
            t_alloc = min(remaining, 10)
            a_opt, _ = spectral_flatten(f_pert, time_limit=t_alloc, eta_init=0.002,
                                        top_k=3, blend=0.8, use_symmetry=True)
            try_seq(a_opt, f"polish_{_}")

        # One final Nesterov
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=min(remaining, 20),
                                        lr_init=0.001, use_symmetry=True)
            try_seq(a_opt, "final_nest")

    print(f"  Phase 6 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t6:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
