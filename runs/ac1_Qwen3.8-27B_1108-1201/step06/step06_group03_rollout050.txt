import numpy as np
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    if size <= 1:
        return np.array([a[0] * a[0]])
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smoothed_max(vals, T):
    """Log-sum-exp smoothing of max."""
    if T < 1e-15:
        return float(vals.max())
    vmax = vals.max()
    return float(vmax + T * np.log(np.sum(np.exp((vals - vmax) / T))))


def objective_and_grad_logspace(x, T, n):
    """Compute objective and gradient in log-space parameterization.
    
    a[i] = exp(x[i]) / sum(exp(x))

    a = np.exp(x)
    a = a / a.sum()
    c = fast_convolve(a)
    cmax = c.max()
    c_shifted = (c - cmax) / T
    exp_c = np.exp(c_shifted)
    w = exp_c / exp_c.sum()
    obj = cmax + T * np.log(exp_c.sum())

    # Transform to x-space
    dot_ga_a = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot_ga_a)
    
    return obj, grad_x

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-space L-BFGS with analytic gradient (smoothed max objective)
2. Momentum-based gradient descent in log-space
3. Peak-targeted surgery with first-order safety checks

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smoothed_max_val(vals, T):
    """Compute smoothed max value using log-sum-exp."""
    if T < 1e-15:
        return float(vals.max())
    vmax = vals.max()
    return float(vmax + T * np.log(np.sum(np.exp((vals - vmax) / T))))


def logspace_obj_grad(x, T, n):
    """Compute smoothed-max objective and gradient in log-space.
    
    Parameterization: a[i] = exp(x[i]) / sum(exp(x))
    Objective: smoothed_max(conv(a,a), T)
    Returns (objective_value, gradient_array).
    """
    x = x - x.max()
    a = np.exp(x)
    a = a / a.sum()
    c = fast_convolve(a)
    cmax = c.max()
    c_shifted = (c - cmax) / T
    exp_c = np.exp(c_shifted)
    w = exp_c / exp_c.sum()
    obj = cmax + T * np.log(exp_c.sum())

    # grad_a[i] = 2 * sum_k w[k] * a[k-i] (valid terms only)
    # This equals 2 * np.correlate(w, a, 'full')[:n]
    grad_a = 2.0 * np.correlate(w, a, 'full')[:n]
    # Log-space transformation: grad_x = a * (grad_a - dot(grad_a, a))
    dot_ga = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot_ga)

    return obj, grad_x


def logspace_lbfgs(a0, time_limit, T_list=None):

    a = np.maximum(a0, 1e-15).copy()

        T_list = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

    x = np.log(a)
    x = x - x.max()

    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())
    t0 = time.monotonic()

    for T in T_list:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 3:
            break
        t_alloc = min(remaining, 50)

        def obj_fn(x_in):
            x_in = x_in - x_in.max()
            a_tmp = np.exp(x_in)
            a_tmp = a_tmp / a_tmp.sum()
            c = fast_convolve(a_tmp)
            cmax = c.max()
            return cmax + T * np.log(np.sum(np.exp((c - cmax) / T)))

        def grad_fn(x_in):
            x_in = x_in - x_in.max()
            a_tmp = np.exp(x_in)
            a_tmp = a_tmp / a_tmp.sum()
            c = fast_convolve(a_tmp)

            return a_tmp * (grad_a - dot_ga)

def resample_seq(a, new_n):
    """Resample a sequence to a new length via linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s

                x = x - x.max()
                start = np.exp(x)
                start = start / start.sum()
                t_alloc = min(remaining, 40)
                a_opt, _ = logspace_lbfgs(start, time_limit=t_alloc,
                                           T_list=[0.001, 0.0005, 0.0002])
                try_seq(a_opt, f"refine_lbfgs_{ri}")
            elif ri % 3 == 1:
                # Momentum
                t_alloc = min(remaining, 30)
                a_opt, _ = momentum_logspace(best_seq, time_limit=t_alloc,
                                              lr_init=0.001, T=0.0003)
                try_seq(a_opt, f"refine_mom_{ri}")
            else:
                # Peak surgery
                t_alloc = min(remaining, 30)
                a_opt, _ = peak_surgery(best_seq, time_limit=t_alloc, seed=1000 + ri)
                try_seq(a_opt, f"refine_peak_{ri}")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Try a few different n values if time remains ===
    print("\n=== Phase 5: Alternative Lengths ===")
    t_phase = time.monotonic()

    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        alt_lengths = [n_best + 64, n_best - 64, n_best + 128, n_best + 256]
        alt_lengths = [l for l in alt_lengths if 128 <= l <= 2500 and l != n_best]
        for n_alt in alt_lengths[:3]:
            remaining = deadline - time.monotonic() - 10
            if remaining < 20:
                break
            start = resample_seq(best_seq, n_alt)
            t_alloc = min(remaining, 40)
            a_opt, _ = logspace_lbfgs(start, time_limit=t_alloc,
                                       T_list=[0.002, 0.001, 0.0005, 0.0002])
            try_seq(a_opt, f"alt_n{n_alt}")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = evaluate_safe(best_seq)

    # Final verification
    fs = evaluate_safe(best_seq)
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
