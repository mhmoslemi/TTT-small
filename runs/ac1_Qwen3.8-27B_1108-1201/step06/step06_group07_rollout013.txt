                      options={'presolve': True, 'time_limit': 10})

def build_sparse_row(f, k, n):
    """Build the row of the constraint matrix for peak k: (f*g)[k] = sum_j f[k-j]*g[j]."""
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    cols = np.arange(js, je + 1)
    vals = f[k - cols]
    return cols, vals


def sca_single_step(f, n, top_k=30, M_scale=1.0):
    """
    One SCA step: solve LP max sum(g) s.t. (f*g)[k] <= M for top_k peaks, g >= 0.
    Returns normalized g or None.
    """
    c = fast_convolve(f)
    M = c.max() * M_scale

    # Select top K peaks
    if top_k >= len(c):
        peaks = np.arange(len(c))
    else:
        peaks = np.argpartition(c, -top_k)[-top_k:]

    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(peaks):
        kcols, kvals = build_sparse_row(f, int(k), n)
        for j_pos in range(len(kcols)):
            rows.append(idx)
            cols.append(kcols[j_pos])
            vals.append(kvals[j_pos])

    if len(rows) == 0:
        return None

    A_sparse = csr_matrix((vals, (rows, cols)), shape=(len(peaks), n))
    c_obj = -np.ones(n)
    b_ub = np.full(len(peaks), M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_sparse, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def sca_optimize(f0, time_limit, top_k=30, verbose=False):
    """
    SCA optimization: iteratively solve LP oracle and blend.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    M_scale = 1.0

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        g = sca_single_step(f, n, top_k=top_k, M_scale=M_scale)

        if g is None:
            M_scale = min(2.0, M_scale * 1.05)
            no_improve += 1
            if no_improve > 30:
                f = best_f.copy()
                M_scale = 1.0
                no_improve = 0
            continue

        # Line search for best blend
        best_val = best_obj
        best_ft = f.copy()
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_ft = ft.copy()

        if best_val < best_obj - 1e-12:
            f = best_ft
            best_obj = best_val
            best_f = f.copy()
            M_scale = max(0.5, M_scale * 0.98)
            no_improve = 0
        else:
            no_improve += 1
            M_scale = min(2.0, M_scale * 0.995)

        if no_improve > 40:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
            scale = 0.002 + 0.005 * min(1.0, (no_improve - 40) / 80.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M_scale = 1.0
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    SCA it={it}, obj={best_obj:.10f}, M_scale={M_scale:.4f}")

    return best_f, best_obj


def mirror_descent(a0, time_limit, eta_init=0.05, top_k=15):
    """Multiplicative weight mirror descent with softmax peak weighting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    eta = eta_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    from scipy.special import softmax

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)
        max_val = conv.max()

        # Top peaks with softmax weights
        k_sel = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_sel)[-k_sel:]
        pv = conv[peaks]
        pw = softmax(pv / (0.01 * max_val + 1e-10))

        # Subgradient
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            w = pw[idx]
            if w < 1e-12:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je + 1] += w * 2.0 * a[k - je:k - js + 1][::-1]

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-18)
        a_new = a_new / a_new.sum()

        # Blend with line search
        best_candidate = a
        best_val = max_val
        for alpha in [0.3, 0.5, 0.7, 1.0]:
            cand = (1 - alpha) * a + alpha * a_new
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_candidate = cand

        a = best_candidate
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            eta *= 0.93
        if eta < 1e-5:
            eta = eta_init * 0.5
            no_improve = 0

        if no_improve > 100:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            eta = eta_init
            no_improve = 0

    return best_a, best_obj


def coordinate_descent(a0, time_limit, max_sweeps=200):
    """
    Coordinate descent: for each coordinate, find optimal value via 1D search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    c = fast_convolve(a)

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_sweeps:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        improved = False
        # Randomize coordinate order
        rng = np.random.default_rng((it * 31337 + 7) % (2**31 - 1))
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit - 1.0:
                break

            a_i = a[i]
            if a_i < 1e-15:
                continue

            # Determine affected c positions
            # c_new[k] = c[k] + 2*(x - a_i)*a[k-i] for k != 2i
            # c_new[2i] = c[2i] + x^2 - a_i^2

            # Precompute: which k are affected?
            # k ranges from i to i+n-1 (where k-i is in [0, n-1])
            # Also k-i must be in [0, n-1], so k in [i, i+n-1] intersected with [0, 2n-2]
            k_start = max(0, i)
            k_end = min(2 * n - 2, i + n - 1)

            # Precompute max of unaffected parts
            if k_start > 0 or k_end < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_start:k_end + 1] = False
                max_unaffected = c[mask].max() if mask.any() else 0.0
            else:
                max_unaffected = 0.0

            # Affected positions
            ka = np.arange(k_start, k_end + 1)
            # For each affected k, the coefficient of (x - a_i) in c_new[k]
            # is 2*a[k-i], except for k=2i where it's quadratic
            coeff = 2.0 * a[ka - i]  # coefficient for linear part

            # The quadratic term at k=2i: x^2 - a_i^2
            k2i = 2 * i
            has_quad = (k2i >= k_start) and (k2i <= k_end)

            # Grid search for x
            x_max = max(3.0 * a_i, 4.0 / n)
            grid = np.linspace(0.0, x_max, 40)
            # Also include some values near a_i
            grid = np.concatenate([grid, np.linspace(max(0, a_i * 0.5), a_i * 1.5, 20)])
            grid = np.unique(grid)

            best_x = a_i
            best_x_obj = best_obj * (a.sum()) ** 2  # Since we work with sum=1, obj = 2n*max(c)

            for x in grid:
                # Compute max of c_new
                max_val = max_unaffected
                # Linear part
                linear_vals = c[ka] + 2.0 * (x - a_i) * coeff
                max_val = max(max_val, linear_vals.max())
                # Quadratic part at k=2i
                if has_quad:
                    quad_val = c[k2i] + x * x - a_i * a_i
                    max_val = max(max_val, quad_val)

                S_new = 1.0 - a_i + x
                if S_new < 1e-15:
                    continue
                obj_val = 2.0 * n * max_val / (S_new * S_new)

                if obj_val < best_x_obj:
                    best_x_obj = obj_val
                    best_x = x

            if abs(best_x - a_i) > 1e-12:
                # Update c
                delta = best_x - a_i
                c[ka] += 2.0 * delta * coeff
                if has_quad:
                    c[k2i] = c[k2i] - 2.0 * delta * a_i + delta * delta
                    # Actually: c[k2i] should be c_old[k2i] + (best_x^2 - a_i^2)
                    # We already added 2*delta*a_i via the linear part, so we need to adjust
                    # c_new[k2i] = c_old[k2i] + 2*delta*a_i + delta^2 = c_old[k2i] + delta*(2*a_i + delta)
                    # But the linear formula gives: c_old[k2i] + 2*delta*a[2i-i] = c_old[k2i] + 2*delta*a_i
                    # So we need to add delta^2 more
                    c[k2i] += delta * delta
                a[i] = best_x
                improved = True

        # Re-normalize and check
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-15:
            break
        a = a / S
        c = fast_convolve(a)
        obj = c.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        if not improved:
            # Try perturbation
            rng = np.random.default_rng((it * 1299709 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            c = fast_convolve(a)

    return best_a, best_obj


def peak_redistribute(a0, time_limit, frac=0.05):
    """
    Targeted peak shaving: reduce the highest peak by moving mass from
    positions that contribute to it to positions that don't.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        c = fast_convolve(a)
        k_peak = int(np.argmax(c))

        # Positions contributing to this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)

        # Contribution of each j to the peak
        contrib = np.zeros(n)
        for j in range(js, je + 1):
            contrib[j] = a[j] * a[k_peak - j]

        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break

        # Move mass from high-contrib to low-contrib positions
        move_frac = frac
        move_amount = np.zeros(n)
        for j in range(n):
            if contrib[j] > 0:
                move_amount[j] -= move_frac * a[j] * (contrib[j] / total_contrib)

        # Find positions NOT in the peak span to add mass
        outside = [j for j in range(n) if j < js or j > je]
        if len(outside) == 0:
            outside = [j for j in range(n) if j != js and j != je]

        if len(outside) > 0:
            total_move = -move_amount.sum()
            if total_move > 0:
                ext_vals = a[outside]
                ext_sum = ext_vals.sum()
                if ext_sum > 1e-15:
                    add = total_move * ext_vals / ext_sum
                else:
                    add = np.full(len(outside), total_move / len(outside))
                for idx, j in enumerate(outside):
                    move_amount[j] += add[idx]

        a_new = a + move_amount
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-15:
            break
        a_new = a_new / S

        new_obj = fast_convolve(a_new).max()
        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 20:
            rng = np.random.default_rng((it * 9973 + 5) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0

    return best_a, best_obj


def subgradient_ls(a0, time_limit, top_k=10):
    """Subgradient with line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = 0.05

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            weights = pv / pv.sum()

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            w = weights[idx]
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je + 1] += w * 2.0 * a[k - je:k - js + 1][::-1]

        direction = project_simplex(a - lr * grad) - a

        # Binary line search
        lo, hi = 0.0, 1.0
        best_alpha = 0.0
        best_val = conv.max()
        for _ in range(10):
            mid = (lo + hi) / 2.0
            cand = a + mid * direction
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                hi = mid
                continue
