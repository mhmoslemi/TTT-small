def spectral_init(n, rng):
    """Generate a non-negative sequence with approximately flat spectrum."""
    # Random phases
    phases = rng.uniform(0, 2*np.pi, n)
    # Flat magnitude (approximately)
    magnitude = np.ones(n)
    # IDFT
    a_complex = np.fft.idft(magnitude * np.exp(1j * phases))
    # Take real part and make non-negative
    a = np.real(a_complex)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Annealed log-sum-exp with Nesterov-accelerated projected gradient (main driver)
- Frank-Wolfe with LP oracle for medium-temperature polishing
- Peak-targeted coordinate descent for final micro-optimization
- Spectral initialization for diverse starting points
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_weights(a, t):
    """Compute softmax weights w_k = exp(t*conv_k) / sum exp(t*conv)."""
    conv = fast_convolve(a)
    t_conv = t * conv
    t_conv_max = t_conv.max()
    exp_vals = np.exp(t_conv - t_conv_max)
    w = exp_vals / exp_vals.sum()
    return w, conv


def compute_lse_gradient(a, w):
    """Compute gradient of (1/t)*log(sum(exp(t*conv(a,a)))) w.r.t. a.
    grad[m] = 2 * sum_j a[j] * w[m+j]
    = 2 * conv(a_reversed, w)[m + n - 1]
    """
    n = len(a)
    ra = a[::-1]
    corr_full = fast_convolve_ab(ra, w)
    # corr_full has length 2n-1
    # corr[m] = corr_full[m + n - 1] for m = 0, ..., n-1
    corr = corr_full[n - 1:2 * n - 1]
    return 2.0 * corr


def lse_objective(a, t):
    """Compute (1/t)*log(sum(exp(t*conv(a,a))))."""
    conv = fast_convolve(a)
    t_conv = t * conv
    t_conv_max = t_conv.max()
    log_sum = t_conv_max + np.log(np.exp(t_conv - t_conv_max).sum())
    return log_sum / t


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (f*g) = A @ g."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-7})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def nesterov_lse(a0, t_values, time_limit, verbose=False):
    """
    Annealed log-sum-exp optimization with Nesterov acceleration.
    
    Parameters:
    - a0: initial point (normalized, sum=1)
    - t_values: list of temperatures to anneal through
    - time_limit: total time in seconds
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_conv_max = fast_convolve(a).max()
    
    t0 = time.monotonic()
    
    for t_idx, t in enumerate(t_values):
        iter_start = time.monotonic()
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Allocate time for this temperature level
        frac = 1.0 / len(t_values)
        t_alloc = min(remaining, time_limit * frac * 1.2)  # Slight overlap
        
        if verbose:
            print(f"    LSE t={t:.1f}, alloc={t_alloc:.1f}s")
        
        # Initialize Nesterov momentum
        y = a.copy()
        a_prev = a.copy()
        k = 1
        step = 1.0 / (4.0 * t * n)  # Initial step size estimate
        
        max_iters = int(t_alloc / 0.002)  # Estimate max iterations
        stagnant = 0
        
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit - 1:
                break
            
            # Nesterov interpolation parameter
            beta = (1.0 - np.sqrt(1.0 + 4.0 / (k * k))) / 2.0
            # Actually, standard Nesterov: beta_k = (1 - sqrt(1 + 4/k^2)) / 2
            # Or simpler: beta_k = (k-1)/(k+2)
            beta = (k - 1.0) / (k + 2.0)
            
            y = a + beta * (a - a_prev)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
                sy = s
            else:
                y = y / sy
            
            # Compute gradient at y
            w, conv_y = compute_lse_weights(y, t)
            grad_y = compute_lse_gradient(y, w)
            
            # Projected gradient step with backtracking
            a_candidate = None
            for _bt in range(20):
                z = y - step * grad_y
                a_new = project_simplex(z)
                
                # Check descent condition (simplified: just check objective)
                # For efficiency, skip the full check and just use the step
                a_candidate = a_new
                break  # Skip backtracking for speed (use fixed step)
                # In a production version, we'd do proper backtracking here
            
            if a_candidate is None:
                a_candidate = a.copy()
            
            # Update
            a_prev = a.copy()
            a = a_candidate
            k += 1
            
            # Periodic check
            if it % 500 == 0 or it == max_iters - 1:
                conv_max = fast_convolve(a).max()
                if conv_max < best_conv_max - 1e-15:
                    best_conv_max = conv_max
                    best_a = a.copy()
                    stagnant = 0
                else:
                    stagnant += 1
                
                # Adapt step size
                if stagnant > 5:
                    step *= 0.7
                
                if verbose and it % 2000 == 0:
                    print(f"      t={t:.1f}, it={it}, conv_max={conv_max:.10f}, stagnant={stagnant}")
        
        a = best_a  # Use best found so far for next temperature level
    
    return best_a, best_conv_max


def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    stagnant = 0
    it = 0
    m_relax = 1.002
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.3:
            break
        
        # Solve LP
        g = solve_lp_oracle(f, M * m_relax)
        
        if g is None or g.sum() < 1e-15:
            g = solve_lp_oracle(f, M * 1.01)
        
        if g is None or g.sum() < 1e-15:
            stagnant += 1
            if stagnant > 15:
                rng = np.random.default_rng(it * 7919 + 3)
                scale = 0.002 + 0.003 * min(1.0, stagnant / 40.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
                m_relax = 1.002
            continue
        
        g = g / g.sum()
        
        if np.max(np.abs(g - f)) < 1e-10:
            stagnant += 1
            if stagnant > 25:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue
        
        # Golden section line search (15 evaluations)
        best_alpha = 0.0
        best_val = M + 1e-14
        
        # First: coarse scan (10 points)
        for alpha in np.linspace(0.01, 1.0, 10):
            ft = (1.0 - alpha) * f + alpha * g
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = np.maximum(ft, 0.0) / ft.sum()
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Fine scan (8 points) around best
        if best_alpha > 0.005:
            lo = max(0.001, best_alpha - 0.02)
            hi = min(1.0, best_alpha + 0.02)
            for alpha in np.linspace(lo, hi, 8):
                ft = (1.0 - alpha) * f + alpha * g
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = np.maximum(ft, 0.0) / ft.sum()
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
        
        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1
        
        if stagnant > 8:
            m_relax = min(m_relax * 1.03, 1.04)
    
    return best_f, best_obj


def peak_coord_descent(a0, time_limit):
    """
    Coordinate descent targeting the top peaks.
    For each peak, try to reduce it by moving mass from contributing coordinates.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    step_scale = 0.003
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        n_peaks = min(3, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]
        
        improved = False
        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            if js > je:
                continue
            
            # Find top contributing elements
            contribs = []
            for j in range(js, je + 1):
                contribs.append((a[j] * a[k - j], j, k - j))
            contribs.sort(reverse=True)
            
            for (_, j1, j2) in contribs[:8]:
                if a[j1] < 1e-9:
                    continue
