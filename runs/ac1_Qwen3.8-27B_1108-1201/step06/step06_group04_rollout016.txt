import numpy as np

from scipy.optimize import minimize_scalar

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_mv(w, a):
    return np.correlate(w, a, mode='valid')

def solve_qp_scipy(w, a_init, n, time_limit=5.0):
    """Solve min a^T W a s.t. sum(a)=1, a>=0 using scipy SLSQP."""
    from scipy.optimize import minimize
    
    W_AA_full = np.array([[w[i+j] if i+j < len(w) else 0 for j in range(n)] for i in range(n)])
    # Symmetrize
    W_AA_full = (W_AA_full + W_AA_full.T) / 2
    
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    def obj(x):
        return x @ W_AA_full @ x
    
    def grad(x):
        return 2 * W_AA_full @ x
    
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - 1, 'jac': lambda x: np.ones(n)}]
    bounds = [(0, None)] * n
    
    try:
        result = minimize(obj, a, jac=grad, method='SLSQP', bounds=bounds,
                         constraints=constraints, options={'maxiter': 100, 'ftol': 1e-12})
        if result.success:
            a_new = np.maximum(result.x, 0)
            a_new = a_new / a_new.sum()
            return a_new
    except Exception:
        pass
    
    return a

def fast_convolve(a):

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
