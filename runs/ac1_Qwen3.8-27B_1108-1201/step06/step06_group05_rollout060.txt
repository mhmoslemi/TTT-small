def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_obj(a):
    """Fast computation of the objective (assuming sum(a)=1)."""
    return 2.0 * len(a) * fast_convolve(a).max()

    a = a.copy()

        conv = fast_convolve(a)
        M = conv.max()
        
        if it % 50 == 0:

        if n_peaks >= len(conv):

        best_move = None

        for k in peak_positions:
            if conv[k] < M * 0.999:
                continue

            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)

                continue

            # contrib[j] = a[j] * a[k-j]
            contrib = a[js] * a[k - js]
            
            # Find the position with the most "excess" mass relative to its contribution
            # We want to move mass from positions where reducing it helps the most
            # without hurting other peaks
            
            # For each source position j, compute the "safety" of moving mass from it
            # A position j is "safe" if it doesn't contribute much to other peaks
            for j_idx in range(min(5, len(js))):  # Try top contributors

                for k2 in peak_positions:

                    js2_end = min(n - 1, k2)
                    for j2 in range(js2_start, js2_end + 1):
                        exposure[j2] = max(exposure[j2], a[k2 - j2])
                
                # Find the safest target (lowest exposure, not the source)
                exposure[j] = np.inf  # Can't move to self
                exposure[k - j] = np.inf  # Avoid the paired position

                if len(safe_targets) == 0:
                    continue
                
                # Pick the safest target
                j_target = safe_targets[np.argmin(exposure[safe_targets])]
                
                # Try different amounts to move
                for frac in [0.01, 0.05, 0.1, 0.2, 0.3]:
                    delta = a[j] * frac

                            best_move = candidate.copy()

        else:
            break

            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
        
        sg = compute_subgradient(a, conv, top_k=5)
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Adaptive learning rate
        lr_t = lr * (1.0 - 0.3 * min(1.0, no_improve / 500.0))
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)
        
        new_M = fast_convolve(a_new).max()
        if new_M <= M + 1e-15:
            a = a_new
        else:
            # Backtrack
            a_new = a - 0.5 * lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a_new = project_simplex(a_new)
            a = a_new
    
    return best_a, best_M

def peak_coordinate_descent(a0, time_limit, n_peaks=3, n_sources=8, n_targets=5):
    """
    Directly reduce the peak by finding mass transfers that decrease max(conv).
    For the peak at k*, find (source, target) pairs where moving mass from source
    to target reduces the peak without increasing other peaks above the current max.
    """
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    rng = np.random.default_rng(12345)
    
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 200:
            # Restart with perturbation
            scale = 0.003 + 0.005 * min(1.0, stall / 400.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            stall = 0
            conv = fast_convolve(a)
            M = conv.max()
            continue
        
        # Find top peaks
        pk = min(n_peaks, len(conv))
        peak_positions = np.argpartition(conv, -pk)[-pk:]
        peak_positions = peak_positions[np.argsort(conv[peak_positions])[::-1]]
        
        # Compute exposure: for each position, how much it contributes to peaks
        exposure = np.zeros(n)
        for k in peak_positions:
            js = range(max(0, k - n + 1), min(n - 1, k) + 1)
            for j in js:
                exposure[j] += a[k - j]
        
        best_improvement = 0
        best_candidate = None
        
        k_star = peak_positions[0]
        js = np.arange(max(0, k_star - n + 1), min(n - 1, k_star) + 1)
        if len(js) == 0:
            break
        contrib = a[js] * a[k_star - js]
        
        # Top sources by contribution
        top_idx = np.argsort(contrib)[::-1][:n_sources]
        top_sources = js[top_idx]
        
        # Safe targets: low exposure positions
        mask = np.ones(n, dtype=bool)
        for k in peak_positions:
            for j in range(max(0, k - n + 1), min(n - 1, k) + 1):
                pass  # Already in exposure
        
        safe_candidates = np.where(exposure < np.percentile(exposure, 60))[0]
        if len(safe_candidates) == 0:
            safe_candidates = np.arange(n)
        safe_sorted = safe_candidates[np.argsort(exposure[safe_candidates])[:n_targets * 2]]
        
        for src in top_sources:
            if a[src] < 1e-10:
                continue
            pair = k_star - src
            for tgt in safe_sorted:
                if tgt == src or tgt == pair:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2, 0.4]:
                    delta = a[src] * frac
                    candidate = a.copy()
                    candidate[src] -= delta
                    candidate[tgt] += delta
                    if candidate[src] < -1e-15 or candidate[tgt] < -1e-15:
                        continue
                    candidate = np.maximum(candidate, 0)
                    s = candidate.sum()
                    if s < 1e-15:
                        continue
                    candidate /= s
                    new_M = fast_convolve(candidate).max()
                    if new_M < M - 1e-14:
                        imp = M - new_M
                        if imp > best_improvement:
                            best_improvement = imp
                            best_candidate = candidate.copy()
        
        if best_candidate is not None:
            a = best_candidate
        else:
            # Small random nudge
            a = project_simplex(a + rng.standard_normal(n) * 0.002)
    
    return best_a, best_M

def solve_lp_oracle(f, M, method='highs'):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method,
                      options={'presolve': True, 'max_time': 20})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def fw_lp_optimize(f0, time_limit, m_tolerance=1.0):
    """Frank-Wolfe with LP oracle."""
    f = f0.copy()
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    m_exp = 1.0
    no_improve = 0
    
    t0 = time.monotonic()
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        g = solve_lp_oracle(f, M * m_exp)
        if g is None:
            m_exp = min(m_exp * 1.5, 5.0)
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 20:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Bisection refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.02)
            hi = min(1.0, best_alpha + 0.02)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            m_exp = 1.0
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)
        
        if no_improve > 40:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.003 + 0.004 * min(1.0, no_improve / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj

def pgd_momentum(a0, time_limit, top_k=15, lr_init=0.03):
    """Projected gradient descent with momentum."""
    a = a0.copy()
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    v = np.zeros(n)
    lr = lr_init
    mom = 0.9
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        sg = compute_subgradient(a, conv, top_k=top_k)
        
        v = mom * v - lr * sg
        best_candidate = a
        best_val = M
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            candidate = project_simplex(a + scale * v)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break
        
        a = best_candidate
        
        if best_val < best_M - 1e-15:
            best_M = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.9
        
        if no_improve > 50:
            lr *= 0.95
            mom = min(0.97, mom + 0.003)
        if no_improve > 300:
            rng = np.random.default_rng((it * 48271 + 17) % (2**31 - 1))
            noise = 0.002 + 0.004 * min(1.0, (no_improve - 300) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init
            mom = 0.9
            no_improve = 0
    
    return best_a, best_M

def active_set_refine(a0, time_limit, tol=1e-13):
    """
    KKT-based refinement: at optimum, for support S and active peaks P:
    - sum_{k in P} mu_k * a[k-j] = nu for j in S
    - (a*a)[k] = t for k in P
    - sum(a) = 1, a >= 0
    Use SVD to find dual variables and construct gradient direction.
    """
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    while time.monotonic() - t0 < time_limit and it < 5000:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
        
        # Find active peaks (within 0.05% of max)
        threshold = M * (1.0 - 0.0005)
        active_peaks = np.where(conv >= threshold)[0]
        
        if len(active_peaks) <= 1:
            # Can't improve with single peak
            break
        
        support = np.where(a > 1e-10)[0]
        if len(support) == 0:
            break
        
        m = len(support)
        p = len(active_peaks)
        if m == 0 or p == 0:
            break
        
        # Build system: for j in support, sum_{k in P} mu_k * a[k-j] = nu
        A_sys = np.zeros((m, p + 1))
        for i, j in enumerate(support):
            A_sys[i, -1] = 1.0
            for k_idx, k in enumerate(active_peaks):
                idx = k - j
                if 0 <= idx < n:
                    A_sys[i, k_idx] = a[idx]
        
        try:
            U, S_vals, Vt = np.linalg.svd(A_sys)
            sol = Vt[-1, :]
            mu_vals = np.maximum(sol[:p], 0)
            mu_sum = mu_vals.sum()
            if mu_sum < 1e-15:
                break
            mu_vals /= mu_sum
        except np.linalg.LinAlgError:
            break
        
        # Gradient direction
        grad = np.zeros(n)
        for k_idx, k in enumerate(active_peaks):
            if mu_vals[k_idx] < 1e-15:
                continue
            js = range(max(0, k - n + 1), min(n - 1, k) + 1)
            for j in js:
                grad[j] -= 2.0 * mu_vals[k_idx] * a[k - j]
        
        # Step
        best_candidate = a
        best_val = M
        for step in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
            candidate = project_simplex(a - step * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                break
        
        a = best_candidate
        
        if abs(M - best_val) < 1e-16:
            break
    
    return best_a, best_M

def interp_sequence(parent, n_target):
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s

def symmetrize(a):
    s = (a + a[::-1]) / 2.0
    s = s / s.sum()
    return s

def get_structured_init(n, style='hann'):
    if style == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1))) if n > 1 else np.ones(1)
    elif style == 'fejer':
        a = 1.0 - np.abs(np.arange(n) - (n - 1) / 2.0) / ((n - 1) / 2.0)
    elif style == 'gaussian':
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    elif style == 'bump':
        a = np.sin(np.pi * np.arange(n) / n) ** 2
    elif style == 'uniform':
        a = np.ones(n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def try_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return best_seq, best_score

        t_alloc = min(remaining, 80)
        a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc, n_peaks=4, n_sources=12, n_targets=8)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "peak_cd_final")
    
    # Adam fine-tuning
    remaining = deadline - time.monotonic() - 60
    if remaining > 10:
        t_alloc = min(remaining, 50)
        a_opt, _ = adam_optimize(best_seq, time_limit=t_alloc, lr=0.1)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "adam_final")
    
    # Active set again
    remaining = deadline - time.monotonic() - 30
    if remaining > 10:
        a_opt, _ = active_set_refine(best_seq, time_limit=min(remaining, 30))
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "active_set_final")
    
    print(f"Phase 5 done: best={best_score:.8f}, elapsed={time.monotonic()-t_phase5:.1f}s")
    
    # Phase 6: Final polish
    print("\n=== Phase 6: Final Polish ===")
    t_phase6 = time.monotonic()
    current_n = len(best_seq)
    
    # Symmetry
    sym_final = symmetrize(best_seq)
    best_seq, best_score = try_and_update(sym_final, best_seq, best_score, "final_sym")
    
    # Tiny perturbations
    for _ in range(8):
        remaining = deadline - time.monotonic() - 15
        if remaining < 8:
            break
        noise = rng.standard_normal(current_n) * 0.001
        f_pert = project_simplex(best_seq + noise)
        t_alloc = min(remaining, 12)
        a_opt, _ = pgd_momentum(f_pert, time_limit=t_alloc, lr_init=0.01)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "perturb")
        
        remaining = deadline - time.monotonic() - 10
        if remaining > 5:
            a_opt2, _ = peak_coordinate_descent(a_opt, time_limit=min(remaining, 8), n_peaks=3, n_sources=5, n_targets=3)
            best_seq, best_score = try_and_update(a_opt2, best_seq, best_score, "perturb_cd")
    
    # One more FW if time
    remaining = deadline - time.monotonic() - 10
    if remaining > 15 and current_n <= 1024:
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=min(remaining, 20))
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "final_fw")
    
    print(f"Phase 6 done: best={best_score:.8f}, elapsed={time.monotonic()-t_phase6:.1f}s")
    
    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Ensure validity
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = best_seq[:min(len(best_seq), 5000)]
    s = best_seq.sum()
    if s < 1e-10:
        best_seq = np.ones(512) / 512
    
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
