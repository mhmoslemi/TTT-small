- Terms where j = j0: a[j0]*a[k-j0] changes from v*a[k-j0] to (v+δ)*a[k-j0], change = δ*a[k-j0]
- Terms where k-j = j0 (i.e., j = k-j0): a[j0]*a[k-j0] is the same term! So we only count it once.

Wait, in the sum sum_j a[j]*a[k-j], when j = j0 we get a[j0]*a[k-j0]. When j = k-j0, we get a[k-j0]*a[j0]. These are the same value, so if j0 != k-j0, the total contribution is 2*a[j0]*a[k-j0]. If j0 = k-j0 (k = 2*j0), the contribution is a[j0]^2.

So:
- If k-j0 is in [0,n-1] and k-j0 != j0: conv[k] includes 2*a[j0]*a[k-j0]. Change: 2*δ*a[k-j0]
- If k = 2*j0: conv[k] includes a[j0]^2. Change: 2*v*δ + δ^2
- Otherwise: no contribution from a[j0]

So:
d_k = 2*a[k-j0] if (0 <= k-j0 < n and k-j0 != j0)
     = 2*v if (k == 2*j0)  [for the linear part]
     = 0 otherwise

And the quadratic part: +δ^2 if k == 2*j0.

So conv_new[k] = conv_old[k] + d_k * δ + (δ^2 if k == 2*j0 else 0)

And sum_new = S + δ

Objective: f(δ) = 2n * max_k (conv_old[k] + d_k*δ + (δ^2 if k==2*j0 else 0)) / (S+δ)^2

We want to minimize this over δ such that v + δ >= 0 (i.e., δ >= -v) and S + δ > 0.

This is a 1D optimization problem. The function is piecewise (the max switches between different k values at certain δ values). We can find the optimum by:
1. Identifying all breakpoints (where two conv values are equal)
2. Evaluating f at each breakpoint and at the boundaries

But there are 2n-1 pieces, and finding all breakpoints could be O(n^2) which is fine for n~768.

Actually, a simpler approach: since we just need to find the minimum, we can use a numerical 1D optimizer (like golden section or scipy.minimize_scalar) on f(δ).

But f is not smooth (it's the max of smooth functions), so we need to be careful. We can use a subgradient method or just do a grid search + refinement.

Actually, for the 1D problem, we can just evaluate f at many points and take the best. Since it's 1D, this is cheap.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def obj_normalized(a):
    """Compute 2n*max(conv)/(sum)^2 for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()

def compute_coord_jacobian(a, j0):
    """
    For coordinate j0, compute d_k such that:
    conv_new[k] = conv_old[k] + d_k * delta + (delta^2 if k == 2*j0 else 0)
    """
    n = len(a)
    conv = fast_convolve(a)
    d = np.zeros(2 * n - 1)
    v = a[j0]
    for k in range(2 * n - 1):
        partner = k - j0
        if 0 <= partner < n:
            if partner == j0:
                d[k] = 2.0 * v  # linear part of (v+delta)^2
            else:
                d[k] = 2.0 * a[partner]
    return conv, d

def coord_optimize(a, j0, num_trials=15):
    """
    Exact 1D coordinate descent for coordinate j0.
    Tries several values and picks the best.
    """
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return a
    
    conv, d = compute_coord_jacobian(a, j0)
    v = a[j0]
    k_quad = 2 * j0  # the lag where the quadratic term appears
    
    best_obj = 2.0 * n * conv.max() / (S * S)
    best_delta = 0.0
    
    # Try various deltas
    # delta ranges from -v (set to 0) to several times v
    if v > 1e-15:
        trials = np.concatenate([
            [-v],  # set to zero
            np.linspace(-v, v, 7),  # from 0 to 2v
            np.array([1.5*v, 2*v, 3*v, 5*v, 10*v])
        ])
    else:
        trials = np.array([1.0/S, 2.0/S, 5.0/S])  # small positive values
    
    for delta in trials:
        new_v = v + delta
        if new_v < -1e-15:
            continue
        new_v = max(0.0, new_v)
        delta = new_v - v  # adjust for clamping
        
        # Update conv
        conv_new = conv + d * delta
        if k_quad < len(conv_new):
            conv_new[k_quad] += delta * delta
        
        S_new = S + delta
        if S_new < 1e-15:
            continue
        
        obj = 2.0 * n * conv_new.max() / (S_new * S_new)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_delta = delta
    
    if abs(best_delta) > 1e-15:
        a = a.copy()
        a[j0] = max(0.0, a[j0] + best_delta)
    
    return a

def coord_descent(a0, time_limit, order='cyclic'):
    """Coordinate descent with 1D exact optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a)) / len(a)
        S = 1.0
    n = len(a)
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max() / (S * S)
    
    t0 = time.monotonic()
    stall = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        improved = False
        
        if order == 'cyclic':
            coords = range(n)
        elif order == 'greedy':
            # Try coordinates in order of current value (larger first)
            coords = np.argsort(a)[::-1]
        else:  # random
            rng = np.random.default_rng(it)
            coords = rng.permutation(n)
        
        for j0 in coords:
            if time.monotonic() - t0 > time_limit - 0.05:
                break
            
            a_new = coord_optimize(a, j0)
            if not np.array_equal(a_new, a):
                S_new = a_new.sum()
                if S_new > 1e-15:
                    score = 2.0 * n * fast_convolve(a_new).max() / (S_new * S_new)
                    if score < best_score - 1e-15:
                        best_score = score
                        best_a = a_new.copy()
                    a = a_new
                    improved = True
        
        if improved:
            stall = 0
        else:
            stall += 1
        
        if stall > 50:
            # Perturb and restart
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            noise = rng.standard_normal(n) * 0.001
            a = np.maximum(best_a + noise, 0.0)
            a = a / a.sum()
            stall = 0
        
        if stall > 200:
            break
    
    return best_a, best_score

def nesterov_smoothed(a0, time_limit, T_init=0.001, lr_init=0.05):
    """Nesterov-accelerated smoothed gradient descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    y = a.copy()
    v = np.zeros(n)
    t = 1.0
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    no_improve = 0
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Nesterov momentum
        alpha = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / alpha
        y = (1.0 + beta) * a - beta * y
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y / ys
        t += 1.0
        
        # Compute gradient at y
        conv_y = fast_convolve(y)
        num_peaks = max(5, int(0.2 * len(conv_y)))
        top_idx = np.argpartition(conv_y, -num_peaks)[-num_peaks:]
        
        scaled = conv_y[top_idx] / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        grad = np.zeros(n)
        for idx, k in enumerate(top_idx):
            w = 2.0 * p[idx]
            js = max(0, int(k) - n + 1)
            je = min(n - 1, int(k))
            for j in range(js, je + 1):
                grad[j] += w * y[int(k) - j]
        
        # Update
        v = beta * v - lr * grad
        a_new = project_simplex(a + v)
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            a_new = a  # revert
        
        a = a_new
        
        # Adaptive T and lr
        if no_improve > 20:
            T *= 1.05
            lr *= 0.95
        if no_improve > 80:
            T *= 0.7
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            y = a.copy()
            v = np.zeros(n)
            t = 1.0
            T = T_init
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj

def pair_perturb(a0, time_limit):
    """
    Perturb pairs of coordinates that jointly affect the top peaks.
    For each top peak, find the pair of coordinates with maximum joint contribution
    and try to reduce them.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    stall = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        # Top 3 peaks
        top_k = min(3, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peaks = peaks[np.argsort(conv[peaks])[::-1]]
        
        improved = False
        
        for k_peak in peaks:
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            
            kp = int(k_peak)
            js = max(0, kp - n + 1)
            je = min(n - 1, kp)
            affected = np.arange(js, je + 1)
            
            if len(affected) < 2:
                continue
            
            # Find the pair with maximum product a[j1]*a[j2] where j1+j2=kp
            # (or more precisely, the pair that contributes most to this peak)
            # The contribution of pair (j1, j2) to conv[kp] is:
            #   a[j1]*a[kp-j1] if j2 = kp-j1 (and j1 != j2)
            #   a[j1]^2 if j1 = kp/2
            
            # So we want to find j that maximizes a[j]*a[kp-j]
            products = []
            for j in affected:

                    products.append((2.0 * a[j] * a[partner], j, partner))
                elif 0 <= partner < n and partner == j:
                    products.append((a[j] ** 2, j, j))
            
            if not products:
                continue
            
            products.sort(reverse=True)
            top_pairs = products[:5]
            
            for contrib, j1, j2 in top_pairs:
                if contrib < 1e-15:
                    continue
                
                # Try to reduce this pair
                for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                    a_test = a.copy()
                    
                    if j1 == j2:
                        a_test[j1] *= (1.0 - frac)
                        # Redistribute to random other positions
                        freed = a_test[j1] was wrong...
                    
                    # Actually, let's just reduce both and redistribute
                    reduce_amt = frac * min(a[j1], a[j2]) if j1 != j2 else frac * a[j1]
                    
                    a_test = a.copy()
                    if j1 != j2:
                        a_test[j1] -= frac * a[j1]
                        a_test[j2] -= frac * a[j2]
                    else:
                        a_test[j1] -= frac * a[j1]
                    
                    # Redistribute to random positions
                    freed = (frac * a[j1] + (frac * a[j2] if j1 != j2 else 0))
                    rng = np.random.default_rng((it * 1000 + kp * 10 + int(j1)) % (2**31))
                    targets = rng.choice(n, size=min(3, n), replace=False)
                    for ti in targets:
                        a_test[ti] += freed / len(targets)
                    
                    a_test = np.maximum(a_test, 0.0)
                    s = a_test.sum()
                    if s < 1e-15:
                        continue
                    a_test = a_test / s
                    c_new = fast_convolve(a_test)
                    if c_new.max() < best_obj - 1e-14:
                        a = a_test
                        best_obj = c_new.max()
                        best_a = a.copy()
                        improved = True
                        break
            
            if improved:
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                rng = np.random.default_rng((it * 48271 + 1) % (2**31))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                stall = 0
            if stall > 150:
                break
    
    return best_a, best_obj
