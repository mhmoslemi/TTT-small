Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-smoothed gradient descent with Nesterov momentum
2. Exact coordinate descent (1D subproblem per coordinate)
3. Random frequency-domain perturbations for diversification
4. Multi-length exploration
5. Mirrored coordinate descent (symmetric pairs)

import numpy as np
from scipy.optimize import minimize_scalar
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_obj_normalized(a):
    """Compute 2*n*max(conv)/(sum)^2 assuming a is non-negative with sum >= 0.01."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_smooth_max(conv, beta):
    """Compute LSE approximation of max and its gradient."""
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        return (1.0/beta) * (bc_max / beta + 0), np.ones(len(conv)) / len(conv)
    lse = (1.0/beta) * (bc_max + np.log(total))
    weights = exp_bc / total
    return lse, weights

def gradient_lse(a, beta):
    """Compute gradient of LSE-smoothed max of auto-convolution."""
    n = len(a)
    conv = fast_convolve(a)
    lse_val, weights = lse_smooth_max(conv, beta)
    
    # Gradient of LSE wrt a[j] = (2/beta) * sum_k weights[k] * a[k-j] (for valid k-j)
    # More precisely: d LSE / d a[j] = sum_k weights[k] * d c[k] / d a[j]
    # d c[k] / d a[j] = 2 * a[k-j] (if 0 <= k-j < n)
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi+1] += w * 2.0 * a[k - np.arange(lo, hi+1)]
    
    return lse_val, grad

def nesterov_gradient_descent(a0, time_limit, beta_start=20.0, beta_end=5000.0, verbose=False):
    """Nesterov accelerated gradient descent with LSE smoothing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    
    beta = beta_start
    lr = 2.0 / (n ** 0.5)
    momentum = 0.9
    
    y = a.copy()  # Nesterov's lookahead point
    v = np.zeros(n)  # velocity
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        # Nesterov step
        _, grad = gradient_lse(y, beta)
        
        # Update velocity
        v = momentum * v - lr * grad
        
        # Project
        y = project_simplex(a + v)
        
        # Evaluate
        obj = compute_obj_normalized(y)
        if obj < best_obj:
            best_obj = obj
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = y
        
        # Adaptive learning rate
        if no_improve > 10:
            lr *= 0.9
        if no_improve > 50:
            # Reset
            a = best_a.copy()
            y = a.copy()
            v = np.zeros(n)
            lr = 2.0 / (n ** 0.5)
            beta = min(beta * 2, beta_end)
            no_improve = 0
        
        # Slowly increase beta
        if beta < beta_end:
            beta *= 1.003
    
    return best_a, best_obj

def coordinate_descent_step(a, j):
    """Solve the 1D subproblem for coordinate j."""
    n = len(a)
    v = a[j]
    S = a.sum()
    S_prime = S - v
    
    conv = fast_convolve(a)
    
    # Objective: F(x) = 2n * max(c_new(x)) / (S_prime + x)^2
    # where c_new(x) is the convolution with a[j] replaced by x
    
    # Determine which conv entries are affected:
    # k ranges from max(0, j) to min(2n-2, j+n-1)
    lo = max(0, j)
    hi = min(2*n-2, j+n-1)
    
    # For k in [lo, hi], k != 2j: c_new[k] = conv[k] + 2*(x-v)*a[k-j]
    # For k = 2j (if in [lo, hi]): c_new[2j] = conv[2j] - v^2 + x^2
    # For k outside [lo, hi]: c_new[k] = conv[k]
    
    two_j = 2 * j
    
    # Precompute for the affected range
    idx = np.arange(lo, hi+1)
    k_minus_j = idx - j  # in [0, min(n-1, 2n-1-lo)]
    
    # Linear coefficients: alpha[k] + beta[k]*x
    alpha = conv[idx].copy()
    beta_coeff = 2.0 * a[k_minus_j].copy()
    alpha -= beta_coeff * v  # alpha[k] = conv[k] - 2*v*a[k-j]
    
    # Handle k = 2j (quadratic)
    if lo <= two_j <= hi:
        pos = two_j - lo
        alpha[pos] = conv[two_j] - v*v
        beta_coeff[pos] = 0.0  # will handle separately
    
    # Unaffected max
    if lo > 0 or hi < 2*n-2:
        unaffected_mask = np.ones(2*n-1, dtype=bool)
        unaffected_mask[lo:hi+1] = False
        unaffected_max = conv[unaffected_mask].max() if unaffected_mask.any() else 0.0
    else:
        unaffected_max = 0.0
    
    def F(x):
        # Compute max of new convolution
        vals = alpha + x * beta_coeff
        # Handle quadratic term at k=2j
        if lo <= two_j <= hi:
            pos = two_j - lo
            vals[pos] = alpha[pos] + x*x
        max_affected = vals.max()
        max_total = max(max_affected, unaffected_max)
        denom = (S_prime + x) ** 2
        if denom < 1e-30:
            return float('inf')
        return max_total / denom
    
    # Find minimum using bounded scalar minimization

    res = minimize_scalar(F, bounds=[0.0, x_upper], method='bounded',

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact coordinate descent (solving 1D subproblems)
2. LSE-smoothed gradient descent with Nesterov momentum for fast initial convergence
3. Frequency-domain random perturbations for diversification
4. Multi-length exploration and interpolation
5. Symmetric coordinate descent (update pairs (j, n-1-j) together)
"""

import numpy as np
from scipy.optimize import minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def coordinate_descent(a0, time_limit, verbose=False):
    """Exact coordinate descent solving 1D subproblems."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_a = a.copy()
    best_obj = 2.0 * n * conv.max()
    
    t0 = time.monotonic()
    pass_num = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_num += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            v = a[j]
            
            # Determine affected convolution range
            lo = j  # k-j >= 0 → k >= j
            hi = min(2 * n - 2, j + n - 1)  # k-j <= n-1 → k <= j+n-1
            two_j = 2 * j
            
            # Precompute coefficients for affected range
            idx = np.arange(lo, hi + 1)
            k_j = idx - j  # valid indices into a, in [0, n-1]
            
            # For k != 2j: conv_new[k] = conv[k] + 2*(x-v)*a[k-j]
            #     = (conv[k] - 2*v*a[k-j]) + x*(2*a[k-j])
            # For k = 2j: conv_new[2j] = conv[2j] - v^2 + x^2
            alpha = conv[idx] - 2.0 * v * a[k_j]
            beta_c = 2.0 * a[k_j]
            
            has_quad = (two_j >= lo) and (two_j <= hi)
            if has_quad:
                quad_pos = two_j - lo
                alpha[quad_pos] = conv[two_j] - v * v
                beta_c[quad_pos] = 0.0
            
            # Max of unaffected entries
            unaff_max = 0.0
            if lo > 0:
                unaff_max = max(unaff_max, conv[:lo].max())
            if hi < 2 * n - 2:
                unaff_max = max(unaff_max, conv[hi + 1:].max())
            
            S_prime = 1.0 - v  # sum of all coords except j (since sum(a)=1)
            
            # Define the 1D objective
            def F(x):
                vals = alpha + x * beta_c
                if has_quad:
                    vals[quad_pos] = alpha[quad_pos] + x * x
                m = vals.max()
                if m < unaff_max:
                    m = unaff_max
                denom = (S_prime + x) * (S_prime + x)
                if denom < 1e-30:
                    return 1e30
                return m / denom
            
            # Bounds for x
            x_lo = 0.0
            x_hi = max(2.0 * v + 0.01 / n, 0.05 / n)
            
            res = minimize_scalar(F, bounds=(x_lo, x_hi), method='bounded',
                                  options={'maxiter': 25})
            x_opt = res.x
            
            if abs(x_opt - v) > 1e-10:
                a[j] = x_opt
                # Renormalize
                s_new = a.sum()
                a = a / s_new
                conv = fast_convolve(a)
                obj = conv.max()
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
        
        if verbose and pass_num % 5 == 0:
            print(f"  CD pass {pass_num}: obj={best_obj:.10f}")
        
        if not improved:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0
    
    return best_a, 2.0 * n * best_obj


def lse_gradient_step(a, beta, lr):
    """One step of gradient descent on LSE-smoothed objective."""
    n = len(a)
    conv = fast_convolve(a)
    
    # LSE weights
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    weights = exp_bc / total
    
    # Gradient: d/dx_j [LSE] = sum_k weights[k] * 2*a[k-j] (for valid k-j)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += w * 2.0 * a[k - np.arange(lo, hi + 1)]
    
    return grad


def gradient_descent(a0, time_limit, beta_start=20.0, beta_end=3000.0, verbose=False):
    """Gradient descent with LSE smoothing and adaptive step size."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_objective(a)
    
    beta = beta_start
    lr = 1.0 / np.sqrt(n)
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        grad = lse_gradient_step(a, beta, lr)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        step = lr
        improved = False
        for _ in range(15):
            a_trial = project_simplex(a - step * grad)
            obj_trial = compute_objective(a_trial)
            if obj_trial < best_obj - 1e-14:
                best_obj = obj_trial
                best_a = a_trial.copy()
                a = a_trial
                improved = True
                no_improve = 0
                break
            step *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve > 15:
                a = best_a.copy()
                lr *= 0.8
                beta = min(beta * 1.5, beta_end)
                no_improve = 0
            if no_improve > 80:
                break
        
        if beta < beta_end and it % 20 == 0:
            beta *= 1.01
    
    return best_a, best_obj


def freq_perturb(a, rng, strength=0.03):
    """Perturb in frequency domain."""
    n = len(a)
    F = np.fft.rfft(a)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(F_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new
