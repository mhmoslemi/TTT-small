    # Actually for auto-conv: d(conv[k])/d(a[j]) = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    # But a[k-j] and a[j-k] are different in general.
    # For k>=j: if k-j<n, contribution is a[k-j]
    # For k<j: if j-k<n, contribution is a[j-k]  
    # For k==j: contribution is 2*a[0]
    
    # The total gradient is:
    # grad[j] = 2n * (sum_{d=0}^{n-1} p[j+d]*a[d] + sum_{d=1}^{n-1} p[j-d]*a[d])
    # where p[j+d] is only included if j+d < 2n-1, and p[j-d] if j-d >= 0.
    
    # This can be computed as a cross-correlation:
    # First term: sum_d p[j+d]*a[d] = (p reversed) correlated with a... 
    # Actually, let's think of it as:
    # grad[j] / (2n) = sum_d a[d] * p[j+d] + sum_d a[d] * p[j-d] (with valid index checks)
    
    # The first sum: for fixed j, sum_{d=0}^{n-1} a[d] * p[j+d]
    # This is the cross-correlation of a and p at lag j: (a ⋆ p)[j] = sum_d a[d] * p[j+d]
    # The second sum: for fixed j, sum_{d=1}^{n-1} a[d] * p[j-d]
    # This is (a ⋆ p')[j] where p'[k] = p[-k]... 
    
    # Actually, let's just compute it directly for small n, or use FFT for large n.
    # For n up to ~2000, direct computation is O(n^2) which is fine.
    
    # Use FFT-based approach:
    # The first term is: for each j, sum_{d=0}^{min(n-1, 2n-2-j)} a[d] * p[j+d]
    # This is the cross-correlation: np.correlate(p, a, 'valid') gives sum_d p[d+j]*a[d] for j=0,...,n-1... 
    # Actually np.correlate(p, a, 'full') gives all lags. Let me think...
    
    # np.correlate(a, p, 'full')[k] = sum_j a[j] * p[j + k - (len(p)-1)]... this is getting confusing.
    
    # Let me just use the direct approach with numpy:
    # For the gradient, I need:
    # grad[j] = 2n * sum_k p[k] * (a[k-j] if 0<=k-j<n else 0 + a[j-k] if 0<=j-k<n else 0)
    
    # Let me define: for each k, the "gradient contribution" to a[j] is:
    #   a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
    # This is the same as the "subgradient" computation in the previous code.
    
    grad = np.zeros(n)
    # For efficiency, process each k with significant p[k]:
    significant_p = np.where(p > 1e-10)[0]
    for k in significant_p:
        weight = 2.0 * n * p[k]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += weight * a[k - idx]
    
    return grad

def lbfgs_optimize(a0, time_limit=100, T=0.001, maxiter=5000):
    """L-BFGS optimization using numerical gradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    t0 = time.monotonic()
    
    # Use L-BFGS-B with bounds [0, None] and a penalty for the sum constraint
    # Or use the fact that we can normalize after each step
    
    def obj_func(x):
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-15:
            return 1e10
        x = x / s
        return compute_obj(x)
    
    def grad_func(x):
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-15:
            return np.zeros(len(x))
        x = x / s
        return compute_grad_pdp(x, T)
    
    # Use L-BFGS-B
    try:
        res = minimize(
            obj_func, a.copy(), method='L-BFGS-B',
            jac=grad_func,
            bounds=[(0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': 10000, 'ftol': 1e-15, 'gtol': 1e-12}
        )
        if res.fun < best_obj:
            a_res = np.maximum(res.x, 0)
            a_res = a_res / a_res.sum()
            best_obj = compute_obj(a_res)
            best_a = a_res
    except Exception as e:
        pass
    
    return best_a, best_obj

def pgd_smoothed(a0, time_limit=100, T=0.001, lr_init=0.01, mom=0.9):
    """Projected gradient descent with momentum on smoothed objective."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        grad = compute_grad_pdp(a, T)
        
        # Momentum step
        v = mom * v - lr * grad
        
        # Backtracking line search
        best_candidate = a
        best_val = best_obj
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            candidate = project_simplex(a + scale * v)
            val = compute_obj(candidate)
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                break
        
        a = best_candidate
        
        if best_val < best_obj - 1e-14:
            best_obj = best_val
