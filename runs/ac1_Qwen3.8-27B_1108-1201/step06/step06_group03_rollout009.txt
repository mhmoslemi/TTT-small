"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-domain optimization
with adaptive log-sum-exp smoothing and multi-phase search.

Key ideas:
1. Parameterize a = softmax(θ) to handle non-negativity and normalization
2. Use log-sum-exp smoothing of max autocorrelation for smooth gradients
3. Adaptive temperature scheduling (start smooth, get sharper)
4. FFT-based gradient computation in O(n log n)
5. Multi-phase search: exploration → exploitation → polishing
"""

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax_obj_and_grad(theta, tau):
    """Compute smoothed max autocorrelation and its gradient w.r.t. theta."""
    n = len(theta)
    
    # Softmax
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    a = exp_theta / exp_theta.sum()
    
    # Autocorrelation
    R = fast_convolve(a)
    
    # Log-sum-exp with weights
    R_shifted = (R - R.max()) / tau
    exp_R = np.exp(R_shifted)
    w = exp_R / exp_R.sum()
    
    # Objective (smoothed max)
    obj = tau * (R.max() / tau + np.log(exp_R.sum()))  # = R.max() + tau * log(sum(exp(R/tau - R.max/tau)))
    # Actually: LSE = max(R) + tau * log(sum(exp((R - max(R))/tau)))
    obj = R.max() + tau * np.log(exp_R.sum())
    
    # R_avg = sum_k w[k] * R[k]
    R_avg = np.dot(w, R)
    
    # cross[m] = sum_k w[k] * a[k-m] = sum_n a[n] * w[n+m]
    # Using FFT: ifft(conj(FFT(a)) * FFT(w))
    N = 1 << (3 * n - 2 - 1).bit_length()
    A_ft = np.fft.fft(a, N)
    W_ft = np.fft.fft(w, N)
    cross = np.fft.ifft(np.conj(A_ft) * W_ft).real[:n]
    
    # Gradient: g[m] = 2*a[m] * (cross[m] - R_avg)
    grad = 2.0 * a * (cross - R_avg)
    
    return obj, grad

def adam_optimize(theta0, tau_schedule, time_limit, verbose=False):
    """Optimize theta using Adam with adaptive temperature."""
    n = len(theta0)
    theta = theta0.copy()
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    best_obj = float('inf')
    best_theta = theta.copy()
    
    t0 = time.time()
    
    while time.time() - t0 < time_limit:
        t += 1
        
        # Get tau from schedule
        if callable(tau_schedule):
            tau = tau_schedule(t)
        else:
            tau = tau_schedule[min(t, len(tau_schedule)-1)]
        
        obj, grad = softmax_obj_and_grad(theta, tau)
        
        if obj < best_obj:
            best_obj = obj
            best_theta = theta.copy()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        lr = 0.01 / (1 + 0.01 * t)
        theta -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if verbose and t % 1000 == 0:
            print(f"    Adam t={t}, tau={tau:.6f}, obj={best_obj:.10f}")
        
        # Check if we can afford another iteration
        if time.time() - t0 > time_limit - 0.01:
            break
    
    return best_theta, best_obj

def lbfqs_optimize(theta0, tau, time_limit, verbose=False):
    """Optimize theta using L-BFGS-B with fixed tau."""
    n = len(theta0)
    
    def objective(theta):
        obj, grad = softmax_obj_and_grad(theta, tau)
        return obj, grad
    
    # Simple maxfees based on time
    start = time.time()
    
    # Use a wrapper that stops when time is up
    results = []
    def callback(theta_k):
        results.append((time.time() - start, theta_k.copy()))
        return (time.time() - start) > time_limit - 1
    
    try:
        res = minimize(objective, theta0, method='L-BFGS-B', 
                      jac=True, callback=callback,
                      options={'maxiter': 100000, 'maxls': 50})
        if res.x is not None:
            obj, _ = softmax_obj_and_grad(res.x, tau)
            return res.x, obj
    except Exception as e:
        pass
    
    # Fallback: return best from callback
    if results:
        best_t = min(results, key=lambda x: softmax_obj_and_grad(x[1], tau)[0])
        return best_t[1], softmax_obj_and_grad(best_t[1], tau)[0]
    
    return theta0, softmax_obj_and_grad(theta0, tau)[0]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def peak_targeted_refinement(a, time_limit):
    """Direct peak reduction by adjusting coordinates."""
    a = a.copy()
    a = a / a.sum()
    n = len(a)
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.time()

        R = fast_convolve(a)
        peak_k = R.argmax()
        peak_val = R[peak_k]
        
        # Find contributing coordinates
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        
        # Contribution of each pair
        contributions = np.array([a[j] * a[peak_k - j] for j in range(js, je + 1)])
        
        if contributions.max() < 1e-15:
            break
        
        # Try reducing the largest contributor
        top_idx = js + contributions.argmax()
        
        # Shift mass from top_idx to a random distant location
        improved = False
        for _ in range(50):
            target = np.random.randint(0, n)
            if target == top_idx or abs(target - top_idx) < n // 10:
                continue

                delta = a[top_idx] * frac
                if delta < 1e-10:
                    continue
                candidate = a.copy()
                candidate[top_idx] -= delta
                candidate[target] += delta
                candidate = np.maximum(candidate, 0)
                cand_sum = candidate.sum()
                if cand_sum < 1e-15:

                    a = candidate
                    best_obj = new_peak
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

            step *= 0.9
            if step < 1e-8:
                break
    
    return best_a, best_obj

def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s

**Idea: Use the fact that the optimal sequence should have a nearly flat autocorrelation**

If the autocorrelation is perfectly flat (all values equal), then `(a*a)[k] = C` for all k. Since `sum_k (a*a)[k] = (sum a)^2 = 1` (after normalization), we'd have `(2n-1)*C = 1`, so `C = 1/(2n-1)`. The score would be `2n/(2n-1) ≈ 2` for large n. Wait, that's worse than what we have!
