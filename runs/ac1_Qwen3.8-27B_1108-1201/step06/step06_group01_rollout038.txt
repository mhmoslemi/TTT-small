This looks good. Let me now think about the **LP kick** mechanism.

The LP oracle gives a direction `g` that maximizes `sum(g)` subject to `conv(f, g)[k] <= M` for all k. This is like finding the "widest" distribution in the set of distributions whose cross-convolution with `f` is bounded.

The idea is: if the current `f` has a high peak, the LP will find a `g` that "spreads out" the mass to reduce the peak when convolved with `f`. Then we interpolate `f -> (1-t)*f + t*g`.

For the LP to be fast, I'll use the symmetry: if `f` is (approximately) palindromic, the constraints `conv(f,g)[k] <= M` and `conv(f,g)[2n-2-k] <= M` are similar. But `g` is not constrained to be palindromic, so we can't directly reduce the LP size.

However, we CAN note that `conv(f,g)[k] = sum_i f_i * g_{k-i}`. If `f` is palindromic, then `conv(f,g)[k] = conv(f, reverse(g))[2n-2-k]`. So `max(conv(f,g)[k]) = max(conv(f,g)[k] for k in range(n))` (we only need to check the first n lags). This halves the LP constraints.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated smoothed
gradient descent with adaptive temperature, momentum, LP-based escape kicks,
and multi-n exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_fft(a, p, T):
    """Compute gradient of smoothed objective using FFT cross-correlation."""
    n = len(a)
    N = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    # cross[i] = sum_j a[j] * p[j+i]
    # corr = ifft(conj(fft(a)) * fft(p))
    corr = np.fft.irfft(np.conj(A) * P, N)[:2 * n - 1]
    return (2.0 / T) * corr[:n]


def smoothed_grad_optimize(a0, time_limit, use_symmetry=False, T_init=None, lr_init=0.005, momentum_init=0.85):
    """
    Smoothed gradient descent with adaptive temperature and momentum.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    if T_init is None:
        T = max(conv.std() / 3.0, conv.max() * 0.005)
    else:
        T = T_init
    
    lr = lr_init
    momentum = momentum_init
    v = np.zeros(n)
    
    best_a = a.copy()
    best_obj = conv.max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Compute softmax distribution over peaks
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.ones(len(conv)) / len(conv)
        else:
            p = p / p_sum
        
        # Compute gradient
        grad = compute_grad_fft(a, p, T)
        
        # Momentum update
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = float(conv_new.max())
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # If no improvement, revert
            a_new = a
        
        a = a_new
        
        # Adaptive parameters
        if no_improve > 30:
            lr *= 0.95
        if no_improve > 80:
            lr *= 0.9
            momentum *= 0.95
        
        # Adapt temperature to current spread
        conv_std = float(conv.std())
        T_target = max(conv_std / 3.0, conv.max() * 0.001)
        T = 0.97 * T + 0.03 * T_target

            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            no_improve = 0

def solve_lp_oracle(f, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: conv(f,g)[k] = sum_i f_i * g_{k-i} <= M
    # For k = 0, ..., 2n-2
    # Row k: g_j appears with coefficient f_{k-j} if 0 <= k-j < n
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g_j appears in conv[f,g][k] for k = j, j+1, ..., j+n-1
        # with coefficient f[k-j]
        for k in range(j, min(2 * n - 1, j + n)):
            A_ub[k, j] = f[k - j]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            return g
    except Exception:
        pass
    return None


def fw_step(f, time_limit=60):
    """One Frank-Wolfe step with LP oracle."""
    a = np.maximum(f.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = float(conv.max())
    
    # Try with current M, then relaxed
    for m_mult in [1.0, 1.1, 1.3, 1.5]:
        g = solve_lp_oracle(a, M * m_mult, time_limit=min(30, time_limit / 4))
        if g is not None and g.sum() > 1e-10:
            g = g / g.sum()
            # Line search
            best_alpha = 0.0
            best_val = M
            for alpha in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
                ft = (1.0 - alpha) * a + alpha * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = float(fast_convolve(ft).max())
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
            
            if best_alpha > 1e-6:
                result = (1.0 - best_alpha) * a + best_alpha * g
                result = np.maximum(result, 0.0)
                result = result / result.sum()
                return result
        if g is not None:
            break
    
    return a.copy()


def gradient_with_lp_kicks(a0, time_limit, lp_interval=60, grad_time=50):

    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 5:
            break
        
        # Gradient phase
        t_grad = min(grad_time, remaining * 0.6)
        a, obj = smoothed_grad_optimize(a, time_limit=t_grad)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if time.monotonic() - t0 > time_limit - 30:
            break
        
        # LP kick phase
        t_lp = min(lp_interval, remaining * 0.4, 40)
        if t_lp > 5:
            a_kicked = fw_step(best_a, time_limit=t_lp)
            obj_kicked = float(fast_convolve(a_kicked).max())
            if obj_kicked < best_obj:
                best_obj = obj_kicked
                best_a = a_kicked.copy()
            a = best_a.copy()
    
    return best_a, best_obj

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_fft(a, p, T):
    """Gradient of smoothed objective T*log(sum exp(conv/T)) via FFT cross-correlation."""
    n = len(a)
    N = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    corr = np.fft.irfft(np.conj(A) * P, N)[:2 * n - 1]
    return (2.0 / T) * corr[:n]

    return best_a, best_obj


def solve_lp_oracle(f, M, time_limit=20):
    """LP: max sum(g) s.t. conv(f,g)[k] <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        A_ub[k_start:k_end, j] = f[0:k_end - k_start]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def fw_step(f, time_limit=60):
    """One Frank-Wolfe step with LP oracle and line search."""
    a = np.maximum(f.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = float(conv.max())

    for m_mult in [1.0, 1.1, 1.3, 1.6]:
        g = solve_lp_oracle(a, M * m_mult, time_limit=max(10, time_limit // 4))
        if g is not None and g.sum() > 1e-10:
            g = g / g.sum()
            if np.max(np.abs(g - a)) < 1e-10:
                continue
            best_alpha = 0.0
            best_val = M
            for alpha in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
                ft = (1.0 - alpha) * a + alpha * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = float(fast_convolve(ft).max())
                if val < best_val:
                    best_val = val
                    best_alpha = alpha

            if best_alpha > 1e-6:
                result = (1.0 - best_alpha) * a + best_alpha * g
                result = np.maximum(result, 0.0)
                result = result / result.sum()
                return result
        if g is not None:
            break

    return a.copy()


def gradient_with_lp_kicks(a0, time_limit, grad_time=50, lp_time=30):
    """Combined smoothed gradient with periodic LP kicks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 10:
            break

        t_grad = min(grad_time, remaining * 0.65)
        a, obj = smoothed_grad_optimize(a, time_limit=t_grad, lr_init=0.005)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        if time.monotonic() - t0 > time_limit - 15:
            break

        t_lp = min(lp_time, remaining * 0.35, 35)
        if t_lp > 8:
            a_kicked = fw_step(best_a, time_limit=t_lp)
            obj_kicked = float(fast_convolve(a_kicked).max())
            if obj_kicked < best_obj:
                best_obj = obj_kicked
                best_a = a_kicked.copy()
            a = best_a.copy()

    return best_a, best_obj


def coordinate_polish(a0, time_limit):
    """Targeted coordinate descent on top peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())

    t0 = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        top_k = min(3, len(conv))
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        improved = False

        for k_peak in peak_indices:
            if time.monotonic() - t0 > time_limit - 1:
                break

            k_int = int(k_peak)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if je - js + 1 < 2:
                continue

            affected = list(range(js, je + 1))
            non_affected = [i for i in range(n) if i < js or i > je]

            if not non_affected:
                continue

            if len(affected) > 15:
                rng_s = np.random.default_rng(it * 13 + k_int)
                affected = list(rng_s.choice(affected, size=15, replace=False))
            if len(non_affected) > 15:
                rng_s = np.random.default_rng(it * 17 + k_int + 5)
                non_affected = list(rng_s.choice(non_affected, size=15, replace=False))

            best_gain = 0.0
            best_move = None

            for src in affected:
                if a[src] < 1e-10:
                    continue
                for dst in non_affected:
                    for frac in [0.03, 0.08, 0.15, 0.25]:
                        delta = a[src] * frac
                        if delta < 1e-10:
                            continue
                        a_test = a.copy()
                        a_test[src] -= delta
                        a_test[dst] += delta
                        a_test = np.maximum(a_test, 0.0)
                        st = a_test.sum()
                        if st < 1e-15:
                            continue
                        a_test = a_test / st
                        c_new = fast_convolve(a_test)
                        new_max = float(c_new.max())
                        gain = best_obj - new_max
                        if gain > best_gain:
                            best_gain = gain
                            best_move = (src, dst, frac)

            if best_move is not None and best_gain > 1e-12:
                src, dst, frac = best_move
                delta = a[src] * frac
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                c_new = fast_convolve(a)
                best_obj = float(c_new.max())
                best_a = a.copy()
                improved = True
                stall = 0

        if not improved:
            stall += 1

        if stall > 40:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            stall = 0

        if stall > 250:
            break

    return best_a, best_obj


def micro_polish(a0, time_limit):
    """Very fine perturbation polish."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())

    t0 = time.monotonic()
    it = 0
    scales = [0.0001, 0.0002, 0.0003, 0.0005, 0.0008]
    si = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        scale = scales[si % len(scales)]
        rng = np.random.default_rng((it * 2654435761 + 3) % (2**32))

        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)

        tried = 0
        while tried < 8 and time.monotonic() - t0 < time_limit - 0.1:
            tried += 1
            noise = rng.standard_normal(n) * scale
            for j in range(js, je + 1):
                partner = peak_k - j
                if 0 <= partner < n:
                    noise[j] -= scale * 1.5 * a[partner]

            candidate = project_simplex(a + noise)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = float(c.max())
                best_a = a.copy()
                break

        if it % 200 == 0:
            si += 1

    return best_a, best_obj
