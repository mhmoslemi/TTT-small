def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    if spread < 1e-15:
        return c_max * 0.01 + 1e-10
    # Target: the target_active-th peak should have weight ~ 1/target_active
    # softmax threshold: c[target-1]/T should be about log(target) below max
    threshold = c_sorted[min(target_active - 1, n_peaks - 1)]
    diff = c_max - threshold
    if diff < 1e-15:
        return spread * 0.5 + 1e-10
    T = diff / max(1.0, np.log(target_active))
    return max(T, 1e-6)


def nesterov_weighted_subgrad(a0, time_limit, T_init=0.01, lr_init=0.03,
                               momentum_init=0.85, polish_every=30,
                               restart_every=300, verbose=False):
    """Nesterov-accelerated weighted subgradient with adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = momentum_init
    T = T_init

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Nesterov lookahead
        y = a + mom * v
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = a.copy()
        else:
            y = y / y_sum

        # Compute convolution
        c = fast_convolve(y)
        cmax = c.max()

        # Adaptive temperature
        T = adaptive_temperature(c, target_active=12)

        # Softmax weights
        scaled = c / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum > 1e-30:
            w = w / w_sum
        else:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0

        # Subgradient
        g = compute_subgradient(y, w)

        # Backtracking line search
        best_candidate = y.copy()
        best_val = cmax
        for trial_lr in [lr, lr * 0.7, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            candidate = project_simplex(y - trial_lr * g)
            val = fast_convolve(candidate).max()
            if val < best_val:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        v = a - y

        # Update best
        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Coordinate polish periodically
        if polish_every > 0 and it % polish_every == 0 and no_improve > 3:
            a, best_obj, best_a = coordinate_polish(a, best_obj, best_a, n_iters=3)
            no_improve = 0

        # Adapt learning rate
        lr = lr_init * (momentum_init / (1.0 + 0.01 * it * momentum_init)) ** 0.5
        lr = max(lr, lr_init * 0.01)

        # Random restart
        if no_improve > restart_every:
            rng = np.random.default_rng((it * 2654435761 + n) % (2**31))
            noise_scale = 0.01 + 0.02 * (no_improve - restart_every) / restart_every
            a = project_simplex(best_a + rng.standard_normal(n) * min(noise_scale, 0.05))
            v = np.zeros(n)
            T = T_init
            no_improve = 0

        # Increase momentum slowly
        mom = min(0.995, mom + 0.00002)

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NS it={it}, obj={best_obj:.10f}, lr={lr:.6f}, T={T:.6f}, "
                  f"no_imp={no_improve}, elapsed={elapsed:.0f}s")

    return best_a, best_obj


def coordinate_polish(a, best_obj, best_a, n_iters=5):
    """Peak-directed coordinate polish: move mass from peak contributors to non-contributors."""
    n = len(a)
    a = np.maximum(a, 0.0)
    a = a / a.sum()

    for _ in range(n_iters):
        c = fast_convolve(a)
        cmax = c.max()
        if cmax < best_obj:
            best_obj = cmax
            best_a = a.copy()

        # Find the top few peaks
        k_peak = int(np.argmax(c))

        # Contributing indices for this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = set(range(js, je + 1))
        non_contrib = [j for j in range(n) if j not in contrib]

        if len(contrib) == 0 or len(non_contrib) == 0:
            break

        improved = False
        # Try moving mass from largest contributors
        sorted_contrib = sorted(contrib, key=lambda j: -a[j])
        for idx in sorted_contrib[:5]:
            val = a[idx]
            if val < 1e-10:
                continue
            for frac in [0.1, 0.2, 0.3, 0.5]:
                shift = val * frac
                target = a.copy()
                target[idx] -= shift
                nc_vals = np.array([a[j] for j in non_contrib])
                nc_sum = nc_vals.sum()
                if nc_sum > 1e-15:
                    for k, j in enumerate(non_contrib):
                        target[j] += shift * nc_vals[k] / nc_sum
                else:
                    for j in non_contrib:
                        target[j] += shift / len(non_contrib)

                target = project_simplex(target)
                new_obj = fast_convolve(target).max()
                if new_obj < best_obj - 1e-14:
                    a = target
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            break

    return a, best_obj, best_a


def smooth_logsumexp_grad(a0, time_limit, T_init=0.005, lr_init=0.02,
                          verbose=False):
    """Smoothed gradient descent with adaptive log-sum-exp temperature."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        c = fast_convolve(a)
        cmax = c.max()

        # Adaptive T
        c_mean = c.mean()
        T = max(1e-6, (cmax - c_mean) * 0.5)

        scaled = c / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient
        grad = compute_subgradient(a, p)

        # Line search
        best_candidate = a
        best_val = cmax
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val:
                best_val = val
                best_candidate = candidate

        a = best_candidate

        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.999
        if no_improve > 50:
            lr *= 0.8
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + n) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            lr = lr_init
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    SLSE it={it}, obj={best_obj:.10f}, T={T:.6f}")

    return best_a, best_obj


def peak_targeted_opt(a0, time_limit, verbose=False):
    """Directly target the highest peaks with 2-coordinate swaps."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        c = fast_convolve(a)
        cmax = c.max()
        k_peak = int(np.argmax(c))

        # Find the "other" peak (second highest)
        c_copy = c.copy()
        c_copy[k_peak] = -np.inf
        k_peak2 = int(np.argmax(c_copy))

        # For the main peak, find contributing indices
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = list(range(js, je + 1))

        if len(contrib) < 2:
            break

        improved = False
        # Try: reduce a[i] and increase a[j] for i in contrib, j not in contrib
        js2 = max(0, k_peak2 - n + 1)
        je2 = min(n - 1, k_peak2)
        contrib2 = set(range(js2, je2 + 1))
        non_contrib = [j for j in range(n) if j not in set(contrib) and j not in contrib2]

        if not non_contrib:
            non_contrib = [j for j in range(n) if j not in set(contrib)]

        if not non_contrib:
            break

        # Pick top contributors to reduce
        top_contrib = sorted(contrib, key=lambda j: -a[j])[:3]
        # Pick targets to increase
        rng = np.random.default_rng(it % (2**31))
        targets = rng.choice(non_contrib, size=min(5, len(non_contrib)), replace=False)

        for i in top_contrib:
            for j in targets:
                if i == j:
                    continue
                ai = a[i]
                if ai < 1e-10:
                    continue
                for frac in [0.05, 0.1, 0.15, 0.25]:
                    delta = ai * frac
                    candidate = a.copy()
                    candidate[i] -= delta
                    candidate[j] += delta
                    candidate = np.maximum(candidate, 0)
                    cs = candidate.sum()
                    if cs < 1e-15:
                        continue
                    candidate = candidate / cs
                    new_obj = fast_convolve(candidate).max()
                    if new_obj < best_obj - 1e-14:
                        a = candidate
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stagnant += 1
            if stagnant > 100:
                # Random kick
                rng = np.random.default_rng((it * 7 + n) % (2**31))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                stagnant = 0
        else:
            stagnant = 0

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Random Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 0.1))

    # Edge concentration (mild)
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge / edge.sum())

    # Edge concentration (strong)
    edge2 = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 2
    inits.append(edge2 / edge2.sum())

    # Edge concentration (very strong)
    edge3 = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 3
    inits.append(edge3 / edge3.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g / g.sum())

    # Wide Gaussian
    gw = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits.append(gw / gw.sum())

    # Two peaks
    g2 = (np.exp(-0.5 * ((x - 0.2 * (n - 1)) / (0.08 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.8 * (n - 1)) / (0.08 * n)) ** 2))
    inits.append(g2 / g2.sum())

    # Three peaks
    g3 = (np.exp(-0.5 * ((x - 0.15 * (n - 1)) / (0.07 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.07 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.85 * (n - 1)) / (0.07 * n)) ** 2))
    inits.append(g3 / g3.sum())

    # Cosine window
    c1 = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    c1 = np.maximum(c1, 0.01)
    inits.append(c1 / c1.sum())

    # Hamming-like
    ham = 0.54 - 0.46 * np.cos(2 * np.pi * x / n)
    ham = np.maximum(ham, 0.01)
    inits.append(ham / ham.sum())

    # Blackman-like
    glm = 0.42 - 0.5 * np.cos(2 * np.pi * x / n) + 0.08 * np.cos(4 * np.pi * x / n)
    glm = np.maximum(glm, 0.005)
    inits.append(glm / glm.sum())

    # Flat top
    ft = np.ones(n)
    n_edge = max(1, n // 10)
    ramp = 0.5 * (1 + np.cos(np.pi * np.arange(n_edge) / n_edge))
    ft[:n_edge] = ramp
    ft[-n_edge:] = ramp[::-1]
    inits.append(ft / ft.sum())

    # Sinc-like (truncated)
    sinc = np.sinc(x / n - 0.5)
    sinc = np.maximum(sinc, 0.01)
    inits.append(sinc / sinc.sum())

    # Parent interpolation
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())

    # Parent + small noise
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = px + rng.standard_normal(n) * 0.01
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())

    return inits


def quick_screen(a, time_limit):
    """Quick optimization to screen initializations."""
    n = len(a)
    a_opt, obj = nesterov_weighted_subgrad(a, time_limit=min(time_limit, 30),
                                            T_init=0.01, lr_init=0.02,
                                            momentum_init=0.8, polish_every=20)
    return a_opt, obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: multi-phase hybrid optimization for auto-convolution minimization.
    Uses Nesterov-accelerated weighted subgradient as the primary method,
    with coordinate polish, smoothed gradient, and peak-targeted optimization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384

    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")

    # === Phase 1: Quick n-exploration and init screening (~250s) ===
    print("\n=== Phase 1: Quick exploration ===")
    t1 = time.monotonic()

    n_candidates = sorted(set([
        n_parent,
        max(128, n_parent - 32),
        n_parent + 32,
        n_parent + 64,
        max(128, n_parent // 2),
        256, 384, 512, 768, 1024,
    ]))

    # For the parent n, try different initializations
    print(f"  Screening initializations for n={n_parent}...")
    inits_parent = generate_initializations(n_parent, parent, rng)
    quick_best_a = parent.copy()
    quick_best_obj = evaluate_sequence(parent.tolist())

    for i, init in enumerate(inits_parent[:8]):  # Screen top 8
        remaining = deadline - time.monotonic() - 1200
        if remaining < 20:
            break
        t_alloc = max(5.0, min(remaining / max(1, 8 - i), 25.0))
        a_opt, obj = quick_screen(init, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < quick_best_obj:
            quick_best_obj = score
            quick_best_a = a_opt.copy()
            print(f"    init {i}: score={score:.10f}")

    try_seq(quick_best_a, "screen_parent_n")

    # Try other n values with the parent interpolated
    for n_try in n_candidates:
        if n_try == n_parent:
            continue
        remaining = deadline - time.monotonic() - 1000
        if remaining < 15:
            break
        start = np.interp(np.linspace(0, 1, n_try),
                          np.linspace(0, 1, n_parent), parent)
        start = np.maximum(start, 0)
        start = start / start.sum()
        t_alloc = min(remaining, 20)
        a_opt, _ = quick_screen(start, time_limit=t_alloc)
        try_seq(a_opt, f"screen_n{n_try}")

    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")

    # === Phase 2: Deep Nesterov weighted subgradient (~800s) ===
    print("\n=== Phase 2: Deep Nesterov Subgradient ===")
    t2 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Long run with different settings
        configs = [
            {'T_init': 0.01, 'lr_init': 0.02, 'mom_init': 0.85, 'polish': 30},
            {'T_init': 0.005, 'lr_init': 0.015, 'mom_init': 0.9, 'polish': 50},
            {'T_init': 0.02, 'lr_init': 0.03, 'mom_init': 0.8, 'polish': 20},
            {'T_init': 0.003, 'lr_init': 0.01, 'mom_init': 0.92, 'polish': 40},
        ]

        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 400
            if remaining < 20:
                break
            t_alloc = min(remaining, int(budget_s * 0.15))
            print(f"  NS config {ci} (T={cfg['T_init']}, lr={cfg['lr_init']}), t={t_alloc:.0f}s")
            a_opt, _ = nesterov_weighted_subgrad(
                best_seq, time_limit=t_alloc,
                T_init=cfg['T_init'], lr_init=cfg['lr_init'],
                momentum_init=cfg['mom_init'], polish_every=cfg['polish'])
            try_seq(a_opt, f"ns_deep_{ci}")

    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")

    # === Phase 3: Smoothed gradient + peak targeted (~350s) ===
    print("\n=== Phase 3: Smoothed Gradient & Peak Targeted ===")
    t3 = time.monotonic()

    if best_seq is not None:
        # Smoothed log-sum-exp gradient
        for T_t in [0.003, 0.01, 0.03]:
            remaining = deadline - time.monotonic() - 200
            if remaining < 15:
                break
            t_alloc = min(remaining, 80)
            print(f"  SLSE T={T_t}, t={t_alloc:.0f}s")
            a_opt, _ = smooth_logsumexp_grad(best_seq, time_limit=t_alloc,
                                              T_init=T_t, lr_init=0.015)
            try_seq(a_opt, f"slse_T{T_t}")

        # Peak targeted
        remaining = deadline - time.monotonic() - 150
        if remaining > 15:
            t_alloc = min(remaining, 100)
            print(f"  Peak targeted, t={t_alloc:.0f}s")
            a_opt, _ = peak_targeted_opt(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_target")

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")

    # === Phase 4: Alternated refinement (~200s) ===
    print("\n=== Phase 4: Alternated Refinement ===")
    t4 = time.monotonic()

    if best_seq is not None:
        # Alternate between NS and coordinate polish
        current = best_seq.copy()
        for round_i in range(6):
            remaining = deadline - time.monotonic() - 100
            if remaining < 10:
                break
            t_ns = min(remaining * 0.4, 25)
            t_polish = min(remaining * 0.3, 15)

            a_opt, _ = nesterov_weighted_subgrad(current, time_limit=t_ns,
                                                  T_init=0.005, lr_init=0.008,
                                                  momentum_init=0.95, polish_every=0)
            score = try_seq(a_opt, f"alt_ns_{round_i}")
            if best_seq is not None:
                current = best_seq.copy()

            a_opt2, _ = coordinate_polish(current, fast_convolve(current).max(),
                                           current.copy(), n_iters=20)
            try_seq(a_opt2, f"alt_polish_{round_i}")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # === Phase 5: Final fine-tuning (~150s) ===
    print("\n=== Phase 5: Final Fine-tuning ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Very small step Nesterov
        remaining = deadline - time.monotonic() - 50
        if remaining > 10:
            t_alloc = min(remaining, 60)
            print(f"  Fine NS, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_weighted_subgrad(best_seq, time_limit=t_alloc,
                                                  T_init=0.001, lr_init=0.003,
                                                  momentum_init=0.97, polish_every=20)
            try_seq(a_opt, "fine_ns")

        # Random perturbations + quick refine
        for pi in range(4):
            remaining = deadline - time.monotonic() - 20
            if remaining < 5:
                break
            noise = 0.001 + 0.001 * pi
            perturbed = project_simplex(best_seq + rng.standard_normal(n_best) * noise)
            t_alloc = min(remaining, 20)
            a_opt, _ = nesterov_weighted_subgrad(perturbed, time_limit=t_alloc,
                                                  T_init=0.002, lr_init=0.005,
                                                  momentum_init=0.9, polish_every=15)
            try_seq(a_opt, f"perturb_{pi}")

        # One final coordinate polish
        remaining = deadline - time.monotonic() - 10
        if remaining > 5:
            c = fast_convolve(best_seq)
            a_opt, _, _ = coordinate_polish(best_seq, c.max(), best_seq.copy(), n_iters=30)
            try_seq(a_opt, "final_polish")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # === Phase 6: Try slightly different n on best solution ===
    print("\n=== Phase 6: n-variation on best ===")
    t6 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        for delta in [16, -16, 32, -32, 48]:
            n_try = n_best + delta
            if n_try < 64:
                continue
            remaining = deadline - time.monotonic() - 10
            if remaining < 5:
                break
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_best), best_seq)
            start = np.maximum(start, 0)
            start = start / start.sum()
            t_alloc = min(remaining, 15)
            a_opt, _ = nesterov_weighted_subgrad(start, time_limit=t_alloc,
                                                  T_init=0.005, lr_init=0.01,
                                                  momentum_init=0.9, polish_every=20)
            try_seq(a_opt, f"nvar_{n_try}")

    print(f"  Phase 6 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t6:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    # Tiny final perturbation check
    n_final = len(best_seq)
    for ns in [0.0001, 0.0003, 0.0005]:
        for _ in range(3):
            pert = project_simplex(best_seq + rng.standard_normal(n_final) * ns)
            ps = float(evaluate_sequence(pert.tolist()))
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_seq = pert.copy()

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
