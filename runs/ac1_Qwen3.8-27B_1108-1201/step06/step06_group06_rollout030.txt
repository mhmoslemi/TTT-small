"""
Optimized step function search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based gradient for peak-targeted optimization, FW-LP with sparse oracle,
simulated annealing with diverse moves, coordinate descent polish, and multi-n exploration.
Key improvement: O(n log n) gradient via FFT instead of O(n^2) loop, and sharper
peak-targeting weights for faster convergence.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_fft(a, w):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a using FFT.
    Returns grad of length n where grad[j] = 2*sum_m a[m]*w[j+m]."""
    n = len(a)
    conv_len = 2 * n - 1
    w_rev = w[::-1].copy()
    full_len = n + conv_len - 1
    fft_size = 1 << (full_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(A * W, fft_size)[:full_len]
    result = conv[n - 1:2 * n - 1][::-1]
    return 2.0 * result


def softmax_weights(conv, T):
    """Compute softmax weights over convolution values with temperature T."""
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    s = p.sum()
    if s < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return p / s


def peak_weights(conv, top_k=20, alpha=0.5):
    """Hybrid weighting: blend uniform with top-k peak concentration."""
    n_conv = len(conv)
    w = np.ones(n_conv) / n_conv
    idx = np.argsort(conv)[::-1][:min(top_k, n_conv)]
    peak_w = np.zeros(n_conv)
    peak_w[idx] = 1.0
    peak_w /= peak_w.sum()
    return (1 - alpha) * w + alpha * peak_w


def build_lp_sparse(n, f):
    """Build sparse constraint matrix for LP oracle: (f*g)[k] <= M."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    A = lil_matrix((2 * n - 1, n))
    if rows:
        A[np.array(rows), np.array(cols)] = np.array(vals)
    return A.tocsr()


def solve_lp_oracle(f, M, n, A_sparse):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns g/sum(g) or None."""
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'max_iter': 50000})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None

            if no_improve > 30:
                rng = np.random.default_rng(it * 7919 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                conv = fast_convolve(f)
                M = conv.max()
                A_sparse = build_lp_sparse(n, f)
                m_exp = 1.0
                no_improve = 0
            continue

        alpha = line_search_conv(f, g)
        if alpha > 1e-10:
            f = (1.0 - alpha) * f + alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            A_sparse = build_lp_sparse(n, f)

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

    return best_f, best_obj


def try_seq(a, label, best_seq, best_score):
    """Evaluate a sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return a_norm.copy(), score
    return best_seq, best_score


def spectral_init(n, seed=42):
    """Create a sequence with approximately flat power spectrum (except DC)."""
    rng = np.random.default_rng(seed)
    # Start with random phase in frequency domain
    freq_size = 1 << (n - 1).bit_length()
    # Create flat spectrum with random phases
    X = np.zeros(freq_size // 2 + 1, dtype=np.complex128)
    X[0] = n  # DC component (sum = n, then normalize)
    # Add flat components with random phases
    for k in range(1, len(X)):
        X[k] = np.exp(1j * rng.uniform(0, 2 * np.pi))
    # Inverse FFT
    a = np.fft.irfft(X, freq_size)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: multi-phase optimization combining FW-LP, peak-targeted gradient,
    simulated annealing, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Using uniform init: score={best_score:.8f}")

    n_parent = len(parent)
    print(f"Starting optimization, target < 1.5027")

    # === Phase 1: Quick exploration of nearby n values ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_phase = time.monotonic()

    # Try parent n and a few neighbors
    n_candidates = sorted(set([
        n_parent,
        n_parent - 16, n_parent + 16,
        n_parent - 32, n_parent + 32,
        512, 1024
    ]))
    n_candidates = [n for n in n_candidates if 128 <= n <= 2048]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 200:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            # Interpolate parent to new length
            start = np.interp(np.linspace(0, 1, n_try),
                             np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try

        # Quick gradient step
        t_alloc = min(remaining * 0.03, 15)
        a_opt, _ = fg_grad_descent(start, time_limit=t_alloc,
                                    T_init=0.0005, top_k=20, alpha_peak=0.8)
        best_seq, best_score = try_seq(a_opt, f"fg_n{n_try}", best_seq, best_score)

        # Quick peak chasing
        remaining2 = deadline - time.monotonic()
        if remaining2 > 150:
            a_opt2, _ = fg_peak_chasing(a_opt, time_limit=min(remaining2 * 0.02, 10),
                                         init_temp=0.001)
            best_seq, best_score = try_seq(a_opt2, f"pc_n{n_try}", best_seq, best_score)

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW-LP on best ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 80:
        t_alloc = min(remaining * 0.30, 400)
        print(f"  Deep FW-LP n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, "deep_fw", best_seq, best_score)

    # Perturbed restarts
    for scale in [0.002, 0.005, 0.01, 0.02]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        n_best = len(best_seq)
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc2 = min(remaining * 0.08, 80)
        a_opt2, _ = fw_lp_optimize(f_pert, time_limit=t_alloc2)
        best_seq, best_score = try_seq(a_opt2, f"fw_p{scale}", best_seq, best_score)

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Peak-targeted gradient (deep) ===
    print("\n=== Phase 3: Peak-Targeted Gradient ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 60:
        t_alloc = min(remaining * 0.20, 200)
        print(f"  FG (sharpened) t={t_alloc:.0f}s")
        a_opt, _ = fg_grad_descent(best_seq, time_limit=t_alloc,
                                    T_init=0.0003, top_k=10, alpha_peak=0.9)
        best_seq, best_score = try_seq(a_opt, "fg_sharp", best_seq, best_score)

    remaining = deadline - time.monotonic()
    if remaining > 50:
        t_alloc = min(remaining * 0.15, 100)
        print(f"  Peak chasing t={t_alloc:.0f}s")
        a_opt, _ = fg_peak_chasing(best_seq, time_limit=t_alloc, init_temp=0.0005)
        best_seq, best_score = try_seq(a_opt, "peak_chase", best_seq, best_score)

    # Multiple T values
    for T_init in [0.0001, 0.0002, 0.001]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        n_best = len(best_seq)
        a_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
        t_alloc = min(remaining * 0.12, 80)
        a_opt, _ = fg_grad_descent(a_pert, time_limit=t_alloc,
                                    T_init=T_init, top_k=15, alpha_peak=0.85)
        best_seq, best_score = try_seq(a_opt, f"fg_T{T_init}", best_seq, best_score)

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Simulated Annealing ===
    print("\n=== Phase 4: Simulated Annealing ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()

    if remaining > 40:
        t_alloc = min(remaining * 0.20, 150)
        print(f"  SA t={t_alloc:.0f}s")
        a_opt, _ = simulated_annealing(best_seq, time_limit=t_alloc, seed=12345)
        best_seq, best_score = try_seq(a_opt, "sa1", best_seq, best_score)

    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_alloc = min(remaining * 0.15, 100)
        print(f"  SA2 t={t_alloc:.0f}s")
        a_opt, _ = simulated_annealing(best_seq, time_limit=t_alloc, seed=99999)
        best_seq, best_score = try_seq(a_opt, "sa2", best_seq, best_score)

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Coordinate Descent & Polish ===
    print("\n=== Phase 5: Polish ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()

    if remaining > 30:
        t_alloc = min(remaining * 0.25, 120)
        print(f"  Coord descent t={t_alloc:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc, seed=55555)
        best_seq, best_score = try_seq(a_opt, "coord", best_seq, best_score)

    # Final gradient polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t_alloc = min(remaining * 0.20, 60)
        print(f"  Fine FG t={t_alloc:.0f}s")
        a_opt, _ = fg_grad_descent(best_seq, time_limit=t_alloc,
                                    T_init=0.0001, top_k=5, alpha_peak=0.95)
        best_seq, best_score = try_seq(a_opt, "fine_fg", best_seq, best_score)

    # Micro perturbations + quick gradient
    n_best = len(best_seq)
    for _ in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        noise = rng.standard_normal(n_best) * 0.0003
        f_pert = project_simplex(best_seq + noise)
        t_alloc = min(remaining * 0.15, 10)
        a_opt, _ = fg_grad_descent(f_pert, time_limit=t_alloc,
                                    T_init=0.0002, top_k=10, alpha_peak=0.9)
        best_seq, best_score = try_seq(a_opt, "micro", best_seq, best_score)

    # Final SA
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_opt, _ = simulated_annealing(best_seq, time_limit=min(remaining * 0.3, 20),
                                        seed=77777)
        best_seq, best_score = try_seq(a_opt, "final_sa", best_seq, best_score)

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Spectral + alternative n ===
    print("\n=== Phase 6: Spectral & Alt N ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()

    # Try spectral initialization at a few n values
    if remaining > 30:
        for n_try in [len(best_seq), len(best_seq) + 64, max(128, len(best_seq) - 64)]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            spec = spectral_init(n_try, seed=n_try)
            t_alloc = min(remaining * 0.10, 20)
            a_opt, _ = fg_grad_descent(spec, time_limit=t_alloc,
                                        T_init=0.001, top_k=20, alpha_peak=0.7)
            best_seq, best_score = try_seq(a_opt, f"spec_n{n_try}", best_seq, best_score)

    # One more FW-LP on absolute best
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t_alloc = min(remaining * 0.3, 40)
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, "final_fw", best_seq, best_score)

    print(f"  Phase 6 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(768) / 768
    fs = float(evaluate_sequence(np.maximum(best_seq, 0.0).tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
        print(f"  Verification improved: {best_score:.8f}")

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
