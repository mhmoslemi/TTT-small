def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def peak_coord_descent(a0, time_limit, top_k=8):

        # Try several perturbations
        for trial in range(15):
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            
            noise = rng.standard_normal(n) * scale
            # Structure: bias toward reducing the main peak
            conv = fast_convolve(a)
            peak_k = int(np.argmax(conv))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            for j in range(js, je + 1):
                partner = peak_k - j
                if 0 <= partner < n:
                    noise[j] -= scale * 1.5 * a[partner]

            candidate = project_simplex(a + noise)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-16:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                break

        if it % 200 == 0:
            si += 1

    return best_a, best_obj


def fw_with_lp(a0, time_limit, max_lp_time=5):
    """
    Frank-Wolfe with LP oracle (limited time per LP solve).
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    from scipy.optimize import linprog

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True,
                                                   'time_limit': max_lp_time})
            if not res.success:
                no_improve += 1
                if no_improve > 20:
                    # Perturb and restart
                    rng = np.random.default_rng((int(time.monotonic()*1000) + 7) % (2**31))
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                    conv = fast_convolve(f)
                    M = conv.max()
                    no_improve = 0
                continue
        except Exception:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        g = np.maximum(res.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            if no_improve > 50:
                break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search:
    Phase 1: Multi-n exploration with structured initializations
    Phase 2: Deep smoothed gradient descent
    Phase 3: FW-LP refinement
    Phase 4: Coordinate descent polish
    Phase 5: Micro-polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    t0_total = time.monotonic()

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t0_total
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return score

    # Load parent
    parent = None
    n_parent = 768
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_parent = len(parent)
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={n_parent}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")
        parent = np.ones(768) / 768
        n_parent = 768

    # === Phase 1: Multi-n exploration with structured initializations (~200s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent, n_parent - 32, n_parent + 32,
                               n_parent - 64, n_parent + 64,
                               640, 704, 832, 896, 960, 1024]))
    candidate_ns = [n for n in candidate_ns if 400 <= n <= 1400]

    for n_try in candidate_ns:
        if time.monotonic() > deadline - 1400:
            break

        t_alloc = min(25, deadline - time.monotonic() - 1400)
        if t_alloc < 5:
            break

        # Multiple initializations for this n
        inits = []
        
        # 1. Interpolated parent
        if parent is not None:
            interp = np.interp(np.linspace(0, 1, n_try),
                               np.linspace(0, 1, n_parent), parent)
            interp = np.maximum(interp, 0)
            if interp.sum() > 1e-10:
                inits.append(interp / interp.sum())
        
        # 2. Gaussian
        j = np.arange(n_try)
        center = (n_try - 1) / 2.0
        sigma = n_try / 4.0
        gauss = np.exp(-0.5 * ((j - center) / sigma) ** 2)
        inits.append(gauss / gauss.sum())
        
        # 3. Raised cosine
        rc = 0.5 + 0.5 * np.cos(2 * np.pi * (j - center) / (n_try - 1))
        rc = np.maximum(rc, 1e-10)
        inits.append(rc / rc.sum())
        
        # 4. Uniform
        inits.append(np.ones(n_try) / n_try)
        
        # 5. Inverse Gaussian (peaked at edges)
        inv_g = 1.0 / (1.0 + ((j - center) / (n_try / 3.0)) ** 2)
        inits.append(inv_g / inv_g.sum())

        best_for_n = None
        best_for_n_score = float('inf')

        for ii, init in enumerate(inits):
            t_sub = min(8, t_alloc / len(inits) + 2)
            if t_sub < 2:
                break
            a_opt, _ = smoothed_grad_descent(init, time_limit=t_sub,
                                              T_init=0.003, T_min=0.0003,
                                              lr_init=0.06)
            sc = evaluate_sequence(a_opt.tolist())
            if np.isfinite(sc) and sc < best_for_n_score:
                best_for_n_score = sc
                best_for_n = a_opt

        if best_for_n is not None:
            try_seq(best_for_n, f"p1_n{n_try}")

    print(f"  Phase 1 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep smoothed gradient descent (~500s) ===
    print("\n=== Phase 2: Deep Smoothed Gradient ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        
        # Main long run
        t_alloc = min(200, deadline - time.monotonic() - 900)
        if t_alloc > 20:
            print(f"  Smooth n={n_best}, t={t_alloc:.0f}s, T_init=0.002")
            a_opt, _ = smoothed_grad_descent(best_seq, time_limit=t_alloc,
                                              T_init=0.002, T_min=0.0002,
                                              lr_init=0.04)
            try_seq(a_opt, "p2_main")

        # Different temperature settings
        for T_i, lr_i in [(0.001, 0.03), (0.01, 0.08), (0.005, 0.05)]:
            t_alloc = min(80, deadline - time.monotonic() - 700)
            if t_alloc < 10:
                break
            print(f"  Smooth T={T_i}, lr={lr_i}, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_grad_descent(best_seq, time_limit=t_alloc,
                                              T_init=T_i, T_min=T_i * 0.1,
                                              lr_init=lr_i)
            try_seq(a_opt, f"p2_T{T_i}")

        # Perturbed restarts
        for scale in [0.003, 0.01, 0.02]:
            t_alloc = min(60, deadline - time.monotonic() - 500)
            if t_alloc < 10:
                break
            rng = np.random.default_rng(scale * 10000)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            a_opt, _ = smoothed_grad_descent(f_pert, time_limit=t_alloc,
                                              T_init=0.003, T_min=0.0003,
                                              lr_init=0.05)
            try_seq(a_opt, f"p2_pert{scale}")

    print(f"  Phase 2 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: FW-LP refinement (~300s) ===
    print("\n=== Phase 3: FW-LP Refinement ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        
        t_alloc = min(200, deadline - time.monotonic() - 300)
        if t_alloc > 30:
            print(f"  FW-LP n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_with_lp(best_seq, time_limit=t_alloc, max_lp_time=5)
            try_seq(a_opt, "p3_fw")

        # Second FW pass with perturbation
        t_alloc = min(80, deadline - time.monotonic() - 200)
        if t_alloc > 15:
            rng = np.random.default_rng(12345)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.005)
            a_opt, _ = fw_with_lp(f_pert, time_limit=t_alloc, max_lp_time=4)
            try_seq(a_opt, "p3_fw_pert")

    print(f"  Phase 3 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Coordinate descent polish (~200s) ===
    print("\n=== Phase 4: Coordinate Descent ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(150, deadline - time.monotonic() - 100)
        if t_alloc > 10:
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_opt, _ = peak_coord_descent(best_seq, time_limit=t_alloc, top_k=10)
            try_seq(a_opt, "p4_coord")

        # Another round from slightly different start
        t_alloc = min(50, deadline - time.monotonic() - 50)
        if t_alloc > 10:
            rng = np.random.default_rng(777)
            f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.001)
            a_opt, _ = peak_coord_descent(f_pert, time_limit=t_alloc, top_k=6)
            try_seq(a_opt, "p4_coord2")

    print(f"  Phase 4 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final micro-polish (~100s) ===
    print("\n=== Phase 5: Micro-Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(60, deadline - time.monotonic() - 20)
        if t_alloc > 5:
            print(f"  Micro polish, t={t_alloc:.0f}s")
            a_opt, _ = micro_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "p5_micro")

        # One more smoothed gradient pass with very low T
        t_alloc = min(30, deadline - time.monotonic() - 10)
        if t_alloc > 5:
            a_opt, _ = smoothed_grad_descent(best_seq, time_limit=t_alloc,
                                              T_init=0.001, T_min=0.0001,
                                              lr_init=0.02)
            try_seq(a_opt, "p5_final_smooth")

        # Final micro-polish on the result
        t_alloc = min(20, deadline - time.monotonic() - 5)
        if t_alloc > 3 and best_seq is not None:
            a_opt, _ = micro_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "p5_micro2")

    print(f"  Phase 5 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, "
          f"total time={time.monotonic()-t0_total:.0f}s)")
    return [float(x) for x in best_seq]
