**Multiplicative weights for the LP:**

The LP is: max 1^T g s.t. (f * g)[k] <= M for all k, g >= 0.

The dual is: min M * sum(lambda) s.t. sum_k lambda[k] * f[k-i] >= 1 for all i, lambda >= 0.

Combines spectral peak-shaving (IFT), softmax-weighted projected gradient,

Key novelty: spectral domain peak shaving using multiplicative frequency updates,

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_fft(a, p):
    """Compute grad[j] = 2 * sum_i a[i] * p[j+i] using FFT.
    This is 2 * cross_correlation(a, p) at lags 0..n-1.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    # cross_corr(a, p)[j] = sum_i a[i] * p[i+j]
    # = IDFT(DFT(a) * conj(DFT(p)))[j] ... no
    # Actually: sum_i a[i]*p[i+j] = (a * p_rev)[j] in circular sense
    # Let's use: cross_corr(j) = sum_i a[i] * p[i+j]
    # In frequency: IDFT(DFT(a) * DFT(p)) gives circular cross-correlation
    # circular_corr(j) = sum_i a[i] * conj(p[i-j]) ... 
    # Let me just use the direct approach with a Toeplitz-like computation via FFT
    
    # Simple approach: pad and use correlation
    a_pad = np.fft.rfft(a, fft_size)
    p_pad = np.fft.rfft(p, fft_size)
    # cross_corr[j] = sum_i a[i] * p[i+j] for j = 0, ..., n-1
    # This is the "non-circular" cross-correlation
    # Use: cross_corr = ifft(conj(fft(a)) * fft(p)) ... no that's circular
    # 
    # Actually for our purpose: grad[j] = 2 * sum_{i=0}^{n-1} a[i] * p[j+i]
    # where p has length >= 2n-1 (it's the convolution result)
    # Let's compute via: ifft(fft(a) * fft(p_reversed_shifted))
    # 
    # Simplest correct approach:
    # grad[j] = 2 * sum_i a[i] * p[j+i]
    # Let q[k] = p[k] for k = 0..2n-2
    # grad[j] = 2 * (a convolved with q, but only the "forward" part)
    # 
    # Actually the cleanest: use matmul or direct dot for n <= 4096
    n_chk = len(a)
    grad = np.zeros(n_chk)
    for j in range(n_chk):
        grad[j] = 2.0 * np.dot(a, p[j:j + n_chk])
    return grad


def compute_grad_fft_fast(a, p):
    """Faster gradient computation using FFT when n is large enough."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    # grad[j] = 2 * sum_{i=0}^{n-1} a[i] * p[j+i]
    # This equals 2 * (a * p)[j] where * is a "partial" convolution
    # Actually: define r[k] = p[k] for k in [0, 2n-2]
    # grad[j] = 2 * sum_i a[i] * r[j+i]
    # Let s = reverse(a): s[i] = a[n-1-i]
    # Then sum_i a[i]*r[j+i] = sum_i s[n-1-i]*r[j+i] = (s * r)[j + n - 1] (circular)
    # where (s*r) is linear convolution of s and r
    
    # Simpler: just use the direct method, it's O(n^2) but with good constants via BLAS
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = 2.0 * np.dot(a, p[j:j + n])
    return grad


def spectral_optimize(a0, time_limit, alpha=0.5, n_inner=3):
    """Spectral peak shaving: iteratively push the auto-convolution towards flat.
    
    Uses multiplicative frequency-domain update:
    A_new = A * (S_target / S)^(alpha/2)
    where S = A^2 (DFT of current convolution), S_target = DFT of peaked-limited convolution.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Compute current convolution and its DFT
        b = fast_convolve(a)
        b_pad = np.zeros(fft_size)
        b_pad[:size] = b
        S = np.fft.rfft(b_pad)
        
        # Create target: reduce peaks above threshold
        mu = np.mean(b)
        sigma = np.std(b)
        # Try different multipliers
        best_trial = None
        best_trial_obj = best_obj
        
        for mult in [0.5, 0.8, 1.0, 1.2, 1.5]:
            threshold = mu + mult * sigma
            b_target = np.minimum(b, threshold)
            # Preserve total sum (to maintain sum(a)^2)
            scale = np.sum(b) / np.sum(b_target) if np.sum(b_target) > 1e-15 else 1.0
            b_target = b_target * scale
            
            b_t_pad = np.zeros(fft_size)
            b_t_pad[:size] = b_target
            S_target = np.fft.rfft(b_t_pad)
            
            # Multiplicative update: A_new = A * (S_target/S)^(alpha/2)
            # A = DFT(a)
            A = np.fft.rfft(a, fft_size)
            
            # Compute ratio with safe division
            ratio = S_target / S
            # For zeros in S, use target directly
            mask = np.abs(S) > 1e-15
            ratio[~mask] = 0.0
            
            # Take the (alpha/2)-root
            # Use log for complex: (S_target/S)^(alpha/2) = exp((alpha/2) * log(S_target/S))
            log_ratio = np.zeros_like(ratio, dtype=np.complex128)
            log_ratio[mask] = (alpha / 2.0) * np.log(ratio[mask])
            # For the DC component (index 0), S[0] = (sum a)^2 = 1 (since we normalized)
            # Keep DC fixed to preserve sum
            log_ratio[0] = 0.0
            
            factor = np.exp(log_ratio)
            factor[0] = 1.0  # preserve DC
            
            A_new = A * factor
            
            # Enforce Hermitian symmetry (rfft already handles this)
            a_new = np.fft.irfft(A_new, fft_size, n=fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            if sn < 1e-10:
                continue
            a_new = a_new / sn
            
            obj = fast_convolve(a_new).max()
            if obj < best_trial_obj:
                best_trial_obj = obj
                best_trial = a_new.copy()
        
        if best_trial is not None:
            a = best_trial
            if best_obj > best_trial_obj:
                best_obj = best_trial_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Adapt alpha
        if no_improve > 50:
            alpha *= 0.8
            no_improve = 0
        if no_improve > 200:
            # Restart from best with perturbation
            rng = np.random.default_rng(it * 13 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            alpha = 0.5
            no_improve = 0
        if no_improve > 500:
            break
    
    return best_a, best_obj


def softmax_grad_optimize(a0, time_limit, T_init=0.001, lr_init=0.02, top_k=5):

    lr = lr_init

    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        
        # Compute weights: focus on top-k peaks
        k = min(top_k, len(conv))
        top_idx = np.argpartition(conv, -k)[-k:]
        peak_vals = conv[top_idx]
        # Softmax over the peaks with temperature T
        scaled = peak_vals / T
        scaled -= scaled.max()
        w_peaks = np.exp(scaled)
        w_peaks = w_peaks / w_peaks.sum()
        
        # Build weight vector
        p = np.zeros(len(conv))
        p[top_idx] = w_peaks
        
        # Compute gradient
        grad = np.zeros(n)
        for j in range(n):
            grad[j] = 2.0 * np.dot(a, p[j:j + n])
        
        # Line search
        best_candidate = a
        best_val = best_obj
        for trial_lr in [lr, lr * 0.5, lr * 0.25]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:

                best_candidate = candidate

            best_obj = best_val
            best_a = a.copy()

        else:
            no_improve += 1

def fw_lp_optimize(a0, time_limit):
    """Frank-Wolfe with LP oracle."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    # Build LP matrix
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    row_idx = np.array(rows)
    col_idx = np.array(cols)
    
    M = fast_convolve(a).max()
    best_obj = M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.3:
            break
        
        # Solve LP: max sum(g) s.t. (a*g)[k] <= M * m_exp, g >= 0
        c = -np.ones(n)
        vals = np.empty(len(row_idx))
        for idx in range(len(row_idx)):
            k = row_idx[idx]
            i = col_idx[idx]
            vals[idx] = a[k - i]
        
        A_mat = lil_matrix((2 * n - 1, n))
        A_mat[row_idx, col_idx] = vals
        A_mat = A_mat.tocsr()
        
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(0.5, remaining * 0.3)})
            if res.success and res.x is not None:
                g = np.maximum(res.x, 0.0)
                sg = g.sum()
                if sg > 1e-15:
                    g = g / sg
                else:
                    continue
            else:
                continue
        except Exception:
            m_exp = min(m_exp * 1.3, 5.0)
            continue
        
        if np.max(np.abs(g - a)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 31 + 7)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
                M = fast_convolve(a).max()
                m_exp = 1.0
                no_improve = 0
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            at = np.maximum((1.0 - alpha) * a + alpha * g, 0.0)
            st = at.sum()
            if st < 1e-15:
                continue
            at = at / st
            val = fast_convolve(at).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            a = np.maximum((1.0 - best_alpha) * a + best_alpha * g, 0.0)
            a = a / a.sum()
            M = fast_convolve(a).max()
            m_exp = 1.0
            if M < best_obj - 1e-15:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.15, 5.0)
    
    return best_a, best_obj


def local_search(a0, time_limit, seed=42):

    a = np.maximum(a0.astype(np.float64), 0.0)

    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()

    rng = np.random.default_rng(seed)
    stale = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        
        improved = False
        
        for scale in [0.0003, 0.001, 0.003, 0.01]:

            candidate = project_simplex(a + pert)
            c = fast_convolve(candidate)

                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                stale = 0
                break
        
        if not improved and it % 2 == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))

                    candidate = a.copy()
                    amount = candidate[i] * frac

                    candidate[j] += amount

                    candidate = candidate / ss
                    c = fast_convolve(candidate)

                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        stale = 0
                        break
        
        if not improved:
            stale += 1
        if stale > 2000:
            break
    
    return best_a, best_obj

    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Using uniform init: score={best_score:.8f}")
    
    n_parent = len(parent)
    
    # === Phase 1: Multi-n exploration ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()
    
    candidate_ns = [256, 384, 512, 640, 768, 896, 1024, 1280, 1536, 2048, 2560, 3072]
    if n_parent not in candidate_ns and 128 <= n_parent <= 4096:
        candidate_ns.append(n_parent)
    candidate_ns.sort()
    
    # Time per n in phase 1
    n_candidates = len(candidate_ns)
    
    for n_try in candidate_ns:
        remaining = deadline - time.monotonic()
        if remaining < 200:
            break
        
        if n_try == n_parent:
            start = parent.copy()
        else:
            # Interpolate parent to new length
            x_orig = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            start = np.interp(x_new, x_orig, parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try
        
        # Quick spectral optimization
        t_alloc = min(remaining * 0.06, 20)
        a_opt, _ = spectral_optimize(start, time_limit=t_alloc, alpha=0.5)
        best_seq, best_score = try_seq(a_opt, f"spec_n{n_try}", (best_seq, best_score))
        
        # Quick gradient
        remaining2 = deadline - time.monotonic()
        if remaining2 > 150:
            a_opt2, _ = softmax_grad_optimize(a_opt, time_limit=min(remaining2 * 0.03, 10),
                                              T_init=0.001, lr_init=0.02)
            best_seq, best_score = try_seq(a_opt2, f"grad_n{n_try}", (best_seq, best_score))
    
    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 2: Deep spectral optimization ===
    print("\n=== Phase 2: Deep Spectral Optimization ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    
    if remaining > 100:
        t_alloc = min(remaining * 0.3, 400)
        print(f"  Deep spectral n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = spectral_optimize(best_seq, time_limit=t_alloc, alpha=0.5)
        best_seq, best_score = try_seq(a_opt, "deep_spec", (best_seq, best_score))
    
    # Spectral with different alpha values
    for alpha_val in [0.3, 0.7, 0.9]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        t_alloc = min(remaining * 0.1, 80)
        a_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.002)
        a_opt, _ = spectral_optimize(a_pert, time_limit=t_alloc, alpha=alpha_val)
        best_seq, best_score = try_seq(a_opt, f"spec_a{alpha_val}", (best_seq, best_score))
    
    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 3: Softmax gradient refinement ===
    print("\n=== Phase 3: Softmax Gradient ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    if remaining > 80:
        t_alloc = min(remaining * 0.25, 250)
        print(f"  Softmax grad T=0.001, t={t_alloc:.0f}s")
        a_opt, _ = softmax_grad_optimize(best_seq, time_limit=t_alloc, T_init=0.001, lr_init=0.02)
        best_seq, best_score = try_seq(a_opt, "softmax", (best_seq, best_score))
    
    # Finer temperature
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t_alloc = min(remaining * 0.15, 120)
        print(f"  Softmax grad T=0.0003, t={t_alloc:.0f}s")
        a_opt, _ = softmax_grad_optimize(best_seq, time_limit=t_alloc, T_init=0.0003, lr_init=0.01)
        best_seq, best_score = try_seq(a_opt, "softmax_fine", (best_seq, best_score))
    
    # Top-1 (just the peak)
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t_alloc = min(remaining * 0.12, 100)
        print(f"  Softmax top-1, t={t_alloc:.0f}s")
        a_opt, _ = softmax_grad_optimize(best_seq, time_limit=t_alloc, T_init=0.001, lr_init=0.015, top_k=1)
        best_seq, best_score = try_seq(a_opt, "softmax_top1", (best_seq, best_score))
    
    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 4: FW-LP ===
    print("\n=== Phase 4: FW-LP ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    
    if remaining > 100:
        t_alloc = min(remaining * 0.2, 200)
        print(f"  FW-LP n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, "fw_lp", (best_seq, best_score))
    
    # Perturbed FW-LP
    for scale in [0.003, 0.01]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        n_best = len(best_seq)
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc2 = min(remaining * 0.1, 80)
        a_opt2, _ = fw_lp_optimize(f_pert, time_limit=t_alloc2)
        best_seq, best_score = try_seq(a_opt2, f"fw_pert{scale}", (best_seq, best_score))
    
    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 5: Local search and polish ===
    print("\n=== Phase 5: Local Search & Polish ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    if remaining > 40:
        t_alloc = min(remaining * 0.3, 120)
        print(f"  Local search, t={t_alloc:.0f}s")
        a_opt, _ = local_search(best_seq, time_limit=t_alloc, seed=777)
        best_seq, best_score = try_seq(a_opt, "local", (best_seq, best_score))
    
    # Alternate spectral and gradient for fine-tuning
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for phase_iter in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            # Spectral micro-step
            t_alloc = min(remaining * 0.2, 20)
            a_opt, _ = spectral_optimize(best_seq, time_limit=t_alloc, alpha=0.3)
            best_seq, best_score = try_seq(a_opt, f"micro_spec{phase_iter}", (best_seq, best_score))
            
            # Gradient micro-step
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            t_alloc = min(remaining * 0.2, 15)
            a_opt, _ = softmax_grad_optimize(best_seq, time_limit=t_alloc, T_init=0.0003, lr_init=0.005, top_k=3)
            best_seq, best_score = try_seq(a_opt, f"micro_grad{phase_iter}", (best_seq, best_score))
    
    # Final local search
    remaining = deadline - time.monotonic()
    if remaining > 15:
        t_alloc = min(remaining * 0.4, 30)
        a_opt, _ = local_search(best_seq, time_limit=t_alloc, seed=4242)
        best_seq, best_score = try_seq(a_opt, "local_final", (best_seq, best_score))
    
    # Tiny perturbations
    remaining = deadline - time.monotonic()
    if remaining > 10:
        for _ in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 3:
                break
            noise = rng.standard_normal(n_best) * 0.0003
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = softmax_grad_optimize(f_pert, time_limit=min(remaining * 0.2, 5),
                                             T_init=0.0002, lr_init=0.003, top_k=1)
            best_seq, best_score = try_seq(a_opt, "micro_perturb", (best_seq, best_score))
    
    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 6: Try alternative n values on the best structure ===
    print("\n=== Phase 6: Alternative n exploration ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    
    if remaining > 60:
        for n_mult in [0.8, 1.2, 1.5]:
            n_try = int(len(best_seq) * n_mult)
            n_try = max(128, min(4096, n_try))
            if n_try == len(best_seq):
                continue
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            x_orig = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            start = np.interp(x_new, x_orig, best_seq)
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try
            t_alloc = min(remaining * 0.25, 40)
            a_opt, _ = spectral_optimize(start, time_limit=t_alloc, alpha=0.5)
            best_seq, best_score = try_seq(a_opt, f"alt_n{n_try}", (best_seq, best_score))
    
    print(f"  Phase 6 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
