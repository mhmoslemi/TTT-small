import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, top_k=5):
    """Compute weighted subgradient of max(conv) using top-k peaks."""
    n = len(a)
    conv = fast_convolve(a)
    # Get top-k peaks
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        weights = np.ones(len(peaks)) / len(peaks)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        # Weight by value (higher peaks get more weight)
        pv = conv[peaks]
        weights = pv / pv.sum()
    
    grad = np.zeros(n)
    for i, k in enumerate(peaks):
        if weights[i] < 1e-10:
            continue
        # d(conv[k])/da[j] = 2*a[k-j] for valid j
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * weights[i] * a[k - idx]
    
    return grad, conv.max()

            candidate = candidate / candidate.sum()
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        if improved:
            stale = 0
        else:
            stale += 1
            if stale % 50 == 0 and stale < 500:
                rng = np.random.default_rng(it * 4567 + 3)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    stale = 0
            elif stale > 500:
                break

    return best_a, best_obj


def resample_sequence(parent, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(parent)
    if old_n == new_n:
        return parent.copy()
    if old_n == 1:
        return np.full(new_n, parent[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, parent)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return new_seq / s


def try_and_update(a, state, label):
    """Evaluate and update state if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a_norm)))


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimized search:
    Phase 1: Quick multi-n exploration (200s)
    Phase 2: Deep smoothed gradient descent (900s)
    Phase 3: Subgradient + simulated annealing (400s)
    Phase 4: Targeted peak reduction + final polish (280s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    state = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.8f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        state = [float(evaluate_sequence(parent.tolist())), parent.copy()]
        print("No parent, using uniform n=384, score={:.8f}".format(state[0]))

    n_parent = len(parent)

    # === Phase 1: Quick Multi-n Exploration (~200s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent, n_parent - 32, n_parent + 32,
                               640, 800, 960, 1024]))
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 2048]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1500
        if remaining < 10:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = resample_sequence(parent, n_try)

        t_alloc = min(remaining, 30)
        print("  Smooth n={}, t={:.0f}s".format(n_try, t_alloc))
        a_opt, _ = local_search_smoothed(start, time_limit=t_alloc,
                                         T_init=0.003, lr_init=0.06)
        try_and_update(a_opt, state, "explore_n{}".format(n_try))

    print("  Phase 1 best: {:.8f} (elapsed {:.0f}s)".format(
        state[0], time.monotonic() - t_phase))

    # === Phase 2: Deep Smoothed Gradient Descent (~900s) ===
    print("\n=== Phase 2: Deep Smoothed Gradient ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])
        remaining = deadline - time.monotonic() - 500
        if remaining > 30:
            t_alloc = min(remaining, 600)
            print("  Deep smooth n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = local_search_smoothed(state[1], time_limit=t_alloc,
                                             T_init=0.001, lr_init=0.05)
            try_and_update(a_opt, state, "deep_smooth_A")

        # Second pass with different params
        remaining = deadline - time.monotonic() - 400
        if remaining > 20:
            t_alloc = min(remaining, 300)
            print("  Deep smooth (alt) n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = local_search_smoothed(state[1], time_limit=t_alloc,
                                             T_init=0.005, lr_init=0.03)
            try_and_update(a_opt, state, "deep_smooth_B")

    # Subgradient method
    if state[1] is not None:
        remaining = deadline - time.monotonic() - 200
        if remaining > 20:
            t_alloc = min(remaining, 120)
            print("  Subgrad n={}, t={:.0f}s".format(len(state[1]), t_alloc))
            a_opt, _ = local_search_subgrad(state[1], time_limit=t_alloc,
                                            top_k=15, lr_init=0.03)
            try_and_update(a_opt, state, "subgrad15")

    print("  Phase 2 best: {:.8f} (elapsed {:.0f}s)".format(
        state[0], time.monotonic() - t_phase))

    # === Phase 3: Simulated Annealing + Perturbs (~400s) ===
    print("\n=== Phase 3: Simulated Annealing ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # SA with different initial temperatures
        for T_s, T_e in [(0.003, 0.0001), (0.001, 0.00005), (0.01, 0.0002)]:
            remaining = deadline - time.monotonic() - 200
            if remaining < 10:
                break
            t_alloc = min(remaining, 120)
            print("  SA T=[{},{}], t={:.0f}s".format(T_s, T_e, t_alloc))
            a_opt, _ = simulated_annealing(state[1], time_limit=t_alloc,
                                           T_start=T_s, T_end=T_e)
            try_and_update(a_opt, state, "sa_{:.4f}".format(T_s))

        # Perturbed restarts with short local search
        for scale in [0.001, 0.003, 0.005, 0.01]:
            remaining = deadline - time.monotonic() - 80
            if remaining < 10:
                break
            f_pert = project_simplex(state[1] + rng.standard_normal(len(state[1])) * scale)
            t_alloc = min(remaining, 40)
            a_opt, _ = local_search_smoothed(f_pert, time_limit=t_alloc,
                                             T_init=0.002, lr_init=0.04)
            try_and_update(a_opt, state, "perturb_{}".format(scale))

    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(
        state[0], time.monotonic() - t_phase))

    # === Phase 4: Targeted Peak Reduction + Final Polish (~280s) ===
    print("\n=== Phase 4: Final Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Targeted peak reduction
        remaining = deadline - time.monotonic() - 100
        if remaining > 15:
            t_alloc = min(remaining, 120)
            print("  Targeted peak red, t={:.0f}s".format(t_alloc))
            a_opt, _ = targeted_peak_reduction(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "targeted")

        # Fine smoothed gradient
        remaining = deadline - time.monotonic() - 50
        if remaining > 10:
            t_alloc = min(remaining, 60)
            print("  Fine smooth, t={:.0f}s".format(t_alloc))
            a_opt, _ = local_search_smoothed(state[1], time_limit=t_alloc,
                                             T_init=0.0005, lr_init=0.01)
            try_and_update(a_opt, state, "fine_smooth")

        # Another targeted pass
        remaining = deadline - time.monotonic() - 20
        if remaining > 10:
            t_alloc = min(remaining, 40)
            print("  Targeted pass 2, t={:.0f}s".format(t_alloc))
            a_opt, _ = targeted_peak_reduction(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "targeted2")

        # Micro perturbations
        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            for _ in range(3):
                remaining = deadline - time.monotonic() - 3
                if remaining < 3:
                    break
                noise = rng.standard_normal(len(state[1])) * 0.0003
                f_pert = project_simplex(state[1] + noise)
                a_opt, _ = local_search_smoothed(f_pert, time_limit=min(remaining, 5),
                                                 T_init=0.0001, lr_init=0.005)
                try_and_update(a_opt, state, "micro")

    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(
        state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(384) / 384
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
