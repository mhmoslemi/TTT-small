def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_soft_gradient(a, conv, T, top_k=None):
    """
    Compute the gradient of T * log(sum_k exp(conv[k]/T)) w.r.t. a.
    Returns gradient vector of same length as a.
    """
    n = len(a)
    conv_len = len(conv)
    
    # Compute softmax weights
    conv_shifted = conv / T - (conv.max() / T)  # for numerical stability
    exp_vals = np.exp(conv_shifted)
    w = exp_vals / exp_vals.sum()
    
    if top_k is not None and top_k < conv_len:
        # Zero out all but top-k
        thresh = np.sort(w)[::-1][top_k - 1]
        w[w < thresh] = 0
        w = w / w.sum()
    
    # Compute gradient
    # grad[i] = sum_k w[k] * 2 * a[k-i] (for valid k-i)
    # This is a weighted sum of shifted copies of a
    grad = np.zeros(n)
    
    # Efficient computation using FFT for the blurring
    # grad[i] = sum_k w[k] * 2 * a[k-i] where k-i is valid
    # = 2 * (w * a)[i] (linear cross-correlation-like)
    
    # Actually, let's think about this more carefully.
    # For each k, the contribution to grad[i] is w[k] * 2 * a[k-i] if 0 <= k-i < n
    # This means: for a fixed k, grad[i] += w[k] * 2 * a[k-i] for i in [k-n+1, k] intersect [0, n-1]
    # i.e., i in [max(0, k-n+1), min(n-1, k)]
    # And a[k-i] where k-i is in [0, n-1]
    
    # Let's define: for each valid pair (i, k), grad[i] += w[k] * 2 * a[k-i]
    # where the validity depends on the convolution structure.
    
    # Actually, I realize this is exactly the cross-correlation of w (padded to length 2n-1) with a.
    # (w * a_cross)[i] = sum_k w[k] * a[k-i] (with appropriate bounds)
    
    # Let's use FFT for efficiency
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    
    # Cross-correlation: sum_k w[k] * a[k-i]
    # This is the cross-correlation of w and a, which is conv(w, reverse(a))
    # Or equivalently, it's ifft(W * conj(A))... let me think.
    
    # Actually: (w ⊛ a)[i] = sum_k w[k] * a[k+i] (cross-correlation)
    # What I want is: sum_k w[k] * a[k-i] = sum_k w[k] * a[-(i-k)] 
    # = (w ⊛ reverse(a))[n-1-i]... this is getting complicated.
    
    # Let me just do it directly for clarity, since n is at most ~2048
    # and the number of non-zero w entries is limited by top_k
    
    nonzero_k = np.where(w > 1e-15)[0]
    for k in nonzero_k:
        wk = w[k]
        if wk < 1e-15:
            continue
        # For this k, valid i range: max(0, k-n+1) to min(n-1, k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        # a[k-i] for i in [i_lo, i_hi]: k-i ranges from k-i_lo to k-i_hi
        # k-i_lo = k - max(0, k-n+1)
        # k-i_hi = k - min(n-1, k)
        # For k < n: i_lo=0, i_hi=k, so k-i ranges from k to 0
        # For k >= n: i_lo=k-n+1, i_hi=n-1, so k-i ranges from n-1 to k-n+1
        
        idx = k - np.arange(i_lo, i_hi + 1)  # these are valid indices into a
        grad[i_lo:i_hi+1] += wk * 2.0 * a[idx]
    
    return grad

def mirror_descent_optimize(a0, time_limit, T_init=0.1, T_min=0.001, eta_init=0.1, verbose=False):
    """
    Mirror descent (exponentiated gradient) with soft-max smoothing.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    T = T_init
    eta = eta_init
    t0 = time.monotonic()
    it = 0
    stagnant = 0
    
    # Adaptive parameters
    T_decay = 0.995  # temperature decay per iteration
    eta_decay = 0.99
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        
        conv = fast_convolve(a)
        current_max = conv.max()
        
        # Compute gradient with current temperature
        grad = compute_soft_gradient(a, conv, T)
        grad_norm = np.linalg.norm(grad)
        
        if grad_norm < 1e-15:
            break
        
        # Mirror descent update: a_i <- a_i * exp(-eta * grad_i)
        # Clip for numerical stability
        log_update = -eta * grad / grad_norm  # normalize gradient
        # Re-normalize to keep sum(a) = 1
        a_new = a * np.exp(log_update)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            # Numerical issue, reset
            a = best_a.copy()
            T = T_init
            eta = eta_init
            stagnant = 0
            continue
        a_new = a_new / a_new_sum
        
        # Check improvement
        new_conv = fast_convolve(a_new)
        new_max = new_conv.max()
        
        if new_max < current_max:
            a = a_new
            if new_max < best_obj - 1e-15:
                best_obj = new_max
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            # Step size too large, reduce it
            eta *= 0.7
            stagnant += 1
        
        # Adapt temperature (anneal)
        T = max(T_min, T * T_decay)
        
        # Adapt eta
        if stagnant > 20:
            eta = eta_init * 0.5
            T = max(T_min, T * 1.1)
            stagnant = 0
        
        # Periodic restart with noise
        if stagnant > 100:
            rng = np.random.default_rng(it * 7919)
            noise = rng.standard_normal(n) * 0.003
            a = np.maximum(best_a + noise, 1e-10)
            a = a / a.sum()
            T = T_init * 0.5
            eta = eta_init
            stagnant = 0
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it}, obj={best_obj:.10f}, T={T:.5f}, eta={eta:.5f}, stagnant={stagnant}, time={elapsed:.0f}s")
    
    return best_a, best_obj

def peak_flattening_descent(a0, time_limit):

    t0 = time.monotonic()

        conv = fast_convolve(a)

        # Find contributing pairs for this peak
        i_lo = max(0, k_peak - n + 1)
        i_hi = min(n - 1, k_peak)
        
        # Contributions: a[i] * a[k_peak - i] for i in [i_lo, i_hi]
        idx_range = np.arange(i_lo, i_hi + 1)
        contributions = a[idx_range] * a[k_peak - idx_range]
        
        # Find the top contributing pairs
        top_idx = np.argsort(contributions)[::-1][:3]
        
        improved = False

def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s

def diverse_starts(n, parent, rng):

    for _ in range(3):

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient(a, conv, top_k=10):
    """
    Compute the subgradient of max_k conv[k] at point a.
    Uses top_k peaks with equal weighting.
    Returns gradient vector of length n.
    """
    n = len(a)
    conv_len = len(conv)
    
    # Get top-k peak indices
    if top_k < conv_len:
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
    else:
        peak_indices = np.arange(conv_len)
    
    # Create weight vector over the full convolution
    w = np.zeros(conv_len)
    w[peak_indices] = 1.0 / len(peak_indices)

    # grad[i] = 2 * sum_k w[k] * a[k-i] (valid terms)
    # = 2 * correlate(w, a, mode='valid')[i]
    grad = 2.0 * correlate(w, a, mode='valid')
    
    return grad


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        current_max = conv.max()
        
        # Compute subgradient
        grad = compute_subgradient(a, conv, top_k=current_top_k)
        
        # Nesterov momentum
        v = momentum * v - lr * grad
        candidate = project_simplex(a + v)
        
        # Backtracking line search
        accepted = False
        for bt in range(backtracking_max):
            alpha = 1.0 / (2 ** bt)
            trial = project_simplex(a + alpha * v)
            trial_conv = fast_convolve(trial)
            if trial_conv.max() < current_max - 1e-15 or bt == backtracking_max - 1:
                a = trial
                if trial_conv.max() < best_obj - 1e-15:
                    best_obj = trial_conv.max()
                    best_a = a.copy()
                    stagnant = 0
                else:
                    stagnant += 1
                accepted = True
                break
        
        if not accepted:
            a = project_simplex(a + v)
            stagnant += 1
        
        # Adapt parameters
        if stagnant > 30:
            lr *= 0.5
            current_top_k = max(3, current_top_k // 2)
            stagnant = 0
        elif stagnant > 100:
            # Reset momentum, try wider peaks
            v = np.zeros(n)
            current_top_k = min(20, current_top_k * 3)
            lr = lr_init * 0.5
            stagnant = 0
        elif stagnant > 300:
            # Restart with perturbation
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            scale = 0.002 + 0.003 * min(1.0, (stagnant - 300) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            lr = lr_init
            current_top_k = top_k
            stagnant = 0
        
        # Gradually narrow the active set
        if it % 100 == 0 and current_top_k > 3:
            current_top_k = max(3, current_top_k - 1)
        
        if verbose and it % 200 == 0:
            print(f"    GD it={it}, obj={best_obj:.10f}, lr={lr:.6f}, top_k={current_top_k}, "
                  f"stagnant={stagnant}, time={time.monotonic()-t0:.0f}s")
    
    return best_a, best_obj


def adam_softmax_optimize(a0, time_limit, lr_init=0.005, top_k=10,
                          beta1=0.9, beta2=0.999, eps=1e-8, verbose=False):
    """
    Adam optimizer on softmax parameterization.
    a = softmax(theta). Optimizes theta directly.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-12)
    a = a / a.sum()
    theta = np.log(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    stagnant = 0
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break
        
        # Recompute a from theta
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        a = exp_theta / exp_theta.sum()
        
        # Compute convolution
        conv = fast_convolve(a)
        current_max = conv.max()
        
        # Subgradient of max
        grad_a = compute_subgradient(a, conv, top_k=top_k)
        
        # Transform to theta-space: grad_theta = a * (grad_a - a^T grad_a)
        grad_theta = a * (grad_a - np.dot(a, grad_a))
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * grad_theta ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        theta -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Evaluate
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        a_new = exp_theta / exp_theta.sum()
        new_conv = fast_convolve(a_new)
        new_max = new_conv.max()
        
        if new_max < best_obj - 1e-15:
            best_obj = new_max
            best_a = a_new.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Adaptive
        if stagnant > 50:
            lr *= 0.6
            stagnant = 0
        elif stagnant > 200:
            rng = np.random.default_rng((t * 104729 + 7) % (2**31))
            theta = np.log(np.maximum(best_a + rng.standard_normal(n) * 0.003, 1e-12))
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = lr_init
            stagnant = 0
        
        if verbose and t % 200 == 0:
            print(f"    Adam it={t}, obj={best_obj:.10f}, lr={lr:.6f}, "
                  f"stagnant={stagnant}, time={time.monotonic()-t0:.0f}s")
    
    return best_a, best_obj


def mirror_descent_optimize(a0, time_limit, lr_init=0.05, top_k=10, verbose=False):
    """
    Mirror descent (exponentiated gradient) for simplex-constrained problem.
    a_i <- a_i * exp(-eta * grad_i), then renormalize.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-12)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    stagnant = 0
    eta = lr_init
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.01:
            break
        
        conv = fast_convolve(a)
        current_max = conv.max()
        
        grad = compute_subgradient(a, conv, top_k=top_k)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Mirror descent step
        log_update = -eta * grad / grad_norm
        # Clip for numerical stability
        log_update = np.clip(log_update, -5.0, 5.0)
        a_new = a * np.exp(log_update)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15 or not np.isfinite(a_new_sum):
            eta *= 0.5
            stagnant += 1
            continue
        a_new = a_new / a_new_sum
        
        new_conv = fast_convolve(a_new)
        new_max = new_conv.max()
        
        if new_max < current_max - 1e-15:
            a = a_new
            if new_max < best_obj - 1e-15:
                best_obj = new_max
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            eta *= 0.7
            stagnant += 1
        
        if stagnant > 50:
            eta = lr_init * 0.3
            stagnant = 0
        elif stagnant > 200:
            rng = np.random.default_rng((int(stagnant * 65537) % (2**31)))
            a = np.maximum(best_a + rng.standard_normal(n) * 0.003, 1e-12)
            a = a / a.sum()
            eta = lr_init
            stagnant = 0
        
        if verbose and stagnant % 500 == 0:
            print(f"    MD stagnant={stagnant}, obj={best_obj:.10f}, time={time.monotonic()-t0:.0f}s")
    
    return best_a, best_obj


def peak_coordinate_descent(a0, time_limit, verbose=False):
    """
    Peak-targeted coordinate descent.
    For the current max peak, try redistributing mass from contributing
    coordinates to distant locations.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    step_scale = 0.02
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break
        
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        
        # Contributing range
        i_lo = max(0, k_peak - n + 1)
        i_hi = min(n - 1, k_peak)
        idx_range = np.arange(i_lo, i_hi + 1)
        contributions = a[idx_range] * a[k_peak - idx_range]
        
        # Top contributors
        top_count = min(5, len(idx_range))
        top_local = np.argsort(contributions)[::-1][:top_count]
        
        improved = False
        for ti in top_local:
            i = idx_range[ti]
            j = k_peak - i
            if i == j:
                continue
            if a[i] < 1e-10 or a[j] < 1e-10:
                continue
            
            src = i if a[i] >= a[j] else j
            other = j if src == i else i
            
            # Try moving mass to several candidate targets
            # Deterministic targets: midpoint, far end, offset
            targets = [
                (src + n // 3) % n,
                (src + 2 * n // 3) % n,
                n - 1 - src,
                (src + n // 2) % n,
            ]
            # Add a random target
            rng = np.random.default_rng((it * 1000033 + src * 7919 + k_peak * 31) % (2**31))
            targets.append(int(rng.integers(0, n)))
            
            for target in targets:
                if target == src or target == other:
                    continue
                
                for frac in [step_scale, step_scale * 0.5, step_scale * 2.0]:
                    delta = a[src] * frac
                    if delta < 1e-12:
                        continue
                    candidate = a.copy()
                    candidate[src] -= delta
                    candidate[target] += delta
                    candidate = np.maximum(candidate, 0)
                    csum = candidate.sum()
                    if csum < 1e-15:
                        continue
                    candidate = candidate / csum
                    
                    new_conv = fast_convolve(candidate)
                    if new_conv.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = new_conv.max()
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            step_scale *= 0.9
            if step_scale < 1e-7:
                break
            # Periodic global perturbation
            if it % 100 == 0:
                rng = np.random.default_rng(it * 48271)
                candidate = project_simplex(a + rng.standard_normal(n) * step_scale * 3)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
            elif it > 3000:
                break
        
        if verbose and it % 500 == 0:
            print(f"    CD it={it}, obj={best_obj:.10f}, step={step_scale:.6f}, "
                  f"time={time.monotonic()-t0:.0f}s")
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def diverse_starts(n, parent, rng):
    """Generate diverse initializations for length n."""
    starts = []
    x = np.arange(n, dtype=np.float64)
    
    if parent is not None:
        starts.append(resample_sequence(parent, n))
    
    # Uniform
    starts.append(np.ones(n) / n)
    
    # Cosine window (symmetric)
    w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    w = np.maximum(w, 0.01)
    starts.append(w / w.sum())
    
    # Asymmetric: concentrated at start
    w = np.exp(-x / (0.3 * n))
    starts.append(w / w.sum())
    
    # Two bumps (asymmetric)
    g1 = np.exp(-((x - 0.25 * n) ** 2) / (0.08 * n) ** 2)
    g2 = np.exp(-((x - 0.65 * n) ** 2) / (0.12 * n) ** 2)
    w = g1 + g2 + 0.01
    starts.append(w / w.sum())
    
    # Sine modulation
    w = 1.0 + 0.5 * np.sin(2 * np.pi * x / n)
    w = np.maximum(w, 0.01)
    starts.append(w / w.sum())
    
    # Random Dirichlet
    starts.append(rng.dirichlet(np.ones(n) * 0.3))
    starts.append(rng.dirichlet(np.ones(n) * 1.0))
    
    return starts


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search combining:
    1. Quick exploration across lengths with multiple optimizers
    2. Focused deep optimization on best candidate
    3. Coordinate descent polishing
    4. Final micro-refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")
    
    n_parent = len(parent)
    
    # === Phase 1: Quick exploration (40% of budget) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_phase = time.monotonic()
    phase1_budget = (budget_s - 30) * 0.40
    
    lengths = sorted(set([n_parent, 512, 640, 768, 896, 1024]))
    lengths = [l for l in lengths if 256 <= l <= 1280]
    
    for n_try in lengths:
        remaining_for_phase = t_phase + phase1_budget - time.monotonic()
        if remaining_for_phase < 15:
            break
        
        starts = diverse_starts(n_try, parent, rng)
        
        for si, start in enumerate(starts[:4]):
            remaining_for_phase = t_phase + phase1_budget - time.monotonic()
            if remaining_for_phase < 8:
                break
            
            t_alloc = min(remaining_for_phase / (4 - si), 25)
            if t_alloc < 3:
                break
            
            # Try gradient descent
            a_opt, _ = gradient_descent_main(start, time_limit=t_alloc, top_k=10, lr_init=0.01)
            try_seq(a_opt, f"gd_n{n_try}_s{si}")
    
    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 2: Deep optimization on best (30% of budget) ===
    print("\n=== Phase 2: Deep Optimization ===")
    t_phase = time.monotonic()
    phase2_budget = (budget_s - 30) * 0.30
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Try multiple optimizers with different hyperparameters
        optimizer_configs = [
            ('gd', {'top_k': 10, 'lr_init': 0.01, 'momentum': 0.9}),
            ('gd', {'top_k': 5, 'lr_init': 0.005, 'momentum': 0.95}),
            ('gd', {'top_k': 20, 'lr_init': 0.02, 'momentum': 0.8}),
            ('adam', {'lr_init': 0.003, 'top_k': 10}),
            ('adam', {'lr_init': 0.001, 'top_k': 5}),
            ('mirror', {'lr_init': 0.03, 'top_k': 10}),
        ]
        
        for oi, (opt_name, cfg) in enumerate(optimizer_configs):
            remaining_phase2 = t_phase + phase2_budget - time.monotonic()
            if remaining_phase2 < 10:
                break
            
            t_alloc = min(remaining_phase2 / (len(optimizer_configs) - oi), 60)
            if t_alloc < 5:
                break
            
            print(f"  Optimizer {opt_name} cfg={cfg}, t={t_alloc:.0f}s")
            
            if opt_name == 'gd':
                a_opt, _ = gradient_descent_main(best_seq, time_limit=t_alloc, **cfg)
            elif opt_name == 'adam':
                a_opt, _ = adam_softmax_optimize(best_seq, time_limit=t_alloc, **cfg)
            elif opt_name == 'mirror':
                a_opt, _ = mirror_descent_optimize(best_seq, time_limit=t_alloc, **cfg)
            else:
                continue
            
            try_seq(a_opt, f"{opt_name}_{oi}")
    
    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 3: Coordinate descent polish (20% of budget) ===
    print("\n=== Phase 3: Coordinate Descent Polish ===")
    t_phase = time.monotonic()
    phase3_budget = (budget_s - 30) * 0.20
    
    if best_seq is not None:
        remaining_phase3 = t_phase + phase3_budget - time.monotonic()
        if remaining_phase3 > 10:
            t_alloc = min(remaining_phase3, 120)
            print(f"  Peak CD, t={t_alloc:.0f}s")
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "peak_cd")
            
            # Second round of CD from the polished result
            remaining_phase3 = t_phase + phase3_budget - time.monotonic()
            if remaining_phase3 > 10 and best_seq is not None:
                t_alloc = min(remaining_phase3, 60)
                a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc)
                try_seq(a_opt, "peak_cd2")
    
    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 4: Final refinement (remaining time) ===
    print("\n=== Phase 4: Final Refinement ===")
    t_phase = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Alternating short rounds of different optimizers with small perturbations
        for ri in range(10):
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            
            noise_scale = 0.0005 + 0.001 * ri / 10
            noise = rng.standard_normal(n_best) * noise_scale
            start = project_simplex(best_seq + noise)
            
            t_alloc = min(remaining * 0.25, 20)
            
            if ri % 3 == 0:
                a_opt, _ = gradient_descent_main(start, time_limit=t_alloc,
                                                  top_k=5, lr_init=0.003, momentum=0.95)
                try_seq(a_opt, f"refine_gd_{ri}")
            elif ri % 3 == 1:
                a_opt, _ = adam_softmax_optimize(start, time_limit=t_alloc,
                                                  lr_init=0.001, top_k=5)
                try_seq(a_opt, f"refine_adam_{ri}")
            else:
                a_opt, _ = peak_coordinate_descent(start, time_limit=t_alloc)
                try_seq(a_opt, f"refine_cd_{ri}")
    
    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
