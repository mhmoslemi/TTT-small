Is Q_k PSD? Q_k is the matrix with 1s on the k-th anti-diagonal. This is a permutation matrix (or a sub-matrix of one). A permutation matrix is not generally PSD. For example, the matrix [[0,1],[1,0]] has eigenvalues 1 and -1, so it's indefinite.

So a^T Q_k a is NOT convex in general. The problem is non-convex.

def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])

            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0
    
    return best_a, best_obj

def coordinate_descent(a0, time_limit, max_passes=200):
    """
    Coordinate descent for minimizing max_k (a*a)[k] on the simplex.
    For each coordinate i, find the optimal value using golden-section search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s if s > 1e-10 else np.ones(len(a)) / len(a)
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Precompute indices for fast convolution update
    # For coordinate i, the affected lags are:
    # k = 2i (quadratic)
    # k in [i, i+n-1] \ {2i} (linear, coefficient 2*a[k-i])
    
    pass_count = 0
    while time.monotonic() - t0 < time_limit and pass_count < max_passes:
        pass_count += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        conv = fast_convolve(a)
        improvements = 0
        
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            
            v = a[i]
            if v < 1e-15:
                # Try setting to small positive value
                x_try = max(a.max() * 0.01, 1e-6)
                S = 1.0 - v  # sum of others
                # Compute h(x_try)
                c_new = conv.copy()
                c_new[2*i] += x_try*x_try - v*v
                k_lo = max(0, i)
                k_hi = min(2*n-2, i+n-1)
                valid_k = np.arange(k_lo, k_hi+1)
                mask = valid_k != 2*i
                if mask.any():
                    c_new[valid_k[mask]] += 2*(x_try-v)*a[valid_k[mask]-i]
                h_try = c_new.max() / (S + x_try)**2
                h_cur = conv.max() / S**2 if S > 1e-10 else conv.max()
                
                if h_try < h_cur - 1e-14:
                    # Accept
                    a[i] = x_try
                    # Rescale others
                    scale = S / (S + x_try)
                    a[np.arange(n) != i] *= scale
                    a[i] /= (S + x_try) * scale * (S+x_try) / x_try  # this is getting messy
                    # Simpler: just set and renormalize
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    conv = fast_convolve(a)
                    improvements += 1
                continue
            
            # Golden section search for optimal x
            S = 1.0 - v  # sum of others (since a is normalized)
            
            # Define the objective function h(x)
            def h_of_x(x):
                if x < 0:
                    return 1e10
                c_new = conv.copy()
                c_new[2*i] += x*x - v*v
                k_lo = max(0, i)
                k_hi = min(2*n-2, i+n-1)
                valid_k = np.arange(k_lo, k_hi+1)
                mask = valid_k != 2*i
                if mask.any():
                    c_new[valid_k[mask]] += 2*(x-v)*a[valid_k[mask]-i]
                denom = (S + x)**2
                if denom < 1e-20:
                    return 1e10
                return c_new.max() / denom
            
            # Search range: [0, 2*v + a.max()] or something reasonable
            x_lo = 0.0
            x_hi = min(1.0, max(2*v + 0.01, a.max() * 3))
            
            # Golden section search
            golden = (np.sqrt(5) - 1) / 2
            c1 = x_hi - golden * (x_hi - x_lo)
            c2 = x_hi - (1 - golden) * (x_hi - x_lo)
            f1 = h_of_x(c1)
            f2 = h_of_x(c2)
            
            for _ in range(40):
                if f1 <= f2:
                    x_hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = x_hi - golden * (x_hi - x_lo)
                    f1 = h_of_x(c1)
                else:
                    x_lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = x_hi - (1 - golden) * (x_hi - x_lo)
                    f2 = h_of_x(c2)
            
            x_opt = (x_lo + x_hi) / 2
            f_opt = h_of_x(x_opt)
            
            # Current value
            f_cur = h_of_x(v)
            
            if f_opt < f_cur - 1e-14:
                # Update a[i] and renormalize
                a[i] = x_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv = fast_convolve(a)
                improvements += 1
        
        # Check improvement
        obj = conv.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        elif improvements == 0:
            # No improvement this pass, try a smaller perturbation
            if pass_count > 10:
                break
    
    return best_a, best_obj

def local_search_anneal(a0, time_limit, T_start=0.001, T_end=0.0001):
    """Simulated annealing local search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_obj = best_obj
    
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    it = 0
    
    total_iters_estimate = 20000
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        
        # Temperature schedule
        progress = min(1.0, it / total_iters_estimate)
        T = T_start * (T_end / T_start) ** progress
        
        # Random perturbation
        scale = T * 10
        pert = rng.standard_normal(n) * scale
        candidate = project_simplex(a + pert)
        
        c = fast_convolve(candidate)
        new_obj = c.max()
        delta = new_obj - cur_obj
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = candidate
            cur_obj = new_obj
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_a = a.copy()
    
    return best_a, best_obj

def lbfgs_softmax(a0, time_limit, T=0.05):
    """L-BFGS optimization using softmax parameterization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    # Initialize z such that softmax(z) ≈ a
    # z = log(a) + noise (to avoid log(0))
    z0 = np.log(np.maximum(a, 1e-10)) + np.random.default_rng(42).standard_normal(n) * 0.01
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    def objective(z):
        if time.monotonic() - t0 > time_limit:
            return 1e10
        a_z = softmax(z)
        conv = fast_convolve(a_z)
        # Log-sum-exp smoothed max
        max_c = conv.max()
        scaled = T * (conv - max_c)
        return max_c + (1.0/T) * np.log(np.exp(scaled).sum())
    
    def gradient(z):
        a_z = softmax(z)
        conv = fast_convolve(a_z)
        max_c = conv.max()
        scaled = T * (conv - max_c)
        exp_s = np.exp(scaled)
        p = exp_s / exp_s.sum()
        grad_a = hankel_grad(a_z, p)
        # Chain rule for softmax
        grad_z = a_z * (grad_a - np.dot(grad_a, a_z))
        return grad_z
    
    try:
        result = minimize(objective, z0, jac=gradient, method='L-BFGS-B',
                         options={'maxiter': 5000, 'maxfun': 5000,
                                  'ftol': 1e-15, 'gtol': 1e-12,
                                  'maxls': 50})
        a_opt = softmax(result.x)
        a_opt = a_opt / a_opt.sum()
        c = fast_convolve(a_opt)
        if c.max() < best_obj:
            best_obj = c.max()
            best_a = a_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")
    
    return best_a, best_obj
