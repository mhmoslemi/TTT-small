"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_conv_and_peaks(a, rel_tol=0.005, abs_tol=1e-12):
    """Compute convolution and identify active peaks."""
    conv = fast_convolve(a)
    M = conv.max()
    threshold = M * (1.0 - rel_tol) + abs_tol
    active = np.where(conv >= threshold)[0]
    return conv, M, active


def solve_lp_oracle(f, M, method='highs'):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method,
                      options={'presolve': True, 'max_time': 20})
        if res.success:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def pgd_optimize(a0, time_limit, top_k=20, lr_init=0.02, mom_init=0.9):
    """Projected gradient descent with momentum and adaptive step size."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = mom_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        obj = conv.max()

        # Subgradient from top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            pv = conv.copy()
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * a[k - idx_arr]

        v = mom * v - lr * sg
        best_candidate = a
        best_val = obj
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            candidate = project_simplex(a + scale * v)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.9

        if no_improve > 30:
            lr *= 0.95
        if no_improve > 200:
            rng = np.random.default_rng((it * 48271 + 17) % (2**31 - 1))
            noise = 0.002 + 0.003 * min(1.0, (no_improve - 200) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init
            mom = mom_init
            no_improve = 0

    return best_a, best_obj


def fw_optimize(a0, time_limit, use_symmetry=False):
    """Frank-Wolfe with LP oracle and adaptive M relaxation."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        g = solve_lp_oracle(f, M * m_exp)
        if g is None:
            m_exp *= 1.3
            if m_exp > 5.0:
                m_exp = 1.0
            no_improve += 1
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 20:
                break
            m_exp = min(m_exp * 1.02, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            lo = max(0.0, best_alpha - 0.02)
            hi = min(1.0, best_alpha + 0.02)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 40:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.004 * min(1.0, no_improve / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def kkt_refine(a0, time_limit, rel_tol=0.002, max_newton=200):
    """
    KKT active-set Newton refinement.
    Solves the full KKT system using block Newton with damped steps.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    outer_it = 0

    while time.monotonic() - t0 < time_limit:
        outer_it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        conv, M, active = compute_conv_and_peaks(a, rel_tol=rel_tol)
        S = np.where(a > 1e-10)[0]
        if len(S) < 2 or len(active) < 1:
            break

        m_s = len(S)
        p = len(active)

        # Solve F=0 for mu, lambda (linear system)
        # F_j = sum_{k in P} mu_k * 2*a[k-j] - lambda = 0 for j in S
        # A_sys is m_s x (p+1)
        A_sys = np.zeros((m_s, p + 1))
        for i, j in enumerate(S):
            A_sys[i, p] = -1.0  # lambda column (with negative sign)
            for k_idx, k in enumerate(active):
                idx = k - j
                if 0 <= idx < n:
                    # Interior: factor is 2, boundary: factor is 1
                    # Check if j is at boundary for this k
                    i_start = max(0, k - n + 1)
                    i_end = min(n - 1, k)
                    if i_start <= j <= i_end and (k - j) != j:
                        A_sys[i, k_idx] = 2.0 * a[idx]
                    elif i_start <= j <= i_end:
                        # j == k-j, i.e., 2j == k
                        A_sys[i, k_idx] = 2.0 * a[idx]
                    else:
                        # j might appear as the second factor only
                        # Check if k-j is in valid range but j is not as first factor
                        pass

        # Solve least squares
        try:
            sol, residuals, rank, sv = np.linalg.lstsq(A_sys, np.zeros(m_s), rcond=None)
        except np.linalg.LinAlgError:
            break

        mu = np.maximum(sol[:p], 0.0)
        lam = sol[p]
        mu_sum = mu.sum()
        if mu_sum < 1e-15:
            break
        mu = mu / mu_sum

        # Compute gradient direction from mu
        grad = np.zeros(n)
        for k_idx, k in enumerate(active):
            w = mu[k_idx]
            if w < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            # d/d a_j of c_k = 2*a[k-j] (interior) or a[k-j] (boundary)
            for jj in idx_arr:
                kk_idx = k - jj
                # Check if this is interior
                i_lo = max(0, k - n + 1)
                i_hi = min(n - 1, k)
                if kk_idx != jj and i_lo <= kk_idx <= i_hi:
                    grad[jj] += w * 2.0 * a[kk_idx]
                else:
                    grad[jj] += w * a[kk_idx]

        # Damped Newton step: move a in direction that reduces max conv
        # We want to reduce peaks, so move against the gradient
        best_candidate = a
        best_val = M
        for step in [0.001, 0.003, 0.01, 0.03, 0.05, 0.1, 0.2, 0.3]:
            candidate = a - step * grad
            candidate = np.maximum(candidate, 0.0)
            cs = candidate.sum()
            if cs < 1e-15:
                continue
            candidate = candidate / cs
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate

        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            # Success: maybe tighten the active set tolerance
            if best_obj < M * 0.999:
                rel_tol = max(0.0005, rel_tol * 0.8)
        else:
            # No improvement: loosen active set and try different peaks
            rel_tol = min(0.05, rel_tol * 1.5)
            # Try adding more peaks
            if outer_it % 10 == 0:
                # Slight perturbation to escape
                rng = np.random.default_rng(outer_it * 31 + 7)
                a = project_simplex(a + rng.standard_normal(n) * 0.001)

        # Check convergence
        if abs(M - best_val) < 1e-13 * M:
            if best_obj < M * (1.0 + 1e-9):
                # Try with tighter tolerance
                rel_tol = max(0.0001, rel_tol * 0.5)
                continue
            break

    return best_a, best_obj


def kkt_block_newton(a0, time_limit, rel_tol=0.003):
    """
    Block Newton for the full KKT system.
    Alternates between:
    - Solving for mu, lambda (given a): linear system
    - Solving for a, M (given mu): nonlinear system via Newton
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    outer = 0
    current_tol = rel_tol

    while time.monotonic() - t0 < time_limit:
        outer += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        conv, M, active = compute_conv_and_peaks(a, rel_tol=current_tol)
        S = np.where(a > 1e-10)[0]
        p = len(active)
        m_s = len(S)

        if m_s < 2 or p < 1:
            break

        # Step 1: Solve for mu, lambda
        A_sys = np.zeros((m_s, p + 1))
        for i, j in enumerate(S):
            A_sys[i, p] = -1.0
            for k_idx, k in enumerate(active):
                idx = k - j
                if 0 <= idx < n:
                    A_sys[i, k_idx] = 2.0 * a[idx]

        try:
            sol, _, rank, _ = np.linalg.lstsq(A_sys, np.zeros(m_s), rcond=None)
        except np.linalg.LinAlgError:
            break

        mu = np.maximum(sol[:p], 0.0)
        if mu.sum() < 1e-15:
            break
        mu_norm = mu / mu.sum()

        # Step 2: Solve for a, M using Newton
        # G_k = sum_i a_i a_{k-i} - M = 0 for k in active
        # H = sum a_j - 1 = 0
        # Unknowns: a (n vals, but constrained to S), M
        # We solve on the full space with a >= 0 constraint

        a_new = a.copy()
        M_new = M

        for newton_it in range(30):
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            conv_new = fast_convolve(a_new)
            resid_G = np.array([conv_new[k] - M_new for k in active])
            resid_H = a_new.sum() - 1.0

            # Check convergence
            if np.max(np.abs(resid_G)) < 1e-12 * max(1, M_new) and abs(resid_H) < 1e-12:
                break

            # Build Jacobian for [a, M]
            # dG_k / da_i = 2*a_{k-i} (interior)
            # dG_k / dM = -1
            # dH / da_i = 1
            # dH / dM = 0

            jac = np.zeros((p + 1, n + 1))
            for k_idx, k in enumerate(active):
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                for jj in range(js, je + 1):
                    idx = k - jj
                    if 0 <= idx < n:
                        jj_lo = max(0, k - n + 1)
                        jj_hi = min(n - 1, k)
                        if idx != jj and jj_lo <= idx <= jj_hi:
                            jac[k_idx, jj] = 2.0 * a_new[idx]
                        else:
                            jac[k_idx, jj] = a_new[idx]
                jac[k_idx, n] = -1.0  # dG/dM

            jac[p, :n] = 1.0  # dH/da
            jac[p, n] = 0.0  # dH/dM

            # Solve jac @ delta = -resid
            resid = np.concatenate([resid_G, [resid_H]])
            try:
                delta = np.linalg.solve(jac + 1e-14 * np.eye(p + 1), -resid)
            except np.linalg.LinAlgError:
                try:
                    delta, _, _, _ = np.linalg.lstsq(jac, -resid, rcond=None)
                except:
                    break

            # Damped step
            best_step = 0.0
            best_resid = np.max(np.abs(resid))
            for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
                a_trial = a_new + alpha * delta[:n]
                M_trial = M_new + alpha * delta[n]
                if M_trial < 0:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                cs = a_trial.sum()
                if cs < 1e-15:
                    continue
                a_trial = a_trial / cs
                M_trial = fast_convolve(a_trial).max()
                conv_trial = fast_convolve(a_trial)
                resid_trial = np.array([conv_trial[k] - M_trial for k in active])
                r_norm = max(np.max(np.abs(resid_trial)), abs(a_trial.sum() - 1))
                if r_norm < best_resid:
                    best_resid = r_norm
                    best_step = alpha
                    a_new = a_trial
                    M_new = M_trial

            if best_step < 1e-10:
                break

        # Evaluate
        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_a = a_new.copy()
            current_tol = max(0.0005, current_tol * 0.85)
        else:
            current_tol = min(0.05, current_tol * 1.3)
            if outer % 5 == 0:
                rng = np.random.default_rng(outer * 43 + 11)
                a = project_simplex(a + rng.standard_normal(n) * 0.002)

    return best_a, best_obj

        candidate = project_simplex(candidate)
        # Re-enforce symmetry after projection
        candidate = (candidate + candidate[::-1]) / 2.0
        candidate = candidate / candidate.sum()

        c = fast_convolve(candidate)
        val = c.max()

        if val < best_obj - 1e-15:
            best_obj = val
            best_a = candidate.copy()
            no_improve = 0
            a = candidate
        else:
            no_improve += 1
            lr *= 0.95
            v *= 0.8
            a = best_a

        if no_improve > 100:
            rng = np.random.default_rng(no_improve * 7 + 3)
            pert = rng.standard_normal(n) * 0.003
            pert = (pert + pert[::-1]) / 2.0
            a = project_simplex(best_a + pert)
            a = (a + a[::-1]) / 2.0
            a = a / a.sum()
            lr = 0.01
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_obj


def try_and_update(a, best_state, label=""):
    """Evaluate sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_state
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    best_seq, best_score = best_state
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return (best_seq, best_score)


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Hybrid search using FW, KKT Newton, and multi-start.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(parent)

    # Phase 1: Quick FW + PGD (~150s)
    print("\n=== Phase 1: Quick Optimization ===")
    t_start = time.monotonic()

    # Symmetrized parent
    sym_parent = (parent + parent[::-1]) / 2.0
    sym_parent = sym_parent / sym_parent.sum()
    (best_seq, best_score) = try_and_update(sym_parent, (best_seq, best_score), "sym")

    # Quick PGD
    remaining = deadline - time.monotonic() - 1300
    if remaining > 20:
        t_alloc = min(remaining, 50)
        a_opt, _ = pgd_optimize(best_seq, time_limit=t_alloc, top_k=20, lr_init=0.03)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "pgd1")

    # Quick FW
    remaining = deadline - time.monotonic() - 1200
    if remaining > 20:
        t_alloc = min(remaining, 60)
        a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "fw1")

    # Quick KKT
    remaining = deadline - time.monotonic() - 1100
    if remaining > 15:
        t_alloc = min(remaining, 40)
        a_opt, _ = kkt_refine(best_seq, time_limit=t_alloc, rel_tol=0.005)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "kkt1")

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Phase 2: Multi-n exploration (~250s)
    print("\n=== Phase 2: Multi-n Exploration ===")
    t_start = time.monotonic()

    n_best = len(best_seq)
    candidate_ns = sorted(set([n_best, n_best // 2, n_best * 2, 512, 768, 896, 1024]))
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 2048]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 900
        if remaining < 30:
            break

        if n_try == n_best:
            start = best_seq.copy()
        else:
            start = interp_sequence(best_seq, n_try)

        # PGD
        t_alloc = min(remaining, 40)
        a_opt, _ = pgd_optimize(start, time_limit=t_alloc, top_k=15, lr_init=0.02)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"pgd_n{n_try}")

        # FW
        remaining = deadline - time.monotonic() - 850
        if remaining > 15:
            a_opt2, _ = fw_optimize(a_opt, time_limit=min(remaining, 30))
            (best_seq, best_score) = try_and_update(a_opt2, (best_seq, best_score), f"fw_n{n_try}")

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Phase 3: Deep KKT refinement (~400s) - THE KEY PHASE
    print("\n=== Phase 3: Deep KKT Refinement ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    # Multiple KKT runs with different tolerances
    for rel_tol in [0.01, 0.005, 0.003, 0.002, 0.001]:
        remaining = deadline - time.monotonic() - 500
        if remaining < 20:
            break
        t_alloc = min(remaining, 80)
        print(f"  KKT refine, tol={rel_tol}, t={t_alloc:.0f}s")
        a_opt, _ = kkt_refine(best_seq, time_limit=t_alloc, rel_tol=rel_tol)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"kkt_t{rel_tol}")

    # Block Newton
    remaining = deadline - time.monotonic() - 450
    if remaining > 20:
        t_alloc = min(remaining, 100)
        print(f"  Block Newton, t={t_alloc:.0f}s")
        a_opt, _ = kkt_block_newton(best_seq, time_limit=t_alloc, rel_tol=0.003)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "block_newton")

    # FW + KKT alternating
    remaining = deadline - time.monotonic() - 400
    if remaining > 40:
        a_tmp = best_seq.copy()
        for _ in range(3):
            remaining = deadline - time.monotonic() - 380
            if remaining < 30:
                break
            a_tmp, _ = fw_optimize(a_tmp, time_limit=min(remaining, 30))
            (best_seq, best_score) = try_and_update(a_tmp, (best_seq, best_score), "fw_kkt_alt")
            remaining = deadline - time.monotonic() - 370
            if remaining < 15:
                break
            a_tmp, _ = kkt_refine(a_tmp, time_limit=min(remaining, 30), rel_tol=0.002)
            (best_seq, best_score) = try_and_update(a_tmp, (best_seq, best_score), "kkt_fw_alt")

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Phase 4: Symmetric optimization + perturbation restarts (~300s)
    print("\n=== Phase 4: Symmetry & Restarts ===")
    t_start = time.monotonic()

    # Symmetric optimization
    remaining = deadline - time.monotonic() - 300
    if remaining > 20:
        t_alloc = min(remaining, 60)
        a_opt, _ = symmetric_optimize(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "sym_opt")
        # Also try the non-symmetric version of the symmetric result
        a_opt_ns = (1.0 - 0.3) * best_seq + 0.3 * a_opt
        a_opt_ns = project_simplex(a_opt_ns)
        (best_seq, best_score) = try_and_update(a_opt_ns, (best_seq, best_score), "sym_mix")

    # Perturbation restarts with KKT
    for scale in [0.002, 0.005, 0.01, 0.02]:
        remaining = deadline - time.monotonic() - 200
        if remaining < 20:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc = min(remaining, 40)
        a_opt, _ = kkt_refine(f_pert, time_limit=t_alloc, rel_tol=0.005)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"kkt_pert_{scale}")

        # Also try FW on the perturbed
        remaining = deadline - time.monotonic() - 180
        if remaining > 15:
            a_opt2, _ = fw_optimize(f_pert, time_limit=min(remaining, 20))
            (best_seq, best_score) = try_and_update(a_opt2, (best_seq, best_score), f"fw_pert_{scale}")

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Phase 5: Intensive final refinement (~250s)
    print("\n=== Phase 5: Final Refinement ===")
    t_start = time.monotonic()

    # KKT with very tight tolerance
    for rel_tol in [0.001, 0.0005, 0.0003, 0.0002, 0.0001]:
        remaining = deadline - time.monotonic() - 100
        if remaining < 15:
            break
        t_alloc = min(remaining, 50)
        a_opt, _ = kkt_refine(best_seq, time_limit=t_alloc, rel_tol=rel_tol)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"kkt_tight_{rel_tol}")

    # Block Newton with tight tolerance
    remaining = deadline - time.monotonic() - 80
    if remaining > 15:
        a_opt, _ = kkt_block_newton(best_seq, time_limit=min(remaining, 50), rel_tol=0.001)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "block_tight")

    # Tiny perturbations
    for _ in range(4):
        remaining = deadline - time.monotonic() - 40
        if remaining < 15:
            break
        noise = rng.standard_normal(n_best) * 0.0005
        f_pert = project_simplex(best_seq + noise)
        t_alloc = min(remaining, 20)
        a_opt, _ = kkt_refine(f_pert, time_limit=t_alloc, rel_tol=0.001)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "tiny_pert")

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Phase 6: Last-ditch efforts (~100s)
    print("\n=== Phase 6: Final Polish ===")
    t_start = time.monotonic()

    # One more FW-KKT cycle
    remaining = deadline - time.monotonic() - 30
    if remaining > 20:
        a_tmp, _ = fw_optimize(best_seq, time_limit=min(remaining, 30))
        (best_seq, best_score) = try_and_update(a_tmp, (best_seq, best_score), "final_fw")

    remaining = deadline - time.monotonic() - 20
    if remaining > 10:
        a_opt, _ = kkt_refine(best_seq, time_limit=min(remaining, 20), rel_tol=0.0005)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "final_kkt")

    # Symmetry check
    sym_final = (best_seq + best_seq[::-1]) / 2.0
    sym_final = sym_final / sym_final.sum()
    (best_seq, best_score) = try_and_update(sym_final, (best_seq, best_score), "final_sym")

    # Asymmetric nudge
    nudge = best_seq + rng.standard_normal(len(best_seq)) * 0.0003
    nudge = project_simplex(nudge)
    (best_seq, best_score) = try_and_update(nudge, (best_seq, best_score), "final_nudge")

    print(f"  Phase 6 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
