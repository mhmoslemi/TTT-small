def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])

One more consideration: the MWU algorithm. Let me be more precise about the update.

In a standard multiplicative weight update for a zero-sum game:
- Row player (us) chooses p from a set P (the simplex)
- Column player (adversary) chooses k from {0, ..., 2n-2}
- Payoff matrix: M[p, k] = (p*p)[k]

We want to find p that minimizes max_k M[p,k].

The MWU (or Hedge) algorithm for the column player:
- Maintain weights W_k for each column k
- At each iteration t:
  - Column player plays q_t = W_t / sum(W_t)
  - Row player responds with p_t = argmin_{p in P} sum_k q_t[k] * M[p,k]
  - Update: W_{t+1}[k] = W_t[k] * exp(-eta * M[p_t, k])

The regret bound gives: max_k M[p_avg, k] <= V + O(sqrt(log(K)/T)) where V is the game value.

For our problem, K = 2n-1, so log(K) ~ log(2n). For n=768, log(K) ~ 11.

To get error <- 0.001 (i.e., 1.5017 vs target 1.5027), we need sqrt(11/T) < 0.001, so T > 11M. That's too many iterations.

But in practice, the convergence is much faster than the worst-case bound. And we can use the "best iterate so far" which often converges faster.

Also, we can use a larger eta for faster convergence at the cost of worse worst-case guarantees. In practice, eta = 1 to 5 works well.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Soft alternating minimization (primal QP + dual softmax update)
- FISTA with log-sum-exp smoothing
- Exact subgradient polish with adaptive peak weights
- Multi-n exploration with warm starts
"""

import numpy as np
from scipy.special import softmax
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_multiply(q, p, fft_size=None):
    """Compute H(q) * p where H(q)[i,j] = q[i+j].
    This equals the cross-correlation (q ⋆ p)[i] = sum_j q[i+j]*p[j].
    """
    n = len(p)
    if fft_size is None:
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
    p_rev = p[::-1]
    Q = np.fft.rfft(q, fft_size)
    P = np.fft.rfft(p_rev, fft_size)
    result = np.fft.irfft(Q * P, fft_size)[:2 * n - 1]
    return result[n - 1 : 2 * n - 1]  # indices n-1 to 2n-2


def hankel_qp_solve(q, p_init, n, max_iters=300, tol=1e-12):
    """Solve min p^T H(q) p s.t. p >= 0, sum(p) = 1 using accelerated projected gradient."""
    p = p_init.astype(np.float64).copy()
    p = project_simplex(p)
    
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    lr = 0.25  # 1 / (2 * ||H||) where ||H|| <= 2
    
    # Nesterov momentum
    t = 1.0
    y = p.copy()
    
    for it in range(max_iters):
        # Compute gradient at y: 2 * H(q) * y
        Hy = hankel_multiply(q, y, fft_size)
        grad_y = 2.0 * Hy
        
        # Gradient step
        x_new = y - lr * grad_y
        x_new = project_simplex(x_new)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - p)
        y = np.maximum(y, 0.0)
        y = y / y.sum()
        
        # Check convergence
        if np.max(np.abs(x_new - p)) < tol:
            p = x_new
            break
        p = x_new
        t = t_new
    
    return p


def soft_alternating(n, p_init, time_limit, verbose=False):
    """Soft alternating: primal QP + dual softmax."""
    p = p_init.astype(np.float64).copy()
    p = project_simplex(p)
    q = np.ones(2 * n - 1) / (2 * n - 1)
    beta = 50.0
    best_p = p.copy()
    best_obj = fast_convolve(p).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Step 1: Solve QP
        p_new = hankel_qp_solve(q, p, n, max_iters=200, tol=1e-13)
        
        # Evaluate
        conv = fast_convolve(p_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_p = p_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Step 2: Update q (softmax)
        # Use a temperature that increases over time
        q = softmax(beta * conv)
        q = q / q.sum()
        
        # Increase beta
        beta *= 1.02
        if beta > 1e5:
            beta = 1e5
        
        p = p_new
        
        if verbose and it % 100 == 0:
            print(f"    SA it={it}, obj={best_obj:.10f}, beta={beta:.1f}")
        
        if no_improve > 200:
            # Reset with slightly higher beta
            no_improve = 0
    
    return best_p, best_obj


def fista_smoothed(n, p_init, time_limit, mu_init=50.0):
    """FISTA on log-sum-exp smoothed objective."""
    p = p_init.astype(np.float64).copy()

    mu = mu_init
    best_p = p.copy()
    best_obj = fast_convolve(p).max()

    t_fista = 1.0
    y = p.copy()
    no_improve = 0

        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv_y = fast_convolve(y)
        
        # Compute weights (softmax with temperature 1/mu)
        # f_mu(p) = (1/mu) * log(sum exp(mu * (p*p)[k]))
        # grad f_mu = sum_k w_k * grad (p*p)[k] where w_k = exp(mu*conv[k]) / sum exp(mu*conv)
        shifted = mu * (conv_y - conv_y.max())  # for numerical stability
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()

        # = 2 * (w ⋆ p)[i]

        x_new = y - lr * grad
        x_new = project_simplex(x_new)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista))
        y = x_new + ((t_fista - 1.0) / t_new) * (x_new - p)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = np.ones(n) / n
        else:
            y = y / ys
        
        p = x_new
        t_fista = t_new
        
        # Periodic exact evaluation
        if it % 50 == 0:
            conv_p = fast_convolve(p)
            obj = conv_p.max()

                best_obj = obj
                best_p = p.copy()
                no_improve = 0
            else:
                no_improve += 1

            mu *= 2.0
            t_fista = 1.0
            y = p.copy()

            mu = 1e6
    
    # Final evaluation
    conv_p = fast_convolve(p)
    obj = conv_p.max()
    if obj < best_obj:

def exact_subgradient(p0, time_limit, top_k=15):
    """Exact subgradient with adaptive peak weights and line search."""
    p = p0.astype(np.float64).copy()
    p = project_simplex(p)
    n = len(p)
    best_p = p.copy()
    best_obj = fast_convolve(p).max()
    
    t0 = time.monotonic()
    it = 0
    lr = 0.1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(p)
        max_val = conv.max()
        
        # Select top peaks with softmax weights
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            pw = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            # Softmax with temperature
            temp = 0.01 * max_val + 1e-10
            pw = np.exp(pv / temp)
            pw = pw / pw.sum()
        
        # Compute subgradient
        grad = np.zeros(n)
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        # Use the hankel_multiply trick
        # grad = 2 * (w_peaks ⋆ p) where w_peaks is the weight vector over conv indices
        w_full = np.zeros(2 * n - 1)
        for idx, k in enumerate(peaks):
            w_full[int(k)] = pw[idx]
        grad = 2.0 * hankel_multiply(w_full, p, fft_size)
        
        # Direction (projected)
        direction = project_simplex(p - lr * grad) - p
        
        # Line search (exponential backtracking)
        best_alpha = 0.0
        best_val = max_val
        alpha = 1.0
        for _ in range(8):
            cand = p + alpha * direction
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs < 1e-15:
                alpha *= 0.5
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
            alpha *= 0.5
        
        if best_alpha > 1e-12:
            p = p + best_alpha * direction
            p = np.maximum(p, 0.0)
            p = p / p.sum()
            obj = fast_convolve(p).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_p = p.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Adapt learning rate
        if no_improve > 20:
            lr *= 0.9
        if lr < 0.001:
            # Restart with perturbation
            rng = np.random.default_rng(it * 7919 + 13)
            p = project_simplex(best_p + rng.standard_normal(n) * 0.002)
            lr = 0.1
            no_improve = 0
        if no_improve > 200:
            break
    
    return best_p, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # From parent (resampled)
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Edge-heavy
    edge_w = 1.0 / (np.minimum(x, n - 1 - x).astype(float) + 1.0) ** 1.5
    inits.append(edge_w / edge_w.sum())
    
    # Two-peak (helps reduce center)
    g2 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())
    
    # Anti-correlation pattern
    t = x / max(n - 1, 1)
    asym = (1.0 + 0.3 * np.cos(4 * np.pi * t)) ** 2 + 0.05
    inits.append(asym / asym.sum())
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    return inits
