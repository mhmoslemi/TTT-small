def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def scipy_smooth(a0, time_limit, T=0.001, verbose=False):
    """Use scipy SLSQP to minimize log-sum-exp surrogate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    
    def obj_func(x):
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-15:
            return 1e10
        x = x / s
        conv = fast_convolve(x)
        scaled = conv / T
        return float(logsumexp(scaled) * T)
    
    def grad_func(x):
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-15:
            return np.ones(n)
        x = x / s
        conv = fast_convolve(x)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        # Gradient of logsumexp(T * conv/T) = sum_k p[k] * grad conv[k]
        # grad conv[k] wrt x[i] = 2*x[k-i] (with bounds)
        # But we also need to account for the normalization... 
        # Actually, since we're optimizing over the simplex, let's just compute
        # the gradient of the unnormalized objective and let the constraint handle it.
        # grad[i] = 2 * sum_k p[k] * x[k-i] (with bounds)
        grad = 2.0 * np.correlate(p, x, mode='valid')
        return grad
    
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - 1.0, 'jac': lambda x: np.ones(n)}]
    bounds = [(0, None)] * n
    
    # Run SLSQP
    result = minimize(obj_func, a, method='SLSQP', jac=grad_func,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 200, 'ftol': 1e-12})
    
    if result.success and result.fun < 1e9:
        a_opt = np.maximum(result.x, 0)
        a_opt = a_opt / a_opt.sum()
        obj = fast_convolve(a_opt).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
    
    return best_a, best_obj

            break

        cmax = conv.max()

        k_use = min(n_peaks, len(conv))
        peak_indices = np.argpartition(conv, -k_use)[-k_use:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        # Use a simple approach: for the dominant peak, shift mass away.

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = np.arange(js, je + 1)
        non_contrib = np.setdiff1d(np.arange(n), contrib)
        
        if len(non_contrib) == 0:
            break

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_mv(p, v):
    """Compute Hankel matrix-vector product: (H@v)[i] = sum_j p[i+j]*v[j]."""
    return np.correlate(p, v, mode='valid')

def compute_smooth_grad(a, T):
    """Compute gradient and peak weights for logsumexp smoothed objective."""
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()
    g = (2.0 / T) * hankel_mv(p, a)
    return g, p, conv

def hessian_mv(v, p, g, T):
    """Apply Hessian of smoothed objective to vector v.
    H = (2/T)*P_hankel + g*g^T

    return (2.0 / T) * hankel_mv(p, v) + np.dot(g, v) * g

def _cg_solve(matvec, b, p, g, T, lam, max_iters=50):
    """Conjugate gradient solver."""
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p_vec = r.copy()
    rs = np.dot(r, r)
    for _ in range(max_iters):
        Ap = matvec(p_vec, p, g, T, lam)
        alpha = rs / (np.dot(p_vec, Ap) + 1e-30)
        x += alpha * p_vec
        r -= alpha * Ap
        rs_new = np.dot(r, r)
        if rs_new < 1e-14 * rs:
            break
        p_vec = r + (rs_new / (rs + 1e-30)) * p_vec
        rs = rs_new
    return x

def peak_equalize_fast(a0, time_limit, n_peaks=8, verbose=False):

        # Targeted moves: reduce worst contributor, add to best non-contributor
        k0 = peak_idx[0]
        js0 = max(0, k0 - n + 1)
        je0 = min(n - 1, k0)
        contrib0 = np.arange(js0, je0 + 1)
        non_contrib0 = np.setdiff1d(np.arange(n), contrib0)

            # Sort contributors by damage * value
            c_score = damage[contrib0] * a[contrib0]
            nc_score = damage[non_contrib0]
            
            worst_sources = contrib0[np.argsort(c_score)[::-1][:5]]
            best_targets = non_contrib0[np.argsort(nc_score)[:5]]
            
            for src in worst_sources:
                for tgt in best_targets:
                    for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                        shift = a[src] * frac
                        if shift < 1e-12:
                            break
                        a_try = a.copy()
                        a_try[src] -= shift
                        a_try[tgt] += shift
                        a_try = np.maximum(a_try, 0)
                        a_try = a_try / a_try.sum()
                        obj = fast_convolve(a_try).max()
                        if obj < best_obj - 1e-15:
                            a = a_try
                            best_obj = obj
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        # Random moves if no targeted improvement
        if not improved:
            for _ in range(20):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                frac = rng.uniform(0.002, 0.03)
                # Move from higher damage to lower damage preferred
                if damage[i] < damage[j]:
                    i, j = j, i
                shift = a[i] * frac
                if shift < 1e-13:
                    continue
                a_try = a.copy()
                a_try[i] -= shift
                a_try[j] += shift
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                obj = fast_convolve(a_try).max()
                if obj < best_obj - 1e-15:
                    a = a_try
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved and it > 500:
            break
    
    return best_a, best_obj

def multiplicative_weight(a0, time_limit, eta_init=0.05, p=1.0, verbose=False):
    """Multiplicative weight / exponentiated gradient update."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    eta = eta_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        it += 1
        
        conv = fast_convolve(a)
        cmax = conv.max()
        
        # Compute weights: w[k] = (conv[k]/cmax)^p
        w = (conv / cmax) ** p
        w_sum = w.sum()
        if w_sum < 1e-15:
            break
        w = w / w_sum
        
        # Gradient: G[i] = 2 * sum_k w[k] * a[k-i]
        G = 2.0 * hankel_mv(w, a)
        
        # Multiplicative update
        a_new = a * np.exp(-eta * G)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                eta *= 0.85
            if no_improve > 50:
                rng = np.random.default_rng((it * 7919 + 3) % (2**31))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
                eta = eta_init
                no_improve = 0
        
        eta = min(eta, eta_init)
        
        if verbose and it % 200 == 0:
            print(f"      MW it={it}, obj={best_obj:.10f}, eta={eta:.6f}")
    
    return best_a, best_obj

def coarse_to_fine(coarse_a, n_fine, time_limit):
    """Lift a coarse solution to fine resolution and optimize."""
    n_coarse = len(coarse_a)
    # Interpolate to fine resolution
    fine_a = np.interp(np.linspace(0, 1, n_fine),
                       np.linspace(0, 1, n_coarse), coarse_a)
    fine_a = np.maximum(fine_a, 0)
    fine_a = fine_a / fine_a.sum()
    
    # Quick Nesterov to refine
    result, _ = nesterov_smooth(fine_a, time_limit=time_limit, T_init=0.003, lr_init=0.01)
    return result

def interpolate_sequence(a, n_target):
    """Interpolate a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    result = np.interp(np.linspace(0, 1, n_target),
                       np.linspace(0, 1, n_src), a)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    return result

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid search:
    Phase 1: Load parent & quick n-exploration (~150s)
    Phase 2: Newton + Nesterov on best candidates (~600s)
    Phase 3: Peak equalization + multiplicative weights (~500s)
    Phase 4: Coarse-to-fine & final polish (~500s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(768) / 768
    
    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")
    
    # === Phase 1: Quick n-exploration (~150s) ===
    print("\n=== Phase 1: Quick n-exploration ===")
    t1 = time.monotonic()
    
    # Try a range of n values around the parent
    candidate_ns = sorted(set([
        n_parent,
        max(128, n_parent - 128),
        n_parent + 128,
        max(128, n_parent - 256),
        n_parent + 256,
        512, 768, 1024, 1536, 2048,
    ]))
    candidate_ns = [n for n in candidate_ns if 64 <= n <= 4096]
    
    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1000
        if remaining < 20:
            break
        
        start = interpolate_sequence(parent, n_try)
        t_alloc = min(remaining, 12)
        print(f"  Quick Nesterov n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = nesterov_smooth(start, time_limit=t_alloc, T_init=0.01, lr_init=0.02)
        try_seq(a_opt, f"quick_n{n_try}")
    
    print(f"  Phase 1 done. Best: {best_score:.10f} (t={time.monotonic()-t1:.0f}s)")
    
    # === Phase 2: Deep Newton + Nesterov (~600s) ===
    print("\n=== Phase 2: Newton + Nesterov ===")
    t2 = time.monotonic()
    
    if best_seq is not None:
        # Newton method (exploits Hessian structure)
        remaining = deadline - time.monotonic() - 700
        if remaining > 30:
            t_alloc = min(remaining, 200)
            print(f"  Newton T=0.01, t={t_alloc:.0f}s")
            a_opt, _ = newton_smooth(best_seq, time_limit=t_alloc, T_init=0.01)
            try_seq(a_opt, "newton_T0.01")
            
            # Continue with lower T
            remaining = deadline - time.monotonic() - 500
            if remaining > 20:
                t_alloc = min(remaining, 150)
                print(f"  Newton T=0.003, t={t_alloc:.0f}s")
                a_opt, _ = newton_smooth(best_seq, time_limit=t_alloc, T_init=0.003)
                try_seq(a_opt, "newton_T0.003")
        
        # Nesterov with different settings
        for T_i, lr_i in [(0.005, 0.015), (0.002, 0.008), (0.01, 0.025)]:
            remaining = deadline - time.monotonic() - 400
            if remaining < 15:
                break
            t_alloc = min(remaining, 120)
            print(f"  Nesterov T={T_i}, lr={lr_i}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_smooth(best_seq, time_limit=t_alloc, T_init=T_i, lr_init=lr_i)
            try_seq(a_opt, f"nest_T{T_i}")
    
    # Try Newton from parent with different n
    for n_try in [n_parent, n_parent + 128]:
        if n_try == len(best_seq) if best_seq is not None else False:
            continue
        remaining = deadline - time.monotonic() - 350
        if remaining < 15:
            break
        start = interpolate_sequence(parent, n_try)
        t_alloc = min(remaining, 80)
        print(f"  Newton n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = newton_smooth(start, time_limit=t_alloc, T_init=0.01)
        try_seq(a_opt, f"newton_n{n_try}")
    
    print(f"  Phase 2 done. Best: {best_score:.10f} (t={time.monotonic()-t2:.0f}s)")
    
    # === Phase 3: Peak equalization + Multiplicative weight (~500s) ===
    print("\n=== Phase 3: Peak Equalization + MW ===")
    t3 = time.monotonic()
    
    if best_seq is not None:
        # Peak equalization
        remaining = deadline - time.monotonic() - 300
        if remaining > 20:
            t_alloc = min(remaining, 150)
            print(f"  Peak equalize (8 peaks), t={t_alloc:.0f}s")
            a_opt, _ = peak_equalize_fast(best_seq, time_limit=t_alloc, n_peaks=8)
            try_seq(a_opt, "peak_eq_8")
        
        remaining = deadline - time.monotonic() - 250
        if remaining > 20:
            t_alloc = min(remaining, 100)
            print(f"  Peak equalize (16 peaks), t={t_alloc:.0f}s")
            a_opt, _ = peak_equalize_fast(best_seq, time_limit=t_alloc, n_peaks=16)
            try_seq(a_opt, "peak_eq_16")
        
        # Multiplicative weight
        for p_val in [1.0, 2.0, 0.5]:
            remaining = deadline - time.monotonic() - 150
            if remaining < 15:
                break
            t_alloc = min(remaining, 80)
            print(f"  MW p={p_val}, t={t_alloc:.0f}s")
            a_opt, _ = multiplicative_weight(best_seq, time_limit=t_alloc, eta_init=0.03, p=p_val)
            try_seq(a_opt, f"mw_p{p_val}")
    
    print(f"  Phase 3 done. Best: {best_score:.10f} (t={time.monotonic()-t3:.0f}s)")
    
    # === Phase 4: Coarse-to-fine + Final polish (~400s) ===
    print("\n=== Phase 4: Coarse-to-Fine + Final Polish ===")
    t4 = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Coarse-to-fine: try lifting to 2x
        if n_best <= 2048:
            remaining = deadline - time.monotonic() - 200
            if remaining > 30:
                n_fine = n_best * 2
                if n_fine <= 8192:
                    t_alloc = min(remaining, 100)
                    print(f"  Coarse-to-fine {n_best}->{n_fine}, t={t_alloc:.0f}s")
                    a_fine = coarse_to_fine(best_seq, n_fine, time_limit=t_alloc)
                    try_seq(a_fine, f"c2f_{n_fine}")
        
        # Fine Nesterov polish
        for T_i in [0.001, 0.0005, 0.002]:
            remaining = deadline - time.monotonic() - 80
            if remaining < 10:
                break
            t_alloc = min(remaining, 60)
            print(f"  Final Nesterov T={T_i}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_smooth(best_seq, time_limit=t_alloc, T_init=T_i, lr_init=0.005)
            try_seq(a_opt, f"final_T{T_i}")
        
        # Re-interpolate to several n and do quick Newton
        for n_try in [768, 1024, 1536]:
            if n_try == n_best:
                continue
            remaining = deadline - time.monotonic() - 50
            if remaining < 10:
                break
            start = interpolate_sequence(best_seq, n_try)
            t_alloc = min(remaining, 40)
            print(f"  Re-interp Newton n={n_try}, t={t_alloc:.0f}s")
            a_opt, _ = newton_smooth(start, time_limit=t_alloc, T_init=0.005)
            try_seq(a_opt, f"reinterp_n{n_try}")
        
        # Final perturbation attempts
        for _ in range(3):
            remaining = deadline - time.monotonic() - 10
            if remaining < 5:
                break
            noise = 0.001 + 0.001 * _
            f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * noise)
            t_alloc = min(remaining, 20)
            a_opt, _ = nesterov_smooth(f_pert, time_limit=t_alloc, T_init=0.001, lr_init=0.003)
            try_seq(a_opt, f"perturb_{_}")
    
    print(f"  Phase 4 done. Best: {best_score:.10f} (t={time.monotonic()-t4:.0f}s)")
    
    # === Phase 5: Last-ditch exploration ===
    print("\n=== Phase 5: Last-ditch ===")
    
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            # Try a completely different n with full optimization pipeline
            n_alt = max(256, len(best_seq) // 2)
            start = interpolate_sequence(best_seq, n_alt)
            t_alloc = min(remaining, 40)
            a_opt, _ = newton_smooth(start, time_limit=t_alloc, T_init=0.01)
            a_opt2, _ = nesterov_smooth(a_opt, time_limit=min(remaining - t_alloc - 5, 30), T_init=0.002)
            try_seq(a_opt2, f"alt_n{n_alt}")
        
        # One more peak equalization round
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_alloc = min(remaining, 30)
            a_opt, _ = peak_equalize_fast(best_seq, time_limit=t_alloc, n_peaks=4)
            try_seq(a_opt, "final_peak_eq")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
