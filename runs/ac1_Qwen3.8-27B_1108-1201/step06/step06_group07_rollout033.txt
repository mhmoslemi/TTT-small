import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def project_nonneg_sum1(v):
    """Project onto {x >= 0, sum(x) = 1} (same as simplex for our case)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v / s


def build_conv_matrix(n):
    """Build sparse matrix for convolution constraints.
    Returns (row_idx, col_idx, values_func) where the matrix A has
    A[k, j] = 1 if max(0, k-n+1) <= j <= min(n-1, k), else 0.
    The actual constraint is (f * g)[k] = sum_j f[k-j] * g[j] <= M
    So for fixed f, A[k,j] = f[k-j].
    """
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_sca_lp(f, M, n, row_idx, col_idx):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g (unnormalized) or None.
    """
    k_arr = row_idx
    i_arr = col_idx
    vals = f[k_arr - i_arr]
    
    A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def spectral_equalize(a, time_limit, alpha=0.3):
    """
    Frequency-domain spectral equalization.
    Iteratively flatten the power spectrum |FFT(a)|^2 and project back.
    """
    a = project_nonneg_sum1(a.copy())
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Compute power spectrum
        A = np.fft.rfft(a, fft_size)
        P = (A * np.conj(A)).real  # |A|^2
        
        # Target: make P flat (for indices 1 to fft_size//2)
        # Keep P[0] = 1 (DC component = sum(a) = 1)
        P_flat = P.copy()
        P_nonzero = P[1:]
        mean_P = P_nonzero.mean()
        
        # Gradually flatten
        P_target = P.copy()
        P_target[1:] = (1 - alpha) * P[1:] + alpha * mean_P
        # Ensure non-negative
        P_target = np.maximum(P_target, 0)
        
        # New magnitude, keep phase
        mag_new = np.sqrt(P_target)
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        
        # Inverse FFT
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Project onto non-negative simplex
        a_new = project_simplex(a_new)
        
        obj = fast_convolve(a_new).max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            # Try different alpha
            alpha *= 0.8
            if alpha < 0.01:
                alpha = 0.5
            no_improve = 0
        
        a = a_new
    
    return best_a, best_obj


def coordinate_peak_descent(a, time_limit, n_steps=20):
    """
    Coordinate descent: for each coordinate, find the optimal 1D move
    that reduces the maximum peak.
    """
    a = project_nonneg_sum1(a.copy())
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        improved = False
        # Shuffle coordinates
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            if a[j] < 1e-10:
                continue
            
            # Try moving mass from j to some other coordinate
            # The effect on conv[k]: delta * (a[k-j'] - a[k-j]) for each k
            # We want to reduce the max peak
            
            # Pick target: a coordinate j' that is "far" from j (reduces peak contribution)
            # Or pick randomly
            j_prime = (j + n // 2) % n  # Far coordinate
            if j_prime == j:
                j_prime = (j + 1) % n
            
            # Binary search for best fraction to move
            lo, hi = 0.0, 0.5 * a[j]
            base_val = fast_convolve(a).max()
            
            if base_val < best_obj - 1e-14:
                best_obj = base_val
                best_a = a.copy()
            
            best_frac = 0.0
            best_val = base_val
            
            for _ in range(8):
                mid = (lo + hi) / 2
                a_test = a.copy()
                a_test[j] -= mid
                a_test[j_prime] += mid
                a_test = np.maximum(a_test, 0)
                st = a_test.sum()
                if st < 1e-15:
                    hi = mid
                    continue
                a_test = a_test / st
                val = fast_convolve(a_test).max()
                if val < best_val:
                    best_val = val
                    best_frac = mid
                    lo = mid
                else:
                    hi = mid
            
            if best_frac > 1e-10:
                a[j] -= best_frac
                a[j_prime] += best_frac
                a = np.maximum(a, 0)
                a = a / a.sum()
                improved = True
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3:
                break
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Improved SCA with LP oracle, adaptive M, and reduced constraints
- Bisection on target M for tighter convergence
- Frequency-domain spectral equalization for diversity
- Deflation LP for targeted peak reduction
- Multi-n exploration with diverse initializations
- Bisection-based M tightening in the final phase

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_indices(n):
    """Precompute row/col indices for convolution matrix entries."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_sca_lp(f, M, n, row_idx, col_idx, time_limit_lp=20):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g (unnormalized) or None.
    """
    k_arr = row_idx
    i_arr = col_idx
    vals = f[k_arr - i_arr]

    A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit_lp})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit_lp})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def sca_optimize(f0, time_limit, use_reduced=False, reduced_frac=0.5, verbose=False):
    """
    SCA optimization with adaptive M and optional reduced constraints.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    row_idx, col_idx = build_conv_indices(n)

    conv = fast_convolve(f)
    M = conv.max() * 1.02
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lp_solves = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        # Decide whether to use reduced LP
        if use_reduced and n > 1000:
            conv_now = fast_convolve(f)
            threshold = conv_now.max() * (1.0 - reduced_frac * 0.3)
            active_k = set(np.argwhere(conv_now >= threshold).flatten().tolist())
            if len(active_k) < max(20, n // 4):
                active_k = set(np.argpartition(conv_now, -max(20, n // 4))[-max(20, n // 4):].tolist())
            g_raw = solve_sca_lp_reduced(f, M, n, active_k, row_idx, col_idx,
                                         time_limit_lp=min(10, remaining - 1))
        else:
            g_raw = solve_sca_lp(f, M, n, row_idx, col_idx,
                                 time_limit_lp=min(20, remaining - 1))

        if g_raw is not None:
            lp_solves += 1

        if g_raw is None:
            M *= 1.05
            no_improve += 1
            if no_improve > 30:
                conv_now = fast_convolve(f)
                M = conv_now.max() * 1.01
                no_improve = 0
            continue

        g = g_raw / g_raw.sum()

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 25:
                rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
                perturb_scale = 0.002 + 0.002 * min(1.0, no_improve / 50.0)
                f = project_simplex(best_f + rng.standard_normal(n) * perturb_scale)
                M = fast_convolve(f).max() * 1.01
                no_improve = 0
            continue

        # Line search over blending alphas
        best_val = fast_convolve(f).max()
        best_ft = f.copy()

        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_ft = ft.copy()

        if best_val < best_obj - 1e-15:
            f = best_ft
            best_obj = best_val
            best_f = f.copy()
            # Tighten M
            M = best_val * 0.999
            no_improve = 0
        else:
            no_improve += 1
            M *= 0.995

        if no_improve > 50:
            rng = np.random.default_rng((it * 31831 + 7) % (2**31 - 1))
            scale = 0.003 + 0.004 * min(1.0, (no_improve - 50) / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max() * 1.01
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    SCA it={it}, obj={best_obj:.10f}, M={M:.10f}, lp={lp_solves}")

    return best_f, best_obj


def bisection_tighten(a0, time_limit, tol=1e-6, verbose=False):
    """
    Use bisection on M to find the tightest achievable M.
    For a given M, check if SCA converges to a point with max_conv <= M.
    """
    n = len(a0)
    row_idx, col_idx = build_conv_indices(n)

    # Initial bounds
    conv0 = fast_convolve(a0)
    M_upper = conv0.max() * 1.001
    M_lower = 1.0 / (2 * n - 1) * 0.99  # theoretical minimum times a bit

    best_f = a0.copy()
    best_M = M_upper

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5.0:
            break

        M_mid = (M_lower + M_upper) / 2.0
        alloc = min(remaining - 2, 30.0)

        # Try to achieve M_mid using SCA
        res_f, res_M = _sca_fixed_M(a0, M_mid, n, row_idx, col_idx, alloc)

        if res_M <= M_mid + 1e-10:
            # Feasible: we can achieve M_mid
            best_M = min(best_M, res_M)
            best_f = res_f
            M_lower = M_mid
        else:
            M_upper = M_mid

        if abs(M_upper - M_lower) < tol:
            break

        if verbose and it % 5 == 0:
            print(f"    Bisection it={it}, M_lower={M_lower:.8f}, M_upper={M_upper:.8f}, best={best_M:.8f}")

    return best_f, best_M


def _sca_fixed_M(f0, target_M, n, row_idx, col_idx, time_limit):
    """
    Run SCA with a fixed target M. Returns (best_f, best_achieved_M).
    """
    f = f0.copy()
    f = f / f.sum()

    t0 = time.monotonic()
    best_M = fast_convolve(f).max()
    best_f = f.copy()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        g_raw = solve_sca_lp(f, target_M, n, row_idx, col_idx,
                             time_limit_lp=min(10, remaining - 0.5))

        if g_raw is None:
            # Infeasible at this M
            break

        g = g_raw / g_raw.sum()

        if np.max(np.abs(g - f)) < 1e-12:
            break

        # Try blends
        best_local = fast_convolve(f).max()
        best_local_f = f.copy()

        for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_local:
                best_local = val
                best_local_f = ft.copy()

        if best_local < best_M - 1e-14:
            f = best_local_f
            best_M = best_local
            best_f = f.copy()
        else:
            f = best_local_f
            # Check if we've converged
            if abs(fast_convolve(f).max() - best_M) < 1e-12:
                break

    return best_f, best_M


def spectral_equalize(a0, time_limit, base_alpha=0.2):
    """
    Frequency-domain spectral equalization.
    Flatten the power spectrum and project back to non-negative simplex.
    """
    a = project_simplex(a0.astype(np.float64).copy())
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    alpha = base_alpha

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        A = np.fft.rfft(a, fft_size)
        P = (A.real ** 2 + A.imag ** 2)

        # Flatten non-DC components
        if len(P) > 1:
            P_non_dc = P[1:]
            mean_P = P_non_dc.mean()
            P_new = P.copy()
            P_new[1:] = (1.0 - alpha) * P[1:] + alpha * mean_P
            P_new = np.maximum(P_new, 0.0)
        else:
            P_new = P

        mag_new = np.sqrt(P_new)
        phase = np.angle(A)
        A_new = mag_new * (np.cos(phase) + 1j * np.sin(phase))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = project_simplex(a_new)

        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 40:
            alpha = alpha * 0.7
            if alpha < 0.02:
                alpha = 0.5
            no_improve = 0
            # Perturb
            rng = np.random.default_rng(it * 48271 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            no_improve = 0

        if no_improve > 150:
            break

    return best_a, best_obj


def deflation_refine(a0, time_limit, max_peaks=8, eps=0.005):
    """
    Deflation LP: target top peaks, find delta that reduces them.
    """
    a = project_simplex(a0.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        conv = fast_convolve(a)
        peak_val = conv.max()

        if peak_val < best_obj - 1e-14:
            best_obj = peak_val
            best_a = a.copy()
            no_improve = 0
            continue

        # Top peaks above 85% of max
        threshold = 0.85 * peak_val
        peaks = np.where(conv >= threshold)[0]
        if len(peaks) > max_peaks:
            top_idx = np.argsort(conv[peaks])[::-1][:max_peaks]
            peaks = peaks[top_idx]

        # Build deflation LP
        # Variables: delta[0..n-1], t
        # min -t s.t. sum_j a[j]*delta[k-j] + 0.5*t <= 0 for each peak k
        #           sum(delta) = 0
        #           -a[j] <= delta[j] <= eps
        #           t >= 0
        c_obj = np.zeros(n + 1)
        c_obj[-1] = -1.0

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        rows = []
        b_ub_list = []
        for k in peaks:
            row = np.zeros(n + 1)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                row[j] = a[j]
            row[-1] = 0.5
            rows.append(row)
            b_ub_list.append(0.0)

        A_ub_arr = np.array(rows) if rows else None
        b_ub_arr = np.array(b_ub_list) if b_ub_list else None

        bounds = [(-a[j], eps) for j in range(n)]
        bounds.append((0.0, None))

        try:
            res = linprog(c_obj, A_ub=A_ub_arr, b_ub=b_ub_arr,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': min(5, remaining - 1)})
            if res.success and res.x is not None and res.x[-1] > 1e-10:
                delta = res.x[:n]
                # Line search
                best_alpha = 0.0
                best_val = peak_val
                for alpha_try in [0.1, 0.3, 0.5, 0.7, 1.0]:
                    cand = a + alpha_try * delta
                    cand = np.maximum(cand, 0)
                    cs = cand.sum()
                    if cs < 1e-15:
                        continue
                    cand = cand / cs
                    val = fast_convolve(cand).max()
                    if val < best_val:
                        best_val = val
                        best_alpha = alpha_try

                if best_alpha > 0:
                    a = a + best_alpha * delta
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 30:
            break
        if no_improve > 15:
            eps = max(0.001, eps * 0.7)

    return best_a, best_obj


def subgradient_multistart(a0, time_limit, top_k=10):
    """Simple subgradient with adaptive learning rate."""
    a = project_simplex(a0.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = 0.03
    eta_decay = 0.9995

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            weights = pv / pv.sum()

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            w = weights[idx]
            if w < 1e-12:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je + 1] += w * 2.0 * a[k - je:k - js + 1][::-1]

        a_new = a - lr * grad
        a_new = project_simplex(a_new)

        # Line search
        best_val = conv.max()
        best_alpha = 0.0
        for alpha_try in [0.3, 0.5, 0.7, 1.0]:
            cand = (1 - alpha_try) * a + alpha_try * a_new
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha_try

        if best_alpha > 0:
            a = (1 - best_alpha) * a + best_alpha * a_new
            a = np.maximum(a, 0)
            a = a / a.sum()
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        lr *= eta_decay
        if lr < 0.003:
            lr = 0.03

        if no_improve > 100:
            rng = np.random.default_rng(it * 9973 + 1)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0
            lr = 0.03

    return best_a, best_obj


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations for exploration."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge-heavy
    edge_w = 1.0 / (np.minimum(x, n - 1 - x).astype(float) + 1.0) ** 1.2
    inits.append(edge_w / edge_w.sum())

    # Sinusoidal modulation
    t = x / max(n - 1, 1)
    for freq in [1, 2, 3]:
        sig = 1.0 + 0.3 * np.sin(2 * np.pi * freq * t)
        sig = np.maximum(sig, 0.1)
        inits.append(sig / sig.sum())

    # Two-peak
    g2 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())

    # Dirichlet with concentration
    for conc in [0.5, 1.0, 3.0]:
        inits.append(rng.dirichlet(np.ones(n) * conc))

    # Parent perturbation
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        for scale in [0.02, 0.05, 0.1]:
            pert = np.maximum(px + rng.standard_normal(n) * scale, 0)
            ps = pert.sum()
            if ps > 1e-10:
                inits.append(pert / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Multi-phase hybrid optimization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s
        = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(768) / 768

    # === Phase 1: Quick SCA across multiple n values (250s) ===
    print("\n=== Phase 1: Multi-n SCA Exploration ===")
    t1 = time.monotonic()

    n_parent = len(parent)
    ns_to_try = sorted(set([n_parent, 512, 640, 768, 896, 1024, 1200, 1536, 2048]))
    ns_to_try = [n for n in ns_to_try if 256 <= n <= 2500]

    # Sort by proximity to parent n first, then by size
    ns_to_try.sort(key=lambda n: (abs(n - n_parent), n))

    for n_try in ns_to_try:
        remaining_after = deadline - time.monotonic()
        if remaining_after < 300:
            break

        if n_try == n_parent and parent is not None:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            start = start / start.sum()

        t_alloc = min(25, remaining_after - 280)
        if t_alloc < 5:
            break
        print(f"  SCA n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = sca_optimize(start, time_limit=t_alloc)
        try_seq(a_opt, f"sca_n{n_try}")

    print(f"  P1 done: best={best_score:.8f}, t={time.monotonic()-t1:.0f}s")

    # === Phase 2: Deep SCA at best n (500s) ===
    print("\n=== Phase 2: Deep SCA ===")
    t2 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 400
        if remaining > 20:
            t_alloc = min(remaining, 300)
            print(f"  Deep SCA n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = sca_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_sca_1")

            remaining = deadline - time.monotonic() - 200
            if remaining > 15:
                t_alloc2 = min(remaining, 150)
                a_opt2, _ = sca_optimize(a_opt, time_limit=t_alloc2)
                try_seq(a_opt2, "deep_sca_2")

    print(f"  P2 done: best={best_score:.8f}, t={time.monotonic()-t2:.0f}s")

    # === Phase 3: Spectral Equalization + Re-optimize (250s) ===
    print("\n=== Phase 3: Spectral Equalization ===")
    t3 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        remaining = deadline - time.monotonic() - 200
        if remaining > 20:
            t_spec = min(remaining, 80)
            print(f"  Spectral n={n_best}, t={t_spec:.0f}s")
            a_spec, _ = spectral_equalize(best_seq, time_limit=t_spec)
            try_seq(a_spec, "spectral")

            # Re-optimize from spectral result
            remaining = deadline - time.monotonic() - 150
            if remaining > 15:
                t_re = min(remaining, 80)
                a_re, _ = sca_optimize(a_spec, time_limit=t_re)
                try_seq(a_re, "spectral+sca")

            # Also try spectral from a perturbed version
            remaining = deadline - time.monotonic() - 100
            if remaining > 15:
                pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.03)
                t_spec2 = min(remaining, 50)
                a_spec2, _ = spectral_equalize(pert, time_limit=t_spec2)
                try_seq(a_spec2, "spectral_pert")

    print(f"  P3 done: best={best_score:.8f}, t={time.monotonic()-t3:.0f}s")

    # === Phase 4: Deflation + Subgradient (200s) ===
    print("\n=== Phase 4: Deflation + Subgradient ===")
    t4 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        remaining = deadline - time.monotonic() - 150
        if remaining > 15:
            t_def = min(remaining, 60)
            print(f"  Deflation t={t_def:.0f}s")
            a_def, _ = deflation_refine(best_seq, time_limit=t_def, max_peaks=8)
            try_seq(a_def, "deflation")

            # Re-optimize
            remaining = deadline - time.monotonic() - 120
            if remaining > 15:
                t_re = min(remaining, 50)
                a_re, _ = sca_optimize(a_def, time_limit=t_re)
                try_seq(a_re, "deflation+sca")

            remaining = deadline - time.monotonic() - 80
            if remaining > 10:
                t_sg = min(remaining, 50)
                print(f"  Subgradient t={t_sg:.0f}s")
                a_sg, _ = subgradient_multistart(best_seq, time_limit=t_sg, top_k=15)
                try_seq(a_sg, "subgrad")

    print(f"  P4 done: best={best_score:.8f}, t={time.monotonic()-t4:.0f}s")

    # === Phase 5: Bisection tightening (150s) ===
    print("\n=== Phase 5: Bisection Tighten ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 80
        if remaining > 20:
            t_bis = min(remaining, 100)
            print(f"  Bisection n={n_best}, t={t_bis:.0f}s")
            a_bis, M_bis = bisection_tighten(best_seq, time_limit=t_bis, tol=1e-7)
            try_seq(a_bis, "bisection")

            # Final SCA from bisection result
            remaining = deadline - time.monotonic() - 50
            if remaining > 10:
                a_final, _ = sca_optimize(a_bis, time_limit=min(remaining, 30))
                try_seq(a_final, "bisection+sca")

    print(f"  P5 done: best={best_score:.8f}, t={time.monotonic()-t5:.0f}s")

    # === Phase 6: Diversified restarts (80s) ===
    print("\n=== Phase 6: Diversified Restarts ===")
    t6 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        inits = generate_diverse_inits(n_best, best_seq, rng)
        # Add extra perturbations
        for scale in [0.01, 0.03, 0.05, 0.08]:
            inits.append(project_simplex(best_seq + rng.standard_normal(n_best) * scale))

        for i, init in enumerate(inits[:6]):
            remaining = deadline - time.monotonic() - 30
            if remaining < 10:
                break
            t_alloc = min(12, remaining - 25)
            if t_alloc < 5:
                break
            print(f"  Restart {i} t={t_alloc:.0f}s")
            a_opt, _ = sca_optimize(init, time_limit=t_alloc)
            try_seq(a_opt, f"restart_{i}")

    print(f"  P6 done: best={best_score:.8f}, t={time.monotonic()-t6:.0f}s")

    # === Phase 7: Final polish (remaining) ===
    print("\n=== Phase 7: Final Polish ===")
    t7 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            t_alloc = min(remaining - 5, 30)
            if t_alloc > 5:
                a_opt, _ = sca_optimize(best_seq, time_limit=t_alloc)
                try_seq(a_opt, "final_sca")

            remaining = deadline - time.monotonic() - 5
            if remaining > 5:
                a_opt, _ = subgradient_multistart(best_seq, time_limit=min(remaining, 15), top_k=5)
                try_seq(a_opt, "final_sg")

    print(f"  P7 done: best={best_score:.8f}, t={time.monotonic()-t7:.0f}s")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence([float(max(0, x)) for x in best_seq]))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")
    return [float(max(0.0, x)) for x in best_seq]
