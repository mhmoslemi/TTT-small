def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_nonneg_sum(v, target_sum=1.0):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) / len(v) * target_sum
    return v / s * target_sum


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_from_conv(n, max_b, sum_a):

def mirror_descent(a0, time_limit, seeds=None, verbose=False):
    """Multiplicative weight mirror descent for minimizing max autocorrelation."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    rng = np.random.default_rng(seeds[0])

            break
        
        conv = fast_convolve(a)
        max_val = conv.max()
        obj = 2.0 * n * max_val  # since sum(a)=1
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
        
        # Get top peaks
        conv_len = len(conv)
        top_k = min(20, conv_len)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        # Temperature-based weighting
        temp = 0.01 * max_val + 1e-10
        pw = np.exp(pv / temp - pv.max() / temp)
        pw /= pw.sum()
        
        # Subgradient
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            w = pw[idx]
            if w < 1e-10:
                continue
            k = int(peaks[idx])

def coordinate_descent(a0, time_limit, verbose=False):

        # We want to find a[j] that minimizes max_k (conv[k] + delta_j * a[k-j])
        
        # Simple: try a few values
        current = a[j]
        candidates = [current * 0.5, current * 0.75, current * 0.9, 
                      current, current * 1.1, current * 1.25, current * 1.5,
                      current + 0.001, current - 0.001]
        candidates = [max(0, c) for c in candidates]
        
        best_cand = current
        best_cand_obj = obj
        
        for c in candidates:
            a_test = a.copy()
            a_test[j] = c
            a_test = a_test / a_test.sum()

            if obj_test < best_cand_obj:

                best_cand = c

            a[j] = best_cand

def subgradient_method(a0, time_limit, top_k=10, verbose=False):
    """Subgradient method with adaptive step size."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_obj = float('inf')
    best_a = a.copy()
    lr = 0.1
    rng = np.random.default_rng(7)
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit - 0.1:
        it += 1
        conv = fast_convolve(a)
        max_val = conv.max()
        obj = 2.0 * n * max_val
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            no_improve = 0
            lr = 0.1
        
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            weights = pv / pv.sum()
        
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            w = weights[idx]
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += w * a[k - j]
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Step
        a_new = a - lr * grad
        a_new = project_simplex(a_new)
        
        # Line search
        best_alpha = 1.0
        best_val = 2.0 * n * fast_convolve(a_new).max()
        for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
            cand = (1 - alpha) * a + alpha * a_new
            cand = project_simplex(cand)
            val = 2.0 * n * fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        a = (1 - best_alpha) * a + best_alpha * a_new
        a = project_simplex(a)
        
        lr = max(0.01, lr * 0.999)
        
        if verbose and it % 100 == 0:
            print(f"  SG it={it}, obj={2*n*fast_convolve(a).max():.8f}")
    
    return best_a, best_obj

            best_obj = obj

        # To reduce the peak, decrease a[j] for j that contributes most

        top_contributors = np.argsort(contrib)[::-1][:min(5, je - js + 1)]

def build_lp_matrix(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def sca_optimize(f0, time_limit, verbose=False):
    """SCA with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    row_idx, col_idx = build_lp_matrix(n)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit - 0.1:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        # Solve LP
        c = -np.ones(n)
        k_arr = row_idx
        i_arr = col_idx
        vals = f[k_arr - i_arr]
        A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True})
            if not res.success:
                M *= 1.02
                no_improve += 1
                if no_improve > 30:
                    M = fast_convolve(f).max() * 1.01
                    no_improve = 0
                continue
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs < 1e-15:
                M *= 1.02
                continue
            g = g / gs
        except Exception:
            M *= 1.02
            no_improve += 1
            continue
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                rng = np.random.default_rng(it * 7919)
                f = project_simplex(f + rng.standard_normal(n) * 0.002)
                M = fast_convolve(f).max() * 1.01
                no_improve = 0
            continue
        
        # Line search
        best_val = fast_convolve(f).max()
        best_ft = f.copy()
        for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_ft = ft.copy()
        
        if best_val < best_obj - 1e-14:
            f = best_ft
            best_obj = best_val
            best_f = f.copy()
            M = best_val * 0.999
            no_improve = 0
        else:
            no_improve += 1
            M *= 0.998
        
        if no_improve > 40:
            rng = np.random.default_rng(it * 31831 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            M = fast_convolve(f).max() * 1.01
            no_improve = 0
        
        if verbose and it % 50 == 0:
            print(f"    SCA it={it}, obj={best_obj:.10f}")
    
    return best_f, best_obj

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(n):
    """Precompute the sparsity pattern for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp(f, M, n, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    k_arr = row_idx
    i_arr = col_idx
    vals = f[k_arr - i_arr]
    A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def sca_optimize(f0, time_limit, verbose=False):
    """SCA optimization with aggressive adaptive M."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

            M *= 1.05

                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                M = fast_convolve(f).max() * 1.01
                no_improve = 0
            continue

        consecutive_fail = 0
        g = g_raw / g_raw.sum()

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 25:

            print(f"    SCA it={it}, obj={best_obj:.10f}, M={M:.10f}, no_improve={no_improve}")

    return best_f, best_obj


def generate_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Alternating (1,0,1,0,...)
    alt = np.ones(n)
    alt[1::2] = 0.0
    alt = np.maximum(alt, 0.01)
    inits.append(alt / alt.sum())

    # Periodic: blocks of 1s and 0s
    for period in [3, 4, 5, 7, 8]:
        pat = np.zeros(n)
        pat[np.arange(0, n, period) < n] = 1.0
        # Add some ones within the period
        pat[np.arange(1, n, period) < n] = 0.5
        pat = np.maximum(pat, 0.01)
        inits.append(pat / pat.sum())

    # Edges heavy
    edge_w = 1.0 / (np.minimum(x, n - 1 - x).astype(float) + 1.0)**2
    inits.append(edge_w / edge_w.sum())

    # Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.1))
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 5.0))

    # Sinusoidal
    t = x / max(n - 1, 1)
    for freq in [1, 2, 3, 5]:
        sw = np.abs(np.sin(np.pi * freq * t))**2 + 0.05
        inits.append(sw / sw.sum())

    # Random binary-like
    rb = rng.integers(0, 2, n).astype(float)
    rb = np.maximum(rb, 0.1)
    inits.append(rb / rb.sum())

    # Two-peak
    g2 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n))**2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n))**2)
    inits.append(g2 / g2.sum())

    return inits


def mirror_descent_fast(a0, time_limit, verbose=False):
    """Fast mirror descent with peak targeting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_obj = float('inf')
    best_a = a.copy()
    rng = np.random.default_rng(2024)

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    eta = 0.05

    while time.monotonic() - t0 < time_limit - 0.1:
        it += 1

        conv = fast_convolve(a)
        max_val = conv.max()
        obj = 2.0 * n * max_val

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
            no_improve = 0
            eta = 0.05
            continue

        # Top peaks
        top_k = min(15, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        pw = np.exp((pv - pv.max()) / (0.005 * max_val + 1e-10))
        pw /= pw.sum()

        # Gradient
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            w = pw[idx]
            if w < 1e-12:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += w * a[k - j]

        # Multiplicative update
        a_new = a * np.exp(-eta * grad / (max_val * n + 1e-10))
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        # Blend
        a = 0.5 * a + 0.5 * a_new
        a = np.maximum(a, 0)
        a = a / a.sum()

        eta *= 0.999
        if eta < 0.005:
            eta = 0.05

    return best_a, best_obj
