"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Continuation method with L-BFGS on smoothed max (soft.conv)
2. Frank-Wolfe with adaptive sparse LP oracle
3. Newton steps using Hankel Hessian structure (FFT-accelerated)
4. Multi-n exploration and perturbation restarts

import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_matvec(p, v):
    """Compute sum_m p[j+m]*v[m] for j=0..n-1. p has length 2n-1, v has length n."""
    n = len(v)
    w = v[::-1]
    L = len(p) + n - 1
    L_fft = 1 << (L - 1).bit_length()
    P = np.fft.rfft(p, L_fft)
    W = np.fft.rfft(w, L_fft)
    conv = np.fft.irfft(P * W, L_fft)[:L]
    return conv[n-1:2*n-1]

def smooth_grad_and_conv(a, T):
    """Compute gradient of T*log(sum(exp(conv/T))) and the convolution."""
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()
    # Gradient of sum_k p[k] * conv[k] = 2 * sum_k p[k] * a[k-j] (for valid j)
    # = 2 * hankel_matvec(p, a)
    grad = 2.0 * hankel_matvec(p, a)
    return grad, conv, p

def lbfgs_continuation(a0, time_limit, T_init=None, T_min=1e-6):
    """
    Minimize max(conv(a,a)) using continuation on T (soft-max smoothing).
    Uses projected gradient with momentum and adaptive T.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    if T_init is None:
        T_init = M / 20.0
    T = T_init
    
    v = np.zeros(n)
    lr = 0.01
    mom = 0.9
    best_a = a.copy()
    best_obj = M
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break
        
        grad, conv, p = smooth_grad_and_conv(a, T)
        obj = conv.max()
        
        # Backtracking line search with momentum
        v = mom * v - lr * grad
        best_candidate = a
        best_val = obj
        accepted = False
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.01]:
            candidate = project_simplex(a + scale * v)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                accepted = True
                break
        
        a = best_candidate
        
        if best_val < best_obj - 1e-12:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.5
        
        # Adapt T and lr
        if no_improve > 20:
            T *= 0.9
            lr *= 0.95
        if no_improve > 50:
            lr *= 0.8
            mom = min(0.95, mom + 0.01)
        if no_improve > 150:
            # Restart with noise
            rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
            noise_scale = 0.001 + 0.003 * min(1.0, (no_improve - 150) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            lr = 0.01
            mom = 0.9
            T = max(T_min, fast_convolve(a).max() / 20.0)
            no_improve = 0
        
        # If T is very small and we're converging, try to reduce further
        if no_improve == 0 and T > T_min:
            T = max(T_min, T * 0.98)
    
    return best_a, best_obj


def sparse_lp_oracle(f, M, active_peaks):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_peaks, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    K = len(active_peaks)
    A_ub = np.zeros((K, n))
    for idx, k in enumerate(active_peaks):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A_ub[idx, js:je+1] = f[k-je:k-js+1][::-1]
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'max_time': 10})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_sparse_lp(f0, time_limit, max_active=200):
    """Frank-Wolfe with sparse LP oracle and adaptive active set."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        # Find active peaks
        threshold = M * (1.0 - 0.01)
        active = np.where(conv >= threshold)[0]
        if len(active) > max_active:
            # Keep only the top max_active
            active = np.argpartition(conv, -max_active)[-max_active:]
        
        g = sparse_lp_oracle(f, M, active)
        if g is None:
            # Try with fewer constraints
            active_top = np.argpartition(conv, -min(50, len(active)))[-min(50, len(active)):]
            g = sparse_lp_oracle(f, M, active_top)
            if g is None:
                no_improve += 1
                if no_improve > 20:
                    break
                continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-12:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 40:
            rng = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
    
    return best_f, best_obj


def newton_step(a, p, max_cg=50, tol=1e-10):
    """
    Compute Newton direction for the smoothed objective.
    Hessian H[j,m] = 2*p[j+m]. Solve H @ d = -grad.
    Uses conjugate gradient with the Hankel structure.
    """
    n = len(a)
    conv = fast_convolve(a)
    T = max(conv.max() / 50.0, 1e-8)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()
    
    # Gradient
    grad = 2.0 * hankel_matvec(p, a)
    
    # CG to solve H @ d = -grad
    # H @ v = 2 * hankel_matvec(p, v)
    r = -grad.copy()
    d = r.copy()
    rr = r @ r
    x = np.zeros(n)
    
    for _ in range(max_cg):
        if np.sqrt(rr) < tol * np.linalg.norm(r, ord=np.inf) or rr < 1e-30:
            break
        Hv = 2.0 * hankel_matvec(p, d)
        dHd = d @ Hv
        if dHd < 1e-30:
            break
        alpha = rr / dHd
        x += alpha * d
        r += alpha * Hv
        rr_new = r @ r
        if rr_new >= rr:
            # Break down, restart
            r = -grad - 2.0 * hankel_matvec(p, x)
            rr = r @ r
            d = r.copy()
        beta = rr_new / rr if rr > 0 else 0
        d = r + beta * d
        rr = rr_new
    
    return x, p

def interp_sequence(parent, n_target):
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Continuation method with L-BFGS-B on smoothed max objective
2. Frank-Wolfe with adaptive sparse LP oracle (active-set cutting plane)
3. Newton steps exploiting Hankel Hessian structure via FFT
4. Momentum-based projected gradient with adaptive top-k peaks
5. Multi-n exploration and perturbation restarts
"""
import numpy as np
from scipy.optimize import linprog, minimize as sp_minimize
import time


def fast_convolve(a):
    """FFT-based auto-convolution, returns full linear convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_matvec(p, v):
    """
    Compute out[j] = sum_m p[j+m]*v[m] for j=0..n-1.
    p has length 2n-1, v has length n. Uses FFT for O(n log n) complexity.
    """
    n = len(v)
    w = v[::-1]
    L = len(p) + n - 1
    L_fft = 1 << (L - 1).bit_length()
    P = np.fft.rfft(p, L_fft)
    W = np.fft.rfft(w, L_fft)
    conv = np.fft.irfft(P * W, L_fft)[:L]
    return conv[n - 1:2 * n - 1]


def smooth_objective_and_grad(x, T):
    """
    Compute the smoothed max objective T*log(sum(exp(conv/T))) and its gradient.
    x is assumed to be on the simplex (a >= 0, sum = 1).
    """
    n = len(x)
    conv = fast_convolve(x)
    max_conv = conv.max()
    scaled = (conv - max_conv) / T
    exp_sums = np.exp(scaled)
    p = exp_sums / exp_sums.sum()
    obj = max_conv + T * np.log(exp_sums.sum())
    grad = 2.0 * hankel_matvec(p, x)
    return obj, grad, conv, p


def continuation_lbfgsb(a0, time_limit, T_init=None, T_min=1e-7, max_inner_iter=500):
    """
    Continuation method: minimize max(conv) by solving a sequence of
    smoothed problems with decreasing T using L-BFGS-B.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    if T_init is None:
        T_init = M / 10.0
    T = T_init

    best_a = a.copy()
    best_obj = M

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit and T > T_min:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        # Solve smoothed problem at current T
        state = {'t0': time.monotonic(), 'time_limit': min(remaining, 60.0)}

        def obj_grad(x):
            if time.monotonic() - state['t0'] > state['time_limit']:
                raise RuntimeError("time")
            xn = np.maximum(x, 0.0)
            sx = xn.sum()
            if sx < 1e-15:
                return 1e10, np.zeros(n)
            xn = xn / sx
            obj, grad, _, _ = smooth_objective_and_grad(xn, T)
            return obj, grad

        try:
            result = sp_minimize(
                obj_grad, a, method='L-BFGS-B',
                bounds=[(0, None)] * n,
                options={'maxiter': max_inner_iter, 'gtol': 1e-12, 'maxfun': 3000}
            )
            a_candidate = np.maximum(result.x, 0.0)
            if a_candidate.sum() > 1e-15:
                a_candidate = a_candidate / a_candidate.sum()
                obj_c = fast_convolve(a_candidate).max()
                if obj_c < best_obj:
                    best_obj = obj_c
                    best_a = a_candidate.copy()
                a = a_candidate
        except RuntimeError:
            pass
        except Exception:
            pass

        # Decrease T
        T = max(T_min, T * 0.7)

    return best_a, best_obj


def momentum_pgd(a0, time_limit, top_k=20, lr_init=0.02, mom_init=0.9):
    """Projected gradient descent with momentum and multi-peak subgradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = mom_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * a[k - idx_arr]

        v = mom * v - lr * sg
        best_candidate = a
        best_val = conv.max()
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03]:
            candidate = project_simplex(a + scale * v)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-14:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate

        if best_val < best_obj - 1e-13:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.85

        if no_improve > 30:
            lr *= 0.93
            mom = min(0.97, mom + 0.003)
        if no_improve > 200:
            rng_local = np.random.default_rng((it * 48271 + 17) % (2**31 - 1))
            noise = 0.002 + 0.003 * min(1.0, (no_improve - 200) / 300.0)
            a = project_simplex(best_a + rng_local.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init
            mom = mom_init
            no_improve = 0

    return best_a, best_obj


def build_lp_matrix(f, active_peaks, n):
    """Build constraint matrix for LP: (f*g)[k] for k in active_peaks."""
    K = len(active_peaks)
    A = np.zeros((K, n))
    for idx, k in enumerate(active_peaks):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[idx, js:je + 1] = f[k - je:k - js + 1][::-1]
    return A


def sparse_lp_oracle(f, M, active_peaks):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_peaks, g >= 0."""
    n = len(f)
    K = len(active_peaks)
    if K == 0:
        return np.ones(n) / n
    c = -np.ones(n)
    A_ub = build_lp_matrix(f, active_peaks, n)
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'max_time': 15})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_sparse_lp(f0, time_limit, max_active=300):
    """Frank-Wolfe with sparse LP oracle and adaptive active set management."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    active_set = None

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        # Identify or update active peaks
        threshold = M * (1.0 - 0.005)
        new_active = np.where(conv >= threshold)[0]
        if len(new_active) > max_active:
            new_active = np.argpartition(conv, -max_active)[-max_active:]

        # Solve LP with current active set
        g = sparse_lp_oracle(f, M, new_active)

        if g is None:
            # Try with fewer constraints
            top_k = min(50, len(new_active))
            if top_k > 0:
                top_active = np.argpartition(conv, -top_k)[-top_k:]
                g = sparse_lp_oracle(f, M, top_active)
            if g is None:
                no_improve += 1
                if no_improve > 30:
                    break
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 25:
                break
            continue

        # Line search (coarse then refine)
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Bisection refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            rng_local = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def newton_hankel(a0, time_limit):
    """
    Newton method using Hankel Hessian structure.
    H[j,m] = 2*p[j+m], H@v computed via FFT.
    Uses CG to solve the Newton system.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_obj = M

    t0 = time.monotonic()
    it = 0
    T = M / 30.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        # Compute weights and gradient
        scaled = (conv - M) / T
        exp_vals = np.exp(scaled)
        p = exp_vals / exp_vals.sum()
        grad = 2.0 * hankel_matvec(p, a)

        # CG to solve H @ d = -grad where H @ v = 2 * hankel_matvec(p, v)
        r = -grad.copy()
        d_cg = r.copy()
        rr = r @ r
        x_newton = np.zeros(n)
        converged = False

        for cg_iter in range(40):
            if np.sqrt(rr) < 1e-12 * max(1.0, np.linalg.norm(r, np.inf)):
                converged = True
                break
            Hv = 2.0 * hankel_matvec(p, d_cg)
            dHd = d_cg @ Hv
            if dHd < 1e-30:
                break
            alpha_cg = rr / dHd
            x_newton += alpha_cg * d_cg
            r_new = r + alpha_cg * Hv
            rr_new = r_new @ r_new
            if rr_new >= rr * 0.99:
                # Restart CG
                r = -grad - 2.0 * hankel_matvec(p, x_newton)
                rr = r @ r
                d_cg = r.copy()
            else:
                beta_cg = rr_new / rr if rr > 1e-30 else 0.0
                d_cg = r_new + beta_cg * d_cg
                r = r_new
                rr = rr_new

        # Check if Newton direction is descent
        if not converged and np.dot(x_newton, -grad) < 0:
            # Newton step not descent; use gradient
            direction = -grad
        else:
            direction = x_newton

        # Line search
        best_candidate = a
        best_val = M
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            candidate = project_simplex(a + alpha * direction)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-14:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate

        if best_val < best_obj - 1e-13:
            best_obj = best_val
            best_a = a.copy()
            conv = fast_convolve(a)
            M = conv.max()
            T = max(T * 0.85, 1e-8)
        else:
            # No improvement, adjust
            T = max(T * 1.2, M / 100.0)
            if it > 100:
                rng_local = np.random.default_rng(it * 31 + 7)
                a = project_simplex(best_a + rng_local.standard_normal(n) * 0.001)
                T = M / 30.0

    return best_a, best_obj


def evaluate_and_update(a, best_seq, best_score, label=""):
    """Evaluate sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length, maintain simplex."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s
