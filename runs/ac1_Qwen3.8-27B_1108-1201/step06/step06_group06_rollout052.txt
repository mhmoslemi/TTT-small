```python
Optimized step function search using accelerated smoothed gradient descent,
greedy peak reduction, and adaptive multi-n exploration.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(a, T):
    """
    Compute smoothed objective value and its gradient.
    Objective: T * log(sum_k exp(conv[k]/T))
    Gradient: 2 * (a cross-correlated with p) where p = softmax(conv/T)
    """
    n = len(a)
    conv = fast_convolve(a)
    
    # Softmax
    scaled = conv / T
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    
    # Log-sum-exp for objective
    obj = T * (scaled.max() + np.log(exp_scaled.sum()))
    
    # Gradient via cross-correlation
    # grad[i] = 2 * sum_j a[j] * p[j+i]
    # FFT size >= n + (2n-1) - 1 = 3n-2
    N = 1 << (3 * n - 2 - 1).bit_length()
    F = np.fft.rfft(a, N)
    G = np.fft.rfft(p, N)
    corr = np.fft.irfft(np.conj(F) * G, N)
    grad = 2.0 * corr[:n]
    
    return obj, grad, conv, p


def peak_reduction_move(a, conv, n, num_peaks=5, strength=0.5):
```

```python
"""
Optimized step function search using accelerated smoothed gradient descent,
FFT-based gradient computation, greedy peak reduction, coordinate search,
and adaptive multi-n exploration. Targets minimization of peak autocorrelation.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(a, T):
    """
    Compute smoothed objective T*log(sum exp(conv/T)) and its gradient.
    Returns (obj, grad, conv, p).
    """
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    p = exp_s / exp_s.sum()
    obj = T * (scaled.max() + np.log(exp_s.sum()))
    # Gradient: grad[i] = 2 * sum_j a[j] * p[j+i]
    N = 1 << (3 * n - 2).bit_length()
    F = np.fft.rfft(a, N)
    G = np.fft.rfft(p, N)
    corr = np.fft.irfft(np.conj(F) * G, N)
    grad = 2.0 * corr[:n]
    return obj, grad, conv, p


def peak_shave(a, conv, n, top_k=8, steps=3):
    """
    Greedy peak reduction: for each top peak lag, reduce contributing pairs
    and redistribute to less critical positions.
    """
    a_new = a.copy()
    sorted_lags = np.argsort(conv)[::-1][:top_k]
    center = n - 1

    for k in sorted_lags:
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start > i_end:
            continue
        i_vals = np.arange(i_start, i_end + 1)
        contribs = a_new[i_vals] * a_new[k - i_vals]
        max_c = contribs.max()
        if max_c <= 0:
            continue

        for _ in range(steps):
            i_vals = np.arange(i_start, i_end + 1)
            contribs = a_new[i_vals] * a_new[k - i_vals]
            if contribs.max() <= 0:
                break
            im = i_vals[np.argmax(contribs)]
            jm = k - im
            amt = min(a_new[im], a_new[jm]) * 0.05
            if amt < 1e-12:
                break
            if im == jm:
                a_new[im] -= amt
                tgt = (im + n // 2) % n
                a_new[tgt] += amt
            else:
                a_new[im] -= amt * 0.6
                a_new[jm] -= amt * 0.4
                t1 = (im + n // 3) % n
                t2 = (jm + 2 * n // 3) % n
                a_new[t1] += amt * 0.3
                a_new[t2] += amt * 0.3
                a_new[(im + jm) % n] += amt * 0.4

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new

                                       T_init=1e-7, T_min=1e-9, use_momentum=False)
        best_seq, best_score = try_seq(a_opt, "ultra_fine", (best_seq, best_score))

    # Peak shave + coordinate
    rem = deadline - time.monotonic()
    if rem > 30:
        conv = fast_convolve(best_seq)
        shaved = peak_shave(best_seq, conv, n_best, top_k=10, steps=5)
        best_seq, best_score = try_seq(shaved, "peak_shave", (best_seq, best_score))

    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.25, 60)
        a_opt, _ = coordinate_search(best_seq, time_limit=t_alloc, seed=31415)
        best_seq, best_score = try_seq(a_opt, "coord_final", (best_seq, best_score))

    # Micro-perturbations
    rem = deadline - time.monotonic()
    if rem > 15:
        for _ in range(6):
            rem = deadline - time.monotonic()
            if rem < 4:
                break
            noise = rng.standard_normal(n_best) * 0.0002
            fp = project_simplex(best_seq + noise)
            t_alloc = min(rem * 0.12, 5)
            a_opt, _ = accelerated_descent(fp, time_limit=t_alloc, T_init=1e-7)
            best_seq, best_score = try_seq(a_opt, "micro", (best_seq, best_score))

    # One more coordinate pass
    rem = deadline - time.monotonic()
    if rem > 10:
        a_opt, _ = coordinate_search(best_seq, time_limit=min(rem * 0.3, 20), seed=99999)
        best_seq, best_score = try_seq(a_opt, "coord_last", (best_seq, best_score))

    print(f"  Phase 5: {best_score:.8f} ({time.monotonic()-t_ph:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final check
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total = time.monotonic() - t_start
    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)}, t={total:.0f}s)")
    return [float(x) for x in best_seq]
```
