def project_simplex(v, tol=1e-10):
    """Euclidean projection onto the simplex {x | x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    # Sort the components
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u * np.arange(1, n+1) > cssv)[0][-1]
    if rho == 0:
        theta = cssv[rho] / (rho + 1)
    else:
        theta = 0
    # Return projection
    return max( (v - theta).clip(tol, float('inf')), 0)
