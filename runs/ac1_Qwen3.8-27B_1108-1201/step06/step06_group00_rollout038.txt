import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize_to_sum(a, target=1.0):
    """Normalize array to have given sum, maintaining non-negativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a * (target / s)


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_top_peaks(conv, k=10):
    """Get indices of top-k peaks in convolution."""
    k = min(k, len(conv))
    return np.argpartition(conv, -k)[-k:]


def targeted_peak_reduction(f, time_budget, n_peaks=5, rng=None):
    """
    For each top peak, identify contributing coordinate pairs and
    adjust them to reduce the peak while maintaining sum=1.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(f)
    f = normalize_to_sum(f.copy(), 1.0)
    conv = fast_convolve(f)
    best_obj = conv.max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_budget:
        conv = fast_convolve(f)
        M = conv.max()
        if M < best_obj - 1e-12:
            best_obj = M
        
        # Get top peaks
        peaks = compute_top_peaks(conv, n_peaks)
        peak_improved = False
        
        for k in peaks:
            if time.monotonic() - t0 > time_budget - 1:
                break
            k = int(k)
            
            # Find contributing pairs (i, k-i) with i+j=k
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo > hi:
                continue
            
            # Compute contributions
            js = np.arange(lo, hi + 1)
            contributions = f[js] * f[k - js]
            total = contributions.sum()
            
            if total < 1e-15:
                continue
            
            # Find the pair with maximum contribution
            idx_max = np.argmax(contributions)
            i_j = js[idx_max]
            j_j = k - i_j
            
            # Strategy: reduce the larger of the two and add to a "safe" position
            if f[i_j] >= f[j_j]:
                reduce_idx = i_j
                other_idx = j_j
            else:
                reduce_idx = j_j
                other_idx = i_j
            
            # Find a safe position to add mass (one that doesn't contribute to this peak)
            # A position p is safe if p + k - p != k (always true) but we want
            # p such that conv at nearby peaks doesn't increase
            # Simple heuristic: add to a position far from both i_j and j_j
            dist = np.abs(np.arange(n) - reduce_idx) + np.abs(np.arange(n) - other_idx)
            # Also avoid positions that contribute to other top peaks
            for pk in peaks:
                pk = int(pk)
                dist += 0.1 * np.abs(np.arange(n) - pk)
            
            safe_candidates = np.argsort(dist)[::-1][:20]
            safe_candidates = [c for c in safe_candidates if c != reduce_idx and c != other_idx]
            
            if not safe_candidates:
                continue
            
            # Try moving small amounts of mass
            for frac in [0.05, 0.02, 0.01, 0.005]:
                delta = f[reduce_idx] * frac
                if delta < 1e-12:
                    continue
                
                for dest in safe_candidates[:5]:
                    f_trial = f.copy()
                    f_trial[reduce_idx] -= delta
                    f_trial[dest] += delta
                    if f_trial[reduce_idx] < 0:
                        continue
                    f_trial = np.maximum(f_trial, 0)
                    f_trial = normalize_to_sum(f_trial, 1.0)
                    
                    conv_trial = fast_convolve(f_trial)
                    if conv_trial.max() < M - 1e-12:
                        f = f_trial
                        M = conv_trial.max()
                        if M < best_obj:
                            best_obj = M
                        peak_improved = True
                        break
                if peak_improved:
                    break
    
    return normalize_to_sum(f, 1.0), best_obj


def coordinate_descent_pass(f, time_budget, n_trials=3):
    """
    Full coordinate descent: for each coordinate, try several values
    and keep the one that minimizes the peak convolution.
    """
    n = len(f)
    f = normalize_to_sum(f.copy(), 1.0)
    conv = fast_convolve(f)
    best_obj = conv.max()
    t0 = time.monotonic()
    
    # Precompute: for each coordinate i, which k values are affected
    # conv_k depends on a[i] if k-i is in [0, n-1], i.e., k in [i, i+n-1]
    
    for coord in range(n):
        if time.monotonic() - t0 > time_budget - 0.5:
            break
        
        base_val = f[coord]
        if base_val < 1e-12:
            # Try adding small mass
            deltas = [base_val + 0.001, base_val + 0.0005]
        else:
            # Try reducing and slightly increasing
            deltas = [base_val * (1 - 0.1), base_val * (1 - 0.05),
                      base_val * 1.1, base_val * 1.05, base_val]
        
        best_delta = base_val
        best_conv_max = fast_convolve(f).max()
        
        for d in deltas:
            if d < 0 or d > 1:
                continue
            f[coord] = d
            f = normalize_to_sum(f, 1.0)
            conv = fast_convolve(f)
            cmax = conv.max()
            if cmax < best_conv_max - 1e-12:
                best_conv_max = cmax
                best_delta = d
            f[coord] = base_val
        
        f[coord] = best_delta
        f = normalize_to_sum(f, 1.0)
        if best_conv_max < best_obj:
            best_obj = best_conv_max
    
    return normalize_to_sum(f, 1.0), best_obj


def spectral_perturb(f, rng, strength=0.05):
    """Perturb f in frequency domain, maintaining non-negativity and sum."""
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    # Don't perturb DC
    perturb = rng.standard_normal(len(mag)) * strength
    perturb[0] = 0
    mag_new = np.maximum(mag * (1.0 + perturb), 0)
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def phase_perturb(f, rng, strength=0.1):
    """Perturb the phase of f in frequency domain."""
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    phase_pert = phase + rng.standard_normal(len(phase)) * strength
    phase_pert[0] = 0  # Keep DC phase
    if len(phase_pert) > n // 2:
        phase_pert[-1] = 0  # Keep Nyquist
    F_new = mag * np.exp(1j * phase_pert)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new

def make_initialization(n, init_type, rng, alpha=0.3):
    """Create various initializations."""
    if init_type == 'uniform':
        return np.ones(n) / n
    elif init_type == 'linear':
        j = np.arange(n)
        a = 1.0 + alpha * (2.0 * j - (n - 1)) / max(1, n - 1)
        return normalize_to_sum(a, 1.0)
    elif init_type == 'quadratic':
        j = np.arange(n)
        x = (2.0 * j - (n - 1)) / max(1, n - 1)
        a = 1.0 + alpha * x * x
        return normalize_to_sum(a, 1.0)
    elif init_type == 'sine':
        j = np.arange(n)
        a = 1.0 + alpha * np.sin(2 * np.pi * j / n)
        a = np.maximum(a, 0.1)
        return normalize_to_sum(a, 1.0)
    elif init_type == 'cosine':
        j = np.arange(n)
        a = 1.0 + alpha * np.cos(2 * np.pi * j / n)
        a = np.maximum(a, 0.1)
        return normalize_to_sum(a, 1.0)
    elif init_type == 'random':
        a = rng.random(n) + 0.1
        return normalize_to_sum(a, 1.0)
    elif init_type == 'exponential':
        j = np.arange(n)
        a = np.exp(-alpha * j / n)
        return normalize_to_sum(a, 1.0)
    elif init_type == 'bilateral_exp':
        j = np.arange(n)
        center = n / 2.0
        a = np.exp(-alpha * np.abs(j - center) / (n / 2.0))
        return normalize_to_sum(a, 1.0)
    elif init_type == 'ramp_dip':
        # Ramp up then dip
        j = np.arange(n)
        a = j / max(1, n - 1)
        a[n//2:] *= 0.5
        return normalize_to_sum(a, 1.0)
    else:
        return np.ones(n) / n


def try_sequence(a, best_tracker):

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f), endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_budget:
        it += 1
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        # Decide what to do based on progress
        phase = it % 6
        
        if phase in [0, 1]:
            # Targeted peak reduction
            t_alloc = min(remaining * 0.4, 60)
            if t_alloc > 3:
                f_new, obj = targeted_peak_reduction(f, t_alloc, n_peaks=5, rng=rng)
                conv = fast_convolve(f_new)
                if conv.max() < best_obj - 1e-10:
                    best_obj = conv.max()
                    f = f_new
                    no_improve = 0
                elif conv.max() < best_obj:
                    # Tiny improvement, still accept to avoid getting stuck
                    best_obj = conv.max()
                    f = f_new
                    no_improve = 0
                else:
                    no_improve += 1
        
        elif phase in [2, 3]:
            # Coordinate descent
            t_alloc = min(remaining * 0.3, 45)
            if t_alloc > 2:
                f_new, obj = coordinate_descent_pass(f, t_alloc)
                conv = fast_convolve(f_new)
                if conv.max() < best_obj - 1e-10:
                    best_obj = conv.max()
                    f = f_new
                    no_improve = 0
                    cd_passes += 1
                else:
                    no_improve += 1
        
        elif phase == 4:
            # Subgradient step with adaptive beta
            conv = fast_convolve(f)
            beta = 5.0 + 10.0 * min(it / 20.0, 5.0)
            lr = 0.5 / np.sqrt(n)
            for _ in range(5):
                f_new = subgradient_step(f, conv, beta, top_k=15, lr=lr)
                conv_new = fast_convolve(f_new)
                if conv_new.max() < best_obj - 1e-10:
                    best_obj = conv_new.max()
                    f = f_new
                    no_improve = 0
                    conv = conv_new
                else:
                    lr *= 0.7
            if no_improve > 8:
                no_improve += 1
        
        else:
            # Spectral perturbation + brief refinement
            if no_improve > 5:
                strength = 0.02 + 0.01 * min(no_improve / 10.0, 3.0)
                f_pert = spectral_perturb(f, rng, strength=strength)
            else:
                f_pert = spectral_perturb(f, rng, strength=0.01)
            
            # Brief targeted refinement after perturbation
            t_alloc = min(remaining * 0.2, 20)
            if t_alloc > 2:
                f_new, obj = targeted_peak_reduction(f_pert, t_alloc, n_peaks=3, rng=rng)
                conv_new = fast_convolve(f_new)
                if conv_new.max() < best_obj - 1e-10:
                    best_obj = conv_new.max()
                    f = f_new
                    no_improve = 0
                else:
                    # Try the perturbation directly
                    conv_direct = fast_convolve(f_pert)
                    if conv_direct.max() < best_obj - 1e-10:
                        best_obj = conv_direct.max()
                        f = f_pert
                        no_improve = 0
                    else:
                        no_improve += 1
        
        # Occasional full coordinate descent reset
        if no_improve > 15:
            no_improve = 0
            # Reset to best and try a bigger perturbation
            f = project_simplex(f + rng.standard_normal(n) * 0.01)
    
    return normalize_to_sum(f, 1.0), best_obj


def refine_with_cd(f, time_budget, n_passes_max=50):
    """Intensive coordinate descent refinement."""
    n = len(f)
    f = normalize_to_sum(f.copy(), 1.0)
    t0 = time.monotonic()
    
    for p in range(n_passes_max):
        if time.monotonic() - t0 > time_budget:
            break
        
        f_pass, obj = coordinate_descent_pass(f, time_budget - (time.monotonic() - t0))
        f = f_pass
    
    return f


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Multi-initialization, multi-length exploration
    Phase 2: Main iterative optimization (targeted peak reduction + CD + spectral)
    Phase 3: Intensive coordinate descent refinement
    Phase 4: Final polish with small perturbations
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(512) / 512.0
        try_sequence(parent, best_tracker)
    
    # === Phase 1: Multi-initialization exploration ===
    print("\n=== Phase 1: Multi-initialization ===")
    t1_end = time.monotonic() + (deadline - time.monotonic()) * 0.15
    
    n_parent = len(parent)
    n_options = sorted(set([
        n_parent,
        max(256, n_parent - 128),
        min(2000, n_parent + 128),
        max(256, int(n_parent * 0.75)),
        min(2000, int(n_parent * 1.25)),
    ]))
    # Keep to at most 4 lengths
    n_options = n_options[:4]
    
    # Try different initializations for the parent length first
    init_types = ['linear', 'sine', 'quadratic', 'random', 'exponential', 'ramp_dip']
    alphas = [0.2, 0.3, 0.5]
    
    t_alloc_per_init = (t1_end - time.monotonic()) / (len(n_options) * 3 + 1)
    
    for nc in n_options:
        if time.monotonic() > t1_end - 10:
            break
        
        # Start from interpolated parent
        if nc == n_parent:
            f_init = parent.copy()
        else:
            f_init = interpolate_to_length(parent, nc)
        
        # Quick optimization
        t_alloc = min(t_alloc_per_init * 2, 30)
        if t_alloc > 3:
            print(f"  Quick opt n={nc}")
            f_opt, obj = optimize_single_length(f_init, t_alloc, rng, verbose=False)
            try_sequence(f_opt, best_tracker)
        
        # Try a structured initialization
        for itype in init_types[:2]:
            if time.monotonic() > t1_end - 5:
                break
            for alpha in alphas[:2]:
                if time.monotonic() > t1_end - 3:
                    break
                f_init2 = make_initialization(nc, itype, rng, alpha=alpha)
                t_alloc2 = min(15, t_alloc_per_init)
                if t_alloc2 > 3:
                    f_opt2, _ = optimize_single_length(f_init2, t_alloc2, rng, verbose=False)
                    try_sequence(f_opt2, best_tracker)
    
    # === Phase 2: Main optimization on best length ===
    print(f"\n=== Phase 2: Main optimization ===")
    print(f"  Current best: {best_tracker[0]:.8f}")
    
    t2_end = time.monotonic() + (deadline - time.monotonic()) * 0.55
    
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)
    
    # Main optimization loop with increasing intensity
    total_phase2_time = t2_end - time.monotonic()
    
    # Divide into iterations
    n_iters = max(5, int(total_phase2_time / 30))
    iter_budget = total_phase2_time / n_iters
    
    for it in range(n_iters):
        if time.monotonic() > t2_end - 5:
            break
        
        f_start = current_best.copy()
        t_alloc = min(iter_budget, t2_end - time.monotonic() - 2)
        if t_alloc < 3:
            break
        
        # Alternate strategies
        if it % 3 == 0:
            # Standard optimization
            f_opt, obj = optimize_single_length(f_start, t_alloc, rng, verbose=False)
        elif it % 3 == 1:
            # Start from perturbed version
            f_pert = spectral_perturb(f_start, rng, strength=0.03)
            f_opt, obj = optimize_single_length(f_pert, t_alloc, rng, verbose=False)
        else:
            # Start from a different structured init
            itype = init_types[it % len(init_types)]
            alpha = 0.3 + 0.1 * (it % 3)
            f_struct = make_initialization(n_cur, itype, rng, alpha=alpha)
            # Blend with current best
            blend = 0.5 * f_start + 0.5 * f_struct
            f_opt, obj = optimize_single_length(normalize_to_sum(blend, 1.0), t_alloc, rng, verbose=False)
        
        score = try_sequence(f_opt, best_tracker)
        if score < best_tracker[0] + 1e-10:
            current_best = f_opt
    
    # === Phase 3: Intensive coordinate descent refinement ===
    print(f"\n=== Phase 3: CD Refinement ===")
    print(f"  Current best: {best_tracker[0]:.8f}")
    
    t3_end = time.monotonic() + (deadline - time.monotonic()) * 0.25
    
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc_cd = min(120, t3_end - time.monotonic())
    if t_alloc_cd > 5:
        print(f"  CD refinement, t={t_alloc_cd:.0f}s")
        f_refined = refine_with_cd(current_best, t_alloc_cd, n_passes_max=30)
        try_sequence(f_refined, best_tracker)
        
        # Also try CD from a slightly perturbed version
        if time.monotonic() < t3_end - 10:
            f_pert = spectral_perturb(current_best, rng, strength=0.01)
            t_alloc_cd2 = min(60, t3_end - time.monotonic() - 5)
            if t_alloc_cd2 > 5:
                f_refined2 = refine_with_cd(f_pert, t_alloc_cd2, n_passes_max=20)
                try_sequence(f_refined2, best_tracker)
    
    # === Phase 4: Final polish ===
    print(f"\n=== Phase 4: Final polish ===")
    
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)
    
    # Try many small perturbations + quick optimization
    t_rem = deadline - time.monotonic()
    n_attempts = max(5, int(t_rem / 5))
    
    for att in range(n_attempts):
        if time.monotonic() > deadline - 3:
            break
        
        # Different perturbation types
        if att % 3 == 0:
            f_pert = spectral_perturb(current_best, rng, strength=0.005)
        elif att % 3 == 1:
            f_pert = phase_perturb(current_best, rng, strength=0.05)
        else:
            f_pert = project_simplex(current_best + rng.standard_normal(n_cur) * 0.003)
        
        # Quick targeted refinement
        t_alloc = min(3, deadline - time.monotonic() - 2)
        if t_alloc > 1:
            f_opt, _ = targeted_peak_reduction(f_pert, t_alloc, n_peaks=3, rng=rng)
            try_sequence(f_opt, best_tracker)
        
        # Also evaluate the perturbation directly
        try_sequence(f_pert, best_tracker)
        
        # Update current best if improved
        if best_tracker[1] is not None and len(best_tracker[1]) == n_cur:
            # Check if we should update our working copy
            conv_check = fast_convolve(best_tracker[1])
            conv_cur = fast_convolve(current_best)
            if conv_check.max() < conv_cur.max():
                current_best = best_tracker[1].copy()
    
    # === Final: Try a few different lengths with quick optimization ===
    if time.monotonic() < deadline - 15:
        print(f"\n=== Final: Multi-length check ===")
        for nc in [n_cur - 64, n_cur + 64]:
            if nc < 128 or time.monotonic() > deadline - 8:
                continue
            if abs(nc - n_cur) > 128:
                continue
            f_interp = interpolate_to_length(best_tracker[1] if best_tracker[1] is not None else parent, nc)
            t_alloc = min(5, deadline - time.monotonic() - 3)
            if t_alloc > 2:
                f_opt, _ = targeted_peak_reduction(f_interp, t_alloc, n_peaks=5, rng=rng)
                try_sequence(f_opt, best_tracker)
    
    # === Return best ===
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            # Rescale to avoid floating point issues
            s = final_seq.sum()
            if s > 0.01:
                return [float(x) for x in final_seq]
    
    # Fallback
    print("WARNING: Using fallback")
    fallback = np.ones(512) / 512.0
    return [float(x) for x in fallback]
