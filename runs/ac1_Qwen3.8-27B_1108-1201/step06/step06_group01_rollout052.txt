def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smoothed_grad(a, T):
    """Compute (1/T)log(sum exp(T*conv)) and its gradient."""
    n = len(a)

    log_sum = np.log(exp_val.sum())
    smoothed = conv_max + log_sum / T
    p = exp_val / exp_val.sum()

    grad = 2.0 * n * 2.0 * corr  # = 4n * corr

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
- Sum-exp smoothed gradient with momentum and adaptive temperature annealing
- BMESD-style targeted pair-swap local search focusing on peak-contributing indices
- Multi-n exploration with resampling from parent
- Multi-restart local search with diverse perturbations
- Final intensive polish alternating between methods
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A,.fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_and_grad(a, T):
    """Compute smoothed max of auto-convolution and its gradient w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = conv - conv_max
    exp_val = np.exp(T * shifted)
    log_sum = np.log(exp_val.sum())
    smoothed = conv_max + log_sum / T
    p = exp_val / exp_val.sum()

    # Gradient of 2n * smoothed = 4n * (p cross-correlated with a)
    L = 1 << ((3 * n - 2) - 1).bit_length()
    P_fft = np.fft.rfft(p, L)
    A_fft = np.fft.rfft(a, L)
    corr = np.fft.irfft(P_fft * np.conj(A_fft), L)[:n]
    grad = 4.0 * n * corr

    return smoothed, grad


def smooth_grad_optimize(a0, time_limit, T_start=0.5, T_end=50.0, lr_init=0.008):
    """Optimize using sum-exp smoothed gradient with momentum and backtracking."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    momentum = np.zeros(n)
    lr = lr_init
    beta = 0.85
    T = T_start

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        smoothed, grad = compute_smoothed_and_grad(a, T)
        current_obj = 2.0 * n * fast_convolve(a).max()

        # Backtracking line search with momentum
        best_candidate = None
        best_val = current_obj
        best_momentum = None

        for trial_lr in [lr, lr * 0.6, lr * 0.35, lr * 0.15, lr * 0.05]:
            mom_trial = beta * momentum - trial_lr * grad
            cand = project_simplex(a + mom_trial)
            val = 2.0 * n * fast_convolve(cand).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = cand
                best_momentum = mom_trial
                if val < best_obj - 1e-12:
                    break

        if best_candidate is not None:
            a = best_candidate
            momentum = best_momentum
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            lr = min(lr * 1.03, 0.08)
        else:
            no_improve += 1
            momentum = np.zeros(n)
            lr *= 0.75
            if lr < 0.0005:
                lr = lr_init * 0.5

        # Temperature annealing: slowly increase T
        if it % 30 == 0:
            T = min(T * 1.15, T_end)

        # Adaptive restart
        if no_improve > 150:
            rng = np.random.default_rng((it * 48271 + 12345) % (2**31))
            scale = 0.002 + 0.002 * min(1.0, no_improve / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            momentum = np.zeros(n)
            lr = lr_init
            T = max(T * 0.4, T_start)
            no_improve = 0

    return best_a, best_obj


def bmesd_optimize(a0, time_limit, top_k=15, sample_k=25):
    """BMESD-style pair swap optimization targeting peak-contributing indices."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        M = conv[peak_k]

        # Indices contributing to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contributors = np.arange(js, je + 1)
        contrib_vals = a[peak_k - contributors]

        # Top contributors
        n_top = min(top_k, len(contributors))
        top_order = np.argsort(contrib_vals)[::-1][:n_top]
        top_contributors = contributors[top_order]

        # Non-contributors (don't affect this peak)
        non_contrib_mask = np.ones(n, dtype=bool)
        non_contrib_mask[contributors] = False
        non_contribs = np.where(non_contrib_mask)[0]

        if len(non_contribs) == 0:
            break

        rng = np.random.default_rng((it * 31 + 7) % (2**31))
        sample_size = min(sample_k, len(non_contribs))
        sampled_nc = rng.choice(non_contribs, size=sample_size, replace=False)

        best_gain = 0.0
        best_move = None

        # Try transferring from top contributors to non-contributors
        for i in top_contributors:
            if a[i] < 1e-9:
                continue
            for j in sampled_nc:
                for frac in [0.02, 0.04, 0.08, 0.15, 0.25]:
                    delta = a[i] * frac
                    if delta < 1e-10:
                        continue
                    a_test = a.copy()
                    a_test[i] -= delta
                    a_test[j] += delta
                    a_test = np.maximum(a_test, 0.0)
                    st = a_test.sum()
                    if st < 1e-15:
                        continue
                    a_test = a_test / st
                    c_new = fast_convolve(a_test)
                    new_max = c_new.max()
                    gain = M - new_max
                    if gain > best_gain:
                        best_gain = gain
                        best_move = (i, j, delta)

        # Also try between two contributors (redistribute within peak)
        if len(top_contributors) >= 2:
            for ii in range(min(n_top, 8)):
                for jj in range(ii + 1, min(n_top, 8)):
                    i1 = top_contributors[ii]
                    i2 = top_contributors[jj]
                    if a[i1] < 1e-9 or a[i2] < 1e-9:
                        continue
                    # Transfer from the one with larger contribution to smaller
                    c1 = a[peak_k - i1] if 0 <= peak_k - i1 < n else 0
                    c2 = a[peak_k - i2] if 0 <= peak_k - i2 < n else 0
                    if c1 > c2:
                        src, dst = i1, i2
                    else:
                        src, dst = i2, i1
                    for frac in [0.03, 0.06, 0.12]:
                        delta = a[src] * frac
                        if delta < 1e-10:
                            continue
                        a_test = a.copy()
                        a_test[src] -= delta
                        a_test[dst] += delta
                        a_test = np.maximum(a_test, 0.0)
                        st = a_test.sum()
                        if st < 1e-15:
                            continue
                        a_test = a_test / st
                        c_new = fast_convolve(a_test)
                        new_max = c_new.max()
                        gain = M - new_max
                        if gain > best_gain:
                            best_gain = gain
                            best_move = (src, dst, delta)

        if best_move is not None and best_gain > 1e-12:
            i, j, delta = best_move
            a[i] -= delta
            a[j] += delta
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            new_obj = 2.0 * n * fast_convolve(a).max()
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if stall > 40:
            rng = np.random.default_rng((it * 97 + 13) % (2**31))
            scale = 0.001 + 0.002 * min(stall / 100.0, 2.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            stall = 0

        if stall > 400:
            break

    return best_a, best_obj

    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, total time={time.monotonic()-t0_total:.0f}s)")
    return [float(x) for x in best_seq]
