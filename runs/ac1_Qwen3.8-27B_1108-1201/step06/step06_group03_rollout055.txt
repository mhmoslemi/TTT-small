```python
def solve_lp_minmax(f, n):
    """Solve LP: min t s.t. conv_matrix(f) @ g <= t, sum(g) = 1, g >= 0."""
    A = build_conv_matrix(f, n)
    # Variables: g_0, ..., g_{n-1}, t
    # c = [0, ..., 0, 1]  (minimize t)
    c = np.concatenate([np.zeros(n), [1.0]])
    # Constraints: A @ g - t <= 0  =>  [A, -1] @ [g; t] <= 0
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    # Equality: sum(g) = 1  =>  [1, ..., 1, 0] @ [g; t] = 1
    A_eq = np.concatenate([np.ones(n), [0.0]])
    b_eq = np.array([1.0])
    # Bounds: g_i >= 0, t free
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[n]
    return None, None
```

```python
import numpy as np
```

```python
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (f*g)[k] = sum_j A[k,j]*g[j]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_minmax(f, n):
    """
    Solve LP: min t s.t. conv_matrix(f) @ g <= t, sum(g) = 1, g >= 0.
    Returns (g, t) where g is the optimal direction and t is the optimal max convolution.
    """
    A = build_conv_matrix(f, n)
    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.concatenate([np.zeros(n), [1.0]])
    
    # Inequality: A @ g - t <= 0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.concatenate([np.ones(n), [0.0]])
    b_eq = np.array([1.0])
    
    # Bounds: g_i >= 0, t >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x[:n], 0.0)
            t = res.x[n]
            return g, t
    except Exception:
        pass
    return None, None

            peaks = np.arange(len(conv))

        pv = conv[peaks]
        pv_min = pv.min()
        w = np.exp((pv - pv_min) / (T * 0.5))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / w_sum

        # Compute weighted subgradient: grad_j = 2 * sum_k w[k] * a[k-j]
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            # c_k = sum_j a_j * a[k-j], dc_k/da_j = 2*a[k-j]

            je = min(n - 1, k)
            if js <= je:

                grad[idx_arr] += 2.0 * w[i] * a[k - idx_arr]

        # Momentum update
        v = mom * v - lr * grad
        candidate = project_simplex(a + v)
        
        # Backtracking
        best_candidate = a
        best_val = conv.max()

            val = fast_convolve(cand).max()

                best_candidate = cand
                break

def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def spectral_init(n, rng):
    """Generate a sequence with approximately flat power spectrum, projected to non-negative."""
    # Random phase
    phases = rng.uniform(0, 2 * np.pi, n)
    # Flat magnitude (with slight taper to ensure non-negativity after ifft)
    mags = np.ones(n)
    # Slight cosine taper
    x = np.arange(n)
    mags *= 0.5 * (1 + np.cos(np.pi * x / (n - 1)))
    A = mags * np.exp(1j * phases)
    a = np.real(np.fft.ifft(A))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
```

```python
import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_correlate(p, a):
    """Compute sum_k p[k]*a[k-j] for j=0..n-1 using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    # Cross-correlation: ifft(conj(P) * A) gives sum_k p[k]*a[k-j]
    result = np.fft.irfft(np.conj(P) * A, fft_size)[:n]
    return result


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (f*g)[k] = sum_j A[k,j]*g[j]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_minmax(f, n):
    """
    Solve LP: min t s.t. conv_matrix(f) @ g <= t, sum(g) = 1, g >= 0, t >= 0.
    Returns (g, t, duals) where g is optimal direction, t is optimal max conv.
    """
    A = build_conv_matrix(f, n)
    c = np.concatenate([np.zeros(n), [1.0]])
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.concatenate([np.ones(n), [0.0]])
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10.0})
        if res.success and res.x is not None:
            g = np.maximum(res.x[:n], 0.0)
            t = float(res.x[n])
            duals = np.zeros(2 * n - 1)
            if hasattr(res, 'uneqlin') and res.uneqlin is not None:
                duals = np.abs(np.asarray(res.uneqlin.marginals))
            return g, t, duals
    except Exception:
        pass
    return None, None, None

def smoothed_subgradient(a0, time_limit, T_init=0.001, lr_init=0.01, top_k_frac=0.05):
    """Smoothed subgradient with FFT-accelerated gradient and momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(3, int(n * top_k_frac))

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.90

    t0 = time.monotonic()
    stagnant = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        # Compute weighted subgradient using top-k peaks
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))

        pv = conv[peaks]
        pv_min = pv.min()
        w = np.exp((pv - pv_min) / (T * 0.5))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / w_sum

        # Build weighted peak vector for FFT computation
        p_vec = np.zeros(2 * n - 1)
        for i in range(len(peaks)):
            if w[i] > 1e-14:
                p_vec[int(peaks[i])] = w[i]

        # Gradient: grad_j = 2 * sum_k p_k * a[k-j]
        grad = 2.0 * fast_cross_correlate(p_vec, a)

        # Momentum + backtracking
        v = mom * v - lr * grad
        best_candidate = a
        best_val = conv.max()
        for frac in [1.0, 0.7, 0.5, 0.3, 0.1]:
            cand = project_simplex(a + frac * v)
            val = fast_convolve(cand).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = cand
                break

        a = best_candidate
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            lr *= 0.997

        if stagnant > 40:
            T *= 0.7
            lr = max(lr, lr_init * 0.5)
        if stagnant > 150:
            T *= 1.5
            lr = lr_init
            v = np.zeros(n)
            mom = 0.92
        if stagnant > 400:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise = 0.003 + 0.005 * min(1.0, (stagnant - 400) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            v = np.zeros(n)
            mom = 0.90
            stagnant = 0

    return best_a, best_obj

def resample_sequence(a, new_n):
    """Resample a sequence to a new length via interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def spectral_init(n, rng):
    """Generate sequence with approximately flat power spectrum, projected non-negative."""
    phases = rng.uniform(0, 2 * np.pi, n)
    mags = np.ones(n)
    x = np.arange(n)
    mags *= 0.5 * (1 + np.cos(np.pi * x / (n - 1)))
    A = mags * np.exp(1j * phases)
    a = np.real(np.fft.ifft(A))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

            if remaining < 25:

            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 50)
            a_opt, _ = fw_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"fw_pert_{scale}")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Smoothed subgradient (250s) ===
    print("\n=== Phase 3: Subgradient ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        configs = [
            {'T_init': 0.0005, 'lr_init': 0.012, 'top_k_frac': 0.03},
            {'T_init': 0.002, 'lr_init': 0.01, 'top_k_frac': 0.05},
            {'T_init': 0.008, 'lr_init': 0.007, 'top_k_frac': 0.1},
        ]
        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 200
            if remaining < 20:
                break
            t_alloc = min(remaining, 75)
            print(f"  Subgrad config {ci} (T={cfg['T_init']}), t={t_alloc:.0f}s")
            a_opt, _ = smoothed_subgradient(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"subgrad_{ci}")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Coordinate descent (200s) ===
    print("\n=== Phase 4: Coordinate Descent ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 100
        if remaining > 20:
            t_alloc = min(remaining, 180)
            print(f"  Peak coord descent, t={t_alloc:.0f}s")
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_descent")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final interleaved polish ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        polish_count = 0
        while time.monotonic() < deadline - 15:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            polish_count += 1

            if polish_count % 3 == 1:
                # FW with tiny perturbation
                noise = rng.standard_normal(n_best) * 0.0008
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining - 5, 30)
                if t_alloc < 5:
                    break
                a_opt, _ = fw_optimize(start, time_limit=t_alloc)
                try_seq(a_opt, f"polish_fw_{polish_count}")
            elif polish_count % 3 == 2:
                # Subgradient
                noise = rng.standard_normal(n_best) * 0.0005
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining - 5, 25)
                if t_alloc < 5:
                    break
                a_opt, _ = smoothed_subgradient(start, time_limit=t_alloc,
                                                T_init=0.0003, lr_init=0.005,
                                                top_k_frac=0.02)
                try_seq(a_opt, f"polish_sg_{polish_count}")
            else:
                # Coordinate descent
                t_alloc = min(remaining - 5, 20)
                if t_alloc < 5:
                    break
                a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"polish_cd_{polish_count}")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Final coordinate descent with remaining time
    remaining = deadline - time.monotonic()
    if remaining > 10 and best_seq is not None:
        a_opt, _ = peak_coordinate_descent(best_seq, time_limit=min(remaining, 20))
        try_seq(a_opt, "final_coord")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
