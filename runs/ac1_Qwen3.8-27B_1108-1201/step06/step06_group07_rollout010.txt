def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_conv(a, beta):
    """Compute (1/beta)*log(sum exp(beta*conv)) and its gradient w.r.t. a."""
    conv = fast_convolve(a)
    n = len(a)
    
    # For numerical stability, shift by max
    m = conv.max()
    shifted = beta * (conv - m)
    log_sum = m * beta + np.log(np.sum(np.exp(shifted)))
    obj = log_sum / beta  # = m + (1/beta)*log(sum exp(beta*(conv-m)))
    
    # Softmax weights
    p = np.exp(shifted)
    p = p / p.sum()
    
    # Gradient: grad[j] = sum_k p[k] * a[k-j] (for valid k-j)
    grad = np.zeros(n)
    # For efficiency, only process significant p[k]
    threshold = p.max() * 1e-10
    significant = p > threshold
    if significant.sum() < 3:
        significant = p > 0
    
    for k in np.where(significant)[0]:
        pk = p[k]
        if pk < 1e-15:
            continue
        js = max(0, int(k) - n + 1)
        je = min(n - 1, int(k))
        grad[js:je+1] += pk * a[k - je:k - js + 1][::-1]
    
    return obj, grad, conv
