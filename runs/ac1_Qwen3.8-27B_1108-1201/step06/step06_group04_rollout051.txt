def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero-cond)[0][-1] if cond.any() else 0
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_peaks_mv(peaks, weights, a):
    n = len(a)
    result = np.zeros(n)
    for idx in range(len(peaks)):
        k = int(peaks[idx])
        w = weights[idx]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            result[js:je + 1] += w * a[k - np.arange(js, je + 1)]
    return result


def solve_qp_nesterov(peaks, weights, a_init, n, max_iters=30000, time_limit=8.0):
    t0 = time.monotonic()
    a = project_simplex(a_init.copy())
    prev_a = a.copy()
    
    lr = 0.5  # 1/L where L = 2*lambda_max(W) <= 2
    t_nest = 1.0
    best_a = a.copy()
    best_val = float(a @ hankel_peaks_mv(peaks, weights, a))
    
    it = 0
    while it < max_iters and time.monotonic() - t0 < time_limit:
        it += 1
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / t_new
        t_nest = t_new
        
        y = a + beta * (a - prev_a)
        y = project_simplex(y)
        prev_a = a
        
        grad = 2.0 * hankel_peaks_mv(peaks, weights, y)
        a_new = project_simplex(y - lr * grad)
        a = a_new
        
        val = float(a @ hankel_peaks_mv(peaks, weights, a))
        if val < best_val:
            best_val = val
            best_a = a.copy()
    
    return best_a, best_val

def smoothed_gradient_descent(a0, time_limit, T_init=0.001, lr_init=0.02, verbose=False):

    n = len(a)
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Gradient: 2 * sum_k p_k * shift_k(a)
        grad = np.zeros(n)
        # Use the fact that grad[i] = 2 * sum_k p_k * a[k-i]
        # This is a correlation
        for k_idx in range(len(p)):
            if p[k_idx] < 1e-15:
                continue
            k = k_idx
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                grad[js:je + 1] += 2.0 * p[k] * a[k - np.arange(js, je + 1)]
        
        best_candidate = a
        best_val = best_obj
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            val = float(fast_convolve(candidate).max())
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                break
        
        a = best_candidate
        obj = float(fast_convolve(a).max())
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 20:
            T *= 1.2
            lr *= 0.9
        if no_improve > 60:
            T *= 0.7
        if no_improve > 150:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj

def generate_inits(n, parent, seed=42):

    # Triangle
    tri = 1.0 - 2.0 * np.abs(x / n - 0.5)
    tri = np.maximum(tri, 0.001)
    inits.append(tri / tri.sum())

    h2 = np.exp(-((x - 0.7 * (n-1)) / (0.2 * n))**2)
    two = h1 + h2

    # Alternating (high-low pattern smoothed)
    alt = np.ones(n)
    alt[::2] = 1.0
    alt[1::2] = 0.5
    # Smooth it
    kernel = np.ones(5) / 5
    alt_smooth = np.convolve(alt, kernel, mode='same')

    if parent is not None:

        else:
            px = parent.copy()
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())
    
    return inits

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
