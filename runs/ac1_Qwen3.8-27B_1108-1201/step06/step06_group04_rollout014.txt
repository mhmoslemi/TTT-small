**Algorithm: Mirror Descent / Multiplicative Weights for Auto-Convolution Minimization**

Given:
- `n`: sequence length
- Time budget `T`

1. Initialize `a = ones(n) / n` (or from parent)
2. Initialize `p = ones(2n-1) / (2n-1)` (uniform over lags)
3. For each iteration:

   b. Update `p`: `p = softmax(c / T_p)` (or `p = c / sum(c)` for the "natural" dual update)
   c. Compute `grad = 2 * correlate(p, a, 'valid')` (length `n`)
   d. Update `a`: `a = a * exp(-eta * grad)` then normalize
   e. Track best `a` (minimizing `max(conv(a,a))`)
4. Return best `a`

The key hyperparameters are:
- `eta`: learning rate for the primal update
- `T_p`: temperature for the dual update (controls how "peaked" p is)
- Number of iterations

For the dual update, using `p = c / sum(c)` (proportional to the convolution values) is the "natural" choice that corresponds to the worst-case lag. Using `p = softmax(c / T_p)` with small `T_p` approximates this.

def qp_refine(a0, p, n, time_limit=2.0):
    """
    Refine a by solving min a^T W a s.t. sum(a)=1, a>=0
    where W[i,j] = p[i+j]. Active-set method.
    """
    a = a0.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    t0 = time.monotonic()
    active = set(np.where(a > 1e-10)[0])
    if not active:
        active = set(range(n))
    
    for _ in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        if len(active) == 0:
            return np.ones(n) / n
        if len(active) == 1:
            a = np.zeros(n)
            a[list(active)[0]] = 1.0
            return a
        
        act = sorted(active)
        k = len(act)
        
        # Build W_A (Hankel submatrix)
        W_AA = np.array([[p[act[i] + act[j]] for j in range(k)] for i in range(k)])
        W_AA += 1e-12 * np.eye(k)
        
        try:
            x = np.linalg.solve(W_AA, np.ones(k))
        except np.linalg.LinAlgError:
            x = np.ones(k) / k
        
        x_sum = x.sum()
        if abs(x_sum) < 1e-14:
            x = np.ones(k) / k
            x_sum = 1.0
        
        a_act = x / x_sum
        
        # Remove most negative
        min_idx = -1
        min_val = float('inf')
        for i, idx in enumerate(act):
            if a_act[i] < min_val:
                min_val = a_act[i]
                min_idx = idx
        
        if min_val < -1e-10:
            active.discard(min_idx)
            a = np.zeros(n)
            for i, idx in enumerate(act):
                if idx != min_idx:
                    a[idx] = max(0.0, a_act[i])
            s2 = a.sum()
            if s2 > 1e-15:
                a = a / s2
            continue
        
        # Form solution
        a = np.zeros(n)
        for i, idx in enumerate(act):
            a[idx] = max(0.0, a_act[i])
        
        # Gradient check
        Wa = np.correlate(p, a, mode='valid')
        mu = np.mean([2.0 * Wa[idx] for idx in act])
        
        # Check KKT for inactive
        worst_viol = 0.0
        worst_j = -1
        for j in range(n):
            if j not in active:
                viol = 2.0 * Wa[j] - mu
                if viol < worst_viol:
                    worst_viol = viol
                    worst_j = j
        
        if worst_viol < -1e-8:
            active.add(worst_j)
            continue
        
        return a
    
    return a

def gradient_descent_refine(a0, p, n, time_limit=2.0, lr=0.01):
    """Gradient descent on the simplex with projection."""
    a = a0.copy() / a0.sum()
    best_a = a.copy()
    best_obj = fast_linear_conv(a).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:

        grad = 2.0 * Wa
        a_new = project_simplex(a - lr * grad)
        obj = fast_linear_conv(a_new).max()

            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.9
        if lr < 1e-6:
            break
    
    return best_a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- FFT-accelerated multiplicative weights (mirror descent) on the simplex
- Hedge-style exponential weights dual update for lag concentration
- Periodic active-set QP refinement with Hankel structure
- Multi-n exploration with adaptive time allocation
- Gradient descent polish with line search
"""

import numpy as np
import time


def fast_linear_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute W @ a where W[i,j] = p[i+j] via correlation (valid mode)."""
    return np.correlate(p, a, mode='valid')


def multiplicative_weights(a0, n, time_limit, eta=0.5, T_dual=0.01,
                           refine_every=1000, verbose=False):
    """
    Mirror descent with multiplicative weights update for minimizing
    max(conv(a,a)) s.t. a in simplex. Uses exponential weights (Hedge)
    for the dual variable (lag distribution).
    """
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_linear_conv(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Compute convolution
        c = fast_linear_conv(a)
        cmax = c.max()

        if cmax < best_obj - 1e-12:
            best_obj = cmax
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Dual update: softmax concentration on current peaks
        if T_dual > 0.0005:
            scaled = c / T_dual
            scaled -= scaled.max()
            p = np.exp(scaled)
            p_sum = p.sum()
            if p_sum < 1e-30:
                p = c / c.sum()
            else:
                p = p / p_sum
        else:
            p = c / c.sum()

        # Primal update: multiplicative weights (Hedge on simplex)
        grad = 2.0 * np.correlate(p, a, mode='valid')
        grad_centered = grad - grad.mean()
        grad_scale = max(np.abs(grad_centered).max(), 1e-10)
        eta_adapt = eta / grad_scale * 0.5
        a = a * np.exp(-eta_adapt * grad_centered)
        a_sum = a.sum()
        if a_sum < 1e-30:
            a = best_a.copy()
            continue
        a = a / a_sum

        # Periodic QP refinement
        if refine_every > 0 and it % refine_every == 0:
            remaining = time_limit - (time.monotonic() - t0)
            if remaining > 3:
                a_qp = qp_refine(a, p, n, time_limit=min(3.0, remaining * 0.5))
                c_qp = fast_linear_conv(a_qp)
                if c_qp.max() < best_obj - 1e-12:
                    best_obj = c_qp.max()
                    best_a = a_qp.copy()
                    no_improve = 0
                a = a_qp

        # Adapt temperature: cool down over time, reheat on stagnation
        if no_improve > 300:
            T_dual *= 0.7
        if no_improve > 800:
            T_dual *= 0.5

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MW it={it}, obj={best_obj:.10f}, T_d={T_dual:.6f}, no_imp={no_improve}, t={elapsed:.1f}s")

    return best_a, best_obj


def qp_refine(a0, p, n, time_limit=5.0):
    """
    Active-set QP: min a^T W a s.t. sum(a)=1, a>=0, W[i,j]=p[i+j].
    """
    a = a0.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    t0 = time.monotonic()
    active = set(np.where(a > 1e-10)[0])
    if not active:
        active = set(range(n))

    for _ in range(100):
        if time.monotonic() - t0 > time_limit:
            break
        if len(active) == 0:
            return np.ones(n) / n
        if len(active) == 1:
            a = np.zeros(n)
            a[list(active)[0]] = 1.0
            return a

        act = sorted(active)
        k = len(act)

        if k > 500:
            # For large active sets, use gradient step instead
            Wa = np.correlate(p, a, mode='valid')
            grad = 2.0 * Wa
            mu = np.mean([grad[idx] for idx in act])
            a_new = project_simplex(a - 0.01 * (grad - mu))
            return a_new

        W_AA = np.array([[p[act[i] + act[j]] for j in range(k)] for i in range(k)])
        W_AA += 1e-12 * np.eye(k)

        try:
            x = np.linalg.solve(W_AA, np.ones(k))
        except np.linalg.LinAlgError:
            try:
                x = np.linalg.lstsq(W_AA, np.ones(k), rcond=None)[0]
            except:
                x = np.ones(k) / k

        x_sum = x.sum()
        if abs(x_sum) < 1e-14:
            x = np.ones(k) / k
            x_sum = 1.0

        a_act = x / x_sum

        # Find most negative
        min_val = float('inf')
        min_pos = -1
        for i in range(k):
            if a_act[i] < min_val:
                min_val = a_act[i]
                min_pos = i

        if min_val < -1e-10:
            active.discard(act[min_pos])
            a = np.zeros(n)
            for i in range(k):
                if i != min_pos:
                    a[act[i]] = max(0.0, a_act[i])
            s2 = a.sum()
            if s2 > 1e-15:
                a = a / s2
            continue

        # Form solution
        a = np.zeros(n)
        for i in range(k):
            a[act[i]] = max(0.0, a_act[i])

        # KKT check
        Wa = np.correlate(p, a, mode='valid')
        mu = np.mean([Wa[idx] for idx in act])

        worst_viol = 0.0
        worst_j = -1
        for j in range(n):
            if j not in active:
                viol = Wa[j] - mu
                if viol < worst_viol:
                    worst_viol = viol
                    worst_j = j

        if worst_viol < -1e-8:
            active.add(worst_j)
            continue

        return a

    return a


def gradient_polish(a0, n, time_limit=10.0, verbose=False):
    """
    Gradient descent with backtracking line search on the simplex.
    Uses smoothed (softmax) surrogate for the max.
    """
    a = a0.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = fast_linear_conv(a).max()

    t0 = time.monotonic()
    T = 0.005
    lr = 0.02
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        c = fast_linear_conv(a)
        cmax = c.max()
        if cmax < best_obj - 1e-12:
            best_obj = cmax
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Smoothed gradient
        scaled = c / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = 2.0 * np.correlate(p, a, mode='valid')

        # Line search
        best_candidate = a
        best_val = cmax
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_linear_conv(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate

        # Adapt
        if no_improve > 20:
            T *= 1.15
            lr *= 0.9
        if no_improve > 100:
            T = max(0.001, T * 0.6)
        if no_improve > 500:
            rng_local = np.random.default_rng(it * 7919 + 3)
            a = project_simplex(best_a + rng_local.standard_normal(n) * 0.002)
            T = 0.005
            lr = 0.02
            no_improve = 0

    return best_a


def peak_targeted_push(a0, n, time_limit=10.0):
    """
    For each dominant peak lag, try to reduce it by shifting mass
    from contributing positions to non-contributing ones.
    """
    a = a0.copy() / a0.sum()
    best_a = a.copy()
    best_obj = fast_linear_conv(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        c = fast_linear_conv(a)
        k_peak = int(np.argmax(c))

        # Contributing indices for this lag
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = np.arange(js, je + 1)
        non_contrib = np.setdiff1d(np.arange(n), contrib)

        if len(contrib) == 0 or len(non_contrib) == 0:
            break

        improved = False
        # Try shifting from largest contributor to non-contributors
        idx_max = contrib[np.argmax(a[contrib])]
        val = a[idx_max]

        if val > 1e-10:
            for frac in [0.01, 0.03, 0.05, 0.1, 0.15]:
                shift = val * frac
                delta = np.zeros(n)
                delta[idx_max] -= shift
                nc_vals = a[non_contrib]
                nc_sum = nc_vals.sum()
                if nc_sum > 1e-15:
                    delta[non_contrib] += shift * nc_vals / nc_sum
                else:
                    delta[non_contrib] += shift / len(non_contrib)

                candidate = project_simplex(a + delta)
                c_new = fast_linear_conv(candidate)
                if c_new.max() < best_obj - 1e-14:
                    a = candidate
                    best_obj = c_new.max()
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            # Try random perturbation on contributing positions
            rng_local = np.random.default_rng(it * 31 + k_peak)
            delta = np.zeros(n)
            delta[contrib] = rng_local.standard_normal(len(contrib)) * 0.003
            candidate = project_simplex(a + delta)
            c_new = fast_linear_conv(candidate)
            if c_new.max() < best_obj - 1e-14:
                a = candidate
                best_obj = c_new.max()
                best_a = a.copy()
                improved = True

        if not improved and it > 200:
            break

    return best_a


def resample_to_n(parent, n):
    """Resample a sequence to length n via linear interpolation."""
    if len(parent) == n:
        return parent.copy()
    x_old = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n)
    resampled = np.interp(x_new, x_old, parent)
    resampled = np.maximum(resampled, 0)
    s = resampled.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return resampled / s
