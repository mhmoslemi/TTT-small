def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_qp_hankel(p):
    """
    Solve QP: min a^T H a s.t. a >= 0, sum(a) = 1
    where H[i,j] = p[i+j] (Hankel matrix).
    
    Uses active set method.
    """
    n = len(p) // 2  # p has length 2n-1, so n = (len(p)+1)//2
    # Actually, p has length 2n-1 where n is the sequence length
    # H[i,j] = p[i+j] for i,j in {0,...,n-1}
    # So p needs to have indices 0 to 2n-2, length 2n-1
    
    # Build H
    idx = np.arange(n)
    H = p[idx[:, None] + idx[None, :]]  # Hankel matrix
    
    # Active set method
    active = np.ones(n, dtype=bool)  # Start with all active
    max_iters = 50
    
    for _ in range(max_iters):
        if not active.any():
            a = np.ones(n) / n
            break
        
        H_aa = H[np.ix_(active, active)]
        # Solve H_aa * a_a = lambda * 1 (from KKT: 2 H a = lambda * 1 on active set)
        # Actually: 2 H a - lambda * 1 - mu = 0, on active set mu = 0
        # So: 2 H_aa a_a = lambda * 1
        # And: sum(a_a) = 1
        # From first: a_a = lambda * (2 H_aa)^{-1} 1
        # From second: lambda * 1^T (2 H_aa)^{-1} 1 = 1
        # So: lambda = 1 / (1^T (2 H_aa)^{-1} 1)
        #     a_a = (2 H_aa)^{-1} 1 / (1^T (2 H_aa)^{-1} 1)
        
        try:
            H_inv_1 = np.linalg.solve(H_aa, np.ones(active.sum()))
            denom = H_inv_1.sum()
            if abs(denom) < 1e-15:
                a = project_simplex(np.ones(n) * 0.1)
                break
            a_a = H_inv_1 / denom
        except np.linalg.LinAlgError:
            # Fallback: use pseudoinverse
            H_pinv = np.linalg.pinv(H_aa)
            H_inv_1 = H_pinv @ np.ones(active.sum())
            denom = H_inv_1.sum()
            if abs(denom) < 1e-15:
                a = project_simplex(np.ones(n) * 0.1)
                break
            a_a = H_inv_1 / denom
        
        # Reconstruct a
        a = np.zeros(n)
        a[active] = a_a
        
        # Check KKT conditions
        # mu = 2 * H @ a - a[active].mean() * 2 * ... 
        # Actually: mu_i = 2*(H@a)_i - lambda for inactive i, should be >= 0
        # lambda = 2 * a_a.mean() ... no
        # lambda = 1 / (0.5 * 1^T H_aa^{-1} 1) ... let me redo
        # KKT: 2 H a - lambda * 1 - mu = 0
        # On active set: 2 H a = lambda * 1
        # lambda = 2 * (H@a)_{active} (any active index)
        # Actually: 2 * (H@a)[active] = lambda * ones, so lambda = 2*(H@a)[active][0]
        
        Ha = H @ a
        lambda_val = 2.0 * Ha[active].mean()
        mu_inactive = 2.0 * Ha[~active] - lambda_val
        
        # Check if any a_a < 0 (constraint violation)
        if (a_a < -1e-12).any():
            # Remove the most violated variable
            worst_idx = np.where(active)[0][np.argmin(a_a)]
            active[worst_idx] = False
            continue
        
        # Check if any mu_inactive < 0 (should free variable)
        if mu_inactive.size > 0 and (mu_inactive < -1e-12).any():
            worst_idx = np.where(~active)[0][np.argmin(mu_inactive)]
            active[worst_idx] = True
            continue
        
        # Converged
        break
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    return a

But wait, there's a subtlety. In the game-theoretic framework:
- Player 1 minimizes E_p[f(a)] = a^T H(p) a
- Player 2 maximizes f(a) = max_k a^T B_k a

At equilibrium, both players are best-responding. The iteration I described is:
1. Given p, find a = argmin a^T H(p) a (Player 1 best response)
2. Given a, find p = argmax_k a^T B_k a (Player 2 best response, which is a point mass on the argmax)

If we use a point mass for p (hard update), then p is a single k*, and H(p) has H[i,j] = 1 if i+j=k*, else 0. This is a very sparse matrix, and a^T H a = conv_a[k*]. Minimizing this subject to a >= 0, sum a = 1 means: minimize the k*-th convolution coefficient. The solution would put all mass on a single element (if k* is even, at position k*/2; if odd, split between two positions). Then the convolution at k* would be 1 (or 1/4 for the split case). But this doesn't help because other convolution positions would be large.

So we need to use a softmax (or some smoothing) for p, not a hard update. The temperature T controls this.

Actually, the correct game-theoretic iteration is:
1. Given p, solve QP to get a.
2. Update p: p[k] proportional to conv_a[k] (or exponentiated for smoothing).

This is like a multiplicative weights update. The equilibrium is a Nash equilibrium of the game.

Game-theoretic QP optimization for minimum peak auto-convolution.
Uses exact QP oracle (active set) within a multiplicative weights game iteration,
combined with FW-LP exploration and local search polish.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_qp_hankel(p, n, eps_reg=1e-10):
    """
    Solve QP: min a^T H a s.t. a >= 0, sum(a) = 1
    where H[i,j] = p[i+j] (Hankel matrix of size n x n).
    
    Uses active set method with regularization fallback.
    """
    idx = np.arange(n)
    row_idx = idx[:, None]
    col_idx = idx[None, :]
    H = p[row_idx + col_idx]
    
    # Add small regularization for numerical stability
    H = H + eps_reg * np.eye(n)
    
    active = np.ones(n, dtype=bool)
    a = np.ones(n) / n
    max_iters = n + 10
    
    for _ in range(max_iters):
        if not active.any():
            a = np.ones(n) / n
            break
        
        H_aa = H[np.ix_(active, active)]
        n_active = active.sum()
        ones_act = np.ones(n_active)
        
        try:
            H_inv_1 = np.linalg.solve(H_aa, ones_act)
        except np.linalg.LinAlgError:
            try:
                H_inv_1 = np.linalg.lstsq(H_aa, ones_act, rcond=None)[0]
            except:
                a = np.zeros(n)
                a[active] = 1.0 / n_active
                break
        
        denom = H_inv_1.sum()
        if abs(denom) < 1e-14:
            a = np.zeros(n)
            a[active] = 1.0 / n_active
            break
        
        a_a = H_inv_1 / denom
        
        # Reconstruct
        a = np.zeros(n)
        a[active] = a_a
        
        # Check KKT
        Ha = H @ a
        lambda_val = 2.0 * Ha[active].mean()
        
        # Check if any active variable is negative
        if a_a.min() < -1e-10:
            # Remove the most negative
            worst_local = np.argmin(a_a)
            worst_global = np.where(active)[0][worst_local]
            active[worst_global] = False
            continue
        
        # Check if any inactive variable should be freed
        if (~active).any():
            mu_inact = 2.0 * Ha[~active] - lambda_val
            if mu_inact.min() < -1e-10:
                worst_local = np.argmin(mu_inact)
                worst_global = np.where(~active)[0][worst_local]
                active[worst_global] = True
                continue
        
        # Converged (all KKT conditions satisfied)
        break
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-14:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def game_qp_optimize(a0, time_limit, T_init=0.003, T_min=0.0001, T_decay=0.995):
    """
    Game-theoretic iteration with exact QP oracle.
    min_a max_k conv_a[k] s.t. a >= 0, sum(a) = 1
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    T = T_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.2:
            break
        
        # Compute p from current a
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Solve QP
        a_new = solve_qp_hankel(p, n)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        obj = conv_new.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Blend: move partially towards a_new
            if obj < fast_convolve(a).max():
                a = 0.5 * a + 0.5 * a_new
                a = a / a.sum()
        
        # Adapt temperature
        T = max(T * T_decay, T_min)
        
        # Periodic restart to escape local minima
        if no_improve > 200:
            rng = np.random.default_rng((it * 7919 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init * 0.5
            no_improve = 0
    
    return best_a, best_obj


def build_lp_matrix(n):
    """Build sparse constraint structure for LP oracle."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_oracle(f, M, n, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    vals = np.empty(len(row_idx))
    for idx in range(len(row_idx)):
        k = row_idx[idx]
        i = col_idx[idx]
        ki = k - i
        if 0 <= ki < n:
            vals[idx] = f[ki]
        else:
            vals[idx] = 0.0

    A_mat = lil_matrix((2 * n - 1, n))
    A_mat[row_idx, col_idx] = vals
    A_mat = A_mat.tocsr()

    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.fun is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def fw_lp_step(f, g, n):
    """Line search between f and g."""
    best_alpha = 0.0
    best_val = fast_convolve(f).max()
    alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
    for alpha in alphas:
        ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    if 0 < best_alpha < 1:
        lo = max(0.0, best_alpha - 0.15)
        hi = min(1.0, best_alpha + 0.15)
        for _ in range(10):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = np.maximum((1 - m1) * f + m1 * g, 0)
            f1 = f1 / f1.sum()
            v1 = fast_convolve(f1).max()
            f2 = np.maximum((1 - m2) * f + m2 * g, 0)
            f2 = f2 / f2.sum()
            v2 = fast_convolve(f2).max()
            if v1 < v2:
                hi = m2
            else:
                lo = m1
        mid = (lo + hi) / 2.0
        ft = np.maximum((1 - mid) * f + mid * g, 0)
        ft = ft / ft.sum()
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = mid
    
    return best_alpha, best_val

def smooth_grad_optimize(a0, time_limit, T_init=0.001, lr_init=0.02):
    """Smoothed gradient descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Gradient of E_p[conv] = 2 * sum_k p[k] * B_k @ a
        # B_k @ a has entries: (B_k @ a)[j] = sum_m a[m] * [j + m = k offset]
        # Actually: conv[k] = sum_i a[i]*a[k-i], d(conv[k])/d(a[j]) = sum_i [i=j or k-i=j] a[i]
        # = a[k-j] + a[j] (with appropriate bounds) ... this is getting complex
        # Simpler: use the Hankel structure
        # grad[j] = 2 * sum_m a[m] * p[j+m] (for valid j+m < 2n-1)
        idx = np.arange(n)
        p_ext = p  # length 2n-1
        grad = np.zeros(n)
        for j in range(n):
            valid = j + idx  # ranges from j to j+n-1
            grad[j] = 2.0 * np.dot(a, p_ext[valid])
        
        # Projected gradient step
        best_candidate = a
        best_val = conv.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
        
        a = best_candidate
        obj = best_val
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 30:
            T *= 1.02
            lr = max(lr * 0.99, 0.001)
        if no_improve > 100:
            T *= 0.7
        if no_improve > 300:
            rng = np.random.default_rng((it * 31 + 7) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            T = T_init
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj


def local_search(a0, time_limit, seed=42):
    """Local search with random perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        
        improved = False
        
        # Multi-scale perturbations
        for scale in [0.0002, 0.0005, 0.001, 0.003, 0.008]:
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(a + pert)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break
        
        # Pair moves
        if not improved and it % 3 == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    candidate = a.copy()
                    amount = candidate[i] * frac
                    candidate[i] -= amount
                    candidate[j] += amount
                    candidate = np.maximum(candidate, 0)
                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break
        
        # Neighborhood swap (move mass from one region to another)
        if not improved and it % 10 == 0:
            block_size = max(1, n // 32)
            start1 = int(rng.integers(0, n - block_size))
            start2 = int(rng.integers(0, n - block_size))
            if start1 != start2:
                for frac in [0.05, 0.1, 0.2]:
                    candidate = a.copy()
                    # Move some mass from block1 to block2
                    mass_to_move = candidate[start1:start1+block_size].sum() * frac
                    per_el = mass_to_move / block_size
                    candidate[start1:start1+block_size] -= per_el
                    candidate[start2:start2+block_size] += per_el
                    candidate = np.maximum(candidate, 0)
                    ss = candidate.sum()
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved and it > 5000:
            break
    
    return best_a, best_obj


def try_seq(a, label, best_seq, best_score):
    """Evaluate a sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:

        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return a_norm.copy(), score
    return best_seq, best_score

            a = np.ones(n) / n
            break
        
        H_aa = H[np.ix_(active, active)]
        n_act = active.sum()
        
        # Check condition number
        try:
            eigmin = np.linalg.eigvalsh(H_aa).min()
            if eigmin < 1e-12:
                # Too ill-conditioned, add more regularization
                H_aa += 1e-8 * np.eye(n_act)
        except:
            pass
        
        try:
            H_inv_1 = np.linalg.solve(H_aa, np.ones(n_act))
        except np.linalg.LinAlgError:
            H_inv_1 = np.linalg.pinv(H_aa) @ np.ones(n_act)
        
        denom = H_inv_1.sum()
        if abs(denom) < 1e-12:
            a = np.zeros(n)
            a[active] = 1.0 / n_act
            break
        
        a_a = H_inv_1 / denom
        a = np.zeros(n)
        a[active] = a_a
        
        # KKT check
        Ha = H @ a
        lambda_val = 2.0 * Ha[active].mean()
        
        # Remove negative variables
        neg_mask = a_a < -1e-10
        if neg_mask.any():
            # Remove the most negative
            worst = np.where(active)[0][np.argmin(a_a)]
            active[worst] = False
            continue
        
        # Free variables that should be active
        if (~active).any():
            mu = 2.0 * Ha[~active] - lambda_val
            free_mask = mu < -1e-10
            if free_mask.any():
                worst = np.where(~active)[0][np.argmin(mu)]
                active[worst] = True
                continue
        
        break
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

"""
Game-theoretic QP optimization for minimum peak auto-convolution.
Uses exact QP oracle (active set method) within a multiplicative weights game iteration,
combined with FW-LP exploration, smoothed gradient, and local search polish.
Key innovation: exact QP solve in the game-theoretic loop (replacing projected gradient),
plus top-k p-update variant and adaptive multi-n exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_qp_hankel(p, n, eps_reg=1e-8):
    """
    Solve QP: min a^T H a s.t. a >= 0, sum(a) = 1
    where H[i,j] = p[i+j] (Hankel matrix, n x n).
    Uses active set method with regularization.
    """
    idx = np.arange(n)
    H = p[idx[:, None] + idx[None, :]].astype(np.float64).copy()
    H += eps_reg * np.eye(n)

    active = np.ones(n, dtype=bool)
    a = np.ones(n) / n
    max_iters = n + 20

    for it in range(max_iters):
        if not active.any():
            a = np.ones(n) / n
            break

        n_act = active.sum()
        H_aa = H[np.ix_(active, active)]

        # Regularize if needed
        if n_act > 1:
            try:
                if np.linalg.cond(H_aa) > 1e12:
                    H_aa += 1e-6 * np.eye(n_act)
            except:
                pass

        try:
            H_inv_1 = np.linalg.solve(H_aa, np.ones(n_act))
        except np.linalg.LinAlgError:
            try:
                H_inv_1 = np.linalg.pinv(H_aa) @ np.ones(n_act)
            except:
                a = np.zeros(n)
                a[active] = 1.0 / n_act
                break

        denom = H_inv_1.sum()
        if abs(denom) < 1e-14:
            a = np.zeros(n)
            a[active] = 1.0 / n_act
            break

        a_a = H_inv_1 / denom
        a = np.zeros(n)
        a[active] = a_a

        # KKT check
        Ha = H @ a
        lambda_val = 2.0 * Ha[active].mean()

        # Remove most negative active variable
        if a_a.min() < -1e-10:
            worst = np.where(active)[0][np.argmin(a_a)]
            active[worst] = False
            continue

        # Free most violated inactive variable
        if (~active).any():
            mu = 2.0 * Ha[~active] - lambda_val
            if mu.min() < -1e-10:
                worst = np.where(~active)[0][np.argmin(mu)]
                active[worst] = True
                continue

        # Converged
        break

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-14:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def game_qp_optimize(a0, time_limit, T_init=0.003, T_min=0.0001, T_decay=0.995,
                     topk_blend=0.0, restart_scale=0.003):
    """
    Game-theoretic iteration with exact QP oracle.
    Optionally blends top-k hard update with softmax.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    T = T_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)

        # Compute p: blend of softmax and top-k
        scaled = conv / T
        scaled -= scaled.max()
        p_soft = np.exp(scaled)
        p_soft = p_soft / p_soft.sum()

        if topk_blend > 0:
            k = max(1, int(topk_blend * n))
            topk_idx = np.argsort(conv)[-k:]
            p_hard = np.zeros(2 * n - 1)
            p_hard[topk_idx] = 1.0 / k
            p = (1.0 - topk_blend) * p_soft + topk_blend * p_hard
        else:
            p = p_soft

        a_new = solve_qp_hankel(p, n)
        conv_new = fast_convolve(a_new)
        obj = conv_new.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try a blend for potential escape
            obj_blend = fast_convolve(0.5 * a + 0.5 * a_new).max()
            obj_blend = min(obj_blend, fast_convolve(a).max())
            if obj_blend < fast_convolve(a).max():
                a = 0.5 * a + 0.5 * a_new
                a = a / a.sum()

        T = max(T * T_decay, T_min)

        if no_improve > 150:
            rng = np.random.default_rng((it * 7919 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * restart_scale)
            T = max(T_init * 0.3, T_min)
            no_improve = 0

    return best_a, best_obj


def build_lp_matrix(n):
    """Build sparse constraint structure for LP oracle."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_oracle(f, M, n, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    vals = np.empty(len(row_idx))
    for idx in range(len(row_idx)):
        k = row_idx[idx]
        i = col_idx[idx]
        ki = k - i
        vals[idx] = f[ki] if 0 <= ki < n else 0.0

    A_mat = lil_matrix((2 * n - 1, n))
    A_mat[row_idx, col_idx] = vals
    A_mat = A_mat.tocsr()

    b = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.fun is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    row_idx, col_idx = build_lp_matrix(n)
    M = fast_convolve(f).max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        g = solve_lp_oracle(f, M * m_exp, n, row_idx, col_idx)
        if g is None:
            m_exp = min(m_exp * 1.4, 8.0)
            if m_exp >= 8.0:
                rng = np.random.default_rng(it * 31 + 7)
                g = project_simplex(f + rng.standard_normal(n) * 0.005)
                m_exp = 1.0
            else:
                continue

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                rng = np.random.default_rng(it * 7919 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                M = fast_convolve(f).max()
                m_exp = 1.0
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            ft = ft / ft.sum()
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if 0 < best_alpha < 1:
            lo, hi = max(0.0, best_alpha - 0.1), min(1.0, best_alpha + 0.1)
            for _ in range(10):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.maximum((1 - m1) * f + m1 * g, 0)
                f1 = f1 / f1.sum()
                f2 = np.maximum((1 - m2) * f + m2 * g, 0)
                f2 = f2 / f2.sum()
                if fast_convolve(f1).max() < fast_convolve(f2).max():
                    hi = m2
                else:
                    lo = m1
            mid = (lo + hi) / 2.0
            ft = np.maximum((1 - mid) * f + mid * g, 0)
            ft = ft / ft.sum()
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = mid

        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            m_exp = 1.0
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.15, 8.0)

    return best_f, best_obj


def smooth_grad_optimize(a0, time_limit, T_init=0.001, lr_init=0.02):
    """Smoothed gradient descent with softmax weighting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    idx = np.arange(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient: grad[j] = 2 * sum_m a[m] * p[j+m]
        grad = np.zeros(n)
        for j in range(n):
            grad[j] = 2.0 * np.dot(a, p[idx + j])

        best_candidate = a
        best_val = conv.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            T *= 1.02
            lr = max(lr * 0.99, 0.001)
        if no_improve > 100:
            T *= 0.7
        if no_improve > 300:
            rng = np.random.default_rng((it * 31 + 7) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def local_search(a0, time_limit, seed=42):
    """Local search with multi-scale perturbations and structured moves."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        improved = False

        for scale in [0.0002, 0.0005, 0.001, 0.003, 0.008]:
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(a + pert)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        if not improved and it % 3 == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    candidate = a.copy()
                    amount = candidate[i] * frac
                    candidate[i] -= amount
                    candidate[j] += amount
                    candidate = np.maximum(candidate, 0)
                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break

        if not improved and it % 15 == 0:
            block_size = max(2, n // 32)
            s1 = int(rng.integers(0, n - block_size))
            s2 = int(rng.integers(0, n - block_size))
            if s1 != s2:
                for frac in [0.05, 0.1, 0.2]:
                    candidate = a.copy()
                    mass = candidate[s1:s1 + block_size].sum() * frac
                    per_el = mass / block_size
                    candidate[s1:s1 + block_size] -= per_el
                    candidate[s2:s2 + block_size] += per_el
                    candidate = np.maximum(candidate, 0)
                    candidate = candidate / candidate.sum()
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break

        if not improved and it > 5000:
            break

    return best_a, best_obj


def try_seq(a, label, best_seq, best_score):
    """Evaluate a sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return a_norm.copy(), score
    return best_seq, best_score
