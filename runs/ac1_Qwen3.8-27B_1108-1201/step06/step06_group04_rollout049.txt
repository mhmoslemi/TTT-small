def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Fenchel primal-dual with multiplicative weights (dual) and projected gradient QP (primal)
- FFT-accelerated Hankel matrix-vector products
- Nesterov-accelerated subgradient as complementary method
- Coordinate descent polish
- Multi-n exploration with adaptive time allocation

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv_fft(q, a, fft_cache=None):
    """
    Compute W @ a where W[i,j] = q[i+j]. q has length 2n-1, a has length n.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    r = q[::-1]  # reverse q
    conv_len = n + (2*n - 1) - 1  # = 3n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    
    if fft_cache is not None and fft_cache.get('size') == fft_size:
        R = fft_cache['R']
    else:
        R = np.fft.rfft(r, fft_size)
        fft_cache = {'R': R, 'size': fft_size}
    
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * R, fft_size)[:conv_len]
    # Wa[i] = conv[2n-2-i] for i = 0, ..., n-1
    return conv[2*n-2 : n-2 : -1]


def solve_qp_pg(q, a_init, n, max_iters=300, verbose=False):
    """
    Solve min a^T W a s.t. sum(a)=1, a>=0 using projected gradient with backtracking.
    W is Hankel: W[i,j] = q[i+j].
    Returns the solution a.
    """
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    fft_cache = {}
    
    # Precompute the reversed q FFT
    r = q[::-1]
    conv_len = 3 * n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    R = np.fft.rfft(r, fft_size)
    
    lr = 1.0
    prev_f = float('inf')
    
    for it in range(max_iters):
        # Compute Wa = W @ a
        A = np.fft.rfft(a, fft_size)
        conv = np.fft.irfft(A * R, fft_size)[:conv_len]
        Wa = conv[2*n-2 : n-2 : -1]
        
        f = a @ Wa  # a^T W a
        grad = 2.0 * Wa
        
        # Backtracking line search
        step = lr
        improved = False
        for _ in range(30):
            a_new = project_simplex(a - step * grad)
            # Compute new objective
            A_new = np.fft.rfft(a_new, fft_size)
            conv_new = np.fft.irfft(A_new * R, fft_size)[:conv_len]
            Wa_new = conv_new[2*n-2 : n-2 : -1]
            f_new = a_new @ Wa_new
            if f_new < f - 1e-15:
                a = a_new
                improved = True
                break
            step *= 0.5
        
        if not improved:
            a = project_simplex(a - 1e-10 * grad)
            a = a / a.sum()
        
        lr = min(lr * 1.3, 5.0)
        
        if it > 5 and abs(prev_f - f) < 1e-12 * max(1.0, abs(f)):
            break
        prev_f = f
    
    return a


def primal_dual(a_init, time_limit, T_init=0.01, eta_init=1.0,
                qp_iters=200, verbose=False):
    """
    Fenchel primal-dual iteration:
    - Dual: q is a distribution over {0,...,2n-2}, updated by multiplicative weights
    - Primal: solve min a^T W(q) a s.t. a in simplex
    - Dual update: q <- normalize(q * exp(eta * conv(a,a)))
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    # Initialize dual variable q
    conv = fast_convolve(a)
    T = T_init
    scaled = conv / T
    scaled -= scaled.max()
    q = np.exp(scaled)
    q = q / q.sum()
    
    eta = eta_init
    best_a = a.copy()
    best_obj = conv.max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break
        
        # Solve inner QP
        a_qp = solve_qp_pg(q, a, n, max_iters=qp_iters)
        
        # Compute new convolution
        new_conv = fast_convolve(a_qp)
        new_obj = new_conv.max()
        
        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = a_qp.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Keep a close to best if QP didn't improve
            a_qp = best_a.copy()
            new_conv = fast_convolve(a_qp)
            new_obj = best_obj
        
        # Dual update (multiplicative weights)
        scaled = new_conv / T
        scaled -= scaled.max()
        q_new = np.exp(scaled)
        q_new = q_new / q_new.sum()
        
        # Blend for stability
        q = 0.5 * q + 0.5 * q_new
        
        a = a_qp
        
        # Adapt parameters
        if no_improve > 20:
            T *= 1.2
        if no_improve > 50:
            T *= 0.8
            eta *= 0.9
        if no_improve > 100:
            # Random restart
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            s = a.sum()
            a = a / s
            T = T_init
            no_improve = 0
            # Reset q
            conv = fast_convolve(a)
            scaled = conv / T
            scaled -= scaled.max()
            q = np.exp(scaled)
            q = q / q.sum()
        
        if verbose and it % 20 == 0:
            print(f"    PD it={it}, obj={best_obj:.10f}, T={T:.4f}, no_imp={no_improve}")

def nesterov_smoothed(a_init, time_limit, T_init=0.001, lr_init=0.01,
                       verbose=False):
    """
    Nesterov-accelerated method with smoothed subgradient (log-sum-exp).
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    v = np.zeros(n)
    lr = lr_init
    mom = 0.9
    y = a.copy()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Nesterov lookahead
        y = a + mom * v
        
        # Compute smoothed gradient using log-sum-exp
        conv_y = fast_convolve(y)
        scaled = conv_y / T_init
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Gradient of smoothed objective: sum_k p[k] * grad(conv[k])
        # grad(conv[k])[i] = (2 - delta_{2i,k}) * y[k-i] for valid k-i
        grad = np.zeros(n)
        for idx in np.where(p > 1e-10)[0]:
            k = int(idx)
            pk = p[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            for i in range(js, je + 1):
                coeff = 1.0 if 2 * i == k else 2.0
                grad[i] += pk * coeff * y[k - i]
        
        # Update
        v_new = project_simplex(y - lr * grad) - y
        a = y + v_new
        a = project_simplex(a)
        v = v_new
        
        # Evaluate
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt
        lr = max(0.0001, lr * 0.9995)
        mom = min(0.99, mom + 0.0001)
        
        if no_improve > 200:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            nr = a.sum()
            a = a / nr
            lr = lr_init
            mom = 0.9
            no_improve = 0
        
        if verbose and it % 100 == 0:
            print(f"    Nesterov it={it}, obj={best_obj:.10f}, lr={lr:.6f}")
    
    return best_a, best_obj


def coordinate_polish(a_init, time_limit, verbose=False):
    """
    Coordinate-wise optimization: for each coordinate, try to reduce the max conv.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    improved = True
    
    while time.monotonic() - t0 < time_limit and improved:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        
        improved = False
        
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 1.0:
                break
            
            # Try a few step sizes
            for delta_frac in [0.01, 0.05, 0.1, 0.2, -0.01, -0.05, -0.1, -0.2]:
                if a[i] < 1e-10 and delta_frac < 0:
                    continue
                a[i] += a[i] * delta_frac
                if a[i] < 0:
                    a[i] = 0
                a = a / a.sum()
                obj = fast_convolve(a).max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    # Revert
                    # This is getting complicated; let me just reset
                    a = best_a.copy()
            
            a = best_a.copy()
        
        if verbose and it % 5 == 0:
            print(f"    Coord it={it}, obj={best_obj:.10f}")
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Uses Fenchel primal-dual with multiplicative weights,
    Nesterov-accelerated smoothed subgradient, and coordinate polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score
