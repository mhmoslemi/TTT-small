Let me now think about the Adam optimizer integration.

Adam state: m (first moment), v (second moment), t (step count).
Update:
- m = beta1 * m + (1-beta1) * grad
- v = beta2 * v + (1-beta2) * grad^2
- m_hat = m / (1 - beta1^t)
- v_hat = v / (1 - beta2^t)
- a_new = a - lr * m_hat / (sqrt(v_hat) + eps)

Then project onto simplex.

The issue with Adam + projection is that the projection can undo some of the Adam updates. A common workaround is to do the projection and then "re-center" the Adam state, but this is heuristic.

An alternative is to use the mirror descent / mirror prox approach, which naturally handles the simplex constraint. In the log-domain:
- u = log(a + eps)
- Update u using Adam
- a_new = exp(u) / sum(exp(u))

This automatically ensures a >= 0 and sum(a) = 1.

Let me use this approach.

Actually, there's a subtlety: if a[i] = 0, then u[i] = -inf, which is problematic. I can use a small epsilon: u[i] = log(a[i] + eps).

But then the gradient in the u-domain is:
dLSE/du[i] = (dLSE/da[i]) * (da[i]/du[i]) = (dLSE/da[i]) * a[i]

Wait, that's for the softmax parametrization. Let me be more careful.

If a[i] = exp(u[i]) / Z where Z = sum_j exp(u[j]), then:
da[i]/du[j] = a[i] * (delta[i,j] - a[j])

So the gradient in u-space is:
du[j] = sum_i (dL/da[i]) * da[i]/du[j] = sum_i (dL/da[i]) * a[i] * (delta[i,j] - a[j])
     = (dL/da[j]) * a[j] - a[j] * sum_i (dL/da[i]) * a[i]

This is the "natural gradient" times a[i] minus a mean correction.

For simplicity, let me just use the projected approach (project onto simplex after each step). It's simpler and works well in practice.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-smoothed gradient descent with Adam optimizer
2. Adaptive beta schedule (annealing)
3. Multi-start with diverse initializations
4. Peak-specific coordinate refinement
5. Multi-length exploration
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_objective_and_grad(f, beta):
    """Compute LSE-smoothed objective and its gradient."""
    n = len(f)
    conv = fast_convolve(f)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w_sum = exp_bc.sum()
    if w_sum < 1e-300:
        w = np.ones(len(conv)) / len(conv)
        obj = bc_max / beta
    else:
        w = exp_bc / w_sum
        obj = (bc_max + np.log(w_sum)) / beta
    
    # Gradient: grad[i] = 2 * sum_j f[j] * w[i+j]
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(f, w[i:i+n])
    
    return obj, grad, conv

def adam_lse_optimize(f0, time_limit, beta_start=5.0, beta_end=50000.0, 
                       lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                       beta_increase=1.01, patience=50):
    """Adam optimizer with LSE smoothing and adaptive beta."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    # Initial evaluation
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta = beta_start
    no_improve = 0
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        # Compute LSE objective and gradient
        lse_val, grad, conv = lse_objective_and_grad(f, beta)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Learning rate decay
        lr_t = lr / (1 + 0.001 * t)
        
        # Update
        f_new = f - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)
        
        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        
        # Increase beta
        if t % 10 == 0:
            beta = min(beta * beta_increase, beta_end)
        
        # Patience: if no improvement for too long, restart from best
        if no_improve > patience:
            f = best_f + np.random.default_rng(t).standard_normal(n) * 0.001
            f = project_simplex(f)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            beta = min(beta * 2.0, beta_end)
            lr = max(lr * 0.7, 0.001)
        
        # Periodic reporting
        if t % 100 == 0:
            print(f"    t={t}, beta={beta:.1f}, obj={obj_new:.8f}, best={best_obj:.8f}, lr={lr_t:.6f}")
    
    return best_f, best_obj

def peak_coordinate_refine(f, time_limit, n_iterations=500):
    """Refine by adjusting individual coordinates to reduce the top peak."""
    f = np.maximum(f.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for it in range(n_iterations):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Find the current peak
        peak_idx = conv.argmax()
        
        # Find which coordinates contribute most to this peak
        # conv[peak_idx] = sum_i f[i] * f[peak_idx - i]
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        contributions = np.zeros(n)
        contributions[lo:hi+1] = f[lo:hi+1] * f[peak_idx - np.arange(lo, hi+1)]
        
        # Pick the coordinate with the largest contribution
        j = lo + np.argmax(contributions[lo:hi+1])
        
        if contributions[j] < 1e-15:
            break
        
        # Try to reduce f[j] by moving mass to a random location
        # that doesn't increase other peaks
        for _ in range(5):
            j_dest = rng.integers(0, n)
            if j_dest == j or j_dest == peak_idx - j:
                continue
            
            # Try different fractions of mass to move
            for frac in [0.01, 0.05, 0.1, 0.2]:
                delta = f[j] * frac
                if delta < 1e-15:
                    continue
                
                f_trial = f.copy()
                f_trial[j] -= delta
                f_trial[j_dest] += delta
                f_trial = np.maximum(f_trial, 0)
                f_trial = f_trial / f_trial.sum()
                
                conv_trial = fast_convolve(f_trial)
                if conv_trial.max() < best_obj - 1e-15:
                    best_obj = conv_trial.max()
                    best_f = f_trial.copy()
                    f = f_trial.copy()
                    conv = conv_trial
                    break
        
        # Also try increasing coordinates that are at zero
        # (if the simplex projection put some at zero)
        zero_coords = np.where(f < 1e-10)[0]
        if len(zero_coords) > 0:
            j_src = lo + rng.integers(0, hi - lo + 1)
            j_dest = zero_coords[rng.integers(0, len(zero_coords))]
            for frac in [0.01, 0.05]:
                delta = f[j_src] * frac
                if delta < 1e-15:
                    continue
                f_trial = f.copy()
                f_trial[j_src] -= delta
                f_trial[j_dest] += delta
                f_trial = np.maximum(f_trial, 0)
                f_trial = f_trial / f_trial.sum()
                conv_trial = fast_convolve(f_trial)
                if conv_trial.max() < best_obj - 1e-15:
                    best_obj = conv_trial.max()
                    best_f = f_trial.copy()
                    f = f_trial.copy()
                    conv = conv_trial
                    break
    
    return best_f, best_obj

def freq_perturb(f, rng, strength=0.03):
    """Perturb in frequency domain."""
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new

def try_sequence(a, best_tracker):

def interpolate_to_length(f, new_n):
    """Interpolate to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated LSE-smoothed gradient with Adam optimizer
2. Aggressive beta annealing schedule
3. Peak-chasing local refinement (direct peak reduction)
4. Multi-start, multi-length exploration
5. Multiplicative weights (mirror descent) as alternate step
6. Symmetry exploitation and structured initializations
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    return np.maximum(result, 0.0)


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_grad_fft(f, beta, top_k=None):
    """Compute LSE-smoothed objective, gradient (via FFT cross-correlation), and convolution."""
    n = len(f)
    conv = fast_convolve(f)

    if top_k is not None and top_k < len(conv):
        idx = np.argpartition(conv, -top_k)[-top_k:]
        conv_masked = np.zeros_like(conv)
        conv_masked[idx] = conv[idx]
        bc = beta * conv_masked
    else:
        bc = beta * conv

    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w_sum = exp_bc.sum()
    if w_sum < 1e-300:
        w = np.ones(len(conv)) / len(conv)
    else:
        w = exp_bc / w_sum

    obj = (bc_max + np.log(max(w_sum, 1e-300))) / beta

    # Gradient via FFT cross-correlation
    # grad[i] = 2 * sum_j f[j] * w[i+j]
    # Use: (conv(f, w_rev))[2n-2-i] = sum_j f[j] * w[i+j]
    w_rev = w[::-1]
    fft_size = 1 << (2 * n - 2 - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W_rev = np.fft.rfft(w_rev, fft_size)
    cross = np.fft.irfft(F * W_rev, fft_size)
    grad = 2.0 * cross[2 * n - 2: 2 * n - 2 - n: -1]

    return obj, grad, conv, w


def adam_lse_optimize(f0, time_limit, beta_start=20.0, beta_end=100000.0,
                      lr=0.01, beta1=0.85, beta2=0.999, eps=1e-8,
                      beta_increase=1.03, top_k=None, patience=80,
                      verbose=False):
    """Adam optimizer with LSE smoothing, adaptive beta, and restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta = beta_start
    no_improve = 0
    t0 = time.monotonic()
    best_lr = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        _, grad, conv_now, _ = lse_grad_fft(f, beta, top_k=top_k)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad ** 2
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Adaptive learning rate with warmup
        lr_t = best_lr * np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t)
        lr_t = min(lr_t, best_lr * 2.0)

        f_new = f - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        # Beta schedule: geometric increase
        if t % 5 == 0:
            beta = min(beta * beta_increase, beta_end)

        # Reporting
        if verbose and t % 50 == 0:
            print(f"      t={t}, beta={beta:.1f}, obj={obj_new:.10f}, "
                  f"best={best_obj:.10f}, lr={lr_t:.6f}")

        # Restart mechanism
        if no_improve > patience:
            rng = np.random.default_rng(t * 31 + 7)
            pert_scale = 0.002 + 0.003 * min(1.0, no_improve / (patience * 2))
            f = best_f + rng.standard_normal(n) * pert_scale
            f = project_simplex(f)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            beta = min(beta * 2.0, beta_end)
            best_lr = max(best_lr * 0.6, 0.0005)

    return best_f, best_obj


def multiplicative_weights_optimize(f0, time_limit, beta_start=50.0,
                                    beta_end=50000.0, step_init=0.1,
                                    beta_increase=1.04, top_k=None,
                                    patience=100, verbose=False):
    """Multiplicative weights (mirror descent in exponential family)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    step = step_init
    no_improve = 0
    t = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        _, grad, conv_now, w = lse_grad_fft(f, beta, top_k=top_k)

        # Multiplicative update: f[i] *= exp(-step * grad[i] / max_grad)
        max_grad = np.abs(grad).max()
        if max_grad < 1e-15:
            break
        log_update = -step * grad / max_grad
        log_update -= log_update.max()  # numerical stability
        f_new = f * np.exp(log_update)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / f_new_sum

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            step = min(step * 1.01, 1.0)
        else:
            no_improve += 1
            step *= 0.95

        f = f_new

        if t % 5 == 0:
            beta = min(beta * beta_increase, beta_end)

        if verbose and t % 50 == 0:
            print(f"      t={t}, beta={beta:.1f}, obj={obj_new:.10f}, "
                  f"best={best_obj:.10f}, step={step:.6f}")

        if no_improve > patience:
            rng = np.random.default_rng(t * 17 + 3)
            f = best_f + rng.standard_normal(n) * 0.002
            f = project_simplex(f)
            no_improve = 0
            beta = min(beta * 1.5, beta_end)
            step = max(step * 0.5, 0.01)

    return best_f, best_obj


def peak_direct_refine(f, time_limit, n_rounds=500):
    """Directly reduce the top peak by adjusting contributing coordinates."""
    f = np.maximum(f.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    for _ in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        # Find top peak
        peak_idx = int(conv.argmax())

        # Contributions to this peak
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        indices = np.arange(lo, hi + 1)
        contribs = f[indices] * f[peak_idx - indices]

        if contribs.max() < 1e-15:
            break

        # Try reducing the largest contributor
        j = int(indices[np.argmax(contribs)])
        partner = peak_idx - j

        # Generate candidate destination indices (far from peak contributors)
        n_candidates = min(20, n)
        candidates = rng.choice(n, size=n_candidates, replace=False)

        for j_dest in candidates:
            if j_dest == j or j_dest == partner:
                continue
            # Check if j_dest is not a major contributor to this peak
            if lo <= j_dest <= hi and contribs[j_dest - lo] > 0.3 * contribs.max():
                continue

            for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
                delta = f[j] * frac
                if delta < 1e-15:
                    continue
                f_trial = f.copy()
                f_trial[j] -= delta
                f_trial[j_dest] += delta
                if f_trial[j] < -1e-15:
                    continue
                f_trial = np.maximum(f_trial, 0.0)
                f_trial_sum = f_trial.sum()
                if f_trial_sum < 1e-15:
                    continue
                f_trial /= f_trial_sum

                conv_trial = fast_convolve(f_trial)
                if conv_trial.max() < best_obj - 1e-15:
                    best_obj = conv_trial.max()
                    best_f = f_trial.copy()
                    f = f_trial.copy()
                    conv = conv_trial
                    break

        # Also try small symmetric adjustments around the peak
        if partner != j and 0 <= partner < n:
            for delta_frac in [0.01, 0.03, 0.05]:
                # Reduce both j and partner slightly, redistribute
                d1 = f[j] * delta_frac
                d2 = f[partner] * delta_frac
                if d1 < 1e-15 and d2 < 1e-15:
                    continue
                f_trial = f.copy()
                f_trial[j] -= d1
                f_trial[partner] -= d2
                # Redistribute to random locations
                dests = rng.choice(n, size=2, replace=False)
                f_trial[dests[0]] += d1
                f_trial[dests[1]] += d2
                f_trial = np.maximum(f_trial, 0.0)
                f_trial_sum = f_trial.sum()
                if f_trial_sum < 1e-15:
                    continue
                f_trial /= f_trial_sum
                conv_trial = fast_convolve(f_trial)
                if conv_trial.max() < best_obj - 1e-15:
                    best_obj = conv_trial.max()
                    best_f = f_trial.copy()
                    f = f_trial.copy()
                    conv = conv_trial
                    break

    return best_f, best_obj


def alternating_peak_minimize(f, time_limit, n_iterations=200):
    """Alternate between minimizing different peaks."""
    f = np.maximum(f.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    # Get top peaks
    top_k = min(10, 2 * n - 1)
    peak_indices = np.argpartition(conv, -top_k)[-top_k:]

    for it in range(n_iterations):
        if time.monotonic() - t0 > time_limit:
            break

        # Cycle through peaks
        k = int(peak_indices[it % len(peak_indices)])

        # Gradient of c[k] with respect to f
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad = np.zeros(n)
        grad[lo:hi + 1] = 2.0 * f[k - np.arange(lo, hi + 1)]

        # Take a small step in the negative gradient direction
        step = 0.01
        f_new = f - step * grad
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        if conv_new.max() < best_obj - 1e-15:
            best_obj = conv_new.max()
            best_f = f_new.copy()
            f = f_new
            conv = conv_new
        else:
            # Try smaller step
            for step in [0.005, 0.002, 0.001]:
                f_new = f - step * grad
                f_new = project_simplex(f_new)
                conv_new = fast_convolve(f_new)
                if conv_new.max() < best_obj - 1e-15:
                    best_obj = conv_new.max()
                    best_f = f_new.copy()
                    f = f_new
                    conv = conv_new
                    break

    return best_f, best_obj


def freq_perturb(f, rng, strength=0.02):
    """Perturb in frequency domain for diversification."""
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0.0)
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.10f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function using:
    1. FFT-accelerated Adam + LSE optimization (primary)
    2. Multiplicative weights as alternate optimizer
    3. Direct peak reduction and alternating peak minimization
    4. Multi-start and multi-length exploration
    5. Freq-domain perturbation for diversification
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={parent_score:.10f}")
            best_tracker = [parent_score, parent.copy()]
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    total_budget = float(budget_s) - 15.0

    # === Phase 1: Adam + LSE (primary optimization) ===
    print("\n=== Phase 1: Adam + LSE optimization ===")
    t1_end = time.monotonic() + total_budget * 0.35

    configs = [
        {'beta_start': 20.0, 'beta_end': 50000.0, 'lr': 0.015,
         'beta_increase': 1.04, 'top_k': None, 'patience': 100},
        {'beta_start': 50.0, 'beta_end': 200000.0, 'lr': 0.008,
         'beta_increase': 1.05, 'top_k': 20, 'patience': 80},
        {'beta_start': 100.0, 'beta_end': 500000.0, 'lr': 0.005,
         'beta_increase': 1.06, 'top_k': 10, 'patience': 60},
    ]

    for idx, cfg in enumerate(configs):
        t_rem = t1_end - time.monotonic()
        if t_rem < 20:
            break
        n_configs_remaining = len(configs) - idx
        t_alloc = t_rem / n_configs_remaining * 0.92
        if t_alloc < 10:
            break
        print(f"  Adam-LSE config {idx}: beta {cfg['beta_start']:.0f}→{cfg['beta_end']:.0f}, "
              f"lr={cfg['lr']}, top_k={cfg['top_k']}, t={t_alloc:.0f}s")
        f_opt, obj = adam_lse_optimize(parent, time_limit=t_alloc, verbose=(idx == 0), **cfg)
        try_sequence(f_opt, best_tracker)

    # === Phase 2: Multiplicative weights ===
    print("\n=== Phase 2: Multiplicative weights ===")
    t2_end = t1_end + total_budget * 0.15
    current_best = best_tracker[1].copy()
    t_alloc = min(150, t2_end - time.monotonic())
    if t_alloc > 15:
        print(f"  MW: n={len(current_best)}, t={t_alloc:.0f}s")
        f_opt, obj = multiplicative_weights_optimize(
            current_best, time_limit=t_alloc,
            beta_start=50.0, beta_end=100000.0,
            step_init=0.05, beta_increase=1.05, top_k=15)
        try_sequence(f_opt, best_tracker)

    # === Phase 3: Peak-specific refinement ===
    print("\n=== Phase 3: Peak-specific refinement ===")
    t3_end = t2_end + total_budget * 0.2
    current_best = best_tracker[1].copy()

    # Direct peak refinement
    t_alloc = min(80, (t3_end - time.monotonic()) * 0.5)
    if t_alloc > 10:
        print(f"  PeakDirect: t={t_alloc:.0f}s")
        f_opt, obj = peak_direct_refine(current_best, time_limit=t_alloc, n_rounds=2000)
        try_sequence(f_opt, best_tracker)

    # Alternating peak minimization
    current_best = best_tracker[1].copy()
    t_alloc = min(60, t3_end - time.monotonic() - 5)
    if t_alloc > 10:
        print(f"  AltPeak: t={t_alloc:.0f}s")
        f_opt, obj = alternating_peak_minimize(current_best, time_limit=t_alloc, n_iterations=500)
        try_sequence(f_opt, best_tracker)

    # === Phase 4: Multi-length exploration ===
    print("\n=== Phase 4: Multi-length exploration ===")
    t4_end = t3_end + total_budget * 0.15
    current_best = best_tracker[1].copy()
    n_cur = len(current_best)

    candidate_ns = sorted(set([
        n_cur,
        max(256, n_cur - 64),
        min(2500, n_cur + 64),
        max(256, int(n_cur * 0.75)),
        min(2500, int(n_cur * 1.33)),
    ]), key=lambda x: abs(x - n_cur))

    explored = set()
    for nc in candidate_ns:
        if nc == n_cur or nc in explored:
            continue
        explored.add(nc)
        t_rem = t4_end - time.monotonic()
        if t_rem < 25:
            break
        t_alloc = min(50, t_rem * 0.5)
        if t_alloc < 12:
            break

        f_init = interpolate_to_length(current_best, nc)
        print(f"  MultiLen n={nc} (from {n_cur}): t={t_alloc:.0f}s")
        f_opt, obj = adam_lse_optimize(f_init, time_limit=t_alloc,
                                        beta_start=30.0, beta_end=100000.0,
                                        lr=0.012, beta_increase=1.04, top_k=15)
        try_sequence(f_opt, best_tracker)

    # === Phase 5: Intensive alternating refinement ===
    print("\n=== Phase 5: Intensive alternating refinement ===")
    t5_end = deadline - 40
    cycles = 0
    while time.monotonic() < t5_end and cycles < 25:
        cycles += 1
        current_best = best_tracker[1].copy()
        remaining = t5_end - time.monotonic()
        if remaining < 20:
            break

        if cycles % 5 == 1:
            # High-beta Adam
            t_alloc = min(70, remaining * 0.35)
            if t_alloc > 12:
                beta_s = 200.0 + 300.0 * cycles
                print(f"  C{cycles}: Adam hi-beta={beta_s:.0f}, t={t_alloc:.0f}s")
                f_opt, _ = adam_lse_optimize(current_best, time_limit=t_alloc,
                                              beta_start=beta_s, beta_end=1000000.0,
                                              lr=0.003, beta_increase=1.08, top_k=5)
                try_sequence(f_opt, best_tracker)

        elif cycles % 5 == 2:
            # Peak direct
            t_alloc = min(50, remaining * 0.25)
            if t_alloc > 8:
                print(f"  C{cycles}: PeakDirect, t={t_alloc:.0f}s")
                f_opt, _ = peak_direct_refine(current_best, time_limit=t_alloc, n_rounds=1500)
                try_sequence(f_opt, best_tracker)

        elif cycles % 5 == 3:
            # Multiplicative weights
            t_alloc = min(50, remaining * 0.25)
            if t_alloc > 10:
                print(f"  C{cycles}: MW, t={t_alloc:.0f}s")
                f_opt, _ = multiplicative_weights_optimize(
                    current_best, time_limit=t_alloc,
                    beta_start=100.0 + 200.0 * cycles, beta_end=200000.0,
                    step_init=0.02, beta_increase=1.06, top_k=8)
                try_sequence(f_opt, best_tracker)

        elif cycles % 5 == 4:
            # Freq perturb + Adam
            t_alloc = min(60, remaining * 0.3)
            if t_alloc > 15:
                print(f"  C{cycles}: FreqPerturb+Adam, t={t_alloc:.0f}s")
                for attempt in range(2):
                    t_rem = t5_end - time.monotonic() - 8
                    if t_rem < 12:
                        break
                    strength = 0.008 + 0.004 * attempt
                    f_pert = freq_perturb(current_best, rng, strength=strength)
                    t_sub = (t_rem - 5) / (2 - attempt) * 0.85
                    if t_sub > 8:
                        f_opt, _ = adam_lse_optimize(f_pert, time_limit=t_sub,
                                                      beta_start=50.0, beta_end=100000.0,
                                                      lr=0.01, beta_increase=1.04, top_k=10)
                        try_sequence(f_opt, best_tracker)

        else:
            # Alternating peak
            t_alloc = min(40, remaining * 0.2)
            if t_alloc > 8:
                print(f"  C{cycles}: AltPeak, t={t_alloc:.0f}s")
                f_opt, _ = alternating_peak_minimize(current_best, time_limit=t_alloc, n_iterations=300)
                try_sequence(f_opt, best_tracker)

    # === Phase 6: Final polish ===
    print("\n=== Phase 6: Final polish ===")
    current_best = best_tracker[1].copy()

    # Very high beta Adam (focus on single top peak)
    t_alloc = min(60, deadline - time.monotonic() - 25)
    if t_alloc > 10:
        print(f"  Final Adam (beta=1000+), t={t_alloc:.0f}s")
        f_opt, _ = adam_lse_optimize(current_best, time_limit=t_alloc,
                                      beta_start=1000.0, beta_end=1000000.0,
                                      lr=0.002, beta_increase=1.1, top_k=3)
        try_sequence(f_opt, best_tracker)

    # Final peak refinement
    current_best = best_tracker[1].copy()
    t_alloc = min(30, deadline - time.monotonic() - 10)
    if t_alloc > 5:
        print(f"  Final PeakRefine, t={t_alloc:.0f}s")
        f_opt, _ = peak_direct_refine(current_best, time_limit=t_alloc, n_rounds=1000)
        try_sequence(f_opt, best_tracker)

    # Quick random perturbations
    current_best = best_tracker[1].copy()
    t_rem = deadline - time.monotonic()
    if t_rem > 10:
        print(f"  Final perturbations, t={t_rem:.0f}s")
        n_b = len(current_best)
        for _ in range(20):
            if time.monotonic() > deadline - 3:
                break
            f_pert1 = freq_perturb(current_best, rng, strength=0.003)
            try_sequence(f_pert1, best_tracker)
            f_pert2 = project_simplex(current_best + rng.standard_normal(n_b) * 0.001)
            try_sequence(f_pert2, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.12f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]
        else:
            print(f"\nWARNING: Final eval returned {final_score}")

    print("WARNING: Using fallback")
    return [float(x) for x in parent]
```
