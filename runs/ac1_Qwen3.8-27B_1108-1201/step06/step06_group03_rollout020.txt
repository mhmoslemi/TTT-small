import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (f*g)[k] = sum_j A[k,j]*g[j]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

                break

        if best_candidate is not None:
            a = best_candidate
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        # Adaptive parameters
        if stagnant > 30:
            lr *= 0.96
            temp *= 1.05
        if stagnant > 100:
            temp *= 1.3
            lr *= 0.9
        if stagnant > 400:
            # Restart with perturbation
            scale = 0.002 + 0.003 * min(1.0, (stagnant - 400) / 500.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            lr = lr_init
            temp = temp_init
            stagnant = 0

    return best_a, best_obj


def fw_optimize_short(f0, time_limit):
    """Short Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    stagnant = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        g = solve_lp_oracle(f, M * 1.001)

        if g is None or g.sum() < 1e-15:
            g = solve_lp_oracle(f, M * 1.005)

        if g is None or g.sum() < 1e-15:
            stagnant += 1
            if stagnant > 10:
                break
            continue

        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-10:
            stagnant += 1
            if stagnant > 15:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M + 1e-14
        for alpha in np.linspace(0.01, 1.0, 20):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1
            if stagnant > 10:
                break

    return best_f, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def peak_targeted_cd(a0, time_limit):
    """
    Peak-targeted coordinate descent: identify the exact pairs contributing
    most to the peak and strategically shift mass.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    it = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)

        # Find pairs contributing most to the peak
        contrib = np.array([a[j] * a[k_star - j] for j in range(js, je + 1)])
        if contrib.max() < 1e-15:
            break

        # Sort contributions
        order = np.argsort(contrib)[::-1]
        top_pairs = order[:8]

        improved = False
        for oi in top_pairs:
            j = js + int(oi)
            partner = k_star - j
            if partner < 0 or partner >= n:
                continue

            # Try to reduce a[j] or a[partner] by moving mass to a distant location
            for src in [j, partner]:
                if a[src] < 1e-10:
                    continue

                # Compute optimal reduction
                c_peak = a[k_star - src] if 0 <= k_star - src <= n - 1 else 0.0
                if c_peak < 1e-12:
                    continue
                d_peak = conv[k_star] - a[src] * c_peak
                S_rest = 1.0 - a[src]
                t_opt = S_rest - 2.0 * d_peak / c_peak
                if t_opt < 0:
                    t_opt = 0.0
                t_opt = max(t_opt, a[src] * 0.3)  # Don't reduce by more than 70%

                delta = a[src] - t_opt
                if delta < 1e-12:
                    continue

                # Try several target locations
                for _ in range(5):
                    j_target = int(rng.integers(0, n))
                    if j_target == src or j_target == partner:
                        continue
                    # Prefer locations where a[k_star - j_target] is small
                    candidate = a.copy()
                    candidate[src] = t_opt
                    candidate[j_target] += delta
                    candidate = np.maximum(candidate, 0.0)
                    cs = candidate.sum()
                    if cs < 1e-10:
                        continue
                    candidate = candidate / cs
                    val = fast_convolve(candidate).max()
                    if val < best_obj - 1e-16:
                        a = candidate
                        best_obj = val
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stagnant += 1
            if stagnant > 200:
                # Global perturbation
                scale = 0.001 + 0.002 * min(1.0, stagnant / 500.0)
                a = project_simplex(best_a + rng.standard_normal(n) * scale)
                best_obj = fast_convolve(a).max()
                if best_obj > fast_convolve(best_a).max() * 1.0005:
                    a = best_a.copy()
                    best_obj = fast_convolve(a).max()
                stagnant = 0
            elif stagnant > 2000:
                break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: multi-phase optimization combining subgradient, coordinate
    descent, and FW methods with careful time allocation and multi-length
    exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})", flush=True)
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}", flush=True)
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}", flush=True)

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")

    n_parent = len(parent)

    # === Phase 1: Subgradient refinement at parent length (500s) ===
    print("\n=== Phase 1: Subgradient at n={} ===".format(n_parent), flush=True)
    t_phase = time.monotonic()

    configs = [
        {'lr_init': 0.008, 'momentum': 0.94, 'n_peaks': 8, 'temp_init': 0.0006},
        {'lr_init': 0.004, 'momentum': 0.96, 'n_peaks': 5, 'temp_init': 0.0003},
        {'lr_init': 0.012, 'momentum': 0.92, 'n_peaks': 12, 'temp_init': 0.001},
        {'lr_init': 0.003, 'momentum': 0.97, 'n_peaks': 3, 'temp_init': 0.0005},
    ]

    for ci, cfg in enumerate(configs):
        remaining = deadline - time.monotonic() - 900
        if remaining < 30:
            break
        t_alloc = min(remaining, 150)
        print(f"  Subgrad config {ci}, t={t_alloc:.0f}s", flush=True)
        a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc, **cfg)
        try_seq(a_opt, f"subgrad_{ci}")

    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)", flush=True)

    # === Phase 2: Exact coordinate descent (300s) ===
    print("\n=== Phase 2: Coordinate Descent ===", flush=True)
    t_phase = time.monotonic()

    remaining = deadline - time.monotonic() - 600
    if remaining > 30:
        t_alloc = min(remaining, 250)
        print(f"  Peak-targeted CD, t={t_alloc:.0f}s", flush=True)
        a_opt, _ = peak_targeted_cd(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "peak_cd")

        # General CD
        remaining = deadline - time.monotonic() - 400
        if remaining > 30:
            t_alloc = min(remaining, 100)
            print(f"  General CD, t={t_alloc:.0f}s", flush=True)
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "gen_cd")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)", flush=True)

    # === Phase 3: Multi-length exploration (250s) ===
    print("\n=== Phase 3: Multi-Length Exploration ===", flush=True)
    t_phase = time.monotonic()

    candidate_lengths = sorted(set([
        n_parent - 3, n_parent - 2, n_parent - 1,
        n_parent,
        n_parent + 1, n_parent + 2, n_parent + 3,
        n_parent + 7, n_parent - 7,
    ]))
    candidate_lengths = [l for l in candidate_lengths if 128 <= l <= 3000]

    for n_try in candidate_lengths:
        if n_try == n_parent:
            continue
        remaining = deadline - time.monotonic() - 350
        if remaining < 20:
            break

        a_resamp = resample_sequence(best_seq, n_try)
        t_alloc = min(remaining, 40)
        print(f"  n={n_try}, t={t_alloc:.0f}s", flush=True)

        # Brief subgradient
        a_opt, _ = subgradient_optimize(a_resamp, time_limit=t_alloc,
                                         lr_init=0.005, momentum=0.94,
                                         n_peaks=6, temp_init=0.0005)
        try_seq(a_opt, f"len_{n_try}")

        # Brief CD
        remaining = deadline - time.monotonic() - 250
        if remaining > 15:
            t_alloc = min(remaining, 20)
            a_opt, _ = coordinate_descent_polish(a_resamp, time_limit=t_alloc)
            try_seq(a_opt, f"cd_{n_try}")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)", flush=True)

    # === Phase 4: FW on best (200s) ===
    print("\n=== Phase 4: Frank-Wolfe ===", flush=True)
    t_phase = time.monotonic()

    remaining = deadline - time.monotonic() - 200
    if remaining > 30:
        t_alloc = min(remaining, 180)
        print(f"  FW n={len(best_seq)}, t={t_alloc:.0f}s", flush=True)
        a_opt, _ = fw_optimize_short(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "fw")

        # Perturbed FW
        for scale in [0.002, 0.005]:
            remaining = deadline - time.monotonic() - 100
            if remaining < 20:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * scale)
            t_alloc = min(remaining, 50)
            a_opt, _ = fw_optimize_short(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"fw_pert_{scale}")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)", flush=True)

    # === Phase 5: Final intensive polishing (remaining ~200s) ===
    print("\n=== Phase 5: Final Polish ===", flush=True)
    t_phase = time.monotonic()

    n_best = len(best_seq)

    # Alternate between subgradient and CD with very small steps
    polish_cycles = 0
    while time.monotonic() < deadline - 30:
        polish_cycles += 1
        remaining = deadline - time.monotonic() - 20

        if polish_cycles % 3 == 1:
            # Very fine subgradient
            t_alloc = min(remaining, 40)
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc,
                                             lr_init=0.002, momentum=0.97,
                                             n_peaks=4, temp_init=0.0003)
            try_seq(a_opt, f"polish_sg_{polish_cycles}")
        elif polish_cycles % 3 == 2:
            # Peak-targeted CD
            t_alloc = min(remaining, 40)
            a_opt, _ = peak_targeted_cd(best_seq, time_limit=t_alloc)
            try_seq(a_opt, f"polish_cd_{polish_cycles}")
        else:
            # General CD
            t_alloc = min(remaining, 40)
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, f"polish_gcd_{polish_cycles}")

    print(f"  Phase 5 done: best={best_score:.10f}, cycles={polish_cycles} ({time.monotonic()-t_phase:.0f}s)", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
