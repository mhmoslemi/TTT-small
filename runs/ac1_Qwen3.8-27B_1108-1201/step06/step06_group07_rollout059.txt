```python
- SCA with LP oracle for local minima escape

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_shaping(a, time_limit, verbose=False):
    """
    Iterative spectral shaping to flatten the power spectrum.
    Alternates between:
    1. Modifying the amplitude spectrum to reduce peaks
    2. Projecting back to non-negative simplex
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2 * n * fast_convolve(a).max() / a.sum()**2
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    lam = 0.1
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Compute spectrum
        fft_size = 1 << (2*n - 2).bit_length()
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        power = mag**2
        
        # Modify amplitude to flatten power spectrum
        # Reduce peaks: scale down frequencies with high power
        p_max = power.max()
        p_mean = power.mean()
        if p_max > 1e-15:
            # We want to reduce |A| where |A|^2 is high
            # New |A|^2 = |A|^2 * (1 - lam * (|A|^2/p_max - p_mean/p_max))
            # Simplified: cap the power
            scale = 1.0 - lam * np.maximum(0, power/p_max - 0.5)
            scale = np.maximum(scale, 0.1)
            new_mag = mag * np.sqrt(scale)
        else:
            new_mag = mag
        
        # Reconstruct
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(new_A, fft_size)[:n]
        
        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        obj = 2 * n * fast_convolve(a_new).max() / a_new.sum()**2
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            # Try smaller lambda
            lam *= 0.95
            if lam < 0.01:
                lam = 0.1
                no_improve = 0
        
        if no_improve > 100:
            break
    
    return best_a, best_obj
```

```python
                    fc = eval_f(c)
                else:
                    t_lo = c
                    c = d
                    fc = fd
                    d = t_lo + gr * (t_hi - t_lo)
                    fd = eval_f(d)
            
            t_star = (t_lo + t_hi) / 2.0
            
            # Check against the coarse grid best
            f_star = eval_f(t_star)
            if f_star > best_f:
                t_star = best_t
                f_star = best_f
            
            # Update a[i] and conv
            if abs(t_star - old) > 1e-15:
                for k in range(k_min, k_max + 1):
                    if k == 2 * i:
                        conv[k] += t_star * t_star - old * old
                    else:
                        conv[k] += 2.0 * (t_star - old) * a[k - i]
                a[i] = t_star
        
        # End of sweep
        obj = 2.0 * n * conv.max() / a.sum()**2
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 5:
            scale = min(0.02, 0.002 * no_improve)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            conv = fast_convolve(a)
            no_improve = 0
    
    return best_a, best_obj
```
