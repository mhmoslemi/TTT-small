"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, p):
    """Compute gradient of sum_k p[k]*conv[k] w.r.t. a.
    grad[i] = 2 * sum_m a[m] * p[m+i]
    Uses FFT for efficiency.
    """
    n = len(a)
    p_len = len(p)
    # Pad a and p for FFT
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p[:fft_size] if p_len <= fft_size else np.zeros(fft_size), fft_size)
    # The gradient is 2 * a * p (cross-correlation style)
    # grad[i] = 2 * sum_m a[m] * p[m+i]
    # This is the cross-correlation of a and p, computed as:
    # ifft(conj(A) * P) gives sum_m a[m] * p[m+i] (circular)
    # We need linear, so use sufficient zero padding
    fft_size2 = 1 << (n + p_len - 1 - 1).bit_length()
    A2 = np.fft.rfft(a, fft_size2)
    P2 = np.fft.rfft(p, fft_size2)
    corr = np.fft.irfft(np.conj(A2) * P2, fft_size2)
    # corr[i] = sum_m a[m] * p[m+i] (for the valid range)
    grad = 2.0 * corr[:n]
    return grad


def smoothed_grad_descent(a0, time_limit, T_init=0.002, lr_init=0.05, top_k=20,
                          momentum=0.7, seed=42):
    """
    Smoothed gradient descent with top-k lag focusing, Nesterov momentum,
    and adaptive parameters.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    T = T_init
    lr = lr_init
    no_improve = 0
    it = 0
    vel = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        # Compute top-k lag weights
        k_eff = min(top_k, len(conv))
        top_indices = np.argpartition(conv, -k_eff)[-k_eff:]
        top_vals = conv[top_indices]
        # Softmax over top-k with temperature T
        scaled = top_vals / T
        scaled -= scaled.max()
        weights = np.exp(scaled)
        weights /= weights.sum()

        # Build sparse p vector
        p = np.zeros(len(conv))
        p[top_indices] = weights

        # Nesterov lookahead
        a_look = a + momentum * vel

        # Compute gradient
        grad = compute_gradient(a_look, p)

        # Adaptive step size via backtracking
        alpha = lr
        for _ in range(15):
            candidate = project_simplex(a - alpha * grad)
            if candidate is not None:
                c = fast_convolve(candidate)
                new_obj = c.max()
                if new_obj < best_obj or alpha < lr * 0.01:
                    vel = momentum * vel - alpha * grad
                    a = candidate
                    obj = new_obj
                    break
            alpha *= 0.5

        else:
            # Fallback: small step
            a = project_simplex(a - lr * 0.01 * grad)
            vel = momentum * vel - lr * 0.01 * grad
            obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameter updates
        if no_improve > 30:
            T *= 1.05
            lr *= 0.97
            momentum = min(momentum + 0.01, 0.9)
        if no_improve > 100:

            lr = lr_init * 0.5
            momentum = 0.5
            top_k = min(top_k * 2, 200)

            vel = np.zeros(n)
            T = T_init
            lr = lr_init
            momentum = 0.7
            top_k = 20
            no_improve = 0

def dual_primal_iterate(a0, time_limit, T_init=0.001, seed=42):
    """

    Dual: maintain distribution p over lags (focus on peaks).
    Primal: optimize a to minimize p^T conv(a).
    """

    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    p_len = 2 * n - 1

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    T = T_init
    lr = 0.03
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Update dual: p based on current peaks
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p /= p.sum()

        # Primal update: few gradient steps on min p^T conv(a)
        for _ in range(3):
            grad = compute_gradient(a, p)
            # Line search
            alpha = lr
            for _ in range(10):
                candidate = project_simplex(a - alpha * grad)
                new_p_obj = np.dot(p, fast_convolve(candidate))
                old_p_obj = np.dot(p, fast_convolve(a))
                if new_p_obj < old_p_obj:
                    a = candidate
                    break
                alpha *= 0.5
            else:
                break

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 20:
            T *= 1.03
        if no_improve > 80:
            T *= 0.7
            lr *= 0.8
        if no_improve > 200:
            noise = rng.standard_normal(n) * 0.002
            a = project_simplex(best_a + noise)
            T = T_init
            lr = 0.03
            no_improve = 0

def coordinate_descent(a0, time_limit, seed=42):
    """Coordinate descent with pairwise swaps."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        improved = False

        # Try moving mass between pairs
        for _ in range(50):
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                candidate = a.copy()
                amount = candidate[i] * frac
                if amount < 1e-15:
                    continue
                candidate[i] -= amount
                candidate[j] += amount
                candidate = np.maximum(candidate, 0)
                ss = candidate.sum()
                if ss < 1e-10:
                    continue
                candidate = candidate / ss
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break

        # Try adding noise
        if not improved:
            for scale in [0.001, 0.003, 0.008, 0.02]:
                pert = rng.standard_normal(n) * scale
                candidate = project_simplex(a + pert)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and it > 200:
            # Larger perturbation to escape
            pert = rng.standard_normal(n) * 0.05
            a = project_simplex(best_a + pert)

    return best_a, best_obj


def asymmetric_structure(n, seed=42):
    """Generate an asymmetric structure that might reduce peak autocorrelation."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)

    # Try various asymmetric profiles
    # Skewed Gaussian
    center = rng.uniform(0.2, 0.8) * (n - 1)
    width = rng.uniform(0.1, 0.3) * n
    left_width = width * rng.uniform(0.5, 2.0)
    a = np.zeros(n)
    for i in range(n):
        if x[i] < center:
            a[i] = np.exp(-0.5 * ((x[i] - center) / left_width) ** 2)
        else:
            a[i] = np.exp(-0.5 * ((x[i] - center) / width) ** 2)

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a

    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    for factor in [2, 4, 8]:
        if time.monotonic() - t0 > time_limit - 30:
            break
        n_low = max(16, n // factor)
        # Downsample by averaging
        a_low = np.zeros(n_low)
        for i in range(n_low):
            start = int(i * n / n_low)
            end = int((i + 1) * n / n_low)
            a_low[i] = a[start:end].mean()
        a_low = np.maximum(a_low, 0)
        s = a_low.sum()
        if s > 1e-10:
            a_low /= s

        # Optimize at low resolution
        t_alloc = min(30, (time_limit - (time.monotonic() - t0)) * 0.5)
        a_low_opt, _ = smoothed_grad_descent(a_low, t_alloc, T_init=0.01, lr_init=0.1, top_k=5, seed=seed + factor)

        # Upsample and blend
        a_up = np.interp(np.linspace(0, 1, n, endpoint=False),
                         np.linspace(0, 1, n_low, endpoint=False), a_low_opt)
        a_up = np.maximum(a_up, 0)
        ss = a_up.sum()
        if ss > 1e-10:
            a_up /= ss

        # Blend with current best
        for alpha in [0.3, 0.5, 0.7, 1.0]:
            blend = (1 - alpha) * best_a + alpha * a_up
            blend = project_simplex(blend)
            c = fast_convolve(blend)
            if c.max() < best_obj - 1e-15:
                best_obj = c.max()
                best_a = blend

    return best_a, best_obj


def try_seq(a, label, best_seq, best_score):
    """Evaluate a sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return a_norm.copy(), score
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point: multi-phase hybrid search for minimum peak autocorrelation.
    Uses smoothed gradient with top-k focusing, dual-primal iteration,
    coordinate descent, multi-scale refinement, and local search.
    Explores multiple n values starting from parent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Using uniform init: n={n_init}, score={best_score:.8f}")

    # === Phase 1: Quickly try different n values ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()
    n_parent = len(parent)

    n_candidates = [512, 640, 768, 896, 1024, 1200, 1280, 1536, 2048]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)
    # Sort by proximity to parent
    n_candidates.sort(key=lambda x: abs(x - n_parent))

    for n_try in n_candidates[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 200:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            # Interpolate from parent
            start = np.interp(np.linspace(0, 1, n_try, endpoint=False),
                              np.linspace(0, 1, n_parent, endpoint=False), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try

        # Quick smoothed gradient
        t_alloc = min(remaining * 0.1, 30)
        a_opt, _ = smoothed_grad_descent(start, t_alloc, T_init=0.005, lr_init=0.08, top_k=10, seed=seed + n_try)
        best_seq, best_score = try_seq(a_opt, f"sg_n{n_try}", best_seq, best_score)

        # Quick coordinate descent
        remaining2 = deadline - time.monotonic()
        if remaining2 > 150:
            a_opt2, _ = coordinate_descent(a_opt, min(remaining2 * 0.05, 15), seed=seed + n_try + 1)
            best_seq, best_score = try_seq(a_opt2, f"cd_n{n_try}", best_seq, best_score)

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep optimization on best sequence ===
    print("\n=== Phase 2: Deep Optimization ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()

    if remaining > 200:
        # Smoothed gradient with fine parameters
        t_alloc = min(remaining * 0.3, 400)
        print(f"  Smoothed gradient (fine), n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = smoothed_grad_descent(best_seq, t_alloc, T_init=0.001, lr_init=0.02,
                                         top_k=30, momentum=0.7, seed=100)
        best_seq, best_score = try_seq(a_opt, "deep_sg", best_seq, best_score)

    # Dual-primal iteration
    remaining = deadline - time.monotonic()
    if remaining > 150:
        t_alloc = min(remaining * 0.25, 300)
        print(f"  Dual-primal, t={t_alloc:.0f}s")
        a_opt, _ = dual_primal_iterate(best_seq, t_alloc, T_init=0.001, seed=200)
        best_seq, best_score = try_seq(a_opt, "dual_primal", best_seq, best_score)

    # Another smoothed gradient pass with different parameters
    remaining = deadline - time.monotonic()
    if remaining > 100:
        t_alloc = min(remaining * 0.2, 200)
        print(f"  Smoothed gradient (T=0.0005, top_k=50), t={t_alloc:.0f}s")
        a_opt, _ = smoothed_grad_descent(best_seq, t_alloc, T_init=0.0005, lr_init=0.01,
                                         top_k=50, momentum=0.8, seed=300)
        best_seq, best_score = try_seq(a_opt, "sg_fine", best_seq, best_score)

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Multi-restart with perturbations ===
    print("\n=== Phase 3: Multi-Restart ===")
    t_phase = time.monotonic()

    for restart_i in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break

        # Different perturbation strategies
        n_best = len(best_seq)
        if restart_i % 3 == 0:
            # Gaussian noise
            scale = 0.002 * (restart_i + 1)
            pert = rng.standard_normal(n_best) * scale
            start = project_simplex(best_seq + pert)
        elif restart_i % 3 == 1:
            # Asymmetric structure blend
            struct = asymmetric_structure(n_best, seed=400 + restart_i)
            alpha = 0.3 + 0.1 * restart_i
            start = project_simplex((1 - alpha) * best_seq + alpha * struct)
        else:
            # Random frequency-domain perturbation
            A = np.fft.rfft(best_seq)
            A *= 1.0 + rng.standard_normal(len(A)) * 0.01
            start = np.fft.irfft(A, len(best_seq) * 2)[:n_best]
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else best_seq.copy()

        t_alloc = min(remaining * 0.15, 120)
        a_opt, _ = smoothed_grad_descent(start, t_alloc, T_init=0.002, lr_init=0.03,
                                         top_k=20, seed=500 + restart_i)
        best_seq, best_score = try_seq(a_opt, f"restart_{restart_i}", best_seq, best_score)

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Local search and coordinate refinement ===
    print("\n=== Phase 4: Local Search ===")
    t_phase = time.monotonic()

    remaining = deadline - time.monotonic()
    if remaining > 80:
        t_alloc = min(remaining * 0.3, 200)
        print(f"  Coordinate descent, t={t_alloc:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, t_alloc, seed=900)
        best_seq, best_score = try_seq(a_opt, "coord", best_seq, best_score)

    # Multi-scale refinement
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t_alloc = min(remaining * 0.2, 100)
        print(f"  Multi-scale, t={t_alloc:.0f}s")
        a_opt, _ = multi_scale_refine(best_seq, t_alloc, seed=1000)
        best_seq, best_score = try_seq(a_opt, "multi_scale", best_seq, best_score)

    # Fine gradient polish
    remaining = deadline - time.monotonic()
    if remaining > 50:
        t_alloc = min(remaining * 0.2, 100)
        print(f"  Final gradient polish (T=0.0001), t={t_alloc:.0f}s")
        a_opt, _ = smoothed_grad_descent(best_seq, t_alloc, T_init=0.0001, lr_init=0.005,
                                         top_k=10, momentum=0.9, seed=1100)
        best_seq, best_score = try_seq(a_opt, "polish", best_seq, best_score)

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final attempts ===
    print("\n=== Phase 5: Final Attempts ===")
    t_phase = time.monotonic()

    # Try a slightly different n
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_best = len(best_seq)
        for n_alt in [n_best + 128, n_best - 128, n_best * 2, n_best // 2]:
            if n_alt < 256 or n_alt > 4096:
                continue
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            start = np.interp(np.linspace(0, 1, n_alt, endpoint=False),
                              np.linspace(0, 1, n_best, endpoint=False), best_seq)
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_alt) / n_alt
            t_alloc = min(remaining * 0.3, 60)
            a_opt, _ = smoothed_grad_descent(start, t_alloc, T_init=0.003, lr_init=0.04,
                                             top_k=15, seed=1200 + n_alt % 100)
            best_seq, best_score = try_seq(a_opt, f"alt_n{n_alt}", best_seq, best_score)

    # Final tiny perturbations
    remaining = deadline - time.monotonic()
    if remaining > 20:
        n_best = len(best_seq)
        for i in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            noise = rng.standard_normal(n_best) * 0.0005
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = smoothed_grad_descent(f_pert, min(remaining * 0.15, 10),
                                             T_init=0.0003, lr_init=0.005,
                                             top_k=10, seed=1300 + i)
            best_seq, best_score = try_seq(a_opt, f"micro_{i}", best_seq, best_score)

    # Final coordinate search
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_opt, _ = coordinate_descent(best_seq, min(remaining * 0.3, 20), seed=1400)
        best_seq, best_score = try_seq(a_opt, "final_coord", best_seq, best_score)

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
