```python
    return e / e.sum()

def compute_grad_b(b, tau):
    """Compute gradient of f_tau(b) = tau * log(sum exp(c/tau)) where c = conv(softmax(b), softmax(b))."""
    a = softmax(b)
    n = len(a)
    c = fast_convolve(a)
    # p = softmax(c / tau)
    c_scaled = c / tau
    c_scaled -= c_scaled.max()
    p = np.exp(c_scaled)
    p /= p.sum()
    # grad_a[j] = 2 * sum_k p[k+j] * a[k] = 2 * correlate(p, a, 'valid')[j]
    grad_a = 2.0 * np.correlate(p, a, mode='valid')
    # grad_b = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_b = a * (grad_a - S)
    return grad_b

def objective_b(b, tau):
    a = softmax(b)
    c = fast_convolve(a)
    c_scaled = c / tau
    c_scaled -= c_scaled.max()
    return tau * (c_scaled.max() + np.log(np.exp(c_scaled).sum()))

def lbfgs_smoothed(a_init, tau_values, max_iter_per_tau=500, time_limit=300):
    """Run L-BFGS on the smoothed objective with decreasing tau."""
    a = a_init.copy()
    n = len(a)
    t0 = time.monotonic()
    
    for tau in tau_values:
        if time.monotonic() - t0 > time_limit - 10:
            break
        # Convert a to b (log of a, with handling of zeros)
        a_safe = np.maximum(a, 1e-15)
        b = np.log(a_safe / a_safe.sum())  # This gives softmax(b) ≈ a
        # Actually, softmax(b) = exp(b)/sum(exp(b)) = (a_safe/sum(a_safe)) / sum(a_safe/sum(a_safe)) = a/sum(a)
        # So b = log(a) - log(sum(a)) gives softmax(b) = a/sum(a). Let me verify:
        # exp(b) = a_safe/sum(a_safe), sum(exp(b)) = sum(a_safe)/sum(a_safe) = 1. So softmax(b) = a_safe/sum(a_safe) = a_safe (if a is normalized).
        # Good.
        
        def obj(b_):
            return objective_b(b_, tau)
        
        def grad(b_):
            return compute_grad_b(b_, tau)
        
        remaining = time_limit - (time.monotonic() - t0) - 10
        max_time = min(remaining, max_iter_per_tau * 0.1)  # rough time per iter
        
        res = minimize(obj, b, jac=grad, method='L-BFGS-B',
                       options={'maxiter': max_iter_per_tau, 'maxfun': max_iter_per_tau * 5,
                                'ftol': 1e-14, 'gtol': 1e-8})
        a_new = softmax(res.x)
        # Re-normalize
        a_new = a_new / a_new.sum()
        
        # Check improvement
        old_obj = fast_convolve(a).max()
        new_obj = fast_convolve(a_new).max()
        if new_obj <= old_obj + 1e-15:
            a = a_new
        print(f"  L-BFGS tau={tau:.6f}: obj {old_obj:.10f} -> {new_obj:.10f}")
    
    return a
```

Wait, there's an issue. The objective `tau * log(sum exp(c/tau))` is not the same as `max(c)` even in the limit. It's an upper bound on max(c). As τ → 0, it approaches max(c), but for finite τ, it's slightly larger.

The actual objective we want to minimize is `2n * max(c) / (sum(a))^2`. With sum(a)=1, it's `2n * max(c)`. The L-BFGS minimizes `tau * log(sum exp(c/tau))` which is ≥ max(c). So minimizing the smoothed objective also reduces the actual objective (or at least doesn't increase it in the limit).

But during the optimization, the direction might be slightly suboptimal for the true (non-smooth) objective. This is the standard trade-off with smoothing.

To get the best result, I should:
1. Use L-BFGS with moderate τ to get close to the optimum.
2. Then use a non-smooth method (FW, subgradient, or coordinate descent) for the final polish.

Let me also think about the FW method more carefully. The LP oracle in FW is:
- Given current a and M = max(conv(a,a)), solve: max sum(g) s.t. conv_matrix(a) @ g ≤ M, g ≥ 0.
- The optimal g is on the simplex (after normalization).
- The FW step is: a_new = (1-t) * a + t * g, with t chosen by line search.

The issue with FW is that it can be slow to converge (linear convergence rate, and the rate depends on the geometry of the problem). The L-BFGS should converge faster (superlinear) when close to the optimum.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Frank-Wolfe with LP oracle for global progress
- L-BFGS on log-sum-exp smoothed objective for fast local convergence
- Fine coordinate descent for final polish
- Multi-length exploration
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_stable(b):
    """Numerically stable softmax."""
    b = b - b.max()
    e = np.exp(b)
    return e / e.sum()


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n matrix A where (f*g)[k] = sum_j A[k,j]*g[j]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M):
    """LP: max 1^T g s.t. conv_matrix(f) g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_step(f, M, m_relax=1.001):
    """One Frank-Wolfe step: solve LP oracle and do line search."""
    n = len(f)
    g = solve_lp_oracle(f, M * m_relax)
    if g is None or g.sum() < 1e-15:
        return None, None
    g = g / g.sum()
    
    # Line search
    best_alpha = 0.0
    best_val = M + 1e-14
    for alpha in np.linspace(0.01, 1.0, 20):
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        ft = ft / ft.sum()
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    if best_alpha > 0:
        ft = (1.0 - best_alpha) * f + best_alpha * g
        ft = np.maximum(ft, 0.0)
        ft = ft / ft.sum()
        return ft, best_val
    return None, None


def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    stagnant = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        # Try a few m_relax values
        found = False
        for m_relax in [1.001, 1.005, 1.02]:
            result = fw_step(f, M, m_relax)
            if result[0] is not None:
                f_new, val = result
                if val < M:
                    f = f_new
                    M = val
                    if M < best_obj:
                        best_obj = M
                        best_f = f.copy()
                        stagnant = 0
                    found = True
                    break
        
        if not found:
            stagnant += 1
            if stagnant > 30:
                rng = np.random.default_rng(it * 7919 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                M = fast_convolve(f).max()
                stagnant = 0
    
    return best_f, best_obj


def compute_smoothed_obj_and_grad(b, tau):
    """Compute f_tau(b) and its gradient."""
    a = softmax_stable(b)
    n = len(a)
    c = fast_convolve(a)
    
    # Log-sum-exp of c/tau
    c_scaled = c / tau
    c_max = c_scaled.max()
    lse = c_max + np.log(np.exp(c_scaled - c_max).sum())
    obj = tau * lse
    
    # Gradient
    p = np.exp(c_scaled - c_max)
    p /= p.sum()  # p = softmax(c/tau)
    grad_a = 2.0 * np.correlate(p, a, mode='valid')
    S = np.dot(grad_a, a)
    grad_b = a * (grad_a - S)
    
    return obj, grad_b


def lbfgs_smoothed(a_init, tau_list, time_limit, verbose=True):
    """L-BFGS on smoothed objective with decreasing tau."""
    a = np.maximum(a_init, 0.0)
    a = a / a.sum()
    n = len(a)
    t0 = time.monotonic()
    
    for tau in tau_list:
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        # Initialize b from a
        a_safe = np.maximum(a, 1e-15)
        b0 = np.log(a_safe) - np.log(a_safe.sum())
        
        def obj(b_):
            val, _ = compute_smoothed_obj_and_grad(b_, tau)
            return val
        
        def grad(b_):
            _, g = compute_smoothed_obj_and_grad(b_, tau)
            return g
        
        remaining = time_limit - (time.monotonic() - t0) - 5
        res = minimize(obj, b0, jac=grad, method='L-BFGS-B',
                       options={'maxiter': 300, 'maxfun': 2000,
                                'ftol': 1e-16, 'gtol': 1e-10,
                                'maxls': 50})
        
        a_new = softmax_stable(res.x)
        a_new = a_new / a_new.sum()
        
        old_max = fast_convolve(a).max()
        new_max = fast_convolve(a_new).max()
        
        if verbose:
            print(f"  L-BFGS tau={tau:.2e}: max_conv {old_max:.10f} -> {new_max:.10f} "
                  f"(iters={res.niter}, status={res.message})")
        
        if new_max <= old_max + 1e-15:
            a = a_new
    
    return a


def coord_descent_polish(a_init, time_limit, n_top_peaks=3, max_rounds=5000):
    """Coordinate descent: for each top peak, try adjusting pairs of coordinates."""
    a = np.maximum(a_init, 0.0).copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    for round_i in range(max_rounds):
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        peaks = np.argsort(conv)[::-1][:n_top_peaks]
        improved = False
        
        for k in peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            # Try shifting mass from high-contributing coordinates to low ones
            contrib = np.zeros(n)
            for j in range(js, je + 1):
                contrib[j] += a[j] * a[k - j]
            
            # Find the top contributing coordinate and a low one to shift to
            top_j = js + int(np.argmax(contrib[js:je+1]))
            if a[top_j] < 1e-10:
                continue
            
            # Try moving a small amount to various targets
            for delta_frac in [0.01, 0.005, 0.001]:
                delta = a[top_j] * delta_frac
                for target in range(n):
                    if target == top_j:
                        continue
                    candidate = a.copy()
                    candidate[top_j] -= delta
                    candidate[target] += delta
                    if candidate[top_j] < -1e-12:
                        continue
                    candidate = np.maximum(candidate, 0)
                    candidate = candidate / candidate.sum()
                    val = fast_convolve(candidate).max()
                    if val < best_obj - 1e-15:
                        a = candidate
                        best_obj = val
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try random perturbations
            rng = np.random.default_rng(round_i * 12345 + 678)
            for _ in range(10):
                j_src = int(rng.integers(0, n))
                j_dst = int(rng.integers(0, n))
                if j_src == j_dst:
                    continue
                delta = a[j_src] * 0.005
                candidate = a.copy()
                candidate[j_src] -= delta
                candidate[j_dst] += delta
                candidate = np.maximum(candidate, 0)
                candidate = candidate / candidate.sum()
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
                    improved = True
                    break
            
            if not improved and round_i > 200:
                break
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s
