def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(n):
    """Precompute the sparsity pattern for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_oracle(f, M, n, row_idx=None, col_idx=None):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g (unnormalized) or None.
    """
    if row_idx is None or col_idx is None:
        row_idx, col_idx = build_lp_matrix(n)

    c = -np.ones(n)
    # A_ub[k, i] = f[k-i] for valid (k,i) pairs
    vals = f[col_idx[::-1].copy()]
    # We need A_ub where A_ub[k, i] = f[k-i]
    # row_idx[t] = k, col_idx[t] = i, value = f[k-i]
    k_arr = row_idx
    i_arr = col_idx
    vals = f[k_arr - i_arr]

    from scipy.sparse import csr_matrix
    A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def mirror_descent(a0, time_limit, eta_init=0.05):
    """Multiplicative weight mirror descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    eta = eta_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        max_val = conv.max()

        # Top peaks with softmax weights
        top_k = min(15, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        pw = softmax(pv / (0.005 * max_val + 1e-10))

        # Subgradient
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            w = pw[idx]
            if w < 1e-12:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je + 1] += w * 2.0 * a[k - je:k - js + 1][::-1]

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        # Blend with line search
        best_candidate = a
        best_val = max_val
        for alpha in [0.3, 0.5, 0.7, 1.0]:
            cand = (1 - alpha) * a + alpha * a_new
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_candidate = cand

        a = best_candidate
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 20:
            eta *= 0.95
        if eta < 1e-4:
            eta = eta_init * 0.5
            no_improve = 0

        if no_improve > 150:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            eta = eta_init
            no_improve = 0

    return best_a, best_obj


def subgradient_ls(a0, time_limit, top_k=10):
    """Subgradient with exact binary line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = 0.05

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            weights = pv / pv.sum()

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            w = weights[idx]
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je + 1] += w * 2.0 * a[k - je:k - js + 1][::-1]

        # Direction
        direction = project_simplex(a - lr * grad) - a

        # Binary line search
        lo, hi = 0.0, 1.0
        best_alpha = 0.0
        best_val = conv.max()
        for _ in range(12):
            mid = (lo + hi) / 2.0
            cand = a + mid * direction
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                hi = mid
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
                lo = mid
            else:
                hi = mid

        if best_alpha > 1e-10:
            a = a + best_alpha * direction
            a = np.maximum(a, 0)
            a = a / a.sum()
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        lr *= 0.999
        if lr < 0.002:
            lr = 0.05

        if no_improve > 80:
            rng = np.random.default_rng((it * 1299709 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            no_improve = 0
            lr = 0.05

    return best_a, best_obj


def peak_shave(a0, time_limit):
    """
    Targeted peak shaving: directly reduce the highest peak by
    redistributing mass from positions that contribute to it.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        peak_val = conv[k_peak]

        # Positions contributing to this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)

        # Contribution of each position j to the peak: a[j] * a[k_peak - j]
        contrib = np.zeros(n)
        for j in range(js, je + 1):
            contrib[j] = a[j] * a[k_peak - j]

        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break

        # Fraction to redistribute from each position
        # Move mass proportional to contribution
        frac = 0.1  # Try to move 10% of the peak-causing mass
        # Remove mass from high-contrib positions, add to low-contrib positions
        move_amount = np.zeros(n)
        for j in range(n):
            if contrib[j] > 0:
                move_amount[j] = -frac * a[j] * (contrib[j] / total_contrib)

        # Determine where to add: positions NOT in the peak span
        outside = np.array([j for j in range(n) if j < js or j > je])
        if len(outside) == 0:
            outside = np.array([j for j in range(n) if j != js and j != je])

        if len(outside) > 0:
            total_move = -move_amount.sum()
            if total_move > 0:
                # Distribute to outside positions proportionally to their current value (or uniformly)
                ext_vals = a[outside]
                ext_sum = ext_vals.sum()
                if ext_sum > 1e-15:
                    add_amount = total_move * ext_vals / ext_sum
                else:
                    add_amount = np.full(len(outside), total_move / len(outside))
                for i, j in enumerate(outside):
                    move_amount[j] += add_amount[i]

        a_new = a + move_amount
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()

        new_obj = fast_convolve(a_new).max()
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            rng = np.random.default_rng((it * 9973 + 5) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            no_improve = 0

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # From parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Uniform
    inits.append(np.ones(n) / n)

    # Dirichlet varied
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    # Edge-heavy (small center)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x).astype(float) + 1.0) ** 1.5
    inits.append(edge_w / edge_w.sum())

    # Anti-symmetric-ish (helps reduce center peak)
    t = x / (n - 1)
    asym = (1.0 + 0.5 * np.sin(2 * np.pi * t)) ** 2 + 0.1
    inits.append(asym / asym.sum())

    # Two-peak
    g2 = np.exp(-0.5 * ((x - 0.2 * (n - 1)) / (0.08 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.8 * (n - 1)) / (0.08 * n)) ** 2)
    inits.append(g2 / g2.sum())

    # Random perturbation of parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px + rng.standard_normal(n) * 0.03, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits
