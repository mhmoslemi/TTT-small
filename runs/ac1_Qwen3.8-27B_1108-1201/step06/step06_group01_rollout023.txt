def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_matvec(p, a):
    """Compute (P(p) @ a)_j = sum_m p_{j+m} * a_m using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    # Reverse p: r_k = p_{size-1-k}
    r = p[::-1]
    # Circular convolution of a and r, length = fft_size
    A = np.fft.rfft(a, fft_size)
    R = np.fft.rfft(r, fft_size)
    conv = np.fft.irfft(A * R, fft_size)[:fft_size]
    # (P(p)@a)_j = conv[fft_size - 1 - j] for j = 0..n-1
    # Actually: (a*r)_k where r is reversed p, circular conv of size fft_size
    # We need (a*r)_{size-1-j} = sum_m a_m * r_{size-1-j-m} = sum_m a_m * p_{j+m}
    result = np.zeros(n)
    for j in range(n):
        idx = (size - 1 - j) % fft_size
        result[j] = conv[idx]
    return result


def hankel_matvec_fast(p, a):
    """Faster Hankel matrix-vector product using direct FFT trick."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    # We want (P(p)@a)_j = sum_{m=0}^{n-1} a_m * p_{j+m}
    # This equals (a * p_rev)_{size-1-j} where p_rev[k] = p[size-1-k]
    # Using circular convolution of length >= 2n-1
    p_rev = p[::-1]
    A = np.fft.rfft(a, fft_size)
    R = np.fft.rfft(p_rev, fft_size)
    circ_conv = np.fft.irfft(A * R, fft_size)
    # Extract: (P(p)@a)_j = circ_conv[(size-1-j) % fft_size]
    indices = np.zeros(n, dtype=int)
    for j in range(n):
        indices[j] = (size - 1 - j) % fft_size
    return circ_conv[indices]


def compute_2var_optimal(vals, target_sum, cost_func):
    """
    Given two variables with current values vals[0], vals[1] that should sum to target_sum,
    find the split that minimizes cost_func(x, target_sum - x).
    Uses golden section search.
    """
    lo = max(0.0, vals[0] - target_sum)
    hi = min(target_sum, vals[0] + target_sum)
    lo = max(0.0, lo)
    hi = min(target_sum, hi)
    if hi - lo < 1e-14:
        return vals.copy()
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = cost_func(c, target_sum - c)
    fd = cost_func(d, target_sum - d)
    
    for _ in range(40):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = cost_func(c, target_sum - c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = cost_func(d, target_sum - d)
    
    x = (lo + hi) / 2.0
    return np.array([x, target_sum - x])

def peak_coord_descent(a0, time_limit, top_k=8, pairs_per_peak=6):
    """
    Peak-directed block coordinate descent.
    For each of the top peaks, find variable pairs that contribute most,
    and optimize their split to reduce the peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    t0 = time.monotonic()
    stall = 0
    it = 0

        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        conv = fast_convolve(a)
        # Get top peaks
        k_peaks = min(top_k, len(conv))
        peak_indices = np.argpartition(conv, -k_peaks)[-k_peaks:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        improved = False

        for k_peak in peak_indices:
            if time.monotonic() - t0 > time_limit - 2.0:
                break
            kp = int(k_peak)

            # Variables affecting this peak: a_j * a_{kp-j} for valid j
            js = max(0, kp - n + 1)
            je = min(n - 1, kp)
            
            # Find the pair with largest product
            products = np.zeros(je - js + 1)
            for idx, j in enumerate(range(js, je + 1)):
                partner = kp - j
                if 0 <= partner < n:
                    products[idx] = a[j] * a[partner]
                    # If j == partner, it's a square term
                    if j == partner:
                        products[idx] = a[j] * a[j]
            
            # Sort by product (descending)
            order = np.argsort(products)[::-1]
            
            for _ in range(pairs_per_peak):
                if time.monotonic() - t0 > time_limit - 2.0:
                    break
                if _ >= len(order):
                    break
                idx = order[_]
                j = js + idx
                partner = kp - j
                
                if j == partner:
                    # Square term: a_j^2. Can't split, skip.
                    continue
                if j < 0 or j >= n or partner < 0 or partner >= n:
                    continue
                if j == partner:
                    continue
                # Avoid double-processing
                if j > partner:
                    continue

                # Try to reduce a[j]*a[partner] by moving mass
                # Keep a[j] + a[partner] constant
                total = a[j] + a[partner]
                if total < 1e-15:
                    continue

                # Try several splits
                current_val = a[j] * a[partner]
                best_gain = 0
                best_split = None

                # The product is maximized when a[j] = a[partner] = total/2
                # So to reduce it, move toward the extremes

                    x_new = total * frac
                    y_new = total * (1.0 - frac)
                    if abs(x_new - a[j]) < 1e-12:
                        continue
                    a_test = a.copy()
                    a_test[j] = x_new
                    a_test[partner] = y_new
                    a_test = np.maximum(a_test, 0.0)
                    st = a_test.sum()
                    if st < 1e-15:
                        continue
                    a_test = a_test / st
                    c_test = fast_convolve(a_test)
                    new_max = c_test.max()
                    gain = best_obj - new_max
                    if gain > best_gain:
                        best_gain = gain
                        best_split = (x_new, y_new)

                if best_split is not None and best_gain > 1e-13:
                    a[j] = best_split[0]
                    a[partner] = best_split[1]
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    c_new = fast_convolve(a)
                    new_max = c_new.max()
                    if new_max < best_obj:

    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    k_current = max_k

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)

        # Adaptive k: increase when no improvement
        k_use = min(k_current, len(conv))
        peak_indices = np.argpartition(conv, -k_use)[-k_use:]
        pv = conv[peak_indices]
        # Weight by value (higher peaks get more weight)
        w = pv - pv.min() + 1e-10
        w = w / w.sum()

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peak_indices)):
            if w[i] < 1e-15:
                continue
            k = int(peak_indices[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                partner = k - j
                if 0 <= partner < n:
                    sg[j] += w[i] * a[partner]
                    if j != partner:
                        sg[partner] += w[i] * a[j]

        # Momentum update
        v = momentum * v - lr * sg
        a_new = a + v
        a_new = project_simplex(a_new)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            # Decrease k as we converge (focus on fewer peaks)
            if k_current > 5 and it % 500 == 0:
                k_current = max(5, k_current - 1)
        else:
            no_improve += 1
            a_new = a

        a = a_new
        lr *= (1.0 - 0.00005)

        # Adaptive restart
        if no_improve > 80:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.001 * min(3.0, 1.0 + no_improve / 80.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            lr = lr_init
            k_current = max_k
            no_improve = 0

    return best_a, best_obj


def micro_polish(a0, time_limit):
    """Very fine structured perturbations targeting the current peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    base_scale = 0.0003

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        # Focus on top 3 peaks
        top3 = np.argpartition(conv, -3)[-3:]
        
        scale = base_scale * (1.0 + 0.5 * np.sin(it * 0.1))
        rng = np.random.default_rng((it * 2654435761 + 42) % (2**32))

        for _ in range(8):
            if time.monotonic() - t0 > time_limit - 0.2:
                break
            noise = rng.standard_normal(n) * scale
            # Bias against peak contributors
            for kp in top3:
                kp = int(kp)
                js = max(0, kp - n + 1)
                je = min(n - 1, kp)
                for j in range(js, je + 1):
                    partner = kp - j
                    if 0 <= partner < n:
                        noise[j] -= scale * 1.5 * a[partner]
            
            candidate = project_simplex(a + noise)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                break

    return best_a, best_obj


def fw_lp_step(f, M, n, time_per_lp=5.0):
    """Single FW step with LP oracle."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_per_lp})
        if res.success:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None


def fw_short(a0, time_limit, max_iters=50):
    """Short FW with LP oracle - used for final refinement."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_f = f.copy()
    best_obj = M

    t0 = time.monotonic()
    m_exp = 1.0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit - 2.0:
            break

        remaining = time_limit - (time.monotonic() - t0)
        t_lp = min(5.0, max(1.0, remaining / 10.0))

        g = fw_lp_step(f, M * m_exp, n, time_per_lp=t_lp)
        if g is None:
            m_exp *= 1.2
            if m_exp > 5.0:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-12:
            # Fine line search
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj:
                best_obj = M
                best_f = f.copy()

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search combining:
    - Smoothed gradient (log-sum-exp) with annealing
    - Dual ascent (peak-weight alternation)
    - Momentum subgradient
    - Peak-directed coordinate descent
    - FW with LP for final refinement
    - Micro-polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    t0_total = time.monotonic()

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={time.monotonic()-t0_total:.1f}s)")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform")

    n_parent = len(parent)

    # === Phase 1: n-exploration (3 min) ===
    print("\n=== Phase 1: n-exploration ===")
    t_phase = time.monotonic()

    n_candidates = sorted(set([
        n_parent, n_parent - 64, n_parent + 64,
        n_parent - 128, n_parent + 128,
        640, 700, 832, 896
    ]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 1200]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 1500:
            break
        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss > 1e-10:
                start = start / ss
            else:
                start = np.ones(n_try) / n_try

        t_alloc = min(120.0, deadline - time.monotonic() - 1450)
        if t_alloc < 10:
            break
        print(f"  Smoothed grad n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = smoothed_gradient_optimize(start, time_limit=t_alloc,
                                                T_init=0.002, lr_init=0.04)
        try_seq(a_opt, f"smooth_n{n_try}")

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # === Phase 2: Momentum subgradient (12 min) ===
    print("\n=== Phase 2: Momentum Subgradient ===")
    t_phase = time.monotonic()

    configs_2 = [
        {'lr_init': 0.003, 'max_k': 20, 'momentum': 0.9},
        {'lr_init': 0.005, 'max_k': 15, 'momentum': 0.92},
        {'lr_init': 0.002, 'max_k': 30, 'momentum': 0.88},
    ]
    for ci, cfg in enumerate(configs_2):
        t_alloc = min(240, deadline - time.monotonic() - 900)
        if t_alloc < 20:
            break
        print(f"  Momentum lr={cfg['lr_init']}, k={cfg['max_k']}, t={t_alloc:.0f}s")
        a_opt, _ = momentum_subgradient(best_seq, time_limit=t_alloc, **cfg)
        try_seq(a_opt, f"mom_{ci}")

    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Dual ascent (12 min) ===
    print("\n=== Phase 3: Dual Ascent ===")
    t_phase = time.monotonic()

    configs_3 = [
        {'inner_iters': 30, 'inner_lr': 0.015},
        {'inner_iters': 50, 'inner_lr': 0.01},
        {'inner_iters': 20, 'inner_lr': 0.02},
    ]
    for ci, cfg in enumerate(configs_3):
        t_alloc = min(240, deadline - time.monotonic() - 700)
        if t_alloc < 20:
            break
        print(f"  Dual ascent cfg{ci}, t={t_alloc:.0f}s")
        a_opt, _ = dual_ascent_optimize(best_seq, time_limit=t_alloc, **cfg)
        try_seq(a_opt, f"dual_{ci}")

    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Smoothed gradient (12 min) ===
    print("\n=== Phase 4: Smoothed Gradient ===")
    t_phase = time.monotonic()

    configs_4 = [
        {'T_init': 0.001, 'lr_init': 0.05},
        {'T_init': 0.003, 'lr_init': 0.03},
        {'T_init': 0.0005, 'lr_init': 0.08},
    ]
    for ci, cfg in enumerate(configs_4):
        t_alloc = min(240, deadline - time.monotonic() - 500)
        if t_alloc < 20:
            break
        print(f"  Smoothed T={cfg['T_init']}, lr={cfg['lr_init']}, t={t_alloc:.0f}s")
        a_opt, _ = smoothed_gradient_optimize(best_seq, time_limit=t_alloc, **cfg)
        try_seq(a_opt, f"smooth2_{ci}")

    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Peak coordinate descent (10 min) ===
    print("\n=== Phase 5: Peak Coordinate Descent ===")
    t_phase = time.monotonic()

    t_alloc = min(300, deadline - time.monotonic() - 350)
    if t_alloc > 20:
        print(f"  Coord descent, t={t_alloc:.0f}s")
        a_opt, _ = peak_coord_descent(best_seq, time_limit=t_alloc,
                                       top_k=6, pairs_per_peak=5)
        try_seq(a_opt, "coord")

    # Another round with different params
    t_alloc = min(180, deadline - time.monotonic() - 200)
    if t_alloc > 10:
        print(f"  Coord descent v2, t={t_alloc:.0f}s")
        a_opt, _ = peak_coord_descent(best_seq, time_limit=t_alloc,
                                       top_k=10, pairs_per_peak=4)
        try_seq(a_opt, "coord2")

    print(f"  Phase 5 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: FW with LP (10 min) ===
    print("\n=== Phase 6: FW with LP ===")
    t_phase = time.monotonic()

    t_alloc = min(300, deadline - time.monotonic() - 200)
    if t_alloc > 20:
        print(f"  FW-LP, t={t_alloc:.0f}s")
        a_opt, _ = fw_short(best_seq, time_limit=t_alloc, max_iters=40)
        try_seq(a_opt, "fw")

    # Perturbed FW
    for scale in [0.003, 0.008]:
        t_alloc = min(120, deadline - time.monotonic() - 100)
        if t_alloc < 10:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * scale)
        print(f"  FW perturbed scale={scale}, t={t_alloc:.0f}s")
        a_opt, _ = fw_short(f_pert, time_limit=t_alloc, max_iters=20)
        try_seq(a_opt, f"fw_p{scale}")

    print(f"  Phase 6 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 7: Final polishing (8 min) ===
    print("\n=== Phase 7: Final Polish ===")
    t_phase = time.monotonic()

    # Micro polish
    t_alloc = min(180, deadline - time.monotonic() - 120)
    if t_alloc > 10:
        print(f"  Micro polish, t={t_alloc:.0f}s")
        a_opt, _ = micro_polish(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "micro")

    # Very low T smoothed gradient
    t_alloc = min(120, deadline - time.monotonic() - 60)
    if t_alloc > 10:
        print(f"  Final smoothed, t={t_alloc:.0f}s")
        a_opt, _ = smoothed_gradient_optimize(best_seq, time_limit=t_alloc,
                                                T_init=0.0003, lr_init=0.02)
        try_seq(a_opt, "final_smooth")

    # One more micro polish
    t_alloc = min(60, deadline - time.monotonic() - 30)
    if t_alloc > 5:
        a_opt, _ = micro_polish(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "micro2")

    # Final FW
    t_alloc = min(30, deadline - time.monotonic() - 15)
    if t_alloc > 5:
        a_opt, _ = fw_short(best_seq, time_limit=t_alloc, max_iters=10)
        try_seq(a_opt, "final_fw")

    print(f"  Phase 7 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy() if parent is not None else np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, total time={time.monotonic()-t0_total:.1f}s)")
    return [float(x) for x in best_seq]
