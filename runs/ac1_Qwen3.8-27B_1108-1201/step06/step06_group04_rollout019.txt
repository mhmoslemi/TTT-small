**Better idea: Use the dual directly with a different strategy**

The dual problem is:
$$\max_p \min_{a: \sum a=1, a\geq 0} \sum_k p_k (a*a)[k]$$

The inner min is the QP. The outer max is over $p \in \Delta_{2n-1}$.

By the minimax theorem (since the problem is convex-concave after appropriate reformulation), the optimal value of the dual equals the optimal value of the primal.

A simple approach:
1. Initialize $p$ uniformly.
2. At each step:
   a. Solve the QP to get $a^*$.
   b. Compute the subgradient: $g_k = (a^* * a^*)[k]$.
   c. Update: $p \leftarrow \text{project}(p + \eta \cdot g)$.
3. The Converged value of $\sum_k p_k (a^* * a^*)[k]$ is the optimal value.

For Nesterov acceleration, we use:
- $t_k = (1 + \sqrt{1 + 4t_{k-1}^2}) / 2$
- $y_k = p_k + \frac{t_{k-1} - 1}{t_k} (p_k - p_{k-1})$
- Solve QP at $y_k$ to get $a^*_k$
- $p_{k+1} = \text{project}(y_k - \eta \cdot g_k)$

The step size for Nesterov is typically $\eta = 1/(L \cdot t_k)$ where $L$ is the Lipschitz constant.

The Lipschitz constant of the dual function: The subgradient is $g = a^* * a^*$, and $|g_k| \leq (a^* * a^*)[k] \leq (\sum a_i)^2 = 1$ (since $a_i \geq 0$ and $\sum a_i = 1$). So the subgradient is bounded in $[0, 1]$ componentwise, and its $\ell_2$ norm is bounded by $\sqrt{2n-1}$.

Actually, for the dual problem on the simplex, the relevant Lipschitz constant for the subgradient is the $\ell_\infty$ norm of the subgradient (when using Euclidean projection), which is $\max_k g_k \leq 1$.

So we can use $\eta = 1/t_k$ or similar.

def dual_nesterov(n, time_limit, p_init=None):
    """Solve the dual problem using Nesterov-accelerated subgradient."""
    if p_init is None:
        p = np.ones(2*n-1) / (2*n-1)
    else:
        p = p_init.copy()
    
    a = np.ones(n) / n  # warm start for primal
    p_prev = p.copy()
    t = 1.0
    t_prev = 1.0
    
    best_val = float('inf')
    best_a = a.copy()
    best_p = p.copy()
    
    t0 = time.monotonic()
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov lookahead
        y = p + (t_prev - 1) / t * (p - p_prev)
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        # Solve QP at y
        a = solve_qp_active(y, a, n)
        
        # Compute subgradient
        g = fast_convolve(a)  # auto-convolution
        
        # Dual objective
        val = np.dot(y, g)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            best_p = y.copy()
        
        # Update
        p_new = project_simplex(y - g / t)
        p_prev = p
        t_prev = t
        t = (1 + np.sqrt(1 + 4*t*t)) / 2
        p = p_new

import numpy as np

def fast_convolve(a):

    n = len(a)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])

    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]

    return np.maximum(v - theta, 0.0)


def solve_qp_hankel(p, a_init, n, max_iters=300):

    Solve min a^T P a s.t. sum(a)=1, a>=0 using active set method.
    P is Hankel: P_{i,j} = p[i+j].

        a = a / s

        active = set(range(n))

        if len(active) == 0:
            return np.ones(n) / n
        if len(active) == 1:

            a[list(active)[0]] = 1.0
            return a

        act = sorted(active)
        k = len(act)

            x = np.linalg.solve(P_SS, np.ones(k))

            except:
                x = np.ones(k) / k

            a[act] = 1.0 / k

def dual_subgrad(n, time_limit, p_init=None, a_init=None, verbose=False):

def coord_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent: for each pair (i,j), try to transfer mass from i to j
    to reduce the max auto-convolution.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        improved = False

        # Try to reduce the peak by moving mass
        # The peak at k_peak is sum_i a[i] * a[k_peak - i]
        # To reduce it, move mass from a position that contributes to the peak
        # to a position that doesn't.
        
        # Find positions contributing to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = set(range(js, je + 1))
        non_contrib = set(range(n)) - contrib

        if not non_contrib:
            break

        # Try moving a small amount from the largest contributing position
        # to the position that causes the least increase elsewhere
        idx_from = js + int(np.argmax(a[js:je+1]))
        val = a[idx_from]
        if val < 1e-12:
            break

        # Try different fractions and targets
        best_frac = None
        best_target = None
        best_new_obj = best_obj

        for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
            shift = val * frac
            if shift < 1e-14:
                continue
            for j in list(non_contrib)[:min(50, len(non_contrib))]:
                a_trial = a.copy()
                a_trial[idx_from] -= shift
                a_trial[j] += shift
                if a_trial[idx_from] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                c = fast_convolve(a_trial)
                if c.max() < best_new_obj - 1e-15:
                    best_new_obj = c.max()
                    best_frac = frac
                    best_target = j

        if best_frac is not None and best_target is not None:
            a[idx_from] -= val * best_frac
            a[best_target] += val * best_frac
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_obj = best_new_obj
            best_a = a.copy()
            improved = True

        if not improved and it > 200:
            # Try random perturbations
            rng = np.random.default_rng((it * 7919 + k_peak) % (2**31))
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                shift = a[i] * 0.01
                if shift < 1e-14:
                    continue
                a_trial = a.copy()
                a_trial[i] -= shift
                a_trial[j] += shift
                if a_trial[i] < 0 or a_trial[j] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                c = fast_convolve(a_trial)
                if c.max() < best_obj - 1e-15:
                    a = a_trial
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and it > 500:
            break

    return best_a, best_obj
