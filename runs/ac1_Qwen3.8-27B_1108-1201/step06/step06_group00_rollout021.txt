"""
Spectral flattening + projected subgradient + coordinate descent optimization
for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def normalize(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def spectral_flatten(a, alpha=0.5, beta=0.5):
    """Apply spectral flattening: reduce dynamic range of |FFT(a)|."""
    n = len(a)
    F = rfft(a)
    mag = np.abs(F)
    mag_max = mag.max()
    if mag_max < 1e-15:
        return a.copy()
    # Flatten: new_mag = mag / (1 + alpha * (mag/mag_max)^beta)
    # This reduces peaks while preserving overall shape
    ratio = mag / mag_max
    scale = 1.0 / (1.0 + alpha * ratio**beta)
    mag_new = mag * scale
    F_new = mag_new * np.exp(1j * np.angle(F))
    a_new = irfft(F_new, n)
    return a_new

def spectral_flatten_power(a, beta=0.7):
    """Power-based flattening: new_mag = mag^(1-beta) * mag^(beta) normalized."""
    n = len(a)
    F = rfft(a)
    mag = np.abs(F)
    mag_max = mag.max()
    if mag_max < 1e-15:
        return a.copy()
    # Normalize mag to [0, 1], apply power, scale back
    mag_norm = mag / mag_max
    mag_new_norm = mag_norm ** (1.0 - beta)
    mag_new = mag_new_norm * mag_max
    F_new = mag_new * np.exp(1j * np.angle(F))
    a_new = irfft(F_new, n)
    return a_new

def subgradient_step(a, peaks, weights, step_size):
    """Take a projected subgradient step to reduce peak autocorrelation."""
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(peaks, weights):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += w * 2.0 * a[k - i_range]
    a_new = np.maximum(a - step_size * grad, 0.0)
    return a_new

def coordinate_descent(a, n_coords=100, time_limit=30.0):
    """Coordinate descent: try moving mass from high-contributing coords to low ones."""
    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    top_peaks = np.argsort(conv)[-min(20, 2*n-1):]
    
    for _ in range(n_coords):
        if time.monotonic() - t0 > time_limit:
            break
        k = int(top_peaks[np.random.randint(len(top_peaks))])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo >= hi:
            continue
        i_vals = np.arange(lo, hi + 1)
        contributions = a[i_vals] * a[k - i_vals]
        # Find coordinate with highest contribution
        j = i_vals[np.argmax(contributions)]
        # Try moving small amount to a random low-value coordinate
        dest = np.random.randint(n)
        for frac in [0.05, 0.1, 0.2, 0.02]:
            delta = a[j] * frac
            if delta < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[j] -= delta
            a_trial[dest] += delta
            if a_trial[j] < 0:
                continue
            a_trial = normalize(a_trial)
            conv_trial = fast_convolve(a_trial)
            if conv_trial.max() < best_obj - 1e-15:
                best_obj = conv_trial.max()
                best_a = a_trial.copy()
                a = a_trial.copy()
                break
    
    return best_a, best_obj

            break
        
        # Progressively increase flattening strength
        progress = min(1.0, it / max(1, int(time_limit / 0.5)))
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        beta = beta_start + (beta_end - beta_start) * progress
        
        # Spectral flattening step
        a_flat = spectral_flatten(a, alpha=alpha, beta=beta)
        a_new = normalize(a_flat)

            a_pow = normalize(a_pow)
            # Take the better one

                a_new = a_pow

        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:

            a = a_new.copy()
            no_improve = 0
        else:

                a_mixed = normalize(mix * best_a + (1 - mix) * a_new)
                conv_mix = fast_convolve(a_mixed)
                if conv_mix.max() < best_obj - 1e-15:
                    best_obj = conv_mix.max()
                    best_a = a_mixed.copy()
                    a = a_mixed.copy()
                    no_improve = 0

def subgradient_refine(a_init, time_limit, top_k=50, beta=50.0):

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    no_improve = 0

        if time.monotonic() - t0 > time_limit - 1:
            break
        
        conv = fast_convolve(a)
        # LSE weights

        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        w_peaks = w[peak_idx]
        w_sum = w_peaks.sum()
        if w_sum > 1e-15:
            w_peaks = w_peaks / w_sum
        else:
            w_peaks = np.ones(k_top) / k_top
        
        # Compute gradient

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    if len(f) == new_n:
        return f.copy()
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    return normalize(f_new)

def generate_from_spectrum(target_mag, n, phase=None):
    """Generate a sequence from a target magnitude spectrum."""
    if phase is None:

        phase = 2 * np.pi * rng.random(len(target_mag))
    F = target_mag * np.exp(1j * phase)
    a = irfft(F, n)
    return normalize(a)

def lowpass_template(n, bandwidth=0.3):

    n_mag = n // 2 + 1

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def spectral_flatten(a, alpha=0.5, beta=0.5):
    """Apply spectral flattening: reduce dynamic range of |FFT(a)|."""
    n = len(a)
    F = rfft(a)
    mag = np.abs(F)
    mag_max = mag.max()
    if mag_max < 1e-15:
        return a.copy()
    ratio = mag / mag_max
    scale = 1.0 / (1.0 + alpha * ratio ** beta)
    mag_new = mag * scale
    F_new = mag_new * np.exp(1j * np.angle(F))
    a_new = irfft(F_new, n)
    return a_new


def spectral_flatten_power(a, beta=0.7):
    """Power-based flattening: compress dynamic range via power law."""
    n = len(a)
    F = rfft(a)
    mag = np.abs(F)
    mag_max = mag.max()
    if mag_max < 1e-15:
        return a.copy()
    mag_norm = mag / mag_max
    mag_new_norm = mag_norm ** (1.0 - beta)
    mag_new = mag_new_norm * mag_max
    F_new = mag_new * np.exp(1j * np.angle(F))
    a_new = irfft(F_new, n)
    return a_new


def spectral_shape_towards_target(a, target_mag, gamma=0.5):
    """Shape spectrum geometrically towards a target magnitude profile."""
    n = len(a)
    F = rfft(a)
    mag = np.abs(F)
    # Avoid division by zero
    safe_mag = np.maximum(mag, 1e-15)
    safe_target = np.maximum(target_mag, 1e-15)
    # Geometric interpolation: F_new = F * (target/mag)^gamma
    ratio = (safe_target / safe_mag) ** gamma
    F_new = F * ratio
    # Ensure F[0] is preserved (DC component = sum)
    F_new[0] = F[0]
    a_new = irfft(F_new, n)
    return a_new


def flat_target(n):
    """Flat spectral target (unit magnitude at all bins)."""
    return np.ones(n // 2 + 1)


def cosine_target(n, roll_off=0.5):
    """Raised-cosine spectral target."""
    n_mag = n // 2 + 1
    freqs = np.arange(n_mag) / n
    return np.cos(np.pi * freqs / roll_off) ** 2


def generate_from_spectrum(target_mag, n, seed=0):
    """Generate a non-negative sequence from a target magnitude spectrum."""
    rng = np.random.default_rng(seed)
    phase = 2 * np.pi * rng.random(len(target_mag))
    phase[0] = 0.0
    F = target_mag * np.exp(1j * phase)
    a = irfft(F, n)
    return normalize(a)


def subgradient_refine(a_init, time_limit, top_k=50, beta_start=30.0, beta_end=2000.0):
    """Subgradient refinement with LSE-smoothed peak weighting."""
    a = normalize(a_init.copy())
    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = 0.5 / np.sqrt(n)
    beta = beta_start
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)
        # LSE weights
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        k_top = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        w_peaks = w[peak_idx]
        w_sum = w_peaks.sum()
        if w_sum > 1e-15:
            w_peaks = w_peaks / w_sum
        else:
            w_peaks = np.ones(k_top) / k_top

        # Compute subgradient
        grad = np.zeros(n)
        for idx, k in enumerate(peak_idx):
            k = int(k)
            wp = w_peaks[idx]
            if wp < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            i_range = np.arange(lo, hi + 1)
            grad[i_range] += wp * 2.0 * a[k - i_range]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm

        # Backtracking line search
        step = lr
        improved = False
        for _ in range(10):
            a_new = project_simplex(a - step * grad)
            conv_new = fast_convolve(a_new)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                no_improve = 0
                lr = min(lr * 1.01, 3.0 / np.sqrt(n))
                break
            step *= 0.55

        if not improved:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.85
            if no_improve > 50:
                a = best_a.copy()
                lr = max(0.5 / np.sqrt(n), 0.001)
                # Diversify via spectral perturbation
                F = rfft(a)
                mag = np.abs(F)
                phase = np.angle(F)
                rng_local = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
                mag_pert = np.maximum(mag * (1.0 + rng_local.standard_normal(len(mag)) * 0.01), 0)
                a = normalize(irfft(mag_pert * np.exp(1j * phase), n))
                no_improve = 0

        if beta < beta_end:
            beta *= 1.005

    return best_a, best_obj


def spectral_optimize(a_init, time_limit, alpha_start=0.3, beta_start=0.5,
                     alpha_end=1.2, beta_end=0.85, use_shaping=False):
    """Main spectral flattening/shaping optimization loop."""
    a = normalize(a_init.copy())
    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(42)

    # Target spectrum for shaping
    target = None
    if use_shaping:
        target = cosine_target(n, roll_off=0.4)
        target = target / target[0]  # Normalize DC to 1

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        progress = min(1.0, it / max(10, int(time_limit / 0.3)))
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        beta = beta_start + (beta_end - beta_start) * progress

        candidates = []

        # Candidate 1: Standard flattening
        a_flat = spectral_flatten(a, alpha=alpha, beta=beta)
        candidates.append(normalize(a_flat))

        # Candidate 2: Power-based flattening
        a_pow = spectral_flatten_power(a, beta=0.4 + 0.4 * progress)
        candidates.append(normalize(a_pow))

        # Candidate 3: Target shaping
        if use_shaping and target is not None:
            a_shape = spectral_shape_towards_target(a, target, gamma=0.3 + 0.3 * progress)
            candidates.append(normalize(a_shape))

        # Select the best candidate
        best_candidate = a
        best_candidate_obj = conv.max()
        for cand in candidates:
            conv_c = fast_convolve(cand)
            obj_c = conv_c.max()
            if obj_c < best_candidate_obj:
                best_candidate_obj = obj_c
                best_candidate = cand

        # Try mixing with best
        a_new = best_candidate
        for mix_frac in [0.3, 0.5, 0.7]:
            a_mix = normalize(mix_frac * best_a + (1.0 - mix_frac) * a_new)
            conv_mix = fast_convolve(a_mix)
            if conv_mix.max() < best_candidate_obj:
                a_new = a_mix
                best_candidate_obj = conv_mix.max()

        if best_candidate_obj < best_obj - 1e-15:
            best_obj = best_candidate_obj
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            a = normalize(a_new)

        # Periodic diversification
        if no_improve > 15:
            F = rfft(a)
            mag = np.abs(F)
            phase = np.angle(F)
            scale = 0.015 + 0.005 * min(3, no_improve // 15)
            mag_pert = np.maximum(mag * (1.0 + rng.standard_normal(len(mag)) * scale), 0)
            F_pert = mag_pert * np.exp(1j * phase)
            a_pert = normalize(irfft(F_pert, n))
            conv_pert = fast_convolve(a_pert)
            if conv_pert.max() < best_obj:
                best_obj = conv_pert.max()
                best_a = a_pert.copy()
            a = a_pert.copy()
            no_improve = 0

    return best_a, best_obj


def coordinate_descent(a_init, n_iterations=500, time_limit=60.0):
    """Coordinate descent: move mass to reduce top peaks."""
    a = normalize(a_init.copy())
    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)

    for _ in range(n_iterations):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        # Sample from top peaks with soft weighting
        top_k = min(30, 2 * n - 1)
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peak_idx]
        # Weight by excess over mean
        excess = np.maximum(peak_vals - best_obj * 0.95, 0.0)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            excess = np.ones(top_k) / top_k
        else:
            excess = excess / total_excess
        probs = excess / excess.sum()
        k = int(peak_idx[rng.choice(top_k, p=probs)])

        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo >= hi:
            continue

        i_vals = np.arange(lo, hi + 1)
        contributions = a[i_vals] * a[k - i_vals]
        j = i_vals[np.argmax(contributions)]

        # Try moving mass to a few candidate destinations
        for frac in [0.03, 0.08, 0.15]:
            delta = a[j] * frac
            if delta < 1e-15:
                continue
            for _ in range(3):
                dest = rng.integers(n)
                if dest == j:
                    continue
                a_trial = a.copy()
                a_trial[j] -= delta
                a_trial[dest] += delta
                if a_trial[j] < 0:
                    continue
                a_trial = normalize(a_trial)
                conv_trial = fast_convolve(a_trial)
                if conv_trial.max() < best_obj - 1e-15:
                    best_obj = conv_trial.max()
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    break

    return best_a, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    if len(f) == new_n:
        return f.copy()
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    return normalize(f_new)


def try_sequence(a, best_tracker):
    """Evaluate and track best."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization across lengths and spectral shapes
    Phase 2: Spectral flattening (primary optimizer)
    Phase 3: Subgradient refinement
    Phase 4: Alternating cycles (spectral + subgradient + coordinate)
    Phase 5: Final polish and verification
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    total_time = float(budget_s) - 25.0
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        try_sequence(parent, best_tracker)

    n_parent = len(parent)

    # === Phase 1: Diverse initialization ===
    print("\n=== Phase 1: Diverse initialization ===")
    t1_end = time.monotonic() + total_time * 0.12

    lengths = sorted(set([
        n_parent,
        max(128, n_parent - 128),
        min(2500, n_parent + 128),
        max(128, int(n_parent * 0.65)),
        min(2500, int(n_parent * 1.35)),
        512, 768, 1024, 1280, 1536, 2048,
    ]))
    lengths = [l for l in lengths if 128 <= l <= 3000]
    # Limit to 8 lengths
    lengths = lengths[:8]

    for nc in lengths:
        if time.monotonic() > t1_end:
            break

        # Interpolated parent
        f1 = interpolate_to_length(parent, nc)
        try_sequence(f1, best_tracker)

        # Shaped spectrum: cosine
        f2 = generate_from_spectrum(cosine_target(nc, roll_off=0.35), nc, seed=nc)
        try_sequence(f2, best_tracker)

        # Shaped spectrum: flat with noise
        flat_mag = np.ones(nc // 2 + 1)
        flat_mag += 0.2 * rng.random(nc // 2 + 1)
        f3 = generate_from_spectrum(flat_mag, nc, seed=nc + 1)
        try_sequence(f3, best_tracker)

    print(f"  After Phase 1: best={best_tracker[0]:.8f}")

    # === Phase 2: Spectral flattening on top candidates ===
    print(f"\n=== Phase 2: Spectral flattening ===")
    t2_end = time.monotonic() + total_time * 0.30

    # Run spectral optimization on the current best
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)
    t_alloc = min(120, t2_end - time.monotonic())
    if t_alloc > 15:
        print(f"  Spectral flatten n={n_cur}, t={t_alloc:.0f}s")
        f_opt, _ = spectral_optimize(current_best, time_limit=t_alloc,
                                      alpha_start=0.3, beta_start=0.5,
                                      alpha_end=1.2, beta_end=0.85,
                                      use_shaping=False)
        try_sequence(f_opt, best_tracker)

    # Try with shaping
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(80, t2_end - time.monotonic())
    if t_alloc > 15:
        print(f"  Spectral shape n={len(current_best)}, t={t_alloc:.0f}s")
        f_opt, _ = spectral_optimize(current_best, time_limit=t_alloc,
                                      alpha_start=0.2, beta_start=0.4,
                                      alpha_end=0.8, beta_end=0.7,
                                      use_shaping=True)
        try_sequence(f_opt, best_tracker)

    # Try a different length
    n_alt = max(256, len(best_tracker[1]) + 192) if best_tracker[1] is not None else 1024
    if n_alt <= 3000 and time.monotonic() < t2_end - 15:
        f_init = interpolate_to_length(best_tracker[1] if best_tracker[1] is not None else parent, n_alt)
        t_alloc = min(60, t2_end - time.monotonic())
        if t_alloc > 10:
            print(f"  Spectral (new length) n={n_alt}, t={t_alloc:.0f}s")
            f_opt, _ = spectral_optimize(f_init, time_limit=t_alloc,
                                          alpha_start=0.3, beta_start=0.5,
                                          alpha_end=1.0, beta_end=0.8,
                                          use_shaping=False)
            try_sequence(f_opt, best_tracker)

    print(f"  After Phase 2: best={best_tracker[0]:.8f}")

    # === Phase 3: Subgradient refinement ===
    print(f"\n=== Phase 3: Subgradient refinement ===")
    t3_end = time.monotonic() + total_time * 0.25

    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(100, t3_end - time.monotonic())
    if t_alloc > 15:
        print(f"  Subgradient n={len(current_best)}, t={t_alloc:.0f}s")
        f_opt, _ = subgradient_refine(current_best, time_limit=t_alloc,
                                       top_k=60, beta_start=20.0, beta_end=1000.0)
        try_sequence(f_opt, best_tracker)

    # Second pass with higher beta (focus on top few peaks)
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(60, t3_end - time.monotonic())
    if t_alloc > 10:
        print(f"  Subgradient (high beta) n={len(current_best)}, t={t_alloc:.0f}s")
        f_opt, _ = subgradient_refine(current_best, time_limit=t_alloc,
                                       top_k=25, beta_start=100.0, beta_end=5000.0)
        try_sequence(f_opt, best_tracker)

    print(f"  After Phase 3: best={best_tracker[0]:.8f}")

    # === Phase 4: Alternating cycles ===
    print(f"\n=== Phase 4: Alternating cycles ===")
    t4_end = deadline - 150
    cycles = 0

    while time.monotonic() < t4_end and cycles < 25:
        cycles += 1
        current_best = best_tracker[1] if best_tracker[1] is not None else parent
        remaining = t4_end - time.monotonic()
        if remaining < 20:
            break

        if cycles % 4 == 1:
            # Spectral step
            t_alloc = min(50, remaining * 0.35)
            if t_alloc > 8:
                print(f"  Cycle {cycles}: Spectral n={len(current_best)}, t={t_alloc:.0f}s")
                f_opt, _ = spectral_optimize(current_best, time_limit=t_alloc,
                                               alpha_start=0.5, beta_start=0.6,
                                               alpha_end=1.5, beta_end=0.9,
                                               use_shaping=False)
                try_sequence(f_opt, best_tracker)

        elif cycles % 4 == 2:
            # Subgradient step
            t_alloc = min(60, remaining * 0.35)
            if t_alloc > 8:
                beta_s = 40.0 + 40.0 * cycles
                print(f"  Cycle {cycles}: Subgrad n={len(current_best)}, t={t_alloc:.0f}s, beta={beta_s:.0f}")
                f_opt, _ = subgradient_refine(current_best, time_limit=t_alloc,
                                               top_k=50, beta_start=beta_s, beta_end=beta_s * 20)
                try_sequence(f_opt, best_tracker)

        elif cycles % 4 == 3:
            # Coordinate descent
            t_alloc = min(40, remaining * 0.25)
            if t_alloc > 8:
                print(f"  Cycle {cycles}: CoordDesc n={len(current_best)}, t={t_alloc:.0f}s")
                f_opt, _ = coordinate_descent(current_best, n_iterations=400, time_limit=t_alloc)
                try_sequence(f_opt, best_tracker)

        else:
            # Spectral perturbation + quick refine
            t_alloc = min(40, remaining * 0.25)
            if t_alloc > 8:
                print(f"  Cycle {cycles}: Perturb+Refine n={len(current_best)}, t={t_alloc:.0f}s")
                for attempt in range(3):
                    if time.monotonic() > t4_end - 10:
                        break
                    F = rfft(current_best)
                    mag = np.abs(F)
                    phase = np.angle(F)
                    strength = 0.008 + 0.005 * attempt
                    mag_pert = np.maximum(mag * (1.0 + rng.standard_normal(len(mag)) * strength), 0)
                    f_pert = normalize(irfft(mag_pert * np.exp(1j * phase), len(current_best)))
                    t_sub = max(5, (t_alloc - 5) / 3 - attempt * 5)
                    if t_sub > 3:
                        f_opt, _ = spectral_optimize(f_pert, time_limit=t_sub,
                                                      alpha_start=0.5, beta_start=0.6,
                                                      alpha_end=1.0, beta_end=0.8)
                        try_sequence(f_opt, best_tracker)

    print(f"  After Phase 4: best={best_tracker[0]:.8f}")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent

    # Final coordinate descent
    t_rem = deadline - time.monotonic()
    if t_rem > 30:
        t_alloc = min(50, t_rem - 15)
        print(f"  Final coord descent, t={t_alloc:.0f}s")
        f_opt, _ = coordinate_descent(current_best, n_iterations=600, time_limit=t_alloc)
        try_sequence(f_opt, best_tracker)

    # Tiny perturbations around the best
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)
    t_rem = deadline - time.monotonic()
    if t_rem > 20:
        for _ in range(30):
            if time.monotonic() > deadline - 5:
                break
            # Spectral micro-perturbation
            F = rfft(current_best)
            mag = np.abs(F)
            phase = np.angle(F)
            pert = rng.standard_normal(len(mag)) * 0.002
            mag_pert = np.maximum(mag * (1.0 + pert), 0)
            f_pert = normalize(irfft(mag_pert * np.exp(1j * phase), n_cur))
            try_sequence(f_pert, best_tracker)

            # Time-domain micro-perturbation
            f_pert2 = project_simplex(current_best + rng.standard_normal(n_cur) * 0.0008)
            try_sequence(f_pert2, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using fallback")
    fallback = np.ones(448) / 448.0
    return [float(x) for x in fallback]
