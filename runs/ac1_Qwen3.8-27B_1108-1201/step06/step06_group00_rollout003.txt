    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def build_full_lp_matrix(f):
    """Build sparse constraint matrix for FULL LP. Shape: (2n-1, n)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    if not rows:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_all_constraints(f, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for ALL k, g >= 0.
    Returns the LP solution or None."""
    n = len(f)
    A = build_full_lp_matrix(f)
    if A.shape[0] == 0:
        return np.ones(n) / n
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def solve_lp_topk(f, M, top_k=40, time_limit=10):
    """Solve LP with top-K constraints only."""
    n = len(f)
    conv = fast_convolve(f)
    peak_idx = np.argpartition(conv, -top_k)[-top_k:]
    center = n - 1
    if center not in peak_idx:
        peak_idx = np.append(peak_idx, center)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(peak_idx):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([idx] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    if not rows:
        return np.ones(n) / n
    A = csr_matrix((vals, (rows, cols)), shape=(len(peak_idx), n))
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_all_constraints(f0, time_limit, use_all=True, verbose=False):
    """Frank-Wolfe with all constraints and BB step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    # BB step tracking
    prev_f = f.copy()
    prev_g = None
    bb_alpha = 2.0 / (n ** 0.5)

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        lp_time = min(12, max(2, remaining * 0.1))
        if use_all and n <= 2048:
            g = solve_lp_all_constraints(f, M, time_limit=lp_time)
        else:
            g = solve_lp_topk(f, M, top_k=50, time_limit=lp_time)

        if g is None:
            no_improve += 2
            if no_improve > 40:
                # Diversify
                rng = np.random.default_rng(it * 613 + 7)
                f = project_simplex(f + rng.standard_normal(n) * 0.01)
                conv = fast_convolve(f)
                M = conv.max()
                if M < best_obj:
                    best_obj = M
                    best_f = f.copy()
                no_improve = 0
            continue

        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        diff = np.max(np.abs(g - f))
        if diff < 1e-11:
            no_improve += 1
            if no_improve > 50:
                rng = np.random.default_rng(it * 997 + 3)
                f = project_simplex(f + rng.standard_normal(n) * 0.015)
                conv = fast_convolve(f)
                M = conv.max()
                if M < best_obj:
                    best_obj = M
                    best_f = f.copy()
                no_improve = 0
            continue

        # BB step size
        if prev_g is not None:
            s_vec = f - prev_f
            y_vec = g - prev_g
            sy = s_vec @ y_vec
            yy = y_vec @ y_vec
            ss = s_vec @ s_vec
            if yy > 1e-15:
                bb1 = sy / yy
                bb2 = ss / sy if abs(sy) > 1e-15 else bb1
                bb_alpha = bb1 if bb1 > 1e-4 else (bb2 if bb2 > 1e-4 else 2.0 / np.sqrt(n))
            bb_alpha = np.clip(bb_alpha, 0.05, 1.0)
        prev_g = g.copy()
        prev_f = f.copy()

        # Line search with BB initial guess
        best_alpha = 0.0
        best_val = M
        # Try BB step and neighbors
        alphas_to_try = [bb_alpha,
                         max(0.05, bb_alpha - 0.1), min(1.0, bb_alpha + 0.1),
                         max(0.05, bb_alpha * 0.7), min(1.0, bb_alpha * 1.4),
                         0.1, 0.3, 0.5, 0.7]
        alphas_to_try = [a for a in alphas_to_try if 0 < a <= 1.0]
        seen_alphas = set()
        for alpha in alphas_to_try:
            alpha_r = round(alpha, 3)
            if alpha_r in seen_alphas:
                continue
            seen_alphas.add(alpha_r)
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Refine around best
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft /= st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-6:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f /= f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if verbose and it % 20 == 0:
            print(f"    FW it={it}, M={M:.8f}, best={best_obj:.8f}, no_impr={no_improve}")

    return best_f, best_obj


def gs_spectral_init(n, n_iterations=200, target_ratio=1.0, seed=42):
    """Gerchberg-Saxton style spectral initialization.
    Tries to equalize the autocorrelation peaks."""
    rng = np.random.default_rng(seed)
    # Start with uniform
    a = np.ones(n) / n

    for it in range(n_iterations):
        conv = fast_convolve(a)
        peak = conv.max()
        if peak < 1e-15:
            break

        # Target: all peaks should be <= peak * target_ratio
        # In frequency domain, try to reduce peaks
        # Use: a_new = IFFT(|FFT(a)|^2 adjusted)
        # Actually, use the "power method" on the peak
        # Compute gradient of the peak
        k_peak = np.argmax(conv)
        lo = max(0, k_peak - n + 1)
        hi = min(n - 1, k_peak)
        grad = np.zeros(n)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * a[k_peak - j]

        # Update: reduce in direction of peak gradient
        step = 0.05 / (np.linalg.norm(grad) / n + 1e-10)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a
        else:
            a_new /= s

        # Only accept if max peak decreased
        conv_new = fast_convolve(a_new)
        if conv_new.max() < peak:
            a = a_new
        else:
            # Try smaller step
            a_new = a - step * 0.3 * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new /= s
                conv_new = fast_convolve(a_new)
                if conv_new.max() < peak:
                    a = a_new

    return a


def freq_diversify(f, rng, strength=0.03):
    """Perturb in frequency domain."""
    n = len(f)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, fft_size)[:n]
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def peak_peeling(f0, time_limit, n_peaks_to_target=5):
    """Target coordinate peeling: for each top peak, try moves that reduce it."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        top_peaks = np.argsort(conv)[-n_peaks_to_target:]
        improved = False

        for k in top_peaks:
            k = int(k)
            if conv[k] < best_obj * 0.9:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo >= hi:
                continue

            # Find the pair (j, k-j) that contributes most to this peak
            js = np.arange(lo, hi + 1)
            products = f[js] * f[k - js]
            j_max = js[np.argmax(products)]
            j_other = k - j_max

            if j_other < 0 or j_other >= n:
                continue

            # Try to move mass from j_max to a position that minimally affects peaks
            # Candidate destinations: positions with low "peak contribution"
            # Score each position by how much it would increase total peak if we add mass there
            n_candidates = min(20, n)
            candidates = []
            for _ in range(n_candidates):
                j_dest = np.random.randint(0, n)
                candidates.append(j_dest)
            candidates = list(set(candidates))

            for j_dest in candidates:
                if j_dest == j_max or j_dest == j_other:
                    continue
                # Try moving a small fraction of mass
                for frac in [0.02, 0.05, 0.1, 0.15]:
                    delta = f[j_max] * frac
                    if delta < 1e-12:
                        continue
                    f_trial = f.copy()
                    f_trial[j_max] -= delta
                    f_trial[j_dest] += delta
                    if f_trial[j_max] < -1e-12:
                        continue
                    f_trial = np.maximum(f_trial, 0)
                    f_trial /= f_trial.sum()
                    conv_trial = fast_convolve(f_trial)
                    if conv_trial.max() < best_obj - 1e-12:
                        best_obj = conv_trial.max()
                        best_f = f_trial.copy()
                        f = f_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and it % 50 == 0:
            # Reset to best and diversify
            f = best_f.copy()
            rng = np.random.default_rng(it * 31)
            f = freq_diversify(f, rng, strength=0.02)

    return best_f, best_obj


def subgradient_lse(f0, time_limit, beta_start=20.0, beta_end=8000.0, top_k=40):
    """Subgradient with LSE smoothing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    beta = beta_start
    lr = 3.0 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        conv = fast_convolve(f)
        k_top = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]

        # LSE weights
        bc = beta * conv[peak_idx]
        bc -= bc.max()
        exp_bc = np.exp(bc)
        w = exp_bc / exp_bc.sum()

        # Subgradient
        grad = np.zeros(n)
        for idx, k in enumerate(peak_idx):
            k = int(k)
            wi = w[idx]
            if wi < 1e-10:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            js = np.arange(lo, hi + 1)
            grad[js] += wi * 2.0 * f[k - js]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_normed = grad / gnorm

        # Backtracking
        step = lr
        improved = False
        for _ in range(12):
            f_new = project_simplex(f - step * grad_normed)
            conv_new = fast_convolve(f_new)
            if conv_new.max() < best_obj - 1e-14:
                best_obj = conv_new.max()
                best_f = f_new.copy()
                f = f_new
                improved = True
                no_improve = 0
                lr = min(lr * 1.003, 15.0 / np.sqrt(n))
                break
            step *= 0.5

        if not improved:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.85
            if no_improve > 60:
                f = best_f.copy()
                lr = max(3.0 / np.sqrt(n), 0.01)
                beta = min(beta * 3.0, beta_end)
                no_improve = 0
                rng = np.random.default_rng(it * 17 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)

        if beta < beta_end:
            beta *= 1.005

    return best_f, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    if len(f) == new_n:
        return f.copy()
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def try_sequence(a, best_tracker):
    """Evaluate and update best tracker."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main optimization:
    Phase 1: Multi-start with spectral init + brief FW (all constraints)
    Phase 2: Intensive FW with all constraints + BB step
    Phase 3: Peak peeling + subgradient alternation
    Phase 4: Final polish with different lengths
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    total_budget = float(budget_s) - 30.0

    # === Phase 1: Multi-start exploration (25% time) ===
    print("\n=== Phase 1: Multi-start exploration ===")
    t1_end = time.monotonic() + total_budget * 0.25

    # Try different starting points
    starts = [('parent', parent)]

    # Frequency perturbed versions
    for i, strength in enumerate([0.03, 0.06, 0.1]):
        f_pert = freq_diversify(parent, rng, strength=strength)
        starts.append((f'freq_{i}', f_pert))

    # Different lengths
    for factor in [0.85, 0.9, 1.1, 1.2]:
        n_new = int(n_parent * factor)
        n_new = max(128, min(2048, n_new))
        if n_new != n_parent:
            f_interp = interpolate_to_length(parent, n_new)
            starts.append((f'len_{n_new}', f_interp))

    # Spectrum init for parent length
    t_remain = t1_end - time.monotonic()
    if t_remain > 30:
        f_gs = gs_spectral_init(n_parent, n_iterations=100, seed=seed)
        starts.append(('gs_init', f_gs))

    # Allocate time for FW on each start
    n_starts = len(starts)
    t_per_start = max(20, (t1_end - time.monotonic()) / n_starts - 5)

    for name, f_start in starts:
        t_rem = t1_end - time.monotonic()
        if t_rem < 15:
            break
        t_alloc = min(t_per_start, t_rem - 5)
        if t_alloc < 15:
            break
        print(f"  FW [{name}] n={len(f_start)}, t={t_alloc:.0f}s")
        f_opt, _ = fw_all_constraints(f_start, time_limit=t_alloc, use_all=True)
        try_sequence(f_opt, best_tracker)

    # === Phase 2: Intensive FW with all constraints (40% time) ===
    print(f"\n=== Phase 2: Intensive FW (all constraints) ===")
    t2_end = t1_end + total_budget * 0.40
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)

    t_alloc = t2_end - time.monotonic()
    if t_alloc > 30:
        print(f"  Intensive FW n={n_cur}, t={t_alloc:.0f}s")
        f_opt, _ = fw_all_constraints(current_best, time_limit=t_alloc, use_all=True)
        try_sequence(f_opt, best_tracker)

        # Second round if time remains
        current_best = best_tracker[1] if best_tracker[1] is not None else current_best
        t_rem = t2_end - time.monotonic()
        if t_rem > 30:
            print(f"  Intensive FW round 2, t={t_rem:.0f}s")
            f_opt, _ = fw_all_constraints(current_best, time_limit=t_rem, use_all=True)
            try_sequence(f_opt, best_tracker)

    # === Phase 3: Peak peeling + subgradient alternation (25% time) ===
    print(f"\n=== Phase 3: Peak peeling + subgradient ===")
    t3_end = t2_end + total_budget * 0.25
    cycles = 0
    while time.monotonic() < t3_end and cycles < 20:
        cycles += 1
        current_best = best_tracker[1] if best_tracker[1] is not None else parent
        n_cur = len(current_best)
        remaining = t3_end - time.monotonic()
        if remaining < 25:
            break

        if cycles % 3 == 1:
            t_alloc = min(60, remaining * 0.4)
            if t_alloc > 15:
                print(f"  Cycle {cycles}: PeakPeel n={n_cur}, t={t_alloc:.0f}s")
                f_opt, _ = peak_peeling(current_best, time_limit=t_alloc, n_peaks_to_target=5)
                try_sequence(f_opt, best_tracker)

        elif cycles % 3 == 2:
            t_alloc = min(80, remaining * 0.45)
            if t_alloc > 15:
                beta_s = 50.0 + 50.0 * cycles
                print(f"  Cycle {cycles}: Subgrad n={n_cur}, t={t_alloc:.0f}s, beta={beta_s:.0f}")
                f_opt, _ = subgradient_lse(current_best, time_limit=t_alloc,
                                           beta_start=beta_s, beta_end=15000.0, top_k=50)
                try_sequence(f_opt, best_tracker)

        else:
            t_alloc = min(50, remaining * 0.4)
            if t_alloc > 15:
                print(f"  Cycle {cycles}: FW-small n={n_cur}, t={t_alloc:.0f}s")
                f_opt, _ = fw_all_constraints(current_best, time_limit=t_alloc, use_all=True)
                try_sequence(f_opt, best_tracker)

    # === Phase 4: Final polish (10% time) ===
    print(f"\n=== Phase 4: Final polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)

    # Try slightly different lengths
    t_rem = deadline - time.monotonic()
    if t_rem > 40:
        for delta in [-32, 32, -64, 64]:
            n_try = n_cur + delta
            if n_try < 128 or n_try > 2048:
                continue
            t_alloc = min(30, t_rem * 0.2)
            if t_alloc < 15:
                break
            t_rem = deadline - time.monotonic()
            f_try = interpolate_to_length(current_best, n_try)
            print(f"  Try n={n_try}, t={t_alloc:.0f}s")
            f_opt, _ = fw_all_constraints(f_try, time_limit=t_alloc, use_all=True)
            try_sequence(f_opt, best_tracker)

    # Final FW pass on best
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_rem = deadline - time.monotonic()
    if t_rem > 20:
        print(f"  Final FW n={len(current_best)}, t={t_rem - 5:.0f}s")
        f_opt, _ = fw_all_constraints(current_best, time_limit=t_rem - 5, use_all=True)
        try_sequence(f_opt, best_tracker)

    # Very short peeling
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_rem = deadline - time.monotonic()
    if t_rem > 15:
        print(f"  Final peeling, t={t_rem - 3:.0f}s")
        f_opt, _ = peak_peeling(current_best, time_limit=t_rem - 3, n_peaks_to_target=3)
        try_sequence(f_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using uniform fallback")
    fallback = np.ones(448) / 448.0
    return [float(x) for x in fallback]
