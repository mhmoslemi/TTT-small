def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_value(a):
    """Compute 2*n*max(conv)/(sum)^2 for non-negative a."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def build_lp_matrix(f, peaks):
    """Build constraint matrix for LP: (f*g)[k] <= M for k in peaks."""
    n = len(f)
    K = len(peaks)
    A = np.zeros((K, n))
    for idx, k in enumerate(peaks):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[idx, js:je + 1] = f[k - je:k - js + 1][::-1]
    return A


def solve_lp_direction(f, M, K=10, max_time=5.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-K peaks, g >= 0."""
    n = len(f)
    conv = fast_convolve(f)
    if K >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -K)[-K:]
    peaks = np.sort(peaks)

    A = build_lp_matrix(f, peaks)
    c = -np.ones(n)
    b_ub = np.full(len(peaks), M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'max_time': max_time})
        if res.success:
            g = np.maximum(res.x, 0.0)
            return g, conv, peaks
    except Exception:
        pass
    return None, conv, peaks


def fw_optimize(f0, time_limit, K=10, verbose=False):
    """Frank-Wolfe with top-K LP oracle and adaptive line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_f = f.copy()
    best_obj = M

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    current_K = K

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        g, conv, peaks = solve_lp_direction(f, M, current_K)
        if g is None:
            no_improve += 1
            if no_improve > 8:
                current_K = min(30, current_K + 5)
                no_improve = 0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 15:
                break
            continue

        # Line search with golden-section refinement
        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Refine around best_alpha
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            M = fast_convolve(f).max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                current_K = K
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 25:
            rng = np.random.default_rng((it * 31 + 7) % (2**31 - 1))
            scale = 0.003 + 0.004 * min(1.0, no_improve / 50.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max()
            no_improve = 0
            current_K = K

        if verbose and it % 50 == 0:
            print(f"    FW it={it}, obj={M:.10f}, best={best_obj:.10f}, K={current_K}")

    return best_f, best_obj


def subgrad_optimize(f0, time_limit, top_k=20, lr_init=0.02, verbose=False):
    """Projected subgradient with momentum and multi-peak weighting."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = 0.9
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        obj = conv.max()

        # Top-k peaks with harmonic weighting
        k_eff = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_eff)[-k_eff:]
        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(k_eff)
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.3
        ws = weights.sum()
        weights /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(k_eff):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * a[k - idx_arr]

        # Momentum step with backtracking
        v_new = mom * v - lr * sg
        best_candidate = a
        best_val = obj
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            candidate = project_simplex(a + scale * v_new)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                v = v_new
                break

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.95

        # Adapt
        if no_improve > 40:
            lr *= 0.93
            mom = min(0.97, mom + 0.003)
        if no_improve > 80:
            top_k = min(50, top_k + 5)
        if no_improve > 250:
            rng = np.random.default_rng((it * 48271 + 17) % (2**31 - 1))
            noise = 0.002 + 0.003 * min(1.0, (no_improve - 250) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init
            mom = 0.9
            top_k = 20
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    SubGrad it={it}, obj={fast_convolve(a).max():.10f}")

    return best_a, best_obj


def active_set_newton(a0, time_limit, max_active=15):
    """Active-set Newton using KKT conditions on binding constraints."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < 1000:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Find active peaks (within 1% of max)
        threshold = M * (1.0 - 0.002)
        active_peaks = np.where(conv >= threshold)[0]
        if len(active_peaks) == 0:
            break
        if len(active_peaks) > max_active:
            active_peaks = np.argpartition(conv, -max_active)[-max_active:]

        support = np.where(a > 1e-10)[0]
        if len(support) <= 2:
            break

        # Compute weighted gradient from active peaks
        p = len(active_peaks)
        # Use weights proportional to how close to max each peak is
        peak_vals = conv[active_peaks]
        w = peak_vals / peak_vals.sum()

        grad = np.zeros(n)
        for i in range(p):
            k = int(active_peaks[i])
            wi = w[i]
            if wi < 1e-14:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je + 1] += wi * a[k - je:k - js + 1][::-1]

        # Projected gradient descent with adaptive step
        best_candidate = a
        best_val = M
        steps = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]
        for step in steps:
            candidate = project_simplex(a - step * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
        else:
            # Try smaller steps or break
            improved = False
            for step in [0.0005, 0.001, 0.002]:
                candidate = project_simplex(a - step * grad)
                c = fast_convolve(candidate)
                val = c.max()
                if val < M - 1e-16:
                    a = candidate
                    M = val
                    if M < best_obj - 1e-15:
                        best_obj = M
                        best_a = a.copy()
                    improved = True
                    break
            if not improved:
                break

    return best_a, best_obj


def softmax_smoothed_grad(a0, time_limit, T_init=0.005, lr_init=0.01):
    """Smoothed gradient using softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init
    lr = lr_init

    v = np.zeros(n)
    mom = 0.92
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()

    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        # Softmax weights over all peaks
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient: sum_k p[k] * a[k-j] + sum_k p[k] * a[j-k]
        # = correlation + convolution terms
        term1 = np.correlate(p, a, mode='valid')  # sum_m p[j+m]*a[m]
        term2 = np.convolve(a, p[:n], 'full')[:n]  # sum_m a[m]*p[j-m] for m<=j
        grad = 2.0 * (term1 + term2)

        v = mom * v - lr * grad
        best_candidate = a
        best_val = conv.max()
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            candidate = project_simplex(a + scale * v)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.92

        if no_improve > 30:
            T *= 1.1
            lr *= 0.95
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31 - 1))
            noise = 0.003 + 0.003 * min(1.0, (no_improve - 200) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            v = np.zeros(n)

def window_init(n, kind='raised_cosine'):
    """Generate initial window sequences of length n."""
    j = np.arange(n, dtype=np.float64)
    if kind == 'uniform':
        return np.ones(n)
    elif kind == 'raised_cosine':
        a = np.sin(np.pi * (j + 0.5) / n) ** 2
    elif kind == 'hamming':
        a = 0.54 - 0.46 * np.cos(2 * np.pi * j / max(1, n - 1))
    elif kind == 'hanning':
        a = 0.5 - 0.5 * np.cos(2 * np.pi * j / max(1, n - 1))
    elif kind == 'kaiser_4':
        from scipy.signal import kaiser
        a = kaiser(n, 4.0)
    elif kind == 'kaiser_2':
        from scipy.signal import kaiser
        a = kaiser(n, 2.0)
    elif kind == 'kaiser_6':
        from scipy.signal import kaiser
        a = kaiser(n, 6.0)
    elif kind == 'trapezoid':
        ramp = max(1, n // 8)
        a = np.ones(n)
        a[:ramp] = np.linspace(0.3, 1.0, ramp)
        a[-ramp:] = np.linspace(1.0, 0.3, ramp)
    elif kind == 'bump':
        # Concentrated in middle
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
    else:
        return np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n)
    return a / s


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s

        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return (best_seq, best_score)
    return best_state

    print(f"  Phase 2 best: {state[1]:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 3: Subgradient + Smoothed gradient (250s) ===
    print("\n=== Phase 3: Subgradient Methods ===")
    t_start = time.monotonic()

    remaining = deadline - time.monotonic() - 500
    if remaining > 30:
        t_alloc = min(remaining, 100)
        print(f"  SubGrad top_k=20, t={t_alloc:.0f}s")
        a_opt, _ = subgrad_optimize(state[0], time_limit=t_alloc, top_k=20, lr_init=0.015)
        state = try_and_update(a_opt, state, "subgrad")

    remaining = deadline - time.monotonic() - 400
    if remaining > 20:
        t_alloc = min(remaining, 80)
        print(f"  SoftGrad T=0.005, t={t_alloc:.0f}s")
        a_opt, _ = softmax_smoothed_grad(state[0], time_limit=t_alloc, T_init=0.005, lr_init=0.008)
        state = try_and_update(a_opt, state, "softmax_sgrad")

    remaining = deadline - time.monotonic() - 350
    if remaining > 20:
        t_alloc = min(remaining, 60)
        print(f"  SoftGrad T=0.002, t={t_alloc:.0f}s")
        a_opt, _ = softmax_smoothed_grad(state[0], time_limit=t_alloc, T_init=0.002, lr_init=0.005)
        state = try_and_update(a_opt, state, "softmax_sgrad_T2")

    print(f"  Phase 3 best: {state[1]:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 4: Active-set Newton (150s) ===
    print("\n=== Phase 4: Active-Set Newton ===")
    t_start = time.monotonic()

    remaining = deadline - time.monotonic() - 300
    if remaining > 20:
        t_alloc = min(remaining, 120)
        print(f"  ActiveSet, t={t_alloc:.0f}s")
        a_opt, _ = active_set_newton(state[0], time_limit=t_alloc, max_active=15)
        state = try_and_update(a_opt, state, "active_set")

    remaining = deadline - time.monotonic() - 250
    if remaining > 15:
        t_alloc = min(remaining, 50)
        a_opt, _ = active_set_newton(state[0], time_limit=t_alloc, max_active=8)
        state = try_and_update(a_opt, state, "active_set_tight")

    print(f"  Phase 4 best: {state[1]:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Multi-n exploration (400s) ===
    print("\n=== Phase 5: Multi-n Exploration ===")
    t_start = time.monotonic()

    n_current = len(state[0])
    candidate_ns = sorted(set([n_current,
                                max(128, n_current // 2),
                                min(3000, n_current * 2),
                                512, 640, 896, 1024, 1280, 1536, 2048]))
    # Filter to reasonable range
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 2500]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 200
        if remaining < 25:
            break

        if n_try == n_current:
            start = state[0].copy()
        else:
            start = interp_sequence(state[0], n_try)

        t_alloc = min(remaining, 70)
        print(f"  n={n_try}, t={t_alloc:.0f}s")

        # FW
        a_opt, _ = fw_optimize(start, time_limit=t_alloc * 0.5, K=10)
        state = try_and_update(a_opt, state, f"fw_n{n_try}")

        # Quick active-set
        remaining = deadline - time.monotonic() - 200
        if remaining > 15:
            a_opt2, _ = active_set_newton(a_opt, time_limit=min(remaining, 20), max_active=10)
            state = try_and_update(a_opt2, state, f"as_n{n_try}")

    print(f"  Phase 5 best: {state[1]:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 6: Intensive local search (200s) ===
    print("\n=== Phase 6: Local Search ===")
    t_start = time.monotonic()

    n_best = len(state[0])

    # Perturbation + FW restarts
    for i in range(8):
        remaining = deadline - time.monotonic() - 60
        if remaining < 15:
            break
        noise_scale = 0.003 * (1 + i * 0.3)
        f_pert = project_simplex(state[0] + rng.standard_normal(n_best) * noise_scale)
        t_alloc = min(remaining, 35)
        a_opt, _ = fw_optimize(f_pert, time_limit=t_alloc, K=12)
        state = try_and_update(a_opt, state, f"pert_{i}")

    # Final active-set pass
    remaining = deadline - time.monotonic() - 40
    if remaining > 10:
        t_alloc = min(remaining, 40)
        a_opt, _ = active_set_newton(state[0], time_limit=t_alloc, max_active=10)
        state = try_and_update(a_opt, state, "final_as")

    print(f"  Phase 6 best: {state[1]:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 7: Final polishing (60s) ===
    print("\n=== Phase 7: Final Polish ===")
    t_start = time.monotonic()

    # Symmetry
    sym = (state[0] + state[0][::-1]) / 2.0
    sym = sym / sym.sum()
    state = try_and_update(sym, state, "final_sym")

    # One more FW
    remaining = deadline - time.monotonic() - 10
    if remaining > 10:
        a_opt, _ = fw_optimize(state[0], time_limit=min(remaining, 25), K=15)
        state = try_and_update(a_opt, state, "final_fw")

    # Active-set
    remaining = deadline - time.monotonic() - 5
    if remaining > 5:
        a_opt, _ = active_set_newton(state[0], time_limit=min(remaining, 10), max_active=5)
        state = try_and_update(a_opt, state, "final_as2")

    print(f"  Phase 7 best: {state[1]:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    best_seq, best_score = state

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    return [float(x) for x in best_seq]
