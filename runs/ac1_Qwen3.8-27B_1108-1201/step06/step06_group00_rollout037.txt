import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size].real

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient(f, peak_indices, weights):
    """Compute weighted gradient of b_k for k in peak_indices."""
    n = len(f)
    grad = np.zeros(n)
    for idx, k in enumerate(peak_indices):
        k = int(k)
        w = weights[idx]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += w * 2.0 * f[k - i_range]
    return grad

def hessian_matvec(f, v, peak_indices, weights, beta):
    """Compute H * v where H = beta * Cov(∇b_k; weights w).
    H*v = beta * (sum_k w_k * ((∇b_k)^T v) * ∇b_k - (sum_k w_k * ∇b_k)^T v * (sum_k w_k * ∇b_k))
    """
    n = len(f)
    # Precompute ∇b_k for each peak
    term1 = np.zeros(n)  # sum_k w_k * ((∇b_k)^T v) * ∇b_k
    for idx, k in enumerate(peak_indices):
        k = int(k)
        w = weights[idx]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        # ∇b_k[i] = 2 * f[k-i] for i in [lo, hi]
        grad_bk = 2.0 * f[k - i_range]
        # (∇b_k)^T v
        dot = dot_v = np.dot(grad_bk, v[i_range])
        term1[i_range] += w * dot_v * grad_bk
    
    # g = sum_k w_k * ∇b_k (the gradient of the smoothed objective)
    g = np.zeros(n)
    for idx, k in enumerate(peak_indices):
        k = int(k)
        w = weights[idx]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        g[i_range] += w * 2.0 * f[k - i_range]
    
    term2 = beta * np.dot(g, v) * g
    return beta * term1 - term2

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frank-Wolfe with full constraint set (all 2n-1 convolution peaks)
2. Highs LP solver for exact direction finding
3. Golden section / backtracking line search
4. Multi-length exploration

6. Final coordinate descent polish

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size].real

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    # Scale to sum=1, project, scale back
    v_scaled = v / target_sum
    u = np.sort(v_scaled)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v_scaled - theta, 0.0) * target_sum

def build_full_constraint_matrix(f):
    """Build the full (2n-1) x n constraint matrix for (f*g)_k <= M."""
    n = len(f)
    num_rows = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(num_rows):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    return csr_matrix((vals, (rows, cols)), shape=(num_rows, n))

                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

The away-step FW algorithm (Lan & Zhou, 2020; Beck et al., 2015):
1. Maintain a set of active indices S (where f_i > 0) and coefficients λ_s
2. Compute the FW direction g

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frank-Wolfe with FULL constraint set (all 2n-1 convolution peaks)
2. HiGHS LP solver for exact direction finding
3. Refined line search (backtracking + bisection)
4. Away-step FW variant for faster convergence
5. Subgradient refinement with LSE smoothing
6. Multi-length exploration and interpolation
7. Final coordinate-level polish

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size].real


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    if n == 1:
        val = max(float(v[0]), 0.0)
        return np.array([val])
    v_scaled = v / target_sum
    u = np.sort(v_scaled)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v_scaled - theta, 0.0) * target_sum


def build_constraint_matrix(f):
    """Build full (2n-1) x n matrix where row k is the constraint (f*g)_k <= M."""
    n = len(f)
    num_rows = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(num_rows):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
        vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(num_rows, n))


def solve_fw_direction(f, M, time_limit=10.0):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for ALL k, g >= 0."""
    n = len(f)
    A = build_constraint_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def solve_topk_direction(f, M, top_k=40, time_limit=10.0):
    """Solve LP with top-K constraints for efficiency on longer sequences."""
    n = len(f)
    conv = fast_convolve(f)
    k_sel = min(top_k, 2 * n - 1)
    peak_idx = np.argpartition(conv, -k_sel)[-k_sel:]
    # Always include the center
    center = n - 1
    if center not in peak_idx:
        peak_idx = np.append(peak_idx, center)
    peak_idx = np.sort(peak_idx)

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(peak_idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        rows.extend([idx] * len(js))
        cols.extend(js.tolist())
        vals.extend(f[k - js].tolist())

    A = csr_matrix((vals, (rows, cols)), shape=(len(peak_idx), n))
    c = -np.ones(n)
    b_ub = np.full(len(peak_idx), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def refined_line_search(f, g, target_sum, max_evals=30):
    """Find best alpha in [0,1] minimizing max(conv((1-alpha)f + alpha*g))."""
    n = len(f)
    best_alpha = 0.0

    conv_f = fast_convolve(f)
    best_val = conv_f.max()

    # Coarse backtracking
    alphas = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]
    candidates = []
    for alpha in alphas:

        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            continue
        conv_new = fast_convolve(f_new)
        val = conv_new.max()
        candidates.append((alpha, val))
        if val < best_val:
            best_val = val
            best_alpha = alpha

    if best_alpha == 0.0 or len(candidates) < 2:
        return best_alpha, best_val

    # Fine bisection around best

    best_cand = sorted_cands[0]

def subgradient_refine(f0, time_limit, beta_start=200.0, beta_end=10000.0, top_k=25):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n)

        f = f / s * n

    t0 = time.monotonic()

    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        conv = fast_convolve(f)
        cur_obj = conv.max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0

        # Top peaks with softmax weights
        k_top = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        bc = beta * conv[peaks]
        bc_max = bc.max()
        w = np.exp(bc - bc_max)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_top) / k_top
        else:
            w = w / w_sum

        # Gradient

        for idx, k in enumerate(peaks):
            k = int(k)
            w_k = w[idx]
            if w_k < 1e-15:
                continue

            i_range = np.arange(lo, hi + 1)
            grad[i_range] += w_k * 2.0 * f[k - i_range]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            no_improve += 100
            break
        grad_dir = grad / gnorm

        # Backtracking line search
        step = lr
        improved = False
        for _ in range(12):
            f_new = f - step * grad_dir
            f_new = project_simplex(f_new, target_sum=n)
            conv_new = fast_convolve(f_new)
            val = conv_new.max()
            if val < cur_obj - 1e-14:
                f = f_new
                improved = True
                no_improve = 0
                lr = min(lr * 1.005, 2.0)

            no_improve += 1
            lr *= 0.92

    n = len(f)
    if s < 1e-15:
        f = np.ones(n)
    else:
        f = f / s * n

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    # Get top peaks
    top_k = 10
    peaks = np.argpartition(conv, -top_k)[-top_k:]

    for _ in range(2000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        cur_obj = conv.max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()

        # For each top peak, find the coordinate that most contributes
        for k in peaks:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo >= hi:
                continue
            i_range = np.arange(lo, hi + 1)
            contribs = f[i_range] * f[k - i_range]
            j_max = lo + np.argmax(contribs)
            partner = k - j_max
            if partner < 0 or partner >= n or partner == j_max:
                continue

            # Try moving mass from j_max to several candidate destinations
            # Choose destinations that are "far" from the peak
            dist_to_peak_right = min(n - 1 - j_max, abs(j_max - k))
            dist_to_peak_left = min(j_max, abs(j_max - k))

            # Candidate destinations: random positions
            rng = np.random.default_rng(hash((k, j_max, int(cur_obj * 1e6))) % (2**31))
            n_try = 5
            for _ in range(n_try):
                j_dest = rng.integers(0, n)
                if j_dest == j_max or j_dest == partner:
                    continue
                # Check: would this increase the peak at k?
                # New contribution at k: (f[j_max] - d) * f[partner] + f[j_dest] * f[k-j_dest] (if valid)
                delta = 0.05 * f[j_max]
                if delta < 1e-15:
                    continue
                f_trial = f.copy()
                f_trial[j_max] -= delta
                f_trial[j_dest] += delta
                if f_trial[j_max] < 0:
                    continue
                f_trial = np.maximum(f_trial, 0.0)
                f_trial = f_trial / f_trial.sum() * n
                conv_trial = fast_convolve(f_trial)
                if conv_trial.max() < cur_obj - 1e-13:
                    f = f_trial
                    break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-15:
        f_new = f_new / s
    else:
        f_new = np.ones(new_n)
    return f_new


def evaluate_candidate(a, best_score_holder):
    """Evaluate a sequence and update best if improved. Returns score."""
    a = np.clip(a, 0.0, 1000.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_score_holder[0]:
        best_score_holder[0] = score
        best_score_holder[1] = a.copy()
        print(f"  *** NEW BEST: {score:.10f} (n={len(a)}) ***")
    return score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function using full-constraint Frank-Wolfe with refined line search,
    subgradient polishing, and away-step refinement.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    best_score_holder = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            score = evaluate_candidate(parent, best_score_holder)
            print(f"Parent loaded: n={len(parent)}, score={score:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768)
        score = evaluate_candidate(parent, best_score_holder)
        print(f"Uniform fallback: n=768, score={score:.10f}")

    n_parent = len(parent)
    print(f"\nStarting optimization. Parent n={n_parent}, current best={best_score_holder[0]:.10f}")

    # === Phase 1: Quick multi-length exploration (300s) ===
    print("\n=== Phase 1: Multi-length FW exploration ===")
    t1_start = time.monotonic()
    t1_budget = 300.0

    # Candidate lengths centered around parent
    candidate_lengths = sorted(set([
        n_parent,
        max(256, n_parent - 128),
        min(1500, n_parent + 128),
        max(256, n_parent - 256),
        min(1500, n_parent + 256),
        max(256, int(n_parent * 0.8)),
        min(1500, int(n_parent * 1.2)),
    ]))

    # Sort by distance from parent (try closest first)
    candidate_lengths.sort(key=lambda x: abs(x - n_parent))

    for nc in candidate_lengths[:5]:
        t_elap = time.monotonic() - t1_start
        if t_elap >= t1_budget - 30:
            break

        t_alloc = min(80.0, (t1_budget - t_elap) * 0.45)
        if t_alloc < 20:
            continue

        if nc == n_parent:
            f_init = parent.copy()
        else:
            f_init = interpolate_sequence(parent, nc)

        print(f"  FW n={nc}, time={t_alloc:.0f}s")
        f_opt, obj = fw_optimize_full(f_init, time_limit=t_alloc,
                                       use_full_constraints=(nc <= 1000), verbose=False)
        evaluate_candidate(f_opt, best_score_holder)

    # === Phase 2: Focused optimization on best length (600s) ===
    print(f"\n=== Phase 2: Focused FW on best (n={len(best_score_holder[1])}) ===")
    t2_start = time.monotonic()
    t2_budget = 600.0

    best_f = best_score_holder[1].copy()
    n_best = len(best_f)

    # Run FW in chunks to allow progress reporting
    chunk_size = 120.0
    chunks = 0
    while time.monotonic() - t2_start < t2_budget:
        t_rem = t2_budget - (time.monotonic() - t2_start)
        if t_rem < 30:
            break
        t_alloc = min(chunk_size, t_rem)
        chunks += 1
        print(f"  Chunk {chunks}: t={t_alloc:.0f}s")
        f_opt, obj = fw_optimize_full(best_f, time_limit=t_alloc,
                                       use_full_constraints=(n_best <= 1000), verbose=True)
        sc = evaluate_candidate(f_opt, best_score_holder)
        if best_score_holder[1] is not None:
            best_f = best_score_holder[1].copy()
            n_best = len(best_f)

    # === Phase 3: Subgradient refinement (400s) ===
    print(f"\n=== Phase 3: Subgradient refinement ===")
    t3_start = time.monotonic()
    t3_budget = 400.0

    best_f = best_score_holder[1].copy()
    n_best = len(best_f)
    t_rem = t3_budget - (time.monotonic() - t3_start)
    if t_rem > 30:
        t_alloc = min(200.0, t_rem)
        print(f"  Subgrad n={n_best}, beta=200->5000, t={t_alloc:.0f}s")
        f_opt, _ = subgradient_refine(best_f, time_limit=t_alloc,
                                       beta_start=200.0, beta_end=5000.0, top_k=25)
        evaluate_candidate(f_opt, best_score_holder)

    # Second subgrad round with higher beta
    best_f = best_score_holder[1].copy()
    t_rem = t3_budget - (time.monotonic() - t3_start)
    if t_rem > 30:
        t_alloc = min(150.0, t_rem)
        print(f"  Subgrad n={len(best_f)}, beta=2000->20000, t={t_alloc:.0f}s")
        f_opt, _ = subgradient_refine(best_f, time_limit=t_alloc,
                                       beta_start=2000.0, beta_end=20000.0, top_k=15)
        evaluate_candidate(f_opt, best_score_holder)

    # === Phase 4: Away-step + alternating polish (400s) ===
    print(f"\n=== Phase 4: Away-step and alternating polish ===")
    t4_start = time.monotonic()
    t4_budget = 400.0

    it = 0
    while time.monotonic() - t4_start < t4_budget:
        t_rem = t4_budget - (time.monotonic() - t4_start)
        if t_rem < 25:
            break

        best_f = best_score_holder[1].copy()
        n_cur = len(best_f)

        it += 1
        if it % 3 == 1:
            # FW step
            t_alloc = min(80.0, t_rem * 0.4)
            if t_alloc > 15:
                print(f"  Alt {it}: FW n={n_cur}, t={t_alloc:.0f}s")
                f_opt, _ = fw_optimize_full(best_f, time_limit=t_alloc,
                                             use_full_constraints=(n_cur <= 1000),
                                             verbose=False)
                evaluate_candidate(f_opt, best_score_holder)

        elif it % 3 == 2:
            # Subgradient step
            t_alloc = min(70.0, t_rem * 0.4)
            if t_alloc > 15:
                beta_s = 500.0 + 200.0 * it
                print(f"  Alt {it}: Subgrad n={n_cur}, beta={beta_s:.0f}, t={t_alloc:.0f}s")
                f_opt, _ = subgradient_refine(best_f, time_limit=t_alloc,
                                               beta_start=beta_s, beta_end=beta_s * 20,
                                               top_k=15)
                evaluate_candidate(f_opt, best_score_holder)

        else:
            # Away-step refinement
            t_alloc = min(50.0, t_rem * 0.3)
            if t_alloc > 10:
                print(f"  Alt {it}: AwayStep n={n_cur}, t={t_alloc:.0f}s")
                f_opt, _ = away_step_refine(best_f, time_limit=t_alloc)
                evaluate_candidate(f_opt, best_score_holder)

    # === Phase 5: Final ultra-precise polish (remaining time) ===
    print(f"\n=== Phase 5: Final polish ===")
    t5_start = time.monotonic()
    t5_budget = max(30.0, deadline - time.monotonic())

    best_f = best_score_holder[1].copy()
    n_best = len(best_f)

    # Very high beta subgradient (focus on top 5 peaks)
    t_rem = t5_budget - (time.monotonic() - t5_start)
    if t_rem > 20:
        t_alloc = min(80.0, t_rem)
        print(f"  High-beta subgrad n={n_best}, beta=5000->50000, t={t_alloc:.0f}s")
        f_opt, _ = subgradient_refine(best_f, time_limit=t_alloc,
                                       beta_start=5000.0, beta_end=50000.0, top_k=8)
        evaluate_candidate(f_opt, best_score_holder)

    # Final FW with full constraints (if possible)
    best_f = best_score_holder[1].copy()
    t_rem = t5_budget - (time.monotonic() - t5_start)
    if t_rem > 20 and len(best_f) <= 1000:
        t_alloc = min(70.0, t_rem)
        print(f"  Final FW-full n={len(best_f)}, t={t_alloc:.0f}s")
        f_opt, _ = fw_optimize_full(best_f, time_limit=t_alloc,
                                     use_full_constraints=True, verbose=True)
        evaluate_candidate(f_opt, best_score_holder)

    # Tiny perturbations
    best_f = best_score_holder[1].copy()
    t_rem = deadline - time.monotonic()
    if t_rem > 10:
        print(f"  Micro-perturbations, t={t_rem:.0f}s")
        for _ in range(20):
            if time.monotonic() > deadline - 3:
                break
            n_cur = len(best_f)
            f_pert = project_simplex(best_f + rng.standard_normal(n_cur) * 0.0005 * n_cur,
                                      target_sum=n_cur)
            evaluate_candidate(f_pert, best_score_holder)
            if best_score_holder[1] is not None:
                best_f = best_score_holder[1].copy()

    # === Final result ===
    if best_score_holder[1] is not None:
        final = np.clip(best_score_holder[1], 0.0, 1000.0)
        final_score = float(evaluate_sequence(final.tolist()))
        print(f"\n{'='*50}")
        print(f"FINAL RESULT: score={final_score:.10f}, n={len(final)}")
        print(f"{'='*50}")
        return [float(x) for x in final]

    # Fallback
    print("WARNING: Using uniform fallback")
    return [float(x) for x in np.ones(768)]
