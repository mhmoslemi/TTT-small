**Subgradient descent with smoothed max:**

"""
Optimized subgradient descent for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous:
- Adaptive temperature schedule based on convergence progress
- Symmetry exploitation (a[i] = a[n-1-i])
- Multi-scale optimization (coarse-to-fine)
- Peak-equalization heuristic
- Nesterov momentum with adaptive step size
- Diverse initialization including optimized bump functions
"""

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad(a, T):
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return grad, conv

def subgradient_descent(a0, time_limit, T_init=0.001, lr_init=0.05, 
                        momentum=0.9, use_symmetry=True, 
                        T_decay=0.99997, lr_decay=0.99995,
                        T_min=1e-6, restart_interval=500):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    T = T_init
    lr = lr_init
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    prev_obj = best_obj
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        
        grad, conv = compute_grad(a, T)
        obj = conv.max()
        
        # Nesterov momentum step
        v = momentum * v - lr * grad
        a_new = a + v
        
        if use_symmetry:
            a_new = (a_new + a_new[::-1]) / 2.0
        
        a_new = project_simplex(a_new)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            # Slightly increase lr
            lr = min(lr * 1.001, lr_init * 2)
        else:
            no_improve += 1
            # Decrease lr
            lr *= 0.98
            # Don't update a if worse
            a_new = a  # reject the step
            v = v * 0.5  # dampen momentum
        
        a = a_new
        
        # Decay temperature
        T = max(T * T_decay, T_min)
        
        # Restart if stuck
        if no_improve > restart_interval:
            rng = np.random.default_rng((it * 7919) % (2**31))
            noise_scale = 0.001 + 0.003 * min(1.0, no_improve / (2 * restart_interval))
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            if use_symmetry:
                a = (a + a[::-1]) / 2.0
                a = project_simplex(a)
            v = np.zeros(n)
            T = T_init * 0.5
            lr = lr_init * 0.7
            no_improve = 0
    
    return best_a, best_obj

def peak_equalization(a0, time_limit, top_k=5):
    """Try to equalize the top-k peaks of the convolution."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    lr = 0.02
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        
        conv = fast_convolve(a)
        
        # Find top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peak_idx = np.argpartition(conv, -top_k)[-top_k:]
            # Weight by inverse rank (top peak gets highest weight)
            sorted_peaks = conv[peak_idx][np.argsort(conv[peak_idx])[::-1]]
            ranks = np.argsort(conv[peak_idx])[::-1]
            weights = 1.0 / (ranks + 1)
            weights /= weights.sum()
            peaks = peak_idx
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for k, w in zip(peaks, weights):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += w * 2.0 * a[k - idx]
        
        # Gradient step
        a_new = a - lr * grad
        a_new = (a_new + a_new[::-1]) / 2.0
        a_new = project_simplex(a_new)
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, 0.1)
        else:
            lr *= 0.95
            if lr < 1e-6:
                break
    
    return best_a, best_obj

def bump_initialization(n, width_frac=0.5, sharpness=2.0):
    """Generate a smooth bump initialization."""
    x = np.arange(n, dtype=np.float64)
    center = 0.5 * (n - 1)
    width = width_frac * n
    a = np.exp(-sharpness * ((x - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a

def multi_scale_optimize(a0, time_limit, use_symmetry=True):
    """Optimize at half resolution, then refine at full resolution."""
    n = len(a0)
    half_n = n // 2
    
    # Downsample
    a_half = np.zeros(half_n)
    for i in range(half_n):
        a_half[i] = a0[2*i] + a0[2*i+1] if 2*i+1 < n else a0[2*i]
    a_half = np.maximum(a_half, 0)
    a_half = a_half / a_half.sum()
    
    # Optimize at half resolution
    a_half_opt, _ = subgradient_descent(a_half, time_limit=time_limit*0.3,
                                         T_init=0.005, lr_init=0.08,
                                         use_symmetry=use_symmetry)
    
    # Upsample by interpolation
    a_full = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, half_n), a_half_opt)
    a_full = np.maximum(a_full, 0)
    a_full = a_full / a_full.sum()
    
    # Refine at full resolution
    a_final, _ = subgradient_descent(a_full, time_limit=time_limit*0.7,
                                       T_init=0.001, lr_init=0.03,
                                       use_symmetry=use_symmetry)
    
    return a_final

def coordinate_descent_polish(a0, time_limit):
    """Polish using coordinate descent on the most impactful coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        
        # Find the coordinate that contributes most to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        
        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]
        
        # Find a coordinate to transfer to (lowest value, outside the affected region)
        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            outside = np.arange(n)
        j_min = outside[np.argmin(a[outside])]
        
        improved = False
        for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
            delta = frac * a[j_max]
            if delta < 1e-10:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min] += delta
            candidate = np.maximum(candidate, 0)
            candidate[j_min] += candidate[j_max] if candidate[j_max] < 0 else 0
            candidate[j_max] = max(candidate[j_max], 0)
            s = candidate.sum()
            if s < 1e-15:
                continue
            candidate /= s
            
            obj = fast_convolve(candidate).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = candidate.copy()
                a = candidate
                improved = True
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                # Random perturbation
                rng = np.random.default_rng((it * 31 + 7) % (2**31))
                candidate = project_simplex(a + rng.standard_normal(n) * 0.002)
                obj = fast_convolve(candidate).max()
                if obj < best_obj - 1e-15:
                    a = candidate
                    best_a = candidate.copy()
                    best_obj = obj
                    no_improve = 0
                elif no_improve > 300:
                    break
    
    return best_a, best_obj
