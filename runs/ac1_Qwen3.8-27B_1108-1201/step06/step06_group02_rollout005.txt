Optimized search for minimum peak autocorrelation non-negative sequences.
Uses U-shaped initializations, multiplicative weight descent, peak-targeted
subgradient methods, and coordinate descent polishing.

import numpy as np

import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgrad_smoothed(a, T, top_k=None):
    """Compute smoothed subgradient of max_k (a*a)[k]."""
    n = len(a)
    conv = fast_convolve(a)
    size = len(conv)
    
    if top_k is not None and top_k < size:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        w = np.exp(pv / T - pv.max() / T)
        w = w / w.sum()
        peaks = peaks[np.argsort(pv)]  # sort by value
        w = w[np.argsort(pv)]
    else:
        peaks = np.arange(size)
        w = np.exp(conv / T - conv.max() / T)
        w = w / w.sum()
    
    g = np.zeros(n)
    for k, wk in zip(peaks, w):
        if wk < 1e-15:
            continue
        k = int(k)
        # g_j += 2 * wk * a[k-j] for valid j
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            g[j] += 2.0 * wk * a[k - j]
    
    return g, conv.max()

        # So it's a[k-js : k-je-1 : -1] reversed... 
        # Actually: for j from js to je, k-j goes from k-js to k-je (decreasing)

        # That's a[je... no.

        start = k - je
        end = k - js
        g[js:je+1] += 2.0 * wk * a[start:end+1][::-1]

def projected_gradient_descent(a0, time_limit, lr0=0.05, T0=0.01, top_k=50):

def coordinate_descent_polish(a0, time_limit):

def u_shaped_init(n, shape='cosine'):
    """Generate U-shaped initialization."""
    x = np.arange(n, dtype=np.float64)
    if shape == 'cosine':
        a = 1.0 - np.cos(np.pi * x / (n - 1))
    elif shape == 'gaussian_edges':
        a = np.exp(-0.5 * (x / (0.2 * n))**2) + np.exp(-0.5 * ((x - (n-1)) / (0.2 * n))**2)
    elif shape == 'power':
        d = np.minimum(x, n - 1 - x) / (n / 2.0)
        a = d ** 0.5
    elif shape == 'v':
        d = np.minimum(x, n - 1 - x) / (n / 2.0)
        a = d
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()
