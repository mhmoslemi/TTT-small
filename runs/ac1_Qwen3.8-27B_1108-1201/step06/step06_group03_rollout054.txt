import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (f*g) = A @ g."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M, n=None):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    if n is None:
        n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle, away-step, and extensive line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    # Away-step tracking
    support = set(range(n))
    away_f = f.copy()

    t0 = time.monotonic()
    stagnant = 0
    it = 0
    m_relax = 1.001

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.3:
            break

        # Solve LP with current M
        g = solve_lp_oracle(f, M * m_relax, n)

        if g is None or g.sum() < 1e-15:
            g = solve_lp_oracle(f, M * 1.005, n)

        if g is None or g.sum() < 1e-15:
            stagnant += 1
            if stagnant > 15:
                rng = np.random.default_rng(it * 7919 + 3)
                scale = 0.002 + 0.003 * min(1.0, stagnant / 40.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.001
                stagnant = 0
                support = set(range(n))
            continue

        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-10:
            stagnant += 1
            if stagnant > 25:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue

        # Extensive line search with away-step option
        best_alpha = 0.0
        best_val = M + 1e-14
        best_is_away = False
        best_beta = 0.0

        # Standard FW step
        for alpha in np.linspace(0.02, 1.0, 25):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
                best_is_away = False

        # Away-step: move away from away_f
        if np.max(away_f) > 0.05:
            for beta in np.linspace(0.1, 0.8, 12):
                ft = f + beta * (f - away_f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = beta
                    best_is_away = True
                    best_beta = beta

        # Fine refinement
        if best_alpha > 0.01 and not best_is_away:
            lo = max(0.0, best_alpha - 0.02)
            hi = min(1.0, best_alpha + 0.02)
            for alpha in np.linspace(lo, hi, 12):
                ft = (1.0 - alpha) * f + alpha * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
                    best_is_away = False

        if best_alpha > 1e-8:
            if best_is_away:
                f = f + best_beta * (f - away_f)
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                # Update away point
                away_f = g.copy()
            else:
                f = (1.0 - best_alpha) * f + best_alpha * g
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                away_f = g.copy()

            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.001

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        if stagnant > 8:
            m_relax = min(m_relax * 1.04, 1.04)

        if verbose and it % 50 == 0:
            print(f"    FW it={it}, obj={best_obj:.10f}, stag={stagnant}")

    return best_f, best_obj


def mirror_descent(a0, time_limit, T_init=0.0003, lr_init=0.01):
    """Mirror descent with exponential family updates and adaptive temperature."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    stagnant = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        n_conv = len(conv)

        # Find top peaks
        top_k = min(5, n_conv)
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peak_idx]
        pv_max = pv.max()
        w = np.exp((pv - pv_max) / (T * 0.05))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(peak_idx)) / len(peak_idx)
        else:
            w = w / w_sum

        # Compute weighted subgradient
        grad = np.zeros(n)
        for i in range(len(peak_idx)):
            if w[i] < 1e-14:
                continue
            k = int(peak_idx[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += w[i] * 2.0 * a[k - j]

        # Exponential family update: p_new[i] ∝ p[i] * exp(-lr * grad[i] / T)
        log_a = np.log(a + 1e-30)
        update = log_a - (lr / T) * grad
        update -= update.max()  # numerical stability
        a_new = np.exp(update)
        a_new = a_new / a_new.sum()

        # Also try the projected gradient version
        a_proj = project_simplex(a - lr * grad / (T + 1e-10))

        # Choose the better one
        val_new = fast_convolve(a_new).max()
        val_proj = fast_convolve(a_proj).max()
        val_cur = conv.max()

        if val_new < val_proj and val_new < val_cur:
            a = a_new
        elif val_proj < val_cur:
            a = a_proj
        else:
            # No improvement, try smaller step
            for frac in [0.5, 0.25, 0.1]:
                a_try = a_proj if val_proj < val_new else a_new
                a_small = (1 - frac) * a + frac * a_try
                a_small = np.maximum(a_small, 0)
                a_small = a_small / a_small.sum()
                val_small = fast_convolve(a_small).max()
                if val_small < val_cur:
                    a = a_small
                    break

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        # Adapt
        if stagnant > 15:
            T *= 1.08
            lr *= 0.95
        if stagnant > 50:
            T *= 0.5
            lr *= 1.5
        if stagnant > 150:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            stagnant = 0

    return best_a, best_obj


def spectral_shaping(a0, time_limit):
    """
    Use FFT to identify frequencies that contribute most to the peak autocorrelation,
    then apply targeted adjustments to reduce spectral energy at those frequencies.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    step = 0.003

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        # Find the global peak
        k_peak = int(np.argmax(conv))

        # Compute the "excess" spectrum at the peak lag
        # R[k] = (1/N) sum_m |P(m)|^2 * exp(2*pi*i*m*k/N)
        # The "bad" frequencies are those where cos(2*pi*m*k/N) > 0
        # We want to reduce |P(m)|^2 at those frequencies
        fft_size = 1 << (2 * n - 1).bit_length()
        P = np.fft.rfft(a, fft_size)
        S = np.abs(P) ** 2

        # Identify frequencies that make R[k_peak] large
        m_vals = np.arange(len(S))
        phase = np.cos(2 * np.pi * m_vals * k_peak / fft_size)
        # Bad frequencies: phase > 0 (they add positively to R[k_peak])
        bad_mask = phase > 0.3
        bad_mask[0] = False  # Don't touch DC

        # Perturb: reduce a[i] where it contributes to bad frequencies
        # Gradient of sum_{m:bad} S[m] w.r.t. a[i]
        grad = np.zeros(n)
        bad_freqs = np.where(bad_mask)[0]
        if len(bad_freqs) > 0:
            # For each bad frequency m, d|P(m)|^2/da[i] = 2*Re(P(m) * exp(2*pi*i*m*i/fft_size))
            # Use FFT to compute efficiently
            phase_vec = np.exp(2j * np.pi * np.outer(np.arange(n), bad_freqs) / fft_size)
            P_bad = P[bad_freqs]
            contrib = 2.0 * np.real(P_bad[:, None] * phase_vec)
            grad = contrib.sum(axis=0)

        # Move a in direction that reduces bad spectral energy
        # We want to DECREASE a where grad is positive (reduce contribution to bad freqs)
        direction = -grad
        # Normalize
        d_norm = np.linalg.norm(direction)
        if d_norm > 1e-15:
            direction = direction / d_norm

        a_new = project_simplex(a + step * direction)
        val_new = fast_convolve(a_new).max()

        if val_new < best_obj - 1e-15:
            a = a_new
            best_obj = val_new
            best_a = a.copy()
        else:
            # Try opposite direction
            a_new2 = project_simplex(a - step * direction)
            val_new2 = fast_convolve(a_new2).max()
            if val_new2 < best_obj - 1e-15:
                a = a_new2
                best_obj = val_new2
                best_a = a.copy()
            else:
                step *= 0.92
                if step < 1e-7:
                    break

        # Periodic annealing: add small random perturbation
        if it % 100 == 0:
            rng = np.random.default_rng(it * 48271)
            a_pert = project_simplex(a + rng.standard_normal(n) * step * 0.5)
            val_pert = fast_convolve(a_pert).max()
            if val_pert < fast_convolve(a).max():
                a = a_pert

    return best_a, best_obj


def peak_mass_transfer(a0, time_limit):
    """
    For each peak in the autocorrelation, try to reduce it by transferring mass
    from contributing elements to non-contributing positions.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    step_scale = 0.004
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        n_peaks = min(6, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        improved = False
        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)

            if js > je:
                continue

            # Find top contributing pairs
            contributions = []
            for j in range(js, je + 1):
                c = a[j] * a[k - j]
                contributions.append((c, j, k - j))
            contributions.sort(reverse=True)

            for (c, j1, j2) in contributions[:8]:
                if a[j1] < 1e-9 and a[j2] < 1e-9:
                    continue

                # Try moving mass from the larger contributor to a random distant position
                src = j1 if a[j1] >= a[j2] else j2
                if a[src] < 1e-9:
                    continue

                # Choose target: prefer positions far from both j1 and j2
                # and also far from the peak lag k
                candidates = []
                for _ in range(10):
                    rng_t = np.random.default_rng((it * 9973 + k * 31 + src * 7 + _ * 13) % (2**31))
                    tgt = int(rng_t.integers(0, n))
                    dist1 = abs(tgt - j1)
                    dist2 = abs(tgt - j2)
                    dist_peak = min(abs(tgt - k), abs(2 * n - 2 - k - tgt), n)
                    candidates.append((min(dist1, dist2, dist_peak), tgt))

                candidates.sort(reverse=True)

                for (_, tgt) in candidates[:3]:
                    if tgt == src or tgt == j1 or tgt == j2:
                        continue
                    for frac in [step_scale, step_scale * 0.4, step_scale * 2.0]:
                        delta = a[src] * frac
                        candidate = a.copy()
                        candidate[src] -= delta
                        candidate[tgt] += delta
                        if candidate[src] < -1e-12:
                            continue
                        candidate = np.maximum(candidate, 0)
                        candidate = candidate / candidate.sum()
                        val = fast_convolve(candidate).max()
                        if val < best_obj - 1e-16:
                            a = candidate
                            best_obj = val
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            step_scale *= 0.94
            if step_scale < 1e-7:
                break
            if it % 200 == 0:
                rng = np.random.default_rng(it * 48271)
                candidate = project_simplex(a + rng.standard_normal(n) * step_scale * 3)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
            elif it > 3000:
                break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def spectral_init(n, freq_frac, amplitude, rng):
    """
    Create an initial sequence with controlled spectral shape.
    freq_frac: fundamental frequency as fraction of n
    amplitude: modulation depth
    """
    x = np.arange(n, dtype=np.float64)
    # Base + cosine modulation to create spectral peaks at specific frequencies
    p = 1.0 + amplitude * np.cos(2 * np.pi * freq_frac * x)
    # Add small random perturbation
    p += rng.standard_normal(n) * amplitude * 0.1
    p = np.maximum(p, 0.01)
    p = p / p.sum()
    return p


def generate_diverse_starts(n, parent, rng):
    """Generate diverse initializations."""
    starts = []

    # Parent resampled
    if parent is not None:
        starts.append(resample_sequence(parent, n))

    # Uniform
    starts.append(np.ones(n) / n)

    # Spectral designs targeting negative autocorrelation at center
    for freq_frac in [1.0/3, 1.0/4, 2.0/5, 1.0/5, 0.3]:
        for amp in [0.3, 0.5, 0.7]:
            starts.append(spectral_init(n, freq_frac, amp, rng))

    # Multi-frequency
    for f1, f2, a1, a2 in [(1/3, 2/5, 0.4, 0.3), (1/4, 1/3, 0.3, 0.4), (0.3, 0.7, 0.3, 0.3)]:
        x = np.arange(n, dtype=np.float64)
        p = 1.0 + a1 * np.cos(2*np.pi*f1*x) + a2 * np.cos(2*np.pi*f2*x)
        p = np.maximum(p, 0.01)
        starts.append(p / p.sum())

    # Edge concentration
    x = np.arange(n, dtype=np.float64)
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    starts.append(edge / edge.sum())

    # Center concentration
    center = np.exp(-0.5 * ((x - (n-1)/2) / (0.15*n))**2)
    starts.append(center / center.sum())

    # Random
    for _ in range(2):
        starts.append(rng.dirichlet(np.ones(n) * 0.5))

    return starts


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search for minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")

    n_parent = len(parent)

    # === Phase 1: Spectral initialization + quick optimization (250s) ===
    print("\n=== Phase 1: Spectral Init + Quick Opt ===")
    t_phase = time.monotonic()

    lengths = sorted(set([n_parent, max(256, n_parent - 64), n_parent + 64, 640, 768, 896, 1024]))
    lengths = [l for l in lengths if 192 <= l <= 1200]

    for n_try in lengths[:5]:
        remaining = deadline - time.monotonic() - 1000
        if remaining < 20:
            break

        starts = generate_diverse_starts(n_try, parent, rng)

        for si, start in enumerate(starts[:6]):
            remaining = deadline - time.monotonic() - 950
            if remaining < 8:
                break
            t_alloc = min(remaining, 15)
            # Quick mirror descent
            a_opt, _ = mirror_descent(start, time_limit=t_alloc, T_init=0.001, lr_init=0.02)
            label = f"md_n{n_try}_s{si}"
            try_seq(a_opt, label)

    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW with LP oracle (700s) ===
    print("\n=== Phase 2: Deep FW ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Main deep FW
        remaining = deadline - time.monotonic() - 550
        if remaining > 30:
            t_alloc = min(remaining, 450)
            print(f"  Deep FW n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_fw")

        # Also try FW from spectral starts
        starts2 = generate_diverse_starts(n_best, best_seq, rng)
        for si, start in enumerate(starts2[1:5]):  # Skip parent
            remaining = deadline - time.monotonic() - 300
            if remaining < 20:
                break
            t_alloc = min(remaining, 60)
            print(f"  FW spectral start {si}, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(start, time_limit=t_alloc)
            try_seq(a_opt, f"fw_spec_{si}")

        # Perturbed restarts
        for scale in [0.002, 0.005, 0.01]:
            remaining = deadline - time.monotonic() - 200
            if remaining < 20:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 40)
            a_opt, _ = fw_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"fw_pert_{scale}")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Spectral refinement (300s) ===
    print("\n=== Phase 3: Spectral Shaping ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 300
        if remaining > 30:
            t_alloc = min(remaining, 150)
            print(f"  Spectral shaping, t={t_alloc:.0f}s")
            a_opt, _ = spectral_shaping(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "spectral")

        # Mirror descent from best
        remaining = deadline - time.monotonic() - 150
        if remaining > 20:
            t_alloc = min(remaining, 100)
            print(f"  Mirror descent, t={t_alloc:.0f}s")
            a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc, T_init=0.0005, lr_init=0.01)
            try_seq(a_opt, "mirror")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Peak-targeted mass transfer (200s) ===
    print("\n=== Phase 4: Peak Mass Transfer ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 100
        if remaining > 20:
            t_alloc = min(remaining, 150)
            print(f"  Peak mass transfer, t={t_alloc:.0f}s")
            a_opt, _ = peak_mass_transfer(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_mt")

    # === Phase 5: Final polish (remaining) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Short FW rounds with tiny perturbations
        for ri in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            noise = rng.standard_normal(n_best) * 0.0005
            start = project_simplex(best_seq + noise)
            t_alloc = min(remaining, 20)
            a_opt, _ = fw_optimize(start, time_limit=t_alloc)
            try_seq(a_opt, f"polish_fw_{ri}")

        # Final peak transfer
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_alloc = min(remaining, 30)
            a_opt, _ = peak_mass_transfer(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_peak")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Try nearby lengths for the best sequence ===
    print("\n=== Phase 6: Length Tuning ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        for delta in [-32, -16, 16, 32, -64, 64]:
            n_new = n_best + delta
            if n_new < 128 or n_new > 2048:
                continue
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            a_resampled = resample_sequence(best_seq, n_new)
            t_alloc = min(remaining, 25)
            a_opt, _ = fw_optimize(a_resampled, time_limit=t_alloc)
            try_seq(a_opt, f"len_tune_{n_new}")

    print(f"  Phase 6 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
