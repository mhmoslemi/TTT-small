def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (f*g) = A @ g."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def subgradient_optimize(a0, time_limit, T_init=0.0005, lr_init=0.02, top_k_frac=0.05):

    a = np.maximum(a0.astype(np.float64), 0.0)

        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(5, int(n * top_k_frac))

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.92

    t0 = time.monotonic()
    stagnant = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        # Top-k peaks with softmax weighting
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))

        pv = conv[peaks]
        pv_max = pv.max()
        w = np.exp((pv - pv_max) / (T * 0.1))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / w_sum

        # Compute weighted subgradient
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += w[i] * 2.0 * a[k - idx_arr]

        # Momentum + backtracking
        v = mom * v - lr * grad
        best_candidate = a
        best_val = conv.max()

        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            candidate = project_simplex(a + frac * v)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        # Adapt hyperparameters
        if stagnant > 20:
            T *= 1.06
            lr *= 0.97
        if stagnant > 60:
            T *= 0.6
            lr *= 1.3
            mom = 0.95

            lr = lr_init
            v = np.zeros(n)

def peak_coordinate_descent(a0, time_limit):

        conv = fast_convolve(a)
        n_peaks = min(4, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        improved = False
        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)

            if js > je:
                continue

            # Top contributing pairs for this peak
            pairs = []
            for j in range(js, je + 1):
                pairs.append((a[j] * a[k - j], j, k - j))
            pairs.sort(reverse=True)

            # Try top few pairs

                    continue

                            candidate = candidate / candidate.sum()
                            val = fast_convolve(candidate).max()
                            if val < best_obj - 1e-15:
                                a = candidate
                                best_obj = val

def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_starts(n, parent, rng):
    """Generate diverse initializations for FW."""
    starts = []

    # Parent (resampled)
    if parent is not None:
        starts.append(resample_sequence(parent, n))

    # Uniform
    starts.append(np.ones(n) / n)

    # Cosine taper
    x = np.arange(n, dtype=np.float64)
    cos_t = np.cos(np.pi * x / (n - 1))
    w = 0.5 * (1 + cos_t)
    w = np.maximum(w, 0.01)
    starts.append(w / w.sum())

    # Gaussian
    g = np.exp(-0.5 * ((x - (n - 1) / 2) / (0.2 * n)) ** 2)
    starts.append(g / g.sum())

    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    starts.append(edge / edge.sum())

    # Two peaks
    g1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.08 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.08 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())

    # Random Dirichlet
    for _ in range(2):
        starts.append(rng.dirichlet(np.ones(n) * 0.3))

    return starts
