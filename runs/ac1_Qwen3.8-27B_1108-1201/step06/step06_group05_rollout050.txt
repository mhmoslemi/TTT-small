Let me also think about the line search more carefully. In Frank-Wolfe, we have:
a_new(t) = (1-t)*a + t*g

The objective is f(a_new(t)) = 2*n*max(conv(a_new(t), a_new(t))) / (sum(a_new(t)))^2

But since both a and g are on the simplex (sum=1), a_new(t) is also on the simplex. So we're just minimizing max(conv(a_new(t), a_new(t))).

The function t -> max(conv(a_new(t), a_new(t))) is convex in t (since each conv entry is quadratic in t, and max of convex functions is convex). So we can use ternary search or golden section search for the line search!

This is a good improvement. Let me use golden section search for the line search.

Actually, each entry of conv(a_new(t), a_new(t)) is a quadratic in t:
conv(a_new(t), a_new(t))[k] = sum_j ((1-t)*a[j] + t*g[j]) * ((1-t)*a[k-j] + t*g[k-j])
= (1-t)^2 * conv(a,a)[k] + 2t(1-t) * conv(a,g)[k] + t^2 * conv(g,g)[k]

So it's a quadratic in t, and the max of quadratics is convex. Golden section search should work well.

"""

import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def conv_cross(a, b):
    """FFT-based cross-convolution a * b."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fw_line_search(a, g, tol=1e-14):
    """Golden section search for optimal t in a_new = (1-t)*a + t*g."""
    # conv(a_new, a_new)[k] = (1-t)^2 * Caa[k] + 2t(1-t) * Cab[k] + t^2 * Cbb[k]
    Caa = fast_convolve(a)
    Cbb = fast_convolve(g)
    Cab = conv_cross(a, g)
    
    # f(t) = max_k [(1-t)^2 * Caa[k] + 2t(1-t) * Cab[k] + t^2 * Cbb[k]]
    # Each term is quadratic in t, so f(t) is convex on [0,1]
    
    inv_phi = (np.sqrt(5) - 1) / 2  # golden ratio inverse
    
    def obj(t):
        val = (1-t)**2 * Caa + 2*t*(1-t) * Cab + t**2 * Cbb
        return val.max()
    
    lo = 0.0
    hi = 1.0
    c = hi - inv_phi * (hi - lo)
    d = lo + inv_phi * (hi - lo)
    fc = obj(c)
    fd = obj(d)
    
    for _ in range(60):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - inv_phi * (hi - lo)
            fc = obj(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + inv_phi * (hi - lo)
            fd = obj(d)
    
    t_opt = (lo + hi) / 2
    return t_opt, obj(t_opt)

def solve_lp_oracle(f, M, max_time=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'max_time': max_time, 'dual_feasibility_tolerance': 1e-10})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def fw_optimize(f0, time_limit, m_factor=1.0, verbose=False):
    """Frank-Wolfe with LP oracle and golden-section line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        g = solve_lp_oracle(f, M * m_factor, max_time=max(5, remaining * 0.3))
        if g is None:
            m_factor = min(m_factor * 1.2, 5.0)
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 10:
                break
            m_factor = min(m_factor * 1.01, 5.0)
            continue
        
        # Golden section line search
        t_opt, best_val = fw_line_search(f, g)
        
        if t_opt > 1e-10 and best_val < M:
            f = (1.0 - t_opt) * f + t_opt * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_factor = 1.0
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    FW it={it}, obj={M:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_factor = min(m_factor * 1.05, 5.0)
        
        if no_improve > 30:
            # Restart with perturbation
            rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
            scale = 0.001 + 0.003 * min(1.0, no_improve / 50.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_factor = 1.0
            no_improve = 0
    
    return best_f, best_obj

def subgrad_optimize(a0, time_limit, top_k=20, lr_init=0.02, mom_init=0.9, verbose=False):
    """Multi-peak subgradient with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    lr = lr_init
    mom = mom_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        obj = conv.max()
        
        # Multi-peak subgradient
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            pv = conv
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
        
        # Exponential weights
        pv_max = pv.max()
        weights = np.exp(-(pv_max - pv) / max(pv_max * 0.01, 1e-10))
        ws = weights.sum()
        if ws < 1e-15:
            weights = np.zeros(len(peaks))
            weights[np.argmax(pv)] = 1.0
        else:
            weights /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * a[k - idx_arr]
        
        # Momentum step
        v = mom * v - lr * sg
        
        # Backtracking line search
        best_candidate = a
        best_val = obj
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            candidate = project_simplex(a + scale * v)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                break
        
        a = best_candidate
        
        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.8
            
        # Adapt
        if no_improve > 20:
            lr *= 0.95
            mom = min(0.97, mom + 0.003)
        if no_improve > 100:
            rng = np.random.default_rng((it * 48271 + 17) % (2**31 - 1))
            noise = 0.002 + 0.005 * min(1.0, (no_improve - 100) / 200.0)

            v = np.zeros(n)
            lr = lr_init
            mom = mom_init
            no_improve = 0
    
    return best_a, best_obj

def coord_descent(a0, time_limit, verbose=False):
    """Coordinate descent: try adjusting pairs of coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    rng = np.random.default_rng(12345)
    
    # Precompute some useful quantities
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Pick a random coordinate to modify
        j = rng.integers(0, n)
        
        # Try moving mass from j to several other positions
        # We'll try: reduce a[j] by δ and increase a[i] by δ
        aj = a[j]
        if aj < 1e-15:
            continue
        
        best_delta = 0.0
        best_i = -1
        best_val = best_obj
        
        # Try a few fractions of a[j] to move
        for frac in [0.1, 0.25, 0.5, 0.75]:
            delta = frac * aj
            # Try several target positions
            for i in rng.integers(0, n, size=min(20, n)):
                if i == j:
                    continue
                a[j] = aj - delta
                a[i] += delta
                val = fast_convolve(a).max()
                if val < best_val - 1e-14:
                    best_val = val
                    best_delta = delta
                    best_i = int(i)
                    best_frac = frac
            a[j] = aj
        
        if best_delta > 0:
            a[j] -= best_delta
            a[best_i] += best_delta
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                break
            a = a / s
            best_obj = fast_convolve(a).max()
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break
    
    return best_a, best_obj

def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s

def try_and_update(a, best_state, label=""):
    """Evaluate sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_state
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    best_seq, best_score = best_state
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return (best_seq, best_score)

Let me also add a "dual certificate checking" step: at the end, check if the current solution satisfies the KKT conditions approximately. If it does, we know it's (approximately) optimal.

The KKT conditions for the problem:
- Primal feasibility: (a*a)[k] <= t for all k, sum(a) = 1, a >= 0
- Dual feasibility: λ_k >= 0 for all k
- Complementary slackness: λ_k * ((a*a)[k] - t) = 0 for all k
- Stationarity: For j in supp(a): sum_k λ_k * 2*a[k-j] = ν (Lagrange multiplier for sum constraint)
  For j not in supp(a): sum_k λ_k * 2*a[k-j] <= ν

So the "dual certificate" is: find λ_k >= 0 such that:
- λ_k > 0 only when (a*a)[k] = t (active peaks)
- For all j in supp(a): sum_k λ_k * 2*a[k-j] = ν
- For all j: sum_k λ_k * 2*a[k-j] <= ν

If we can find such a λ, we've proved optimality.

This is exactly what the LP oracle gives us! The dual of the LP:

is:

The j-th column of the constraint matrix has entries a[j] at positions k where 0<=k-j<n. So the dual constraint is:
a[j] * sum_{k: 0<=k-j<n} λ[k] >= 1 for all j

If a[j] > 0, this becomes: sum_{k: 0<=k-j<n} λ[k] >= 1/a[j]

And the KKT condition for the original problem requires:

These are related but not identical. The LP dual gives a "relaxed" certificate.
