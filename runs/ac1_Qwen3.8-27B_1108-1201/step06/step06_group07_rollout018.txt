def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smooth_max(x, tau=0.01):
    """Log-sum-exp approximation of max."""
    x_max = np.max(x)
    if x_max == 0:
        return 0.0
    return x_max + tau * np.log(np.sum(np.exp((x - x_max) / tau)))

def smooth_max_grad(x, tau=0.01):
    """Gradient of log-sum-exp approximation of max."""
    x_max = np.max(x)
    weights = np.exp((x - x_max) / tau)
    weights /= np.sum(weights)
    return weights

def convolve_subgradient(a, peaks, weights):

        for j in range(js, je + 1):

def spectral_init(n, rng):
    """Generate initial sequences using spectral methods."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Sine-based (reduces center peak)
    x = np.arange(n) / (n - 1)
    for freq in [1, 2, 3, 5]:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * x + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())
    
    # Cosine window (reduces edges)
    a = np.cos(np.pi * x) ** 2
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())
    
    # Dirichlet
    for alpha_val in [0.1, 0.5, 1.0, 2.0, 5.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha_val))
    
    # Two-hump
    centers = [0.25, 0.75]
    widths = [0.08, 0.12]
    a = np.zeros(n)
    for c, w in zip(centers, widths):
        a += np.exp(-0.5 * ((x - c) / w) ** 2)
    inits.append(a / a.sum())
    
    # Random with structure
    a = np.abs(rng.standard_normal(n)) + 0.1
    # Smooth it
    from scipy.ndimage import uniform_filter1d
    a = uniform_filter1d(a, size=max(3, n // 20))
    inits.append(a / a.sum())
    
    return inits

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_max(x, tau):
    """Log-sum-exp approximation of max."""
    x_max = np.max(x)
    if x_max <= 0:
        return 0.0
    shifted = (x - x_max) / tau
    shifted = np.clip(shifted, -50, 50)
    return x_max + tau * np.log(np.sum(np.exp(shifted)))


def smooth_max_weights(x, tau):
    """Weights (softmax) of the log-sum-exp."""
    x_max = np.max(x)
    shifted = (x - x_max) / tau
    shifted = np.clip(shifted, -50, 50)
    w = np.exp(shifted)
    w = w / np.sum(w)
    return w


def conv_grad(a, weights):
    """Gradient of weighted sum of conv entries: sum_k w[k]*conv[k]."""
    n = len(a)
    grad = np.zeros(n)
    conv_len = len(weights)
    for k in range(conv_len):
        wk = weights[k]
        if wk < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad[j] += 2.0 * wk * a[k - j]
    return grad


def active_set_lp(f, M, n, frac=0.35):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0."""
    # Determine active constraints: use current conv as proxy
    conv_f = fast_convolve(f)
    n_total = 2 * n - 1
    n_active = max(80, int(n_total * frac))
    # Select top constraints by conv_f value (likely to be tight)
    k_indices = np.argsort(conv_f)[::-1][:n_active]
    k_indices = np.sort(k_indices)
    
    all_rows = []
    all_cols = []
    all_vals = []
    for idx, k in enumerate(k_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            all_rows.append(idx)
            all_cols.append(j)
            all_vals.append(f[k - j])
    
    if len(all_rows) == 0:
        return None
    
    rows = np.array(all_rows)
    cols = np.array(all_cols)
    vals = np.array(all_vals)
    A_sparse = csr_matrix((vals, (rows, cols)), shape=(len(k_indices), n))
    b_ub = np.full(len(k_indices), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def full_lp(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n_c = 2 * n - 1
    all_rows = []
    all_cols = []
    all_vals = []
    for k in range(n_c):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            all_rows.append(k)
            all_cols.append(j)
            all_vals.append(f[k - j])
    
    rows = np.array(all_rows)
    cols = np.array(all_cols)
    vals = np.array(all_vals)
    A_sparse = csr_matrix((vals, (rows, cols)), shape=(n_c, n))
    b_ub = np.full(n_c, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def sca_optimize(f0, time_limit, use_active=True, active_frac=0.35):
    """
    SCA optimization with active set LP and adaptive M.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max() * 0.998
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        # Solve LP
        if use_active and n > 400:
            g_raw = active_set_lp(f, M, n, frac=active_frac)
        else:
            g_raw = full_lp(f, M, n)
        
        if g_raw is None:
            M *= 1.015
            no_improve += 1
            if no_improve > 40:
                M = fast_convolve(f).max() * 1.002
                no_improve = 0
            continue
        
        g = g_raw / g_raw.sum()
        if g.sum() < 1e-15:
            g = np.ones(n) / n
        
        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 25:
                rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
                f = np.maximum(f + rng.standard_normal(n) * 0.004, 0)
                f = f / f.sum()
                M = fast_convolve(f).max() * 0.998
                no_improve = 0
            continue
        
        # Line search for best blend
        best_val = fast_convolve(f).max()
        best_ft = f.copy()
        
        for alpha_try in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha_try) * f + alpha_try * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_ft = ft.copy()
        
        if best_val < best_obj - 1e-15:
            f = best_ft
            best_obj = best_val
            best_f = f.copy()
            M = best_val * 0.997
            no_improve = 0
        else:
            no_improve += 1
            M *= 0.994
        
        if no_improve > 50:
            rng = np.random.default_rng((it * 31831 + 7) % (2**31 - 1))
            scale = 0.003 + 0.002 * min(3.0, (no_improve - 50) / 20.0)
            f = np.maximum(best_f + rng.standard_normal(n) * scale, 0)
            f = f / f.sum()
            M = fast_convolve(f).max() * 0.998
            no_improve = 0
    
    return best_f, best_obj


def lbfgs_smoothed(a0, time_limit, tau_init=0.08):
    """
    L-BFGS optimization of smoothed max(conv) using continuation method.
    Parameterizes on simplex via log-softmax.
    """
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    tau = tau_init
    theta = np.log(a)
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        maxiter = max(50, int(remaining * 8))
        
        def obj_and_grad(th):
            th_shifted = th - np.max(th)
            a_th = np.exp(th_shifted)
            a_th = a_th / a_th.sum()
            conv = fast_convolve(a_th)
            obj = smooth_max(conv, tau)
            pw = smooth_max_weights(conv, tau)
            grad_a = conv_grad(a_th, pw)
            grad_a_dot_a = np.dot(grad_a, a_th)
            grad_th = a_th * (grad_a - grad_a_dot_a)
            return obj, grad_th
        
        try:
            result = minimize(obj_and_grad, theta, jac=True, method='L-BFGS-B',
                            options={'maxiter': maxiter, 'gtol': 1e-9, 'maxls': 30})
            theta = result.x
        except Exception:
            # Try with smaller maxiter
            try:
                result = minimize(obj_and_grad, theta, jac=True, method='L-BFGS-B',
                                options={'maxiter': 30, 'gtol': 1e-7})
                theta = result.x
            except Exception:
                break
        
        # Evaluate true objective
        a_new = np.exp(theta - np.max(theta))
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0)
        obj_true = fast_convolve(a_new).max()
        
        if obj_true < best_obj - 1e-13:
            best_obj = obj_true
            best_a = a_new.copy()
        
        # Decrease tau (continuation)
        tau *= 0.82
        if tau < 0.003:
            # Check convergence at smallest tau
            if obj_true > best_obj - 1e-11:
                break
            tau = 0.003
    
    return best_a, best_obj


def gradient_descent_shave(a0, time_limit, lr_init=0.01):
    """
    Gradient-based peak shaving with line search.
    Directly reduces the dominant peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = lr_init
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_val] if False else conv[peak_k]
        
        # Subgradient at the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        grad = np.zeros(n)
        for j in range(js, je + 1):
            grad[j] = 2.0 * a[peak_k - j]
        
        # Project gradient onto tangent space of simplex
        mean_grad = np.mean(grad)
        grad_tan = grad - mean_grad
        
        # Step
        a_new = a - lr * grad_tan
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        new_obj = fast_convolve(a_new).max()
        
        if new_obj < best_obj - 1e-14:
            a = a_new
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.995
        if lr < 0.001:
            lr = lr_init
        
        if no_improve > 40:
            rng = np.random.default_rng((it * 9973 + 5) % (2**31 - 1))
            a = np.maximum(best_a + rng.standard_normal(n) * 0.003, 0)
            a = a / a.sum()
            no_improve = 0
            lr = lr_init
    
    return best_a, best_obj


def spectral_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Sine-modulated
    for freq in [1, 2, 3]:
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.4 * np.sin(2 * np.pi * freq * x + phase)
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())
    
    # Cosine window
    a = np.cos(np.pi * x) ** 2
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())
    
    # Dirichlet
    for alpha_val in [0.2, 0.5, 2.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha_val))
    
    # Two-hump
    a = np.exp(-((x - 0.3) / 0.1) ** 2) + np.exp(-((x - 0.7) / 0.1) ** 2) + 0.05
    inits.append(a / a.sum())
    
    # Edge-heavy
    edge = 1.0 / (x * (1 - x) + 0.05)
    inits.append(edge / edge.sum())
    
    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function combining multiple optimization strategies.
    """

    best_seq = None

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Parent load failed: {e}")
    
    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "default")
    
    n_parent = len(parent)
    
    # === Phase 1: Multi-n Active Set SCA (180s) ===
    print("\n=== Phase 1: Multi-n Active Set SCA ===")
    t1 = time.monotonic()
    
    ns_to_try = sorted(set([n_parent, 512, 640, 768, 896, 1024, 1280]))

                              np.linspace(0, 1, n_parent), parent)

            start = start / start.sum()
        
        t_alloc = min(22, remaining_after - 180)

        try_seq(a_opt, f"sca_n{n_try}")
    
    print(f"  P1 done: best={best_score:.8f}, t={time.monotonic()-t1:.0f}s")
    
    # === Phase 2: Deep Full-Constraint SCA (350s) ===
    print("\n=== Phase 2: Deep Full SCA ===")

    if best_seq is not None:

            print(f"  Full SCA n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = sca_optimize(best_seq, time_limit=t_alloc, use_active=False)
