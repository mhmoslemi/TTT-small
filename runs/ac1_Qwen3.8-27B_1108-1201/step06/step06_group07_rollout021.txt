def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgrad_descent(a0, time_limit, top_k=10, eta_init=0.1):

    s = a.sum()

        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        peaks = np.argpartition(conv, -k)[-k:]

        pw = softmax(pv / (0.01 * max_val + 1e-10))

            lo = max(0, kk - n + 1)
            hi = min(n - 1, kk)

            no_improve = 0

Let me also think about the coordinate descent approach more carefully.

**Coordinate Descent:**
At each step, pick a random subset of 2-5 coordinates. Fix all others. Optimize the selected coordinates to minimize the max convolution.

For 2 coordinates `a_i, a_j` with `a_i + a_j = s` fixed:
- `c_k = base_k + (terms involving a_i or a_j)`
- The terms involving `a_i`: `a_i * a_{k-i}` (appears twice: once as `a_i * a_{k-i}` and once as `a_{k-i} * a_i`, unless `k-i = i`)
- Similarly for `a_j`.

So `c_k(x) = base_k + x * f_k + (s-x) * g_k + h_k * x * (s-x)` where `x = a_i`, and `f_k, g_k, h_k` depend on `k` and the fixed coordinates.

Actually, let me just compute `c_k` as a function of `x`:
- `c_k = sum_p a_(p)(x) * a_(k-p)(x)` where `a_(i)(x) = x`, `a_(j)(x) = s-x`, `a_(p)(x) = a_p` for `p ≠ i,j`.

The terms involving `x`:
1. `p = i`: `x * a_(k-i)(x)`. If `k-i = i`: `x^2`. If `k-i = j`: `x*(s-x)`. Else: `x * a_{k-i}`.
2. `p = j`: `(s-x) * a_(k-j)(x)`. If `k-j = i`: `(s-x)*x`. If `k-j = j`: `(s-x)^2`. Else: `(s-x) * a_{k-j}`.
3. `p ≠ i,j` and `k-p = i`: `a_p * x`
4. `p ≠ i,j` and `k-p = j`: `a_p * (s-x)`

So:
`c_k(x) = base_k + x * (coeff of x) + (s-x) * (coeff of (s-x)) + x^2 * I[k-2i=0] + (s-x)^2 * I[k-2j=0] + 2*x*(s-x) * I[k-i=j]`

This is a quadratic in `x`. The max over `k` of these quadratics is a piecewise quadratic function. Its minimum can be found by checking:
- The minimum of each quadratic (if it's in the valid range `[0, s]`)
- The intersection points of pairs of quadratics

For 2D, this is tractable. For higher dimensions, it becomes harder.

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgrad(a, peaks, weights):
    """Compute weighted sum of subgradients of c_k at a for given peaks and weights."""
    n = len(a)
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        w = weights[idx]
        if w < 1e-15:
            continue
        k = int(peaks[idx])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi+1] += 2.0 * w * a[k - hi:k - lo + 1][::-1]
    return grad

def subgrad_nesterov(a0, time_limit, top_k=15, eta_init=0.05):
    """Subgradient descent with Nesterov momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)
    eta = eta_init
    momentum = 0.9
    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv_y = fast_convolve(y)
        max_y = conv_y.max()
        
        # Top-k peaks
        k = min(top_k, len(conv_y))
        peaks = np.argpartition(conv_y, -k)[-k:]
        pv = conv_y[peaks]
        # Softmax weights
        temp = 0.01 * max_y + 1e-10
        pw = np.exp(pv / temp)
        pw /= pw.sum()
        
        # Subgradient at y
        grad = compute_subgrad(y, peaks, pw)
        
        # Nesterov update
        v = momentum * v + grad
        y_old = y.copy()
        y = y - eta * v
        a_new = project_simplex(y)
        
        # Update y for next iteration (Nesterov)
        y = a_new + momentum * (a_new - a)
        a = a_new
        
        # Check improvement
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt step size
        if no_improve > 20:
            eta *= 0.9
        if eta < 1e-5:
            eta = eta_init * 0.3
            no_improve = 0
        
        if no_improve > 100:
            rng = np.random.default_rng((it * 7919) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
            y = a.copy()
            v = np.zeros(n)
            eta = eta_init
            no_improve = 0
    
    return best_a, best_obj

Now let me think about the L-BFGS approach with the log-sum-exp surrogate.

The key idea: minimize `f(a) = eps * log(sum_k exp(c_k/eps))` subject to `a >= 0`, `sum(a) = 1`.

For the constraint, I'll use the "log-barrier" parameterization: `a_i = exp(z_i) / sum_j exp(z_j)` (softmax parameterization). This automatically ensures `a >= 0` and `sum(a) = 1`.

The objective in terms of `z`:
`F(z) = eps * log(sum_k exp(c_k(z)/eps))` where `c_k(z) = sum_i,j z_ij * ...`

Wait, `c_k = sum_i a_i * a_{k-i}` where `a = softmax(z)`. This is a composition of a quadratic form with softmax, which is differentiable.

The gradient:
`dF/dz_m = (1/eps) * sum_k exp(c_k/eps) / sum_j exp(c_j/eps) * dc_k/dz_m`

`dc_k/dz_m = sum_i (da_i/dz_m) * a_{k-i} + sum_i a_i * (da_{k-i}/dz_m)`

`da_i/dz_m = a_i * (delta_im - a_m)`

So `dc_k/dz_m = sum_i a_i * (delta_im - a_m) * a_{k-i} + sum_i a_i * a_{k-i} * (delta_{k-i,m} - a_m)`
`= a_m * a_{k-m} - a_m * sum_i a_i * a_{k-i} + a_{k-m} * a_m - a_m * sum_i a_i * a_{k-i}`
`= 2 * a_m * a_{k-m} - 2 * a_m * c_k`
`= 2 * a_m * (a_{k-m} - c_k)`

(where `a_{k-m} = 0` if `k-m` is out of range)

So: `dF/dz_m = (2/eps) * a_m * sum_k w_k * (a_{k-m} - c_k)`

where `w_k = exp(c_k/eps) / sum_j exp(c_j/eps)`.

`= (2/eps) * a_m * (sum_k w_k * a_{k-m} - c_k * sum_k w_k)`

Wait, `c_k` is indexed by `k`, not a constant. Let me redo.

`dF/dz_m = (1/eps) * sum_k w_k * dc_k/dz_m = (1/eps) * sum_k w_k * 2 * a_m * (a_{k-m} - c_k)`

`= (2*a_m/eps) * sum_k w_k * (a_{k-m} - c_k)`

`= (2*a_m/eps) * (sum_k w_k * a_{k-m} - sum_k w_k * c_k)`

The first term `sum_k w_k * a_{k-m}` is the cross-correlation of `w` and `a` at lag `m`.
The second term `sum_k w_k * c_k` is a scalar (call it `W_c`).

So: `dF/dz_m = (2*a_m/eps) * (corr[m] - W_c)`

where `corr[m] = sum_k w_k * a_{k-m}`.

This can be computed in O(n log n) using FFT!
