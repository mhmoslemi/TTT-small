def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_conv_obj(a, w):

    conv = fast_convolve(a)

def optimize_single_n(a0, time_limit, T_init=0.002, lr_init=0.05, momentum=0.9):

        a = np.ones(n) / n
    else:
        a = a / s

    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    T = T_init
    lr = lr_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)
    
    # Momentum buffer
    v = np.zeros(n)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        
        # Compute weights: softmax of conv / T
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()

            a = candidate
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
        elif new_obj < conv.max() + 1e-10:
            # Accept even if not strictly better than best (for exploration)
            a = candidate
            no_improve += 1
        else:
            # Reject, reduce step size
            lr *= 0.9
            no_improve += 1
        
        # Periodic FW step
        if it % 10 == 0 and no_improve > 5:
            # FW direction: vertex with smallest gradient component
            j_star = np.argmin(grad)
            alpha = min(1.0 / (it % 100 + 20), 0.1)
            fw_candidate = (1 - alpha) * a + alpha * np.zeros(n)
            fw_candidate[j_star] += alpha
            fw_candidate = np.maximum(fw_candidate, 0)
            fw_candidate = fw_candidate / fw_candidate.sum()
            fw_obj = fast_convolve(fw_candidate).max()
            if fw_obj < best_obj - 1e-15:

                best_obj = fw_obj
                best_a = a.copy()
                no_improve = 0

            T *= 1.05
            lr *= 0.95
        if no_improve > 80:

        if no_improve > 200:

            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(best_a + noise)
            v = np.zeros(n)
            T = T_init
            lr = lr_init
            no_improve = 0
        
        # Slowly increase T to explore more broadly
        if it % 500 == 0 and no_improve < 50:

    return best_a, best_obj


def fw_lp_optimize_fast(a0, time_limit):

"""
Multi-phase optimization for minimum peak autocorrelation step function.
Uses FFT-based gradient computation, adaptive smoothed gradient descent,
Frank-Wolfe vertex steps, peak-focused reweighting, and local search.
Explores multiple sequence lengths with adaptive time allocation.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_correlate_w(a, w):
    """Compute grad[j] = 2 * sum_m a[m] * w[j+m] for j=0..n-1.
    This is the gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a[j].
    """
    n = len(a)
    L = 1
    while L < 4 * n - 2:
        L *= 2
    c = np.zeros(L)
    c[:2 * n - 1] = w
    d = np.zeros(L)
    d[:n] = a[::-1]
    e = np.fft.irfft(np.fft.rfft(c) * np.fft.rfft(d), L)
    return 2.0 * e[n - 1:2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_weights(conv, T, peak_fraction=0.0):
    """Compute weights for the smoothed objective.
    If peak_fraction > 0, only weight the top fraction of peaks.
    """
    if peak_fraction > 0:
        n_peaks = max(1, int(peak_fraction * len(conv)))
        top_indices = np.argsort(conv)[::-1][:n_peaks]
        w = np.zeros_like(conv)
        top_vals = conv[top_indices]
        scaled = top_vals / T
        scaled -= scaled.max()
        w[top_indices] = np.exp(scaled)
        w /= w.sum()
    else:
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()
    return w

                  peak_frac=0.0, use_fw=True):

    Adaptive smoothed gradient descent with optional FW vertex steps.
    Minimizes max_k conv(a,a)[k] subject to sum(a)=1, a>=0.
    """

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    T = T_init
    lr = lr_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(it + 12345)
    v = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Compute weights
        pf = peak_frac if it > 50 else 0.0
        w = compute_weights(conv, T, peak_fraction=pf)

        # Gradient
        grad = cross_correlate_w(a, w)

        # Momentum step
        v = momentum * v + grad
        candidate = project_simplex(a - lr * v)

        new_conv = fast_convolve(candidate)
        new_obj = new_conv.max()

        if new_obj < best_obj - 1e-15:
            a = candidate
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
        elif new_obj < M + 1e-10:
            a = candidate
            no_improve += 1
        else:
            lr *= 0.85
            v *= 0.5
            no_improve += 1

        # Periodic FW vertex step
        if use_fw and it % 8 == 0 and no_improve > 3:
            j_star = np.argmin(grad)
            best_alpha = 0.0
            best_fw_val = M
            for alpha in [0.02, 0.05, 0.1, 0.2, 0.4]:
                fc = (1 - alpha) * a + alpha * np.zeros(n)
                fc[j_star] += alpha
                fc = np.maximum(fc, 0)
                fc = fc / fc.sum()
                fv = fast_convolve(fc).max()
                if fv < best_fw_val:
                    best_fw_val = fv
                    best_alpha = alpha
            if best_alpha > 0 and best_fw_val < best_obj - 1e-15:
                a = (1 - best_alpha) * a + best_alpha * np.zeros(n)
                a[j_star] += best_alpha
                a = np.maximum(a, 0)
                a /= a.sum()
                best_obj = best_fw_val
                best_a = a.copy()
                no_improve = 0

        # Adaptive parameters
        if no_improve > 25:
            T *= 1.04
            lr *= 0.96
        if no_improve > 70:
            T = max(T * 0.7, 0.0003)
            lr = min(lr * 1.3, 0.1)
        if no_improve > 180:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(best_a + noise)
            v = np.zeros(n)
            T = T_init
            lr = lr_init
            no_improve = 0
            rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))

        # Gradually increase T for broader exploration
        if it % 400 == 0 and no_improve < 30:
            T *= 1.005

    return best_a, best_obj


def fw_optimize(a0, time_limit):
    """
    Frank-Wolfe with vertex directions and adaptive step size.
    Focuses on reducing the maximum convolution peak.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(424242)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Focus on top peaks
        threshold = M * 0.995
        peak_lags = np.where(conv >= threshold)[0]
        w = np.zeros(2 * n - 1)
        w[peak_lags] = 1.0 / len(peak_lags)

        grad = cross_correlate_w(a, w)
        j_star = np.argmin(grad)

        # Try step sizes
        best_alpha = 0.0
        best_val = M
        for alpha in [0.01, 0.03, 0.07, 0.15, 0.3, 0.5]:
            fc = a.copy() * (1 - alpha)
            fc[j_star] += alpha
            fc = np.maximum(fc, 0)
            ss = fc.sum()
            if ss < 1e-10:
                continue
            fc /= ss
            fv = fast_convolve(fc).max()
            if fv < best_val:
                best_val = fv
                best_alpha = alpha

        if best_alpha > 0:
            a = a.copy() * (1 - best_alpha)
            a[j_star] += best_alpha
            a = np.maximum(a, 0)
            a /= a.sum()
            if best_val < best_obj - 1e-15:
                best_obj = best_val
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            # Switch to gradient step
            T = 0.003
            w_smooth = compute_weights(conv, T, peak_fraction=0.05)
            grad_smooth = cross_correlate_w(a, w_smooth)
            found = False
            for lr in [0.005, 0.01, 0.02, 0.04]:
                candidate = project_simplex(a - lr * grad_smooth)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
                    no_improve = 0
                    found = True
                    break
            if not found:
                noise = rng.standard_normal(n) * 0.004
                a = project_simplex(best_a + noise)
                no_improve = 0

    return best_a, best_obj


def local_search(a0, time_limit, seed=42):
    """Multi-scale local search with gradient steps, random perturbations, and structured moves."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        improved = False

        # Gradient step
        conv = fast_convolve(a)
        w = compute_weights(conv, 0.003, peak_fraction=0.03)
        grad = cross_correlate_w(a, w)

        for lr in [0.003, 0.008, 0.015, 0.03, 0.06]:
            candidate = project_simplex(a - lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_obj - 1e-15:
                a = candidate
                best_obj = val
                best_a = a.copy()
                improved = True
                break

        # Random perturbation
        if not improved:
            scale = 0.001 * (1 + 0.5 * (it % 5))
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(a + pert)
            val = fast_convolve(candidate).max()
            if val < best_obj - 1e-15:
                a = candidate
                best_obj = val
                best_a = a.copy()
                improved = True

        # Coordinate move
        if not improved:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j and a[i] > 0.005:
                for frac in [0.03, 0.08, 0.15, 0.3]:
                    candidate = a.copy()
                    amt = candidate[i] * frac
                    candidate[i] -= amt
                    candidate[j] += amt
                    candidate = np.maximum(candidate, 0)
                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate /= ss
                    val = fast_convolve(candidate).max()
                    if val < best_obj - 1e-15:
                        a = candidate
                        best_obj = val
                        best_a = a.copy()
                        improved = True
                        break

        # Smoothing (reduce high-frequency content)
        if not improved and it % 30 == 0:
            k = 7
            kernel = np.hanning(k)
            kernel /= kernel.sum()
            sm = np.convolve(a, kernel, mode='same')
            sm = np.maximum(sm, 0)
            sm /= sm.sum()
            for alpha in [0.1, 0.3, 0.5, 0.7]:
                candidate = (1 - alpha) * a + alpha * sm
                candidate = project_simplex(candidate)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stagnation = 0
        else:
            stagnation += 1

        if stagnation > 2000:
            noise = rng.standard_normal(n) * 0.008
            a = project_simplex(best_a + noise)
            stagnation = 0

    return best_a, best_obj


def try_seq(a, label, best_seq, best_score):
    """Evaluate and track best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return a_norm.copy(), score
    return best_seq, best_score
