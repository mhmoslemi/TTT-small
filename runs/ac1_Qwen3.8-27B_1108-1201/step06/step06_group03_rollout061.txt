Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Log-sum-exp smoothed gradient descent with Nesterov momentum
- Frank-Wolfe with LP oracle for convergence acceleration
- Adaptive temperature annealing
- Peak-shaving coordinate descent

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def solve_lp_oracle(f, M):
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
- POCS (Projection onto Convex Sets) with DC-concentration in frequency domain
- Log-sum-exp smoothed gradient descent with Nesterov momentum
- Frank-Wolfe with LP oracle for fast convergence
- Symmetric subspace optimization to reduce effective dimensionality
- Peak-shaving coordinate descent for final polishing
- Multi-length exploration with resampling
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def project_symmetric_simplex(v):
    """Project onto simplex intersected with symmetric subspace (a[j] = a[n-1-j])."""
    n = len(v)
    half = (n + 1) // 2
    # Average with reverse to enforce symmetry
    v_sym = 0.5 * (v + v[::-1])
    # Now project the "half" onto the appropriate simplex
    # If v is symmetric, sum(v) = 2*sum(v[:half]) + (v[n//2] if n odd else 0)
    # We need sum(v_sym) = 1, v_sym >= 0, v_sym[j] = v_sym[n-1-j]
    # Equivalent: optimize over half-vector h with 2*sum(h) - (h[mid] if odd) = 1
    # Simpler: just project v_sym onto simplex (which may break symmetry slightly)
    # then re-symmetrize and renormalize
    v_proj = project_simplex(v_sym)
    v_proj = 0.5 * (v_proj + v_proj[::-1])
    s = v_proj.sum()
    if s < 1e-15:
        v_proj = np.ones(n) / n
    else:
        v_proj = v_proj / s
    return v_proj


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def smoothed_objective_and_grad(a, temperature):
    """Compute log-sum-exp smoothed objective and gradient.
    Returns (obj, grad) where grad is w.r.t. a (not projected)."""
    n = len(a)
    conv = fast_convolve(a)
    # Log-sum-exp for numerical stability
    c_max = conv.max()
    shifted = (conv - c_max) / temperature
    # Clip for numerical safety
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    log_sum_exp = c_max + temperature * (np.log(exp_vals.sum()) )
    
    # Weights for gradient
    weights = exp_vals / exp_vals.sum()
    
    # Gradient: grad[j] = 2 * sum_k weights[k] * a[k-j] (with bounds)
    # Efficient computation via cross-correlation
    # grad[j] = 2 * sum_{m=0}^{n-1} a[m] * weights[m+j] (for m+j < 2n-1)
    # Use FFT for efficiency
    L = 1 << (3 * n).bit_length()
    A_fft = np.fft.rfft(a, L)
    W_fft = np.fft.rfft(weights, L)
    # cross-correlation: (a cross weights)[j] = sum_m a[m] * weights[m+j]
    # = irfft(conj(A_fft) * W_fft)[j] ... let me verify
    # Actually: irfft(A_fft * conj(W_fft))[k] = sum_m a[m] * weights[m-k mod L]
    # We want sum_m a[m] * weights[m+j] = sum_m a[m] * weights[m-(-j)]
    # So it's the correlation at lag -j, which is the reverse.
    # Let me use: corr = irfft(conj(W_fft) * A_fft)
    # corr[k] = sum_m weights[m] * a[m+k mod L] (for linear, with L large enough)
    # We want: sum_m a[m] * weights[m+j] = sum_m weights[m+j] * a[m]
    # Let m' = m+j: = sum_{m'} weights[m'] * a[m'-j] (for m'-j >= 0)
    # = sum_{m'} a[m'-j] * weights[m']
    # This is (a cross-correlated with weights) at lag j... 
    
    # Let me just use the straightforward FFT approach:
    # We want g[j] = sum_{m=0}^{n-1} a[m] * w[m+j] where w has length 2n-1
    # This is equivalent to: pad a to length 2n-1 (zeros), then:
    # g[j] = <a_padded, w_shifted_by_j>
    
    # Simpler: use np.correlate
    # np.correlate(a, weights, 'full')[2n-2 : 2n-2+n] gives sum_m a[m]*weights[m+j]
    
    full_corr = np.correlate(a, weights, mode='full')
    offset = len(weights) - 1  # = 2n-2
    grad = 2.0 * full_corr[offset:offset + n]
    
    return log_sum_exp, grad, conv


def pocs_step(a, strength=0.15, mode='dc_boost'):
    """One POCS step: modify frequency domain, project back to simplex."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, L)
    power = np.abs(A) ** 2
    total_power = power.sum()
    
    if total_power < 1e-30:
        return a.copy()
    
    new_power = power.copy()
    
    if mode == 'dc_boost':
        # Boost DC component to make convolution flatter
        dc_frac = power[0] / total_power
        boost = 1.0 + strength
        new_power[0] = min(dc_frac * boost, 1.0) * total_power
        remaining = (1.0 - new_power[0] / total_power) * total_power
        non_dc = power[1:].sum()
        if non_dc > 1e-30:
            new_power[1:] = power[1:] * (remaining / non_dc)
        else:
            new_power[1:] = remaining / max(len(power) - 1, 1)
    elif mode == 'flatten':
        # Flatten non-DC power (reduce dynamic range)
        non_dc = power[1:]
        mean_non_dc = non_dc.mean()
        new_power[1:] = (1 - strength) * non_dc + strength * mean_non_dc
    elif mode == 'soft_threshold':
        # Soft-threshold the power spectrum (reduce sidelobes)
        threshold = strength * power[0]
        new_power[1:] = np.maximum(new_power[1:] - threshold, 0)
        # Renormalize to maintain total power
        # Actually, don't renormalize - the scaling will be handled by projection
    
    phase = A / np.maximum(np.abs(A), 1e-30)
    A_new = phase * np.sqrt(np.maximum(new_power, 0))
    a_new = np.fft.irfft(A_new, L)[:n]
    return project_simplex(a_new)


def fw_single_step(f, M, time_check=None):
    """Single Frank-Wolfe step with LP oracle."""
    g = solve_lp_oracle(f, M * 1.002)
    if g is None or g.sum() < 1e-15:
        g = solve_lp_oracle(f, M * 1.01)
    if g is None or g.sum() < 1e-15:
        return None, None
    g = g / g.sum()
    
    # Line search
    best_alpha = 0.0
    best_val = fast_convolve(f).max()
    for alpha in np.linspace(0.05, 1.0, 15):
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    if best_alpha < 1e-8:
        return None, None
    ft = (1.0 - best_alpha) * f + best_alpha * g
    ft = np.maximum(ft, 0.0)
    ft = ft / ft.sum()
    return ft, ft  # return new point


def smoothed_gradient_optimize(a0, time_limit, T_init=0.0003, lr_init=0.03, use_nesterov=True):
    """Smoothed gradient descent with adaptive temperature and Nesterov momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.9
    y = a.copy()  # Nesterov look-ahead
    
    t0 = time.monotonic()
    stagnant = 0
    it = 0
    T_min = T_init * 0.01
    T_max = T_init * 100
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Nesterov look-ahead
        if use_nesterov:
            y = a + mom * (a - (a - v / max(lr, 1e-15)))  # simplified
            # Actually: y = a + mom * v / lr (standard Nesterov)
            # But v already includes the learning rate, so: y = a + mom * v
            y = a + mom * v
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys > 1e-15:
                y = y / ys
        
        # Compute smoothed objective and gradient
        obj_smooth, grad, conv = smoothed_objective_and_grad(y, T)
        
        # Gradient step
        v = mom * v - lr * grad
        candidate = project_simplex(y + v)
        
        # Backtracking
        actual_obj = 2.0 * n * fast_convolve(candidate).max()
        if actual_obj > best_obj * 1.01 + 1e-6:
            # Step was too large, try smaller
            candidate = project_simplex(y + 0.5 * v)
            actual_obj = 2.0 * n * fast_convolve(candidate).max()
        
        a = candidate
        obj = 2.0 * n * fast_convolve(a).max()
        
        if obj < best_obj - 1e-9:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Adaptive temperature and learning rate
        if stagnant > 30:
            T = max(T_min, T * 0.95)
            lr *= 0.98
        if stagnant > 100:
            T = min(T_max, T * 1.5)
            lr *= 1.2
            mom = min(0.95, mom + 0.01)
        if stagnant > 300:
            # Restart with perturbation
            rng = np.random.default_rng((it * 7919) % (2**31))
            noise = 0.003 + 0.005 * min(1.0, (stagnant - 300) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            T = T_init
            lr = lr_init
            mom = 0.9
            stagnant = 0
        
        # Periodic POCS step
        if it % 50 == 0:
            a_pocs = pocs_step(a, strength=0.05, mode='dc_boost')
            obj_pocs = 2.0 * n * fast_convolve(a_pocs).max()
            if obj_pocs < obj:
                a = a_pocs
                obj = obj_pocs
                if obj < best_obj - 1e-9:
                    best_obj = obj
                    best_a = a.copy()
                    stagnant = 0
    
    return best_a, best_obj


def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    t0 = time.monotonic()
    stagnant = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        
        g = solve_lp_oracle(f, M * 1.002)
        if g is None or g.sum() < 1e-15:
            g = solve_lp_oracle(f, M * 1.01)
        
        if g is None or g.sum() < 1e-15:
            stagnant += 1
            if stagnant > 30:
                rng = np.random.default_rng((it * 7919 + 3) % (2**31))
                scale = 0.002 + 0.004 * min(1.0, stagnant / 80.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue
        
        g = g / g.sum()
        
        if np.max(np.abs(g - f)) < 1e-10:
            stagnant += 1
            if stagnant > 40:
                rng = np.random.default_rng((it * 31 + 7) % (2**31))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in np.linspace(0.02, 1.0, 20):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            obj = 2.0 * n * M
            
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1
        
        if stagnant > 15:
            # Try a POCS step to escape
            f_pocs = pocs_step(f, strength=0.08, mode='dc_boost')
            obj_pocs = 2.0 * n * fast_convolve(f_pocs).max()
            if obj_pocs < best_obj:
                f = f_pocs
                best_obj = obj_pocs
                best_f = f.copy()
                M = fast_convolve(f).max()
                stagnant = 0
        
        if verbose and it % 200 == 0:
            print(f"    FW it={it}, obj={best_obj:.10f}, stagnant={stagnant}")
    
    return best_f, 2.0 * n * best_f.max() if best_f is not None else best_obj


def peak_coordinate_descent(a0, time_limit):
    """Coordinate descent targeting the top peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    step_scale = 0.003
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        n_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]
        
        improved = False
        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            if js > je:
                continue
            
            # Find top contributing pairs
            contribs = []
            for j in range(js, je + 1):
                contribs.append((a[j] * a[k - j], j, k - j))
            contribs.sort(reverse=True)
            
            for (_, j1, j2) in contribs[:6]:
                if a[j1] < 1e-10 and a[j2] < 1e-10:
                    continue
                
                # Try moving mass from the larger contributor to a random location
                src = j1 if a[j1] >= a[j2] else j2
                if a[src] < 1e-10:
                    continue
                
                for _attempt in range(5):
                    rng = np.random.default_rng((it * 9973 + k * 31 + src * 7 + _attempt) % (2**31))
                    j_target = int(rng.integers(0, n))
                    if j_target == src:
                        continue
                    
                    for delta_frac in [step_scale, step_scale * 0.5, step_scale * 2.0]:
                        delta = a[src] * delta_frac
                        candidate = a.copy()
                        candidate[src] = max(0, candidate[src] - delta)
                        candidate[j_target] += delta
                        candidate = np.maximum(candidate, 0)
                        cs = candidate.sum()
                        if cs < 1e-15:
                            continue
                        candidate = candidate / cs
                        val = 2.0 * n * fast_convolve(candidate).max()
                        if val < best_obj - 1e-12:
                            a = candidate
                            best_obj = val
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            step_scale *= 0.95
            if step_scale < 1e-7:
                break
            if it % 100 == 0:
                rng = np.random.default_rng(it * 48271 + 13)
                candidate = project_simplex(a + rng.standard_normal(n) * step_scale * 3)
                val = 2.0 * n * fast_convolve(candidate).max()
                if val < best_obj - 1e-12:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
                    improved = True
            elif it > 3000 and not improved:
                break
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_hann(n):
    """Hann window as probability vector."""
    x = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    w = np.maximum(w, 0.01)
    return w / w.sum()


def generate_starts(n, parent, rng):
    """Generate diverse initializations."""
    starts = []
    
    if parent is not None:
        starts.append(resample_sequence(parent, n))
    
    starts.append(np.ones(n) / n)
    starts.append(generate_hann(n))
    
    # Gaussian centered
    x = np.arange(n, dtype=np.float64)
    g = np.exp(-0.5 * ((x - (n - 1) / 2) / (0.25 * n)) ** 2)
    starts.append(g / g.sum())
    
    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x).astype(float) + 1.0)
    starts.append(edge / edge.sum())
    
    # Two peaks
    g1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.1 * n)) ** 2)
    starts.append((g
