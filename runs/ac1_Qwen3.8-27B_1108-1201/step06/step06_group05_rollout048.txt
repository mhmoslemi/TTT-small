def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(a, peaks, weights):
    """Compute subgradient sum_l weights[l] * a[peak_l - i] for valid indices."""
    n = len(a)
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        w = weights[idx]
        if w < 1e-15:
            continue
        k = int(peaks[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += w * a[k - np.arange(js, je + 1)]
    return 2.0 * sg


def get_top_peaks(conv, k=15):
    """Get indices of top-k peaks in convolution."""
    if k >= len(conv):
        return np.arange(len(conv))
    return np.argpartition(conv, -k)[-k:]


def pgd_optimize(a0, time_limit, top_k=20, lr_init=0.025, mom_init=0.91):
    """Adaptive projected subgradient descent with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = mom_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        obj = conv.max()

        pk = get_top_peaks(conv, top_k)
        pv = conv[pk]
        # Inverse-rank weighting
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(pk))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (1.0 + 0.5 * rank)
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        sg = compute_subgrad(a, pk, weights)

        # Momentum update
        v = mom * v - lr * sg

        # Backtracking line search

        best_val = obj

            no_improve = 0

            v *= 0.85

def spectral_shaping(a0, time_limit, n_iter_max=5000):
    """
    Iterative spectral shaping: in the frequency domain, reduce power at
    frequencies that contribute most to the time-domain peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    conv_size = 2 * n - 1
    fft_size = 1 << (conv_size - 1).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < n_iter_max:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Compute spectrum
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A)**2

        # Current convolution (time domain)
        conv = fast_convolve(a)
        obj = conv.max()

        # Find the peak position
        peak_idx = np.argmax(conv)

        # The contribution of each frequency bin to conv[peak_idx]:
        # conv[peak_idx] = sum_k (1/fft_size) * power[k] * exp(2*pi*i*k*peak_idx/fft_size)
        # = sum_k (1/fft_size) * |A[k]|^2 * cos(2*pi*k*peak_idx/fft_size) (since conv is real)
        # Actually, conv = IFFT(|A|^2), so:
        # conv[m] = (1/fft_size) * sum_k |A[k]|^2 * exp(2*pi*i*k*m/fft_size)
        # For real conv: conv[m] = (1/fft_size) * (|A[0]|^2 + 2*sum_{k=1}^{N/2} |A[k]|^2 * cos(2*pi*k*m/fft_size))

        # Gradient of conv[peak_idx] w.r.t. power[k]:
        # d(conv[peak])/d(power[k]) = (1/fft_size) * cos(2*pi*k*peak/fft_size) for k>0
        #                            = (1/fft_size) for k=0

        # To reduce conv[peak], we should decrease power at bins where cos > 0
        # and increase where cos < 0 (but keep total power fixed for sum(a)=1)

        # Compute the "driving" frequency response
        freqs = np.arange(len(power))
        cos_vals = np.cos(2 * np.pi * freqs * peak_idx / fft_size)
        cos_vals[0] = 1.0  # DC is always positive contribution

        # Direction: reduce power where cos > 0, increase where cos < 0
        # But we need to keep a >= 0, so work in time domain
        # Equivalent: multiply spectrum by (1 - eta * cos_vals) and inverse FFT

        eta = 0.3
        for _ in range(5):
            A_new = A * np.sqrt(np.maximum(1.0 - eta * cos_vals, 0.1))
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new = a_new / s
            obj_new = fast_convolve(a_new).max()
            if obj_new < obj - 1e-18:
                a = a_new
                obj = obj_new
                if obj < best_obj - 1e-16:
                    best_obj = obj
                    best_a = a.copy()
                A = np.fft.rfft(a, fft_size)
                break
            else:
                eta *= 0.5

        if eta < 0.001:
            # Stuck, add small perturbation
            rng = np.random.default_rng(it * 31 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            A = np.fft.rfft(a, fft_size)
            obj = fast_convolve(a).max()

    return best_a, best_obj


def newton_refine(a0, time_limit, tol=1e-13, max_iter=500):
    """
    Newton-type local optimization.
    At optimum with active peaks P and support S:
    - conv[k] = T for k in P
    - For i in S (with a[i] > 0): sum_{k in P} lambda_k * (d conv[k]/d a[i]) = nu
    We use SVD to find the multiplier structure and take gradient steps.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    stalling = 0

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Active peaks: within 0.5% of max
        threshold = M * (1.0 - 0.005)
        active = np.where(conv >= threshold)[0]

        if len(active) < 2:
            break

        # Build the KKT system
        # For support elements: sum_k lambda_k * 2*a[k-i] = nu
        support = np.where(a > 1e-12)[0]
        if len(support) < 2:
            break

        m = len(support)
        p = len(active)

        # A_sys[i, l] = 2 * a[active[l] - support[i]] (if valid)
        A_sys = np.zeros((m, p + 1))
        for i_idx, j in enumerate(support):
            A_sys[i_idx, -1] = 1.0  # nu column (with minus sign)
            for l_idx, k in enumerate(active):
                idx = k - j
                if 0 <= idx < n:
                    A_sys[i_idx, l_idx] = 2.0 * a[idx]

        # Solve: find (lambda, nu) such that A_sys @ [lambda; -nu] ≈ 0
        # Use SVD - smallest right singular vector
        try:
            U, S_vals, Vt = np.linalg.svd(A_sys)
            sol = Vt[-1, :]
            lam = np.maximum(sol[:p], 0.0)
        except np.linalg.LinAlgError:
            break

        lam_sum = lam.sum()
        if lam_sum < 1e-12:
            lam = np.ones(p) / p
        else:
            lam /= lam_sum

        # Gradient direction: -sum_l lam[l] * 2*a[peak_l - i]
        grad = np.zeros(n)
        for l_idx, k in enumerate(active):
            if lam[l_idx] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je + 1] -= 2.0 * lam[l_idx] * a[k - np.arange(js, je + 1)]

        # Line search
        improved = False
        for step in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
            candidate = project_simplex(a + step * grad)
            cv = fast_convolve(candidate)
            val = cv.max()
            if val < M - 1e-16:
                a = candidate
                if val < best_obj - 1e-16:
                    best_obj = val
                    best_a = a.copy()
                improved = True
                break

        if not improved:
            stalling += 1
            if stalling > 30:
                break
        else:
            stalling = 0

    return best_a, best_obj


def variance_polish(a0, time_limit):
    """
    After the max is low, minimize the variance of conv to get a more uniform shape.
    This can help find better local minima for the max.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    conv0 = fast_convolve(a)
    best_obj = conv0.max()

    t0 = time.monotonic()
    lr = 0.01
    v = np.zeros(n)
    mom = 0.9

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        obj = conv.max()

        # Gradient of variance (as secondary objective)
        mean = conv.mean()
        # d(var(conv))/d(a[i]) = 2 * sum_k (conv[k] - mean) * d(conv[k])/d(a[i])
        #                        = 2 * sum_k (conv[k] - mean) * 2*a[k-i]
        dev = conv - mean
        grad = np.zeros(n)
        for k in range(len(dev)):
            if abs(dev[k]) < 1e-10:
                continue
            idx = k
            js = max(0, idx - n + 1)
            je = min(n - 1, idx)
            grad[js:je + 1] += 2.0 * dev[k] * a[idx - np.arange(js, je + 1)]
        grad *= 2.0

        # Combine: primary is max reduction, secondary is variance
        pk = np.argmax(conv)
        sg_max = np.zeros(n)
        js = max(0, pk - n + 1)
        je = min(n - 1, pk)
        sg_max[js:je + 1] = 2.0 * a[pk - np.arange(js, je + 1)]

        combined = sg_max + 0.1 * grad

        v = mom * v - lr * combined
        for scale in [1.0, 0.5, 0.25, 0.125]:
            candidate = project_simplex(a + scale * v)
            cv = fast_convolve(candidate)
            val = cv.max()
            if val < obj - 1e-16:
                a = candidate
                if val < best_obj - 1e-16:
                    best_obj = val
                    best_a = a.copy()
                break

        lr *= 0.999

    return best_a, best_obj


def make_window(n, kind='raised_cosine'):
    """Generate various window functions as initializations."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'raised_cosine':
        j = np.arange(n)
        w = 0.5 * (1 + np.cos(2 * np.pi * j / (n - 1))) if n > 1 else np.array([1.0])
        w = np.maximum(w, 0)
        s = w.sum()
        return w / s if s > 0 else np.ones(n) / n
    elif kind == 'sin_squared':
        j = np.arange(n)
        w = np.sin(np.pi * j / n)**2
        s = w.sum()
        return w / s if s > 0 else np.ones(n) / n
    elif kind == 'parabolic':
        j = np.arange(n)
        center = (n - 1) / 2.0
        w = 1.0 - ((j - center) / center)**2
        w = np.maximum(w, 0)
        s = w.sum()
        return w / s if s > 0 else np.ones(n) / n
    elif kind == 'gaussian':
        j = np.arange(n)
        center = (n - 1) / 2.0
        sigma = n / 4.0
        w = np.exp(-0.5 * ((j - center) / sigma)**2)
        s = w.sum()
        return w / s if s > 0 else np.ones(n) / n
    elif kind == 'hamming':
        j = np.arange(n)
        w = 0.54 - 0.46 * np.cos(2 * np.pi * j / (n - 1)) if n > 1 else np.array([1.0])
        w = np.maximum(w, 0)
        s = w.sum()
        return w / s if s > 0 else np.ones(n) / n
    elif kind == 'hann':
        j = np.arange(n)
        w = 0.5 * (1 - np.cos(2 * np.pi * j / (n - 1))) if n > 1 else np.array([1.0])
        w = np.maximum(w, 0)
        s = w.sum()
        return w / s if s > 0 else np.ones(n) / n
    elif kind == 'blackman':
        j = np.arange(n)
        if n > 2:
            w = 0.42 - 0.5 * np.cos(2 * np.pi * j / (n - 1)) + 0.08 * np.cos(4 * np.pi * j / (n - 1))
        else:
            w = np.ones(n)
        w = np.maximum(w, 0)
        s = w.sum()
        return w / s if s > 0 else np.ones(n) / n
    elif kind == 'power_law':
        j = np.arange(n)
        center = (n - 1) / 2.0
        x = np.abs(j - center) / center
        w = (1 - x)**2
        w = np.maximum(w, 0)
        s = w.sum()
        return w / s if s > 0 else np.ones(n) / n
    elif kind == 'super_gaussian':
        j = np.arange(n)
        center = (n - 1) / 2.0
        w = np.exp(-np.abs(j - center)**4 / (n**4 / 16.0))
        s = w.sum()
        return w / s if s > 0 else np.ones(n) / n
    elif kind == 'flat_top':
        j = np.arange(n)
        if n > 1:
            w = 1.0 - 1.93 * np.cos(2 * np.pi * j / (n - 1)) + 1.29 * np.cos(4 * np.pi * j / (n - 1)) - 0.388 * np.cos(6 * np.pi * j / (n - 1))
        else:
            w = np.array([1.0])
        w = np.maximum(w, 0)
        s = w.sum()
        return w / s if s > 0 else np.ones(n) / n
    else:
        return np.ones(n) / n


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def evaluate_safe(seq):
    """Wrapper that safely evaluates a sequence."""

        s = [float(max(0.0, min(1000.0, x))) for x in seq]
        if len(s) == 0 or sum(s) < 0.01:
            return float('inf')
        return float(evaluate_sequence(s))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Multi-phase hybrid optimization:
    Phase 1: Diverse initializations + quick screening
    Phase 2: Local optimization (PGD + spectral)
    Phase 3: Deep refinement (Newton + variance polish)
    Phase 4: Multi-restart exploration
    Phase 5: Final polishing
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return
        a = a / s
        score = evaluate_safe(a.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_safe(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = evaluate_safe(best_seq.tolist())
        print(f"Default parent: score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Diverse Initializations (~200s) ===
    print("\n=== Phase 1: Diverse Initializations ===")
    t_start = time.monotonic()

    window_types = ['raised_cosine', 'sin_squared', 'parabolic', 'gaussian',
                    'hamming', 'hann', 'blackman', 'flat_top', 'super_gaussian']
    n_candidates = [n_parent]
    for n_try in [n_parent // 2, n_parent * 2, 384, 512, 640, 768, 896, 1024, 1280, 1536]:
        if n_try not in n_candidates and 128 <= n_try <= 2048:
            n_candidates.append(n_try)

    candidates = []

    # Try windows at various sizes
    for n_try in n_candidates[:6]:
        remaining = deadline - time.monotonic() - 1400
        if remaining < 5:
            break
        for wtype in window_types:
            w = make_window(n_try, wtype)
            score = try_update(w, f"win_{wtype}_n{n_try}")
            if score is not None and score < best_score + 0.05:
                candidates.append(w.copy())

    # Also try interpolated parent
    for n_try in n_candidates[:4]:
        if n_try != n_parent:
            interp = interp_sequence(parent, n_try)
            score = try_update(interp, f"interp_n{n_try}")
            if score is not None and score < best_score + 0.05:
                candidates.append(interp.copy())

    # Keep top candidates
    if not candidates:
        candidates = [best_seq.copy()]

    # Sort by score and keep top 10
    scored_candidates = []
    for c in candidates:
        s = evaluate_safe(c.tolist())
        if np.isfinite(s):
            scored_candidates.append((s, c))
    scored_candidates.sort(key=lambda x: x[0])
    top_candidates = [c for _, c in scored_candidates[:10]]

    print(f"  Phase 1: {len(scored_candidates)} candidates, best={best_score:.8f}")
    print(f"  Elapsed: {time.monotonic()-t_start:.0f}s")

    # === Phase 2: Local Optimization on Top Candidates (~500s) ===
    print("\n=== Phase 2: Local Optimization ===")
    t_start = time.monotonic()

    for idx, cand in enumerate(top_candidates[:5]):
        remaining = deadline - time.monotonic() - 900
        if remaining < 30:
            break

        n_c = len(cand)
        t_alloc = min(remaining, 100)
        label = f"cand{idx}_n{n_c}"
        print(f"  Optimizing {label}, t={t_alloc:.0f}s, current_best={best_score:.8f}")

        # PGD
        t_pgd = t_alloc * 0.5
        a_opt, _ = pgd_optimize(cand, time_limit=t_pgd, top_k=20, lr_init=0.025, mom_init=0.91)
        try_update(a_opt, f"pgd_{label}")

        # Spectral shaping
        remaining = deadline - time.monotonic() - 900
        if remaining > 20:
            t_spec = min(remaining, t_alloc * 0.3)
            a_opt2, _ = spectral_shaping(a_opt, time_limit=t_spec)
            try_update(a_opt2, f"spec_{label}")

        # Quick PGD again
        remaining = deadline - time.monotonic() - 900
        if remaining > 15:
            a_opt3, _ = pgd_optimize(a_opt2, time_limit=min(remaining, 30), top_k=15, lr_init=0.015, mom_init=0.93)
            try_update(a_opt3, f"pgd2_{label}")

    print(f"  Phase 2 best: {best_score:.8f}")
    print(f"  Elapsed: {time.monotonic()-t_start:.0f}s")

    # === Phase 3: Deep Refinement (~400s) ===
    print("\n=== Phase 3: Deep Refinement ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    # PGD with fine parameters
    remaining = deadline - time.monotonic() - 350
    if remaining > 20:
        t_alloc = min(remaining, 120)
        print(f"  Fine PGD n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = pgd_optimize(best_seq, time_limit=t_alloc, top_k=30, lr_init=0.01, mom_init=0.95)
        try_update(a_opt, "fine_pgd")

    # Newton refinement
    remaining = deadline - time.monotonic() - 300
    if remaining > 15:
        t_alloc = min(remaining, 100)
        print(f"  Newton refine, t={t_alloc:.0f}s")
        a_opt, _ = newton_refine(best_seq, time_limit=t_alloc)
        try_update(a_opt, "newton")

    # Spectral shaping on best
    remaining = deadline - time.monotonic() - 250
    if remaining > 15:
        t_alloc = min(remaining, 80)
        print(f"  Spectral shape, t={t_alloc:.0f}s")
        a_opt, _ = spectral_shaping(best_seq, time_limit=t_alloc)
        try_update(a_opt, "spectral_deep")

    # More PGD
    remaining = deadline - time.monotonic() - 200
    if remaining > 15:
        t_alloc = min(remaining, 80)
        a_opt, _ = pgd_optimize(best_seq, time_limit=t_alloc, top_k=40, lr_init=0.008, mom_init=0.93)
        try_update(a_opt, "pgd_deep")

    # Variance polish
    remaining = deadline - time.monotonic() - 150
    if remaining > 15:
        t_alloc = min(remaining, 60)
        a_opt, _ = variance_polish(best_seq, time_limit=t_alloc)
        try_update(a_opt, "var_polish")

    print(f"  Phase 3 best: {best_score:.8f}")
    print(f"  Elapsed: {time.monotonic()-t_start:.0f}s")

    # === Phase 4: Multi-Restart Exploration (~250s) ===
    print("\n=== Phase 4: Multi-Restart ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    # Perturbation restarts with different scales
    perturb_scales = [0.003, 0.007, 0.015, 0.03, 0.05]
    for scale in perturb_scales:
        remaining = deadline - time.monotonic() - 100
        if remaining < 20:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc = min(remaining, 60)

        # PGD from perturbation
        a_opt, _ = pgd_optimize(f_pert, time_limit=t_alloc * 0.6, top_k=20, lr_init=0.02, mom_init=0.9)
        try_update(a_opt, f"pert_{scale}")

        # Quick spectral
        remaining = deadline - time.monotonic() - 100
        if remaining > 10:
            a_opt2, _ = spectral_shaping(a_opt, time_limit=min(remaining, 20))
            try_update(a_opt2, f"pert_spec_{scale}")

    # Try different n values from the best sequence
    for n_try in [n_best * 2, n_best // 2, 1024, 2048]:
        if n_try == n_best or n_try < 64 or n_try > 4096:
            continue
        remaining = deadline - time.monotonic() - 80
        if remaining < 20:
            break
        interp = interp_sequence(best_seq, n_try)
        t_alloc = min(remaining, 60)
        a_opt, _ = pgd_optimize(interp, time_limit=t_alloc, top_k=20, lr_init=0.02, mom_init=0.9)
        try_update(a_opt, f"resize_n{n_try}")

    print(f"  Phase 4 best: {best_score:.8f}")
    print(f"  Elapsed: {time.monotonic()-t_start:.0f}s")

    # === Phase 5: Final Polishing (~200s) ===
    print("\n=== Phase 5: Final Polish ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    # Iterative refinement loop
    for round_num in range(5):
        remaining = deadline - time.monotonic() - 30
        if remaining < 15:
            break

        # PGD with very fine parameters
        t_alloc = min(remaining, 50)
        a_opt, _ = pgd_optimize(best_seq, time_limit=t_alloc, top_k=25, lr_init=0.005, mom_init=0.96)
        try_update(a_opt, f"polish_pgd_{round_num}")

        # Newton
        remaining = deadline - time.monotonic() - 30
        if remaining > 10:
            a_opt2, _ = newton_refine(best_seq, time_limit=min(remaining, 30))
            try_update(a_opt2, f"polish_newton_{round_num}")

        # Tiny perturbation
        remaining = deadline - time.monotonic() - 30
        if remaining > 10:
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            a_opt3, _ = pgd_optimize(f_pert, time_limit=min(remaining, 25), top_k=15, lr_init=0.008, mom_init=0.94)
            try_update(a_opt3, f"polish_pert_{round_num}")

    # Symmetry check
    sym_final = (best_seq + best_seq[::-1]) / 2.0
    sym_final = sym_final / sym_final.sum()
    try_update(sym_final, "final_sym")

    # One final PGD on the absolute best
    remaining = deadline - time.monotonic() - 10
    if remaining > 5:
        a_opt, _ = pgd_optimize(best_seq, time_limit=min(remaining, 30), top_k=50, lr_init=0.003, mom_init=0.97)
        try_update(a_opt, "absolute_final")

    print(f"  Phase 5 best: {best_score:.8f}")
    print(f"  Elapsed: {time.monotonic()-t_start:.0f}s")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = evaluate_safe(best_seq.tolist())

    # Final verification
    fs = evaluate_safe(best_seq.tolist())
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Total time: {time.monotonic() - (deadline - budget_s + 30):.0f}s")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
