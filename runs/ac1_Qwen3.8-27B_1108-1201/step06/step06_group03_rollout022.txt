def logsumexp(x):
    """Numerically stable log-sum-exp."""
    x_max = np.max(x)
    if x_max > 700:
        return x_max + np.log(np.sum(np.exp(x - x_max)))
    return np.log(np.sum(np.exp(x)))


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    return e / e.sum()

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            best_candidate = a
            best_val = fast_convolve(a).max()

            for shrink in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
                step = lr * shrink
                new_a = a - step * grad
                new_a = project_simplex(new_a)
                val = fast_convolve(new_a).max()
                if val < best_val - 1e-15:
                    best_val = val
                    best_candidate = new_a
                    break

            # Update momentum
            v = 0.9 * v + (a - best_candidate)
            a = best_candidate
            y = project_simplex(a + 0.5 * v)

            # Track best
            cur_obj = fast_convolve(a).max()
            if cur_obj < best_obj - 1e-15:
                best_obj = cur_obj
                best_a = a.copy()

            if verbose and it % 200 == 0:
                print(f"    NGD it={it}, T={T:.6f}, obj={best_obj:.10f}")

        if _time.monotonic() - t0 > time_limit - 0.1:
            break

    return best_a, best_obj


def lbfgs_smoothed(a0, T, time_limit, max_iter=5000):
    """
    L-BFGS optimization of the smoothed objective using softmax parameterization.
    Optimizes over z where a = softmax(z).
    """
    n = len(a0)
    a_init = np.maximum(a0, 1e-15)
    a_init = a_init / a_init.sum()
    z0 = np.log(a_init)  # log of the initial point

    best_a = a_init.copy()
    best_obj = fast_convolve(a_init).max()
    t0 = _time.monotonic()

    def objective(z):
        a = softmax(z)
        conv = fast_convolve(a)
        return conv.max() + T * np.log(np.sum(np.exp(conv / T - conv.max())))

    # Actually, let's just minimize the max (hard) via smooth approximation
    def obj_and_grad(z):
        a = softmax(z)
        conv = fast_convolve(a)
        # Smooth objective: T * logsumexp(conv/T)
        conv_shifted = conv - conv.max()
        w_unnorm = np.exp(conv_shifted / T)
        w_sum = w_unnorm.sum()
        f_val = conv.max() + T * np.log(w_sum)

        # Gradient wrt z: grad_z[j] = a[j] * (grad_a[j] - sum_m grad_a[m]*a[m])
        w = w_unnorm / w_sum
        N = 1 << (3 * n - 2 - 1).bit_length()
        W_ft = np.fft.rfft(w, N)
        A_ft = np.fft.rfft(a, N)
        term1 = np.fft.irfft(W_ft * np.conj(A_ft), N)[:n]
        term2 = np.fft.irfft(W_ft * A_ft, N)[:n]
        grad_a = term1 + term2

        grad_z = a * (grad_a - np.dot(grad_a, a))
        return f_val, grad_z

    # Run L-BFGS
    result = minimize(obj_and_grad, z0, method='L-BFGS-B',
                      jac=True, options={'maxiter': max_iter, 'maxfun': 50000,
                                         'maxls': 30, 'maxcor': 20})

    if result.success or result.fun < objective(z0):
        a_opt = softmax(result.x)
        a_opt = np.maximum(a_opt, 0)
        a_opt = a_opt / a_opt.sum()
        obj = fast_convolve(a_opt).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt

    # Check time
    if _time.monotonic() - t0 > time_limit - 0.5:
        return best_a, best_obj

    return best_a, best_obj


def flat_spectrum_init(n, n_iters=200, rng=None):
    """
    Gerchberg-Saxton style flat-spectrum initialization.
    Alternates between:
    - Frequency domain: enforce flat magnitude spectrum
    - Time domain: enforce non-negativity and correct length
    """
    if rng is None:
        rng = np.random.default_rng(42)

    N = 2 * n - 1  # Circular convolution length
    a = np.zeros(N)
    a[:n] = rng.random(n)
    a = np.maximum(a, 0)
    a = a / a.sum()

    for it in range(n_iters):
        # Frequency domain: enforce flat magnitude
        A = np.fft.rfft(a)
        mag = np.abs(A)
        # Target magnitude: uniform
        target_mag = mag.mean()
        # New phase-preserving spectrum with target magnitude
        phase = A / (mag + 1e-30)
        A_new = target_mag * phase
        # IFFT
        a_new = np.fft.irfft(A_new * N, N)  # scale by N for inverse
        # Actually use standard: irfft(rfft(x)) = x
        a_new = np.fft.irfft(A_new, N) * N  # normalize
        # Hmm, let me just use the standard approach
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        angle = np.angle(A)
        target_mag = np.full_like(mag, mag.mean())
        A_flat = target_mag * np.exp(1j * angle)
        a_new = np.fft.irfft(A_flat, N)

        # Time domain: keep first n, clip to non-negative
        a = a_new[:n].copy()
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = rng.random(n)
        a = a / s

    return a


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle and extensive line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = _time.monotonic()
    stagnant = 0
    it = 0
    m_relax = 1.002

    while _time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (_time.monotonic() - t0)
        if remaining < 0.5:
            break

        g = solve_lp_oracle(f, M * m_relax)
        if g is None or g.sum() < 1e-15:
            g = solve_lp_oracle(f, M * 1.01)

        if g is None or g.sum() < 1e-15:
            stagnant += 1
            if stagnant > 15:
                rng = np.random.default_rng(it * 7919 + 3)
                scale = 0.002 + 0.004 * min(1.0, stagnant / 50.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.002
                stagnant = 0
                continue
            continue

        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-10:
            stagnant += 1
            if stagnant > 30:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = M + 1e-14
        for alpha in np.linspace(0.01, 1.0, 20):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine search
        if best_alpha > 0.005:
            lo = max(0.0, best_alpha - 0.02)
            hi = min(1.0, best_alpha + 0.02)
            for alpha in np.linspace(lo, hi, 10):
                ft = (1.0 - alpha) * f + alpha * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha

        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.002

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        if stagnant > 10:
            m_relax = min(m_relax * 1.05, 1.05)

        if verbose and it % 50 == 0:
            print(f"    FW it={it}, obj={best_obj:.10f}, stagnant={stagnant}")

    return best_f, best_obj


def peak_coordinate_descent(a0, time_limit):
    """Coordinate descent targeting top peaks via mass redistribution."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = _time.monotonic()
    it = 0
    step_scale = 0.003

    while _time.monotonic() - t0 < time_limit:
        it += 1
        if _time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        n_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        improved = False
        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue

            # Find top contributing pairs
            contribs = []
            for j in range(js, je + 1):
                contribs.append((a[j] * a[k - j], j, k - j))
            contribs.sort(reverse=True)

            for (_, j1, j2) in contribs[:6]:
                if a[j1] < 1e-8 and a[j2] < 1e-8:
                    continue

                rng = np.random.default_rng((it * 9973 + k * 31 + j1 * 7) % (2**31))
                for _attempt in range(3):
                    j_target = int(rng.integers(0, n))
                    if j_target == j1 or j_target == j2:
                        continue
                    for src in [j1, j2]:
                        if a[src] < 1e-8:
                            continue
                        for delta_frac in [step_scale, step_scale * 0.5, step_scale * 2.0]:
                            delta = a[src] * delta_frac
                            candidate = a.copy()
                            candidate[src] -= delta
                            candidate[j_target] += delta
                            if candidate[src] < -1e-12:
                                continue
                            candidate = np.maximum(candidate, 0)
                            candidate = candidate / candidate.sum()
                            val = fast_convolve(candidate).max()
                            if val < best_obj - 1e-15:
                                a = candidate
                                best_obj = val
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            step_scale *= 0.92
            if step_scale < 1e-7:
                break
            if it % 100 == 0:
                rng = np.random.default_rng(it * 48271)
                candidate = project_simplex(a + rng.standard_normal(n) * step_scale * 3)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
                    step_scale = max(step_scale, 0.001)
            elif it > 3000:
                break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_starts(n, parent, rng):
    """Generate diverse initializations."""
    starts = []
    if parent is not None:
        starts.append(resample_sequence(parent, n))

    # Uniform
    starts.append(np.ones(n) / n)

    # Cosine taper
    x = np.arange(n, dtype=np.float64)
    cos_t = 0.5 * (1 + np.cos(np.pi * x / max(1, n - 1)))
    starts.append(cos_t / cos_t.sum())

    # Gaussian
    g = np.exp(-0.5 * ((x - (n - 1) / 2) / (0.2 * n)) ** 2)
    starts.append(g / g.sum())

    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    starts.append(edge / edge.sum())

    # Flat spectrum init
    fs = flat_spectrum_init(n, n_iters=100, rng=rng)
    starts.append(fs)

    # Another flat spectrum with different seed
    rng2 = np.random.default_rng(rng.integers(0, 2**31))
    fs2 = flat_spectrum_init(n, n_iters=100, rng=rng2)
    starts.append(fs2)

    # Random Dirichlet
    starts.append(rng.dirichlet(np.ones(n) * 0.5))

    return starts


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search for minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = _time.monotonic() + max(1.0, float(budget_s) - 60.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")

    n_parent = len(parent)

    # === Phase 1: Flat-spectrum + Nesterov GD exploration ===
    print("\n=== Phase 1: Flat-Spectrum + Nesterov GD ===")
    t_phase = _time.monotonic()

    lengths = sorted(set([n_parent, max(256, n_parent - 64), min(1024, n_parent + 64),
                          512, 640, 768, 896, 1024, 1280]))
    lengths = [l for l in lengths if 128 <= l <= 1536]

    for n_try in lengths:
        remaining = deadline - _time.monotonic() - 900
        if remaining < 20:
            break

        starts = generate_starts(n_try, parent, rng)

        for si, start in enumerate(starts[:5]):
            remaining = deadline - _time.monotonic() - 850
            if remaining < 10:
                break

            # Nesterov GD with adaptive T
            t_alloc = min(remaining, 30)
            conv_init = fast_convolve(start)
            M_init = conv_init.max()
            T_schedule = [
                (M_init * 0.1, M_init * 0.03, 300),
                (M_init * 0.03, M_init * 0.01, 300),
                (M_init * 0.01, M_init * 0.003, 300),
            ]
            a_opt, _ = nesterov_gradient_descent(start, T_schedule, time_limit=t_alloc)
            try_seq(a_opt, f"ngd_n{n_try}_s{si}")

    print(f"  Phase 1 done: best={best_score:.10f} ({_time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: L-BFGS on smoothed objective ===
    print("\n=== Phase 2: L-BFGS Smoothed ===")
    t_phase = _time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        conv_best = fast_convolve(best_seq)
        M_best = conv_best.max()

        # Multiple T values
        for T_frac in [0.05, 0.02, 0.01]:
            remaining = deadline - _time.monotonic() - 600
            if remaining < 20:
                break
            T = M_best * T_frac
            t_alloc = min(remaining, 120)
            print(f"  L-BFGS T={T:.6f}, t={t_alloc:.0f}s")
            a_opt, _ = lbfgs_smoothed(best_seq, T, time_limit=t_alloc, max_iter=3000)
            try_seq(a_opt, f"lbfgs_T{T_frac}")

    print(f"  Phase 2 done: best={best_score:.10f} ({_time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Frank-Wolfe with LP oracle ===
    print("\n=== Phase 3: Frank-Wolfe ===")
    t_phase = _time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Main deep FW
        remaining = deadline - _time.monotonic() - 400
        if remaining > 30:
            t_alloc = min(remaining, 300)
            print(f"  Deep FW n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "fw_deep")

        # Perturbed restarts
        for scale in [0.002, 0.005, 0.01, 0.02]:
            remaining = deadline - _time.monotonic() - 250
            if remaining < 20:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 50)
            a_opt, _ = fw_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"fw_pert_{scale}")

    print(f"  Phase 3 done: best={best_score:.10f} ({_time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Nesterov GD with fine T ===
    print("\n=== Phase 4: Fine Nesterov GD ===")
    t_phase = _time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        conv_best = fast_convolve(best_seq)
        M_best = conv_best.max()

        T_schedule_fine = [
            (M_best * 0.005, M_best * 0.001, 500),
            (M_best * 0.001, M_best * 0.0003, 500),
            (M_best * 0.0003, M_best * 0.0001, 500),
        ]
        remaining = deadline - _time.monotonic() - 200
        if remaining > 20:
            t_alloc = min(remaining, 200)
            print(f"  Fine NGD t={t_alloc:.0f}s")
            a_opt, _ = nesterov_gradient_descent(best_seq, T_schedule_fine,
                                                 time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "ngd_fine")

    print(f"  Phase 4 done: best={best_score:.10f} ({_time.monotonic()-t_phase:.0f}s)")

    print("\n=== Phase 5: Peak Coord Descent ===")

            try_seq(a_opt, "peak_coord")

    print(f"  Phase 5 done: best={best_score:.10f} ({_time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Multi-length + final polish ===
    print("\n=== Phase 6: Multi-Length + Final Polish ===")
    t_phase = _time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Try nearby lengths
        for delta in [-32, 32, -64, 64, -128, 128]:
            remaining = deadline - _time.monotonic() - 60
            if remaining < 15:
                break
            n_try = n_best + delta
            if n_try < 128 or n_try > 2048:
                continue
            resampled = resample_sequence(best_seq, n_try)
            t_alloc = min(remaining, 30)

            # Quick NW optimization
            conv_r = fast_convolve(resampled)
            M_r = conv_r.max()
            T_sch = [(M_r * 0.05, M_r * 0.01, 200), (M_r * 0.01, M_r * 0.003, 200)]
            a_opt, _ = nesterov_gradient_descent(resampled, T_sch, time_limit=t_alloc)
            try_seq(a_opt, f"multilen_n{n_try}")

        # Final alternating polish
        for ri in range(4):
            remaining = deadline - _time.monotonic()
            if remaining < 10:
                break
            if ri % 2 == 0:
                noise = rng.standard_normal(n_best) * 0.0005
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 20)
                a_opt, _ = fw_optimize(start, time_limit=t_alloc)
                try_seq(a_opt, f"polish_fw_{ri}")
            else:
                noise = rng.standard_normal(n_best) * 0.0003
                start = project_simplex(best_seq + noise)
                conv_s = fast_convolve(start)
                M_s = conv_s.max()
                T_sch = [(M_s * 0.003, M_s * 0.0005, 300)]

                a_opt, _ = nesterov_gradient_descent(start, T_sch, time_limit=t_alloc)
                try_seq(a_opt, f"polish_ngd_{ri}")

    print(f"  Phase 6 done: best={best_score:.10f} ({_time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite
