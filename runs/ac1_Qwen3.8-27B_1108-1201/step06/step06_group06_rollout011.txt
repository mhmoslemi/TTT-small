from scipy.optimize import linprog
from scipy.sparse import lil_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def convolve_score(a):
    """Compute the normalized score for a sequence (assumes sum=1)."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def peak_directed_subgrad(a0, time_limit, T_init=0.0008, P_top=5):
    """
    Peak-directed projected subgradient descent.
    Targets the top P lags of the convolution, cycling through them.
    Uses a smoothed objective with adaptive temperature.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    T = T_init
    lr = 0.02
    no_improve = 0
    lag_offset = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        n_lags = len(conv)

        # Compute softmax weights over lags (top-P emphasized)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Emphasize top-P lags
        top_idx = np.argsort(conv)[::-1][:P_top]
        for idx in top_idx:
            p[idx] *= 3.0
        p = p / p.sum()

        # Compute gradient: grad[i] = sum_k p[k] * a[k-i] (if k-i valid)
        # Use FFT-based computation for speed
        # grad[i] = sum_k p[k] * a[k-i] = (p * a_flipped)[i] in some sense
        # Actually: grad[i] = sum_k p[k] * (a[k-i] if 0<=k-i<n else 0)
        # This is the cross-correlation of p (padded) with a
        # grad = correlate(p_full, a) where p_full is the full convolution-length array

        # Efficient computation via FFT
        fft_size = 1 << (n_lags + n - 1 - 1).bit_length()
        P_ft = np.fft.rfft(p, fft_size)
        A_ft = np.fft.rfft(a, fft_size)
        # grad[i] = sum_k p[k] * a[k-i] = (p * a_rev)[i + n - 1] ... 
        # Let me just do it directly for correctness
        # grad[i] = sum over valid k of p[k] * a[k-i]
        # = sum over j=0..n-1 of p[i+j] * a[j]  (where j = k-i, so k = i+j)
        # So grad[i] = sum_j p[i+j] * a[j] for j=0..n-1, with p[i+j] = 0 if i+j >= n_lags

        # This is a sliding dot product: grad[i] = dot(a, p[i:i+n])
        # Use FFT: pad p to length n_lags + n - 1, correlate with a
        p_pad = np.zeros(n_lags + n - 1)
        p_pad[:n_lags] = p
        # grad[i] = sum_j a[j] * p[i+j] = correlate(p_pad, a)[i] (with appropriate indexing)
        # correlate(p_pad, a)[i] = sum_j a[j] * p_pad[i+j] ... actually numpy correlate
        # Let me use direct FFT approach
        # convolve(a, p_pad_reversed)[i] = sum_j a[j] * p_pad[i-j] ... not quite
        # 
        # Simplest: use the fact that grad = ifft(fft(a) * conj(fft(p_shifted)))
        # Or just compute via the Hankel structure
        #
        # For efficiency with n up to ~4096, direct computation is O(n^2) per step
        # which is too slow. Use FFT convolution:
        # grad[i] = sum_j a[j] * p[i+j]
        # Let q[j] = a[j]. Then grad[i] = sum_j q[j] * p[i+j]
        # This is the cross-correlation of q and p (reversed):
        # grad[i] = (q * p_flipped)[n-1-i] ... let me just verify
        #
        # Alternative: grad = np.correlate(a, p[:n], mode='valid') won't work directly
        # because p has length 2n-1 and we need p[i+j] for i=0..n-1, j=0..n-1
        # 
        # Let's use: create array r of length n_lags + n - 1 where r[k] = p[k]
        # Then grad[i] = sum_{j=0}^{n-1} a[j] * r[i+j]
        # This is the "cross-correlation" computed as:
        # flip_r = r[::-1]
        # conv = convolve(a, flip_r)  -- this gives (a * flip_r)[k] = sum_j a[j]*flip_r[k-j]
        # We want sum_j a[j]*r[i+j] = sum_j a[j]*flip_r[n_lags-1-i-j]
        # = (a * flip_r)[n_lags-1-i]
        # So grad[i] = convolve(a, r[::-1])[n_lags-1-i]

        r_flip = p_pad[:n_lags][::-1]
        fft_sz2 = 1 << (n + n_lags - 1 - 1).bit_length()
        A2 = np.fft.rfft(a, fft_sz2)
        R2 = np.fft.rfft(r_flip, fft_sz2)
        conv_ar = np.fft.irfft(A2 * R2, fft_sz2)[:n + n_lags - 1]
        # grad[i] = conv_ar[n_lags - 1 - i]
        grad = conv_ar[n_lags - 1 : n_lags - 1 - n : -1]

        # Projected gradient step with line search
        best_candidate = a
        best_val = conv.max()

        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            c_val = fast_convolve(candidate).max()
            if c_val < best_val - 1e-15:
                best_val = c_val
                best_candidate = candidate

        a = best_candidate

        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt parameters
        if no_improve > 15:
            T *= 1.04
            lr *= 0.99
        if no_improve > 80:
            T *= 0.7
            lr = 0.015
        if no_improve > 200:
            rng = np.random.default_rng((it * 104729 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = 0.02
            no_improve = 0

    return best_a, best_obj


def cyclic_lag_reduction(a0, time_limit, P_top=3):
    """
    Cyclically target top-P convolution lags for reduction.
    For each target lag k*, compute the specific gradient and take a step.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        n_lags = 2 * n - 1

        # Get top P lags
        top_lags = np.argsort(conv)[::-1][:P_top]

        improved = False
        for k_star in top_lags:
            if time.monotonic() - t0 > time_limit - 0.1:
                break

            # Gradient of c[k*] with respect to a[i]:
            # c[k*] = sum_i a[i] * a[k*-i]
            # d(c[k*])/d(a[i]) = a[k*-i] + (if k*-i != i) a[i] ... 
            # Actually: c[k*] = sum_i a[i]*a[k*-i]
            # d/da[i] = a[k*-i] (from the term where a[i] is the first factor)
            #           + a[k*-i] (from the term where a[k*-i] is the first factor, i.e., j = k*-i)
            # Wait: c[k*] = sum_j a[j]*a[k*-j]. 
            # d/da[i]: the term a[i]*a[k*-i] contributes a[k*-i].
            # Also the term a[k*-i]*a[i] contributes a[i] (if k*-i != i).
            # So d/da[i] = a[k*-i] + (i != k*-i) * a[i] ... no.
            # c[k*] = sum_j a[j] * a[k*-j]
            # The derivative with respect to a[i]:
            #   From j=i: a[k*-i]
            #   From j=k*-i (if k*-i != i and 0 <= k*-i < n): a[i]
            # So d/da[i] = a[k*-i] (if 0<=k*-i<n) + a[i] (if 0<=k*-i<n and k*-i != i)
            # Simplify: if 0 <= k*-i < n:
            #   if i == k*-i (i.e., 2i = k*): d/da[i] = 2*a[i]
            #   else: d/da[i] = a[k*-i] + a[i]
            # else: d/da[i] = 0

            grad = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    if i == j:
                        grad[i] = 2.0 * a[i]
                    else:
                        grad[i] = a[j] + a[i]

            # Take a step to reduce c[k*]
            for trial_lr in [0.01, 0.005, 0.002, 0.001]:
                candidate = project_simplex(a - trial_lr * grad)
                c_new = fast_convolve(candidate).max()
                if c_new < fast_convolve(a).max() - 1e-15:
                    a = candidate
                    improved = True
                    break

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            rng = np.random.default_rng((it * 31627 + 7) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            no_improve = 0

    return best_a, best_obj


def swap_based_search(a0, time_limit, seed=42):
    """
    Local search using swap and transfer moves that are convolution-aware.
    Preferentially swaps/transfers between positions that are far apart
    in the convolution sense (to reduce peak alignment).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        improved = False

        # Strategy 1: Transfer mass away from peak-contributing positions
        # Positions i where a[i]*a[k_peak-i] is large contribute most to the peak
        contributions = np.zeros(n)
        for i in range(n):
            j = k_peak - i
            if 0 <= j < n:
                contributions[i] = a[i] * a[j]

        # Find positions with high contribution to the peak
        high_contrib = np.argsort(contributions)[::-1]

        for i in high_contrib[:10]:
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            j_peak = k_peak - i
            if not (0 <= j_peak < n):
                continue

            # Try transferring mass from i to a random position far from j_peak
            for _ in range(3):
                target = int(rng.integers(0, n))
                if target == i or target == j_peak:
                    continue
                # Check if target is "far" from j_peak (in convolution sense)
                for frac in [0.1, 0.2, 0.3, 0.5]:
                    amount = a[i] * frac
                    if amount < 1e-10:
                        break
                    candidate = a.copy()
                    candidate[i] -= amount
                    candidate[target] += amount
                    candidate = np.maximum(candidate, 0)
                    cs = candidate.sum()
                    if cs < 1e-10:
                        break
                    candidate = candidate / cs
                    c_val = fast_convolve(candidate).max()
                    if c_val < best_obj - 1e-15:
                        a = candidate
                        best_obj = c_val
                        best_a = a.copy()
                        improved = True
                        break

        # Strategy 2: Random swap of values at two positions
        if not improved:
            for _ in range(5):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                # Prefer swaps between positions far apart
                candidate = a.copy()
                candidate[i], candidate[j] = candidate[j], candidate[i]
                c_val = fast_convolve(candidate).max()
                if c_val < best_obj - 1e-15:
                    a = candidate
                    best_obj = c_val
                    best_a = a.copy()
                    improved = True
                    break

        # Strategy 3: Pair transfer (move mass from i to j and from k to l simultaneously)
        if not improved and it % 3 == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                for frac in [0.05, 0.1, 0.2]:
                    amount = min(a[i], a[j]) * frac
                    if amount < 1e-12:
                        break
                    candidate = a.copy()
                    candidate[i] -= amount
                    candidate[j] += amount
                    candidate = np.maximum(candidate, 0)
                    cs = candidate.sum()
                    candidate = candidate / cs
                    c_val = fast_convolve(candidate).max()
                    if c_val < best_obj - 1e-15:
                        a = candidate
                        best_obj = c_val
                        best_a = a.copy()
                        improved = True
                        break

        # Strategy 4: Sinusoidal perturbation
        if not improved and it % 20 == 0:
            freq = rng.uniform(0.5, 10.0)
            phase = rng.uniform(0, 2 * np.pi)
            x = np.arange(n)
            pattern = np.sin(2 * np.pi * freq * x / n + phase) * 0.002
            candidate = project_simplex(a + pattern)
            c_val = fast_convolve(candidate).max()
            if c_val < best_obj - 1e-15:
                a = candidate
                best_obj = c_val
                best_a = a.copy()
                improved = True

        if not improved and it > 5000:
            break

    return best_a, best_obj


def simulated_annealing_search(a0, time_limit, seed=99, T_start=0.01, T_end=0.0001):
    """
    Simulated annealing with convolution-aware perturbations.
    Accepts worsening moves with probability based on temperature.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    current_obj = best_obj

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    n_steps = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        progress = (time.monotonic() - t0) / time_limit
        T = T_start * (T_end / T_start) ** progress

        # Generate perturbation
        pert_type = rng.integers(0, 4)

        if pert_type == 0:
            # Random Gaussian perturbation
            scale = 0.005 * (1.0 - 0.5 * progress)
            candidate = project_simplex(a + rng.standard_normal(n) * scale)
        elif pert_type == 1:
            # Two-point transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            amount = a[i] * rng.uniform(0.1, 0.5)
            candidate = a.copy()
            candidate[i] -= amount
            candidate[j] += amount
            candidate = np.maximum(candidate, 0)
            cs = candidate.sum()
            if cs < 1e-10:
                continue
            candidate = candidate / cs
        elif pert_type == 2:
            # Block shift
            block_start = int(rng.integers(0, n - 10))
            block_len = int(rng.integers(5, 20))
            block_len = min(block_len, n - block_start)
            shift = int(rng.integers(-5, 6))
            if shift == 0:
                continue
            candidate = a.copy()
            block = candidate[block_start:block_start + block_len].copy()
            new_start = block_start + shift
            new_end = new_start + block_len
            if new_start < 0 or new_end > n:
                continue
            candidate[block_start:block_start + block_len] = candidate[new_start:new_end]
            candidate[new_start:new_end] = block
            candidate = np.maximum(candidate, 0)
            cs = candidate.sum()
            if cs < 1e-10:
                continue
            candidate = candidate / cs
        else:
            # Swap two random positions
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            candidate = a.copy()
            candidate[i], candidate[j] = candidate[j], candidate[i]

        new_obj = fast_convolve(candidate).max()
        n_steps += 1

        # Acceptance criterion
        if new_obj < current_obj:
            a = candidate
            current_obj = new_obj
            if new_obj < best_obj - 1e-15:
                best_obj = new_obj
                best_a = a.copy()
        else:
            # Metropolis acceptance
            delta = new_obj - current_obj
            if delta > 0 and T > 1e-10:
                accept_prob = np.exp(-delta / max(T, 1e-10))
                if rng.random() < accept_prob:
                    a = candidate
                    current_obj = new_obj

    return best_a, best_obj


def fw_lp_fast(a0, time_limit):
    """
    Fast Frank-Wolfe with LP oracle (simplified version for speed).
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    # Pre-build the LP matrix structure
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    row_idx = np.array(rows)
    col_idx = np.array(cols)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0
        c = -np.ones(n)
        vals = np.empty(len(row_idx))
        for idx in range(len(row_idx)):
            k = row_idx[idx]
            i = col_idx[idx]
            vals[idx] = f[k - i]

        try:
            A_mat = lil_matrix((2 * n - 1, n))
            A_mat[row_idx, col_idx] = vals
            A_mat = A_mat.tocsr()
            b = np.full(2 * n - 1, M)
            bounds = [(0, None)] * n
            res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(0.5, time_limit - (time.monotonic() - t0) - 2)})
            if res.success:
                g = np.maximum(res.x, 0.0)
                gs = g.sum()
                if gs > 1e-15:
                    g = g / gs
                    # Line search
                    best_alpha = 0.0
                    best_val = M
                    for alpha in [0.1, 0.3, 0.5, 0.7, 0.9]:
                        ft = np.maximum((1 - alpha) * f + alpha * g, 0)
                        ft = ft / ft.sum()
                        val = fast_convolve(ft).max()
                        if val < best_val:
                            best_val = val
                            best_alpha = alpha

                    if best_alpha > 1e-10:
                        f = np.maximum((1 - best_alpha) * f + best_alpha * g, 0)
                        f = f / f.sum()
                        obj = fast_convolve(f).max()
                        if obj < best_obj - 1e-14:
                            best_obj = obj
                            best_f = f.copy()
                            no_improve = 0
                        else:
                            no_improve += 1
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 30:
            rng = np.random.default_rng(it * 1299703 + 17)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            no_improve = 0

    return best_f, best_obj


def interpolation_diversify(a_best, n_target, method='linear'):
    """Create a sequence of different length by interpolation."""
    n_src = len(a_best)
    if n_target == n_src:
        return a_best.copy()
    if method == 'linear':
        x_src = np.linspace(0, 1, n_src)
        x_tgt = np.linspace(0, 1, n_target)
        result = np.interp(x_tgt,
