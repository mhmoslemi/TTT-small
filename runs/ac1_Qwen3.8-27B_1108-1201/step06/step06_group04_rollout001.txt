import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(w, a):
    """Compute W @ a where W_{i,j} = w[i+j]. w has length 2n-1, a has length n."""
    return np.correlate(w, a, mode='valid')


def solve_qp_hankel_fast(w, a_init, n, max_iters=300, time_limit=3.0):
    """
    Solve min a^T W a s.t. sum(a)=1, a>=0 using active set with FFT-based ops.
    W is Hankel: W_{i,j} = w[i+j].
    """
    t0 = time.monotonic()
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    active = set(np.where(a > 1e-10)[0])
    if not active:
        active = set(range(n))

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if len(active) == 0:
            return np.ones(n) / n
        if len(active) == 1:
            a = np.zeros(n)
            a[list(active)[0]] = 1.0
            return a

        act = sorted(active)
        k = len(act)

        # Build submatrix W_AA
        W_AA = np.array([[w[act[i] + act[j]] for j in range(k)] for i in range(k)])
        W_AA_reg = W_AA + 1e-14 * np.eye(k)

        try:
            x = np.linalg.solve(W_AA_reg, np.ones(k))
        except np.linalg.LinAlgError:
            try:
                x = np.linalg.lstsq(W_AA_reg, np.ones(k), rcond=None)[0]
            except Exception:
                x = np.ones(k) / k

        x_sum = x.sum()
        if abs(x_sum) < 1e-14:
            x = np.ones(k) / k
            x_sum = 1.0

        a_act = x / x_sum

        # Check for negative components
        min_val = float('inf')
        min_idx = -1
        for i, idx in enumerate(act):
            if a_act[i] < min_val:
                min_val = a_act[i]
                min_idx = idx

        if min_val < -1e-10:
            active.discard(min_idx)
            a_new = np.zeros(n)
            for i, idx in enumerate(act):
                if idx != min_idx:
                    a_new[idx] = max(0.0, a_act[i])
            s2 = a_new.sum()
            if s2 > 1e-15:
                a = a_new / s2
            continue

        # Form full solution
        a = np.zeros(n)
        for i, idx in enumerate(act):
            a[idx] = max(0.0, a_act[i])

        # Compute gradient
        Wa = hankel_mv(w, a)
        grad = 2.0 * Wa
        mu = np.mean([grad[idx] for idx in act])

        # KKT check for inactive
        inactive = set(range(n)) - active
        if inactive:
            min_viol = 0.0
            min_j = -1
            for j in inactive:
                viol = grad[j] - mu
                if viol < min_viol:
                    min_viol = viol
                    min_j = j
            if min_viol < -1e-8:
                active.add(min_j)
                continue

        return a

    return a


def qp_primal_dual(a0, time_limit, T_init=0.01, use_topk=False, topk=15,
                   blend_init=0.5, blend_end=0.2, verbose=False):
    """
    Primal-dual QP iteration with adaptive blending schedule.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    total_iters_planned = 500

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        conv = fast_convolve(a)
        cmax = conv.max()

        # Adaptive blending: start high, decay toward blend_end
        progress = min(1.0, it / total_iters_planned)
        alpha = blend_end + (blend_init - blend_end) * (1.0 - progress)

        # Compute weights - multiple strategies
        strategy = it % 3
        if strategy == 0 and use_topk:
            k_use = min(topk, len(conv))
            peaks = np.argpartition(conv, -k_use)[-k_use:]
            w = np.zeros(len(conv))
            w[peaks] = 1.0 / k_use
        elif strategy == 1:
            # Exponential tilting: emphasize peaks more aggressively
            scaled = conv / (T * 0.5)
            scaled -= scaled.max()
            w = np.exp(scaled)
            w = w / w.sum()
        else:
            scaled = conv / T
            scaled -= scaled.max()
            w = np.exp(scaled)
            w = w / w.sum()

        # Solve QP
        qp_time = min(2.0, remaining * 0.25)
        a_qp = solve_qp_hankel_fast(w, a, n, max_iters=80, time_limit=qp_time)

        # Blend
        a_new = (1.0 - alpha) * a + alpha * a_qp
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a_qp
        else:
            a_new = a_new / s_new

        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0

            no_improve += 1

        a = a_new

def lse_gradient(a0, time_limit, T_init=0.005, lr_init=0.02, momentum=0.9,
                 verbose=False):
    """
    Nesterov-accelerated log-sum-exp gradient descent.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = momentum
    T = T_init
    y = a.copy()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.03:
            break

        # Nesterov lookahead
        y = a + mom * v

        # Compute log-sum-exp gradient
        conv_y = fast_convolve(y)
        scaled = conv_y / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient: 2 * correlation(p, y)
        grad = 2.0 * hankel_mv(p, y)

        # Update with projection
        v_new = project_simplex(y - lr * grad) - y
        a = y + v_new
        v = v_new
        a = project_simplex(a)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        decay = 0.9997
        lr *= decay
        mom = min(0.995, mom + 0.00003)
        T *= 0.9998

        if no_improve > 200:
            lr *= 0.85
            T *= 0.7
        if no_improve > 500:
            rng = np.random.default_rng((it * 6151 + 7) % (2**31))
            # Fourier perturbation restart
            A = np.fft.rfft(best_a)
            n_perturb = max(5, n // 15)
            freqs = rng.choice(len(A), size=n_perturb, replace=False)
            for f in freqs:
                A[f] += rng.standard_normal() * 0.005 * np.abs(A[0])
            A[0] = best_a.sum()
            a = project_simplex(np.fft.irfft(A, n))
            v = np.zeros(n)
            lr = lr_init * 0.5
            mom = momentum
            T = T_init
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    lse it={it}, obj={best_obj:.10f}, lr={lr:.6f}")

    return best_a, best_obj

            best_alpha = a_i_old
            best_peak = conv.max()

            # Try a few values around current
            alphas = np.array([0.0, s, s*0.5, a_i_old + s*0.1, a_i_old - s*0.1,
                               a_i_old + s*0.3, a_i_old - s*0.3])
            alphas = np.clip(alphas, 0.0, s)
            # Add random values
            alphas = np.concatenate([alphas, s * rng.random(5)])
            alphas = np.clip(alphas, 0.0, s)

            for alpha in alphas:
                a[i] = alpha
                a[j] = s - alpha
                c = fast_convolve(a)
                peak = c.max()
                if peak < best_peak - 1e-15:
                    best_peak = peak
                    best_alpha = alpha

            a[i] = best_alpha
            a[j] = s - best_alpha

            if best_peak < best_obj - 1e-15:
                best_obj = best_peak
                best_a = a.copy()
                improved = True
                improved_total += 1

        if not improved and it > 100:
            break

        if verbose and it % 20 == 0:
            print(f"    coord it={it}, obj={best_obj:.10f}, improved={improved_total}")

    return best_a, best_obj


def fourier_restart(a0, n_freqs, rng):
    """Perturb a sequence in the frequency domain."""
    n = len(a0)
    A = np.fft.rfft(a0)
    freqs = rng.choice(len(A), size=min(n_freqs, len(A)), replace=False)
    for f in freqs:
        if f == 0:
            continue
        A[f] += rng.standard_normal() * 0.02 * np.abs(A[0])
    A[0] = a0.sum()
    a_new = np.fft.irfft(A, n)
    return project_simplex(a_new)


def try_and_track(a, best_state, label=""):
    """Evaluate sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_state[0]:
        best_state[0] = score
        best_state[1] = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search for auto-convolution minimization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_score = float('inf')
    best_seq = None

    def try_seq(a, label=""):
        nonlocal best_score, best_seq
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")

    # === Phase 1: Aggressive n-exploration (250s) ===
    print("\n=== Phase 1: n-exploration ===")
    t1 = time.monotonic()

    # Candidate n values: around parent and some other values
    candidate_ns = []
    for delta in range(-5, 6):
        candidate_ns.append(n_parent + delta)
    candidate_ns.extend([n_parent // 2, n_parent * 2, 384, 512, 640, 800, 900, 1024])
    candidate_ns = sorted(set([n for n in candidate_ns if 64 <= n <= 2048]))

    # Sort by expected promise: closer to parent first
    candidate_ns.sort(key=lambda x: abs(x - len(best_seq)) if best_seq is not None else abs(x - n_parent))

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 10:
            break
        if n_try == n_parent and best_seq is not None and len(best_seq) == n_parent:
            start = best_seq.copy()
        else:
            base = best_seq if best_seq is not None else parent
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, len(base)), base)
            start = np.maximum(start, 0)
            start = start / start.sum()

        t_alloc = min(remaining, 25)
        a_opt, _ = lse_gradient(start, time_limit=t_alloc, T_init=0.005, lr_init=0.02)
        try_seq(a_opt, f"lse_n{n_try}")

    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")

    # === Phase 2: Deep QP primal-dual (800s) ===
    print("\n=== Phase 2: Deep QP Primal-Dual ===")
    t2 = time.monotonic()

    if best_seq is not None:
        configs = [
            {'T_init': 0.003, 'use_topk': False, 'topk': 20, 'blend_init': 0.5, 'blend_end': 0.15},
            {'T_init': 0.01, 'use_topk': True, 'topk': 8, 'blend_init': 0.4, 'blend_end': 0.2},
            {'T_init': 0.008, 'use_topk': False, 'topk': 15, 'blend_init': 0.6, 'blend_end': 0.25},
            {'T_init': 0.02, 'use_topk': True, 'topk': 5, 'blend_init': 0.5, 'blend_end': 0.1},
            {'T_init': 0.005, 'use_topk': False, 'topk': 10, 'blend_init': 0.3, 'blend_end': 0.1},
        ]

        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 400
            if remaining < 15:
                break
            t_alloc = min(remaining, 150)
            print(f"  QP-PD config {ci} (T={cfg['T_init']}, topk={'yes' if cfg['use_topk'] else 'no'}:{cfg['topk']}), t={t_alloc:.0f}s")
            a_opt, _ = qp_primal_dual(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"qp_{ci}")

    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")

    # === Phase 3: Log-sum-exp gradient with multiple settings (400s) ===
    print("\n=== Phase 3: LSE Gradient ===")
    t3 = time.monotonic()

    if best_seq is not None:
        lse_configs = [
            {'T_init': 0.002, 'lr_init': 0.01, 'momentum': 0.92},
            {'T_init': 0.008, 'lr_init': 0.025, 'momentum': 0.88},
            {'T_init': 0.004, 'lr_init': 0.015, 'momentum': 0.95},
        ]

        for li, cfg in enumerate(lse_configs):
            remaining = deadline - time.monotonic() - 250
            if remaining < 15:
                break
            t_alloc = min(remaining, 120)
            print(f"  LSE config {li} (T={cfg['T_init']}, lr={cfg['lr_init']}), t={t_alloc:.0f}s")
            a_opt, _ = lse_gradient(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"lse_{li}")

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s}")

    # === Phase 4: Coordinate descent polish (250s) ===
    print("\n=== Phase 4: Coordinate Descent ===")
    t4 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 100
        if remaining > 15:
            t_alloc = min(remaining, 200)
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord")

            # Second round from improved point
            remaining = deadline - time.monotonic() - 60
            if remaining > 10 and best_seq is not None:
                t_alloc2 = min(remaining, 80)
                a_opt2, _ = coordinate_descent(best_seq, time_limit=t_alloc2)
                try_seq(a_opt2, "coord2")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # === Phase 5: Fourier-based diversification + re-optimization (150s) ===
    print("\n=== Phase 5: Fourier Diversification ===")
    t5 = time.monotonic()

    if best_seq is not None:
        for restart_i in range(8):
            remaining = deadline - time.monotonic() - 30
            if remaining < 10:
                break

            n_freqs = [5, 10, 20, 3, 15, 8, 25, 12][restart_i]
            perturbed = fourier_restart(best_seq, n_freqs, rng)

            t_alloc = min(remaining, 12)
            # Quick LSE from perturbed point
            a_opt, _ = lse_gradient(perturbed, time_limit=t_alloc, T_init=0.003, lr_init=0.01)
            try_seq(a_opt, f"fourier_{restart_i}")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # === Phase 6: Final intensive QP on best (100s) ===
    print("\n=== Phase 6: Final QP Polish ===")
    t6 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 20
        if remaining > 10:
            t_alloc = min(remaining, 80)
            a_opt, _ = qp_primal_dual(best_seq, time_limit=t_alloc,
                                       T_init=0.002, use_topk=False,
                                       topk=10, blend_init=0.3, blend_end=0.1)
            try_seq(a_opt, "final_qp")

        remaining = deadline - time.monotonic() - 15
        if remaining > 5:
            t_alloc = min(remaining, 30)
            a_opt2, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt2, "final_coord")

    print(f"  Phase 6 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t6:.0f}s)")

    # === Final verification ===
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Verify final answer
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
