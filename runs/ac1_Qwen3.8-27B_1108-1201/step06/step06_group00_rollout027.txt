import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, constraint_indices):
    """Build sparse constraint matrix for LP."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([idx] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    if not rows:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(len(constraint_indices), n))


def solve_lp_direction(f, M, top_k=30, time_limit=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-K peaks, g >= 0."""
    n = len(f)
    conv = fast_convolve(f)
    top_k = min(top_k, len(conv))
    peak_idx = np.argpartition(conv, -top_k)[-top_k:]
    center = n - 1
    if center not in peak_idx:
        peak_idx = np.append(peak_idx, center)
    A = build_lp_matrix(f, peak_idx)
    if A.shape[0] == 0:
        return np.ones(n) / n
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lse_weights(conv, beta):
    """Compute LSE softmax weights."""
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_bc / total


def compute_subgradient(f, active_peaks, weights):
    """Compute weighted gradient of LSE-smoothed max autocorrelation."""
    n = len(f)
    grad = np.zeros(n)
    for idx, k in enumerate(active_peaks):
        k = int(k)
        w = weights[idx]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += w * 2.0 * f[k - i_range]
    return grad


def spectral_flatten(f, n_iters=300, time_limit=60.0):
    """Iteratively flatten the power spectrum to reduce autocorrelation peak."""
    n = len(f)
    f = f.copy().astype(np.float64)
    s_target = f.sum()
    t0 = time.monotonic()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    fft_len = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        F = np.fft.rfft(f, fft_len)
        mag = np.abs(F)

        # Adaptive alpha: start with strong compression, ease off
        progress = it / max(1, n_iters)
        alpha = 0.4 + 0.35 * progress

        mag_new = mag ** alpha

        # Also try subtracting a fraction of the max to reduce peaks further
        peak_fraction = 0.3 * (1.0 - progress)
        mag_new = mag_new - peak_fraction * mag_new.max()
        mag_new = np.maximum(mag_new, 1e-10)

        # Normalize to preserve total energy
        energy_orig = np.sum(mag ** 2)
        energy_new = np.sum(mag_new ** 2)
        if energy_new > 1e-30:
            mag_new *= np.sqrt(energy_orig / energy_new)

        phase = np.angle(F)
        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, fft_len)[:n]

        # Enforce non-negativity and correct sum
        f_new = np.maximum(f_new, 0.0)
        sn = f_new.sum()
        if sn < 1e-15:

        f_new *= s_target / sn

        if obj < best_obj:

        f = f_new

    return best_f, best_obj


def fw_optimize(f0, time_limit, top_k=30):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_top_k = top_k

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining_time = time_limit - (time.monotonic() - t0)
        if remaining_time < 5:
            break

        lp_time = min(10, max(2, remaining_time * 0.04))
        g = solve_lp_direction(f, M, top_k=cur_top_k, time_limit=lp_time)

        if g is None:
            no_improve += 3
            if no_improve > 40:
                rng = np.random.default_rng(it * 7919 + 13)
                F = np.fft.rfft(f)
                mag = np.abs(F)
                phase = np.angle(F)
                scale = 0.01 + 0.005 * min(1.0, no_improve / 60.0)
                mag_pert = np.maximum(mag * (1.0 + rng.standard_normal(len(mag)) * scale), 0)
                F_new = mag_pert * np.exp(1j * phase)
                f_new = np.fft.irfft(F_new, n)
                f_new = np.maximum(f_new, 0)
                f_new = project_simplex(f_new)
                conv = fast_convolve(f_new)
                M = conv.max()
                if M < best_obj:
                    best_obj = M
                    best_f = f_new.copy()
                f = f_new.copy()
                no_improve = 0
                cur_top_k = top_k
            continue

        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        diff = np.max(np.abs(g - f))
        if diff < 1e-12:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 104729 + 7)
                F = np.fft.rfft(f)
                mag = np.abs(F)
                phase = np.angle(F)
                scale = 0.01 + 0.008 * min(1.0, no_improve / 50.0)
                mag_pert = np.maximum(mag * (1.0 + rng.standard_normal(len(mag)) * scale), 0)
                F_new = mag_pert * np.exp(1j * phase)
                f_new = np.fft.irfft(F_new, n)
                f_new = np.maximum(f_new, 0)
                f_new = project_simplex(f_new)
                conv = fast_convolve(f_new)
                M = conv.max()
                if M < best_obj:
                    best_obj = M
                    best_f = f_new.copy()
                f = f_new.copy()
                no_improve = 0
                cur_top_k = top_k
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = np.linspace(0.05, 1.0, 15)
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-6:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft /= st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f /= f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                cur_top_k = max(20, cur_top_k - 3)
            else:
                no_improve += 1
                cur_top_k = min(cur_top_k + 8, n)
        else:
            no_improve += 1
            cur_top_k = min(cur_top_k + 15, n)

    return best_f, best_obj


def momentum_subgradient(f0, time_limit, beta_start=20.0, beta_end=8000.0, top_k=50):
    """Momentum-accelerated LSE subgradient with adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)  # momentum
    mu = 0.7  # momentum decay (start moderate, increase over time)
    beta = beta_start
    lr = 1.5 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    f_nesterov = f.copy()

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # Nesterov look-ahead
        f_nest = project_simplex(f + mu * v)

        conv = fast_convolve(f_nest)
        k_top = min(top_k, len(conv))
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        w_full = lse_weights(conv, beta)
        w_peaks = w_full[peak_idx]
        w_sum = w_peaks.sum()
        if w_sum > 1e-15:
            w_peaks = w_peaks / w_sum
        else:
            w_peaks = np.ones(k_top) / k_top

        grad = compute_subgradient(f_nest, peak_idx, w_peaks)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        grad_normed = grad / gnorm

        # Backtracking line search
        new_v = -lr * grad_normed
        improved = False
        best_step_val = best_obj
        best_step_f = None

        for trial in range(8):
            candidate_v = mu * v + (1 - mu) * new_v
            f_trial = project_simplex(f + lr * candidate_v)
            if f_trial is None:
                break
            conv_trial = fast_convolve(f_trial)
            obj_trial = conv_trial.max()
            if obj_trial < best_step_val:
                best_step_val = obj_trial
                best_step_f = f_trial.copy()
                v = candidate_v.copy()

        if best_step_f is not None:
            f = best_step_f
            if best_step_val < best_obj - 1e-15:
                best_obj = best_step_val
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.003, 5.0 / np.sqrt(n))
            else:
                no_improve += 1
        else:
            v = mu * v + (1 - mu) * new_v
            f = project_simplex(f + lr * v)
            conv = fast_convolve(f)
            obj = conv.max()
            if obj < best_obj:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Adaptive momentum
        if no_improve > 20:
            mu = max(0.3, mu * 0.95)
            lr *= 0.92
        elif no_improve == 0:
            mu = min(0.9, mu + 0.005)

        # Increase beta gradually
        beta = min(beta * 1.003, beta_end)

        # Reset if stuck
        if no_improve > 60:
            f = best_f.copy()
            v = np.zeros(n)
            mu = 0.5
            lr = 1.5 / np.sqrt(n)
            no_improve = 0
            rng = np.random.default_rng(it * 31 + 17)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)

    return best_f, best_obj


def coordinate_descent(f0, time_limit, n_rounds=None):
    """Coordinate descent targeting peak autocorrelation values."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f.copy()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    # Get top peaks
    n_peaks = min(30, len(conv))
    top_peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]

    # Precompute: for each peak k, which coordinates affect it
    peak_coords = {}
    for k in top_peaks:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        peak_coords[k] = (lo, hi)

    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        # Shuffle coordinate order
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            # Try reducing f[i] and redistributing
            if f[i] < 1e-10:
                continue

            # Determine how much to reduce
            reductions = [0.01, 0.03, 0.05, 0.08, 0.1]
            for frac in reductions:
                delta = frac * f[i]
                if delta < 1e-12:
                    continue

                # Try redistributing to a random set of positions
                n_dest = min(5, n - 1)
                if n_dest <= 0:
                    continue
                dests = rng.choice([j for j in range(n) if j != i], size=n_dest, replace=False) \
                    if n > 1 else np.array([])

                for dest in dests:
                    f_trial = f.copy()
                    f_trial[i] -= delta
                    f_trial[dest] += delta
                    if f_trial[i] < 0:
                        continue
                    f_trial = np.maximum(f_trial, 0.0)
                    f_trial /= f_trial.sum()

                    conv_trial = fast_convolve(f_trial)
                    obj_trial = conv_trial.max()

                    if obj_trial < best_obj - 1e-14:
                        best_obj = obj_trial
                        best_f = f_trial.copy()
                        f = f_trial.copy()
                        # Update top peaks occasionally
                        break

        # Update top peaks every few passes
        if it % 50 == 0 if 'it' in dir() else False:
            conv = fast_convolve(f)
            top_peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
            peak_coords = {}
            for k in top_peaks:
                k = int(k)
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                peak_coords[k] = (lo, hi)

    return best_f, best_obj


def target_peak_descent(f0, time_limit):
    """Targeted descent: repeatedly find the worst peak and try to reduce it."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f.copy()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        max_k = int(np.argmax(conv))
        max_val = conv[max_k]

        if max_val < best_obj * 1.001:
            # Not at the peak location, recheck
            pass

        # The peak is at lag max_k: (a*a)[max_k] = sum_{i=lo}^{hi} a[i]*a[max_k-i]
        lo = max(0, max_k - n + 1)
        hi = min(n - 1, max_k)

        # Find the coordinate that most contributes to this peak
        contributions = np.zeros(n)
        for i in range(lo, hi + 1):
            contributions[i] = f[i] * f[max_k - i]

        # Try reducing the top contributing coordinates
        top_contrib = np.argsort(contributions)[::-1][:10]

        improved = False
        for i in top_contrib:
            if f[i] < 1e-10:
                continue
            # Try small reductions with redistribution
            for frac in [0.02, 0.05, 0.1, 0.2]:
                delta = frac * f[i]
                if delta < 1e-12:
                    continue
                # Redistribute to positions far from the peak
                dist = np.abs(np.arange(n) - max_k)
                # Also avoid positions that would create new peaks
                far_positions = np.where(dist > n // 4)[0]
                far_positions = far_positions[far_positions != i]
                if len(far_positions) == 0:
                    continue

                # Pick a few candidates
                n_cand = min(3, len(far_positions))
                candidates = rng.choice(far_positions, size=n_cand, replace=False)

                for dest in candidates:
                    f_trial = f.copy()
                    f_trial[i] -= delta
                    f_trial[dest] += delta
                    f_trial = np.maximum(f_trial, 0.0)
                    f_trial /= f_trial.sum()

                    conv_trial = fast_convolve(f_trial)
                    if conv_trial.max() < best_obj - 1e-14:
                        best_obj = conv_trial.max()
                        best_f = f_trial.copy()
                        f = f_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try a global perturbation
            f_pert = f + rng.standard_normal(n) * 0.003
            f_pert = np.maximum(f_pert, 0)
            f_pert /= f_pert.sum()
            conv_pert = fast_convolve(f_pert)
            if conv_pert.max() < best_obj:
                best_obj = conv_pert.max()
                best_f = f_pert.copy()
                f = f_pert.copy()

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

            f_pert = np.maximum(f_pert, 0)
            f_pert = project_simplex(f_pert)
            try_sequence(f_pert, best_tracker)

            # Small time-domain perturbation
            f_pert2 = project_simplex(current_best + rng.standard_normal(n_cur) * 0.001)
            try_sequence(f_pert2, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    print("WARNING: Using fallback")
    fallback = np.ones(448) / 448.0
    return [float(x) for x in fallback]
