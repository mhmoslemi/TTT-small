def fast_convolve(a):
    """Linear auto-convolution via FFT."""

        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute sum_k p[k] * (conv(a,a))[k] gradient w.r.t. a."""
    # d/da[m] of sum_k p[k] * sum_j a[j]*a[k-j] = 2 * sum_k p[k] * a[k-m]
    n = len(a)
    # For each m, sum over k of p[k] * a[k-m] where k-m is valid index
    # This is a valid correlation
    return 2.0 * np.correlate(p, a, mode='full')[:n]


def solve_lp_oracle(f, M, top_k=None, time_limit=10.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k (or top-k), g >= 0."""
    n = len(f)
    # Use top-k constraints if specified
    conv_f = fast_convolve(f)  # f * f conv, but we need the constraint matrix
    # Actually the constraint is (f * g)[k] <= M where * is convolution
    # (f*g)[k] = sum_j f[j] * g[k-j] for valid j
    # This is a linear constraint in g with coefficients from f
    
    # Build constraint matrix: A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    # We only need the top-k rows
    if top_k is not None and top_k < 2 * n - 1:
        # Get top-k lag indices
        # We need to know which lags will be active - use current conv as proxy
        all_lags = np.arange(2 * n - 1)
        # Use current f's self-conv to prioritize
        proxy = fast_convolve(f)
        top_indices = np.argpartition(proxy, -min(top_k, 2*n-1))[-min(top_k, 2*n-1):]
    else:
        top_indices = np.arange(2 * n - 1)
    
    n_constraints = len(top_indices)
    A_ub = np.zeros((n_constraints, n))
    for idx, k in enumerate(top_indices):
        # (f*g)[k] = sum_j f[j] * g[k-j]
        # For g[j] to appear: k-j must be a valid index of f, i.e., 0 <= k-j < n
        # So j ranges from max(0, k-n+1) to min(n-1, k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A_ub[idx, js:je+1] = f[k - je:k - js + 1][::-1]
    
    c = -np.ones(n)
    b_ub = np.full(n_constraints, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def nesterov_fw(a0, time_limit, top_k_frac=0.3, use_top_k=True):
    """Nesterov-accelerated Frank-Wolfe with LP oracle."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()
    
    # Nesterov momentum
    beta = 0.5  # initial momentum
    t_iters = 1.0
    y = a.copy()  # Nesterov extrapolated point
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    top_k = max(10, int(top_k_frac * (2 * n - 1))) if use_top_k else None
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break
        
        # Update Nesterov parameters
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_iters * t_iters)) / 2.0
        beta = (t_iters - 1.0) / t_next
        t_iters = t_next
        
        # Solve LP oracle at y (Nesterov point) or a (standard)
        oracle_point = y if beta > 0.1 else a
        g = solve_lp_oracle(oracle_point, M * m_exp, top_k=top_k, time_limit=min(5.0, remaining * 0.1))
        
        if g is None:
            m_exp = min(m_exp * 1.3, 5.0)
            if m_exp > 4.0:
                m_exp = 1.0
                no_improve += 1
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - a)) < 1e-12:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 3.0)
            if no_improve > 40:
                # Restart with perturbation
                rng = np.random.default_rng(it * 7919 + 13)
                scale = 0.002 + 0.005 * min(1.0, no_improve / 100.0)
                a = project_simplex(best_a + rng.standard_normal(n) * scale)
                y = a.copy()
                t_iters = 1.0
                beta = 0.5
                conv = fast_convolve(a)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        
        # Line search: minimize max(conv((1-ta)*a + ta*g)) for ta in [0, 1]
        # Coarse search
        best_alpha = 0.0
        best_val = M
        alphas = np.concatenate([np.linspace(0, 1, 15), np.linspace(0.05, 0.5, 10)])
        for alpha in alphas:
            ft = (1.0 - alpha) * a + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Fine search if needed
        if 0.01 < best_alpha < 0.95:
            lo = max(0.0, best_alpha - 0.04)
            hi = min(1.0, best_alpha + 0.04)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * a + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            a_new = (1.0 - best_alpha) * a + best_alpha * g
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            conv_new = fast_convolve(a_new)
            M_new = conv_new.max()
            
            # Update Nesterov point
            y = a_new + beta * (a_new - a)
            y = np.maximum(y, 0.0)
            ys = y.sum()
            if ys > 1e-15:
                y = y / ys
            else:
                y = a_new.copy()
            
            a = a_new
            M = M_new
            m_exp = 1.0
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)
            y = a.copy()
    
    return best_a, best_obj


def smoothed_grad_descent(a0, time_limit, T_init=0.001, lr_init=0.03):
    """Smoothed gradient descent using softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        grad = hankel_mv(p, a)
        
        best_candidate = a
        best_val = conv.max()
        
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
        
        a = best_candidate
        obj = best_val
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 15:
            T *= 1.08
            lr *= 0.96
        if no_improve > 50:
            T *= 0.7
        if no_improve > 120:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.001 + 0.004 * min(1.0, (no_improve - 120) / 150.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = T_init
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj


def spectral_equalize(a0, time_limit, n_modes=20):
    """Try to flatten the DFT magnitude to reduce convolution peak."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    
    # Work in frequency domain: a = IDFT(A), we want |A| flat
    # But a >= 0 constraint makes this hard. Use iterative projection.
    for it in range(50):
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        A = np.fft.rfft(a, 2 * n - 1)
        mag = np.abs(A)
        # Target: flat magnitude (geometric mean)
        target_mag = np.geometric_mean(np.maximum(mag, 1e-10))
        
        # Scale phases to target magnitude
        A_new = A * (target_mag / np.maximum(mag, 1e-10))
        a_new = np.fft.irfft(A_new, 2 * n - 1)[:n]
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn
        
        # Mix with current
        for mix in [0.3, 0.5, 0.7, 1.0]:
            candidate = (1 - mix) * a + mix * a_new
            candidate = np.maximum(candidate, 0.0)
            cs = candidate.sum()
            if cs < 1e-15:
                continue
            candidate = candidate / cs
            val = fast_convolve(candidate).max()
            if val < best_obj - 1e-15:
                best_obj = val
                best_a = candidate.copy()
                a = candidate
                break
    
    return best_a, best_obj


def targeted_peak_reduction(a0, time_limit):
    """Reduce the highest peak by targeted coordinate transfers."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    stale = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        
        if len(affected) == 0:
            break
        
        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]
        
        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            # Transfer within affected to reduce peak
            j_min_cand = affected[np.argmin(a[affected])]
        else:
            j_min_cand = outside[np.argmin(a[outside])]
        
        improved = False
        for delta in [0.0003, 0.0008, 0.002, 0.005, 0.01, 0.02]:
            if a[j_max] < delta:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min_cand] += delta
            candidate = np.maximum(candidate, 0.0)
            candidate = candidate / candidate.sum()
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            stale = 0
        else:
            stale += 1
            if stale % 20 == 0:
                rng = np.random.default_rng(it * 4567 + 3)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    stale = 0
                elif stale > 300:
                    break
    
    return best_a, best_obj


def multi_peak_subgrad(a0, time_limit, top_k=20, lr_init=0.003, mom=0.93):
    """Multi-peak subgradient with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        if ws > 0:
            weights /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]
        
        v = mom * v - lr * sg
        a = project_simplex(a + v)
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.9998
        mom = min(0.97, mom + 0.00005)
        
        if no_improve > 150:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.001 + 0.003 * min(1.0, (no_improve - 150) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init * 0.7
            no_improve = 0
    
    return best_a, best_obj


def try_and_update(a, state, label):
    """Evaluate a sequence and update state if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a)))


def interp_sequence(parent, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(parent)
    if n_old == n_new:
        return parent.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    result = np.interp(x_new, x_old, parent)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(n_new) / n_new
    return result


def diverse_initializations(n, parent, rng):
    """Generate diverse initializations for a given n."""
    inits = []
    inits.append(np.ones(n) / n)
    
    # Interpolated parent
    if parent is not None:
        inits.append(interp_sequence(parent, n))
    
    # Random Dirichlet
    for alpha in [0.2, 0.5, 1.0, 2.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha))
    
    # Structured
    x = np.arange(n, dtype=np.float64)
    
    # Edge-weighted
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Asymmetric: ramp up then plateau
    ramp = np.minimum(x / (n * 0.3), 1.0)
    inits.append(ramp / ramp.sum())
    
    # Asymmetric: plateau then ramp down
    ramp2 = np.minimum((n - 1 - x) / (n * 0.3), 1.0)
    inits.append(ramp2 / ramp2.sum())
    
    # Cosine asymmetric
    h = 1.0 + 0.5 * np.cos(2.0 * np.pi * x / n - np.pi / 3)
    h = np.maximum(h, 0.01)
    inits.append(h / h.sum())
    
    # Two bumps
    g1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.15 * n)) ** 2)
    two_bump = g1 + g2
    inits.append(two_bump / two_bump.sum())
    
    # Quarter wave
    qw = np.sin(np.pi * x / (2 * n))
    qw = np.maximum(qw, 0.001)
    inits.append(qw / qw.sum())
    
    return inits

        # Multiple smoothed gradient configs
        for T_init, lr_init in [(0.0002, 0.05), (0.0008, 0.03), (0.002, 0.02),
                                 (0.005, 0.01)]:
            remaining = deadline - time.monotonic() - 300
            if remaining < 8:
                break
            t_alloc = min(remaining, 50)
            a_opt, _ = smoothed_grad_descent(state[1], time_limit=t_alloc,
                                             T_init=T_init, lr_init=lr_init)
            try_and_update(a_opt, state, "sg_T{:.4f}".format(T_init))
        
        # Multi-peak subgradient
        for top_k, lr, mom in [(10, 0.004, 0.92), (30, 0.002, 0.94),
                                (50, 0.001, 0.95), (80, 0.002, 0.93)]:
            remaining = deadline - time.monotonic() - 150
            if remaining < 8:
                break
            t_alloc = min(remaining, 40)
            a_opt, _ = multi_peak_subgrad(state[1], time_limit=t_alloc,
                                          top_k=top_k, lr_init=lr, mom=mom)
            try_and_update(a_opt, state, "mpsg_k{}".format(top_k))
    
    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(
        state[0], time.monotonic() - t_phase))
    
    # === Phase 4: Targeted peak reduction + alternating (250s) ===
    print("\n=== Phase 4: Targeted + Alternating ===")
    t_phase = time.monotonic()
    
    if state[1] is not None:
        # Targeted peak reduction
        remaining = deadline - time.monotonic() - 200
        if remaining > 10:
            t_alloc = min(remaining, 80)
            print("  Targeted peak reduction, t={:.0f}s".format(t_alloc))
            a_opt, _ = targeted_peak_reduction(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "targeted")
        
        # Alternate: targeted -> FW -> targeted
        for round_num in range(4):
            remaining = deadline - time.monotonic() - 100
            if remaining < 10:
                break
            t1 = min(remaining // 3, 30)
            t2 = min(remaining // 3, 40)
            
            if t1 > 3:
                a_opt, _ = targeted_peak_reduction(state[1], time_limit=t1)
                try_and_update(a_opt, state, "alt_t_{}".format(round_num))
            
            if t2 > 3:
                a_opt, _ = nesterov_fw(state[1], time_limit=t2)
                try_and_update(a_opt, state, "alt_fw_{}".format(round_num))
        
        # Fine smoothed gradient
        remaining = deadline - time.monotonic() - 50
        if remaining > 5:
            a_opt, _ = smoothed_grad_descent(state[1], time_limit=min(remaining, 30),
                                             T_init=0.0001, lr_init=0.005)
            try_and_update(a_opt, state, "fine_smooth")
    
    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(
        state[0], time.monotonic() - t_phase))
    
    # === Phase 5: Final experiments (150s) ===
    print("\n=== Phase 5: Final Experiments ===")
    t_phase = time.monotonic()
    
    # Try slightly different n values near the best
    if state[1] is not None:
        n_best = len(state[1])
        for delta_n in [-64, -32, 32, 64, 128]:
            n_try = n_best + delta_n
            if n_try < 128 or n_try > 2500:
                continue
            remaining = deadline - time.monotonic() - 50
            if remaining < 5:
                break
            start = interp_sequence(state[1], n_try)
            t_alloc = min(remaining, 30)
            a_opt, _ = nesterov_fw(start, time_limit=t_alloc)
            try_and_update(a_opt, state, "final_n{}".format(n_try))
        
        # Tiny perturbations with quick polish
        for _ in range(8):
            remaining = deadline - time.monotonic() - 20
            if remaining < 3:
                break
            noise = rng.standard_normal(len(state[1])) * 0.0003
            f_pert = project_simplex(state[1] + noise)
            a_opt, _ = smoothed_grad_descent(f_pert, time_limit=min(remaining, 5),
                                             T_init=0.0002, lr_init=0.003)
            try_and_update(a_opt, state, "perturb_fine")
            a_opt2, _ = targeted_peak_reduction(state[1], time_limit=min(remaining, 5))
            try_and_update(a_opt2, state, "perturb_target")
    
    # One final FW pass on the absolute best
    remaining = deadline - time.monotonic() - 10
    if remaining > 5 and state[1] is not None:
        a_opt, _ = nesterov_fw(state[1], time_limit=min(remaining, 20), top_k_frac=0.15)
        try_and_update(a_opt, state, "absolute_final")
    
    print("  Phase 5 best: {:.8f} (elapsed {:.0f}s)".format(
        state[0], time.monotonic() - t_phase))
    
    # Fallback
    if state[1] is None:
        state[1] = np.ones(384) / 384
        state[0] = float(evaluate_sequence(state[1].tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs
    
    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
