    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:

    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(k, a):
    """Return vector v where v[j] = a[k-j] if 0<=k-j<n, else 0. Length n."""
    n = len(a)
    result = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if js > je:
        return result
    result[js:je + 1] = a[k - js:k - je + 1][::-1]
    return result


def compute_smoothed_grad(a, T, top_k=None):
    """Compute gradient of T*log(sum exp(b[k]/T)) w.r.t. a using FFT."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()

    if top_k is not None and top_k < len(p):
        idx = np.argpartition(p, -top_k)[-top_k:]
        p_masked = np.zeros_like(p)
        p_masked[idx] = p[idx]
        s = p_masked.sum()
        if s > 1e-30:
            p = p_masked / s

    # grad[j] = 2 * sum_k p[k] * a[k-j] = 2 * (p * flip(a))[n-1+j]
    a_flip = a[::-1].copy()
    size = len(p) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    Af = np.fft.rfft(a_flip, fft_size)
    conv_result = np.fft.irfft(P * Af, fft_size)
    grad = conv_result[n - 1:n - 1 + n]
    return 2.0 * grad


def smooth_gd(a0, time_limit, T_init=0.005, eta_init=0.05, top_k=None,
              T_decay=0.99995, eta_decay=0.99999, restart_every=500):
    """Smoothed gradient descent with adaptive temperature and restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    eta = eta_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        grad = compute_smoothed_grad(a, T, top_k=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size with line search
        step_found = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            trial_eta = eta * factor
            candidate = project_simplex(a - trial_eta * grad)
            c = fast_convolve(candidate)
            if c.max() < obj - 1e-15:
                a = candidate
                step_found = True
                break

        if not step_found:
            no_improve += 5

        T = max(T * T_decay, 0.0001)
        eta = eta * eta_decay

        # Restart if stuck
        if no_improve > restart_every:
            noise_scale = 0.002 + 0.004 * min(1.0, no_improve / (restart_every * 3))
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = T_init
            eta = eta_init
            no_improve = 0

        # Gradually cool down
        elapsed = time.monotonic() - t0
        if elapsed > time_limit * 0.7:
            T = max(T * 0.999, 0.00005)
            eta = eta * 0.999

    return best_a, best_obj


def fw_lp_step(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0. Returns g or None."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_optimize(a0, time_limit, lp_interval=5):
    """Frank-Wolfe with periodic LP oracle and gradient steps between."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_obj = M
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(67890)
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Every lp_interval iterations, try LP oracle
        if it % lp_interval == 1:
            g = fw_lp_step(a, M * m_exp)
            if g is not None:
                gs = g.sum()
                if gs > 1e-15:
                    g = g / gs
                    # Line search
                    best_alpha = 0.0
                    best_val = M
                    for alpha in np.linspace(0.05, 1.0, 15):
                        ft = (1.0 - alpha) * a + alpha * g
                        ft = np.maximum(ft, 0.0)
                        st = ft.sum()
                        if st < 1e-15:
                            continue
                        ft = ft / st
                        val = fast_convolve(ft).max()
                        if val < best_val:
                            best_val = val
                            best_alpha = alpha

                    if best_alpha > 1e-8:
                        a = (1.0 - best_alpha) * a + best_alpha * g
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        conv = fast_convolve(a)
                        M = conv.max()
                        m_exp = 1.0
                        if M < best_obj - 1e-14:
                            best_obj = M
                            best_a = a.copy()
                            no_improve = 0
                        else:
                            no_improve += 1
                        continue
            m_exp = min(m_exp * 1.2, 4.0)
        else:
            # Gradient step
            T = 0.001
            grad = compute_smoothed_grad(a, T, top_k=20)
            for factor in [0.1, 0.05, 0.02, 0.01]:
                candidate = project_simplex(a - factor * grad)
                c = fast_convolve(candidate)
                if c.max() < M - 1e-15:
                    a = candidate
                    M = c.max()
                    if M < best_obj - 1e-14:
                        best_obj = M
                        best_a = a.copy()
                    no_improve = 0
                    break
            else:
                no_improve += 1

        if no_improve > 100:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            m_exp = 1.0

    return best_a, best_obj


def peak_reducer(a0, time_limit, top_peaks=5):
    """Reduce top peaks by moving mass from high-contribution to low-contribution coords."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    stale = 0
    rng = np.random.default_rng(424242)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        # Get top peaks
        if top_peaks >= len(conv):
            peak_indices = np.arange(len(conv))
        else:
            peak_indices = np.argpartition(conv, -top_peaks)[-top_peaks:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        improved = False
        for k_peak in peak_indices:
            if conv[k_peak] < best_obj * 0.99:
                continue

            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            affected = np.arange(js, je + 1)
            if len(affected) == 0:
                continue

            contributions = a[affected] * a[k_peak - affected]
            j_max = affected[np.argmax(contributions)]

            outside = np.setdiff1d(np.arange(n), affected)
            if len(outside) == 0:
                continue
            j_min = outside[np.argmin(a[outside]) + 0]  # coordinate with smallest a

            if a[j_max] < 1e-10:
                continue

            for delta in [0.0001, 0.0003, 0.001, 0.003, 0.01]:
                if a[j_max] < delta:
                    break
                candidate = a.copy()
                candidate[j_max] -= delta
                candidate[j_min] += delta
                candidate = np.maximum(candidate, 0.0)
                ss = candidate.sum()
                if ss < 1e-15:
                    break
                candidate = candidate / ss
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stale = 0
        else:
            stale += 1
            if stale % 50 == 0:
                candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    stale = 0
            elif stale > 500:
                break

    return best_a, best_obj


def multi_peak_sg(a0, time_limit, top_k=20, lr_init=0.01, momentum=0.9):
    """Multi-peak subgradient with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(99999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]

        v = momentum * v - lr * sg
        candidate = project_simplex(a + v)

        c = fast_convolve(candidate)
        if c.max() < best_obj - 1e-14:
            a = candidate
            best_obj = c.max()
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9998
        momentum = min(0.97, momentum + 0.00005)

        if no_improve > 200:
            noise = 0.001 + 0.003 * min(1.0, (no_improve - 200) / 400.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init * 0.5
            no_improve = 0

    return best_a, best_obj


def try_and_update(a, state, label):
    """Evaluate and update state if improved. state = [best_score, best_seq]."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.10f} (n={})".format(label, score, len(a)))


def interpolate_to_n(parent, n):
    """Interpolate parent sequence to length n."""
    px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
    px = np.maximum(px, 0)
    ps = px.sum()
    if ps > 1e-10:
        return px / ps
    return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):

    state = [float('inf'), None]

        p = np.clip(p, 0.0, 1000.0)

            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))

                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.10f}".format(len(parent), ps))

        pass

    if parent is None:
        parent = np.ones(768) / 768
        state = [float(evaluate_sequence(parent.tolist())), parent.copy()]
        print("No parent, using uniform n=768")

                                 eta_decay=0.99999, restart_every=2000)
            try_and_update(a_opt, state, "fine_sg")

            if remaining < 5:

            perturbed = project_simplex(state[1] + rng.standard_normal(len(state[1])) * scale)
            t_alloc = min(remaining, 20)
            a_opt, _ = smooth_gd(perturbed, time_limit=t_alloc, T_init=0.001,
                                 eta_init=0.02, top_k=50, T_decay=0.99999,

    # Final verification

    return [float(x) for x in state[1]]
