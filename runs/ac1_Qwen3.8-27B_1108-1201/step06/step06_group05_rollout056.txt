import numpy as np

def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_gradient(p, a):
    """Compute grad[j] = 2 * sum_k p[k] * a[k-j] for j=0..n-1.
    This is the gradient of sum_k p[k]*c[k] where c = conv(a,a)."""
    n = len(a)
    corr_full = np.correlate(p, a, mode='full')
    return 2.0 * corr_full[:n]


def hankel_gradient_fft(p, a):
    """FFT-accelerated version of hankel_gradient."""
    n = len(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(P * np.conj(A), fft_size)
    return 2.0 * corr[:n]

def symmetric_project(v, n):
    """Project v onto the symmetric simplex: a[j] = a[n-1-j], a >= 0, sum = 1."""
    half = (n + 1) // 2
    # Aggregate: for each pair (j, n-1-j), the sum constraint is a[j] + a[n-1-j] = 2*a[j]
    # So we project onto simplex of size 'half' with the understanding that
    # the full sum is 2*sum(half_part) - (middle if n odd) = 1
    # Actually simpler: just build the symmetric vector and project
    sym = np.zeros(n)
    for j in range(half):
        sym[j] = v[j]
        sym[n - 1 - j] = v[j]
    # Project sym onto simplex
    return project_simplex(sym)


def solve_lp_oracle(f, M, time_per_lp=5.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g (sum normalized to 1) or None."""
    n = len(f)
    c_obj = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[j] * g[k-j]... 
    # Actually (f*g)[k] = sum_j f[k-j] * g[j] = sum_j g[k-j] * f[j]
    # For each k: sum_j f[max(0, k-n+1)..min(n-1,k)][j - max(0,k-n+1)] * g[j] <= M
    # Simpler: (f*g)[k] = sum_{j=0}^{n-1} g[j] * f[k-j] where f[m]=0 if m<0 or m>=n
    # So the constraint matrix A has A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] contributes to (f*g)[k] for k = j, j+1, ..., j+n-1
        # (f*g)[k] gets contribution g[j]*f[k-j]
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_per_lp, 'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None

def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_gradient(p, a):
    """Compute grad[j] = 2 * sum_k p[k] * a[k-j] for j=0..n-1."""
    n = len(a)
    corr_full = np.correlate(p, a, mode='full')
    return 2.0 * corr_full[:n]


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def solve_lp_oracle(f, M, time_per_lp=5.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_per_lp, 'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None
