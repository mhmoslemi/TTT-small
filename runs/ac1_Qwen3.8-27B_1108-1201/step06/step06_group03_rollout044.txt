def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    for j in range(js, je + 1):

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (f*g) = A @ g."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M, n):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def compute_balance_vector(a, active_peaks, weights):
    """Compute S[j] = sum_k w_k * a[k-j] for all j."""
    n = len(a)
    S = np.zeros(n)
    for idx, k in enumerate(active_peaks):
        w = weights[idx]
        if w < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            S[j] += w * a[k - j]
    return S

def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n < 2:
        return np.ones(new_n) / new_n
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s

        try_seq(parent, "uniform_768")

    n_parent = len(parent)

    # === Phase 1: Quick exploration (200s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_phase = time.monotonic()

    # Try a few lengths
    lengths = sorted(set([n_parent, max(256, n_parent - 32), n_parent + 32,
                          512, 640, 768, 896, 1024, 1152]))
    lengths = [l for l in lengths if 128 <= l <= 1280]

    for n_try in lengths:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 10:
            break

        starts = generate_diverse_starts(n_try, parent, rng)
        # Try top 3 starts with quick FW
        for si, start in enumerate(starts[:3]):
            remaining = deadline - time.monotonic() - 1150
            if remaining < 5:
                break
            t_alloc = min(remaining, 15)
            a_opt, _ = fw_optimize_improved(start, time_limit=t_alloc)
            try_seq(a_opt, f"expl_n{n_try}_s{si}")

    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW on best (500s) ===
    print("\n=== Phase 2: Deep FW ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Main deep FW
        remaining = deadline - time.monotonic() - 700
        if remaining > 30:
            t_alloc = min(remaining, 400)
            print(f"  Deep FW n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize_improved(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_fw")

        # Symmetric optimization
        remaining = deadline - time.monotonic() - 500
        if remaining > 20:
            t_alloc = min(remaining, 60)
            print(f"  Symmetric opt, t={t_alloc:.0f}s")
            a_opt, _ = symmetric_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "symmetric")

        # Perturbed restarts
        for scale in [0.001, 0.003, 0.01]:
            remaining = deadline - time.monotonic() - 400
            if remaining < 20:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 50)
            a_opt, _ = fw_optimize_improved(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"fw_pert_{scale}")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Balance-condition multiplicative updates (400s) ===
    print("\n=== Phase 3: Balance Updates ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple configurations
        configs = [
            {'lr_init': 0.05, 'delta': 0.02, 'power': 2.0},
            {'lr_init': 0.03, 'delta': 0.01, 'power': 3.0},
            {'lr_init': 0.08, 'delta': 0.05, 'power': 1.5},
            {'lr_init': 0.02, 'delta': 0.005, 'power': 4.0},
        ]

        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 350
            if remaining < 15:
                break
            t_alloc = min(remaining, 80)
            print(f"  Balance config {ci} (lr={cfg['lr_init']}, delta={cfg['delta']}), t={t_alloc:.0f}s")
            a_opt, _ = balance_update(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"balance_{ci}")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Subgradient refinement (250s) ===
    print("\n=== Phase 4: Subgradient ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        configs = [
            {'T_init': 0.0003, 'lr_init': 0.01},
            {'T_init': 0.001, 'lr_init': 0.008},
            {'T_init': 0.003, 'lr_init': 0.005},
        ]
        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 250
            if remaining < 15:
                break
            t_alloc = min(remaining, 70)
            print(f"  Subgrad config {ci}, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize_v2(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"subgrad_{ci}")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Peak coordinate descent (250s) ===
    print("\n=== Phase 5: Peak Coord Descent ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 150
        if remaining > 15:
            t_alloc = min(remaining, 150)
            print(f"  Peak coord descent, t={t_alloc:.0f}s")
            a_opt, _ = peak_coordinate_descent_v2(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_coord")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Final alternating polish (remaining) ===
    print("\n=== Phase 6: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        round_num = 0
        while time.monotonic() < deadline - 5:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            if round_num % 4 == 1:
                # Tiny perturb + FW
                noise = rng.standard_normal(n_best) * 0.0005
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 20)
                a_opt, _ = fw_optimize_improved(start, time_limit=t_alloc)
                try_seq(a_opt, f"polish_fw_{round_num}")
            elif round_num % 4 == 2:
                # Balance update
                t_alloc = min(remaining, 20)
                a_opt, _ = balance_update(best_seq, time_limit=t_alloc,
                                          lr_init=0.02, delta=0.01, power=3.0)
                try_seq(a_opt, f"polish_bal_{round_num}")
            elif round_num % 4 == 3:
                # Subgradient
                t_alloc = min(remaining, 15)
                a_opt, _ = subgradient_optimize_v2(best_seq, time_limit=t_alloc,
                                                   T_init=0.0002, lr_init=0.005)
                try_seq(a_opt, f"polish_sg_{round_num}")
            else:
                # Coord descent
                t_alloc = min(remaining, 15)
                a_opt, _ = peak_coordinate_descent_v2(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"polish_cd_{round_num}")

    print(f"  Phase 6 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
