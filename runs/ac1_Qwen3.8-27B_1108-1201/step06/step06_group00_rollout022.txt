    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_minimax_lp(f, M):
    """Solve the minimax LP:
    minimize t
    subject to (f*g)[k] <= t for all k
               sum(g) = 1
               g >= 0, t >= 0
    Returns the optimal g (normalized to sum=1) or None.
    """
    n = len(f)
    size = 2 * n - 1
    
    # Build A_ub: size x (n+1) matrix
    # A_ub[k, j] = f[k-j] for valid j, A_ub[k, n] = -1
    rows = []
    cols = []
    vals = []
    
    for k in range(size):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
        # Add the -1 for t
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(size, n + 1))
    b_ub = np.zeros(size)
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Bounds: g_i >= 0, t >= 0
    bounds = [(0.0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success and res.x is not None:
            g = np.maximum(res.x[:n], 0.0)
            gs = g.sum()
            if gs > 1e-15:
                g /= gs
                return g
    except Exception:
        pass
    return None

Actually, I realize I should also think about the "peak carving" phase more carefully. After FW converges, the sequence is at a local minimum (or near it). To further improve, we can try:

1. **Coordinate descent**: For each coordinate i, try to change a[i] (while keeping the sum constant) to reduce the max peak.
2. **Pair swap**: For each pair (i, j), try to swap some mass between a[i] and a[j].
3. **Block move**: Move a block of consecutive elements.

The coordinate descent approach:
- For coordinate i, the constraint is that the sum stays 1. So if we increase a[i] by δ, we must decrease some other coordinate by δ.
- The most effective approach: for each pair (i, j), try to move δ from i to j (or vice versa) and check if the max peak decreases.

This is O(n^2) per sweep, and each evaluation is O(n log n) (for the FFT convolution). For n=768, that's about 590K evaluations per sweep, each taking ~10μs (for the FFT), so ~6 seconds per sweep. That's too slow for multiple sweeps.

A better approach: only evaluate the peaks that are close to the maximum. If we have 10 active peaks, we only need to check those 10 values after each change. Each check is O(n) (computing the convolution at a specific lag), so 10 * 768 = 7680 operations per evaluation. That's much faster.

    n = len(f)

    best_obj = conv.max()
    best_f = f.copy()

    top_k = np.argpartition(conv, -n_top_peaks)[-n_top_peaks:]
    
    sweep = 0
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        # Update top peaks
        conv = fast_convolve(f)
        top_k = np.argpartition(conv, -n_top_peaks)[-n_top_peaks:]
        current_max = conv.max()
        
        made_progress = False

        for i in range(n):
