"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent (multiplicative weights) with adaptive temperature,
lag-targeted quadratic optimization, and simulated annealing.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0]*a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad_smoothed(a, T):
    """Compute the gradient of the smoothed objective T*log(sum exp(c_k/T))."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()
    # grad_j = 2 * sum_k p[k] * a[k-j] (where valid)
    grad = np.zeros(n)
    for k in range(len(conv)):
        if p[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # a[k-j] for j in [js, je]
        partners = k - np.arange(js, je + 1)
        grad[js:je+1] += 2.0 * p[k] * a[partners]
    return grad, conv, p

def mirror_descent(a0, time_limit, T_init=0.001, T_min=0.0001, lr_init=0.02):
    """Mirror descent (multiplicative weights) with adaptive temperature."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        grad, conv, p = compute_grad_smoothed(a, T)
        
        # Multiplicative update
        log_update = -lr * grad
        # Clip to avoid overflow
        log_update = np.clip(log_update, -50, 50)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        obj = fast_convolve(a_new).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive temperature and learning rate
        if no_improve > 50:
            T = max(T_min, T * 0.95)
            lr *= 0.98
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            noise = rng.standard_normal(n) * 0.002
            a = project_simplex(best_a + noise)
            T = T_init * 0.5
            lr = lr_init * 0.7
            no_improve = 0
    
    return best_a, best_obj

def lag_targeted_descent(a0, time_limit):
    """Target the maximum lag and try to reduce it via pairwise adjustments."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]
        
        # Find pairs (j, k_star - j) that contribute to the peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        
        # Try to reduce the peak by adjusting pairs
        improved = False
        
        # Sample some pairs to try
        candidates = list(range(js, je + 1))
        if len(candidates) > 40:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
            candidates = list(rng.choice(candidates, size=40, replace=False))
        
        for j in candidates:
            l = k_star - j
            if l < 0 or l >= n or l == j:
                continue
            if a[j] < 1e-10 or a[l] < 1e-10:
                continue
            
            # Try moving mass from the larger to the smaller
            # This reduces 2*a[j]*a[l] (the contribution to b[k_star])
            # but may increase other b[k] values
            if a[j] > a[l]:
                src, dst = j, l
            else:
                src, dst = l, j
            
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                delta = a[src] * frac
                a_test = a.copy()
                a_test[src] -= delta
                a_test[dst] += delta
                a_test = np.maximum(a_test, 0.0)
                a_test = a_test / a_test.sum()
                c_new = fast_convolve(a_test)
                new_max = c_new.max()
                
                if new_max < best_obj - 1e-14:
                    a = a_test
                    best_obj = new_max
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Try a different strategy: reduce the peak by adding mass
            # to positions that don't contribute to it
            if stall > 5:
                # Find positions that don't contribute to k_star
                contributing = set(range(js, je + 1)) | set(k_star - j for j in range(js, je + 1))
                non_contributing = [j for j in range(n) if j not in contributing and a[j] > 0]
                
                if non_contributing:
                    # Move mass from a contributing position to a non-contributing one
                    src_candidates = [j for j in range(js, je + 1) if a[j] > 1e-5]
                    if src_candidates:
                        rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                        src = int(rng.choice(src_candidates))
                        dst = int(rng.choice(non_contributing))
                        for frac in [0.02, 0.05, 0.1]:
                            delta = a[src] * frac
                            a_test = a.copy()
                            a_test[src] -= delta
                            a_test[dst] += delta
                            a_test = np.maximum(a_test, 0.0)
                            a_test = a_test / a_test.sum()
                            c_new = fast_convolve(a_test)
                            if c_new.max() < best_obj - 1e-14:
                                a = a_test
                                best_obj = c_new.max()
                                best_a = a.copy()
                                stall = 0
                                break
                
                if stall > 30:
                    # Random perturbation
                    rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                    noise_scale = 0.001 * (1 + stall / 50.0)
                    a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
                    stall = 0
        else:
            stall = 0
        
        if stall > 200:
            break
    
    return best_a, best_obj

def simulated_annealing(a0, time_limit, T_init=0.01, T_min=0.0001, cool_rate=0.999):
    """Simulated annealing with structured moves."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    current_obj = best_obj
    
    T = T_init
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        rng = np.random.default_rng((it * 62771 + 3) % (2**31))
        
        # Generate a move
        move_type = rng.integers(0, 4)
        
        if move_type == 0:
            # Single element perturbation
            a_test = a.copy()
            j = int(rng.integers(n))
            delta = rng.standard_normal() * 0.01
            a_test[j] = max(0, a_test[j] + delta)
        elif move_type == 1:
            # Pair transfer
            j1 = int(rng.integers(n))
            j2 = int(rng.integers(n))
            if j1 == j2:
                continue
            frac = rng.uniform(0.01, 0.2)
            a_test = a.copy()
            delta = a_test[j1] * frac
            a_test[j1] -= delta
            a_test[j2] += delta
        elif move_type == 2:
            # Small block shift
            block_size = int(rng.integers(1, min(5, n // 4)))
            start = int(rng.integers(max(1, n - block_size)))
            a_test = a.copy()
            a_test[start:start+block_size] = a_test[start-1:start-1+block_size]
            a_test[start-1] = 0  # or distribute
        else:
            # Gaussian noise on a few elements
            n_perturb = int(rng.integers(1, min(5, n // 10)))
            indices = rng.choice(n, size=n_perturb, replace=False)

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_fft(a, T):
    """Compute gradient of smoothed objective using FFT cross-correlation.
    grad_j = 2 * sum_k p[k] * a[k-j] where p[k] = softmax(conv/T)
    Returns (grad, conv, p, peak_lag).

    A = np.fft.rfft(a, L)

                a_test[src] -= delta
                a_test[dst] += delta
                a_test = np.maximum(a_test, 0.0)
                a_test = a_test / a_test.sum()
                c_new = fast_convolve(a_test)
                new_max = c_new.max()
                gain = best_obj - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_move = (src, dst, frac)

        if best_move is not None and best_gain > 1e-14:
            src, dst, frac = best_move
            delta = a[src] * frac
            a[src] -= delta
            a[dst] += delta
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            best_obj = fast_convolve(a).max()
            best_a = a.copy()
            improved = True
            stall = 0
        else:
            stall += 1

        # Strategy 2: Move mass to non-contributing positions
        if not improved and stall >= 3:
            contributing = set(range(js, je + 1))
            non_contrib = [j for j in range(n) if j not in contributing and a[j] >= 0]

            if non_contrib:
                src_candidates = [j for j in range(js, je + 1) if a[j] > 1e-6]
                if src_candidates:
                    rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31 - 1))
                    for _ in range(min(10, len(src_candidates))):
                        src = int(rng.choice(src_candidates))
                        dst = int(rng.choice(non_contrib))
                        for frac in (0.05, 0.1, 0.2):
                            delta = a[src] * frac
                            a_test = a.copy()
                            a_test[src] -= delta
                            a_test[dst] += delta
                            a_test = np.maximum(a_test, 0.0)
                            a_test = a_test / a_test.sum()
                            c_new = fast_convolve(a_test)
                            if c_new.max() < best_obj - 1e-14:
                                a = a_test
                                best_obj = c_new.max()
                                best_a = a.copy()
                                stall = 0
                                improved = True
                                break
                        if improved:
                            break

        if not improved:
            # Random restart perturbation
            if stall > 20:
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31 - 1))
                noise_scale = 0.001 * (1.0 + min(stall / 40.0, 3.0))
                a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
                stall = 0

        if stall > 300:
            break

    return best_a, best_obj


def simulated_annealing(a0, time_limit, T_init=0.008, T_min=0.00005, cool_rate=0.9995):
    """Simulated annealing with structured moves targeting peak reduction."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    current_obj = best_obj

    T = T_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        rng = np.random.default_rng((it * 15485863 + 5) % (2**31 - 1))
        move_type = rng.integers(0, 5)

        a_test = a.copy()

        if move_type == 0:
            # Peak-directed transfer
            conv = fast_convolve(a)
            k_star = int(np.argmax(conv))
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            src = int(rng.integers(js, je + 1))
            # Move to a non-peak-contributing position
            contributing = set(range(js, je + 1))
            non_contrib = [j for j in range(n) if j not in contributing]
            if non_contrib:
                dst = int(rng.choice(non_contrib))
            else:
                dst = int(rng.integers(n))
            frac = rng.uniform(0.02, 0.15)
            delta = a_test[src] * frac
            a_test[src] -= delta
            a_test[dst] += delta

        elif move_type == 1:
            # Pair swap
            j1 = int(rng.integers(n))
            j2 = int(rng.integers(n))
            if j1 != j2:
                a_test[j1], a_test[j2] = a_test[j2], a_test[j1]

        elif move_type == 2:
            # Small Gaussian perturbation on 1-3 elements
            n_pert = int(rng.integers(1, 4))
            indices = rng.choice(n, size=n_pert, replace=False)
            a_test[indices] += rng.standard_normal(n_pert) * 0.008

        elif move_type == 3:
            # Block shift (move a small block by 1-3 positions)
            block_len = int(rng.integers(1, min(4, n // 8) + 1))
            start = int(rng.integers(1, max(2, n - block_len - 1)))
            shift = int(rng.integers(1, 4)) * (1 if rng.random() > 0.5 else -1)
            new_start = start + shift
            if 0 <= new_start and new_start + block_len <= n:
                a_test = a_test.copy()
                block = a_test[start:start + block_len].copy()
                a_test[start:start + block_len] = a_test[start + shift:start + shift + block_len]
                a_test[start + shift:start + shift + block_len] = block

        else:
            # Entropy-based perturbation (inverse temperature move)
            j = int(rng.integers(n))
            k = int(rng.integers(n))
            if j != k:
                transfer = min(a_test[j], a_test[k]) * rng.uniform(0.1, 0.5)
                a_test[j] -= transfer
                a_test[k] += transfer

        a_test = np.maximum(a_test, 0.0)
        s = a_test.sum()
        if s < 1e-15:
            continue
        a_test = a_test / s

        new_obj = fast_convolve(a_test).max()
        delta_obj = new_obj - current_obj

        if delta_obj < 0 or (T > T_min and rng.random() < np.exp(-delta_obj / (T + 1e-15))):
            a = a_test
            current_obj = new_obj
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_a = a.copy()
                no_improve = 0
        else:
            no_improve += 1

        T = max(T_min, T * cool_rate)

        # Reheat if stuck
        if no_improve > 200:
            T = T_init * 0.2
            no_improve = 0

    return best_a, best_obj


def directed_gradient_descent(a0, time_limit, lr=0.01, max_k=20):
    """Gradient descent using only top-k peak subgradients with exact line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        k_sel = np.argpartition(conv, -max_k)[-max_k:]
        pv = conv[k_sel]
        weights = pv / pv.sum()

        # Subgradient: gradient of max approximated by weighted sum of peak gradients
        sg = np.zeros(n)
        for i in range(len(k_sel)):
            if weights[i] < 1e-12:
                continue
            k = int(k_sel[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            partners = k - np.arange(js, je + 1)
            sg[js:je + 1] += weights[i] * 2.0 * a[partners]

        # Line search
        best_candidate = a
        best_val = best_obj
        for trial_lr in (lr, lr * 0.5, lr * 0.25, lr * 0.1):
            candidate = project_simplex(a - trial_lr * sg)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate

        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.05, 0.1)
        else:
            no_improve += 1
            lr *= 0.95

        if no_improve > 80:
            rng = np.random.default_rng((it * 31 + 7) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            lr = 0.01
            no_improve = 0

    return best_a, best_obj


def diverse_init(n, seed):
    """Generate diverse initial sequences with different structural patterns."""
    rng = np.random.default_rng(seed)
    pattern = seed % 7

    if pattern == 0:
        a = np.ones(n)
    elif pattern == 1:
        # Asymmetric ramp
        a = np.linspace(0.3, 2.0, n)
    elif pattern == 2:
        # Bimodal (peaks at quartiles)
        a = np.ones(n)
        peak1 = n // 4
        peak2 = 3 * n // 4
        a[peak1] = 2.0
        a[peak2] = 2.0
        # Smooth
        kernel = np.ones(15) / 15
        a = np.convolve(a, kernel, mode='same')
    elif pattern == 3:
        # Anti-palindromic tendency
        a = np.linspace(2.0, 0.3, n)
    elif pattern == 4:
        # Periodic modulation
        a = 1.0 + 0.4 * np.sin(2 * np.pi * 3 * np.arange(n) / n)
    elif pattern == 5:
        # Random smooth
        noise = rng.standard_normal(n)
        kernel = np.ones(20) / 20
        noise = np.convolve(noise, kernel, mode='same')
        a = 1.0 + 0.5 * noise
    else:
        a = np.abs(rng.standard_normal(n)) + 0.2

    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a
