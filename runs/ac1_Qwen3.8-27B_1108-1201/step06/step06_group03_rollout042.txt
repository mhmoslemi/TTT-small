"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth max (log-sum-exp) gradient descent with adaptive temperature
2. Frank-Wolfe with LP oracle for polishing
3. Multi-length exploration with resampling
4. Peak equalization via active-set quadratic methods
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 1 << (2*n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_gradient(a, T):
    """Compute gradient of smooth max of conv(a,a)."""
    n = len(a)
    conv = fast_convolve(a)
    conv_max = conv.max()
    # p = softmax(T * conv)
    exp_T_conv = np.exp(T * (conv - conv_max))
    p = exp_T_conv / exp_T_conv.sum()
    
    # grad[j] = 2 * sum_m p[j+m] * a[m] = 2 * cross_correlate(p, a)[j]
    N = 1 << (3*n - 2).bit_length()
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    grad = 2.0 * np.fft.irfft(P * A.conj(), N)[:n]
    
    return grad, conv.max()


def smooth_max_optimize(a0, time_limit, T_start=50.0, T_end=2000.0, verbose=False):
    """Minimize max(conv(a,a)) using smooth max + projected gradient with line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    stagnant = 0
    
    # Use Adam-like optimizer
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 0.1:
            break
        
        # Anneal temperature
        progress = min(1.0, elapsed / (time_limit * 0.7))
        T = T_start * (T_end / T_start) ** (progress ** 0.5)
        
        # Base learning rate (decreasing)
        lr = 0.05 / (1.0 + 0.001 * it)
        
        grad, _ = smooth_max_gradient(a, T)
        
        # Adam update
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad**2
        m_hat = m_v / (1 - beta1**it)
        v_hat = v_v / (1 - beta2**it)
        adam_step = m_hat / (np.sqrt(v_hat) + eps)
        
        # Line search with Adam direction
        best_val = fast_convolve(a).max()
        best_alpha = 0.0
        base_lr = lr / (np.max(np.abs(adam_step)) + 1e-8)
        for alpha in [base_lr, base_lr*0.5, base_lr*0.25, base_lr*0.1]:
            candidate = project_simplex(a - alpha * adam_step)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            a = project_simplex(a - best_alpha * adam_step)
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1
        
        # Periodic perturbation if stuck
        if stagnant % 50 == 50:
            rng = np.random.default_rng(it * 7919 + 3)
            noise_level = 0.001 + 0.003 * min(1.0, stagnant / 200.0)
            a = project_simplex(a + rng.standard_normal(n) * noise_level)
            m_v = np.zeros(n)
            v_v = np.zeros(n)
            stagnant = 0
        
        if stagnant > 500:
            break
        
        if verbose and it % 50 == 0:
            print(f"    SmoothMax it={it}, T={T:.1f}, obj={best_obj:.10f}, stagnant={stagnant}")
    
    return best_a, best_obj


def build_conv_matrix(f, n):
    A = np.zeros((2*n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A


def solve_lp_oracle(f, M):
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

            f = np.maximum(f, 0.0)

            conv = fast_convolve(f)

def subgradient_optimize(a0, time_limit, T_init=0.0005, lr_init=0.02):
    """Smoothed subgradient with momentum (top-k peaks)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(8, int(n * 0.05))
    
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.9
    
    t0 = time.monotonic()
    stagnant = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))
        
        pv = conv[peaks]
        pv_max = pv.max()
        w = np.exp((pv - pv_max) / (T * 0.1))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += w[i] * 2.0 * a[k - idx_arr]
        
        v = mom * v - lr * grad
        best_candidate = a
        best_val = conv.max()
        
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            candidate = project_simplex(a + frac * v)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
        
        a = best_candidate
        obj = fast_convolve(a).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 30:
            T *= 1.05
            lr *= 0.98
        if stagnant > 100:
            T *= 0.5
            lr *= 1.2
            mom = 0.93
        if stagnant > 300:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise = 0.002 + 0.003 * min(1.0, (stagnant - 300) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            v = np.zeros(n)
            mom = 0.9
            stagnant = 0
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def peak_equalization(a0, time_limit):
    """Try to equalize the top peaks by solving a small QP."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        # Top 5 peaks
        n_peaks = min(5, len(conv))
        peak_idx = np.argsort(conv)[::-1][:n_peaks]
        
        improved = False
        for k in peak_idx:
            k = int(k)
            json_ = max(0, k - n + 1)
            jend = min(n - 1, k)
            idx = np.arange(json_, jend + 1)
            
            # For peak k: conv[k] = sum_{j in idx} a[j] * a[k-j]
            # Try to reduce it by shifting mass from high contributors to low ones
            pairs = [(a[j] * a[k-j], j, k-j) for j in idx]
            pairs.sort(reverse=True)
            
            for contrib, j1, j2 in pairs[:8]:
                if a[j1] < 1e-7:
                    continue
                for step_frac in [0.01, 0.005, 0.002, 0.001]:
                    for src in [j1, j2]:
                        if a[src] < 1e-7:
                            continue
                        delta = a[src] * step_frac
                        # Move to a random distant location
                        rng = np.random.default_rng((it * 9973 + k * 31 + src * 7 + int(step_frac*100000)) % (2**31))
                        j_target = int(rng.integers(0, n))
                        if abs(j_target - src) < n // 10:
                            continue
                        
                        candidate = a.copy()
                        candidate[src] -= delta
                        candidate[j_target] += delta
                        if candidate[src] < -1e-12:
                            continue
                        candidate = np.maximum(candidate, 0)
                        candidate = candidate / candidate.sum()
                        val = fast_convolve(candidate).max()
                        if val < best_obj - 1e-15:
                            a = candidate
                            best_obj = val
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            
            if improved:
                break
        
        if not improved:
            # Try a small random perturbation
            rng = np.random.default_rng(it * 48271)
            candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
            val = fast_convolve(candidate).max()
            if val < best_obj - 1e-15:
                a = candidate
                best_obj = val
                best_a = a.copy()
                improved = True
        
        if not improved and it > 500:
            break
    
    return best_a, best_obj


def generate_starts(n, parent, rng):
    starts = []
    if parent is not None:
        starts.append(resample_sequence(parent, n))
    starts.append(np.ones(n) / n)
    
    x = np.arange(n, dtype=np.float64)
    # Cosine taper
    cos_t = 0.5 * (1 + np.cos(np.pi * x / max(n - 1, 1)))
    w = np.maximum(cos_t, 0.01)
    starts.append(w / w.sum())
    
    # Gaussian
    g = np.exp(-0.5 * ((x - (n - 1) / 2) / (0.2 * n)) ** 2)
    starts.append(g / g.sum())
    
    # Edge
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    starts.append(edge / edge.sum())
    
    # Bump
    bump = np.exp(-0.5 * (2 * x / (n - 1) - 1) ** 4)
    starts.append(bump / bump.sum())
    
    # Random Dirichlet
    starts.append(rng.dirichlet(np.ones(n) * 0.5))
    starts.append(rng.dirichlet(np.ones(n) * 0.2))
    
    return starts

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_gradient(a, T):
    """Compute gradient of smooth max of conv(a,a) and the current max."""
    n = len(a)
    conv = fast_convolve(a)

    N = 1 << (3 * n - 2).bit_length()
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    grad = 2.0 * np.fft.irfft(P * A.conj(), N)[:n]
    return grad, conv_max, conv

        pass

            m_relax = 1.0005

        if verbose and it % 50 == 0:
            print(f"    FW it={it}, obj={best_obj:.10f}, stg={stagnant}")

    return best_f, best_obj


def subgradient_optimize(a0, time_limit, T_init=0.0005, lr_init=0.02):
    """Top-k smoothed subgradient with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(10, int(n * 0.06))

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.9
    t0 = time.monotonic()
    stagnant = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)

        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))

        pv = conv[peaks]
        pv_max = pv.max()
        w = np.exp((pv - pv_max) / (T * 0.1))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += w[i] * 2.0 * a[k - idx_arr]

        v = mom * v - lr * grad
        best_candidate = a
        best_val = conv.max()

        for frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            candidate = project_simplex(a + frac * v)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 25:
            T *= 1.06
            lr *= 0.98
        if stagnant > 80:
            T *= 0.5
            lr *= 1.3
            mom = 0.93
        if stagnant > 250:
            rng = np.random.default_rng((it * 31 + 7) % (2 ** 31))
            noise = 0.002 + 0.003 * min(1.0, (stagnant - 250) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            v = np.zeros(n)
            mom = 0.9
            stagnant = 0

    return best_a, best_obj


def peak_equalization(a0, time_limit):
    """Targeted mass redistribution to reduce top peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    step_scale = 0.008

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        n_peaks = min(6, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        improved = False
        for ki in range(n_peaks):
            if improved:
                break
            k = int(peak_indices[ki])
            json_ = max(0, k - n + 1)
            jend = min(n - 1, k)

            if json_ > jend:
                continue

            pairs = []
            for j in range(json_, jend + 1):
                pairs.append((a[j] * a[k - j], j, k - j))
            pairs.sort(reverse=True)

            for contrib, j1, j2 in pairs[:10]:
                if improved:
                    break
                if a[j1] < 1e-7:
                    continue
                for step_frac in [step_scale, step_scale * 0.5, step_scale * 0.25]:
                    if improved:
                        break
                    for src in [j1, j2]:
                        if improved:
                            break
                        if a[src] < 1e-7:
                            continue
                        delta = a[src] * step_frac
                        rng = np.random.default_rng((it * 9973 + k * 31 + src * 7 + int(step_frac * 1e6)) % (2 ** 31))
                        for _ in range(5):
                            j_target = int(rng.integers(0, n))
                            if abs(j_target - src) < max(1, n // 15):
                                continue
                            candidate = a.copy()
                            candidate[src] -= delta
                            candidate[j_target] += delta
                            if candidate[src] < -1e-12:
                                continue
                            candidate = np.maximum(candidate, 0)
                            candidate = candidate / candidate.sum()
                            val = fast_convolve(candidate).max()
                            if val < best_obj - 1e-15:
                                a = candidate
                                best_obj = val
                                best_a = a.copy()
                                improved = True
                                break

        if not improved:
            step_scale *= 0.94
            if step_scale < 5e-5:
                break
            if it % 150 == 0:
                rng = np.random.default_rng(it * 48271)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.002)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
            elif it > 3000:
                break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_starts(n, parent, rng):
    """Generate diverse initializations."""
    starts = []
    if parent is not None:
        starts.append(resample_sequence(parent, n))
    starts.append(np.ones(n) / n)

    x = np.arange(n, dtype=np.float64)
    cos_t = 0.5 * (1.0 + np.cos(np.pi * x / max(n - 1, 1)))
    w = np.maximum(cos_t, 0.01)
    starts.append(w / w.sum())

    g = np.exp(-0.5 * ((x - (n - 1) / 2) / (0.2 * n)) ** 2)
    starts.append(g / g.sum())

    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    starts.append(edge / edge.sum())

    bump = np.exp(-0.5 * (2.0 * x / max(n - 1, 1) - 1.0) ** 4)
    starts.append(bump / bump.sum())

    starts.append(rng.dirichlet(np.ones(n) * 0.5))
    starts.append(rng.dirichlet(np.ones(n) * 0.2))
    return starts


def shared_best():
    """Global state holder to avoid closures."""
    return {'seq': None, 'score': float('inf')}


def try_seq(state, a, label=""):
    """Evaluate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state['score']:
        state['score'] = score
        state['seq'] = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search combining smooth max gradient, Frank-Wolfe,
    subgradient, and peak equalization across multiple lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    state = shared_best()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(state, parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(state, parent, "uniform_768")

    n_parent = len(parent)

    # === Phase 1: Multi-length smooth max (350s) ===
    print("\n=== Phase 1: Smooth Max Multi-Length ===")
    t_phase = time.monotonic()

    lengths = sorted(set([n_parent, max(256, n_parent - 128), n_parent + 128,
                          512, 640, 768, 896, 1024, 1280]))
    lengths = [l for l in lengths if 128 <= l <= 2048]

    for n_try in lengths:
        remaining = deadline - time.monotonic() - 1000
        if remaining < 30:
            break
        starts = generate_starts(n_try, parent, rng)
        for si, start in enumerate(starts[:3]):
            remaining = deadline - time.monotonic() - 950
            if remaining < 15:
                break
            t_alloc = min(remaining, 45)
            a_opt, _ = smooth_max_optimize(start, time_limit=t_alloc)
            try_seq(state, a_opt, f"sm_n{n_try}_s{si}")

    print(f"  Phase 1 done: best={state['score']:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW + Smooth Max (500s) ===
    print("\n=== Phase 2: Deep Optimization ===")
    t_phase = time.monotonic()

    if state['seq'] is not None:
        n_best = len(state['seq'])

        remaining = deadline - time.monotonic() - 400
        if remaining > 30:
            t_alloc = min(remaining, 200)
            print(f"  Smooth max deep n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = smooth_max_optimize(state['seq'], time_limit=t_alloc,
                                           T_start=100, T_end=5000)
            try_seq(state, a_opt, "sm_deep")

        remaining = deadline - time.monotonic() - 200
        if remaining > 25:
            t_alloc = min(remaining, 180)
            print(f"  FW deep n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(state['seq'], time_limit=t_alloc, verbose=True)
            try_seq(state, a_opt, "fw_deep")

        # Perturbed restarts
        for scale in [0.002, 0.005, 0.01]:
            remaining = deadline - time.monotonic() - 100
            if remaining < 20:
                break
            f_pert = project_simplex(state['seq'] + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 50)
            a_opt, _ = fw_optimize(f_pert, time_limit=t_alloc)
            try_seq(state, a_opt, f"fw_pert_{scale}")

    print(f"  Phase 2 done: best={state['score']:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Subgradient + Peak Equalization (250s) ===
    print("\n=== Phase 3: Subgradient + Peak EQ ===")
    t_phase = time.monotonic()

    if state['seq'] is not None:
        n_best = len(state['seq'])

        remaining = deadline - time.monotonic() - 150
        if remaining > 20:
            t_alloc = min(remaining, 100)
            print(f"  Subgradient t={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize(state['seq'], time_limit=t_alloc)
            try_seq(state, a_opt, "subgrad")

        remaining = deadline - time.monotonic() - 80
        if remaining > 15:
            t_alloc = min(remaining, 80)
            print(f"  Peak EQ t={t_alloc:.0f}s")
            a_opt, _ = peak_equalization(state['seq'], time_limit=t_alloc)
            try_seq(state, a_opt, "peak_eq")

    print(f"  Phase 3 done: best={state['score']:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Alternating polish (remaining) ===
    print("\n=== Phase 4: Final Polish ===")
    t_phase = time.monotonic()

    if state['seq'] is not None:
        n_best = len(state['seq'])
        round_num = 0
        while time.monotonic() < deadline:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break

            if round_num % 4 == 1:
                t_alloc = min(remaining, 60)
                a_opt, _ = smooth_max_optimize(state['seq'], time_limit=t_alloc,
                                               T_start=200, T_end=4000)
                try_seq(state, a_opt, f"pol_sm_{round_num}")
            elif round_num % 4 == 2:
                t_alloc = min(remaining, 60)
                a_opt, _ = fw_optimize(state['seq'], time_limit=t_alloc)
                try_seq(state, a_opt, f"pol_fw_{round_num}")
            elif round_num % 4 == 3:
                t_alloc = min(remaining, 45)
                a_opt, _ = peak_equalization(state['seq'], time_limit=t_alloc)
                try_seq(state, a_opt, f"pol_peak_{round_num}")
            else:
                t_alloc = min(remaining, 45)
                a_opt, _ = subgradient_optimize(state['seq'], time_limit=t_alloc,
                                                T_init=0.0002, lr_init=0.01)
                try_seq(state, a_opt, f"pol_sg_{round_num}")

    print(f"  Phase 4 done: best={state['score']:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if state['seq'] is None:
        state['seq'] = np.ones(768) / 768
        state['score'] = float(evaluate_sequence(state['seq'].tolist()))

    # Final check
    fs = float(evaluate_sequence(state['seq'].tolist()))
    if np.isfinite(fs) and fs < state['score']:
        state['score'] = fs
    print(f"\nFINAL SCORE: {state['score']:.10f} (n={len(state['seq'])})")
    return [float(x) for x in state['seq']]
