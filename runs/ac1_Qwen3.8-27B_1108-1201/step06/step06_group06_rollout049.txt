Multi-phase optimization for minimum peak self-convolution of non-negative sequences.
Uses Nesterov-accelerated smoothed gradient descent, peak-targeted local search,
and simulated annealing. Explores multiple sequence lengths.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear convolution a*a via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    ind = np.nonzero(cond)[0][-1]
    theta = (cssv[ind] - 1.0) / (ind + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(a, T):
    """Compute (smoothed_obj, weights, gradient) for T*log(sum(exp(c/T)))."""
    n = len(a)
    c = fast_convolve(a)
    size_c = len(c)

    max_c = c.max()
    shifted = (c - max_c) / T
    w = np.exp(shifted)
    w_sum = w.sum()
    w = w / w_sum
    
    obj = T * (max_c / T + np.log(w_sum))

    fft_len = 1 << (size_c + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, fft_len)
    A = np.fft.rfft(a, fft_len)
    corr = np.fft.irfft(W * np.conj(A), fft_len)[:n]
    grad = 2.0 * corr
    
    return obj, w, grad, c

        # Momentum update
        v = mu * v + lr * grad
        a_new = project_simplex(a_look - v)
        
        # Evaluate actual peak
        new_peak = fast_convolve(a_new).max()
        
        a = a_new
        if new_peak < best_peak - 1e-14:
            best_peak = new_peak
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive parameters
        if no_improve == 30:
            T *= 1.02
            lr *= 0.99
        elif no_improve == 80:
            T *= 0.8
            lr *= 0.7
            v *= 0.5
        elif no_improve == 150:
            T = max(T_init * 0.5, 0.0001)
            lr = lr_init * 0.3
            v = np.zeros(n)
        elif no_improve == 250:
            rng = np.random.default_rng((it * 104729 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            best_peak = fast_convolve(a).max()
            T = T_init
            lr = lr_init
            v = np.zeros(n)
            no_improve = 0
        
        # Gradually decrease T for convergence
        if it % 500 == 0 and T > 0.0001:
            T *= 0.95
    
    return best_a, best_peak


def peak_targeted_search(a0, time_limit, seed=42):
    """Local search targeting peak positions in the convolution."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(len(a)) / len(a)
    n = len(a)
    
    best_a = a.copy()
    best_conv = fast_convolve(a)
    best_peak = best_conv.max()
    
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        
        improved = False
        
        # Strategy 1: Peak-targeted mass moves
        c = best_conv
        top_k = np.argpartition(c, -5)[-5:]  # 5 largest
        top_k = top_k[np.argsort(c[top_k])[::-1]]
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit - 0.02:
                break
            
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_end < i_start:
                continue
            
            # Find the pair (i, k-i) with largest product
            prods = a[i_start:i_end+1] * a[k - np.arange(i_start, i_end+1)]
            opt_i = i_start + int(np.argmax(prods))
            
            # Try moving mass from opt_i to various target positions
            for frac in [0.01, 0.03, 0.05, 0.1]:
                candidate = a.copy()
                amount = candidate[opt_i] * frac
                if amount < 1e-15:
                    continue
                candidate[opt_i] -= amount
                
                # Target: position that minimizes impact on peaks
                # Simple: try a few random positions
                for j in [int(rng.integers(0, n)), int(rng.integers(0, n)),
                           (opt_i + n // 2) % n]:
                    candidate = a.copy()
                    candidate[opt_i] -= amount
                    candidate[j] += amount
                    candidate = np.maximum(candidate, 0)
                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate = candidate / ss
                    new_c = fast_convolve(candidate)
                    if new_c.max() < best_peak - 1e-14:
                        a = candidate
                        best_peak = new_c.max()
                        best_a = a.copy()
                        best_conv = new_c
                        improved = True
                        break
                if improved:
                    break
        
        if improved:
            stall = 0
            continue
        
        # Strategy 2: Random perturbations at multiple scales
        for scale in [0.001, 0.003, 0.01]:
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(a + pert)
            new_c = fast_convolve(candidate)
            if new_c.max() < best_peak - 1e-14:
                a = candidate
                best_peak = new_c.max()
                best_a = a.copy()
                best_conv = new_c
                improved = True
                break
        
        # Strategy 3: Coordinate pair moves
        if not improved and it % 2 == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                for frac in [0.05, 0.1, 0.3]:
                    candidate = a.copy()
                    amount = candidate[i] * frac
                    candidate[i] -= amount
                    candidate[j] += amount
                    candidate = np.maximum(candidate, 0)
                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate = candidate / ss
                    new_c = fast_convolve(candidate)
                    if new_c.max() < best_peak - 1e-14:
                        a = candidate
                        best_peak = new_c.max()
                        best_a = a.copy()
                        best_conv = new_c
                        improved = True
                        break
        
        if improved:
            stall = 0
        else:
            stall += 1
        
        if stall > 2000:
            # Restart from best with perturbation
            pert = rng.standard_normal(n) * 0.005
            a = project_simplex(best_a + pert)
            best_conv = fast_convolve(a)
            best_peak = best_conv.max()
            stall = 0
    
    return best_a, best_peak


def simulated_annealing(a0, time_limit, seed=42):
    """Simulated annealing with adaptive moves."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(len(a)) / len(a)
    n = len(a)
    
    best_a = a.copy()
    best_peak = fast_convolve(a).max()
    curr_peak = best_peak
    
    T_start = 0.005 * best_peak
    T_end = 0.00005 * best_peak
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        
        # Annealing temperature
        frac = min(it / 30000.0, 1.0)
        T = T_start * (T_end / T_start) ** frac
        
        # Generate move
        r = rng.random()
        candidate = a.copy()
        
        if r < 0.4:
            # Gaussian perturbation
            scale = 0.005 * T / T_start + 0.001
            candidate += rng.standard_normal(n) * scale
        elif r < 0.7:
            # Coordinate move
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                amt = candidate[i] * rng.uniform(-0.5, 0.5)
                candidate[i] -= amt
                candidate[j] += amt
        else:
            # Peak-targeted move
            c = fast_convolve(a)
            k = int(np.argmax(c))
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_end > i_start:
                i = i_start + int(rng.integers(0, i_end - i_start + 1))
                j = int(rng.integers(0, n))
                amt = candidate[i] * rng.uniform(0.01, 0.15)
                candidate[i] -= amt
                candidate[j] += amt
        
        candidate = np.maximum(candidate, 0)
        ss = candidate.sum()
        if ss < 1e-10:
            continue
        candidate = candidate / ss
        
        new_peak = fast_convolve(candidate).max()
        delta = new_peak - curr_peak
        
        # Metropolis acceptance
        if delta < 0 or (T > 1e-20 and rng.random() < np.exp(-delta / T)):
            a = candidate
            curr_peak = new_peak
        
        if new_peak < best_peak - 1e-14:
            best_peak = new_peak
            best_a = a.copy()
    
    return best_a, best_peak


def subgradient_method(a0, time_limit):
    """
    Subgradient method for the peak: directly minimize max_k c[k].
    Uses the top K peaks for a "soft" subgradient.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(len(a)) / len(a)
    n = len(a)
    
    best_a = a.copy()
    best_peak = fast_convolve(a).max()
    
    t0 = time.monotonic()
    lr = 0.01
    it = 0
    K = 10  # number of top peaks to average
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        
        c = fast_convolve(a)
        peak = c.max()
        size_c = len(c)
        
        # Get top K peaks (or all if fewer)
        if size_c <= K:
            top_idx = np.arange(size_c)
        else:
            top_idx = np.argpartition(c, -K)[-K:]
        
        # Average the subgradients at these peak positions
        # subgrad_k[j] = 2*a[k-j] (for valid k-j)
        # Efficient: for each k in top_idx, add 2*a[k-j] to grad[j]
        # This is: grad = sum_{k in top} 2 * cross_corr(unit_k, a)
        # = 2 * cross_corr(sum_{k in top} unit_k, a)
        # where sum of unit vectors at top_idx positions
        
        top_vec = np.zeros(size_c)
        top_vec[top_idx] = 1.0 / len(top_idx)
        
        fft_len = 1 << (size_c + n - 1 - 1).bit_length()
        TV = np.fft.rfft(top_vec, fft_len)
        AV = np.fft.rfft(a, fft_len)
        corr = np.fft.irfft(TV * np.conj(AV), fft_len)[:n]
        grad = 2.0 * corr
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm
        
        # Step
        a_new = project_simplex(a - lr * grad * np.linalg.norm(fast_convolve(a) - fast_convolve(a) + 1))  # scale
        # Actually, let's use a proper step size
        # The subgradient has magnitude ~ O(1), and the objective is ~ O(1/n)
        # So step size should be ~ O(1/n)
        step = lr / n
        a_new = project_simplex(a - step * 2.0 * corr * n)  # hmm, let me think about scaling
        
        # Actually, the correct subgradient of max_k c_k at a point where the max is at set S is:
        # grad = (1/|S|) * sum_{k in S} grad_k
        # where grad_k[j] = 2*a[k-j]
        # The magnitude of this gradient is roughly O(1) (since a is normalized).
        # The objective value is ~ 1/n (for good sequences).
        # So a step of size O(1/n) should be appropriate.
        
        # Let me just use the gradient as-is with a small step
        a_new = project_simplex(a - step * 2.0 * corr * len(c))  # scale by size_c to normalize
        
        new_peak = fast_convolve(a_new).max()
        if new_peak < best_peak - 1e-14:
            best_peak = new_peak
            best_a = a_new.copy()
        
        a = a_new
        
        # Decay step size
        if it % 100 == 0:
            lr *= 0.98
    
    return best_a, best_peak
