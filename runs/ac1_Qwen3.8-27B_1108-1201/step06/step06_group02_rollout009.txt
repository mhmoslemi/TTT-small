def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])

OK so the gradient computation is:
`grad_j = 2 * sum_{m=0}^{n-1} w_{j+m} * a_m`

where `w_k` are the softmax weights of the convolution values. This can be computed as:
`grad = 2 * np.correlate(w, a, mode='valid')`

where w has length 2n-1 and a has length n.

This is O(n * (2n-1)) = O(n^2) using the naive correlate, but we can use FFT to make it O(n log n).

Actually, `np.correlate(w, a, mode='valid')` computes `sum_m w_{j+m} * a_m` for j = 0, ..., len(w)-len(a). If len(w) = 2n-1 and len(a) = n, we get n values. This is what we want.

For FFT-based computation:
`correlate(w, a, 'valid') = ifft(fft(w, 2L) * conj(fft(a, 2L)))` (with appropriate slicing)

Let me implement the mirror descent approach:

Mirror descent with entropy regularization:
- Update: a_j <- a_j * exp(-lr * grad_j) / Z (where Z is the normalizer)
- This is equivalent to: log(a_j) <- log(a_j) - lr * grad_j (up to normalization)

The update in log-space is: `log_a_j <- log_a_j - lr * grad_j`, then exponentiate and normalize.

This is the "exponential weights" or "Hedge" algorithm, adapted for our problem.

Advantages:
- Naturally keeps values positive (no projection needed)
- The effective step size adapts to the scale of each coordinate
- Can be more stable than Euclidean gradient descent

Now let me think about the block coordinate descent approach more carefully:

If we fix all coordinates except a block of B consecutive coordinates, the objective becomes a function of B variables. The convolution values that change are those k where at least one of the block indices j satisfies 0 <= k-j < n. This is a range of k values.

For a block at positions [s, s+B), the affected convolution values are k in [s, s+B+n-2]. The values outside this range don't change.

So the subproblem is:
minimize max(max_k (a'*a')_k for affected k, max_k (a'*a')_k for unaffected k)
= max(max_affected, max_unaffected)
= max(max_affected, C) where C is a constant (the max of unaffected values)

This is: minimize max(max_affected, C) = minimize max_affected (if max_affected > C) or C (if max_affected <= C).

If C is already below our current best, we just need to make sure the affected values don't exceed C. Otherwise, we need to minimize the max of the affected values.

This is a small optimization problem (B variables) that we can solve with scipy.optimize.minimize using the Nelder-Mead or L-BFGS-B method.

The challenge is the normalization constraint: if we change the block, the sum changes. We can parameterize the block as `a[s:s+B] = base + x` where x is the optimization variable, and enforce the sum constraint by adjusting one coordinate (or by using a reparameterization).

Actually, a simpler approach: parameterize the block as `a[s:s+B] = softmax(x) * target_sum` where `target_sum = 1 - sum(a[:s]) - sum(a[s+B:])`. This automatically satisfies the non-negativity and sum constraints for the block.

"""
Improved hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements:
1. Mirror descent (multiplicative weights) alongside Euclidean gradient
2. Top-k subgradient with momentum for sharper peak targeting
3. Block coordinate descent with local Nelder-Mead
4. Wider n exploration with diverse initializations
5. Targeted single-coordinate and pair-coordinate descent for final polish
6. Simulated annealing restarts with adaptive perturbation
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad(a, weights):
    """Compute gradient given weights w_k for each convolution position.
    grad_j = 2 * sum_k w_k * a_{k-j} (for valid k-j)
    """
    return 2.0 * np.correlate(weights, a, mode='valid')


def get_top_k_weights(conv, k):
    """Get uniform weights on the top-k convolution positions."""
    n_conv = len(conv)
    k = min(k, n_conv)
    idx = np.argpartition(conv, -k)[-k:]
    w = np.zeros(n_conv)
    w[idx] = 1.0 / k
    return w


def get_softmax_weights(conv, T):
    """Get softmax weights on convolution positions."""
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    w = w / w.sum()
    return w


def smoothed_grad_descent(a_init, time_limit, T_init, T_end, lr_init, lr_end, topk=None):
    """Smoothed gradient descent with adaptive temperature.
    If topk is set, uses top-k subgradient; otherwise uses softmax.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    v = np.zeros(n)  # momentum
    mom = 0.9

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break

        frac = min(1.0, it / max(1, int(time_limit / 0.005)))
        T = T_init * (T_end / T_init) ** frac
        lr = lr_init * (lr_end / lr_init) ** frac

        conv = fast_convolve(a)
        if topk is not None:
            k = int(topk * n)
            k = max(3, min(k, 2 * n - 1))
            w = get_top_k_weights(conv, k)
        else:
            w = get_softmax_weights(conv, T)

        grad = compute_grad(a, w)
        v = mom * v - lr * grad
        a_new = project_simplex(a + v)

        # Backtracking line search
        best_candidate = a_new
        best_val = fast_convolve(a_new).max()
        for alpha in [0.5, 0.25, 0.125, 0.0625]:
            cand = project_simplex(a - alpha * lr * grad)
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_candidate = cand

        a = best_candidate
        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 80:
            T = max(T * 1.1, T_end * 10)
        if no_improve > 200:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            noise_scale = 0.001 + 0.006 * min(1.0, (no_improve - 200) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            lr = lr_init * 0.7
            no_improve = 0

    return best_a, best_obj


def mirror_descent(a_init, time_limit, lr_init, lr_end, topk_frac=0.1):
    """Mirror descent (multiplicative weights) with top-k subgradient."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-10).copy()
    a = a / a.sum()
    log_a = np.log(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break

        frac = min(1.0, it / max(1, int(time_limit / 0.005)))
        lr = lr_init * (lr_end / lr_init) ** frac

        conv = fast_convolve(a)
        k = max(5, int(topk_frac * n))
        k = min(k, 2 * n - 1)
        w = get_top_k_weights(conv, k)
        grad = compute_grad(a, w)

        # Mirror descent update in log-space
        log_a_new = log_a - lr * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        val = fast_convolve(a_new).max()
        if val < best_obj - 1e-14:
            best_obj = val
            best_a = a_new.copy()
            a = a_new
            log_a = np.log(np.maximum(a, 1e-300))
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            lr *= 0.95
        if no_improve > 150:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            scale = 0.002 + 0.004 * min(1.0, (no_improve - 150) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            log_a = np.log(np.maximum(a, 1e-300))
            no_improve = 0
            lr = lr_init * 0.6

    return best_a, best_obj


def block_coordinate_descent(a_init, time_limit, block_size=16, max_iter_per_block=200):
    """Block coordinate descent with Nelder-Mead on each block."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    stagnant_cycles = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        stride = block_size

        for start in range(0, n - block_size + 1, stride):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            end = start + block_size
            fixed_sum = a[:start].sum() + a[end:].sum()
            target_sum = 1.0 - fixed_sum
            if target_sum < 1e-10:
                continue

            def block_obj(x):
                x_sh = x - x.max()
                bv = np.exp(x_sh)
                bv /= bv.sum()
                bv *= target_sum
                candidate = a.copy()
                candidate[start:end] = bv
                return fast_convolve(candidate).max()

            block_vals = np.maximum(a[start:end], 1e-15)
            x0 = np.log(block_vals)

            try:
                res = minimize(block_obj, x0, method='Nelder-Mead',
                               options={'maxiter': max_iter_per_block,
                                        'xatol': 1e-10, 'fatol': 1e-13})
                if res.fun < best_obj - 1e-13:
                    x_sh = res.x - res.x.max()
                    bv = np.exp(x_sh)
                    bv /= bv.sum()
                    bv *= target_sum
                    a[start:end] = bv
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    best_obj = fast_convolve(a).max()
                    best_a = a.copy()
                    improved = True
            except Exception:
                pass

        if improved:
            stagnant_cycles = 0
        else:
            stagnant_cycles += 1
            if stagnant_cycles > 5:
                break

    return best_a, best_obj


def targeted_coord_descent(a_init, time_limit):
    """Targeted single and pair coordinate moves to reduce peaks."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    stale = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Find the coordinate contributing most to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        if len(affected) == 0:
            break
        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]

        # Find a good target to move mass to
        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            break
        # Target: minimize increase in other peaks, prefer low-value positions
        j_min = outside[np.argmin(a[outside])]

        improved = False
        for delta_frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.03]:
            delta = delta_frac * a[j_max]
            if delta < 1e-12:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min] += delta
            candidate = np.maximum(candidate, 0.0)
            candidate = candidate / candidate.sum()
            val = fast_convolve(candidate).max()
            if val < best_obj - 1e-14:
                a = candidate
                best_obj = val
                best_a = a.copy()
                improved = True
                break

        if improved:
            stale = 0
        else:
            stale += 1
            if stale % 50 == 0:
                rng = np.random.default_rng((stale * 4567 + 3) % (2**31))
                candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-14:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
                    stale = 0
            if stale > 500:
                break

    return best_a, best_obj


def diverse_initializations(n, parent, rng):
    """Generate diverse initializations for a given n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Interpolated parent
    if parent is not None and len(parent) > 1:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Dirichlet (spread)
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    # Cosine taper
    x = np.arange(n, dtype=np.float64)
    cos_t = np.cos(np.pi * x / n) ** 2
    cos_t = np.maximum(cos_t, 1e-6)
    inits.append(cos_t / cos_t.sum())

    # Linear ramp (asymmetric)
    ramp = x + 1.0
    inits.append(ramp / ramp.sum())

    # Quadratic
    quad = (x + 1.0) ** 2
    inits.append(quad / quad.sum())

    # Bimodal
    bim = np.exp(-0.5 * ((x - 0.15 * n) / (0.1 * n)) ** 2) + np.exp(-0.5 * ((x - 0.85 * n) / (0.1 * n)) ** 2)
    bim = np.maximum(bim, 1e-6)
    inits.append(bim / bim.sum())

    # Random Fourier
    M = min(8, n // 2)
    x = np.arange(n, dtype=np.float64)
    base = np.ones(n) / n
    for m in range(1, M + 1):
        c = rng.standard_normal() * 0.1
        d = rng.standard_normal() * 0.1
        base += (c * np.cos(2.0 * np.pi * m * x / n) +
                 d * np.sin(2.0 * np.pi * m * x / n)) / n
    base = np.maximum(base, 1e-6)
    inits.append(base / base.sum())

    return inits
