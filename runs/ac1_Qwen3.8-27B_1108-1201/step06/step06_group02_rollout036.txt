Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2:
- Subgradient method with multi-peak weighting and Polyak averaging
- LP-based Frank-Wolfe for local refinement
- Structured initialization (Fourier, Gaussian, periodic)
- Adaptive n exploration and random restarts
- CMA-ES-style population search for global exploration
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def objective_val(a):
    """Compute max(conv(a,a)) for normalized a."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a = a / s
    return fast_convolve(a).max()

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_multi_peak(a, top_k=5):
    """Compute weighted multi-peak subgradient."""
    n = len(a)
    conv = fast_convolve(a)
    size = len(conv)
    
    if top_k >= size:
        peaks = np.arange(size)
        weights = np.ones(size) / size
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        # Exponential weighting based on rank
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = np.exp(-rank * 0.5)
        weights /= weights.sum()
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-14:
            continue
        k = int(peaks[i])
        # grad += 2 * weights[i] * H_k @ a
        # H_k @ a: (H_k @ a)(i) = a(k-i) if 0 <= k-i < n else 0
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad[j] += 2.0 * weights[i] * a[k - j]
    
    return grad, conv.max()

def sg_method(a0, time_limit, top_k, lr0, decay, avg_start):
    """Subgradient method with Polyak averaging."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = objective_val(a)
    
    # Polyak averaging
    sum_a = np.zeros(n)
    count = 0
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        grad, obj = subgradient_multi_peak(a, top_k)
        
        # Adaptive step size
        lr = lr0 / (1.0 + decay * it)
        
        # Try a few step sizes
        best_candidate = a
        best_val = obj
        for trial_lr in [lr, lr * 0.5, lr * 2.0]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
        
        a = best_candidate
        obj = best_val
        
        # Polyak averaging
        if it >= avg_start:
            sum_a += a
            count += 1
            avg_a = sum_a / count
            avg_obj = objective_val(avg_a)
            if avg_obj < best_obj:
                best_obj = avg_obj
                best_a = avg_a.copy()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            # Restart
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.005 * min(1.0, (no_improve - 100) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            no_improve = 0
            sum_a = np.zeros(n)
            count = 0
    
    return best_a, best_obj

def solve_lp_oracle(f, M, n):
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 45})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def fw_optimize(f0, time_limit):
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        g = solve_lp_oracle(f, M, n)
        if g is None:
            # Try with relaxed M
            g = solve_lp_oracle(f, M * 1.1, n)
            if g is None:
                no_improve += 1
                if no_improve > 30:
                    break
                continue
        
        g = g / g.sum()
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in np.linspace(0.005, 1.0, 20):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            ft = ft / ft.sum()
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Golden section
        if 0.005 < best_alpha < 0.995:
            lo = max(0.0, best_alpha - 0.04)
            hi = min(1.0, best_alpha + 0.04)
            for _ in range(15):
                m1 = lo + 0.382 * (hi - lo)
                m2 = lo + 0.618 * (hi - lo)
                f1 = fast_convolve(np.maximum((1-m1)*f + m1*g, 0)).max()
                f2 = fast_convolve(np.maximum((1-m2)*f + m2*g, 0)).max()
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            best_alpha = (lo + hi) / 2
            ft = (1.0 - best_alpha) * f + best_alpha * g
            ft = np.maximum(ft, 0.0)
            best_val = fast_convolve(ft).max()
        
        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 80:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.001 + 0.004 * min(1.0, (no_improve - 80) / 150.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max()
            no_improve = 0
    
    return best_f, best_obj

def fourier_param_optimize(a0, time_limit, n_modes):
    """Optimize using Fourier parametrization with Nelder-Mead."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    M = min(n_modes, n // 2 - 1)
    
    x = np.arange(n, dtype=np.float64)
    a_centered = a - 1.0 / n
    
    init_coeffs = np.zeros(2 * M)
    for m in range(1, M + 1):
        init_coeffs[2*(m-1)] = 2.0 * np.sum(a_centered * np.cos(2.0*np.pi*m*x/n))
        init_coeffs[2*(m-1)+1] = 2.0 * np.sum(a_centered * np.sin(2.0*np.pi*m*x/n))
    
    best_a = a.copy()
    best_obj = objective_val(a)
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    n_restarts = max(2, int(time_limit // 40) + 1)
    for r in range(n_restarts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        if r == 0:
            x0 = init_coeffs.copy()
        else:
            scale = 0.01 * (1 + 0.5 * r)
            x0 = init_coeffs + rng.standard_normal(2*M) * scale
        
        def obj_func(coeffs):
            base = np.ones(n) / n
            for m in range(1, M + 1):
                c = coeffs[2*(m-1)]
                d = coeffs[2*(m-1)+1]
                base += (c * np.cos(2.0*np.pi*m*x/n) + d * np.sin(2.0*np.pi*m*x/n)) / n
            base = np.maximum(base, 0.0)
            s = base.sum()
            if s < 1e-15:
                return 10.0
            base = base / s
            return fast_convolve(base).max()
        
        # Simple coordinate descent / Nelder-Mead
        from scipy.optimize import minimize
        try:
            res = minimize(obj_func, x0, method='Nelder-Mead',
                           options={'maxiter': 5000, 'xatol': 1e-8, 'fatol': 1e-10})
            if res.fun < best_obj:
                best_obj = res.fun
                # Reconstruct
                base = np.ones(n) / n
                for m in range(1, M + 1):
                    c = res.x[2*(m-1)]
                    d = res.x[2*(m-1)+1]
                    base += (c * np.cos(2.0*np.pi*m*x/n) + d * np.sin(2.0*np.pi*m*x/n)) / n
                base = np.maximum(base, 0.0)
                base = base / base.sum()
                best_a = base
                print("    Fourier r={}: obj={:.10f}".format(r, best_obj))
        except Exception:
            pass
    
    return best_a, best_obj

def ia_method(a0, time_limit):
    """Iterative Amplitude method (spectral domain)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = objective_val(a)
    
    t0 = time.monotonic()
    eta = 0.5
    T = 1.0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # FFT
        A = np.fft.fft(a)
        power = np.abs(A) ** 2
        
        # Modify power spectrum to reduce peak autocorrelation
        # Use softmax-like suppression
        conv = fast_convolve(a)
        max_conv = conv.max()
        target = conv.mean() * 1.5  # Target slightly above mean
        
        # In frequency domain: reduce high power where it contributes to peaks
        # Simple approach: reduce peak of power spectrum
        # Actually, let's use the time-domain approach:
        # a_new = a * (1 - eta * exp(conv_peak_contrib / T))
        
        # Compute contribution of each a(i) to the peak
        k_peak = np.argmax(conv)
        contributions = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        for j in range(js, je + 1):
            contributions[j] = a[k_peak - j]
        
        # Also consider other high peaks
        top_peaks = np.argpartition(conv, -5)[-5:]
        for k in top_peaks:
            if k == k_peak:
                continue
            weight = conv[k] / max_conv
            js2 = max(0, int(k) - n + 1)
            je2 = min(n - 1, int(k))
            for j in range(js2, je2 + 1):
                contributions[j] += weight * a[int(k) - j]
        
        # Update: reduce contributions to peaks
        scale = np.exp(-contributions / (T * max_conv + 1e-10))
        a_new = a * scale
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
        
        # Anneal
        T *= 0.995
        eta = min(eta * 1.001, 0.9)
    
    return best_a, best_obj

def targeted_peak_reduction(a0, time_limit):
    """Reduce peaks by targeted coordinate swaps."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = objective_val(a)
    
    t0 = time.monotonic()
    it = 0
    stale = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        
        if len(affected) == 0:
            break
        
        # Find the pair (j_max, j_min) that gives best improvement
        contributions = a[affected] * a[k_peak - affected]
        
        # Try moving mass from high-contribution positions to low-value positions
        sorted_affected = affected[np.argsort(contributions)[::-1]]
        outside = np.setdiff1d(np.arange(n), affected)
        
        improved = False
        for j_max in sorted_affected[:5]:
            if a[j_max] < 1e-10:
                continue
            # Find best position to move to
            best_delta = 0
            best_pos = -1
            best_val = best_obj
            for delta in [0.001, 0.005, 0.01, 0.02]:
                if a[j_max] < delta:
                    continue
                # Try moving to the position that minimizes impact
                for j_min in outside[:min(10, len(outside))]:
                    candidate = a.copy()
                    candidate[j_max] -= delta
                    candidate[j_min] += delta
                    c = fast_convolve(candidate)
                    if c.max() < best_val:
                        best_val = c.max()
                        best_delta = delta
                        best_pos = j_min
            if best_pos >= 0:
                a[j_max] -= best_delta
                a[best_pos] += best_delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_obj = best_val
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            stale = 0
        else:
            stale += 1
            if stale > 100:
                # Random perturbation
                rng = np.random.default_rng(it * 4567 + 3)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.002)
                c = fast_convolve(candidate)
                if c.max() < best_obj:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    stale = 0
                elif stale > 500:
                    break
    
    return best_a, best_obj

def population_search(a0, time_limit, pop_size=20):
    """Simple population-based search."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0)
    a0 = a0 / a0.sum()
    
    # Initialize population
    rng = np.random.default_rng(123)
    population = [a0.copy()]
    for _ in range(pop_size - 1):
        noise = rng.standard_normal(n) * 0.01
        individual = project_simplex(a0 + noise)
        population.append(individual)
    
    best_a = a0.copy()
    best_obj = objective_val(a0)
    
    t0 = time.monotonic()
    generation = 0
    
    while time.monotonic() - t0 < time_limit:
        generation += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        
        # Evaluate all
        for i in range(pop_size):
            obj = objective_val(population[i])
            if obj < best_obj:
                best_obj = obj
                best_a = population[i].copy()
        
        # Create next generation
        new_pop = [best_a.copy()]
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break
        
        for _ in range(pop_size - 1):
            if time.monotonic() - t0 > time_limit - 1.0:
                break
            # Crossover + mutation
            i1 = rng.integers(0, pop_size)
            i2 = rng.integers(0, pop_size)
            alpha = rng.random()
            child = (1 - alpha) * population[i1] + alpha * population[i2]
            child = np.maximum(child, 0.0)
            child = project_simplex(child + rng.standard_normal(n) * 0.005)
            new_pop.append(child)
        
        population = new_pop
    
    return best_a, best_obj

def try_and_update(a, state):
    """Evaluate and update state."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST: {:.8f} (n={})".format(score, len(a)))
