    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M):
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fista_smooth(a0, time_limit, T_init=None, T_final=1e-7, T_decay=0.97,
                 L_init=10.0, verbose=False):
    """FISTA with log-sum-exp smoothing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    r = fast_autoconv(a)
    M_curr = r.max()
    if T_init is None:
        T_init = M_curr / 5.0
    
    T = T_init
    L = L_init
    best_a = a.copy()
    best_obj = M_curr
    best_fT = T * np.log(np.sum(np.exp(r / T)))
    
    x = a.copy()
    x_prev = a.copy()
    t_moment = 1.0
    
    t0 = time.monotonic()
    it = 0
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # FISTA momentum
        t_prev = t_moment
        t_moment = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev))
        beta = (t_prev - 1.0) / t_moment
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = np.ones(n) / n
        else:
            y = y / sy  # Keep in simplex for stability (approximate)
        
        # Compute gradient of f_T at y
        r_y = fast_autoconv(y)
        # Numerical stability for softmax
        r_shifted = r_y - r_y.max()
        w = np.exp(r_shifted / T)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.ones(len(r_y)) / len(r_y)
        else:
            w = w / w_sum
        
        # grad[m] = 2 * sum_k w[k] * y[k-m] for valid k
        grad = np.zeros(n)
        for m in range(n):
            k_start = m
            k_end = min(2 * n - 1, m + n)
            if k_start < k_end:
                grad[m] = np.dot(w[k_start:k_end], y[k_start - m:k_end - m])
        grad *= 2.0
        
        # FISTA step with backtracking
        z = project_simplex(y - grad / L)
        
        # Backtracking (limited to 10 iterations)
        for _bt in range(10):
            r_z = fast_autoconv(z)
            rz_shifted = r_z - r_z.max()
            fT_z = T * np.log(np.sum(np.exp(rz_shifted / T))) + T * r_z.max()
            fT_y = T * np.log(np.sum(np.exp(r_shifted / T))) + T * r_y.max()
            # Check: fT(z) <= fT(y) + grad^T(z-y) + L/2 * ||z-y||^2
            linear_bound = fT_y + np.dot(grad, z - y) + 0.5 * L * np.dot(z - y, z - y)
            if fT_z <= linear_bound + 1e-10:
                break
            L *= 1.5
            z = project_simplex(y - grad / L)
        
        x_prev = x
        x = z
        
        # Update T (geometric decay)
        if it % 5 == 0:
            T = max(T * T_decay, T_final)
        
        # Track best (use actual max, not smoothed)
        r_x = fast_autoconv(x)
        obj = r_x.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = x.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Adaptive L
        if stagnant > 50:
            L *= 0.9  # Try smaller L for faster steps
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA it={it}, T={T:.6f}, obj={best_obj:.10f}, L={L:.2f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid multi-phase approach:
1. FISTA with log-sum-exp smoothing (continuation method) for smooth optimization
2. Frank-Wolfe with LP oracle for convex relaxation at each step
3. Peak-targeted coordinate descent for final polishing
4. Multi-length exploration with diverse initializations including spectral methods
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_autoconv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def compute_fista_grad(a, w):
    """Compute gradient of T*log(sum exp(r/T)): grad[m] = 2*sum_j w[m+j]*a[j]."""
    n = len(a)
    grad = np.zeros(n)
    for m in range(n):
        grad[m] = np.dot(w[m:m + n], a)
    grad *= 2.0
    return grad


def fista_optimize(a0, time_limit, T_init=None, T_final=1e-8, T_decay=0.96,
                   L_init=8.0, verbose=False):
    """FISTA with log-sum-exp smoothing and continuation method."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    r = fast_autoconv(a)
    M_curr = r.max()
    if T_init is None:
        T_init = M_curr / 4.0

    T = T_init
    L = L_init
    best_a = a.copy()
    best_obj = M_curr

    x = a.copy()
    x_prev = a.copy()
    t_mom = 1.0

    t0 = time.monotonic()
    it = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # FISTA momentum
        t_prev = t_mom
        t_mom = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev))
        beta = (t_prev - 1.0) / t_mom
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 0.0)

        # Compute autocorrelation and softmax weights at y
        r_y = fast_autoconv(y)
        r_max_y = r_y.max()
        r_shifted = r_y - r_max_y
        exp_vals = np.exp(r_shifted / T)
        w_sum = exp_vals.sum()
        if w_sum < 1e-300:
            w = np.ones(len(r_y)) / len(r_y)
        else:
            w = exp_vals / w_sum

        # Compute gradient
        grad = compute_fista_grad(y, w)

        # FISTA step with backtracking
        z = project_simplex(y - grad / L)
        bt_count = 0
        for _bt in range(8):
            r_z = fast_autoconv(z)
            rz_max = r_z.max()
            rz_shifted = r_z - rz_max
            fT_z = T * (rz_max + np.log(np.exp(rz_shifted / T).sum() / len(r_z)) + np.log(len(r_z)))
            fT_y = T * (r_max_y + np.log(exp_vals.sum() / len(r_y)) + np.log(len(r_y)))
            # Simplified: fT = T * log(sum(exp(r/T))) = T*(r_max + log(sum(exp((r-r_max)/T))))
            fT_z = T * (rz_max + np.log(np.exp(rz_shifted / T).sum()))
            fT_y = T * (r_max_y + np.log(exp_vals.sum()))
            linear_bound = fT_y + np.dot(grad, z - y) + 0.5 * L * np.dot(z - y, z - y)
            if fT_z <= linear_bound + 1e-8:
                break
            L *= 1.5
            z = project_simplex(y - grad / L)
            bt_count += 1

        x_prev = x
        x = z

        # Adaptive T decay
        if it % 10 == 0:
            T = max(T * T_decay, T_final)

        # Track best using actual max
        r_x = fast_autoconv(x)
        obj = r_x.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = x.copy()
            stagnant = 0
        else:
            stagnant += 1

        # Adaptive L
        if stagnant > 100:
            L *= 0.85
        if stagnant > 300:
            rng_s = np.random.default_rng((it * 7919 + 13) % (2**31))
            x = project_simplex(best_a + rng_s.standard_normal(n) * 0.003)
            x_prev = x.copy()
            t_mom = 1.0
            T = max(T * 5.0, T_final)
            L = L_init
            stagnant = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA it={it}, T={T:.6e}, obj={best_obj:.10f}, L={L:.2f}, t={elapsed:.0f}s")

    return best_a, best_obj
