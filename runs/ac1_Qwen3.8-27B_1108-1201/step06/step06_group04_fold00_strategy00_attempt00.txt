def project_simplex(v):
    """
    Euclidean projection onto the probability simplex.
    Projects a N-dimensional vector onto the set {x | sum(x) = 1, x >= 0}.

    Args:
        v: Input vector (numpy array)

    Returns:
        simplex projection (numpy array)
    """
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0)])
    else:
        v_sorted = np.sort(v)[::-1]
        cssv = np.cumsum(v_sorted)
        cond = v_sorted[:-1] > (cssv[:-1] - v_sorted[-1])
        if not cond.any():
            return np.ones(n) / n
        else:
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            if theta < v_sorted[rho]:
                theta = v_sorted[rho] * (1 - 1e-10)
                if theta < 1e-24:
                    theta = 0.0
            y = np.maximum(v_sorted - theta, 0.0)
            return y
