def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hio_initialize(n, n_iters=100, alpha_init=0.5, alpha_final=0.9, seed=42):

def build_lp_matrix(f):

    n = len(f)
    T = np.zeros((2 * n - 1, n))

        # When m is in the valid range, k-m is in [0, n-1]
        for m in range(js, je + 1):
            T[k, m] = f[k - m]
    return T

def solve_lp_direction(f, max_time=5.0):

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local(a):
    """Compute objective locally without calling evaluate_sequence."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def build_toeplitz(f):
    """Build (2n-1) x n matrix T where T[k,m] = f[k-m] if 0<=k-m<n."""
    n = len(f)
    T = np.zeros((2 * n - 1, n))
    for m in range(n):
        length = min(n, 2 * n - 1 - m)
        T[m:m + length, m] = f[:length]
    return T


def hio_run(a_init, n_iters=100, alpha_start=0.5, alpha_end=0.92, seed=42):
    """Run HIO iterations to flatten the auto-convolution spectrum."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    fft_size = 1 << (2 * n - 2).bit_length()
    rng = np.random.default_rng(seed)
    
    for it in range(n_iters):
        t_frac = it / max(1, n_iters - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * t_frac
        
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        mean_power = power.mean()
        new_power = alpha * power + (1.0 - alpha) * mean_power
        new_amp = np.sqrt(new_power)
        A_new = new_amp * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = np.ones(n) / n
            continue
        a_new = a_new / s_new
        
        if it % 25 == 24:
            a_new += rng.standard_normal(n) * 0.001
            a_new = project_simplex(a_new)
        
        a = a_new
    
    return a


def solve_lp_direction(f, max_time=3.0):
    """
    Solve LP: min t s.t. (f*g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None).
    """
    n = len(f)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    T = build_toeplitz(f)
    A_ub = np.hstack([T, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max_time})
        if res.success:
            g = np.maximum(res.x[:n], 0.0)
            gs = g.sum()
            if gs < 1e-15:
                return None, None
            g = g / gs
            return g, float(res.x[-1])
    except Exception:
        pass
    return None, None


def frank_wolfe(a0, time_limit, max_lp_time=3.0, restart_every=50, seed=0):
    """Frank-Wolfe with LP oracle."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        g, t_star = solve_lp_direction(a, max_time=min(max_lp_time, remaining - 1.0))
        
        if g is None:
            no_improve += 1
            if no_improve > 15:
                rng = np.random.default_rng(it * 37 + seed + 11)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                M = fast_convolve(a).max()
                no_improve = 0
            continue
        
        if np.max(np.abs(g - a)) < 1e-10:
            no_improve += 1
            if no_improve > 15:
                rng = np.random.default_rng(it * 41 + seed + 13)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                M = fast_convolve(a).max()
                no_improve = 0
            continue
        
        best_alpha = 0.0
        best_val = M
        alphas = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = (1.0 - alpha) * a + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.04)
            hi = min(1.0, best_alpha + 0.04)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * a + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-12:
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            M = fast_convolve(a).max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > restart_every:
            rng = np.random.default_rng(it * 53 + seed + 17)
            scale = 0.002 + 0.005 * min(1.0, no_improve / (2 * restart_every))
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            M = fast_convolve(a).max()
            no_improve = 0
    
    return best_a, best_obj


def multi_peak_pgd(a0, time_limit, top_k=20, lr_init=0.02, mom_init=0.9, seed=0):
    """Projected gradient with multi-peak subgradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    lr = lr_init
    mom = mom_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        obj = conv.max()
        
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            weights = pv / pv.sum()
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * a[k - idx_arr]
        
        v = mom * v - lr * sg
        best_candidate = a
        best_val = obj
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            candidate = project_simplex(a + scale * v)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break
        
        a = best_candidate
        
        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.9
        
        if no_improve > 30:
            lr *= 0.95
        if no_improve > 150:
            rng = np.random.default_rng((it * 48271 + seed + 17) % (2**31 - 1))
            noise = 0.002 + 0.003 * min(1.0, (no_improve - 150) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init
            mom = mom_init
            no_improve = 0
    
    return best_a, best_obj


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def try_update(a, best_seq, best_score, label=""):
    """Try to update best state. Returns (best_seq, best_score)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_seq, best_score
    a_norm = a / s
    sc = eval_local(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def hio_variant(n, seed, n_iters, init_type="uniform"):
    """Create an initial sequence and run HIO on it."""
    if init_type == "uniform":
        a = np.ones(n) / n
    elif init_type == "hann":
        x = np.arange(n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
        a = a / a.sum()
    elif init_type == "cosine_quadratic":
        x = np.arange(n) / max(1, n - 1)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif init_type == "exponential":
        decay = np.linspace(0.3, 1.0, n)
        a = np.exp(-decay * 3.0)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    
    return hio_run(a, n_iters=n_iters, alpha_start=0.5, alpha_end=0.92, seed=seed)

            parent = p / p.sum()
            ps = eval_local(parent)

            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = eval_local(best_seq)
        print(f"Default: n={len(best_seq)}, score={best_score:.8f}")
    
    # === Phase 1: Diverse HIO initialization ===
    print("\n=== Phase 1: HIO Initialization ===")
    t_start = time.monotonic()
    
    n_parent = len(best_seq)
    target_ns = sorted(set([
        n_parent,
        n_parent // 2 if n_parent >= 256 else 256,
        n_parent * 2 if n_parent * 2 <= 2048 else 2048,
        512, 768, 1024, 1536
    ]))
    target_ns = [n for n in target_ns if 256 <= n <= 2048]
    
    candidates = {}
    
    for n_try in target_ns:
        remaining = deadline - time.monotonic() - 1100

        (best_seq, best_score) = try_update(candidates.get((n_try, "uniform"), (None, float('inf')))[0] if (n_try, "uniform") in candidates else np.ones(n_try)/n_try, best_seq, best_score, f"hio_n{n_try}")
    
    # Keep top candidates
    sorted_candidates = sorted(candidates.items(), key=lambda x: x[1][1])
