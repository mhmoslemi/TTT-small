def mw_primal_dual(a0, time_limit, T_init=0.01, eta_init=0.1, verbose=False):
    """
    Primal-dual with multiplicative weights (exponentiated gradient).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    T = T_init
    eta = eta_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 100 == 0:
            if time.monotonic() - t0 > time_limit - 0.05:
                break
        
        conv = fast_convolve(a)
        cmax = conv.max()
        
        # Dual weights
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()
        
        # Gradient: grad[i] = 2 * sum_j w[i+j] * a[j]
        grad = 2.0 * hankel_mv(w, a)
        
        # Multiplicative weights update
        # a_new[i] prop to a[i] * exp(-eta * grad[i])
        # Normalize so sum = 1
        log_a = np.log(np.maximum(a, 1e-300)) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        
        # Blend
        alpha = min(1.0, 0.3 + 0.7 * (1.0 - no_improve / 100.0))
        a_blend = (1.0 - alpha) * a + alpha * a_new
        a_blend = np.maximum(a_blend, 0.0)
        a_blend = a_blend / a_blend.sum()
        
        obj = fast_convolve(a_blend).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_blend.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Revert to previous
            a_blend = a  # Don't update if no improvement
        
        a = a_blend
        
        # Adapt
        if no_improve > 20:
            eta *= 0.9
            T *= 1.1
        if no_improve > 50:
            T *= 0.8  # Re-focus
        if no_improve > 100:
            # Restart with perturbation
            rng = np.random.default_rng((it * 104729 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            T = T_init
            eta = eta_init
            no_improve = 0
        
        if verbose and it % 200 == 0:
            print(f"    MW it={it}, obj={best_obj:.10f}, T={T:.6f}, eta={eta:.6f}")
    
    return best_a, best_obj

def solve_qp_hankel(w, a_init, n, max_iters=300, time_limit=5.0):
    """
    Solve min a^T W a s.t. sum(a)=1, a>=0 using active set method.
    W is Hankel: W_{i,j} = w[i+j].
    """
    t0 = time.monotonic()
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    active = set(np.where(a > 1e-12)[0])
    if not active:
        active = set(range(n))

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if len(active) == 0:
            return np.ones(n) / n
        if len(active) == 1:
            a = np.zeros(n)
            a[list(active)[0]] = 1.0
            return a

        act = sorted(active)
        k = len(act)

        W_AA = np.array([[w[act[i] + act[j]] for j in range(k)] for i in range(k)])
        W_AA_reg = W_AA + 1e-12 * np.eye(k)

        try:
            x = np.linalg.solve(W_AA_reg, np.ones(k))
        except np.linalg.LinAlgError:
            try:
                x = np.linalg.lstsq(W_AA_reg, np.ones(k), rcond=None)[0]
            except Exception:
                x = np.ones(k) / k

        x_sum = x.sum()
        if abs(x_sum) < 1e-14:
            x = np.ones(k) / k
            x_sum = 1.0

        a_act = x / x_sum

        removed = False
        min_val = float('inf')
        min_idx = -1
        for i, idx in enumerate(act):
            if a_act[i] < 0:
                if a_act[i] < min_val:
                    min_val = a_act[i]
                    min_idx = idx
        if min_idx >= 0:
            active.discard(min_idx)
            a_new = np.zeros(n)
            for i, idx in enumerate(act):
                if idx != min_idx:
                    a_new[idx] = max(0.0, a_act[i])
            s2 = a_new.sum()
            if s2 > 1e-15:
                a = a_new / s2
            continue

        a = np.zeros(n)
        for i, idx in enumerate(act):
            a[idx] = max(0.0, a_act[i])

        Wa = hankel_mv(w, a)
        grad = 2.0 * Wa
        mu = np.mean([grad[idx] for idx in act])

        inactive = set(range(n)) - active
        if inactive:
            min_viol = 0.0
            min_j = -1
            for j in inactive:
                viol = grad[j] - mu
                if viol < min_viol:
                    min_viol = viol
                    min_j = j
            if min_viol < -1e-8:
                active.add(min_j)
                continue

        return a

    return a


def mw_primal_dual(a0, time_limit, T_init=0.01, eta_init=0.05, verbose=False):
    """
    Multiplicative weights (exponentiated gradient) primal-dual method.
    Much faster per iteration than QP-based approach.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    eta = eta_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        cmax = conv.max()

        # Dual weights (softmax)
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()

        # Gradient of a^T W a: 2 * W @ a
        grad = 2.0 * hankel_mv(w, a)

        # Multiplicative weights update
        # a_new[i] prop to a[i] * exp(-eta * grad[i])
        log_ratio = -eta * grad
        log_ratio -= log_ratio.max()  # numerical stability
        a_new = a * np.exp(log_ratio)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Evaluate
        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            if no_improve < 10:
                # Don't update
                pass
            # a stays the same

        # Adapt
        if no_improve > 10:
            eta *= 0.95
            T *= 1.05
        if no_improve > 40:
            T *= 0.8  # Re-focus on peaks
            eta *= 0.8
        if no_improve > 100:
            # Restart with structured perturbation
            rng = np.random.default_rng((it * 104729 + 7) % (2**31))
            # Perturb in a structured way: shift mass from high to low peak contributors
            conv = fast_convolve(best_a)
            k_peak = int(np.argmax(conv))
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            perm = rng.permutation(n)
            delta = np.zeros(n)
            for idx in perm[:n//2]:
                if js <= idx <= je and best_a[idx] > 1e-8:
                    delta[idx] = -best_a[idx] * 0.05
                elif not (js <= idx <= je):
                    delta[idx] = best_a[idx] * 0.05 + 1e-8
            a = project_simplex(best_a + delta)
            T = T_init
            eta = eta_init
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    MW it={it}, obj={best_obj:.10f}, T={T:.6f}, eta={eta:.6f}, no_imp={no_improve}")

    return best_a, best_obj


def nesterov_subgrad(a0, time_limit, lr_init=0.003, momentum_init=0.9, verbose=False):
    """Nesterov-accelerated subgradient method."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = momentum_init
    y = a.copy()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        y = a + mom * v
        y = project_simplex(y)

        conv_y = fast_convolve(y)
        k_peak = int(np.argmax(conv_y))
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * y[k_peak - idx]

        v_new = project_simplex(y - lr * grad) - y
        a = y + v_new
        a = project_simplex(a)
        v = v_new

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        mom = min(0.995, mom + 0.00003)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- SDP relaxation (CVXPY/SCS) for warm-starting
- Multiplicative weights (exponentiated gradient) primal-dual (fast inner loop)
- QP primal-dual with active set (precise inner loop)
- Adaptive dual (softmax with annealing, top-k)
- Nesterov subgradient polish

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(w, a):
    """Compute W @ a where W_{i,j} = w[i+j]. w has length 2n-1, a has length n."""
    return np.correlate(w, a, mode='valid')


def solve_qp_hankel(w, a_init, n, max_iters=300, time_limit=5.0):
    """Solve min a^T W a s.t. sum(a)=1, a>=0 via active set. W_{i,j}=w[i+j]."""
    t0 = time.monotonic()
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    active = set(np.where(a > 1e-12)[0])
    if not active:
        active = set(range(n))

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if len(active) == 0:
            return np.ones(n) / n
        if len(active) == 1:
            a = np.zeros(n)
            a[list(active)[0]] = 1.0
            return a

        act = sorted(active)
        k = len(act)
        W_AA = np.array([[w[act[i] + act[j]] for j in range(k)] for i in range(k)])
        W_AA_reg = W_AA + 1e-12 * np.eye(k)

        try:
            x = np.linalg.solve(W_AA_reg, np.ones(k))
        except np.linalg.LinAlgError:
            try:
                x = np.linalg.lstsq(W_AA_reg, np.ones(k), rcond=None)[0]
            except Exception:
                x = np.ones(k) / k

        x_sum = x.sum()
        if abs(x_sum) < 1e-14:
            x = np.ones(k) / k
            x_sum = 1.0
        a_act = x / x_sum

        min_val = float('inf')
        min_idx = -1
        for i, idx in enumerate(act):
            if a_act[i] < 0 and a_act[i] < min_val:
                min_val = a_act[i]
                min_idx = idx
        if min_idx >= 0:
            active.discard(min_idx)
            a_new = np.zeros(n)
            for i, idx in enumerate(act):
                if idx != min_idx:
                    a_new[idx] = max(0.0, a_act[i])
            s2 = a_new.sum()
            if s2 > 1e-15:
                a = a_new / s2
            continue

        a = np.zeros(n)
        for i, idx in enumerate(act):
            a[idx] = max(0.0, a_act[i])

        Wa = hankel_mv(w, a)
        grad = 2.0 * Wa
        mu = np.mean([grad[idx] for idx in act])

        inactive = set(range(n)) - active
        if inactive:
            min_viol = 0.0
            min_j = -1
            for j in inactive:
                viol = grad[j] - mu
                if viol < min_viol:
                    min_viol = viol
                    min_j = j
            if min_viol < -1e-8:
                active.add(min_j)
                continue

        return a

    return a


def mw_primal_dual(a0, time_limit, T_init=0.005, eta_init=0.03, verbose=False):
    """Multiplicative weights (exponentiated gradient) primal-dual. Very fast per iteration."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    eta = eta_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    T_min = T_init * 0.01

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        cmax = conv.max()

        # Dual weights
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()

        # Gradient of a^T W a
        grad = 2.0 * hankel_mv(w, a)

        # Multiplicative weights: a_new[i] prop a[i] * exp(-eta * grad[i])
        log_r = -eta * grad
        log_r -= log_r.max()
        a_new = a * np.exp(log_r)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Slightly decrease T to focus more
            T = max(T * 0.995, T_min)
        else:
            no_improve += 1

        if no_improve > 5:
            eta *= 0.97
        if no_improve > 30:
            T = max(T * 0.9, T_min)
        if no_improve > 80:
            # Structured restart: perturb peak-contributing coordinates
            rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
            conv_b = fast_convolve(best_a)
            top_peaks = np.argpartition(conv_b, -5)[-5:]
            delta = np.zeros(n)
            for kp in top_peaks:
                js = max(0, kp - n + 1)
                je = min(n - 1, kp)
                for i in range(js, je + 1):
                    delta[i] -= rng.random() * 0.01 * best_a[i]
                # Add mass to non-peak positions
                non_peak = np.setdiff1d(np.arange(n), np.arange(js, je + 1))
                if len(non_peak) > 0:
                    targets = rng.choice(non_peak, size=min(5, len(non_peak)), replace=False)
                    for i in targets:
                        delta[i] += 0.005
            a = project_simplex(best_a + delta)
            T = T_init
            eta = eta_init
            no_improve = 0

        if verbose and it % 1000 == 0:
            print(f"    MW it={it}, obj={best_obj:.10f}, T={T:.6f}, eta={eta:.6f}, no_imp={no_improve}")

    return best_a, best_obj


def nesterov_subgrad(a0, time_limit, lr_init=0.002, momentum_init=0.9, verbose=False):
    """Nesterov-accelerated subgradient for the max-convolution objective."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = momentum_init
    y = a.copy()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        y = a + mom * v
        y = project_simplex(y)

        conv_y = fast_convolve(y)
        k_peak = int(np.argmax(conv_y))
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * y[k_peak - idx]

        v_new = project_simplex(y - lr * grad) - y
        a = y + v_new
        a = project_simplex(a)
        v = v_new

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.99995
        mom = min(0.997, mom + 0.00002)

        if no_improve > 300:
            lr *= 0.85
            mom *= 0.97
        if no_improve > 800:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = lr_init * 0.5
            mom = momentum_init
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    nesterov it={it}, obj={best_obj:.10f}, lr={lr:.7f}")

    return best_a, best_obj


def qp_primal_dual(a0, time_limit, T_init=0.01, use_topk=False, topk=20,
                   blend=0.4, verbose=False):
    """QP-based primal-dual with active set inner solver."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s
    n = len(a)

    T = T_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        conv = fast_convolve(a)

        if use_topk:
            k_use = min(topk, len(conv))
            peaks = np.argpartition(conv, -k_use)[-k_use:]
            w = np.zeros(len(conv))
            w[peaks] = 1.0 / k_use
        else:
            scaled = conv / T
            scaled -= scaled.max()
            w = np.exp(scaled)
            w = w / w.sum()

        qp_time = min(4.0, remaining * 0.25)
        a_qp = solve_qp_hankel(w, a, n, max_iters=300, time_limit=qp_time)

        a_new = (1.0 - blend) * a + blend * a_qp
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a_qp
        else:
            a_new = a_new / s_new

        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 10:
            T *= 1.08
        if no_improve > 35:
            T *= 0.8
        if no_improve > 90:
            rng = np.random.default_rng((it * 104729 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)

            no_improve = 0

        if verbose and it % 30 == 0:

    return best_a, best_obj


def sdp_warmstart(n, time_limit=60.0):
    """
    Solve SDP relaxation:
    minimize t
    s.t. sum_{i+j=k} Z[i,j] <= t for all k
         Z PSD, Z[n,n]=1, Z[:n,n] >= 0, sum(Z[:n,n]) = 1
    Returns a feasible (primal) solution extracted from Z[:n,n].
    """
    try:
        import cvxpy as cp
    except ImportError:
        print("    cvxpy not available, skipping SDP")
        return None

    t0 = time.monotonic()
    try:
        Z = cp.Variable((n + 1, n + 1), symmetric=True)
        t = cp.Variable()
        a_vec = Z[:n, n]

        constraints = [
            Z >> 0,
            Z[n, n] == 1,
            cp.sum(a_vec) == 1,
            a_vec >= 0,
        ]

        # For each lag k, sum of Z[i,j] over i+j=k (with i,j in [0,n))
        for k in range(2 * n - 1):
            i_vals = np.arange(max(0, k - n + 1), min(n, k + 1))
            j_vals = k - i_vals
            # Sum Z[i,j] for these pairs
            # Use sum of Z[i_vals, j_vals] element-wise
            constraints.append(cp.sum(Z[i_vals, j_vals]) <= t)

        prob = cp.Problem(cp.Minimize(t), constraints)

        # Try SCS first (fast for this size)
        solver_timeout = max(5.0, time_limit - 5.0)
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=5000, eps=1e-5)

        if time.monotonic() - t0 > time_limit:
            return None

        if Z.value is not None:
            a_sol = Z[:n, n].value.copy()
            a_sol = np.maximum(a_sol, 0.0)
            s = a_sol.sum()
            if s > 1e-10:
                a_sol = a_sol / s
                obj = fast_convolve(a_sol).max()
                print(f"    SDP: n={n}, t*={float(t.value):.8f}, max_conv={obj:.8f}")
                return a_sol
    except Exception as e:
        print(f"    SDP failed: {e}")

    return None


def smoothed_grad(a0, time_limit, T_init=0.001, lr_init=0.02, verbose=False):
    """Smoothed gradient descent with log-sum-exp surrogate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = 2.0 * hankel_mv(p, a)

        # Backtracking line search
        best_candidate = a
        best_val = conv.max()
        trial = lr
        for _ in range(6):
            candidate = project_simplex(a - trial * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                break
            trial *= 0.5

        a = best_candidate
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 20:
            T *= 1.1
            lr *= 0.9
        if no_improve > 60:
            T = max(T * 0.7, 1e-5)
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    smooth it={it}, obj={best_obj:.10f}, T={T:.6f}")

    return best_a, best_obj


def peak_push(a0, time_limit, verbose=False):
    """Push mass away from peak-contributing coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = np.arange(js, je + 1)
        non_contrib = np.setdiff1d(np.arange(n), contrib)

        if len(contrib) == 0 or len(non_contrib) == 0:
            break

        improved = False
        idx_max = contrib[np.argmax(a[contrib])]
        val = a[idx_max]
        if val < 1e-10:
            break

        for frac in [0.01, 0.02, 0.03, 0.05, 0.08, 0.12]:
            shift = val * frac
            if shift < 1e-12:
                continue
            nc_vals = a[non_contrib]
            nc_sum = nc_vals.sum()
            if nc_sum < 1e-15:
                delta = np.zeros(n)
                delta[idx_max] -= shift
                delta[non_contrib] += shift / len(non_contrib)
            else:
                delta = np.zeros(n)
                delta[idx_max] -= shift
                delta[non_contrib] += shift * nc_vals / nc_sum

            candidate = project_simplex(a + delta)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        if not improved and it % 200 == 0:
            rng = np.random.default_rng((it * 31 + k_peak) % (2**31 - 1))
            for _ in range(5):
                delta = np.zeros(n)
                delta[contrib] = rng.standard_normal(len(contrib)) * 0.001
                delta[non_contrib] += rng.standard_normal(len(non_contrib)) * 0.0005
                candidate = project_simplex(a + delta)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and it > 1000:
            break

    return best_a, best_obj


def interpolate_to_n(a_src, n_target):
    """Interpolate a sequence to a new length."""
    if len(a_src) == n_target:
        return a_src.copy()
    x_src = np.linspace(0, 1, len(a_src))
    x_tgt = np.linspace(0, 1, n_target)
    a_tgt = np.interp(x_tgt, x_src, a_src)
    a_tgt = np.maximum(a_tgt, 0.0)
    s = a_tgt.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return a_tgt / s

    # Multiple MW runs with different parameters
    mw_configs = [
        {'T_init': 0.003, 'eta_init': 0.02, 't_frac': 0.35},
        {'T_init': 0.008, 'eta_init': 0.05, 't_frac': 0.30},
        {'T_init': 0.001, 'eta_init': 0.01, 't_frac': 0.20},
        {'T_init': 0.015, 'eta_init': 0.08, 't_frac': 0.15},
    ]

    for ci, cfg in enumerate(mw_configs):
        remaining = deadline - time.monotonic() - (total_budget * 0.30)
        if remaining < 20:
            break
        t_alloc = min(remaining, total_budget * cfg['t_frac'])
        start = work_seq.copy()
        if ci > 0 and best_seq is not None:
            start = best_seq.copy()
        print(f"  MW config {ci} (T={cfg['T_init']}, eta={cfg['eta_init']}), t={t_alloc:.0f}s")
        a_opt, _ = mw_primal_dual(start, time_limit=t_alloc,
                                   T_init=cfg['T_init'], eta_init=cfg['eta_init'])
        try_seq(a_opt, f"mw_{ci}")
        if best_seq is not None:
            work_seq = best_seq
            n_work = len(work_seq)

    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")

    # === Phase 3: QP Primal-Dual (~400s) ===
    print(f"\n=== Phase 3: QP Primal-Dual (n={n_work}) ===")
    t3 = time.monotonic()

    if best_seq is not None:
        qp_configs = [
            {'T_init': 0.003, 'use_topk': False, 'topk': 20, 'blend': 0.4, 't_frac': 0.35},
            {'T_init': 0.01, 'use_topk': False, 'topk': 30, 'blend': 0.3, 't_frac': 0.30},
            {'T_init': 0.005, 'use_topk': True, 'topk': 8, 'blend': 0.5, 't_frac': 0.20},
            {'T_init': 0.002, 'use_topk': True, 'topk': 4, 'blend': 0.3, 't_frac': 0.15},
        ]

        for ci, cfg in enumerate(qp_configs):
            remaining = deadline - time.monotonic() - (total_budget * 0.18)
            if remaining < 20:
                break
            t_alloc = min(remaining, total_budget * cfg['t_frac'])
            print(f"  QP config {ci} (T={cfg['T_init']}, topk={cfg['use_topk']}/{cfg['topk']}), t={t_alloc:.0f}s")
            a_opt, _ = qp_primal_dual(best_seq, time_limit=t_alloc,
                                       T_init=cfg['T_init'], use_topk=cfg['use_topk'],
                                       topk=cfg['topk'], blend=cfg['blend'])
            try_seq(a_opt, f"qp_{ci}")

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")

    # === Phase 4: Nesterov + Smoothed Gradient (~250s) ===
    print(f"\n=== Phase 4: Nesterov & Smoothed Grad (n={len(best_seq) if best_seq is not None else n_work}) ===")
    t4 = time.monotonic()

    if best_seq is not None:
        # Nesterov
        for lr_i in [0.001, 0.003, 0.005]:
            remaining = deadline - time.monotonic() - (total_budget * 0.10)
            if remaining < 15:
                break
            t_alloc = min(remaining, total_budget * 0.10)
            print(f"  Nesterov lr={lr_i}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=t_alloc, lr_init=lr_i)
            try_seq(a_opt, f"nest_{lr_i}")

        # Smoothed gradient
        for T_s in [0.0003, 0.001, 0.003]:
            remaining = deadline - time.monotonic() - (total_budget * 0.05)
            if remaining < 10:
                break
            t_alloc = min(remaining, total_budget * 0.05)
            print(f"  Smooth T={T_s}, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_grad(best_seq, time_limit=t_alloc, T_init=T_s, lr_init=0.015)
            try_seq(a_opt, f"smooth_{T_s}")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # === Phase 5: Peak Push + Fine MW (~150s) ===
    print(f"\n=== Phase 5: Peak Push & Fine MW ===")
    t5 = time.monotonic()

    if best_seq is not None:
        # Peak pushing
        remaining = deadline - time.monotonic() - (total_budget * 0.05)
        if remaining > 10:
            t_alloc = min(remaining, total_budget * 0.06)
            print(f"  Peak push, t={t_alloc:.0f}s")
            a_opt, _ = peak_push(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_push")

        # Fine MW with very low T
        remaining = deadline - time.monotonic() - (total_budget * 0.03)
        if remaining > 10:
            t_alloc = min(remaining, total_budget * 0.04)
            print(f"  Fine MW T=0.0005, t={t_alloc:.0f}s")
            a_opt, _ = mw_primal_dual(best_seq, time_limit=t_alloc, T_init=0.0005, eta_init=0.01)
            try_seq(a_opt, "fine_mw")

        # Another QP with very focused top-k
        remaining = deadline - time.monotonic() - (total_budget * 0.02)
        if remaining > 10:
            t_alloc = min(remaining, total_budget * 0.03)
            print(f"  QP top-3, t={t_alloc:.0f}s")
            a_opt, _ = qp_primal_dual(best_seq, time_limit=t_alloc, T_init=0.005,
                                       use_topk=True, topk=3, blend=0.4)
            try_seq(a_opt, "qp_top3")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # === Phase 6: n-sweep (~100s) ===
    print("\n=== Phase 6: n-Sweep ===")
    t6 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        # Try nearby n values
        n_candidates = sorted(set([
            n_best - 32, n_best - 16, n_best,
            n_best + 16, n_best + 32, n_best + 64,
            n_best // 2, n_best * 2,
            512, 640, 832, 896,
        ]))
        # Filter to reasonable range
        n_candidates = [n for n in n_candidates if 64 <= n <= 2048]
        n_candidates = sorted(set(n_candidates))

        for n_try in n_candidates:
            remaining = deadline - time.monotonic() - 10
            if remaining < 8:
                break
            if n_try == n_best:
                continue

            a_interp = interpolate_to_n(best_seq, n_try)
            t_alloc = min(remaining, 12)
            # Quick MW refinement
            a_opt, _ = mw_primal_dual(a_interp, time_limit=t_alloc, T_init=0.003, eta_init=0.03)
            try_seq(a_opt, f"nsweep_n{n_try}")

    print(f"  Phase 6 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t6:.0f}s)")

    # === Phase 7: Final polish with best seed (~80s) ===
    print("\n=== Phase 7: Final Polish ===")
    t7 = time.monotonic()

    if best_seq is not None:
        n_fin = len(best_seq)

        # Several short MW runs with different seeds
        for i in range(4):
            remaining = deadline - time.monotonic() - 5
            if remaining < 5:
                break
            noise = 0.001 * (i + 1)
            perturbed = project_simplex(best_seq + rng.standard_normal(n_fin) * noise)
            t_alloc = min(remaining, 20)
            a_opt, _ = mw_primal_dual(perturbed, time_limit=t_alloc, T_init=0.002, eta_init=0.02)
            try_seq(a_opt, f"polish_mw_{i}")

        # Final Nesterov
        remaining = deadline - time.monotonic() - 3
        if remaining > 5:
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=min(remaining, 15), lr_init=0.001)
            try_seq(a_opt, "final_nest")

    print(f"  Phase 7 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t7:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
