    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u).astype(np.float64)
    rho = np.arange(1, n + 1)
    cond = (u * rho) > (cssv - 1.0)
    if not cond.any():

    r = np.nonzero(cond)[0][-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

def obj_value(a):
    n = len(a)
    b = fast_convolve(a)
    s = a.sum()
    return 2.0 * n * b.max() / (s * s)

    n = len(a)

"""
Optimized search for minimum peak autocorrelation step function.
Uses smoothed gradient descent (log-sum-exp), targeted local search,
SPSA exploration, and multi-n refinement.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u).astype(np.float64)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * rho) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    r = np.nonzero(cond)[0][-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_value(a):
    """Compute the objective: 2*n*max_b/sum_a^2."""
    n = len(a)
    b = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * b.max() / (s * s)


def gradient_smoothed(a, T):
    """
    Gradient of (1/T)*log(sum_k exp(T*b[k])) w.r.t. a, where b=conv(a,a).
    Uses the identity: grad[m] = 2 * sum_j a[j] * p[m+j]
    where p[k] = softmax(T*b)[k].
    """
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b.astype(np.float64)
    Tb -= Tb.max()
    exp_Tb = np.exp(Tb)
    p = exp_Tb / exp_Tb.sum()

    grad = np.zeros(n)
    for m in range(n):
        j_end = min(n, 2 * n - 1 - m)
        if j_end > 0:
            grad[m] = 2.0 * np.dot(a[:j_end], p[m:m + j_end])
    return grad


def smoothed_grad_descent(a0, time_limit, T_init=0.05, T_final=0.001, lr_init=0.03):
    """Projected gradient descent on log-sum-exp smoothed objective."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_obj = obj_value(a)

    t0 = time.monotonic()
    T = T_init
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        grad = gradient_smoothed(a, T)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 10
        else:
            # Backtracking line search
            alpha = lr
            improved = False
            for _ in range(25):
                candidate = project_simplex(a - alpha * grad)
                obj = obj_value(candidate)
                if obj < best_obj - 1e-15:
                    a = candidate
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
                if obj < obj_value(a) - 1e-15:
                    a = candidate
                    improved = True
                    break
                alpha *= 0.5
                if alpha < 1e-10:
                    break

            if not improved:
                no_improve += 1
            else:
                no_improve = 0

        # Adaptive T: decrease over time, with restarts
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T_sched = T_init * (T_final / T_init) ** frac
        T = max(T * 0.997, T_sched)
        lr = lr_init * max(0.3, 1.0 - 0.5 * frac)

        if no_improve > 30:
            # Perturb best and restart
            rng = np.random.default_rng((it * 6271 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            T = T_init * 0.5
            lr = lr_init
            no_improve = 0

        if no_improve > 200:
            break

    return best_a, best_obj


def targeted_local_search(a0, time_limit, seed=42):
    """Local search with targeted peak reduction and coordinate moves."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_obj = obj_value(a)

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        improved = False

        # Strategy 1: Targeted peak reduction
        if it % 3 == 0:
            b = fast_convolve(a)
            peak_idx = int(np.argmax(b))

            if 0 <= peak_idx < n:
                i_start = max(0, peak_idx - (n - 1))
                i_end = min(n - 1, peak_idx)
                if i_end >= i_start:
                    # b[peak_idx] = sum_{i=i_start}^{i_end} a[i] * a[i + (peak_idx - i)]
                    # Actually b[k] = sum_i a[i]*a[k-i] for valid i
                    # For peak_idx: valid i are max(0, peak_idx-(n-1)) to min(peak_idx, n-1)
                    i_lo = max(0, peak_idx - (n - 1))
                    i_hi = min(peak_idx, n - 1)
                    if i_hi >= i_lo:
                        products = np.array([a[i] * a[peak_idx - i] for i in range(i_lo, i_hi + 1)])
                        top_i = i_lo + int(np.argmax(products))
                        # Move weight from top_i to a random other position
                        for _ in range(5):
                            j = int(rng.integers(0, n))
                            if j == top_i:
                                continue
                            for frac in [0.005, 0.01, 0.02, 0.05]:
                                amt = a[top_i] * frac
                                if amt < 1e-14:
                                    break
                                candidate = a.copy()
                                candidate[top_i] -= amt
                                candidate[j] += amt
                                candidate = np.maximum(candidate, 0)
                                candidate /= candidate.sum()
                                obj = obj_value(candidate)
                                if obj < best_obj - 1e-13:
                                    a = candidate
                                    best_obj = obj
                                    best_a = a.copy()
                                    improved = True
                                    break
                                elif obj < obj_value(a) - 1e-13:
                                    a = candidate
                                    improved = True
                                    break
                            if improved:
                                break

        # Strategy 2: Random coordinate moves
        if not improved:
            for _ in range(8):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j or a[i] < 1e-12:
                    continue
                for frac in [0.01, 0.03, 0.1, 0.3]:
                    amt = a[i] * frac
                    candidate = a.copy()
                    candidate[i] -= amt
                    candidate[j] += amt
                    candidate = np.maximum(candidate, 0)
                    candidate /= candidate.sum()
                    obj = obj_value(candidate)
                    if obj < best_obj - 1e-13:
                        a = candidate
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                        break
                    elif obj < obj_value(a) - 1e-13:
                        a = candidate
                        improved = True
                        break
                if improved:
                    break

        # Strategy 3: Random perturbation
        if not improved:
            for scale in [0.0003, 0.001, 0.003, 0.01]:
                pert = rng.standard_normal(n) * scale
                candidate = project_simplex(a + pert)
                obj = obj_value(candidate)
                if obj < best_obj - 1e-13:
                    a = candidate
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                rng2 = np.random.default_rng(it * 31 + 7)
                a = project_simplex(best_a + rng2.standard_normal(n) * 0.005)
                no_improve = 0
                if it > 5000:
                    break

    return best_a, best_obj


def spsa_optimize(a0, time_limit, c0=0.005, step0=0.02):
    """Simultaneous Perturbation Stochastic Approximation."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_obj = obj_value(a)

    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    it = 0
    A_param = 50
    alpha = 0.6
    gamma = 0.602

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        step = step0 / (1.0 + it / A_param) ** alpha
        perturb = c0 / (1.0 + it) ** gamma

        delta = rng.integers(0, 2, size=n) * 2 - 1

        a_plus = a + perturb * delta
        a_plus = np.maximum(a_plus, 0)
        a_plus = a_plus / a_plus.sum()

        a_minus = a - perturb * delta
        a_minus = np.maximum(a_minus, 0)
        a_minus = a_minus / a_minus.sum()

        obj_plus = obj_value(a_plus)
        obj_minus = obj_value(a_minus)

        # SPSA gradient estimate and update
        gradient_est = (obj_plus - obj_minus) / (2.0 * perturb)
        a_new = a - step * gradient_est * delta
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-12:
            a_new = a_new / s
        else:
            a_new = a.copy()
        a = a_new

        obj = obj_value(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def refined_smoothed_grad(a0, time_limit, n_T=8):
    """Multi-temperature smoothed gradient descent."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_obj = obj_value(a)

    t0 = time.monotonic()
    total_time = time_limit
    per_T = total_time / n_T

    for t_idx in range(n_T):
        T_curr = 0.05 * (0.001 / 0.05) ** (t_idx / max(1, n_T - 1))
        T_curr = max(T_curr, 0.0005)
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        t_alloc = min(per_T, remaining)

        a_opt, obj_opt = smoothed_grad_descent(a, t_alloc, T_init=T_curr,
                                                T_final=T_curr * 0.3, lr_init=0.02)
        if obj_opt < best_obj:

            best_a = a_opt.copy()
            a = a_opt
        else:

            a_opt2, obj_opt2 = smoothed_grad_descent(best_a, t_alloc * 0.5,
                                                      T_init=T_curr,
                                                      T_final=T_curr * 0.3, lr_init=0.02)
