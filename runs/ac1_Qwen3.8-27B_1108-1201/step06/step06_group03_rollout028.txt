"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Gerchberg-Saxton spectral initialization (flat magnitude → flat autocorrelation)
- Smoothed subgradient descent (softmax approximation of max convolution) with adaptive scheduling
- LP-based Frank-Wolfe for final polishing
- Multi-length, multi-start exploration with diverse initializations
- Peak-targeted coordinate descent for micro-polishing
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gerchberg_saxton(n, n_iters=150, variant=0, rng=None):
    """
    Gerchberg-Saxton style algorithm: iteratively enforce flat |DFT| magnitude
    and non-negativity + normalization constraints.
    Different variants use different target magnitudes.
    """
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    if rng is None:
        rng = np.random.default_rng(42)

    # Various initializations
    if variant == 0:
        a = np.ones(n) / n
    elif variant == 1:
        x = np.arange(n, dtype=np.float64)
        a = np.cos(np.pi * x / (n - 1)) ** 2
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif variant == 2:
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * ((x - n / 2) / (n / 4)) ** 2)
        a = a / a.sum()
    elif variant == 3:
        a = rng.dirichlet(np.ones(n) * 0.5)
    else:
        a = np.ones(n) / n

    target_flat = 1.0 / (2 * n - 1)

    for it in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        phase = np.angle(A)
        mag = np.abs(A)

        if variant % 3 == 0:
            # Flat target magnitude
            target_mag = np.sqrt(target_flat) * np.ones_like(mag)
        elif variant % 3 == 1:
            # Slightly tapered (emphasize low freq)
            freqs = np.fft.rfftfreq(fft_size)
            taper = 1.0 + 0.3 * np.exp(-100 * freqs ** 2 * (fft_size ** 2))
            target_mag = np.sqrt(target_flat) * taper
        else:
            # Blend current magnitude with flat
            target_mag = 0.7 * mag + 0.3 * np.sqrt(target_flat)

        A_new = target_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a = np.ones(n) / n
            break
        a = a_new / s

    return a

    """
    Subgradient using only top-k peaks with softmax weighting within the top-k.
    Faster than full softmax when top_k << 2n-1.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(8, int((2 * n - 1) * top_k_frac))

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    momentum = 0.92
    stagnant = 0

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        # Select top-k peaks
        if top_k < len(conv):
            peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peak_indices = np.arange(len(conv))

        pv = conv[peak_indices]
        pv_max = pv.max()
        w = np.exp((pv - pv_max) / (T * 0.05))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / w_sum

        # Build sparse weights array
        weights = np.zeros(2 * n - 1)
        for i in range(len(peak_indices)):
            weights[peak_indices[i]] = w[i]

        # Gradient
        grad_raw = np.correlate(weights, a, mode='valid')
        grad = grad_raw / T

        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-30:
            grad = grad / grad_norm

        v = momentum * v - lr * grad
        best_val = conv.max()
        best_candidate = a
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            c = project_simplex(a + frac * v)
            val = fast_convolve(c).max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = c

        a = best_candidate
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 20:
            T *= 1.06
            lr *= 0.95
        if stagnant > 80:
            T = T_init * 0.4
            lr = lr_init * 1.4
            v = np.zeros(n)
            stagnant = 0
        if stagnant > 250:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            T = T_init
            lr = lr_init
            v = np.zeros(n)
            stagnant = 0

    return best_a, best_obj


def fw_lp_optimize(a0, time_limit, max_iters=25, relax=1.002):
    """Frank-Wolfe with LP oracle for final polishing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_obj = M

    t0 = time.monotonic()
    it = 0

    while it < max_iters and time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        # Build convolution matrix
        A_mat = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_mat[j:j + n, j] = a

        c_vec = -np.ones(n)
        b_vec = np.full(2 * n - 1, M * relax)
        bounds = [(0, None)] * n

        try:
            lp_time = min(remaining - 5, 30)
            res = linprog(c_vec, A_ub=A_mat, b_ub=b_vec, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': lp_time})
            if not (res.success and res.x is not None):
                continue
            g = np.maximum(res.x, 0.0)
            g_sum = g.sum()
            if g_sum < 1e-15:
                continue
            g = g / g_sum

            if np.max(np.abs(g - a)) < 1e-10:
                break

            # Line search
            best_alpha = 0.0
            best_val = M
            for alpha in np.linspace(0.05, 1.0, 25):
                ft = (1 - alpha) * a + alpha * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha

            if best_alpha > 1e-8:
                a = (1 - best_alpha) * a + best_alpha * g
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M = fast_convolve(a).max()
                if M < best_obj:
                    best_obj = M
                    best_a = a.copy()
                    relax = 1.002
                else:
                    relax = min(relax * 1.03, 1.02)
        except Exception:
            break

    return best_a, best_obj


def peak_coordinate_descent(a0, time_limit):
    """Coordinate descent targeting top peaks by shifting mass."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    step_scale = 0.003

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        n_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        improved = False
        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue

            # Find top contributing pairs
            contributions = []
            for j in range(js, je + 1):
                contributions.append((a[j] * a[k - j], j, k - j))
            contributions.sort(reverse=True)

            for (_, j1, j2) in contributions[:6]:
                if a[j1] < 1e-8 and a[j2] < 1e-8:
                    continue

                rng = np.random.default_rng((it * 9973 + k * 31 + j1 * 7 + j2 * 13) % (2**31))
                for _att in range(3):
                    j_target = int(rng.integers(0, n))
                    if j_target in (j1, j2):
                        continue
                    for src in [j1, j2]:
                        if a[src] < 1e-8:
                            continue
                        for df in [step_scale, step_scale * 0.5, step_scale * 2.0]:
                            delta = a[src] * df
                            candidate = a.copy()
                            candidate[src] -= delta
                            candidate[j_target] += delta
                            if candidate[src] < -1e-12:
                                continue
                            candidate = np.maximum(candidate, 0)
                            cs = candidate.sum()
                            if cs < 1e-15:
                                continue
                            candidate = candidate / cs
                            val = fast_convolve(candidate).max()
                            if val < best_obj - 1e-15:
                                a = candidate
                                best_obj = val
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            step_scale *= 0.94
            if step_scale < 1e-7:
                break
            if it % 300 == 0:
                rng = np.random.default_rng(it * 48271)
                candidate = project_simplex(a + rng.standard_normal(n) * step_scale * 3)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
            elif it > 3000:
                break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length via interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_initializations(n, parent, rng):
    """Generate diverse initializations for optimization."""
    starts = []

    # Parent resampled
    if parent is not None:
        starts.append(resample_sequence(parent, n))
        starts.append(resample_sequence(parent[::-1], n))

    # Gerchberg-Saxton variants
    for v in range(4):
        gs = gerchberg_saxton(n, n_iters=80, variant=v, rng=rng)
        starts.append(gs)

    # Uniform
    starts.append(np.ones(n) / n)

    # Cosine taper
    x = np.arange(n, dtype=np.float64)
    cos_t = np.cos(np.pi * x / (n - 1))
    w = 0.5 * (1 + cos_t)
    w = np.maximum(w, 0.01)
    starts.append(w / w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - (n - 1) / 2) / (0.15 * n)) ** 2)
    starts.append(g / g.sum())

    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    starts.append(edge / edge.sum())

    # Two peaks
    g1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.06 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.06 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())

    # Random Dirichlet
    starts.append(rng.dirichlet(np.ones(n) * 0.4))
    starts.append(rng.dirichlet(np.ones(n) * 0.2))

    # Lattice-like (periodic)
    period = max(3, n // 30)
    lat = np.array([1.0 if (i % period) < period // 2 else 0.1 for i in range(n)])
    starts.append(lat / lat.sum())

    return starts


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search for minimizing
    2*n*max(conv(a,a))/(sum(a))^2.

    Phases:
    1. Diverse initialization + quick subgradient screening (250s)
    2. Deep smoothed subgradient on best candidates (500s)
    3. TopK subgradient refinement (250s)
    4. LP-based Frank-Wolfe polishing (400s)
    5. Peak coordinate descent micro-polish (remaining time)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_seq = None
    best_score = float('inf')
    best_obj_internal = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score, best_obj_internal
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        internal_obj = fast_convolve(a_norm).max()
        if internal_obj < best_obj_internal:
            best_obj_internal = internal_obj
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: score={score:.10f}, internal={internal_obj:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")

    n_parent = len(parent)

    # === Phase 1: Diverse initialization + quick screening (250s) ===
    print("\n=== Phase 1: Diverse Initialization + Quick Screening ===")
    t_phase = time.monotonic()

    # Determine candidate lengths
    lengths = sorted(set([n_parent, max(256, n_parent - 128), n_parent + 128,
                          512, 640, 768, 896, 1024, 1280]))
    lengths = [l for l in lengths if 192 <= l <= 1536]

    quick_results = []  # (score, sequence) from quick optimization

    for n_try in lengths:
        if time.monotonic() > deadline - 900:
            break

        starts = generate_initializations(n_try, parent, rng)

        for si, start in enumerate(starts[:6]):  # Max 6 starts per n
            if time.monotonic() > deadline - 850:
                break
            t_alloc = min(25.0, deadline - time.monotonic() - 850)
            if t_alloc < 5:
                break
            a_opt, obj = smoothed_subgradient(start, time_limit=t_alloc,
                                              T_init=0.001, lr_init=0.01)
            score = try_seq(a_opt, f"quick_n{n_try}_s{si}")
            if np.isfinite(score):
                quick_results.append((score, a_opt))

    # Sort by score
    quick_results.sort(key=lambda x: x[0])
    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep smoothed subgradient (500s) ===
    print("\n=== Phase 2: Deep Smoothed Subgradient ===")
    t_phase = time.monotonic()

    # Work with top candidates from Phase 1
    candidates = []
    if best_seq is not None:
        candidates.append(best_seq)
    for sc, seq in quick_results[:3]:
        if len(seq) != len(candidates[-1]) if candidates else True:
            candidates.append(seq)

    seen_n = set()
    final_candidates = []
    for c in candidates:
        if len(c) not in seen_n:
            final_candidates.append(c)
            seen_n.add(len(c))

    for ci, cand in enumerate(final_candidates[:3]):
        if time.monotonic() > deadline - 450:
            break

        n_c = len(cand)
        t_alloc = min(180.0, deadline - time.monotonic() - 450)
        if t_alloc < 20:
            break

        print(f"  Deep SS candidate {ci} (n={n_c}), t={t_alloc:.0f}s")
        a_opt, obj = smoothed_subgradient(cand, time_limit=t_alloc,
                                          T_init=0.0005, lr_init=0.007)
        try_seq(a_opt, f"deep_ss_{ci}")

        # Also try with different parameters
        if time.monotonic() < deadline - 420:
            t_alloc2 = min(100.0, deadline - time.monotonic() - 420)
            if t_alloc2 > 15:
                a_opt2, obj2 = topk_subgradient(cand, time_limit=t_alloc2,
                                                 T_init=0.0004, lr_init=0.012,
                                                 top_k_frac=0.03)
                try_seq(a_opt2, f"topk_{ci}")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: TopK subgradient with multiple configs (250s) ===
    print("\n=== Phase 3: TopK Subgradient Refinement ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        configs = [
            {'T_init': 0.0003, 'lr_init': 0.01, 'top_k_frac': 0.02},
            {'T_init': 0.0008, 'lr_init': 0.008, 'top_k_frac': 0.05},
            {'T_init': 0.002, 'lr_init': 0.006, 'top_k_frac': 0.1},
            {'T_init': 0.0005, 'lr_init': 0.015, 'top_k_frac': 0.03},
        ]

        for ci, cfg in enumerate(configs):
            if time.monotonic() > deadline - 350:
                break
            t_alloc = min(70.0, deadline - time.monotonic() - 350)
            if t_alloc < 15:
                break
            print(f"  TopK config {ci} (T={cfg['T_init']}), t={t_alloc:.0f}s")
            a_opt, _ = topk_subgradient(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"topk_refine_{ci}")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: LP-based Frank-Wolfe polishing (350s) ===
    print("\n=== Phase 4: LP Frank-Wolfe Polishing ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Main FW
        t_alloc = min(250.0, deadline - time.monotonic() - 150)
        if t_alloc > 30:
            print(f"  Main FW, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, max_iters=20, relax=1.002)
            try_seq(a_opt, "fw_main")

        # Perturbed restarts
        n_best = len(best_seq)
        for scale in [0.002, 0.005, 0.01]:
            if time.monotonic() > deadline - 100:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(60.0, deadline - time.monotonic() - 100)
            if t_alloc < 20:
                break
            print(f"  FW perturb scale={scale}, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc, max_iters=10, relax=1.003)
            try_seq(a_opt, f"fw_pert_{scale}")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Peak coordinate descent micro-polish (remaining) ===
    print("\n=== Phase 5: Peak Coordinate Descent ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(80.0, deadline - time.monotonic() - 20)
        if t_alloc > 10:
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_final")

        # One more round of quick SS
        if time.monotonic() < deadline - 15:
            t_alloc2 = min(30.0, deadline - time.monotonic() - 10)
            if t_alloc2 > 10:
                noise = rng.standard_normal(len(best_seq)) * 0.001
                start = project_simplex(best_seq + noise)
                a_opt2, _ = smoothed_subgradient(start, time_limit=t_alloc2,
                                                  T_init=0.0003, lr_init=0.004)
                try_seq(a_opt2, "final_ss")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Try a different n if time permits ===
    print("\n=== Phase 6: Different Length Exploration ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_n = n_best + 128 if n_best < 1024 else n_best - 128
        if 256 <= alt_n <= 1536 and time.monotonic() < deadline - 60:
            t_alloc = min(50.0, deadline - time.monotonic() - 30)
            if t_alloc > 20:
                alt_start = resample_sequence(best_seq, alt_n)
                a_opt, _ = smoothed_subgradient(alt_start, time_limit=t_alloc,
                                                T_init=0.001, lr_init=0.01)
                try_seq(a_opt, f"alt_n{alt_n}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
