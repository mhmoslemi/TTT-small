```python
import numpy as np
import time
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def hankel_mv(w, a):
    """Compute W @ a where W_{i,j} = w[i+j]. w has length 2n-1, a has length n."""
    return np.correlate(w, a, mode='valid')

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
- Continuation method on temperature
- Final polish with scipy
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def hankel_mv(w, a):
    """Compute W @ a where W_{i,j} = w[i+j]. w has length 2n-1, a has length n."""
    return np.correlate(w, a, mode='valid')


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_qp_fista(w, a_init, n, time_limit=10.0, verbose=False):
    """
    Solve min a^T H a s.t. sum(a)=1, a>=0
    where H_{i,j} = w[i+j] (Hankel matrix, PSD).
    Uses FISTA with FFT-accelerated matrix-vector products.
    """
    t0 = time.monotonic()
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    t_mom = 1.0

    # Compute Lipschitz constant L = 2 * max_row_sum(H)
    # max_row_sum = max_i sum_j w[i+j]
    # For efficiency, compute using the structure
    row_sums = np.correlate(w, np.ones(n), mode='valid')
    L = 2.0 * row_sums.max()
    L = max(L, 1e-10)

    best_a = a.copy()
    best_obj = np.dot(a, hankel_mv(w, a))
    no_improve = 0

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # FISTA update
        Hy = hankel_mv(w, y)
        grad = 2.0 * Hy
        z = y - (1.0 / L) * grad
        a_new = project_simplex(z)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2)) / 2.0
        y = a_new + ((t_mom - 1.0) / t_new) * (a_new - a)
        a = a_new
        t_mom = t_new

        obj = np.dot(a, hankel_mv(w, a))
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 5000 or it > 100000:
            break

    return best_a


def primal_dual(a0, time_limit, T_init=0.01, use_topk=False, topk=20,
                blend=0.5, verbose=False):
    """
    Primal-dual iteration for min-max convolution.
    - Dual: compute weights w from conv(a,a)
    - Primal: solve QP via FISTA
    - Blend and iterate
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        conv = fast_convolve(a)
        cmax = conv.max()

        # Compute dual weights
        if use_topk:
            k_use = min(topk, len(conv))
            peaks = np.argpartition(conv, -k_use)[-k_use:]
            w = np.zeros(len(conv))
            w[peaks] = 1.0 / k_use
        else:
            scaled = conv / T
            scaled -= scaled.max()
            w = np.exp(scaled)
            w = w / w.sum()

        # Solve QP via FISTA
        qp_time = min(remaining * 0.5, 30.0)
        a_qp = solve_qp_fista(w, a, n, time_limit=qp_time)

        # Blend
        a_new = (1.0 - blend) * a + blend * a_qp
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a_qp
        else:
            a_new = a_new / s_new

        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adapt temperature and blend
        if no_improve > 5:
            T *= 1.3
            blend = max(0.1, blend * 0.9)
        if no_improve > 20:
            T *= 0.5  # Re-focus
            blend = min(0.9, blend * 1.2)
        if no_improve > 80:
            # Restart from best with perturbation
            rng = np.random.default_rng((it * 104729 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
            T = T_init
            blend = 0.5
            no_improve = 0

        if verbose and it % 10 == 0:
            print(f"    PD it={it}, obj={best_obj:.10f}, T={T:.6f}, no_imp={no_improve}")

    return best_a, best_obj


def symmetric_primal_dual(a0, time_limit, T_init=0.01, verbose=False):
    """
    Primal-dual with symmetry constraint a_i = a_{n-1-i}.
    Reduces the problem to n/2 variables.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    half = (n + 1) // 2

    # Work with the first half
    a_half = a[:half].copy()
    if n % 2 == 0:
        a_half = 0.5 * (a[:n//2] + a[n//2-1::-1])
        a_half = a_half[:n//2]
    else:
        a_half = 0.5 * (a[:half] + a[:half-1:-1])
        a_half = a_half[:half]
    
    # Ensure symmetry
    def expand(half_a, n):
        if n % 2 == 0:
            return np.concatenate([half_a, half_a[::-1]])
        else:
            return np.concatenate([half_a, half_a[-2::-1]])

    a = expand(a_half, n)
    a = a / a.sum()

    T = T_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    blend = 0.5

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        conv = fast_convolve(a)
        cmax = conv.max()

        # Dual weights
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()

        # QP with FISTA
        qp_time = min(remaining * 0.5, 20.0)
        a_qp = solve_qp_fista(w, a, n, time_limit=qp_time)

        # Enforce symmetry
        a_qp = expand(0.5 * (a_qp[:half] + a_qp[:half-1:-1])[:half] if n % 2 else
                      0.5 * (a_qp[:n//2] + a_qp[n//2-1::-1])[:n//2], n)
        a_qp = a_qp / a_qp.sum()

        # Blend
        a_new = (1.0 - blend) * a + blend * a_qp
        a_new = project_simplex(a_new)
        # Enforce symmetry
        a_new = expand(0.5 * (a_new[:half] + a_new[:half-1:-1])[:half] if n % 2 else
                       0.5 * (a_new[:n//2] + a_new[n//2-1::-1])[:n//2], n)
        a_new = a_new / a_new.sum()

        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 10:
            T *= 1.2
            blend = max(0.1, blend * 0.9)
        if no_improve > 50:
            rng = np.random.default_rng((it * 7919 + 1) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            T = T_init
            blend = 0.5
            no_improve = 0

    return best_a, best_obj


def multi_peak_duality(a0, time_limit, verbose=False):
    """
    Alternative dual update: use the top peaks with adaptive weights.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    T = 0.01

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        conv = fast_convolve(a)
        cmax = conv.max()

        # Adaptive: use weighted combination of top-k peaks with softmax
        k = min(10, len(conv))
        top_vals = np.sort(conv)[::-1][:k]
        # Weights proportional to how much each peak exceeds the average
        avg = conv.mean()
        exceed = np.maximum(conv - avg, 0)
        if exceed.sum() > 1e-15:
            w = exceed / exceed.sum()
        else:
            w = np.ones(len(conv)) / len(conv)

        # QP
        qp_time = min(remaining * 0.4, 15.0)
        a_qp = solve_qp_fista(w, a, n, time_limit=qp_time)

        # Step size based on duality gap
        a_new = project_simplex(a + 0.3 * (a_qp - a))
        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 30:
            rng = np.random.default_rng((it * 6151 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            no_improve = 0

    return best_a, best_obj


def smoothed_convex_opt(a0, time_limit, T_schedule=None, verbose=False):
    """
    Direct optimization of the smooth surrogate:
    f_T(a) = T * log(sum_k exp(conv_k / T))
    subject to a in simplex.
    Uses FISTA with gradient = 2 * sum_k p_k * M^(k) a = 2 * hankel_mv(p, a) * 2...
    Actually grad = 2 * H(p) * a where p[k] = softmax(conv/T).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    if T_schedule is None:
        T_schedule = np.geomspace(1.0, 0.001, 10)

    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    for T in T_schedule:
        T_time = max(1.0, min(30.0, (time_limit - (time.monotonic() - t0)) / len(T_schedule)))
        
        # Compute L for this T
        # L = 2 * lambda_max(H(p)) where p depends on a
        # Start with L = 2 * max_row_sum
        for _ in range(20):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_convolve(a)
            scaled = conv / T
            scaled -= scaled.max()
            p = np.exp(scaled)
            p = p / p.sum()
            
            row_sums = np.correlate(p, np.ones(n), mode='valid')
            L = 2.0 * row_sums.max() * 2.0  # factor of 2 for gradient
            L = max(L, 1e-10)
            
            # FISTA for a few steps
            y = a.copy()
            t_mom = 1.0
            for _ in range(100):
                if time.monotonic() - t0 > time_limit:
                    break
                Hy = hankel_mv(p, y)
                grad = 2.0 * Hy
                z = y - (1.0 / L) * grad
                a_new = project_simplex(z)
                t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2)) / 2.0
                y = a_new + ((t_mom - 1.0) / t_new) * (a_new - a)
                a = a_new
                t_mom = t_new
            
            obj = fast_convolve(a).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    return best_a, best_obj


def scipy_polish(a0, time_limit, verbose=False):
    """
    Final polish using scipy SLSQP on the log-sum-exp surrogate.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = 0.005

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    def objective(b):
        if time.monotonic() - t0 > time_limit:
            return 1e10
        aa = np.exp(b)
        aa = aa / aa.sum()
        conv = fast_convolve(aa)
        # Log-sum-exp for numerical stability
        m = conv.max()
        val = m + T * np.log(np.sum(np.exp((conv - m) / T)))
        return val

    def gradient(b):
        aa = np.exp(b)
        aa = aa / aa.sum()
        conv = fast_convolve(aa)
        m = conv.max()
        p = np.exp((conv - m) / T)
        p = p / p.sum()
        # d/db_k [T * log sum exp(conv/T)] = sum_j (da_j/db_k) * d(a^T H a)/da_j
        # da/db = diag(a) - a a^T
        # grad_a = 2 H(p) a
        # grad_b = (diag(a) - a a^T)^T grad_a = a * grad_a - a * sum(a * grad_a)
        grad_a = 2.0 * hankel_mv(p, aa)
        grad_b = aa * grad_a - aa * np.dot(aa, grad_a)
        return grad_b

    # Start from log(a)
    b0 = np.log(np.maximum(a, 1e-10))

    result = scipy_minimize(objective, b0, method='SLSQP',
                            jac=gradient,
                            options={'maxiter': 50, 'ftol': 1e-10})
    
    if result.success:
        a_new = np.exp(result.x)
        a_new = a_new / a_new.sum()
        score = float(evaluate_sequence(a_new.tolist()))
        if score < best_score:
            best_score = score
            best_a = a_new

    return best_a, best_score
```

def project_simplex(v):
    """Euclidean projection onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])

    cssv = np.cumsum(u)

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]

    return np.maximum(v - theta, 0.0)
