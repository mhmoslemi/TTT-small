Hybrid optimization for minimizing 2*n*max(f*f)/(sum f)^2:
- Active-set Frank-Wolfe with adaptive constraint selection
- Coordinate-wise 1D optimization sweeps
- Smoothed gradient descent with multiple temperatures
- Multi-peak subgradient with momentum
- Cosine/Gaussian window initializations for diversity
- Final polish with combined methods

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]."""
    return np.correlate(p, a, mode='valid')


def solve_active_lp(f, M, active_indices):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_indices, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((len(active_indices), n))
    for i, k in enumerate(active_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A_ub[i, js:je+1] = f[je-js::-1]  # f[k - j] for j in [js, je]
    b_ub = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

                scale = 0.001 + 0.005 * min(1.0, no_improve / 100.0)

                m_exp = 1.0

        best_val = M

            if st < 1e-15:

            val = fast_convolve(ft).max()
            if val < best_val:

            conv = fast_convolve(f)

            m_exp = 1.0

def coordinate_wise_optimize(a0, time_limit, n_sweeps_max=None):
    """Coordinate-wise 1D optimization with renormalization."""
    a = np.maximum(a0.astype(np.float64), 0.0)

    sweep = 0

    while time.monotonic() - t0 < time_limit:

            if time.monotonic() - t0 > time_limit - 0.05:
                return best_a, best_obj
            
            best_local = a[i]
            best_local_obj = 2.0 * n * fast_convolve(a).max() / (a.sum()**2)

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]."""
    return np.correlate(p, a, mode='valid')


def score_of(a):
    """Compute the evaluation score (without calling evaluate_sequence)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def solve_active_lp(f, M, active_indices):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_indices, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((len(active_indices), n))
    for i, k in enumerate(active_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A_ub[i, js:je + 1] = f[js:je + 1][::-1]
    b_ub = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def coordinate_wise_optimize(a0, time_limit, multipliers=None):
    """
    Coordinate-wise optimization with O(n) incremental convolution update.
    Key insight: changing a_i by delta only affects n entries of the convolution.
    """
    if multipliers is None:
        multipliers = [0.0, 0.2, 0.5, 0.7, 0.9, 1.1, 1.3, 1.6, 2.0, 3.0]

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()

    t0 = time.monotonic()
    sweep = 0
    stale = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 0.05:
                break

            best_mult = 1.0
            best_local_obj = best_obj
            v = a[i]

            if v < 1e-12:
                # Only try positive multipliers for zero coordinates
                try_mults = [0.01, 0.05, 0.1, 0.3, 1.0]
            else:
                try_mults = multipliers

            for mult in try_mults:
                if mult == 1.0:
                    continue
                v_new = v * mult
                if v_new < 0:
                    continue
                s_new = s - v + v_new
                if s_new < 1e-10:
                    continue

                delta = v_new - v
                # Update convolution incrementally
                new_conv = conv.copy()
                lo = i
                hi = min(i + n, 2 * n - 1)
                if lo < hi:
                    new_conv[lo:hi] += delta * a[hi - 1 - lo:hi - 1]
                if 2 * i < 2 * n - 1:
                    new_conv[2 * i] -= delta * a[0]

                obj = 2.0 * n * new_conv.max() / (s_new * s_new)
                if obj < best_local_obj - 1e-13:
                    best_local_obj = obj
                    best_mult = mult
                    new_conv_best = new_conv.copy()
                    s_best = s_new

            if best_mult != 1.0:
                a[i] = v * best_mult
                s = s_best
                conv = new_conv_best
                improved = True

        if improved:
            current_obj = 2.0 * n * conv.max() / (s * s)
            if current_obj < best_obj - 1e-13:
                best_obj = current_obj
                best_a = a.copy()
                stale = 0
            else:
                stale += 1
        else:
            stale += 1

        if stale > 50:
            rng = np.random.default_rng(sweep * 12345 + 7)
            scale = 0.001 + 0.003 * min(1.0, stale / 100.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            s = a.sum()
            conv = fast_convolve(a)
            best_obj = 2.0 * n * conv.max() / (s * s)
            best_a = a.copy()
            stale = 0

    return best_a, best_obj

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)

def multi_peak_sg(a0, time_limit, top_k, lr_init, mom):
    """Multi-peak subgradient with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]

        v = mom * v - lr * sg
        a = project_simplex(a + v)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        mom = min(0.98, mom + 0.0001)

        if no_improve > 150:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.001 + 0.003 * min(1.0, (no_improve - 150) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init * 0.5
            no_improve = 0

    return best_a, best_obj


def generate_structured_windows(n):
    """Generate diverse structured window initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / (n - 1) if n > 1 else np.array([0.5])

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine window (Hann)
    w = 0.5 * (1 + np.cos(2 * np.pi * xn))
    inits.append(w / w.sum())

    # Flat-top with cosine taper
    taper = 0.3
    w = np.ones(n)
    t = taper * n
    for i in range(int(t)):
        w[i] = 0.5 * (1 - np.cos(np.pi * i / t))
        w[n - 1 - i] = w[i]
    inits.append(w / w.sum())

    # Gaussian centered
    sigma = 0.2
    w = np.exp(-0.5 * ((xn - 0.5) / sigma) ** 2)
    inits.append(w / w.sum())

    # Raised cosine (Blackman)
    w = 0.42 + 0.5 * np.cos(2 * np.pi * xn) + 0.08 * np.cos(4 * np.pi * xn)
    w = np.maximum(w, 0.01)
    inits.append(w / w.sum())

    # Two-lobe structure
    w = np.zeros(n)
    half = n // 2
    gap = max(1, n // 20)
    w[:half - gap] = 1.0
    w[half + gap:] = 1.0
    if w.sum() > 0:
        inits.append(w / w.sum())

    # Power-law decay from center
    dist = np.abs(xn - 0.5) / 0.5
    w = (1 - dist) ** 2
    inits.append(w / w.sum())

    # Sinc-like (truncated)
    w = np.sinc((xn - 0.5) * 2)
    w = np.maximum(w, 0.0)
    if w.sum() > 0:
        inits.append(w / w.sum())

    return inits

    # === Phase 1: Multi-n, Multi-structure Exploration (~200s) ===
    print("\n=== Phase 1: Multi-n, Multi-structure Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent, 256, 384, 512, 640, 768, 1024]))
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 2048]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1300
        if remaining < 20:
            break

        # Generate structured initializations
        windows = generate_structured_windows(n_try)
        # Also interpolate parent
        if parent is not None and n_try != n_parent:
            px = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_parent), parent)
            px = np.maximum(px, 0)
            if px.sum() > 0:
                windows.append(px / px.sum())

        best_for_n = None
        best_obj_n = float('inf')

        for wi, w in enumerate(windows):
            remaining = deadline - time.monotonic() - 1300
            if remaining < 5:
                break
            t_alloc = min(remaining, 15)
            a_opt, obj = fw_active_lp(w, time_limit=t_alloc, K_frac=0.3)
            if obj < best_obj_n:
                best_obj_n = obj
                best_for_n = a_opt

        if best_for_n is not None:
            try_and_update(best_for_n, state, "phase1_n{}_fw".format(n_try))

            remaining = deadline - time.monotonic() - 1250
            if remaining > 5:
                a_opt2, _ = smoothed_grad_descent(best_for_n, time_limit=min(remaining, 10),
                                                   T_init=0.001, lr_init=0.03)
                try_and_update(a_opt2, state, "phase1_n{}_sg".format(n_try))

    print("  Phase 1 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 2: Deep Active-Set FW-LP (~500s) ===
    print("\n=== Phase 2: Deep Active-Set FW-LP ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])
        remaining = deadline - time.monotonic() - 800
        if remaining > 30:
            t_alloc = min(remaining, 400)
            print("  Deep FW-LP n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = fw_active_lp(state[1], time_limit=t_alloc, K_frac=0.5)
            try_and_update(a_opt, state, "deep_fw")

        # Perturbed restarts
        for scale in [0.003, 0.008, 0.015, 0.025]:
            remaining = deadline - time.monotonic() - 600
            if remaining < 15:
                break
            f_pert = project_simplex(state[1] + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 50)
            a_opt2, _ = fw_active_lp(f_pert, time_limit=t_alloc, K_frac=0.4)
            try_and_update(a_opt2, state, "fw_pert_{}".format(scale))

    print("  Phase 2 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 3: Coordinate-wise Optimization (~500s) ===
    print("\n=== Phase 3: Coordinate-wise Optimization ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        remaining = deadline - time.monotonic() - 400
        if remaining > 10:
            t_alloc = min(remaining, 400)
            print("  Coordinate-wise, t={:.0f}s".format(t_alloc))
            mults = [0.0, 0.1, 0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.6, 2.0, 3.0]
            a_opt, _ = coordinate_wise_optimize(state[1], time_limit=t_alloc, multipliers=mults)
            try_and_update(a_opt, state, "coord_wise")

        # Second pass with different multipliers
        remaining = deadline - time.monotonic() - 300
        if remaining > 10 and state[1] is not None:
            t_alloc = min(remaining, 100)
            mults2 = [0.0, 0.5, 0.8, 1.2, 1.5]
            a_opt2, _ = coordinate_wise_optimize(state[1], time_limit=t_alloc, multipliers=mults2)
            try_and_update(a_opt2, state, "coord_wise2")

    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 4: Smoothed Gradient + Multi-peak SG (~350s) ===
    print("\n=== Phase 4: Gradient Methods ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        for T_init, lr_init in [(0.0003, 0.04), (0.001, 0.03), (0.003, 0.02), (0.0005, 0.05)]:
            remaining = deadline - time.monotonic() - 200
            if remaining < 10:
                break
            t_alloc = min(remaining, 60)
            print("  Smoothed grad T={}, lr={}, t={:.0f}s".format(T_init, lr_init, t_alloc))
            a_opt, _ = smoothed_grad_descent(state[1], time_limit=t_alloc,
                                             T_init=T_init, lr_init=lr_init)
            try_and_update(a_opt, state, "smooth_T{}".format(T_init))

        sg_configs = [
            (10, 0.003, 0.93),
            (25, 0.002, 0.94),
            (50, 0.001, 0.95),
            (100, 0.001, 0.96),
        ]
        for top_k, lr, mom in sg_configs:
            remaining = deadline - time.monotonic() - 100
            if remaining < 10:
                break
            t_alloc = min(remaining, 50)
            print("  Multi-peak SG k={}, t={:.0f}s".format(top_k, t_alloc))
            a_opt, _ = multi_peak_sg(state[1], time_limit=t_alloc,
                                     top_k=top_k, lr_init=lr, mom=mom)
            try_and_update(a_opt, state, "mpsg_k{}".format(top_k))

    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 5: Final Polish (~150s) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Coordinate-wise final polish
        remaining = deadline - time.monotonic() - 50
        if remaining > 5:
            t_alloc = min(remaining, 60)
            print("  Final coordinate-wise, t={:.0f}s".format(t_alloc))
            mults_final = [0.0, 0.3, 0.7, 0.9, 1.05, 1.1, 1.2, 1.5]
            a_opt, _ = coordinate_wise_optimize(state[1], time_limit=t_alloc, multipliers=mults_final)
            try_and_update(a_opt, state, "final_coord")

        # Quick FW
        remaining = deadline - time.monotonic() - 30
        if remaining > 5:
            t_alloc = min(remaining, 30)
            print("  Final FW, t={:.0f}s".format(t_alloc))
            a_opt, _ = fw_active_lp(state[1], time_limit=t_alloc, K_frac=0.6)
            try_and_update(a_opt, state, "final_fw")

        # Fine smoothed gradient
        remaining = deadline - time.monotonic() - 15
        if remaining > 5:
            t_alloc = min(remaining, 15)
            a_opt, _ = smoothed_grad_descent(state[1], time_limit=t_alloc,
                                             T_init=0.0001, lr_init=0.005)
            try_and_update(a_opt, state, "final_smooth")

        # Tiny perturbations
        remaining = deadline - time.monotonic() - 5
        if remaining > 3:
            for _ in range(3):
                remaining = deadline - time.monotonic() - 3
                if remaining < 2:
                    break
                noise = rng.standard_normal(len(state[1])) * 0.0005
                f_pert = project_simplex(state[1] + noise)
                a_opt, _ = coordinate_wise_optimize(f_pert, time_limit=min(remaining, 5),
                                                     multipliers=[0.5, 0.8, 1.2, 1.5])
                try_and_update(a_opt, state, "perturb_fine")

    print("  Phase 5 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(384) / 384
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(state[1].tolist()))
        if np.isfinite(fs) and fs < state[0]:
            state[0] = fs
    except Exception:
        pass

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
