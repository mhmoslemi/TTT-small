**Key insight for improvement: The "all-peaks" subgradient**

Instead of just using the argmax, use a softmax-weighted combination of subgradients from all peaks:

g[j] = sum_k p[k] * 2*a[k-j] (for valid j)

where p[k] = exp(c[k]/T) / sum exp(c[k']/T)

This is the gradient of the log-sum-exp smooth approximation to max(c[k]). As T -> 0, it converges to the argmax subgradient.

The advantage of using all peaks is that it provides a smoother, more informative gradient direction.

**Another key insight: The problem has a lot of symmetry**

If a is a solution, then reversed(a) is also a solution (since the auto-convolution is the same). Also, any cyclic shift changes the auto-convolution in a predictable way.

We can exploit this by working in a "symmetry-reduced" space.

```python
def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
def optimized_subgrad(a0, time_limit, T_init=0.001, T_min=0.0001, lr_init=0.01,
                      use_nesterov=True, verbose=False):
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    T = T_init
    lr = lr_init
    beta = 0.9  # momentum
    
    a_prev = a.copy()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit - 1:
        it += 1
        
        # Nesterov lookahead
        if use_nesterov:
            y = a + beta * (a - a_prev)
        else:
            y = a.copy()
        
        # Compute smoothed gradient
        c = fast_convolve(y)
        scaled = c / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        grad = 2.0 * np.correlate(p, y, mode='full')[:n]
        
        # Step
        a_new = project_simplex(y - lr * grad)
        
        # Update
        a_prev = a.copy()
        a = a_new
        beta = min(0.99, beta + 0.0001)
        
        # Evaluate
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt
        if no_improve > 20:
            T = max(T_min, T * 0.95)
            lr *= 0.98
        if no_improve > 100:
            T = T_init * 0.5
            lr = lr_init * 0.5
            beta = 0.9
        if no_improve > 300:
            # Restart
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            a_prev = a.copy()
            T = T_init
            lr = lr_init
            beta = 0.9
            no_improve = 0
        
        if verbose and it % 1000 == 0:
            print(f"  subgrad it={it}, obj={best_obj:.10f}, T={T:.6f}, lr={lr:.6f}")
    
    return best_a, best_obj
```

**Approach: Coordinate descent with exact 1D optimization**

For each coordinate j, if we fix all other coordinates, the objective becomes:

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(a, T):
    """Compute (smoothed objective value, gradient, softmax weights, conv)."""
    c = fast_convolve(a)
    scaled = c / T
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    f = T * np.log(exp_scaled.sum())  # = max(c) + T*log(sum(exp(scaled))) - max(c)
    # Actually f_T = T * logsumexp(c/T), the constant max(c) doesn't matter for optimization
    grad = 2.0 * np.correlate(p, a, mode='full')[:len(a)]
    return f, grad, p, c

        a = a_trial

def peak_shave(a0, time_limit, verbose=False, seed=42):
    """
    Peak-shaving: iteratively identify the dominant peak and shift mass
    from high-contributing positions to low-contributing ones.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    rng = np.random.default_rng(seed)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit - 1:
        it += 1

        c = fast_convolve(a)
        k_peak = int(np.argmax(c))
        peak_val = c[k_peak]

        # Positions contributing to this peak
        js_lo = max(0, k_peak - n + 1)
        js_hi = min(n - 1, k_peak)
        contrib = np.arange(js_lo, js_hi + 1)
        non_contrib = np.setdiff1d(np.arange(n), contrib)

        if len(contrib) == 0 or len(non_contrib) == 0:
            break

        # Contribution of each position j to peak k_peak: a[j] * a[k_peak - j]
        contrib_vals = a[contrib] * a[k_peak - contrib]
        total_contrib = contrib_vals.sum()

        if total_contrib < 1e-15:
            break

        # Find position with highest contribution
        idx_max_local = np.argmax(contrib_vals)
        idx_max = contrib[idx_max_local]
        max_contrib = contrib_vals[idx_max_local]

        if max_contrib < 1e-12:
            break

        # Try shifting a fraction of a[idx_max] to non-contrib positions
        # Weighted by how little they contribute to ALL peaks
        a_shifted = a.copy()
        improved = False

        for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
            shift = a[idx_max] * frac
            if shift < 1e-13:
                continue

            # Distribute to non-contrib positions proportional to (1 - their peak contribution)
            # Simple: distribute uniformly to non-contrib
            a_shifted = a.copy()
            a_shifted[idx_max] -= shift
            for j in non_contrib:
                a_shifted[j] += shift / len(non_contrib)

            a_shifted = project_simplex(a_shifted)
            c_new = fast_convolve(a_shifted)
            obj_new = c_new.max()

            if obj_new < best_obj - 1e-14:
                a = a_shifted
                best_obj = obj_new
                best_a = a.copy()
                improved = True
                break

        if not improved:
            # Try random perturbation focused on contributing positions
            if it % 50 == 0 or it > 500:
                delta = np.zeros(n)
                delta[contrib] = rng.standard_normal(len(contrib)) * 0.001
                a_trial = project_simplex(a + delta)
                c_trial = fast_convolve(a_trial)
                if c_trial.max() < best_obj - 1e-14:
                    a = a_trial
                    best_obj = c_trial.max()
                    best_a = a.copy()
                    improved = True

            if not improved and it > 2000:
                break

        if verbose and it % 200 == 0:
            print(f"    peak_shave it={it}, obj={best_obj:.10f}, k_peak={k_peak}")

    return best_a, best_obj


def coordinate_refine(a0, time_limit, verbose=False, seed=42):
    """
    Coordinate descent: for each coordinate j, find the best value
    that minimizes the peak auto-convolution, holding others fixed.
    Uses the parameterization where we swap mass between j and a random partner.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    rng = np.random.default_rng(seed)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    sweep = 0

    while time.monotonic() - t0 < time_limit - 2:
        sweep += 1
        improved_this_sweep = False

        # Random order for diversity
        order = rng.permutation(n) if sweep % 3 == 0 else np.arange(n)

        for j in order:
            if time.monotonic() - t0 > time_limit - 2:
                break

            if a[j] < 1e-10:
                continue

            # Try transferring mass from j to a few other positions
            # Pick partner positions that are "far" from j in index (to spread mass)
            partners = [j]
            for d in [n // 3, 2 * n // 3, n // 4, 3 * n // 4, 1, n // 2]:
                p = (j + d) % n
                if p != j and p not in partners:
                    partners.append(p)

            for p in partners:
                if p == j or a[j] < 1e-10:
                    continue

                # Try several transfer fractions
                best_local = None
                best_local_obj = fast_convolve(a).max()

                for frac in [0.01, 0.02, 0.05, 0.1, 0.15]:
                    delta = a[j] * frac
                    if delta < 1e-13:
                        continue
                    a_trial = a.copy()
                    a_trial[j] -= delta
                    a_trial[p] += delta
                    a_trial = project_simplex(a_trial)
                    c_trial = fast_convolve(a_trial)
                    obj_trial = c_trial.max()

                    if obj_trial < best_local_obj - 1e-14:
                        best_local_obj = obj_trial
                        best_local = a_trial.copy()

                if best_local is not None:
                    a = best_local
                    if best_local_obj < best_obj - 1e-14:
                        best_obj = best_local_obj
                        best_a = a.copy()
                        improved_this_sweep = True

        if not improved_this_sweep and sweep > 5:
            break

        if verbose and sweep % 5 == 0:
            print(f"    coord_refine sweep={sweep}, obj={best_obj:.10f}")

    return best_a, best_obj


def generate_inits(n, parent, rng):

    inits.append(("dirichlet_beta", rng.dirichlet(np.ones(n) * 2.0)))

    # Cosine window
    x = np.arange(n, dtype=np.float64)
    cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    cos_w = np.maximum(cos_w, 0.01)
    inits.append(("cosine", cos_w / cos_w.sum()))

    # Gaussian centered

    inits.append(("gaussian", g / g.sum()))

          np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2))
    inits.append(("two_peaks", g2 / g2.sum()))

    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    inits.append(("edge", edge / edge.sum()))

    # Sine-based (non-negative)
    sine = np.maximum(0.3 + 0.7 * np.sin(2 * np.pi * x / n + np.pi / 4), 0.01)
    inits.append(("sine", sine / sine.sum()))

    # Interpolated parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0.0)
        if px.sum() > 1e-10:
            inits.append(("parent_interp", px / px.sum()))
        # Parent with smoothing
        from scipy.ndimage import uniform_filter1d
        ps = uniform_filter1d(parent, size=max(2, len(parent) // 32))
        ps_interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), ps)
        ps_interp = np.maximum(ps_interp, 0.0)
        if ps_interp.sum() > 1e-10:
            inits.append(("parent_smooth", ps_interp / ps_interp.sum()))

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search:
    Phase 1: Diverse initialization + quick screening
    Phase 2: Nesterov LSE optimization with adaptive cooling
    Phase 3: Peak shaving + coordinate refinement
    Phase 4: Final polish with very low temperature
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  *** NEW BEST [{label}]: {score:.10f} (n={len(a)}) ***")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        print("No parent available, using uniform init")

    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")
    print(f"Budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")

    # === Phase 1: Quick n-exploration and screening (~200s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t1 = time.monotonic()

    candidate_ns = sorted(set([
        max(128, n_parent),
        384, 512, 768, 1024,
    ]))

    # Filter to reasonable sizes
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 2048]

    quick_results = []
    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 30:
            break

        inits = generate_inits(n_try, parent, rng)
        t_alloc = max(10, min(remaining / max(1, len(candidate_ns) - candidate_ns.index(n_try)), 40))

        # Quick screen: run short Nesterov on a few inits
        for name, init_a in inits[:4]:
            if time.monotonic() > deadline - 1200:
                break
            t_quick = min(15, remaining * 0.1)
            a_opt, obj = nesterov_lse(init_a, time_limit=t_quick, T_init=0.02,
                                      T_final=0.005, lr_init=0.03, seed=n_try + hash(name) % 10000)
            score = try_seq(a_opt, f"quick_{name}_n{n_try}")
            quick_results.append((score, a_opt, n_try))

        # Also evaluate inits directly
        for name, init_a in inits[4:]:
            score = try_seq(init_a, f"init_{name}_n{n_try}")
            quick_results.append((score, init_a, n_try))

    quick_results.sort(key=lambda x: x[0])
    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")
    print(f"  Top 3 quick results: {[f'{r[0]:.8f}(n={r[2]})' for r in quick_results[:3]]}")

    # === Phase 2: Deep Nesterov optimization (~1000s) ===
    print("\n=== Phase 2: Deep Nesterov Optimization ===")
    t2 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
    else:
        best_seq = parent.copy()
        n_best = len(best_seq)

    # Run multiple configurations
    configs = [
        # (T_init, T_final, lr_init, momentum, time_frac, label)
        (0.01, 0.0003, 0.02, 0.85, 0.25, "main"),
        (0.02, 0.001, 0.03, 0.90, 0.20, "high_T"),
        (0.005, 0.0002, 0.015, 0.88, 0.20, "low_T"),
        (0.01, 0.0005, 0.025, 0.92, 0.15, "high_mom"),
        (0.03, 0.001, 0.04, 0.80, 0.10, "aggressive"),
    ]

    for ci, (T_i, T_f, lr_i, mom_i, t_frac, label) in enumerate(configs):
        remaining = deadline - time.monotonic() - 400

                                  lr_init=lr_i, momentum_init=mom_i,
