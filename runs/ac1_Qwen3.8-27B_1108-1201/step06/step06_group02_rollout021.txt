Optimized minimization of 2*n*max(conv(a,a))/(sum(a))^2 using:

- Non-palindromic FW for breaking symmetry
- Adaptive line search with warm-start alpha

- Strategic dual-informed perturbations

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_palindromic(h):
    """Create palindromic sequence from half."""
    return np.concatenate([h, h[::-1]])


def build_pal_A(f, n):
    """Build constraint matrix for palindromic LP oracle.
    A[k, i] = sum of f[j] where g[k-j] = g_half[i], for k in [0,n), i in [0,n/2).
    """
    half = n // 2
    A = np.zeros((n, half))
    f_arr = np.asarray(f, dtype=np.float64)
    for i in range(half):
        j1 = np.arange(n) - i
        valid1 = (j1 >= 0) & (j1 < n)
        vals1 = np.zeros(n)
        vals1[valid1] = f_arr[j1[valid1]]
        j2 = np.arange(n) - n + 1 + i
        valid2 = (j2 >= 0) & (j2 < n)
        vals2 = np.zeros(n)
        vals2[valid2] = f_arr[j2[valid2]]
        A[:, i] = vals1 + vals2
    return A


def solve_pal_lp(f, M, n):
    """Solve LP: max sum(g_half) s.t. (f*g)[k] <= M for k=0..n-1, g_half >= 0.
    Returns g_half or None on failure.
    """
    half = n // 2
    A = build_pal_A(f, n)
    c = -np.ones(half)
    b_ub = np.full(n, M)
    bounds = [(0, None)] * half
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def solve_std_lp(f, M, n):

    c = -np.ones(n)

    for j in range(n):

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:

        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()
    alpha_prev = 0.1

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.05:
            break

        g = solve_std_lp(a, M * m_exp, n)
        if g is None:
            m_exp = min(m_exp * 1.3, 5.0)
            if m_exp > 4.5:
                m_exp = 1.0
                no_improve += 10
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - a)) < 1e-12:
            no_improve += 5
            if no_improve > 40:
                rng = np.random.default_rng(it * 7919 + 13)
                scale = 0.002 + 0.005 * min(1.0, no_improve / 100.0)
                a_pert = project_simplex(best_a + rng.standard_normal(n) * scale)
                conv = fast_convolve(a_pert)
                M = conv.max()
                a = a_pert
                m_exp = 1.0
                no_improve = 0
                continue
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = np.concatenate([
            np.linspace(0.01, min(1.0, alpha_prev * 3), 8),
            np.linspace(max(0.0, alpha_prev - 0.2), 1.0, 7)
        ])
        alphas = np.unique(np.clip(alphas, 0.005, 1.0))

        for alpha in alphas:
            ft = (1.0 - alpha) * a + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val - 1e-15:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-8:
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            alpha_prev = best_alpha
            conv = fast_convolve(a)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 500 == 0:
                    print("    FW-std it={}: obj={:.10f}".format(it, best_obj))
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 80:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            scale = 0.003 + 0.007 * min(1.0, (no_improve - 80) / 100.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            conv = fast_convolve(a)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
            alpha_prev = 0.1

    return best_a, best_obj


def coordinate_polish(a0, time_limit):
    """Coordinate-wise optimization: try moving mass from high-contributing to low-contributing coords."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)

        if len(affected) == 0:
            break

        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]

        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            outside = np.arange(n)
        j_min_idx = outside[np.argmin(a[outside])]

        improved = False
        for delta in [0.0002, 0.0005, 0.001, 0.002, 0.005]:
            if a[j_max] < delta:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min_idx] += delta
            candidate = np.maximum(candidate, 0.0)
            cs = candidate.sum()
            if cs < 1e-15:
                continue
            candidate = candidate / cs
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        if not improved:
            # Try random pair swaps
            rng = np.random.default_rng(it * 12345 + 7)
            for _ in range(5):
                if time.monotonic() - t0 > time_limit - 0.02:
                    break
                i1, i2 = rng.integers(0, n, 2)
                if i1 == i2:
                    continue
                delta = 0.001 * a[i1]
                if delta < 1e-8 or a[i1] < delta:
                    continue
                candidate = a.copy()
                candidate[i1] -= delta
                candidate[i2] += delta
                candidate = np.maximum(candidate, 0.0)
                candidate = candidate / candidate.sum()
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break

            if not improved:
                if it % 10000 == 0:
                    break

    return best_a, best_obj


def smoothed_grad_descent(a0, time_limit, T_init=0.001, lr_init=0.02):
    """Smoothed gradient descent using softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.9

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient: grad_i = 2 * sum_k p[k] * a[k-i] (for valid k-i)
        # This is 2 * (p * a) reversed, cropped
        grad = 2.0 * np.correlate(p, a, mode='valid')

        v = mom * v - lr * grad
        candidate = project_simplex(a + v)

        c = fast_convolve(candidate)
        val = c.max()

        if val < best_obj - 1e-14:
            a = candidate
            best_obj = val
            best_a = a.copy()
            no_improve = 0
        else:
            # Try smaller step
            for factor in [0.5, 0.25, 0.1]:
                candidate2 = project_simplex(a - factor * lr * grad)
                c2 = fast_convolve(candidate2)
                if c2.max() < best_obj - 1e-14:
                    a = candidate2
                    best_obj = c2.max()
                    best_a = a.copy()
                    no_improve = 0
                    break
            else:
                no_improve += 1

        lr *= 0.9998
        mom = min(0.98, mom + 0.00005)

        if no_improve > 30:
            T *= 1.1
        if no_improve > 60:
            T *= 0.7
        if no_improve > 200:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.002 + 0.004 * min(1.0, (no_improve - 200) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            T = T_init
            lr = lr_init
            mom = 0.9
            no_improve = 0

    return best_a, best_obj


def interpolate_seq(parent, n_target):
    """Interpolate parent sequence to target length."""
    n_src = len(parent)
    if n_src == n_target:
        return parent.copy()
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_target)
    result = np.interp(x_dst, x_src, parent)
    result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return result / s


def try_and_update(a, state, label):
    """Evaluate a sequence and update state if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.10f} (n={})".format(label, score, len(a_norm)))


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search:
    Phase 1: Quick palindromic FW on multiple n values
    Phase 2: Deep palindromic FW on best n
    Phase 3: Non-palindromic FW (break symmetry)
    Phase 4: Smoothed gradient + coordinate descent
    Phase 5: Final polish with all methods
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    state = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.10f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        state[1] = parent.copy()
        state[0] = float(evaluate_sequence(parent.tolist()))

    n_parent = len(parent)
    print("Starting: n={}, score={:.10f}".format(n_parent, state[0]))

    # === Phase 1: Quick multi-n exploration with palindromic FW (~120s) ===
    print("\n=== Phase 1: Quick Multi-n Exploration (palindromic FW) ===")
    t_phase = time.monotonic()

    # Try a range of n values (all even)
    n_candidates = sorted(set([
        n_parent if n_parent % 2 == 0 else n_parent - 1,
        512, 640, 768, 896, 1024, 1280, 1536, 2048
    ]))
    n_candidates = [n for n in n_candidates if 256 <= n <= 2048]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic() - (budget_s * 0.6)
        if remaining < 10:
            break

        if abs(n_try - n_parent) <= 1:
            start = parent.copy()
        else:
            start = interpolate_seq(parent, n_try)

        # Make palindromic
        if n_try % 2 != 0:
            n_try -= 1
            start = start[:n_try]

        t_alloc = min(remaining, 25)
        print("  Pal-FW n={}, t={:.0f}s".format(n_try, t_alloc))
        a_opt, _ = fw_palindromic(start, time_limit=t_alloc)
        try_and_update(a_opt, state, "pal_n{}".format(n_try))

    print("  Phase 1 best: {:.10f} (t={:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 2: Deep palindromic FW on best n (~400s) ===
    print("\n=== Phase 2: Deep Palindromic FW ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])
        if n_best % 2 != 0:
            n_best -= 1
            state[1] = state[1][:n_best]
            state[1] = state[1] / state[1].sum()

        # Check if palindromic helps
        h = (state[1][:n_best // 2] + state[1][n_best // 2:][::-1]) / 2.0
        pal_a = make_palindromic(h)
        pal_a = pal_a / pal_a.sum()
        pal_score = float(evaluate_sequence(pal_a.tolist()))
        if np.isfinite(pal_score) and pal_score < state[0]:
            state = [pal_score, pal_a.copy()]
            print("  Palindromic symmetrization improved: {:.10f}".format(pal_score))

        remaining = deadline - time.monotonic() - (budget_s * 0.35)
        if remaining > 30:
            t_alloc = min(remaining, 300)
            print("  Deep Pal-FW n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = fw_palindromic(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "deep_pal")

        # Perturbed restarts
        for scale in [0.003, 0.008, 0.015, 0.025]:
            remaining = deadline - time.monotonic() - (budget_s * 0.25)
            if remaining < 10:
                break
            h_best = state[1][:len(state[1]) // 2]
            h_pert = h_best + rng.standard_normal(len(h_best)) * scale
            h_pert = np.maximum(h_pert, 0)
            if h_pert.sum() < 1e-15:
                continue
            h_pert = h_pert / h_pert.sum()
            a_pert = make_palindromic(h_pert)
            t_alloc = min(remaining, 50)
            a_opt, _ = fw_palindromic(a_pert, time_limit=t_alloc)
            try_and_update(a_opt, state, "pal_pert_{}".format(scale))

    print("  Phase 2 best: {:.10f} (t={:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 3: Non-palindromic FW (~300s) ===
    print("\n=== Phase 3: Non-Palindromic FW ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])
        remaining = deadline - time.monotonic() - (budget_s * 0.2)
        if remaining > 30:
            t_alloc = min(remaining, 200)
            print("  Std FW n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = fw_standard(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "std_fw")

        # Try a non-palindromic perturbation
        remaining = deadline - time.monotonic() - (budget_s * 0.1)
        if remaining > 20:
            a_pert = project_simplex(state[1] + rng.standard_normal(n_best) * 0.005)
            t_alloc = min(remaining, 80)
            print("  Std FW (perturbed) n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = fw_standard(a_pert, time_limit=t_alloc)
            try_and_update(a_opt, state, "std_fw_pert")

    print("  Phase 3 best: {:.10f} (t={:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 4: Smoothed gradient + coordinate descent (~250s) ===
    print("\n=== Phase 4: Gradient + Coordinate ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Multiple temperature/lr settings
        for T_init, lr_init in [(0.0005, 0.015), (0.001, 0.02), (0.003, 0.01)]:
            remaining = deadline - time.monotonic() - (budget_s * 0.1)
            if remaining < 10:
                break
            t_alloc = min(remaining, 60)
            print("  SmoothGrad T={}, lr={}, t={:.0f}s".format(T_init, lr_init, t_alloc))
            a_opt, _ = smoothed_grad_descent(state[1], time_limit=t_alloc,
                                             T_init=T_init, lr_init=lr_init)
            try_and_update(a_opt, state, "sg_T{}".format(int(T_init * 10000)))

        # Coordinate polish
        remaining = deadline - time.monotonic() - (budget_s * 0.05)
        if remaining > 10:
            t_alloc = min(remaining, 80)
            print("  CoordPolish t={:.0f}s".format(t_alloc))
            a_opt, _ = coordinate_polish(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "coord")

    print("  Phase 4 best: {:.10f} (t={:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 5: Final polish (~remaining) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Alternate between methods
        n_best = len(state[1])

        # More palindromic FW
        if n_best % 2 == 0:
            remaining = deadline - time.monotonic() - 10
            if remaining > 10:
                t_alloc = min(remaining, 60)
                a_opt, _ = fw_palindromic(state[1], time_limit=t_alloc)
                try_and_update(a_opt, state, "final_pal")

        # Fine coordinate descent
        remaining = deadline - time.monotonic() - 10
        if remaining > 5:
            t_alloc = min(remaining, 40)
            a_opt, _ = coordinate_polish(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "final_coord")

        # Tiny perturbations with quick optimization
        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            for trial in range(min(10, int(remaining // 5))):
                remaining = deadline - time.monotonic() - 2
                if remaining < 3:
                    break
                noise = rng.standard_normal(n_best) * 0.0003
                a_pert = project_simplex(state[1] + noise)
                t_alloc = min(remaining, 4)
                a_opt, _ = smoothed_grad_descent(a_pert, time_limit=t_alloc,
                                                 T_init=0.0002, lr_init=0.005)
                try_and_update(a_opt, state, "fine_{}".format(trial))

        # One final palindromic FW if even
        if n_best % 2 == 0:
            remaining = deadline - time.monotonic() - 2
            if remaining > 5:
                a_opt, _ = fw_palindromic(state[1], time_limit=min(remaining, 20))
                try_and_update(a_opt, state, "very_final_pal")

    print("  Phase 5 best: {:.10f} (t={:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(768) / 768
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.10f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
