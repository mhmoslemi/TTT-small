def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (f*g) = A @ g."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def spectral_init(n, rng):
    """Create a sequence with approximately flat autocorrelation using spectral shaping."""
    # Start with a random sequence
    a = rng.standard_normal(n)
    a = np.abs(a) + 0.1
    a = a / a.sum()
    
    # Try to flatten the |FFT(a)|^2 spectrum
    for _ in range(100):
        A = np.fft.rfft(a, 2*n-1)
        mag = np.abs(A)
        # Target: constant magnitude
        target_mag = np.ones_like(mag) * mag.mean()
        # Update phase to reduce variance of |FFT|^2
        # Use a simple iterative approach
        new_a = np.fft.irfft(A * (target_mag / (mag + 1e-10)), 2*n-1)[:n]
        new_a = np.maximum(new_a, 0)

        if s < 1e-10:
            break
        new_a = new_a / s
        if np.max(np.abs(new_a - a)) < 1e-10:
            break
        a = new_a
    
    return a

def peak_shave(a0, time_limit):
    """
    Directly target peaks in the convolution by redistributing mass.
    For each peak, identify the most contributing pair and try to
    reduce it by moving mass to a location that doesn't contribute
    to any current peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    step_frac = 0.005
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        # Find the top peaks
        n_peaks = min(6, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]
        
        # Find "cold" positions (low convolution values) as targets
        cold_indices = np.argsort(conv)[:n]  # n coldest positions
        
        improved = False
        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            # Find the pair with maximum product
            best_product = -1
            best_j = -1
            best_kj = -1
            for j in range(js, je + 1):
                prod = a[j] * a[k - j]
                if prod > best_product:
                    best_product = prod
                    best_j = j
                    best_kj = k - j
            
            if best_j < 0 or a[best_j] < 1e-10:
                continue
            
            # Try moving mass from best_j to a cold position
            for ci in range(min(3, len(cold_indices))):
                target = int(cold_indices[ci])
                if target == best_j or target == best_kj:
                    continue
                
                for frac in [step_frac, step_frac * 0.5, step_frac * 2]:
                    delta = a[best_j] * frac
                    if delta < 1e-12:
                        continue
                    candidate = a.copy()
                    candidate[best_j] -= delta
                    candidate[target] += delta
                    if candidate[best_j] < -1e-12:
                        continue

def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Spectral equalization initialization (flat |FFT|^2 target)
- Frank-Wolfe with LP oracle and adaptive relaxation
- Frequency-domain water-filling step

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (f*g) = A @ g."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

    Iteratively equalize the magnitude spectrum of a non-negative sequence.
    Uses a Gerchberg-Saxton-like approach: alternate between
    enforcing flat |FFT|^2 in frequency domain and non-negativity in time domain.
    """
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        # Target: flat magnitude
        target_mag = mag.mean() * np.ones(N)
        # Update phase-preserving
        phase = A / (mag + 1e-30)
        A_new = target_mag * phase
        a_new = np.fft.ifft(A_new).real[:n]
        # Enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        if np.max(np.abs(a_new - a)) < 1e-12:
            break
        a = a_new

    return a


def freq_domain_step(a0, n_iters=50, alpha=0.3):
    """
    Frequency-domain water-filling: blend current sequence with
    its spectrally-equalized version.
    """
    a_eq = spectral_equalization(a0, n_iters=n_iters)
    a_blend = (1.0 - alpha) * a0 + alpha * a_eq
    a_blend = np.maximum(a_blend, 0.0)
    s = a_blend.sum()
    if s < 1e-15:
        return a0
    return a_blend / s


def init_variants(n, parent, rng):
    """Generate diverse initializations."""
    starts = []
    x = np.arange(n, dtype=np.float64)

    # Parent resampled
    if parent is not None:
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        p_new = np.interp(x_new, x_old, parent)
        p_new = np.maximum(p_new, 0)
        s = p_new.sum()
        if s > 1e-10:
            starts.append(p_new / s)

    # Uniform
    starts.append(np.ones(n) / n)

    # Spectral equalization of random start
    rnd = np.abs(rng.standard_normal(n)) + 0.1
    starts.append(spectral_equalization(rnd / rnd.sum(), n_iters=100))

    # Cosine window
    w = 0.5 * (1 + np.cos(np.pi * x / max(n - 1, 1)))
    w = np.maximum(w, 0.01)
    starts.append(w / w.sum())

    # Two-lobe
    g1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.1 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())

    # Edge-weighted
    edge = 1.0 / (np.minimum(x, n - 1 - x).astype(float) + 1.0)
    starts.append(edge / edge.sum())

    # Random Dirichlet
    starts.append(rng.dirichlet(np.ones(n) * 0.5))

    # Spectral equalization of Dirichlet
    d = rng.dirichlet(np.ones(n) * 0.3)
    starts.append(spectral_equalization(d, n_iters=100))

    return starts


def fw_optimize(f0, time_limit, m_relax_init=1.001, verbose=False):
    """Frank-Wolfe with LP oracle and adaptive relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    stagnant = 0
    it = 0
    m_relax = m_relax_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.3:
            break

        g = solve_lp_oracle(f, M * m_relax)
        if g is None or g.sum() < 1e-15:
            g = solve_lp_oracle(f, M * 1.005)
        if g is None or g.sum() < 1e-15:
            stagnant += 1
            if stagnant > 12:
                rng = np.random.default_rng(it * 7919 + 3)
                scale = 0.001 + 0.003 * min(1.0, stagnant / 25.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = m_relax_init
                stagnant = 0
            continue

        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-10:
            stagnant += 1
            if stagnant > 15:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue

        # Two-phase line search
        best_alpha = 0.0
        best_val = M + 1e-14

        # Coarse grid
        for alpha in np.linspace(0.02, 1.0, 15):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine grid
        if best_alpha > 0.01:
            lo = max(0.0, best_alpha - 0.02)
            hi = min(1.0, best_alpha + 0.02)
            for alpha in np.linspace(lo, hi, 8):
                ft = (1.0 - alpha) * f + alpha * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha

        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = m_relax_init

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        if stagnant > 6:
            m_relax = min(m_relax * 1.04, 1.05)

        if verbose and it % 50 == 0:
            print(f"    FW it={it}, obj={best_obj:.10f}, stagnant={stagnant}")

    return best_f, best_obj


def peak_shave(a0, time_limit):
    """
    Directly target peaks in the convolution by redistributing mass
    from contributing positions to cold positions.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    step_frac = 0.004

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        n_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        # Cold positions as targets
        cold_indices = np.argsort(conv)[:max(n // 4, 10)]

        improved = False
        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)

            # Find max-contributing pair
            best_product = -1
            best_j = -1
            best_kj = -1
            for j in range(js, je + 1):
                prod = a[j] * a[k - j]
                if prod > best_product:
                    best_product = prod
                    best_j = j
                    best_kj = k - j

            if best_j < 0 or a[best_j] < 1e-10:
                continue

            # Try moving mass from best_j to cold positions
            for ci in range(min(5, len(cold_indices))):
                target = int(cold_indices[ci])
                if target == best_j or target == best_kj:
                    continue

                for frac in [step_frac, step_frac * 0.5, step_frac * 1.5]:
                    delta = a[best_j] * frac
                    if delta < 1e-13:
                        continue
                    candidate = a.copy()
                    candidate[best_j] -= delta
                    candidate[target] += delta
                    if candidate[best_j] < -1e-12:
                        continue
                    candidate = np.maximum(candidate, 0)
                    candidate = candidate / candidate.sum()
                    val = fast_convolve(candidate).max()
                    if val < best_obj - 1e-12:
                        a = candidate
                        best_obj = val
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

            if improved:
                break

        if not improved:
            step_frac *= 0.94
            if step_frac < 1e-8:
                break
            if it % 150 == 0:
                rng = np.random.default_rng(it * 48271 + 17)
                candidate = project_simplex(a + rng.standard_normal(n) * step_frac * 4)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-12:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
            elif it > 4000:
                break

    return best_a, best_obj


def subgradient_optimize(a0, time_limit, T_init=0.0005, lr_init=0.02, top_k_frac=0.05):
    """Smoothed subgradient with momentum and adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(5, int(n * top_k_frac))

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.9

    t0 = time.monotonic()
    stagnant = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))

        pv = conv[peaks]
        pv_max = pv.max()
        w = np.exp((pv - pv_max) / (T * 0.1))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += w[i] * 2.0 * a[k - idx_arr]

        v = mom * v - lr * grad

        best_candidate = a
        best_val = conv.max()
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            candidate = project_simplex(a + frac * v)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 25:
            T *= 1.08
            lr *= 0.96
        if stagnant > 80:
            T *= 0.5
            lr *= 1.4
            v = np.zeros(n)
            stagnant = 0
        if stagnant > 250:
            rng = np.random.default_rng((it * 31 + 7) % (2 ** 31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            v = np.zeros(n)
            mom = 0.9
            stagnant = 0

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search combining:
    - Spectral equalization for initialization
    - Frank-Wolfe with LP oracle
    - Frequency-domain water-filling
    - Peak shaving
    - Multi-length exploration
    - Interleaved polishing
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 120.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return float('inf')
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")

    n_parent = len(parent)

    # === Phase 1: Multi-length exploration with spectral init (250s) ===
    print("\n=== Phase 1: Multi-length Exploration ===")
    t_phase = time.monotonic()

    base_lengths = [256, 320, 384, 448, 512, 576, 640, 704, 768, 832, 896, 960, 1024, 1152, 1280, 1536, 1920, 2048]
    if n_parent in base_lengths:
        base_lengths.remove(n_parent)
    base_lengths.insert(0, n_parent)
    base_lengths = sorted(set([l for l in base_lengths if 192 <= l <= 2560]))

    results = {}
    for n_try in base_lengths:
        remaining = deadline - time.monotonic() - 600
        if remaining < 10:
            break

        starts = init_variants(n_try, parent, rng)

        best_for_n = None
        best_obj_for_n = float('inf')

        n_starts = min(4, len(starts))
        per_start = max(5.0, min(20.0, remaining / (n_starts + 1)))

        for si in range(n_starts):
            remaining_inner = deadline - time.monotonic() - 600
            if remaining_inner < 3:
                break
            t_alloc = min(per_start, remaining_inner)
            a_opt, obj = fw_optimize(starts[si], time_limit=t_alloc)
            if obj < best_obj_for_n:
                best_obj_for_n = obj
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            results[n_try] = best_for_n.copy()
            try_seq(best_for_n, f"explore_n{n_try}")

    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic() - t_phase:.0f}s)")

    # === Phase 2: Deep FW on best candidate (350s) ===
    print("\n=== Phase 2: Deep FW ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        remaining = deadline - time.monotonic() - 500
        if remaining > 20:
            t_alloc = min(remaining, 300)
            print(f"  Deep FW n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_fw")

        # Frequency-domain step on best
        remaining = deadline - time.monotonic() - 400
        if remaining > 10:
            a_freq = freq_domain_step(best_seq, n_iters=100, alpha=0.2)
            try_seq(a_freq, "freq_step")
            # Follow up with FW
            t_alloc = min(remaining - 10, 60)
            if t_alloc > 5:
                a_opt, _ = fw_optimize(a_freq, time_limit=t_alloc)
                try_seq(a_opt, "freq_then_fw")

        # Perturbed restarts
        for scale in [0.001, 0.004, 0.008]:
            remaining = deadline - time.monotonic() - 300
            if remaining < 15:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 40)
            a_opt, _ = fw_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"fw_pert_{scale}")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic() - t_phase:.0f}s)")

    # === Phase 3: Peak shaving (150s) ===
    print("\n=== Phase 3: Peak Shaving ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 250
        if remaining > 10:
            t_alloc = min(remaining, 120)
            print(f"  Peak shaving, t={t_alloc:.0f}s")
            a_opt, _ = peak_shave(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_shave")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic() - t_phase:.0f}s)")

    # === Phase 4: Subgradient (120s) ===
    print("\n=== Phase 4: Subgradient ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        configs = [
            {'T_init': 0.0003, 'lr_init': 0.015, 'top_k_frac': 0.03},
            {'T_init': 0.001, 'lr_init': 0.01, 'top_k_frac': 0.05},
            {'T_init': 0.0001, 'lr_init': 0.008, 'top_k_frac': 0.02},
        ]
        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 100
            if remaining < 10:
                break
            t_alloc = min(remaining, 45)
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"subgrad_{ci}")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic() - t_phase:.0f}s)")

    # === Phase 5: Interleaved polish (remaining) ===
    print("\n=== Phase 5: Interleaved Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        round_i = 0
        while time.monotonic() < deadline - 15:
            round_i += 1
            remaining = deadline - time.monotonic() - 15
            if remaining < 8:
                break

            method = round_i % 5

            if method == 0:
                # FW with tiny perturbation
                noise = rng.standard_normal(n_best) * 0.0004
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 18)
                a_opt, _ = fw_optimize(start, time_limit=t_alloc)
                try_seq(a_opt, f"polish_fw_{round_i}")
            elif method == 1:
                # Peak shave
                t_alloc = min(remaining, 25)
                a_opt, _ = peak_shave(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"polish_peak_{round_i}")
            elif method == 2:
                # Frequency domain step + FW
                a_freq = freq_domain_step(best_seq, n_iters=50, alpha=0.1)
                t_alloc = min(remaining, 15)
                a_opt, _ = fw_optimize(a_freq, time_limit=t_alloc)
                try_seq(a_opt, f"polish_freq_{round_i}")
            elif method == 3:
                # Subgradient
                noise = rng.standard_normal(n_best) * 0.0003
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 12)
                a_opt, _ = subgradient_optimize(start, time_limit=t_alloc,
                                                T_init=0.0002, lr_init=0.005,
                                                top_k_frac=0.02)
                try_seq(a_opt, f"polish_sg_{round_i}")
            else:
                # Spectral re-initialization + FW
                a_spec = spectral_equalization(best_seq, n_iters=50)
                t_alloc = min(remaining, 15)
                a_opt, _ = fw_optimize(a_spec, time_limit=t_alloc)
                try_seq(a_opt, f"polish_spec_{round_i}")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic() - t_phase:.0f}s)")

    # === Phase 6: Cross-length refinement ===
    print("\n=== Phase 6: Cross-length Refinement ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        for n_try, seq in sorted(results.items()):
            if len(seq) == len(best_seq):
                continue
            remaining = deadline - time.monotonic() - 5
            if remaining < 8:
                break
            # Resample best to this length and do quick FW
            resampled = resample_sequence(best_seq, n_try)
            t_alloc = min(remaining, 15)
            a_opt, _ = fw_optimize(resampled, time_limit=t_alloc)
            try_seq(a_opt, f"cross_n{n_try}")
            remaining = deadline - time.monotonic() - 5
            if remaining < 8:
                break

    print(f"  Phase 6 done: best={best_score:.10f} ({time.monotonic() - t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
