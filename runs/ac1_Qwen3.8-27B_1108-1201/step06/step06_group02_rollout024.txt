import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]."""
    return np.correlate(p, a, mode='valid')


def build_conv_matrix(f):
    """Build the (2n-1) x n convolution constraint matrix."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M, time_limit=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    if M <= 0:
        return None
    c = -np.ones(n)
    A_ub = build_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
        elif res.status == 3:  # time limit
            if res.x is not None:
                return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def coord_descent_quadratic(a0, time_limit):
    """
    Coordinate descent using exact quadratic constraint solving.
    For each coordinate i, compute the range of allowed values given
    the constraint that all convolution terms stay <= M.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_obj = M
    best_score = 2 * n * M

    t0 = time.monotonic()
    pass_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        pass_count += 1
        improved = False

        # Randomize coordinate order
        rng = np.random.default_rng(pass_count * 1000 + 42)
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            # Current value
            ai = a[i]
            if ai < 1e-15:
                continue

            # For coordinate i, new_conv[k] = conv[k] + 2*delta*a[k-i] + delta^2*I(k=2i)
            # We want new_max <= M (or as small as possible)
            # Find max delta such that all constraints are satisfied

            # For k where 0 <= k-i < n (i.e., a[k-i] is a valid index):
            #   conv[k] + 2*delta*a[k-i] + delta^2*I(k=2i) <= M
            #   => 2*delta*a[k-i] + delta^2*I(k=2i) <= M - conv[k]

            max_delta = 1.0  # upper bound (can't increase by more than 1 in normalized space)
            min_delta = -ai  # can't go below 0

            ks = np.arange(2 * n - 1)
            ki = ks - i  # k - i values
            valid = (ki >= 0) & (ki < n)

            if not valid.any():
                continue

            kv = ks[valid]
            a_ki = a[ki[valid]]

            # For k != 2i: delta <= (M - conv[k]) / (2 * a[k-i]) if a[k-i] > 0
            # For k == 2i: delta^2 + 2*a[i]*delta + (conv[2i] - M) <= 0
            #   => delta <= -a[i] + sqrt(a[i]^2 + M - conv[2i]) (taking positive root)

            mask_non_diag = valid & (kv != 2 * i)
            if mask_non_diag.any():
                a_ki_nd = a_ki[mask_non_diag]
                slack = M - conv[kv[mask_non_diag]]
                # delta <= slack / (2 * a_ki) for a_ki > 0
                pos = a_ki_nd > 1e-15
                if pos.any():
                    bounds_delta = slack[pos] / (2.0 * a_ki_nd[pos])
                    max_delta = min(max_delta, bounds_delta.min())

            # Diagonal constraint (k = 2i)
            if 0 <= 2 * i < 2 * n - 1:
                slack_diag = M - conv[2 * i]
                # delta^2 + 2*a[i]*delta + (conv[2i] - M) <= 0
                # delta^2 + 2*a[i]*delta + (conv[2i] - M) <= 0
                # Roots: delta = -a[i] ± sqrt(a[i]^2 + M - conv[2i])
                disc = a[i] * a[i] + slack_diag
                if disc >= 0:
                    root = np.sqrt(disc)
                    upper_root = -a[i] + root
                    max_delta = min(max_delta, upper_root)

            max_delta = max(max_delta, 0.0)  # Don't decrease

            # Try increasing a[i]
            if max_delta > 1e-12:
                # Use a fraction of max_delta for stability
                candidates = [max_delta * 0.5, max_delta * 0.8, max_delta * 1.0]
                for frac_delta in candidates:
                    new_ai = ai + frac_delta
                    if new_ai < 0:
                        continue
                    new_a = a.copy()
                    new_a[i] = new_ai
                    new_a = new_a / new_a.sum()
                    new_conv = fast_convolve(new_a)
                    new_M = new_conv.max()
                    if new_M < M - 1e-10:
                        a = new_a
                        conv = new_conv
                        M = new_M
                        improved = True
                        break
                    elif new_M <= M + 1e-10:
                        # Check if sum increased (score decreased)
                        pass

            # Also try setting to 0 (peeling)
            if ai > 1e-8:
                test_a = a.copy()
                test_a[i] = 0.0
                ts = test_a.sum()
                if ts > 1e-10:
                    test_a = test_a / ts
                    test_conv = fast_convolve(test_a)
                    test_M = test_conv.max()
                    # Score comparison: 2n*M_test/(sum_test)^2 vs 2n*M/(sum_a)^2
                    # Since we normalize, score = 2n * M_normalized
                    if test_M < M - 1e-10:

                        conv = test_conv

        if improved:

            score = 2 * n * M

        else:

                no_improve = 0

    return best_a, best_obj


def geometric_init(n, r_start=0.95, r_end=0.999):
    """Generate geometric decay initializations."""
    x = np.arange(n, dtype=np.float64)
    seqs = []
    for r in np.linspace(r_start, r_end, 8):
        a = r ** x
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            seqs.append(a / s)
    # Reverse geometric
    for r in np.linspace(r_start, r_end, 4):
        a = r ** (n - 1 - x)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            seqs.append(a / s)
    return seqs


def sawtooth_init(n, cycles=2):
    """Generate sawtooth initializations."""
    x = np.arange(n, dtype=np.float64)
    seqs = []
    for period in range(3, min(20, n // 2)):
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * cycles * x / period)
        a = np.maximum(a, 0.01)
        s = a.sum()
        seqs.append(a / s)
    return seqs

            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[idx] * 2.0 * a[k - idx_arr]

        v = mom * v - lr * sg
        a = project_simplex(a + v)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        mom = min(0.99, mom + 0.0001)

        if no_improve > 150:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.002 + 0.003 * min(1.0, (no_improve - 150) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init * 0.5
            no_improve = 0

    return best_a, best_obj


def smoothed_grad_descent(a0, time_limit, T_init, lr_init):
    """Smoothed gradient descent using softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = 2.0 * hankel_mv(p, a)

        best_candidate = a
        best_val = conv.max()

        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            T *= 1.08
            lr *= 0.95
        if no_improve > 100:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.003 + 0.004 * min(1.0, (no_improve - 100) / 100.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def rescale_to_n(parent, n_new):
    """Rescale a parent sequence to a new length using interpolation."""
    if len(parent) == n_new:
        return parent.copy()
    x_old = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_new)
    new_seq = np.interp(x_new, x_old, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s > 1e-10:
        new_seq = new_seq / s
    else:
        new_seq = np.ones(n_new) / n_new
    return new_seq


def try_and_update(a, state, label):
    """Evaluate a sequence and update state if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a_norm)))


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid search with coordinate descent and diverse initializations.
    Phase 1: Quick multi-n exploration with FW-LP
    Phase 2: Deep FW-LP + coordinate descent on best
    Phase 3: Multi-start perturbation with re-optimization
    Phase 4: Final polish (coord descent + fine FW)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    state = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.8f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        # Default: use a geometric initialization
        parent = np.array([0.99 ** i for i in range(768)])
        parent = parent / parent.sum()
        ps = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(ps):
            state = [ps, parent.copy()]
            print("Default parent: n={}, score={:.8f}".format(len(parent), ps))

    n_parent = len(parent)
    print("Initial best: {:.8f} (n={})".format(state[0], n_parent))

    # === Phase 1: Quick Multi-n Exploration ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    # Generate diverse initializations for the parent n
    inits = [parent.copy()]
    inits.extend(geometric_init(n_parent, 0.93, 0.995))
    inits.extend(sawtooth_init(n_parent, cycles=1))
    inits.append(rng.dirichlet(np.ones(n_parent) * 0.5))

    # Try different n values
    n_candidates = sorted(set([max(128, min(2048, n)), n for n in
                                [n_parent - 64, n_parent, n_parent + 64,
                                 n_parent - 128, n_parent + 128,
                                 512, 640, 896, 1024]]))

    for init_idx, init_seq in enumerate(inits[:4]):  # Limit number of inits
        remaining = deadline - time.monotonic() - 1200
        if remaining < 10:
            break

        n_try = n_parent if init_idx < 2 else n_candidates[min(init_idx, len(n_candidates) - 1)]
        start = rescale_to_n(init_seq, n_try) if len(init_seq) != n_try else init_seq.copy()

        t_alloc = min(remaining, 30)
        print("  FW n={}, init={}, t={:.0f}s".format(n_try, init_idx, t_alloc))
        a_opt, _ = fw_lp_optimize(start, time_limit=t_alloc)
        try_and_update(a_opt, state, "expl_p1_{}".format(init_idx))

        # Quick coord descent
        remaining = deadline - time.monotonic() - 1150
        if remaining > 5:
            a_cd, _ = coord_descent_quadratic(a_opt, time_limit=min(remaining, 10))
            try_and_update(a_cd, state, "cd_p1_{}".format(init_idx))

    print("  Phase 1 best: {:.8f}".format(state[0]))

    # === Phase 2: Deep Optimization on Best ===
    print("\n=== Phase 2: Deep FW + Coord Descent ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        a_best = state[1]
        n_best = len(a_best)

        # Deep FW
        remaining = deadline - time.monotonic() - 600
        if remaining > 30:
            t_alloc = min(remaining, 250)
            print("  Deep FW-LP n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = fw_lp_optimize(a_best, time_limit=t_alloc, aggressive=True)
            try_and_update(a_opt, state, "deep_fw")

        # Coordinate descent
        remaining = deadline - time.monotonic() - 400
        if remaining > 10:
            t_alloc = min(remaining, 100)
            print("  Coord descent, t={:.0f}s".format(t_alloc))
            a_cd, _ = coord_descent_quadratic(state[1], time_limit=t_alloc)
            try_and_update(a_cd, state, "cd_deep")

        # Smoothed gradient
        remaining = deadline - time.monotonic() - 350
        if remaining > 10:
            t_alloc = min(remaining, 50)
            print("  Smooth grad, t={:.0f}s".format(t_alloc))
            a_sg, _ = smoothed_grad_descent(state[1], time_limit=t_alloc,
                                            T_init=0.0005, lr_init=0.02)
            try_and_update(a_sg, state, "smooth_deep")

    print("  Phase 2 best: {:.8f}".format(state[0]))

    # === Phase 3: Multi-Start Perturbation ===
    print("\n=== Phase 3: Multi-Start Perturbation ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        a_cur = state[1]
        n_cur = len(a_cur)

        # Various perturbation scales
        pert_scales = [0.002, 0.005, 0.01, 0.02, 0.03]

        for scale in pert_scales:
            remaining = deadline - time.monotonic() - 200
            if remaining < 15:
                break

            # Multiple restarts at this scale
            n_restarts = 2
            for r in range(n_restarts):
                remaining = deadline - time.monotonic() - 150
                if remaining < 10:
                    break

                rng_r = np.random.default_rng(seed + r * 1000 + int(scale * 10000))
                perturbed = project_simplex(a_cur + rng_r.standard_normal(n_cur) * scale)

                # Short FW
                t_alloc = min(remaining, 40)
                a_opt, _ = fw_lp_optimize(perturbed, time_limit=t_alloc)
                try_and_update(a_opt, state, "pert_{:.3f}_r{}".format(scale, r))

                # Quick coord descent
                remaining = deadline - time.monotonic() - 100
                if remaining > 5:
                    a_cd, _ = coord_descent_quadratic(a_opt, time_limit=min(remaining, 15))
                    try_and_update(a_cd, state, "cd_pert_{:.3f}_r{}".format(scale, r))

        # Multi-peak subgradient
        for top_k, lr, mom in [(5, 0.003, 0.93), (20, 0.002, 0.95)]:
            remaining = deadline - time.monotonic() - 100
            if remaining < 10:
                break
            t_alloc = min(remaining, 40)
            print("  MPSG k={}, t={:.0f}s".format(top_k, t_alloc))
            a_opt, _ = multi_peak_sg(state[1], time_limit=t_alloc,
                                     top_k=top_k, lr_init=lr, mom=mom)
            try_and_update(a_opt, state, "mpsg_k{}".format(top_k))

    print("  Phase 3 best: {:.8f}".format(state[0]))

    # === Phase 4: Try different n around best ===
    print("\n=== Phase 4: n-Refinement ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_cur = len(state[1])
        n_options = [n_cur - 32, n_cur - 16, n_cur, n_cur + 16, n_cur + 32]
        n_options = [n for n in n_options if 128 <= n <= 2048]
        n_options = sorted(set(n_options))

        for n_try in n_options:
            if n_try == n_cur:
                continue
            remaining = deadline - time.monotonic() - 80
            if remaining < 15:
                break

            start = rescale_to_n(state[1], n_try)
            t_alloc = min(remaining, 30)
            print("  n-refine n={}, t={:.0f}s".format(n_try, t_alloc))
            a_opt, _ = fw_lp_optimize(start, time_limit=t_alloc)
            try_and_update(a_opt, state, "nref_{}".format(n_try))

            remaining = deadline - time.monotonic() - 60
            if remaining > 5:
                a_cd, _ = coord_descent_quadratic(a_opt, time_limit=min(remaining, 10))
                try_and_update(a_cd, state, "cdnref_{}".format(n_try))

    print("  Phase 4 best: {:.8f}".format(state[0]))

    # === Phase 5: Final Polish ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Final coord descent
        remaining = deadline - time.monotonic() - 40
        if remaining > 10:
            t_alloc = min(remaining, 60)
            print("  Final coord descent, t={:.0f}s".format(t_alloc))
            a_cd, _ = coord_descent_quadratic(state[1], time_limit=t_alloc)
            try_and_update(a_cd, state, "final_cd")

        # Final FW (short)
        remaining = deadline - time.monotonic() - 20
        if remaining > 5:
            t_alloc = min(remaining, 30)
            print("  Final FW, t={:.0f}s".format(t_alloc))
            a_opt, _ = fw_lp_optimize(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "final_fw")

        # Ultra-fine perturbations
        remaining = deadline - time.monotonic() - 5
        if remaining > 3:
            for scale in [0.0005, 0.001]:
                remaining = deadline - time.monotonic() - 3
                if remaining < 2:
                    break
                rng_f = np.random.default_rng(seed + 99999)
                a_pert = project_simplex(state[1] + rng_f.standard_normal(len(state[1])) * scale)
                a_opt, _ = smoothed_grad_descent(a_pert, time_limit=min(remaining, 5),
                                                 T_init=0.0002, lr_init=0.005)
                try_and_update(a_opt, state, "fine_{:.4f}".format(scale))

    print("  Phase 5 best: {:.8f}".format(state[0]))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(384) / 384
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL: {:.8f} (n={}) time={:.0f}s".format(
        state[0], len(state[1]), time.monotonic() - (deadline - budget_s + 20)))
    return [float(x) for x in state[1]]
