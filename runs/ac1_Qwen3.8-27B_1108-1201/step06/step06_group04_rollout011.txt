import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, p):
    """
    Compute gradient of sum_k p[k] * (a*a)[k] w.r.t. a.
    p has length 2n-1, a has length n.
    Returns array of length n.
    """
    n = len(a)
    # grad[j] = 2 * sum_m p[j+m] * a[m] for m = 0..n-1
    # = 2 * np.correlate(p, a, mode='valid')[j]
    cross = np.correlate(p, a, mode='valid')
    return 2.0 * cross


def logsumexp_grad(a0, time_limit, T_init=0.001, lr_init=0.01, 
                   T_max=0.1, restart_every=200):
    """
    Log-sum-exp smoothed gradient descent with adaptive temperature and restarts.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        grad = compute_gradient(a, p)
        
        # Backtracking line search
        best_candidate = a
        best_val = conv.max()
        step = lr
        for _ in range(10):
            candidate = project_simplex(a - step * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                break
            step *= 0.5
            if step < 1e-10:
                break
        
        a = best_candidate
        obj = best_val
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt temperature and learning rate
        if no_improve > 20:
            T = min(T * 1.3, T_max)
            lr *= 0.9
        if no_improve > 60:
            T *= 0.5  # Re-focus on peaks
        if no_improve > restart_every:
            rng = np.random.default_rng((it * 2654435761 + 12345) % (2**31))
            noise = 0.001 + 0.002 * min(no_improve / restart_every, 1.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj


def peak_targeting(a0, time_limit, topk=5):
    """
    Directly target the top-K peaks of the autocorrelation.
    At each step, compute the gradient of the sum of top-K peaks
    and take a projected gradient step with line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = 0.01
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        k_use = min(topk, len(conv))
        peaks = np.argpartition(conv, -k_use)[-k_use:]
        
        # Subgradient of sum of top-K peaks
        grad = np.zeros(n)
        for k in peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * a[k - idx]
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Line search
        best_candidate = a
        best_val = conv.max()
        step = lr
        for _ in range(8):
            candidate = project_simplex(a - step * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                break
            step *= 0.5
            if step < 1e-12:
                break
        
        a = best_candidate
        obj = best_val
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.05, 0.1)
        else:
            no_improve += 1
            lr *= 0.95
        
        if no_improve > 100:
            # Restart with perturbation
            rng = np.random.default_rng((it * 7919 + 31) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            lr = 0.01
            no_improve = 0
    
    return best_a, best_obj

        power = mag ** 2  # |A(f)|^2

        # We want to make this flat.
        # In the frequency domain, "flat autocorrelation" means power is concentrated at f=0.
        # But power[0] = (sum a)^2 = 1 is fixed.
        # So we want to minimize sum_{f != 0} power[f].
        # This is equivalent to maximizing power[0] / sum(power) = 1 / sum(power).
        # But sum(power) = N * sum(|a|^2) by Parseval.
        # So we want to minimize sum(|a|^2), i.e., make a as "spread out" as possible.
        # But we also need a >= 0 and sum(a) = 1.
        # The minimum of sum(a^2) given sum(a) = 1 and a >= 0 is 1/n (uniform).
        # So the spectral approach would just give us the uniform sequence!
        
        # That's not helpful. Let me think differently.
        # The issue is that "flat autocorrelation" in the time domain doesn't correspond to
        # a simple constraint in the frequency domain.
        
        # Alternative: use the water-filling idea.
        # The autocorrelation at lag k is: b[k] = (1/N) * sum_f power[f] * exp(2*pi*i*k*f/N)
        # We want all b[k] to be equal to 1/(2n-1).
        # This means the IDFT of power should be a constant (for k in the range of interest).
        
        # A different approach: iteratively reduce the peaks.
        # 1. Compute b = IDFT(power)[:2n-1]
        # 2. Find the peak at k*.
        # 3. Reduce power at frequencies that contribute most to b[k*].
        # 4. Renormalize power to maintain sum(a) = 1.

        # But we need to maintain power[0] = 1 (since sum a = 1)
        
        # Actually, this approach is not straightforward. Let me try a simpler version:
        # Just do a gradient step in the a domain targeting the peak.
        # This is essentially the same as peak_targeting.
        
        # Let me try a different spectral approach:
        # Adjust the phase of A to make a more "uniform" while maintaining |A|.
        # This is like a phase retrieval problem.
        
        # Actually, I think the spectral approach as stated is not directly useful.
        # Let me replace it with a "moment matching" approach.
