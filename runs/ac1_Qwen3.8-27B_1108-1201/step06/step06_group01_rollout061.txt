```
def project_simplex(v):
    n=len(v)
    if n==0: return np.array([])
    if n==1: return np.array([max(float(v[0]),0.0)])
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    inds=np.arange(1,n+1)
    cond = u*inds > (cssv - 1.0)
    if not np.any(cond): return np.ones(n)/n
    rho=np.nonzero(cond)[0][-1]
    theta=(cssv[rho]-1.0)/(rho+1.0)
```

```python
"""
Black-box convex / derivative-free optimizer for the step-function height
sequence.  The program:
  - probes the evaluator a little to detect the presence of the zero-lag term,
  - builds many diverse initial candidates from the parent (if available),
  - runs a hybrid Frank-Wolfe / projected-direction local search that uses:
       * cheap autocorrelation-lag directional estimates,
       * frequent true-evaluator finite-difference directional estimates,
       * adaptive line searches toward points, pairs, uniform, and gradient-like
         projected directions.
  - keeps the best sequence by the actual evaluator and returns it.
"""

import numpy as np
import time
import math


def safe_score(a):
    """Call the global evaluate_sequence in a defensive way."""
    try:
        arr = np.asarray(a, dtype=float)
        if arr.size == 0:
            return float("inf")
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.sum() < 0.01:
            return float("inf")
        return float(evaluate_sequence(arr.tolist()))
    except Exception:
        return float("inf")


def normalize_heights(a, target_sum=1.0):
    """Return a nonnegative float vector with sum target_sum."""
    arr = np.asarray(a, dtype=float)
    if arr.size == 0:
        return arr
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = arr.sum()
    if s <= 1e-15:
        arr = np.ones_like(arr)
        s = arr.sum()
    return (target_sum / s) * arr


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    arr = np.asarray(v, dtype=float)
    n = len(arr)
    if n == 0:
        return arr
    if n == 1:
        return np.array([max(float(arr[0]), 0.0)])
    arr = np.where(np.isfinite(arr), arr, 0.0)
    u = np.sort(arr)[::-1]
    cssv = np.cumsum(u)
    inds = np.arange(1, n + 1)
    cond = u * inds > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(arr - theta, 0.0)


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=float)
    n = len(a)
    size = 2 * n - 1
    m = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, m)
    out = np.fft.irfft(A * A, m)
    return out[:size]


def resample_heights(parent, n):
    """Linearly resample a height vector to length n (boundary-safe)."""
    parent = np.asarray(parent, dtype=float)
    n_old = len(parent)
    if n_old == n:
        return parent.copy()
    if n_old < 1:
        return np.ones(n)
    if n_old == 1:
        return np.full(n, float(parent[0]))
    lo, hi = 0.02, 0.98
    x_old = np.linspace(lo, hi, n_old)
    x_new = np.linspace(lo, hi, n)
    out = np.interp(x_new, x_old, parent)
    return np.maximum(out, 0.0)


def top_active_lags(bp, include_zero, k):
    """
    bp is b[n-1:] from full auto-convolution, so bp[d] corresponds to lag d.
    Return up to k lag indices d, possibly excluding d=0.
    """
    m = len(bp)
    if m <= 0:
        return np.array([], dtype=int)
    if include_zero:
        if k >= m:
            return np.arange(m)
        return np.argpartition(bp, -k)[-k:]
    else:
        if m <= 1:
            return np.array([], dtype=int)
        if k >= m - 1:
            return np.arange(1, m)
        return np.argpartition(bp[1:], -k)[-k:] + 1


def proxy_pq(a, bp, active):
    """
    Cheap heuristic subgradient-like values for moving mass toward pure points:
      V_i approx directional derivative toward e_i - a for a max-of-lags proxy.
    Returns p=min V, q=max V, and V.
    """
    a = np.asarray(a, dtype=float)
    n = len(a)
    if n == 0:
        return 0, 0, np.array([])
    vals = np.full(n, -np.inf)
    for dv in active:
        d = int(dv)
        if d < 0 or d >= n:
            continue
        bd = float(bp[d])
        if not math.isfinite(bd):
            continue
        if d == 0:
            val_d = 2.0 * a - bd
        else:
            val_d = -np.full(n, bd)
            m = n - d
            if m > 0:
                val_d[:m] += a[d:]
                val_d[d:] += a[:m]
        finite = np.isfinite(val_d)
        vals[finite] = np.maximum(vals[finite], val_d[finite])
    if not np.any(np.isfinite(vals)):
        vals = np.zeros(n)
    else:
        vals = np.where(np.isfinite(vals), vals, 0.0)
    p = int(np.argmin(vals))
    q = int(np.argmax(vals))
    return p, q, vals


def zero_sum_direction(vals):
    """Convert heuristic values V_i to a normalized zero-sum direction -V."""
    v = np.asarray(vals, dtype=float)
    if v.size == 0:
        return None
    v = np.where(np.finite(v), v, 0.0)
    d = -(v - float(np.mean(v)))
    sc = float(np.max(np.abs(d)))
    if sc <= 1e-15:
        return None
    return d / sc


def target_candidate(a, t, alpha):
    """Candidate (1-alpha)*a + alpha*t, projected safely to the simplex."""
    c = (1.0 - alpha) * np.asarray(a, dtype=float) + alpha * np.asarray(t, dtype=float)
    c = np.maximum(c, 0.0)
    s = c.sum()
    if s <= 1e-15:
        return None
    return c / s


def pair_candidate(a, p, q, alpha):
    """Candidate a + alpha*(e_p - e_q), safely clipped."""
    c = np.asarray(a, dtype=float).copy()
    c[p] += alpha
    c[q] -= alpha
    c = np.maximum(c, 0.0)
    s = c.sum()
    if s <= 1e-15:
        return None
    return c / s


def additive_candidate(a, d, alpha):
    """Candidate a + alpha*d projected onto the simplex."""
    c = np.asarray(a, dtype=float) + alpha * np.asarray(d, dtype=float)
    return project_simplex(c)


def line_search_target(a, t, F, max_evals=18, tol=1e-10):
    """Line search on (1-alpha)*a + alpha*t."""
    grid = [0.001, 0.003, 0.01, 0.03, 0.1, 0.25, 0.5, 0.75, 0.9, 0.97]
    best_a = np.asarray(a, dtype=float).copy()
    best_F = float(F)
    best_alpha = 0.0
    best_idx = -1
    evals = 0

    for idx, alpha in enumerate(grid):
        if evals >= max(0, max_evals - 8):
            break
        c = target_candidate(a, t, alpha)
        val = float("inf") if c is None else safe_score(c)
        evals += 1
        if math.isfinite(val) and val < best_F - tol:
            best_F = val
            best_alpha = alpha
            best_idx = idx
            best_a = c.copy()

    if best_idx < 0:
        return best_a, best_F, 0.0

    lo = 0.0 if best_idx == 0 else max(0.0, grid[best_idx - 1] * 0.6 - 0.001)
    hi = 1.0
    if best_idx + 1 < len(grid):
        hi = min(1.0, grid[best_idx + 1] * 1.4 + 0.002)
    if hi <= lo:
        hi = min(1.0, max(0.02, best_alpha * 1.5 + 0.002))
    if hi < best_alpha:
        hi = min(1.0, best_alpha + max(0.001, 0.5 * (best_alpha - lo)))

    iters = min(5, max(0, (max_evals - evals) // 2))
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        c1 = target_candidate(a, t, m1)
        f1 = float("inf") if c1 is None else safe_score(c1)
        c2 = target_candidate(a, t, m2)
        f2 = float("inf") if c2 is None else safe_score(c2)
        evals += 1
        if math.isfinite(f1) and f1 < best_F - tol:
            best_F = f1
            best_alpha = m1
            best_a = c1.copy()
        if math.isfinite(f2) and f2 < best_F - tol:
            best_F = f2
            best_alpha = m2
            best_a = c2.copy()
        if (not math.isfinite(f1) and math.isfinite(f2)) or (math.isfinite(f1) and f1 < f2):
            hi = m2
        else:
            lo = m1

    if evals < max_evals:
        xm = 0.5 * (lo + hi)
        cm = target_candidate(a, t, xm)
        fm = float("inf") if cm is None else safe_score(cm)
        evals += 1
        if math.isfinite(fm) and fm < best_F - tol:

            best_alpha = xm
            best_a = cm.copy()

    return best_a, best_F, best_alpha

def line_search_additive(a, d, F, max_evals=8, tol=1e-10):
    """Line search on projected a + alpha*d."""
    a = np.asarray(a, dtype=float)
    d = np.asarray(d, dtype=float)
    alphas = [1e-5, 1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 1e-1]
    best_a = a.copy()

    evals = 0
    for alpha in alphas:
        if evals >= max_evals:
            break
        c = additive_candidate(a, d, alpha)
        val = float("inf") if c is None else safe_score(c)
        evals += 1
        if math.isfinite(val) and val < best_F - tol:
            best_F = val
            best_a = c.copy()
    return best_a, best_F


def estimate_h_full(a, F, h, deadline=None):
    """
    One-sided directional derivatives toward pure points:
      H_i = (F((1-h)a + h e_i) - F(a)) / h.
    Only differences matter for many simplex methods.
    """
    a = np.asarray(a, dtype=float)
    n = len(a)
    vals = np.full(n, 1e100)
    base = (1.0 - h) * a
    for i in range(n):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = base.copy()
        c[i] += h
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s <= 1e-15:
            continue
        c = c / s
        val = safe_score(c)
        if math.isfinite(val):
            vals[i] = (val - float(F)) / h
    return vals


def local_pair_fw(a0, budget, use_zero, seed=123, active_k=10, full_grad_every=8):
    """Hybrid local search using the true evaluator as objective."""
    a = normalize_heights(a0)
    F = safe_score(a)
    if not math.isfinite(F):
        return a, F

    best_a = a.copy()
    best_F = float(F)
    n = len(a)
    t0 = time.monotonic()
    it = 0
    stall = 0
    h = 0.01
    uniform = np.ones(n) / n
    rng = np.random.default_rng(int(seed) + n)

    while True:
        elapsed = time.monotonic() - t0
        rem = float(budget) - elapsed
        if rem <= 1.0:
            break

        it += 1
        b = fast_convolve(a)
        bp = b[n - 1:]
        active = top_active_lags(bp, use_zero, active_k)
        p, q, vals_proxy = proxy_pq(a, bp, active)

        improved = False

        # Pairwise FW-style move: from q to p.
        if p != q and float(a[q]) > 1e-10:
            ca, cf, _ = line_search_pair(a, p, q, F, max_evals=14)
            if math.isfinite(cf) and cf < F - 1e-10:
                a = ca
                F = cf
                improved = True

        # Pure-point FW-style move.
        if not improved:
            t = np.zeros(n)
            t[p] = 1.0
            ca, cf, _ = line_search_target(a, t, F, max_evals=12)
            if math.isfinite(cf) and cf < F - 1e-10:
                a = ca
                F = cf
                improved = True

        # Projected gradient-like move from cheap proxy values.
        if not improved and vals_proxy is not None:
            d = zero_sum_direction(vals_proxy)
            if d is not None:
                ca, cf = line_search_additive(a, d, F, max_evals=7)
                if math.isfinite(cf) and cf < F - 1e-10:
                    a = ca
                    F = cf
                    improved = True

        # Full true-evaluator directional estimate every so often.
        if not improved and (it % full_grad_every == 0 or stall > 5):
            if rem > 3.0 and n <= 1500:
                vals = estimate_h_full(a, F, h, deadline=time.monotonic() + min(10.0, max(0.5, rem - 1.0)))
                vals = np.where(np.isfinite(vals), vals, 1e100)
                p2 = int(np.argmin(vals))
                q2 = int(np.argmax(vals))

                if p2 != q2 and float(a[q2]) > 1e-10:
                    ca, cf, _ = line_search_pair(a, p2, q2, F, max_evals=12)
                    if math.isfinite(cf) and cf < F - 1e-10:
                        a = ca
                        F = cf
                        improved = True

                if not improved:
                    t2 = np.zeros(n)
                    t2[p2] = 1.0
                    ca, cf, _ = line_search_target(a, t2, F, max_evals=10)
                    if math.isfinite(cf) and cf < F - 1e-10:
                        a = ca
                        F = cf
                        improved = True

                if not improved:
                    d2 = zero_sum_direction(vals)
                    if d2 is not None:
                        ca, cf = line_search_additive(a, d2, F, max_evals=8)
                        if math.isfinite(cf) and cf < F - 1e-10:
                            a = ca
                            F = cf
                            improved = True

        # Blend toward uniform; useful when the true objective rewards spreading mass.
        if not improved:
            ca, cf, _ = line_search_target(a, uniform, F, max_evals=8)
            if math.isfinite(cf) and cf < F - 1e-10:
                a = ca
                F = cf
                improved = True

        if F < best_F - 1e-12:
            best_F = float(F)
            best_a = a.copy()
            stall = 0
            h = max(0.0005, h * 0.85)
        else:
            stall += 1
            h = min(0.05, h * 1.25)

            if stall > 40:
                scale = 0.0008 + 0.0035 * min(1.0, float(stall - 40) / 120.0)
                a2 = project_simplex(best_a + rng.standard_normal(n) * scale)
                F2 = safe_score(a2)
                if math.isfinite(F2) and F2 < best_F + 1e-7:
                    a = a2
                    F = F2
                    stall = 0
                if stall > 260:
                    break

        if best_F < 1e-7:
            break

    return best_a, float(best_F)


def build_initial_candidates(parent, desired_ns, rng):
    """Generate a diverse set of well-normalized starting candidates."""
    cands = []

    if parent is not None:
        pn = len(parent)
        pnorm = normalize_heights(parent)
        cands.append(pnorm)

        for n in desired_ns:
            if 16 <= n <= 1600:
                cands.append(normalize_heights(resample_heights(parent, n)))

        # Useful structured candidates at the parent length.
        cands.append(np.ones(pn) / pn)

        if pn > 1:
            eps = 1e-8
            ad = np.full(pn, eps / (pn - 1))
            ad[0] = 1.0 - eps
            cands.append(normalize_heights(ad))

        for d in [1, pn // 4, pn // 3, pn // 2]:
            if 0 < d < pn:
                at = np.zeros(pn)
                at[0] = 0.5
                at[d] = 0.5
                cands.append(at)

        idx = np.arange(pn)
        for w in [max(1.0, pn / 8.0), max(1.0, pn / 4.0), max(1.0, pn / 2.0)]:
            ag = np.exp(-((idx - pn / 2.0) ** 2) / (2.0 * w * w))
            cands.append(normalize_heights(ag))

        ae = np.exp(-idx / max(1.0, pn / 10.0))
        cands.append(normalize_heights(ae))

        for pp in [0.5, 2.0]:
            ap = idx.astype(float) ** pp
            cands.append(normalize_heights(ap))

    for n in desired_ns:
        if n < 16:
            continue
        cands.append(np.ones(n) / n)

        # Random constant-weight binary candidate.
        m = max(1, n // 2)
        rb = np.zeros(n)
        rb[0] = 1.0
        if n > 1:
            inds = rng.integers(0, n, size=min(m, n - 1), replace=False)
            rb[inds] = 1.0
        cands.append(normalize_heights(rb))

        # Smoothed nonnegative random candidate.
        rs = rng.normal(size=n)
        kernel = np.exp(-((np.linspace(0.0, 1.0, n) - 0.5) ** 2) * 10.0)
        rs = np.convolve(rs, kernel, mode="same")
        rs = np.where(rs > 0.0, rs, 0.0)
        s = rs.sum()
        if s > 1e-15:
            cands.append(rs / s)
        else:
            cands.append(np.ones(n) / n)

    return cands


def finalize_sequence(a):
    """Choose between raw nonnegative output and a tiny positive floor."""
    a = normalize_heights(a)
    if a.size == 0:
        a = np.ones(384) / 384

    raw = a.copy()
    alternatives = [raw]
    for eps in [1e-12, 1e-9, 1e-7]:
        c = a + eps
        c = c / c.sum()
        alternatives.append(c)

    best = raw
    best_s = safe_score(raw)
    for c in alternatives[1:]:
        s = safe_score(c)
        if math.isfinite(s) and (not math.isfinite(best_s) or s < best_s - 1e-12):
            best = c
            best_s = s

    if not math.isfinite(best_s):
        return raw
    return best


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for a low-score nonnegative height sequence and return it as a
    Python list of floats.
    """
    seed = int(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(seed)

    # Load parent if present.
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=float)
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, parent_score={safe_score(parent):.8f}")
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(768) / 768

    # Candidate lengths.
    des = [384, 512, 640, 768, 896, 1024]
    pn = len(parent)
    for nn in [max(64, pn - 128), pn, max(64, pn + 128)]:
        if 64 <= nn <= 1600:
            des.append(int(nn))
    des = sorted(set(des))
    des = [n for n in des if 64 <= n <= 1280]
    if not des:
        des = [768]

    # Probe the evaluator for zero-lag behavior.
    n_probe = 48
    uni_probe = np.ones(n_probe) / n_probe
    f_uni_probe = safe_score(uni_probe)
    delta_probe = np.zeros(n_probe)
    delta_probe[0] = 1.0
    f_delta_probe = safe_score(delta_probe)
    use_zero = bool(math.isfinite(f_uni_probe) and math.isfinite(f_delta_probe) and f_delta_probe > 1.5 * f_uni_probe)
    print(f"Probe: uniform={f_uni_probe:.8f}, delta={f_delta_probe:.8f}, use_zero={use_zero}")

    # Build and score initial candidates.
    candidates = build_initial_candidates(parent, des, rng)
    cand_list = []
    seen = set()
    for a in candidates:
        a = normalize_heights(a)
        if a.size < 16:
            continue
        key = (len(a), tuple(np.round(a, 12).tolist()))
        if key in seen:
            continue
        seen.add(key)
        s = safe_score(a)
        if math.isfinite(s):
            cand_list.append((float(s), a))

    if not cand_list:
        a = np.ones(768) / 768
        s = safe_score(a)
        if not math.isfinite(s):
            return [float(x) for x in a]
        cand_list.append((float(s), a))

    scores = np.array([x[0] for x in cand_list], dtype=float)
    order = np.argsort(scores)

    best_a = cand_list[order[0]][1].copy()
    best_F = float(scores[order[0]])
    print(f"Initial best: {best_F:.8f}, n={len(best_a)}")

    # If something trivially excellent is found, stop early.
    if best_F < 0.25:
        out = finalize_sequence(best_a)
        print(f"Early trivial-good return: {safe_score(out):.8f}, n={len(out)}")
        return [float(x) for x in out]

    # Select a few distinct lengths to optimize.
    selected = []
    seen_ns = set()
    for idx in order:
        s = float(scores[idx])
        if not math.isfinite(s):
            continue
        a = cand_list[idx][1]
        n = len(a)
        if n in seen_ns:
            continue
        seen_ns.add(n)
        selected.append((s, a))
        if len(selected) >= 4:
            break

    phase_fracs = [0.62, 0.23, 0.15]
    phase_idx = 0
    for frac, (s0, a0) in zip(phase_fracs, selected):
        rem = deadline - time.monotonic()
        if rem < 20.0:
            break
        budget = float(frac) * rem
        if budget < 12.0:
            continue
        print(f"\nPhase {phase_idx}: n={len(a0)}, initial={s0:.8f}, budget={budget:.0f}s")
        a1, f1 = local_pair_fw(
            a0,
            budget,
            use_zero,
            seed=seed + 1000 * (phase_idx + 1),
            active_k=10,
            full_grad_every=8
        )
        if math.isfinite(f1) and f1 < best_F:
            best_F = float(f1)
            best_a = a1.copy()
            print(f"  Phase {phase_idx} improved: {best_F:.8f}, n={len(best_a)}")
        phase_idx += 1

    # Final polish on the best candidate.
    rem = deadline - time.monotonic()
    if rem > 40.0 and math.isfinite(best_F):
        print(f"\nFinal polish: n={len(best_a)}, current={best_F:.8f}, remaining={rem:.0f}s")
        a1, f1 = local_pair_fw(
            best_a,
            min(120.0, rem - 10.0),
            use_zero,
            seed=seed + 99991,
            active_k=8,
            full_grad_every=12
        )
        if math.isfinite(f1) and f1 < best_F:
            best_F = float(f1)
            best_a = a1.copy()
            print(f"  Final polish improved: {best_F:.8f}, n={len(best_a)}")

    out = finalize_sequence(best_a)
    final_score = safe_score(out)
    print(f"\nFINAL: score={final_score:.8f}, n={len(out)}")
    return [float(x) for x in out]
```
