            a = a / a.sum()

        else:

    return a

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad(a, conv, T):
    """Compute gradient of log-sum-exp smoothed max in O(n) using correlation."""
    n = len(a)
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    w = w / w.sum()
    # grad[j] = 2 * sum_{m} w[j+m] * a[m]  (valid m, j+m)
    # This is 2 * correlate(w, a, 'valid')
    grad = 2.0 * np.correlate(w, a, mode='valid')[:n]
    return grad, w

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    if choice == 0:
        # Uniform
        a = np.ones(n)
    elif choice == 1:
        # Cosine-weighted (symmetric)
        j = np.arange(n)
        a = 1.0 + np.cos(2.0 * np.pi * j / n)
    elif choice == 2:
        # Asymmetric linear ramp
        j = np.arange(n)
        a = 1.0 + 2.0 * j / (n - 1)
    elif choice == 3:
        # Bimodal (mass at both ends, spread)
        j = np.arange(n)
        center = (n - 1) / 2.0
        a = np.exp(-0.5 * ((j - 0.0) / (n * 0.1)) ** 2) + np.exp(-0.5 * ((j - (n - 1)) / (n * 0.1)) ** 2)
    elif choice == 4:
        # Random
        a = rng.random(n)
    else:
        # Asymmetric Gaussian (lopsided)
        j = np.arange(n)
        center = n * 0.3  # Off-center
        a = np.exp(-0.5 * ((j - center) / (n * 0.2)) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

    deadline = t_start + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_eval = list(np.clip(a, 0.0, 1000.0))
        score = float(evaluate_sequence(a_eval))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"No parent: {e}")

    # Phase 1: Quick multi-n exploration (~200s)
    print("\n=== Phase 1: Multi-n Quick Exploration ===")
    t_phase1_end = t_start + 200

    n_candidates = [384, 512, 640, 768, 896, 1024]
    if parent is not None:
        n_candidates.insert(0, len(parent))

    phase1_best = {}
    for n_try in sorted(set(n_candidates)):
        if time.monotonic() > t_phase1_end - 15:
            break
        t_alloc = min(25, t_phase1_end - time.monotonic() - 10)
        if t_alloc < 5:
            break

        # Try 2 different initializations for this n
        for init_id in range(2):
            if time.monotonic() > t_phase1_end - 5:
                break
            if init_id == 0 and parent is not None:
                # Interpolate parent to length n_try
                if n_try == len(parent):
                    start = parent.copy()
                else:
                    start = np.interp(np.linspace(0, 1, n_try),
                                      np.linspace(0, 1, len(parent)), parent)
                    start = np.maximum(start, 0)
                    ss = start.sum()
                    if ss > 1e-10:
                        start = start / ss
                    else:
                        start = np.ones(n_try) / n_try
            else:
                start = structured_init(n_try, rng)

            t_sub = max(3, t_alloc / 2 - 1)
            a_opt, _ = nesterov_optimize(start, time_limit=t_sub, lr0=0.08)
            sc = try_seq(a_opt, f"p1_n{n_try}_{init_id}")
            if n_try not in phase1_best or sc < phase1_best[n_try]:
                phase1_best[n_try] = sc

    print(f"  Phase 1 done. Best: {best_score:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # Phase 2: Deep optimization on top candidates (~800s)
    print("\n=== Phase 2: Deep Nesterov Optimization ===")
    t_phase2_end = t_start + 1000

    if best_seq is not None:
        n_best = len(best_seq)
        # Long run on the best
        t_alloc = min(300, t_phase2_end - time.monotonic() - 200)
        if t_alloc > 30:
            print(f"  Deep Nesterov n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_optimize(best_seq, time_limit=t_alloc, lr0=0.03)
            try_seq(a_opt, "deep_nest")

        # Multi-temperature
        t_alloc = min(150, t_phase2_end - time.monotonic() - 150)
        if t_alloc > 30:
            print(f"  Multi-temp n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = multi_temp_optimize(a_opt, time_limit=t_alloc)
            try_seq(a_opt, "multi_temp")

    # Try other promising n values
    for n_try in sorted(phase1_best.keys(), key=lambda k: phase1_best[k]):
        if time.monotonic() > t_phase2_end - 30:
            break
        if n_try == len(best_seq) if best_seq is not None else False:
            continue
        t_alloc = min(80, t_phase2_end - time.monotonic() - 20)
        if t_alloc < 20:
            break
        if best_seq is not None:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, len(best_seq)), best_seq)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss > 1e-10:
                start = start / ss
            else:
                start = np.ones(n_try) / n_try
        else:
            start = structured_init(n_try, rng)
        print(f"  Deep n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = nesterov_optimize(start, time_limit=t_alloc, lr0=0.04)
        try_seq(a_opt, f"deep_n{n_try}")

    print(f"  Phase 2 done. Best: {best_score:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # Phase 3: Peak coordinate descent polish (~400s)
    print("\n=== Phase 3: Peak Coordinate Descent ===")
    t_phase3_end = t_start + 1450

    if best_seq is not None:
        t_alloc = min(200, t_phase3_end - time.monotonic() - 50)
        if t_alloc > 20:
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_d")

    print(f"  Phase 3 done. Best: {best_score:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # Phase 4: Final refinement with very fine optimization (~300s)
    print("\n=== Phase 4: Final Refinement ===")
    t_phase4_end = deadline

    if best_seq is not None:
        # Very fine Nesterov with small lr
        t_alloc = min(150, t_phase4_end - time.monotonic() - 50)
        if t_alloc > 20:
            print(f"  Fine Nesterov, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_optimize(best_seq, time_limit=t_alloc,
                                         T0=1.0 / (len(best_seq) * 5), lr0=0.01)
            try_seq(a_opt, "fine_nest")

        # One more coordinate descent pass
        t_alloc = min(100, t_phase4_end - time.monotonic() - 20)
        if t_alloc > 15:
            print(f"  Final coord, t={t_alloc:.0f}s")
            a_opt, _ = peak_coordinate_descent(a_opt, time_limit=t_alloc)
            try_seq(a_opt, "final_coord")

        # Micro perturbation polish
        t_alloc = min(50, t_phase4_end - time.monotonic() - 5)
        if t_alloc > 10:
            n = len(a_opt)
            a_cur = a_opt.copy()
            t0_micro = time.monotonic()
            it = 0
            while time.monotonic() - t0_micro < t_alloc:
                it += 1
                for scale in [0.001, 0.003, 0.0005]:
                    rng_m = np.random.default_rng(it * 2654435761 + scale * 10000)
                    pert = a_cur + rng_m.standard_normal(n) * scale
                    pert = project_simplex(pert)
                    c = fast_convolve(pert)
                    if c.max() < fast_convolve(a_cur).max() - 1e-15:
                        a_cur = pert
                if it % 50 == 0:
                    try_seq(a_cur, f"micro_{it}")
            try_seq(a_cur, "micro_final")

    print(f"  Phase 4 done. Best: {best_score:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(list(np.clip(best_seq, 0.0, 1000.0))))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_start:.0f}s")
    return [float(x) for x in best_seq]
