```python
def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def md_gradient(a, p):
    """Compute smoothed gradient: 2 * sum_k p[k] * a[k-i] for each i."""
```

This is a simpler LP (fewer variables if n is small relative to 2n-1... well, they're the same order). But the dual has a simpler structure: the constraint matrix A^T has n rows and 2n-1 columns, with each row i having non-zeros in columns from i to i+n-1 (shifted).

Actually, I think solving the primal directly is fine. Let me just optimize the matrix construction.

```python
def build_lp_indices(n):
    """Precompute row/col indices for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

def solve_lp_oracle_fast(f, M, n, row_idx, col_idx):
    """Solve LP oracle with precomputed indices."""
    c = -np.ones(n)
    # Values: A[k,i] = f[k-i]
    vals = f[row_idx - col_idx]  # vectorized!
    
    from scipy.sparse import csr_matrix
    A_mat = csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))
    
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None
```

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def md_gradient(a, p):
