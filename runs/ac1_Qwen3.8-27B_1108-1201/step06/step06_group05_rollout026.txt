def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])

def hankel_mv(p, a):
    """Compute sum_k p[k]*a[k-m] for each m. This is the gradient of sum_k p[k]*conv_k."""
    return np.correlate(p, a, mode='valid')


def weighted_grad(a, conv, alpha=2.0):
    """Compute gradient with geometric peak weighting.
    Weights decay geometrically from the top peaks.
    """
    n = len(a)
    cmax = conv.max()
    if cmax < 1e-30:
        return np.zeros(n)
    # Geometric weighting: w_k = (conv_k / cmax)^alpha
    # This emphasizes peaks while still considering all lags
    w = (conv / cmax) ** alpha
    w = w / w.sum()
    return 2.0 * hankel_mv(w, a)


def peak_shaving_grad(a, conv, top_k=10, decay=0.7):
    """Gradient that strongly emphasizes the top-k peaks with geometric decay."""
    n = len(a)
    cmax = conv.max()
    if cmax < 1e-30:
        return np.zeros(n)
    # Get top k peak indices
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    # Sort peaks by value
    peaks = peaks[np.argsort(conv[peaks])[::-1]]
    # Assign geometric weights
    w = np.zeros(len(conv))
    for rank, k in enumerate(peaks):
        w[k] = decay ** rank
    w = w / (w.sum() + 1e-30)
    return 2.0 * hankel_mv(w, a)


def bb_step(prev_a, curr_a, prev_grad, curr_grad):
    """Barzilai-Borwein step size."""
    s = curr_a - prev_a
    y = curr_grad - prev_grad
    sTs = np.dot(s, s)
    sTy = np.dot(s, y)
    if sTs < 1e-30:
        return 0.01
    # BB+ (use max of the two BB parameters)
    if sTy > 1e-15:
        bb_plus = sTs / sTy
    else:
        bb_plus = 0.01
    if sTy < -1e-15:
        bb_minus = sTs / (-sTy)
    else:
        bb_minus = 0.01
    # Bound the step
    return np.clip(max(bb_plus, bb_minus), 1e-5, 0.5)

def solve_lp_oracle(f, M, max_time=20):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'max_time': max_time})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

            rng = np.random.default_rng(it * 104729 + 13)
            scale = 0.003 + 0.005 * min(1.0, no_improve / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def genetic_populate(pop, time_limit, seed=42, pop_size=12, elite_frac=0.3):
    """Genetic algorithm with median crossover and adaptive mutation."""
    rng = np.random.default_rng(seed)
    n = len(pop[0])
    elite_count = max(2, int(pop_size * elite_frac))

    # Evaluate initial population
    scores = []
    for i in range(pop_size):
        conv = fast_convolve(pop[i])
        scores.append(conv.max())
    scores = np.array(scores)

    best_idx = np.argmin(scores)
    best_a = pop[best_idx].copy()
    best_score = scores[best_idx]

    t0 = time.monotonic()
    gen = 0

    while time.monotonic() - t0 < time_limit:
        gen += 1
        if time.monotonic() - t0 > time_limit - 2.0:
            break

        # Sort by score
        order = np.argsort(scores)
        pop_sorted = [pop[i].copy() for i in order]
        scores_sorted = scores[order]

        # Elite survive
        elites = pop_sorted[:elite_count]

        # Generate offspring
        new_pop = list(elites)
        for _ in range(pop_size - elite_count):
            if time.monotonic() - t0 > time_limit - 1.0:
                break
            # Select parents (tournament of 3)
            parents = []
            for _ in range(2):
                candidates = rng.integers(0, min(8, pop_size), size=3)
                best_c = candidates[np.argmin(scores[candidates])]
                parents.append(pop[best_c])

            # Median crossover
            child = np.median(np.stack(parents), axis=0)
            child = np.maximum(child, 0.0)
            cs = child.sum()
            if cs < 1e-15:
                continue
            child = child / cs

            # Mutation: structured noise
            mut_rate = 0.02 + 0.03 * min(1.0, gen / 50.0)
            noise = rng.standard_normal(n) * mut_rate
            child = project_simplex(child + noise)

            new_pop.append(child)

        pop = new_pop[:pop_size]

        # Quick local optimization on a few of the new ones
        for i in range(len(pop)):
            if i >= elite_count and time.monotonic() - t0 < time_limit - 5:
                a_opt, _ = optimize_adaptive_grad(pop[i], time_limit=2.0, alpha_init=2.0, top_k_init=8)
                pop[i] = a_opt

        # Re-evaluate
        scores = np.array([fast_convolve(p).max() for p in pop])
        new_best = scores.min()
        if new_best < best_score - 1e-12:
            best_score = new_best
            best_a = pop[np.argmin(scores)].copy()

    return best_a, best_score


def spectral_flatten(a, iters=50, time_limit=10.0):
    """Try to flatten the power spectrum to reduce peak auto-conv."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    for it in range(iters):
        if time.monotonic() - t0 > time_limit:
            break
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        # Target: flat power spectrum
        # Compute "excess" power at each frequency
        mean_power = power.mean()
        # Gradient of (sum (power - mean)^2) w.r.t. a
        # This is related to the 4th moment of the spectrum
        excess = power - mean_power
        # Inverse FFT to get gradient
        # d/d a of sum_m |A[m]|^4 = 4 * IFFT(|A|^2 * A) (up to constants)
        grad_spectrum = 4.0 * np.fft.irfft(excess * A, fft_size)[:n]
        grad = np.real(grad_spectrum)

        # Project gradient step
        lr = 0.005
        candidate = project_simplex(a - lr * grad)
        c = fast_convolve(candidate)
        if c.max() < best_obj - 1e-14:
            best_obj = c.max()
            best_a = candidate.copy()
            a = candidate
        else:
            lr *= 0.5

    return best_a, best_obj


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def try_and_update(a, best_state, label=""):
    """Evaluate sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_state
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    best_seq, best_score = best_state
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return (best_seq, best_score)


def peak_shaving_coordinate(a, max_iter=500, time_limit=30.0):
    """Direct coordinate moves to shave the top peak."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        peak_k = np.argmax(conv)
        peak_val = conv[peak_k]

        # Find the position j that most contributes to the peak
        # conv[peak_k] = sum_j a[j]*a[peak_k - j]
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contributions = a[idx] * a[peak_k - idx]
        top_j = idx[np.argmax(contributions)]

        # Find a position to transfer mass to
        # We want to increase a[i] where a[peak_k - i] is small
        # and doesn't create new peaks
        best_gain = 0
        best_i = -1
        # Sample some candidate positions
        candidates = np.random.default_rng(it).integers(0, n, size=min(n, 200))
        for i in candidates:
            if i == top_j:
                continue
            if a[i] < 1e-10:
                # Moving to a zero position
                d = a[top_j] * 0.1
            else:
                d = min(a[top_j], a[i]) * 0.05
            if d < 1e-10:
                continue
            # Estimate change in peak
            # conv[peak_k] changes by: d*(a[peak_k - i] - a[peak_k - top_j]) approximately
            pk_i = peak_k - i
            pk_j = peak_k - top_j
            if 0 <= pk_i < n and 0 <= pk_j < n:
                change = d * (a[pk_i] - a[pk_j])
            else:
                change = -d * a[pk_j] if 0 <= pk_j < n else 0
            if change < -best_gain:
                best_gain = -change
                best_i = i

        if best_i < 0 or best_gain < 1e-10:
            break

        # Make the move
        d = min(a[top_j], max(a[best_i], a[top_j] * 0.1)) * 0.1
        a[top_j] -= d
        a[best_i] += d
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        new_conv = fast_convolve(a)
        if new_conv.max() < best_obj - 1e-12:
            best_obj = new_conv.max()
            best_a = a.copy()
        else:
            # Revert
            a = best_a.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search:
    Phase 1: Quick optimization on parent (150s)
    Phase 2: Multi-n exploration (300s)
    Phase 3: Genetic population search (400s)
    Phase 4: Deep FW-LP + adaptive gradient (500s)
    Phase 5: Coordinate peak-shaving + spectral flattening (300s)
    Phase 6: Final polishing (150s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(parent)
    print(f"Starting optimization. n={n_parent}, initial score={best_score:.8f}")

    # === Phase 1: Quick optimization (~150s) ===
    print("\n=== Phase 1: Quick Optimization ===")
    t_start = time.monotonic()

    remaining = deadline - time.monotonic() - 1100
    if remaining > 20:
        t_alloc = min(remaining, 60)
        print(f"  Adaptive gradient (t={t_alloc:.0f}s)")
        a_opt, _ = optimize_adaptive_grad(best_seq, time_limit=t_alloc, alpha_init=3.0, top_k_init=15)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "phase1_grad")

    remaining = deadline - time.monotonic() - 1050
    if remaining > 20:
        t_alloc = min(remaining, 50)
        print(f"  Softmax gradient (t={t_alloc:.0f}s)")
        a_opt, _ = optimize_softmax(best_seq, time_limit=t_alloc, T_init=0.002, lr_init=0.01)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "phase1_softmax")

    remaining = deadline - time.monotonic() - 1000
    if remaining > 20:
        t_alloc = min(remaining, 40)
        print(f"  FW-LP quick (t={t_alloc:.0f}s)")
        a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "phase1_fw")

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 2: Multi-n exploration (~300s) ===
    print("\n=== Phase 2: Multi-n Exploration ===")
    t_start = time.monotonic()

    n_best = len(best_seq)
    # Try a range of n values around the parent
    candidate_ns = sorted(set([
        n_best,
        max(128, n_best - n_best // 8),
        n_best + n_best // 8,
        max(128, n_best - n_best // 4),
        n_best + n_best // 4,
        n_best * 2,
        n_best // 2 if n_best >= 256 else n_best,
    ]))
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 4096]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 600
        if remaining < 30:
            break

        start = interp_sequence(best_seq, n_try)
        t_alloc = min(remaining, 50)
        print(f"  n={n_try}, t={t_alloc:.0f}s")

        a_opt, _ = optimize_adaptive_grad(start, time_limit=t_alloc * 0.6, alpha_init=2.5, top_k_init=12)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"n{n_try}_grad")

        remaining = deadline - time.monotonic() - 550
        if remaining > 15:
            a_opt2, _ = fw_optimize(a_opt, time_limit=min(remaining, t_alloc * 0.4))
            (best_seq, best_score) = try_and_update(a_opt2, (best_seq, best_score), f"n{n_try}_fw")

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 3: Genetic population search (~400s) ===
    print("\n=== Phase 3: Genetic Population ===")
    t_start = time.monotonic()

    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 500
    if remaining > 60:
        # Build population from diverse starting points
        pop_size = 14
        pop = [best_seq.copy()]
        # Add perturbed versions
        for i in range(1, pop_size):
            noise_scale = 0.01 + 0.03 * (i / pop_size)
            pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            pop.append(pert)
        # Add a couple of interpolated-to-different-n versions (resized back)
        for i in range(2):
            n_alt = n_best + rng.integers(-n_best//4, n_best//4 + 1)
            n_alt = max(128, n_alt)
            interp = interp_sequence(best_seq, n_alt)
            # Resize back to n_best
            pop.append(interp_sequence(interp, n_best))

        t_alloc = min(remaining, 300)
        print(f"  Genetic pop size={pop_size}, t={t_alloc:.0f}s")
        a_opt, _ = genetic_populate(pop, time_limit=t_alloc, seed=seed + 1, pop_size=pop_size)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "genetic")

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 4: Deep FW-LP + adaptive gradient (~500s) ===
    print("\n=== Phase 4: Deep Optimization ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    # Deep FW-LP
    remaining = deadline - time.monotonic() - 350
    if remaining > 30:
        t_alloc = min(remaining, 200)
        print(f"  Deep FW-LP (t={t_alloc:.0f}s)")
        a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "deep_fw")

    # Deep adaptive gradient
    remaining = deadline - time.monotonic() - 250
    if remaining > 20:
        t_alloc = min(remaining, 150)
        print(f"  Deep adaptive grad (t={t_alloc:.0f}s)")
        a_opt, _ = optimize_adaptive_grad(best_seq, time_limit=t_alloc, alpha_init=4.0, top_k_init=20, use_bb=True)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "deep_grad")

    # Softmax with very low temperature
    remaining = deadline - time.monotonic() - 150
    if remaining > 15:
        t_alloc = min(remaining, 100)
        print(f"  Fine softmax (t={t_alloc:.0f}s)")
        a_opt, _ = optimize_softmax(best_seq, time_limit=t_alloc, T_init=0.0005, lr_init=0.008)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "fine_softmax")

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Peak-shaving + spectral flattening (~300s) ===
    print("\n=== Phase 5: Peak Shaving & Spectral ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    # Peak shaving coordinate descent
    remaining = deadline - time.monotonic() - 200
    if remaining > 15:
        t_alloc = min(remaining, 80)
        print(f"  Peak shaving (t={t_alloc:.0f}s)")
        a_opt, _ = peak_shaving_coordinate(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "peak_shave")

    # Spectral flattening
    remaining = deadline - time.monotonic() - 150
    if remaining > 10:
        t_alloc = min(remaining, 50)
        print(f"  Spectral flatten (t={t_alloc:.0f}s)")
        a_opt, _ = spectral_flatten(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "spectral")

    # Second round of adaptive gradient on the result
    remaining = deadline - time.monotonic() - 100
    if remaining > 15:
        t_alloc = min(remaining, 60)
        a_opt, _ = optimize_adaptive_grad(best_seq, time_limit=t_alloc, alpha_init=5.0, top_k_init=10, use_bb=True)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "phase5_grad")

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 6: Final polishing (~150s) ===
    print("\n=== Phase 6: Final Polish ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    # Multiple short optimization runs with different parameters
    configs = [
        (3.0, 15, 0.01),
        (5.0, 8, 0.005),
        (2.0, 30, 0.02),
    ]
    for alpha, tk, lr in configs:
        remaining = deadline - time.monotonic() - 40
        if remaining < 10:
            break
        t_alloc = min(remaining, 30)
        a_opt, _ = optimize_adaptive_grad(best_seq, time_limit=t_alloc, alpha_init=alpha, top_k_init=tk)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"polish_a{alpha}")

    # Tiny perturbation + quick optimization
    for _ in range(3):
        remaining = deadline - time.monotonic() - 20
        if remaining < 8:
            break
        noise = rng.standard_normal(n_best) * 0.002
        f_pert = project_simplex(best_seq + noise)
        t_alloc = min(remaining, 15)
        a_opt, _ = optimize_adaptive_grad(f_pert, time_limit=t_alloc, alpha_init=4.0, top_k_init=12)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "perturb")

    # Final FW
    remaining = deadline - time.monotonic() - 10
    if remaining > 8:
        a_opt, _ = fw_optimize(best_seq, time_limit=min(remaining, 20))
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "final_fw")

    # Symmetrization check (might not help since non-symmetric is better, but check anyway)
    sym_final = (best_seq + best_seq[::-1]) / 2.0
    sym_final = sym_final / sym_final.sum()
    (best_seq, best_score) = try_and_update(sym_final, (best_seq, best_score), "sym_check")

    print(f"  Phase 6 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
