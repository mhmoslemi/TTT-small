"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k=20, temperature=0.01):
    """Compute weighted subgradient of max conv using top peaks with softmax weights."""
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        weights = np.exp(conv - conv.max())
        weights /= weights.sum()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        logits = (pv - pv.max()) / (temperature * pv.max() + 1e-15)
        weights = np.exp(logits)
        weights /= weights.sum()

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        w = weights[idx]
        if w < 1e-12:
            continue
        k = int(peaks[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # dR[k]/da[j] = a[k-j] for j in [js, je]
        grad[js:je + 1] += w * a[k - je:k - js + 1][::-1]

    return 2.0 * grad


def solve_lp_with_dual(f, M, n, row_idx, col_idx):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns (g, dual_vars) or (None, None).
    """
    c = -np.ones(n)
    k_arr = row_idx
    i_arr = col_idx
    vals = f[k_arr - i_arr]

    A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                dual = res.ineqlin if res.ineqlin is not None else np.zeros(2*n-1)
                return g, dual
    except Exception:
        pass
    return None, None

def momentum_subgradient(a0, time_limit, top_k=20, momentum_init=0.5):
    """Momentum-based subgradient with backtracking line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    momentum = momentum_init
    velocity = np.zeros(n)
    lr = 0.02

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)
        grad = compute_subgradient(a, conv, top_k=top_k)

        # Momentum update
        velocity = momentum * velocity + (1 - momentum) * grad
        direction = project_simplex(a - lr * velocity) - a

        # Backtracking line search
        best_alpha = 0.0
        best_val = conv.max()
        alpha = 1.0
        for _ in range(15):
            cand = a + alpha * direction
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                alpha *= 0.5
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
            alpha *= 0.5
            if alpha < 1e-6:
                break

        if best_alpha > 1e-8:
            a = a + best_alpha * direction
            a = np.maximum(a, 0)
            a = a / a.sum()
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve > 10:
            lr *= 0.9
        if lr < 0.001:
            lr = 0.02
            no_improve = 0

        if no_improve > 100:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            velocity = np.zeros(n)
            lr = 0.02
            momentum = 0.5
            no_improve = 0

    return best_a, best_obj


def smooth_objective_descent(a0, time_limit):
    """
    Minimize sum(R[k]^2) as a smooth proxy, then refine with SCA.
    This can escape local optima of the min-max problem.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)

        # Gradient of sum(conv[k]^2) = 2 * sum_k conv[k] * dconv[k]/da
        # dconv[k]/da[i] = a[k-i] for valid k-i
        # grad[i] = 2 * sum_k conv[k] * a[k-i] (for valid indices)
        # This is 2 * (conv cross-correlated with a)[i]
        # = 2 * sum_j conv[i+j] * a[j] for j=0..n-1
        # Compute via FFT: it's the cross-correlation
        conv_rev = conv[::-1]
        corr_full = np.fft.irfft(np.fft.rfft(a, 4*n) * np.fft.rfft(conv_rev, 4*n), 4*n)
        # corr_full[n-1-i] = sum_j a[j] * conv_rev[n-1-i-j] = sum_j a[j] * conv[i+j]
        # So grad[i] = 2 * corr_full[n-1-i]
        grad = 2.0 * corr_full[n-1 : 2*n-1 : -1]
        # Clip to valid range
        grad = np.maximum(grad, -np.inf)  # grad should be non-negative since conv >= 0 and a >= 0

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Projected gradient step with line search
        direction = project_simplex(a - lr * grad) - a
        best_alpha = 0.0
        best_val = conv.max()
        alpha = 1.0
        for _ in range(10):
            cand = a + alpha * direction
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                alpha *= 0.5
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
            alpha *= 0.5

        if best_alpha > 1e-8:
            a = a + best_alpha * direction
            a = np.maximum(a, 0)
            a = a / a.sum()
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            lr *= 0.8
            if lr < 0.001:
                rng = np.random.default_rng((it * 99991 + 3) % (2**31 - 1))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                lr = 0.01
                no_improve = 0

    return best_a, best_obj


def generate_diverse_inits(n, parent, rng, num_inits=8):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # From parent (interpolated)
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine perturbation of various frequencies
    for freq in [1, 2, 3]:
        c = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x / n)
        inits.append(c / c.sum())

    # Edge-heavy
    edge_w = 1.0 / (np.minimum(x, n - 1 - x).astype(float) + 1.0) ** 1.2
    inits.append(edge_w / edge_w.sum())

    # Two-peak
    g2 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Harmonic (helps with spectral flatness)
    harm = 1.0/(x + 1.0) ** 0.5
    inits.append(harm / harm.sum())

    # Sine wave
    sine = 1.0 + 0.5 * np.sin(2 * np.pi * x / n + rng.uniform(0, 2*np.pi))
    inits.append(np.maximum(sine, 0.1) / np.maximum(sine, 0.1).sum())

    return inits


def try_sequence(a, best_state, label=""):

        best_state[0] = score
        best_state[1] = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return score

            a_opt, _ = sca_optimize(start, time_limit=t_alloc)
            try_sequence(a_opt, best_state, f"alt_n{n_alt}")

    print(f"  P4 done: best={best_state[0]:.8f}, t={time.monotonic()-t4:.0f}s")

    # === Phase 5: Final polish (remaining time) ===
    print("\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()

    if best_state[1] is not None:
        remaining = deadline - time.monotonic()
        if remaining > 5:
            t_alloc = min(remaining * 0.5, 60)
            print(f"  Final SCA t={t_alloc:.0f}s")
            a_opt, _ = sca_optimize(best_state[1], time_limit=t_alloc)
            try_sequence(a_opt, best_state, "final_sca")

        remaining = deadline - time.monotonic()
        if remaining > 5:
            t_alloc = min(remaining * 0.3, 30)
            print(f"  Final SG t={t_alloc:.0f}s")
            a_opt2, _ = momentum_subgradient(best_state[1], time_limit=t_alloc, top_k=30)
            try_sequence(a_opt2, best_state, "final_sg")

        remaining = deadline - time.monotonic()
        if remaining > 5:
            t_alloc = min(remaining, 20)
            a_opt3, _ = sca_optimize(best_state[1], time_limit=t_alloc)
            try_sequence(a_opt3, best_state, "final_sca2")

    print(f"  P5 done: best={best_state[0]:.8f}, t={time.monotonic()-t5:.0f}s")

    # Fallback
    if best_state[1] is None:
        best_state[1] = np.ones(768) / 768
        best_state[0] = float(evaluate_sequence(best_state[1].tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_state[1].tolist()))
    if np.isfinite(final_score) and final_score < best_state[0]:
        best_state[0] = final_score

    print(f"\nFINAL: {best_state[0]:.8f} (n={len(best_state[1])})")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")
    return [float(x) for x in best_state[1]]
