import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import diags

def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def conv_gradient(a, k, n=None):
    """Gradient of (a*a)[k] w.r.t. a. Returns vector of length n."""
    if n is None:
        n = len(a)
    grad = np.zeros(n)
    # d(a*a)[k]/da[j] = 2*a[k-j] if 0 <= k-j < n, else 0
    js = np.arange(n)
    valid = (k - js >= 0) & (k - js < n)
    grad[valid] = 2.0 * a[k - js[valid]]
    return grad

def conv_gradient_multi(a, peaks, weights=None):
    """Weighted sum of gradients at multiple peaks."""
    n = len(a)
    if weights is None:
        weights = np.ones(len(peaks))
    grad = np.zeros(n)
    for w, k in zip(weights, peaks):
        if abs(w) < 1e-15:
            continue
        k = int(k)
        js = np.arange(n)
        valid = (k - js >= 0) & (k - js < n)
        grad[valid] += w * 2.0 * a[k - js[valid]]
    return grad

def solve_sdp_small_n(n, time_limit=60):
    """Solve SDP relaxation for small n. Returns optimal a if successful."""
    try:
        import cvxpy as cp
    except ImportError:
        return None
    
    # Variables
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    constraints = [X >> 0, cp.sum(cp.diagonal(X)) == 1]
    
    # Add convolution constraints
    for k in range(2 * n - 1):
        # Build the constraint: sum_{i+j=k, valid} X_{i,j} * (factor) <= t
        # For i != j: factor is 1 (since X is symmetric, X_{i,j} = X_{j,i})
        # For i == j: factor is 1
        # The actual (a*a)[k] = sum_{i+j=k} a_i * a_j
        # In terms of X: for i < j with i+j=k: contributes 2*X_{i,j} (since a_i*a_j = X_{i,j} appears twice)
        # For i = j with 2i=k: contributes X_{i,i}
        # So (a*a)[k] = 2 * sum_{i<j, i+j=k} X_{i,j} + sum_{i: 2i=k} X_{i,i}
        
        expr = 0
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            j = k - i
            if i == j:
                expr += X[i, i]
            elif i < j:
                expr += 2 * X[i, j]
            # if i > j, already counted when i and j were swapped
        constraints.append(expr <= t)
    
    # Solve
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, max_time=time_limit)
        if X.value is not None:
            # Extract a from the diagonal
            a = np.maximum(np.real(cp.diagonal(X.value)), 0)
            s = a.sum()
            if s > 1e-10:
                a = a / s
                return a
    except Exception:
        pass
    return None

                if np.max(np.abs(g - a)) < 1e-12:
                    no_improve += 1
                    if no_improve > 20:
                        break
                    m_relax = min(m_relax * 1.05, 3.0)
                    continue
                
                # Line search
                best_alpha = 0.0
                best_val = M
                for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
                    ft = (1.0 - alpha) * a + alpha * g
                    ft = np.maximum(ft, 0.0)
                    st = ft.sum()
                    if st < 1e-15:
                        continue
                    ft = ft / st
                    val = fast_convolve(ft).max()
                    if val < best_val:
                        best_val = val
                        best_alpha = alpha
                
                if best_alpha > 0:
                    # Bisection refinement
                    lo = max(0.0, best_alpha - 0.05)
                    hi = min(1.0, best_alpha + 0.05)
                    for _ in range(8):
                        mid = (lo + hi) / 2.0
                        ft = (1.0 - mid) * a + mid * g
                        ft = np.maximum(ft, 0.0)
                        st = ft.sum()
                        if st < 1e-15:
                            break
                        ft = ft / st
                        val = fast_convolve(ft).max()
                        if val < best_val:
                            best_val = val
                            best_alpha = mid
                            lo = mid
                        else:
                            hi = mid
                    
                    a = (1.0 - best_alpha) * a + best_alpha * g
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    conv = fast_convolve(a)
                    M = conv.max()
                    m_relax = 1.0
                    
                    if M < best_obj - 1e-15:
                        best_obj = M
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
                    m_relax = min(m_relax * 1.1, 4.0)
            else:
                no_improve += 1
                m_relax = min(m_relax * 1.2, 4.0)
        except Exception:
            no_improve += 1
            m_relax = min(m_relax * 1.2, 4.0)
        
        if no_improve > 40:
            rng = np.random.default_rng(it * 6151 + 7)
            scale = 0.003 + 0.005 * min(1.0, no_improve / 100.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            conv = fast_convolve(a)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0
        
        if it % 50 == 0:
            print(f"    FW it={it}, M={M:.10f}, best={best_obj:.10f}")

    return best_a, best_obj

def smoothed_gradient_descent(a0, time_limit, alpha_init=50.0, lr_init=0.01):
    """
    Minimize log-sum-exp(alpha * c_k) where c = a*a, using Adam.
    This smoothly approximates max_k c_k as alpha increases.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    alpha = alpha_init
    lr = lr_init
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t_adam = 0
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t_adam += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Compute smoothed objective and gradient
        conv = fast_convolve(a)
        
        # Log-sum-exp for numerical stability
        c_max = conv.max()
        scaled = alpha * (conv - c_max)
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Smoothed gradient: alpha * sum_k p_k * grad_k
        grad = conv_gradient_multi(a, np.arange(2*n-1), weights=alpha * p)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t_adam)
        v_hat = v / (1 - beta2**t_adam)
        
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Backtracking line search
        best_candidate = a
        best_val = conv.max()
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            candidate = project_simplex(a - scale * step)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break
        
        a = best_candidate
        
        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt
        if no_improve > 30:
            alpha *= 1.1  # Increase sharpness
            lr *= 0.95
        if no_improve > 100:
            alpha = alpha_init
            lr = lr_init
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            no_improve = 0
            # Small perturbation
            rng = np.random.default_rng(t_adam + int(time.monotonic()*1000))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
        
        if t_adam % 100 == 0:
            print(f"    SmoothGrad it={t_adam}, M={conv.max():.10f}, best={best_obj:.10f}, alpha={alpha:.1f}")

    return best_a, best_obj

def newton_active_set(a0, time_limit, tol=1e-10):
    """
    Newton method on the active set of peaks.
    At optimum: for support S and active peaks P with multipliers mu:
    - sum_{k in P} mu_k * 2*a[k-j] = lambda for all j in S (KKT for a_j > 0)
    - (a*a)[k] = M for all k in P
    - sum_{j in S} a_j = 1
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(500):
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Find active peaks (within 0.1% of max)
        threshold = M * (1.0 - 0.001)
        active = np.where(conv >= threshold)[0]
        support = np.where(a > 1e-12)[0]
        
        if len(active) == 0 or len(support) == 0:
            break
        
        # Build the KKT system
        # For j in support: sum_{k in active} mu_k * 2*a[k-j] = lambda
        # (a*a)[k] = M for k in active
        # sum a_j = 1 for j in support
        
        # We'll use an iterative approach:
        # 1. Given active peaks, compute the "ideal" gradient direction
        # 2. Take a Newton step
        
        # Compute the weighted gradient from active peaks
        # Use equal weights initially, then refine
        p = len(active)
        m = len(support)
        
        if p == 1:
            # Single active peak: simple direction
            k = active[0]
            grad = conv_gradient(a, k, n)
            # Project onto tangent space (sum delta = 0, a + delta >= 0)
            grad_centered = grad - np.mean(grad)  # approximate
            step_size = 0.01
            for ss in [0.1, 0.05, 0.01, 0.005, 0.001]:
                candidate = project_simplex(a - ss * grad)
                val = fast_convolve(candidate).max()
                if val < M - 1e-14:
                    step_size = ss
                    break
            a_new = project_simplex(a - step_size * grad)
        else:
            # Multiple active peaks: solve for weights
            # Build matrix B where B[i, k_idx] = 2*a[active[k_idx] - support[i]] (if valid)
            B = np.zeros((m, p))
            for i, j in enumerate(support):
                for k_idx, k in enumerate(active):
                    idx = k - j
                    if 0 <= idx < n:
                        B[i, k_idx] = 2.0 * a[idx]
            
            # We want to find mu >= 0 such that B @ mu is constant (equal to lambda)
            # This is: B @ mu - lambda * 1 = 0
            # Augmented system: [B, -1; 1^T, 0] @ [mu; lambda] = [0; 0] (homogeneous)
            # But we also want mu >= 0. Use SVD for the null space.
            
            A_sys = np.zeros((m + 1, p + 1))
            A_sys[:m, :p] = B
            A_sys[:m, p] = -1.0
            A_sys[m, :m] = 1.0  # sum of mu is free (we normalize later)
            
            try:
                U, S_vals, Vt = np.linalg.svd(A_sys)
                sol = Vt[-1, :]
                mu_vals = np.maximum(sol[:p], 0)
                mu_sum = mu_vals.sum()
                if mu_sum < 1e-15:
                    mu_vals = np.ones(p)
                else:
                    mu_vals = mu_vals / mu_sum
            except:
                mu_vals = np.ones(p) / p
            
            grad = conv_gradient_multi(a, active, weights=mu_vals)
            
            # Newton step: minimize the smooth approximation
            # Use a small step with backtracking
            best_candidate = a
            best_val = M
            for ss in [0.1, 0.05, 0.02, 0.01, 0.005, 0.001, 0.0005]:
                candidate = project_simplex(a - ss * grad)
                val = fast_convolve(candidate).max()
                if val < best_val - 1e-14:
                    best_val = val
                    best_candidate = candidate
                    break
            a_new = best_candidate
        
        a = a_new
        new_obj = fast_convolve(a).max()
        
        if new_obj < best_obj - 1e-13:
            best_obj = new_obj
            best_a = a.copy()
        elif abs(new_obj - best_obj) < 1e-12:
            # Check if we're at a local optimum
            if it > 10:
                break
    
    return best_a, best_obj
