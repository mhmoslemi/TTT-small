- If `m != k-m` (i.e., `k != 2m`): `sg[m] = a[k-m] + a[k-m] = 2*a[k-m]`
- If `m == k-m` (i.e., `k == 2m`): `sg[m] = a[m]` (only counted once)

And if `k-m` is not in `[0, n-1]`: `sg[m] = 0`

Let me vectorize this:

```python
def compute_subgradient(a, k):
    """Subgradient of (a*a)[k] w.r.t. a."""
    n = len(a)
    sg = np.zeros(n)
    # For each j in valid range, sg[j] += a[k-j], and if k-j != j, sg[k-j] += a[j]
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    js = np.arange(j_start, j_end + 1)
    others = k - js
    sg[js] += a[others]
    mask = others != js
    sg[others[mask]] += a[js[mask]]
    return sg
```

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad_peaks(a, peaks):
    """Compute weighted subgradient for multiple peaks with softmax weighting."""
    n = len(a)
    conv = fast_convolve(a)
    peak_vals = conv[peaks]
    # Use softmax weighting (temperature based on gap)
    gap = peak_vals.max() - peak_vals.min()
    if gap < 1e-15:
        T = peak_vals.max() * 0.1 + 1e-10
    else:
        T = gap * 0.5
    scaled = peak_vals / (T + 1e-30)
    scaled -= scaled.max()
    weights = np.exp(scaled)
    weights /= weights.sum()

    sg = np.zeros(n)
    for i, k in enumerate(peaks):
        w = weights[i]
        if w < 1e-10:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        others = k - js
        sg[js] += w * a[others]
        mask = others != js
        sg[others[mask]] += w * a[js[mask]]
    return sg, weights


def mirror_step(a, sg, eta):
    """Mirror descent step: a_new proportional to a * exp(-eta * sg)."""
    log_ratio = -eta * sg
    log_ratio -= log_ratio.max()
    a_new = a * np.exp(log_ratio)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    a_new /= s
    return a_new


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

        f = f / s
    n = len(f)

        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        g = solve_lp_oracle(f, M * m_exp)
        if g is None:
            m_exp = min(m_exp * 1.3, 5.0)
            if m_exp > 4.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 7919 + 13)
                scale = 0.002 + 0.004 * min(1.0, no_improve / 100.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
                continue
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in np.linspace(0.05, 1.0, 15):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.005 * min(1.0, (no_improve - 60) / 100.0)

            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def coordinate_descent_polish(a0, time_limit):
    """Targeted coordinate descent: move mass from high-contribution to low-contribution positions."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stale = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Find the position that contributes most to the peak
        j_start = max(0, k_peak - n + 1)
        j_end = min(n - 1, k_peak)
        js = np.arange(j_start, j_end + 1)
        others = k_peak - js
        contributions = a[js] * a[others]
        j_max = js[np.argmax(contributions)]

        # Find position with least contribution to any peak (use the peak)
        outside = np.setdiff1d(np.arange(n), js)
        if len(outside) == 0:
            break

        # Also minimize impact on other lags
        # Use the minimum value outside as target
        j_min_cand = outside[np.argmin(a[outside])]

        improved = False
        for delta_frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
            delta = a[j_max] * delta_frac * 0.01
            if delta < 1e-10:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min_cand] += delta
            candidate = np.maximum(candidate, 0.0)
            cs = candidate.sum()
            if cs < 1e-15:
                continue
            candidate /= cs
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        if improved:
            stale = 0
        else:
            stale += 1
            if stale % 50 == 0:
                rng = np.random.default_rng(it * 4567 + 3)
                # Try multiple random pairs
                for _ in range(20):
                    i = rng.integers(0, n)
                    j = rng.integers(0, n)
                    if i == j:
                        continue
                    for delta in [0.001, 0.005, 0.01]:
                        candidate = a.copy()
                        candidate[i] -= delta * a[i]
                        candidate[j] += delta * a[i]
                        candidate = np.maximum(candidate, 0.0)
                        cs = candidate.sum()
                        if cs < 1e-15:
                            break
                        candidate /= cs
                        c = fast_convolve(candidate)
                        if c.max() < best_obj - 1e-15:
                            a = candidate
                            best_obj = c.max()
                            best_a = a.copy()
                            stale = 0
                            break
                    if stale == 0:
                        break
                if stale > 300:
                    # Larger perturbation
                    noise = 0.01 * rng.standard_normal(n)
                    a = project_simplex(a + noise)
                    c = fast_convolve(a)
                    if c.max() < best_obj:
                        best_obj = c.max()
                        best_a = a.copy()
                        stale = 0
                    elif stale > 500:
                        break

    return best_a, best_obj

def fourier_refine(a0, time_limit, n_fourier=8):
    """Optimize over low-frequency Fourier coefficients."""
    from scipy.optimize import minimize
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    M = min(n_fourier, n // 2)
    x = np.arange(n, dtype=np.float64)

    # Extract initial Fourier coefficients
    a_centered = a - 1.0 / n
    init_coeffs = np.zeros(2 * M)
    for m in range(1, M + 1):
        init_coeffs[2 * (m - 1)] = 2.0 * np.sum(a_centered * np.cos(2.0 * np.pi * m * x / n))
        init_coeffs[2 * (m - 1) + 1] = 2.0 * np.sum(a_centered * np.sin(2.0 * np.pi * m * x / n))

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    def fourier_to_seq(coeffs):
        base = np.ones(n) / n
        for m in range(1, M + 1):
            c = coeffs[2 * (m - 1)]
            d = coeffs[2 * (m - 1) + 1]
            base += (c * np.cos(2.0 * np.pi * m * x / n) +
                     d * np.sin(2.0 * np.pi * m * x / n)) / n
        base = np.maximum(base, 0.0)
        s = base.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return base / s

    def fourier_objective(coeffs):
        seq = fourier_to_seq(coeffs)
        conv = fast_convolve(seq)
        return conv.max()

    t0 = time.monotonic()
    n_restarts = max(2, int((time_limit // 25)) + 1)

    for r in range(n_restarts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        if r == 0:
            x0 = init_coeffs.copy()
        else:
            rng = np.random.default_rng(r * 12345 + 7)
            scale = 0.003 * (1 + r * 0.3)
            x0 = init_coeffs + rng.standard_normal(2 * M) * scale

        try:
            res = minimize(fourier_objective, x0, method='Nelder-Mead',
                           options={'maxiter': 5000, 'xatol': 1e-12, 'fatol': 1e-14,
                                    'maxfev': 30000})
            if res.fun < best_obj:
                best_obj = res.fun
                best_a = fourier_to_seq(res.x)
                print("    Fourier restart {}: obj={:.10f}".format(r, best_obj))
        except Exception:
            pass

    return best_a, best_obj

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.10f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384

    n_parent = len(parent)
    print("Starting optimization, n_parent={}".format(n_parent))

    # === Phase 1: Quick Multi-n Exploration (~250s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    # Try parent n and several nearby/alternative n values
    n_candidates = sorted(set([
        n_parent,
        n_parent - 32, n_parent + 32,
        n_parent - 64, n_parent + 64,
        512, 640, 768, 896, 1024,
        767, 770, 792, 800, 832,   # primes and other values
    ]))
    n_candidates = [n for n in n_candidates if 128 <= n <= 3000]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 10:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss > 1e-10:
                start = start / ss
            else:
                start = np.ones(n_try) / n_try

        t_alloc = min(remaining, 25)
        a_opt, _ = mirror_descent_optimize(start, time_limit=t_alloc,
                                           T_init=0.005, eta_init=0.03, top_k=8)
        try_and_update(a_opt, state, "md_n{}".format(n_try))

    print("  Phase 1 best: {:.10f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 2: Deep Optimization (~1100s) ===
    print("\n=== Phase 2: Deep Optimization ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])

        # Dual-guided optimization (main workhorse)
        remaining = deadline - time.monotonic() - 700
        if remaining > 20:
            t_alloc = min(remaining, 400)
            print("  Dual-guided n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = dual_guided_optimization(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "dual")

        # FW-LP for global improvement
        remaining = deadline - time.monotonic() - 500
        if remaining > 15:
            t_alloc = min(remaining, 200)
            print("  FW-LP n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = fw_lp_optimize(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "fw")

        # More dual-guided from current best
        remaining = deadline - time.monotonic() - 350
        if remaining > 15:
            t_alloc = min(remaining, 200)
            print("  Dual-guided round 2, t={:.0f}s".format(t_alloc))
            a_opt, _ = dual_guided_optimization(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "dual2")

        # Perturbed restarts with quick optimization
        for scale in [0.002, 0.005, 0.01, 0.02]:
            remaining = deadline - time.monotonic() - 200
            if remaining < 10:
                break
            f_pert = state[1] * np.exp(rng.standard_normal(n_best) * scale)
            f_pert = f_pert / f_pert.sum()
            t_alloc = min(remaining, 60)
            a_opt, _ = dual_guided_optimization(f_pert, time_limit=t_alloc)
            try_and_update(a_opt, "pert_{}".format(scale))

    print("  Phase 2 best: {:.10f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 3: Fourier + Coordinate Descent (~350s) ===
    print("\n=== Phase 3: Fourier + Coord Descent ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Fourier refinement
        remaining = deadline - time.monotonic() - 150
        if remaining > 15:
            t_alloc = min(remaining, 150)
            print("  Fourier refine, t={:.0f}s".format(t_alloc))
            a_opt, _ = fourier_refine(state[1], time_limit=t_alloc, n_fourier=10)
            try_and_update(a_opt, state, "fourier_10")

        remaining = deadline - time.monotonic() - 80
        if remaining > 10:
            t_alloc = min(remaining, 80)
            print("  Fourier refine (more modes), t={:.0f}s".format(t_alloc))
            a_opt, _ = fourier_refine(state[1], time_limit=t_alloc, n_fourier=16)
            try_and_update(a_opt, state, "fourier_16")

        # Coordinate descent polish
        remaining = deadline - time.monotonic() - 50
        if remaining > 10:
            t_alloc = min(remaining, 100)
            print("  Coord descent, t={:.0f}s".format(t_alloc))
            a_opt, _ = coordinate_descent_polish(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "coord")

    print("  Phase 3 best: {:.10f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 4: Final Polish (~200s) ===
    print("\n=== Phase 4: Final Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Very fine mirror descent
        remaining = deadline - time.monotonic() - 50
        if remaining > 5:
            t_alloc = min(remaining, 80)
            print("  Fine mirror descent, t={:.0f}s".format(t_alloc))
            a_opt, _ = mirror_descent_optimize(state[1], time_limit=t_alloc,
                                               T_init=0.001, eta_init=0.01, top_k=5)
            try_and_update(a_opt, state, "fine_md")

        # Another dual pass
        remaining = deadline - time.monotonic() - 30
        if remaining > 5:
            t_alloc = min(remaining, 60)
            print("  Final dual, t={:.0f}s".format(t_alloc))
            a_opt, _ = dual_guided_optimization(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "final_dual")

        # Tiny perturbations
        remaining = deadline - time.monotonic() - 10
        if remaining > 5:
            for attempt in range(10):
                remaining = deadline - time.monotonic() - 5
                if remaining < 3:
                    break
                noise = rng.standard_normal(len(state[1])) * 0.0003
                f_pert = state[1] * np.exp(noise)
                f_pert = f_pert / f_pert.sum()
                try_and_update(f_pert, state, "micro_pert_{}".format(attempt))
                if remaining < 10:
                    # Quick polish
                    a_opt, _ = dual_guided_optimization(f_pert, time_limit=min(remaining, 5))
                    try_and_update(a_opt, state, "micro_polish_{}".format(attempt))

        # Final coordinate descent
        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            t_alloc = min(remaining, 30)
            a_opt, _ = coordinate_descent_polish(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "final_coord")

    print("  Phase 4 best: {:.10f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(384) / 384
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.10f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
