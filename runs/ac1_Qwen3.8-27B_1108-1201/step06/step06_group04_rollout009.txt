        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = np.arange(js, je + 1)
        non_contrib = np.setdiff1d(np.arange(n), contrib)
        
        if len(contrib) == 0 or len(non_contrib) == 0:
            break
        
        improved = False
        
        # Try shifting from largest contributor to non-contributors
        idx_max = contrib[np.argmax(a[contrib])]
        val = a[idx_max]
        
        if val < 1e-10:
            break
        
        for frac in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3]:
            shift = val * frac
            if shift < 1e-12:
                continue
            
            delta = np.zeros(n)
            delta[idx_max] -= shift
            nc_vals = a[non_contrib]
            nc_sum = nc_vals.sum()
            if nc_sum > 1e-15:
                for j in non_contrib:
                    delta[j] += shift * nc_vals[j] / nc_sum
            else:
                for j in non_contrib:
                    delta[j] += shift / len(non_contrib)
            
            candidate = a + delta
            if use_sym:
                candidate = symmetrize(candidate)
            candidate = project_simplex(candidate)
            c = fast_convolve(candidate)
            
            if c.max() < best_obj - 1e-14:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try random perturbations
            if it % 50 == 0:
                rng = np.random.default_rng((it * 31 + k_peak) % (2**31))
                for _ in range(5):
                    delta = rng.standard_normal(n) * 0.001
                    if use_sym:
                        delta = symmetrize(delta)
                    candidate = project_simplex(a + delta)
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-14:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break
            
            if not improved and it > 200:
                break
    
    return best_a, best_obj
