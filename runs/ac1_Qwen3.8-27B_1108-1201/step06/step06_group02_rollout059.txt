"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2:
- KKT-guided iterative balancing (exploiting dual structure)
- Frank-Wolfe with LP oracle (banded constraint structure)
- Exponential (mirror) descent variant
- Smoothed gradient descent (softmax-weighted active peaks)
- Multi-scale exploration
- Targeted peak reduction for final polish
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def exp_proj(v):
    """Exponential mapping to simplex: exp(v)/sum(exp(v))."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(v), 1.0 / len(v))
    return e / s


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m].
    Equivalent to: result[j] = sum_m p[j+m] * a[m]
    """
    n = len(a)
    lp = len(p)
    result = np.zeros(n)
    for j in range(n):
        # sum_m p[j+m] * a[m] where 0 <= j+m < lp and 0 <= m < n
        lo = max(0, 0)
        hi = min(n, lp - j)
        if hi > lo:
            result[j] = np.dot(p[j:j+hi], a[lo:hi])
    return result


def hankel_mv_fast(p, a):
    """Fast version using FFT.
    result[j] = sum_m p[j+m] * a[m] for m=0..n-1 where j+m < len(p).
    """
    n = len(a)
    lp = len(p)
    size = n + lp - 1
    fft_size = 1 << (size - 1).bit_length()
    # We want cross-correlation: result[j] = sum_m a[m] * p[j+m]
    # This is np.correlate(p, a, 'valid') if len(p) >= len(a)
    if lp >= n:
        return np.correlate(p, a, mode='valid')[:n]
    else:
        # Pad
        return np.correlate(p, np.zeros(n), mode='full')[n-1:n-1+n]  # fallback


def solve_lp_oracle(f, M, time_limit_lp=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns g (normalized to sum=1) or None.
    """
    n = len(f)
    c = -np.ones(n)
    
    # Build constraint matrix (banded)
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    # For column j: A[j:j+n, j] = f[0:n]
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit_lp})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def kkt_balance(a0, time_limit, init_temp=0.1):
    """KKT-guided iterative balancing.
    
    Exploits the KKT condition: at optimal a, the weighted autocorrelation
    (y * a) should be constant on the support of a, where y is the dual
    distribution over active peaks.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    lr = 0.1
    temp = init_temp
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        gmax = conv.max()
        
        # Dual: softmax distribution over peaks
        scaled = conv / temp
        scaled -= scaled.max()
        y = np.exp(scaled)
        y_s = y.sum()
        if y_s < 1e-300:
            y = np.zeros(2 * n - 1)
            y[np.argmax(conv)] = 1.0
        else:
            y = y / y_s
        
        # Compute w[i] = sum_k y[k] * a[k-i] = correlate(y, a, 'valid')[i]
        w = np.correlate(y, a, mode='valid')[:n]
        
        # KKT: w should be constant on support of a
        support = a > 1e-12
        if support.sum() > 0:
            w_target = w[support].mean()
            # Gradient direction: where w > target, decrease a; where w < target, increase
            grad = np.zeros(n)
            grad[support] = 2.0 * (w[support] - w_target)
            
            # Update
            alpha = lr  # step size
            a_new = a - alpha * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            
            # Check improvement
            conv_new = fast_convolve(a_new)
            obj_new = conv_new.max()
            
            if obj_new < best_obj - 1e-15:
                a = a_new
                best_obj = obj_new
                best_a = a.copy()
                no_improve = 0
                # Decrease temperature to focus more
                temp *= 0.99
            elif obj_new < gmax - 1e-15:
                a = a_new
                no_improve += 1
            else:
                # Try smaller step
                a_new2 = a - (alpha * 0.5) * grad
                a_new2 = np.maximum(a_new2, 0.0)
                a_new2 = a_new2 / a_new2.sum()
                conv_new2 = fast_convolve(a_new2)
                if conv_new2.max() < gmax - 1e-15:
                    a = a_new2
                    no_improve += 1
                else:
                    no_improve += 1
                    lr *= 0.95
                    temp *= 1.02
            
            if no_improve > 50:
                # Perturb
                rng = np.random.default_rng((it * 31 + 7) % (2**31))
                noise = 0.001 + 0.002 * min(1.0, (no_improve - 50) / 100.0)
                a = project_simplex(best_a + rng.standard_normal(n) * noise)
                conv = fast_convolve(a)
                if conv.max() < best_obj:
                    best_obj = conv.max()
                    best_a = a.copy()
                lr = 0.1
                temp = init_temp
                no_improve = 0
        
        # Decay learning rate slowly
        lr *= 0.999
        temp = max(0.001, min(1.0, temp))
    
    return best_a, best_obj


def mirror_descent(a0, time_limit, T_init=0.01, lr_init=0.05):
    """Exponential (mirror) descent for the smooth approximation."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        
        # Softmax weights on peaks
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Gradient: 2 * (p-weighted autocorrelation)
        # grad[i] = 2 * sum_k p[k] * a[k-i]
        grad = 2.0 * np.correlate(p, a, mode='valid')[:n]
        
        # Mirror step: a_new = exp(a - lr*grad) / sum(exp(a - lr*grad))
        # But we need to be careful: a is already on the simplex
        # Use: a_new = exp(log(a) - lr*grad) / sum(...) = a * exp(-lr*grad) / sum(...)
        # This is the multiplicative weight update!
        log_a = np.where(a > 0, np.log(a + 1e-300), -100.0)
        a_new = np.exp(log_a - lr * grad)
        a_new = a_new / a_new.sum()
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        elif obj_new < conv.max() - 1e-14:
            a = a_new
            no_improve += 1
        else:
            # Backtrack
            lr *= 0.7
            a_new2 = np.exp(log_a - lr * grad)
            a_new2 = a_new2 / a_new2.sum()
            conv_new2 = fast_convolve(a_new2)
            if conv_new2.max() < conv.max() - 1e-14:
                a = a_new2
                no_improve += 1
            else:
                no_improve += 1
        
        if no_improve > 30:
            T *= 1.05
        if no_improve > 100:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.002 * min(1.0, (no_improve - 100) / 200.0) + 0.001
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break
        
        # Adaptive M expansion
        g = solve_lp_oracle(f, M * m_exp, time_limit_lp=min(15, remaining * 0.1))
        if g is None:
            m_exp = min(m_exp * 1.5, 10.0)
            if m_exp > 8.0:
                m_exp = 1.0
            continue
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 7919 + 13)
                scale = 0.001 + 0.003 * min(1.0, no_improve / 100.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
                continue
            m_exp = min(m_exp * 1.1, 5.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        
        for alpha in np.linspace(0.01, 1.0, 15):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Golden section refinement
        if 0.02 < best_alpha < 0.98:
            lo = max(0.0, best_alpha - 0.04)
            hi = min(1.0, best_alpha + 0.04)
            for _ in range(10):
                mid = 0.618 * lo + 0.382 * hi
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    hi = mid
                else:
                    lo = mid
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.2, 10.0)
        
        if no_improve > 60:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            scale = 0.002 + 0.004 * min(1.0, (no_improve - 60) / 150.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj


def smoothed_grad_descent(a0, time_limit, T_init=0.001, lr_init=0.03):
    """Smoothed gradient descent with Euclidean projection."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        gmax = conv.max()
        
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        grad = 2.0 * np.correlate(p, a, mode='valid')[:n]
        
        best_candidate = a
        best_val = gmax
        
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
        
        a = best_candidate
        obj = best_val
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 20:
            T *= 1.05
            lr *= 0.98
        if no_improve > 50:
            T = max(0.0001, T * 0.8)
        if no_improve > 150:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.002 + 0.004 * min(1.0, (no_improve - 150) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = T_init
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj


def targeted_peak_reduction(a0, time_limit):
    """Reduce the highest peak by targeted coordinate transfers."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    stale = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        
        if len(affected) == 0:
            break
        
        # Find the pair that contributes most to the peak
        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]
        
        # Find where to move mass to
        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            break
        j_min_cand = outside[np.argmin(a[outside])]
        
        improved = False
        for delta in [0.0003, 0.0005, 0.001, 0.002, 0.005, 0.01]:
            if a[j_max] < delta * 1.1:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min_cand] += delta
            candidate = np.maximum(candidate, 0.0)
            ss = candidate.sum()
            if ss < 1e-15:
                continue
            candidate = candidate / ss
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            stale = 0
        else:
            stale += 1
            if stale % 20 == 0:
                rng = np.random.default_rng(it * 4567 + 3)
                # Try random pair swap
                idx = rng.choice(n, size=2, replace=False)
                amt = min(a[idx[0]], a[idx[1]]) * 0.1
                candidate = a.copy()
                candidate[idx[0]] -= amt
                candidate[idx[1]] += amt
                candidate = np.maximum(candidate, 0.0)
                candidate = candidate / candidate.sum()
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    stale = 0
                elif stale > 300:
                    break
    
    return best_a, best_obj


def multi_peak_subgrad(a0, time_limit, top_k=20, lr_init=0.003, mom=0.93):
    """Multi-peak subgradient with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        
        # Select top peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            # Weight by value
            weights = pv / pv.sum()
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]
        
        v = mom * v - lr * sg
        a_new = project_simplex(a + v)
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            # Still update if it reduces current obj
            if obj < conv.max() - 1e-14:
                a = a_new
        
        lr *= 0.9995
        mom = min(0.97, mom + 0.00005)
        
        if no_improve > 100:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.001 + 0.003 * min(1.0, (no_improve - 100) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init * 0.7
            no_improve = 0
    
    return best_a, best_obj


def upsample(a, new_n):
    """Upsample sequence to new length using linear interpolation."""
    old_n = len(a)
    if new_n == old_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def try_and_update(a, state, label):
    """Evaluate and update best state."""

    s = a.sum()
    if s < 1e-15:

    score = float(evaluate_sequence(a_norm.tolist()))

        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a)))
