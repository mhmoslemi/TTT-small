def directed_line_search(a, d, n):
    """Find best step size alpha for direction d from point a on simplex."""
    # d should satisfy sum(d) = 0
    # Find alpha_max
    neg = d < -1e-15
    if neg.any():
        alpha_max = np.min(-a[neg] / d[neg]) - 1e-12
    else:
        alpha_max = 1.0
    
    alpha_max = max(alpha_max, 0.0)
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, alpha_max
    
    def f(alpha):
        x = a + alpha * d
        x = np.maximum(x, 0)  # safety
        c = fast_convolve(x)
        return c.max()
    
    # Evaluate at a few points first to narrow down
    alphas = np.linspace(0, alpha_max, 20)
    vals = [f(al) for al in alphas]
    best_idx = np.argmin(vals)
    
    # Narrow to neighborhood of best
    if best_idx == 0:
        lo, hi = 0, alphas[1]
    elif best_idx == len(alphas) - 1:
        lo, hi = alphas[-2], alphas[-1]
    else:
        lo, hi = alphas[max(0, best_idx-2):min(len(alphas)-1, best_idx+2)]
        lo, hi = lo[0], hi[-1]
    
    # Golden section
    c = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc, fd = f(c), f(d_pt)
    
    for _ in range(30):
        if fc < fd:
            hi = d_pt
            d_pt = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = f(d_pt)
    
    best_alpha = (lo + hi) / 2
    return best_alpha, f(best_alpha)

def solve_qp_cvxpy(w, n, a_warm):
    """Solve min a^T W a s.t. sum(a)=1, a>=0 using CVXPY."""
    a = cp.Variable(n)
    # Build W as a Hankel matrix - but for CVXPY, we need to express a^T W a
    # Since W is Hankel with W[i,j] = w[i+j], we can write:
    # a^T W a = sum_i sum_j a[i] * w[i+j] * a[j]
    # This is a quadratic form. In CVXPY, we can use cp.quad_form(a, W)
    # But building W explicitly for large n is expensive.
    
    # Alternative: use the fact that a^T W a = sum_k w[k] * (sum_{i+j=k} a[i]*a[j])
    # = sum_k w[k] * (a*a)[k]
    # But in CVXPY, we can't directly express this without building W.
    
    # For n up to ~1024, building W (n x n) is fine (1M elements).
    W = np.array([[w[i+j] for j in range(n)] for i in range(n)])
    # Ensure PSD (add small regularization)
    W = (W + W.T) / 2 + 1e-12 * np.eye(n)
    
    objective = cp.Minimize(cp.quad_form(a, W))
    constraints = [cp.sum(a) == 1, a >= 0]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS)  # or ECOS, MOSEK, etc.
    
    if a.value is not None:
        return np.maximum(a.value, 0)
    return a_warm

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- QP primal-dual initialization with active-set method
- Riemannian gradient descent with adaptive line search on the simplex
- Multi-peak subgradient with height-weighted combination
- Momentum with periodic restarts for escaping local minima
- Fine polish phase with single-peak targeting and very small steps

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(w, a):
    """Compute W @ a where W_{i,j} = w[i+j]. w has length 2n-1, a has length n."""
    return np.correlate(w, a, mode='valid')


def solve_qp_active_set(w, a_init, n, max_iters=300, time_limit=5.0):
    """
    Solve min a^T W a s.t. sum(a)=1, a>=0 using active set method.
    W is Hankel: W_{i,j} = w[i+j] where w has length 2n-1.
    """
    t0 = time.monotonic()
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    active = set(np.where(a > 1e-10)[0])
    if not active:
        active = set(range(min(n, 20)))

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if len(active) == 0:
            return np.ones(n) / n
        if len(active) == 1:
            a = np.zeros(n)
            a[list(active)[0]] = 1.0
            return a

        act = np.array(sorted(active))
        k = len(act)

        # Build submatrix W_A
        W_AA = np.array([[w[act[i] + act[j]] for j in range(k)] for i in range(k)])
        W_AA_reg = W_AA + 1e-12 * np.eye(k)

        try:
            x = np.linalg.solve(W_AA_reg, np.ones(k))
        except np.linalg.LinAlgError:
            try:
                x = np.linalg.lstsq(W_AA_reg, np.ones(k), rcond=None)[0]
            except Exception:
                x = np.ones(k) / k

        x_sum = x.sum()
        if abs(x_sum) < 1e-14:
            x = np.ones(k) / k
            x_sum = 1.0

        a_act = x / x_sum

        # Remove most negative component
        min_idx_local = np.argmin(a_act)
        if a_act[min_idx_local] < -1e-12:
            active.discard(act[min_idx_local])
            a_new = np.zeros(n)
            for i in range(k):
                if i != min_idx_local:
                    a_new[act[i]] = max(0.0, a_act[i])
            s2 = a_new.sum()
            if s2 > 1e-15:
                a = a_new / s2
            continue

        # Form full solution
        a = np.zeros(n)
        for i in range(k):
            a[act[i]] = max(0.0, a_act[i])

        # Compute KKT multipliers
        Wa = hankel_mv(w, a)
        grad = 2.0 * Wa
        mu = np.mean([grad[idx] for idx in act])

        # Check KKT for inactive
        inactive_mask = np.ones(n, dtype=bool)
        inactive_mask[act] = False
        inactive = np.where(inactive_mask)[0]
        if len(inactive) > 0:
            violations = grad[inactive] - mu
            min_viol_idx = inactive[np.argmin(violations)]
            if violations.min() < -1e-10:
                active.add(min_viol_idx)
                continue

        return a

    return a


def qp_primal_dual(a0, time_limit, T_init=0.01, use_topk=False, topk=15,
                   blend=0.5, seed=0):
    """Primal-dual QP iteration."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        conv = fast_convolve(a)
        cmax = conv.max()

        if use_topk:
            k_use = min(topk, len(conv))
            peaks = np.argpartition(conv, -k_use)[-k_use:]
            w = np.zeros(len(conv))
            w[peaks] = 1.0 / k_use
        else:
            scaled = conv / T
            scaled -= scaled.max()
            w = np.exp(scaled)
            w = w / w.sum()

        qp_time = min(3.0, remaining * 0.3)
        a_qp = solve_qp_active_set(w, a, n, max_iters=150, time_limit=qp_time)

        a_new = (1.0 - blend) * a + blend * a_qp
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a_qp
        else:
            a_new = a_new / s_new

        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 15:
            T *= 1.2
        if no_improve > 40:
            T *= 0.7
        if no_improve > 100:
            rng = np.random.default_rng((it * 104729 + seed) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            no_improve = 0

    return best_a, best_obj


def gradient_line_search(a0, time_limit, K=3, lr_scale=1.0, n_grid=30,
                         momentum=0.0, seed=0, verbose=False):
    """
    Gradient descent with line search targeting top-K peaks.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 0.5 * lr_scale

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        # Find top K peaks
        if K >= len(conv):
            peak_indices = np.arange(len(conv))
            peak_weights = np.ones(len(peak_indices))
        else:
            peak_indices = np.argpartition(conv, -K)[-K:]
            peak_weights = conv[peak_indices].astype(np.float64)
            peak_weights = peak_weights / peak_weights.sum()

        # Compute weighted gradient
        grad = np.zeros(n)
        for ki, k in enumerate(peak_indices):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * peak_weights[ki] * a[k - idx]

        # Project to tangent space (sum = 0)
        grad = grad - grad.mean()

        # Add momentum
        if momentum > 0:
            v = momentum * v + (1 - momentum) * grad
            direction = -v
        else:
            direction = -grad

        # Rescale direction
        d_norm = np.linalg.norm(direction)
        if d_norm < 1e-15:
            break
        direction = direction / d_norm * lr

        # Find max valid step (maintain non-negativity)
        neg_mask = direction < -1e-15
        if neg_mask.any():
            alpha_max = np.min(-a[neg_mask] / direction[neg_mask])
        else:
            alpha_max = 1.0
        alpha_max = max(alpha_max * 0.999, 0.0)

        if alpha_max < 1e-15:
            # Try a smaller direction
            direction = -grad / d_norm * lr * 0.1
            neg_mask = direction < -1e-15
            if neg_mask.any():
                alpha_max = np.min(-a[neg_mask] / direction[neg_mask])
            else:
                alpha_max = 1.0
            alpha_max = max(alpha_max * 0.999, 0.0)
            if alpha_max < 1e-15:
                break

        # Grid search for best alpha
        alphas = np.linspace(0, alpha_max, n_grid)
        best_val = conv.max()
        best_alpha = 0.0
        for al in alphas:
            x = a + al * direction
            x = np.maximum(x, 0)
            val = fast_convolve(x).max()
            if val < best_val:
                best_val = val
                best_alpha = al

        if best_alpha == 0:
            lr *= 0.7
            continue

        a = a + best_alpha * direction
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.95

        # Adaptive K
        if no_improve > 50:
            K = min(K + 1, 10)
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + seed) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = 0.5 * lr_scale
            K = 3
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    grad_ls it={it}, obj={best_obj:.10f}, alpha={best_alpha:.6f}, no_imp={no_improve}")

    return best_a, best_obj


def fine_polish(a0, time_limit, n_grid=50, max_steps=500, seed=0):
    """
    Fine polish: single peak targeting with very precise line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_steps:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Gradient at the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad = np.zeros(n)
        grad[idx] = 2.0 * a[k_peak - idx]

        # Project to tangent space
        grad = grad - grad.mean()
        d_norm = np.linalg.norm(grad)
        if d_norm < 1e-15:
            break

        direction = -grad / d_norm

        # Try multiple step sizes
        neg_mask = direction < -1e-15
        if neg_mask.any():
            alpha_max = np.min(-a[neg_mask] / direction[neg_mask]) * 0.999
        else:
            alpha_max = 1.0

        if alpha_max < 1e-15:
            # Perturb and retry
            rng = np.random.default_rng((it * 7 + seed) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.0005)
            continue

        # Dense grid
        alphas = np.linspace(0, alpha_max, n_grid)
        best_val = conv.max()
        best_alpha = 0.0
        for al in alphas:
            x = np.maximum(a + al * direction, 0)
            val = fast_convolve(x).max()
            if val < best_val:
                best_val = val
                best_alpha = al

        if best_alpha < 1e-10:
            # No improvement possible in this direction, try perturbation
            if it % 10 == 0:
                rng = np.random.default_rng((it * 13 + seed) % (2**31))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0003)
            continue

        a = np.maximum(a + best_alpha * direction, 0)
        a = a / a.sum()

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def coordinate_descent(a0, time_limit, seed=0, verbose=False):
    """
    Coordinate descent: for each coordinate, find optimal value given others fixed.
    Uses the structure of the auto-convolution for efficient updates.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    conv = fast_convolve(a)

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            # When we change a[i] by delta, the sum changes by delta.
            # To maintain sum=1, we need to scale the rest, or adjust another coord.
            # Easier: parameterize a[i] = t, and scale all other coords to sum to 1-t.
            # This is complex. Instead, do a 2-coordinate swap.
            
            # Find the peak
            k_peak = int(np.argmax(conv))

            # Does coordinate i contribute to the peak?
            if not (max(0, k_peak - n + 1) <= i <= min(n - 1, k_peak)):
                continue

            # Try shifting mass from i to a non-contributing coordinate
            # Find best non-contributing coordinate
            non_contrib = [j for j in range(n)
                          if not (max(0, k_peak - n + 1) <= j <= min(n - 1, k_peak))]
            if not non_contrib:
                continue

            j = non_contrib[i % len(non_contrib)]

            # Try several fractions
            for frac in [0.005, 0.01, 0.02, 0.05]:
                delta = a[i] * frac
                if delta < 1e-12:
                    continue
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < 0:
                    continue
                c = fast_convolve(a_try)
                if c.max() < best_obj - 1e-14:
                    a = a_try
                    conv = c
                    best_obj = conv.max()
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and it > 200:
            # Perturb to escape
            rng = np.random.default_rng((it * 17 + seed) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            conv = fast_convolve(a)

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge / edge.sum())
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())
    # Two peaks
    g2 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())
    # Cosine
    c1 = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    c1 = np.maximum(c1, 0.01)
    inits.append(c1 / c1.sum())
    # Symmetric triangle
    tri = 1.0 - 2.0 * np.abs(x - (n - 1) / 2.0) / (n - 1)
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())

    return inits

    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")

    # === Phase 1: Exploration with multiple n and initializations (~250s) ===
    print("\n=== Phase 1: Exploration ===")
    t1 = time.monotonic()

    candidate_ns = sorted(set([
        n_parent,
        max(256, n_parent - 128),
        min(1536, n_parent + 128),
        384, 512, 768, 1024,
    ]))

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 900
        if remaining < 30:
            break

        # Interpolate parent to new size
        if n_try == n_parent and parent is not None:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            start = start / start.sum()

        t_alloc = min(remaining, 35)
        print(f"  QP n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = qp_primal_dual(start, time_limit=t_alloc, T_init=0.01,
                                   use_topk=False, topk=15, blend=0.5, seed=seed)
        try_seq(a_opt, f"qp_n{n_try}")

        remaining = deadline - time.monotonic() - 850
        if remaining > 15:
            a_opt2, _ = qp_primal_dual(a_opt, time_limit=min(remaining, 20),
                                        T_init=0.005, use_topk=True, topk=8,
                                        blend=0.6, seed=seed + 1)
            try_seq(a_opt2, f"qp_topk_n{n_try}")

    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")

    # === Phase 2: Deep gradient line search (~900s) ===
    print("\n=== Phase 2: Gradient Line Search ===")
    t2 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple configurations
        configs = [
            {'K': 1, 'lr_scale': 1.0, 'momentum': 0.0, 't_frac': 0.25},
            {'K': 3, 'lr_scale': 0.8, 'momentum': 0.3, 't_frac': 0.2},
            {'K': 5, 'lr_scale': 0.6, 'momentum': 0.5, 't_frac': 0.2},
            {'K': 2, 'lr_scale': 1.2, 'momentum': 0.2, 't_frac': 0.15},
            {'K': 1, 'lr_scale': 0.5, 'momentum': 0.7, 't_frac': 0.15},
        ]

        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 300
            if remaining < 20:
                break
            t_alloc = min(remaining, int(budget_s * cfg['t_frac']))
            if t_alloc < 15:
                continue
            print(f"  GradLS K={cfg['K']}, lr={cfg['lr_scale']}, "
                  f"mom={cfg['momentum']}, t={t_alloc:.0f}s")
            a_opt, _ = gradient_line_search(best_seq, time_limit=t_alloc,
                                             K=cfg['K'], lr_scale=cfg['lr_scale'],
                                             momentum=cfg['momentum'], seed=seed + ci)
            try_seq(a_opt, f"gradls_{ci}")

    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")

    # === Phase 3: QP refinement on best (~300s) ===
    print("\n=== Phase 3: QP Refinement ===")
    t3 = time.monotonic()

    if best_seq is not None:
        qp_configs = [
            {'T_init': 0.003, 'use_topk': False, 'topk': 10, 'blend': 0.4},
            {'T_init': 0.001, 'use_topk': True, 'topk': 5, 'blend': 0.3},
            {'T_init': 0.01, 'use_topk': True, 'topk': 8, 'blend': 0.5},
            {'T_init': 0.005, 'use_topk': False, 'topk': 15, 'blend': 0.6},
        ]

        for ci, cfg in enumerate(qp_configs):
            remaining = deadline - time.monotonic() - 150
            if remaining < 15:
                break
            t_alloc = min(remaining, 60)
            print(f"  QP refine cfg{ci}, t={t_alloc:.0f}s")
            a_opt, _ = qp_primal_dual(best_seq, time_limit=t_alloc, seed=seed + 100 + ci, **cfg)
            try_seq(a_opt, f"qp_refine_{ci}")

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")

    # === Phase 4: Fine polish (~150s) ===
    print("\n=== Phase 4: Fine Polish ===")
    t4 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 50
        if remaining > 10:
            t_alloc = min(remaining, 80)
            print(f"  Fine polish, t={t_alloc:.0f}s")
            a_opt, _ = fine_polish(best_seq, time_limit=t_alloc,
                                    n_grid=40, max_steps=1000, seed=seed)
            try_seq(a_opt, "fine_polish")

        remaining = deadline - time.monotonic() - 30
        if remaining > 10:
            t_alloc = min(remaining, 40)
            print(f"  Coordinate descent, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc, seed=seed)
            try_seq(a_opt, "coord_descent")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # === Phase 5: Final micro-optimations (~50s) ===
    print("\n=== Phase 5: Final Micro ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        try_seq_fine_configs = [
            {'K': 1, 'lr_scale': 0.3, 'momentum': 0.0},
            {'K': 1, 'lr_scale': 0.1, 'momentum': 0.0},
            {'K': 2, 'lr_scale': 0.2, 'momentum': 0.5},
        ]

        for ci, cfg in enumerate(try_seq_fine_configs):
            remaining = deadline - time.monotonic() - 10
            if remaining < 5:
                break
            t_alloc = min(remaining, 15)
            a_opt, _ = gradient_line_search(best_seq, time_limit=t_alloc,
                                             K=cfg['K'], lr_scale=cfg['lr_scale'],
                                             momentum=cfg['momentum'],
                                             n_grid=60, seed=seed + 200 + ci)
            try_seq(a_opt, f"micro_{ci}")

        # Tiny random perturbations
        for _ in range(3):
            remaining = deadline - time.monotonic() - 5
            if remaining < 3:
                break
            noise = 0.0003
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise)
            t_alloc = min(remaining, 8)
            a_opt, _ = fine_polish(f_pert, time_limit=t_alloc, n_grid=30,
                                    max_steps=200, seed=seed + _ + 300)
            try_seq(a_opt, f"perturb_{_}")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
