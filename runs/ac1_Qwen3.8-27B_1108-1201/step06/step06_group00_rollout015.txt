"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-smoothed projected gradient descent with backtracking line search
2. Adaptive beta schedule (increasing smoothness parameter)
3. Coordinate descent polishing
4. Multi-length exploration
5. Peak-aware random restarts

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex_sum(v, s):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - s)
    if not cond.any():
        return np.full(n, s / n)
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - s) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_score(a):
    """Compute the actual score."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    C = fast_convolve(a)
    return 2.0 * n * C.max() / (S * S)

def lse_projected_gradient(a, time_limit, beta_start=10.0, beta_end=20000.0, top_k=80, verbose=True):
    """Main optimization: LSE-smoothed projected gradient."""
    a = np.maximum(a.copy(), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / S
    
    n = len(a)
    C = fast_convolve(a)
    best_obj = C.max()
    best_a = a.copy()
    
    beta = beta_start
    lr = 1.0 / np.sqrt(n) * 2
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    momentum = 0.9
    v_mom = np.zeros(n)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 200 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"      it={it}, beta={beta:.1f}, best={best_obj:.8f}, lr={lr:.5f}")
        
        C = fast_convolve(a)
        k_top = min(top_k, 2*n-1)
        peak_idx = np.argpartition(C, -k_top)[-k_top:]
        
        # LSE weights
        C_peaks = C[peak_idx]
        bc = beta * C_peaks
        bc_max = bc.max()
        w = np.exp(bc - bc_max)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.ones(len(peak_idx)) / len(peak_idx)
        else:
            w /= w_sum
        
        # Gradient
        grad = np.zeros(n)
        for idx in range(len(peak_idx)):
            wi = w[idx]
            if wi < 1e-15:
                continue
            k = int(peak_idx[idx])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            i_range = np.arange(lo, hi + 1)
            grad[i_range] += wi * 2.0 * a[k - i_range]
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        # Nesterov-style momentum
        v_mom = momentum * v_mom + grad / gn
        direction = v_mom
        
        dn = np.linalg.norm(direction)
        if dn < 1e-15:
            direction = grad / gn
            dn = 1.0
        
        # Backtracking line search
        step = lr
        improved = False
        best_trial_obj = best_obj
        best_trial_a = a.copy()
        
        for _ in range(10):
            a_new = project_simplex_sum(a - step * direction / dn, 1.0)
            C_new = fast_convolve(a_new)
            obj_new = C_new.max()
            if obj_new < best_trial_obj:
                best_trial_obj = obj_new
                best_trial_a = a_new
            if obj_new < best_obj * (1.0 - 1e-12):
                improved = True
                break
            step *= 0.5
        
        if improved:
            a = best_trial_a
            if best_trial_obj < best_obj - 1e-14:
                best_obj = best_trial_obj
                best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.01, 10.0 / np.sqrt(n))
        else:
            # Accept the best trial even if not better than global best
            a = best_trial_a
            no_improve += 1
            if no_improve >= 20:
                lr *= 0.9
            if no_improve > 100:
                rng = np.random.default_rng(it * 13 + 7)
                a = project_simplex_sum(best_a + rng.standard_normal(n) * (0.005 + 0.002 * min(no_improve/100, 1.0)), 1.0)
                v_mom = np.zeros(n)
                lr = 1.0 / np.sqrt(n) * 2
                no_improve = 0
                beta = min(beta * 5.0, beta_end)
        
        # Increase beta slowly
        if beta < beta_end:
            beta *= 1.002
    
    return best_a, 2.0 * n * best_obj
