Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_gradient_descent(a0, time_limit, top_k=15, T_init=50.0):
    """

    no_improve = 0

        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_conv(a)
        max_val = conv.max()
        
        # Softmax weights over peaks
        K = min(top_k, len(conv))
        peaks = np.argpartition(conv, -K)[-K:]
        pv = conv[peaks]
        weights = softmax(pv * T)
        
        # Subgradient: weighted average of peak gradients
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            w = weights[idx]
            if w < 1e-12:
                continue

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                # d(c[k])/d(a[j]) = 2*a[k-j] (if 0 <= k-j < n)
                grad[j] += w * 2.0 * a[k - j]
        
        # Projected gradient step on simplex
        a_new = project_simplex(a - lr * grad)
        
        # Line search
        best_candidate = a
        best_val = max_val

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Active-set SCA (top-K LP constraints) for fast iteration
- Nesterov-accelerated softmax subgradient descent
- Peak-directed redistribution when stuck
- Multi-n exploration with adaptive time allocation
- Coordinate descent for final polishing

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.special import softmax
import time


def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, peaks, weights):
    """Compute weighted subgradient of max(c) at peaks with weights."""
    n = len(a)
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        w = weights[idx]
        if w < 1e-14:
            continue
        k = int(peaks[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad[j] += w * 2.0 * a[k - j]
    return grad

def nesterov_softmax_gd(a0, time_limit, top_k=20, T=60.0):
    """
    Nesterov-accelerated gradient descent using smoothed max (softmax).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    a_prev = a.copy()
    a_mom = a.copy()
    best_a = a.copy()
    best_obj = fast_conv(a).max()

    t0 = time.monotonic()
    lr = 0.008
    beta = 0.9
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # Evaluate gradient at momentum point
        a_mom = np.maximum(a_mom, 0.0)
        a_mom = a_mom / a_mom.sum()
        conv = fast_conv(a_mom)
        max_val = conv.max()

        K = min(top_k, len(conv))
        peaks = np.argpartition(conv, -K)[-K:]
        pv = conv[peaks]
        weights = softmax(pv * T)

        grad = compute_subgradient(a_mom, conv, peaks, weights)

        # Nesterov step
        a_new_stencil = a_mom - lr * grad
        a_new_stencil = project_simplex(a_new_stencil)
        a_mom = a_new_stencil + beta * (a_new_stencil - a)
        a_mom = project_simplex(a_mom)
        a = a_new_stencil

        obj = fast_conv(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            lr *= 0.9
        if lr < 0.0005:
            lr = 0.008
            beta = 0.9
            no_improve = 0

        if no_improve > 150:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            a_mom = a.copy()
            a_prev = a.copy()
            lr = 0.008
            beta = 0.9
            no_improve = 0

    return best_a, best_obj

        lo, hi = 0.01, 0.3
