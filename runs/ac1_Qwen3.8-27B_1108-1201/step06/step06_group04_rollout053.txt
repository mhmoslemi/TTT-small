def hankel_mv(w, a):
    """Compute H @ a where H[i,j] = w[i+j]."""
    n = len(a)
    # (H a)[i] = sum_j w[i+j] * a[j]
    # This is a valid correlation
    return np.correlate(w, a, mode='valid')

def hankel_mv_transpose(w, b):
    """Compute H^T @ b where H[i,j] = w[i+j]. Note H is symmetric if w is symmetric."""
    # For a Hankel matrix, H = H^T, so this is the same
    return np.correlate(w, b, mode='valid')

def solve_eq_qp(w, n, a_init=None, max_cg_iters=100):
    """Solve min a^T H a s.t. 1^T a = 1 using CG.
    H[i,j] = w[i+j]
    Returns the unconstrained (equality-only) solution.
    """
    # Solution: a* = H^{-1} 1 / (1^T H^{-1} 1)
    # Use CG to solve H x = 1
    b = np.ones(n)
    x = a_init.copy() if a_init is not None else np.ones(n)/n
    
    # CG iteration
    r = b - hankel_mv(w, x)
    p = r.copy()
    rs_old = r @ r
    
    for k in range(max_cg_iters):
        Ap = hankel_mv(w, p)
        pAp = p @ Ap
        if pAp < 1e-15:
            break
        alpha = rs_old / pAp
        x += alpha * p
        r -= alpha * Ap
        rs_new = r @ r
        if rs_new < 1e-20 * rs_old:
            break
        p = r + (rs_new / rs_old) * p
        rs_old = rs_new
    
    # Normalize
    s = x.sum()
    if abs(s) < 1e-15:
        return np.ones(n)/n
    a = x / s
    return a

import numpy as np
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(w, a):
    """Compute H @ a where H[i,j] = w[i+j]. w has length >= 2n-1, a has length n."""
    return np.correlate(w, a, mode='valid')


def solve_sdp_autocorr(n, solver='CLARABEL', time_limit=180.0):
    """
    Solve SDP relaxation:
      min t
      s.t. sum_{i+j=k} X[i,j] <= t for all k
           sum X[i,j] = 1
           X >= 0 (PSD), X[i,j] >= 0
    Returns (a, t_value) where a is the extracted solution.
    """
    t0 = time.monotonic()
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    ones = np.ones(n)

    constraints = [X >> 0, X >= 0, ones @ X @ ones == 1]

    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        terms = [X[i, k - i] for i in range(i_min, i_max + 1)]
        constraints.append(cp.sum(terms) <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver='MOSEK')
        except Exception:
            return None, None

    if time.monotonic() - t0 > time_limit:
        return None, None

    if X.value is None:
        return None, None

    X_val = (X.value + X.value.T) / 2.0
    eigvals, eigvecs = np.linalg.eigh(X_val)
    a = np.abs(eigvecs[:, -1])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    t_val = float(t.value) if t.value is not None else None
    return a, t_val


def qp_cg_solve(w, a_init, n, max_iters=200, tol=1e-12):
    """
    Solve min a^T H a s.t. 1^T a = 1, a >= 0
    where H[i,j] = w[i+j] (Hankel matrix).
    Uses active-set method with CG inner solver.
    """
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    active = np.ones(n, dtype=bool)
    active[a > 1e-10] = True
    if not active.any():
        active[:] = True

    for outer in range(max_iters):
        act_idx = np.where(active)[0]
        k_active = len(act_idx)
        if k_active == 0:
            return np.ones(n) / n
        if k_active == 1:
            a = np.zeros(n)
            a[act_idx[0]] = 1.0
            return a

        # Solve subproblem on active set using CG
        # We need to solve: H_A x = 1_A, then normalize
        # H_A is a submatrix of the Hankel matrix (not Hankel itself for subsets)
        # For small k_active, use direct solve; for large, use CG with full H and masking

        if k_active <= 64:
            # Direct solve
            H_sub = np.array([[w[act_idx[i] + act_idx[j]] for j in range(k_active)]
                              for i in range(k_active)])
            H_sub += 1e-14 * np.eye(k_active)
            try:
                x_sub = np.linalg.solve(H_sub, np.ones(k_active))
            except np.linalg.LinAlgError:
                x_sub = np.linalg.lstsq(H_sub, np.ones(k_active), rcond=None)[0]
        else:
            # CG on the full system with active set masking
            b_full = np.zeros(n)
            b_full[act_idx] = 1.0
            x = np.zeros(n)
            x[act_idx] = 1.0 / k_active  # Initial guess

            def matvec(v):
                # H @ v but zero out inactive rows
                Hv = hankel_mv(w, v)
                Hv[~active] = 0.0
                return Hv

            r = b_full - matvec(x)
            p = r.copy()
            rs_old = r @ r
            for cg_it in range(min(100, 2 * k_active)):
                if abs(rs_old) < tol * (1.0 + x @ x):
                    break
                Ap = matvec(p)
                pAp = p @ Ap
                if pAp < 1e-16:
                    break
                alpha = rs_old / pAp
                x += alpha * p
                r_new = r - alpha * Ap
                rs_new = r_new @ r_new
                r = r_new
                if rs_new < 1e-30:
                    break
                p = r + (rs_new / rs_old) * p
                rs_old = rs_new

            x_sub = x[act_idx]

        x_sum = x_sub.sum()
        if abs(x_sum) < 1e-14:
            x_norm = np.ones(k_active) / k_active
        else:
            x_norm = x_sub / x_sum

        # Check for negative components in active set
        min_val = x_norm.min()
        if min_val < -1e-10:
            # Remove the most negative
            worst_local = np.argmin(x_norm)
            worst_global = act_idx[worst_local]
            active[worst_global] = False
            # Update a
            a_new = np.zeros(n)
            a_new[act_idx] = np.maximum(x_norm, 0.0)
            s2 = a_new.sum()
            if s2 > 1e-15:
                a = a_new / s2
            continue

        # Form candidate
        a_candidate = np.zeros(n)
        a_candidate[act_idx] = np.maximum(x_norm, 0.0)
        s_cand = a_candidate.sum()
        if s_cand < 1e-15:
            a_candidate[act_idx] = 1.0 / k_active
        else:
            a_candidate = a_candidate / s_cand

        # KKT check: for inactive j, grad[j] >= mu
        Wa = hankel_mv(w, a_candidate)
        grad = 2.0 * Wa
        mu = np.mean(grad[act_idx])

        inactive_idx = np.where(~active)[0]
        if len(inactive_idx) > 0:
            violations = grad[inactive_idx] - mu
            min_viol = violations.min()
            if min_viol < -1e-10:
                # Add the most violating inactive variable
                worst_inactive = inactive_idx[np.argmin(violations)]
                active[worst_inactive] = True
                a = a_candidate
                continue

        # Converged
        a = a_candidate
        break

    return a


def primal_dual_qp(a0, time_limit, T_init=0.01, topk_mode=False, topk=15,
                   blend=0.5, verbose=False):
    """
    Primal-dual iteration with CG-based QP solver.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.3:
            break

        conv = fast_convolve(a)
        cmax = conv.max()

        # Compute weights
        if topk_mode:
            k_use = min(topk, len(conv))
            peaks = np.argpartition(conv, -k_use)[-k_use:]
            w = np.zeros(len(conv))
            w[peaks] = 1.0 / k_use
        else:
            scaled = conv / T
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.ones(len(conv)) / len(conv)
            else:
                w = w / w_sum

        # Solve QP
        a_qp = qp_cg_solve(w, a, n, max_iters=150)

        # Blend
        a_new = (1.0 - blend) * a + blend * a_qp
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a_qp
        else:
            a_new = a_new / s_new

        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adapt temperature
        if no_improve > 8:
            T *= 1.12
        if no_improve > 30:
            T *= 0.7
        if no_improve > 80:
            rng = np.random.default_rng((it * 104729 + 3) % (2 ** 31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    pd_qp it={it}, obj={best_obj:.10f}, T={T:.6f}")

    return best_a, best_obj


def nesterov_multi(a0, time_limit, lr_init=0.005, K=5, verbose=False):
    """
    Nesterov-accelerated subgradient with top-K peak convex combination.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = 0.9
    y = a.copy()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        y = a + mom * v

        conv_y = fast_convolve(y)
        # Top-K peaks with softmax weights
        k_use = min(K, len(conv_y))
        top_k_idx = np.argpartition(conv_y, -k_use)[-k_use:]
        top_k_vals = conv_y[top_k_idx]
        T_sub = max(1e-6, (top_k_vals.max() - top_k_vals.min()) * 0.1)
        p = np.exp((top_k_vals - top_k_vals.max()) / T_sub)
        p = p / p.sum()

        # Compute convex combination of gradients
        grad = np.zeros(n)
        for pk_weight, pk in zip(p, top_k_idx):
            pk_int = int(pk)
            js = max(0, pk_int - n + 1)
            je = min(n - 1, pk_int)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * pk_weight * y[pk_int - idx]

        # Update
        y_next = project_simplex(y - lr * grad)
        a = y_next
        v = a - y

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9999
        mom = min(0.995, mom + 0.00003)

        if no_improve > 80:
            lr *= 0.7
            mom *= 0.9
        if no_improve > 250:
            rng = np.random.default_rng((it * 48271 + 1) % (2 ** 31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = lr_init * 0.5
            mom = 0.9
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    nesterov_K{K} it={it}, obj={best_obj:.10f}")

    return best_a, best_obj


def admm_minmax(a0, time_limit, rho_init=1.0, verbose=False):

    # Simpler ADMM:
    # min 0 s.t. a in simplex, conv(a,a) <= t*1
    # Lagrangian with dual variable lambda >= 0 for the constraint
    # This doesn't work directly because the constraint is on a nonlinear function.

    # Alternative: use proximal gradient on f(a) = max_k conv(a,a)[k]

    return nesterov_multi(a0, time_limit, lr_init=lr_init, K=3, verbose=verbose)


def coordinate_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent: try adjusting one coordinate at a time.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Try adjusting coordinates that contribute to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = np.arange(js, je + 1)

        # Random subset for efficiency
        if len(contrib) > 50:
            trial = rng.choice(contrib, size=50, replace=False)
        else:
            trial = contrib

        improved = False
        for idx in trial:
            if a[idx] < 1e-12:
                continue
            # Try reducing a[idx] and distributing to others
            for frac in [0.005, 0.01, 0.02, 0.05]:
                shift = a[idx] * frac
                if shift < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[idx] -= shift
                # Distribute to non-contributing positions
                non_contrib = np.setdiff1d(np.arange(n), contrib)
                if len(non_contrib) == 0:
                    continue
                # Give to positions with lowest contribution to peak
                a_trial[non_contrib] += shift / len(non_contrib)
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                c = fast_convolve(a_trial).max()
                if c < best_obj - 1e-14:
                    a = a_trial
                    best_obj = c
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved and it % 200 == 0:
            break

    return best_a, best_obj


def evolutionary_search(a0, time_limit, pop_size=20, verbose=False):
    """
    Simple evolutionary strategy for diversification.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    rng = np.random.default_rng(999)

    # Initialize population
    population = [a.copy()]
    for _ in range(pop_size - 1):
        noise_scale = rng.uniform(0.001, 0.01)
        mutant = project_simplex(a + rng.standard_normal(n) * noise_scale)
        population.append(mutant)

    # Evaluate
    population = sorted(population, key=lambda x: fast_convolve(x).max())
    best_a = population[0].copy()
    best_obj = fast_convolve(best_a).max()

    t0 = time.monotonic()
    gen = 0

    while time.monotonic() - t0 < time_limit:
        gen += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        new_pop = [population[0].copy()]  # Elitism

        half = max(2, len(population) // 2)
        for _ in range(len(population) - 1):
            # Crossover: interpolate two parents
            p1 = population[rng.integers(0, half)]
            p2 = population[rng.integers(0, half)]
            alpha = rng.uniform(0.3, 0.7)
            child = alpha * p1 + (1 - alpha) * p2
            # Mutation
            noise = rng.standard_normal(n) * rng.uniform(0.0005, 0.005)
            child = project_simplex(child + noise)
            new_pop.append(child)

        # Evaluate and select
        new_pop = sorted(new_pop, key=lambda x: fast_convolve(x).max())
        population = new_pop[:pop_size]

        if fast_convolve(population[0]).max() < best_obj - 1e-14:
            best_obj = fast_convolve(population[0]).max()
            best_a = population[0].copy()

    return best_a, best_obj


def smoothed_grad(a0, time_limit, T_init=0.001, lr_init=0.03, verbose=False):
    """Smoothed gradient descent with log-sum-exp surrogate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient: 2 * correlate(p, a)
        grad = 2.0 * hankel_mv(p, a)

        # Line search
        best_candidate = a
        best_val = conv.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            T *= 1.1
            lr *= 0.9
        if no_improve > 60:
            T *= 0.6
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2 ** 31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    smoothed it={it}, obj={best_obj:.10f}")

    return best_a, best_obj

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384

    n_target = len(parent)
    print(f"Starting: n_target={n_target}, current_best={best_score:.10f}")

    # === Phase 1: SDP warm start (~250s) ===
    print("\n=== Phase 1: SDP Warm Start ===")
    t1 = time.monotonic()

    sdp_results = {}
    for n_sdp in [128, 256]:
        remaining = deadline - time.monotonic() - 900
        if remaining < 60:
            break
        if n_sdp > n_target:
            n_sdp = n_target
        t_alloc = min(remaining, 120)
        print(f"  Solving SDP n={n_sdp} (budget {t_alloc:.0f}s)...")
        try:
            a_sdp, t_sdp = solve_sdp_autocorr(n_sdp, time_limit=t_alloc)
            if a_sdp is not None:
                sdp_results[n_sdp] = a_sdp
                s = a_sdp.sum()
                obj = fast_convolve(a_sdp / s).max()
                print(f"  SDP n={n_sdp}: t_bound={t_sdp:.10f}, obj_on_simplex={obj:.10f}")
                # Interpolate to target n
                if n_sdp != n_target:
                    a_interp = np.interp(np.linspace(0, 1, n_target),
                                         np.linspace(0, 1, n_sdp), a_sdp / s)
                    a_interp = np.maximum(a_interp, 0.0)
                    a_interp = a_interp / a_interp.sum()
                    try_seq(a_interp, f"sdp_interp_{n_sdp}")
                else:
                    try_seq(a_sdp / s, f"sdp_{n_sdp}")
        except Exception as e:
            print(f"  SDP n={n_sdp} failed: {e}")

    # Also try a few structured inits
    inits = [np.ones(n_target) / n_target]
    x = np.arange(n_target, dtype=np.float64)
    edge = 1.0 / (np.minimum(x, n_target - 1 - x) + 1.0)
    inits.append(edge / edge.sum())
    g = np.exp(-0.5 * ((x - 0.5 * (n_target - 1)) / (0.2 * n_target)) ** 2)
    inits.append(g / g.sum())
    for ii, init in enumerate(inits):
        try_seq(init, f"init_{ii}")

    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")

    # Use best from Phase 1 for further optimization
    if best_seq is not None:
        start_a = best_seq.copy()
    else:
        start_a = parent.copy()

    # === Phase 2: Primal-dual QP (main, ~700s) ===
    print("\n=== Phase 2: Primal-Dual QP ===")
    t2 = time.monotonic()

    configs = [
        {'T_init': 0.005, 'topk_mode': False, 'topk': 20, 'blend': 0.5, 't_frac': 0.25},
        {'T_init': 0.01, 'topk_mode': False, 'topk': 30, 'blend': 0.4, 't_frac': 0.2},
        {'T_init': 0.02, 'topk_mode': True, 'topk': 10, 'blend': 0.5, 't_frac': 0.15},
        {'T_init': 0.003, 'topk_mode': True, 'topk': 5, 'blend': 0.3, 't_frac': 0.15},
        {'T_init': 0.008, 'topk_mode': False, 'topk': 50, 'blend': 0.6, 't_frac': 0.15},
    ]

    for ci, cfg in enumerate(configs):
        remaining = deadline - time.monotonic() - 500
        if remaining < 20:
            break
        t_alloc = min(remaining, int(budget_s * cfg['t_frac']))
        print(f"  PD-QP config {ci} (T={cfg['T_init']}, topk={cfg.get('topk')}, "
              f"blend={cfg['blend']}), t={t_alloc:.0f}s")
        t_params = {k: v for k, v in cfg.items() if k != 't_frac'}
        a_opt, _ = primal_dual_qp(start_a, time_limit=t_alloc, **t_params)
        try_seq(a_opt, f"pdqp_{ci}")
        if best_seq is not None:
            start_a = best_seq.copy()

    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")

    # === Phase 3: Nesterov + Smoothed Gradient (~400s) ===
    print("\n=== Phase 3: Nesterov & Smoothed Gradient ===")
    t3 = time.monotonic()

    if best_seq is not None:
        for K in [3, 5, 8]:
            remaining = deadline - time.monotonic() - 250
            if remaining < 15:
                break
            t_alloc = min(remaining, 80)
            print(f"  Nesterov K={K}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_multi(best_seq, time_limit=t_alloc, lr_init=0.004, K=K)
            try_seq(a_opt, f"nest_K{K}")

        for T_val, lr_val in [(0.0005, 0.02), (0.002, 0.015), (0.005, 0.01)]:
            remaining = deadline - time.monotonic() - 150
            if remaining < 15:
                break
            t_alloc = min(remaining, 60)
            print(f"  Smoothed T={T_val}, lr={lr_val}, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_grad(best_seq, time_limit=t_alloc, T_init=T_val, lr_init=lr_val)
            try_seq(a_opt, f"smooth_T{T_val}")

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")

    # === Phase 4: Evolutionary + Coordinate Descent (~250s) ===
    print("\n=== Phase 4: Evolutionary & Coordinate Descent ===")
    t4 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 150
        if remaining > 20:
            t_alloc = min(remaining, 80)
            print(f"  Evolutionary, t={t_alloc:.0f}s")
            a_opt, _ = evolutionary_search(best_seq, time_limit=t_alloc, pop_size=15)
            try_seq(a_opt, "evo")

        remaining = deadline - time.monotonic() - 100
        if remaining > 10:
            t_alloc = min(remaining, 60)
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord")

        # More QP with the best so far
        remaining = deadline - time.monotonic() - 80
        if remaining > 10:
            t_alloc = min(remaining, 50)
            print(f"  Fine PD-QP, t={t_alloc:.0f}s")
            a_opt, _ = primal_dual_qp(best_seq, time_limit=t_alloc,
                                      T_init=0.002, topk_mode=False, topk=20, blend=0.4)
            try_seq(a_opt, "fine_pdqp")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # === Phase 5: Final Polish (~100s) ===
    print("\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Tiny Nesterov passes
        for lr in [0.001, 0.002]:
            remaining = deadline - time.monotonic() - 10
            if remaining < 5:
                break
            t_alloc = min(remaining, 20)
            a_opt, _ = nesterov_multi(best_seq, time_limit=t_alloc, lr_init=lr, K=3)
            try_seq(a_opt, f"final_nest_lr{lr}")

        # Tiny perturbations
        for _ in range(3):
            remaining = deadline - time.monotonic() - 5
            if remaining < 3:
                break
            noise = rng.standard_normal(n_best) * 0.0005
            perturbed = project_simplex(best_seq + noise)
            t_alloc = min(remaining, 10)
            a_opt, _ = smoothed_grad(perturbed, time_limit=t_alloc, T_init=0.0003, lr_init=0.003)
            try_seq(a_opt, f"perturb_{_}")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
