def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])

    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    return A

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build (2n-1) x n matrix A where (f*g)[k] = sum_j A[k,j]*g[j]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, n=None):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    if n is None:
        n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def compute_cross_corr(w, a):
    """Compute cross-correlation (w ⊛ a)[j] = sum_k w[k+j]*a[k] via FFT."""
    n_w = len(w)
    n_a = len(a)
    N = 1 << (n_w + n_a - 1 - 1).bit_length() if n_w + n_a - 1 > 1 else 1
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    cross = np.fft.ifft(W * np.conj(A)).real[:n_w]
    return cross


def smooth_max_gradient(a, temp):
    """
    Compute smooth max of conv(a,a) and its gradient wrt a.
    Uses log-sum-exp approximation.
    Returns (value, gradient).
    """
    n = len(a)
    conv = fast_convolve(a)
    cmax = conv.max()
    if temp < 1e-15:
        idx = np.argmax(conv)
        grad = np.zeros(n)
        js = max(0, idx - n + 1)
        je = min(n - 1, idx)
        grad[js:je + 1] = 2.0 * a[idx - js:idx - je + 1][::-1]
        return cmax, grad

    shifted = (conv - cmax) / temp
    exp_shifted = np.exp(shifted - shifted.max())
    w = exp_shifted / exp_shifted.sum()
    val = cmax + temp * np.log(exp_shifted.sum())

    # grad[j] = 2 * sum_k w[k] * a[k-j] (bounded) = 2 * cross_corr(w, a)[j]
    cross = compute_cross_corr(w, a)
    grad = 2.0 * cross[:n]
    return val, grad

def peak_coord_descent(a0, time_limit):
    """
    Targeted coordinate descent: for each peak, try redistributing mass
    from contributing indices to reduce the peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    step_scale = 0.005
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit - 1.0:
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        it += 1

        conv = fast_convolve(a)
        # Focus on top peaks
        n_peaks = min(6, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        improved = False
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue

            # Find top contributing pairs
            contribs = []
            for j in range(js, je + 1):
                contribs.append((a[j] * a[k - j], j, k - j))
            contribs.sort(reverse=True)

            for (_, j1, j2) in contribs[:4]:
                if a[j1] < 1e-10:
                    continue
                # Try several random target positions
                for _trial in range(6):
                    j_target = int(rng.integers(0, n))
                    if j_target == j1 or j_target == j2:
                        continue
                    src = j1 if a[j1] >= a[j2] else j2
                    if a[src] < 1e-10:
                        continue
                    for delta_frac in [step_scale, step_scale * 0.3, step_scale * 2.0]:
                        delta = a[src] * delta_frac
                        if delta < 1e-12:
                            continue
                        candidate = a.copy()
                        candidate[src] -= delta
                        candidate[j_target] += delta
                        candidate = np.maximum(candidate, 0)
                        candidate = candidate / candidate.sum()
                        val = fast_convolve(candidate).max()
                        if val < best_obj - 1e-15:
                            a = candidate
                            best_obj = val
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            step_scale *= 0.95
            if step_scale < 1e-7:
                break
            if it % 100 == 0:
                # Global perturbation
                candidate = project_simplex(a + rng.standard_normal(n) * step_scale * 3)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
            elif it > 5000:
                break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_starts(n, parent, rng):
    """Generate diverse initializations."""
    starts = []
    x = np.arange(n, dtype=np.float64)

    # Parent resampled
    if parent is not None and len(parent) != n:
        starts.append(resample_sequence(parent, n))
    elif parent is not None:
        starts.append(parent.copy())

    # Uniform
    starts.append(np.ones(n) / n)

    # Cosine taper
    cos_t = np.cos(np.pi * x / max(n - 1, 1))
    w = 0.5 * (1 + cos_t)
    w = np.maximum(w, 0.005)
    starts.append(w / w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - (n - 1) / 2) / (0.25 * n)) ** 2)
    starts.append(g / g.sum())

    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    starts.append(edge / edge.sum())

    # Two peaks
    g1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.1 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())

    # Triangle
    tri = 1.0 - 2.0 * np.abs(x - (n - 1) / 2) / max(n - 1, 1)
    tri = np.maximum(tri, 0.01)
    starts.append(tri / tri.sum())

    # Split cosine (alternating)
    alt = np.where(x % 2 == 0, 1.0, 0.3)
    starts.append(alt / alt.sum())

    # Random Dirichlet
    starts.append(rng.dirichlet(np.ones(n) * 0.5))
    starts.append(rng.dirichlet(np.ones(n) * 0.2))

    return starts


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search for minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        try:
            score = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")

    n_parent = len(parent)

    # === Phase 1: Quick exploration across n values (200s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_phase = time.monotonic()

    lengths = [512, 640, 768, 896, 1024, 1280, 1536]
    lengths = [l for l in lengths if 256 <= l <= 2048]
    if n_parent not in lengths:
        lengths.append(n_parent)
        lengths.sort()

    for n_try in lengths:
        remaining = deadline - time.monotonic() - 900
        if remaining < 20:
            break

        starts = generate_starts(n_try, parent, rng)
        # Use top 3 starts
        for si, start in enumerate(starts[:3]):
            remaining = deadline - time.monotonic() - 850
            if remaining < 10:
                break
            t_alloc = min(remaining, 30)
            a_opt, _ = fw_optimize(start, time_limit=t_alloc)
            try_seq(a_opt, f"fw_n{n_try}_s{si}")

    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW on best (500s) ===
    print("\n=== Phase 2: Deep FW ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Main deep FW
        remaining = deadline - time.monotonic() - 550
        if remaining > 30:
            t_alloc = min(remaining, 450)
            print(f"  Deep FW n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_fw")

        # Perturbed restarts with different scales
        for scale in [0.001, 0.003, 0.01, 0.02]:
            remaining = deadline - time.monotonic() - 400
            if remaining < 20:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 80)
            a_opt, _ = fw_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"fw_pert_{scale}")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Smooth refinement (300s) ===
    print("\n=== Phase 3: Smooth Refinement ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Run smooth refine with different initial temperatures
        for temp_factor in [0.05, 0.02, 0.01]:
            remaining = deadline - time.monotonic() - 300
            if remaining < 15:
                break
            t_alloc = min(remaining, 100)
            print(f"  Smooth refine (temp_factor={temp_factor}), t={t_alloc:.0f}s")
            a_opt, _ = smooth_refine(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, f"smooth_{temp_factor}")

        # Also try from a slightly perturbed start
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 200
        if remaining > 20:
            perturbed = project_simplex(best_seq + rng.standard_normal(n_best) * 0.005)
            t_alloc = min(remaining, 80)
            a_opt, _ = smooth_refine(perturbed, time_limit=t_alloc)
            try_seq(a_opt, "smooth_pert")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Peak coordinate descent (250s) ===
    print("\n=== Phase 4: Peak Coord Descent ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 150
        if remaining > 10:
            t_alloc = min(remaining, 200)
            print(f"  Peak coord descent, t={t_alloc:.0f}s")
            a_opt, _ = peak_coord_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_coord")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final polish - alternate methods (remaining) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        round_num = 0
        while time.monotonic() < deadline - 5.0:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            if round_num % 4 == 1:
                # Small FW
                noise = rng.standard_normal(n_best) * 0.001
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 30)
                a_opt, _ = fw_optimize(start, time_limit=t_alloc)
                try_seq(a_opt, f"polish_fw_{round_num}")
            elif round_num % 4 == 2:
                # Smooth refine
                t_alloc = min(remaining, 40)
                a_opt, _ = smooth_refine(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"polish_smooth_{round_num}")
            elif round_num % 4 == 3:
                # Coord descent
                t_alloc = min(remaining, 30)
                a_opt, _ = peak_coord_descent(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"polish_coord_{round_num}")
            else:
                # Medium perturbation + FW
                noise = rng.standard_normal(n_best) * 0.005
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 40)
                a_opt, _ = fw_optimize(start, time_limit=t_alloc)
                try_seq(a_opt, f"polish_med_{round_num}")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
