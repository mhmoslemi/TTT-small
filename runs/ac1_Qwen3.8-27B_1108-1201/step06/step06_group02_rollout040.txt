import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute the gradient contribution: for each j, sum over k of p[k] * 2*a[k-j]."""
    n = len(a)
    # p has length 2n-1, a has length n
    # grad[j] = sum_k p[k] * d(conv_k)/da[j] ≈ 2 * sum_k p[k] * a[k-j] (with bounds)
    # This is a cross-correlation
    return 2.0 * np.correlate(p, a, mode='valid')


def solve_lp_oracle_multilevel(f, M, top_k, k_gap):

    bounds = [(0, None)] * n

                      options={'presolve': True, 'time_limit': 20})
        if res.success:

    except Exception:
        pass
    return None

def lse_optimize(a0, time_limit, alpha_start=50, alpha_end=500, lr_start=0.02):
    """
    Optimize using log-sum-exp as smooth surrogate for max.
    Uses continuation on alpha (start low, increase to sharp).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    # Continuation: alpha goes from alpha_start to alpha_end
    # We'll use a logarithmic schedule
    total_steps_estimate = int(time_limit / 0.01)  # rough estimate
    alpha = alpha_start
    lr = lr_start

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        
        # Compute LSE gradient
        # Use softmax weights with temperature 1/alpha
        shifted = conv - conv.max()
        weights = np.exp(alpha * shifted)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            weights = np.zeros_like(conv)
            weights[np.argmax(conv)] = 1.0
            w_sum = 1.0
        else:
            weights /= w_sum

        grad = 2.0 * hankel_mv(weights, a)

        # Gradient step with projection
        best_candidate = a
        best_val = conv.max()
        
        for trial_lr in [lr, lr * 0.5, lr * 0.25]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive alpha and learning rate
        if no_improve > 15:
            alpha = min(alpha * 1.1, alpha_end)
            lr *= 0.95
        if no_improve > 50:
            lr = max(lr * 0.9, lr_start * 0.1)
        if no_improve > 150:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise = 0.002 + 0.003 * min(1.0, (no_improve - 150) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            alpha = alpha_start
            lr = lr_start
            no_improve = 0

    return best_a, best_obj


def peak_shaving(a0, time_limit):
    """
    Directly attack the highest peaks by moving mass from contributing
    positions to non-contributing ones.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Find the most contributing pair for this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]

        # Find a position to move mass to (one that doesn't contribute to the top peaks)
        # Use the bottom 10% of a as targets
        threshold_idx = int(0.1 * n)
        bottom_positions = np.argsort(a)[:threshold_idx]
        
        # Check which bottom positions don't contribute much to the peak
        best_target = None
        best_gain = 0
        for j_t in bottom_positions:
            # Check if moving mass to j_t would reduce the peak
            # The change in conv[k_peak] is approximately:
            # -a[j_max]*a[k_peak-j_max] + a[j_max_new]*a[k_peak-j_t] (rough)
            # For simplicity, just try it
            pass

        # Simpler approach: try moving a small amount from j_max to the minimum position
        j_min = int(np.argmin(a))
        if j_min == j_max:
            stall += 1
            if stall > 100:
                break
            continue

        improved = False
        for delta_frac in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            delta = a[j_max] * delta_frac
            if delta < 1e-10:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min] += delta
            candidate = np.maximum(candidate, 0.0)
            st = candidate.sum()
            if st < 1e-15:
                continue
            candidate = candidate / st
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall % 20 == 0:
                # Try random perturbations
                rng = np.random.default_rng(stall * 137 + 1)
                for _ in range(5):
                    j_src = rng.integers(0, n)
                    j_dst = rng.integers(0, n)
                    if j_src == j_dst:
                        continue
                    delta = a[j_src] * 0.05
                    candidate = a.copy()
                    candidate[j_src] -= delta
                    candidate[j_dst] += delta
                    candidate = np.maximum(candidate, 0.0)
                    st = candidate.sum()
                    if st > 1e-15:
                        candidate = candidate / st
                        c = fast_convolve(candidate)
                        if c.max() < best_obj - 1e-15:
                            a = candidate
                            best_obj = c.max()
                            best_a = a.copy()
                            stall = 0
                            break
            if stall > 500:
                break

    return best_a, best_obj


def lse_multipeak_optimize(a0, time_limit, top_m=5):

def generate_structured_inits(n, parent, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine window
    c = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits.append(c / c.sum())
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5*(n-1)) / (0.2*n))**2)
    inits.append(g / g.sum())
    
    # Edge-enhanced
    e = 1.0 / (np.minimum(x, n-1-x) + 1.0)
    inits.append(e / e.sum())
    
    # Multi-plateau: 3 equal plateaus with gaps
    plateaus = np.zeros(n)
    seg = n // 5
    plateaus[0:seg] = 1.0
    plateaus[2*seg:3*seg] = 1.0
    plateaus[4*seg:n] = 1.0
    inits.append(plateaus / plateaus.sum())
    
    # Symmetric with center peak
    sym = np.exp(-0.5 * ((x - 0.5*(n-1)) / (0.1*n))**2) + 0.3
    inits.append(sym / sym.sum())
    
    # Fourier-based: low frequency dominant
    for freq in [2, 3, 5]:
        h = 1.0 + 0.2 * np.cos(2 * np.pi * freq * x / n)
        h = np.maximum(h, 0.01)
        inits.append(h / h.sum())
    
    # Parent interpolation
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # "Spectral" initialization: design in frequency domain
    # Want |A(f)|^2 to be as flat as possible
    # This means A(f) should have constant magnitude
    # Approximate with a window that has flat spectrum
    nfft = 1 << (2*n).bit_length()
    target_spectrum = np.ones(nfft // 2 + 1)
    # Add some structure
    A_target = target_spectrum * np.exp(1j * rng.standard_normal(nfft // 2 + 1) * 0.1)
    a_spec = np.fft.irfft(A_target, nfft)[:n]
    a_spec = np.maximum(a_spec, 0)
    s = a_spec.sum()
    if s > 1e-10:
        inits.append(a_spec / s)
    
    return inits


def try_and_update(a, state, label):
    """Evaluate a sequence and update state if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a_norm)))

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute gradient: for each j, sum_k p[k] * 2*a[k-j]."""
    return 2.0 * np.correlate(p, a, mode='valid')


def solve_lp_oracle(f, M, time_limit_lp=15):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit_lp})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def solve_lp_oracle_multilevel(f, M, top_k, k_gap, time_limit_lp=15):
    """
    LP with multi-level bounds: top-k conv values get progressively tighter bounds.
    b_ub[rank_r] = M + k_gap * r for the (r+1)-th largest conv value.
    """
    n = len(f)
    conv = fast_convolve(f)
    b_ub = np.full(2 * n - 1, M + k_gap * top_k)
    order = np.argsort(conv)[::-1]
    for rank in range(min(top_k, len(order))):
        b_ub[order[rank]] = M + k_gap * rank
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit_lp})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def solve_lp_oracle_soft(f, M, lam, time_limit_lp=15):
    """
    LP with soft multi-level: bound = M + lam * (conv_k - M)_+ for each k.
    This gives tighter bounds where conv is already high.
    """
    n = len(f)
    conv = fast_convolve(f)
    b_ub = M + lam * np.maximum(conv - M, 0.0)
    b_ub = np.maximum(b_ub, M)  # at least M
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit_lp})

            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def lse_optimize(a0, time_limit, alpha_start=50.0, alpha_end=1000.0, lr_start=0.015):
    """
    LSE (log-sum-exp) based optimization with continuation on alpha.
    Minimizes a smooth approximation to max(conv) using gradient descent.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    alpha = alpha_start
    lr = lr_start

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        shifted = conv - conv.max()
        weights = np.exp(alpha * shifted)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            weights = np.zeros_like(conv)
            weights[np.argmax(conv)] = 1.0
            w_sum = 1.0
        else:
            weights /= w_sum

        grad = hankel_mv(weights, a)

        best_candidate = a
        best_val = conv.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 12:
            alpha = min(alpha * 1.12, alpha_end)
            lr *= 0.95
        if no_improve > 40:
            lr = max(lr * 0.93, lr_start * 0.05)
        if no_improve > 130:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise = 0.002 + 0.004 * min(1.0, (no_improve - 130) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            alpha = alpha_start
            lr = lr_start
            no_improve = 0

    return best_a, best_obj


def peak_shaving(a0, time_limit):
    """
    Directly attack the highest peak by moving mass from the most
    contributing source position to a low-value sink position.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]

        j_min = int(np.argmin(a))
        if j_min == j_max:
            j_min = int(np.argmin(np.delete(a, j_max)))
            if j_min >= j_max:
                j_min += 1

        improved = False
        for delta_frac in [0.4, 0.25, 0.15, 0.08, 0.04, 0.02]:
            delta = a[j_max] * delta_frac
            if delta < 1e-10:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min] += delta
            candidate = np.maximum(candidate, 0.0)
            st = candidate.sum()
            if st < 1e-15:
                continue
            candidate = candidate / st
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall % 15 == 0 and stall < 400:
                rng = np.random.default_rng(stall * 199 + 3)
                for _ in range(3):
                    j_src = int(rng.integers(0, n))
                    j_dst = int(rng.integers(0, n))
                    if j_src == j_dst or a[j_src] < 1e-8:
                        continue
                    delta = a[j_src] * 0.05
                    candidate = a.copy()
                    candidate[j_src] -= delta
                    candidate[j_dst] += delta
                    candidate = np.maximum(candidate, 0.0)
                    st = candidate.sum()
                    if st > 1e-15:
                        candidate = candidate / st
                        c = fast_convolve(candidate)
                        if c.max() < best_obj - 1e-15:
                            a = candidate
                            best_obj = c.max()
                            best_a = a.copy()
                            stall = 0
                            break
            if stall > 400:
                break

    return best_a, best_obj


def fourier_refine(a0, time_limit, n_modes=15):
    """
    Optimize over Fourier coefficients for structured improvement.
    Parameterizes a[i] = (1/M + sum c_m cos + s_m sin) projected to simplex.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    M = min(n_modes, n // 2 - 1)
    x = np.arange(n, dtype=np.float64)

    a_centered = a - 1.0 / n
    init_coeffs = np.zeros(2 * M)
    for m in range(1, M + 1):
        init_coeffs[2 * (m - 1)] = 2.0 * np.sum(a_centered * np.cos(2.0 * np.pi * m * x / n))
        init_coeffs[2 * (m - 1) + 1] = 2.0 * np.sum(a_centered * np.sin(2.0 * np.pi * m * x / n))

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    n_restarts = max(2, int(time_limit / 25))

    for r in range(n_restarts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 8:
            break

        if r == 0:
            x0 = init_coeffs.copy()
        else:
            rng = np.random.default_rng(r * 12345 + 7)
            scale = 0.003 * (1 + r * 0.5)
            x0 = init_coeffs + rng.standard_normal(2 * M) * scale

        def obj_func(coeffs):
            base = np.ones(n) / n
            for m in range(1, M + 1):
                c = coeffs[2 * (m - 1)]
                d = coeffs[2 * (m - 1) + 1]
                base += (c * np.cos(2.0 * np.pi * m * x / n) +
                         d * np.sin(2.0 * np.pi * m * x / n)) / n
            base = np.maximum(base, 0.0)
            s = base.sum()
            if s < 1e-15:
                return 10.0
            base = base / s
            return fast_convolve(base).max()

        try:
            res = minimize(obj_func, x0, method='Nelder-Mead',
                           options={'maxiter': 5000, 'xatol': 1e-10, 'fatol': 1e-12,
                                    'maxfev': 30000})
            if res.fun < best_obj:
                best_obj = res.fun
                base = np.ones(n) / n
                for m in range(1, M + 1):
                    c = res.x[2 * (m - 1)]
                    d = res.x[2 * (m - 1) + 1]
                    base += (c * np.cos(2.0 * np.pi * m * x / n) +
                             d * np.sin(2.0 * np.pi * m * x / n)) / n
                base = np.maximum(base, 0.0)
                best_a = base / base.sum()
                print("    Fourier restart {}: obj={:.10f}".format(r, best_obj))
        except Exception:
            pass

    return best_a, best_obj


def generate_structured_inits(n, parent, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)

    c = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits.append(c / c.sum())

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    e = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(e / e.sum())

    # Multi-plateau
    plateaus = np.zeros(n)
    seg = n // 6
    plateaus[0:seg] = 1.0
    plateaus[2 * seg:3 * seg] = 1.0
    plateaus[4 * seg:5 * seg] = 1.0
    if plateaus.sum() > 0:
        inits.append(plateaus / plateaus.sum())

    # Log-spaced
    ls = 1.0 / (np.minimum(x + 1, n - x) ** 0.3 + 1)
    inits.append(ls / ls.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # Frequency-domain design
    nfft = 1 << (2 * n).bit_length()
    A_target = np.ones(nfft // 2 + 1, dtype=np.complex128)
    phases = rng.standard_normal(nfft // 2 + 1) * 0.05
    A_target = A_target * np.exp(1j * phases)
    a_spec = np.fft.irfft(A_target, nfft)[:n]
    a_spec = np.maximum(a_spec, 0)
    s = a_spec.sum()
    if s > 1e-10:
        inits.append(a_spec / s)

    return inits

        state = [evaluate_sequence(parent.tolist()), parent.copy()]

    n_parent = len(parent)

    # === Phase 1: Multi-n exploration (~280s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase1 = time.monotonic()

    # Focus on n values around and above parent
    candidate_ns = sorted(set([n_parent, 640, 768, 896, 1024, 1280, 1536]))
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 2048]
    # Sort: parent first, then by closeness to parent
    candidate_ns.sort(key=lambda x: (x != n_parent, abs(x - n_parent)))

    n_phase1 = 280.0
    time_per_n = n_phase1 / len(candidate_ns)

    for n_try in candidate_ns:
        remaining = min(time_per_n, n_phase1 - (time.monotonic() - t_phase1),
                        deadline - time.monotonic() - 900)
        if remaining < 15:
            break

        inits = generate_structured_inits(n_try, parent, rng)
        best_a_this_n = None
        best_obj_this_n = float('inf')

        # Try up to 3 initializations
        n_inits_to_try = min(3, len(inits))
        t_per_init = max(4.0, remaining / (n_inits_to_try + 1))

        for ii in range(n_inits_to_try):
            rem = min(t_per_init, deadline - time.monotonic() - 850)
            if rem < 3:
                break
            a_opt, obj = fw_lp_optimize(inits[ii], time_limit=rem)
            if obj < best_obj_this_n:
                best_obj_this_n = obj
                best_a_this_n = a_opt.copy()
            # Quick LSE polish
            rem2 = min(5.0, deadline - time.monotonic() - 850)
            if rem2 > 2:
                a_opt2, _ = lse_optimize(a_opt, time_limit=rem2,
                                         alpha_start=100, alpha_end=500, lr_start=0.01)
                if fast_convolve(a_opt2).max() < best_obj_this_n:
                    best_obj_this_n = fast_convolve(a_opt2).max()
                    best_a_this_n = a_opt2.copy()

        if best_a_this_n is not None:
            try_and_update(best_a_this_n, state, "p1_n{}".format(n_try))

    print("  Phase 1 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase1))

    # === Phase 2: Deep FW-LP (~520s) ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase2 = time.monotonic()
    n_phase2 = 520.0

    if state[1] is not None:
        n_best = len(state[1])

        # Standard FW
        remaining = min(n_phase2 * 0.35, deadline - time.monotonic() - 500)
        if remaining > 30:
            print("  Standard FW n={}, t={:.0f}s".format(n_best, remaining))
            a_opt, _ = fw_lp_optimize(state[1], time_limit=remaining, oracle_mode='standard')
            try_and_update(a_opt, state, "fw_std")

        # Multi-level FW
        remaining = min(n_phase2 * 0.25, deadline - time.monotonic() - 450)
        if remaining > 20:
            print("  Multi-level FW t={:.0f}s".format(remaining))
            a_opt, _ = fw_lp_optimize(state[1], time_limit=remaining,
                                      oracle_mode='multilevel', top_k=10, k_gap_frac=0.015)
            try_and_update(a_opt, state, "fw_ml")

        # Soft FW
        remaining = min(n_phase2 * 0.2, deadline - time.monotonic() - 400)
        if remaining > 20:
            print("  Soft FW t={:.0f}s".format(remaining))
            a_opt, _ = fw_lp_optimize(state[1], time_limit=remaining, oracle_mode='soft')
            try_and_update(a_opt, state, "fw_soft")

        # Perturbed restarts
        for scale in [0.003, 0.008, 0.015, 0.03]:
            remaining = min(n_phase2 * 0.08, deadline - time.monotonic() - 350)
            if remaining < 12:
                break
            f_pert = project_simplex(state[1] + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 40)
            a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
            try_and_update(a_opt, state, "fw_p_{}".format(scale))

    print("  Phase 2 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase2))

    # === Phase 3: LSE + Fourier (~400s) ===
    print("\n=== Phase 3: LSE + Fourier Refinement ===")
    t_phase3 = time.monotonic()
    n_phase3 = 400.0

    if state[1] is not None:
        # LSE with different settings
        lse_configs = [
            (50, 500, 0.02),
            (100, 1000, 0.01),
            (200, 2000, 0.005),
        ]
        for alpha_s, alpha_e, lr_s in lse_configs:
            remaining = min(n_phase3 * 0.2, deadline - time.monotonic() - 250)
            if remaining < 12:
                break
            t_alloc = min(remaining, 100)
            print("  LSE a=[{},{}], lr={}, t={:.0f}s".format(alpha_s, alpha_e, lr_s, t_alloc))
            a_opt, _ = lse_optimize(state[1], time_limit=t_alloc,
                                    alpha_start=alpha_s, alpha_end=alpha_e, lr_start=lr_s)
            try_and_update(a_opt, state, "lse_{}_{}".format(alpha_s, alpha_e))

        # Fourier refinement
        remaining = min(n_phase3 * 0.25, deadline - time.monotonic() - 200)
        if remaining > 15:
            t_alloc = min(remaining, 150)
            print("  Fourier M=15, t={:.0f}s".format(t_alloc))
            a_opt, _ = fourier_refine(state[1], time_limit=t_alloc, n_modes=15)
            try_and_update(a_opt, state, "fourier_15")

        # More Fourier modes
        remaining = min(n_phase3 * 0.2, deadline - time.monotonic() - 150)
        if remaining > 15:
            t_alloc = min(remaining, 100)
            print("  Fourier M=25, t={:.0f}s".format(t_alloc))
            a_opt, _ = fourier_refine(state[1], time_limit=t_alloc, n_modes=25)
            try_and_update(a_opt, state, "fourier_25")

    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase3))

    # === Phase 4: Peak shaving + final polish (~380s) ===
    print("\n=== Phase 4: Final Polish ===")
    t_phase4 = time.monotonic()
    n_phase4 = 380.0

    if state[1] is not None:
        # Peak shaving
        remaining = min(n_phase4 * 0.25, deadline - time.monotonic() - 120)
        if remaining > 10:
            t_alloc = min(remaining, 100)
            print("  Peak shaving t={:.0f}s".format(t_alloc))
            a_opt, _ = peak_shaving(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "peak_shave")

        # Final FW
        remaining = min(n_phase4 * 0.2, deadline - time.monotonic() - 100)
        if remaining > 10:
            t_alloc = min(remaining, 80)
            print("  Final FW t={:.0f}s".format(t_alloc))
            a_opt, _ = fw_lp_optimize(state[1], time_limit=t_alloc, oracle_mode='standard')
            try_and_update(a_opt, state, "final_fw")

        # LSE very high alpha (almost exact max)
        remaining = min(n_phase4 * 0.2, deadline - time.monotonic() - 70)
        if remaining > 10:
            t_alloc = min(remaining, 80)
            print("  LSE high-alpha t={:.0f}s".format(t_alloc))
            a_opt, _ = lse_optimize(state[1], time_limit=t_alloc,
                                    alpha_start=500, alpha_end=5000, lr_start=0.003)
            try_and_update(a_opt, state, "lse_high")

        # Alternating: peak shave -> FW -> peak shave
        remaining = min(n_phase4 * 0.15, deadline - time.monotonic() - 40)
        if remaining > 10:
            t_sub = remaining / 3
            print("  Alternating polish t={:.0f}s each".format(t_sub))
            a1, _ = peak_shaving(state[1], time_limit=t_sub)
            try_and_update(a1, state, "alt_ps1")
            a2, _ = fw_lp_optimize(state[1], time_limit=t_sub)
            try_and_update(a2, state, "alt_fw")
            a3, _ = peak_shaving(state[1], time_limit=t_sub)
            try_and_update(a3, state, "alt_ps2")

        # Tiny perturbations with quick polish
        remaining = min(n_phase4 * 0.1, deadline - time.monotonic() - 15)
        if remaining > 5:
            for pi in range(4):
                rem = deadline - time.monotonic() - 8
                if rem < 3:
                    break
                noise = rng.standard_normal(len(state[1])) * 0.0003
                f_pert = project_simplex(state[1] + noise)
                a_opt, _ = lse_optimize(f_pert, time_limit=min(rem, 6),
                                        alpha_start=300, alpha_end=1000, lr_start=0.004)
                try_and_update(a_opt, state, "pert_{}".format(pi))

    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase4))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(512) / 512
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
