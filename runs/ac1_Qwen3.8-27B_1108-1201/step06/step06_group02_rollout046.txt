Hybrid optimization combining spectral flattening, fast coordinate descent,
and smoothed subgradient methods to minimize 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_flatten(a0, time_limit, alpha=0.5):
    """Spectral flattening method (Fienup-like)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()  # next power of 2 >= 2n-1
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        a_padded = np.zeros(L)
        a_padded[:n] = a
        A = np.fft.rfft(a_padded)
        
        # Flatten the spectrum
        mag = np.abs(A)
        mag_mean = mag.mean()
        # Soft flatten: compress dynamic range
        mag_new = mag ** (1 - alpha) * mag_mean ** alpha
        # Or: mag_new = np.minimum(mag, mag_mean * factor)
        # Let's use a power-law compression
        phase = A / (mag + 1e-30)
        A_new = mag_new * phase
        
        a_new = np.fft.irfft(A_new, L)[:n]

        s_new = a_new.sum()
        if s_new < 1e-15:

        a_new = a_new / s_new

def coord_descent_peak(a0, time_limit):
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    # Compute initial convolution
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step_size = 0.01
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        
        k_star = int(np.argmax(conv))

        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        if js > je:
            break
        
        affected = np.arange(js, je + 1)
        contributions = a[affected] * a[k_star - affected]
        j_star = affected[np.argmax(contributions)]

        # We want an index whose removal from peaks is minimal
        # Simple heuristic: move to the index with the smallest value

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
- Spectral flattening (Fienup-like method in frequency domain)
- Fast coordinate descent with adaptive mass transfer
- Smoothed subgradient with momentum
- Multi-n exploration with diverse initializations
- Final precision polishing
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_flatten(a0, time_limit, alpha_init=0.4):
    """Spectral flattening: iteratively compress the dynamic range of the spectrum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    alpha = alpha_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        A = np.fft.rfft(a_padded)
        mag = np.abs(A)

        # Compress dynamic range of the spectrum
        mag_max = mag.max()
        mag_min = mag[mag > 1e-15].min() if np.any(mag > 1e-15) else 1e-15
        # Power-law compression: large values get reduced more
        mag_compressed = mag_max * (mag / mag_max) ** (1.0 + alpha)
        # Ensure we don't go below a floor
        mag_compressed = np.maximum(mag_compressed, mag_min * 0.1)

        phase = A / (mag + 1e-30)
        A_new = mag_compressed * phase

        a_new = np.fft.irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                alpha = 0.2 + 0.6 * (1.0 - 0.9 * (alpha - 0.2) / 0.6)
            if no_improve > 80:
                rng = np.random.default_rng(it * 7919 + 13)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                alpha = alpha_init
                no_improve = 0

    return best_a, best_obj


def coord_descent(a0, time_limit, step_frac=0.1):
    """Fast coordinate descent: move mass from peak contributors to cold spots."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    step_scale = 1.0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        k_star = int(np.argmax(conv))
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        if js > je:
            break

        affected = np.arange(js, je + 1)
        contributions = a[affected] * a[k_star - affected]
        j_star = int(affected[np.argmax(contributions)])

        # Choose target: minimize impact on other peaks
        # Score each candidate m by the max convolution increase
        # Approximation: use the value of a at positions that would be affected
        # For speed, just pick from the smallest-valued indices not near peaks
        # Top-10 smallest values
        sorted_idx = np.argsort(a)[:20]
        target_found = False
        for m in sorted_idx:
            if m == j_star:
                continue
            # Check: does adding to m increase any currently-high convolution value?
            # conv[k] increases by delta * a[k-m] for valid k
            # The max increase is delta * max(a) at best case
            # But we want to avoid positions where conv is already high
            # Quick check: conv values in [m, m+n-1] range
            conv_m_start = max(0, m)
            conv_m_end = min(2 * n - 2, m + n - 1)
            local_max = conv[conv_m_start:conv_m_end + 1].max()
            if local_max < best_obj * 0.95:  # "cold" enough
                target = m
                target_found = True
                break

        if not target_found:
            # Fallback: use smallest value index that's not j_star
            target = int(sorted_idx[0])
            if target == j_star:
                target = int(sorted_idx[1]) if len(sorted_idx) > 1 else 0

        delta = step_frac * a[j_star] * step_scale
        if delta < 1e-12:
            break
        delta = min(delta, a[j_star] * 0.4)

        a[j_star] -= delta
        a[target] += delta
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-15:
            break
        a = a / a_sum

        conv = fast_convolve(a)
        obj = conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            # Try to increase step
            step_scale = min(step_scale * 1.05, 3.0)
        else:
            no_improve += 1
            step_scale = max(step_scale * 0.95, 0.1)
            if no_improve > 200:
                # Perturb to escape
                rng = np.random.default_rng(it * 4567 + 3)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(a)
                best_obj = conv.max()
                best_a = a.copy()
                step_scale = 1.0
                no_improve = 0

    return best_a, best_obj


def smoothed_subgrad(a0, time_limit, T_init=0.001, lr_init=0.03, top_k=5):
    """Smoothed subgradient descent with softmax-weighted peaks and momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    v = np.zeros(n)
    T = T_init
    lr = lr_init
    mom = 0.9

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        # Get top-k peaks with softmax weights
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        weights = np.exp((pv - pv.max()) / T)
        weights = weights / weights.sum()

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-10:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]

        v = mom * v - lr * sg
        a_new = project_simplex(a + v)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        mom = min(0.98, mom + 0.0002)

        if no_improve > 30:
            T *= 1.08
        if no_improve > 60:
            T *= 0.9
        if no_improve > 200:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31 - 1))
            noise = 0.001 + 0.004 * min(1.0, (no_improve - 200) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            T = T_init
            lr = lr_init * 0.7
            mom = 0.9
            no_improve = 0

    return best_a, best_obj


def multi_scale_coord_descent(a0, time_limit):
    """Coordinate descent with multiple step sizes for robust convergence."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    t0 = time.monotonic()
    remaining = time_limit - (time.monotonic() - t0)

    # Coarse phase with larger steps
    t1 = min(remaining * 0.4, remaining)
    a, obj1 = coord_descent(a, time_limit=t1, step_frac=0.15)

    remaining = time_limit - (time.monotonic() - t0)
    if remaining < 5:
        return a, obj1

    # Medium phase
    t2 = min(remaining * 0.4, remaining)

    remaining = time_limit - (time.monotonic() - t0)
    if remaining < 5:
        return a, obj2

    # Fine phase
    a, obj3 = coord_descent(a, time_limit=remaining, step_frac=0.02)

    return a, obj3


def diverse_initializations(n, parent, rng):
    """Generate diverse initial sequences."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Resampled parent
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Edge-weighted (U-shape)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.12 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine modulation
    for freq in [2, 3, 5]:
        h = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * x / n)
        h = np.maximum(h, 0.01)
        inits.append(h / h.sum())

    # Two-peak structure
    p1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.08 * n)) ** 2)
    p2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.08 * n)) ** 2)
    two_peak = p1 + p2
    inits.append(two_peak / two_peak.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.1))

    # Block structure
    block_size = max(1, n // 8)
    block = np.zeros(n)
    block[0:block_size] = 1.0
    block[n - block_size:] = 1.0
    inits.append(block / block.sum())

    return inits
