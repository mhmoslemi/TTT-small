import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

def solve_lp_oracle(f, M, n, row_idx, col_idx):
    c = -np.ones(n)
    k_arr = row_idx
    i_arr = col_idx
    vals = f[k_arr - i_arr]
    A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def coordinate_descent(a0, time_limit, verbose=False):

    a = np.maximum(a0.astype(np.float64), 0.0)

        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]

        # Transfer epsilon from i to j: a[i] -= eps, a[j] += eps
        # Effect on conv[k]: eps * (a[k-j] - a[k-i]) for the peak position
        # We want to reduce conv[peak_k], so we want a[peak_k - j] > a[peak_k - i]
        # i.e., move mass from position i to position j where a[peak_k-j] > a[peak_k-i]
        
        # Also consider secondary peaks
        top_peaks = np.argsort(conv)[-5:][::-1]
        
        best_delta = 0
        best_i, best_j = -1, -1
        best_eps = 0.0
        
        # Sample pairs (for efficiency)
        rng = np.random.default_rng(it + 42)
        n_samples = min(2000, n * n)
        
        for _ in range(n_samples):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if a[i] < 1e-10:
                continue
            
            # Effect on top peaks
            effect = 0
            for pk in top_peaks:
                # conv[pk] changes by eps * (a[pk-j] - a[pk-i]) (approx)
                # where a[pk-j] means a at position (pk-j) mod... no, it's the actual index
                # conv[pk] = sum_m a[m] * a[pk-m]
                # d(conv[pk])/da[j] = a[pk-j] if 0 <= pk-j < n
                # d(conv[pk])/da[i] = a[pk-i] if 0 <= pk-i < n
                
                effect_j = a[pk - j] if 0 <= pk - j < n else 0.0
                effect_i = a[pk - i] if 0 <= pk - i < n else 0.0
                
                # We change a[j] += eps, a[i] -= eps
                # d(conv[pk]) = eps * effect_j - eps * effect_i = eps * (effect_j - effect_i)

                effect += (effect_i - effect_j)  # This is -d(conv[pk])/d(eps)
            
            # Also consider effect on secondary peaks (we don't want to increase them)
            # For simplicity, just use the effect on the main peak
            if effect > best_delta:
                # Determine max epsilon
                max_eps = a[i] * 0.5  # Don't transfer more than half
                best_delta = effect
                best_i, best_j = i, j
                best_eps = max_eps

                rng2 = np.random.default_rng((it * 9973 + 5) % (2**31 - 1))
                a = project_simplex(best_a + rng2.standard_normal(n) * 0.003)
                no_improve = 0
            continue
        
        # Line search on epsilon
        lo, hi = 0.0, best_eps
        best_v = peak_val
        best_e = 0.0
        for _ in range(15):
            mid = (lo + hi) / 2
            a_new = a.copy()
            a_new[best_i] -= mid
            a_new[best_j] += mid
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                hi = mid
                continue
            a_new = a_new / s
            v = fast_convolve(a_new).max()
            if v < best_v:
                best_v = v
                best_e = mid
                lo = mid
            else:
                hi = mid
        
        if best_e > 1e-10:
            a[best_i] -= best_e
            a[best_j] += best_e
            a = np.maximum(a, 0)
            a = a / a.sum()
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 100:

    return best_a, best_obj

def freq_domain_smooth(a0, time_limit, verbose=False):
    """Reduce spectral peaks by smoothing in frequency domain, then project."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        # Compute FFT
        A = np.fft.rfft(a)
        
        # The power spectrum is |A|^2
        # We want to reduce the peak of ifft(|A|^2)
        # In frequency domain, we can try to "flatten" |A|^2
        # but we need to maintain A[0] = sum(a) = 1 and conjugate symmetry
        
        # Strategy: reduce the magnitude of the largest frequency components
        # (except DC), which should reduce the peak in time domain
        
        power = np.abs(A) ** 2
        total_power = power.sum()
        
        # Soft threshold: reduce large components
        # This is like a "box" constraint on the power spectrum
        threshold = np.percentile(power[1:], 85)  # Keep top 15% as is, reduce rest
        
        A_new = A.copy()
        for k in range(1, len(A)):
            if power[k] > threshold:
                # Reduce this component
                reduction = (power[k] - threshold) * 0.1
                scale = np.sqrt(max(0, power[k] - reduction) / power[k]) if power[k] > 0 else 1.0
                A_new[k] *= scale
        
        # Convert back
        a_new = np.fft.irfft(A_new, n)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            no_improve += 1
            continue
        a_new = a_new / s
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            # Try less aggressive reduction
            no_improve += 1
        
        if no_improve > 30:

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Adaptive SCA with LP oracle and binary line search
- Pairwise coordinate descent for fine-tuning
- Frequency-domain spectral flattening
- Multi-n exploration with diverse initializations
- Simulated annealing restarts

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(n):
    """Precompute the sparsity pattern for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_oracle(f, M, n, row_idx, col_idx):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g (unnormalized) or None.
    """
    c = -np.ones(n)
    k_arr = row_idx
    i_arr = col_idx
    vals = f[k_arr - i_arr]
    A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

    """
    SCA optimization with adaptive M and binary line search.
    """

    if s < 1e-15:

    else:
        f = f / s
    n = len(f)
    row_idx, col_idx = build_lp_matrix(n)
    conv = fast_convolve(f)
    M = conv.max() * 1.001 if M_init is None else M_init
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    M_decrease = 0.999
    M_increase = 1.005
    
    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        g_raw = solve_lp_oracle(f, M, n, row_idx, col_idx)
        
        if g_raw is None:
            M *= M_increase
            no_improve += 1
            if no_improve > 20:
                M = fast_convolve(f).max() * 1.002
                no_improve = 0
            continue
        
        g = g_raw / g_raw.sum()
        
        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 15:
                rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
                f = project_simplex(f + rng.standard_normal(n) * 0.002)
                M = fast_convolve(f).max() * 1.002
                no_improve = 0
            continue
        
        # Binary line search for best alpha
        best_val = fast_convolve(f).max()
        best_ft = f.copy()
        best_alpha = 0.0
        
        # Coarse search
        alphas = [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0]
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_ft = ft.copy()
                best_alpha = alpha
        
        # Fine search around best_alpha
        if best_alpha > 0:
            lo = max(0.001, best_alpha * 0.5)
            hi = min(1.0, best_alpha * 2.0)
            for _ in range(5):
                mid = (lo + hi) / 2
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    hi = mid
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_ft = ft.copy()
                    lo = mid
                else:
                    hi = mid
        
        if best_val < best_obj - 1e-15:
            f = best_ft
            best_obj = best_val
            best_f = f.copy()
            if it % 5 == 0:
                M = best_obj * 0.9995
            else:
                M = min(M, best_obj * 1.001)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 5 == 0:
                M = max(M * M_decrease, best_obj * 0.999)
        
        if no_improve > 30:
            rng = np.random.default_rng((it * 31831 + 7) % (2**31 - 1))
            scale = 0.001 + 0.003 * min(1.0, (no_improve - 30) / 50.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max() * 1.002
            no_improve = 0
        
        if verbose and it % 500 == 0:
            print(f"    SCA it={it}, obj={best_obj:.12f}, M={M:.12f}")
    
    return best_f, best_obj


def coordinate_descent(a0, time_limit, verbose=False):
    """
    Pairwise coordinate descent: transfer mass between pairs to reduce peak.
    Uses gradient information to select promising pairs.
    """

    a = a / a.sum()
    n = len(a)
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]
        
        # Get top peaks for secondary constraint checking
        top_k = min(5, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peaks_sorted = peaks[np.argsort(conv[peaks])[::-1]]
        
        # Find best pair (i, j) to swap
        # For each pair, the effect on the peak is:
        #   delta = eps * (a[peak_k - j] - a[peak_k - i])
        # We want delta < 0, so a[peak_k - j] < a[peak_k - i]
        # i.e., transfer from position i to position j where a[peak_k - i] > a[peak_k - j]
        
        # Precompute the "influence" of each position on the peak
        influence = np.zeros(n)
        for pk in peaks_sorted:
            for j in range(n):
                if 0 <= pk - j < n:
                    influence[j] += a[pk - j]

        rng = np.random.default_rng(it * 31 + 17)
        n_samples = min(5000, n * n)
        
        best_score = 0.0

            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j or a[i] < 1e-12:
                continue
            
            # Effect on main peak: eps * (a[peak_k - j] - a[peak_k - i])
            # We transfer from i to j (a[i] -= eps, a[j] += eps)
            # d(conv[peak_k])/d(eps) = a[peak_k - j] - a[peak_k - i]
            # We want this to be negative
            
            val_j = a[peak_k - j] if 0 <= peak_k - j < n else 0.0
            val_i = a[peak_k - i] if 0 <= peak_k - i < n else 0.0
            main_effect = val_i - val_j  # Positive means transfer reduces peak
