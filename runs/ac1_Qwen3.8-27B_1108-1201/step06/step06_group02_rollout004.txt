def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_mv(p, a):
    return np.correlate(p, a, mode='valid')

def build_lp_matrix(f, active_set):
    """Build the constraint matrix for the LP oracle."""
    n = len(f)
    A = np.zeros((len(active_set), n))
    for idx, k in enumerate(active_set):
        # (f*g)[k] = sum_i f[k-i]*g[i] where 0<=k-i<n and 0<=i<n
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # g[i] appears with coefficient f[k-i]
        # i ranges from max(0, k-(n-1)) to min(n-1, k)
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        A[idx, i_lo:i_hi+1] = f[k - i_hi:k - i_lo + 1][::-1]
    return A

            if val < best_val - 1e-15:

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_mv(p, a):
    """Compute the Hankel matrix-vector product: (M @ a)[i] = sum_k p[k]*a[k-i]."""
    return np.correlate(p, a, mode='valid')

def fw_lazy_lp(f0, time_limit, k_init=50, k_max=300):
    """Frank-Wolfe with lazy LP oracle (top-k most violated constraints)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

        f = np.ones(len(f)) / len(f)

    M = conv.max()

    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Adaptive k
        k_act = min(k_max, k_init + it // 5)
        k_eff = min(k_act, len(conv))
        
        # Top-k constraints
        top_idx = np.argpartition(conv, -k_eff)[-k_eff:]
        top_idx = sorted(top_idx.tolist())
        
        # Build LP
        c_obj = -np.ones(n)
        n_cons = len(top_idx)
        A_ub = np.zeros((n_cons, n))
        for idx, k in enumerate(top_idx):
            i_lo = max(0, k - (n - 1))
            i_hi = min(n - 1, k)
            A_ub[idx, i_lo:i_hi + 1] = f[k - i_hi:k - i_lo + 1][::-1]
        b_ub = np.full(n_cons, M)
        bounds = [(0, None)] * n
        
        try:
            lp_time = max(0.5, min(remaining / 10, 10.0))
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': lp_time})
            if not res.success:
                no_improve += 1
                continue
            g = np.maximum(res.x, 0.0)
        except Exception:
            no_improve += 1
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            no_improve += 1
            continue

                ft = np.maximum(ft, 0.0)

                    best_val = val
                    best_alpha = alpha

                f = (1.0 - best_alpha) * f + best_alpha * g
                f = np.maximum(f, 0.0)

                if best_val < best_obj - 1e-14:
                    best_obj = best_val

                    no_improve = 0
                else:

            f = project_simplex(best_f + rng.standard_normal(n) * scale)

            M = conv.max()
            no_improve = 0
    
    return best_f, best_obj
