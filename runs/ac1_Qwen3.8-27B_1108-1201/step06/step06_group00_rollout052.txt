"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_weights(b, beta):
    """Compute softmax weights on convolution values for LSE smoothing."""
    bc = beta * b
    bc_max = bc.max()
    if bc_max > 700 / beta:
        idx = np.argmax(b)
        w = np.zeros(len(b))
        w[idx] = 1.0
        return w
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.ones(len(b)) / len(b)
    return exp_bc / total


def compute_subgradient(a, peaks, weights):
    """Compute weighted sum of gradients of b[k] for k in peaks."""
    n = len(a)
    grad = np.zeros(n)
    for idx, k in enumerate(peaks):
        k = int(k)
        w = weights[idx]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        contrib = 2.0 * a[k - i_range]
        # Fix diagonal: where k-i_range == i_range, i.e., k - i == i => i = k//2
        diag_pos = []
        for j, i in enumerate(i_range):
            if k == 2 * i:
                diag_pos.append(j)
        for j in diag_pos:
            contrib[j] = a[i_range[j]]
        grad[i_range] += w * contrib
    return grad


def nesterov_lse_optimize(a_init, time_limit, beta_start=5.0, beta_end=20000.0,
                           top_k=40, verbose=True):
    """Nesterov-accelerated LSE subgradient with backtracking line search."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    b = fast_convolve(a)
    best_obj = b.max()
    best_a = a.copy()

    beta = beta_start
    lr = 1.0 / np.sqrt(n)
    momentum = 0.0
    y = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    nesterov_t = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Update beta
        progress = min(1.0, it / max(1, int(time_limit / 0.5)))
        beta = beta_start * (beta_end / beta_start) ** progress

        # Compute LSE weights (focus on top peaks)
        k_top = min(top_k + 10, 2 * n - 1)
        peak_idx = np.argpartition(b, -k_top)[-k_top:]
        w_full = lse_weights(b, beta)
        w_peaks = w_full[peak_idx]
        w_sum = w_peaks.sum()
        if w_sum > 1e-15:
            w_peaks = w_peaks / w_sum
        else:
            w_peaks = np.ones(len(peak_idx)) / len(peak_idx)

        # Compute gradient at y (Nesterov lookahead point)
        b_y = fast_convolve(y)
        w_full_y = lse_weights(b_y, beta)
        peak_idx_y = np.argpartition(b_y, -k_top)[-k_top:]
        w_peaks_y = w_full_y[peak_idx_y]
        ws = w_peaks_y.sum()
        if ws > 1e-15:
            w_peaks_y = w_peaks_y / ws
        else:
            w_peaks_y = np.ones(len(peak_idx_y)) / len(peak_idx_y)

        grad = compute_subgradient(y, peak_idx_y, w_peaks_y)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Backtracking line search
        step = lr
        improved = False
        for _ in range(15):
            candidate = y - step * grad
            candidate = project_simplex(candidate)
            b_new = fast_convolve(candidate)
            obj_new = b_new.max()
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = candidate.copy()
                improved = True
            if obj_new < b_y.max() * (1 - 1e-4):
                break
            step *= 0.5

        if improved:
            # Nesterov momentum update
            new_t = (1.0 + np.sqrt(1.0 + 4.0 * nesterov_t ** 2)) / 2.0
            momentum_factor = (nesterov_t - 1.0) / new_t
            a_new = candidate
            y = a_new + momentum_factor * (a_new - y)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy > 1e-15:
                y /= sy
            a = a_new
            nesterov_t = new_t
            no_improve = 0
            lr = min(lr * 1.002, 5.0 / np.sqrt(n))
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.85
            if no_improve > 50:
                # Restart from best with perturbation
                rng = np.random.default_rng(it * 104729)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                y = a.copy()
                nesterov_t = 1.0
                lr = max(lr, 1.0 / np.sqrt(n))
                no_improve = 0
            update_y = (no_improve % 5 == 0)
            if update_y:
                y = a.copy()
                nesterov_t = 1.0

        # Update b
        b = fast_convolve(a)

def eg_optimize(a_init, time_limit, beta_start=20.0, beta_end=5000.0, top_k=50):
    """Exponential Gradient (mirror descent) method for non-negative variables."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    b = fast_convolve(a)
    best_obj = b.max()
    best_a = a.copy()

    beta = beta_start
    lr_base = 2.0 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        progress = min(1.0, it / max(1, int(time_limit / 0.3)))
        beta = beta_start * (beta_end / beta_start) ** progress

        k_top = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(b, -k_top)[-k_top:]
        w_full = lse_weights(b, beta)
        w_peaks = w_full[peak_idx]
        ws = w_peaks.sum()
        if ws > 1e-15:
            w_peaks = w_peaks / ws
        else:
            w_peaks = np.ones(k_top) / k_top

        grad = compute_subgradient(a, peak_idx, w_peaks)

        # Exponential update (mirror descent with KL divergence)
        step = lr_base
        improved = False
        for _ in range(10):
            log_update = -step * grad
            log_update -= log_update.max()  # numerical stability
            update = np.exp(log_update)
            a_new = a * update
            s_new = a_new.sum()
            if s_new < 1e-15:
                step *= 0.5
                continue
            a_new = a_new / s_new
            b_new = fast_convolve(a_new)
            obj_new = b_new.max()
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                improved = True
            if obj_new < b.max() * (1 - 1e-4):
                break
            step *= 0.5

        if improved:
            a = a_new
            b = b_new
            no_improve = 0
            lr_base = min(lr_base * 1.001, 5.0 / np.sqrt(n))
        else:
            no_improve += 1
            if no_improve >= 15:
                lr_base *= 0.85
            if no_improve > 60:
                a = best_a.copy()
                b = fast_convolve(a)
                lr_base = max(lr_base, 1.0 / np.sqrt(n))
                no_improve = 0

    return best_a, best_obj

    sweep = 0
    no_improve_sweeps = 0

    delta_fracs = np.array([-1.0, -0.5, -0.3, -0.1, -0.05, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0])

                break

            if ai < 1e-12:
                continue

            # b[k] involves a[i] when k-i is in [0, n-1], i.e., k in [i, i+n-1]

            # a[k-i] for k in [k_lo, k_hi] is a[0], a[1], ..., a[n-1]
            contrib = 2.0 * a[0:n]  # a[k - i] for k = i, i+1, ..., i+n-1

                if k_lo <= diag_k <= k_hi:
                    pos = diag_k - k_lo
                    contrib[pos] = a[i]  # instead of 2*a[i]

            # Try different deltas
            best_delta = 0.0
            best_peak = b[k_range].max()

            # Also consider small absolute deltas for coordinates with small values
            abs_deltas = np.array([ai * 0.01, ai * 0.05, ai * 0.1, -ai * 0.01, -ai * 0.05])
            all_deltas = np.concatenate([ai * delta_fracs, abs_deltas])
            all_deltas = all_deltas[all_deltas > -ai + 1e-15]  # ensure a[i]+delta >= 0
            all_deltas = np.unique(all_deltas)

            for delta in all_deltas:
                if delta == 0:
                    continue
                # Update b[k] for affected k values
                b_new_vals = b[k_range] + delta * contrib
                new_peak = b_new_vals.max()
                if new_peak < best_peak - 1e-15:
                    best_peak = new_peak
                    best_delta = delta

            if best_delta != 0:
                # Apply the update
                b[k_range] += best_delta * contrib
                a[i] += best_delta
                if a[i] < 0:
                    a[i] = 0
                    # Recompute b for correctness
                    b = fast_convolve(a)
                sweep_improved = True

        # End of sweep: check improvement
        current_obj = b.max()
        if current_obj < best_obj - 1e-15:
            best_obj = current_obj
            best_a = a.copy()
            no_improve_sweeps = 0
        elif current_obj < sweep_start_obj - 1e-10:
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1

        # Periodic renormalization
        if sweep % 50 == 0:
            s = a.sum()
            if s > 1e-15:
                a = a / s
                b = fast_convolve(a)
                if b.max() < best_obj - 1e-15:
                    best_obj = b.max()
                    best_a = a.copy()

        # If stuck, try a perturbation
        if no_improve_sweeps > 30:
            # Perturb and restart
            pert_strength = 0.001 + 0.002 * min(1.0, no_improve_sweeps / 50.0)
            a_pert = best_a + rng.standard_normal(n) * pert_strength
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            b = fast_convolve(a_pert)
            if b.max() < best_obj + 0.01:  # accept if not much worse
                a = a_pert
                no_improve_sweeps = 0
            else:
                a = best_a.copy()
                b = fast_convolve(a)
                no_improve_sweeps = 0
                # Try different delta set
                delta_fracs = np.array([-0.5, -0.2, -0.1, -0.05, -0.02,
                                        0.02, 0.05, 0.1, 0.2, 0.5])

        # Early stop check
        if no_improve_sweeps > 100:
            break

    return best_a, best_obj


def pair_coordinate_descent(a_init, time_limit, n_rounds=500):
    """Try pairs of coordinates (i, j) and transfer mass between them."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    b = fast_convolve(a)
    best_obj = b.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    # Precompute which peaks are currently active
    top_peaks = np.argsort(b)[-20:]

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        # Pick a peak to reduce
        k = int(top_peaks[rng.integers(len(top_peaks))])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo >= hi:
            continue

        # Find the pair (i, j) with i+j=k that maximizes a[i]*a[j]
        i_vals = np.arange(lo, hi + 1)
        prod = a[i_vals] * a[k - i_vals]
        i_best = i_vals[np.argmax(prod)]
        j_best = k - i_best
        if j_best < 0 or j_best >= n or j_best == i_best:
            continue

        # Try to reduce a[i_best] and increase a[random_other]
        # The other position should not contribute to this peak
        for trial in range(5):
            j_dest = int(rng.integers(n))
            if j_dest == i_best:
                continue
            # Check if j_dest + i_best == k (would still contribute)
            # If so, the transfer might not help
            for frac in [0.05, 0.1, 0.2, 0.3]:
                delta = a[i_best] * frac
                if delta < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i_best] -= delta
                a_trial[j_dest] += delta
                if a_trial[i_best] < 0:
                    continue
                b_trial = fast_convolve(a_trial)
                if b_trial.max() < best_obj - 1e-15:
                    best_obj = b_trial.max()
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    b = b_trial
                    break

        # Update top peaks periodically
        if rnd % 50 == 0:
            top_peaks = np.argsort(b)[-20:]

    return best_a, best_obj


def freq_perturb(a, rng, strength=0.03):
    """Perturb in frequency domain for diversification."""
    n = len(a)
    F = np.fft.rfft(a)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(F_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new


def make_initialization(n, style, rng):
    """Create various initializations."""
    if style == 'uniform':
        return np.ones(n) / n
    elif style == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x / 2) ** 2
        return a / a.sum()
    elif style == 'gaussian':
        x = np.linspace(-3, 3, n)
        a = np.exp(-x ** 2 / 4)
        return a / a.sum()
    elif style == 'gap':
        # Alternating high/low pattern
        a = np.ones(n)
        a[::2] = 2.0
        a = np.maximum(a + rng.standard_normal(n) * 0.1, 0)
        return a / a.sum()
    elif style == 'broadened':
        # Slightly broader than uniform in spectral domain
        a = np.ones(n) / n
        F = np.fft.rfft(a)
        # Boost low frequencies
        boost = 1.0 + 0.3 * np.exp(-np.arange(len(F)) / (n * 0.1))
        F *= boost
        a = np.fft.irfft(F, n)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a /= s
        else:
            a = np.ones(n) / n
        return a
    else:
        return np.ones(n) / n


def try_update(a, best_tracker):
    """Evaluate and update best tracker."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.10f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    """Interpolate normalized sequence to new length."""
    if len(f) == new_n:
        return f.copy()
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Main optimization: multi-phase search combining Nesterov LSE,
    coordinate descent, EG, and diversification.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all():
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
            try_update(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768.0
        try_update(parent, best_tracker)

    n_parent = len(parent)

    # Define phases with time allocation
    total_time = float(budget_s) - 30

    # Phase 1: Nesterov LSE on parent (30% of time)
    print("\n=== Phase 1: Nesterov LSE optimization ===")
    t1 = total_time * 0.25
    t1_end = time.monotonic() + t1
    print(f"  n={n_parent}, time={t1:.0f}s")
    f_opt, obj = nesterov_lse_optimize(parent, time_limit=t1,
                                        beta_start=5.0, beta_end=10000.0,
                                        top_k=40)
    try_update(f_opt, best_tracker)

    # Phase 2: Coordinate descent polish (25% of time)
    print(f"\n=== Phase 2: Coordinate descent polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t2 = total_time * 0.25
    t2_end = time.monotonic() + t2
    print(f"  n={len(current_best)}, time={t2:.0f}s")
    f_opt, obj = coordinate_descent_polish(current_best, time_limit=t2)
    try_update(f_opt, best_tracker)

    # Phase 3: EG optimization (15% of time)
    print(f"\n=== Phase 3: Exponential Gradient ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t3 = total_time * 0.15
    t3_end = time.monotonic() + t3
    print(f"  n={len(current_best)}, time={t3:.0f}s")
    f_opt, obj = eg_optimize(current_best, time_limit=t3,
                              beta_start=20.0, beta_end=5000.0, top_k=50)
    try_update(f_opt, best_tracker)

    # Phase 4: Iterative refinement cycles (25% of time)
    print(f"\n=== Phase 4: Iterative refinement cycles ===")
    t4_end = deadline - 50
    cycle = 0
    methods_cycle = ['cd', 'nesterov', 'eg', 'cd', 'pair', 'cd']

    while time.monotonic() < t4_end and cycle < 30:
        cycle += 1
        current_best = best_tracker[1] if best_tracker[1] is not None else parent
        remaining = t4_end - time.monotonic()
        if remaining < 15:
            break

        method = methods_cycle[cycle % len(methods_cycle)]
        t_alloc = min(remaining * 0.3, 120)
        if t_alloc < 10:
            break

        if method == 'cd':
            print(f"  Cycle {cycle}: CoordDesc n={len(current_best)}, t={t_alloc:.0f}s")
            f_opt, _ = coordinate_descent_polish(current_best, time_limit=t_alloc)
        elif method == 'nesterov':
            print(f"  Cycle {cycle}: Nesterov n={len(current_best)}, t={t_alloc:.0f}s")
            beta_s = 50.0 + 50.0 * cycle
            f_opt, _ = nesterov_lse_optimize(current_best, time_limit=t_alloc,
                                              beta_start=beta_s, beta_end=20000.0, top_k=30)
        elif method == 'eg':
            print(f"  Cycle {cycle}: EG n={len(current_best)}, t={t_alloc:.0f}s")
            f_opt, _ = eg_optimize(current_best, time_limit=t_alloc,
                                    beta_start=50.0 + 20.0 * cycle, beta_end=8000.0, top_k=40)
        elif method == 'pair':
            print(f"  Cycle {cycle}: PairCD n={len(current_best)}, t={t_alloc:.0f}s")
            f_opt, _ = pair_coordinate_descent(current_best, time_limit=t_alloc, n_rounds=200)
            try_update(f_opt, best_tracker)
            continue  # Skip the common try_update below for pair

        try_update(f_opt, best_tracker)

    # Phase 5: Diversification - try different initializations (10% of time)
    print(f"\n=== Phase 5: Diversification ===")
    t5_end = deadline - 30
    initializations = ['cosine', 'gap', 'broadened', 'gaussian']

    for init_style in initializations:
        if time.monotonic() > t5_end - 10:
            break
        t_alloc = min(40, t5_end - time.monotonic() - 5)
        if t_alloc < 10:
            break

        # Start from parent but perturb with new initialization
        n_try = n_parent
        a_init = make_initialization(n_try, init_style, rng)
        # Blend with parent
        blend = 0.3 + 0.1 * rng.random()
        a_init = (1 - blend) * parent + blend * a_init
        a_init = np.maximum(a_init, 0)
        a_init = a_init / a_init.sum()

        print(f"  Init: {init_style}, blend={blend:.2f}, t={t_alloc:.0f}s")
        f_opt, _ = nesterov_lse_optimize(a_init, time_limit=t_alloc * 0.5,
                                          beta_start=10.0, beta_end=5000.0, top_k=30)
        try_update(f_opt, best_tracker)

        # Follow up with coordinate descent if time allows
        if time.monotonic() < t5_end - 5:
            t_cd = min(20, t5_end - time.monotonic() - 3)
            if t_cd > 5:
                f_opt2, _ = coordinate_descent_polish(f_opt, time_limit=t_cd)
                try_update(f_opt2, best_tracker)

    # Phase 6: Final intensive coordinate descent on best (remaining time)
    print(f"\n=== Phase 6: Final coordinate descent ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_rem = deadline - time.monotonic()
    if t_rem > 10:
        print(f"  n={len(current_best)}, time={t_rem:.0f}s")
        f_opt, _ = coordinate_descent_polish(current_best, time_limit=t_rem)
        try_update(f_opt, best_tracker)

    # Final: try a few quick perturbations
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    for _ in range(5):
        if time.monotonic() > deadline:
            break
        f_pert = freq_perturb(current_best, rng, strength=0.005)
        try_update(f_pert, best_tracker)
        f_pert2 = project_simplex(current_best + rng.standard_normal(len(current_best)) * 0.001)
        try_update(f_pert2, best_tracker)

    # Final verification
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using fallback")
    fallback = np.ones(768) / 768.0
    return [float(x) for x in fallback]
