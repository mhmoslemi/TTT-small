import time

def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def logsumexp_obj(a, T):
    """Compute (1/T) * log(sum(exp(T * c_k))) and its gradient."""
    conv = fast_convolve(a)
    # For numerical stability
    conv_max = conv.max()
    scaled = T * (conv - conv_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -100, 100)
    log_sum = np.log(np.exp(scaled).sum()) + T * conv_max
    obj = log_sum / T
    
    # Gradient: (1/T) * sum_k exp(T*c_k - log_sum) * grad_c_k
    # = sum_k p_k * grad_c_k where p_k = softmax(T*conv)
    p = np.exp(scaled - (scaled - scaled.max()))  # = exp(T*(conv - conv_max))
    p = p / p.sum()
    
    # grad_c_k at position j = 2 * a[k-j] (if 0 <= k-j < n)
    n = len(a)
    grad = np.zeros(n)
    for k in range(len(p)):
        if p[k] < 1e-15:
            continue
        # grad_c_k[j] = 2*a[k-j] for j where 0 <= k-j < n
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        grad[js:je+1] += p[k] * 2.0 * a[k - (js:je+1)]
    
    return obj, grad, conv

"""

import numpy as np

def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, T):
    """
    Compute the log-sum-exp smoothed objective and its exact gradient.
    Objective: (1/T) * log(sum_k exp(T * c_k(a)))
    Gradient: sum_k p_k * grad c_k where p_k = softmax(T * c)
    """
    n = len(a)
    conv = fast_convolve(a)
    conv_max = conv.max()
    scaled = T * (conv - conv_max)
    scaled = np.clip(scaled, -500, 500)
    log_sum = np.log(np.exp(scaled).sum()) + T * conv_max
    obj = log_sum / T

    p = np.exp(scaled - scaled.max())
    p = p / p.sum()

    grad = np.zeros(n)
    # Only iterate over significant peaks
    thresh = p.max() * 1e-8
    active = np.where(p > thresh)[0]
    for k in active:
        pk = p[k]
        if pk < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        grad[js:je + 1] += pk * 2.0 * a[k - (js:je + 1)]

    return obj, grad, conv, p


def exact_subgradient(a, tolerance=1e-10):
    """Compute the exact subgradient of max_k c_k(a) using uniform weights on ties."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    active = np.where(conv >= M - tolerance * M)[0]
    weights = np.ones(len(active)) / len(active)

    grad = np.zeros(n)
    for i, k in enumerate(active):
        w = weights[i]
        if w < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        grad[js:je + 1] += w * 2.0 * a[k - (js:je + 1)]

    return grad, conv, active


def kkt_feasibility_check(a, active_peaks, support, tol=1e-6):
    """
    Check KKT feasibility: find lambda >= 0, sum lambda = 1 such that
    sum_k lambda_k * a[k-j] = c (constant) for all j in support.
    Returns True if feasible, and the lambda, c values.
    """
    s = len(support)
    p = len(active_peaks)
    if s == 0 or p == 0:
        return False, None, 0.0

    # Build constraint matrix: s+1 equations, p+1 variables (lambda_0..lambda_{p-1}, c)
    # sum_k lambda_k * a[active_peaks[k] - support[j]] - c = 0 for each j in support
    # sum_k lambda_k = 1
    A_eq = np.zeros((s + 1, p + 1))
    b_eq = np.zeros(s + 1)

    for i, j in enumerate(support):
        for k_idx, k in enumerate(active_peaks):
            idx = k - j
            if 0 <= idx < len(a):
                A_eq[i, k_idx] = a[idx]
        A_eq[i, p] = -1.0  # -c

    # Normalization: sum lambda = 1
    A_eq[s, :p] = 1.0
    b_eq[s] = 1.0

    # LP: minimize 0 subject to A_eq @ x = b_eq, x[:p] >= 0, x[p] free
    c_obj = np.zeros(p + 1)
    bounds = [(0, None)] * p + [(None, None)]  # lambda >= 0, c free

    try:
        res = linprog(c_obj, A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                      method='highs', options={'presolve': True, 'max_time': 10})
        if res.success:
            return True, res.x[:p], res.x[p]
    except Exception:
        pass
    return False, None, 0.0


def lse_pgd_optimize(a0, time_limit, T_init=5.0, T_min=0.01, lr_init=0.05,

        # Nesterov lookahead

def exact_subgrad_polish(a0, time_limit, max_iter=50000):
    """
    Polish using exact subgradient of max with adaptive step size.
    Uses Polyak-like step based on gap to lower bound.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)

    no_improve = 0

    while time.monotonic() - t0 < time_limit and it < max_iter:

            break

        grad, conv, active = exact_subgradient(a)
        M = conv.max()

        # Polyak step size: lr = (f(x) - f*) / ||g||^2
        gap = M - lb
        gnorm2 = np.dot(grad, grad)
        if gnorm2 > 1e-30:
            lr_polyak = gap / gnorm2
            lr_step = min(0.1, max(1e-4, lr_polyak * 0.5))
        else:
            lr_step = 1e-4

        candidate = project_simplex(a - lr_step * grad)

            a = candidate

                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Try smaller step
            for scale in [0.5, 0.1, 0.01]:
                candidate2 = project_simplex(a - lr_step * scale * grad)

                    a = candidate2
                    if M2 < best_M - 1e-14:

                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                    break
            else:
                no_improve += 1
                lr *= 0.9

        if no_improve > 200:
            # Perturb and restart
            rng = np.random.default_rng(it * 37 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)

def active_set_refine(a0, time_limit, peak_tol=0.005):
    """
    Active-set refinement:
    1. Identify active peaks and support.
    2. Solve LP for KKT multipliers.
    3. If infeasible, adjust the active set.
    4. Use the multipliers to compute a refined direction.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    conv = fast_convolve(a)
    best_M = conv.max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < 200:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        conv = fast_convolve(a)
        M = conv.max()
        support = np.where(a > 1e-8)[0]
        if len(support) < 2:
            break

        # Active peaks: within peak_tol fraction of max
        threshold = M * (1.0 - peak_tol)
        active_peaks = np.where(conv >= threshold)[0]

        if len(active_peaks) == 0:
            break

        # Check KKT feasibility
        feasible, lam, c_val = kkt_feasibility_check(a, active_peaks, support)

        if feasible and lam is not None:
            # We're at (or very near) a KKT point.
            # Compute the gradient using these multipliers for a final check
            grad = np.zeros(n)
            for k_idx, k in enumerate(active_peaks):
                if lam[k_idx] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                grad[js:je + 1] += lam[k_idx] * 2.0 * a[k - (js:je + 1)]

            # The gradient should be constant on the support
            resd = grad[support] - grad[support].mean()
            if np.abs(resd).max() < 1e-4:
                # Converged
                break
        else:
            # KKT infeasible: need to adjust
            # Try adding more peaks or relaxing tolerance
            peak_tol = min(0.1, peak_tol * 1.5)

            # Use the current active peaks for a gradient step
            weights = np.ones(len(active_peaks)) / len(active_peaks)
            grad = np.zeros(n)
            for k_idx, k in enumerate(active_peaks):
                w = weights[k_idx]
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                grad[js:je + 1] += w * 2.0 * a[k - (js:je + 1)]

            # Line search
            best_candidate = a
            best_val = M
            for step in [0.001, 0.005, 0.01, 0.02, 0.05]:
                candidate = project_simplex(a - step * grad)
                cv = fast_convolve(candidate)
                val = cv.max()
                if val < best_val - 1e-16:
                    best_val = val
                    best_candidate = candidate
                    break

            a = best_candidate

            if best_val < best_M - 1e-14:
                best_M = best_val
                best_a = a.copy()
            else:

                peak_tol = min(0.15, peak_tol * 1.3)

        # Periodically try to shrink the support
        if it % 20 == 0:
            # Zero out small components
            a[a < 1e-10] = 0.0
            a = a / a.sum()

    conv_final = fast_convolve(best_a)
    return best_a, conv_final.max()


def freq_domain_perturb(a, rng, scale=0.01):
    """Perturb in frequency domain for diverse restarts."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    # Perturb the magnitudes (keep DC fixed)
    phase = np.angle(A)
    mag = np.abs(A)
    mag[1:] *= (1.0 + rng.standard_normal(len(mag) - 1) * scale)
    mag = np.maximum(mag, 0)
    A_new = mag * np.exp(1j * phase)
    A_new[0] = A[0]  # Keep DC (sum) fixed
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def interp_to_length(a, n_target):
    """Interpolate sequence to target length, preserving auto-convolution flatness."""
    n_orig = len(a)
    if n_orig == n_target:
        return a.copy()
    x_orig = np.linspace(0, 1, n_orig)
    x_new = np.linspace(0, 1, n_target)
    a_new = np.interp(x_new, x_orig, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return a_new / s

    score = float(evaluate_sequence(a_norm.tolist()))

        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return (best_seq, best_score)

    # One last exact subgrad on the very best
    remaining = deadline - time.monotonic() - 10
    if remaining > 5:
        a_final, _ = exact_subgrad_polish(best_seq, time_limit=min(remaining, 20), max_iter=10000)
        (best_seq, best_score) = evaluate_and_report(a_final, "very_final", (best_seq, best_score))

    print(f"  Phase 5: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # === Final verification ===
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
