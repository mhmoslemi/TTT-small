So the active set QP is non-convex. This makes it harder.

Alternative: Instead of minimizing the sum of peaks exactly, we can use it as a heuristic - solve the non-convex QP with a local solver (like scipy.optimize.minimize with SLSQP), and accept the answer as a candidate.

Actually, for practical purposes, if we start from a good point and the Hessian is "mostly" PSD (which it is when the peaks are well-separated and the sequence is already close to optimal), the local solver should find a good solution.

Optimized search for minimum peak autocorrelation non-negative step function.
Uses improved smoothed gradient with momentum, active-set QP refinement,
and multi-n exploration.

import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient(a, p):
    """Gradient of sum_k p[k] * (a*a)[k] wrt a.
    d/d a[j] (a*a)[k] = 2*a[k-j] (out-of-range = 0)
    So grad[j] = 2 * sum_k p[k] * a[k-j]
    """
    n = len(a)
    # For each j, grad[j] = 2 * sum_k p[k] * a[k-j] where 0 <= k-j < n
    # = 2 * sum_k p[k] * a[(k-j) mod n]... no, not mod, just valid range
    # = 2 * (p * a_flipped)[j + n - 1]... let me think
    # grad[j] = 2 * sum_{k: 0<=k-j<n} p[k] * a[k-j]
    # Let m = k-j, so k = m+j, m in [0, n-1]
    # grad[j] = 2 * sum_{m=0}^{n-1} p[m+j] * a[m] (where p[m+j] = 0 if m+j >= 2n-1)
    
    # This is a convolution-like operation
    # p has length 2n-1, a has length n
    # grad[j]/2 = sum_m a[m] * p[m+j]
    # This is the cross-correlation of a and p, evaluated at j
    
    # Using FFT: cross-correlation = ifft(fft(a) * conj(fft(p)))
    # But it's easier to just use the direct formula with the Hankel matrix
    
    # Actually, the simplest way:
    # grad = 2 * (Hankel correlation)
    # Let's use the FFT approach
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    # cross-correlation: sum_m a[m] * p[m+j] for j = 0, ..., n-1
    # This is the "anti-diagonal" sum, which is:
    # corr[j] = sum_m a[m] * p[m+j]
    # = sum_m a[m] * p[n-1+m+j-(n-1)]... 
    
    # Actually let me use a different approach:
    # (a*a)[k] = sum_i a[i]*a[k-i]
    # d/d a[j] (a*a)[k] = 2*a[k-j] (valid range)
    # grad[j] = 2 * sum_k p[k] * a[k-j] (valid range)
    
    # Let's flip a: a_flip[i] = a[n-1-i]
    # Then a[k-j] (for 0<=k-j<n) = a_flip[n-1-(k-j)] = a_flip[n-1-k+j]
    # grad[j] = 2 * sum_k p[k] * a_flip[n-1-k+j]
    # Let l = n-1-k, so k = n-1-l:
    # grad[j] = 2 * sum_l p[n-1-l] * a_flip[l+j]
    # This is the cross-correlation of a_flip and p_flipped (where p_flipped[l] = p[n-1-l])
    
    # Cross-correlation: (a_flip ⋆ p_flipped)[j] = sum_l a_flip[l] * p_flipped[l-j]
    # Hmm, this is getting confusing. Let me just use the direct O(n^2) approach for small n
    # or the Hankel matrix approach.
    
    # Simplest: use the formula grad[j] = 2 * sum_{k=j}^{j+n-1} p[k] * a[k-j]
    # where p[k] = 0 for k >= 2n-1
    # This is: for each j, take p[j:j+n] and dot with a.
    
    # This can be done with a sliding window or with FFT
    # Using the " Toeplitz/Hankel" structure:
    # grad/2 = [p j] @ a for j = 0, ..., n-1
    # where [p j] is the vector (p[j], p[j+1], ..., p[j+n-1]) (zero-padded)
    
    # This is equivalent to: grad/2 = ifft(fft(a) * flip(fft(p_shifted)))
    # ... this is getting too complicated. Let me just use the O(n^2) approach
    # which is fine for n <= 4096.
    
    grad = np.zeros(n)
    for j in range(n):
        # grad[j] = 2 * sum_{m=0}^{n-1} a[m] * p[m+j] (p zero-padded to length 2n-1)
        # p has length 2n-1, so p[m+j] is valid for m+j <= 2n-2
        grad[j] = 2.0 * np.dot(a[:2*n-1-j], p[j:2*n-1])
    return grad

def smooth_max_obj(a, T):
    """Smoothed max of autocorrelation."""
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    return T * np.log(np.exp(scaled).sum())

def smooth_max_grad(a, T):
    """Gradient of smoothed max of autocorrelation."""
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()
    return compute_gradient(a, p)

def momentum_grad_descent(a0, time_limit, T_start=0.001, lr_start=0.01, 
                          momentum=0.9, T_end=0.0001, T_decay=0.9995):
    """Smoothed gradient descent with Nesterov momentum and adaptive step size."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    T = T_start
    lr = lr_start
    v = np.zeros(n)  # momentum
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Nesterov lookahead
        a_look = a - momentum * v
        a_look = project_simplex(a_look)
        
        grad = smooth_max_grad(a_look, T)
        
        # Step with backtracking line search
        v = momentum * v + lr * grad
        candidate = project_simplex(a - v)
        
        # Check improvement
        obj = fast_convolve(candidate).max()
        if obj < best_obj - 1e-15:
            a = candidate
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            # Halve step
            v = v * 0.5
            candidate = project_simplex(a - v)
            obj = fast_convolve(candidate).max()
            if obj < best_obj - 1e-15:
                a = candidate
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                # Reset momentum if stuck
                if no_improve > 50:
                    v = np.zeros(n)
                    T = max(T_end, T * 0.9)
                    lr = max(0.001, lr * 0.9)
                
                if no_improve > 150:
                    # Perturb
                    rng = np.random.default_rng((it * 104729 + 3) % (2**31 - 1))
                    a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                    v = np.zeros(n)
                    T = T_start * 0.5
                    lr = lr_start
                    no_improve = 0
        
        # Decay T slowly
        T = max(T_end, T * T_decay)
    
    return best_a, best_obj

def active_set_qp(a0, time_limit, m_peaks=5):
    """Active set QP: minimize sum of top m autocorrelation peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        # Find top m peaks
        peak_indices = np.argsort(conv)[::-1][:m_peaks]
        
        # Build the objective: sum of top m peaks
        def obj_and_grad(x):
            x = np.maximum(x, 0)
            s = x.sum()
            if s < 1e-15:
                return 0, np.zeros(n)
            x = x / s
            c = fast_convolve(x)
            val = sum(c[k] for k in peak_indices)
            # Gradient
            p = np.zeros(2*n-1)
            for k in peak_indices:
                p[k] = 1.0
            g = compute_gradient(x, p)
            # Adjust for normalization (chain rule)
            g = g - g.sum() * x  # This isn't quite right for the simplex constraint
            return val, g
        
        # Use SLSQP
        res = minimize(obj_and_grad, a, method='SLSQP',
                      bounds=[(0, None)] * n,
                      constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - 1,
                                    'jac': lambda x: np.ones(n)}],
                      options={'maxiter': 50, 'ftol': 1e-8})
        
        if res.success:
            a_new = np.maximum(res.x, 0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                obj_new = fast_convolve(a_new).max()
                if obj_new < best_obj - 1e-12:
                    a = a_new
                    best_obj = obj_new
                    best_a = a.copy()
                else:
                    break
        else:
            break
    
    return best_a, best_obj

def local_search(a0, time_limit, seed=42):
    """Local search with diverse perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        
        improved = False
        
        # Multiple perturbation scales
        for scale in [0.0002, 0.0005, 0.001, 0.002, 0.005]:
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(a + pert)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break
        
        # Coordinate moves
        if not improved and it % 3 == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    candidate = a.copy()
                    amount = candidate[i] * frac
                    candidate[i] -= amount
                    candidate[j] += amount
                    candidate = np.maximum(candidate, 0)
                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break
        
        # Pair swap (move mass from high-conv contribution to low)
        if not improved and it % 5 == 0:
            conv = fast_convolve(a)
            # Find positions that contribute most to peaks
            peak_k = np.argmax(conv)
            # Contributing pairs: (i, peak_k - i)
            contrib = np.zeros(n)
            for i in range(n):
                j = peak_k - i
                if 0 <= j < n:
                    contrib[i] += a[i] * a[j]
            # Move mass from high-contrib to low-contrib
            src = int(np.argmax(contrib * a))
            dst = int(np.argmin(contrib))
            if src != dst:
                for frac in [0.01, 0.03, 0.05]:
                    candidate = a.copy()
                    amount = candidate[src] * frac
                    candidate[src] -= amount
                    candidate[dst] += amount
                    candidate = np.maximum(candidate, 0)
                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved:
            stall += 1
            if stall > 200:
                break
        else:
            stall = 0
    
    return best_a, best_obj

def try_score(a):

    a = a / a.sum()

    return 2.0 * n * conv.max() / (a.sum()**2), a
