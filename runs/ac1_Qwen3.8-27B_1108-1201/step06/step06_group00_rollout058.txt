    """Log-sum-exp approximation of max."""
    mx = conv.max()
    if beta * mx > 700:
        return (mx + np.log(np.exp(beta * (conv - mx)).sum()) / beta)
    return (mx + np.log(np.exp(beta * (conv - mx)).sum()) / beta)

    w = np.exp(beta * (conv - conv.max()))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(conv)) / len(conv)
    else:
        w /= w_sum
    return w

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact coordinate descent (1D subgradient bisection per coordinate)
2. Log-sum-exp smoothed gradient descent with Nesterov momentum
3. LP-based Frank-Wolfe with adaptive constraint selection
4. Multi-length exploration with interpolation
5. Diverse initialization from parent + structured perturbations

import numpy as np
from scipy.optimize import linprog

import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_weights(conv, beta):
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_bc / total


def compute_sparse_subgradient(f, peak_idx, weights):
    n = len(f)
    grad = np.zeros(n)
    for idx in range(len(peak_idx)):
        k = int(peak_idx[idx])
        w = weights[idx]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += w * 2.0 * f[k - i_range]
    return grad


def lse_gradient_descent(f0, time_limit, beta_start=20.0, beta_end=8000.0, top_k=80):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

    else:

    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    v = np.zeros(n)
    beta = beta_start
    lr = 2.0 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    mu = 0.85

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:

        k_top = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        w_full = lse_weights(conv, beta)
        w_peaks = w_full[peak_idx]
        w_sum = w_peaks.sum()
        if w_sum > 1e-15:
            w_peaks = w_peaks / w_sum
        else:
            w_peaks = np.ones(k_top) / k_top
        grad = compute_sparse_subgradient(f, peak_idx, w_peaks)
        gnorm = np.linalg.norm(grad)

            no_improve += 100
            break
        grad_normed = grad / gnorm

        if v_norm < 1e-15:
            no_improve += 100
            break
        v = v / v_norm
        step = lr / (1.0 + 0.1 * np.sqrt(mu))
        improved = False
        for _ in range(12):
            f_new = project_simplex(f - step * v)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                improved = True
                no_improve = 0
                lr = min(lr * 1.01, 15.0 / np.sqrt(n))
                break
            step *= 0.5
        if not improved:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.85
            if no_improve > 80:

    return best_f, best_obj


def build_lp_matrix(f, constraint_indices):
    n = len(f)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([idx] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    if not rows:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(len(constraint_indices), n))


def solve_lp_direction(f, M, top_k=50, time_limit=15):
    n = len(f)
    conv = fast_convolve(f)
    pk = min(top_k, 2 * n - 1)
    peak_idx = np.argpartition(conv, -pk)[-pk:]
    center = n - 1
    if center not in peak_idx:
        peak_idx = np.append(peak_idx, center)
    A = build_lp_matrix(f, peak_idx)
    if A.shape[0] == 0:
        return np.ones(n) / n
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def coordinate_descent(f0, max_passes=50, iters_per_coord=35, random_order=True, seed=42):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    for p in range(max_passes):
        if random_order:
            coords = rng.permutation(n)
        else:
            coords = np.arange(n)
        improved = False
        for i in coords:
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            affected_k = np.arange(k_min, k_max + 1)
            B = np.zeros(len(affected_k))
            C = np.zeros(len(affected_k))
            for idx in range(len(affected_k)):
                k = affected_k[idx]
                ko = k - i
                if 0 <= ko < n:
                    if k == 2 * i:
                        C[idx] = 1.0
                    else:
                        B[idx] = 2.0 * f[ko]
            current_val = f[i]
            A = conv[affected_k] - B * current_val - C * current_val * current_val
            x_lo = 0.0
            x_hi = max(3.0 * current_val, 2.0 / n)
            for _ in range(iters_per_coord):
                x_mid = (x_lo + x_hi) / 2.0
                vals = A + B * x_mid + C * x_mid * x_mid
                max_val = vals.max()
                max_idx = np.argwhere(vals >= max_val - 1e-14)[:, 0]
                sgradients = B[max_idx] + 2.0 * C[max_idx] * x_mid
                if sgradients.min() > 1e-12:
                    x_hi = x_mid
                elif sgradients.max() < -1e-12:
                    x_lo = x_mid
                else:
                    x_lo = x_hi = x_mid
                    break
            x_new = (x_lo + x_hi) / 2.0
            vals_new = A + B * x_new + C * x_new * x_new
            new_max_local = vals_new.max()
            if len(affected_k) < len(conv):
                other_max = np.max(np.delete(conv, affected_k))
            else:
                other_max = 0.0
            new_max = max(new_max_local, other_max)
            if new_max < best_obj - 1e-14:
                f[i] = x_new
                improved = True
            elif abs(x_new - f[i]) > 1e-12:
                f[i] = x_new
                improved = True
            conv = fast_convolve(f)
            if conv.max() < best_obj:
                best_obj = conv.max()
                best_f = f.copy()
        if not improved:
            break
    return best_f, best_obj


def interpolate_to_length(f, new_n):
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def freq_perturb(f, rng, strength=0.03):
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new
