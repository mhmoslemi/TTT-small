"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Full softmax projected gradient with temperature annealing
- Frank-Wolfe with LP oracle for large moves
- Multiplicative weight updates
- Structured initialization (geometric, anti-symmetric, bimodal)

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (f*g) = A @ g."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def compute_softmax_gradient(a, t):
    """Compute gradient of soft-max objective w.r.t. a (on simplex)."""
    n = len(a)
    b = fast_convolve(a)
    bm = b.max()
    p = np.exp((b - bm) / t)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.ones(len(b)) / len(b)
    else:
        p = p / p_sum
    
    # grad_j = 2n * sum_m p[j+m] * a[m]
    grad = np.zeros(n)
    for j in range(n):
        m_max = min(n, 2 * n - 1 - j)
        if m_max > 0:
            grad[j] = np.dot(p[j:j + m_max], a[:m_max])
    grad *= 2.0 * n
    return grad, b, p


def softmax_projected_gradient(a0, time_limit, t_init=0.005, t_min=0.0001, lr_init=0.01):
    """Projected gradient with full softmax and temperature annealing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    t = t_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    stagnant = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        grad, b, p = compute_softmax_gradient(a, t)
        
        # Backtracking line search
        best_candidate = a
        best_val = b.max()
        
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            candidate = project_simplex(a - lr * frac * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
        
        a = best_candidate
        obj = fast_convolve(a).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
            # Slowly decrease t

            lr = min(lr * 1.01, 0.05)
        else:
            stagnant += 1
            # Adaptive: increase t when stuck (exploration)

                t = min(t * 1.15, t_init * 2)
                lr *= 0.95
            if stagnant > 100:
                t = t_init
                lr = lr_init
                # Random perturbation
                rng = np.random.default_rng((it * 31 + 7) % (2**31))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                stagnant = 0
    
    return best_a, best_obj


def multiplicative_weights(a0, time_limit, eta_init=0.1):
    """Multiplicative weight updates targeting peak reduction."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()

    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:

        b = fast_convolve(a)
        b_max = b.max()
        
        # Find top peaks
        n_peaks = min(10, len(b))
        peak_indices = np.argsort(b)[::-1][:n_peaks]
        
        # Compute "pressure" on each coordinate
        pressure = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            # Weight proportional to how much this peak exceeds the target
            weight = max(0, b[k] - b_max * 0.9)
            if weight < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                pressure[j] += weight * a[k - j]
        
        if pressure.max() < 1e-15:
            stagnant += 1
        else:
            # Multiplicative update: reduce coordinates with high pressure
            a_new = a * np.exp(-eta * pressure / (pressure.max() + 1e-15))
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a
            else:
                a_new = a_new / s_new
            
            obj_new = fast_convolve(a_new).max()
            if obj_new < best_obj - 1e-15:
                a = a_new
                best_obj = obj_new
                best_a = a.copy()
                stagnant = 0
                eta = min(eta * 1.02, 1.0)
            else:
                stagnant += 1
                eta *= 0.97
        
        if stagnant > 50:
            eta = eta_init
            t = min(t_init, t * 1.1) if 't' in dir() else eta_init
            rng = np.random.default_rng((it * 41 + 3) % (2**31))

    return best_a, best_obj


def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle, extensive line search, and adaptive M."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    stagnant = 0
    it = 0
    m_relax = 1.002

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.3:
            break

        g = solve_lp_oracle(f, M * m_relax)

        if g is None or g.sum() < 1e-15:
            g = solve_lp_oracle(f, M * 1.01)

        if g is None or g.sum() < 1e-15:
            stagnant += 1
            if stagnant > 20:
                rng = np.random.default_rng(it * 7919 + 3)
                scale = 0.002 + 0.004 * min(1.0, stagnant / 50.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.002
                stagnant = 0
                continue
            continue

        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-10:
            stagnant += 1
            if stagnant > 30:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue

        # Extensive line search
        best_alpha = 0.0
        best_val = M + 1e-14

        for alpha in np.linspace(0.01, 1.0, 30):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0.005:
            lo = max(0.0, best_alpha - 0.015)
            hi = min(1.0, best_alpha + 0.015)
            for alpha in np.linspace(lo, hi, 15):
                ft = (1.0 - alpha) * f + alpha * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha

        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.002

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        if stagnant > 10:
            m_relax = min(m_relax * 1.05, 1.05)

    return best_f, best_obj

            break

        conv = fast_convolve(a)
        n_peaks = min(5, len(conv))

def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_structured_starts(n, parent, rng):
    """Generate diverse structured initializations."""
    starts = []
    x = np.arange(n, dtype=np.float64)
    
    # Parent (resampled)
    if parent is not None:
        starts.append(resample_sequence(parent, n))
    
    # Uniform
    starts.append(np.ones(n) / n)
    
    # Geometric (concentrated at start)
    for r in [0.97, 0.95, 0.9]:
        g = r ** x
        starts.append(g / g.sum())
    
    # Geometric (concentrated at end)
    for r in [0.97, 0.95]:
        g = r ** (n - 1 - x)
        starts.append(g / g.sum())
    
    # Anti-symmetric bias: more mass on left half
    w = np.exp(-0.5 * ((x - 0.3 * n) / (0.15 * n)) ** 2)
    starts.append(w / w.sum())
    
    # Bimodal: two well-separated peaks
    g1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.08 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.08 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())
    
    # Cosine taper
    cos_t = np.cos(np.pi * x / (n - 1))
    w = 0.5 * (1 + cos_t)
    w = np.maximum(w, 0.01)
    starts.append(w / w.sum())
    
    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    starts.append(edge / edge.sum())
    
    # Random Dirichlet
    for _ in range(3):
        starts.append(rng.dirichlet(np.ones(n) * 0.2))
    
    # "Sawtooth" - alternating high/low
    saw = np.where(x % 2 == 0, 1.0, 0.3)
    starts.append(saw / saw.sum())
    
    # Quarter concentration
    w = np.exp(-0.5 * ((x - 0.2 * n) / (0.1 * n)) ** 2)
    starts.append(w / w.sum())
    
    return starts

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (f*g) = A @ g."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def compute_softmax_grad(a, t):
    """Compute gradient of soft-max objective w.r.t. a (a on simplex)."""
    n = len(a)
    b = fast_convolve(a)
    bm = b.max()
    p = np.exp((b - bm) / t)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.ones(len(b)) / len(b)
    else:
        p = p / p_sum
    # grad_j = 2n * sum_m p[j+m] * a[m] for valid m
    grad = np.zeros(n)
    for j in range(n):
        m_max = min(n, 2 * n - 1 - j)
        if m_max > 0:
            grad[j] = np.dot(p[j:j + m_max], a[:m_max])
    grad *= 2.0 * n
    return grad, b, p

            if stagnant > 20:
                rng = np.random.default_rng(it * 7919 + 3)
                scale = 0.002 + 0.004 * min(1.0, stagnant / 50.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.002
                stagnant = 0
                continue
            continue

        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-10:
            stagnant += 1
            if stagnant > 30:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue

        # Extensive line search
        best_alpha = 0.0
        best_val = M + 1e-14

        for alpha in np.linspace(0.01, 1.0, 25):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0.005:
            lo = max(0.0, best_alpha - 0.02)
            hi = min(1.0, best_alpha + 0.02)
            for alpha in np.linspace(lo, hi, 12):
                ft = (1.0 - alpha) * f + alpha * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha

        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.002

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        if stagnant > 10:
            m_relax = min(m_relax * 1.05, 1.06)

    return best_f, best_obj


def peak_coord_descent(a0, time_limit):
    """Coordinate descent targeting the top peaks directly."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    step_scale = 0.003

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        n_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        improved = False
        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)

            if js > je:
                continue

            pairs = []
            for j in range(js, je + 1):
                pairs.append((a[j] * a[k - j], j, k - j))
            pairs.sort(reverse=True)

            for (_, j1, j2) in pairs[:10]:
                if a[j1] < 1e-8:
                    continue

                for _attempt in range(8):
                    rng = np.random.default_rng((it * 9973 + k * 31 + j1 * 7) % (2**31))
                    j_target = int(rng.integers(0, n))
                    if j_target == j1 or j_target == j2:
                        continue

                    for src in [j1, j2]:
                        if a[src] < 1e-8:
                            continue
                        for delta_frac in [step_scale, step_scale * 0.5, step_scale * 1.5, step_scale * 0.25]:
                            delta = a[src] * delta_frac
                            candidate = a.copy()
                            candidate[src] -= delta
                            candidate[j_target] += delta
                            if candidate[src] < -1e-12:
                                continue
                            candidate = np.maximum(candidate, 0)
                            candidate = candidate / candidate.sum()
                            val = fast_convolve(candidate).max()
                            if val < best_obj - 1e-15:
                                a = candidate
                                best_obj = val
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break

                if improved:
                    break

            if improved:
                break

        if not improved:
            step_scale *= 0.94
            if step_scale < 1e-7:
                break
            if it % 200 == 0:
                rng = np.random.default_rng(it * 48271)
                candidate = project_simplex(a + rng.standard_normal(n) * step_scale * 3)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
            elif it > 4000:
                break

    return best_a, best_obj


def resample_seq(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def gen_starts(n, parent, rng):
    """Generate diverse structured initializations."""
    starts = []
    x = np.arange(n, dtype=np.float64)

    if parent is not None:
        starts.append(resample_seq(parent, n))

    starts.append(np.ones(n) / n)

    # Geometric decays (asymmetric - key for reducing center peak)
    for r in [0.97, 0.95, 0.92, 0.9]:
        g = r ** x
        starts.append(g / g.sum())

    # Geometric from right
    for r in [0.97, 0.95]:
        g = r ** (n - 1 - x)
        starts.append(g / g.sum())

    # Off-center Gaussian (anti-symmetric bias)
    for center in [0.25, 0.35, 0.65, 0.75]:
        for width in [0.12, 0.18]:
            w = np.exp(-0.5 * ((x - center * n) / (width * n)) ** 2)
            starts.append(w / w.sum())

    # Bimodal
    g1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.07 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.8 * n) / (0.07 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())

    # Cosine taper
    cos_t = np.cos(np.pi * x / (n - 1))
    w = 0.5 * (1 + cos_t)
    w = np.maximum(w, 0.01)
    starts.append(w / w.sum())

    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    starts.append(edge / edge.sum())

    # Random Dirichlet
    for _ in range(3):
        starts.append(rng.dirichlet(np.ones(n) * 0.25))

    # Spectral: random frequency domain
    for _ in range(2):
        freq = rng.standard_normal(n // 2 + 1) + 1j * rng.standard_normal(n // 2 + 1)
        freq[0] = abs(freq[0])
        sig = np.fft.irfft(freq, n)
        sig = np.maximum(sig, 0)
        if sig.sum() > 1e-10:
            starts.append(sig / sig.sum())

    return starts


def try_sequence(a, best_score, best_seq, label=""):
    """Try a sequence and update best if improved. Returns (score, best_score, best_seq)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf'), best_score, best_seq
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
    return score, best_score, best_seq


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search combining:
    - Full softmax projected gradient with annealing
    - Frank-Wolfe with LP oracle
    - Multiplicative weight updates
    - Peak coordinate descent
    - Structured initializations and multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            _, best_score, best_seq = try_sequence(parent, best_score, best_seq, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        _, best_score, best_seq = try_sequence(parent, best_score, best_seq, "uniform_768")

    n_parent = len(parent)

    # === Phase 1: Quick exploration across n and starts (180s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_phase = time.monotonic()

    lengths = sorted(set([n_parent, max(256, n_parent - 128), n_parent + 128, 512, 640, 768, 896, 1024]))
    lengths = [l for l in lengths if 192 <= l <= 2048]

    for n_try in lengths:
        remaining = deadline - time.monotonic() - 1300
        if remaining < 15:
            break

        starts = gen_starts(n_try, parent, rng)

        for si, start in enumerate(starts[:6]):
            remaining = deadline - time.monotonic() - 1200
            if remaining < 8:
                break
            t_alloc = min(remaining, 18)
            a_opt, _ = softmax_pg(start, time_limit=t_alloc, t_init=0.01, lr_init=0.012)
            _, best_score, best_seq = try_sequence(a_opt, best_score, best_seq, f"sg_n{n_try}_s{si}")

    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep optimization (600s) ===
    print("\n=== Phase 2: Deep Optimization ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # 2a: Softmax PG with careful annealing
        remaining = deadline - time.monotonic() - 500
        if remaining > 20:
            t_alloc = min(remaining, 250)
            print(f"  Softmax PG n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = softmax_pg(best_seq, time_limit=t_alloc,
                                  t_init=0.004, t_min=0.00003, lr_init=0.006)
            _, best_score, best_seq = try_sequence(a_opt, best_score, best_seq, "deep_sg")

        # 2b: Frank-Wolfe with LP
        remaining = deadline - time.monotonic() - 300
        if remaining > 20:
            t_alloc = min(remaining, 220)
            print(f"  FW n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc)
            _, best_score, best_seq = try_sequence(a_opt, best_score, best_seq, "deep_fw")

        # 2c: Multiplicative weights
        remaining = deadline - time.monotonic() - 150
        if remaining > 15:
            t_alloc = min(remaining, 100)
            print(f"  MultW n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = mult_weight_opt(best_seq, time_limit=t_alloc, eta_init=0.06)
            _, best_score, best_seq = try_sequence(a_opt, best_score, best_seq, "multw")

        # 2d: Perturbed restarts with softmax PG
        for scale in [0.002, 0.005, 0.01, 0.02]:
            remaining = deadline - time.monotonic() - 100
            if remaining < 15:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 50)
            a_opt, _ = softmax_pg(f_pert, time_limit=t_alloc, t_init=0.003, lr_init=0.005)
            _, best_score, best_seq = try_sequence(a_opt, best_score, best_seq, f"pert_{scale}")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Alternating refinement (250s) ===
    print("\n=== Phase 3: Alternating Refinement ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        for mi in range(8):
            remaining = deadline - time.monotonic() - 100
            if remaining < 25:
                break

            if mi % 3 == 0:
                t_alloc = min(remaining, 50)
                a_opt, _ = softmax_pg(best_seq, time_limit=t_alloc,
                                      t_init=0.002, t_min=0.00001, lr_init=0.003)
                _, best_score, best_seq = try_sequence(a_opt, best_score, best_seq, f"alt_sg_{mi}")
            elif mi % 3 == 1:
                t_alloc = min(remaining, 50)
                a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc)
                _, best_score, best_seq = try_sequence(a_opt, best_score, best_seq, f"alt_fw_{mi}")
            else:
                t_alloc = min(remaining, 40)
                a_opt, _ = mult_weight_opt(best_seq, time_limit=t_alloc, eta_init=0.04)
                _, best_score, best_seq = try_sequence(a_opt, best_score, best_seq, f"alt_mw_{mi}")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Peak coordinate descent (150s) ===
    print("\n=== Phase 4: Peak Coordinate Descent ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 50
        if remaining > 15:
            t_alloc = min(remaining, 120)
            print(f"  Peak CD, t={t_alloc:.0f}s")
            a_opt, _ = peak_coord_descent(best_seq, time_limit=t_alloc)
            _, best_score, best_seq = try_sequence(a_opt, best_score, best_seq, "peak_cd")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final micro-polish (remaining) ===
    print("\n=== Phase 5: Final Micro-Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        for ri in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            if ri % 2 == 0:
                noise = rng.standard_normal(n_best) * 0.0004
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 15)
                a_opt, _ = softmax_pg(start, time_limit=t_alloc,
                                      t_init=0.001, t_min=0.00001, lr_init=0.002)
                _, best_score, best_seq = try_sequence(a_opt, best_score, best_seq, f"fin_sg_{ri}")
            else:
                t_alloc = min(remaining, 12)
                a_opt, _ = peak_coord_descent(best_seq, time_limit=t_alloc)
                _, best_score, best_seq = try_sequence(a_opt, best_score, best_seq, f"fin_cd_{ri}")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
