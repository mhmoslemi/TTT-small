import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rng_arr = np.arange(1, n + 1)
    cond = u * rng_arr > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, k_peak):
    """Compute subgradient of b[k_peak] = sum_j a[j]*a[k_peak-j] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    # b[k] = sum_j a[j]*a[k-j], db[k]/da[i] = 2*a[k-i] if 0<=k-i<n
    shift = k_peak
    for i in range(n):
        idx = shift - i
        if 0 <= idx < n:
            grad[i] = 2.0 * a[idx]
    return grad


def compute_weighted_subgradient(a, peaks, weights):
    """Compute weighted combination of subgradients at multiple peaks."""
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(peaks, weights):
        if w < 1e-15:
            continue
        for i in range(n):
            idx = k - i
            if 0 <= idx < n:
                grad[i] += w * 2.0 * a[idx]
    return grad


def solve_lp_direction(f, M, time_limit=5):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None

            remaining = deadline - time.monotonic() - 400
            if remaining < 15:
                break
            f_pert = project_simplex(state[1] + rng.standard_normal(len(state[1])) * scale)
            t_alloc = min(remaining, 80)
            a_opt, _ = subgradient_optimize(f_pert, time_limit=t_alloc,
                                             T_init=0.001, lr_init=0.025)
            try_and_update(a_opt, state, "sg_pert_{}".format(int(scale*1000)))

    print("  Phase 2 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 3: LP-Augmented Refinement (~500s) ===
    print("\n=== Phase 3: LP-Augmented ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Main LP optimization
        remaining = deadline - time.monotonic() - 200
        if remaining > 30:
            t_alloc = min(remaining, 250)
            print("  LP-opt n={}, t={:.0f}s".format(len(state[1]), t_alloc))
            a_opt, _ = lp_augmented_optimize(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "lp_main")

        # LP from perturbed starting points
        for scale in [0.005, 0.01]:
            remaining = deadline - time.monotonic() - 150
            if remaining < 20:
                break
            f_pert = project_simplex(state[1] + rng.standard_normal(len(state[1])) * scale)
            t_alloc = min(remaining, 100)
            a_opt, _ = lp_augmented_optimize(f_pert, time_limit=t_alloc)
            try_and_update(a_opt, state, "lp_pert_{}".format(int(scale*1000)))

        # Fast subgradient + LP alternating
        remaining = deadline - time.monotonic() - 100
        if remaining > 20:
            a_cur = state[1].copy()
            t_sub = min(remaining, 60)
            a_opt, _ = subgradient_optimize(a_cur, time_limit=t_sub, T_init=0.0005, lr_init=0.02)
            try_and_update(a_opt, state, "sg_sub")
            if state[1] is not a_opt:
                remaining = deadline - time.monotonic() - 60
                if remaining > 10:
                    a_opt2, _ = lp_augmented_optimize(state[1], time_limit=min(remaining, 40))
                    try_and_update(a_opt2, state, "lp_after_sg")

    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 4: Final Polish (~150s) ===
    print("\n=== Phase 4: Final Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Try nearby n values by resampling
        for frac in [0.9, 1.1, 1.25]:
            remaining = deadline - time.monotonic() - 50
            if remaining < 10:
                break
            n_new = int(len(state[1]) * frac)
            n_new = max(128, min(2048, n_new))
            if n_new == len(state[1]):
                continue
            a_resized = np.interp(np.linspace(0, 1, n_new),
                                  np.linspace(0, 1, len(state[1])), state[1])
            a_resized = np.maximum(a_resized, 0)
            s = a_resized.sum()
            if s > 1e-10:
                a_resized /= s
                # Quick optimization
                t_alloc = min(remaining, 25)
                a_opt, _ = subgradient_optimize(a_resized, time_limit=t_alloc,
                                                 T_init=0.001, lr_init=0.02)
                try_and_update(a_opt, state, "resize_n{}".format(n_new))

        # Tiny perturbations with quick polish
        remaining = deadline - time.monotonic() - 20
        if remaining > 5:
            n_best = len(state[1])
            n_trials = min(10, int(remaining // 3) + 1)
            for trial in range(n_trials):
                remaining = deadline - time.monotonic() - 5
                if remaining < 3:
                    break
                noise = rng.standard_normal(n_best) * 0.001
                f_pert = project_simplex(state[1] + noise)
                a_opt, _ = subgradient_optimize(f_pert, time_limit=min(remaining, 5),
                                                 T_init=0.0003, lr_init=0.01)
                try_and_update(a_opt, state, "polish_{}".format(trial))

        # One final LP pass
        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            a_opt, _ = lp_augmented_optimize(state[1], time_limit=min(remaining, 10))
            try_and_update(a_opt, state, "final_lp")

    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(384) / 384
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
