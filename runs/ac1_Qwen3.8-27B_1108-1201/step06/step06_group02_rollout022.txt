"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Multi-initialization, adaptive subgradient descent with
simulated annealing, and multi-scale n exploration.
Key insight: Non-palindromic sequences can break the score >= 2 barrier.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the positive simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(a):
    """Compute 2*n*max_b/(sum_a)^2 for a non-negative sequence."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * len(a) * b.max() / (s * s))


def subgrad_step(a, b, top_K=10, lr=0.01):
    """One subgradient step: compute grad from top-K peaks, project."""
    n = len(a)
    b = fast_convolve(a) if b is None else b
    
    if top_K >= len(b):
        peaks = np.arange(len(b))
        w = np.ones(len(peaks)) / len(peaks)
    else:
        idx = np.argpartition(b, -top_K)[-top_K:]
        peaks = idx
        pv = b[peaks]
        # Smooth weighting: weight proportional to peak value
        w = pv / pv.sum()
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        j_arr = np.arange(js, je + 1)
        grad[j_arr] += w[i] * 2.0 * a[k - j_arr]
    
    candidate = project_simplex(a - lr * grad)
    return candidate


def sgd_optimize(a0, time_limit, top_K=10, lr_init=0.02, anneal=False):
    """Subgradient descent with adaptive lr and optional simulated annealing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    T_anneal = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Try multiple step sizes (backtracking)
        best_candidate = a
        best_val = max_b
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            cand = subgrad_step(a, b, top_K, trial_lr)
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_candidate = cand
        
        a = best_candidate
        score = compute_score(a)
        
        # Simulated annealing
        if anneal and it % 100 == 0 and T_anneal > 0.01:
            rng = np.random.default_rng(it * 7919 + 3)
            noise = 0.005 * T_anneal
            cand = project_simplex(a + rng.standard_normal(n) * noise)
            val = compute_score(cand)
            if val < best_score * (1 + 0.02 * T_anneal):
                a = cand
            T_anneal *= 0.99
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive lr
        if no_improve > 10:
            lr *= 0.95
        if no_improve % 20 == 0 and no_improve > 10:
            lr = max(0.001, lr * 1.1)
        if no_improve > 300:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise = 0.003 + 0.005 * min(1.0, (no_improve - 300) / 300)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            lr = lr_init
            no_improve = 0
    
    return best_a, best_score


def multi_peak_smoothed(a0, time_limit, T_init=0.001, lr_init=0.03, top_K=20):
    """Smoothed multi-peak subgradient with softmax weighting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    T = T_init
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        b = fast_convolve(a)
        
        # Softmax-weighted peaks
        scaled = b / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Gradient
        grad = np.zeros(n)
        # Use top-K for efficiency
        K = min(top_K, len(b))
        idx = np.argsort(p)[::-1][:K]
        for i in idx:
            if p[i] < 1e-14:
                continue
            k = int(i)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            j_arr = np.arange(js, je + 1)
            grad[j_arr] += p[i] * 2.0 * a[k - j_arr]
        
        # Line search
        best_candidate = a
        best_val = b.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.1]:
            cand = project_simplex(a - trial_lr * grad)
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_candidate = cand
        
        a = best_candidate
        score = compute_score(a)
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 15:
            T *= 1.05
            lr *= 0.97
        if no_improve > 40:
            T *= 0.7
        if no_improve > 150:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise = 0.002 + 0.004 * min(1.0, (no_improve - 150) / 200)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            no_improve = 0
    
    return best_a, best_score


def coordinate_descent(a0, time_limit):
    """Coordinate descent: optimize one coordinate at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit - 0.05:
                break
            
            # Try transferring mass from j to a few other positions
            # Find the position with minimum a that is not j
            # Actually, try small perturbations
            for delta in [0.001, 0.002, 0.005]:
                if a[j] < delta:
                    continue
                # Move delta from j to the position with smallest a
                j_other = int(np.argmin(a))
                if j_other == j:
                    # Second smallest
                    a_temp = a.copy()
                    a_temp[j] = 0
                    j_other = int(np.argmin(a_temp))
                
                cand = a.copy()
                cand[j] -= delta
                cand[j_other] += delta
                cand = np.maximum(cand, 0)
                s = cand.sum()
                if s < 1e-15:
                    continue
                cand = cand / s
                
                score = compute_score(cand)
                if score < best_score - 1e-15:
                    a = cand
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            # Try adding noise
            rng = np.random.default_rng(it * 4567 + 3)
            cand = project_simplex(a + rng.standard_normal(n) * 0.001)
            score = compute_score(cand)
            if score < best_score - 1e-15:
                a = cand
                best_score = score
                best_a = a.copy()
    
    return best_a, best_score


def make_initializations(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Tilted uniform (breaks left-right symmetry)
    for alpha in [0.5, 1.0, 2.0, -0.5, -1.0]:
        t = np.exp(alpha * x / max(n - 1, 1))
        inits.append(t / t.sum())
    
    # Two-peak
    for c1_frac, c2_frac in [(0.25, 0.75), (0.2, 0.8), (0.15, 0.85)]:
        c1 = c1_frac * (n - 1)
        c2 = c2_frac * (n - 1)
        sigma = 0.1 * n
        two = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
        inits.append(two / two.sum())
    
    # Edge concentration
    for p in [2, 4, 6]:
        edge = np.minimum(x, n - 1 - x).astype(float)
        # Invert: more mass at edges
        edge_inv = 1.0 / (edge / (0.5 * n) + 0.1) ** p
        inits.append(edge_inv / edge_inv.sum())
    
    # Cosine-based (non-symmetric)
    for freq in [1, 2, 3]:
        c = np.cos(2 * np.pi * freq * x / n)
        h = 1.0 + 0.5 * c
        h = np.maximum(h, 0.01)
        inits.append(h / h.sum())
    
    # Resampled parent
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 0.1))
    
    return inits


def try_and_update(a, state, label):
    """Evaluate and update state if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = compute_score(a_norm)
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a)))
