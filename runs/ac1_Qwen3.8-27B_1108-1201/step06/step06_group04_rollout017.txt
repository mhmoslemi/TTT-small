def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_mv(w, a):
    """Compute W @ a where W_{i,j} = w[i+j]."""
    return np.correlate(w, a, mode='valid')

def build_hankel(w, n):
    """Build n x n Hankel matrix H where H[i,j] = w[i+j]."""
    H = np.zeros((n, n))
    for i in range(n):
        H[i, :] = w[i:i+n]
    return H

def solve_qp_cvxpy(w, n, a_warm=None, verbose=False):
    """
    Solve min a^T H w a s.t. sum(a)=1, a>=0
    where H is Hankel with H[i,j] = w[i+j].

    # Try to build the matrix efficiently
    # Actually, for CVXPY, we can use the quad_form
    
    a = cp.Variable(n)
    
    # Build the Hessian - it's Hankel
    # H[i,j] = w[i+j]
    # quad_form(a, H) = a^T H a

    H = np.array([[w[i+j] for j in range(n)] for i in range(n)])

def mirror_descent(a0, time_limit, K=5, eta_init=0.01, verbose=False):
    """
    Mirror descent (exponential weights) on the simplex for min max_k conv(a,a)[k].
    Uses top-K peaks for the subgradient.
    """

    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    eta = eta_init
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        cmax = conv.max()
        
        # Get top-K peaks
        if K < len(conv):
            top_k = np.argpartition(conv, -K)[-K:]
        else:
            top_k = np.arange(len(conv))
        
        # Compute weighted subgradient
        # Weight by how close each peak is to the max
        peak_vals = conv[top_k]
        weights = np.exp((peak_vals - cmax) * 10.0)  # Softmax-like weighting
        weights = weights / weights.sum()
        
        grad = np.zeros(n)
        for wk, k in zip(weights, top_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * wk * a[k - idx]
        
        # Mirror descent update: a_new[i] = a[i] * exp(-eta * grad[i]) / Z
        log_update = -eta * grad
        log_update -= log_update.max()  # Numerical stability
        update = np.exp(log_update)
        a_new = a * update
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        obj = fast_convolve(a_new).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            eta = min(eta * 1.05, 0.5)  # Increase step slightly
        else:
            no_improve += 1
            eta *= 0.95  # Decrease step
            # Revert
            a_new = a  # Keep old a
        
        a = a_new
        
        if no_improve > 50:
            # Reset with perturbation
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            eta = eta_init
            no_improve = 0
        
        if verbose and it % 100 == 0:
            print(f"    MD it={it}, obj={best_obj:.10f}, eta={eta:.6f}")
    
    return best_a, best_obj

    a = a / a.sum()
    n = len(a)

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    # We'll use a "frozen" set of peaks and update their weights

    conv = fast_convolve(a)
    top_k = np.argpartition(conv, -K)[-K:]
    dual_weights = np.ones(K) / K
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2.0:
            break

        for wk, k in zip(dual_weights, top_k):
            w[k] = wk

        H = np.array([[w[i+j] for j in range(n)] for i in range(n)])
        H = (H + H.T) / 2

        prob = cp.Problem(cp.Minimize(quad), constraints)
        
        try:
            prob.solve(solver=cp.CLARABEL)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                prob.solve(solver=cp.ECOS)
        
        if a_var.value is None:

        a_qp = a_qp / a_qp.sum()

            peak_vals = conv_new[top_k]

            dual_weights = np.exp((peak_vals - cmax) * 5.0)

        if verbose and it % 10 == 0:
            print(f"    CVXPY-PD it={it}, obj={best_obj:.10f}, K={K}, no_imp={no_improve}")
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase hybrid approach:
- Phase 1: Fast mirror descent (multiplicative weights) for initial optimization
- Phase 2: CVXPY-based primal-dual with adaptive active set
- Phase 3: Fine optimization with large active sets and small steps
- Phase 4: Final polish and verification
"""

import numpy as np
import time
import cvxpy as cp

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_mv(w, a):
    return np.correlate(w, a, mode='valid')

def mirror_descent(a0, time_limit, K=5, eta_init=0.005, verbose=False):
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    eta = eta_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        conv = fast_convolve(a)
        cmax = conv.max()
        if K < len(conv):
            top_k = np.argpartition(conv, -K)[-K:]
        else:
            top_k = np.arange(len(conv))
        peak_vals = conv[top_k]
        weights = np.exp((peak_vals - cmax) * 10.0)
        weights = weights / weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * wk * a[k - idx]
        log_update = -eta * grad
        log_update -= log_update.max()
        update = np.exp(log_update)
        a_new = a * update
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            eta = min(eta * 1.02, 0.1)
        else:
            no_improve += 1
            eta *= 0.97
        a = a_new
        if no_improve > 100:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            eta = eta_init
            no_improve = 0
        if verbose and it % 200 == 0:
            print(f"    MD it={it}, obj={best_obj:.10f}, eta={eta:.6f}")
    return best_a, best_obj

def solve_qp_cvxpy(w, n, a_warm=None):
    """Solve min a^T H a s.t. sum(a)=1, a>=0 where H[i,j]=w[i+j]."""
    # Build Hankel matrix
    H = np.zeros((n, n))
    for i in range(n):
        H[i, :] = w[i:i+n]
    H = (H + H.T) / 2.0
    
    a_var = cp.Variable(n)
    if a_warm is not None:
        a_var = cp.Variable(n, initial_value=a_warm)
    
    try:
        quad = cp.quad_form(a_var, H, assume_PSD=True)
    except Exception:
        # Fallback: express as sum of products
        quad = 0
        for i in range(n):
            quad += cp.multiply(a_var[i], H[i, :] @ a_var)
        quad = quad / 2.0  # Since we're double counting
    
    constraints = [cp.sum(a_var) == 1, a_var >= 0]
    prob = cp.Problem(cp.Minimize(quad), constraints)
    
    solvers = [cp.CVXOPT, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if a_var.value is not None:
                result = np.maximum(a_var.value, 0)
                s = result.sum()
                if s > 1e-15:
                    result = result / s
                return result
        except Exception:
            continue
    
    # Fallback: return warm start or uniform
    if a_warm is not None:
        return a_warm
    return np.ones(n) / n

def cvxpy_primal_dual(a0, time_limit, K_init=10, K_max=50, verbose=False):
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    K = K_init
    blend = 0.6
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 2.0:
            break
        it += 1
        conv = fast_convolve(a)
        cmax = conv.max()
        # Get top-K peaks
        if K < len(conv):
            top_k = np.argpartition(conv, -K)[-K:]
        else:
            top_k = np.arange(len(conv))
        # Dual weights
        peak_vals = conv[top_k]
        weights = np.exp((peak_vals - cmax) * 5.0)
        weights = weights / weights.sum()
        # Build weight vector
        w = np.zeros(2 * n - 1)
        for wk, k in zip(weights, top_k):
            w[k] = wk
        # Solve QP
        a_qp = solve_qp_cvxpy(w, n, a_warm=a)
        # Blend
        a_new = (1.0 - blend) * a + blend * a_qp
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        else:
            a_new = a.copy()
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
