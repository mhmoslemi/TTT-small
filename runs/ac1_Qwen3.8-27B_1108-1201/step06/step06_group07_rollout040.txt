For each coordinate i:
1. Compute the current convolution.
2. For each k where a[i] contributes (i.e., 0 <= k-i < n or k = 2i), compute the new conv[k] as a function of x.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using accelerated coordinate descent
with 1D subgradient search, upsampling to larger n, and diverse initializations.
Key idea: coordinate descent is O(n^2) per pass (vs O(n^3) for LP-based SCA),
allowing thousands of passes within the time budget.
"""

import numpy as np
import time


def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a, conv):
    """Compute objective 2*n*max(conv)/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * conv.max() / (s * s)


def coord_desc_pass(a, conv, rng, num_trials=25):
    """
    One pass of coordinate descent.
    Updates a and conv in place.
    Returns the objective value.
    """
    n = len(a)
    C = a.sum()
    order = np.random.permutation(n, random_state=rng)
    
    for i in order:
        x_old = a[i]
        # Determine search range for d = new_x - old_x
        # x must be >= 0, so d >= -x_old
        # Upper bound: no need to exceed 2*C/n * 3 (a few times the mean)
        d_max = max(3.0 * C / n, 2.0 * x_old) if x_old > 0 else 3.0 * C / n
        
        # Generate candidate d values
        if x_old > 1e-10:
            d_candidates = np.concatenate([
                np.array([0.0, -x_old * 0.5, -x_old]),
                np.linspace(-x_old, d_max, num_trials)
            ])
        else:
            d_candidates = np.linspace(0, d_max, num_trials)
        
        best_d = 0.0
        best_f = 2.0 * n * conv.max() / (C * C)
        
        # Precompute M_unaff
        # Unaffected: k in [0, i-1] and k in [i+n, 2n-2]
        m_unaff = 0.0
        if i > 0:
            m_unaff = max(m_unaff, conv[:i].max())
        if i + n < 2 * n - 1:
            m_unaff = max(m_unaff, conv[i + n:].max())
        
        for d in d_candidates:
            x_new = x_old + d
            if x_new < 0:
                continue
            d = x_new - x_old
            if abs(d) < 1e-12:
                continue
            
            # Compute new max conv
            affected = conv[i:i + n] + 2.0 * d * a
            affected[i] += d * d
            max_new = max(m_unaff, affected.max())
            
            new_C = C + d
            if new_C < 1e-15:
                continue
            f_new = 2.0 * n * max_new / (new_C * new_C)
            
            if f_new < best_f:
                best_f = f_new
                best_d = d
        
        # Apply best update
        if abs(best_d) > 1e-12:
            x_new = max(x_old + best_d, 0.0)
            d = x_new - x_old
            conv[i:i + n] += 2.0 * d * a
            if 2 * i < 2 * n - 1:
                conv[2 * i] += d * d
            a[i] = x_new
            C += d
    
    return 2.0 * n * conv.max() / (C * C)


def coord_descent(a_init, time_limit, rng, verbose=False, max_passes_per_phase=100):
    """
    Run coordinate descent until time_limit.
    Returns (best_a, best_conv, best_obj).
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    # Don't normalize - objective is scale invariant, keep as is
    conv = fast_convolve(a)
    n = len(a)
    
    best_a = a.copy()
    best_conv = conv.copy()
    best_obj = obj_value(a, conv)
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        obj = coord_desc_pass(a, conv, rng, num_trials=20)
        
        if obj < best_obj - 1e-12 * best_obj:
            best_obj = obj
            best_a = a.copy()
            best_conv = conv.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and it % 50 == 0:
            print(f"    CD it={it}, obj={best_obj:.10f}, no_improve={no_improve}")
        
        if no_improve > 30:
            # Perturb
            scale = 0.001 + 0.005 * min(1.0, no_improve / 50.0)
            a_pert = a + rng.standard_normal(n) * scale * a.mean()
            a_pert = np.maximum(a_pert, 0)
            if a_pert.sum() > 1e-15:
                a = a_pert
                conv = fast_convolve(a)
            no_improve = 0
        
        if no_improve > 200:
            # Larger perturbation
            a_pert = a * (1.0 + rng.standard_normal(n) * 0.1)
            a_pert = np.maximum(a_pert, 0)
            if a_pert.sum() > 1e-15:
                a = a_pert
                conv = fast_convolve(a)
            no_improve = 0
    
    return best_a, best_conv, best_obj


def upsample(a, new_n):
    """Upsample using linear interpolation."""
    old_n = len(a)
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=np.float64)
    x_new = np.linspace(0, old_n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def make_initializations(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    t = x / (n - 1) if n > 1 else np.array([0.0])
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Hann window
    hann = 0.5 * (1 + np.cos(2 * np.pi * t))
    inits.append(hann / hann.sum())
    
    # Blackman window
    blackman = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    blackman = np.maximum(blackman, 0)
    if blackman.sum() > 1e-15:
        inits.append(blackman / blackman.sum())
    
    # Gaussian (varied sigma)
    for sigma in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
        inits.append(g / g.sum())
    
    # Cosine raised (smooth bump)
    cos_bump = np.cos(np.pi * (t - 0.5)) ** 2
    inits.append(cos_bump / cos_bump.sum())
    
    # Raised cosine with parameter
    for alpha in [1.0, 2.0, 4.0, 8.0]:
        rc = np.cos(np.pi * (t - 0.5)) ** (2 * alpha)
        rc = np.maximum(rc, 0)
        if rc.sum() > 1e-15:
            inits.append(rc / rc.sum())
    
    # From parent (interpolated)
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-15:
            inits.append(px / px.sum())
        # Perturbed parent
        px2 = np.maximum(px + rng.standard_normal(n) * 0.05, 0)
        if px2.sum() > 1e-15:
            inits.append(px2 / px2.sum())
    
    # Random Dirichlet
    for alpha in [0.3, 1.0, 3.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha))
    
    # Two-hump
    g2 = np.exp(-0.5 * ((t - 0.25) / 0.1) ** 2) + np.exp(-0.5 * ((t - 0.75) / 0.1) ** 2)
    inits.append(g2 / g2.sum())
    
    # Edge-heavy
    edge_w = 1.0 / (np.minimum(x, n - 1 - x).astype(float) + 1.0) ** 1.0
    inits.append(edge_w / edge_w.sum())
    
    return inits
