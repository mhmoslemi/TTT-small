def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def shifted_convolve_weights(a, peak_indices):
    """
    For given peak indices k, compute the subgradient contributions.
    grad(c[k])/a[j] = a[k-j] (if valid)
    Returns weighted sum: sum_k w[k] * a[k-j]
    """
    n = len(a)
    result = np.zeros(n)
    for k_idx, k in enumerate(peak_indices):
        k_int = int(k)
        js = max(0, k_int - n + 1)
        je = min(n - 1, k_int)
        if js > je:
            continue
        idx_arr = np.arange(js, je + 1)
        result[idx_arr] += a[k_int - idx_arr]
    return result

            c_lp = np.zeros(n_vars)
            c_lp[-1] = 1.0  # minimize spread

            A_ub = []
            b_ub = []
            Bmu = B  # m x p, (B@mu)_i for i in support
            for i in range(m):
                for j in range(i + 1, m):
                    row = np.zeros(n_vars)
                    row[:p] = B[i, :] - B[j, :]
                    row[-1] = -1.0
                    A_ub.append(row)
                    b_ub.append(0.0)

            if len(A_ub) > 0:
                A_ub = np.array(A_ub)
                b_ub = np.array(b_ub)
            else:
                A_ub = np.zeros((1, n_vars))
                b_ub = np.zeros(1)

            # Equality: sum(mu) = 1
            A_eq = np.zeros((1, n_vars))
            A_eq[0, :p] = 1.0
            b_eq = np.array([1.0])

            bounds = [(0, None)] * p + [(0, None)]
            try:
                res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                              bounds=bounds, method='highs',
                              options={'presolve': True, 'max_time': 5})
                if res.success:
                    mu = np.maximum(res.x[:p], 0)
                    mu_sum = mu.sum()
                    if mu_sum > 1e-15:
                        mu = mu / mu_sum
                    else:
                        mu = np.ones(p) / p
                else:
                    mu = np.ones(p) / p
            except Exception:
                mu = np.ones(p) / p
        else:
            # Fallback: uniform weights
            mu = np.ones(p) / p

        # Compute gradient direction: -sum_k mu[k] * a[k-j]
        grad = np.zeros(n)
        for k_idx, k in enumerate(active):
            w = mu[k_idx]
            if w < 1e-15:
                continue
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] -= w * a[k_int - idx_arr]

        # Also subtract the mean to ensure sum-orthogonal (for simplex projection)
        # Take step
        best_cand = a
        best_val = M
        for step in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:
            cand = project_simplex(a + step * grad)
            val = fast_convolve(cand).max()
            if val < best_val - 1e-16:
                best_val = val
                best_cand = cand
                break

        a = best_cand

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()

        if M - best_val < 1e-14:
            break

    return best_a, best_obj


def kkt_newton(a0, time_limit, max_iter=2000):
    """
    Newton-type method solving KKT conditions directly.
    At optimum: c[k]=t for active k, and grad condition holds on support.
    Uses linearization of the quadratic constraints.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1

        conv = fast_convolve(a)
        M = conv.max()
        support = np.where(a > 1e-10)[0]
        if len(support) == 0:
            break

        # Active peaks within 1e-4 relative of max
        threshold = M - 0.001 * M
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])

        p = len(active)
        m = len(support)

        if p == 1:
            # Single active peak: move mass away from it
            k = int(active[0])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad = np.zeros(n)
            grad[idx_arr] = -a[k - idx_arr]
            best_cand = a
            best_val = M
            for step in [0.01, 0.005, 0.001]:
                cand = project_simplex(a + step * grad)
                val = fast_convolve(cand).max()
                if val < best_val:
                    best_val = val
                    best_cand = cand
            a = best_cand
            if M < best_obj - 1e-15:
                best_obj = M
                best_a = a.copy()
            continue

        # Linearized Newton: solve for delta_a such that
        # c[k] + 2*(a * delta_a)[k] = t for k in active
        # sum(delta_a) = 0 (to preserve normalization)
        # This gives: for each active k: 2*sum_j a[j]*delta_a[k-j] = t - M (if c[k]=M)
        # or more generally: 2*(a*delta_a)[k] = t - c[k] for k in active

        # Unknown: t and delta_a (on support)
        # System: 2*(a*delta_a)[k] + t = c[k] for k in active
        #         sum_{j in S} delta_a[j] = 0

        # (a*delta_a)[k] = sum_{j in S} a[j] * delta_a[k-j] (if k-j in S)
        # Build linear system for delta_a on support

        # Matrix: for each active k, row is: [a[j] * 1_{k-j in S} for j in S]
        # Unknowns: delta_a[j] for j in S, and t
        # Equations: 2 * sum_j B_k[i] * delta_a[support[i]] + t = c[k]
        # Plus: sum delta_a = 0

        A_sys = np.zeros((p + 1, m + 1))
        b_sys = np.zeros(p + 1)

        for k_idx, k in enumerate(active):
            k_int = int(k)
            for i, j in enumerate(support):
                idx = k_int - j
                if 0 <= idx < n:
                    A_sys[k_idx, i] = 2.0 * a[idx]
            A_sys[k_idx, m] = 1.0  # t column
            b_sys[k_idx] = conv[k]

        # Sum constraint
        A_sys[p, :m] = 1.0
        b_sys[p] = 0.0

        # Solve
        try:
            sol = np.linalg.solve(A_sys, b_sys)
        except np.linalg.LinAlgError:
            try:
                sol, _, _, _ = np.linalg.lstsq(A_sys, b_sys, rcond=None)
            except Exception:
                break

        delta_support = sol[:m]
        delta_a = np.zeros(n)
        delta_a[support] = delta_support

        # Take step with damping
        best_cand = a
        best_val = M
        for damped in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            cand = a + damped * delta_a
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val - 1e-16:
                best_val = val
                best_cand = cand
                break

        a = best_cand

        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()

        if M - best_val < 1e-15:
            break

    return best_a, best_obj


def spectral_init(n, rng):
    """
    Create a sequence whose DFT has a designed magnitude spectrum
    to promote flat auto-convolution.
    """
    # Flat magnitude spectrum: |A(k)| = const for all k
    # This gives c[k] = const (for circular conv), close to flat for linear
    freqs = np.fft.rfftfreq(n)
    # Design: slightly suppress low frequencies to reduce DC dominance
    magnitude = 1.0 / (1.0 + 0.5 * np.abs(freqs) * n)
    # Add random phase
    phase = rng.uniform(0, 2 * np.pi, len(magnitude))
    A = magnitude * np.exp(1j * phase)
    a = np.fft.irfft(A * n, n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def skewed_init(n, alpha, rng):
    """Create a skewed (exponential-like) initialization."""
    j = np.arange(n)
    # Exponential decay from one side
    a = np.exp(-alpha * j / n)
    # Add small noise to break symmetry
    a += rng.standard_normal(n) * 0.01
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def power_law_init(n, alpha, rng):
    """Power-law initialization."""
    j = np.arange(1, n + 1)
    a = j ** (-alpha)
    a += rng.standard_normal(n) * 0.005
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def try_and_update(a, best_seq, best_score, label=""):
    """Evaluate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_seq, best_score
    a_n = a / s
    score = float(evaluate_sequence(a_n.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_n.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_n)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search with diverse initializations,
    Nesterov PGD, peak equalization, and KKT Newton refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = skewed_init(512, 2.0, rng)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    # === Phase 1: Diverse initializations + quick optimization (~200s) ===
    print("\n=== Phase 1: Diverse Inits ===")
    t_start = time.monotonic()

    n_base = len(best_seq)
    inits = []

    # Parent and symmetrized
    inits.append(("parent", best_seq.copy()))
    sym = (best_seq + best_seq[::-1]) / 2.0
    inits.append(("sym", sym / sym.sum()))

    # Skewed inits at various alphas
    for alpha in [1.0, 2.0, 3.0, 5.0]:
        inits.append((f"skew_{alpha}", skewed_init(n_base, alpha, rng)))

    # Power law
    for alpha in [0.5, 1.0, 1.5]:
        inits.append((f"pow_{alpha}", power_law_init(n_base, alpha, rng)))

    # Spectral
    inits.append(("spectral", spectral_init(n_base, rng)))

    # Uniform with noise
    noisy = np.ones(n_base) + rng.standard_normal(n_base) * 0.3
    noisy = np.maximum(noisy, 0)
    inits.append(("noisy", noisy / noisy.sum()))

    # Quick PGD on each (30s each, but limited total)
    t_per_init = 40.0
    for label, init in inits:
        remaining = deadline - time.monotonic() - 600
        if remaining < 10:
            break
        t_alloc = min(t_per_init, remaining * 0.8)
        a_opt, _ = nesterov_pgd(init, time_limit=t_alloc, top_k=30, lr_init=0.02)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"pgd_{label}")

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 2: Multi-n exploration (~300s) ===
    print("\n=== Phase 2: Multi-n Exploration ===")
    t_start = time.monotonic()

    n_best = len(best_seq)
    candidate_ns = sorted(set([
        n_best, n_best // 2, n_best * 2,
        384, 512, 640, 768, 896, 1024, 1280, 1536
    ]))
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 3072]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 500
        if remaining < 30:
            break

        if n_try == n_best:
            start = best_seq.copy()
        else:
            start = interp_sequence(best_seq, n_try)

        t_alloc = min(remaining, 45)
        # Nesterov PGD
        a_opt, _ = nesterov_pgd(start, time_limit=t_alloc * 0.5, top_k=25, lr_init=0.02)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"pgd_n{n_try}")

        # Peak equalize
        remaining = deadline - time.monotonic() - 450
        if remaining > 15:
            a_opt2, _ = peak_equalize(a_opt, time_limit=min(remaining, t_alloc * 0.5))
            best_seq, best_score = try_and_update(a_opt2, best_seq, best_score, f"peq_n{n_try}")

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 3: Deep refinement on best sequence (~400s) ===
    print("\n=== Phase 3: Deep Refinement ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    # KKT Newton
    remaining = deadline - time.monotonic() - 300
    if remaining > 20:
        t_alloc = min(remaining, 150)
        print(f"  KKT Newton, t={t_alloc:.0f}s")
        a_opt, _ = kkt_newton(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "kkt1")

    # Peak equalize
    remaining = deadline - time.monotonic() - 250
    if remaining > 20:
        t_alloc = min(remaining, 120)
        a_opt, _ = peak_equalize(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "peq1")

    # Nesterov with fine params
    remaining = deadline - time.monotonic() - 200
    if remaining > 15:
        t_alloc = min(remaining, 100)
        a_opt, _ = nesterov_pgd(best_seq, time_limit=t_alloc, top_k=50, lr_init=0.005, momentum_init=0.95)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "pgd_fine")

    # KKT Newton again
    remaining = deadline - time.monotonic() - 150
    if remaining > 15:
        t_alloc = min(remaining, 80)
        a_opt, _ = kkt_newton(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "kkt2")

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 4: Perturbation restarts (~300s) ===
    print("\n=== Phase 4: Perturbation Restarts ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    for pert_scale in [0.002, 0.005, 0.01, 0.02, 0.03, 0.05]:
        remaining = deadline - time.monotonic() - 150
        if remaining < 15:
            break

        f_pert = best_seq + rng.standard_normal(n_best) * pert_scale
        f_pert = project_simplex(f_pert)

        t_alloc = min(remaining, 40)
        a_opt, _ = nesterov_pgd(f_pert, time_limit=t_alloc * 0.5, top_k=30, lr_init=0.01)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"pert_{pert_scale}")

        remaining = deadline - time.monotonic() - 120
        if remaining > 10:
            a_opt2, _ = kkt_newton(a_opt, time_limit=min(remaining, t_alloc * 0.3))
            best_seq, best_score = try_and_update(a_opt2, best_seq, best_score, f"pert_kkt_{pert_scale}")

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Final intensive polishing (~200s) ===
    print("\n=== Phase 5: Final Polish ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    # Alternating peak equalize and KKT Newton
    for round_idx in range(4):
        remaining = deadline - time.monotonic() - 50
        if remaining < 10:
            break

        t_half = min(remaining / 2, 30)
        a_opt, _ = peak_equalize(best_seq, time_limit=t_half)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"polish_peq_{round_idx}")

        remaining = deadline - time.monotonic() - 30
        if remaining < 5:
            break
        a_opt, _ = kkt_newton(best_seq, time_limit=min(remaining, t_half))
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"polish_kkt_{round_idx}")

    # Tiny perturbations
    for _ in range(3):
        remaining = deadline - time.monotonic() - 20
        if remaining < 10:
            break
        noise = rng.standard_normal(n_best) * 0.001
        f_pert = project_simplex(best_seq + noise)
        a_opt, _ = kkt_newton(f_pert, time_limit=min(remaining, 15))
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "micro_pert")

    # Symmetry check
    sym_final = (best_seq + best_seq[::-1]) / 2.0
    sym_final = sym_final / sym_final.sum()
    best_seq, best_score = try_and_update(sym_final, best_seq, best_score, "final_sym")

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
