    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact coordinate descent with incremental convolution updates
2. Multi-length exploration via interpolation
3. Frequency-domain perturbation for diversification  
4. Frequency-domain optimization (direct spectral shaping)

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    S = a.sum()
    if S < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max() / (S * S)

def coord_descent(a, time_limit, verbose=False, n_coarse=20, n_fine=30, step_scale=1.0):
    """
    Exact coordinate descent minimizing 2n*max(conv(a,a))/sum(a)^2.
    Works with a normalized to sum=1.
    Uses incremental conv updates for efficiency.
    """
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a)) / len(a)
        S = 1.0
    else:
        a = a / S
        S = 1.0
    
    n = len(a)
    conv = fast_convolve(a)
    current_obj = 2.0 * n * conv.max() / (S * S)
    best_obj = current_obj
    best_a = a.copy()
    t0 = time.monotonic()
    
    it = 0
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        cycle_improved = False
        cycle_best = best_obj
        
        # Coordinate order: random after first iteration
        if it == 1:
            order = np.arange(n)
        else:
            order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            
            ai = a[i]
            # Determine search range for t = delta in a[i]
            t_lo = -ai  # Ensure a[i] + t >= 0
            t_hi = ai * step_scale + 1.0/n  # Allow some growth
            
            # Special case: if ai is very small, search in a small positive range

                t_hi = max(2.0/n, 1e-6)
            
            # Coarse grid
            best_t = 0.0
            best_val = current_obj
            
            # Include t=0 in the grid
            ts = np.unique(np.concatenate([np.linspace(t_lo, t_hi, n_coarse), [0.0]]))
            
            for t in ts:
                # Compute new max conv efficiently
                # conv_new[k] = conv[k] + 2*t*a[k-i] for k in [i, i+n-1], k != 2i
                # conv_new[2i] = conv[2i] + 2*t*a[i] + t^2
                
                if abs(t) < 1e-15:
                    val = current_obj
                else:
                    # Max in affected region [i, i+n-1]
                    affected = conv[i:i+n] + 2.0 * t * a
                    affected[i] += t * t  # position 2i in conv = position i in affected
                    aff_max = affected.max()
                    
                    # Max outside affected region
                    if i > 0:
                        left_max = conv[:i].max()
                    else:
                        left_max = -np.inf
                    if i + n < len(conv):
                        right_max = conv[i+n:].max()
                    else:
                        right_max = -np.inf
                    
                    overall_max = max(aff_max, left_max, right_max)
                    new_S = S + t
                    if new_S < 0.01:
                        val = float('inf')
                    else:
                        val = 2.0 * n * overall_max / (new_S * new_S)
                
                if val < best_val - 1e-15:
                    best_val = val
                    best_t = t
            
            # Fine grid around best_t
            if abs(best_t) > 1e-13:
                grid_size = (t_hi - t_lo) / max(n_coarse - 1, 1)
                fine_lo = max(t_lo, best_t - grid_size)
                fine_hi = min(t_hi, best_t + grid_size)
                ts_fine = np.linspace(fine_lo, fine_hi, n_fine)
                
                for t in ts_fine:
                    if abs(t) < 1e-15:
                        val = current_obj
                    else:
                        affected = conv[i:i+n] + 2.0 * t * a
                        affected[i] += t * t
                        aff_max = affected.max()
                        if i > 0:

                        else:
                            left_max = -np.inf
                        if i + n < len(conv):

                        else:
                            right_max = -np.inf
                        overall_max = max(aff_max, left_max, right_max)
                        new_S = S + t
                        if new_S < 0.01:
                            val = float('inf')
                        else:
                            val = 2.0 * n * overall_max / (new_S * new_S)
                    
                    if val < best_val - 1e-15:
                        best_val = val
                        best_t = t
            
            # Apply update
            if abs(best_t) > 1e-14:
                # Update conv incrementally
                conv[i:i+n] += 2.0 * best_t * a
                conv[2*i] += best_t * best_t
                a[i] += best_t
                a[i] = max(a[i], 0.0)
                S += best_t
                current_obj = best_val
                
                if best_val < cycle_best - 1e-15:
                    cycle_best = best_val
                    cycle_improved = True
                
                if best_obj > best_val:
                    best_obj = best_val
                    best_a = a.copy()
        
        # Normalize periodically to prevent drift
        if it % 100 == 0:
            a = a / a.sum()
            S = 1.0
            conv = fast_convolve(a)
            current_obj = 2.0 * n * conv.max() / (S * S)
            if current_obj < best_obj:
                best_obj = current_obj
                best_a = a.copy()
        
        if cycle_improved:
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and it % 20 == 0:
            print(f"      CD iter={it}, obj={best_obj:.8f}, no_improve={no_improve}")
        
        if no_improve > 200:
            break
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact coordinate descent with incremental convolution updates (O(n) per evaluation)
2. Multi-length exploration via interpolation
3. Frequency-domain perturbation for diversification
4. Multi-start strategy with diverse initializations
5. Periodic re-normalization to prevent drift
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_from_conv(conv, S, n):
    """Compute 2*n*max(conv)/S^2."""
    if S < 0.01:
        return float('inf')
    return 2.0 * n * conv.max() / (S * S)


def eval_seq_score(a):
    """Compute the objective directly."""
    S = a.sum()
    if S < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max() / (S * S)


def coord_descent(a, time_limit, verbose=False, n_coarse=20, n_fine=30,
                  step_scale=1.5, recompute_every=30, rng=None):
    """
    Exact coordinate descent minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses incremental conv updates: when a[i] changes by t,
      conv[i:i+n] += 2*t*a  and  conv[2*i] += t^2
    """
    if rng is None:
        rng = np.random.default_rng(42)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a)) / len(a)
        S = 1.0
    else:
        a = a / S
        S = 1.0

    n = len(a)
    conv = fast_convolve(a)
    current_obj = objective_from_conv(conv, S, n)
    best_obj = current_obj
    best_a = a.copy()
    t0 = time.monotonic()

    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        cycle_improved = False
        cycle_best_obj = best_obj

        if it == 1:
            order = np.arange(n)
        else:
            order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit - 1.0:
                break

            ai = a[i]

            # Search range for t (change in a[i])
            t_lo = -ai
            t_hi = ai * step_scale + 1.0 / n

            if ai < 1e-10:
                t_lo = 0.0
                t_hi = max(3.0 / n, 1e-7)

            # Coarse grid search
            best_t = 0.0
            best_val = current_obj

            ts = np.linspace(t_lo, t_hi, n_coarse)

            for t in ts:
                if abs(t) < 1e-16:
                    val = current_obj
                else:
                    # Compute new max conv
                    aff_max = (conv[i:i + n] + 2.0 * t * a).max()
                    # Add t^2 at the correct position
                    pos_in_slice = 2 * i - i  # = i
                    if pos_in_slice >= 0 and pos_in_slice < n:
                        aff_max = max(aff_max, conv[2 * i] + 2.0 * t * a[i] + t * t)

                    if i > 0:
                        left_max = conv[:i].max()
                    else:
                        left_max = -np.inf
                    if i + n < len(conv):
                        right_max = conv[i + n:].max()
                    else:
                        right_max = -np.inf

                    overall_max = max(aff_max, left_max, right_max)
                    new_S = S + t
                    if new_S < 0.01:
                        val = float('inf')
                    else:
                        val = 2.0 * n * overall_max / (new_S * new_S)

                if val < best_val - 1e-15:
                    best_val = val
                    best_t = t

            # Fine grid around best_t
            if abs(best_t) > 1e-14:
                grid_width = (t_hi - t_lo) / max(n_coarse - 1, 1)
                fine_lo = max(t_lo, best_t - grid_width)
                fine_hi = min(t_hi, best_t + grid_width)
                ts_fine = np.linspace(fine_lo, fine_hi, n_fine)

                for t in ts_fine:
                    if abs(t) < 1e-16:
                        val = current_obj
                    else:
                        aff_max = (conv[i:i + n] + 2.0 * t * a).max()
                        pos_in_slice = i
                        if pos_in_slice >= 0 and pos_in_slice < n:
                            aff_max = max(aff_max, conv[2 * i] + 2.0 * t * a[i] + t * t)
                        if i > 0:
                            left_max = conv[:i].max()
                        else:
                            left_max = -np.inf
                        if i + n < len(conv):
                            right_max = conv[i + n:].max()
                        else:
                            right_max = -np.inf
                        overall_max = max(aff_max, left_max, right_max)
                        new_S = S + t
                        if new_S < 0.01:
                            val = float('inf')
                        else:
                            val = 2.0 * n * overall_max / (new_S * new_S)

                    if val < best_val - 1e-15:
                        best_val = val
                        best_t = t

            # Apply update
            if abs(best_t) > 1e-15:
                old_ai = a[i]
                conv[i:i + n] += 2.0 * best_t * a
                conv[2 * i] += best_t * best_t
                a[i] = old_ai + best_t
                if a[i] < 0:
                    a[i] = 0.0
                S += best_t
                current_obj = best_val

                if best_val < cycle_best_obj - 1e-15:
                    cycle_best_obj = best_val
                    cycle_improved = True
                if best_obj > best_val:
                    best_obj = best_val
                    best_a = a.copy()

        # Periodic recomputation to prevent numerical drift
        if it % recompute_every == 0:
            a = np.maximum(a, 0.0)
            S = a.sum()
            if S < 1e-10:
                a = np.ones(n) / n
                S = 1.0
            else:
                a = a / S
                S = 1.0
            conv = fast_convolve(a)
            current_obj = objective_from_conv(conv, S, n)
            if current_obj < best_obj:
                best_obj = current_obj
                best_a = a.copy()

        if cycle_improved:
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 10 == 0:
            print(f"        CD iter={it}, obj={best_obj:.8f}, no_improve={no_improve}")

        if no_improve > 150:
            break

    return best_a, best_obj


def coord_descent_fine(a, time_limit, verbose=False):
    """Fine-grained coordinate descent with smaller steps for final polish."""
    return coord_descent(a, time_limit, verbose=verbose,
                         n_coarse=30, n_fine=50, step_scale=0.5,

def freq_perturb(a, rng, strength=0.03):
    """Perturb the power spectrum and recover a non-negative sequence."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)

    # Perturb the magnitude
    noise = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + noise), 0.0)

    # Reconstruct
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new


def interp_to_length(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new

def smooth_initializer(n, rng, mode='parzen'):
    """Create a smooth initial sequence."""
    if mode == 'parzen':
        x = np.linspace(0, 1, n)
        a = np.zeros(n)
        # Parzen window: piecewise cubic
        for lx in [-0.5, 0.0, 0.5]:
            d = x - lx
            mask = np.abs(d) < 1.5
            a[mask] += (1 - np.abs(d[mask])) ** 3
        a = np.maximum(a, 0.0)
    elif mode == 'raised_cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif mode == 'gaussian':
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / (2 * 0.15 ** 2))
    elif mode == 'random_smooth':
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        amps = rng.uniform(0.3, 1.0, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.0)
    else:
        a = np.ones(n)

    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a

        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]
        # If the tracked best is somehow invalid, re-evaluate
        print(f"WARNING: tracked best invalid, trying fallback")

    # Fallback
    print("FALLBACK: using parent or uniform")
    if parent is not None:
        return [float(x) for x in np.clip(parent, 0, 1000)]
    return [1.0 / 768.0] * 768
