"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with adaptive LSE smoothing (fast local optimization)
2. LP-based Frank-Wolfe with adaptive active set (large moves)
3. Greedy coordinate descent for peak reduction (fine polish)
4. Multi-length exploration and frequency-domain diversification

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_lse_gradient(a, conv, beta, top_k=100):
    """Compute LSE-smoothed gradient of max(conv)."""
    n = len(a)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        # All weights equal
        k = n - 1
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        grad = 2.0 * a[k - i_range]
        full_grad = np.zeros(n)
        full_grad[i_range] = grad
        return full_grad
    
    weights = exp_bc / total
    # Only use top_k peaks for efficiency
    k_sel = min(top_k, len(conv))
    peak_idx = np.argpartition(weights, -k_sel)[-k_sel:]
    w_sel = weights[peak_idx]
    w_sum = w_sel.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w_sel = w_sel / w_sum
    
    grad = np.zeros(n)
    for idx, k in enumerate(peak_idx):
        k = int(k)
        w = w_sel[idx]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += w * 2.0 * a[k - i_range]
    return grad

def fista_with_lse(a0, time_limit=60, beta_start=20, beta_end=10000, top_k=80):
    """
    Accelerated gradient descent (FISTA) using LSE smoothing of the max.
    Returns (best_a, best_max_conv).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    beta = beta_start
    # Estimate Lipschitz constant: for LSE with parameter beta,
    # the gradient norm is bounded by about 2*beta*max(a) (rough)
    L = 2.0 * beta * max(a.max() * 2, 1.0 / n)
    t_fista = 1.0
    y = a.copy()
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Compute gradient at y
        conv_y = fast_convolve(y)
        grad = compute_lse_gradient(y, conv_y, beta, top_k=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # FISTA step
        a_new = project_simplex(y - grad / L)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista)) / 2.0
        y = a_new + ((t_fista - 1.0) / t_new) * (a_new - a)
        a = a_new
        t_fista = t_new
        
        # Evaluate
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
        
        # Adapt beta (increase slowly to sharpen the LSE)
        if it % 50 == 0:
            beta = min(beta * 1.5, beta_end)
            L = 2.0 * beta * max(a.max() * 2, 1.0 / n)
            t_fista = 1.0  # Reset momentum when beta changes
            y = a.copy()
        
        # Periodic restart if stuck
        if it % 200 == 0:
            if fast_convolve(a).max() > best_obj * 1.001:
                a = best_a.copy()
                y = a.copy()
                t_fista = 1.0
    
    return best_a, best_obj

def fw_lp_step(a, M, top_k=40, time_limit=10):
    """
    One Frank-Wolfe step: solve LP to find best direction.
    max sum(g) s.t. (a*g)[k] <= M for top-k peaks, g >= 0.
    """
    n = len(a)
    conv = fast_convolve(a)
    # Select active constraints: peaks within 5% of max
    threshold = M * 0.95
    active = np.where(conv >= threshold)[0]
    if len(active) > top_k:
        active = np.argpartition(conv, -top_k)[-top_k:]
    # Always include the center
    center = n - 1
    if center not in active:
        active = np.append(active, center)
    
    # Build constraint matrix (sparse)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        rows.extend([idx] * len(js))
        cols.extend(js.tolist())
        vals.extend((a[k - js]).tolist())
    
    if not rows:
        return np.ones(n) / n
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def coordinate_descent(a0, time_limit=60, n_peaks=10):
    """
    Greedy coordinate descent: reduce the highest peaks by moving mass
    from contributing positions to "safe" positions.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        if M < best_obj - 1e-15:
            best_obj = M
            best_a = a.copy()
        
        # Find top peaks
        k_sel = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -k_sel)[-k_sel:]
        
        improved = False
        for k in peaks[:5]:  # Try to reduce top 5 peaks
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo >= hi:
                continue
            
            # Find the position i in [lo, hi] that contributes most to this peak
            products = a[lo:hi+1] * a[k - np.arange(lo, hi+1)]
            i_max = lo + np.argmax(products)
            j_other = k - i_max
            
            if a[i_max] < 1e-8:
                continue
            
            # Try moving a fraction of a[i_max] to a random position
            for _ in range(5):
                # Pick a destination that's far from this peak
                j_dest = rng.integers(0, n)
                if j_dest == i_max or j_dest == j_other:
                    continue
                # Check if dest would create a worse peak
                # Moving delta from i_max to j_dest changes conv by:
                # -delta * a[k'-i_max] + delta * a[k'-j_dest] for each k'
                # Net change at k' = delta * (a[k'-j_dest] - a[k'-i_max])
                
                delta = a[i_max] * 0.15
                a_trial = a.copy()
                a_trial[i_max] -= delta
                a_trial[j_dest] += delta
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:

                a_trial /= st

                    break

            for _ in range(3):

                    a_trial[i_max] -= delta

                if best_dest >= 0 and best_val < M - 1e-12:

def freq_perturb(a, rng, strength=0.03):
    """Perturb in frequency domain for diversification."""
    n = len(a)
    F = np.fft.rfft(a)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(F_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new

def interpolate_to_length(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new
