def project_simplex(v):
    """Project a list of non-negative values onto the probability simplex.
    This function ensures that the sum of elements is fixed at 1 and all are non-negative.
    It is based on the algorithm from [this paper](https://people.eecs.berkeley.edu/~vazic/pubs/SINOUZI2011projection.pdf)."""
    n = len(v)
    # Convert to numpy array
    v = np.array(v)
    # Sum and sort in descending order
    sum_val = np.sum(v)
    if sum_val <= 0:
        return np.ones(n) / n
    
    v = v * np.sqrt(len(v)) / sum_val  # Adjust for simplex after evaluation
    
    # Sort in descending order
    sorted_indices = np.argsort(-v)
    v_sorted = v[sorted_indices]
    
    # Allowed deviations
    deviations = np.zeros(n)
    for i in range(min(n, len(v_sorted))):
        # Force non-negativity and distribute mass
        if i > 0:
            # Compute the minimum possible sum from v_sorted[i,0] onward if we were to set previous to 0
            avail_sum = v_sorted[0] + np.sum(v_sorted[1:i]) * (i - (i-1)) / i
            # Adjust deviations if the current sum is too large
            target_deviations = (avail_sum - v_sorted[0]) / (2*i-1)
            if target_deviations < deviations[i-1]:
                deviations[i-1] = target_deviations
    
    # Create solution with deviations
    solution = np.zeros(n)
    for i in range(min(n, len(v_sorted))):
        if deviations[i] > 0:
            solution[sorted_indices[i]] = v_sorted[i] - deviations[i] + 1e-10
        else:
            solution[sorted_indices[i]] = v_sorted[i]
    
    return solution
