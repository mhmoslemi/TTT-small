import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_grad(p, a):
    """Compute gradient: 2 * sum_k p[k] * a[k-j] for valid indices.
    Uses FFT-based convolution for O(n log n) complexity.
    p has length 2n-1, a has length n.
    Returns array of length n.
    """
    n = len(a)
    a_flip = a[::-1]
    # conv = a_flip * p (linear convolution, length n + (2n-1) - 1 = 3n-2)
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = np.fft.rfft(a_flip, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(A * P, fft_size)[:3 * n - 2]
    # g[j] = conv[j + n - 1] for j = 0, ..., n-1
    g = conv[n - 1:2 * n - 1]
    return 2.0 * g

        k_peak = int(np.argmax(b))

        # Find the element contributing most to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]

        # Find element to move mass to
        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            break
        j_min = outside[np.argmin(a[outside])]

        improved = False
        # Try multiple transfer amounts
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
            delta = a[j_max] * frac
            if delta < 1e-15:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min] += delta
            candidate = np.maximum(candidate, 0.0)
            ss = candidate.sum()
            if ss < 1e-15:
                continue
            candidate = candidate / ss
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        if improved:
            stale = 0
        else:
            stale += 1
            if stale > 100:
                # Try random pairwise swaps
                rng = np.random.default_rng(it * 7919 + 13)
                for _ in range(10):
                    i, j = rng.integers(0, n, 2)
                    if i == j or a[i] < 1e-10:
                        continue
                    delta = a[i] * 0.05
                    candidate = a.copy()
                    candidate[i] -= delta
                    candidate[j] += delta
                    candidate = np.maximum(candidate, 0.0)
                    candidate = candidate / candidate.sum()
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        stale = 0
                        break
                if stale > 500:
                    break

    return best_a, best_obj


def fw_simple_oracle(f, M):
    """Simple Frank-Wolfe oracle: find direction that reduces max convolution.
    Uses greedy approach: for each constraint b[k] <= M, find the j that
    most reduces b[k] when we increase g[j]."""
    n = len(f)
    # The LP is: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
    # Simple approximation: start with g = 0, greedily increase coordinates
    g = np.zeros(n)
    b = np.full(2 * n - 1, M)  # remaining budget

    # Try to add to each coordinate
    order = np.argsort(f)[::-1]  # try largest f entries first
    for j in order:
        # Max we can add at position j
        # (f*g)[k] += f[k-j]*g[j] for valid k, and f[j-k]*g[j]
        # So we need f[k-j]*delta <= b[k] for all valid k
        max_delta = np.inf
        for k in range(2 * n - 1):
            idx1 = k - j
            idx2 = j - k
            contrib = 0.0
            if 0 <= idx1 <= n - 1:
                contrib += f[idx1]
            if 0 <= idx2 <= n - 1 and idx2 != idx1:
                contrib += f[idx2]
            if contrib > 1e-15:
                max_delta = min(max_delta, b[k] / contrib)
        max_delta = max(0.0, max_delta)
        if max_delta > 1e-10:
            g[j] += max_delta
            # Update b
            for k in range(2 * n - 1):
                idx1 = k - j
                idx2 = j - k
                contrib = 0.0
                if 0 <= idx1 <= n - 1:
                    contrib += f[idx1]
                if 0 <= idx2 <= n - 1 and idx2 != idx1:
                    contrib += f[idx2]
                b[k] -= contrib * max_delta

    return g


def frank_wolfe_simple(f0, time_limit):
    """Frank-Wolfe with simple greedy oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        b = fast_convolve(f)
        M = b.max()

        # Get direction from oracle
        g = fw_simple_oracle(f, M)
        g_sum = g.sum()
        if g_sum < 1e-15:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-8:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val - 1e-14:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-6:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            if no_improve > 30:
                break

    return best_f, best_obj


def fourier_parametrize(a0, n_modes):
    """Extract Fourier coefficients from a sequence."""
    n = len(a0)
    x = np.arange(n, dtype=np.float64)
    a_centered = a0 - 1.0 / n
    coeffs = np.zeros(2 * n_modes)
    for m in range(1, n_modes + 1):
        coeffs[2 * (m - 1)] = 2.0 * np.sum(a_centered * np.cos(2.0 * np.pi * m * x / n))
        coeffs[2 * (m - 1) + 1] = 2.0 * np.sum(a_centered * np.sin(2.0 * np.pi * m * x / n))
    return coeffs


def fourier_to_sequence(coeffs, n, n_modes):
    """Convert Fourier coefficients back to a sequence."""
    x = np.arange(n, dtype=np.float64)
    base = np.ones(n) / n
    for m in range(1, n_modes + 1):
        c = coeffs[2 * (m - 1)]
        d = coeffs[2 * (m - 1) + 1]
        base += (c * np.cos(2.0 * np.pi * m * x / n) +
                 d * np.sin(2.0 * np.pi * m * x / n)) / n
    base = np.maximum(base, 0.0)
    s = base.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return base / s


def fourier_optimize(a0, time_limit, n_modes=15):
    """Optimize in Fourier coefficient space using Nelder-Mead."""
    n = len(a0)
    n_modes = min(n_modes, n // 2 - 1)
    init_coeffs = fourier_parametrize(a0, n_modes)

    best_a = a0.copy()
    best_obj = fast_convolve(a0).max()

    t0 = time.monotonic()
    n_restarts = max(2, int((time_limit // 40)) + 1)

    for r in range(n_restarts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        if r == 0:
            x0 = init_coeffs.copy()
        else:
            rng = np.random.default_rng(r * 999983 + 7)
            scale = 0.003 * (1.0 + 0.5 * r)
            x0 = init_coeffs + rng.standard_normal(2 * n_modes) * scale

        def obj_func(c):
            seq = fourier_to_sequence(c, n, n_modes)
            return fast_convolve(seq).max()

        try:
            res = minimize(obj_func, x0, method='Nelder-Mead',
                           options={'maxiter': 5000, 'xatol': 1e-10, 'fatol': 1e-12,
                                    'maxfev': 30000})
            if res.fun < best_obj - 1e-14:
                best_obj = res.fun
                best_a = fourier_to_sequence(res.x, n, n_modes)
                print("    Fourier restart {}: obj={:.10f}".format(r, best_obj))
        except Exception:
            pass

    return best_a, best_obj


def try_and_update(a, state, label):
    """Evaluate and update best state."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a)))


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Load parent, evaluate
    2. Quick gradient descent on parent
    3. Multi-n exploration
    4. Fourier refinement
    5. Coordinate descent polish
    6. Final gradient descent
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    state = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.8f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        state = [float(evaluate_sequence(parent.tolist())), parent.copy()]
        print("No parent, using uniform n=768, score={:.8f}".format(state[0]))

    n_parent = len(parent)

    # === Phase 1: Quick gradient descent on parent (~200s) ===
    print("\n=== Phase 1: Gradient Descent on Parent ===")
    t_phase = time.monotonic()

    remaining = deadline - time.monotonic() - 1400
    if remaining > 20:
        t_alloc = min(remaining, 180)
        print("  Grad descent n={}, T=0.001->0.0001, t={:.0f}s".format(n_parent, t_alloc))
        a_opt, _ = grad_descent_smoothed(parent, time_limit=t_alloc,
                                          T_start=1.0 / n_parent, T_end=0.001 / n_parent,
                                          lr_start=0.05)
        try_and_update(a_opt, state, "grad_parent")

    print("  Phase 1 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 2: Multi-n exploration (~400s) ===
    print("\n=== Phase 2: Multi-n Exploration ===")
    t_phase = time.monotonic()

    # Try various lengths
    candidate_ns = sorted(set([
        n_parent,
        max(128, n_parent - 128),
        n_parent + 128,
        n_parent + 256,
        512, 640, 768, 896, 1024, 1280, 1536, 1800, 2048
    ]))
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 2500]
    # Sort by proximity to parent
    candidate_ns.sort(key=lambda x: abs(x - n_parent))

    for n_try in candidate_ns[:8]:
        remaining = deadline - time.monotonic() - 800
        if remaining < 20:
            break

        if n_try == n_parent:
            start = state[1] if state[1] is not None else parent
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, len(state[1]) if state[1] is not None else n_parent),
                              state[1] if state[1] is not None else parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss > 1e-10:
                start = start / ss
            else:
                start = np.ones(n_try) / n_try

        t_alloc = min(remaining, 60)
        print("  Grad descent n={}, t={:.0f}s".format(n_try, t_alloc))
        a_opt, _ = grad_descent_smoothed(start, time_limit=t_alloc,
                                          T_start=1.0 / n_try, T_end=0.001 / n_try,
                                          lr_start=0.05)
        try_and_update(a_opt, state, "grad_n{}".format(n_try))

    print("  Phase 2 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 3: Frank-Wolfe on best (~300s) ===
    print("\n=== Phase 3: Frank-Wolfe ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        remaining = deadline - time.monotonic() - 400
        if remaining > 30:
            t_alloc = min(remaining, 250)
            print("  FW n={}, t={:.0f}s".format(len(state[1]), t_alloc))
            a_opt, _ = frank_wolfe_simple(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "fw")

    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 4: Fourier refinement (~250s) ===
    print("\n=== Phase 4: Fourier Refinement ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        remaining = deadline - time.monotonic() - 250
        if remaining > 20:
            t_alloc = min(remaining, 200)
            print("  Fourier n={}, modes=15, t={:.0f}s".format(len(state[1]), t_alloc))
            a_opt, _ = fourier_optimize(state[1], time_limit=t_alloc, n_modes=15)
            try_and_update(a_opt, state, "fourier15")

        remaining = deadline - time.monotonic() - 150
        if remaining > 15:
            t_alloc = min(remaining, 100)
            print("  Fourier n={}, modes=25, t={:.0f}s".format(len(state[1]), t_alloc))
            a_opt, _ = fourier_optimize(state[1], time_limit=t_alloc, n_modes=25)
            try_and_update(a_opt, state, "fourier25")

    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 5: Coordinate descent polish (~200s) ===
    print("\n=== Phase 5: Coordinate Descent Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        remaining = deadline - time.monotonic() - 100
        if remaining > 10:
            t_alloc = min(remaining, 150)
            print("  Coord descent n={}, t={:.0f}s".format(len(state[1]), t_alloc))
            a_opt, _ = coordinate_descent_polish(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "coord")

        # Another gradient pass after coord descent
        remaining = deadline - time.monotonic() - 50
        if remaining > 10:
            t_alloc = min(remaining, 60)
            a_opt, _ = grad_descent_smoothed(state[1], time_limit=t_alloc,
                                              T_start=0.001 / len(state[1]),
                                              T_end=0.0001 / len(state[1]),
                                              lr_start=0.02)
            try_and_update(a_opt, state, "grad_after_coord")

    print("  Phase 5 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 6: Final fine-tuning (~50s) ===
    print("\n=== Phase 6: Final Fine-tuning ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            t_alloc = min(remaining, 40)
            a_opt, _ = grad_descent_smoothed(state[1], time_limit=t_alloc,
                                              T_start=0.0005 / len(state[1]),
                                              T_end=0.00005 / len(state[1]),
                                              lr_start=0.01)
            try_and_update(a_opt, state, "final_grad")

            remaining = deadline - time.monotonic() - 3
            if remaining > 5:
                a_opt, _ = coordinate_descent_polish(state[1], time_limit=min(remaining, 10))
                try_and_update(a_opt, state, "final_coord")

    print("  Phase 6 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(768) / 768
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
