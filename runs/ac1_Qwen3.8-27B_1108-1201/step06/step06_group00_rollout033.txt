"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral domain shaping combined with projected gradient
2. Adaptive Frank-Wolfe with improved LP oracle

4. Coordinate descent on peak-contributing pairs

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(f, constraint_indices):
    """Build sparse constraint matrix for LP."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([idx] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    if not rows:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(len(constraint_indices), n))

def solve_lp_direction(f, M, top_k=30, time_limit=15, weighted=False):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for selected peaks, g >= 0."""
    n = len(f)
    conv = fast_convolve(f)
    k_select = min(top_k, len(conv))
    peak_idx = np.argpartition(conv, -k_select)[-k_select:]
    center = n - 1
    if center not in peak_idx:
        peak_idx = np.append(peak_idx, center)
        peak_idx = peak_idx[:k_select + 1]
    A = build_lp_matrix(f, peak_idx)
    if A.shape[0] == 0:
        return np.ones(n) / n
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def compute_subgradient(f, active_peaks, weights=None):
    """Compute weighted subgradient."""
    n = len(f)
    if weights is None:
        weights = np.ones(len(active_peaks)) / len(active_peaks)
    grad = np.zeros(n)
    for idx, k in enumerate(active_peaks):
        k = int(k)
        w = weights[idx]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += w * 2.0 * f[k - i_range]
    return grad

def lse_weights(conv, beta):
    """Compute LSE softmax weights."""
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_bc / total

def spectral_shaping(f, n_iters=50, time_limit=30.0):
    """
    Perform spectral shaping: modify the power spectrum to flatten the autocorrelation.
    Key idea: in the frequency domain, the autocorrelation is the IFFT of |FFT(a)|^2.
    We iteratively adjust |FFT(a)| to reduce peaks in the time domain.
    """
    n = len(f)
    f_work = f.copy()
    f_work = np.maximum(f_work, 0)
    f_work = f_work / f_work.sum()
    
    fft_size = 1 << (2 * n - 2).bit_length()
    best_obj = fast_convolve(f_work).max()
    best_f = f_work.copy()
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        F = np.fft.rfft(f_work, fft_size)
        power = np.abs(F) ** 2
        phase = np.angle(F)
        
        # Compute current autocorrelation
        c = np.fft.irfft(power, fft_size)[:2*n-1]
        c_max = c.max()
        
        # Compute gradient of max(c) with respect to power spectrum
        # c[k] = (1/fft_size) * sum_f power[f] * exp(2*pi*i*k*f/fft_size)
        # To reduce c at the peak location, reduce power at frequencies that
        # constructively interfere at that lag.
        
        peak_lag = int(np.argmax(c))
        
        # The contribution of frequency f to c[peak_lag] is:
        # (1/fft_size) * power[f] * exp(2*pi*i*peak_lag*f/fft_size).real
        # We want to reduce power where this contribution is positive (constructive)
        # and increase where negative (but we can't increase total power arbitrarily)
        
        # Simpler approach: use a "tilt" in the power spectrum
        # Reduce power at frequencies that contribute most to the peak
        freqs = np.arange(len(power))
        # The phase factor for the peak lag
        cos_factor = np.cos(2 * np.pi * peak_lag * freqs / fft_size)
        
        # Reduce power where cos_factor is positive (constructive at peak)
        # Increase power where cos_factor is negative (destructive at peak)
        # But keep total power roughly constant
        
        alpha = 0.1  # shaping strength
        weight = 1.0 - alpha * cos_factor
        weight = np.maximum(weight, 0.1)
        
        new_power = power * weight
        # Renormalize to keep sum(f) = 1
        # sum(f) = c at lag... actually sum(f) = F[0] / n... 
        # F[0] = sum(f), so we need F[0] to stay the same
        # F[0]^2 = new_power[0], so new_power[0] should stay the same
        new_power[0] = power[0]
        
        # Reconstruct
        F_new = np.sqrt(new_power) * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, fft_size)[:n]
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new = f_new / s
        
        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_f = f_new.copy()
            f_work = f_new.copy()
            alpha = min(alpha * 1.02, 0.5)  # increase strength slightly
        else:
            alpha = max(alpha * 0.9, 0.01)  # decrease strength
    
    return best_f, best_obj

        if best_alpha > 1e-6:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft /= st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f /= f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                cur_top_k = max(15, cur_top_k - 3)
            else:
                no_improve += 1
                cur_top_k = min(cur_top_k + 10, n)
        else:
            no_improve += 1
            cur_top_k = min(cur_top_k + 20, n)

    return best_f, best_obj

def subgradient_optimize(f0, time_limit, beta_start=10.0, beta_end=5000.0, top_k=60):
    """Subgradient method with LSE smoothing and adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    beta = beta_start
    lr = 2.0 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        conv = fast_convolve(f)
        w_full = lse_weights(conv, beta)
        k_top = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        w_peaks = w_full[peak_idx]
        w_sum = w_peaks.sum()
        if w_sum > 1e-15:
            w_peaks = w_peaks / w_sum
        else:
            w_peaks = np.ones(k_top) / k_top
        grad = compute_subgradient(f, peak_idx, w_peaks)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            no_improve += 100
            break
        grad_normed = grad / gnorm

        step = lr
        improved = False
        for _ in range(12):
            f_new = project_simplex(f - step * grad_normed)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                improved = True
                no_improve = 0
                lr = min(lr * 1.01, 10.0 / np.sqrt(n))
                break
            step *= 0.5

        if not improved:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.85
            if no_improve > 60:
                f = best_f.copy()
                lr = max(2.0 / np.sqrt(n), 0.005)
                beta = min(beta * 2.0, beta_end)
                no_improve = 0
                rng = np.random.default_rng(it * 31 + 17)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)

        if beta < beta_end:
            beta *= 1.003

    return best_f, best_obj

def local_search(f, time_limit=30.0):
    """
    Coordinate descent / local search on individual and paired coordinates.
    For each coordinate i, try to find the optimal value given all others fixed.
    The objective is max_k sum_j a[j]*a[k-j], and for fixed a[-i], changing a[i]
    affects specific entries of the convolution.
    """
    f = np.maximum(f.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False
        
        # Try coordinate-wise updates
        # For coordinate i, the convolution entries that depend on a[i] are:
        # c[k] = ... + a[i]*a[k-i] + ... for valid k-i
        # If we change a[i] to a[i]+d, then c[k] changes by d*a[k-i] for each valid k
        # We want to choose d to minimize the new max(c)
        
        # Subsample coordinates for speed on large n
        if n > 1000:
            indices = np.random.choice(n, size=200, replace=False)
        else:
            indices = np.arange(n)
        
        np.random.shuffle(indices)
        
        for i in indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Current value
            a_i = f[i]
            if a_i < 1e-10:
                continue
            
            # The convolution entries that involve a[i]:
            # c[k] includes a[i]*a[k-i] if 0 <= k-i < n and 0 <= k < 2n-1
            # So k ranges from i to i+n-1
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            
            # If we change a[i] to a[i] + d:
            # c[k] -> c[k] + d * f[k-i] for k in [i, i+n-1]
            # Also c[k] -> c[k] + d * f[i-(k-n+1)] ... wait let me be more careful
            
            # Actually, c[k] = sum_j f[j] * f[k-j]
            # Terms involving f[i]: f[i] * f[k-i] (when 0 <= k-i < n)
            #              and f[k-i] * f[i] (same thing since it's auto-conv)
            # Wait, no. c[k] = sum_{j=0}^{n-1} f[j] * f[k-j] where f is 0 outside [0,n-1]
            # Terms where j = i: f[i] * f[k-i] (valid if 0 <= k-i <= n-1, i.e., i <= k <= i+n-1)
            # Terms where k-j = i, i.e., j = k-i: f[k-i] * f[i] (valid if 0 <= k-i <= n-1)
            # These are the same term! So c[k] has exactly one term involving f[i]: f[i]*f[k-i]
            # UNLESS i = k-i, i.e., k = 2i, in which case the term is f[i]^2 (appears once)
            
            # So if we change f[i] to f[i]+d:
            # For k != 2i: c[k] changes by d * f[k-i]
            # For k = 2i: c[2i] changes by 2*d*f[i] + d^2
            
            # We want to find d that minimizes max(c + d*contrib)
            # This is a 1D optimization!
            
            # Get the contributions
            ks = np.arange(k_lo, k_hi + 1)
            contribs = np.zeros(len(ks))
            for idx, k in enumerate(ks):
                j = k - i
                if 0 <= j < n:
                    if k == 2 * i:
                        # quadratic term
                        contribs[idx] = 2.0 * f[i]  # linear part, we'll handle quadratic separately
                    else:
                        contribs[idx] = f[j]
            
            # Current conv values at these positions
            conv_vals = conv[ks]
            
            # We also need to consider: changing f[i] changes sum(f), which affects
            # the normalization. But for a local search, let's first try without
            # re-normalizing (small changes), then renormalize at the end.
            
            # Binary search for optimal d
            # Try a range of d values
            max_d = f[i] * 0.9  # don't go negative
            min_d = -f[i] * 0.5  # allow some decrease
            
            best_d = 0.0
            best_local_obj = conv.max()
            
            # Sample d values
            d_candidates = np.linspace(min_d, max_d, 20)
            for d in d_candidates:
                new_conv = conv.copy()
                new_conv[ks] = conv_vals + d * contribs
                if 2 * i in ks:
                    idx_2i = np.where(ks == 2 * i)[0]
                    if len(idx_2i) > 0:
                        new_conv[2 * i] += d * d  # quadratic correction
                new_max = new_conv.max()
                if new_max < best_local_obj:
                    best_local_obj = new_max
                    best_d = d
            
            if best_d != 0:
                f_new = f.copy()
                f_new[i] = f[i] + best_d
                f_new = np.maximum(f_new, 0)
                f_new = f_new / f_new.sum()
                conv_new = fast_convolve(f_new)
                obj_new = conv_new.max()
                if obj_new < best_obj - 1e-15:
                    best_obj = obj_new
                    best_f = f_new.copy()
                    f = f_new.copy()
                    conv = conv_new
                    improved = True
        
        if not improved and it > 20:
            # Try random restart within local region
            rng = np.random.default_rng(it * 1009 + 3)
            f_trial = f + rng.standard_normal(n) * 0.002
            f_trial = project_simplex(f_trial)
            conv_trial = fast_convolve(f_trial)
            if conv_trial.max() < best_obj:
                best_obj = conv_trial.max()
                best_f = f_trial.copy()
                f = f_trial.copy()
    
    return best_f, best_obj

def freq_perturb(f, rng, strength=0.05):
    """Perturb f in the frequency domain."""
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

def multiplicative_update(f, time_limit=30.0):
    """
    Multiplicative (exponentiated gradient) update.
    At each step, compute the convolution, identify peaks, and
    reduce coordinates that contribute to peaks multiplicatively.
    """
    f = np.maximum(f.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    eta = 0.01  # learning rate
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        
        # Compute the "potential" for each coordinate:
        # How much does coordinate i contribute to the max?
        # contrib[i] = max over k of f[i]*f[k-i] (for valid k)
        # Better: use the LSE-soft version
        beta = 50.0
        w = lse_weights(conv, beta)
        
        # For each i, its contribution to the weighted sum of peaks:
        # sum_k w[k] * f[k-i] (for valid k where i participates)
        # This is essentially (w * f_reversed)[i] in some sense
        
        # Actually, the gradient of sum_k w[k] * c[k] wrt f[i] is:
        # sum_k w[k] * f[k-i] for valid k
        # = sum_j w[i+j] * f[j] (letting j = k-i)
        
        # Let's compute this efficiently using convolution
        # grad[i] = sum_j w[i+j] * f[j] for valid (i,j)
        # This is like a cross-correlation of w with f
        
        w_full = np.zeros(2 * n - 1)
        w_full[:] = w
        
        # grad[i] = sum_{j: 0<=j<n, 0<=i+j<2n-1} w_full[i+j] * f[j]
        # = sum_j w_full[i+j] * f[j]
        # This is (w_full * f)[-i-1+n-1] ... let me just compute it directly with FFT
        # Actually it's a valid convolution/correlation
        
        # Let's use: define w_ext of length 2n-1 (already have it)
        # grad[i] = sum_j f[j] * w_ext[i+j] for j in range(n), i+j in range(2n-1)
        # = sum_j f[j] * w_ext[i+j]
        # This is the cross-correlation of f with w_ext, evaluated at specific lags
        
        # Simpler: just compute it
        grad = np.zeros(n)
        for i in range(n):
            j_max = min(n - 1, 2 * n - 2 - i)
            if j_max >= 0:
                grad[i] = np.dot(f[:j_max+1], w_ext[:j_max+1][::-1])  # Wrong, let me fix
        
        # Actually, let's just use the subgradient approach
        k_top = min(30, 2*n-1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        w_peaks = w[peak_idx]
        w_peaks = w_peaks / w_peaks.sum()
        grad = compute_subgradient(f, peak_idx, w_peaks)
        
        # Multiplicative update: f[i] *= exp(-eta * grad[i])
        f_new = f * np.exp(-eta * grad)
        f_new = np.maximum(f_new, 1e-15)  # avoid zeros
        f_new = f_new / f_new.sum()
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new.copy()
            eta = min(eta * 1.01, 0.1)
        else:
            eta = max(eta * 0.95, 0.001)
    
    return best_f, best_obj

"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, constraint_indices):
    """Build sparse constraint matrix for LP. Shape: (num_constr, n)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([idx] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    if not rows:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(len(constraint_indices), n))


def solve_lp_direction(f, M, top_k=30, time_limit=12):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-K peaks, g >= 0."""
    n = len(f)
    conv = fast_convolve(f)
    k_select = min(top_k, len(conv))
    peak_idx = np.argpartition(conv, -k_select)[-k_select:]
    center = n - 1
    if center not in peak_idx:
        peak_idx = np.append(peak_idx, center)
    A = build_lp_matrix(f, peak_idx)
    if A.shape[0] == 0:
        return np.ones(n) / n
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def compute_subgradient(f, active_peaks, weights=None):
    """Compute weighted subgradient of max(conv)."""
    n = len(f)
    if weights is None:
        weights = np.ones(len(active_peaks)) / len(active_peaks)
    grad = np.zeros(n)
    for idx, k in enumerate(active_peaks):
        k = int(k)
        w = weights[idx]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += w * 2.0 * f[k - i_range]
    return grad


def lse_weights(conv, beta):
    """Compute LSE softmax weights on convolution peaks."""
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_bc / total


def spectral_shaping(f, n_iters=80, time_limit=30.0):
    """
    Spectral domain shaping: modify power spectrum to flatten autocorrelation.
    Reduces power at frequencies constructively interfering at the peak lag.
    """
    n = len(f)
    f_work = f.copy()
    f_work = np.maximum(f_work, 0)
    f_work = f_work / f_work.sum()

    fft_size = 1 << (2 * n - 2).bit_length()
    best_obj = fast_convolve(f_work).max()
    best_f = f_work.copy()
    t0 = time.monotonic()
    alpha = 0.08

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        F = np.fft.rfft(f_work, fft_size)
        power = np.abs(F) ** 2
        phase = np.angle(F)

        c = np.fft.irfft(power, fft_size)[:2 * n - 1]
        peak_lag = int(np.argmax(c))

        freqs = np.arange(len(power))
        cos_factor = np.cos(2 * np.pi * peak_lag * freqs / fft_size)

        # Also use the top 3 peaks for more robust shaping
        top_lags = np.argpartition(c, -3)[-3:]
        cos_combined = np.zeros(len(power))
        for lag in top_lags:
            cos_combined += np.cos(2 * np.pi * int(lag) * freqs / fft_size)
        cos_combined /= len(top_lags)

        weight = 1.0 - alpha * cos_combined
        weight = np.maximum(weight, 0.05)

        new_power = power * weight
        new_power[0] = power[0]  # preserve DC component (sum)

        F_new = np.sqrt(new_power) * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, fft_size)[:n]
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            alpha *= 0.9
            continue
        f_new = f_new / s

        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f_work = f_new.copy()
            alpha = min(alpha * 1.03, 0.4)
        else:
            alpha = max(alpha * 0.92, 0.01)

    return best_f, best_obj

def subgradient_optimize(f0, time_limit, beta_start=10.0, beta_end=5000.0, top_k=60):

    no_improve = 0

        grad = compute_subgradient(f, peak_idx, w_peaks)

def local_coordinate_search(f, time_limit=30.0):
    """
    Exact 1D coordinate descent. For each coordinate i, find the optimal
    value of a[i] (others fixed) by solving a 1D minimax problem.
    """
    f = np.maximum(f.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve_rounds = 0

    while time.monotonic() - t0 < time_limit and no_improve_rounds < 5:
        no_improve_rounds += 1
        round_improved = False

        # Get top peaks for prioritization
        top_k = min(20, 2 * n - 1)
        peak_lags = np.argpartition(conv, -top_k)[-top_k:]

        # Build list of coordinates sorted by their contribution to top peaks
        contrib = np.zeros(n)
        for k in peak_lags:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            contrib[lo:hi + 1] += f[lo:hi + 1] * f[k - np.arange(lo, hi + 1)]

        # Process coordinates in order of contribution
        order = np.argsort(contrib)[::-1]
        if n > 800:
            order = order[:400]  # limit for large n

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            a_i = f[i]
            if a_i < 1e-12:
                continue

            # Convolution entries involving a[i]:
            # c[k] for k in [i, i+n-1] (clamped to [0, 2n-2])
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            # Contribution vector: how c[k] changes when a[i] -> a[i] + d
            # c[k] += d * f[k-i] for k != 2i
            # c[2i] += 2*d*f[i] + d^2 (quadratic)
            ks = np.arange(k_lo, k_hi + 1)
            contribs = np.zeros(len(ks))
            for idx, k in enumerate(ks):
                j = k - i
                if 0 <= j < n:
                    if k == 2 * i:
                        contribs[idx] = 2.0 * f[i]
                    else:
                        contribs[idx] = f[j]

            has_quadratic = (2 * i >= k_lo and 2 * i <= k_hi)
            quad_idx = 2 * i - k_lo if has_quadratic else -1

            conv_vals = conv[ks].copy()
            other_max = conv.copy()
            other_max[k_lo:k_hi + 1] = -np.inf
            other_max_val = other_max.max() if np.any(np.isfinite(other_max)) else 0.0

            # 1D minimax: minimize max(conv_vals + d*contribs + [d^2 at quad], other_max_val)
            # Binary search / golden section on d in [-a_i, a_i * 2]
            d_lo = -a_i * 0.9
            d_hi = a_i * 1.5

            # First, sample to bracket
            best_d = 0.0
            best_local = conv.max()
            d_samples = np.linspace(d_lo, d_hi, 15)
            for d in d_samples:
                new_vals = conv_vals + d * contribs
                if has_quadratic:
                    new_vals[quad_idx] += d * d
                local_max = max(new_vals.max(), other_max_val)
                if local_max < best_local:
                    best_local = local_max
                    best_d = d

            if best_d != 0:
                # Golden section refine
                lo = best_d - (d_hi - d_lo) / 14
                hi = best_d + (d_hi - d_lo) / 14
                lo = max(lo, d_lo)
                hi = min(hi, d_hi)
                for _ in range(10):
                    m1 = lo + 0.382 * (hi - lo)
                    m2 = lo + 0.618 * (hi - lo)
                    v1_ks = conv_vals + m1 * contribs
                    if has_quadratic:
                        v1_ks[quad_idx] += m1 * m1
                    v1 = max(v1_ks.max(), other_max_val)
                    v2_ks = conv_vals + m2 * contribs
                    if has_quadratic:
                        v2_ks[quad_idx] += m2 * m2
                    v2 = max(v2_ks.max(), other_max_val)
                    if v1 < v2:
                        hi = m2
                    else:
                        lo = m1

                best_d = (lo + hi) / 2.0

                # Apply update
                f_new = f.copy()
                f_new[i] = max(0.0, f[i] + best_d)
                f_new = f_new / f_new.sum()
                conv_new = fast_convolve(f_new)
                obj_new = conv_new.max()
                if obj_new < best_obj - 1e-15:
                    best_obj = obj_new
                    best_f = f_new.copy()
                    f = f_new.copy()
                    conv = conv_new
                    round_improved = True

        if round_improved:
            no_improve_rounds = 0

    return best_f, best_obj


def paired_coordinate_search(f, time_limit=30.0, n_pairs=200):
    """
    Simultaneously optimize pairs of coordinates (i, j) that both contribute
    to the same peak. This allows mass transfer that single-coordinate
    search cannot achieve.
    """
    f = np.maximum(f.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    # Get top peaks
    top_k = min(15, 2 * n - 1)
    peak_lags = np.argpartition(conv, -top_k)[-top_k:]

    for pair_iter in range(n_pairs):
        if time.monotonic() - t0 > time_limit:
            break

        # Pick a random peak
        k = int(peak_lags[rng.integers(len(peak_lags))])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi - lo < 2:
            continue

        # Find the pair (i, j) with i + j = k that maximizes f[i]*f[j]
        # (i.e., the most "expensive" pair contributing to this peak)
        i_vals = np.arange(lo, hi + 1)
        j_vals = k - i_vals
        products = f[i_vals] * f[j_vals]
        top_pair_idx = np.argsort(products)[::-1][:5]

        for idx in top_pair_idx:
            i = int(i_vals[idx])
            j = int(j_vals[idx])
            if i == j:
                continue
            if i >= n or j >= n:
                continue

            a_i = f[i]
            a_j = f[j]
            if a_i < 1e-10 or a_j < 1e-10:
                continue

            # Try transferring mass from i to j and vice versa
            # Also try spreading to other locations
            for dest in [j, rng.integers(n), rng.integers(n)]:
                dest = int(dest)
                if dest == i:
                    continue

                # Try moving fraction of a[i] to dest
                for frac in [0.1, 0.2, 0.3, 0.5]:
                    delta = a_i * frac
                    f_trial = f.copy()
                    f_trial[i] -= delta
                    f_trial[dest] += delta
                    if f_trial[i] < 0:
                        continue
                    f_trial = np.maximum(f_trial, 0)
                    f_trial /= f_trial.sum()
                    conv_trial = fast_convolve(f_trial)
                    obj_trial = conv_trial.max()
                    if obj_trial < best_obj - 1e-15:
                        best_obj = obj_trial
                        best_f = f_trial.copy()
                        f = f_trial.copy()
                        conv = conv_trial
                        break

    return best_f, best_obj


def multiplicative_update(f, time_limit=30.0):
    """
    Exponentiated gradient / multiplicative weight update.
    f[i] <- f[i] * exp(-eta * grad[i]) / Z
    """
    f = np.maximum(f.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    eta = 0.02

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()

        k_top = min(40, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        w = lse_weights(conv[peak_idx], 30.0)
        w = w / w.sum()

        grad = compute_subgradient(f, peak_idx, w)
        grad = grad / (np.linalg.norm(grad) + 1e-15)

        f_new = f * np.exp(-eta * grad)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new.copy()
            eta = min(eta * 1.02, 0.15)
        else:
            eta = max(eta * 0.93, 0.002)

    return best_f, best_obj


def freq_perturb(f, rng, strength=0.05):
    """Perturb f in the frequency domain."""
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})", flush=True)
    return score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Main multi-phase optimization:
    Phase 1: Multi-length FW-LP exploration
    Phase 2: Spectral shaping + subgradient
    Phase 3: Local coordinate search (exact 1D)
    Phase 4: Alternating (FW + subgrad + local + spectral)
    Phase 5: Paired coordinate + multiplicative update
    Phase 6: Final high-precision polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}", flush=True)
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768.0
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    total_budget = float(budget_s) - 20.0

    # === Phase 1: Multi-length FW exploration (18%) ===
    print("\n=== Phase 1: Multi-length FW exploration ===", flush=True)
    t1_end = time.monotonic() + total_budget * 0.18

    n_candidates = sorted(set([
        n_parent,
        max(256, n_parent - 100),
        min(1100, n_parent + 100),
        max(256, int(n_parent * 0.8)),
        min(1100, int(n_parent * 1.2)),
    ]), reverse=True)
    seen = set()
    n_candidates = [x for x in n_candidates if not (x in seen or seen.add(x))]

    for nc in n_candidates[:4]:
        t_remaining = t1_end - time.monotonic()
        if t_remaining < 18:
            break
        t_alloc = min(70, t_remaining * 0.4)
        if t_alloc < 10:
            break

        if nc == n_parent:
            f_init = parent.copy()
        else:
            f_init = interpolate_to_length(parent, nc)

        print(f"  FW n={nc}, t={t_alloc:.0f}s", flush=True)
        f_opt, obj = fw_lp_optimize(f_init, time_limit=t_alloc, top_k=35)
        try_sequence(f_opt, best_tracker)

    # === Phase 2: Spectral + Subgradient (12%) ===
    print(f"\n=== Phase 2: Spectral + Subgradient ===", flush=True)
    t2_end = t1_end + total_budget * 0.12

    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(40, (t2_end - time.monotonic()) * 0.45)
    if t_alloc > 8:
        print(f"  Spectral n={len(current_best)}, t={t_alloc:.0f}s", flush=True)
        f_opt, _ = spectral_shaping(current_best, n_iters=100, time_limit=t_alloc)
        try_sequence(f_opt, best_tracker)

    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t_alloc = min(40, t2_end - time.monotonic())
    if t_alloc > 8:
        print(f"  Subgrad n={len(current_best)}, t={t_alloc:.0f}s", flush=True)
        f_opt, _ = subgradient_optimize(current_best, time_limit=t_alloc,
                                         beta_start=20.0, beta_end=8000.0, top_k=50)
        try_sequence(f_opt, best_tracker)

    # === Phase 3: Local coordinate search (15%) ===
    print(f"\n=== Phase 3: Local coordinate search ===", flush=True)
    t3_end = t2_end + total_budget * 0.15

    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(60, t3_end - time.monotonic())
    if t_alloc > 8:
        print(f"  CoordSearch n={len(current_best)}, t={t_alloc:.0f}s", flush=True)
        f_opt, _ = local_coordinate_search(current_best, time_limit=t_alloc)
        try_sequence(f_opt, best_tracker)

    # === Phase 4: Alternating optimization (30%) ===
    print(f"\n=== Phase 4: Alternating optimization ===", flush=True)
    t4_end = deadline - total_budget * 0.15
    cycles = 0
    while time.monotonic() < t4_end and cycles < 25:
        cycles += 1
        current_best = best_tracker[1] if best_tracker[1] is not None else parent
        n_cur = len(current_best)
        remaining = t4_end - time.monotonic()
        if remaining < 15:
            break

        mod = cycles % 6
        if mod == 1:
            t_alloc = min(50, remaining * 0.25)
            if t_alloc > 8:
                print(f"  C{cycles}: FW n={n_cur} t={t_alloc:.0f}s", flush=True)
                f_opt, _ = fw_lp_optimize(current_best, time_limit=t_alloc, top_k=40)
                try_sequence(f_opt, best_tracker)

        elif mod == 2:
            t_alloc = min(40, remaining * 0.25)
            if t_alloc > 8:
                beta_s = 50.0 + 40.0 * cycles
                print(f"  C{cycles}: Subgrad n={n_cur} t={t_alloc:.0f}s b={beta_s:.0f}", flush=True)
                f_opt, _ = subgradient_optimize(current_best, time_limit=t_alloc,
                                                  beta_start=beta_s, beta_end=20000.0, top_k=60)
                try_sequence(f_opt, best_tracker)

        elif mod == 3:
            t_alloc = min(40, remaining * 0.25)
            if t_alloc > 8:
                print(f"  C{cycles}: CoordSearch n={n_cur} t={t_alloc:.0f}s", flush=True)
                f_opt, _ = local_coordinate_search(current_best, time_limit=t_alloc)
                try_sequence(f_opt, best_tracker)

        elif mod == 4:
            t_alloc = min(30, remaining * 0.2)
            if t_alloc > 8:
                print(f"  C{cycles}: Spectral n={n_cur} t={t_alloc:.0f}s", flush=True)
                f_opt, _ = spectral_shaping(current_best, n_iters=60, time_limit=t_alloc)
                try_sequence(f_opt, best_tracker)

        elif mod == 5:
            t_alloc = min(30, remaining * 0.2)
            if t_alloc > 8:
                print(f"  C{cycles}: MultUpdate n={n_cur} t={t_alloc:.0f}s", flush=True)
                f_opt, _ = multiplicative_update(current_best, time_limit=t_alloc)
                try_sequence(f_opt, best_tracker)

        else:
            # Freq perturbation + quick refine
            t_alloc = min(40, remaining * 0.25)
            if t_alloc > 8:
                print(f"  C{cycles}: FreqPerturb n={n_cur} t={t_alloc:.0f}s", flush=True)
                for attempt in range(3):
                    t_rem = t4_end - time.monotonic() - 8
                    if t_rem < 6:
                        break
                    strength = 0.008 + 0.004 * attempt
                    f_pert = freq_perturb(current_best, rng, strength=strength)
                    t_sub = (t_rem - 4) / 2.0
                    if t_sub > 5:
                        f_opt, _ = local_coordinate_search(f_pert, time_limit=t_sub)
                        try_sequence(f_opt, best_tracker)

    # === Phase 5: Paired coordinate + final subgradient (12%) ===
    print(f"\n=== Phase 5: Paired + final subgradient ===", flush=True)
    t5_end = deadline - 30
    current_best = best_tracker[1] if best_tracker[1] is not None else parent

    t_alloc = min(40, (t5_end - time.monotonic()) * 0.5)
    if t_alloc > 8:
        print(f"  PairedSearch n={len(current_best)} t={t_alloc:.0f}s", flush=True)
        f_opt, _ = paired_coordinate_search(current_best, time_limit=t_alloc, n_pairs=300)
        try_sequence(f_opt, best_tracker)

    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t_alloc = min(30, t5_end - time.monotonic())
    if t_alloc > 8:
        print(f"  HighBeta Subgrad n={len(current_best)} t={t_alloc:.0f}s", flush=True)
        f_opt, _ = subgradient_optimize(current_best, time_limit=t_alloc,
                                         beta_start=2000.0, beta_end=80000.0, top_k=15)
        try_sequence(f_opt, best_tracker)

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final polish ===", flush=True)
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)

    # Very targeted local search
    t_alloc = min(25, deadline - time.monotonic() - 10)
    if t_alloc > 5:
        print(f"  Final CoordSearch n={n_cur} t={t_alloc:.0f}s", flush=True)
        f_opt, _ = local_coordinate_search(current_best, time_limit=t_alloc)
        try_sequence(f_opt, best_tracker)

    # Tiny perturbations
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t_rem = deadline - time.monotonic()
    if t_rem > 8:
        print(f"  Final perturbations t={t_rem:.0f}s", flush=True)
        for _ in range(20):
            if time.monotonic() > deadline - 3:
                break
            f_pert = freq_perturb(current_best, rng, strength=0.002)
            try_sequence(f_pert, best_tracker)
            f_pert2 = project_simplex(current_best + rng.standard_normal(n_cur) * 0.001)
            try_sequence(f_pert2, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}", flush=True)
            return [float(x) for x in final_seq]

    print("WARNING: Using uniform fallback", flush=True)
    fallback = np.ones(768) / 768.0
    return [float(x) for x in fallback]
