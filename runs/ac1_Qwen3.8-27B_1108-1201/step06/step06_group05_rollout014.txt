"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Frank-Wolfe with adaptive LP oracle (improved)
2. Spectral balancing in frequency domain
3. Coordinate descent with bisection for 1D subproblems
4. Multi-n exploration
5. Simulated annealing for escape from local minima
"""
import numpy as np
from scipy.optimize import linprog
import time
import math

def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def evaluate_normalized(p):
    """Evaluate the normalized sequence (sum=1, non-negative)."""
    n = len(p)
    conv = fast_convolve(p)
    max_c = conv.max()
    return 2.0 * n * max_c

def hankel_mv(p, a):
    """Compute the 'Hankel' matrix-vector product: M(p) @ a where M(p)_{j,m} = p[j+m]."""
    return np.correlate(p, a, mode='valid')

def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[j] * g[k-j] for valid j,k-j
    # For each k (0 to 2n-2), the constraint is: sum_j f[j] * g[k-j] <= M
    # where j ranges over max(0,k-n+1) to min(n-1,k)
    # In matrix form: A_ub[k, j] = f[j] if 0 <= k-j < n, else 0
    # A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] appears in constraints k = j, j+1, ..., j+n-1
        # with coefficient f[k-j] = f[0], f[1], ..., f[n-1]
        A_ub[j:j+n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'max_time': 60})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle and adaptive M."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        # Solve LP with current M (or slightly relaxed)
        g = solve_lp_oracle(f, M * m_exp)
        if g is None:
            m_exp = min(m_exp * 1.5, 10.0)
            if m_exp > 10.0:
                break
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Check if g is close to f (convergence)
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.02, 5.0)
            continue
        
        # Line search: find best alpha
        best_alpha = 0.0
        best_val = M
        for alpha in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Bisection refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.02)
            hi = min(1.0, best_alpha + 0.02)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            obj = 2.0 * n * M
            
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)
        
        if no_improve > 40:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.005 * min(1.0, no_improve / 80.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
        
        if verbose and it % 50 == 0:
            print(f"    FW it={it}, obj={best_obj:.8f}, M={M:.8f}, m_exp={m_exp:.2f}")
    
    return best_f, best_obj

def subgrad_descent(a0, time_limit, top_k=20, lr_init=0.02, mom_init=0.9):
    """Projected subgradient descent with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    lr = lr_init
    mom = mom_init
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        obj = conv.max()
        
        # Compute subgradient from top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            # Exponential weighting
            p_sorted = np.sort(pv)[::-1]
            weights = np.exp(-np.arange(len(peaks)) * 0.1)  # After sorting by value
            # Reorder weights to match peaks
            order = np.argsort(pv)[::-1]
            w_reordered = np.zeros(len(peaks))
            for rank, idx in enumerate(order):
                w_reordered[idx] = weights[idx]
            weights = w_reordered / w_reordered.sum()
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * a[k - idx_arr]
        
        # Momentum step with backtracking
        v_new = mom * v - lr * sg
        best_candidate = a
        best_val = obj
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            candidate = project_simplex(a + scale * v_new)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                v = v_new
                break
        
        a = best_candidate
        
        if best_val < best_obj / (2.0 * n) - 1e-15:
            best_obj = 2.0 * n * best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.95
        
        # Adapt
        if no_improve > 30:
            lr *= 0.95
            mom = min(0.97, mom + 0.003)
        if no_improve > 150:
            rng = np.random.default_rng((it * 48271 + 17) % (2**31 - 1))
            noise = 0.002 + 0.004 * min(1.0, (no_improve - 150) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init
            mom = mom_init
            no_improve = 0
    
    return best_a, best_obj

def spectral_balance(a0, time_limit, iterations=500):
    """
    Spectral balancing: work in frequency domain to flatten the autocorrelation.
    Key idea: the autocorrelation is the IDFT of |F(p)|^2.
    We iteratively adjust the "amplitude" of the spectrum to flatten the time-domain signal.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit and it < iterations:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Compute spectrum
        P = np.fft.fft(a)
        S = np.abs(P) ** 2  # Power spectrum
        
        # The periodic autocorrelation is IDFT of S
        # c_periodic = IDFT(S)
        # We want to minimize max(c_periodic)
        
        # Iteratively: reduce the peaks in time domain by adjusting S
        # This is similar to Gerchberg-Saxton / error reduction
        
        c = np.fft.ifft(S).real  # Periodic autocorrelation
        
        # Find peaks and try to reduce them
        # Adjust S to reduce peaks: S_new = S - eta * gradient
        # Gradient of max(c) w.r.t. S is the DFT basis at the peak location
        peak_idx = np.argmax(c)
        
        # Simple adjustment: reduce S at frequencies that contribute most to the peak
        # This is heuristic
        eta = 0.1 / (it + 10)
        S[peak_idx] -= eta * c[peak_idx]
        S[n - peak_idx] -= eta * c[n - peak_idx] if peak_idx != 0 else 0
        S = np.maximum(S, 0)
        
        # Recover amplitude and recompute p
        # We need |P_new|^2 = S_new, and P_new = FFT(p) with p >= 0
        # This is phase retrieval - use iterative projection
        for _ in range(3):
            P = np.sqrt(S) * np.exp(1j * np.angle(P))  # Update amplitude, keep phase
            p_new = np.fft.ifft(P).real
            p_new = np.maximum(p_new, 0)
            s = p_new.sum()
            if s < 1e-15:
                break
            p_new = p_new / s
            P = np.fft.fft(p_new)
            S = np.abs(P) ** 2
        
        a = p_new
        obj = 2.0 * n * fast_convolve(a).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj

def coordinate_descent(a0, time_limit, block_size=1):
    """
    Coordinate descent: update one (or a few) coordinates at a time.
    For a single coordinate, the 1D subproblem can be solved by bisection.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        # Select coordinate to update
        # Strategy: prioritize coordinates that are at the boundary of the support
        # or that have high "potential" for reduction
        conv = fast_convolve(a)
        max_c = conv.max()
        peak_k = np.argmax(conv)
        
        # The subgradient at peak_k for coordinate i is 2*a[peak_k - i]
        # We want to decrease coordinates with high subgradient at the peak
        # and increase coordinates with low subgradient
        if 0 <= peak_k < n:
            sg = 2.0 * a[peak_k - np.arange(n)]
        else:
            # For peak_k >= n, the subgradient is a[peak_k - i] where 0 <= peak_k - i < n
            # i ranges from peak_k - n + 1 to n - 1
            sg = np.zeros(n)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            sg[js:je+1] = 2.0 * a[peak_k - np.arange(js, je+1)]
        
        # Pick the coordinate with the highest subgradient to decrease
        # and the one with the lowest to increase
        if a.sum() > 0:
            # Find candidates for decrease (high sg, high value)
            candidate_decrease = np.argmax(sg * np.maximum(a, 1e-10))
            candidate_increase = np.argmin(sg * np.maximum(a, 1e-10))  # or where a is low
            
            # Actually, let's do a random coordinate update for diversity
            if rng.random() < 0.5:
                # Random coordinate
                i = rng.integers(0, n)
            else:
                # Subgradient-guided: decrease high-sg coordinate
                i = candidate_decrease
            
            # Solve 1D subproblem: change a[i] to optimize
            # Moving mass from i to a random j (or to minimize the max)
            # Simple approach: try decreasing a[i] by delta and increasing a[j] by delta
            # for a few candidate j values
            
            delta_options = [0.001, 0.005, 0.01, 0.02, 0.05]
            best_delta = 0
            best_j = -1
            best_val = max_c
            
            # Try a few target coordinates
            target_candidates = [candidate_increase]
            target_candidates += [int(x) for x in rng.integers(0, n, size=5)]
            
            for j in set(target_candidates):
                if j == i:
                    continue
                if a[j] < 0 and a[i] < 0:
                    continue
                for delta in delta_options:
                    if a[i] < delta:
                        continue
                    a_new = a.copy()
                    a_new[i] -= delta
                    a_new[j] += delta
                    a_new = np.maximum(a_new, 0)
                    a_new = a_new / a_new.sum()
                    val = fast_convolve(a_new).max()
                    if val < best_val:
                        best_val = val
                        best_delta = delta
                        best_j = j
            
            if best_j >= 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0)
                a = a / a.sum()
                obj = 2.0 * n * fast_convolve(a).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
    
    return best_a, best_obj

def sim_annealing(a0, time_limit, T_init=0.01, T_min=0.0001, cool_rate=0.995):
    """Simulated annealing on the probability simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    curr_obj = best_obj
    
    T = T_init
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    while time.monotonic() - t0 < time_limit and T > T_min:
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Propose a move: random exchange
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = T * 2 * a[i]  # Scale perturbation with temperature
        if delta > a[i] * 0.9:
            delta = a[i] * 0.9
        
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        
        new_obj = 2.0 * n * fast_convolve(a_new).max()
        delta_obj = new_obj - curr_obj
        
        if delta_obj < 0 or rng.random() < math.exp(-delta_obj / T):
            a = a_new
            curr_obj = new_obj
            if curr_obj < best_obj:
                best_obj = curr_obj
                best_a = a.copy()
        
        T *= cool_rate
    
    return best_a, best_obj

def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s

def try_and_update(a, best_seq, best_score, label=""):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return best_seq, best_score

def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search:
    1. Quick exploration with parent and symmetrization
    2. Frank-Wolfe with LP oracle (main workhorse)
    3. Subgradient descent with adaptive peaks
    4. Spectral balancing
    5. Simulated annealing
    6. Multi-n exploration
    7. Final polishing
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")
    
    n_parent = len(parent)
    
    # === Phase 1: Quick exploration (~100s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_start = time.monotonic()
    
    # Symmetrized version
    sym_parent = (parent + parent[::-1]) / 2.0
    sym_parent = sym_parent / sym_parent.sum()
    best_seq, best_score = try_and_update(sym_parent, best_seq, best_score, "sym")
    
    # Quick subgradient on parent
    remaining = deadline - time.monotonic() - 1400
    if remaining > 20:
        t_alloc = min(remaining, 50)
        a_opt, _ = subgrad_descent(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "sg_quick")
    
    # Quick FW
    remaining = deadline - time.monotonic() - 1300
    if remaining > 20:
        t_alloc = min(remaining, 50)
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "fw_quick")
    
    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")
    
    # === Phase 2: Multi-n exploration (~200s) ===
    print("\n=== Phase 2: Multi-n Exploration ===")
    t_start = time.monotonic()
    
    n_best = len(best_seq)
    # Try several n values
    n_candidates = set()
    for n_try in [256, 384, 512, 640, 768, 896, 1024, 1280, 1536]:
        if 128 <= n_try <= 2048:
            n_candidates.add(n_try)
    n_best in n_candidates or n_candidates.add(n_best)
    
    for n_try in sorted(n_candidates):
        remaining = deadline - time.monotonic() - 900
        if remaining < 30:
            break
        
        if n_try == n_best:
            start = best_seq.copy()
        else:
            start = interp_sequence(best_seq, n_try)
        
        t_alloc = min(remaining, 40)
        print(f"  Exploring n={n_try}, t={t_alloc:.0f}s")
        
        # Subgradient
        a_opt, _ = subgrad_descent(start, time_limit=t_alloc * 0.5)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"sg_n{n_try}")
        
        # FW (if time permits)
        remaining = deadline - time.monotonic() - 800
        if remaining > 20:
            a_opt2, _ = fw_lp_optimize(a_opt, time_limit=min(remaining, t_alloc * 0.5))
            best_seq, best_score = try_and_update(a_opt2, best_seq, best_score, f"fw_n{n_try}")
    
    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")
    
    # === Phase 3: Deep FW optimization (~500s) ===
    print("\n=== Phase 3: Deep FW ===")
    t_start = time.monotonic()
    
    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 500
    if remaining > 30:
        t_alloc = min(remaining, 400)
        print(f"  Deep FW n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, verbose=True)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "deep_fw")
    
    # FW with perturbations
    for scale in [0.003, 0.008, 0.015, 0.025]:
        remaining = deadline - time.monotonic() - 350
        if remaining < 20:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc = min(remaining, 60)
        a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"fw_pert_{scale}")
    
    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")
    
    # === Phase 4: Spectral + Subgradient refinement (~300s) ===
    print("\n=== Phase 4: Spectral & Subgradient ===")
    t_start = time.monotonic()
    
    n_best = len(best_seq)
    
    # Spectral balancing
    remaining = deadline - time.monotonic() - 300
    if remaining > 20:
        t_alloc = min(remaining, 100)
        print(f"  Spectral balancing, t={t_alloc:.0f}s")
        a_opt, _ = spectral_balance(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "spectral")
    
    # Multiple subgradient configs
    for top_k, lr_init, mom_init in [(10, 0.015, 0.93), (20, 0.01, 0.95), (40, 0.005, 0.92)]:
        remaining = deadline - time.monotonic() - 200
        if remaining < 15:
            break
        t_alloc = min(remaining, 60)
        a_opt, _ = subgrad_descent(best_seq, time_limit=t_alloc, top_k=top_k, lr_init=lr_init, mom_init=mom_init)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"sg_k{top_k}")
    
    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")
    
    # === Phase 5: Simulated annealing + local search (~200s) ===
    print("\n=== Phase 5: SA & Local ===")
    t_start = time.monotonic()
    
    n_best = len(best_seq)
    
    # Simulated annealing
    remaining = deadline - time.monotonic() - 150
    if remaining > 20:
        t_alloc = min(remaining, 80)
        print(f"  Simulated annealing, t={t_alloc:.0f}s")
        a_opt, _ = sim_annealing(best_seq, time_limit=t_alloc, T_init=0.005)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "sa")
    
    # Coordinate descent
    remaining = deadline - time.monotonic() - 100
    if remaining > 15:
        t_alloc = min(remaining,
