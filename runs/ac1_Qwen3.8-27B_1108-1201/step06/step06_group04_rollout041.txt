```python
"""
Search for a non-negative step-function height sequence minimizing
    2*n*max(convolve(a,a))/(sum(a))^2.

The approach is a hybrid local/global search:
- Fast FFT convolutions and gradients.
- Mirror-descent on a softmax-smoothed max-conv objective.
- Direct top-peak subgradient polishing.
- Multi-n exploration starting from the parent construction,
  anti-palindromic/sinusoidal seeds, and interpolated parent seeds.
"""

import numpy as np
import time


def _score_key(item):
    return item[0]


def next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    N = next_pow2(L)
    fa = np.fft.rfft(a, N)
    out = np.fft.irfft(fa * fa, N)[:L]
    return np.maximum(out, 0.0)


def fast_conv_with_N(a, N):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    L = 2 * n - 1
    fa = np.fft.rfft(a, N)
    out = np.fft.irfft(fa * fa, N)[:L]
    return np.maximum(out, 0.0)


def fast_score(a):
    a = normalize_nonneg(a)
    n = int(a.size)
    if n == 0:
        return float("inf")
    c = fast_conv(a)
    return float(2.0 * n * np.max(c))


def logsumexp_scaled(c, T):
    c = np.asarray(c, dtype=np.float64)
    if T <= 0.0:
        return float(np.max(c))
    x = c / T
    xm = float(np.max(x))
    return float(xm + np.log(np.sum(np.exp(x - xm))))


def softmax_from_c(c, T):
    c = np.asarray(c, dtype=np.float64)
    L = int(c.size)
    if L == 0:
        return np.array([], dtype=np.float64)
    if T <= 1e-300:
        m = float(np.max(c))
        tol = max(1e-15, 1e-12 * abs(m))
        idx = np.where(c >= m - tol)[0]
        p = np.zeros(L, dtype=np.float64)
        p[idx] = 1.0 / float(idx.size)
        return p
    x = c / T
    xm = float(np.max(x))
    e = np.exp(x - xm)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros(L, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
        return p
    return e / s


def interpolate_sequence(a, n):
    a = normalize_nonneg(a)
    m = int(a.size)
    n = int(n)
    if n <= 0:
        return np.array([1.0], dtype=np.float64)
    if m == n:
        return a.copy()
    if m == 1:
        return np.array([a[0]], dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    vals = np.interp(x_new, x_old, a)
    return normalize_nonneg(vals)


def make_uniform(n):
    n = int(n)
    if n <= 0:
        return np.array([1.0], dtype=np.float64)
    return np.ones(n, dtype=np.float64) / float(n)


def make_antipal(n):
    n = int(n)
    if n <= 0:
        return np.array([1.0], dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    v = np.sin(np.pi * (2.0 * x - 1.0))
    if n > 4:
        v = 0.80 * v + 0.35 * np.sin(3.0 * np.pi * (2.0 * x - 1.0))
    vals = 1.0 + 0.62 * v
    vals = np.maximum(vals, 1e-6)
    return normalize_nonneg(vals)


def make_periodic(n, freq, phase, amp):
    n = int(n)
    if n <= 0:
        return np.array([1.0], dtype=np.float64)
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    vals = 1.0 + float(amp) * np.sin(2.0 * np.pi * float(freq) * x + float(phase))
    vals = np.maximum(vals, 1e-6)
    return normalize_nonneg(vals)


def make_ramp(n, direction):
    n = int(n)
    if n <= 0:
        return np.array([1.0], dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    if int(direction) >= 0:
        vals = 0.25 + 1.75 * x
    else:
        vals = 2.0 - 1.75 * x
    return normalize_nonneg(vals)


def optimize_soft(a0, max_time, seed=0, verbose=False):
    """
    Smoothing softmax mirror descent.

    Minimize F_T(a) = log sum_k exp(c_k(a)/T), where c_k(a) are
    linear auto-convolution coefficients, under sum(a)=1, a>=0.
    """
    a = normalize_nonneg(a0)
    n = int(a.size)
    if n <= 1 or max_time <= 0.2:
        return a

    L = 2 * n - 1
    N = next_pow2(3 * n - 2)
    idx = 2 * n - 2 - np.arange(n)

    loga = np.log(np.maximum(a, 1e-300))
    a_fft = np.fft.rfft(a, N)
    c = fast_conv_with_N(a, N)
    M = float(np.max(c))

    best_a = a.copy()
    best_M = M

    alpha = 0.30
    t_frac = 0.20
    min_tf = 1.0e-3
    max_tf = 0.80
    clip = 25.0

    no_imp = 0
    imp_streak = 0
    it = 0
    t0 = time.monotonic()
    limit = max(0.1, float(max_time))

    while time.monotonic() - t0 < limit - 0.05:
        it += 1

        T = t_frac * max(M, 1.0 / float(n))
        if T <= 1e-300:
            T = 1e-300

        p = softmax_from_c(c, T)
        F = logsumexp_scaled(c, T)

        p_fft = np.fft.rfft(p[::-1], N)
        g = np.fft.irfft(a_fft * p_fft, N)
        grad = 2.0 * g[idx]

        gmax = float(np.max(np.abs(grad)))
        accepted = False

        if np.isfinite(gmax) and gmax > 1e-30:
            eta_base = alpha / gmax
            multipliers = (1.0, 0.5, 0.25, 0.125, 0.0625)

            for mult in multipliers:
                eta = eta_base * mult
                delta = -eta * grad
                delta -= float(np.mean(delta))
                delta = np.clip(delta, -clip, clip)

                la = loga + delta
                la_m = float(np.max(la))
                a_new = np.exp(la - la_m)
                s = float(np.sum(a_new))
                if not np.isfinite(s) or s <= 1e-300:
                    continue
                a_new = a_new / s

                a_new_fft = np.fft.rfft(a_new, N)
                c_new = fast_conv_with_N(a_new, N)
                M_new = float(np.max(c_new))
                if not np.isfinite(M_new):
                    continue

                accept = False
                if M_new < M - 1e-15:
                    accept = True
                else:
                    F_new = logsumexp_scaled(c_new, T)
                    if F_new < F and M_new <= M + 1e-12:
                        accept = True

                if accept:
                    a = a_new
                    c = c_new
                    M = M_new
                    loga = np.log(np.maximum(a, 1e-300))
                    a_fft = a_new_fft
                    accepted = True
                    break

        if accepted:
            no_imp = 0
            imp_streak += 1
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
        else:
            no_imp += 1
            imp_streak = 0

        if imp_streak > 10:
            t_frac = max(min_tf, t_frac * 0.98)
            alpha = min(1.8, alpha * 1.04)
            imp_streak = 0

        if no_imp > 15:
            t_frac = min(max_tf, t_frac * 1.25)
            alpha = max(0.03, alpha * 0.85)

        if no_imp > 45:
            clip = max(6.0, clip * 0.85)

        if no_imp > 120:
            rng = np.random.default_rng(int(seed) + 1000003 * (it % 1000003))
            noise = rng.standard_normal(n) * 0.10
            diff = best_a - best_a[::-1]
            ds = float(np.std(diff))
            if ds > 1e-15:
                noise = noise + 0.25 * diff / ds
            noise -= float(np.mean(noise))
            noise = np.clip(noise, -2.0, 2.0)
            a = best_a * np.exp(noise)
            a = normalize_nonneg(a)
            loga = np.log(np.maximum(a, 1e-300))
            a_fft = np.fft.rfft(a, N)
            c = fast_conv_with_N(a, N)
            M = float(np.max(c))
            t_frac = max(t_frac, 0.18)
            alpha = 0.30
            clip = 25.0
            no_imp = 0
            imp_streak = 0

        if verbose and it % 2000 == 0:
            print(f"      soft it={it}, n={n}, M={M:.10g}, score={2.0*float(n)*M:.10f}, Tfrac={t_frac:.4g}")

    return best_a


def optimize_peaks(a0, max_time, seed=0, q_start=12, verbose=False):
    """
    Direct top-peak mirror descent.

    At each step, weight the current top-q convolution coefficients
    uniformly and take a multiplicative step to reduce their values.
    """
    a = normalize_nonneg(a0)
    n = int(a.size)
    if n <= 1 or max_time <= 0.2:
        return a

    L = 2 * n - 1
    N = next_pow2(3 * n - 2)
    idx = 2 * n - 2 - np.arange(n)

    loga = np.log(np.maximum(a, 1e-300))
    a_fft = np.fft.rfft(a, N)
    c = fast_conv_with_N(a, N)
    M = float(np.max(c))

    best_a = a.copy()
    best_M = M

    q = int(min(max(1, q_start), L))
    alpha = 0.12
    clip = 15.0
    no_imp = 0
    acc_streak = 0
    it = 0
    t0 = time.monotonic()
    limit = max(0.1, float(max_time))

    while time.monotonic() - t0 < limit - 0.05:
        it += 1

        if q >= L:
            peaks = np.arange(L)
        else:
            peaks = np.argpartition(c, L - q)[L - q:]

        p = np.zeros(L, dtype=np.float64)
        p[peaks] = 1.0 / float(q)

        p_fft = np.fft.rfft(p[::-1], N)
        g = np.fft.irfft(a_fft * p_fft, N)
        grad = 2.0 * g[idx]

        gmax = float(np.max(np.abs(grad)))
        accepted = False

        if np.isfinite(gmax) and gmax > 1e-30:
            eta_base = alpha / gmax
            multipliers = (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125)

            for mult in multipliers:
                eta = eta_base * mult
                delta = -eta * grad
                delta -= float(np.mean(delta))
                delta = np.clip(delta, -clip, clip)

                la = loga + delta
                la_m = float(np.max(la))
                a_new = np.exp(la - la_m)
                s = float(np.sum(a_new))
                if not np.isfinite(s) or s <= 1e-300:
                    continue
                a_new = a_new / s

                a_new_fft = np.fft.rfft(a_new, N)
                c_new = fast_conv_with_N(a_new, N)
                M_new = float(np.max(c_new))
                if not np.isfinite(M_new):
                    continue

                if M_new < M - 1e-15:
                    a = a_new
                    c = c_new
                    M = M_new
                    loga = np.log(np.maximum(a, 1e-300))
                    a_fft = a_new_fft
                    accepted = True
                    break

        if accepted:
            no_imp = 0
            acc_streak += 1
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            if acc_streak > 15:
                q = max(1, q // 2)
                alpha = min(0.8, alpha * 1.05)
                acc_streak = 0
        else:
            no_imp += 1
            acc_streak = 0
            if no_imp > 4:
                q = min(L, q * 2)
                alpha = max(0.02, alpha * 0.85)

        if no_imp > 40:
            clip = max(5.0, clip * 0.85)

        if no_imp > 100:
            rng = np.random.default_rng(int(seed) + 2000003 * (it % 1000003))
            noise = rng.standard_normal(n) * 0.10
            diff = best_a - best_a[::-1]
            ds = float(np.std(diff))
            if ds > 1e-15:
                noise = noise + 0.30 * diff / ds
            noise -= float(np.mean(noise))
            noise = np.clip(noise, -2.0, 2.0)
            a = best_a * np.exp(noise)
            a = normalize_nonneg(a)
            loga = np.log(np.maximum(a, 1e-300))
            a_fft = np.fft.rfft(a, N)
            c = fast_conv_with_N(a, N)
            M = float(np.max(c))
            q = max(5, q // 2)
            alpha = 0.12
            clip = 15.0
            no_imp = 0
            acc_streak = 0

        if verbose and it % 2000 == 0:
            print(f"      peaks it={it}, n={n}, M={M:.10g}, score={2.0*float(n)*M:.10f}, q={q}")

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Entry point. Must return the best non-negative list found before timeout.
    """
    t_start = time.monotonic()
    total = max(60.0, float(budget_s))
    deadline = t_start + max(60.0, total - 60.0)

    # Load parent construction if present.
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = normalize_nonneg(p)
        if p.size > 0 and finite_ok(p):
            parent = p
    except Exception:
        parent = None

    if parent is None:
        parent = make_uniform(768)

    n_parent = int(parent.size)
    best_a = parent.copy()
    best_score = fast_score(best_a)

    print(f"Parent n={n_parent}, initial fast_score={best_score:.10f}")

    pool = []
    pool.append([best_score, best_a.copy(), "parent"])

    # Deterministic anti/symmetric perturbations of the parent.
    if n_parent > 1:
        diff = best_a - best_a[::-1]
        ds = float(np.std(diff))
        if ds > 1e-15:
            for eps in (0.05, 0.15, 0.30):
                cand = best_a * np.exp(eps * diff / ds)
                cand = normalize_nonneg(cand)
                sc = fast_score(cand)
                if np.isfinite(sc):
                    pool.append([sc, cand.copy(), f"mulanti{eps}"])
        sym = (best_a + best_a[::-1]) * 0.5
        sym = normalize_nonneg(sym)
        sc = fast_score(sym)
        if np.isfinite(sc):
            pool.append([sc, sym.copy(), "sym"])

    # Candidate lengths.
    ns_set = set()
    ns_set.add(n_parent)
    for d in (-128, -64, 64, 128, 256):
        v = n_parent + d
        if 64 <= v <= 6144:
            ns_set.add(v)
    for v in (256, 512, 768, 1024, 1536, 2048, 3072, 4096, 6144):
        ns_set.add(v)

    ns = []
    for v in ns_set:
        v = int(v)
        if 64 <= v <= 6144:
            ns.append(v)
    ns.sort()

    rng = np.random.default_rng(int(seed))
    starts = []

    for n in ns:
        if parent is not None:
            starts.append((interpolate_sequence(parent, n), f"inter{n}"))
        starts.append((make_antipal(n), f"anti{n}"))
        starts.append((make_periodic(n, 1, 0.0, 0.80), f"per1{n}"))
        starts.append((make_periodic(n, 2, 1.0, 0.60), f"per2{n}"))
        starts.append((make_ramp(n, 1), f"ramp{n}"))
        if n <= 2048:
            if n <= 1024:
                starts.append((rng.dirichlet(np.ones(n) * 1.0), f"rand{n}"))
        if n <= 1536:
            x = np.linspace(0.0, 1.0, n)
            vals = 1.0 + 0.45 * np.cos(2.0 * np.pi * x) + 0.25 * np.sin(4.0 * np.pi * x)
            starts.append((normalize_nonneg(vals), f"mix{n}"))

    # Initial direct scores.
    for a, label in starts:
        sc = fast_score(a)
        if np.isfinite(sc):
            pool.append([sc, a, label])

    pool.sort(key=_score_key)
    if len(pool) > 80:
        pool = pool[:80]

    # Quick screening of promising starts.
    topK = min(12, len(pool))
    screened = 0
    for i in range(topK):
        rem = deadline - time.monotonic()
        if rem < 800.0:
            break
        sc, a, label = pool[i]
        # Skip very bad initial starts except the first few.
        if i > 2 and sc > best_score + 1.0:
            continue
        qtime = min(15.0, max(3.0, (rem - 800.0) / max(1, topK - i)))
        if qtime < 3.0:
            break
        a_opt = optimize_soft(a, qtime, seed=int(seed) * 1000 + i)
        sc_opt = fast_score(a_opt)
        if np.isfinite(sc_opt):
            if sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt.copy()
                print(f"  NEW BEST quick {label}: {sc_opt:.10f} n={len(best_a)}")
            pool.append([sc_opt, a_opt.copy(), label + "_opt"])
            pool.sort(key=_score_key)
            if len(pool) > 100:
                pool = pool[:100]
        screened += 1

    print(f"After screening: best={best_score:.10f} n={len(best_a)} screened={screened}")

    # Build a small list of deep candidates.
    deep_list = []
    deep_list.append((best_a.copy(), "deep_best"))

    for item in pool[:6]:
        if len(deep_list) >= 7:
            break
        deep_list.append((item[1].copy(), item[2] + "_deep"))

    # Make sure at least one longer-than-parent candidate is considered
    # if it has been screened or initialized reasonably.
    large_count = 0
    for a, _ in deep_list:
        if int(a.size) > int(n_parent * 1.2):
            large_count += 1
    for item in pool:
        if large_count >= 2:
            break
        if int(item[1].size) > int(n_parent * 1.2):
            deep_list.append((item[1].copy(), item[2] + "_large"))
            large_count += 1

    # Deep optimization of top candidates.
    for di, (a_cand, label) in enumerate(deep_list):
        rem = deadline - time.monotonic()
        if rem < 250.0:
            break
        per = max(40.0, (rem - 250.0) / max(1, len(deep_list) - di))

        t_soft = min(160.0, per * 0.65)
        if t_soft < 15.0:
            break
        a2 = optimize_soft(a_cand, t_soft, seed=int(seed) * 7777 + di)
        s2 = fast_score(a2)
        if np.isfinite(s2):
            if s2 < best_score:
                best_score = s2
                best_a = a2.copy()
                print(f"  NEW BEST deep soft {label}: {s2:.10f} n={len(best_a)}")
            pool.append([s2, a2.copy(), label + "_softdeep"])
            pool.sort(key=_score_key)
            if len(pool) > 120:
                pool = pool[:120]

        rem = deadline - time.monotonic()
        t_peak = min(80.0, max(15.0, per * 0.35))
        if rem > 120.0:
            a3 = optimize_peaks(a2, t_peak, seed=int(seed) * 8888 + di, q_start=10)
            s3 = fast_score(a3)
            if np.isfinite(s3):
                if s3 < best_score:
                    best_score = s3
                    best_a = a3.copy()
                    print(f"  NEW BEST deep peaks {label}: {s3:.10f} n={len(best_a)}")
                pool.append([s3, a3.copy(), label + "_peakdeep"])
                pool.sort(key=_score_key)
                if len(pool) > 120:
                    pool = pool[:120]

    # Final iterative polishing, alternating smooth and peak methods.
    method_iter = 0
    stagnant = 0

    while deadline - time.monotonic() > 95.0:
        rem = deadline - time.monotonic()
        if rem < 95.0:
            break

        start = best_a.copy()

        # Occasional perturbation to escape local minima.
        if stagnant >= 3:
            rng2 = np.random.default_rng(int(seed) + 999983 * (method_iter + 1))
            noise = rng2.standard_normal(len(start)) * 0.06
            if len(start) > 1:
                d = start - start[::-1]
                ds = float(np.std(d))
                if ds > 1e-15:
                    noise = noise + 0.35 * d / ds
            noise -= float(np.mean(noise))
            noise = np.clip(noise, -2.0, 2.0)
            start = normalize_nonneg(start * np.exp(noise))

        t_step = min(220.0, rem - 45.0)
        if t_step < 20.0:
            break

        if method_iter % 2 == 0:
            a2 = optimize_soft(start, t_step, seed=int(seed) + 500000 + method_iter)
            label = "final_soft"
        else:
            a2 = optimize_peaks(start, t_step, seed=int(seed) + 600000 + method_iter, q_start=6)
            label = "final_peaks"

        s2 = fast_score(a2)
        if np.isfinite(s2) and s2 < best_score - 1e-15:
            best_score = s2
            best_a = a2.copy()
            stagnant = 0
            print(f"  NEW BEST final {label}: {best_score:.10f} n={len(best_a)}")
            pool.append([best_score, best_a.copy(), label])
            pool.sort(key=_score_key)
            if len(pool) > 120:
                pool = pool[:120]
        else:
            stagnant += 1

        method_iter += 1

    # Final verification using the official evaluator on a few top candidates.
    final_best_a = best_a.copy()
    final_best_score = float("inf")

    check_items = []
    check_items.append(best_a)
    for item in pool[:5]:
        check_items.append(item[1])

    for a in check_items:
        a = np.asarray(a, dtype=np.float64)
        vals = [float(x) for x in a]
        try:
            sc = evaluate_sequence(vals)
        except Exception:
            continue
        if np.isfinite(sc) and sc < final_best_score:
            final_best_score = float(sc)
            final_best_a = a.copy()

    if not np.isfinite(final_best_score):
        final_best_score = best_score
        final_best_a = best_a

    print(f"FINAL fast_score={best_score:.10f}, evaluated={final_best_score:.10f}, n={len(final_best_a)}")
    return [float(x) for x in final_best_a]


def finite_ok(a):
    try:
        return bool(np.all(np.isfinite(a)))
    except Exception:
        return False
```
