    """Build the (2n-1) x n matrix A where A[k,j] = f[k-j] if 0 <= k-j < n."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M, A_ub=None, lp_time=5.0):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns (g_primal, mu_dual) or (None, None) on failure.
    """
    n = len(f)
    c = -np.ones(n)
    if A_ub is None:
        A_ub = build_constraint_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                     method='highs',
                     options={'presolve': True, 'time_limit': lp_time,
                              'dual_feasibility_tolerance': 1e-10,
                              'primal_feasibility_tolerance': 1e-10})
        if res.status in (0, 5) and res.x is not None:  # 0=optimal, 5=time limit
            g = np.maximum(res.x, 0.0)
            mu = np.zeros(2 * n - 1)
            if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                if hasattr(res.ineqlin, 'marginals') and res.ineqlin.marginals is not None:
                    mu = np.maximum(res.ineqlin.marginals, 0.0)
            return g, mu
    except Exception:
        pass
    return None, None


def fw_with_lp(a0, time_limit, lp_time=5.0, verbose=False):
    """
    Frank-Wolfe with LP oracle and adaptive step size.
    Normalizes input to sum=1. Returns (best_sequence, best_max_conv).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0  # relaxation factor for M

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 1.0:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Solve LP with possibly relaxed M
        g, mu = solve_lp_oracle(a, M * m_relax, lp_time=min(lp_time, remaining * 0.25))

        if g is None or g.sum() < 1e-15:
            no_improve += 1
            m_relax *= 1.2
            if m_relax > 4.0:
                m_relax = 1.0
            continue

        g = g / g.sum()

        # Check if LP solution is same as current
        if np.max(np.abs(g - a)) < 1e-13:
            no_improve += 1
            m_relax = min(m_relax * 1.05, 3.0)
            if no_improve > 50:
                break
            continue

        # Line search: try several step sizes
        best_alpha = 0.0
        best_val = M
        candidate_best = a

        # Coarse search
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * a + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
                candidate_best = ft.copy()

        # Fine search around best_alpha
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(15):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * a + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    candidate_best = ft.copy()
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            a = candidate_best
            m_relax = 1.0

            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.1, 5.0)

        # Restart if stuck
        if no_improve > 40:
            rng = np.random.default_rng(it * 31 + 67)
            scale = 0.002 + 0.004 * min(1.0, (no_improve - 40) / 100.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            m_relax = 1.0
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    FW it={it}, obj={best_obj:.10f}, no_improve={no_improve}")

    return best_a, best_obj
