import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient_step(a, T, lr):
    """
    One step of projected gradient descent on the log-sum-exp surrogate.
    Returns (new_a, conv, cmax).
    """
    n = len(a)
    conv = fast_convolve(a)
    cmax = conv.max()
    
    # Compute softmax weights
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()
    
    # Gradient: 4n * cross_correlation(p, a)
    cross = np.correlate(p, a, mode='valid')
    grad = 4.0 * n * cross
    
    # Projected gradient step
    a_new = project_simplex(a - lr * grad)
    return a_new, conv, cmax

        it += 1
        
        # Choose pair strategy: cycle through peaks and random
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        
        # Indices contributing to peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = np.arange(js, je + 1)
        
        # Pick a pair: one from contributing (high), one from non-contributing
        if len(contrib) > 0:
            # Find the index in contrib with largest a value
            i_idx = contrib[np.argmax(a[contrib])]
            # Find non-contributing index with smallest a value (to receive mass)
            non_contrib = np.setdiff1d(np.arange(n), contrib)
            if len(non_contrib) > 0:
                j_idx = non_contrib[np.argmin(a[non_contrib])]
            else:
                j_idx = (i_idx + 1) % n
        else:
            i_idx = it % n
            j_idx = (it + 1) % n
        
        # Golden section search for optimal d
        # a[i] += d, a[j] -= d, d in [-a[i], a[j]]
        d_lo = -a[i_idx]
        d_hi = a[j_idx]
        
        if d_hi - d_lo < 1e-15:
            continue
        
        # Precompute coefficients
        # conv_new[k] = base[k] + d * lin[k] + d^2 * quad[k]
        base = conv  # current conv values
        k_range = np.arange(0, 2 * n - 1)
        
        # linear coefficient: 2*(a[k-i] - a[k-j])
        lin = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            val = 0.0
            ki = k - i_idx
            kj = k - j_idx
            if 0 <= ki < n:
                val += a[ki]
            if 0 <= kj < n:
                val -= a[kj]
            lin[k] = 2.0 * val
        
        # quadratic coefficient: 1[2i=k] - 2*1[i+j=k] + 1[2j=k]
        quad = np.zeros(2 * n - 1)
        if 2 * i_idx < 2 * n - 1:
            quad[2 * i_idx] += 1.0
        if i_idx + j_idx < 2 * n - 1:
            quad[i_idx + j_idx] -= 2.0
        if 2 * j_idx < 2 * n - 1:
            quad[2 * j_idx] += 1.0
        
        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        d1 = d_hi - gr * (d_hi - d_lo)
        d2 = d_lo + gr * (d_hi - d_lo)
        f1 = np.max(base + d1 * lin + d1 * d1 * quad)
        f2 = np.max(base + d2 * lin + d2 * d2 * quad)
        
        for _ in range(50):
            if f1 < f2:
                d_hi = d2
                d2 = d1
                f2 = f1
                d1 = d_hi - gr * (d_hi - d_lo)
                f1 = np.max(base + d1 * lin + d1 * d1 * quad)
            else:
                d_lo = d1
                d1 = d2
                f1 = f2
                d2 = d_lo + gr * (d_hi - d_lo)
                f2 = np.max(base + d2 * lin + d2 * d2 * quad)
        
        d_opt = (d_lo + d_hi) / 2.0
        f_opt = np.max(base + d_opt * lin + d_opt * d_opt * quad)
        
        if f_opt < best_obj - 1e-15:
            a[i_idx] += d_opt
            a[j_idx] -= d_opt
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                break
            a = a / s
            obj = fast_convolve(a).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        
        # Also try random pairs periodically
        if it % 10 == 0:
            for _ in range(3):
                i_r = rng.integers(0, n)
                j_r = rng.integers(0, n)
                if i_r == j_r:
                    continue
                conv = fast_convolve(a)
                d_lo = -a[i_r]
                d_hi = a[j_r]
                if d_hi - d_lo < 1e-15:
                    continue
                
                base = conv
                lin = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    val = 0.0
                    ki = k - i_r
                    kj = k - j_r
                    if 0 <= ki < n:
                        val += a[ki]
                    if 0 <= kj < n:
                        val -= a[kj]
                    lin[k] = 2.0 * val
                
                quad = np.zeros(2 * n - 1)
                if 2 * i_r < 2 * n - 1:
                    quad[2 * i_r] += 1.0
                if i_r + j_r < 2 * n - 1:
                    quad[i_r + j_r] -= 2.0
                if 2 * j_r < 2 * n - 1:
                    quad[2 * j_r] += 1.0
                
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                d1 = d_hi - gr * (d_hi - d_lo)
                d2 = d_lo + gr * (d_hi - d_lo)
                f1 = np.max(base + d1 * lin + d1 * d1 * quad)
                f2 = np.max(base + d2 * lin + d2 * d2 * quad)
                
                for _ in range(40):
                    if f1 < f2:
                        d_hi = d2
                        d2 = d1
                        f2 = f1
                        d1 = d_hi - gr * (d_hi - d_lo)
                        f1 = np.max(base + d1 * lin + d1 * d1 * quad)
                    else:
                        d_lo = d1
                        d1 = d2
                        f1 = f2
                        d2 = d_lo + gr * (d_hi - d_lo)
                        f2 = np.max(base + d2 * lin + d2 * d2 * quad)
                
                d_opt = (d_lo + d_hi) / 2.0
                f_opt = np.max(base + d_opt * lin + d_opt * d_opt * quad)
                
                if f_opt < best_obj - 1e-15:
                    a[i_r] += d_opt
                    a[j_r] -= d_opt
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s < 1e-15:
                        break
                    a = a / s
                    obj = fast_convolve(a).max()
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()
    
    return best_a, best_obj


def active_set_qp(a0, time_limit, max_active=50):
    """
    Active set QP: solve min a^T W a s.t. sum(a)=1, a>=0
    where W is a weighted sum of Hankel matrices based on current peaks.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit - 0.5:
        it += 1
        conv = fast_convolve(a)
        
        # Identify top peaks
        n_peaks = min(10, 2 * n - 1)
        top_peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
        
        # Build weight vector: weight peaks by their values
        w = np.zeros(2 * n - 1)
        w[top_peaks] = conv[top_peaks]
        w = w / w.sum()
        
        # Active set: indices with a[i] > threshold
        threshold = 0.005 / n
        active = set(np.where(a > threshold)[0])
        if len(active) == 0:
            active = set(np.argsort(a)[-max_active:])
        active = set(list(active)[:max_active])
        
        if len(active) < 2:
            break
        
        act = np.array(sorted(active))
        k = len(act)
        
        # Build W_AA: W_AA[i,j] = sum_k w[k] * 1[act[i]+act[j]==k]
        W_AA = np.zeros((k, k))
        for ii in range(k):
            for jj in range(ii, k):
                kk = act[ii] + act[jj]
                if kk < 2 * n - 1:
                    W_AA[ii, jj] = w[kk]
                    W_AA[jj, ii] = w[kk]
        
        # Solve KKT: [2W_AA  1] [x]   [0]
        #            [1^T    0] [mu] = [1]
        KKT = np.zeros((k + 1, k + 1))
        KKT[:k, :k] = 2.0 * W_AA + 1e-12 * np.eye(k)
        KKT[:k, k] = 1.0
        KKT[k, :k] = 1.0
        rhs = np.zeros(k + 1)
        rhs[k] = 1.0
        
        try:
            sol = np.linalg.solve(KKT, rhs)
            x = sol[:k]
            mu = sol[k]
        except np.linalg.LinAlgError:
            try:
                sol = np.linalg.lstsq(KKT, rhs, rcond=None)[0]
                x = sol[:k]
            except Exception:
                continue
        
        # Check non-negativity
        if np.min(x) < -1e-10:
            # Remove most negative from active set
            min_idx = np.argmin(x)
            active.discard(act[min_idx])
            # Scale up remaining
            x_pos = np.maximum(x, 0)
            if x_pos.sum() > 1e-15:
                x_pos = x_pos / x_pos.sum()
                a_new = np.zeros(n)
                for ii, idx in enumerate(act):
                    if x[ii] >= 0:
                        a_new[idx] = x[ii]
                s = a_new.sum()
                if s > 1e-15:
                    a_new = a_new / s
                    obj = fast_convolve(a_new).max()
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a_new.copy()
                        a = a_new
            continue
        
        # Form solution
        a_new = np.zeros(n)
        x = np.maximum(x, 0)
        x = x / x.sum()
        for ii, idx in enumerate(act):
            a_new[idx] = x[ii]
        
        # Check KKT for inactive: gradient[i] - mu >= 0
        grad = np.zeros(n)
        for ii, idx in enumerate(act):
            for jj, idx2 in enumerate(act):
                kk = idx + idx2
                if kk < 2 * n - 1:
                    grad[idx] += 2.0 * w[kk] * a_new[idx2]
        
        min_viol = float('inf')
        min_j = -1
        for j in range(n):
            if j not in active:
                viol = grad[j] - mu
                if viol < min_viol:
                    min_viol = viol
                    min_j = j
        
        if min_viol < -1e-8:
            active.add(min_j)
            a = a_new
            continue
        
        # Converged KKT
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            # No improvement, perturb and continue
            break
    
    return best_a, best_obj


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        return rng.dirichlet(np.ones(n) * 0.3)
    elif mode == 2:
        # Two bumps
        g1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
        g2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
        a = g1 + g2
        return a / a.sum()
    elif mode == 3:
        # Cosine window
        c = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
        c = np.maximum(c, 0.01)
        return c / c.sum()
    elif mode == 4:
        # Edge heavy
        e = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
        return e / e.sum()
    elif mode == 5:
        # Linear ramp
        a = np.linspace(0.1, 1.0, n)
        return a / a.sum()
    elif mode == 6:
        # Anti-symmetric-ish (high on left, low on right)
        a = np.exp(-x / (n * 0.3))
        return a / a.sum()
    else:
        return rng.dirichlet(np.ones(n) * 0.1)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid search for auto-convolution minimization.
    Uses smooth gradient with annealing, coordinate descent, and QP sub-solves.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        try:
            score = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform")
    
    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, best={best_score:.10f}")
    
    # === Phase 1: Multi-start n-exploration (250s) ===
    print("\n=== Phase 1: Multi-start exploration ===")
    t1 = time.monotonic()
    
    # Try various n values
    n_candidates = sorted(set([
        n_parent,
        max(64, n_parent - 32),
        n_parent + 32,
        max(64, n_parent - 64),
        n_parent + 64,
        511, 512, 767, 768, 769, 1023, 1024, 255, 256,
    ]))
    # Filter to reasonable range
    n_candidates = [n for n in n_candidates if 64 <= n <= 2048]
    
    for n_try in n_candidates:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 20:
            break
        
        # Create initializations
        if n_try == n_parent and parent is not None:
            inits = [parent.copy()]
            inits.append(diverse_init(n_try, rng, 1))
            inits.append(diverse_init(n_try, rng, 2))
        else:
            if parent is not None:
                interp = np.interp(np.linspace(0, 1, n_try),
                                   np.linspace(0, 1, n_parent), parent)
                interp = np.maximum(interp, 0)
                if interp.sum() > 1e-10:
                    interp = interp / interp.sum()
                    inits = [interp]
                else:
                    inits = [diverse_init(n_try, rng, 0)]
            else:
                inits = [diverse_init(n_try, rng, 0)]
            inits.append(diverse_init(n_try, rng, 2))
            inits.append(diverse_init(n_try, rng, 4))
        
        t_per_init = min(15.0, remaining / (len(inits) + 1))
        
        for ii, init in enumerate(inits):
            remaining = deadline - time.monotonic() - 1150
            if remaining < 5:
                break
            t_alloc = min(t_per_init, remaining)
            a_opt, _ = annealed_smooth_optimize(init, time_limit=t_alloc,
                                                 T_start=0.01, T_end=0.001,
                                                 lr_start=0.02, use_reversal=True)
            try_seq(a_opt, f"exp_n{n_try}_i{ii}")
        
        # Break if we've found something much better
        if best_score < 1.45:
            break
    
    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")
    
    # === Phase 2: Deep annealed optimization (900s) ===
    print("\n=== Phase 2: Deep annealed optimization ===")
    t2 = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Multiple annealing schedules
        configs = [
            {'T_start': 0.01, 'T_end': 0.0001, 'lr': 0.02, 'rev': True},
            {'T_start': 0.005, 'T_end': 0.00005, 'lr': 0.015, 'rev': True},
            {'T_start': 0.02, 'T_end': 0.0002, 'lr': 0.025, 'rev': False},
            {'T_start': 0.003, 'T_end': 0.00001, 'lr': 0.01, 'rev': True},
            {'T_start': 0.01, 'T_end': 0.00005, 'lr': 0.03, 'rev': True},
        ]
        
        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 300
            if remaining < 30:
                break
            t_alloc = min(remaining * 0.2, 180)
            print(f"  Anneal cfg{ci} (T:{cfg['T_start']}->{cfg['T_end']}), t={t_alloc:.0f}s")
            
            # Start from current best with tiny perturbation for diversity
            if ci > 0:
                start = project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)
            else:
                start = best_seq.copy()
            
            a_opt, _ = annealed_smooth_optimize(start, time_limit=t_alloc,
                                                 T_start=cfg['T_start'],
                                                 T_end=cfg['T_end'],
                                                 lr_start=cfg['lr'],
                                                 use_reversal=cfg['rev'])
            try_seq(a_opt, f"anneal_{ci}")
    
    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")
    
    # === Phase 3: Coordinate descent (400s) ===
    print("\n=== Phase 3: Coordinate descent ===")
    t3 = time.monotonic()
    
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 150
        if remaining > 30:
            t_alloc = min(remaining, 200)
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_opt, _ = pairwise_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_desc")
        
        # Another round from perturbed start
        remaining = deadline - time.monotonic() - 100
        if remaining > 30:
            start = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.001)
            t_alloc = min(remaining, 100)
            print(f"  Coord descent (perturbed), t={t_alloc:.0f}s")
            a_opt, _ = pairwise_coordinate_descent(start, time_limit=t_alloc)
            try_seq(a_opt, "coord_desc_pert")
    
    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")
    
    # === Phase 4: Active set QP (200s) ===
    print("\n=== Phase 4: Active set QP ===")
    t4 = time.monotonic()
    
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 80
        if remaining > 20:
            t_alloc = min(remaining, 120)
            print(f"  Active set QP, t={t_alloc:.0f}s")
            a_opt, _ = active_set_qp(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "active_qp")
    
    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")
    
    # === Phase 5: Final fine polish (80s) ===
    print("\n=== Phase 5: Final polish ===")
    t5 = time.monotonic()
    
    if best_seq is not None:
        # Very fine annealing
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            t_alloc = min(remaining, 30)
            a_opt, _ = annealed_smooth_optimize(best_seq, time_limit=t_alloc,
                                                 T_start=0.0001, T_end=0.00001,
                                                 lr_start=0.003, use_reversal=True)
            try_seq(a_opt, "fine_anneal")
        
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            t_alloc = min(remaining, 20)
            a_opt, _ = pairwise_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "fine_coord")
        
        # Tiny perturbations
        for _ in range(3):
            remaining = deadline - time.monotonic() - 3
            if remaining < 5:
                break
            pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.0003)
            a_opt, _ = annealed_smooth_optimize(pert, time_limit=min(remaining, 8),
                                                 T_start=0.0001, T_end=0.00001,
                                                 lr_start=0.002, use_reversal=True)
            try_seq(a_opt, f"perturb_{_}")
    
    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
    except Exception:
        pass
    
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
