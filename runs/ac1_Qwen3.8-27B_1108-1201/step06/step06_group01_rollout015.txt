def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]

def solve_lp_oracle(f, M, active_lags=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k in active_lags, g >= 0."""
    n = len(f)
    if active_lags is None:
        active_lags = np.arange(2 * n - 1)
    
    active_lags = np.asarray(active_lags)
    num_constraints = len(active_lags)
    
    c = -np.ones(n)  # minimize -sum(g)
    
    # Build constraint matrix
    A_ub = np.zeros((num_constraints, n))
    for i, k in enumerate(active_lags):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A_ub[i, js:je+1] = f[js:je+1]
    
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success:
            g = np.maximum(res.x, 0.0)
            # Verify all constraints
            full_conv = cross_convolve(f, g)
            if full_conv.max() <= M + 1e-10:
                return g, res  # res contains dual info
            # Constraint violation: add violated constraints and re-solve
            violated = np.where(full_conv > M + 1e-10)[0]
            all_lags = np.concatenate([active_lags, violated])
            all_lags = np.unique(all_lags)
            # Re-solve with more constraints
            return solve_lp_oracle(f, M, active_lags=all_lags)
    except Exception:
        pass
    return None, None

    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
