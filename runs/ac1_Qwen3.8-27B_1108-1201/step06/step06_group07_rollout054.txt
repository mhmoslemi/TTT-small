"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Enhanced SCA with trust region and adaptive M
- Soft peak objective (sum of exponentials) with projected gradient
- Direct coordinate descent on the auto-convolution peaks

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.special import softmax
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    # Clip small negative values from FFT
    result = np.maximum(result, 0)
    return result


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]

def fast_convolve(a):
    """Linear auto-convolution via FFT, clipped to non-negative."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    return np.maximum(result, 0.0)


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_oracle(f, M, n, row_idx, col_idx):

def soft_peak_objective(a, beta):
    """Compute F(a) = mean(exp(beta * b)) and its gradient."""
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    w = np.exp(beta * (b - b_max))
    F = w.mean() * np.exp(beta * b_max)
    # Gradient: dF/da[i] = 2*beta * corr[i] where corr[i] = sum_j w[j+i]*a[j]

    corr = np.fft.irfft(W * A.conjugate(), N)[:size]

    return F, grad, b

    # So b[k] = 2*a[i]*a[k-i] + c_k for k != 2i, and b[2i] = a[i]^2 + c_{2i}

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        b = fast_convolve(a)
        w = np.exp(beta * (b - b.max()))

        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 1.0:
                break

            # Compute the 1D function f(a_i) = sum_k w[k] * exp(beta * (2*a_i*a[k-i] - 2*a[i]*a[k-i]))
            # Actually, when we change a[i] from a[i] to a_i, the change in b[k] is:
            # For k where 0 <= k-i < n: delta_b[k] = 2*(a_i - a[i])*a[k-i] (if i != k-i)
            #                                              = (a_i^2 - a[i]^2) (if i == k-i, i.e., k = 2i)
            # So: f(a_i) = sum_k w[k] * exp(beta * delta_b[k])
            #            = sum_{k: 0<=k-i<n, k!=2i} w[k] * exp(2*beta*(a_i - a[i])*a[k-i])
            #            + w[2i] * exp(beta*(a_i^2 - a[i]^2))  (if 2i < 2n-1)

            # The function is convex in a_i >= 0 (sum of exponentials of linear/quadratic)
            # Find minimum using golden section or Newton

            ai_old = a[i]
            # Effective range: a_i in [0, 2*ai_old] should capture the minimum
            lo = 0.0
            hi = max(2.0 * ai_old, 1.0 / n)

            # Quick check: evaluate at a few points
            # For efficiency, only do full 1D optimization every few iterations
            if it % 3 == 0 or no_improve > 20:
                # Bisection on the derivative
                f_lo = _coord_obj(lo, i, ai_old, a, w, beta, n)
                f_hi = _coord_obj(hi, i, ai_old, a, w, beta, n)
                a_i_best = ai_old
                f_best = f_lo  # we'll also check ai_old
                f_mid = _coord_obj(ai_old, i, ai_old, a, w, beta, n)
                if f_mid < f_best:
                    f_best = f_mid
                    a_i_best = ai_old

                # Golden section search
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                x1 = hi - gr * (hi - lo)
                x2 = lo + gr * (hi - lo)
                fx1 = _coord_obj(x1, i, ai_old, a, w, beta, n)
                fx2 = _coord_obj(x2, i, ai_old, a, w, beta, n)
                for _ in range(30):
                    if fx1 < fx2:
                        hi = x2
                        x2 = x1
                        fx2 = fx1
                        x1 = hi - gr * (hi - lo)
                        fx1 = _coord_obj(x1, i, ai_old, a, w, beta, n)
                    else:
                        lo = x1
                        x1 = x2
                        fx1 = fx2
                        x2 = lo + gr * (hi - lo)
                        fx2 = _coord_obj(x2, i, ai_old, a, w, beta, n)
                a_i_opt = (lo + hi) / 2.0
                f_opt = _coord_obj(a_i_opt, i, ai_old, a, w, beta, n)
                if f_opt < f_best - 1e-12:
                    a_i_best = a_i_opt
                    f_best = f_opt

            if abs(a_i_best - ai_old) > 1e-10:
                a[i] = max(a_i_best, 0.0)
                improved = True

        if improved:
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 5:
            beta *= 1.1
            no_improve = 0

        if no_improve > 50:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0
            beta = 8.0

        if it % 50 == 0:
            print(f"    CoordDesc it={it}, obj={best_obj:.10f}")

    return best_a, best_obj


def _coord_obj(a_i, i, ai_old, a, w, beta, n):
    """Evaluate soft objective contribution for coordinate i at value a_i."""
    # b[k] changes by:
    # For k where 0 <= k-i < n and k-i != i (i.e., k != 2i):
    #   delta_b[k] = 2 * (a_i - ai_old) * a[k-i]
    # For k = 2i (if 2i < 2n-1):
    #   delta_b[k] = a_i^2 - ai_old^2
    # Objective: sum_k w[k] * exp(beta * delta_b[k])
    # We only need to consider k where a[k-i] > 0 (or k = 2i)
    result = 0.0
    for k in range(2 * n - 1):
        ki = k - i
        if 0 <= ki < n and a[ki] > 1e-15:
            if ki == i:
                delta = a_i * a_i - ai_old * ai_old
            else:
                delta = 2.0 * (a_i - ai_old) * a[ki]
            result += w[k] * np.exp(beta * delta)
        elif ki == i and 0 <= ki < n:
            delta = a_i * a_i - ai_old * ai_old
            result += w[k] * np.exp(beta * delta)
        else:
            result += w[k]
    return result


def peak_redistribute(a0, time_limit):
    """
    Directly target the maximum peak: move mass from positions contributing
    to the peak to positions that don't, using a quadratic penalty approach.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    frac = 0.05

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        b = fast_convolve(a)
        k_peak = int(np.argmax(b))
        peak_val = b[k_peak]

        # Contributions to the peak from each position
        # b[k_peak] = sum_j a[j] * a[k_peak - j]
        contrib = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        for j in range(js, je + 1):
            ki = k_peak - j
            if 0 <= ki < n:
                if j == ki:
                    contrib[j] = a[j] * a[j]
                else:
                    contrib[j] = a[j] * a[ki]
                    contrib[ki] = contrib[ki] + a[ki] * a[j] * 0  # avoid double counting
        # Actually let's just use: contrib[j] = a[j] * a[k_peak - j] for valid k_peak - j
        contrib = np.zeros(n)
        for j in range(n):
            ki = k_peak - j
            if 0 <= ki < n:
                contrib[j] = a[j] * a[ki]

        total_c = contrib.sum()
        if total_c < 1e-15:
            break

        # Also check second peak
        b_sorted = np.sort(b)[::-1]
        k_peak2 = int(np.argsort(b)[::-1][1]) if len(b) > 1 else k_peak

        # Find "safe" positions to move mass to: positions with low contribution to top peaks
        safety = np.zeros(n)
        for j in range(n):
            ki1 = k_peak - j
            ki2 = k_peak2 - j
            c1 = a[j] * a[ki1] if 0 <= ki1 < n else 0
            c2 = a[j] * a[ki2] if 0 <= ki2 < n else 0
            safety[j] = a[j] - 0.5 * (c1 + c2) / max(peak_val, 1e-15)

        # Move mass from high-contrib to high-safety positions
        donors = np.where(contrib > np.median(contrib[contrib > 0] + 1e-15))[0]
        donors = donors[contrib[donors] > 1e-10]
        recipients = np.where(safety > np.median(safety))[0]
        recipients = recipients[a[recipients] > 1e-10]

        if len(donors) == 0 or len(recipients) == 0:
            no_improve += 1
            frac = max(0.01, frac * 0.9)
            continue

        move_total = frac * a[donors].sum()
        if move_total < 1e-12:
            no_improve += 1
            continue

        # Remove from donors proportionally to contribution
        donor_total_c = contrib[donors].sum()
        if donor_total_c < 1e-15:
            no_improve += 1
            continue
        remove = a[donors] * (contrib[donors] / donor_total_c) * (move_total / a[donors].sum())
        remove = np.minimum(remove, a[donors] * 0.5)

        # Add to recipients proportionally to safety
        rec_total_s = safety[recipients].max()  # just distribute evenly or by current value
        add = a[recipients] * (safety[recipients] / safety[recipients].sum())

        for j in donors:
            a[j] -= remove[j - donors[0] if isinstance(j, np.integer) else list(donors).index(j)] if j in donors else 0
        # Simpler approach:
        a_new = a.copy()
        donor_c = contrib[donors]
        total_donor_c = donor_c.sum()
        for idx, j in enumerate(donors):
            a_new[j] -= frac * move_total * (donor_c[idx] / total_donor_c) / max(a[j], 1e-15) * 0.5
        a_new[donors] = np.maximum(a_new[donors], 0)

        total_add = -a_new[donors].sum() + a[donors].sum()
        if total_add > 0:
            rec_weight = np.maximum(safety[recipients], 0) + 1e-6
            rec_weight = rec_weight / rec_weight.sum()
            for idx, j in enumerate(recipients):
                a_new[j] += total_add * rec_weight[idx]

        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()

        new_obj = fast_convolve(a_new).max()
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            frac = min(0.1, frac * 1.05)
        else:
            no_improve += 1
            frac = max(0.01, frac * 0.9)

        if no_improve > 40:
            rng = np.random.default_rng((it * 9973 + 5) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            no_improve = 0
            frac = 0.05

        if it % 100 == 0:
            print(f"    PeakRed it={it}, obj={best_obj:.10f}, frac={frac:.4f}")

    return best_a, best_obj


def sca_optimize(f0, time_limit, M_init=None):
    """Enhanced SCA with adaptive M and better blending."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    row_idx, col_idx = build_sparsity(n)

    conv = fast_convolve(f)
    M = conv.max() if M_init is None else M_init
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        g_raw = solve_lp_oracle(f, M, n, row_idx, col_idx)

        if g_raw is None:
            M *= 1.02
            no_improve += 1
            if no_improve > 30:
                conv_now = fast_convolve(f)
                M = conv_now.max() * 1.005
                no_improve = 0
            continue

        g = g_raw / g_raw.sum()

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 20:
                rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                M = fast_convolve(f).max() * 1.005
                no_improve = 0
            continue

        # Multi-alpha blend with evaluation
        best_val = fast_convolve(f).max()
        best_ft = f.copy()
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_ft = ft.copy()

        if best_val < best_obj - 1e-14:
            f = best_ft
            best_obj = best_val
            best_f = f.copy()
            M = best_val * 0.999
            no_improve = 0
        else:
            no_improve += 1
            M *= 0.998

        if no_improve > 40:
            rng = np.random.default_rng((it * 31831 + 7) % (2**31 - 1))
            scale = 0.003 + 0.005 * min(1.0, (no_improve - 40) / 80.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max() * 1.005
            no_improve = 0

        if it % 200 == 0:
            print(f"    SCA it={it}, obj={best_obj:.10f}, M={M:.8f}")

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Asymmetric ramp
    ramp = (x + 1.0) / (x + 1.0).sum()
    inits.append(ramp)

    # Two-sided asymmetric
    t = x / max(n - 1, 1)
    asym = np.exp(-2.0 * (t - 0.3)) + 0.3 * np.exp(-2.0 * (t - 0.8))
    inits.append(asym / asym.sum())

    # Oscillating
    osc = (1.0 + 0.3 * np.sin(4 * np.pi * t)) ** 2 + 0.05
    inits.append(osc / osc.sum())

    # Concentrated center-left
    g1 = np.exp(-0.5 * ((x - 0.3 * (n - 1)) / (0.1 * n)) ** 2)
    g2 = 0.3 * np.exp(-0.5 * ((x - 0.7 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append((g1 + g2) / (g1 + g2).sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: multi-phase hybrid combining SCA, soft-peak gradient,
    coordinate descent, and peak redistribution with adaptive n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def _try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        try:
            score = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            _try_seq(parent, "parent")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)

    # === Phase 1: Quick SCA across n values (250s) ===
    print("\n=== Phase 1: Multi-n SCA ===")
    t1 = time.monotonic()

    ns_to_try = sorted(set([n_parent, 512, 640, 768, 896, 1024, 1280, 1536, 2048]))
    ns_to_try = [n for n in ns_to_try if 256 <= n <= 2560]

    for n_try in ns_to_try:
        remaining_after = deadline - time.monotonic()
        if remaining_after < 250:
            break
        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss < 1e-10:
                start = np.ones(n_try) / n_try
            else:
                start = start / ss

        t_alloc = min(25, remaining_after - 200)
        if t_alloc < 5:
            break
        print(f"  SCA n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = sca_optimize(start, time_limit=t_alloc)
        _try_seq(a_opt, f"sca_n{n_try}")

    print(f"  P1 done: best={best_score:.8f}, t={time.monotonic()-t1:.0f}s")

    # === Phase 2: Deep SCA at best n (350s) ===
    print("\n=== Phase 2: Deep SCA ===")
    t2 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 100:
            t_alloc = min(remaining - 300, 250)
            if t_alloc > 20:
                print(f"  Deep SCA n={len(best_seq)}, t={t_alloc:.0f}s")
                a_opt, _ = sca_optimize(best_seq, time_limit=t_alloc)
                _try_seq(a_opt, "deep_sca")

                remaining = deadline - time.monotonic()
                if remaining > 100:
                    t_alloc2 = min(remaining - 200, 100)
                    if t_alloc2 > 10:
                        a_opt2, _ = sca_optimize(a_opt, time_limit=t_alloc2)
                        _try_seq(a_opt2, "deep_sca2")

    print(f"  P2 done: best={best_score:.8f}, t={time.monotonic()-t2:.0f}s")

    # === Phase 3: Soft-peak gradient descent (350s) ===
    print("\n=== Phase 3: Soft-Peak Gradient ===")
    t3 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 80:
            t_alloc = min(remaining - 200, 200)
            if t_alloc > 15:
                print(f"  SoftPeak t={t_alloc:.0f}s")
                a_opt, _ = soft_peak_descent(best_seq, time_limit=t_alloc, beta=5.0)
                _try_seq(a_opt, "soft_peak")

                remaining = deadline - time.monotonic()
                if remaining > 80:
                    t_alloc2 = min(remaining - 150, 100)
                    if t_alloc2 > 10:
                        a_opt2, _ = soft_peak_descent(a_opt, time_limit=t_alloc2, beta=10.0)
                        _try_seq(a_opt2, "soft_peak2")

    print(f"  P3 done: best={best_score:.8f}, t={time.monotonic()-t3:.0f}s")

    # === Phase 4: Coordinate Descent (250s) ===
    print("\n=== Phase 4: Coordinate Descent ===")
    t4 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t_alloc = min(remaining - 150, 150)
            if t_alloc > 15:
                print(f"  CoordDesc t={t_alloc:.0f}s")
                a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
                _try_seq(a_opt, "coord_desc")

                remaining = deadline - time.monotonic()
                if remaining > 60:
                    t_alloc2 = min(remaining - 100, 80)
                    if t_alloc2 > 10:
                        a_opt2, _ = coordinate_descent(a_opt, time_limit=t_alloc2)
                        _try_seq(a_opt2, "coord_desc2")

    print(f"  P4 done: best={best_score:.8f}, t={time.monotonic()-t4:.0f}s")

    # === Phase 5: Peak Redistribution (200s) ===
    print("\n=== Phase 5: Peak Redistribution ===")
    t5 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 50:
            t_alloc = min(remaining - 80, 120)
            if t_alloc > 10:
                print(f"  PeakRed t={t_alloc:.0f}s")
                a_opt, _ = peak_redistribute(best_seq, time_limit=t_alloc)
                _try_seq(a_opt, "peak_redist")

                remaining = deadline - time.monotonic()
                if remaining > 50:
                    t_alloc2 = min(remaining - 50, 60)
                    if t_alloc2 > 10:
                        a_opt2, _ = peak_redistribute(a_opt, time_limit=t_alloc2)
                        _try_seq(a_opt2, "peak_redist2")

    print(f"  P5 done: best={best_score:.8f}, t={time.monotonic()-t5:.0f}s")

    # === Phase 6: Diversified restarts (150s) ===
    print("\n=== Phase 6: Diversified Restarts ===")
    t6 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        scales = [0.003, 0.008, 0.02, 0.05]
        for scale in scales:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(20, remaining - 40)
            if t_alloc < 5:
                break
            print(f"  Restart scale={scale}, t={t_alloc:.0f}s")
            a_opt, _ = sca_optimize(f_pert, time_limit=t_alloc)
            _try_seq(a_opt, f"restart_{scale}")

    print(f"  P6 done: best={best_score:.8f}, t={time.monotonic()-t6:.0f}s")

    # === Phase 7: Final polish (remaining) ===
    print("\n=== Phase 7: Final Polish ===")
    t7 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_alloc = min(remaining - 10, 30)
            if t_alloc > 5:
                a_opt, _ = soft_peak_descent(best_seq, time_limit=t_alloc, beta=15.0)
                _try_seq(a_opt, "final_soft")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_opt, _ = peak_redistribute(best_seq, time_limit=min(remaining, 15))
            _try_seq(a_opt, "final_peak")

        remaining = deadline - time.monotonic()
        if remaining > 8:
            a_opt, _ = sca_optimize(best_seq, time_limit=min(remaining, 10))
            _try_seq(a_opt, "final_sca")

    print(f"  P7 done: best={best_score:.8f}, t={time.monotonic()-t7:.0f}s")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final evaluation
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
