import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(n):
    """Build sparse constraint matrix for LP oracle. Returns (row_idx, col_idx)."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows), np.array(cols)


def solve_lp_oracle(f, M, n, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0. Returns normalized g or None."""
    c = -np.ones(n)
    # Build values: for row k, col i, value is f[k-i]
    vals = np.empty(len(row_idx))
    for idx in range(len(row_idx)):
        k = row_idx[idx]
        i = col_idx[idx]
        vals[idx] = f[k - i]

    A_mat = lil_matrix((2 * n - 1, n))
    A_mat[row_idx, col_idx] = vals
    A_mat = A_mat.tocsr()

    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.fun is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None

def compute_hankel_grad(a, p):
    """Compute gradient of sum_k p[k]*(a*a)[k] with respect to a.
    grad[j] = 2 * sum_m a[m] * p[j+m]
    """
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = 2.0 * np.dot(a, p[j:j + n])
    return grad

def local_search(a0, time_limit, seed=42):

            candidate = project_simplex(a + pert)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:

                best_a = a.copy()
                improved = True
                break

                    candidate[j] += amount

                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break

        # Structured perturbation: add a sinusoidal pattern
        if not improved and it % 10 == 0:
            freq = rng.uniform(1, 8)
            phase = rng.uniform(0, 2 * np.pi)
            pattern = np.sin(2 * np.pi * freq * np.arange(n) / n + phase) * 0.002

            if c.max() < best_obj - 1e-15:

                best_a = a.copy()

def try_seq(a, label, best_ref):
    """Evaluate a sequence and update best if improved. Returns (best_seq, best_score)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_ref
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_ref[1]:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return (a_norm.copy(), score)
    return best_ref
