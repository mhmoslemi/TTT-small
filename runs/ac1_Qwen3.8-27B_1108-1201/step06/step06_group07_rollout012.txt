import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.special import softmax
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    mu = np.sort(v)[::-1]
    cumsum_mu = np.cumsum(mu)
    rho_candidates = np.arange(1, n + 1)
    cond = mu * rho_candidates > (cumsum_mu - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = cond.nonzero()[0][-1]
    theta = (cumsum_mu[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k=20):
    """Compute weighted subgradient at top-k peaks."""
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        pv = conv[peaks]
        weights = pv / pv.sum()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        weights = pv / pv.sum()

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        w = weights[idx]
        if w < 1e-14:
            continue
        k = int(peaks[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # d(conv[k])/d(a[j]) = 2*a[k-j] for j in [js, je]
        # a[k-je : k-js+1] reversed gives a[k-j] for j=js..je
        grad[js:je + 1] += w * 2.0 * a[k - je:k - js + 1][::-1]
    return grad

    """
    Adaptive subgradient with exponential averaging for the outer average.
    Handles non-negativity via projection onto simplex.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    # Averaging machinery
    avg_a = a.copy()
    avg_count = 1.0
    avg_decay = 0.9995  # Exponential moving average

    t0 = time.monotonic()
    it = 0
    lr = 0.1
    top_k = top_k_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        current_max = conv.max()

        # Update best
        if current_max < best_obj - 1e-15:
            best_obj = current_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive top_k: increase when stuck
        if no_improve > 100:
            top_k = min(50, top_k + 5)
        elif no_improve < 20:
            top_k = max(5, top_k - 1)

        # Compute subgradient
        grad = compute_subgradient(a, conv, top_k)

        # Adaptive step size: polynomial decay with restarts
        lr = 0.5 / (np.sqrt(it + 1) + 1.0)
        if no_improve > 200:
            lr = 0.5 / (np.sqrt(it + 1) + 1.0) * 2.0

        # Update
        a_new = a - lr * grad
        a_new = project_simplex(a_new)

        # Line search: try a few step sizes
        best_candidate = a_new
        best_val = current_max
        for alpha in [0.5, 1.0, 2.0]:
            cand = project_simplex(a - alpha * lr * grad)
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_candidate = cand

        a = best_candidate

        # Update exponential average
        avg_a = avg_decay * avg_a + (1 - avg_decay) * a
        avg_count = avg_decay * avg_count + 1.0
        avg_a = avg_a / avg_a.sum()

        # Check averaged solution
        avg_obj = fast_convolve(avg_a).max()
        if avg_obj < best_obj - 1e-15:

            print(f"    SG it={it}, obj={best_obj:.10f}, t={elapsed:.1f}s, top_k={top_k}")

def peak_coordinate_refine(a0, time_limit, verbose=False):
    """
    Coordinate descent: for each position j, find the best value
    that reduces the max convolution peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            conv = fast_convolve(a)
            current_max = conv.max()

            # Try reducing a[j] and redistributing
            old_val = a[j]
            if old_val < 1e-10:
                continue

            # Try several reduction factors
            best_trial_val = current_max
            best_trial_frac = 0.0  # fraction to keep

            for frac in [0.5, 0.3, 0.7, 0.1, 0.9]:
                trial = a.copy()
                reduction = old_val * (1.0 - frac)
                if reduction < 1e-12:
                    continue
                trial[j] = old_val * frac
                # Redistribute to positions far from the peak
                k_peak = int(np.argmax(conv))
                # Positions that least contribute to this peak
                js = max(0, k_peak - n + 1)
                je = min(n - 1, k_peak)
                # Add to positions outside [js, je] preferentially
                outside = [i for i in range(n) if i < js or i > je]
                if not outside:
                    outside = [i for i in range(n) if i != j]
                if not outside:
                    continue
                per_add = reduction / len(outside)
                for i in outside:
                    trial[i] += per_add
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:

                trial = trial / ts
                val = fast_convolve(trial).max()
                if val < best_trial_val:
                    best_trial_val = val
                    best_trial_frac = frac
                    best_trial = trial.copy()

            if best_trial_frac > 0 and best_trial_val < current_max - 1e-12:
                a = best_trial
                if best_trial_val < best_obj - 1e-14:

                improved = True

        if verbose and sweep % 5 == 0:
            print(f"    CoordRefine sweep={sweep}, obj={best_obj:.10f}")

    return best_a, best_obj


def parametric_init(n, mode=0, rng=None):
    """Generate parametric initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    x = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0

    if mode == 0:
        # Uniform
        a = np.ones(n)
    elif mode == 1:
        # U-shaped (more mass at edges)
        a = 1.0 / (np.abs(x - half) / half + 0.1)
    elif mode == 2:
        # Bimodal
        a = np.exp(-((x - 0.15 * (n-1)) / (0.1 * n))**2) + np.exp(-((x - 0.85 * (n-1)) / (0.1 * n))**2) + 0.01
    elif mode == 3:
        # Cosine window
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
        a = np.maximum(a, 0.01)
    elif mode == 4:
        # Sinc-like
        t = (x - half) / (n / 4)
        a = np.sinc(t * 0.5) ** 2 + 0.05
        a = np.maximum(a, 0.01)
    elif mode == 5:
        # Random Dirichlet
        alpha = 0.3 + rng.random() * 2.0
        a = rng.dirichlet(np.ones(n) * alpha)
    elif mode == 6:
        # Two-scale: large-scale U + small-scale noise
        a = 1.0 / (np.abs(x - half) / half + 0.3)
        a += rng.standard_normal(n) * 0.05
        a = np.maximum(a, 0)
    elif mode == 7:
        # Periodic modulation of uniform
        freq = rng.integers(1, 10)
        amp = rng.random() * 0.3
        a = 1.0 + amp * np.cos(2 * np.pi * freq * x / n)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n) * rng.random(n) + 0.01

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        a_new = np.ones(new_n) / new_n
    else:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: multi-phase hybrid combining SCA (reduced LP), subgradient,
    coordinate refinement, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return float('inf')
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = parametric_init(768, mode=1, rng=rng)

    n_parent = len(parent)

    # === Phase 1: Quick SCA on parent n (120s) ===
    print("\n=== Phase 1: Quick SCA on parent ===")
    t1 = time.monotonic()
    remaining = deadline - time.monotonic()
    t_alloc = min(120, remaining - 600)
    if t_alloc > 30:
        print(f"  SCA n={n_parent}, t={t_alloc:.0f}s, top_k=100")
        a_opt, _ = sca_reduced(parent, time_limit=t_alloc, top_k=100)
        try_seq(a_opt, f"sca_n{n_parent}")
    print(f"  P1 done: best={best_score:.8f}, t={time.monotonic()-t1:.0f}s")

    # === Phase 2: Explore larger n with subgradient (300s) ===
    print("\n=== Phase 2: Multi-n subgradient exploration ===")
    t2 = time.monotonic()

    # Try several n values with different initializations
    n_candidates = []
    for n_try in [1024, 1536, 2048, 3072, 4096]:
        remaining = deadline - time.monotonic()
        if remaining < 200:
            break
        n_candidates.append(n_try)

    best_n_for_deep = n_parent
    best_n_score = best_score

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 150:
            break
        t_alloc = min(60, remaining - 150)
        if t_alloc < 20:
            break

        # Initialize from parent (resampled) + perturbation
        a_init = resample_sequence(parent, n_try)
        a_init = project_simplex(a_init + rng.standard_normal(n_try) * 0.02)

        print(f"  Subgrad n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = subgradient_optimize(a_init, time_limit=t_alloc, top_k_init=15)
        score = try_seq(a_opt, f"sg_n{n_try}")
        if score < best_n_score:
            best_n_score = score
            best_n_for_deep = n_try

    # Also try parametric inits at promising n
    for n_try in [best_n_for_deep, 2048]:
        if n_try == best_n_for_deep:
            continue
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break
        t_alloc = min(40, remaining - 100)
        if t_alloc < 15:
            break
        for mode in [1, 3, 6]:
            a_init = parametric_init(n_try, mode=mode, rng=rng)
            print(f"  Subgrad n={n_try} mode={mode}, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize(a_init, time_limit=t_alloc, top_k_init=15)
            try_seq(a_opt, f"sg_p{n_try}_m{mode}")

    print(f"  P2 done: best={best_score:.8f}, t={time.monotonic()-t2:.0f}s")

    # === Phase 3: Deep SCA on best n (400s) ===
    print("\n=== Phase 3: Deep SCA ===")
    t3 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()
        t_alloc = min(250, remaining - 300)
        if t_alloc > 30:
            print(f"  Deep SCA n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = sca_reduced(best_seq, time_limit=t_alloc, top_k=120)
            try_seq(a_opt, "deep_sca")

            # Second round
            remaining = deadline - time.monotonic()
            t_alloc2 = min(120, remaining - 200)
            if t_alloc2 > 20:
                a_opt2, _ = sca_reduced(a_opt, time_limit=t_alloc2, top_k=100)
                try_seq(a_opt2, "deep_sca2")

    print(f"  P3 done: best={best_score:.8f}, t={time.monotonic()-t3:.0f}s")

    # === Phase 4: Subgradient polish on best (200s) ===
    print("\n=== Phase 4: Subgradient polish ===")
    t4 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_alloc = min(150, remaining - 150)
        if t_alloc > 20:
            print(f"  SG polish n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc, top_k_init=30)
            try_seq(a_opt, "sg_polish")

            # Try with different top_k
            remaining = deadline - time.monotonic()
            t_alloc2 = min(50, remaining - 100)
            if t_alloc2 > 15:
                a_opt2, _ = subgradient_optimize(a_opt, time_limit=t_alloc2, top_k_init=5)
                try_seq(a_opt2, "sg_polish_5")

    print(f"  P4 done: best={best_score:.8f}, t={time.monotonic()-t4:.0f}s")

    # === Phase 5: Coordinate refinement (150s) ===
    print("\n=== Phase 5: Coordinate refinement ===")
    t5 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_alloc = min(100, remaining - 80)
        if t_alloc > 15:
            print(f"  CoordRefine n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, _ = peak_coordinate_refine(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_refine")

            # Another round of SCA after coord refine
            remaining = deadline - time.monotonic()
            t_alloc2 = min(40, remaining - 50)
            if t_alloc2 > 10:
                a_opt2, _ = sca_reduced(a_opt, time_limit=t_alloc2, top_k=80)
                try_seq(a_opt2, "post_coord_sca")

    print(f"  P5 done: best={best_score:.8f}, t={time.monotonic()-t5:.0f}s")

    # === Phase 6: Diversified restarts (150s) ===
    print("\n=== Phase 6: Diversified restarts ===")
    t6 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        restart_configs = [
            (0.003, "sca"),
            (0.008, "sca"),
            (0.015, "sg"),
            (0.005, "sca"),
            (0.02, "sg"),
        ]
        for scale, method in restart_configs:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            t_alloc = min(30, remaining - 30)
            if t_alloc < 10:
                break

            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            if method == "sca":
                a_opt, _ = sca_reduced(f_pert, time_limit=t_alloc, top_k=80)
            else:
                a_opt, _ = subgradient_optimize(f_pert, time_limit=t_alloc, top_k_init=10)
            try_seq(a_opt, f"restart_{scale}_{method}")

    print(f"  P6 done: best={best_score:.8f}, t={time.monotonic()-t6:.0f}s")

    # === Phase 7: Final combined polish (remaining) ===
    print("\n=== Phase 7: Final polish ===")
    t7 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        # Run SCA + SG alternating
        for round_num in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            t_sca = min(20, remaining // 2 - 5)
            t_sg = min(15, remaining - t_sca - 5)
            if t_sca < 5:
                break

            a_opt, _ = sca_reduced(best_seq, time_limit=t_sca, top_k=60)
            try_seq(a_opt, f"final_sca_{round_num}")

            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            t_sg2 = min(15, remaining - 5)
            if t_sg2 > 5:
                a_opt2, _ = subgradient_optimize(a_opt, time_limit=t_sg2, top_k_init=10)
                try_seq(a_opt2, f"final_sg_{round_num}")

    print(f"  P7 done: best={best_score:.8f}, t={time.monotonic()-t7:.0f}s")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
