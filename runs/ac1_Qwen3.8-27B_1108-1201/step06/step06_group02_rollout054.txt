def solve_qp_simplex(H, x0, mu, max_iter=100):
    """Solve min (x-x0)^T M (x-x0) subject to x >= 0, sum(x) = 1
    where M = H + mu*I. Use projected gradient with backtracking."""
    n = len(x0)
    M = H + mu * np.eye(n)
    b = M @ x0  # gradient at x0 is M(x-x0), so we minimize (x-x0)^T M (x-x0)
    # grad = 2*M*(x - x0)
    
    x = x0.copy()
    for it in range(max_iter):
        grad = 2 * M @ (x - x0)
        # Projected gradient step with backtracking
        lr = 1.0 / (2 * np.linalg.norm(H, ord=2) + mu)  # Lipschitz constant bound
        # Actually, use backtracking
        f_current = (x - x0) @ M @ (x - x0)
        for _ in range(50):
            x_new = project_simplex(x - lr * grad)
            f_new = (x_new - x0) @ M @ (x_new - x0)
            if f_new <= f_current - 0.5 * lr * grad @ grad:
                break
            lr *= 0.5
        if np.linalg.norm(x_new - x) < 1e-10:
            break
        x = x_new
    return x
```

def qp_step(x0, p, mu, max_iter=200):
    """One QP step: min (x-x0)^T M (x-x0) s.t. x in simplex, M = H(p) + mu*I"""
    n = len(x0)
    x = x0.copy()
    lr = 1.0 / (2.0 * (1.0 + mu))  # Conservative step size
    f_prev = 0.0
    
    for it in range(max_iter):
        # Compute gradient: 2*(H(p) + mu*I) * (x - x0)
        # H(p) @ (x - x0) = hankel_mv(p, x - x0)
        d = x - x0
        Hd = hankel_mv(p, d)  # This computes (H(p)) @ d
        grad = 2.0 * (Hd + mu * d)
        
        f_current = d @ (Hd + mu * d)  # This is d^T M d
        
        # Backtracking line search
        accepted = False
        for _ in range(30):
            x_new = project_simplex(x - lr * grad)
            d_new = x_new - x0
            Hd_new = hankel_mv(p, d_new)
            f_new = d_new @ (Hd_new + mu * d_new)
            if f_new <= f_current - 0.5 * lr * np.dot(grad, grad):
                accepted = True
                break
            lr *= 0.5
        
        if not accepted:
            break
        
        if np.linalg.norm(x_new - x) < 1e-12:
            x = x_new
            break
        x = x_new
        f_prev = f_new
        lr = min(lr * 1.5, 1.0 / (2.0 * (1.0 + mu)))  # Increase lr cautiously
    
    return x
```

import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_mv(p, x):

QP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: uses quadratic subproblems (exploiting curvature of the
auto-convolution objective) with adaptive softmax-weighted Hankel matrices,
combined with coordinate descent polish and diverse structured restarts.

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, x):
    """Compute H(p) @ x where H(p)_{j,m} = p[j+m]. Uses correlation."""
    return np.correlate(p, x, mode='valid')


def qp_subproblem(x0, p, mu, max_iter=200, tol=1e-14):

    Solve min (x-x0)^T (H(p) + mu*I) (x-x0) subject to x >= 0, sum(x) = 1.
    Uses projected gradient with backtracking line search.
    """

    x = x0.copy()
    max_lr = 0.8 / (1.0 + mu)
    lr = max_lr * 0.5

    for it in range(max_iter):
        d = x - x0
        Hd = hankel_mv(p, d)
        grad = 2.0 * (Hd + mu * d)

        g_sq = grad @ grad
        if g_sq < tol:
            break

        f_val = d @ (Hd + mu * d)

        accepted = False
        for _ in range(35):
            x_trial = project_simplex(x - lr * grad)

                break
            lr *= 0.5

        if not accepted:
            break

        step = np.linalg.norm(x_trial - x)
        x = x_trial

        if step < tol:
            break

        lr = min(lr * 1.6, max_lr)

    return x

def coordinate_descent_polish(a0, time_limit):
    """
    Pairwise coordinate swaps to reduce max convolution.
    Tries moving mass from high-contribution coordinates to low ones.

    n = len(a)

    it = 0

        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Find the coordinate that most contributes to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]

        # Find coordinate to move mass to
        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            outside = np.arange(n)
        j_min = outside[np.argmin(a[outside])]

            if a[j_max] < delta:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min] += delta
            candidate = np.maximum(candidate, 0.0)
            candidate = candidate / candidate.sum()
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate

                best_a = a.copy()

            stale = 0
        else:
            stale += 1
            if stale % 50 == 0:

def fourier_refine(a0, time_limit, n_fourier=15):
    """Optimize over Fourier coefficients for structured improvement."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    M = min(n_fourier, n // 2)
    x = np.arange(n, dtype=np.float64)

    # Extract initial Fourier coefficients
    a_centered = a - 1.0 / n
    init_coeffs = np.zeros(2 * M)
    for m in range(1, M + 1):
        init_coeffs[2 * (m - 1)] = 2.0 * np.sum(a_centered * np.cos(2.0 * np.pi * m * x / n))
        init_coeffs[2 * (m - 1) + 1] = 2.0 * np.sum(a_centered * np.sin(2.0 * np.pi * m * x / n))

    def fourier_to_seq(coeffs):
        base = np.ones(n) / n
        for m in range(1, M + 1):
            c = coeffs[2 * (m - 1)]
            d = coeffs[2 * (m - 1) + 1]
            base += (c * np.cos(2.0 * np.pi * m * x / n) +
                     d * np.sin(2.0 * np.pi * m * x / n)) / n
        base = np.maximum(base, 0.0)
        s = base.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return base / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    def fourier_objective(coeffs):
        seq = fourier_to_seq(coeffs)
        conv = fast_convolve(seq)
        return conv.max()

    t0 = time.monotonic()
    n_restarts = max(3, int((time_limit // 20)) + 1)

    for r in range(n_restarts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        if r == 0:
            x0 = init_coeffs.copy()
        else:
            rng = np.random.default_rng(r * 99991 + 7)
            scale = 0.003 * (1 + r * 0.5)
            x0 = init_coeffs + rng.standard_normal(2 * M) * scale

        try:
            res = minimize(fourier_objective, x0, method='Nelder-Mead',
                           options={'maxiter': 5000, 'xatol': 1e-11, 'fatol': 1e-13,
                                    'maxfev': 30000})
            if res.fun < best_obj:
                best_obj = res.fun
                best_a = fourier_to_seq(res.x)
                print("    Fourier restart {}: obj={:.10f}".format(r, best_obj))
        except Exception:
            pass

    return best_a, best_obj


def subgradient_momentum(a0, time_limit, top_k=20, lr_init=0.003, mom_init=0.93):
    """Multi-peak subgradient descent with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = mom_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)

        # Weighted subgradient from top peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]

        v = mom * v - lr * sg
        a = project_simplex(a + v)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        mom = min(0.97, mom + 0.00005)

        if no_improve > 100:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.001 + 0.003 * min(1.0, (no_improve - 100) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init * 0.5
            no_improve = 0

    return best_a, best_obj


def try_and_update(a, state, label):
    """Evaluate a sequence and update state if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a)))


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase QP-based search:
    Phase 1: Quick multi-n exploration with QP
    Phase 2: Deep QP optimization with multi-scale temperatures
    Phase 3: Coordinate descent + subgradient polish
    Phase 4: Fourier refinement + final QP pass
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    state = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.8f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    n_parent = len(parent)

    # === Phase 1: Quick Multi-n Exploration (~200s) ===
    print("\n=== Phase 1: Quick Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent, 512, 640, 768, 1024, 1280]))
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 2048]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1300
        if remaining < 10:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss > 1e-10:
                start = start / ss
            else:
                start = np.ones(n_try) / n_try

        t_alloc = min(remaining, 35)
        print("  QP n={}, t={:.0f}s".format(n_try, t_alloc))
        a_opt, _ = qp_optimize(start, time_limit=t_alloc, T_init=0.3, mu_init=0.5)
        try_and_update(a_opt, state, "qp_n{}".format(n_try))

    print("  Phase 1 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 2: Deep Multi-Scale QP (~800s) ===
    print("\n=== Phase 2: Deep Multi-Scale QP ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])
        remaining = deadline - time.monotonic() - 600
        if remaining > 30:
            t_alloc = min(remaining, 600)
            print("  Multi-scale QP n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = multi_scale_qp(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "multiscale_qp")

        # Direct QP with warm start
        remaining = deadline - time.monotonic() - 300
        if remaining > 20:
            t_alloc = min(remaining, 200)
            print("  Deep QP (warm) n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = qp_optimize(state[1], time_limit=t_alloc,
                                   T_init=0.1, mu_init=0.3,
                                   T_decay=0.9998, mu_decay=0.9995)
            try_and_update(a_opt, state, "deep_qp_warm")

    print("  Phase 2 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 3: Coordinate Descent + Subgradient (~300s) ===
    print("\n=== Phase 3: Coordinate Descent + Subgradient ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        remaining = deadline - time.monotonic() - 250
        if remaining > 10:
            t_alloc = min(remaining, 100)
            print("  Coordinate descent, t={:.0f}s".format(t_alloc))
            a_opt, _ = coordinate_descent_polish(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "coord_desc")

        # Multiple subgradient configs
        sg_configs = [
            (10, 0.002, 0.92),
            (20, 0.003, 0.94),
            (50, 0.001, 0.95),
            (100, 0.001, 0.96),
        ]
        for top_k, lr, mom in sg_configs:
            remaining = deadline - time.monotonic() - 150
            if remaining < 10:
                break
            t_alloc = min(remaining, 50)
            print("  Subgradient k={}, t={:.0f}s".format(top_k, t_alloc))
            a_opt, _ = subgradient_momentum(state[1], time_limit=t_alloc,
                                            top_k=top_k, lr_init=lr, mom_init=mom)
            try_and_update(a_opt, state, "sg_k{}".format(top_k))

    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 4: Fourier + Final Polish (~300s) ===
    print("\n=== Phase 4: Fourier + Final Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        remaining = deadline - time.monotonic() - 80
        if remaining > 15:
            t_alloc = min(remaining, 120)
            print("  Fourier refine, t={:.0f}s".format(t_alloc))
            a_opt, _ = fourier_refine(state[1], time_limit=t_alloc, n_fourier=15)
            try_and_update(a_opt, state, "fourier_15")

        remaining = deadline - time.monotonic() - 40
        if remaining > 10:
            t_alloc = min(remaining, 60)
            print("  Final QP, t={:.0f}s".format(t_alloc))
            a_opt, _ = qp_optimize(state[1], time_limit=t_alloc,
                                   T_init=0.05, mu_init=0.2)
            try_and_update(a_opt, state, "final_qp")

        # Final coordinate descent polish
        remaining = deadline - time.monotonic() - 20
        if remaining > 5:
            t_alloc = min(remaining, 40)
            print("  Final coord descend, t={:.0f}s".format(t_alloc))
            a_opt, _ = coordinate_descent_polish(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "final_coord")

        # Tiny perturbations
        remaining = deadline - time.monotonic() - 5
        if remaining > 3:
            for i in range(8):
                remaining = deadline - time.monotonic() - 2
                if remaining < 2:
                    break
                noise = rng.standard_normal(len(state[1])) * 0.0003
                f_pert = project_simplex(state[1] + noise)
                a_opt, _ = qp_optimize(f_pert, time_limit=min(remaining, 3),
                                       T_init=0.03, mu_init=0.1)
                try_and_update(a_opt, state, "perturb_{}".format(i))

    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = parent.copy()
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
