The objective `f(a) = 2n * max_k conv[k]` (with `sum(a)=1`) is convex in `a`? Let me check.

`conv[k] = a^T H_k a` where `H_k` is a Hankel matrix. Is `H_k` positive semidefinite?

`H_k[i,j] = 1 if i+j = k, else 0`... no wait.

`conv[k] = sum_i a[i] * a[k-i]`. This can be written as `a^T M_k a` where `(M_k)_{i,j} = 1/2 * (delta_{i+j,k} + delta_{i+j,k})`... hmm.

Actually, `conv[k] = sum_i a[i] a[k-i]`. Let me write this as `a^T B_k a` where `B_k` is a matrix. We have:
`a^T B_k a = sum_{i,j} a[i] B_k[i,j] a[j] = sum_i a[i] a[k-i]`

This means `B_k[i,j] = 1` if `i+j = k` (or more precisely, `j = k-i`), and 0 otherwise. So `B_k` is a "reversed diagonal" matrix (anti-diagonal for k=0, next anti-diagonal for k=1, etc.).

Is `B_k` positive semidefinite? For k=0: `B_0 = I` (identity), which is positive definite. For k=1: `B_1` has 1s on the first super-diagonal and sub-diagonal. The eigenvalues of this matrix are `2*cos(m*pi/(n+1))` for m=1,...,n. Some of these are negative! So `B_k` is NOT positive semidefinite in general.

This means `conv[k]` is NOT a convex function of `a`, and the overall objective is NOT convex. This explains why the problem is hard.

However, if we restrict to symmetric sequences (`a[i] = a[n-1-i]`), things might be different. But I won't rely on this.

1. Log-sum-exp smoothed L-BFGS optimization with continuation
2. Active-set linearized QP refinement

Uses FFT-based convolution and efficient gradient computation via cross-correlation.

import numpy as np

import time

def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_obj_grad(x, T, n):
    """
    Compute the smoothed objective and gradient.
    x: log-parameters (length n)
    T: temperature for log-sum-exp
    Returns (objective, gradient)
    """
    a = softmax(x)
    conv = fast_convolve(a)
    
    # Log-sum-exp of conv/T
    conv_max = conv.max()
    exp_terms = np.exp((conv - conv_max) / T)
    lse = conv_max + T * np.log(exp_terms.sum())
    obj = 2.0 * n * lse
    
    # Softmax weights
    p = exp_terms / exp_terms.sum()
    
    # Gradient of lse w.r.t. a: g[i] = sum_k p[k] * d(conv[k])/da[i]
    # g[i] = 2 * (a * p)[i] - p[2i] * a[i]  (cross-correlation)
    c = np.correlate(a, p, mode='valid')  # c[i] = sum_m a[m] * p[m+i], length n
    g = 2.0 * c - p[2 * np.arange(n)] * a
    
    # Convert to gradient w.r.t. x: grad_x[i] = a[i] * (g[i] - sum_j a[j]*g[j])
    mean_g = np.dot(a, g)
    grad_x = 2.0 * n * a * (g - mean_g)
    
    return obj, grad_x

def run_lbfgs(x0, T, n, max_iter=200, gtol=1e-8):
    """Run L-BFGS on the smoothed objective."""
    a0 = softmax(x0)
    conv0 = fast_convolve(a0)
    
    def fun(x):
        return smooth_obj_grad(x, T, n)[0]
    
    def jac(x):
        return smooth_obj_grad(x, T, n)[1]
    
    result = minimize(fun, x0, jac=jac, method='L-BFGS-B',
                      options={'maxiter': max_iter, 'gtol': gtol})
    return result.x, result.fun

def continuation_optimize(x0, time_limit, T_factor_start=10.0, T_factor_end=200.0, decay=0.85):
    """
    Continuation method: start with large T (smooth) and decrease.
    """
    a0 = softmax(x0)
    conv0 = fast_convolve(a0)
    conv_max = conv0.max()
    
    T = conv_max / T_factor_start
    T_end = conv_max / T_factor_end
    
    x = x0.copy()
    t_start = time.monotonic()
    
    while time.monotonic() - t_start < time_limit:
        # Run L-BFGS for this T
        remaining = time_limit - (time.monotonic() - t_start)
        if remaining < 1.0:
            break
        
        max_iter = min(200, max(20, int(remaining * 5)))
        x, obj = run_lbfgs(x, T, len(x), max_iter=max_iter)
        
        # Check convergence
        a = softmax(x)
        conv = fast_convolve(a)
        actual_max = conv.max()
        
        # Decrease T
        if T > T_end * 1.01:
            T *= decay
        else:
            break
        
        elapsed = time.monotonic() - t_start
        if elapsed > time_limit * 0.8:
            break
    
    return x

def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent) + 1)[:-1]
    x_new = np.linspace(0, 1, n_target + 1)[:-1]
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed L-BFGS with continuation (decreasing temperature)
2. Active-set linearized LP refinement exploiting KKT structure

4. Symmetry exploitation and perturbation restarts
Uses FFT-based convolution and O(n^2) gradient via cross-correlation.

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """FFT-based auto-convolution of non-negative sequence a."""
    n = len(a)
    size = 2 * n - 1
    if size <= 1:
        return a * a
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_obj_grad(x, T, n):
    """
    Compute the log-sum-exp smoothed objective and gradient.
    x: log-parameters (length n), a = softmax(x)
    T: temperature
    Returns (objective_value, gradient_vector)
    """
    a = softmax(x)
    conv = fast_convolve(a)

    # Numerically stable log-sum-exp of conv/T
    conv_max = conv.max()
    scaled = (conv - conv_max) / T
    exp_sum = np.exp(scaled).sum()
    if exp_sum < 1e-300:
        lse = conv_max
    else:
        lse = conv_max + T * np.log(exp_sum)
    obj = 2.0 * n * lse

    # Softmax weights p[k] = exp(conv[k]/T) / sum
    if exp_sum > 1e-300:
        p = np.exp(scaled) / exp_sum
    else:
        p = np.zeros_like(conv)
        p[conv_max == conv] = 1.0 / (conv == conv_max).sum()

    # Gradient of lse w.r.t. a: g[i] = sum_k p[k] * d(conv[k])/da[i]
    # d(conv[k])/da[i] = 2*a[k-i] (bulk) with correction at k=2i
    # g[i] = 2 * correlate(a, p, 'valid')[i] - p[2i] * a[i]
    c = np.correlate(a, p, mode='valid')
    g = 2.0 * c - p[2 * np.arange(n)] * a

    # Convert to gradient w.r.t. x (softmax parameterization)
    mean_g = np.dot(a, g)
    grad_x = 2.0 * n * a * (g - mean_g)

    return obj, grad_x
