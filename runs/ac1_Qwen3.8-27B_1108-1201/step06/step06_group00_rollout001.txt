    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_weights(conv, beta):
    """Compute LSE softmax weights on the convolution peaks."""
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_bc / total


def compute_subgradient(f, active_peaks, weights):
    """Compute weighted sum of gradients of (f*f)[k] for k in active_peaks."""
    n = len(f)
    grad = np.zeros(n)
    for idx, k in enumerate(active_peaks):
        k = int(k)
        w = weights[idx]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += w * 2.0 * f[k - i_range]
    return grad


def subgradient_momentum(f0, time_limit, beta_start=5.0, beta_end=8000.0, 
                          top_k=40, momentum=0.9, verbose=False):
    """Subgradient method with momentum, adaptive beta, and Nesterov averaging."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    v = np.zeros(n)  # momentum
    avg_f = f.copy()  # running average
    lr_base = 2.0 / np.sqrt(n)
    lr = lr_base
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    total_iters = 0
    avg_weight_sum = 0.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        total_iters += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        conv = fast_convolve(f)
        cur_obj = conv.max()
        
        # Update best and average
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Nesterov-style averaging
        w = 1.0 / (it + 1)
        avg_weight_sum += w
        avg_f = avg_f + w * (f - avg_f) / (it * w / avg_weight_sum)
        # Actually, let's use proper Polyak averaging
        # avg_f = (sum of first k iterates) / k
        # This is O(n) per step which is fine

        # Full LSE weights
        w_full = lse_weights(conv, beta)
        
        k_top = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        w_peaks = w_full[peak_idx]
        w_sum = w_peaks.sum()
        if w_sum > 1e-15:
            w_peaks = w_peaks / w_sum
        else:
            w_peaks = np.ones(k_top) / k_top

        grad = compute_subgradient(f, peak_idx, w_peaks)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_normed = grad / gnorm

        # Momentum update
        v = momentum * v - lr * grad_normed
        f_new = project_simplex(f + v)

        # Check improvement
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            lr = min(lr * 1.01, 5.0 / np.sqrt(n))
        else:
            # Backtracking
            f_new = project_simplex(f - lr * grad_normed)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new < best_obj:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                no_improve = 0
            else:
                no_improve += 1
                if no_improve >= 8:
                    lr *= 0.85
                if no_improve > 50:
                    f = best_f.copy()
                    v = np.zeros(n)
                    lr = max(lr_base, 0.01)
                    no_improve = 0
                    # Small perturbation
                    rng = np.random.default_rng(it * 104729 + 7)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.002)

        # Adaptive beta
        if no_improve < 20:
            beta = min(beta * 1.005, beta_end)
        else:
            beta = min(beta * 1.02, beta_end)

    return best_f, best_obj


def swap_local_search(f0, time_limit, n_swaps=5000, verbose=False):
    """Local search using swap operations: move mass from one position to another."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    # Get top peaks for responsibility computation
    n_peaks = min(10, len(conv))
    top_peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
    
    iterations = 0
    while time.monotonic() - t0 < time_limit and iterations < n_swaps:
        iterations += 1
        
        # Pick a random source position weighted by "responsibility" to top peaks
        # Responsibility: for peak k, position i contributes f[i]*f[k-i]
        # We want to find positions that contribute a lot to the top peaks
        
        # Sample a peak
        k = int(top_peaks[rng.integers(n_peaks)])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo >= hi:
            continue
        
        # Compute contributions to this peak
        contributions = f[lo:hi+1] * f[k - np.arange(lo, hi+1)]
        
        # Pick source weighted by contribution
        source_idx = lo + rng.choice(hi - lo + 1, p=contributions / contributions.sum())
        source_idx = int(source_idx)
        
        # Pick destination: a position that contributes little to any peak
        # Simple approach: pick a random position
        dest_idx = int(rng.integers(n))
        if dest_idx == source_idx:
            continue
        
        # Try different fractions of mass to move
        f_src = f[source_idx]
        if f_src < 1e-10:
            continue
            
        for fraction in [0.05, 0.1, 0.2, 0.3, 0.5]:
            delta = f_src * fraction
            f_trial = f.copy()
            f_trial[source_idx] -= delta
            f_trial[dest_idx] += delta
            if f_trial[source_idx] < -1e-15:
                continue
            f_trial = np.maximum(f_trial, 0)
            f_trial = f_trial / f_trial.sum()
            
            conv_trial = fast_convolve(f_trial)
            obj_trial = conv_trial.max()
            
            if obj_trial < best_obj - 1e-15:
                best_obj = obj_trial
                best_f = f_trial.copy()
                f = f_trial.copy()
                # Update top peaks
                top_peaks = np.argpartition(conv_trial, -n_peaks)[-n_peaks:]
                break
            elif obj_trial < best_obj:
                # Accept but don't update best (for exploration)
                pass
    
    return best_f, best_obj


def freq_perturb(f, rng, strength=0.03):
    """Perturb f in the frequency domain."""
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    # Zero out DC to preserve sum (approximately)
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.10f} (n={len(a)})", flush=True)
    return score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    return f_new


def peak_shaving(f, n_peaks=5, n_iterations=100, time_limit=30.0):
    """Aggressively reduce the top peaks by redistributing mass."""
    f = np.maximum(f, 0.0)
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    for _ in range(n_iterations):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Find top peaks
        peak_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
        peak_vals = sorted([conv[k] for k in peak_idx], reverse=True)
        
        if peak_vals[0] - peak_vals[-1] < 1e-8:
            break  # All top peaks are similar
        
        # Focus on the largest peak
        k_max = peak_idx[np.argmax([conv[k] for k in peak_idx])]
        k_max = int(k_max)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        
        if lo >= hi:
            continue
        
        # Find the pair (i, j) with i+j=k_max that contributes most
        i_range = np.arange(lo, hi + 1)
        contributions = f[i_range] * f[k_max - i_range]
        i_max = int(i_range[np.argmax(contributions)])
        j_max = k_max - i_max
        
        if i_max == j_max:
            # Single element at center: try to reduce it
            # Move its mass to a random position far from center
            f_src = f[i_max]
            if f_src < 1e-10:
                continue
            dest = int(rng.integers(n))
            if dest == i_max:
                continue
            for frac in [0.02, 0.05, 0.1]:
                f_trial = f.copy()
                f_trial[i_max] -= f_src * frac
                f_trial[dest] += f_src * frac
                f_trial = np.maximum(f_trial, 0)
                f_trial /= f_trial.sum()
                conv_trial = fast_convolve(f_trial)
                if conv_trial.max() < best_obj:
                    best_obj = conv_trial.max()
                    best_f = f_trial.copy()
                    f = f_trial.copy()
                    conv = conv_trial
                    break
        else:
            # Two elements: reduce the larger one
            if f[i_max] >= f[j_max]:
                src = i_max
                other = j_max
            else:
                src = j_max
                other = i_max
            
            f_src = f[src]
            if f_src < 1e-10:
                continue
            
            # Try moving mass to a position that minimizes increase in other peaks
            # Simple: try a few random destinations
            tried = False
            for _ in range(20):
                dest = int(rng.integers(n))
                if dest == src or dest == other:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    f_trial = f.copy()
                    f_trial[src] -= f_src * frac
                    f_trial[dest] += f_src * frac
                    if f_trial[src] < 0:
                        continue
                    f_trial = np.maximum(f_trial, 0)
                    f_trial /= f_trial.sum()
                    conv_trial = fast_convolve(f_trial)
                    if conv_trial.max() < best_obj:
                        best_obj = conv_trial.max()
                        best_f = f_trial.copy()
                        f = f_trial.copy()
                        conv = conv_trial
                        tried = True
                        break
                if tried:
                    break
    
    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main optimization:
    Phase 1: Subgradient with momentum (40% of time)
    Phase 2: Peak shaving (20% of time)
    Phase 3: Local search with swaps (20% of time)
    Phase 4: Multi-restart + refinement (15% of time)
    Phase 5: Final polish (5% of time)
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.10f}", flush=True)
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768.0
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    total_budget = float(budget_s) - 20.0
    print(f"Starting optimization with n={n_parent}, budget={total_budget:.0f}s", flush=True)

    # === Phase 1: Subgradient with momentum ===
    t1_end = time.monotonic() + total_budget * 0.40
    print(f"\n=== Phase 1: Subgradient (t={total_budget*0.40:.0f}s) ===", flush=True)
    
    t_alloc = t1_end - time.monotonic()
    if t_alloc > 30:
        f_opt, _ = subgradient_momentum(parent, time_limit=t_alloc,
                                          beta_start=5.0, beta_end=10000.0,
                                          top_k=40, momentum=0.9)
        try_sequence(f_opt, best_tracker)

    # Also try with different parameters
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_remaining = t1_end - time.monotonic()
    if t_remaining > 20:
        f_opt, _ = subgradient_momentum(current_best, time_limit=t_remaining,
                                          beta_start=50.0, beta_end=50000.0,
                                          top_k=20, momentum=0.95)
        try_sequence(f_opt, best_tracker)

    # === Phase 2: Peak shaving ===
    t2_end = t1_end + total_budget * 0.20
    print(f"\n=== Phase 2: Peak shaving (t={total_budget*0.20:.0f}s) ===", flush=True)
    
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(120, t2_end - time.monotonic())
    if t_alloc > 15:
        f_opt, _ = peak_shaving(current_best, n_peaks=3, n_iterations=500, time_limit=t_alloc)
        try_sequence(f_opt, best_tracker)
        
        # Run again on the result
        current_best = best_tracker[1] if best_tracker[1] is not None else current_best
        t_rem = t2_end - time.monotonic()
        if t_rem > 15:
            f_opt, _ = peak_shaving(current_best, n_peaks=5, n_iterations=500, time_limit=t_rem)
            try_sequence(f_opt, best_tracker)

    # === Phase 3: Local search with swaps ===
    t3_end = t2_end + total_budget * 0.20
    print(f"\n=== Phase 3: Local search (t={total_budget*0.20:.0f}s) ===", flush=True)
    
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = t3_end - time.monotonic()
    if t_alloc > 15:
        f_opt, _ = swap_local_search(current_best, time_limit=t_alloc, n_swaps=20000)
        try_sequence(f_opt, best_tracker)

    # === Phase 4: Multi-restart ===
    t4_end = t3_end + total_budget * 0.15
    print(f"\n=== Phase 4: Multi-restart (t={total_budget*0.15:.0f}s) ===", flush=True)
    
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_restarts_available = 5
    while time.monotonic() < t4_end and n_restarts_available > 0:
        n_restarts_available -= 1
        for attempt in range(3):
            t_rem = t4_end - time.monotonic() - 5
            if t_rem < 10:
                break
            strength = 0.005 + 0.005 * attempt
            f_pert = freq_perturb(current_best, rng, strength=strength)
            t_sub = (t_rem - 5) / 3.0
            if t_sub > 8:
                f_opt, _ = subgradient_momentum(f_pert, time_limit=t_sub,
                                                  beta_start=100.0, beta_end=20000.0,
                                                  top_k=20, momentum=0.9)
                try_sequence(f_opt, best_tracker)
                f_opt2, _ = peak_shaving(f_opt, n_peaks=3, n_iterations=200, 
                                           time_limit=min(10, t_rem - 5))
                try_sequence(f_opt2, best_tracker)
            n_restarts_available -= 1
        if n_restarts_available <= 0:
            break
        current_best = best_tracker[1] if best_tracker[1] is not None else current_best

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final polish ===", flush=True)
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    
    t_rem = deadline - time.monotonic()
    if t_rem > 10:
        f_opt, _ = peak_shaving(current_best, n_peaks=2, n_iterations=2000, time_limit=t_rem * 0.5)
        try_sequence(f_opt, best_tracker)
    
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t_rem = deadline - time.monotonic()
    if t_rem > 10:
        f_opt, _ = swap_local_search(current_best, time_limit=t_rem, n_swaps=50000)
        try_sequence(f_opt, best_tracker)

    # Final verification
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}", flush=True)
            return [float(x) for x in final_seq]

    print("WARNING: Using uniform fallback", flush=True)
    fallback = np.ones(7
