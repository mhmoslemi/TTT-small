    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, n):
    """Build the constraint matrix for the LP oracle.
    A_ub has shape (2n-1, n), A_ub[k, j] = f[j] if 0 <= k-j < n else 0.
    """
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        row_start = j
        row_end = min(2 * n - 1, j + n)
        A_ub[row_start:row_end, j] = f[:row_end - row_start]
    return A_ub


def solve_lp_oracle(f, n, M, A_ub=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    if A_ub is None:
        A_ub = build_lp_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=' highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='simplex')
            if res.success and res.x is not None:
                return np.maximum(res.x, 0.0)
        except Exception:
            pass
    return None


def golden_line_search(f, g, n, tol=1e-10):
    """Exact 1D line search: minimize max_k conv((1-t)f + t*g, (1-t)f + t*g)_k over t in [0,1].
    Uses golden section search on the convex function phi(t).
    """
    pre_factor = (1.0 + 5.0**0.5) / 2.0
    inv_factor = 1.0 / pre_factor

    lo = 0.0
    hi = 1.0

    def phi(t):
        ft = (1.0 - t) * f + t * g
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            return 1e10
        ft = ft / s
        return fast_convolve(ft).max()

    d = inv_factor
    c = 1.0 - inv_factor
    fc = phi(c)
    fd = phi(d)

    for _ in range(60):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + inv_factor * (hi - lo)
            fc = phi(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + inv_factor * (hi - lo)
            fd = phi(d)

    t_best = (lo + hi) / 2.0
    # Evaluate at a few extra points near the optimum
    candidates = [t_best, max(0, t_best - 0.01), min(1, t_best + 0.01), 0.0, 1.0]
    best_t = t_best
    best_val = phi(t_best)
    for tc in candidates:
        if 0 <= tc <= 1:
            vc = phi(tc)
            if vc < best_val:
                best_val = vc
                best_t = tc
    return best_t


def fw_lp_optimize(f0, time_limit, max_iters=None):
    """Frank-Wolfe with LP oracle, exact line search, and adaptive restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    A_ub = build_lp_matrix(f, n)

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Solve LP oracle
        g = solve_lp_oracle(f, n, M * m_exp, A_ub)
        if g is None:
            m_exp = min(m_exp * 1.2, 3.0)
            if m_exp >= 2.9:
                m_exp = 1.0
                A_ub = build_lp_matrix(f, n)
            continue

        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                _restart(f, best_f, n, it, no_improve)
                f, M, best_obj, best_f, m_exp, A_ub, no_improve = _do_restart(f, best_f, n, it, no_improve)
            continue

        # Golden section line search
        best_alpha = golden_line_search(f, g, n)

        f = (1.0 - best_alpha) * f + best_alpha * g
        f = np.maximum(f, 0.0)
        f = f / f.sum()

        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
            m_exp = 1.0
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.03, 3.0)

        if no_improve > 50:
            _do_restart(f, best_f, n, it, no_improve)
            f = project_simplex(best_f + np.random.default_rng(it * 31).standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
            A_ub = build_lp_matrix(f, n)

        if max_iters and it >= max_iters:
            break

    return best_f, best_obj


def _restart(f, best_f, n, it, no_improve):
    pass


def _do_restart(f, best_f, n, it, no_improve):
    scale = 0.001 + 0.005 * min(1.0, no_improve / 200.0)
    rng = np.random.default_rng(it * 7919 + 13)
    f = project_simplex(best_f + rng.standard_normal(n) * scale)
    conv = fast_convolve(f)
    M = conv.max()
    return f, M, conv.max(), best_f.copy(), 1.0, build_lp_matrix(f, n), 0


def smoothed_grad_descent(a0, time_limit, T_init=0.001, lr_init=0.03):
    """Smoothed gradient descent using softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    n = len(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Compute gradient efficiently
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            w = p[k]
            if w < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += w * 2.0 * a[k - idx]

        improved = False
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_obj - 1e-15:
                a = candidate
                best_obj = val
                best_a = a.copy()
                no_improve = 0
                improved = True
                break

        if not improved:
            no_improve += 1

        if no_improve > 15:
            T = min(T * 1.1, 1.0)
            lr *= 0.96
        if no_improve > 60:
            T *= 0.7
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise = 0.002 + 0.004 * min(1.0, (no_improve - 200) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit):
    """Coordinate descent: iteratively try to reduce the peak by moving mass."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stale = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Find the source (largest contributor to the peak)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        contributions = a[affected] * a[k_peak - affected]
        j_src = affected[np.argmax(contributions)]

        # Find the target (position that most benefits from receiving mass)
        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            outside = np.arange(n)

        # Try several targets and deltas
        best_candidate = None
        best_val = best_obj

        # Sort outside by a_i (prefer moving to where a is small)
        outside_sorted = outside[np.argsort(a[outside])[:min(20, len(outside))]]

        for j_dst in outside_sorted[:10]:
            for delta in [5e-5, 1e-4, 2e-4, 5e-4, 1e-3, 2e-3]:
                if a[j_src] < delta * 1.1:
                    continue
                candidate = a.copy()
                candidate[j_src] -= delta
                candidate[j_dst] += delta
                candidate = np.maximum(candidate, 0.0)
                s = candidate.sum()
                if s < 1e-10:
                    continue
                candidate = candidate / s
                obj = fast_convolve(candidate).max()
                if obj < best_val - 1e-15:
                    best_val = obj
                    best_candidate = candidate

        if best_candidate is not None:
            a = best_candidate
            best_obj = best_val
            best_a = a.copy()
            stale = 0
        else:
            stale += 1
            if stale > 100:
                # Random perturbation
                rng = np.random.default_rng(it * 4567 + 3)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
                obj = fast_convolve(candidate).max()
                if obj < best_obj - 1e-15:
                    a = candidate
                    best_obj = obj
                    best_a = a.copy()
                    stale = 0
                elif stale > 500:
                    break

    return best_a, best_obj


def multi_peak_subgrad(a0, time_limit, top_k=20, lr_init=0.005, mom=0.93):
    """Multi-peak subgradient with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        k = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k)[-k:]
        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(k)
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 2.0
        ws = weights.sum()
        weights /= ws

        sg = np.zeros(n)
        for i in range(k):
            if weights[i] < 1e-12:
                continue
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            sg[idx] += weights[i] * 2.0 * a[kk - idx]

        v = mom * v - lr * sg
        a = project_simplex(a + v)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        mom = min(0.97, mom + 0.00005)

        if no_improve > 150:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.001 + 0.003 * min(1.0, (no_improve - 150) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init * 0.7
            no_improve = 0

    return best_a, best_obj


def upsequence(src, n_target):
    """Upsample a sequence to a different length using interpolation."""
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    interp = np.interp(x_tgt, x_src, src)
    interp = np.maximum(interp, 0)
    s = interp.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return interp / s


def anti_palindromic_init(n, alpha=0.3):
    """Create a sequence with anti-palindromic structure to reduce center peak."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    # Symmetric base (slightly peaked at edges)
    sym = 1.0 + 0.2 * np.cos(2.0 * np.pi * x / n)
    # Anti-symmetric perturbation
    anti = alpha * np.sin(2.0 * np.pi * x / n)
    p = sym + anti
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return p / s


def structured_init(n, rng):
    """Generate a diverse structured initialization."""
    x = np.arange(n, dtype=np.float64)
    choice = rng.integers(0, 5)
    if choice == 0:
        # Edge-concentrated
        p = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    elif choice == 1:
        # Two bumps
        p = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    elif choice == 2:
        # Anti-palindromic
        p = anti_palindromic_init(n, alpha=rng.uniform(0.1, 0.5))
    elif choice == 3:
        # Cosine power
        beta = rng.uniform(0.5, 3.0)
        p = np.cos(np.pi * x / (n - 1)) ** beta
        p = np.maximum(p, 0)
    else:
        # Random smooth
        freq = rng.integers(2, 8)
        p = 1.0 + rng.uniform(0.2, 0.6) * np.cos(2.0 * np.pi * freq * x / n + rng.uniform(0, 2*np.pi))
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return p / s


def score_sequence(a):
    """Compute the evaluation score for a sequence (assumes sum=1, non-negative)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def try_and_update(a, state, label):
    """Evaluate and update best state."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a = a / s
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a)))


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid search:
    Phase 1: Multi-n exploration with diverse initializations + quick FW-LP
    Phase 2: Deep FW-LP with exact line search on best candidates
    Phase 3: Smoothed gradient + multi-peak subgradient refinement
    Phase 4: Coordinate descent polish + final FW-LP
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    state = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.8f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        state = [2.0, parent.copy()]

    n_parent = len(parent)
    print("Starting search. Best initial: {:.8f}".format(state[0]))

    # === Phase 1: Multi-n Exploration (~250s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent, 384, 480, 576, 640, 720, 768, 896, 1024, 1152]))
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 1280]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1100
        if remaining < 20:
            break

        # Generate multiple initializations
        inits = []
        # Interpolated parent
        if n_try != n_parent:
            inits.append(upsequence(parent, n_try))
        else:
            inits.append(parent.copy())
        # Structured
        for _ in range(2):
            inits.append(structured_init(n_try, rng))
        # Anti-palindromic
        inits.append(anti_palindromic_init(n_try, alpha=0.2))

        for ii, init in enumerate(inits):

            if sc < best_score_n:
                best_score_n = sc
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            try_and_update(best_for_n, state, "expl_n{}_best".format(n_try))

    print("  Phase 1 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 2: Deep FW-LP (~600s) ===
    print("\n=== Phase 2: Deep FW-LP ===")

        # Run deep FW-LP on the best

            print("  Deep FW-LP n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = fw_lp_optimize(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "deep_fw")

        # Try a few perturbed restarts
        for scale in [0.005, 0.01, 0.02, 0.03]:
            remaining = deadline - time.monotonic() - 400
            if remaining < 15:

            f_pert = project_simplex(state[1] + rng.standard_normal(n_best) * scale)

            try_and_update(a_opt, state, "coord")

        remaining = deadline - time.monotonic() - 60
        if remaining > 10:

            print("  Final FW-LP, t={:.0f}s".format(t_alloc))

        if remaining > 5:
            t_alloc = min(remaining, 30)
            a_opt, _ = smoothed_grad_descent(state[1], time_limit=t_alloc,
                                             T_init=0.0001, lr_init=0.005)
            try_and_update(a_opt, state, "fine_smooth")

        # Tiny perturbations
        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            for _ in range(3):
                remaining = deadline - time.monotonic() - 3
                if remaining < 3:
                    break
                noise = rng.standard_normal(len(state[1])) * 0.001
                f_pert = project_simplex(state[1] + noise)
                a_opt, _ = smoothed_grad_descent(f_pert, time_limit=min(remaining, 5),
                                                 T_init=0.0002, lr_init=0.003)
                try_and_update(a_opt, state, "perturb_fine")

    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(384) / 384
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
