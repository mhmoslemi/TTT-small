import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad_at_peak(f, k):
    """Gradient of (f*f)[k] wrt f."""
    n = len(f)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    i_range = np.arange(lo, hi + 1)
    grad[i_range] = 2.0 * f[k - i_range]
    return grad

def lse_subgradient_optimize(f0, time_limit, beta_start=50.0, beta_end=10000.0, top_k=40):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = 1.0 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        conv = fast_convolve(f)
        
        # LSE weights on top peaks
        k_top = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        
        # Compute LSE weights
        bc = beta * conv[peak_idx]
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for idx, k in enumerate(peak_idx):
            if w[idx] < 1e-15:
                continue
            grad += w[idx] * compute_grad_at_peak(f, int(k))
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Backtracking line search
        step = lr
        improved = False
        for _ in range(15):
            f_new = project_simplex(f - step * grad)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                improved = True
                no_improve = 0
                lr = min(lr * 1.01, 5.0 / np.sqrt(n))
                break
            step *= 0.5
        
        if not improved:
            no_improve += 1

    return best_f, best_obj

                cols.extend(js.tolist())

        if rows:
            A = csr_matrix((vals, (rows, cols)), shape=(len(peak_idx), n))

            try:

                M = conv.max()
                if M < best_obj:
                    best_obj = M
                    best_f = f_new.copy()
                f = f_new

        if diff < 1e-12:
            no_improve += 1
            if no_improve > 25:
                break
            continue
        
        # Line search (exact for quadratic)
        best_alpha = 0.0
        best_val = M
        # Coarse grid
        alphas = np.linspace(0.05, 1.0, 16)
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:

def interpolate_to_length(f, new_n):
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    return f_new

def freq_perturb(f, rng, strength=0.03):

    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new
