```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond_1 = u * np.arange(1, n + 1) - (cssv - 1.0) > 0
    if not cond_1.any():
        return np.ones(n) / n
    rho = np.nonzero(cond_1)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgrad(a, k_star):
    """Compute subgradient of max_k conv(a,a)[k] at active point k_star."""
    n = len(a)
    # By symmetry, we can assume k_star is in [0, n-1]
    k = min(k_star, 2*n-2-k_star)
    j = np.arange(n)
    return a[np.abs(k - j)]

def subgradient_method(a0, time_limit, lr_init=0.1, decay=0.9995):
    """Projected subgradient method for min max_k conv(a,a)[k] s.t. sum(a)=1, a>=0."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    lr = lr_init
    avg_grad = np.zeros(n)
    count = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        obj = conv.max()
        k_star = int(np.argmax(conv))
        
        grad = compute_subgrad(a, k_star)
        
        # Averaged subgradient
        count += 1
        avg_grad = (count * avg_grad + lr * grad) / (count + 1)
        
        # Two step sizes: current and averaged
        a1 = project_simplex(a - lr * grad)
        a2 = project_simplex(np.maximum(a0, 0.0) / a0.sum() - lr * avg_grad)  # Hmm, this doesn't quite work
        
        # Actually, for averaged subgradient, we track the iterate
        a1_obj = fast_convolve(a1).max()
        
        if a1_obj < best_obj:
            best_obj = a1_obj
            best_a = a1.copy()
        
        a = a1
```

```python
def compute_smooth_grad(a, p):
    """Compute gradient of smooth approximation of max conv.
    a: sequence (length n)
    p: softmax weights on conv values (length 2n-1)
    Returns: gradient (length n)
    """
    n = len(a)
    # grad[j] = 2 * sum_{m=0}^{n-1} a[m] * p[j+m] (where j+m < 2n-1)
    grad = np.zeros(n)
    for j in range(n):
        max_m = min(n, 2*n-1-j)
        grad[j] = 2.0 * np.dot(a[:max_m], p[j:j+max_m])
    return grad
```

Optimized step function search using smooth gradient descent with Hankel gradients,
active-set QP polishing, and diverse structural restarts.
Targets minimization of 2*n*max(convolve(a,a))/(sum(a))^2.
Key insight: gradient of conv(a,a)[k] w.r.t. a[j] = 2*a[k-j] if 0<=k-j<n, else 0.
This gives an efficient Hankel structure for the smoothed gradient.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond_1 = u * np.arange(1, n + 1) - (cssv - 1.0) > 0
    if not cond_1.any():
        return np.ones(n) / n
    rho = np.nonzero(cond_1)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_hankel_grad(a, p):
    """Compute gradient of sum_k p[k]*conv(a,a)[k] w.r.t. a.
    Uses the identity: grad[j] = 2 * sum_m a[m] * p[j+m]
    O(n^2) via numpy vectorization.
    """
    n = len(a)
    grad = np.zeros(n)
    # For each m, add a[m] * p[m:m+n] to grad (truncated)
    for m in range(n):
        am = a[m]
        if am > 0:
            end = min(n, 2 * n - 1 - m)
            grad[:end] += am * p[m:m + end]
    return 2.0 * grad


def solve_qp_active_set(a0, q_weights, max_iter=500):
    """Solve QP: min a^T H a s.t. sum(a)=1, a>=0
    where H is defined by q_weights (Hankel structure).
    Uses active-set method with linear system solves.
    Returns optimal a or None on failure.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Build the H matrix for the quadratic form sum_k q[k] * conv(a,a)[k]
    # H[j,i] = 2 * q[j+i] (from the gradient structure: (Ha)[j] = 2*sum_i q[j+i]*a[i])
    # Actually the quadratic form is a^T H a where (Ha)[j] = 2*sum_i q[j+i]*a[i]
    # So H[j,i] = 2*q[j+i] if j+i < 2n-1, else 0
    # This is a Hankel matrix.
    # For the QP min a^T H a, the Hessian is 2H (since d/d(a) a^T H a = 2Ha if H symmetric)
    # But H is not symmetric in general (Hankel). Let's use the symmetric part.
    # Actually: a^T H a = sum_j a[j] (Ha)[j] = sum_j a[j] * 2 * sum_i q[j+i] a[i]
    # = 2 * sum_{j,i} a[j] q[j+i] a[i]
    # The Hessian is 2*(Q + Q^T) where Q[j,i] = q[j+i]
    # = 2 * (Q + Q^T)
    # For the QP: min (1/2) a^T (2(Q+Q^T)) a = min a^T(Q+Q^T)a
    # So the Hessian matrix is 2(Q + Q^T) where Q[j,i] = q[j+i]

    # Build Q (Hankel matrix)
    Q = np.zeros((n, n))
    for j in range(n):
        end = min(n, 2 * n - 1 - j)
        Q[j, :end] = q_weights[j:j + end]
    H = Q + Q.T  # Symmetric Hessian/2 for the objective a^T H a
    # Objective: min a^T H a s.t. 1^T a = 1, a >= 0
    # KKT: 2Ha + lambda*1 - mu = 0, 1^T a = 1, a>=0, mu>=0, a*mu=0

    # Active set method
    active = np.ones(n, dtype=bool)  # Initially all free
    for _ in range(max_iter):
        free = ~active
        n_free = free.sum()
        if n_free == 0:
            break
        if n_free == 1:
            idx = np.where(free)[0][0]
            a_new = np.zeros(n)
            a_new[idx] = 1.0
            a = a_new
            break

        H_ff = H[np.ix_(free, free)]
        f_vec = np.zeros(n)
        f_vec[free] = 1.0

        try:
            # Solve: minimize a^T H a s.t. 1^T a = 1 over free variables
            # Solution: a_free = H_ff^{-1} 1 / (1^T H_ff^{-1} 1)
            Hff_1 = np.linalg.solve(H_ff + 1e-12 * np.eye(n_free), np.ones(n_free))
            denom = Hff_1.sum()
            if denom < 1e-15:
                # Fallback: uniform on free set
                a_new = np.zeros(n)
                a_new[free] = 1.0 / n_free
                a = a_new
                break
            a_free = Hff_1 / denom
        except np.linalg.LinAlgError:
            a_new = np.zeros(n)
            a_new[free] = 1.0 / n_free
            a = a_new
            break

        a_new = np.zeros(n)
        a_new[free] = a_free
        a_new[active] = 0.0

        if np.all(a_new >= -1e-10):
            a = np.maximum(a_new, 0.0)
            break
        else:
            # Add violated variable to active set
            min_idx = np.argmin(a_new)
            active[min_idx] = True
            a = np.maximum(a_new, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) / n
                break
            a = a / s

    return a / a.sum() if a.sum() > 1e-15 else None


def smooth_gradient_descent(a0, time_limit, T_init=0.5, T_max=500.0, lr_scale=1.0):
    """Smoothed gradient descent with adaptive temperature.
    Minimizes f_T(a) = (1/T) log(sum_k exp(T*conv_k)) which converges to max_k conv_k.
    """
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    T = T_init
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        obj = conv.max()

        # Compute softmax weights
        scaled = T * (conv - conv.max())
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            p = p / p_sum

        # Compute gradient
        grad = compute_hankel_grad(a, p)

        # Gradient descent with projection
        # Step size: use 1/(T * ||grad||^2) heuristic or fixed
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        lr = lr_scale * n / (T * grad_norm * grad_norm + 1e-10)
        lr = min(lr, 0.5 / T)  # Cap step size

        a_new = project_simplex(a - lr * grad)
        a = a_new

        new_obj = fast_convolve(a).max()
        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt T
        if no_improve > 15:
            T = min(T * 1.15, T_max)
            no_improve = 0
        if no_improve > 100:
            T = min(T * 1.5, T_max)
        if no_improve > 300:
            # Restart
            rng = np.random.default_rng((it * 104729 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            no_improve = 0

    return best_a, best_obj


def active_set_refine(a0, time_limit, n_active_frac=0.1):
    """Refine using active-set QP. Finds the active set (peaks) and solves QP."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        obj = conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Find active set: indices where conv is close to max
        threshold = obj - 0.01 * obj + 1e-10
        active_indices = np.where(conv >= threshold)[0]

        if len(active_indices) == 0:
            active_indices = np.array([np.argmax(conv)])

        # Create q weights: uniform on active set
        q = np.zeros(2 * n - 1)
        q[active_indices] = 1.0 / len(active_indices)

        # Solve QP
        a_qp = solve_qp_active_set(a, q)
        if a_qp is not None:
            a_qp_obj = fast_convolve(a_qp).max()
            if a_qp_obj < best_obj - 1e-12:
                best_obj = a_qp_obj
                best_a = a_qp.copy()
                a = a_qp
            elif a_qp_obj < obj + 1e-10:
                # QP didn't improve much; try with expanded active set
                threshold2 = obj - 0.05 * obj
                active_indices2 = np.where(conv >= threshold2)[0]
                if len(active_indices2) > len(active_indices):
                    q2 = np.zeros(2 * n - 1)
                    q2[active_indices2] = 1.0 / len(active_indices2)
                    a_qp2 = solve_qp_active_set(a, q2)
                    if a_qp2 is not None:
                        a_qp2_obj = fast_convolve(a_qp2).max()
                        if a_qp2_obj < best_obj - 1e-12:
                            best_obj = a_qp2_obj
                            best_a = a_qp2.copy()
                            a = a_qp2
                        elif a_qp2_obj < obj + 1e-10:
                            a = a_qp2
            else:
                a = a_qp

        # Small perturbation to avoid cycling
        if it % 20 == 0:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
            a = project_simplex(a + rng.standard_normal(n) * 0.001)

        # Check convergence
        conv2 = fast_convolve(a)
        if abs(conv2.max() - best_obj) < 1e-14 and it > 50:
            break

    return best_a, best_obj


def fw_lp_step(f, g, n):
    """Line search between f and g."""
    best_alpha = 0.0
    best_val = fast_convolve(f).max()
    alphas = [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
    for alpha in alphas:
        ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    return best_alpha, best_val
