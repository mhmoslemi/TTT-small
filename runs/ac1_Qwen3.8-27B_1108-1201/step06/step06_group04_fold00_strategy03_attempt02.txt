def evaluate_sequence(sequence):
    """Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid."""
    if not isinstance(sequence, list):
        return np.inf
    if len(sequence) == 0:
        return np.inf
    try:
        for x in sequence:
            # Reject boolean types (as they are a subclass of int) and any other non-numeric types (like strings or complex numbers)
            if isinstance(x, bool) or not isinstance(x, (int, float, np.number)):
                return np.inf
            # Reject Not-a-Number (NaN) and infiity values
            if np.isnan(x) or np.isinf(x):
                return np.inf
        # Convert all elements to float for consistence
        sequence = [float(x) for x in sequence]
    except Exception:
        return np.inf
    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]
    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence] if any(x > 1000 for x in sequence) else sequence

        return np.inf
