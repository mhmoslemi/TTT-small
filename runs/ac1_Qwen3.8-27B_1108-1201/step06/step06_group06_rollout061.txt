import numpy as np
from scipy.optimize import minimize
from scipy.sparse import lil_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_obj_and_grad(x, T, k_smooth=1):
    """
    Compute the smoothed objective and gradient in the softmax parameterization.
    x: free parameters
    T: temperature for the max smoothing
    k_smooth: number of top peaks to include (for top-k smoothing)
    """
    # Stable softmax
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    
    # Compute convolution
    conv = fast_convolve(a)
    
    # Smoothed max: use log-sum-exp with temperature T
    scaled = conv / T
    scaled_max = scaled.max()
    log_sum_exp = scaled_max + np.log(np.exp(scaled - scaled_max).sum())
    obj = T * log_sum_exp  # This is the smoothed max
    
    # For top-k smoothing:
    # Sort conv values in descending order, take top k, compute their average
    # This is smoother than the pure max
    
    # Gradient: d(obj)/d(a[j]) = 2 * sum_k p[k] * a[k-j] (zero-padded)
    # where p[k] = exp(conv[k]/T) / sum(exp(conv[k']/T))
    
    p = np.exp(conv / T - log_sum_exp)  # normalized
    # Actually: p[k] = exp(conv[k]/T) / sum(exp(conv[k']/T))
    # log_sum_exp = log(sum(exp(conv/T))), so exp(conv[k]/T)/sum = exp(conv[k]/T - log_sum_exp)
    # = exp((conv[k]/T) - (conv[k]/T) - log(sum(exp)))... no
    # p[k] = exp(scaled[k]) / sum(exp(scaled)) = exp(scaled[k] - log_sum_exp_of_scaled)
    # where log_sum_exp_of_scaled = scaled_max + log(sum(exp(scaled - scaled_max)))
    
    p = np.exp(scaled - scaled_max)
    p = p / p.sum()
    
    # Gradient with respect to a:
    # d(conv[k])/da[j] = 2*a[k-j] (zero-padded)
    # So d(obj)/da[j] = 2*sum_k p[k]*a[k-j]
    
    # This is 2 * (p_padded * a) where p_padded is p zero-extended appropriately
    # More precisely: (grad_a)[j] = 2 * sum_{m=0}^{n-1} a[m] * p[j+m]
    
    n = len(a)
    # p has length 2n-1
    # For each j, we need sum_m a[m] * p[j+m] where j+m is in [0, 2n-2]
    # This is a "cross-correlation" between a and p

    # a = softmax(x), so da[i]/dx[j] = a[i]*(delta_ij - a[j])
    # d(obj)/dx[j] = sum_i d(obj)/da[i] * da[i]/dx[j]

    # = grad_a[j]*a[j] - a[j]*sum_i grad_a[i]*a[i]
    
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)
    
    return obj, grad_x

"""

Uses L-BFGS on smoothed log-sum-exp objective with softmax parameterization,
adaptive temperature annealing, peak equalization, basin hopping, and multi-n search.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_a_fft(a, p):
    """Compute grad_a[j] = 2 * sum_m a[m] * p[j+m] using FFT."""
    n = len(a)
    q = p[::-1].copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    Q = np.fft.rfft(q, fft_size)
    conv = np.fft.irfft(A * Q, fft_size)[:size]
    return 2.0 * conv[::-1].copy()


def softmax_to_a(x):
    """Convert log-parameters to probability vector via stable softmax."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return a / s


def a_to_softmax_params(a):
    """Convert probability vector to log-parameters."""
    eps = 1e-15
    return np.log(np.maximum(a, eps))


def lse_obj_grad(x, T):
    """
    Log-sum-exp smoothed max of autocorrelation, with gradient.
    Objective: max(conv) + T * log(sum(exp((conv - max(conv))/T)))
    """
    a = softmax_to_a(x)
    n = len(a)
    conv = fast_convolve(a)
    
    conv_max = conv.max()
    scaled = (conv - conv_max) / T
    # All scaled values are <= 0, so exp is in (0, 1]
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    obj = conv_max + T * log_sum
    
    # Weights: p[k] = exp((conv[k] - max)/T) / sum(...)
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient w.r.t. a
    grad_a = compute_grad_a_fft(a, p)
    
    # Chain rule: d(obj)/dx[j] = sum_i d(obj)/da[i] * da[i]/dx[j]
    # da[i]/dx[j] = a[i] * (delta_ij - a[j])
    # d(obj)/dx[j] = grad_a[j]*a[j] - a[j]*sum_i grad_a[i]*a[i]
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)
    
    return obj, grad_x


def lbfgs_optimize(a_start, T, max_iter=100, tol=1e-12):
    """Run L-BFGS on the smoothed objective for a given temperature."""
    x = a_to_softmax_params(a_start)
    
    def func_and_grad(x):
        return lse_obj_grad(x, T)
    
    result = minimize(func_and_grad, x, method='L-BFGS-B',
                      jac=True, options={'maxiter': max_iter, 'gtol': tol,
                                         'maxls': 30, 'ftol': 1e-15})
    
    a_new = softmax_to_a(result.x)
    return a_new, result


def peak_equalization(a, n_iter=1000, lr_init=0.01):
    """
    Direct peak-targeting: iteratively reduce the maximum autocorrelation value
    using subgradient descent on the true max.
    """
    a = a.copy()
    n = len(a)
    best_a = a.copy()
    best_peak = fast_convolve(a).max()
    lr = lr_init
    
    for it in range(n_iter):
        conv = fast_convolve(a)
        peak_val = conv.max()
        
        # Find all lags at (or near) the peak
        threshold = peak_val * (1 - 1e-6)
        peak_indices = np.where(conv >= threshold)[0]
        
        # Subgradient: average of gradients at peak lags
        # grad R[k] w.r.t. a[j] = 2*a[k-j] (zero-padded)
        # For multiple peaks, average the subgradients
        n_peaks = len(peak_indices)
        grad = np.zeros(n)
        for k in peak_indices:
            # grad[j] += 2*a[k-j] (with zero-padding)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * a[idx]
        
        grad /= n_peaks
        
        # Project subgradient onto tangent space of simplex
        # Tangent space: {d : sum(d) = 0, d[i] >= 0 where a[i] = 0}
        # Simple projection: subtract mean, then set negative components on zero set
        grad_tangent = grad - grad.mean()
        # For zero components of a, the gradient should be non-negative
        # (we can't decrease them below 0)
        zero_mask = a < 1e-15
        if zero_mask.any():
            # If gradient at zero positions is negative, we'd want to increase them
            # But that would require decreasing others, which might not help
            # For simplicity, just zero out the gradient at non-positive positions
            pass
        
        # Take step
        a_new = a - lr * grad_tangent
        a_new = project_simplex(a_new)
        
        new_peak = fast_convolve(a_new).max()
        if new_peak < best_peak:
            best_peak = new_peak
            best_a = a_new.copy()
            a = a_new

        else:
            # Decrease learning rate
            lr *= 0.9
            if lr < 1e-8:
                break

"""
Optimization for minimum peak autocorrelation step function.
Uses L-BFGS on smoothed log-sum-exp objective with softmax parameterization,
adaptive temperature annealing, FFT-based peak equalization, basin hopping,
and multi-n exploration.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_a_fft(a, p):
    """Compute grad_a[j] = 2 * sum_m a[m] * p[j+m] using FFT."""
    n = len(a)
    q = p[::-1].copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    Q = np.fft.rfft(q, fft_size)
    conv = np.fft.irfft(A * Q, fft_size)[:size]
    return 2.0 * conv[::-1].copy()


def softmax_to_a(x):
    """Convert log-parameters to probability vector via stable softmax."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return a / s


def a_to_log_params(a):
    """Convert probability vector to log-parameters for softmax."""
    eps = 1e-15
    return np.log(np.maximum(a, eps))


def lse_obj_grad(x, T):
    """
    Log-sum-exp smoothed max of autocorrelation with gradient.
    """
    a = softmax_to_a(x)
    conv = fast_convolve(a)
    conv_max = conv.max()
    scaled = (conv - conv_max) / T
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    obj = conv_max + T * log_sum
    p = exp_scaled / exp_scaled.sum()
    grad_a = compute_grad_a_fft(a, p)
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)
    return obj, grad_x


def lbfgs_run(a_start, T, max_iter=80):
    """Run L-BFGS on smoothed objective."""
    x = a_to_log_params(a_start)

    def fg(x):
        return lse_obj_grad(x, T)

    result = minimize(fg, x, method='L-BFGS-B', jac=True,
                      options={'maxiter': max_iter, 'gtol': 1e-14,
                               'maxls': 25, 'ftol': 1e-16})
    a_new = softmax_to_a(result.x)
    return a_new


def annealed_lbfgs(a_start, time_limit, T_init=None, T_min=None, n_steps=None):
    """
    Multi-temperature L-BFGS annealing.
    """
    n = len(a_start)
    if T_init is None:
        T_init = 2.0 / n
    if T_min is None:
        T_min = T_init / (n ** 1.5)
    if n_steps is None:
        n_steps = 30

    a = a_start.copy()
    best_a = a.copy()
    best_peak = fast_convolve(a).max()
    t0 = time.monotonic()

    # Geometric T schedule
    if n_steps > 1:
        ratio = (T_min / T_init) ** (1.0 / (n_steps - 1))
    else:
        ratio = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        T = T_init * (ratio ** step)
        max_iter = max(20, min(80, int((time_limit - (time.monotonic() - t0)) * 2)))
        a_new = lbfgs_run(a, T, max_iter=max_iter)
        peak = fast_convolve(a_new).max()
        if peak < best_peak:
            best_peak = peak
            best_a = a_new.copy()
            a = a_new
        else:
            a = a_new  # still update a to follow the landscape

    return best_a, best_peak


def peak_equalization(a, n_iter=3000, lr_init=0.003, T_peak_ratio=0.005):
    """
    FFT-based peak equalization using soft peak weighting.
    """
    a = a.copy()
    n = len(a)
    best_a = a.copy()
    best_peak = fast_convolve(a).max()
    lr = lr_init
    no_improve = 0

    for it in range(n_iter):
        conv = fast_convolve(a)
        peak_val = conv.max()

        # Soft peak indicator
        T_pk = max(peak_val * T_peak_ratio, peak_val * 1e-6)
        scaled = (conv - peak_val) / T_pk
        P = np.exp(scaled)
        P = P / P.sum()

        # Gradient via FFT
        grad = compute_grad_a_fft(a, P)
        # Project to tangent space of simplex
        grad = grad - grad.mean()

        # Backtracking line search
        best_alpha = 0.0
        best_new_peak = best_peak
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            alpha = lr * mult
            a_new = project_simplex(a - alpha * grad)
            new_peak = fast_convolve(a_new).max()
            if new_peak < best_new_peak:
                best_new_peak = new_peak
                best_alpha = alpha

        if best_alpha > 0:
            a = project_simplex(a - best_alpha * grad)
            curr_peak = fast_convolve(a).max()
            if curr_peak < best_peak - 1e-15:
                best_peak = curr_peak
                best_a = a.copy()
                no_improve = 0
                lr = min(lr * 1.005, 0.02)
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 200:
            lr *= 0.7
            no_improve = 0
        if lr < 1e-10:
            break

    return best_a, best_peak


def coordinate_descent(a, n_iter=5000, seed=42):
    """
    Coordinate descent: try adjusting pairs of coordinates.
    """
    rng = np.random.default_rng(seed)
    a = a.copy()
    n = len(a)
    best_a = a.copy()
    best_peak = fast_convolve(a).max()

    for it in range(n_iter):
        improved = False
        # Try moving mass from i to j
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
            amount = a[i] * frac
            if amount < 1e-15:
                continue
            for direction in [1, -1]:
                a_new = a.copy()
                a_new[i] -= direction * amount
                a_new[j] += direction * amount
                if a_new[i] < 0 or a_new[j] < 0:
                    continue
                s = a_new.sum()
                a_new = a_new / s
                new_peak = fast_convolve(a_new).max()
                if new_peak < best_peak - 1e-15:
                    a = a_new
                    best_peak = new_peak
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved and it > 1000:
            # Try random perturbation
            scale = 0.001 * (it / n_iter) + 0.0003
            a_new = project_simplex(a + rng.standard_normal(n) * scale)
            new_peak = fast_convolve(a_new).max()
            if new_peak < best_peak - 1e-15:
                a = a_new
                best_peak = new_peak
                best_a = a.copy()

    return best_a, best_peak


def basin_hopping(a_best, time_limit, seed=123):
    """
    Basin hopping with re-optimization from perturbed points.
    """
    rng = np.random.default_rng(seed)
    n = len(a_best)
    best_a = a_best.copy()
    best_peak = fast_convolve(a_best).max()

def try_seq(a, label, best_seq, best_score):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
