"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
- Structured initialization via parametric profiles (power laws, exponentials)
- Fast Frank-Wolfe with LP oracle for local optimization
- Peak equalization via dual variable updates
- Evolutionary search with structured mutations for global exploration
- Final micro-polish with targeted perturbations

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_profile(n, profile_type, param):
    """Generate a structured profile."""
    j = np.arange(n, dtype=np.float64)
    if profile_type == 'power_asc':
        a = (j + 1) ** param
    elif profile_type == 'power_desc':
        a = (n - j) ** param
    elif profile_type == 'exp_asc':
        a = np.exp(param * j / n)
    elif profile_type == 'exp_desc':
        a = np.exp(-param * j / n)
    elif profile_type == 'cos_asc':
        a = 0.5 + 0.5 * np.sin(param * np.pi * j / (n - 1) / 2)
    elif profile_type == 'bump':
        center = param * n
        a = np.exp(-((j - center) ** 2) / (2 * (n / 4) ** 2))
    elif profile_type == 'uniform':
        a = np.ones(n)
    else:
        a = np.ones(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.3:
            break

        # Solve LP oracle
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        g = None
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': min(10, remaining * 0.5)})
            if res.success:
                g = np.maximum(res.x, 0.0)
        except Exception:

        if g is None:

                if st < 1e-15:

                ft = ft / st

                if val < best_val:

                    hi = mid

            conv = fast_convolve(f)

                no_improve = 0

def peak_equalization(a0, time_limit):
    """
    Iterative peak equalization: adjust the sequence to make all autocorrelation
    values as equal as possible. Uses a multiplicative update inspired by
    information geometry.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        M = conv.max()
        
        # Find all peaks above a threshold
        threshold = M * 0.7
        peaks = np.where(conv > threshold)[0]
        
        # Compute a "pressure" for each variable based on how much it
        # contributes to each peak
        pressure = np.zeros(n)
        for k in peaks:
            excess = conv[k] - M * 0.5  # How much this peak exceeds a target
            if excess < 0:
                continue
            js = max(0, int(k) - n + 1)
            je = min(n - 1, int(k))
            for j in range(js, je + 1):
                partner = int(k) - j
                if 0 <= partner < n:
                    pressure[j] += excess * a[partner]
        
        # Multiplicative update: reduce variables with high pressure
        # a_new[j] = a[j] * exp(-eta * pressure[j])
        eta = 0.01
        update = np.exp(-eta * pressure / (pressure.max() + 1e-15))
        a_new = a * update
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a_new = np.maximum(a_new, 0.0)
        
        new_obj = fast_convolve(a_new).max()
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Don't update a if no improvement
            a = best_a.copy()
        
        a = a_new
        
        if no_improve > 50:
            break

    return best_a, best_obj


def evolved_search(a0, time_limit, pop_size=20):

    """

    rng = np.random.default_rng(12345)

        a_i = project_simplex(a0 + rng.standard_normal(n) * scale)
        pop.append(a_i)

    best_obj = fast_convolve(a0).max()
    
    t0 = time.monotonic()
    gen = 0
    
    while time.monotonic() - t0 < time_limit:
        gen += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        
        # Evaluate all candidates
        objs = []
        for i in range(len(pop)):
            obj = fast_convolve(pop[i]).max()
            objs.append(obj)
            if obj < best_obj:
                best_obj = obj
                best_a = pop[i].copy()
        
        # Sort by fitness
        order = np.argsort(objs)
        pop = [pop[i] for i in order]
        objs = [objs[i] for i in order]
        
        # Elitism: keep top half
        elite_count = max(2, len(pop) // 3)
        
        # Generate new candidates
        new_pop = [pop[i].copy() for i in range(elite_count)]

            # Choose parent (tournament selection)
            t1 = rng.integers(0, min(5, len(pop)))
            t2 = rng.integers(0, min(5, len(pop)))
            parent = pop[t1] if objs[t1] < objs[t2] else pop[t2]
            
            # Choose mutation type
            mut_type = rng.integers(0, 5)
            
            if mut_type == 0:
                # Gaussian mutation

            elif mut_type == 1:
                # Peak-targeted mutation
                conv = fast_convolve(parent)
                peak_k = int(np.argmax(conv))
                noise = rng.standard_normal(n) * 0.003
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                for j in range(js, je + 1):
                    partner = peak_k - j
                    if 0 <= partner < n:
                        noise[j] -= 0.002 * parent[partner]
                child = project_simplex(parent + noise)
            elif mut_type == 2:
                # Block swap: move mass from one region to another
                child = parent.copy()
                block_size = rng.integers(max(2, n // 20), max(3, n // 10))
                src_start = rng.integers(0, n - block_size)
                dst_start = rng.integers(0, n - block_size)
                if abs(src_start - dst_start) < block_size:
                    continue
                mass = child[src_start:src_start + block_size].sum() * 0.1
                child[src_start:src_start + block_size] -= mass / block_size
                child[dst_start:dst_start + block_size] += mass / block_size
                child = np.maximum(child, 0.0)
                s = child.sum()
                if s > 1e-15:
                    child = child / s
            elif mut_type == 3:
                # Crossover with another parent
                idx2 = rng.integers(0, min(5, len(pop)))
                parent2 = pop[idx2]
                point = rng.integers(1, n - 1)
                child = np.concatenate([parent[:point], parent2[point:]])
                child = np.maximum(child, 0.0)
                s = child.sum()
                if s > 1e-15:
                    child = child / s
                child = project_simplex(child)
            else:
                # Asymmetry-preserving mutation: perturb the "shape" parameter
                # Reshape the sequence to a smooth curve and perturb it
                from scipy.interpolate import interp1d
                x_orig = np.linspace(0, 1, n)
                x_dense = np.linspace(0, 1, max(50, n // 4))
                f_interp = np.interp(x_dense, x_orig, parent)
                f_interp += rng.standard_normal(len(f_interp)) * 0.01
                f_interp = np.maximum(f_interp, 0)
                child = np.interp(x_orig, x_dense, f_interp)
                child = project_simplex(child)
            
            new_pop.append(child)
        
        pop = new_pop
        
        # Replace worst with best found so far (to maintain quality)
        if gen % 5 == 0:
            pop[-1] = best_a.copy()

    return best_a, best_obj


def nesterov_subgradient(a0, time_limit, lr_init=0.005, max_k=50):
    """Nesterov-accelerated subgradient with top-k peak tracking."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    y = a.copy()
    v = np.zeros(n)
    t_nest = 1.0
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        alpha = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / alpha
        y = (1.0 + beta) * a - beta * y
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y / ys
        t_nest += 1.0

        conv = fast_convolve(y)
        if max_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -max_k)[-max_k:]

        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * y[k - idx_arr]

        v = beta * v - lr * sg
        a_new = a + v
        a_new = project_simplex(a_new)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            a_new = a

        a = a_new
        lr *= (1.0 - 0.0003)

        if no_improve > 100:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.001 * np.sqrt(no_improve / 100.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            y = a.copy()
            v = np.zeros(n)
            t_nest = 1.0
            no_improve = 0

    return best_a, best_obj
