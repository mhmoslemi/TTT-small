def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """FFT-based cross-convolution of a and b."""
    n = max(len(a), len(b))

    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    best_val = conv.max()

    candidates = [0.0, alpha_max * 0.01, alpha_max * 0.05, alpha_max * 0.1, 
                  alpha_max * 0.2, alpha_max * 0.3, alpha_max * 0.5,
                  alpha_max * 0.7, alpha_max * 0.9, alpha_max]
    
    # Add intersection points of top peaks
    top_k = np.argpartition(conv, -10)[-10:]
    for i in range(len(top_k)):
        for j in range(i + 1, len(top_k)):
            ki, kj = top_k[i], top_k[j]
            di = delta_conv[ki]
            dj = delta_conv[kj]
            if abs(di - dj) > 1e-15:
                alpha_int = (conv[kj] - conv[ki]) / (di - dj)
                if 0 < alpha_int < alpha_max:
                    candidates.append(alpha_int)
    
    for alpha in candidates:
        alpha = max(0.0, min(alpha, alpha_max))
        val = np.max(conv + alpha * delta_conv)
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    # Refine with bisection around best_alpha
    if best_alpha > 0:
        lo = max(0, best_alpha * 0.8)
        hi = min(alpha_max, best_alpha * 1.2)
        for _ in range(20):
            mid = (lo + hi) / 2
            val = np.max(conv + mid * delta_conv)
            if val < best_val:
                best_val = val
                best_alpha = mid
                lo = mid
            else:
                hi = mid - 1e-15
    
    return best_alpha, best_val


def optimize_minsg(a0, time_limit, tol=1e-14, verbose=False):
    """Main optimization loop using minimum-norm subgradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_a = a.copy()
    best_obj = conv.max()
    
    t0 = time.monotonic()
    it = 0
    active_tol = 0.01  # Fraction of max to include in active set
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.05:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        if M < best_obj:
            best_obj = M
            best_a = a.copy()
        
        # Identify active peaks
        threshold = M * (1.0 - active_tol)
        active_peaks = np.where(conv >= threshold)[0]
        
        if len(active_peaks) == 0:
            active_peaks = np.array([np.argmax(conv)])
        
        # Compute minimum-norm subgradient
        d = compute_min_norm_subgrad(a, conv, active_peaks)
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        
        d = d / d_norm  # Normalize for step size interpretation
        
        # Find optimal step size
        alpha, new_val = optimal_step_size(a, d, conv)
        
        if alpha < 1e-12:
            # Try smaller active tolerance
            active_tol *= 0.5
            if active_tol < 1e-6:
                break
            continue
        
        # Take the step
        a_new = a + alpha * d
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        
        # Check improvement
        conv_new = fast_convolve(a_new)
        new_max = conv_new.max()
        
        if new_max < M - 1e-16:
            a = a_new
            # Adjust active tolerance
            if len(active_peaks) > 15:
                active_tol = max(active_tol * 0.95, 0.001)
            else:
                active_tol = min(active_tol * 1.05, 0.05)
        else:
            # Try smaller step
            alpha_half = alpha * 0.5
            a_new = np.maximum(a + alpha_half * d, 0.0)
            a_new = a_new / a_new.sum()
            conv_new = fast_convolve(a_new)
            new_max = conv_new.max()
            if new_max < M - 1e-16:
                a = a_new
                alpha *= 0.5
            else:
                # No improvement, try random perturbation
                if it % 50 == 0:
                    rng = np.random.default_rng(it * 31 + 7)
                    noise = rng.standard_normal(n) * 0.001
                    a = project_simplex(best_a + noise)
                    active_tol = 0.01
                else:
                    break
    
    return best_a, best_obj


def parametric_initial(n, alpha, beta, family):
    """Generate parametric initial sequences."""
    j = np.arange(n, dtype=np.float64)
    if family == 'uniform':
        a = np.ones(n)
    elif family == 'power_edge':
        # a_j ∝ (j+1)^alpha * (n-1-j+1)^beta
        a = (j + 1) ** alpha * (n - j) ** beta
    elif family == 'cosine':
        a = np.cos(np.pi * (j + 0.5) / (2 * n)) ** alpha
    elif family == 'gaussian':
        center = (n - 1) / 2.0
        sigma = alpha * n
        a = np.exp(-((j - center) ** 2) / (2 * sigma ** 2))
    elif family == 'U_shape':
        center = (n - 1) / 2.0
        dist = np.abs(j - center) / center
        a = 1.0 / (1.0 + alpha * dist ** 2)
    elif family == 'sech':
        center = (n - 1) / 2.0
        dist = (j - center) / (alpha * n)
        a = 1.0 / np.cosh(dist)
    elif family == 'bimodal':
        center1 = alpha * n
        center2 = (1 - alpha) * n
        sigma = beta * n
        a = np.exp(-((j - center1) ** 2) / (2 * sigma ** 2)) + \
            np.exp(-((j - center2) ** 2) / (2 * sigma ** 2))
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def symmetric_optimize(a0, time_limit, verbose=False):
    """Optimization enforcing palindromic symmetry."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    
    # Enforce symmetry
    half = (a[:n//2 + 1] + a[n - 1 - np.arange(n//2 + 1)][::-1]) / 2.0
    # Actually, let me just do it properly
    sym = (a + a[::-1]) / 2.0
    s = sym.sum()
    if s < 1e-15:
        sym = np.ones(n) / n
    else:
        sym = sym / s
    
    # Work on the half sequence
    n_half = n // 2 + 1  # Include center if odd
    half_seq = sym[:n_half].copy()
    
    def reconstruct(h):
        # Rebuild full symmetric sequence from half
        full = np.zeros(n)
        full[:n_half] = h
        full[n_half:] = h[:n - n_half][::-1]
        s = full.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return full / s
    
    def obj_half(h):
        full = reconstruct(h)
        return fast_convolve(full).max()
    
    # Gradient-based optimization on the half sequence
    best_half = half_seq.copy()
    best_val = obj_half(half_seq)
    
    t0 = time.monotonic()
    lr = 0.01
    mom = 0.9
    v = np.zeros(n_half)
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        full = reconstruct(half_seq)
        conv = fast_convolve(full)
        M = conv.max()
        
        if M < best_val:
            best_val = M
            best_half = half_seq.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Subgradient for the half sequence
        threshold = M * 0.99
        active_peaks = np.where(conv >= threshold)[0]
        if len(active_peaks) == 0:
            active_peaks = np.array([np.argmax(conv)])
        
        # Compute subgradient for full sequence
        sg_full = np.zeros(n)
        weights = np.ones(len(active_peaks)) / len(active_peaks)
        for i, k in enumerate(active_peaks):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg_full[idx_arr] += weights[i] * 2.0 * full[k - idx_arr]
        
        # Project to half
        sg_half = (sg_full[:n_half] + sg_full[n - 1 - np.arange(n_half)][::-1]) / 2.0
        sg_half = sg_half - np.mean(sg_half)  # Tangent space (for half)
        
        # Momentum step
        v = mom * v - lr * sg_half
        candidate = half_seq + v
        candidate = np.maximum(candidate, 0)
        cs = candidate.sum()
        if cs < 1e-15:
            candidate = half_seq.copy()
        else:
            candidate = candidate / cs
        
        new_val = obj_half(candidate)
        if new_val < M:
            half_seq = candidate
        else:
            v *= 0.5
            # Backtrack
            for scale in [0.5, 0.25, 0.125]:
                candidate = np.maximum(half_seq + scale * v, 0)
                cs = candidate.sum()
                if cs > 1e-15:
                    candidate = candidate / cs
                new_val = obj_half(candidate)
                if new_val < M:
                    half_seq = candidate
                    break
        
        # Adapt
        if no_improve > 30:
            lr *= 0.95
        if no_improve > 100:
            rng = np.random.default_rng(it * 17 + 3)
            half_seq = np.maximum(best_half + rng.standard_normal(n_half) * 0.005, 0)
            half_seq = half_seq / half_seq.sum()
            v = np.zeros(n_half)
            lr = 0.01
            no_improve = 0
    
    best_full = reconstruct(best_half)
    return best_full, obj_half(best_half)


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def coordinate_descent_polish(a0, time_limit, n_iter_max=5000):
    """Simple coordinate descent: try moving mass between pairs of adjacent positions."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    delta = 0.001
    
    while time.monotonic() - t0 < time_limit and it < n_iter_max:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        improved = False
        # Random pair
        rng = np.random.default_rng(it)
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Try moving delta from i to j (and j to i)
        for di, dj in [(delta, -delta), (-delta, delta), (delta/2, -delta/2), (-delta/2, delta/2)]:
            if a[i] + di < 0 or a[j] + dj < 0:
                continue
            a_new = a.copy()
            a_new[i] += di
            a_new[j] += dj
            conv_new = fast_convolve(a_new)
            val = conv_new.max()
            if val < best_obj - 1e-15:
                a = a_new
                best_obj = val
                best_a = a.copy()
                improved = True
                break
        
        if not improved and it % 500 == 0:
            delta *= 0.9
    
    return best_a, best_obj
