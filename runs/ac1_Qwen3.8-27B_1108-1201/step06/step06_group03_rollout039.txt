"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Spectral initialization with random flat-phase spectra
2. Iterative spectral flattening (amplitude compression of DFT)
3. Subgradient descent with adaptive peak weighting
4. Targeted peak-shaving coordinate descent
5. Multi-length exploration
"""
import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_flatten(a, alpha, n_fft=None):
    """
    Apply spectral amplitude compression.
    A_new = |A|^alpha * exp(i*angle(A))
    a_new = real(IDFT(A_new))[:n]
    """
    n = len(a)
    if n_fft is None:
        n_fft = 1 << (2*n - 2).bit_length()
    A = np.fft.rfft(a, n_fft)
    mag = np.abs(A)
    phase = np.exp(1j * np.angle(A))
    # Compress amplitude
    mag_new = mag ** alpha
    A_new = mag_new * phase
    a_new = np.fft.irfft(A_new, n_fft)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_new / s

def subgradient_step(a, c, peaks, lr):
    """
    One subgradient step toward reducing the specified peaks.
    peaks: array of peak indices
    Returns new a (on simplex) and the step's effect.
    """
    n = len(a)
    grad = np.zeros(n)
    for k in peaks:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad[j] += 2.0 * a[k - j]
    # Normalize gradient
    g_norm = np.linalg.norm(grad)
    if g_norm < 1e-30:
        return a.copy()
    grad = grad / g_norm
    v = a - lr * grad
    return project_simplex(v)

def peak_shave(a, max_rounds=100, step_base=0.001):
    """
    Targeted peak shaving: for each top peak, move mass from
    the largest contributing pair to a position that minimizes damage.
    """
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for round_i in range(max_rounds):
        c = fast_convolve(best_a)
        # Get top peaks
        n_peaks = min(5, len(c))
        peak_idx = np.argsort(c)[::-1][:n_peaks]
        
        improved = False
        for k in peak_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            # Find the pair with max product
            max_prod = 0
            best_j1 = -1
            best_j2 = -1
            for j in range(js, je + 1):
                prod = best_a[j] * best_a[k - j]
                if prod > max_prod:
                    max_prod = prod
                    best_j1 = j
                    best_j2 = k - j
            
            if best_j1 < 0 or max_prod < 1e-15:
                continue
            
            # Try moving mass from the larger of the two to a random distant position
            src = best_j1 if best_a[best_j1] >= best_a[best_j2] else best_j2
            if best_a[src] < 1e-10:
                continue
            
            # Find a good target: position that minimizes contribution to peaks
            # Simple heuristic: far from all peaks
            candidates = []
            for _ in range(20):
                j_target = np.random.randint(0, n)
                if abs(j_target - src) < n // 4:
                    continue
                # Check contribution to main peaks
                contrib = 0
                for pk in peak_idx:
                    pk = int(pk)
                    if 0 <= pk - j_target < n:
                        contrib += best_a[pk - j_target]
                    if 0 <= j_target - pk < n:  # This isn't quite right for convolution
                        pass
                candidates.append((contrib, j_target))
            
            if not candidates:
                continue
            candidates.sort()
            
            for contrib, j_target in candidates[:3]:
                for frac in [step_base, step_base * 0.5, step_base * 2.0]:
                    delta = best_a[src] * frac
                    if delta > best_a[src]:
                        continue
                    candidate = best_a.copy()
                    candidate[src] -= delta
                    candidate[j_target] += delta
                    candidate = np.maximum(candidate, 0)
                    candidate = candidate / candidate.sum()
                    val = fast_convolve(candidate).max()
                    if val < best_obj - 1e-15:
                        best_a = candidate
                        best_obj = val
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            step_base *= 0.9
            if step_base < 1e-6:
                break
    
    return best_a, best_obj

def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s

def spectral_optimize(a0, time_limit, alphas=None):
    """
    Iterative spectral optimization: repeatedly apply amplitude compression
    with different exponents and keep the best result.
    """
    if alphas is None:
        alphas = [0.3, 0.5, 0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.5]
    
    n = len(a0)
    best_a = a0.copy()
    best_obj = fast_convolve(a0).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Try different alpha values
        for alpha in alphas:
            if time.monotonic() - t0 > time_limit:
                break
            a_new = spectral_flatten(best_a, alpha)
            obj = fast_convolve(a_new).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = a_new
        
        # Also try random phase perturbation
        if it % 5 == 0:
            A = np.fft.rfft(best_a, 1 << (2*n-2).bit_length())
            A_pert = A * np.exp(1j * rng.standard_normal(len(A)) * 0.01)
            a_pert = np.fft.irfft(A_pert, 1 << (2*n-2).bit_length())[:n]
            a_pert = np.maximum(a_pert, 0)
            s = a_pert.sum()
            if s > 1e-15:
                a_pert = a_pert / s
                obj = fast_convolve(a_pert).max()
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_a = a_pert
        
        # Check convergence
        if it > 100 and it % 50 == 0:
            # Try a fresh random initialization
            A_rand = rng.standard_normal(1 << (2*n-2).bit_length() // 2 + 1) + 1j * 0  # real spectrum

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def spectral_compress(a, alpha):
    """Apply amplitude compression to DFT and return projected sequence."""

    mag_new = mag ** alpha
    A_new = mag_new * phase
    a_new = np.fft.irfft(A_new, n_fft)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:

    return a_new / s

                                             T_init=0.0005, lr_init=0.01, top_k_frac=0.04)
            try_seq(a_opt, f"perturb_{scale}")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Peak-shaving coordinate descent (350s) ===
    print("\n=== Phase 3: Peak-Shaving CD ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 250
        if remaining > 20:
            t_alloc = min(remaining, 300)
            print(f"  Peak shaving, t={t_alloc:.0f}s")
            a_opt, _ = peak_shave_coordinate_descent(best_seq, time_limit=t_alloc, step_base=0.002)
            try_seq(a_opt, "peak_shave")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Spectral refinement (200s) ===
    print("\n=== Phase 4: Spectral Refinement ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 150
        if remaining > 15:
            t_alloc = min(remaining, 180)
            print(f"  Spectral iteration, t={t_alloc:.0f}s")
            a_opt, _ = spectral_iteration(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "spectral")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final polish (remaining) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Alternate short optimization rounds
        for ri in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            if ri % 3 == 0:
                # Short subgradient
                noise = rng.standard_normal(n_best) * 0.001
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 30)
                a_opt, _ = subgradient_optimize(start, time_limit=t_alloc,
                                                T_init=0.0002, lr_init=0.005, top_k_frac=0.02)
                try_seq(a_opt, f"polish_sg_{ri}")
            elif ri % 3 == 1:
                # Peak shaving
                t_alloc = min(remaining, 30)
                a_opt, _ = peak_shave_coordinate_descent(best_seq, time_limit=t_alloc, step_base=0.0005)
                try_seq(a_opt, f"polish_ps_{ri}")
            else:
                # Spectral
                t_alloc = min(remaining, 20)
                a_opt, _ = spectral_iteration(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"polish_spec_{ri}")

        # Try nearby n values
        for dn in [-3, -2, -1, 1, 2, 3, -5, 5, -10, 10]:
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            n_try = n_best + dn
            if n_try < 128 or n_try > 2048:
                continue
            a_resampled = resample_sequence(best_seq, n_try)
            t_alloc = min(remaining, 15)
            a_opt, _ = subgradient_optimize(a_resampled, time_limit=t_alloc,
                                            T_init=0.0003, lr_init=0.008, top_k_frac=0.03)
            try_seq(a_opt, f"n_adj_{dn}")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_np = np.asarray(best_seq, dtype=np.float64).tolist()
    fs = float(evaluate_sequence(final_np))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return final_np
