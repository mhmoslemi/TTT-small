def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def convolve_subgrad(a, k):
    """Subgradient of (a*a)[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    # (a*a)[k] = sum_j a[j]*a[k-j]
    # d/d[a[i]] = a[k-i] + a[k-i] = 2*a[k-i] (if k-i in [0,n-1])
    # But when k-i == i, it's 2*a[i] (same formula works)
    # Actually: if k-i is in [0, n-1], derivative contribution is a[k-i]
    # from the j=i term, plus a[k-i] from the j=k-i term (if k-i != i)
    # When k-i == i: only one term a[i]^2, derivative is 2*a[i]
    # So: if k-i in [0,n-1] and k-i != i: 2*a[k-i]
    #     if k-i == i: 2*a[i] = 2*a[k-i] (same!)
    #     if k-i not in [0,n-1]: 0 (from j=i term) + possibly a[i] if i=k-i... 
    #     Wait, if k-i not in [0,n-1], then the term a[j]*a[k-j] with j=i is a[i]*a[k-i] = 0
    #     and the term with k-j=i means j=k-i, but k-i not in [0,n-1] so that term doesn't exist.
    # So: grad[i] = 2*a[k-i] if 0 <= k-i <= n-1, else 0.
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    for i in range(lo, hi + 1):
        g[i] = 2.0 * a[k - i]
    return g

def smoothed_objective(a, T):
    """Smoothed max of convolution using softmax."""
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()
    # Smoothed max = T * sum(p * conv) ... actually T * log(sum(exp(conv/T)))
    # But for gradient, we use the weighted sum
    smax = np.dot(p, conv)
    return smax, p

def kkt_refine(a0, time_limit):

    a = np.maximum(a0.astype(np.float64), 0.0)

    n = len(a)
    
    best_a = a.copy()

        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        max_val = conv.max()
        
        # Find peaks close to max (within 1% relative)
        threshold = max_val * 0.99
        active_peaks = np.where(conv >= threshold)[0]
        
        if len(active_peaks) == 0:
            break
        
        # For each active peak, compute the subgradient
        # We want to find a direction d such that:
        #   (a+ed)*(a+ed)[k] decreases for active peaks
        #   sum(d) = 0 (stay on simplex)
        #   a + e*d >= 0
        
        # Use the subgradient of the max at active peaks
        g = np.zeros(n)
        for k in active_peaks:
            g += convolve_subgrad(a, k)
        g /= len(active_peaks)
        
        # Project gradient onto simplex tangent: g_t = g - mean(g)
        g_t = g - g.mean()
        
        # Take a small step
        for step in [1e-4, 5e-5, 1e-5, 5e-6]:

            a_trial = np.maximum(a_trial, 0)

                a = a_trial
                best_obj = trial_obj
                best_a = a.copy()
                break
        
        # If no improvement, try a different threshold
        if it % 100 == 0:

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def convolve_subgrad(a, k):
    """Subgradient of (a*a)[k] w.r.t. a. Returns 2*a[k-i] for valid i."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    if lo <= hi:
        g[lo:hi + 1] = 2.0 * a[k - np.arange(lo, hi + 1)]
    return g


def convolve_subgrad_multi(a, peaks, weights):
    """Weighted sum of subgradients for multiple peaks."""
    n = len(a)
    g = np.zeros(n)
    for i, k in enumerate(peaks):
        if weights[i] < 1e-15:
            continue
        g += weights[i] * convolve_subgrad(a, k)
    return g


def softmax_smoothed_max(conv, T):
    """Compute softmax weights and smoothed max of convolution."""
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()
    smax = np.dot(p, conv)
    return smax, p


def exponentiated_gradient(a0, time_limit, T_init=0.001, eta_init=0.05):

    a = np.maximum(a0.astype(np.float64), 0.0)

    n = len(a)

    eta = eta_init

    best_obj = fast_convolve(a).max()

    no_improve = 0
    it = 0
    eta_history = []

    while time.monotonic() - t0 < time_limit:
        it += 1

            no_improve += 1
        else:

def peak_equalization(a0, time_limit, n_target_peaks=8):
    """Equalize top peaks using weighted subgradient descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    step = 1e-4

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        max_val = conv.max()

        # Find peaks to equalize
        sorted_idx = np.argsort(conv)[::-1]
        top_peaks = sorted_idx[:n_target_peaks]
        top_vals = conv[top_peaks]

        # Weights: emphasize peaks that are highest
        mean_top = top_vals.mean()
        spread = top_vals.max() - top_vals.min()
        if spread < 1e-12:
            # All equal, use uniform weights
            weights = np.ones(len(top_peaks)) / len(top_peaks)
        else:
            # Weight by deviation from mean (positive part)
            raw_w = np.maximum(top_vals - mean_top, 0)
            if raw_w.sum() < 1e-15:
                raw_w = np.ones(len(top_peaks))
            weights = raw_w / raw_w.sum()

        # Also include the absolute highest with extra weight
        g = np.zeros(n)
        for i, k in enumerate(top_peaks):
            w = 1.0 + 2.0 * weights[i]
            g += w * convolve_subgrad(a, k)

        # Project to simplex tangent
        g_t = g - g.mean()

        # Try decreasing steps
        improved = False
        for trial_step in [step, step * 0.5, step * 0.1, step * 0.05]:
            a_trial = a - trial_step * g_t
            a_trial = np.maximum(a_trial, 0.0)
            a_trial_sum = a_trial.sum()
            if a_trial_sum < 1e-15:
                continue
            a_trial = a_trial / a_trial_sum
            trial_obj = fast_convolve(a_trial).max()
            if trial_obj < best_obj - 1e-14:
                a = a_trial
                best_obj = trial_obj
                best_a = a.copy()
                improved = True
                break

        if improved:
            step = min(step * 1.1, 5e-3)
        else:
            step *= 0.7
            if step < 1e-8:
                # Perturb and restart
                rng = np.random.default_rng(it * 31 + 7)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
                step = 1e-4

    return best_a, best_obj


def newton_active_peaks(a0, time_limit, n_peaks=5):
    """Newton refinement: solve (a*a)[k_j] = t for active peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()

    it = 0

        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        max_val = conv.max()
        sorted_idx = np.argsort(conv)[::-1]
        active = sorted_idx[:n_peaks]
        active_vals = conv[active]

        # Check if peaks are well-balanced
        if active_vals.max() - active_vals.min() < 1e-10 * max_val:
            # Try to reduce all simultaneously
            g = np.zeros(n)
            for k in active:
                g += convolve_subgrad(a, k)
            g /= len(active)
            g_t = g - g.mean()
            for step in [1e-6, 5e-7, 1e-7, 5e-8]:
                a_trial = np.maximum(a - step * g_t, 0)
                a_trial = a_trial / a_trial.sum()
                obj = fast_convolve(a_trial).max()
                if obj < best_obj - 1e-14:
                    a = a_trial
                    best_obj = obj
                    best_a = a.copy()
                    break
            continue

        # Newton step: we want to make active_vals all equal
        # Target: reduce the spread
        target = active_vals.mean()
        residuals = active_vals - target

        # Compute Jacobian-like matrix
        # J[i, j] = d(conv[active[i]])/d(a[j])
        # This is n_peaks x n, which is large. Use Gauss-Newton with conjugate gradient.
        # Simplification: use the subgradients directly

        g = np.zeros(n)
        for i, k in enumerate(active):
            # Weight by residual: peaks above target get pushed down more
            w = residuals[i]
            if w > 0:
                g += w * convolve_subgrad(a, k)

        if np.max(np.abs(g)) < 1e-15:
            break

        g_t = g - g.mean()

        # Line search
        improved = False
        for step in [1e-5, 5e-6, 1e-6, 5e-7, 1e-7, 5e-8]:
            a_trial = np.maximum(a - step * g_t, 0)
            a_trial_sum = a_trial.sum()
            if a_trial_sum < 1e-15:
                continue
            a_trial = a_trial / a_trial_sum
            trial_obj = fast_convolve(a_trial).max()
            if trial_obj < best_obj - 1e-14:
                a = a_trial
                best_obj = trial_obj
                best_a = a.copy()
                improved = True
                break

        if not improved:
            # Increase n_peaks slightly and retry
            n_peaks = min(n_peaks + 2, 20)
            # Or perturb
            if it % 50 == 0:
                rng = np.random.default_rng(it * 97 + 3)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0005)
                n_peaks = 5

    return best_a, best_obj


def two_level_search(a0, time_limit):
    """Search over two-level sequences: a[i] = L or H, optimize ratio and pattern."""
    n = len(a0)
    best_a = a0.astype(np.float64).copy()
    best_a = np.maximum(best_a, 0)
    best_a = best_a / best_a.sum()
    best_score = 2.0 * n * fast_convolve(best_a).max()

    t0 = time.monotonic()

    # Initialize binary pattern from a0
    threshold = np.median(a0)
    b = (a0 >= threshold).astype(np.float64)
    w = int(b.sum())

    # Optimize ratio r = L/H for this pattern
    def two_level_obj(r, b, n, w):
        """Compute max convolution for two-level sequence with ratio r."""
        H = 1.0 / (n - w + r * w)
        L = r * H
        a = L + (H - L) * b
        conv = fast_convolve(a)
        return conv.max() * (n - w + r * w) ** 2  # Scale back to unnormalized

    # Actually, let's just work with the normalized sequence directly
    def two_level_score(r, b, n, w):
        H = 1.0 / (n - w + r * w)
        L = r * H
        a = L + (H - L) * b
        conv = fast_convolve(a)
        return 2.0 * n * conv.max()

    # Optimize r
    def neg_score(r):
        return -two_level_score(np.exp(r), b, n, w)

    res = minimize(neg_score, x0=np.log(0.5), method='Nelder-Mead',
                   options={'maxiter': 2000, 'xatol': 1e-10, 'fatol': 1e-12})
    r_opt = np.exp(res.x[0])
    H = 1.0 / (n - w + r_opt * w)
    L = r_opt * H
    a_opt = L + (H - L) * b
    a_opt = a_opt / a_opt.sum()
    score = evaluate_sequence(a_opt.tolist())
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_a = a_opt.copy()
        print("    Two-level init: score={:.8f}, r={:.4f}, w={}".format(score, r_opt, w))

    # Local search: flip bits and re-optimize r
    it = 0
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        if no_improve > 50:
            break

        # Pick a random position to flip
        rng = np.random.default_rng(it * 12345 + 7)
        pos = rng.integers(0, n)
        old_val = b[pos]
        b[pos] = 1.0 - old_val
        w_new = int(b.sum())

        if w_new == 0 or w_new == n:
            b[pos] = old_val
            continue

        # Quick re-optimization of r
        res = minimize(neg_score, x0=np.log(r_opt), method='Nelder-Mead',
                       options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-10})
        r_new = np.exp(res.x[0])
        H_new = 1.0 / (n - w_new + r_new * w_new)
        L_new = r_new * H_new
        a_new = L_new + (H_new - L_new) * b
        a_new = a_new / a_new.sum()

        new_score = evaluate_sequence(a_new.tolist())
        if np.isfinite(new_score) and new_score < best_score - 1e-10:
            best_score = new_score
            best_a = a_new.copy()
            r_opt = r_new
            no_improve = 0
        else:
            b[pos] = old_val
            no_improve += 1

    return best_a, best_score


def gradient_descent_simplex(a0, time_limit, lr_init=0.01, momentum=0.9):
    """Standard gradient descent with simplex projection and momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        # Use top-5 peaks with softmax weighting
        T = 0.0005
        smax, p = softmax_smoothed_max(conv, T)

        g = np.zeros(n)
        active = np.where(p > 1e-8)[0]
        for idx in active:
            g += p[idx] * convolve_subgrad(a, idx)

        v = momentum * v - lr * g
        a_trial = project_simplex(a + v)

        obj = fast_convolve(a_trial).max()
        if obj < best_obj - 1e-14:
            a = a_trial
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.5

        lr *= 0.9995
        momentum = min(0.98, momentum + 0.0002)

        if no_improve > 50:
            rng = np.random.default_rng(it * 41 + 13)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = lr_init * 0.5
            no_improve = 0

    return best_a, best_obj


def try_and_update(a, state, label):
    """Evaluate a sequence and update state if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a)))
