"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Fast mirror descent (multiplicative weights) with FFT-based multi-peak gradients
- Periodic QP active-set refinement
- Accelerated mode with Nesterov momentum
- Multi-n exploration and structured initializations
- Adaptive step sizes with intelligent restarts

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fft_gradient(w, a):
    """Compute subgradient g[i] = sum_j a[j]*w[j+i] using FFT. Returns 2*g."""
    n = len(a)
    w_rev = w[::-1]
    conv = np.convolve(a, w_rev, mode='full')
    return 2.0 * conv[n - 1:2 * n - 1][::-1]

def md_optimize(a0, time_limit, eta_init=5.0, T_frac=0.03, accelerated=False,
                momentum_init=0.0, verbose=False):
    """Mirror descent (multiplicative weights) with multi-peak gradient."""
    a = np.maximum(a0.astype(np.float64), 1e-30)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()

    eta = eta_init
    mom = momentum_init
    a_prev = a.copy() if accelerated else None

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit - 0.05:
        it += 1

        cmax = conv.max()

        # Multi-peak gradient via FFT
        T = max(cmax * T_frac, 1e-15)
        weights = np.exp((conv - cmax) / T)
        wsum = weights.sum()
        if wsum < 1e-30:
            k_peak = int(np.argmax(conv))
            g = np.zeros(n)
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            g[idx] = 2.0 * a[k_peak - idx]
        else:
            weights = weights / wsum
            g = fft_gradient(weights, a)

        # Multiplicative update
        if accelerated:
            # Nesterov: use gradient at extrapolated point
            a_extrap = a + mom * (a - a_prev)
            a_extrap = np.maximum(a_extrap, 1e-30)
            a_extrap = a_extrap / a_extrap.sum()
            # Recompute gradient at extrapolated point (approximation: reuse g)
            scaled = eta * g
            scaled = np.clip(scaled, -100, 100)
            a_new = a_extrap * np.exp(-scaled)
        else:
            scaled = eta * g
            scaled = np.clip(scaled, -100, 100)
            a_new = a * np.exp(-scaled)

        s_new = a_new.sum()
        if s_new < 1e-30:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        if accelerated:
            a_prev = a
        a = a_new
        conv = fast_convolve(a)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive step size
        if no_improve == 50:
            eta *= 0.85
        elif no_improve == 200:
            eta *= 1.2
        elif no_improve == 500:
            eta *= 0.7
            if accelerated:
                mom = max(0.0, mom - 0.1)
        elif no_improve == 1000:
            # Restart with perturbation
            rng = np.random.default_rng((it * 7919 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(a)
            eta = eta_init
            if accelerated:
                mom = momentum_init
            a_prev = a.copy()
            no_improve = 0
        elif no_improve > 3000:
            break

        if accelerated and it % 1000 == 0:
            mom = min(0.9, mom + 0.01)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it}, obj={best_obj:.10f}, eta={eta:.3f}, "
                  f"no_imp={no_improve}, t={elapsed:.1f}s, rate={it/elapsed:.0f} it/s")

    return best_a, best_obj

def qp_active_set(w, a_init, n, max_iters=80, time_limit=3.0):
    """Solve min a^T W a s.t. sum(a)=1, a>=0 using active set. W is Hankel from w."""
    t0 = time.monotonic()
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    active = set(np.where(a > 1e-10)[0])
    if not active:
        active = set(range(min(n, 100)))

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if len(active) == 0:
            return np.ones(n) / n
        if len(active) == 1:
            a = np.zeros(n)
            a[list(active)[0]] = 1.0
            return a

        act = sorted(active)
        k = len(act)

        # Build submatrix
        W_AA = np.array([[w[act[i] + act[j]] for j in range(k)] for i in range(k)])
        W_AA_reg = W_AA + 1e-12 * np.eye(k)

        try:
            x = np.linalg.solve(W_AA_reg, np.ones(k))
        except np.linalg.LinAlgError:
            try:
                x = np.linalg.lstsq(W_AA_reg, np.ones(k), rcond=None)[0]
            except Exception:
                x = np.ones(k) / k

        x_sum = x.sum()
        if abs(x_sum) < 1e-14:
            return a

        a_act = x / x_sum

        # Remove most negative
        min_val = float('inf')
        min_idx = -1
        for i, idx in enumerate(act):
            if a_act[i] < min_val:
                min_val = a_act[i]
                min_idx = idx

        if min_val < -1e-8:
            active.discard(min_idx)
            continue

        # Form full solution
        a_new = np.zeros(n)
        for i, idx in enumerate(act):
            a_new[idx] = max(0.0, a_act[i])
        s2 = a_new.sum()
        if s2 > 1e-15:
            a_new = a_new / s2

        # Compute gradient
        Wa = np.zeros(n)
        for i in range(n):
            for j in act:
                Wa[i] += a_new[j] * w[i + j]
        grad = 2.0 * Wa

        # KKT check
        mu = np.mean([grad[idx] for idx in act])
        inactive = list(set(range(n)) - active)
        if inactive:
            min_viol = 0.0
            min_j = -1
            for j in inactive:
                viol = grad[j] - mu
                if viol < min_viol:
                    min_viol = viol
                    min_j = j
            if min_viol < -1e-6:
                active.add(min_j)
                a = a_new
                continue

        a = a_new
        return a

    return a

def qp_primal_dual_cycle(a0, time_limit, T_init=0.01, n_cycles=10, verbose=False):
    """Primal-dual: alternate between QP solve and dual update."""
    a = np.maximum(a0.astype(np.float64), 1e-30)
    a = a / a.sum()
    n = len(a)
    T = T_init

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    for cyc in range(n_cycles):
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        conv = fast_convolve(a)
        cmax = conv.max()

        # Dual update
        T_local = max(cmax * T, 1e-15)
        weights = np.exp((conv - cmax) / T_local)
        weights = weights / weights.sum()

        # Primal: solve QP
        qp_time = min(2.0, remaining * 0.5)
        a_qp = qp_active_set(weights, a, n, max_iters=40, time_limit=qp_time)

        # Blend
        alpha = 0.5
        a_new = (1.0 - alpha) * a + alpha * a_qp
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a_qp
        else:
            a_new = a_new / s_new

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()

        a = a_new

        # Adapt temperature
        if (cyc + 1) % 3 == 0:
            T *= 1.3
        if (cyc + 1) % 7 == 0:
            T *= 0.6

        if verbose:
            print(f"    QP-PD cycle {cyc}: obj={best_obj:.10f}, T={T:.6f}")

    return best_a, best_obj

def structured_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine window (raised cosine)
    x = np.arange(n, dtype=np.float64)
    c = 0.5 + 0.5 * np.cos(2 * np.pi * x / (n - 1))
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    # Two peaks (quarter points)
    g2 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())

    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge / edge.sum())

    # Chirp-based (modulated)
    for freq in [1, 2, 3]:
        chirp = 1.0 + 0.5 * np.cos(2 * np.pi * freq * x / n)
        chirp = np.maximum(chirp, 0.01)
        inits.append(chirp / chirp.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    # Parent interpolated
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())

    return inits

        cmax = conv.max()

        # Focus on top-3 peaks with equal weight
        top_k = np.argsort(conv)[::-1][:min(3, len(conv))]
        weights = np.zeros(len(conv))
        weights[top_k] = 1.0 / len(top_k)

        g = fft_gradient(weights, a)

        # Larger step for top peaks
        scaled = eta * g
        scaled = np.clip(scaled, -100, 100)
        a_new = a * np.exp(-scaled)
        s_new = a_new.sum()
        if s_new < 1e-30:
            a_new = a
        else:
            a_new = a_new / s_new

        a = a_new
        conv = fast_convolve(a)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve == 100:
            eta *= 0.8
        elif no_improve == 400:
            eta *= 1.5
        elif no_improve == 800:
            rng = np.random.default_rng(it * 31 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            eta = 3.0
            no_improve = 0
        elif no_improve > 2000:
            break

        if verbose and it % 500 == 0:
            print(f"    CD it={it}, obj={best_obj:.10f}")

    return best_a, best_obj

def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search for auto-correlation minimization.
    Uses fast mirror descent as the workhorse with periodic QP refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")
    print(f"Budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")

    # === Phase 1: Quick n-exploration with MD ===
    print("\n=== Phase 1: Quick n-exploration ===")
    t1 = time.monotonic()

    candidate_ns = sorted(set([
        n_parent,
        max(256, n_parent - 128),
        n_parent + 128,
        max(256, n_parent // 2),
        min(2048, n_parent * 2),
        512, 768, 1024,
    ]))

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 30:
            break

        inits = structured_inits(n_try, parent, rng)

        best_a_n = None
        best_obj_n = float('inf')

        for init_idx, init in enumerate(inits[:4]):  # Limit to 4 inits per n
            alloc = min(remaining / max(1, len(inits[:4])), 25)
            if alloc < 3:
                break
            a_opt, obj = md_optimize(init, time_limit=alloc, eta_init=5.0, T_frac=0.03)
            if obj < best_obj_n:
                best_obj_n = obj
                best_a_n = a_opt

        if best_a_n is not None:
            try_seq(best_a_n, f"md_n{n_try}")

        # Quick QP-PD cycle on best
        remaining = deadline - time.monotonic() - 1100
        if remaining > 15 and best_a_n is not None:
            a_opt2, _ = qp_primal_dual_cycle(best_a_n, time_limit=min(remaining, 30),
                                              T_init=0.01, n_cycles=5)
            try_seq(a_opt2, f"qp_n{n_try}")

    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.1f}s)")

    # === Phase 2: Deep MD on best sequence ===
    print("\n=== Phase 2: Deep Mirror Descent ===")
    t2 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Long MD run with different settings
        configs = [
            {'eta_init': 5.0, 'T_frac': 0.03, 'accelerated': False, 't_frac': 0.3},
            {'eta_init': 3.0, 'T_frac': 0.01, 'accelerated': True, 'momentum_init': 0.3, 't_frac': 0.25},
            {'eta_init': 8.0, 'T_frac': 0.05, 'accelerated': False, 't_frac': 0.2},
            {'eta_init': 2.0, 'T_frac': 0.02, 'accelerated': True, 'momentum_init': 0.5, 't_frac': 0.25},
        ]

        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 500
            if remaining < 20:
                break
            t_alloc = min(remaining, int(budget_s * cfg['t_frac']))
            print(f"  MD config {ci} (eta={cfg['eta_init']}, T={cfg['T_frac']}, "
                  f"acc={cfg['accelerated']}), t={t_alloc:.0f}s")
            a_opt, _ = md_optimize(best_seq, time_limit=t_alloc,
                                   verbose=(ci == 0))
            try_seq(a_opt, f"md_deep_{ci}")

    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.1f}s)")

    # === Phase 3: QP refinement ===
    print("\n=== Phase 3: QP Primal-Dual Refinement ===")
    t3 = time.monotonic()

    if best_seq is not None:
        # Multiple QP-PD cycles with different temperatures
        for T_init in [0.005, 0.01, 0.02, 0.05]:
            remaining = deadline - time.monotonic() - 300
            if remaining < 20:
                break
            t_alloc = min(remaining, 120)
            print(f"  QP-PD T={T_init}, t={t_alloc:.0f}s")
            a_opt, _ = qp_primal_dual_cycle(best_seq, time_limit=t_alloc,
                                            T_init=T_init, n_cycles=15)
            try_seq(a_opt, f"qp_T{T_init}")

        # Coherence descent variant
        remaining = deadline - time.monotonic() - 200
        if remaining > 20:
            t_alloc = min(remaining, 80)
            print(f"  Coherence descent, t={t_alloc:.0f}s")
            a_opt, _ = coherence_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coherence")

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.1f}s)")

    # === Phase 4: Fine-tuning with very low temperature MD ===
    print("\n=== Phase 4: Fine-tuning ===")
    t4 = time.monotonic()

    if best_seq is not None:
        # Very focused MD (low T = sharp peak focus)
        for T_frac in [0.005, 0.01, 0.001]:
            remaining = deadline - time.monotonic() - 100
            if remaining < 15:
                break
            t_alloc = min(remaining, 60)
            print(f"  Fine MD T_frac={T_frac}, t={t_alloc:.0f}s")
            a_opt, _ = md_optimize(best_seq, time_limit=t_alloc,
                                   eta_init=2.0, T_frac=T_frac)
            try_seq(a_opt, f"fine_T{T_frac}")

        # Accelerated with high momentum
        remaining = deadline - time.monotonic() - 60
        if remaining > 15:
            t_alloc = min(remaining, 50)
            print(f"  Accelerated MD mom=0.7, t={t_alloc:.0f}s")
            a_opt, _ = md_optimize(best_seq, time_limit=t_alloc,
                                   eta_init=3.0, T_frac=0.02,
                                   accelerated=True, momentum_init=0.7)
            try_seq(a_opt, "accel_mom07")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.1f}s)")

    # === Phase 5: Final perturbations and verification ===
    print("\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Tiny perturbations with quick refinement
        for pert in range(8):
            remaining = deadline - time.monotonic() - 10
            if remaining < 5:
                break
            noise_scale = 0.001 + 0.0005 * pert
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            t_alloc = min(remaining, 20)
            a_opt, _ = md_optimize(f_pert, time_limit=t_alloc,
                                   eta_init=3.0, T_frac=0.02)
            try_seq(a_opt, f"perturb_{pert}")

        # One final QP cycle
        remaining = deadline - time.monotonic() - 15
        if remaining > 10:
            t_alloc = min(remaining, 30)
            a_opt, _ = qp_primal_dual_cycle(best_seq, time_limit=t_alloc,
                                            T_init=0.005, n_cycles=5)
            try_seq(a_opt, "final_qp")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.1f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    print(f"Total time used: {time.monotonic() - (deadline - budget_s + 30):.1f}s")
    return [float(x) for x in best_seq]
