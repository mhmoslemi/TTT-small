What distribution makes the sum of two i.i.d. draws uniform? This is related to "sum-free" distributions or "difference sets" in combinatorics.

In finite group theory, a (v,k,lambda) difference set D in a group G of order v is a subset of size k such that every non-identity element is expressed as a difference d1-d2 in exactly lambda ways. For the cyclic group Z_n, if D is a difference set, then the "auto-correlation" of the indicator of D is constant (away from 0).

Hybrid optimization for minimum peak autocorrelation step function.
Uses improved Frank-Wolfe with LP oracle, accelerated mirror descent,
multi-scale n exploration, and difference-set inspired initialization.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def eval_objective(a):
    """Compute 2*n*max_conv/sum^2 (the actual score)."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * len(a) * fast_convolve(a).max() / (s * s)

def build_convolution_matrix_sparse(n):
    """Build the sparse structure for the convolution constraints."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows), np.array(cols)

def solve_lp_direction(a, M, n, row_idx, col_idx):
    """
    LP oracle: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns normalized g or None.
    """
    c = -np.ones(n)
    # A_ub[row, col] = a[col + row_offset] ... 
    # For constraint k: sum_i a[k-i] * g[i] <= M
    # So A_ub[k, i] = a[k-i] (when valid)
    vals = np.empty(len(row_idx))
    for idx in range(len(row_idx)):
        k = row_idx[idx]
        i = col_idx[idx]
        vals[idx] = a[k - i]
    
    from scipy.sparse import lil_matrix
    A_mat = lil_matrix((2 * n - 1, n))
    A_mat[row_idx, col_idx] = vals
    A_mat = A_mat.tocsr()
    
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.fun is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None

def gradient_descent_smoothed(a0, time_limit, T0=0.001, lr0=0.02):
    """Smoothed gradient descent with better adaptive schedule."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    T = T0
    lr = lr0
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Compute gradient: grad[j] = 2 * sum_k p[k] * a[k-j]
        # Using the convolution trick: grad/2 = convolve(p, a_rev)[j+n-1]
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 2 - 1).bit_length()
        P = np.fft.rfft(p, fft_size)
        A_rev = np.fft.rfft(a_rev, fft_size)
        corr = np.fft.irfft(P * A_rev, fft_size)
        grad = 2.0 * corr[n-1 : n-1+n]
        
        # Gradient step with projection
        best_candidate = a
        best_val = conv.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.5]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            if c.max() < best_val - 1e-15:
                best_val = c.max()
                best_candidate = candidate
        
        a = best_candidate
        obj = 2.0 * n * best_val
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 15:
            T *= 1.02
            lr *= 0.995
        if no_improve > 80:
            T *= 0.85
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            T = T0
            lr = lr0
            no_improve = 0
    
    return best_a, best_obj
