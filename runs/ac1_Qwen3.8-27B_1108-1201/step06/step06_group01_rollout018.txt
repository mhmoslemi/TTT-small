def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_oracle(f, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def peak_surgery(a0, time_limit):
    """
    Targeted peak reduction via exact 1D optimization.
    For each top peak, find the direction that most reduces it,
    then do exact line search along that direction.

    a = a / a.sum()
    n = len(a)

    t0 = time.monotonic()
    it = 0
    stall = 0

        it += 1
        if time.monotonic() - t0 > time_limit - 0.2:
            break

        conv = fast_convolve(a)
        # Get top 3 peaks
        top_k = min(3, len(conv))
        peak_lags = np.argsort(conv)[-top_k:]
        peak_lags = peak_lags[np.argsort(conv[peak_lags])[::-1]]

        improved = False

        for k_peak in peak_lags:
            if time.monotonic() - t0 > time_limit - 1.0:
                break

            peak_val = conv[int(k_peak)]
            # Direction that reduces this peak:
            # grad_j = 2 * a[k-j] for j where 0 <= k-j < n
            # We want to move in direction -grad (to reduce the peak)
            # But we need to maintain sum=1 and a>=0
            # So: direction d[j] = -a[k-j] (for valid j), then project to simplex difference

            je = min(n - 1, k_peak)
            valid_j = np.arange(js, je + 1)

            grad_peak[valid_j] = 2.0 * a[partners]

def sdp_relaxation(n, time_limit=60):
    """
    Solve SDP relaxation for the flat convolution problem.
    minimize t
    s.t. sum_{i+j=k} X[i,j] <= t for all k
         sum(X) = 1
         X >= 0 (PSD)
         X[i,j] >= 0 (element-wise)
    Returns optimal t and the optimal X (or None if failed).
    """
    try:
        X = cp.Variable((n, n), symmetric=True)
        t = cp.Variable()

        constraints = [X >> 0]  # PSD

        # Peak constraints
        for k in range(2 * n - 1):
            idx_i = np.arange(max(0, k - n + 1), min(n, k + 1))
            idx_j = k - idx_i
            peak_sum = cp.sum(cp.multiply(X[idx_i, idx_j], 1))
            # Actually we need sum of specific entries
            # Let's use a different approach
            peak_expr = 0
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                j = k - i
                peak_expr = peak_expr + X[i, j]
            constraints.append(peak_expr <= t)

        # Sum constraint: sum of all entries = 1
        constraints.append(cp.sum(X) == 1)

        # Element-wise non-negativity
        constraints.append(X >= 0)

        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)

        # Try CLARAB or SCS
        try:
            prob.solve(solver=cp.SCS, max_iter=5000, eps=1e-6)
        except Exception:
            prob.solve(solver=cp.SCS, max_iter=3000)

        if prob.status in ['optimal', 'optimal_inaccurate']:
            t_val = float(t.value)
            X_val = np.array(X.value)
            X_val = (X_val + X_val.T) / 2.0  # symmetrize
            X_val = np.maximum(X_val, 0)
            return t_val, X_val
    except Exception as e:
        print(f"  SDP failed: {e}")
    return None, None


def sdp_to_vector(X, n):
    """Extract a non-negative vector from SDP solution X (rank-1 approximation)."""
    # Use top eigenvector
    eigenvalues, eigenvectors = np.linalg.eigh(X)
    # Top eigenvector (largest eigenvalue)
    idx = np.argmax(eigenvalues)
    v = eigenvectors[:, idx]
    # Make non-negative
    v = np.abs(v)
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return v / s

    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    scales = [0.0001, 0.0002, 0.0005, 0.001, 0.002]
    si = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        scale = scales[si % len(scales)]
        rng = np.random.default_rng((it * 2654435761 + 1) % (2**32))

        # Try several perturbations
        for _ in range(20):
            if time.monotonic() - t0 > time_limit - 0.1:
                break

            noise = rng.standard_normal(n) * scale
            # Bias toward reducing peaks
            conv = fast_convolve(a)
            peak_k = int(np.argmax(conv))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            for j in range(js, je + 1):
                partner = peak_k - j
                if 0 <= partner < n:
                    noise[j] -= 1.5 * scale * a[partner]

            candidate = project_simplex(a + noise)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                break

        if it % 50 == 0:
            si += 1

    return best_a, best_obj
