def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)

    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_soft_max_obj_grad(z, beta):
    """Compute soft-max(conv(softmax(z), softmax(z))) and gradient w.r.t. z."""
    z_shift = z - z.max()
    a = np.exp(z_shift)
    a = a / a.sum()
    
    conv = fast_convolve(a)
    
    scaled = conv * beta
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    
    log_sum = np.log(exp_scaled.sum())
    obj = conv.max() + (1.0/beta) * log_sum
    
    grad_a = hankel_mv(p, a)
    dz_grad = a * (grad_a - np.dot(a, grad_a))
    
    return obj, dz_grad

def lbfgs_smooth(a0, time_limit, beta_init=50.0, beta_max=500.0):
    """L-BFGS optimization on softmax-smoothed objective."""
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = len(a)
    
    # Initialize z from a: z = log(a) (plus a constant for numerical stability)
    z = np.log(a + 1e-30)
    z = z - z.max()  # center for numerical stability
    
    beta = beta_init
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        # Define objective and gradient for L-BFGS
        def fun(z_vec):
            obj, _ = compute_soft_max_obj_grad(z_vec, beta)
            return obj
        
        def grad(z_vec):
            _, g = compute_soft_max_obj_grad(z_vec, beta)
            return g
        
        try:
            res = minimize(fun, z, jac=grad, method='L-BFGS-B',
                          options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-12,
                                   'maxfun': 500})
        except Exception:
            break
        
        # Convert back to a
        z_new = res.x
        a_new = np.exp(z_new - z_new.max())
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Increase beta (make approximation sharper)
        if no_improve == 0:
            beta = min(beta_max, beta * 1.2)
        else:
            beta = max(beta_init, beta * 0.9)
        
        z = z_new
        
        if no_improve > 20:
            break
    
    return best_a, best_obj

def active_set_lp(a0, time_limit, tol=0.001, max_iter=500):
    """
    Linearized active-set LP refinement.
    At each step:
    1. Identify active peaks P
    2. Solve LP: min t s.t. 2*sum_j a[j]*d[k-j] <= t for k in P, sum(d)=0, d>=-a
    3. Line search
    """
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Identify active peaks
        threshold = M * (1.0 - tol)
        active = np.where(conv >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(conv)])
        
        # Build LP:
        # Variables: d[0..n-1], t (total n+1 variables)
        # Minimize: t
        # Subject to:
        #   2 * sum_j a[j] * d[(k-j) mod ...] <= t  for each k in active
        #   Actually: conv(a+d, a+d)[k] ≈ 2*sum_j a[j]*d[k-j] + conv(a,a)[k]
        #   So: 2*sum_j a[j]*d[k-j] <= t - conv(a,a)[k]
        #   sum(d) = 0
        #   d >= -a (a+d >= 0)
        
        # LP objective: minimize t (variable index n)
        c = np.zeros(n + 1)
        c[n] = 1.0  # minimize t
        
        # Inequality constraints (A_ub @ x <= b_ub):
        # For each active peak k: 2*sum_j a[j]*d[k-j] - t <= -conv(a,a)[k]
        # i.e., sum_j (2*a[j] if (k-j) is valid) * d[j] + (-1)*t <= -conv[k]
        
        A_ub = np.zeros((len(active), n + 1))
        b_ub = np.zeros(len(active))
        
        for i, k in enumerate(active):
            b_ub[i] = -conv[k]
            # For each j: coefficient of d[j] is 2*a[k-j] if 0 <= k-j < n
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[i, j] = 2.0 * a[k - j]
            A_ub[i, n] = -1.0  # coefficient of t
        
        # Equality constraint: sum(d) = 0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        # Bounds: d[j] >= -a[j], t is unbounded below (but will be >= 0 at optimum)
        bounds = [(-a[j], None) for j in range(n)] + [(None, None)]
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if not res.success:
                break
            d = res.x[:n]
        except Exception:
            break
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            candidate = a + alpha * d
            if np.any(candidate < -1e-12):
                continue
            candidate = np.maximum(candidate, 0.0)
            s = candidate.sum()
            if s < 1e-15:
                continue
            candidate = candidate / s
            val = fast_convolve(candidate).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha < 1e-10:
            # Try with tighter active set
            tol *= 0.5
            if tol < 1e-6:
                break
            continue
        
        a = a + best_alpha * d
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        
        conv = fast_convolve(a)
        M = conv.max()
        
        if M < best_obj - 1e-14:
            best_obj = M
            best_a = a.copy()
        else:
            # No improvement, try tightening
            tol *= 0.7
            if tol < 1e-6:
                break
    
    return best_a, best_obj

def spectral_init(n, seed=42):
    """Initialize using flat-spectrum approach."""
    rng = np.random.default_rng(seed)
    
    # Start with random phase, flat magnitude
    mag = np.ones(n + 1)  # rfft has n//2+1 bins for even n
    phase = rng.standard_normal(n + 1)
    # Make it real: spectrum must be Hermitian
    F = (mag + 1j * phase).astype(np.complex128)
    # Ensure Hermitian: F[k] = conjugate(F[n-k])
    # For rfft, we just need F[0] real
    F[0] = np.abs(F[0])
    
    a = np.fft.irfft(F, n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def spectral_flatten(a, n_iter=50, alpha=0.5):
    """Iteratively flatten the power spectrum while maintaining a >= 0."""
    a = np.maximum(a, 0.0).copy()
    a = a / a.sum()
    n = len(a)
    
    for _ in range(n_iter):
        F = np.fft.rfft(a, 2 * n - 1)
        mag2 = np.abs(F) ** 2
        
        # Flatten: reduce peaks in the power spectrum
        target = mag2.mean()
        mag2_new = mag2 + alpha * (target - mag2)
        mag2_new = np.maximum(mag2_new, 1e-10)
        
        # Keep the phase, update magnitude
        phase = np.angle(F)
        F_new = np.sqrt(mag2_new) * np.exp(1j * phase)
        F_new[0] = np.abs(F_new[0])  # DC component must be real
        
        a = np.fft.irfft(F_new, 2 * n - 1)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
    
    return a

def pgd_smooth(a0, time_limit, beta=100.0, lr=0.01, mom=0.9):
    """SGD with momentum on the smooth approximation."""
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break
        
        conv = fast_convolve(a)
        scaled = conv * beta
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        grad = hankel_mv(p, a)
        
        v = mom * v - lr * grad
        candidate = project_simplex(a + v)
        
        c = fast_convolve(candidate)
        val = c.max()
        
        if val < best_obj - 1e-15:
            best_obj = val
            best_a = candidate.copy()
            a = candidate
            no_improve = 0
        else:
            a = candidate  # still move, but track best separately
            no_improve += 1
            v *= 0.95
        
        if no_improve > 50:
            lr *= 0.9
            if lr < 0.001:
                break
        if no_improve > 200:
            break
    
    return best_a, best_obj

def try_and_update(a, best_state, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_state
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    best_seq, best_score = best_state
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return (best_seq, best_score)

def interp_sequence(parent, n_target):
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s

Hmm, let me reconsider. I think the main issues with the previous approach and my current plan are:

1. The LP in the active-set method might be too slow if there are many active peaks
2. The L-BFGS with softmax parameterization might have numerical issues with high beta
3. The multi-start might not be diverse enough

Let me think about what's really the bottleneck. The objective is `2n * max(conv) / (sum)^2`. With sum=1, it's `2n * max(conv)`. For n=768, the target is 1.5027, so `max(conv) ≈ 1.5027/(2*768) ≈ 0.000979`.

The average value of conv is `1/(2n-1) = 1/1535 ≈ 0.000651`. So the max is about 50% above the average. This means the convolution is already quite flat.

For the active-set LP, the key question is: how many active peaks are there? If the convolution is very flat, there might be many peaks close to the maximum. Let me limit the active set to the top few peaks.

Let me also reconsider the approach. Maybe instead of the LP, I should use a direct gradient method that's more tailored to this problem.

**Alternative approach: "Peak shaving" gradient**

The gradient of max(conv) w.r.t. a is:
- If k* is the unique argmax: grad[j] = 2*a[k*-j] (for valid k*-j)
- If there are multiple argmax at {k1, k2, ...}: grad is in the convex hull of {2*a[ki-j]}

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines: (1) L-BFGS on softmax-smoothed max-conv objective with adaptive temperature,
(2) Linearized active-set LP refinement for KKT snapping,
(3) Spectral domain initialization and iterative flattening,
(4) Multi-start with diverse strategies and length exploration,
(5) Alternating smooth/LP schedule to escape plateaus.

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def hankel_mv(p, a):
    """Compute sum_k p[k]*a[k-j] for each j (gradient of soft-max conv w.r.t. a)."""
    return np.correlate(p, a, mode='valid')


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_soft_max_obj_grad(z, beta):
    """Compute soft-max(conv(softmax(z), softmax(z))) and gradient w.r.t. z."""
    z_shift = z - z.max()
    a = np.exp(z_shift)
    a = a / a.sum()

    conv = fast_convolve(a)

    scaled = conv * beta
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()

    log_sum = np.log(exp_scaled.sum())
    obj = conv.max() + (1.0 / beta) * log_sum

    grad_a = hankel_mv(p, a)
    dz_grad = a * (grad_a - np.dot(a, grad_a))

    return obj, dz_grad


def lbfgs_smooth(a0, time_limit, beta_init=50.0, beta_max=2000.0, max_iter_per_call=100):
    """L-BFGS optimization on softmax-smoothed objective with adaptive beta."""
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z = z - z.max()

    beta = beta_init
    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()

    t0 = time.monotonic()
    no_improve = 0
    total_calls = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        def fun(z_vec):
            obj, _ = compute_soft_max_obj_grad(z_vec, beta)
            return obj

        def grad_fn(z_vec):
            _, g = compute_soft_max_obj_grad(z_vec, beta)
            return g

        try:
            res = minimize(fun, z, jac=grad_fn, method='L-BFGS-B',
                           options={'maxiter': max_iter_per_call, 'ftol': 1e-16,
                                    'gtol': 1e-14, 'maxfun': 300})
        except Exception:
            break

        z_new = res.x
        a_new = np.exp(z_new - z_new.max())
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            z = z_new
            no_improve = 0
            # Increase beta to sharpen
            beta = min(beta_max, beta * 1.3)
        else:
            no_improve += 1
            # Keep z at best, try different beta
            beta = max(beta_init * 0.5, beta * 0.8)
            if no_improve > 10:
                # Restart from best with perturbation
                rng_restart = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.max()
                z = z + rng_restart.standard_normal(n) * 0.01
                beta = beta_init
                no_improve = 0

        total_calls += 1
        if total_calls > 200:
            break

    return best_a, best_obj


def active_set_lp(a0, time_limit, tol=0.002, max_iter=300, max_active=30):
    """
    Linearized active-set LP refinement.
    Solves LP to find descent direction that decreases all active peaks.
    """
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()

    t0 = time.monotonic()
    it = 0
    current_tol = tol

    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Identify active peaks (top peaks within tolerance of max)
        threshold = M * (1.0 - current_tol)
        active = np.where(conv >= threshold)[0]

        # Limit to top max_active peaks for LP efficiency
        if len(active) > max_active:
            active_sorted = active[np.argsort(conv[active])[::-1][:max_active]]
            active = active_sorted
        else:
            active = active

        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Build LP:
        # Variables: d[0..n-1], t  (total n+1)
        # Minimize: t
        # Subject to:
        #   2*sum_j a[j]*d[k-j] - t <= -conv[k]  for k in active
        #   sum(d) = 0
        #   d[j] >= -a[j]
        #   t >= 0 (implicit from constraints)

        c = np.zeros(n + 1)
        c[n] = 1.0

        A_ub = np.zeros((len(active), n + 1))
        b_ub = np.zeros(len(active))

        for i, k in enumerate(active):
            b_ub[i] = -conv[k]
            A_ub[i, n] = -1.0
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                A_ub[i, js:je + 1] = 2.0 * a[k - np.arange(js, je + 1)]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        bounds = [(-a[j], None) for j in range(n)] + [(0, None)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(1, remaining * 0.3)})
            if not res.success or res.x is None:
                current_tol *= 0.5
                if current_tol < 1e-7:
                    break
                continue
            d = res.x[:n]
        except Exception:
            current_tol *= 0.5
            if current_tol < 1e-7:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            candidate = a + alpha * d
            if np.min(candidate) < -1e-10:
                continue
            candidate = np.maximum(candidate, 0.0)
            s = candidate.sum()
            if s < 1e-15:
                continue
            candidate = candidate / s
            val = fast_convolve(candidate).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha < 1e-10:
            current_tol *= 0.5
            if current_tol < 1e-7:
                break
            continue

        a = a + best_alpha * d
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        conv = fast_convolve(a)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_a = a.copy()
            # Tighten tolerance as we make progress
            current_tol = max(1e-6, current_tol * 0.9)
        else:
            current_tol = max(current_tol * 0.7, 1e-6)

    return best_a, best_obj


def spectral_flatten(a, n_iter=50, alpha=0.4):
    """Iteratively flatten the power spectrum while maintaining a >= 0."""
    a = np.maximum(a, 0.0).copy()
    a = a / a.sum()
    n = len(a)

    for i in range(n_iter):
        F = np.fft.rfft(a, 2 * n - 1)
        mag2 = np.abs(F) ** 2

        # Flatten: pull magnitudes toward the mean
        target = mag2[1:].mean() if len(mag2) > 1 else mag2[0]
        mag2[0] = mag2[0]  # Keep DC
        mag2[1:] = mag2[1:] + alpha * (target - mag2[1:])
        mag2 = np.maximum(mag2, 1e-20)

        phase = np.angle(F)
        F_new = np.sqrt(mag2) * np.exp(1j * phase)
        F_new[0] = np.abs(F_new[0])

        a = np.fft.irfft(F_new, 2 * n - 1)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

    return a


def pgd_adaptive(a0, time_limit, beta_init=100.0, lr_init=0.005, mom_init=0.92):
    """Adaptive PGD with softmax-smoothed gradient and backtracking."""
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    v = np.zeros(n)
    beta = beta_init
    lr = lr_init
    mom = mom_init
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time.monotonic() and (time_limit - (time.monotonic() - t0))
        if remaining < 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv * beta
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = hankel_mv(p, a)

        v = mom * v - lr * grad

        # Backtracking line search
        best_candidate = a
        best_val = conv.max()
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            candidate = project_simplex(a + scale * v)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
            beta = min(5000.0, beta * 1.1)
        else:
            no_improve += 1
            v *= 0.9
            beta = max(beta_init, beta * 0.95)

        if no_improve > 30:
            lr *= 0.9
            mom = min(0.97, mom + 0.001)
        if no_improve > 100:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
            noise = 0.002 + 0.003 * min(1.0, (no_improve - 100) / 100.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init
            mom = mom_init
            beta = beta_init
            no_improve = 0

    return best_a, best_obj


def try_and_update(a, best_state, label=""):
    """Evaluate sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_state
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    best_seq, best_score = best_state
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return (best_seq, best_score)


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def alternating_refine(a0, time_limit, lbfgs_frac=0.5, aslp_frac=0.3, pgd_frac=0.2):
    """Alternate between L-BFGS, active-set LP, and PGD for robust refinement."""
    a = np.maximum(a0, 0.0)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    cycle = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3.0:
            break

        cycle += 1
        t_lbfgs = min(remaining * 0.4, 30.0)
        t_aslp = min(remaining * 0.3, 20.0)
        t_pgd = min(remaining * 0.2, 15.0)

        # L-BFGS phase
        if t_lbfgs > 1.0:
            a_opt, obj = lbfgs_smooth(a, time_limit=t_lbfgs,
                                      beta_init=50.0 + cycle * 20,
                                      beta_max=min(2000.0, 200.0 + cycle * 100))
            if obj < best_obj:
                best_obj = obj
                best_a = a_opt.copy()
            a = a_opt

        # Active-set LP phase

            a_opt, obj = active_set_lp(a, time_limit=t_aslp, tol=max(0.001, 0.005 / cycle))
            if obj < best_obj:

                best_obj = obj
                best_a = a_opt.copy()
            a = a_opt

    print("\n=== Phase 4: Intensive Refinement ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    # High-beta L-BFGS
    remaining = deadline - time.monotonic() - 250
    if remaining > 15:
        t_alloc = min(remaining, 100)
        a_opt, _ = lbfgs_smooth(best_seq, time_limit=t_alloc, beta_init=200, beta_max=2000)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "lbfgs_deep")

    # Tight active-set LP
    remaining = deadline - time.monotonic() - 200
    if remaining > 10:
        t_alloc = min(remaining, 80)
        a_opt, _ = active_set_lp(best_seq, time_limit=t_alloc, tol=0.001, max_active=20)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "aslp_tight")

    # PGD with high beta
    remaining = deadline - time.monotonic() - 150
    if remaining > 10:
        t_alloc = min(remaining, 60)
        a_opt, _ = pgd_adaptive(best_seq, time_limit=t_alloc, beta_init=500, lr_init=0.002, mom_init=0.95)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "pgd_deep")

    # Another alternating cycle
    remaining = deadline - time.monotonic() - 100
    if remaining > 15:
        t_alloc = min(remaining, 80)
        a_opt, _ = alternating_refine(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "alt_deep")

    print(f"  Phase 4: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Multi-start diversification (~250s) ===
    print("\n=== Phase 5: Multi-Start ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    for attempt in range(12):
        remaining = deadline - time.monotonic() - 120
        if remaining < 15:
            break

        # Diverse starting strategies
        strategy = attempt % 4
        if strategy == 0:
            # Gaussian perturbation
            noise = rng.standard_normal(n_best) * (0.003 + 0.002 * attempt)
            start = project_simplex(best_seq + noise)
        elif strategy == 1:
            # Spectral perturbation
            start = spectral_flatten(best_seq, n_iter=10 + attempt * 5, alpha=0.2 + 0.1 * (attempt % 5))
        elif strategy == 2:
            # Support modification
            start = best_seq.copy()
            n_zeros = min(n_best // 4, 50 + attempt * 10)
            zero_mask = rng.random(n_best) < (n_zeros / n_best)
            start[zero_mask] = 0.0
            start = project_simplex(start)
        else:
            # Blended with uniform
            alpha = 0.1 + 0.05 * attempt
            start = (1 - alpha) * best_seq + alpha * np.ones(n_best) / n_best
            start = project_simplex(start)

        # Quick optimization
        t_alloc = min(remaining, 25)
        a_opt, _ = lbfgs_smooth(start, time_limit=t_alloc * 0.6, beta_init=100 + attempt * 20, beta_max=500 + attempt * 50)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"multi_{attempt}")

        remaining = deadline - time.monotonic() - 100
        if remaining > 8:
            a_opt2, _ = active_set_lp(a_opt, time_limit=min(remaining, t_alloc * 0.3), tol=0.002)
            (best_seq, best_score) = try_and_update(a_opt2, (best_seq, best_score), f"multi_as_{attempt}")

    print(f"  Phase 5: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 6: Final polishing (~120s) ===
    print("\n=== Phase 6: Final Polish ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    # Very tight active-set LP
    remaining = deadline - time.monotonic() - 40
    if remaining > 5:
        t_alloc = min(remaining, 50)
        a_opt, _ = active_set_lp(best_seq, time_limit=t_alloc, tol=0.0005, max_active=15)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "aslp_final")

    # Final L-BFGS
    remaining = deadline - time.monotonic() - 25
    if remaining > 5:
        t_alloc = min(remaining, 40)
        a_opt, _ = lbfgs_smooth(best_seq, time_limit=t_alloc, beta_init=500, beta_max=5000)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "lbfgs_final")

    # Symmetry
    sym_final = (best_seq + best_seq[::-1]) / 2.0
    sym_final = sym_final / sym_final.sum()
    (best_seq, best_score) = try_and_update(sym_final, (best_seq, best_score), "final_sym")

    # Final AS-LP
    remaining = deadline - time.monotonic() - 10
    if remaining > 5:
        a_opt, _ = active_set_lp(best_seq, time_limit=min(remaining, 20), tol=0.0003)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "aslp_last")

    # Symmetry again
    sym_final2 = (best_seq + best_seq[::-1]) / 2.0
    sym_final2 = sym_final2 / sym_final2.sum()
    (best_seq, best_score) = try_and_update(sym_final2, (best_seq, best_score), "sym_last")

    print(f"  Phase 6: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
