def softmax_from_t(t):
    """Convert parameter t to probability vector."""
    t_shift = t - t.max()
    exp_t = np.exp(t_shift)
    return exp_t / exp_t.sum()

def optimize_lse(f_init, beta_start, beta_end, time_limit):
    """Optimize using LSE-smoothed objective with L-BFGS-B."""
    n = len(f_init)
    t0 = time.monotonic()
    
    # Initialize t from f_init (which should sum to 1)
    f_init = np.maximum(f_init, 1e-15)
    f_init = f_init / f_init.sum()
    t = np.log(f_init)
    
    best_obj = fast_convolve(f_init).max() * 2 * n
    best_t = t.copy()
    
    beta = beta_start
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Optimize at current beta
        max_iter = max(10, int(remaining * 5))
        
        def obj_func(t_curr):
            val, _ = lse_objective_and_grad(t_curr, beta)
            return val
        
        def grad_func(t_curr):
            _, g = lse_objective_and_grad(t_curr, beta)
            return g
        
        try:
            res = minimize(obj_func, t, method='L-BFGS-B', jac=grad_func,
                          options={'maxiter': max_iter, 'gtol': 1e-8, 'maxfun': max_iter * 2})
            
            if res.success or res.fun < obj_func(t):
                t = res.x
                f_curr = softmax_from_t(t)
                obj_curr = fast_convolve(f_curr).max() * 2 * n
                if obj_curr < best_obj:
                    best_obj = obj_curr
                    best_t = t.copy()
                    print(f"    beta={beta:.1f}, obj={best_obj:.8f}")
        except Exception as e:
            print(f"    Optimization error at beta={beta:.1f}: {e}")
        
        # Increase beta (annealing)
        beta = min(beta * 1.5, beta_end)
        if beta >= beta_end * 0.99:
            # Do a few more iterations at final beta
            beta = beta_end
            if time.monotonic() - t0 > time_limit * 0.7:
                break
    
    return softmax_from_t(best_t), best_obj

def freq_perturb(f, rng, strength=0.05):

    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()

def sinusoidal_perturb(f, rng, n_harmonics=3):
    """Add sinusoidal perturbations in frequency domain."""
    n = len(f)
    F = np.fft.rfft(f)
    k_max = len(F)
    # Pick random frequency bins and add small perturbations
    for _ in range(n_harmonics):
        k = rng.integers(1, k_max)
        delta_mag = rng.standard_normal() * 0.02
        delta_phase = rng.standard_normal() * 0.05
        F[k] *= (1.0 + delta_mag)
        F[k] *= np.exp(1j * delta_phase)
        # Also perturb the conjugate bin (implicit in rfft)
    f_new = np.fft.irfft(F, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

def coordinate_descent_peaks(f, time_limit, n_peaks=10):
    """Try to reduce the top peaks by coordinate descent."""
    n = len(f)
    f = f.copy()
    f /= f.sum()
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    top_peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for k in top_peaks:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo >= hi:
                continue
            # Try moving mass from the index that contributes most to this peak
            vals = f[lo:hi+1] * f[k - np.arange(lo, hi+1)]
            j_max = lo + np.argmax(vals)
            if f[j_max] < 1e-10:
                continue
            # Try reducing f[j_max] and distributing to other positions
            for frac in [0.01, 0.05, 0.1]:
                delta = f[j_max] * frac
                # Distribute to random positions
                dests = np.random.choice(n, size=min(5, n), replace=False)
                for d in dests:
                    if d == j_max:
                        continue
                    f_trial = f.copy()
                    f_trial[j_max] -= delta
                    f_trial[d] += delta
                    if f_trial[j_max] < 0:
                        continue
                    f_trial = np.maximum(f_trial, 0)
                    f_trial /= f_trial.sum()
                    conv_trial = fast_convolve(f_trial)
                    if conv_trial.max() < best_obj - 1e-12:
                        best_obj = conv_trial.max()
                        best_f = f_trial.copy()
                        f = f_trial.copy()
                        improved = True
                        break
        if not improved:
            # Recompute top peaks
            conv = fast_convolve(f)
            top_peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
    
    return best_f, best_obj * 2 * n

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-smoothed objective with exact gradient + L-BFGS-B in softmax parameterization
2. Multi-length exploration with interpolation
3. Frequency-domain structured restarts for diversification
4. Adaptive beta annealing schedule
5. Coordinate descent on top peaks for fine refinement
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_from_t(t):
    """Convert parameter vector t to probability vector."""
    t_shift = t - t.max()
    exp_t = np.exp(t_shift)
    s = exp_t.sum()
    if s < 1e-300:
        return np.ones(len(t)) / len(t)
    return exp_t / s


def lse_objective_and_grad(t, beta):
    """
    Compute LSE-smoothed objective and gradient in softmax parameterization.
    Objective: 2n * (1/beta) * log(sum_k exp(beta * b_k)) where b = a*a
    Gradient derived analytically.
    """
    n = len(t)
    a = softmax_from_t(t)
    b = fast_convolve(a)
    b_len = len(b)

    beta_b = beta * b
    max_bb = beta_b.max()
    exp_bb = np.exp(beta_b - max_bb)
    S = exp_bb.sum()
    if S < 1e-300:
        obj = 2.0 * n * b.max()
        grad = np.zeros(n)
        return obj, grad

    obj = (2.0 * n / beta) * (max_bb + np.log(S))
    w = exp_bb / S

    # Gradient: dL/dt_j = (4n*a_j/beta) * (c_j - d)
    # where c_j = sum_k w_k * a_{k-j} (cross-correlation)
    # and d = sum_k w_k * b_k (scalar)
    d = np.dot(w, b)

    # Cross-correlation via FFT: corr[j] = sum_k a[k]*w[k+j]
    N = 1 << (2 * n).bit_length()
    A_fft = np.fft.rfft(a, N)
    W_fft = np.fft.rfft(w, N)
    corr = np.fft.irfft(np.conj(A_fft) * W_fft, N)
    c = corr[:n]

    grad = (4.0 * n * a / beta) * (c - d)
    return obj, grad


def optimize_lse(f_init, beta_start, beta_end, time_limit, max_beta_steps=30):
    """
    Optimize using LSE-smoothed objective with L-BFGS-B and adaptive beta annealing.
    Returns (best_f_normalized, best_objective_value).
    """
    n = len(f_init)
    t0 = time.monotonic()

    f_init = np.maximum(f_init, 1e-15)
    f_init = f_init / f_init.sum()
    t = np.log(f_init)

    conv_init = fast_convolve(f_init)
    best_obj = conv_init.max() * 2.0 * n
    best_t = t.copy()
    best_f = f_init.copy()

    beta = beta_start
    step = 0
    total_steps = max_beta_steps
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        step += 1
        # Adaptive beta schedule
        if step < total_steps:
            frac = (step - 1) / max(1, total_steps - 1)
            beta = beta_start * (beta_end / beta_start) ** frac
        else:
            beta = beta_end

        max_iter = max(20, int(remaining * 3))

        def obj_func(t_curr):
            val, _ = lse_objective_and_grad(t_curr, beta)
            return val

        def grad_func(t_curr):
            _, g = lse_objective_and_grad(t_curr, beta)
            return g

        try:
            res = minimize(obj_func, t, method='L-BFGS-B', jac=grad_func,
                           options={'maxiter': max_iter, 'gtol': 1e-10,
                                    'maxfun': max(max_iter * 3, 100)})
        except Exception:
            no_improve_count += 1
            beta = min(beta * 1.3, beta_end)
            continue

        if res.x is not None:
            t = res.x
            f_curr = softmax_from_t(t)
            conv_curr = fast_convolve(f_curr)
            obj_curr = conv_curr.max() * 2.0 * n

            if obj_curr < best_obj - 1e-12:
                best_obj = obj_curr
                best_t = t.copy()
                best_f = f_curr.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1

            # Don't restore the best if we haven't improved much yet (allow exploration)
            if no_improve_count > 5 and time.monotonic() - t0 > time_limit * 0.3:
                # Try a small perturbation to escape
                rng_local = np.random.default_rng(step * 1000 + 7)
                t = best_t + rng_local.standard_normal(n) * 0.01
                no_improve_count = 0

    # Final evaluation
    f_final = softmax_from_t(best_t)
    conv_final = fast_convolve(f_final)
    obj_final = conv_final.max() * 2.0 * n
    return f_final, obj_final


def freq_perturb(f, rng, strength=0.05):
    """Perturb f in the frequency domain (magnitude perturbation)."""
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def phase_perturb(f, rng, strength=0.1):
    """Perturb only the phase in frequency domain."""
    n = len(f)
    F = np.fft.rfft(f)
    pert = rng.standard_normal(len(F)) * strength
    F_new = F * np.exp(1j * pert)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def coordinate_descent_peaks(f, time_limit, n_peaks=15, rng=None):
    """Reduce top peaks by moving mass from contributing indices."""
    if rng is None:
        rng = np.random.default_rng(12345)
    n = len(f)
    f = f.copy()
    f /= f.sum()
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    top_peaks = np.argpartition(conv, -min(n_peaks, len(conv)))[-min(n_peaks, len(conv)):]

    stagnant = 0
    while time.monotonic() - t0 < time_limit:
        improved = False
        # Re-evaluate peaks periodically
        if stagnant > 20:
            conv = fast_convolve(f)
            top_peaks = np.argpartition(conv, -min(n_peaks, len(conv)))[-min(n_peaks, len(conv)):]
            stagnant = 0

        for k in top_peaks:
            k = int(k)
            if time.monotonic() - t0 > time_limit:
                break
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo >= hi:
                continue
            # Find the index contributing most to this peak
            i_range = np.arange(lo, hi + 1)
            vals = f[i_range] * f[k - i_range]
            j_max = i_range[np.argmax(vals)]
            if f[j_max] < 1e-12:
                continue
            # Try moving a fraction of mass from j_max to random destinations
            for frac in [0.02, 0.05, 0.1, 0.2]:
                delta = f[j_max] * frac
                if delta < 1e-14:
                    continue
                # Pick destination: prefer positions far from the peak
                n_dest = min(8, n)
                dests = rng.integers(0, n, size=n_dest)
                for d in dests:
                    if d == j_max:
                        continue
                    f_trial = f.copy()
                    f_trial[j_max] -= delta
                    f_trial[d] += delta
                    f_trial = np.maximum(f_trial, 0)
                    s = f_trial.sum()
                    if s < 1e-15:
                        continue
                    f_trial /= s
                    conv_trial = fast_convolve(f_trial)
                    if conv_trial.max() < best_obj - 1e-13:
                        best_obj = conv_trial.max()
                        best_f = f_trial.copy()
                        f = f_trial.copy()
                        improved = True
                        stagnant = 0
                        break
                if improved:
                    break
        if not improved:
            stagnant += 1
            # Perturb to escape
            f = project_simplex(f + rng.standard_normal(n) * 0.002)

    return best_f, best_obj * 2.0 * n


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    if a.sum() < 0.01:
        return float('inf')
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Main optimization pipeline:
    Phase 1: LSE+L-BFGS on parent
    Phase 2: Multi-length exploration
    Phase 3: Iterative refinement with diverse restarts
    Phase 4: Coordinate descent polish
    Phase 5: Final high-precision LSE
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768.0
        try_sequence(parent, best_tracker)

    n_parent = len(parent)

    # === Phase 1: LSE optimization on parent ===
    print("\n=== Phase 1: LSE+L-BFGS on parent ===")
    t1_end = time.monotonic() + 350
    t_alloc = min(330, t1_end - time.monotonic())
    if t_alloc > 30:
        print(f"  LSE n={n_parent}, t={t_alloc:.0f}s, beta: 3->3000")
        f_opt, obj = optimize_lse(parent, beta_start=3.0, beta_end=3000.0,
                                   time_limit=t_alloc, max_beta_steps=25)
        try_sequence(f_opt, best_tracker)

    # === Phase 2: Multi-length exploration ===
    print("\n=== Phase 2: Multi-length exploration ===")
    t2_end = deadline - 400
    lengths_to_try = sorted(set([
        n_parent - 64, n_parent - 32, n_parent - 16,
        n_parent + 16, n_parent + 32, n_parent + 64,
        n_parent + 100, n_parent + 150,
        640, 700, 800, 850, 900, 1000, 1024, 1100, 1200
    ]))
    lengths_to_try = [nl for nl in lengths_to_try if 128 <= nl <= 2000 and nl != n_parent]

    for nl in lengths_to_try:
        remaining = t2_end - time.monotonic()
        if remaining < 50:
            break
        t_alloc = min(100, remaining * 0.25)
        if t_alloc < 35:
            break
        f_init = interpolate_to_length(parent, nl)
        print(f"  LSE n={nl}, t={t_alloc:.0f}s")
        f_opt, obj = optimize_lse(f_init, beta_start=3.0, beta_end=2000.0,
                                   time_limit=t_alloc, max_beta_steps=20)
        try_sequence(f_opt, best_tracker)

    # === Phase 3: Iterative refinement with diverse restarts ===
    print("\n=== Phase 3: Iterative refinement ===")
    iteration = 0
    while time.monotonic() < deadline - 150 and iteration < 25:
        iteration += 1
        current_best = best_tracker[1] if best_tracker[1] is not None else parent
        n_cur = len(current_best)
        remaining = deadline - 150 - time.monotonic()
        if remaining < 40:
            break

        strategy = iteration % 4
        if strategy == 0:
            # Direct LSE from current best
            t_alloc = min(90, remaining * 0.4)
            if t_alloc > 25:
                print(f"  Iter {iteration}: LSE n={n_cur}, t={t_alloc:.0f}s")
                f_opt, _ = optimize_lse(current_best, beta_start=20.0, beta_end=5000.0,
                                         time_limit=t_alloc, max_beta_steps=15)
                try_sequence(f_opt, best_tracker)

        elif strategy == 1:
            # Frequency perturbation + LSE
            t_alloc = min(90, remaining * 0.4)
            if t_alloc > 25:
                print(f"  Iter {iteration}: FreqPerturb+LSE n={n_cur}, t={t_alloc:.0f}s")
                for attempt in range(3):
                    t_rem = deadline - 150 - time.monotonic()
                    if t_rem < 25:
                        break
                    strength = 0.003 + 0.004 * attempt
                    f_pert = freq_perturb(current_best, rng, strength=strength)
                    t_sub = max(15, (t_rem - 10) / 3.0)
                    f_opt, _ = optimize_lse(f_pert, beta_start=10.0, beta_end=3000.0,
                                             time_limit=t_sub, max_beta_steps=12)
                    try_sequence(f_opt, best_tracker)

        elif strategy == 2:
            # Phase perturbation + LSE
            t_alloc = min(90, remaining * 0.4)
            if t_alloc > 25:
                print(f"  Iter {iteration}: PhasePerturb+LSE n={n_cur}, t={t_alloc:.0f}s")
                for attempt in range(2):
                    t_rem = deadline - 150 - time.monotonic()
                    if t_rem < 25:
                        break
                    f_pert = phase_perturb(current_best, rng, strength=0.02 + 0.02 * attempt)
                    t_sub = max(15, (t_rem - 10) / 2.0)
                    f_opt, _ = optimize_lse(f_pert, beta_start=10.0, beta_end=3000.0,
                                             time_limit=t_sub, max_beta_steps=12)
                    try_sequence(f_opt, best_tracker)

        else:
            # Coordinate descent
            t_alloc = min(60, remaining * 0.25)
            if t_alloc > 15:
                print(f"  Iter {iteration}: CoordDescent n={n_cur}, t={t_alloc:.0f}s")
                f_opt, _ = coordinate_descent_peaks(current_best, time_limit=t_alloc,
                                                     n_peaks=12, rng=rng)
                try_sequence(f_opt, best_tracker)

    # === Phase 4: Coordinate descent polish ===
    print("\n=== Phase 4: Coordinate descent polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)
    t_alloc = min(80, deadline - time.monotonic() - 80)
    if t_alloc > 20:
        print(f"  CoordDescent n={n_cur}, t={t_alloc:.0f}s")
        f_opt, _ = coordinate_descent_peaks(current_best, time_limit=t_alloc,
                                             n_peaks=10, rng=rng)
        try_sequence(f_opt, best_tracker)

    # === Phase 5: Final high-precision LSE ===
    print("\n=== Phase 5: Final high-precision LSE ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)
    t_alloc = min(60, deadline - time.monotonic() - 30)
    if t_alloc > 15:
        print(f"  Final LSE (beta 200->10000) n={n_cur}, t={t_alloc:.0f}s")
        f_opt, _ = optimize_lse(current_best, beta_start=200.0, beta_end=10000.0,
                                 time_limit=t_alloc, max_beta_steps=10)
        try_sequence(f_opt, best_tracker)

    # Quick perturbations
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_rem = deadline - time.monotonic()
    if t_rem > 10:
        n_cur = len(current_best)
        for _ in range(30):
            if time.monotonic() > deadline - 5:
                break
            f_pert = freq_perturb(current_best, rng, strength=0.002)
            try_sequence(f_pert, best_tracker)
            f_pert2 = project_simplex(current_best + rng.standard_normal(n_cur) * 0.001)
            try_sequence(f_pert2, best_tracker)

    # Final verification
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using fallback")
    fallback = parent if parent is not None else np.ones(768) / 768.0
    return [float(x) for x in fallback]
