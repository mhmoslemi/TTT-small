"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses spectral projection, Frank-Wolfe with LP oracle, and coordinate descent.

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_project(a, n_iter=200, t_init=0.3, t_decay=0.995):
    """Spectral projection toward flat autocorrelation."""
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Compute target spectrum magnitude
    f = np.arange(N)
    # target_mag[k] proportional to |sin(pi*(2n-1)*k/N) / sin(pi*k/N)|
    with np.errstate(divide='ignore', invalid='ignore'):
        ratio = np.sin(np.pi * (2 * n - 1) * f / N) / np.sin(np.pi * f / N)
    ratio[0] = 2 * n - 1  # Limit at f=0
    target_mag = np.abs(ratio)
    # Normalize so that target_mag[0] = 1 (matching |A[0]| = sum(a) = 1)
    target_mag = target_mag / target_mag[0]
    
    t = t_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for it in range(n_iter):
        A = np.fft.rfft(a, N)
        A_mag = np.abs(A)
        A_phase = np.angle(A)
        
        # Blend toward target magnitude
        A_new = (1 - t) * A_mag + t * target_mag
        A_new = A_new * np.exp(1j * A_phase)
        
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:

        a_new = a_new / s_new

        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new

def solve_lp_oracle(f, M, time_budget=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix A_ub where row k is the k-th convolution constraint
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_budget})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None
