import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def solve_lp(f, M):
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def line_search_1d(f, g, best_val):
    """Find best alpha in [0, 1] to minimize conv((1-alpha)*f + alpha*g).max()"""
    def obj(alpha):
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            return best_val + 1
        ft = ft / s
        return fast_convolve(ft).max()
    
    res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-4})
    if res.success:
        alpha = res.x
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s > 1e-15:
            ft = ft / s
            return ft, res.fun
    return None, best_val

from scipy.optimize import linprog, minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (A @ g)[k] = sum_j f[k-j]*g[j]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M, time_limit_ms=5000):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit_ms / 1000.0})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search(f, g, current_max):
    """Two-stage line search: coarse grid then bounded scalar minimization."""
    n = len(f)
    best_alpha = 0.0
    best_val = current_max

    # Stage 1: Coarse grid (20 points)
    alphas = np.linspace(0.0, 1.0, 20)
    for alpha in alphas:
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            continue
        ft = ft / s
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = float(alpha)

    # Stage 2: Fine bounded minimization around best
    if 0.0 < best_alpha < 1.0:
        lo = max(0.0, best_alpha - 0.06)
        hi = min(1.0, best_alpha + 0.06)
        if hi - lo > 1e-6:
            def obj(alpha):
                ft = (1.0 - alpha) * f + alpha * g
                ft = np.maximum(ft, 0.0)
                s = ft.sum()
                if s < 1e-15:
                    return current_max + 1.0
                ft = ft / s
                return fast_convolve(ft).max()
            try:
                res = minimize_scalar(obj, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 5e-5, 'maxiter': 30})
                if res.success and res.fun < best_val:
                    best_val = float(res.fun)
                    best_alpha = float(res.x)
            except Exception:
                pass
    elif best_alpha == 0.0:
        # Try small steps from 0
        for alpha in [0.01, 0.02, 0.05, 0.1]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            s = ft.sum()
            if s < 1e-15:
                continue
            ft = ft / s
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
    elif best_alpha == 1.0:
        # Try to go further (already at g)
        pass

    if best_alpha > 1e-10:
        ft = (1.0 - best_alpha) * f + best_alpha * g
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s > 1e-15:
            ft = ft / s
        else:
            ft = f
    else:
        ft = f.copy()

    return ft, best_val


def fw_optimize(f0, time_limit, verbose=False, m_tighten_schedule=None):
    """
    Frank-Wolfe with LP oracle and adaptive M-tightening.
    m_tighten_schedule: if provided, a list of (time_frac, m_factor) pairs
    that define how m_relax changes over time.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    stagnant = 0
    it = 0
    m_relax = 1.003

    # Default tightening schedule
    if m_tighten_schedule is None:
        m_tighten_schedule = [(0.0, 1.003), (0.3, 1.002), (0.6, 1.001), (0.85, 1.0005), (1.0, 1.0003)]

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        if time_limit - elapsed < 0.5:
            break

        # Update m_relax based on progress
        frac = elapsed / time_limit
        for i in range(len(m_tighten_schedule) - 1):
            if m_tighten_schedule[i][0] <= frac <= m_tighten_schedule[i + 1][0]:
                t0_s = m_tighten_schedule[i][0]
                t1_s = m_tighten_schedule[i + 1][0]
                m0 = m_tighten_schedule[i][1]
                m1 = m_tighten_schedule[i + 1][1]
                if t1_s > t0_s:
                    local_frac = (frac - t0_s) / (t1_s - t0_s)
                    m_relax = m0 + local_frac * (m1 - m0)
                else:
                    m_relax = m1
                break

        # Solve LP
        g = solve_lp_oracle(f, M * m_relax)

        if g is None or g.sum() < 1e-15:
            # Try looser
            g = solve_lp_oracle(f, M * (m_relax + 0.005))
            if g is None or g.sum() < 1e-15:
                stagnant += 1
                if stagnant > 15:
                    # Perturb and reset
                    rng = np.random.default_rng(it * 7919 + 13)
                    scale = 0.003 + 0.003 * min(1.0, stagnant / 50.0)
                    f = project_simplex(best_f + rng.standard_normal(n) * scale)
                    conv = fast_convolve(f)
                    M = conv.max()
                    stagnant = 0
                continue

        g = g / g.sum()

        # Check direction quality
        if np.max(np.abs(g - f)) < 1e-10:
            stagnant += 1
            if stagnant > 25:
                rng = np.random.default_rng(it * 97 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue

        # Line search
        f_new, val_new = line_search(f, g, M)

        if val_new < best_obj - 1e-15:
            f = f_new
            M = val_new
            best_obj = M
            best_f = f.copy()
            stagnant = 0
        else:
            # Try a smaller step
            alpha_try = 0.1
            ft = (1.0 - alpha_try) * f + alpha_try * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st > 1e-15:
                ft = ft / st
                val_try = fast_convolve(ft).max()
                if val_try < best_obj - 1e-15:
                    f = ft
                    M = val_try
                    best_obj = M
                    best_f = f.copy()
                    stagnant = 0
                else:
                    stagnant += 1
            else:
                stagnant += 1

        if verbose and it % 50 == 0:
            print(f"    FW it={it}, obj={best_obj:.12f}, m_relax={m_relax:.6f}, stagnant={stagnant}")

    return best_f, best_obj


def spectral_init(n, mode="raised_cosine", rng=None):
    """Generate a frequency-domain-inspired initialization."""
    if rng is None:
        rng = np.random.default_rng(42)

    # Create a frequency profile
    freqs = np.fft.rfftfreq(n) * 2  # normalized to [0, 1]

    if mode == "raised_cosine":
        # Raised cosine in frequency: flat low-freq, rolloff at high-freq
        beta = 0.3
        spec = np.ones_like(freqs)
        mask = freqs > (1.0 - beta)
        spec[mask] = 0.5 * (1 + np.cos(np.pi * (freqs[mask] - (1.0 - beta)) / beta))
    elif mode == "gaussian_spec":
        # Gaussian in frequency
        spec = np.exp(-0.5 * ((freqs - 0.3) / 0.25) ** 2)
    elif mode == "flat":
        spec = np.ones_like(freqs)
    elif mode == "edge_spec":
        # Emphasize edges in frequency
        spec = 0.5 + 0.5 * np.cos(2 * np.pi * freqs * 3)
        spec = np.maximum(spec, 0.05)
    else:
        spec = np.ones_like(freqs)

    spect = spec ** 0.5  # magnitude
    phase = rng.standard_normal(len(spect))
    F = spect * (1.0 + 0j) * np.exp(1j * phase * 0.3)  # small random phase
    a = np.fft.irfft(F, n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def generate_starts(n, parent, rng):
    """Generate diverse initializations."""
    starts = []
    x = np.arange(n, dtype=np.float64)

    # Parent resampled
    if parent is not None:
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        a_res = np.interp(x_new, x_old, parent)
        a_res = np.maximum(a_res, 0.0)
        s = a_res.sum()
        if s > 1e-15:
            starts.append(a_res / s)

    # Uniform
    starts.append(np.ones(n) / n)

    # Cosine taper
    cos_t = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_t = np.maximum(cos_t, 0.01)
    starts.append(cos_t / cos_t.sum())

    # Raised cosine window
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    rc = np.maximum(rc, 0.01)
    starts.append(rc / rc.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - (n - 1) / 2) / (0.15 * n)) ** 2)
    starts.append(g / g.sum())

    # Edge concentration
    edge = 1.0 / (np.minimum(x + 1, n - x) + 1.0)
    starts.append(edge / edge.sum())

    # Spectral: raised cosine
    starts.append(spectral_init(n, "raised_cosine", rng))

    # Spectral: edge
    starts.append(spectral_init(n, "edge_spec", rng))

    # Random Dirichlet (concentrated)
    starts.append(rng.dirichlet(np.ones(n) * 0.5))

    # Two-Gaussian
    g1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.08 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.08 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())

    return starts


def coordinate_descent_polish(a0, time_limit, verbose=False):
    """
    Coordinate descent that targets the top peaks of the convolution.
    For each peak, find the pair of indices contributing most, and try to
    redistribute mass to reduce the peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    step_scale = 0.005
    no_improve = 0

    while time.monotonic() - t0 < time_limit - 0.1:
        it += 1
        conv = fast_convolve(a)
        n_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        improved = False
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue

            # Find top contributing pairs
            contributions = []
            for j in range(js, je + 1):
                c = a[j] * a[k - j]
                contributions.append((c, j, k - j))
            contributions.sort(reverse=True)

            for (_, j1, j2) in contributions[:8]:
                if a[j1] < 1e-9 and a[j2] < 1e-9:
                    continue

                # Try shifting mass from the larger contributor to a random location
                src = j1 if a[j1] >= a[j2] else j2
                if a[src] < 1e-9:
                    continue

                n_trials = 6
                for _t in range(n_trials):
                    rng_local = np.random.default_rng((it * 1000003 + k * 1009 + src * 37 + _t) % (2**31))
                    tgt = int(rng_local.integers(0, n))
                    if tgt == src or tgt == (j1 if src == j2 else j2):
                        continue

                    for frac in [step_scale, step_scale * 0.5, step_scale * 2.0, step_scale * 0.25]:
                        delta = a[src] * frac
                        if delta < 1e-9:
                            continue
                        candidate = a.copy()
                        candidate[src] -= delta
                        candidate[tgt] += delta
                        if candidate[src] < -1e-12:
                            continue
                        candidate = np.maximum(candidate, 0.0)
                        st = candidate.sum()
                        if st < 1e-15:
                            continue
                        candidate = candidate / st
                        val = fast_convolve(candidate).max()
                        if val < best_obj - 1e-14:
                            a = candidate
                            best_obj = val
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                no_improve = 0
                break
            else:
                no_improve += 1

        if not improved:
            step_scale *= 0.95
            if step_scale < 1e-7:
                break
            # Periodic perturbation
            if no_improve % 50 == 0 and no_improve < 500:
                rng = np.random.default_rng(it * 48271 + 7)
                noise_scale = min(0.01, 0.002 * (1 + no_improve / 100.0))
                candidate = project_simplex(a + rng.standard_normal(n) * noise_scale)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-14:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
            elif no_improve > 300:
                break

        if verbose and it % 100 == 0:
            print(f"    CD it={it}, obj={best_obj:.12f}, step={step_scale:.6f}, no_improve={no_improve}")

    return best_a, best_obj


def subgradient_step(a, peaks, weights, lr):
    """One subgradient step targeting weighted peaks."""
    n = len(a)
    conv = fast_convolve(a)
    grad = np.zeros(n)

    for i, k in enumerate(peaks):
        w = weights[i]
        if w < 1e-14:
            continue
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad[j] += w * a[k - j]
            if j != k - j:
                grad[k - j] += w * a[j]

    return grad


def subgradient_optimize(a0, time_limit, T_init=0.0003, lr_init=0.015, top_k=20):
    """Adaptive smoothed subgradient with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.9

    t0 = time.monotonic()
    stagnant = 0
    it = 0

    while time.monotonic() - t0 < time_limit - 0.05:
        it += 1
        conv = fast_convolve(a)

        # Top-k peaks
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))

        pv = conv[peaks]
        pv_max = pv.max()
        # Temperature-based weighting
        w = np.exp((pv - pv_max) / (T * 0.05))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / w_sum

        grad = subgradient_step(a, peaks, w, lr)

        v = mom * v - lr * grad

        # Backtracking line search
        best_candidate = a
        best_val = fast_convolve(a).max()
        fracs = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]
        for frac in fracs:
            candidate = project_simplex(a + frac * v)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        # Adapt
        if stagnant > 30:
            T *= 1.08
            lr *= 0.95
        if stagnant > 80:
            T *= 0.5
            lr *= 1.5
            mom = 0.93
        if stagnant > 250:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise = 0.003 + 0.004 * min(1.0, (stagnant - 250) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            v = np.zeros(n)
            mom = 0.9
            stagnant = 0

    return best_a, best_obj


def resample_to_length(a, new_n):
    """Resample a sequence to a new length via linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search for minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a_norm)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512
        try_seq(parent, "uniform_512")

    # === Phase 1: Multi-length exploration (250s) ===
    print("\n=== Phase 1: Multi-length Exploration ===")
    t_phase = time.monotonic()

    lengths = [384, 512, 640, 768, 896, 1024, 1280]
    # Keep parent length
    n_parent = len(parent)
    if n_parent not in lengths:
        lengths.append(n_parent)
    lengths = sorted(set([l for l in lengths if 256 <= l <= 1600]))

    for n_try in lengths:
        remaining = deadline - time.monotonic() - (budget_s * 0.55)
        if remaining < 15:
            break

        starts = generate_starts(n_try, parent, rng)
        n_starts_to_try = min(5, len(starts))

        for si in range(n_starts_to_try):
            remaining = deadline - time.monotonic() - (budget_s * 0.50)
            if remaining < 10:
                break
            t_alloc = min(remaining, 35)
            if t_alloc < 5:
                break
            a_opt, _ = fw_optimize(starts[si], time_limit=t_alloc)
            label = f"expl_n{n_try}_s{si}"
            try_seq(a_opt, label)

    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW on best (500s) ===
    print("\n=== Phase 2: Deep FW Optimization ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Main deep FW with adaptive tightening
        remaining = deadline - time.monotonic() - (budget_s * 0.30)
        if remaining > 50:
            t_alloc = min(remaining, 450)
            print(f"  Deep FW n={n_best}, t_alloc={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_fw_main")

        # Try at slightly different lengths
        for delta in [-64, 64]:
            n_alt = n_best + delta
            if n_alt < 256:
                continue
            remaining = deadline - time.monotonic() - (budget_s * 0.25)
            if remaining < 20:
                break
            a_resampled = resample_to_length(best_seq, n_alt)
            t_alloc = min(remaining, 80)
            print(f"  FW at n={n_alt}, t_alloc={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(a_resampled, time_limit=t_alloc)
            try_seq(a_opt, f"deep_fw_n{n_alt}")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Perturbed restarts (350s) ===
    print("\n=== Phase 3: Perturbed Restarts ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        perturb_scales = [0.002, 0.005, 0.01, 0.02, 0.04]
        for ps in perturb_scales:
            remaining = deadline - time.monotonic() - (budget_s * 0.15)
            if remaining < 20:
                break
            noise = rng.standard_normal(n_best) * ps
            start = project_simplex(best_seq + noise)
            t_alloc = min(remaining, 60)
            a_opt, _ = fw_optimize(start, time_limit=t_alloc)
            try_seq(a_opt, f"pert_{ps}")

        # Spectral restarts
        for mode in ["raised_cosine", "edge_spec", "gaussian_spec"]:
            remaining = deadline - time.monotonic() - (budget_s * 0.10)
            if remaining < 15:
                break
            start = spectral_init(n_best, mode, rng)
            t_alloc = min(remaining, 40)
            a_opt, _ = fw_optimize(start, time_limit=t_alloc)
            try_seq(a_opt, f"spectral_{mode}")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Subgradient refinement (200s) ===
    print("\n=== Phase 4: Subgradient Refinement ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        configs = [
            {'T_init': 0.0002, 'lr_init': 0.01, 'top_k': 15},
            {'T_init': 0.001, 'lr_init': 0.02, 'top_k': 30},
            {'T_init': 0.005, 'lr_init': 0.008, 'top_k': 50},
        ]
        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - (budget_s * 0.05)
            if remaining < 15:
                break
            t_alloc = min(remaining, 65)
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"subgrad_{ci}")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Coordinate descent polishing (remaining - 30s) ===
    print("\n=== Phase 5: Coordinate Descent Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 10:
            t_alloc = min(remaining, 200)
            print(f"  CD polish, t_alloc={t_alloc:.0f}s")
            a_opt, cd_obj = coordinate_descent_polish(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "cd_polish")

            # One more round: FW then CD
            remaining = deadline - time.monotonic() - 15.0
            if remaining > 20:
                n_best = len(best_seq)
                noise = rng.standard_normal(n_best) * 0.001
                start = project_simplex(best_seq + noise)
                t_fw = min(remaining, 40)
                a_opt_fw, _ = fw_optimize(start, time_limit=t_fw)
                try_seq(a_opt_fw, "fw_micro")

                remaining = deadline - time.monotonic() - 10.0
                if remaining > 10:
                    t_cd = min(remaining, 50)
                    a_opt_cd, _ = coordinate_descent_polish(a_opt_fw, time_limit=t_cd)
                    try_seq(a_opt_cd, "cd_final")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Last-ditch: try a few more FW runs with tight time ===
    print("\n=== Phase 6: Final Micro-Optimization ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        for attempt in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            noise = rng.standard_normal(n_best) * (0.0005 * (attempt + 1))
            start = project_simplex(best_seq + noise)
            t_alloc = min(remaining, 12)
            a_opt, _ = fw_optimize(start, time_limit=t_alloc)
            try_seq(a_opt, f"micro_{attempt}")

    print(f"  Phase 6 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback used: score={best_score:.10f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
