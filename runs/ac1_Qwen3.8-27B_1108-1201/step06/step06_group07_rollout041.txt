def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])

Let me also think about another approach: **conditional gradient (Frank-Wolfe) method**.

The Frank-Wolfe method for minimizing a convex function `f(a)` over the simplex:
1. Compute gradient `g = ∇f(a)`.
2. Find `s = argmin_{v in simplex} <g, v> = e_{argmin(g)}` (a vertex of the simplex).
3. Update: `a_new = (1-t)*a + t*s` for some step size `t`.

The advantage: the "linear oracle" step is trivial (just find the minimum of the gradient). No projection needed!

For the smoothed max objective, the gradient is well-defined, so Frank-Wolfe should work well.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Smoothed max (log-sum-exp) gradient descent with Nesterov momentum and FFT-based subgradients
- Peak-directed mass exchange for local refinement
- Focused subgradient (top-K peaks) for alternating refinement
- Multi-n exploration with diverse initializations
- Adaptive beta (smoothing parameter) scheduling
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smgd_gradient(a, weights, conv_size, xcorr_fft):
    """Compute subgradient via FFT cross-correlation."""
    n = len(a)
    wp = np.zeros(xcorr_fft)
    wp[:conv_size] = weights
    ap = np.zeros(xcorr_fft)
    ap[:n] = a
    W = np.fft.rfft(wp, xcorr_fft)
    A = np.fft.rfft(ap, xcorr_fft)
    xcorr = np.fft.irfft(np.conj(W) * A, xcorr_fft)
    return 2.0 * xcorr[:n]


def smgd(a_init, time_limit, beta_start=20.0, beta_end=250.0, lr_init=0.15, momentum=0.9):
    """
    Smoothed Max Gradient Descent with Nesterov momentum.
    Uses log-sum-exp approximation to max for smooth gradients.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv_size = 2 * n - 1
    conv_fft = 1 << (conv_size - 1).bit_length()
    xcorr_fft = 1 << (3 * n - 3).bit_length()

    best_a = a.copy()
    best_val = fast_convolve(a).max()

    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 0.05:
            break

        progress = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_start + (beta_end - beta_start) * progress

        # Compute convolution
        A = np.fft.rfft(a, conv_fft)
        conv = np.fft.irfft(A * A, conv_fft)[:conv_size]
        cur_max = conv.max()

        # Softmax weights
        z = beta * conv
        z -= z.max()
        w = np.exp(z)
        w_sum = w.sum()
        if w_sum < 1e-300:
            break
        w /= w_sum

        # Gradient via FFT
        grad = compute_smgd_gradient(a, w, conv_size, xcorr_fft)

        # Normalize gradient for stable step
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad_n = grad / gmax

        # Nesterov momentum update
        v = momentum * v - lr * grad_n
        a_new = project_simplex(a + v)

        # Evaluate new point
        AN = np.fft.rfft(a_new, conv_fft)
        conv_new = np.fft.irfft(AN * AN, conv_fft)[:conv_size]
        new_max = conv_new.max()

        if new_max < best_val - 1e-13:
            best_val = new_max
            best_a = a_new.copy()

        if new_max < cur_max * (1.0 - 1e-12):
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                lr *= 0.85
            if no_improve > 80:
                v = np.zeros(n)
                lr = lr_init
                no_improve = 0
            if no_improve > 250:
                rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
                scale = 0.003 + 0.005 * min(1.0, (no_improve - 250) / 50.0)
                a = project_simplex(best_a + rng.standard_normal(n) * scale)
                v = np.zeros(n)
                lr = lr_init
                no_improve = 0

        # Periodically boost alpha to escape plateaus
        if it % 200 == 0:
            lr = min(lr * 1.2, lr_init * 2.0)

    return best_a, best_val


def peak_exchange(a_init, time_limit):
    """
    Peak-directed mass exchange: iteratively move mass from positions
    that strongly contribute to the highest peak toward positions
    that contribute weakly to all top peaks.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_val = fast_convolve(a).max()
    conv = fast_convolve(a)

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Top peaks
        top_k = min(15, len(conv))
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        top_w = conv[top_idx]
        top_w = top_w / top_w.sum()

        # Compute cost for each position: weighted sum of contributions to top peaks
        cost = np.zeros(n)
        for idx in range(top_k):
            kp = int(top_idx[idx])
            wi = top_w[idx]
            js = max(0, kp - n + 1)
            je = min(n - 1, kp)
            for j in range(js, je + 1):
                cost[j] += wi * a[j] * a[kp - j]

        # Find highest-cost position (source)
        j_src = int(np.argmax(cost))
        if cost[j_src] < 1e-15:
            break

        # Find lowest-cost position (target), excluding source
        cost_excl = cost.copy()
        cost_excl[j_src] = np.inf
        j_dst = int(np.argmin(cost_excl))

        # Try moving different fractions
        best_frac = 0.0
        best_new_val = best_val
        for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
            delta = frac * a[j_src]
            if delta < 1e-15:
                continue
            a_new = a.copy()
            a_new[j_src] -= delta
            a_new[j_dst] += delta
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            new_val = fast_convolve(a_new).max()
            if new_val < best_new_val - 1e-14:
                best_new_val = new_val
                best_frac = frac

        if best_frac > 0:
            delta = best_frac * a[j_src]
            a = a.copy()
            a[j_src] -= delta
            a[j_dst] += delta
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            best_val = fast_convolve(a).max()
            best_a = a.copy()
            conv = fast_convolve(a)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Perturb cost by adding noise, try different targets
                rng = np.random.default_rng((it * 9973 + 7) % (2**31 - 1))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(a)
                no_improve = 0
            elif no_improve > 100:
                break

    return best_a, best_val


def focused_subgrad(a_init, time_limit, K=15):
    """
    Subgradient method focusing on top-K peaks with line search.
    Faster than full SCA when only a few peaks are active.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_val = fast_convolve(a).max()

    lr = 0.08
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        cur_max = conv.max()

        # Top K peaks
        if K >= len(conv):
            peaks = np.arange(len(conv))
            pv = conv[peaks]
        else:
            peaks = np.argpartition(conv, -K)[-K:]
            pv = conv[peaks]

        # Sharpened softmax weights
        z = 8.0 * (pv - pv.max())
        w = np.exp(z)
        w /= w.sum()

        # Gradient from top-K peaks
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            k = int(peaks[idx])
            wi = w[idx]
            if wi < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            a_slice = a[k - je:k - js + 1][::-1]
            grad[js:je + 1] += wi * 2.0 * a_slice

        # Direction
        direction = project_simplex(a - lr * grad) - a

        # Line search
        best_alpha = 0.0
        best_new_val = cur_max
        for alpha in [0.1, 0.3, 0.5, 0.8, 1.0]:
            cand = a + alpha * direction
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_new_val - 1e-14:
                best_new_val = val
                best_alpha = alpha

        if best_alpha > 0:
            a = a + best_alpha * direction
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            new_val = fast_convolve(a).max()
            if new_val < best_val - 1e-13:
                best_val = new_val
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            lr *= 0.7
            no_improve = 0
        if lr < 0.003:
            lr = 0.08
            no_improve = 0

        if no_improve > 150:
            rng = np.random.default_rng((it * 48271 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            lr = 0.08
            no_improve = 0

    return best_a, best_val


def generate_inits(n, parent, rng):
    """Generate diverse initializations for a given length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Interpolated parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Uniform
    inits.append(np.ones(n) / n)

    # Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.3) / n)

    # Dirichlet (spread)
    inits.append(rng.dirichlet(np.ones(n) * 3.0) / n)

    # U-shaped (edge-heavy)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x).astype(float) + 1.0) ** 1.5
    inits.append(edge_w / edge_w.sum())

    # Two-peak
    g2 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())

    # Cosine-modulated
    t = x / max(n - 1, 1)
    cos_mod = (1.0 + 0.4 * np.cos(2 * np.pi * t)) ** 2 + 0.05
    inits.append(cos_mod / cos_mod.sum())

    # Random perturbation of parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px + rng.standard_normal(n) * 0.05, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: multi-phase hybrid optimization combining SMGD,
    peak exchange, and focused subgradient with adaptive n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)

    # === Phase 1: Multi-n SMGD exploration (350s) ===
    print("\n=== Phase 1: Multi-n SMGD Exploration ===")
    t1 = time.monotonic()

    ns_to_try = sorted(set([n_parent, 512, 640, 768, 896, 1024, 1280, 1536, 2048]))
    ns_to_try = [n for n in ns_to_try if 256 <= n <= 3072]

    for n_try in ns_to_try:
        remaining_after = deadline - time.monotonic()
        if remaining_after < 300:
            break

        inits = generate_inits(n_try, parent, rng)
        n_init = min(3, len(inits))

        for ii in range(n_init):
            remaining = deadline - time.monotonic() - 250
            if remaining < 15:
                break
            t_alloc = min(20, remaining)
            a_opt, _ = smgd(inits[ii], time_limit=t_alloc,
                            beta_start=20, beta_end=100, lr_init=0.12, momentum=0.88)
            try_seq(a_opt, f"smgd_n{n_try}_i{ii}")
            if best_seq is not None and best_seq is not a_opt:
                # Quick peak exchange on the best
                remaining2 = deadline - time.monotonic() - 200
                if remaining2 > 10:
                    a_pe, _ = peak_exchange(best_seq, time_limit=min(10, remaining2))
                    try_seq(a_pe, f"peak_n{n_try}_i{ii}")

    print(f"  P1 done: best={best_score:.8f}, t={time.monotonic()-t1:.0f}s")

    # === Phase 2: Deep SMGD refinement (500s) ===
    print("\n=== Phase 2: Deep SMGD ===")
    t2 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 500
        if remaining > 50:
            t_alloc = min(remaining, 400)
            print(f"  Deep SMGD n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = smgd(best_seq, time_limit=t_alloc,
                            beta_start=30, beta_end=300, lr_init=0.1, momentum=0.9)
            try_seq(a_opt, "deep_smgd")

            remaining = deadline - time.monotonic() - 300
            if remaining > 30:
                t_alloc2 = min(remaining, 200)
                a_opt2, _ = smgd(a_opt, time_limit=t_alloc2,
                                 beta_start=100, beta_end=500, lr_init=0.05, momentum=0.92)
                try_seq(a_opt2, "deep_smgd2")

    print(f"  P2 done: best={best_score:.8f}, t={time.monotonic()-t2:.0f}s")

    # === Phase 3: Alternating refinement (500s) ===
    print("\n=== Phase 3: Alternating Refinement ===")
    t3 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        phase3_budget = min(500, max(0, (deadline - time.monotonic()) - 300))
        t_phase3_start = time.monotonic()

        # Alternate between peak exchange and focused subgradient
        while time.monotonic() - t_phase3_start < phase3_budget:
            remaining = phase3_budget - (time.monotonic() - t_phase3_start)
            if remaining < 20:
                break

            # Peak exchange
            t_alloc = min(30, remaining - 15)
            if t_alloc > 5:
                a_pe, _ = peak_exchange(best_seq, time_limit=t_alloc)
                try_seq(a_pe, f"peak3_{time.monotonic()-t_phase3_start:.0f}")

            remaining = phase3_budget - (time.monotonic() - t_phase3_start)
            if remaining < 20:
                break

            # Focused subgradient
            K = min(10, len(fast_convolve(best_seq)))
            t_alloc = min(30, remaining - 15)
            if t_alloc > 5:
                a_fs, _ = focused_subgrad(best_seq, time_limit=t_alloc, K=K)
                try_seq(a_fs, f"fs3_{time.monotonic()-t_phase3_start:.0f}")

            remaining = phase3_budget - (time.monotonic() - t_phase3_start)
            if remaining < 20:
                break

            # Short SMGD burst
            t_alloc = min(25, remaining - 10)
            if t_alloc > 5:
                a_sm, _ = smgd(best_seq, time_limit=t_alloc,
                               beta_start=100, beta_end=200, lr_init=0.05, momentum=0.9)
                try_seq(a_sm, f"sm3_{time.monotonic()-t_phase3_start:.0f}")

    print(f"  P3 done: best={best_score:.8f}, t={time.monotonic()-t3:.0f}s")

    # === Phase 4: Diversified restarts (250s) ===
    print("\n=== Phase 4: Diversified Restarts ===")
    t4 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        scales = [0.002, 0.005, 0.01, 0.02, 0.04]
        for scale in scales:
            remaining = deadline - time.monotonic() - 100
            if remaining < 20:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(30, remaining - 30)
            if t_alloc < 5:
                break
            a_opt, _ = smgd(f_pert, time_limit=t_alloc,
                            beta_start=50, beta_end=300, lr_init=0.08, momentum=0.9)
            try_seq(a_opt, f"restart_{scale}")

            # Quick peak exchange after each restart
            remaining = deadline - time.monotonic() - 100
            if remaining > 10 and best_seq is not None:
                a_pe, _ = peak_exchange(best_seq, time_limit=min(10, remaining - 20))
                try_seq(a_pe, f"peak_r_{scale}")

    print(f"  P4 done: best={best_score:.8f}, t={time.monotonic()-t4:.0f}s")

    # === Phase 5: Final polish (remaining) ===
    print("\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine SMGD
        remaining = deadline - time.monotonic() - 30
        if remaining > 15:
            t_alloc = min(remaining, 60)
            a_opt, _ = smgd(best_seq, time_limit=t_alloc,
                            beta_start=200, beta_end=500, lr_init=0.03, momentum=0.93)
            try_seq(a_opt, "polish_smgd")

        # Final peak exchange
        remaining = deadline - time.monotonic() - 15
        if remaining > 10:
            a_pe, _ = peak_exchange(best_seq, time_limit=min(remaining, 30))
            try_seq(a_pe, "polish_peak")

        # Final focused subgradient
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            a_fs, _ = focused_subgrad(best_seq, time_limit=min(remaining, 20), K=10)
            try_seq(a_fs, "polish_fs")

    print(f"  P5 done: best={best_score:.8f}, t={time.monotonic()-t5:.0f}s")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final evaluation
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
