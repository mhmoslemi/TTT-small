    """Softmax approximation of max(conv(a,a))."""
    conv = fast_convolve(a)
    conv_max = conv.max()
    # Shift for numerical stability
    shifted = T * (conv - conv_max)
    log_sum = np.log(np.exp(shifted).sum()) 
    return conv_max + np.log(np.exp(shifted).sum()) / T

def softened_grad(a, T):
    """Gradient of the softmax approximation."""
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = T * (conv - conv_max)
    weights = np.exp(shifted)
    weights /= weights.sum()

Let me also think about a key mathematical insight: **What does a "good" sequence look like?**

For the auto-convolution to be flat, the sequence should have "random-like" autocorrelation properties. In signal processing, this is achieved by:
- Zadoff-Chu sequences (complex, perfect autocorrelation).
- m-sequences (binary, good autocorrelation).
- Costas arrays (for 2D).
- Barker sequences (for short binary).

For real-valued non-negative sequences, the best we can do is related to the "flat-top" design. The sequence should oscillate in amplitude to cancel out the convolution peaks.

But since we need a >= 0, we're restricted. The "oscillation" can only be in the amplitude (between 0 and max), not in sign.

**Key insight**: For a non-negative sequence, the auto-convolution at lag 0 is sum a[i]^2, which is at least 1/n (by Cauchy-Schwarz, since (sum a)^2 <= n * sum a^2). And the auto-convolution at lag n-1 (the center) is sum a[i]*a[n-1-i], which is the "reverse correlation". If the sequence is symmetric (a[i] = a[n-1-i]), then the center value equals sum a[i]^2.

To minimize the max, we want all lags to be approximately equal. The "average" value of the auto-convolution is 1/(2n-1) (since sum of all lags = (sum a)^2 = 1). So the target is for each lag to be about 1/(2n-1), giving an objective of 2n/(2n-1) ≈ 1.

But this is not achievable with non-negative sequences (the lag-0 value is at least 1/n > 1/(2n-1) for n > 1). So the minimum is determined by the "bottleneck" lag.

For the practical problem (n≈768), the minimum seems to be around 1.5, meaning max(conv) ≈ 1.5/(2*768) ≈ 0.000977. The average is 1/(2*768-1) ≈ 0.000651. So the max is about 50% above the average. This makes sense - we can't make it perfectly flat due to the non-negativity constraint and the boundary effects.

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def optimize_direct(a0, T_schedule, time_limit=None):

    L = 1 << (3 * n - 3).bit_length()
    W = np.fft.rfft(weights, L)
    A = np.fft.rfft(a_rev, L)
    C = np.fft.irfft(W * A, L)
    grad_a = 2.0 * C[n - 1 : n - 1 + n]
    
    return value, grad_a


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = int(np.nonzero(cond)[0][-1])
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def lbfgs_step(a0, T, max_iter=200):
    """Run L-BFGS on softmax parameterization."""
    n = len(a0)
    
    def obj_grad(z):
        a_z = softmax_fn(z)
        val, grad_a = compute_value_and_grad(a_z, T)
        mean = float(np.dot(a_z, grad_a))
        grad_z = a_z * (grad_a - mean)
        return val, grad_z
    
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe / a_safe.sum())
    
    try:
        res = minimize(obj_grad, z0, jac=True, method='L-BFGS-B',
                      options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12,
                               'maxls': 50})
        return softmax_fn(res.x)
    except Exception:
        return a0


def annealed_lbfgs(a0, T_list, max_iter_each=200, time_limit=None):
    """Annealed L-BFGS with increasing temperature."""
    a = a0.copy()
    t0 = time.monotonic()
    for T in T_list:
        if time_limit is not None and (time.monotonic() - t0) > time_limit:
            break
        a = lbfgs_step(a, T, max_iter=max_iter_each)
    return a


def natural_gradient_opt(a0, T, n_iters=2000, time_limit=None):
    """Natural gradient (multiplicative weights) with backtracking line search."""
    a = a0.copy()
    t0 = time.monotonic()
    lr = 0.3
    
    for it in range(n_iters):
        if time_limit is not None and (time.monotonic() - t0) > time_limit:
            break
        val, grad = compute_value_and_grad(a, T)
        accepted = False
        for _ in range(40):
            a_new = a * np.exp(-lr * grad)
            s = a_new.sum()
            if s < 1e-300:
                lr *= 0.5
                continue
            a_new /= s
            val_new, _ = compute_value_and_grad(a_new, T)
            if val_new < val - 1e-18:
                a = a_new
                accepted = True
                break
            lr *= 0.5
        if not accepted:
            # Try a random perturbation
            rng = np.random.default_rng(it * 12345)
            a = project_simplex(a + rng.standard_normal(len(a)) * 0.001)
            lr = 0.3
        else:
            lr = min(lr * 1.02, 3.0)
    
    return a


def peak_shave_optimize(a0, time_limit):
    """
    Iterative peak shaving using subgradient at the active peak.
    Projects onto simplex after each step.
    """
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_val = float(fast_convolve(a).max())
    t0 = time.monotonic()
    lr = 0.05
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        val = float(conv[k_star])
        
        # Subgradient: grad[i] = 2 * a[k_star - i] for valid i
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        grad = np.zeros(n)
        for i in range(js, je + 1):
            grad[i] = 2.0 * a[k_star - i]
        
        improved = False
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            step = lr * frac
            a_new = project_simplex(a - step * grad)
            val_new = float(fast_convolve(a_new).max())
            if val_new < val:
                a = a_new
                improved = True
                if val_new < best_val:
                    best_val = val_new
                    best_a = a.copy()
                    stagnant = 0
                break
        if not improved:
            stagnant += 1
            lr *= 0.95
            if stagnant > 100:
                # Perturb and reset
                rng = np.random.default_rng(stagnant * 7919)
                a = project_simplex(a + rng.standard_normal(n) * 0.002)
                lr = 0.05
                stagnant = 0
    
    return best_a, best_val


def generate_starts(n, parent, rng):
    """Generate diverse initializations for a given length n."""
    starts = []
    x = np.arange(n, dtype=np.float64)
    
    # Resample parent
    if parent is not None and len(parent) > 1:
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        a_new = np.interp(x_new, x_old, parent)
        a_new = np.maximum(a_new, 0)
        if a_new.sum() > 1e-15:
            starts.append(a_new / a_new.sum())
    
    # Uniform
    starts.append(np.ones(n) / n)
    
    # Cosine taper
    w = 0.5 * (1.0 + np.cos(np.pi * x / max(n - 1, 1)))
    w = np.maximum(w, 1e-6)
    starts.append(w / w.sum())
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - n / 2) / (n / 4)) ** 2)
    starts.append(g / g.sum())
    
    # Two peaks
    g1 = np.exp(-0.5 * ((x - 0.25 * n) / (n / 10)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75 * n) / (n / 10)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())
    
    # Edge concentration
    dist = np.minimum(x, (n - 1) - x)
    w = 1.0 / (dist + 1.0)
    starts.append(w / w.sum())
    
    # Random Dirichlet with various concentrations
    for alpha in [0.1, 0.3, 1.0, 5.0]:
        starts.append(rng.dirichlet(np.ones(n) * alpha))
    
    # Periodic
    w = 1.0 + 0.5 * np.sin(2 * np.pi * x / 8.0)
    w = np.maximum(w, 1e-6)
    starts.append(w / w.sum())
    
    # Alternating
    w = np.ones(n)
    w[::2] *= 1.5
    starts.append(w / w.sum())
    
    return starts


def resample(a, new_n):
    """Resample sequence to new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s
