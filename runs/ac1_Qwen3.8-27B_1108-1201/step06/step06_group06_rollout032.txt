def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""

    # FFT size >= 3n-2 (length of conv(rev(a), p))

    conv_result = np.fft.irfft(U * P, N)
    r = conv_result[n-1:2*n-1]  # This gives r[0..n-1]
    
    return 2.0 * r


def compute_grad_smoothed_fast(a, p):
    """Fast gradient computation using FFT.
    grad[i] = 2 * sum_j a[j] * p[i+j]
    """
    n = len(a)
    N = 1 << (3 * n - 2 - 1).bit_length()
    u = a[::-1]
    U = np.fft.rfft(u, N)
    P = np.fft.rfft(p, N)
    conv_result = np.fft.irfft(U * P, N)
    r = conv_result[n - 1:2 * n - 1]
    return 2.0 * r


def softmax_smooth(conv, T):
    """Compute softmax(conv/T) with numerical stability."""
    x = conv / T
    x -= x.max()
    p = np.exp(x)
    s = p.sum()
    if s < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return p / s

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_smoothed(a, p):
    """Compute gradient of sum_k p[k]*(a*a)[k] w.r.t. a.

    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        improved = False

        # Random coordinate pair transfers
        for _ in range(200):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j or a[i] < 1e-10:
                continue
            for frac in [0.01, 0.03, 0.08, 0.15, 0.3]:
                amount = a[i] * frac
                candidate = a.copy()
                candidate[i] -= amount
                candidate[j] += amount
                if candidate[i] < 0:
                    continue
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-12:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Perturbation
        if not improved and it % 30 == 0:
            for scale in [0.0005, 0.001, 0.003, 0.008]:
                pert = rng.standard_normal(n) * scale
                candidate = project_simplex(a + pert)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-12:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and it > 300 and (time.monotonic() - t0) > time_limit * 0.85:
            break

    return best_a, best_obj


def try_seq(a, label, best_ref):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_ref
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_ref[1]:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return (a_norm.copy(), score)
    return best_ref


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: Nesterov + Adam smoothed gradient, coordinate descent,
    multi-n exploration, multiple restarts with diverse hyperparameters.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Uniform init: n=768, score={best_score:.8f}")

    n_parent = len(parent)
    print(f"Starting: n={n_parent}, score={best_score:.8f}")

    # === Phase 1: Nesterov with multiple hyperparams ===
    print("\n=== Phase 1: Nesterov Smoothed Gradient ===")
    t_phase = time.monotonic()

    nest_configs = [
        (0.002, 0.03, 0.85),
        (0.005, 0.04, 0.85),
        (0.008, 0.03, 0.9),
        (0.003, 0.05, 0.8),
        (0.012, 0.02, 0.9),
        (0.001, 0.02, 0.95),
    ]

    for T_init, lr_init, beta in nest_configs:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        t_alloc = min(remaining * 0.12, 200)
        print(f"  Nesterov T={T_init}, lr={lr_init}, beta={beta}, t={t_alloc:.0f}s")
        a_opt, _ = nesterov_smoothed_grad(best_seq, time_limit=t_alloc,
                                          T_init=T_init, lr_init=lr_init,
                                          beta_start=beta)
        best_seq, best_score = try_seq(a_opt, f"nest_T{T_init}", (best_seq, best_score))

    # Adam
    print("  --- Adam ---")
    adam_configs = [
        (0.003, 0.004),
        (0.005, 0.003),
        (0.008, 0.002),
        (0.002, 0.006),
        (0.001, 0.008),
    ]

    for T_init, lr_init in adam_configs:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        t_alloc = min(remaining * 0.10, 150)
        print(f"  Adam T={T_init}, lr={lr_init}, t={t_alloc:.0f}s")
        a_opt, _ = adam_smoothed_grad(best_seq, time_limit=t_alloc,
                                      T_init=T_init, lr_init=lr_init)
        best_seq, best_score = try_seq(a_opt, f"adam_T{T_init}", (best_seq, best_score))

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Multi-n exploration ===
    print("\n=== Phase 2: Multi-n Exploration ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)

    n_candidates = sorted(set(
        n_best + d for d in [-128, -64, -32, -16, -8, 8, 16, 32, 64, 128, 256, 512]
        if 128 <= n_best + d <= 3072
    ))
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
        n_candidates.sort()

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        if n_try == n_best:
            start = best_seq.copy()
        else:
            src = best_seq if len(best_seq) == n_best else parent
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, len(src)), src)
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try

        t_alloc = min(remaining * 0.06, 60)
        a_opt, _ = nesterov_smoothed_grad(start, time_limit=t_alloc,
                                          T_init=0.005, lr_init=0.03)
        best_seq, best_score = try_seq(a_opt, f"n{n_try}_nest", (best_seq, best_score))

        remaining2 = deadline - time.monotonic()
        if remaining2 > 40 and n_try != n_best:
            t_alloc2 = min(remaining2 * 0.04, 30)
            a_opt2, _ = adam_smoothed_grad(a_opt, time_limit=t_alloc2,
                                           T_init=0.003, lr_init=0.004)
            best_seq, best_score = try_seq(a_opt2, f"n{n_try}_adam", (best_seq, best_score))

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Deep refinement on best ===
    print("\n=== Phase 3: Deep Refinement ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)

    # Long Nesterov
    remaining = deadline - time.monotonic()
    if remaining > 100:
        t_alloc = min(remaining * 0.22, 350)
        print(f"  Deep Nesterov T=0.001, lr=0.008, t={t_alloc:.0f}s")
        a_opt, _ = nesterov_smoothed_grad(best_seq, time_limit=t_alloc,
                                          T_init=0.001, lr_init=0.008, beta_start=0.95)
        best_seq, best_score = try_seq(a_opt, "deep_nest", (best_seq, best_score))

    # Deep Adam
    remaining = deadline - time.monotonic()
    if remaining > 80:
        t_alloc = min(remaining * 0.15, 200)
        print(f"  Deep Adam T=0.002, lr=0.001, t={t_alloc:.0f}s")
        a_opt, _ = adam_smoothed_grad(best_seq, time_limit=t_alloc,
                                      T_init=0.002, lr_init=0.001)
        best_seq, best_score = try_seq(a_opt, "deep_adam", (best_seq, best_score))

    # Perturbed restarts
    for scale in [0.002, 0.005, 0.01, 0.02]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc = min(remaining * 0.08, 80)
        a_opt, _ = nesterov_smoothed_grad(f_pert, time_limit=t_alloc,
                                          T_init=0.003, lr_init=0.03)
        best_seq, best_score = try_seq(a_opt, f"pert_{scale}", (best_seq, best_score))

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Coordinate descent polish ===
    print("\n=== Phase 4: Coordinate Descent Polish ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)

    remaining = deadline - time.monotonic()
    if remaining > 50:
        t_alloc = min(remaining * 0.25, 300)
        print(f"  Coord descent t={t_alloc:.0f}s")
        a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_alloc, seed=12345)
        best_seq, best_score = try_seq(a_opt, "coord1", (best_seq, best_score))

    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_alloc = min(remaining * 0.15, 100)
        print(f"  Coord descent 2 t={t_alloc:.0f}s")
        a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_alloc, seed=99999)
        best_seq, best_score = try_seq(a_opt, "coord2", (best_seq, best_score))

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final fine-tuning ===
    print("\n=== Phase 5: Final Fine-tuning ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)

    # Very low temperature Nesterov
    remaining = deadline - time.monotonic()
    if remaining > 40:
        t_alloc = min(remaining * 0.3, 200)
        print(f"  Fine Nesterov T=0.0005, t={t_alloc:.0f}s")
        a_opt, _ = nesterov_smoothed_grad(best_seq, time_limit=t_alloc,
                                          T_init=0.0005, lr_init=0.003, beta_start=0.98)
        best_seq, best_score = try_seq(a_opt, "fine_nest", (best_seq, best_score))

    # Very low temperature Adam
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_alloc = min(remaining * 0.2, 100)
        print(f"  Fine Adam T=0.0005, t={t_alloc:.0f}s")
        a_opt, _ = adam_smoothed_grad(best_seq, time_limit=t_alloc,
                                      T_init=0.0005, lr_init=0.0005)
        best_seq, best_score = try_seq(a_opt, "fine_adam", (best_seq, best_score))

    # Final coord descent
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t_alloc = min(remaining * 0.3, 80)
        a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_alloc, seed=77777)
        best_seq, best_score = try_seq(a_opt, "coord_final", (best_seq, best_score))

    # Micro perturbations
    remaining = deadline - time.monotonic()
    if remaining > 10:
        for _ in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 3:
                break
            noise = rng.standard_normal(n_best) * 0.0003
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = nesterov_smoothed_grad(f_pert, time_limit=min(remaining * 0.3, 10),
                                              T_init=0.0003, lr_init=0.002)
            best_seq, best_score = try_seq(a_opt, "micro", (best_seq, best_score))

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={total_time:.0f}s")
    return [float(x) for x in best_seq]
