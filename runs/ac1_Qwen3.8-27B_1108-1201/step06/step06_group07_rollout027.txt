import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.special import softmax
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def obj_value(a):
    """Compute the objective (normalized: sum(a)=1 assumed)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

        w = pw[idx]
        if w < 1e-12:

        # grad[j] += w * 2 * a[k-j] for j in [js, je]
        # a[k-j] for j in [js, je] is a[k-je : k-js+1][::-1]
        grad[js:je+1] += w * 2.0 * a[k-je:k-js+1][::-1]
    return grad

def asfw_step(a, grad, support):
    """
    Away-Step Frank-Wolfe: find best direction in conv(support ∪ {e_min}).
    Returns (direction, step_fraction) where new_a = a + step_fraction * direction.
    """
    n = len(a)
    # Find e_min (vertex with minimum gradient)
    j_min = int(np.argmin(grad))
    
    # Away-step: find coordinate in support to remove
    # Solve: min over v in conv(support ∪ {e_min}) of <grad, v>
    # This is equivalent to: min over alpha >= 0, sum alpha = 1 of <grad, sum alpha_s * e_s + alpha_min * e_min>
    # where s ranges over support and alpha can be negative for away-steps
    
    # Actually, the standard away-step FW solves:
    # min_{v in v-swap simplex} <grad, v>
    # where the v-swap simplex is the convex hull of the support vertices
    # allowing one vertex to be replaced.
    
    # Simpler approach: try forward step (toward e_min) and away step (away from each support element)
    # and pick the best one.
    
    best_dir = None
    best_val = np.inf
    
    # Forward step: move toward e_min
    if a[j_min] < 1.0 - 1e-12:
        dir_f = np.zeros(n)
        dir_f[j_min] = 1.0
        # Subtract mass from support proportionally
        sup_vals = a[support]
        sup_sum = sup_vals.sum()
        if sup_sum > 1e-12:
            dir_f[support] -= a[support] / sup_sum  # Move mass proportionally
        val = np.dot(grad, dir_f)
        if val < best_val:
            best_val = val
            best_dir = dir_f
    
    # Away steps: for each s in support, try moving away from e_s
    for s in support:
        if a[s] < 1e-8:
            continue
        dir_a = np.zeros(n)
        dir_a[s] = -1.0
        # Add the freed mass to e_min (or distribute among support \ {s})
        dir_a[j_min] = 1.0
        val = np.dot(grad, dir_a)
        if val < best_val:
            best_val = val
            best_dir = dir_a
    
    if best_dir is None:
        return None, 0.0
    return best_dir, best_val

def asfw_optimize(a0, time_limit, verbose=False):
    """Away-Step Frank-Wolfe optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = obj_value(a)
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        grad = get_subgradient(a, conv, top_k=20, temperature=0.01)
        
        # Maintain support (non-zero coordinates)
        support = np.where(a > 1e-10)[0]
        if len(support) == 0:
            support = np.arange(n)
        
        dir, _ = asfw_step(a, grad, support)
        if dir is None:
            # Fallback: use full simplex gradient step
            j_min = int(np.argmin(grad))
            dir = np.zeros(n)
            dir[j_min] = 1.0
            dir[a > 0] -= 1.0 / np.sum(a > 0)
        
        # Line search
        best_alpha = 0.0
        best_val = obj_value(a)
        alphas = [1.0/np.sqrt(it+1), 1.0/(it+1), 0.5, 0.3, 0.1]
        for alpha in alphas:
            cand = a + alpha * dir
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-12:
                continue
            cand = cand / cs
            val = obj_value(cand)
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-12:
            a = a + best_alpha * dir
            a = np.maximum(a, 0)
            a = a / a.sum()
            obj = obj_value(a)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 50:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            no_improve = 0
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ASF it={it}, obj={best_obj:.10f}, t={elapsed:.0f}s")
    
    return best_a, best_obj

def projected_subgrad_optimize(a0, time_limit, verbose=False):
    """Projected subgradient with adaptive step size and softmax weighting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = obj_value(a)

        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        grad = get_subgradient(a, conv, top_k=top_k, temperature=temp)
        
        # Projected update
        a_new = project_simplex(a - lr * grad)
        
        # Adaptive step size: try multiple step sizes
        best_val = best_obj
        best_candidate = a
        for alpha in [lr, lr*0.5, lr*2, lr*0.1]:
            cand = (1-alpha) * a + alpha * a_new
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-12:
                continue
            cand = cand / cs
            val = obj_value(cand)
            if val < best_val:
                best_val = val
                best_candidate = cand
        
        a = best_candidate
        if a is best_candidate and best_val < best_obj - 1e-12:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        elif best_val < best_obj - 1e-12:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt parameters
        if no_improve > 20:
            lr *= 0.9
            temp *= 1.5  # Make softmax less peaked (more uniform)
        if no_improve > 100:
            top_k = min(top_k + 5, 50)
        if lr < 1e-5:
            lr = 0.1
            no_improve = 0
            temp = 0.01
            top_k = 20
        
        if no_improve > 200:
            rng = np.random.default_rng((it * 31831 + 7) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)

            print(f"    PSG it={it}, obj={best_obj:.10f}, lr={lr:.6f}, t={elapsed:.0f}s")
    
    return best_a, best_obj

def coordinate_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate i, find the optimal value
    given all other coordinates fixed, while maintaining sum=1.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = obj_value(a)
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            if a[i] < 1e-12:
                continue
            
            # Try moving mass from i to each other coordinate j
            # For efficiency, only try a few candidates
            # Find coordinates where it's beneficial to move mass
            conv = fast_convolve(a)
            k_peak = int(np.argmax(conv))
            
            # Gradient for the peak lag
            grad_i = 2.0 * a[k_peak - i] if 0 <= k_peak - i < n else 0.0
            
            # Candidate targets: coordinates with low gradient contribution
            # (i.e., coordinates where adding mass won't increase the peak much)
            candidates = []
            for j in range(n):
                if j == i:
                    continue
                grad_j = 2.0 * a[k_peak - j] if 0 <= k_peak - j < n else 0.0
                candidates.append((grad_j, j))
            
            # Sort by gradient (prefer moving mass to low-gradient coordinates)
            candidates.sort()
            top_candidates = [c[1] for c in candidates[:5]]
            
            for j in top_candidates:
                # Try moving delta from i to j
                best_delta = 0
                best_val = obj_value(a)
                for frac in [0.01, 0.05, 0.1, 0.2, 0.5]:
                    delta = a[i] * frac
                    if delta <= 0:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= delta
                    a_trial[j] += delta
                    if a_trial[i] < 0:
                        continue
                    val = obj_value(a_trial)
                    if val < best_val - 1e-12:
                        best_val = val
                        best_delta = delta
                
            if best_delta > 0:
                a[i] -= best_delta
                a[top_candidates[0]] += best_delta
                a = np.maximum(a, 0)
                a = a / a.sum()
                improved = True
                obj = obj_value(a)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        
        if no_improve > 5:
            break
        
        if verbose:
            print(f"    CD: obj={best_obj:.10f}, t={time.monotonic()-t0:.0f}s")
    
    return best_a, best_obj

def build_lp_matrix(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

def sca_optimize(f0, time_limit, verbose=False):
    """SCA with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    row_idx, col_idx = build_lp_matrix(n)
    k_arr = row_idx
    i_arr = col_idx
    vals = f[k_arr - i_arr]
    A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        try:
            res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 10})
            if res.success and res.x is not None:
                g = np.maximum(res.x, 0.0)
                gs = g.sum()
                if gs > 1e-15:
                    g = g / gs
                    # Line search
                    best_val = fast_convolve(f).max()
                    best_ft = f.copy()
                    for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
                        ft = (1 - alpha) * f + alpha * g
                        ft = np.maximum(ft, 0)
                        st = ft.sum()
                        if st < 1e-15:
                            continue
                        ft = ft / st
                        val = fast_convolve(ft).max()
                        if val < best_val:
                            best_val = val
                            best_ft = ft.copy()
                    
                    if best_val < best_obj - 1e-14:
                        f = best_ft
                        best_obj = best_val
                        best_f = f.copy()
                        M = best_val * 0.999
                        no_improve = 0
                    else:
                        no_improve += 1
                        M *= 0.998
                else:
                    no_improve += 1
            else:
                no_improve += 1
                M *= 1.01
        except Exception:
            no_improve += 1
        
        if no_improve > 30:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            M = fast_convolve(f).max() * 1.01
            no_improve = 0
            # Update LP matrix
            vals = f[k_arr - i_arr]
            A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Projected subgradient with adaptive softmax-weighted active set
- Away-step Frank-Wolfe for accelerated convergence
- SCA (Successive Convex Approximation) with LP oracle for polishing
- Multi-n exploration with diverse initializations
- Perturbed iterations with adaptive restarts
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.special import softmax
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute 2*n*max(conv)/(sum)^2 for the given sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return float(2.0 * n * conv.max())


def get_subgradient(a, conv, top_k=25, temperature=0.008):
    """
    Compute a subgradient of max_k (a*a)[k] using softmax-weighted
    combination of top-k active lag gradients.
    """
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        weights = np.ones(len(peaks)) / len(peaks)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        max_pv = pv.max()
        if max_pv < 1e-15:
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            weights = softmax(pv / (temperature * max_pv + 1e-12))

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        w = weights[idx]
        if w < 1e-12:
            continue
        k = int(peaks[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        grad[js:je + 1] += w * 2.0 * a[k - je:k - js + 1][::-1]
    return grad


def psg_optimize(a0, time_limit, temp_init=0.008, top_k_init=25, lr_init=0.08, verbose=False):
    """Projected subgradient with adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = lr_init
    temp = temp_init
    top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        grad = get_subgradient(a, conv, top_k=top_k, temperature=temp)

        # Projected gradient step
        a_new = project_simplex(a - lr * grad)

        # Adaptive line search with multiple candidates
        best_val = best_obj
        best_candidate = a
        step_fracs = [1.0, 0.5, 0.25, 0.1, 2.0]
        for frac in step_fracs:
            alpha = lr * frac
            cand = (1.0 - alpha) * a + alpha * a_new
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs < 1e-12:
                continue
            cand = cand / cs
            val = compute_obj(cand)
            if val < best_val:
                best_val = val
                best_candidate = cand

        a = best_candidate
        if best_val < best_obj - 1e-13:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt parameters based on progress
        if no_improve == 10:
            lr *= 0.85
            temp *= 1.3
        elif no_improve == 30:
            lr *= 0.7
            temp *= 1.5
            top_k = min(top_k + 5, 60)
        elif no_improve == 80:
            lr *= 0.5
            temp = min(temp * 2.0, 0.1)
            top_k = min(top_k + 10, 100)
        if lr < 1e-6:
            lr = lr_init * 0.3
            temp = temp_init
            top_k = top_k_init
            no_improve = 0

        # Periodic perturbation to escape local minima
        if no_improve > 150 or (no_improve > 80 and it % 50 == 0):
            rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
            scale = 0.003 + 0.004 * min(1.0, no_improve / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            no_improve = 0
            lr = lr_init
            temp = temp_init
            top_k = top_k_init

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PSG it={it}, obj={best_obj:.10f}, lr={lr:.6f}, tk={top_k}, t={elapsed:.0f}s")

    return best_a, best_obj


def asfw_optimize(a0, time_limit, verbose=False):
    """Away-step Frank-Wolfe with backtracking."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        grad = get_subgradient(a, conv, top_k=20, temperature=0.01)

        # Find best Frank-Wolfe direction
        j_min = int(np.argmin(grad))
        support = np.where(a > 1e-8)[0]

        best_dir = None
        best_dir_val = np.inf

        # Forward step toward e_min
        if len(support) > 0:
            dir_f = np.zeros(n)
            dir_f[j_min] = 1.0
            sup_sum = a[support].sum()
            if sup_sum > 1e-12:
                dir_f[support] = -a[support] / sup_sum
            val_f = np.dot(grad, dir_f)
            if val_f < best_dir_val:
                best_dir_val = val_f
                best_dir = dir_f

        # Away steps
        for s_idx in support:
            if a[s_idx] < 1e-6:
                continue
            dir_a = np.zeros(n)
            dir_a[s_idx] = -1.0
            dir_a[j_min] = 1.0
            val_a = np.dot(grad, dir_a)
            if val_a < best_dir_val:
                best_dir_val = val_a
                best_dir = dir_a

        if best_dir is None:
            best_dir = np.zeros(n)
            best_dir[j_min] = 1.0
            best_dir[a > 0] -= 1.0 / max(1, np.sum(a > 0))

        # Backtracking line search
        best_alpha = 0.0
        best_val = best_obj
        alpha = 2.0 / (it + 1)
        for _ in range(10):
            cand = a + alpha * best_dir
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs < 1e-12:
                alpha *= 0.5
                continue
            cand = cand / cs
            val = compute_obj(cand)
            if val < best_val:
                best_val = val
                best_alpha = alpha
                # Try to increase alpha
                if alpha < 2.0:
                    alpha = min(alpha * 1.5, 2.0)
                else:
                    break
            else:
                alpha *= 0.5

        if best_alpha > 1e-10:
            a = a + best_alpha * best_dir
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            obj = compute_obj(a)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            no_improve = 0

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ASF it={it}, obj={best_obj:.10f}, t={elapsed:.0f}s")

    return best_a, best_obj


def build_lp_sparsity(n):
    """Precompute sparsity pattern for LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def sca_optimize(f0, time_limit, verbose=False):
    """SCA optimization with adaptive M and LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    row_idx, col_idx = build_lp_sparsity(n)
    k_arr = row_idx
    i_arr = col_idx

    conv = fast_convolve(f)
    M = conv.max() * 1.001
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        # Build LP with current f
        vals = f[k_arr - i_arr]
        A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        g = None
        try:
            res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': 15})
            if res.success and res.x is not None:
                g_candidate = np.maximum(res.x, 0.0)
                if g_candidate.sum() > 1e-12:
                    g = g_candidate / g_candidate.sum()
        except Exception:
            pass

        if g is None:
            M *= 1.02
            no_improve += 1
            if no_improve > 40:
                M = fast_convolve(f).max() * 1.005
                no_improve = 0
            continue

        # Line search
        best_val = fast_convolve(f).max()
        best_ft = f.copy()
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-12:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_ft = ft.copy()

        if best_val < best_obj - 1e-14:
            f = best_ft
            best_obj = best_val
            best_f = f.copy()
            M = best_val * 0.999
            no_improve = 0
        else:
            no_improve += 1
            M *= 0.998

        if no_improve > 50:
            rng = np.random.default_rng((it * 1299709 + 3) % (2**31 - 1))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            M = fast_convolve(f).max() * 1.005
            no_improve = 0

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SCA it={it}, obj={best_obj:.10f}, M={M:.10f}, t={elapsed:.0f}s")

    return best_f, best_obj


def peak_directed_redistribution(a0, time_limit, verbose=False):
    """
    Targeted peak reduction: for the dominant peak lag, find the best
    redistribution of mass from high-contributing to low-contributing positions.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Compute contribution of each position to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = np.zeros(n)
        for j in range(n):
            partner = k_peak - j
            if 0 <= partner < n:
                contrib[j] = a[j] * a[partner]

        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break

        # Find best target: position with minimum contribution to peak
        # and maximum "safety margin" (low values in other peaks)
        min_contrib = contrib.min()
        max_contrib = contrib.max()

        if max_contrib - min_contrib < 1e-12:
            no_improve += 1
        else:
            # Move mass from high-contrib to low-contrib positions
            source_candidates = np.where(contrib > 0.5 * max_contrib)[0]
            target_candidates = np.where(contrib < 0.3 * max_contrib)[0]

            if len(source_candidates) > 0 and len(target_candidates) > 0:
                best_move = None
                best_move_val = best_obj

                for src in source_candidates[:5]:
                    for tgt in target_candidates[:5]:
                        if src == tgt:
                            continue
                        for frac in [0.05, 0.1, 0.2, 0.3]:
                            delta = a[src] * frac
                            if delta < 1e-12:
                                continue
                            a_trial = a.copy()
                            a_trial[src] -= delta
                            a_trial[tgt] += delta
                            a_trial = np.maximum(a_trial, 0.0)
                            a_trial = a_trial / a_trial.sum()
                            val = compute_obj(a_trial)
                            if val < best_move_val - 1e-12:
                                best_move_val = val
                                best_move = (src, tgt, delta)

                if best_move is not None:
                    src, tgt, delta = best_move
                    a[src] -= delta
                    a[tgt] += delta
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    if best_move_val < best_obj - 1e-12:
                        best_obj = best_move_val
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if no_improve > 30:
            rng = np.random.default_rng((it * 9973 + 5) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            no_improve = 0

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PDR it={it}, obj={best_obj:.10f}, t={elapsed:.0f}s")

    return best_a, best_obj


def interpolated_init(parent, n_target, rng, mode="linear"):
    """Create an initialization of length n_target from a parent sequence."""
    if mode == "linear":
        x_target = np.linspace(0, 1, n_target)
        x_parent = np.linspace(0, 1, len(parent))
        result = np.interp(x_target, x_parent, parent)
    elif mode == "cubic":
        from scipy.interpolate import CubicSpline
        x_parent = np.linspace(0, 1, len(parent))
        cs = CubicSpline(x_parent, parent, bc_type='natural')
        x_target = np.linspace(0, 1, n_target)
        result = cs(x_target)
    else:
        result = np.interp(np.linspace(0, 1, n_target),
                           np.linspace(0, 1, len(parent)), parent)

    result = np.maximum(result, 0.0)
    s = result.sum()
    if s < 1e-12:
        result = np.ones(n_target) / n_target
    else:
        result = result / s
    return result


def diverse_inits(n, parent, rng):
    """Generate multiple diverse initializations of length n."""
    inits = []

    # Interpolated parent
    inits.append(interpolated_init(parent, n, rng, "linear"))

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge-heavy
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(x - center) / center
    edge_w = 1.0 / (dist**1.5 + 0.1)
    inits.append(edge_w / edge_w.sum())

    # Sine-modulated
    t = x / max(1, n - 1)
    sine_w = (1.0 + 0.4 * np.sin(2 * np.pi * t + 0.5)) ** 2 + 0.05
    inits.append(sine_w / sine_w.sum())

    # Two-peak
    g2 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2) + 0.01
    inits.append(g2 / g2.sum())

    # Perturbed parent
    px = interpolated_init(parent, n, rng, "linear")
    px = px + rng.standard_normal(n) * 0.02
    px = np.maximum(px, 0.0)
    px = px / px.sum()
    inits.append(px)

    return inits

                a_opt, _ = sca_optimize(state["best_seq"], time_limit=t_alloc)
                try_seq_local(a_opt, "sca")

            remaining = total_deadline - time.monotonic() - 130
            if remaining > 20:
                t_alloc2 = min(remaining, 100)
                a_opt2, _ = sca_optimize(state["best_seq"], time_limit=t_alloc2)
                try_seq_local(a_opt2, "sca2")
        else:
            print(f"  Skipping SCA (n={n_best} too large)")

    print(f"  P3 done: best={state['best_score']:.8f}, t={time.monotonic()-t3:.0f}s")

    # === Phase 4: Peak-directed redistribution (200s) ===
    print("\n=== Phase 4: Peak Directed ===")
    t4 = time.monotonic()

    if state["best_seq"] is not None:
        remaining = total_deadline - time.monotonic() - 150
        if remaining > 20:
            t_alloc = min(remaining, 120)
            print(f"  PDR n={len(state['best_seq'])}, t={t_alloc:.0f}s")
            a_opt, _ = peak_directed_redistribution(state["best_seq"], time_limit=t_alloc)
            try_seq_local(a_opt, "pdr")

            remaining = total_deadline - time.monotonic() - 80
            if remaining > 15:
                t_alloc2 = min(remaining, 60)
                a_opt2, _ = psg_optimize(state["best_seq"], time_limit=t_alloc2,
                                         temp_init=0.005, top_k_init=40, lr_init=0.04)
                try_seq_local(a_opt2, "psg_pdr")

    print(f"  P4 done: best={state['best_score']:.8f}, t={time.monotonic()-t4:.0f}s")

    # === Phase 5: Diversified restarts (150s) ===
    print("\n=== Phase 5: Restarts ===")
    t5 = time.monotonic()

    if state["best_seq"] is not None:
        n_best = len(state["best_seq"])
        scales = [0.003, 0.007, 0.012, 0.02, 0.035]
        for scale in scales:
            remaining = total_deadline - time.monotonic() - 80
            if remaining < 20:
                break
            f_pert = project_simplex(state["best_seq"] + rng.standard_normal(n_best) * scale)
            t_alloc = min(18, remaining - 40)
            if t_alloc < 5:
                break
            print(f"  Restart scale={scale}, t={t_alloc:.0f}s")
            a_opt, _ = psg_optimize(f_pert, time_limit=t_alloc,
                                    temp_init=0.01, top_k_init=25, lr_init=0.08)
            try_seq_local(a_opt, f"restart_{scale}")

        # Also try different n
        for n_alt in [int(n_best * 0.8), int(n_best * 1.25)]:
            if n_alt < 256 or n_alt > 2560:
                continue
            remaining = total_deadline - time.monotonic() - 60
            if remaining < 20:
                break
            start = interpolated_init(state["best_seq"], n_alt, rng, "linear")
            t_alloc = min(15, remaining - 30)
            if t_alloc < 5:
                break
            print(f"  Restart n={n_alt}, t={t_alloc:.0f}s")
            a_opt, _ = psg_optimize(start, time_limit=t_alloc)
            try_seq_local(a_opt, f"restart_n{n_alt}")

    print(f"  P5 done: best={state['best_score']:.8f}, t={time.monotonic()-t5:.0f}s")

    # === Phase 6: Final polish (remaining) ===
    print("\n=== Phase 6: Final Polish ===")
    t6 = time.monotonic()

    if state["best_seq"] is not None:
        remaining = total_deadline - time.monotonic()
        if remaining > 30:
            t_alloc = min(remaining - 10, 50)
            print(f"  Final PSG t={t_alloc:.0f}s")
            a_opt, _ = psg_optimize(state["best_seq"], time_limit=t_alloc,
                                    temp_init=0.003, top_k_init=50, lr_init=0.02)
            try_seq_local(a_opt, "final_psg")

        remaining = total_deadline - time.monotonic()
        if remaining > 15:
            t_alloc = min(remaining - 5, 20)
            print(f"  Final ASF t={t_alloc:.0f}s")
            a_opt, _ = asfw_optimize(state["best_seq"], time_limit=t_alloc)
            try_seq_local(a_opt, "final_asf")

    print(f"  P6 done: best={state['best_score']:.8f}, t={time.monotonic()-t6:.0f}s")

    # Fallback
    if state["best_seq"] is None:
        state["best_seq"] = np.ones(768) / 768
        state["best_score"] = float(evaluate_sequence(state["best_seq"].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state["best_seq"].tolist()))
    if np.isfinite(fs) and fs < state["best_score"]:
        state["best_score"] = fs

    print(f"\nFINAL: {state['best_score']:.8f} (n={len(state['best_seq'])})")
    return [float(x) for x in state["best_seq"]]
