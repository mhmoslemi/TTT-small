    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:

    u = np.sort(v)[::-1]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed gradient descent
with log-sum-exp approximation, adaptive annealing, and peak-targeted
coordinate refinement.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_obj_and_grad(a, T):
    """Compute smoothed objective F_T and its gradient."""
    c = fast_convolve(a)
    cmax = c.max()
    exp_shifted = np.exp((c - cmax) / T)
    log_sum_exp = np.log(exp_shifted.sum())
    obj = cmax + T * log_sum_exp
    p = exp_shifted / exp_shifted.sum()
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return obj, grad, c, p


def smoothed_gradient_descent(a_init, time_limit, T_start=None, T_end=None):
    """Smoothed gradient descent with adaptive annealing."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    if T_start is None:
        T_start = 3.0 / n
    if T_end is None:
        T_end = 0.01 / n
    
    T = T_start
    lr = 0.5
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    best_smoothed = float('inf')
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.3:
            break
        
        obj, grad, c, p = compute_smoothed_obj_and_grad(a, T)
        
        # Backtracking line search
        step = lr
        found = False
        for _ in range(25):
            a_new = project_simplex(a - step * grad)
            c_new = fast_convolve(a_new)
            cmax_new = c_new.max()
            exp_new = np.exp((c_new - cmax_new) / T)
            obj_new = cmax_new + T * np.log(exp_new.sum())
            if obj_new < obj - 1e-15:
                a = a_new
                found = True
                break
            step *= 0.5
        
        if not found:
            no_improve += 1
            if no_improve > 50:
                # Anneal T
                T = max(T_end, T * 0.9)
                no_improve = 0
                if T <= T_end * 1.01:
                    # Try perturbation
                    rng = np.random.default_rng(it * 2654435761 + 1)
                    a = project_simplex(a + rng.standard_normal(n) * 0.002)
        else:
            no_improve = 0
        
        # Check actual max (not smoothed)
        actual_max = fast_convolve(a).max()
        if actual_max < best_obj:
            best_obj = actual_max
            best_a = a.copy()
        
        # Periodic annealing
        if it % 200 == 0:
            T = max(T_end, T * 0.97)
    
    return best_a, best_obj

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
