Actually, the Hessian H = 2 * sum_k w[k] * Q_k where Q_k[i,j] = 1 if i+j=k (within bounds). This gives:
H[i,j] = 2 * w[i+j] (if i+j ≤ 2n-2)

So H is a Hankel matrix! H[i,j] = 2*w[i+j]. This is great because Hankel matrices have fast multiplication algorithms.

The QP is:
minimize a^T H a (or (1/2) a^T H a)
subject to a ≥ 0, 1^T a = 1

Since H is a Hankel matrix, we can compute H @ a efficiently using FFT:
(H @ a)[i] = sum_j H[i,j] * a[j] = 2 * sum_j w[i+j] * a[j] = 2 * (w * a)[i] (convolution!)

So we can compute the gradient of the QP objective in O(n log n) time using FFT!

This means we can use a fast projected gradient method for the QP subproblem.

Optimization for minimum peak autocorrelation step function.
Uses FFT-accelerated projected gradient with adaptive log-sum-exp smoothing,
Nesterov momentum, coordinate descent jumps, and multi-start exploration.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_matvec(w, a):
    """Compute H @ a where H[i,j] = w[i+j]. Uses FFT for O(n log n) speed."""
    n = len(a)
    ar = a[::-1]
    conv_size = len(w) + n - 1
    fft_sz = 1 << (conv_size - 1).bit_length()
    W = np.fft.rfft(w, fft_sz)
    AR = np.fft.rfft(ar, fft_sz)
    conv = np.fft.irfft(W * AR, fft_sz)[:conv_size]
    return conv[n - 1:2 * n - 1]


def compute_smoothed_grad(a, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*(a*a)_k)) w.r.t. a."""
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()
    grad = 2.0 * hankel_matvec(p, a)
    return grad, conv


def optimize_pg_nesterov(a0, time_limit, beta_init=30.0, beta_max=500.0,
                         momentum_init=0.85, seed=42):
    """Nesterov-accelerated projected gradient with adaptive log-sum-exp."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = a0.copy()
    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)
    beta = beta_init
    momentum = momentum_init

    best_a = a.copy()
    best_peak = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break

        # Nesterov lookahead
        y = project_simplex(a + momentum * v)

        # Gradient at lookahead point
        grad, conv = compute_smoothed_grad(y, beta)
        peak = conv.max()

        if peak < best_peak - 1e-15:
            best_peak = peak
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Step
        v = momentum * v - grad
        a = project_simplex(y + v)

        # Clamp momentum to prevent divergence
        v_norm = np.linalg.norm(v)
        if v_norm > 1.0:
            v = v / v_norm

        # Adapt temperature and momentum
        if no_improve > 100:
            beta = min(beta * 1.05, beta_max)
        if no_improve > 300:
            momentum = max(momentum * 0.99, 0.5)
        if no_improve > 800:
            # Reheat with perturbation
            beta = beta_init
            momentum = momentum_init
            pert_scale = min(0.03, 0.01 * np.sqrt(no_improve / 400.0))
            a = project_simplex(best_a + rng.standard_normal(n) * pert_scale)
            y = a.copy()
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_peak


def optimize_bb_gradient(a0, time_limit, beta_init=20.0, beta_max=300.0, seed=123):
    """Barzilai-Borwein projected gradient with adaptive temperature."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = a0.copy()
    beta = beta_init
    lr = 0.02

    best_a = a.copy()
    best_peak = fast_convolve(a).max()

    s_prev = None
    g_prev = None

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break

        grad, conv = compute_smoothed_grad(a, beta)
        peak = conv.max()

        if peak < best_peak - 1e-15:
            best_peak = peak
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # BB step size
        if s_prev is not None and g_prev is not None:
            dy = grad - g_prev
            sy = np.dot(s_prev, dy)
            ss = np.dot(s_prev, s_prev)
            if sy > 1e-12 and ss > 1e-12:
                lr_new = ss / sy
                lr = np.clip(lr_new, 1e-5, 0.5)

        # Backtracking line search
        lr_trial = lr
        a_new = a
        for _ in range(8):
            a_new = project_simplex(a - lr_trial * grad)
            conv_new = fast_convolve(a_new)
            if conv_new.max() <= peak + 1e-12 or lr_trial < 1e-6:
                break
            lr_trial *= 0.5

        s_prev = a_new - a
        g_prev = grad
        a = a_new

        # Adapt
        if no_improve > 150:
            beta = min(beta * 1.03, beta_max)
        if no_improve > 600:
            beta = beta_init
            a = project_simplex(best_a + rng.standard_normal(n) * 0.008)
            s_prev = None
            g_prev = None
            no_improve = 0

    return best_a, best_peak


def coordinate_descent_jumps(a0, time_limit, seed=999, max_rounds=5000):
    """Coordinate descent with greedy moves to escape local minima."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_peak = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0

    for it in range(max_rounds):
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # Pick two random coordinates to swap mass
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        # Try moving mass from i to j at various fractions
        improved = False
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
            amount = a[i] * frac
            if amount < 1e-12:
                continue
            candidate = a.copy()
            candidate[i] -= amount
            candidate[j] += amount
            candidate = np.maximum(candidate, 0)
            ss = candidate.sum()
            if ss < 1e-10:
                continue
            candidate /= ss
            conv = fast_convolve(candidate)
            if conv.max() < best_peak - 1e-14:
                a = candidate
                best_peak = conv.max()
                best_a = a.copy()
                no_improve = 0
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 200:
                # Larger random jump
                pert = rng.standard_normal(n) * 0.02
                candidate = project_simplex(a + pert)
                conv = fast_convolve(candidate)
                if conv.max() < best_peak:
                    a = candidate
                    best_peak = conv.max()
                    best_a = a.copy()
                no_improve = 0
                if time.monotonic() - t0 > time_limit - 1.0:
                    break

    return best_a, best_peak


def multi_scale_smooth(a0, time_limit, n_scales=4, seed=555):

            continue

        # Downsample
        a_coarse = np.zeros(n_coarse)
        for i in range(n_coarse):
            a_coarse[i] = a[i * s:(i + 1) * s].mean()

    if remaining > 30:
        t_alloc = min(remaining * 0.20, 60)
        a_opt, _ = optimize_bb_gradient(best_seq, time_limit=t_alloc,
                                        beta_init=100.0, beta_max=500.0,
                                        seed=seed + 995)
        best_seq, best_score = try_seq(a_opt, "final_bb", (best_seq, best_score))

    # Ultra-fine perturbations
    remaining = deadline - time.monotonic()
    if remaining > 15:
        for _ in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 3:
                break
            noise = rng.standard_normal(n_best) * 0.0003
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(remaining * 0.3, 5)
            a_opt, _ = optimize_pg_nesterov(f_pert, time_limit=t_alloc,
                                            beta_init=300.0, beta_max=500.0,
                                            seed=int(time.monotonic() * 1000) % 10000)
            best_seq, best_score = try_seq(a_opt, "micro", (best_seq, best_score),
                                           verbose=False)

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic() - t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
