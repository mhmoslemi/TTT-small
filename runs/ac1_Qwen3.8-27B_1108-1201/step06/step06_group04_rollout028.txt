```python
import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
def block_qp_update(w, a, block_size=64):
    """
    One pass of block coordinate descent for min a^T W a s.t. sum(a)=1, a>=0.
    W_{i,j} = w[i+j].
    """
    n = len(a)
    a = a.copy()
    
    for start in range(0, n, block_size):
        end = min(start + block_size, n)
        block = slice(start, end)
        block_size = end - start
        
        # Fixed part
        a_fixed = a.copy()
        a_fixed[block] = 0
        
        # The objective: (a_block)^T W_BB (a_block) + 2*(W_BF a_fixed)^T a_block + const
        # = a_block^T W_BB a_block + 2 * grad_block^T a_block + const
        # where grad_block = W_BF @ a_fixed = hankel contribution from fixed part
        
        # W_BB[i,j] = w[(start+i) + (start+j)]
        bb = np.arange(start, end)
        W_BB = w[bb[:, None] + bb[None, :]]
        W_BB += 1e-12 * np.eye(block_size)
        
        # grad_block[i] = sum_{j not in block} w[(start+i)+j] * a[j]
        # = (W @ a_fixed)[start:end]
        # = hankel_mv(w, a_fixed)[start:end]
        grad_block = np.correlate(w, a_fixed, mode='valid')[start:end]
        
        # Solve: min a_b^T W_BB a_b + 2*grad_block^T a_b s.t. sum(a_b) = S, a_b >= 0
        # where S = 1 - sum(a_fixed) = a[block].sum() (since sum(a)=1)
        S = a[block].sum()
        
        # KKT: 2*W_BB * a_b + 2*grad_block + mu*1 = 0 (for active set)
        # For full active set: W_BB * a_b + grad_block + (mu/2)*1 = 0
        # a_b = -W_BB^{-1} (grad_block + (mu/2)*1)
        # sum(a_b) = S
        # => -1^T W_BB^{-1} grad_block - (mu/2) * 1^T W_BB^{-1} 1 = S
        # => mu = -2 * (S + 1^T W_BB^{-1} grad_block) / (1^T W_BB^{-1} 1)
        
        try:
            W_inv_1 = np.linalg.solve(W_BB, np.ones(block_size))
            W_inv_g = np.linalg.solve(W_BB, grad_block)
            denom = W_inv_1.sum()
            if abs(denom) < 1e-14:
                continue
            mu_half = -(S + W_inv_g.sum()) / denom
            a_b = -(W_inv_g + mu_half * W_inv_1)
        except np.linalg.LinAlgError:
            continue
        
        # Check non-negativity
        if a_b.min() >= -1e-10:
            a[block] = np.maximum(a_b, 0)
            # Renormalize
            a = a / a.sum()
        else:
            # Would need active set - skip for now (or do a simple projected gradient)
            # Simple: projected gradient on this block
            grad_full = 2 * (W_BB @ a[block] + grad_block)
            for t in [0.01, 0.005, 0.001]:
                candidate = np.maximum(a[block] - t * grad_full, 0)
                if candidate.sum() > 1e-10:
                    a[block] = candidate
                    a = a / a.sum()
                    break
    
    return a
```

```python
import numpy as np

from scipy.optimize import minimize as scipy_minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(w, a):
    """Compute W @ a where W_{i,j} = w[i+j]. w has length 2n-1, a has length n."""
    return np.correlate(w, a, mode='valid')

            for _ in range(10):
                grad = 2.0 * hankel_mv(w, a)
                for t in [0.01, 0.005, 0.001]:
                    candidate = project_simplex(a - t * grad)
                    obj = candidate @ hankel_mv(w, candidate)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = candidate.copy()
                    a = candidate
                    break
            return best_a

        act_arr = np.array(act)
        W_AA = w[act_arr[:, None] + act_arr[None, :]]
        W_AA += 1e-12 * np.eye(k)

        try:
            x = np.linalg.solve(W_AA, np.ones(k))
        except np.linalg.LinAlgError:
            x = np.linalg.lstsq(W_AA, np.ones(k), rcond=None)[0]

        x_sum = x.sum()
        if abs(x_sum) < 1e-14:
            x = np.ones(k) / k
            x_sum = 1.0

        a_act = x / x_sum

        neg_mask = a_act < -1e-10
        if neg_mask.any():
            min_idx_local = int(np.argmin(a_act))
            active.discard(act[min_idx_local])
            a_new = np.zeros(n)
            a_new[act] = np.maximum(a_act, 0)
            s2 = a_new.sum()
            if s2 > 1e-15:
                a = a_new / s2
            continue

        a_full = np.zeros(n)
        a_full[act] = a_act
        obj = a_full @ hankel_mv(w, a_full)
        if obj < best_obj:
            best_obj = obj
            best_a = a_full.copy()

        # KKT check
        Wa = hankel_mv(w, a_full)
        grad = 2.0 * Wa
        mu = float(np.mean(grad[act]))

        inactive = np.setdiff1d(np.arange(n), act_arr)
        if len(inactive) > 0:
            violations = grad[inactive] - mu
            min_viol = float(violations.min())
            if min_viol < -1e-10:
                worst = int(inactive[np.argmin(violations)])
                active.add(worst)
                continue

        return a_full

    return best_a


def primal_dual_norm(a0, time_limit, blend_init=0.5, use_line_search=True, verbose=False):
    """
    Primal-dual with normalized autocorrelation weights.
    Key: w = c/sum(c) guarantees PSD Hankel matrix.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    blend = blend_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        c = fast_convolve(a)
        w = c / c.sum()

        qp_time = min(2.0, remaining * 0.25)
        a_qp = solve_qp_hankel(w, a, time_limit=qp_time)

        if use_line_search:
            # Line search on blend parameter
            best_t = blend
            best_val = best_obj
            for t in [blend, 0.3, 0.5, 0.7, 0.9, 1.0]:
                candidate = (1.0 - t) * a + t * a_qp
                candidate = np.maximum(candidate, 0)
                cs = candidate.sum()
                if cs < 1e-15:
                    continue
                candidate = candidate / cs
                val = fast_convolve(candidate).max()
                if val < best_val - 1e-15:
                    best_val = val
                    best_t = t
            a = (1.0 - best_t) * a + best_t * a_qp
            a = np.maximum(a, 0)
            a = a / a.sum()
            obj = best_val
        else:
            a = (1.0 - blend) * a + blend * a_qp
            a = np.maximum(a, 0)
            a = a / a.sum()
            obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt
        if no_improve > 5:
            blend = max(0.2, blend * 0.95)
        if no_improve > 30:
            blend = min(0.8, blend * 1.1)
        if no_improve > 100:
            rng = np.random.default_rng((it * 104729 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0
            blend = 0.5

        if verbose and it % 50 == 0:
            print(f"    pd_norm it={it}, obj={best_obj:.10f}, no_imp={no_improve}")

    return best_a, best_obj


def scipy_smoothed_optimize(a0, time_limit, T_init=0.01, T_min=0.001, verbose=False):
    """
    Use scipy L-BFGS-B on smoothed objective with softmax parameterization.
    a = softmax(z), minimize 2n*T*logsumexp(conv(a)/T)
    """
    n = len(a0)
    a_init = np.maximum(a0.astype(np.float64), 1e-10)
    a_init = a_init / a_init.sum()
    z_init = np.log(a_init)  # Inverse of softmax

    def obj_and_grad(z):
        z_shifted = z - z.max()
        e = np.exp(z_shifted)
        a = e / e.sum()
        c = fast_convolve(a)
        scaled = c / T_current[0]
        scaled -= scaled.max()
        log_sum = np.log(np.sum(np.exp(scaled))) + scaled.max() * T_current[0] / T_current[0]
        # Actually compute properly:
        val = T_current[0] * np.log(np.sum(np.exp(scaled))) + c.max() * 0  # placeholder
        # Let me redo:
        # f = 2n * T * logsumexp(c/T)
        # logsumexp(c/T) = max(c/T) + log(sum(exp(c/T - max(c/T))))
        m = np.max(scaled)
        lse = m + np.log(np.sum(np.exp(scaled - m)))
        obj_val = 2.0 * n * T_current[0] * lse

        # Gradient: df/dz_i
        # p_k = exp(c_k/T) / sum exp(c_k/T)
        p = np.exp(scaled - m)
        p = p / p.sum()
        # dc_k/da_i = 2*a_{k-i}
        # da_i/dz_i = a_i(1-a_i), da_i/dz_j = -a_i*a_j
        # df/dz_i = 2n * sum_k p_k * dc_k/dz_i
        # dc_k/dz_i = sum_j (dc_k/da_j)(da_j/dz_i)
        # = dc_k/da_i * a_i(1-a_i) - a_i * sum_{j!=i} dc_k/da_j * a_j
        # = a_i * [dc_k/da_i - a_i * sum_j dc_k/da_j]
        # sum_j dc_k/da_j = d/dt c_k(a+t) at t=0 (directional derivative in direction 1)
        # = sum_j 2*a_{k-j} (with appropriate bounds) ... this is complex
        # 
        # Simpler: use the chain rule differently
        # dc_k/da_i = 2*a_{k-i} (for valid indices)
        # grad_c_i = sum_k p_k * 2*a_{k-i} = 2 * hankel_mv(p, a)
        # 
        # For softmax: the gradient w.r.t. z is:
        # df/dz = diag(a) @ (grad_c - (a @ grad_c) * 1)
        # This is because: for f(a) with a=softmax(z), 
        # df/dz = a ⊙ (df/da - a*(a^T df/da))
        # Wait, that's not right either.
        # 
        # For softmax: da_i/dz_j = a_i(δ_ij - a_j)
        # df/dz_j = sum_i (df/da_i) * a_i(δ_ij - a_j)
        # = (df/da)_j * a_j - a_j * sum_i (df/da_i) * a_i
        # = a_j * [(df/da)_j - a · (df/da)]
        
        grad_c = 2.0 * hankel_mv(p, a)  # length n
        df_da = grad_c  # This is df/da for the smoothed objective (factor 2n*T is absorbed)
        # Actually f = 2n*T*sum_k p_k * c_k / T ... no.
        # f = 2n * T * logsumexp(c/T)
        # df/dc_k = 2n * T * (1/T) * p_k = 2n * p_k
        # df/da_i = sum_k (2n*p_k) * dc_k/da_i = 2n * sum_k p_k * 2*a_{k-i} = 4n * hankel_mv(p,a)[i]
        
        df_da = 4.0 * n * hankel_mv(p, a)
        
        # Softmax gradient
        grad_z = a * (df_da - np.dot(a, df_da))
        
        return obj_val, grad_z

    T_current = [T_init]
    
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_convolve(a_init).max()
    t0 = time.monotonic()
    
    # Anneal T
    n_phases = 5
    phase_time = time_limit / n_phases
    
    for phase in range(n_phases):
        T_current[0] = T_init * (T_min / T_init) ** (phase / max(1, n_phases - 1))
        if phase > 0:
            # Re-initialize z from current best
            a_cur = np.maximum(a_init, 1e-10)
            a_cur = a_cur / a_cur.sum()
            z_init = np.log(a_cur)
        
        phase_deadline = t0 + (phase + 1) * phase_time
        
        def local_obj_grad(z):
            T_current[0] = T_init * (T_min / T_init) ** (phase / max(1, n_phases - 1))
            return obj_and_grad(z)
        
        result = scipy_minimize(
            local_obj_grad, z_init, method='L-BFGS-B',
            options={'maxiter': 500, 'maxfun': 2000, 'maxls': 20}
        )
        
        z_opt = result.x
        a_opt = np.exp(z_opt - z_opt.max())
        a_opt = a_opt / a_opt.sum()
        a_opt = np.maximum(a_opt, 0)
        
        score = 2.0 * n * fast_convolve(a_opt).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_opt.copy()
            a_init = a_opt.copy()
        
        if verbose:
            print(f"    scipy phase {phase}: T={T_current[0]:.6f}, score={best_score:.10f}")
        
        if time.monotonic() > phase_deadline:
            break
    
    return best_a, best_score


def block_coordinate_descent(a0, time_limit, block_size=64, verbose=False):
    """
    Block coordinate descent for min max_k c[k] on the simplex.
    For each block, solve a small QP using the current "effective" weights.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    n_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Compute weights from current autocorrelation
        c = fast_convolve(a)
        w = c / c.sum()
        
        improved = False
        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            bb = np.arange(start, end)
            bs = end - start
            
            a_fixed = a.copy()
            a_fixed[bb] = 0.0
            S = a[bb].sum()
            
            if S < 1e-12:
                continue
            
            W_BB = w[bb[:, None] + bb[None, :]]
            W_BB += 1e-12 * np.eye(bs)
            grad_b = hankel_mv(w, a_fixed)[start:end]
            
            # Solve: min a_b^T W_BB a_b + 2*grad_b^T a_b s.t. sum(a_b)=S, a_b>=0
            # Try full active set first
            try:
                W_inv_1 = np.linalg.solve(W_BB, np.ones(bs))
                W_inv_g = np.linalg.solve(W_BB, grad_b)
                denom = W_inv_1.sum()
                if abs(denom) > 1e-12:
                    mu_half = -(S + W_inv_g.sum()) / denom
                    a_b = -(W_inv_g + mu_half * W_inv_1)
                    if a_b.min() >= -1e-8:
                        a[bb] = np.maximum(a_b, 0)
                        a = a / a.sum()
                        improved = True
                        continue
            except np.linalg.LinAlgError:
                pass
            
            # Fallback: projected gradient on this block
            grad_full = 2.0 * (W_BB @ a[bb] + grad_b)
            for t in [0.01 / bs, 0.005 / bs]:
                candidate_block = np.maximum(a[bb] - t * grad_full, 0)
                if candidate_block.sum() > 1e-12:
                    a[bb] = candidate_block
                    a = a / a.sum()
                    improved = True
                    break
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
        elif not improved and n_passes > 20:
            break
    
    return best_a, best_obj


def nesterov_smoothed(a0, time_limit, T_init=0.005, lr_init=0.02, verbose=False):
    """Nesterov-accelerated smoothed subgradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    v = np.zeros(n)
    lr = lr_init
    mom = 0.9
    y = a.copy()
    T = T_init
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        y = a + mom * v
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        c_y = fast_convolve(y)
        # Smoothed gradient
        scaled = c_y / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        grad = 2.0 * hankel_mv(p, y)
        
        v_new = project_simplex(y - lr * grad) - y
        a = y + v_new
        v = v_new
        a = project_simplex(a)
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.9999
        mom = min(0.999, mom + 0.00002)
        T = max(0.0001, T * 0.999)
        
        if no_improve > 200:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = lr_init
            mom = 0.9
            T = T_init
            no_improve = 0
        
        if verbose and it % 200 == 0:
            print(f"    nesterov_smooth it={it}, obj={best_obj:.10f}")
    
    return best_a, best_obj


def generate_diverse_inits(n, parent, seed):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 0.1))
    
    # Edge concentration
    x = np.arange(n, dtype=np.float64)
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge / edge.sum())
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Two peaks
    g2 = (np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2))
    inits.append(g2 / g2.sum())
    
    # Cosine
    c1 = np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * x / n), 0.01)
    inits.append(c1 / c1.sum())
    
    # Parent interpolated
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())
    
    return inits


def try_sequence(seq, best_state):
    """Try a sequence and update best if improved."""
    a = np.maximum(np.asarray(seq, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_state['score']:
        best_state['score'] = score
        best_state['seq'] = a_norm.copy()
        print(f"  NEW BEST: {score:.10f} (n={len(a_norm)})")
    return score

    print(f"  Phase 4 done. Best: {best_state['score']:.10f} ({time.monotonic()-t4:.0f}s)")
    
    # === Phase 5: Final polish (~200s) ===
    print("\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()
    
    if best_state['seq'] is not None:
        n_best = len(best_state['seq'])
        
        # Tiny perturbations + quick re-optimize
        for pi in range(5):
            remaining = deadline - time.monotonic() - 20
            if remaining < 10:
                break
            noise = rng.standard_normal(n_best) * (0.001 + 0.0005 * pi)
            perturbed = project_simplex(best_state['seq'] + noise)
            t_alloc = min(remaining, 25)
            a_opt, _ = primal_dual_norm(perturbed, time_limit=t_alloc,
                                        blend_init=0.3, use_line_search=True)
            try_sequence(a_opt, best_state)
        
        # One final block CD
        remaining = deadline - time.monotonic() - 15
        if remaining > 10:
            t_alloc = min(remaining, 40)
            a_opt, _ = block_coordinate_descent(best_state['seq'], time_limit=t_alloc, block_size=16)
            try_sequence(a_opt, best_state)
    
    print(f"  Phase 5 done. Best: {best_state['score']:.10f} ({time.monotonic()-t5:.0f}s)")
    
    # Fallback
    if best_state['seq'] is None:
        best_state['seq'] = np.ones(384) / 384
        best_state['score'] = float(evaluate_sequence(best_state['seq'].tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_state['seq'].tolist()))
    if np.isfinite(fs):
        best_state['score'] = min(best_state['score'], fs)
    
    print(f"\nFINAL: score={best_state['score']:.10f}, n={len(best_state['seq'])}")
    return [float(x) for x in best_state['seq']]
```
