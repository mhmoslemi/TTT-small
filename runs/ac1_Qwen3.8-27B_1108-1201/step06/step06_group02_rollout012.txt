def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_minmax_lp(a, peak_indices, n_constraints_max=500):
    """
    Solve: min t s.t. (a*y)[k] <= t for k in peak_indices, sum(y)=1, y>=0
    Returns (y, t) or (None, None) if infeasible.
    """
    n = len(a)
    K = peak_indices
    
    if len(K) > n_constraints_max:
        # Too many constraints, use all
        K = np.arange(2*n-1)
    
    # Variables: [y_0, ..., y_{n-1}, t]
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Equality: sum(y) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Inequalities: (a*y)[k] - t <= 0 for each k in K
    A_ub = np.zeros((len(K), n + 1))
    for idx, k in enumerate(K):
        k = int(k)
        # (a*y)[k] = sum_i a[k-i] * y[i] for valid i
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_end >= i_start:
            A_ub[idx, i_start:i_end+1] = a[k - i_end:k - i_start + 1][::-1]
            # Wait, let me be more careful
            # (a*y)[k] = sum_{i=0}^{n-1} a[k-i] * y[i] where 0 <= k-i <= n-1
            # So for each i in [i_start, i_end], the coefficient of y[i] is a[k-i]
            # a[k-i] where i goes from i_start to i_end
            # k-i goes from k-i_start to k-i_end
            # Which is from min(n-1, k) to max(0, k-n+1)
            # So a[k-i] = a[min(n-1,k)] down to a[max(0,k-n+1)]
            # That's a[k-i_start], a[k-i_start-1], ..., a[k-i_end]
            # = a[min(n-1,k)], a[min(n-1,k)-1], ..., a[max(0,k-n+1)]
            # In numpy: a[k-i_start : k-i_end-1 : -1] ... this is getting confusing
            # Let me just do it directly:
            A_ub[idx, i_start:i_end+1] = a[k - np.arange(i_start, i_end+1)]
        A_ub[idx, n] = -1.0  # coefficient of t
    
    b_ub = np.zeros(len(K))
    
    bounds = [(0, None)] * n + [(None, None)]  # y >= 0, t free
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs',
                     options={'presolve': True})
        if res.success:
            y = np.maximum(res.x[:n], 0.0)
            t = res.x[n]
            y = y / y.sum()  # ensure normalization
            return y, t
    except Exception:
        pass
    return None, None

        best_alpha = 0.0

def peak_surgery(a0, time_limit, seed=42):

    a = np.maximum(a0.astype(np.float64), 0.0)

    n = len(a)
    
    best_a = a.copy()

    stale = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        
        # Find contributing coordinates
        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)
        
        if i_end < i_start:
            break

        idx = np.arange(i_start, i_end + 1)
        contribs = a[idx] * a[k_peak - idx]
        
        # Find the coordinate with max contribution
        j_max = idx[np.argmax(contribs)]
        
        # Find "safe" receiver coordinates (low values, far from peak)
        # A coordinate j is "safe" if it doesn't appear in the peak's product
        # and has low value
        in_peak = set(idx.tolist())

        if not safe_indices:

        for j_min in candidate_receivers:
            if j_min == j_max:

                if candidate[j_max] < 0:

"""
Optimized FW-minmax LP approach for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: Uses a reduced LP with only active peak constraints (top-K peaks)
instead of all 2n-1 constraints, enabling many more iterations in the same time.
Combined with a novel peak surgery coordinate descent for final polish.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_minmax_lp(a, peak_indices, max_constraints=150):
    """
    Solve: min t s.t. (a*y)[k] <= t for k in peak_indices, sum(y)=1, y>=0.
    Returns (y, t) or (None, None) on failure.
    """
    n = len(a)
    K = peak_indices[:max_constraints]
    nK = len(K)
    if nK == 0:
        return None, None

    # Variables: [y_0, ..., y_{n-1}, t]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t

    # Equality: sum(y) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Inequalities: (a*y)[k] - t <= 0
    A_ub = np.zeros((nK, n + 1))
    for idx in range(nK):
        k = int(K[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_end >= i_start:
            i_vals = np.arange(i_start, i_end + 1)
            A_ub[idx, i_start:i_end + 1] = a[k - i_vals]
        A_ub[idx, n] = -1.0

    b_ub = np.zeros(nK)
    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res is not None and res.success:
            y = np.maximum(res.x[:n], 0.0)
            t = float(res.x[n])
            ys = y.sum()
            if ys > 1e-15:
                y = y / ys
            return y, t
    except Exception:
        pass
    return None, None


def fw_minmax(a0, time_limit, seed=42):

    s = a.sum()

    n = len(a)

def peak_surgery(a0, time_limit, seed=42):
    """
    Coordinate-level peak reduction via smart mass transfer.
    Targets the highest peak and transfers mass from contributing coordinates
    to safe receiver coordinates.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    rng = np.random.default_rng(seed)

    t0 = time.monotonic()
    it = 0
    stale = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        M = conv[k_peak]

        # Find contributing coordinates to the peak
        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)

        if i_end < i_start:
            break

        idx = np.arange(i_start, i_end + 1)
        contribs = a[idx] * a[k_peak - idx]

        sorted_contrib = np.argsort(contribs)[::-1]
        top_contributors = idx[sorted_contrib[:5]]

        # Determine "danger zone" - coordinates that affect peaks near k_peak
        danger = set()
        for dk in range(-2, 3):
            k2 = k_peak + dk
            if 0 <= k2 <= 2 * n - 2:
                s2 = max(0, k2 - n + 1)
                e2 = min(n - 1, k2)
                danger.update(range(s2, e2 + 1))

        # Find safe receivers (outside danger zone, low value)
        safe_candidates = []
        for j in range(n):
            if j not in danger:
                safe_candidates.append((a[j], j))
        if not safe_candidates:
            safe_candidates = [(a[j], j) for j in range(n)]
        safe_candidates.sort()
        receivers = [jc for _, jc in safe_candidates[:8]]

        improved = False
        for j_max in top_contributors[:3]:
            for j_min in receivers[:3]:
                if j_min == j_max:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
                    delta = a[j_max] * frac
                    if delta < 1e-15 or delta > a[j_max]:
                        continue
                    candidate = a.copy()
                    candidate[j_max] -= delta
                    candidate[j_min] += delta
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stale += 1
            if stale % 50 == 0:
                scale = 0.001 + 0.003 * min(1.0, stale / 500.0)
                a = project_simplex(best_a + rng.standard_normal(n) * scale)
                c = fast_convolve(a)
                if c.max() < best_obj:
                    best_obj = c.max()
                    best_a = a.copy()
                stale = 0
            if stale > 2000:
                break

    return best_a, best_obj


def multi_peak_geometric(a0, time_limit, seed=42):
    """
    Geometric-phase optimization: adjust the geometric mean of coordinates
    at peak-contributing positions to reduce peaks.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    rng = np.random.default_rng(seed)

    t0 = time.monotonic()
    it = 0
    stale = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        M = conv.max()
        k_peak = int(np.argmax(conv))

        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)
        idx = np.arange(i_start, i_end + 1)

        # Try scaling down the top contributors and scaling up non-contributors
        contribs = a[idx] * a[k_peak - idx]
        top_mask = contribs >= 0.5 * contribs.max()
        top_idx = idx[top_mask]
        bottom_idx = idx[~top_mask]

        if len(top_idx) == 0 or len(bottom_idx) == 0:
            stale += 1
            if stale > 200:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                stale = 0
                if stale > 500:
                    break
            continue

        # Try reducing top contributors and boosting bottom ones
        improved = False
        for reduction in [0.05, 0.1, 0.15, 0.2]:
            delta_total = 0.0
            for j in top_idx:
                delta = a[j] * reduction
                delta_total += delta
                a[j] -= delta
            if delta_total < 1e-15:
                continue
            # Distribute to bottom contributors
            per_boost = delta_total / len(bottom_idx)
            for j in bottom_idx:
                a[j] += per_boost
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            c = fast_convolve(a)
            if c.max() < best_obj - 1e-15:
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break
            else:
                # Revert
                for j in top_idx:
                    a[j] += a[j] * reduction / (1.0 - reduction) if reduction < 1 else 0

        if improved:
            stale = 0
        else:
            stale += 1
            if stale > 100:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                stale = 0
            if stale > 1000:
                break

    return best_a, best_obj


def gen_initializations(n, parent, rng):
    """Generate diverse initializations for exploration."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)

    # Edge-weighted (high at edges)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.12 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine modulations
    for freq in [1, 2, 3, 4, 6]:
        h = 1.0 + 0.35 * np.cos(2.0 * np.pi * freq * x / n)
        h = np.maximum(h, 0.01)
        inits.append(h / h.sum())

    # Bimodal
    two_peak = np.exp(-0.5 * ((x - 0.25 * n) / (0.08 * n)) ** 2) + \
               np.exp(-0.5 * ((x - 0.75 * n) / (0.08 * n)) ** 2)
    inits.append(two_peak / two_peak.sum())

    # Sawtooth
    saw = np.maximum(0.1, 1.0 - 2.0 * np.abs((x / n) - 0.5))
    inits.append(saw / saw.sum())

    # Parent interpolation
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1500
        if remaining < 10:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss > 1e-10:
                start = start / ss
            else:
                start = np.ones(n_try) / n_try

        t_alloc = min(n_per, remaining)
        print("  FW-minmax n={}, t={:.0f}s".format(n_try, t_alloc))
        a_opt, _ = fw_minmax(start, time_limit=t_alloc, seed=seed + n_try)
        try_and_update(a_opt, state, "fw_n{}".format(n_try))

    print("  Phase 1 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 2: Deep FW-minmax (~900s) ===
    print("\n=== Phase 2: Deep FW-minmax ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])
        remaining = deadline - time.monotonic() - 700
        if remaining > 30:
            t_alloc = min(remaining, 800)
            print("  Deep FW-minmax n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = fw_minmax(state[1], time_limit=t_alloc, seed=seed)
            try_and_update(a_opt, state, "fw_deep")

            # Perturbed restarts from best
            for scale in [0.001, 0.003, 0.007, 0.012]:
                remaining = deadline - time.monotonic() - 550
                if remaining < 15:
                    break
                f_pert = project_simplex(state[1] + rng.standard_normal(n_best) * scale)
                t_alloc = min(remaining, 80)
                a_opt2, _ = fw_minmax(f_pert, time_limit=t_alloc, seed=seed + int(scale * 1e5))
                try_and_update(a_opt2, state, "fw_pert_{:.4f}".format(scale))

    print("  Phase 2 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 3: Peak Surgery + Geometric (~450s) ===
    print("\n=== Phase 3: Peak Surgery ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Peak surgery
        remaining = deadline - time.monotonic() - 250
        if remaining > 20:
            t_alloc = min(remaining, 250)
            print("  Peak surgery, t={:.0f}s".format(t_alloc))
            a_opt, _ = peak_surgery(state[1], time_limit=t_alloc, seed=seed)
            try_and_update(a_opt, state, "surgery")

        # Geometric optimization
        remaining = deadline - time.monotonic() - 200
        if remaining > 20:
            t_alloc = min(remaining, 150)
            print("  Geometric opt, t={:.0f}s".format(t_alloc))
            a_opt, _ = multi_peak_geometric(state[1], time_limit=t_alloc, seed=seed + 1)
            try_and_update(a_opt, state, "geometric")

        # Another surgery round on the new best
        remaining = deadline - time.monotonic() - 200
        if remaining > 15:
            t_alloc = min(remaining, 100)
            a_opt, _ = peak_surgery(state[1], time_limit=t_alloc, seed=seed + 2)
            try_and_update(a_opt, state, "surgery2")

    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 4: Final Polish (~200s) ===
    print("\n=== Phase 4: Final Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Short FW-minmax for final improvement
        remaining = deadline - time.monotonic() - 80
        if remaining > 10:
            t_alloc = min(remaining, 100)
            print("  Final FW-minmax, t={:.0f}s".format(t_alloc))
            a_opt, _ = fw_minmax(state[1], time_limit=t_alloc, seed=seed + 99)
            try_and_update(a_opt, state, "final_fw")

        # Fine peak surgery
        remaining = deadline - time.monotonic() - 40
        if remaining > 10:
            t_alloc = min(remaining, 50)
            a_opt, _ = peak_surgery(state[1], time_limit=t_alloc, seed=seed + 77)
            try_and_update(a_opt, state, "final_surgery")

        # Micro-perturbations with quick evaluation
        remaining = deadline - time.monotonic() - 20
        if remaining > 5:
            n_cur = len(state[1])
            for i in range(5):
                remaining = deadline - time.monotonic() - 10
                if remaining < 3:
                    break
                noise = rng.standard_normal(n_cur) * (0.0003 + 0.0002 * i)
                f_pert = project_simplex(state[1] + noise)
                t_alloc = min(remaining, 4)
                a_opt, _ = peak_surgery(f_pert, time_limit=t_alloc, seed=seed + 55 + i)
                try_and_update(a_opt, state, "micro_{}".format(i))

    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(384) / 384
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
