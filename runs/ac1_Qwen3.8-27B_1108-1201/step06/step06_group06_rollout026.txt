from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows), np.array(cols)

def solve_lp_oracle(f, M, n, row_idx, col_idx):
    c = -np.ones(n)
    vals = np.empty(len(row_idx))
    for idx in range(len(row_idx)):
        k = row_idx[idx]
        i = col_idx[idx]
        vals[idx] = f[k - i]
    A_mat = lil_matrix((2 * n - 1, n))
    A_mat[row_idx, col_idx] = vals
    A_mat = A_mat.tocsr()
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.fun is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None

def softmax_grad_descent(a0, time_limit, T_init=None, lr_init=0.5, seed=42):
    """Smooth gradient descent with softmax parameterization and Adam."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    
    if T_init is None:
        conv_init = fast_convolve(a)
        T_init = max(conv_init.max() - conv_init.min(), 1e-6) * 0.3
    
    # Initialize b from a
    b = np.log(a + 1e-30)  # b such that softmax(b) ≈ a
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_init
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        a = softmax_stable(b)
        conv = fast_convolve(a)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive temperature
        spread = conv.max() - conv.min()
        T = max(spread * 0.3, 1e-6)
        
        # Compute gradient
        conv_shifted = conv / T - (conv / T).max()
        p = np.exp(conv_shifted)
        p = p / p.sum()
        
        # S[j] = dot(a, p[j:j+n])
        # Use sliding dot product
        # Actually let's just compute it directly
        S = np.array([np.dot(a, p[j:j+n]) for j in range(n)])
        C = np.dot(p, conv)
        grad = 2.0 * a * (S - C)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        b = b - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Learning rate decay

            b = np.log(a_pert + 1e-30)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj

"""
Hybrid optimizer for minimum peak autocorrelation step function.
Uses softmax-parameterized smooth gradient descent with Adam optimizer,
FW-LP with sparse oracle, projected subgradient, and local search polishing.
Multi-n exploration near the parent length with adaptive time allocation.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_stable(b):
    """Numerically stable softmax."""
    b_shifted = b - b.max()
    e = np.exp(b_shifted)
    return e / e.sum()


def build_lp_matrix(n):
    """Build sparse constraint indices for LP oracle."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows), np.array(cols)


def solve_lp_oracle(f, M, n, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0. Returns normalized g or None."""
    c = -np.ones(n)
    vals = np.empty(len(row_idx))
    for idx in range(len(row_idx)):
        k = row_idx[idx]
        i = col_idx[idx]
        vals[idx] = f[k - i]
    A_mat = lil_matrix((2 * n - 1, n))
    A_mat[row_idx, col_idx] = vals
    A_mat = A_mat.tocsr()
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.fun is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None

def softmax_grad_descent(a0, time_limit, T_init=None, lr_init=0.5, seed=42):
    """
    Smooth gradient descent with softmax parameterization and Adam optimizer.
    Optimizes f(b) = T*log(sum_k exp((softmax(b)*softmax(b))[k]/T))
    Gradient: df/db[j] = 2*a[j] * (S[j] - C)
    where S[j] = sum_k p[k]*a[k-j], C = sum_k p[k]*conv[k],
    p[k] = exp(conv[k]/T) / sum exp(conv[k']/T)
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-20)
    a = a / a.sum()

    if T_init is None:
        conv_init = fast_convolve(a)
        T_init = max(conv_init.max() - conv_init.min(), 1e-8) * 0.3

    b = np.log(a)
    b = b - b.mean()  # Center for numerical stability

    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_init

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        a = softmax_stable(b)
        conv = fast_convolve(a)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive temperature
        spread = conv.max() - conv.min()
        T = max(spread * 0.25, 1e-8)

        # Compute probability weights
        conv_scaled = conv / T
        conv_scaled -= conv_scaled.max()
        p = np.exp(conv_scaled)
        p = p / p.sum()

        # Compute S[j] = sum_k p[k] * a[k-j] = dot(a, p[j:j+n])
        # Vectorized using matrix multiplication
        # p[j:j+n] for j=0..n-1 is a sliding window
        # We can compute all at once using a Toeplitz-like matrix
        # But for n=768, direct computation is fine
        S = np.empty(n)
        for j in range(n):
            S[j] = np.dot(a, p[j:j + n])

        C = np.dot(p, conv)
        grad = 2.0 * a * (S - C)

        # Adam update
        t_adam += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t_adam)
        v_hat = v / (1.0 - beta2 ** t_adam)
        b = b - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Learning rate schedule
        if it % 200 == 0:
            lr *= 0.97
        if no_improve > 100 and no_improve % 50 == 0:
            lr *= 0.9

        # Restart on stagnation
        if no_improve > 300:
            noise_scale = 0.002 + 0.005 * (it % 3)
            noise = rng.standard_normal(n) * noise_scale
            a_pert = project_simplex(best_a + noise)
            b = np.log(np.maximum(a_pert, 1e-20))
            b = b - b.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def projected_subgrad(a0, time_limit, T_init=0.001, seed=42):
    """Projected subgradient method with backtracking line search."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    T = T_init
    lr = 0.05

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive T
        spread = conv.max() - conv.min()
        T = max(spread * 0.25, 1e-8)

        # Compute subgradient via smooth max
        conv_scaled = conv / T
        conv_scaled -= conv_scaled.max()
        p = np.exp(conv_scaled)
        p = p / p.sum()

        # Gradient of smooth max w.r.t. a (not b)
        # d/d a[i] of conv[k] = a[k-i] (clipped)
        # So grad_a[i] = sum_k p[k] * a[k-i] (symmetric part)
        # Actually: d/d a[i] sum_k p[k] * conv[k] is tricky since p depends on a
        # Use the direct subgradient of max instead:
        # subgrad[i] = (1/|S|) sum_{k in S} a[k-i]  (for active set S)
        # For smooth: grad_a[i] = sum_k p[k] * a[k-i] + correction
        # Simplified: use grad_a[i] = sum_k p[k] * (a[k-i] + a[i-k]) / 2 ... 
        # Actually let's just use the direction: for each active peak k, the gradient
        # of conv[k] w.r.t. a is the vector with [i] entry = a[k-i] (if valid)
        # Weighted by p:
        grad_a = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] < 1e-10:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            for i in range(i_start, i_end + 1):
                grad_a[i] += p[k] * a[k - i]

        # Scale gradient
        grad_norm = np.linalg.norm(grad_a)
        if grad_norm < 1e-15:
            no_improve += 100
            continue
        grad_a = grad_a / grad_norm

        # Backtracking line search
        best_candidate = a
        best_val = obj
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            candidate = project_simplex(a - trial_lr * grad_norm * grad_a)
            c = fast_convolve(candidate)
            if c.max() < best_val - 1e-15:
                best_val = c.max()
                best_candidate = candidate
                lr = trial_lr
                break

        a = best_candidate

        # Decay and restart
        if no_improve > 50:
            lr *= 0.95
        if no_improve > 200:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(best_a + noise)
            lr = 0.05
            no_improve = 0

    return best_a, best_obj


def local_search(a0, time_limit, seed=42):
    """Local search with multi-scale perturbations and coordinate moves."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        improved = False
        scale = 0.0005 + 0.003 * (stagnant % 4) / 3.0
        pert = rng.standard_normal(n) * scale
        candidate = project_simplex(a + pert)
        c = fast_convolve(candidate)
        if c.max() < best_obj - 1e-15:
            a = candidate
            best_obj = c.max()
            best_a = a.copy()
            improved = True
            stagnant = 0
        else:
            # Coordinate moves
            if it % 3 == 0:
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j:
                    for frac in [0.01, 0.03, 0.05, 0.1]:
                        candidate = a.copy()
                        amt = candidate[i] * frac
                        candidate[i] -= amt
                        candidate[j] += amt
                        candidate = np.maximum(candidate, 0)
                        ss = candidate.sum()
                        if ss < 1e-10:
                            continue
                        candidate = candidate / ss
                        c = fast_convolve(candidate)
                        if c.max() < best_obj - 1e-15:
                            a = candidate
                            best_obj = c.max()
                            best_a = a.copy()
                            improved = True
                            stagnant = 0
                            break

        if not improved:
            stagnant += 1
            if stagnant > 500:
                break

    return best_a, best_obj


def try_seq(a, label, best_seq, best_score):
    """Evaluate a sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return a_norm.copy(), score
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point: multi-phase hybrid search for minimum peak autocorrelation.
    Combines softmax-gradient-descent with Adam, FW-LP, projected subgradient,
    and local search. Explores multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Using uniform init: score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Softmax GD on parent ===
    print("\n=== Phase 1: Softmax GD (parent) ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        t_alloc = min(remaining * 0.25, 350)
        print(f"  Softmax GD, lr=0.5, t={t_alloc:.0f}s")
        a_opt, _ = softmax_grad_descent(parent, time_limit=t_alloc, T_init=0.002, lr_init=0.5)
        best_seq, best_score = try_seq(a_opt, "sgd_p1", best_seq, best_score)

    # Projected subgradient
    remaining = deadline - time.monotonic()
    if remaining > 80:
        t_alloc = min(remaining * 0.12, 150)
        print(f"  Projected subgrad, t={t_alloc:.0f}s")
        a_opt, _ = projected_subgrad(best_seq, time_limit=t_alloc, T_init=0.001)
        best_seq, best_score = try_seq(a_opt, "psg_p1", best_seq, best_score)

    # FW-LP
    remaining = deadline - time.monotonic()
    if remaining > 80:
        t_alloc = min(remaining * 0.15, 200)
        print(f"  FW-LP, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, "fw_p1", best_seq, best_score)

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 2: Multi-n exploration ===
    print("\n=== Phase 2: Multi-n Exploration ===")
    target_ns = [384, 512, 640, 704, 768, 832, 896, 1024, 1280]
    if n_parent not in target_ns:
        target_ns.append(n_parent)
    # Add neighbors of parent
    for delta in [-32, -16, 16, 32]:
        target_ns.append(n_parent + delta)
    target_ns = sorted(set(x for x in target_ns if 128 <= x <= 2048))

    for n_try in target_ns:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        if n_try == len(best_seq):
            start = best_seq.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                             np.linspace(0, 1, len(best_seq)), best_seq)
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try

        # Softmax GD
        remaining = deadline - time.monotonic()
        if remaining > 50:
            t_alloc = min(remaining * 0.04, 30)
            a_opt, _ = softmax_grad_descent(start, time_limit=t_alloc,
                                            T_init=0.002, lr_init=0.4, seed=n_try)
            best_seq, best_score = try_seq(a_opt, f"sgd_n{n_try}", best_seq, best_score)

        # FW-LP
        remaining = deadline - time.monotonic()
        if remaining > 40:
            cur = best_seq if len(best_seq) == n_try else start
            t_alloc = min(remaining * 0.03, 20)
            a_opt, _ = fw_lp_optimize(cur, time_limit=t_alloc)
            best_seq, best_score = try_seq(a_opt, f"fw_n{n_try}", best_seq, best_score)

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 3: Deep optimization with varied hyperparameters ===
    print("\n=== Phase 3: Deep Optimization ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Multiple softmax GD runs with different T and lr
    configs = [
        (0.001, 0.4), (0.003, 0.3), (0.0005, 0.6),
        (0.002, 0.2), (0.005, 0.15), (0.001, 0.8),
    ]
    for i, (T_i, lr_i) in enumerate(configs):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        t_alloc = min(remaining * 0.1, 100)
        a_opt, _ = softmax_grad_descent(best_seq, time_limit=t_alloc,
                                        T_init=T_i, lr_init=lr_i, seed=1000 + i * 137)
        best_seq, best_score = try_seq(a_opt, f"deep_sgd_{i}", best_seq, best_score)

    # Deep FW-LP
    remaining = deadline - time.monotonic()
    if remaining > 80:
        t_alloc = min(remaining * 0.2, 200)
        print(f"  Deep FW-LP, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, "deep_fw", best_seq, best_score)

    # Perturbed FW-LP restarts
    for scale in [0.001, 0.003, 0.008, 0.015]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * scale)
        t_alloc = min(remaining * 0.08, 60)
        a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, f"fw_pert_{scale}", best_seq, best_score)

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 4: Alternative n and structured inits ===
    print("\n=== Phase 4: Structure & Alt-n ===")
    remaining = deadline - time.monotonic()

    # Try adjacent n values with fresh optimization
    for n_try in [len(best_seq) - 16, len(best_seq) + 16, len(best_seq) - 8, len(best_seq) + 8]:
        if n_try < 128 or n_try > 2048:
            continue
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        start = np.interp(np.linspace(0, 1, n_try),
                         np.linspace(0, 1, len(best_seq)), best_seq)
        start = np.maximum(start, 0)
        start = start / start.sum()
        t_alloc = min(remaining * 0.08, 50)
        a_opt, _ = softmax_grad_descent(start, time_limit=t_alloc, T_init=0.001, lr_init=0.4, seed=n_try * 3)
        best_seq, best_score = try_seq(a_opt, f"alt_n_{n_try}", best_seq, best_score)

    # Structured blends
    n_best = len(best_seq)
    x = np.arange(n_best, dtype=np.float64)
    structures = []

    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / (n_best - 1)))
    structures.append(cos_w / cos_w.sum())

    #Raised cosine (Tukey-like)
    tukey = np.ones(n_best)
    taper = int(0.1 * n_best)
    for i in range(taper):
        tukey[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * taper) - 1)))
        tukey[-1 - i] = tukey[i]
    structures.append(tukey / tukey.sum())

    # Logistic-like
    logi = 1.0 / (1.0 + np.exp(-0.05 * (x - n_best / 2)))
    structures.append(logi / logi.sum())

    for si, struct in enumerate(structures):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        for alpha in [0.2, 0.4, 0.6]:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            blend = (1 - alpha) * best_seq + alpha * struct
            blend = project_simplex(blend)
            t_alloc = min(remaining * 0.05, 25)
            a_opt, _ = softmax_grad_descent(blend, time_limit=t_alloc, T_init=0.002, lr_init=0.3, seed=5000 + si * 100 + int(alpha * 100))
            best_seq, best_score = try_seq(a_opt, f"struct{si}_a{alpha}", best_seq, best_score)

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Final polish ===
    print("\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Local search
    if remaining > 40:
        t_alloc = min(remaining * 0.25, 120)
        print(f"  Local search, t={t_alloc:.0f}s")
        a_opt, _ = local_search(best_seq, time_limit=t_alloc, seed=77777)
        best_seq, best_score = try_seq(a_opt, "local1", best_seq, best_score)

    # Fine softmax GD
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_alloc = min(remaining * 0.2, 80)
        a_opt, _ = softmax_grad_descent(best_seq, time_limit=t_alloc, T_init=0.0003, lr_init=0.1, seed=424242)
        best_seq, best_score = try_seq(a_opt, "micro_sgd", best_seq, best_score)

    # Another FW-LP pass
    remaining = deadline - time.monotonic()
    if remaining > 25:
        t_alloc = min(remaining * 0.2, 50)
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, "final_fw", best_seq, best_score)

    # Tiny perturbations + quick optimize
    remaining = deadline - time.monotonic()
    for _ in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 8:
            break
        noise = rng.standard_normal(n_best) * 0.0003
        f_pert = project_simplex(best_seq + noise)
        t_alloc = min(remaining * 0.15, 5)
        a_opt, _ = softmax_grad_descent(f_pert, time_limit=t_alloc, T_init=0.0002, lr_init=0.05, seed=int(rng.integers(1e6)))
        best_seq, best_score = try_seq(a_opt, "micro_pert", best_seq, best_score)

    # Final local search
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_opt, _ = local_search(best_seq, time_limit=min(remaining * 0.4, 30), seed=99999)
        best_seq, best_score = try_seq(a_opt, "local2", best_seq, best_score)

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
