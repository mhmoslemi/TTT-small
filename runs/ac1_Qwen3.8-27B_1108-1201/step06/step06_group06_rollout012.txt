def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

    # corr[j] = sum_{m=0}^{n-1} a[m] * p[j+m] (where p is defined for indices 0..2n-2)
    # Pad both to length at least 3n-2 and use FFT.
    
    L = 1 << (3 * n - 2 - 1).bit_length()
    A = np.fft.rfft(a, L)
    P = np.fft.rfft(p, L)
    # corr[j] = sum_m a[m] * p[j+m]
    # = ifft(fft(p) * conj(fft(a)))[j + n - 1] ... hmm let me think more carefully.
    
    # Actually: ifft(fft(p) * conj(fft(a)))[k] = sum_m p[m] * a[m-k] = sum_m p[m+k] * a[m]
    # So ifft(fft(p) * conj(fft(a)))[k] = sum_m a[m] * p[m+k]
    # We want corr[j] = sum_m a[m] * p[m+j], so corr[j] = ifft(fft(p) * conj(fft(a)))[j]
    
    # Wait: ifft(X*Y)[k] = sum_m x[m] * y[(k-m) mod L]
    # ifft(fft(p) * conj(fft(a)))[k] = sum_m p[m] * a[(k-m) mod L]
    # For circular correlation: sum_m p[m] * a[(k-m) mod L]
    # For linear (with zero padding): sum_m p[m] * a[k-m] (where a is zero outside [0,L-1])
    # = sum_m p[m] * a[k-m] = sum_j p[j] * a[k-j] (substituting j=m)
    # Let j' = k-j: = sum_{j'} p[k-j'] * a[j'] = sum_{j'} a[j'] * p[k-j']
    # Hmm, this gives sum a[j'] * p[k-j'], but we want sum a[m] * p[m+j].
    
    # Let me try the other order: ifft(fft(a) * conj(fft(p)))[k] = sum_m a[m] * p[m-k]
    # We want sum_m a[m] * p[m+j] = sum_m a[m] * p[m-(-j)]
    # So we need ifft(fft(a) * conj(fft(p)))[-j mod L] = ifft(fft(a) * conj(fft(p)))[L-j]
    
    # Or: ifft(conj(fft(a)) * fft(p))[k] = sum_m a[m-k] * p[m] = sum_j a[j] * p[j+k]
    # So ifft(conj(fft(a)) * fft(p))[j] = sum_m a[m] * p[m+j] = what we want!
    
    corr = np.fft.irfft(np.conj(A) * P, L)[:n]
    return 2.0 * corr


def smoothed_objective(a, T):
    """Compute smoothed max of convolution: T * log(sum exp(c/T))."""
    c = fast_convolve(a)
    c_shifted = c - c.max()
    return T * np.sum(np.exp(c_shifted / T)) + c.max()


def adam_smoothed_opt(a0, time_limit, T_start=0.002, T_min=0.00005, lr_start=0.05):
    """Adam-based optimization of smoothed peak convolution."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    T = T_start
    lr = lr_start
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        c = fast_convolve(a)
        c_max = c.max()
        
        # Softmax weights
        scaled = (c - c_max) / T
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-15:
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
        else:
            p = p / p_sum
        
        # Gradient
        grad = hankel_grad(a, p)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        
        # Try multiple learning rates
        best_candidate = a
        best_val = c_max
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 2.0]:
            candidate = project_simplex(a - trial_lr * m_hat / (np.sqrt(v_hat) + eps))
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
        
        a = best_candidate
        
        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt T and lr
        if no_improve > 30:
            T = max(T * 0.95, T_min)
            lr *= 1.01
        if no_improve > 100:
            T = max(T * 0.8, T_min)
        if no_improve > 200:
            # Reset with perturbation
            rng = np.random.default_rng((it * 104729 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            m = np.zeros(n)
            v = np.zeros(n)
            T = T_start
            lr = lr_start
            no_improve = 0
        
        # Periodic T annealing
        if it % 500 == 0:
            T = max(T * 0.9, T_min)
    
    return best_a, best_obj


def fw_gradient_oracle(a, T, time_limit):
    """Frank-Wolfe with gradient-based oracle (no LP)."""
    f = np.maximum(a.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        c = fast_convolve(f)
        c_max = c.max()
        
        # Subgradient: use softmax for smoothing
        scaled = (c - c_max) / T
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Linearized objective: <grad, g> where grad = hankel_grad(f, p)
        grad = hankel_grad(f, p)
        
        # FW oracle: minimize <grad, g> over simplex => g = one-hot at argmin(grad)
        # But this is too greedy. Use: g = project of -grad onto simplex (which is just the min coordinate)
        # Actually for FW on simplex, the oracle is: g_j = 1 if j = argmin(grad), else 0
        min_idx = np.argmin(grad)
        g = np.zeros(n)
        g[min_idx] = 1.0
        
        # Line search
        best_alpha = 0.0
        best_val = c_max
        for alpha in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = (1 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            c_max_new = fast_convolve(f).max()
            if c_max_new < best_obj - 1e-14:
                best_obj = c_max_new
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 50:
            T *= 0.9
            if T < 0.0001:
                rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                T = 0.002
                no_improve = 0
    
    return best_f, best_obj


def local_search(a0, time_limit, seed=42):
    """Local search with random perturbations and structured moves."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        
        improved = False
        
        # Multi-scale random perturbations
        for scale in [0.0002, 0.0005, 0.001, 0.003, 0.008, 0.02]:
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(a + pert)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break
        
        # Coordinate moves
        if not improved and it % 3 == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.5]:
                    candidate = a.copy()
                    amount = candidate[i] * frac
                    candidate[i] -= amount
                    candidate[j] += amount
                    if candidate[i] < 0:
                        continue
                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break
        
        # Block moves: move a chunk of energy from one region to another
        if not improved and it % 5 == 0:
            block_size = int(n * rng.uniform(0.05, 0.2))
            i = int(rng.integers(0, max(1, n - block_size)))
            j = int(rng.integers(0, max(1, n - block_size)))
            if i != j and not (i < i + block_size and j < j + block_size and i + block_size > j):
                for frac in [0.05, 0.1, 0.2, 0.3]:
                    candidate = a.copy()
                    src_energy = candidate[i:i+block_size].sum() * frac
                    if src_energy < 1e-15:
                        continue
                    candidate[i:i+block_size] -= src_energy / block_size
                    candidate = np.maximum(candidate, 0)
                    candidate[j:j+block_size] += src_energy / block_size
                    ss = candidate.sum()
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break
        
        # Sinusoidal perturbation
        if not improved and it % 20 == 0:
            freq = rng.uniform(0.5, 10)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.001, 0.01)
            x = np.arange(n)
            pattern = np.sin(2 * np.pi * freq * x / n + phase) * amp
            candidate = project_simplex(a + pattern)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
        
        if not improved and it > 5000:
            break
    
    return best_a, best_obj


def try_seq(a, label, best_ref):
    """Evaluate a sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_ref
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_ref[1]:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return (a_norm.copy(), score)
    return best_ref

Wait, I need to double-check my `hankel_grad` function. Let me verify:

We want: grad[j] = 2 * sum_{m=0}^{n-1} a[m] * p[j+m] where p has length 2n-1 and j+m must be in [0, 2n-2].

For the FFT approach:
- We claim: ifft(conj(FFT(a)) * FFT(p))[j] = sum_m a[m] * p[m+j]

Let me verify with a small example. Let a = [a0, a1] (n=2), p = [p0, p1, p2] (length 3 = 2n-1).

We want:
grad[0] = 2*(a0*p0 + a1*p1)
grad[1] = 2*(a0*p1 + a1*p2)

"""
Hybrid optimization for minimum peak autocorrelation step function.
Uses Adam-based smoothed gradient descent with adaptive temperature,
diverse structural initializations, multi-n exploration,
FW with fast oracle, and intensive local search.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_grad(a, p):
    """Gradient of sum_k p[k]*(a*a)[k] w.r.t. a[j] = 2*sum_m a[m]*p[j+m]."""
    return 2.0 * np.correlate(p, a, 'valid')


def adam_smoothed(a0, time_limit, T_start=0.002, T_min=0.00005, lr_start=0.05):
    """Adam-based optimization of smoothed peak convolution."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8

    T = T_start
    lr = lr_start
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        c = fast_convolve(a)
        c_max = c.max()

        scaled = (c - c_max) / T
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
        else:
            p = p / p_sum

        grad = hankel_grad(a, p)

        m_vec = beta1 * m_vec + (1.0 - beta1) * grad
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad * grad)
        m_hat = m_vec / (1.0 - beta1 ** it)
        v_hat = v_vec / (1.0 - beta2 ** it)

        best_candidate = a
        best_val = c_max
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            step = m_hat / (np.sqrt(v_hat) + eps)
            candidate = project_simplex(a - trial_lr * step)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate

        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 20:
            T = max(T * 0.97, T_min)
            lr = min(lr * 1.005, 0.3)
        if no_improve > 80:
            T = max(T * 0.85, T_min)
        if no_improve > 200:
            rng = np.random.default_rng((it * 104729 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            m_vec = np.zeros(n)
            v_vec = np.zeros(n)
            T = max(T_start, T_min * 10)
            lr = lr_start
            no_improve = 0

        if it % 1000 == 0:
            T = max(T * 0.9, T_min)

    return best_a, best_obj


def fw_fast_oracle(a0, time_limit, T=0.001):
    """Frank-Wolfe with fast gradient oracle (one-hot at min gradient)."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        c = fast_convolve(f)
        c_max = c.max()

        scaled = (c - c_max) / T
        p = np.exp(scaled)
        p = p / p.sum()

        grad = hankel_grad(f, p)

        # Oracle: one-hot at argmin of gradient
        min_idx = np.argmin(grad)
        g = np.zeros(n)
        g[min_idx] = 1.0

        best_alpha = 0.0
        best_val = c_max
        for alpha in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            new_obj = fast_convolve(f).max()
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 30:
            T *= 0.9
            if T < 0.0001:
                rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                T = 0.001
                no_improve = 0

    return best_f, best_obj


def local_search(a0, time_limit, seed=42):

    a = np.maximum(a0.astype(np.float64), 0.0)

    n = len(a)

    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        improved = False

        # Multi-scale random perturbations
        for scale in [0.0001, 0.0003, 0.0008, 0.002, 0.005, 0.015]:
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(a + pert)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        # Coordinate moves

            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))

                    candidate = a.copy()
                    amount = candidate[i] * frac

                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate = candidate / ss
                    c = fast_convolve(candidate)

                        best_a = a.copy()
                        improved = True

                    if ss < 1e-10:

                    candidate = candidate / ss

                        best_a = a.copy()
                        improved = True

def gen_structures(n):
    """Generate diverse structural initializations of length n."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    structs = []

    # U-shape (high at edges)
    edge_dist = np.minimum(x, n - 1 - x) + 0.5
    u_shape = 1.0 / (edge_dist ** 0.5)
    structs.append(("u_shape", u_shape / u_shape.sum()))

    # Strong U-shape
    u_strong = 1.0 / (edge_dist ** 1.5)
    structs.append(("u_strong", u_strong / u_strong.sum()))

    # Inverted U (Gaussian bell)
    bell = np.exp(-0.5 * (x - center) ** 2 / (0.12 * n) ** 2)
    structs.append(("bell", bell / bell.sum()))

    # Two peaks
    g1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.08 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.8 * n) / (0.08 * n)) ** 2)
    structs.append(("two_peak", (g1 + g2) / (g1 + g2).sum()))

    # Three peaks
    g1 = np.exp(-0.5 * ((x - 0.1 * n) / (0.06 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.5 * n) / (0.06 * n)) ** 2)
    g3 = np.exp(-0.5 * ((x - 0.9 * n) / (0.06 * n)) ** 2)
    structs.append(("three_peak", (g1 + g2 + g3) / (g1 + g2 + g3).sum()))

    # Cosine window
    cos_win = 0.5 * (1.0 + np.cos(2 * np.pi * x / (n - 1)))
    structs.append(("cosine", cos_win / cos_win.sum()))

    # Exponential decay (right)
    exp_r = np.exp(-0.01 * x)
    structs.append(("exp_r", exp_r / exp_r.sum()))

    # Exponential decay (left)
    exp_l = np.exp(-0.01 * (n - 1 - x))
    structs.append(("exp_l", exp_l / exp_l.sum()))

    # Alternating blocks
    block_len = max(4, n // 32)
    alt = np.ones(n)
    for b in range(0, n, 2 * block_len):
        alt[b:b + block_len] = 0.3
    structs.append(("alt_block", alt / alt.sum()))

    # Power law
    pw = 1.0 / (edge_dist + 1.0) ** 0.8
    structs.append(("power", pw / pw.sum()))

    return structs
