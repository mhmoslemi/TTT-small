"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-smoothed gradient descent with Nesterov acceleration and adaptive beta
2. Coordinate descent with 1D ternary search (maintaining simplex constraint)
3. Frequency-domain perturbation for diversification
4. Multi-length exploration
5. Periodic restarts with different optimization parameters

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def evaluate_fast(a):
    """Fast evaluation of the objective without calling evaluate_sequence."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def lse_gradient(f, beta):
    """Compute gradient of (1/beta)*log(sum_k exp(beta*conv(f,f)[k])) wrt f.
    Returns (gradient, conv, lse_value)."""
    n = len(f)
    conv = fast_convolve(f)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    lse_val = (bc_max + np.log(total)) / beta
    if total < 1e-300:
        return np.zeros(n), conv, lse_val
    weights = exp_bc / total
    # Only process significant weights
    thresh = max(1e-12, 1.0 / (len(conv) * 10))
    active = np.where(weights > thresh)[0]
    grad = np.zeros(n)
    for k in active:
        w = weights[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += w * 2.0 * f[k - idx]
    return grad, conv, lse_val


def lse_optimize(f0, time_limit, beta_start=10.0, beta_end=8000.0, seed=42):
    """LSE-smoothed gradient descent with Nesterov acceleration."""
    f = project_simplex(np.maximum(f0.astype(np.float64), 0.0))
    n = len(f)
    rng = np.random.default_rng(seed)

    best_obj = evaluate_fast(f)
    best_f = f.copy()

    t0 = time.monotonic()
    beta = beta_start
    momentum = 0.85
    v = np.zeros(n)
    no_improve = 0
    it = 0
    lr_base = 2.0 / np.sqrt(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.5:
            break

        grad, conv, lse_val = lse_gradient(f, beta)

        # Adaptive step size based on gradient norm
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            no_improve += 10
            break

        # Nesterov step
        lr = lr_base / (1.0 + 0.01 * it)
        v = momentum * v - lr * grad / gnorm
        f_trial = project_simplex(f + v)

        obj_trial = evaluate_fast(f_trial)

        if obj_trial < best_obj - 1e-13:
            best_obj = obj_trial
            best_f = f_trial.copy()
            f = f_trial
            no_improve = 0
        else:
            # Backtracking
            improved = False
            for shrink in [0.5, 0.25, 0.1, 0.05, 0.02]:
                f_trial = project_simplex(f + shrink * v)
                obj_trial = evaluate_fast(f_trial)
                if obj_trial < best_obj - 1e-13:
                    best_obj = obj_trial
                    best_f = f_trial.copy()
                    f = f_trial
                    no_improve = 0
                    improved = True
                    break
            if not improved:
                no_improve += 1

        # Gradually increase beta
        if it % 30 == 0:
            progress = min(1.0, it / max(1, int(time_limit * 2.0)))
            beta = beta_start * (beta_end / beta_start) ** progress

        # Periodic perturbation to escape local minima
        if no_improve > 40:
            strength = 0.01 + 0.008 * min(1.0, no_improve / 60.0)
            F = np.fft.rfft(f)
            mag = np.abs(F)
            phase = np.angle(F)
            mag_pert = np.maximum(mag * (1.0 + rng.standard_normal(len(mag)) * strength), 0)
            F_new = mag_pert * np.exp(1j * phase)
            f_new = np.fft.irfft(F_new, n)
            f_new = project_simplex(np.maximum(f_new, 0))
            obj_new = evaluate_fast(f_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_f = f_new.copy()
            f = f_new
            v = np.zeros(n)
            no_improve = 0
            beta = max(beta_start, beta * 0.6)

    return best_f, best_obj


def coord_descent(f0, time_limit, n_sweeps=None, seed=42):
    """Coordinate descent with 1D ternary search on the simplex."""
    f = project_simplex(np.maximum(f0.astype(np.float64), 0.0))
    n = len(f)
    rng = np.random.default_rng(seed)

    best_obj = evaluate_fast(f)
    best_f = f.copy()
    t0 = time.monotonic()
    sweep = 0

    order = list(range(n))
    rng.shuffle(order)

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        if n_sweeps is not None and sweep > n_sweeps:
            break

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[i]
            if old_val < 1e-10:
                continue

            # Search range for f[i]
            lo = 0.0
            hi = min(1.0 - 1e-10, old_val * 4.0 + 0.5 / n)

            # Ternary search
            a = lo
            b = hi
            for _ in range(20):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0

                # Evaluate at m1: set f[i]=m1, scale others to maintain sum=1
                s1 = 1.0 - m1
                if s1 < 1e-15:
                    f_trial = np.zeros(n)
                    f_trial[i] = 1.0
                    obj1 = evaluate_fast(f_trial)
                else:
                    f_trial = f.copy()
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    scale1 = s1 / max(1e-15, 1.0 - old_val)
                    f_trial[i] = m1
                    f_trial[mask] = f_trial[mask] * scale1
                    obj1 = evaluate_fast(f_trial)

                s2 = 1.0 - m2
                if s2 < 1e-15:
                    f_trial = np.zeros(n)
                    f_trial[i] = 1.0
                    obj2 = evaluate_fast(f_trial)
                else:
                    f_trial = f.copy()
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    scale2 = s2 / max(1e-15, 1.0 - old_val)
                    f_trial[i] = m2
                    f_trial[mask] = f_trial[mask] * scale2
                    obj2 = evaluate_fast(f_trial)

                if obj1 < obj2:
                    b = m2
                else:
                    a = m1

            # Final evaluation at the best point
            best_x = (a + b) / 2.0
            s_best = 1.0 - best_x
            if s_best < 1e-15:
                f_trial = np.zeros(n)
                f_trial[i] = 1.0
                obj_best = evaluate_fast(f_trial)
            else:
                f_trial = f.copy()
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                scale_best = s_best / max(1e-15, 1.0 - old_val)
                f_trial[i] = best_x
                f_trial[mask] = f_trial[mask] * scale_best
                obj_best = evaluate_fast(f_trial)

            # Also check the original value
            if obj_best < best_obj - 1e-14:
                best_obj = obj_best
                best_f = f_trial.copy()
                f = f_trial
    return best_f, best_obj


def freq_perturb(f, rng, strength=0.03):
    """Perturb f in the frequency domain."""
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new
