    Returns (g, dual_vars) or (None, None).
    """
    n = len(f)
    if lags is None:
        lags = np.arange(2 * n - 1)
    lags = np.asarray(lags, dtype=int)
    n_rows = len(lags)

    A = build_sparse_conv(f, lags)
    c = -np.ones(n)
    b_ub = np.full(n_rows, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            g = np.maximum(res.x, 0.0)
            dual = np.zeros(n_rows)
            if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                dual = np.asarray(res.ineqlin.marginals, dtype=np.float64)
            return g, dual
    except Exception:
        pass
    return None, None


def fw_lp_enhanced(f0, time_limit):
    """Enhanced FW with sparse LP oracle and active constraint management."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    dual_weights = np.zeros(2 * n - 1)

    # Pre-build full lag array
    all_lags = np.arange(2 * n - 1)

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Determine which lags to use in LP
        if it > 5 and np.sum(dual_weights > 0) > 0:
            # Use top lags by dual weight + always include top conv lags
            n_top_dual = max(n // 2, min(2 * n - 1, int(1.2 * n)))
            top_dual = np.argpartition(dual_weights, -n_top_dual)[-n_top_dual:]
            n_top_conv = max(n // 2, min(2 * n - 1, int(0.8 * n)))
            top_conv = np.argpartition(conv, -n_top_conv)[-n_top_conv:]
            lags_use = np.unique(np.concatenate([top_dual, top_conv]))
        else:
            lags_use = all_lags

        # Solve LP
        lp_time = min(8.0, remaining * 0.4)
        g, dual = solve_lp_oracle(f, M * m_exp, lags=lags_use, time_limit=lp_time)

        if g is None:
            m_exp *= 1.15
            if m_exp > 4.0:
                m_exp = 1.0
                g = np.ones(n) / n
            else:
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Update dual weights
        if dual is not None:
            new_dw = dual_weights * 0.9
            for i, k in enumerate(lags_use):
                if dual[i] > 1e-6:
                    new_dw[int(k)] = max(new_dw[int(k)], dual[i])
            dual_weights = new_dw

        # Check convergence
        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 25:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.03)
            hi = min(1.0, best_alpha + 0.03)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        # Perturbation on stall
        if no_improve > 30:
            rng = np.random.default_rng(it * 104729 + 7)
            noise = rng.standard_normal(n) * 0.003
            # Bias: reduce peaks
            top_peaks = np.argpartition(conv, -min(15, len(conv)))[-min(15, len(conv)):]
            for k in top_peaks:
                js = max(0, int(k) - n + 1)
                je = min(n - 1, int(k))
                noise[js:je + 1] -= 0.001
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def nesterov_refine(f0, time_limit, lr_init=0.004, max_k=30, temperature=0.001):
    """Nesterov-accelerated subgradient with temperature-based peak weighting."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    y = a.copy()
    v = np.zeros(n)
    t_nest = 1.0
    lr = lr_init
    T = temperature
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Nesterov lookahead
        alpha_n = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / alpha_n
        y = (1.0 + beta) * a - beta * y
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y / ys
        t_nest += 1.0

        # Compute smooth gradient
        conv = fast_convolve(y)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.ones(len(conv)) / len(conv)
        else:
            p = p / p_sum

        # Gradient: grad_j = 2 * sum_k p[k] * (y[k-j] + y[j-k]) / p_sum_handled
        # More precisely: grad of sum_k p[k] * conv[k] w.r.t. y[j]
        # = 2 * sum_k p[k] * y[k-j] (for k-j in [0,n-1])
        grad = np.zeros(n)
        # Use convolution: grad = 2 * (p * y) restricted to valid range
        # p has length 2n-1, y has length n
        # (p * y)[j] = sum_k p[k] * y[k-j] for k-j in [0, n-1]
        # This is a cross-correlation
        # grad[j] = 2 * sum_{k} p[k] * y[k-j] where 0 <= k-j < n and 0 <= k < 2n-1
        # Equivalently: grad = 2 * correlate(p, y, mode='valid')... 
        # Let's just do it explicitly for correctness
        # Actually use FFT:
        # grad[j] = 2 * sum_k p[k] * y[k-j] for valid k,j
        # This is the correlation of p and y, shifted
        fft_size = 1 << (2 * n - 1 + n - 1).bit_length()
        P = np.fft.rfft(p, fft_size)
        Y = np.fft.rfft(y, fft_size)
        corr = np.fft.irfft(P * Y, fft_size)
        # corr[m] = sum_k p[k] * y[(k+m) mod fft_size]
        # We want grad[j] = 2 * sum_k p[k] * y[k-j]
        # = 2 * sum_k p[k] * y[(k-j) mod fft_size]
        # = 2 * corr[j] (by the correlation property)
        # Actually let me verify: if p has length 2n-1 and y has length n,
        # then (p * y) in circular convolution sense gives us what we want
        # corr[j] = sum_k p[k] * y[(k+j) mod fft]
        # We want: sum_k p[k] * y[k-j] = sum_k p[k] * y[(k-j)]
        # Let m = -j: sum_k p[k] * y[(k+m)] = corr[m] = corr[-j mod fft]
        # So grad[j] = 2 * corr[(-j) % fft_size]
        neg_idx = (-np.arange(n)) % fft_size
        grad = 2.0 * corr[neg_idx]

        # Update with Nesterov momentum
        v = beta * v - lr * grad
        a_new = a + v
        a_new = project_simplex(a_new)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            a_new = a

        a = a_new
        lr *= (1.0 - 0.0001)

        # Temperature schedule
        if no_improve > 50:
            T *= 1.05
        if no_improve > 150:
            T *= 0.7

        # Restart
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.001 + 0.0005 * min(1.0, no_improve / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            y = a.copy()
            v = np.zeros(n)
            t_nest = 1.0
            no_improve = 0
            T = temperature

    return best_a, best_obj


def local_quadratic_polish(f0, time_limit):
    """Polish using local quadratic approximation around active peaks.
    
    At the optimum, the max is achieved at a few lags k*. Near the optimum,
    we can approximate the objective as max of quadratics and do a Dinkelbach-style
    iteration or solve a small QP.
    """
    a = np.maximum(f0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Find active peaks (within 99% of max)
        threshold = M * 0.99
        active = np.where(conv >= threshold)[0]

        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])

        # For each active peak, compute the local quadratic:
        # conv[k] = sum_j a[j] * a[k-j]
        # Around a, perturbation d: conv[k] ≈ conv[k] + 2*d^T H_k a + d^T (H_k + H_k^T) d / 2
        # where H_k is the Hankel matrix for lag k
        # But we're on the simplex, so use projected approach

        # Simple approach: for each active peak, try to reduce it
        # by moving mass between pairs of coordinates
        improved = False

        for k in active:
            k = int(k)
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            affected = list(range(js, je + 1))

            if len(affected) < 2:
                continue

            # Sample pairs if too many
            if len(affected) > 40:
                rng_s = np.random.default_rng(it * 997 + k)
                sampled = list(rng_s.choice(affected, size=40, replace=False))
            else:
                sampled = affected

            best_gain = 0.0
            best_move = None

            for ii in range(len(sampled)):
                for jj in range(ii + 1, len(sampled)):
                    j1 = sampled[ii]
                    j2 = sampled[jj]

                    # Contribution of a[j1] to peak k
                    p1 = k - j1
                    c1 = a[p1] if 0 <= p1 < n else 0.0
                    # Contribution of a[j2] to peak k
                    p2 = k - j2
                    c2 = a[p2] if 0 <= p2 < n else 0.0

                    # Move mass from higher contributor to lower
                    if c1 > c2 and a[j1] > 1e-10:
                        src, dst = j1, j2
                        # Optimal delta: minimize (a[src]-d)*c1 * c1' + ... 
                        # Simple: try a few fractions
                        for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                            d = a[src] * frac
                            if d < 1e-12:
                                continue
                            a_t = a.copy()
                            a_t[src] -= d
                            a_t[dst] += d
                            a_t = np.maximum(a_t, 0.0)
                            s_t = a_t.sum()
                            if s_t < 1e-15:
                                continue
                            a_t = a_t / s_t
                            c_t = fast_convolve(a_t)
                            new_max = c_t.max()
                            gain = M - new_max
                            if gain > best_gain:
                                best_gain = gain
                                best_move = (src, dst, frac)
                    elif c2 > c1 and a[j2] > 1e-10:
                        src, dst = j2, j1
                        for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                            d = a[src] * frac
                            if d < 1e-12:
                                continue
                            a_t = a.copy()
                            a_t[src] -= d
                            a_t[dst] += d
                            a_t = np.maximum(a_t, 0.0)
                            s_t = a_t.sum()
                            if s_t < 1e-15:
                                continue
                            a_t = a_t / s_t
                            c_t = fast_convolve(a_t)
                            new_max = c_t.max()
                            gain = M - new_max
                            if gain > best_gain:
                                best_gain = gain
                                best_move = (src, dst, frac)

            if best_move is not None and best_gain > 1e-12:
                src, dst, frac = best_move
                d = a[src] * frac
                a[src] -= d
                a[dst] += d
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                new_max = fast_convolve(a).max()
                if new_max < best_obj - 1e-14:
                    best_obj = new_max
                    best_a = a.copy()
                    improved = True
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1

        if not improved:
            stall += 1

        if stall > 50:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise_scale = 0.0003 + 0.001 * min(1.0, stall / 100.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            stall = 0

        if stall > 300:
            break

    return best_a, best_obj


def micro_polish(f0, time_limit):
    """Very fine structured perturbation polish."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    scales = [0.00005, 0.0001, 0.0002, 0.0005, 0.001]
    si = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        scale = scales[si % len(scales)]
        rng = np.random.default_rng((it * 2654435761 + 1) % (2**32))

        found = False
        for _ in range(20):
            if time.monotonic() - t0 > time_limit - 0.05:
                break
            conv = fast_convolve(a)
            peak_k = int(np.argmax(conv))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            noise = rng.standard_normal(n) * scale
            # Structured bias: reduce values contributing to peak
            for j in range(js, je + 1):
                partner = peak_k - j
                if 0 <= partner < n:
                    noise[j] -= scale * 3.0 * a[partner]

            candidate = project_simplex(a + noise)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                found = True
                break

        if not found and it % 50 == 0:
            si += 1

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase optimization:
    1. Load parent and evaluate
    2. Quick multi-n exploration
    3. Deep FW-LP with sparse oracle
    4. Nesterov refinement
    5. Local quadratic polish
    6. Micro-polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t0_total = time.monotonic()

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={time.monotonic()-t0_total:.0f}s)")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")

    n_parent = len(parent)

    # === Phase 1: Quick multi-n exploration ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent, max(256, n_parent - 128), min(1200, n_parent + 128),
                                512, 640, 768, 896, 1024]))
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 1200]

    for n_try in candidate_ns:
        if time.monotonic() > deadline - 1200:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            # Interpolate parent to new length
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            start = np.interp(x_new, x_old, parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss > 1e-10:
                start = start / ss
            else:
                start = np.ones(n_try) / n_try

        t_alloc = min(25, max(5, deadline - time.monotonic() - 1200))
        print(f"  FW-LP n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_enhanced(start, time_limit=t_alloc)
        try_seq(a_opt, f"explore_n{n_try}")

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW-LP on best ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        t_alloc = min(400, max(20, deadline - time.monotonic() - 700))
        print(f"  Deep FW n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_enhanced(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "deep_fw")

        # Perturbed restarts
        for scale in [0.003, 0.008, 0.015]:
            t_alloc = min(50, max(8, deadline - time.monotonic() - 550))
            if t_alloc < 5:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            a_opt, _ = fw_lp_enhanced(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"fw_pert{scale}")

    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Nesterov Subgradient ===
    print("\n=== Phase 3: Nesterov Subgradient ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        configs = [
            {'lr_init': 0.003, 'max_k': 20, 'temperature': 0.0005},
            {'lr_init': 0.006, 'max_k': 40, 'temperature': 0.001},
            {'lr_init': 0.002, 'max_k': 15, 'temperature': 0.0003},
            {'lr_init': 0.010, 'max_k': 50, 'temperature': 0.002},
        ]
        for ci, cfg in enumerate(configs):
            t_alloc = min(100, max(10, deadline - time.monotonic() - 400))
            if t_alloc < 8:
                break
            print(f"  Nesterov cfg{ci} lr={cfg['lr_init']} T={cfg['temperature']}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_refine(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"nest_{ci}")

    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Local Quadratic Polish ===
    print("\n=== Phase 4: Local Quadratic Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(150, max(10, deadline - time.monotonic() - 200))
        if t_alloc > 5:
            print(f"  Local quad polish, t={t_alloc:.0f}s")
            a_opt, _ = local_quadratic_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "local_quad")

    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Micro-Polish ===
    print("\n=== Phase 5: Micro-Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(80, max(5, deadline - time.monotonic() - 80))
        if t_alloc > 3:
            print(f"  Micro polish, t={t_alloc:.0f}s")
            a_opt, _ = micro_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "micro")

        # Another Nesterov pass with very small temperature
        t_alloc = min(50, max(5, deadline - time.monotonic() - 50))
        if t_alloc > 3:
            a_opt, _ = nesterov_refine(best_seq, time_limit=t_alloc,
                                        lr_init=0.002, max_k=10, temperature=0.0002)
            try_seq(a_opt, "nest_fine")

    print(f"  Phase 5 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Final FW + Polish ===
    print("\n=== Phase 6: Final Refinement ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(60, max(5, deadline - time.monotonic() - 30))
        if t_alloc > 3:
            print(f"  Final FW, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_enhanced(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_fw")

        t_alloc = min(30, max(3, deadline - time.monotonic() - 15))
        if t_alloc > 2:
            a_opt, _ = micro_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_micro")

    print(f"  Phase 6 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, total time={time.monotonic()-t0_total:.0f}s)")
    return [float(x) for x in best_seq]
