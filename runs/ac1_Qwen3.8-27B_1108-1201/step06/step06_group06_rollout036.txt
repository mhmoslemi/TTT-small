import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def coord_descent(a, time_limit, random_order=False, seed=42):
    """Coordinate descent minimizing max of auto-convolution."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    t0 = time.monotonic()
    
    b = fast_convolve(a)
    best_max = b.max()
    
    while time.monotonic() - t0 < time_limit:
        if random_order:
            order = np.random.permutation(n)
        else:
            order = range(n)
        
        any_changed = False
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            
            # Compute D[k] = a[k-j] for valid k (where 0 <= k-j < n)
            # and C[k] = b[k] - old_val * D[k]
            idx = np.arange(2*n-1) - j
            valid_mask = (idx >= 0) & (idx < n)
            D = np.zeros(2*n-1)
            D[valid_mask] = a[idx[valid_mask]]
            
            # But we need to be careful: when k-j == j, the term is a[j]^2, 
            # and changing a[j] by delta changes it by 2*delta*a[j] - delta^2
            # For the linear approximation: C[k] = b[k] - old_val * D[k]
            # This is exact for the linear part. The quadratic part (delta^2) 
            # only appears when k-j = j.
            
            # For simplicity, use the linear approximation and verify
            C = b - old_val * D
            
            # Find t in [0, t_hi] minimizing max(C + t*D)
            # Check t=0
            f0 = C.max()
            
            # Upper bound for t
            t_hi = max(old_val * 3, 0.01)
            f_hi = (C + t_hi * D).max()
            if f_hi >= f0:
                # T=0 is better
                if old_val > 1e-10:
                    a[j] = 0
                    # Update b
                    b = fast_convolve(a)
                    any_changed = True
                continue
            
            # Bisection/ternary search
            t_lo = 0.0
            for _ in range(25):
                t1 = t_lo + (t_hi - t_lo) / 3.0
                t2 = t_hi - (t_hi - t_lo) / 3.0
                f1 = (C + t1 * D).max()
                f2 = (C + t2 * D).max()
                if f1 < f2:
                    t_hi = t2
                else:
                    t_lo = t1
            
            t_opt = (t_lo + t_hi) / 2.0
            new_val = max(0.0, t_opt)
            
            if abs(new_val - old_val) > 1e-12:
                a[j] = new_val
                any_changed = True
        
        # Renormalize
        s = a.sum()
        if s > 1e-15:
            a = a / s
        
        b = fast_convolve(a)
        new_max = b.max()
        
        if new_max < best_max - 1e-14:
            best_max = new_max
        elif not any_changed:
            break
    
    return a

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Optimized search for minimum peak autocorrelation step function.
Uses smoothed gradient with Nesterov momentum, adaptive temperature scheduling,
simulated annealing acceptance, diverse structured restarts, and multi-n exploration.
Key improvements over previous: momentum, better T schedule, structured perturbations,
and more aggressive exploration of n-space.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_fft(a, p, fft_size=None):
    """Compute gradient of sum_k p[k]*b[k] w.r.t. a.
    grad[j] = 2 * sum_m a[m] * p[j+m]
    Uses FFT-based cross-correlation.
    """
    n = len(a)
    if fft_size is None:
        fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(A * np.conj(P), fft_size)
    return 2.0 * cross[:n]


def smooth_grad_optimize(a0, time_limit, T_init=0.001, lr_init=0.03,
                         momentum=0.9, seed=42, use_annealing=True,
                         anneal_T0=0.01, anneal_T1=0.0001):
    """Smoothed gradient descent with Nesterov momentum and simulated annealing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    T = T_init
    lr = lr_init
    v = np.zeros(n)  # momentum
    no_improve = 0
    it = 0
    anneal_T = anneal_T0

    # Precompute FFT size
    fft_size = 1 << (3 * n - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        current_max = conv.max()

        # Softmax weights
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            p = p / p_sum

        # Gradient
        grad = compute_grad_fft(a, p, fft_size)

        # Nesterov momentum update
        v = momentum * v + lr * grad
        a_candidate = project_simplex(a - v)

        # Evaluate candidate
        new_conv = fast_convolve(a_candidate)
        new_obj = new_conv.max()

        # Accept/reject (simulated annealing)
        acceptable = False
        if new_obj < current_max - 1e-15:
            acceptable = True
        elif use_annealing and anneal_T > 1e-6:
            delta = new_obj - current_max
            if delta < 0.01:  # Only accept moderately worse
                prob = np.exp(-delta / anneal_T)
                if rng.random() < prob:
                    acceptable = True

        if acceptable:
            a = a_candidate
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            # Reduce momentum slightly when rejected
            v = 0.9 * v

        # Adaptive parameters
        if no_improve > 30:
            T = min(T * 1.03, 0.1)
            lr *= 0.98
        if no_improve > 100:
            T *= 0.8
            lr = max(lr * 0.8, 0.001)
            v = 0.5 * v
        if no_improve > 200:
            # Aggressive restart
            scale = rng.uniform(0.002, 0.01)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            T = T_init
            lr = lr_init
            no_improve = 0
        if no_improve > 400:
            # Very aggressive restart
            scale = rng.uniform(0.01, 0.05)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            T = T_init
            lr = lr_init
            no_improve = 0

        # Anneal acceptance temperature
        if use_annealing:
            progress = min(1.0, (time.monotonic() - t0) / time_limit)
            anneal_T = anneal_T0 * (anneal_T1 / anneal_T0) ** progress

    return best_a, best_obj


def fw_lp_optimize(f0, time_limit, seed=42):
    """Frank-Wolfe with LP oracle (simplified version)."""
    from scipy.optimize import linprog
    from scipy.sparse import lil_matrix

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    # Build sparse constraint matrix
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    row_idx = np.array(rows)
    col_idx = np.array(cols)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        conv = fast_convolve(f)
        M = conv.max() * m_exp

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0
        c = -np.ones(n)
        vals = np.empty(len(row_idx))
        for idx in range(len(row_idx)):
            k = row_idx[idx]
            i = col_idx[idx]
            vals[idx] = f[k - i]

        A_mat = lil_matrix((2 * n - 1, n))
        A_mat[row_idx, col_idx] = vals
        A_mat = A_mat.tocsr()
        b = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                          options={'presolve': True})
            if res.success and res.x is not None:
                g = np.maximum(res.x, 0.0)
                gs = g.sum()
                if gs > 1e-15:
                    g = g / gs
                else:
                    g = None
            else:
                g = None
        except Exception:
            g = None

        if g is None:
            m_exp = min(m_exp * 1.5, 8.0)
            if m_exp >= 8.0:
                rng = np.random.default_rng(it * 31 + 7 + seed)
                g = project_simplex(f + rng.standard_normal(n) * 0.01)
                m_exp = 1.0
            else:
                continue

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 7919 + 13 + seed)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                m_exp = 1.0
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv_new = fast_convolve(f)
            M_new = conv_new.max()
            m_exp = 1.0

            if M_new < best_obj - 1e-15:
                best_obj = M_new
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.15, 8.0)

    return best_f, best_obj


def local_search(a0, time_limit, seed=42):
    """Local search with diverse move types."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        improved = False

        # Gaussian perturbation at multiple scales
        for scale in [0.0005, 0.001, 0.003, 0.008, 0.02]:
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(a + pert)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        # Symmetric pair move: transfer mass between i and n-1-i
        if not improved:
            i = int(rng.integers(0, n // 2))
            j = n - 1 - i
            for frac in [0.05, 0.1, 0.2, 0.4]:
                for direction in [1, -1]:
                    candidate = a.copy()
                    amt = candidate[i] * frac * direction
                    candidate[i] -= amt
                    candidate[j] += amt
                    candidate = np.maximum(candidate, 0)
                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        # Block transfer
        if not improved and it % 3 == 0:
            block_size = int(rng.integers(max(2, n // 32), max(3, n // 8)))
            i = int(rng.integers(0, n - block_size))
            j = int(rng.integers(0, n - block_size))
            if abs(i - j) > block_size:
                for frac in [0.1, 0.3, 0.5]:
                    candidate = a.copy()
                    block = candidate[i:i + block_size].sum() * frac
                    if block > 1e-15:
                        candidate[i:i + block_size] -= candidate[i:i + block_size] * frac
                        candidate[j:j + block_size] += candidate[j:j + block_size] * frac + block * (1 - frac)
                        candidate = np.maximum(candidate, 0)
                        ss = candidate.sum()
                        if ss > 1e-10:
                            candidate = candidate / ss
                            c = fast_convolve(candidate)
                            if c.max() < best_obj - 1e-15:
                                a = candidate
                                best_obj = c.max()
                                best_a = a.copy()
                                improved = True
                                break

        # Sinusoidal pattern addition
        if not improved and it % 5 == 0:
            freq = rng.uniform(1, 12)
            phase = rng.uniform(0, 2 * np.pi)
            for amp in [0.001, 0.003, 0.01]:
                pattern = np.sin(2 * np.pi * freq * np.arange(n) / n + phase) * amp
                candidate = project_simplex(a + pattern)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            if stall > 200:
                # Restart from best with larger perturbation
                scale = rng.uniform(0.01, 0.05)
                a = project_simplex(best_a + rng.standard_normal(n) * scale)
                stall = 0
        else:
            stall = 0

    return best_a, best_obj


def try_seq(a, label, best_ref):
    """Evaluate a sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_ref
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_ref
    if np.isfinite(score) and score < best_ref[1]:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return (a_norm.copy(), score)
    return best_ref


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point: Multi-phase hybrid search.
    Phase 1: Quick multi-n exploration
    Phase 2: Deep smoothed gradient + FW-LP on best n
    Phase 3: Diverse restarts with structured perturbations
    Phase 4: Final polish with local search
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Using uniform init: score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Quick multi-n exploration ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([256, 384, 512, 640, 768, 896, 1024, 1280, 1536, 2048]))

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic()
        if remaining < 200:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try

        # Quick smoothed gradient
        t_alloc = min(remaining * 0.03, 15)
        a_opt, _ = smooth_grad_optimize(start, time_limit=t_alloc,
                                         T_init=0.001, lr_init=0.03,
                                         momentum=0.9, seed=seed + n_try)
        best_seq, best_score = try_seq(a_opt, f"sg_n{n_try}", (best_seq, best_score))

        # Quick FW-LP
        remaining2 = deadline - time.monotonic()
        if remaining2 > 150:
            t_alloc2 = min(remaining2 * 0.03, 12)
            a_opt2, _ = fw_lp_optimize(a_opt, time_limit=t_alloc2, seed=seed)
            best_seq, best_score = try_seq(a_opt2, f"fw_n{n_try}", (best_seq, best_score))

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep optimization on best n ===
    print("\n=== Phase 2: Deep Optimization ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    # Deep smoothed gradient with momentum
    if remaining > 200:
        t_alloc = min(remaining * 0.3, 400)
        print(f"  Deep smooth grad n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = smooth_grad_optimize(best_seq, time_limit=t_alloc,
                                         T_init=0.0005, lr_init=0.02,
                                         momentum=0.92, seed=seed + 100,
                                         use_annealing=True,
                                         anneal_T0=0.005, anneal_T1=0.0001)
        best_seq, best_score = try_seq(a_opt, "deep_sg_mom", (best_seq, best_score))

    # Deep FW-LP
    remaining = deadline - time.monotonic()
    if remaining > 150:
        t_alloc = min(remaining * 0.25, 300)
        print(f"  Deep FW-LP n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, seed=seed + 200)
        best_seq, best_score = try_seq(a_opt, "deep_fw", (best_seq, best_score))

    # Smoothed gradient with different params
    remaining = deadline - time.monotonic()
    if remaining > 100:
        t_alloc = min(remaining * 0.15, 150)
        print(f"  Smooth grad (high momentum, low T), t={t_alloc:.0f}s")
        a_opt, _ = smooth_grad_optimize(best_seq, time_limit=t_alloc,
                                         T_init=0.0002, lr_init=0.01,
                                         momentum=0.95, seed=seed + 300,
                                         use_annealing=True,
                                         anneal_T0=0.002, anneal_T1=0.00005)
        best_seq, best_score = try_seq(a_opt, "sg_high_mom", (best_seq, best_score))

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Diverse restarts ===
    print("\n=== Phase 3: Diverse Restarts ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)

    # Structured perturbations
    restart_configs = [
        # (perturbation_type, params, T_init, lr_init, momentum)
        ("gaussian", {"scale": 0.003}, 0.001, 0.03, 0.9),
        ("gaussian", {"scale": 0.01}, 0.001, 0.03, 0.9),
        ("gaussian", {"scale": 0.03}, 0.002, 0.02, 0.85),
        ("sinusoidal", {"freq": 3.0, "amp": 0.01}, 0.001, 0.03, 0.9),
        ("sinusoidal", {"freq": 7.0, "amp": 0.005}, 0.001, 0.03, 0.9),
        ("block", {"block": n_best // 16, "frac": 0.2}, 0.001, 0.03, 0.9),
    ]

    for ri, (ptype, params, T_i, lr_i, mom) in enumerate(restart_configs):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        # Generate perturbed start
        if ptype == "gaussian":
            pert_start = project_simplex(best_seq + rng.standard_normal(n_best) * params["scale"])
        elif ptype == "sinusoidal":
            pattern = np.sin(2 * np.pi * params["freq"] * np.arange(n_best) / n_best) * params["amp"]
            pert_start = project_simplex(best_seq + pattern)
        elif ptype == "block":
            pert_start = best_seq.copy()
            bs = params["block"]
            bi = int(rng.integers(0, max(1, n_best - bs)))
            bj = int(rng.integers(0, max(1, n_best - bs)))
            if abs(bi - bj) > bs:
                amt = pert_start[bi:bi + bs].sum() * params["frac"]
                pert_start[bi:bi + bs] *= (1 - params["frac"])
                pert_start[bj:bj + bs] += amt
                pert_start = np.maximum(pert_start, 0)
                pert_start = pert_start / pert_start.sum()
            else:
                pert_start = project_simplex(best_seq + rng.standard_normal(n_best) * 0.005)
        else:
            pert_start = project_simplex(best_seq + rng.standard_normal(n_best) * 0.005)

        t_alloc = min(remaining * 0.15, 100)
        a_opt, _ = smooth_grad_optimize(pert_start, time_limit=t_alloc,
                                         T_init=T_i, lr_init=lr_i,
                                         momentum=mom, seed=seed + 1000 + ri * 17,
                                         use_annealing=True,
                                         anneal_T0=0.003, anneal_T1=0.0001)
        best_seq, best_score = try_seq(a_opt, f"restart_{ptype}_{ri}", (best_seq, best_score))

    # FW-LP from perturbed starts
    for scale in [0.005, 0.015, 0.04]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * scale)
        t_alloc = min(remaining * 0.12, 80)
        a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc, seed=seed + 5000)
        best_seq, best_score = try_seq(a_opt, f"fw_pert_{scale}", (best_seq, best_score))

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Alternative n values ===
    print("\n=== Phase 4: Alternative n ===")
    t_phase = time.monotonic()

    for n_try in [len(best_seq) * 2, len(best_seq) // 2, len(best_seq) + 128, len(best_seq) - 128]:
        if n_try < 128 or n_try > 4096:
            continue
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        start = np.interp(np.linspace(0, 1, n_try),
                          np.linspace(0, 1, len(best_seq)), best_seq)
        start = np.maximum(start, 0)
        ss = start.sum()
        start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try

        t_alloc = min(remaining * 0.15, 100)
        a_opt, _ = smooth_grad_optimize(start, time_limit=t_alloc,
                                         T_init=0.001, lr_init=0.03,
                                         momentum=0.9, seed=seed + 8000 + n_try)
        best_seq, best_score = try_seq(a_opt, f"alt_n{n_try}", (best_seq, best_score))

        remaining = deadline - time.monotonic()
        if remaining > 60 and len(best_seq) == n_try:
            t_alloc2 = min(remaining * 0.1, 60)
            a_opt2, _ = fw_lp_optimize(best_seq, time_limit=t_alloc2, seed=seed + 9000)
            best_seq, best_score = try_seq(a_opt2, f"fw_alt_n{n_try}", (best_seq, best_score))

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final polish ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    # Local search
    if remaining > 40:
        t_alloc = min(remaining * 0.25, 100)
        print(f"  Local search, t={t_alloc:.0f}s")
        a_opt, _ = local_search(best_seq, time_limit=t_alloc, seed=12345)
        best_seq, best_score = try_seq(a_opt, "local1", (best_seq, best_score))

    # Fine smoothed gradient
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_alloc = min(remaining * 0.2, 80)
        print(f"  Fine smooth (T=0.0001), t={t_alloc:.0f}s")
        a_opt, _ = smooth_grad_optimize(best_seq, time_limit=t_alloc,
                                         T_init=0.0001, lr_init=0.005,
                                         momentum=0.95, seed=seed + 15000,
                                         use_annealing=True,
                                         anneal_T0=0.001, anneal_T1=0.00001)
        best_seq, best_score = try_seq(a_opt, "fine_smooth", (best_seq, best_score))

    # Another FW-LP pass
    remaining = deadline - time.monotonic()
    if remaining > 25:
        t_alloc = min(remaining * 0.2, 60)
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, seed=seed + 16000)
        best_seq, best_score = try_seq(a_opt, "final_fw", (best_seq, best_score))

    # Micro perturbations
    remaining = deadline - time.monotonic()
    if remaining > 20:
        for micro_i in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            noise = rng.standard_normal(n_best) * 0.0003
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = smooth_grad_optimize(f_pert, time_limit=min(remaining * 0.25, 10),
                                             T_init=0.0002, lr_init=0.003,
                                             momentum=0.95, seed=seed + 17000 + micro_i)
            best_seq, best_score = try_seq(a_opt, f"micro_{micro_i}", (best_seq, best_score))

    # Final local search
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_opt, _ = local_search(best_seq, time_limit=min(remaining * 0.3, 30), seed=99999)
        best_seq, best_score = try_seq(a_opt, "local2", (best_seq, best_score))

    # Tiny final smooth
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_opt, _ = smooth_grad_optimize(best_seq, time_limit=min(remaining * 0.4, 15),
                                         T_init=0.00005, lr_init=0.002,
                                         momentum=0.97, seed=seed + 18000,
                                         use_annealing=True,
                                         anneal_T0=0.0005, anneal_T1=0.00001)
        best_seq, best_score = try_seq(a_opt, "tiny_final", (best_seq, best_score))

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
