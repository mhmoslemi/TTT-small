"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Gerchberg-Saxton style initialization for flat power spectrum
2. Frank-Wolfe with full constraint LP (all 2n-1 auto-correlation constraints)
3. Multi-length exploration with interpolation between lengths
4. Golden section line search for precise step selection
5. Peak-targeted coordinate refinement for final polish
6. Diversified restarts with frequency-domain perturbation
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_full_lp_matrix(f):
    """Build sparse constraint matrix for LP with ALL 2n-1 constraints."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            i_range = np.arange(lo, hi + 1)
            rows.extend([k] * len(i_range))
            cols.extend(i_range.tolist())
            vals.extend(f[k - i_range].tolist())
    if not rows:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def gs_init(n, n_iters=300, rng=None):
    """Gerchberg-Saxton style initialization aiming for flat power spectrum."""
    if rng is None:
        rng = np.random.default_rng(42)

    a = rng.random(n) + 0.05
    a /= a.sum()

    best_a = a.copy()
    best_conv_max = fast_convolve(a).max()

    for it in range(n_iters):
        F = rfft(a)
        mag = np.abs(F)
        phase = np.angle(F)

        # Target: approximately flat power spectrum
        # Keep DC component, flatten the rest
        target_mag = np.ones(len(mag))
        target_mag[0] = 1.0  # DC component (sum of a = 1)

        # Adaptive blending: more aggressive early, less later
        alpha = 0.3 * (1.0 - it / n_iters) + 0.1
        new_mag = alpha * target_mag + (1.0 - alpha) * mag
        new_mag[0] = mag[0]  # Keep DC exact

        F_new = new_mag * np.exp(1j * phase)
        a_new = irfft(F_new, n)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s

        conv_max = fast_convolve(a_new).max()
        if conv_max < best_conv_max:
            best_conv_max = conv_max
            best_a = a_new.copy()

        if np.max(np.abs(a_new - a)) < 1e-12:
            break
        a = a_new

    return np.maximum(best_a, 0.0)


def fw_full_constraints(f0, time_limit, verbose=True):
    """Frank-Wolfe with LP oracle using ALL auto-correlation constraints."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    phi = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Build LP matrix with current f
        A = build_full_lp_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        # Solve LP
        lp_time = min(remaining * 0.4, 20.0)
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': lp_time})
            if not res.success or res.x is None:
                no_improve += 5
                if no_improve > 40:
                    break
                M *= 1.001  # Relax slightly
                continue
        except Exception:
            no_improve += 10
            if no_improve > 30:
                break
            continue

        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        if gs < 1e-15:
            break
        g /= gs

        # Check if direction is meaningful
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            if no_improve > 25:
                break
            continue

        # Golden section line search
        a_lo = 0.0
        a_hi = 1.0

        def obj_fn(alpha):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                return float('inf')
            ft /= st
            return fast_convolve(ft).max()

        c_pt = a_hi - phi * (a_hi - a_lo)
        d_pt = a_lo + phi * (a_hi - a_lo)
        c_val = obj_fn(c_pt)
        d_val = obj_fn(d_pt)

        for _ in range(35):
            if abs(a_hi - a_lo) < 1e-7:
                break
            if c_val < d_val:
                a_hi = d_pt
                d_pt = c_pt
                d_val = c_val
                c_pt = a_hi - phi * (a_hi - a_lo)
                c_val = obj_fn(c_pt)
            else:
                a_lo = c_pt
                c_pt = d_pt
                c_val = d_val
                d_pt = a_lo + phi * (a_hi - a_lo)
                d_val = obj_fn(d_pt)

        best_alpha = (a_lo + a_hi) / 2.0
        # Also check alpha = 1.0
        val_1 = obj_fn(1.0)
        val_best = obj_fn(best_alpha)
        if val_1 < val_best:
            best_alpha = 1.0
            best_alpha_val = val_1
        else:
            best_alpha_val = val_best

        f_new = (1.0 - best_alpha) * f + best_alpha * g
        f_new = np.maximum(f_new, 0.0)
        f_new /= f_new.sum()

        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()

        if M_new < best_obj - 1e-15:
            if verbose and it % 20 == 0:
                print(f"    FW iter {it}: M={M_new:.8f} (alpha={best_alpha:.4f})")
            best_obj = M_new
            best_f = f_new.copy()
            f = f_new
            M = M_new
            no_improve = 0
        else:
            no_improve += 1
            if verbose and no_improve % 10 == 0:
                print(f"    FW iter {it}: no improve ({no_improve}), M={M_new:.8f}")
            if no_improve > 50:
                # Try relaxing M slightly
                M *= 1.002
            if no_improve > 100:
                break

    return best_f, best_obj


def peak_coordinate_refine(f0, time_limit, n_rounds=200):
    """Targeted coordinate descent on the top auto-correlation peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f /= f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        # Get top 5 peaks
        k_top = min(5, len(conv))
        peak_idx = np.argsort(conv)[-k_top:][::-1]

        improved_this_round = False
        for k in peak_idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo >= hi:
                continue

            # (f*f)[k] = sum_{i=lo}^{hi} f[i] * f[k-i]
            # Try reducing the largest contributing pair
            pairs = f[lo:hi + 1] * f[k - np.arange(lo, hi + 1)]
            if pairs.max() < 1e-15:
                continue

            j_max = lo + int(np.argmax(pairs))
            j_other = k - j_max
            if j_other < 0 or j_other >= n or j_other == j_max:
                continue

            # Move small amount of mass from j_max to a random location
            delta = f[j_max] * 0.02
            if delta < 1e-15:
                continue

            # Try a few random destination positions
            for _ in range(5):
                j_dest = int(rng.integers(0, n))
                if j_dest == j_max or j_dest == j_other:
                    continue

                f_trial = f.copy()
                f_trial[j_max] = f[j_max] - delta
                f_trial[j_dest] = f[j_dest] + delta
                if f_trial[j_max] < 0:
                    continue
                f_trial = np.maximum(f_trial, 0.0)
                f_trial /= f_trial.sum()
                conv_trial = fast_convolve(f_trial)
                if conv_trial.max() < best_obj - 1e-12:
                    best_obj = conv_trial.max()
                    best_f = f_trial.copy()
                    f = f_trial.copy()
                    improved_this_round = True
                    break

        if not improved_this_round and rnd > 50:
            # Add small noise to escape
            f = f + rng.standard_normal(n) * 0.001
            f = np.maximum(f, 0.0)
            f /= f.sum()

    return best_f, best_obj


def freq_perturb(f, rng, strength=0.03):
    """Perturb f in the frequency domain for diversification."""
    n = len(f)
    F = rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0.0)
    mag_new[0] = mag[0]  # Keep DC
    F_new = mag_new * np.exp(1j * phase)
    f_new = irfft(F_new, n)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def subgradient_polish(f0, time_limit, beta_start=100.0, beta_end=5000.0, top_k=40):
    """Subgradient method with LSE smoothing for final polish."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 0.5 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    def project_simplex(v):
        nn = len(v)
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.arange(1, nn + 1)
        cond = u * rho > (cssv - 1)
        if not cond.any():
            return np.ones(nn) / nn
        rho_val = np.nonzero(cond)[0][-1]
        theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
        return np.maximum(v - theta, 0.0)

    def lse_weights(conv_val, beta_val):
        bc = beta_val * conv_val
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        total = exp_bc.sum()
        if total < 1e-300:
            return np.ones(len(conv_val)) / len(conv_val)
        return exp_bc / total

    def compute_grad(f_cur, peak_idx, weights):
        nn = len(f_cur)
        grad = np.zeros(nn)
        for idx, k in enumerate(peak_idx):
            k = int(k)
            w = weights[idx]
            if w < 1e-15:
                continue
            lo = max(0, k - nn + 1)
            hi = min(nn - 1, k)
            i_range = np.arange(lo, hi + 1)
            grad[i_range] += w * 2.0 * f_cur[k - i_range]
        return grad

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        conv = fast_convolve(f)
        w_full = lse_weights(conv, beta)
        k_top = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        w_peaks = w_full[peak_idx]
        w_sum = w_peaks.sum()
        if w_sum > 1e-15:
            w_peaks = w_peaks / w_sum
        else:
            w_peaks = np.ones(k_top) / k_top

        grad = compute_grad(f, peak_idx, w_peaks)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_normed = grad / gnorm

        # Backtracking line search
        step = lr
        improved = False
        for _ in range(12):
            f_new = project_simplex(f - step * grad_normed)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                improved = True
                no_improve = 0
                lr = min(lr * 1.01, 5.0 / np.sqrt(n))
                break
            step *= 0.5

        if not improved:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.85
            if no_improve > 60:
                f = best_f.copy()
                lr = max(0.3 / np.sqrt(n), 0.002)
                no_improve = 0
                rng = np.random.default_rng(it * 31 + 17)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)

        if beta < beta_end:
            beta *= 1.005

    return best_f, best_obj

    # Budget: give more time to lengths near parent
    total_p1 = min(budget_s * 0.45, 700.0)
    t1_end = time.monotonic() + total_p1

    for i, nc in enumerate(all_lengths):
        remaining_p1 = t1_end - time.monotonic()
        if remaining_p1 < 30:
            break

        # Allocate more time to parent length
        if nc == n_parent:
            t_alloc = min(remaining_p1 * 0.35, 200.0)
        else:
            t_alloc = min(remaining_p1 / max(1, len(all_lengths) - i), 120.0)

        if t_alloc < 15:
            continue

        print(f"  Length n={nc}, time={t_alloc:.0f}s")

        # Initialize
        if nc == n_parent and parent is not None:
            f_init = parent.copy()
        else:
            f_init = gs_init(nc, n_iters=150, rng=rng)
            # Also try interpolating parent
            f_interp = interpolate_to_length(parent, nc)
            # Pick the better initialization
            conv_gs = fast_convolve(f_init).max()
            conv_int = fast_convolve(f_interp).max()
            if conv_int < conv_gs:
                f_init = f_interp

        # FW optimization
        f_opt, obj = fw_full_constraints(f_init, time_limit=t_alloc * 0.7, verbose=False)
        try_sequence(f_opt, best_tracker)

        # Quick subgradient polish
        t_rem = t1_end - time.monotonic()
        if t_rem > 15:
            f_sg, _ = subgradient_polish(f_opt, time_limit=min(t_rem * 0.5, 40.0),
                                         beta_start=50.0, beta_end=2000.0, top_k=30)
            try_sequence(f_sg, best_tracker)

    # Phase 2: Refined optimization on best candidate
    print(f"\n=== Phase 2: Refined optimization (best n={len(best_tracker[1]) if best_tracker[1] is not None else 'N/A'}) ===")
    t2_end = deadline - 200.0
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)
    remaining_p2 = t2_end - time.monotonic()
    if remaining_p2 > 30:
        print(f"  FW refinement n={n_cur}, time={remaining_p2:.0f}s")
        # Start from best, do more careful FW
        f_opt, obj = fw_full_constraints(current_best, time_limit=remaining_p2 * 0.6, verbose=True)
        try_sequence(f_opt, best_tracker)

        # Subgradient with higher beta
        t_rem = t2_end - time.monotonic()
        if t_rem > 20:
            print(f"  Subgradient polish n={n_cur}, time={t_rem*0.6:.0f}s")
            f_sg, _ = subgradient_polish(best_tracker[1], time_limit=t_rem * 0.6,
                                         beta_start=200.0, beta_end=10000.0, top_k=20)
            try_sequence(f_sg, best_tracker)

    # Phase 3: Peak coordinate refinement
    print(f"\n=== Phase 3: Peak coordinate refinement ===")
    t3_end = deadline - 100.0
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    remaining_p3 = t3_end - time.monotonic()
    if remaining_p3 > 10:
        print(f"  Coordinate refine n={len(current_best)}, time={remaining_p3:.0f}s")
        f_coord, _ = peak_coordinate_refine(current_best, time_limit=remaining_p3)
        try_sequence(f_coord, best_tracker)

    # Phase 4: Diversified restarts
    print(f"\n=== Phase 4: Diversified restarts ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    remaining_p4 = deadline - time.monotonic()
    n_restarts = min(8, max(1, int(remaining_p4 / 60)))
    print(f"  {n_restarts} restarts, total time={remaining_p4:.0f}s")

    for r in range(n_restarts):
        t_rem = deadline - time.monotonic()
        if t_rem < 30:
            break
        t_alloc = min(t_rem / (n_restarts - r), 120.0)
        if t_alloc < 15:
            break

        strength = 0.01 + 0.005 * r
        f_pert = freq_perturb(current_best, rng, strength=strength)
        print(f"  Restart {r+1}: strength={strength:.4f}, time={t_alloc:.0f}s")
        f_opt, _ = fw_full_constraints(f_pert, time_limit=t_alloc * 0.7, verbose=False)
        try_sequence(f_opt, best_tracker)

        # Also try time-domain perturbation
        f_pert2 = current_best + rng.standard_normal(len(current_best)) * 0.005
        f_pert2 = np.maximum(f_pert2, 0.0)
        f_pert2 /= f_pert2.sum()
        t_rem2 = deadline - time.monotonic()
        if t_rem2 > 20:
            f_opt2, _ = fw_full_constraints(f_pert2, time_limit=min(t_rem2 * 0.3, 40.0), verbose=False)
            try_sequence(f_opt2, best_tracker)

        current_best = best_tracker[1] if best_tracker[1] is not None else current_best

    # Phase 5: Final high-precision polish
    print(f"\n=== Phase 5: Final polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_rem = deadline - time.monotonic()
    if t_rem > 20:
        print(f"  Final FW (very careful) n={len(current_best)}, time={t_rem*0.5:.0f}s")
        f_final, _ = fw_full_constraints(current_best, time_limit=t_rem * 0.5, verbose=False)
        try_sequence(f_final, best_tracker)

        t_rem = deadline - time.monotonic()
        if t_rem > 15:
            print(f"  Final subgradient n={len(current_best)}, time={t_rem*0.7:.0f}s")
            f_sg, _ = subgradient_polish(best_tracker[1], time_limit=t_rem * 0.7,
                                         beta_start=500.0, beta_end=20000.0, top_k=15)
            try_sequence(f_sg, best_tracker)

        t_rem = deadline - time.monotonic()
        if t_rem > 10:
            print(f"  Final coordinate refine, time={t_rem:.0f}s")
            f_coord, _ = peak_coordinate_refine(best_tracker[1], time_limit=t_rem, n_rounds=500)
            try_sequence(f_coord, best_tracker)

    # Final verification
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.12f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    print("WARNING: Using fallback")
    fallback = np.ones(768) / 768.0
    return [float(x) for x in fallback]
