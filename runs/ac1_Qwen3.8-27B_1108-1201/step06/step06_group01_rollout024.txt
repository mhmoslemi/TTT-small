"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Fast multiplicative peak reduction (exponentiated gradient with adaptive K)
2. Projected gradient with log-sum-exp smoothing (FFT-accelerated)
3. Targeted coordinate descent with peak-aware mass transfers
4. Multi-n exploration, structured restarts, and fine polishing

Key innovations: replaces expensive LP oracle with O(K*n) multiplicative updates,
uses FFT-accelerated gradients, adaptive peak tracking for smooth convergence.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    d = np.nonzero(cond)[0][-1]
    theta = (cssv[d] - 1.0) / (d + 1.0)
    return np.maximum(v - theta, 0.0)


def multiplicative_peak_reduce(a_init, time_limit, K_init=8, eta_init=0.2):
    """
    Iteratively reduce peaks using multiplicative weight updates.
    a[j] <- a[j] * exp(-eta * pressure[j]) / norm
    where pressure[j] = sum over top-K peaks of weight_k * a[k-j]
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    eta = eta_init
    K = K_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        current_max = conv.max()

        if K >= len(conv):
            peak_idx = np.arange(len(conv))
            peak_vals = conv.copy()
        else:
            peak_idx = np.argpartition(conv, -K)[-K:]
            peak_vals = conv[peak_idx]

        # Weight by excess above median (focus on tallest peaks)
        med = np.median(peak_vals)
        excess = np.maximum(peak_vals - med, 0.0)
        if excess.sum() < 1e-15:
            weights = np.ones(len(peak_idx)) / len(peak_idx)
        else:
            weights = excess / excess.sum()

        # Compute pressure vector
        pressure = np.zeros(n)
        for idx in range(len(peak_idx)):
            w = weights[idx]
            if w < 1e-14:
                continue
            k = int(peak_idx[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            partners = k - np.arange(js, je + 1)
            pressure[js:je + 1] += w * a[partners]

        max_p = pressure.max()
        if max_p < 1e-15:
            break

        # Adaptive eta scaling
        effective_eta = eta * (1.0 / max_p) * min(1.0, 0.1 / max_p)
        effective_eta = min(effective_eta, 2.0)
        if effective_eta < 1e-6:
            effective_eta = eta * 0.01

        update = np.exp(-effective_eta * pressure)
        a_new = a * update
        s_new = a_new.sum()
        if s_new < 1e-15:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        new_max = fast_convolve(a_new).max()
        if new_max < current_max - 1e-14:
            a = a_new
            if new_max < best_obj:
                best_obj = new_max
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Adaptive parameters
        if no_improve > 2:
            eta = max(eta * 0.9, 0.01)
        if no_improve > 10:
            eta = max(eta * 0.75, 0.005)
        if no_improve > 30:
            K = min(K + 5, 60)
            eta = eta_init * 0.5
            no_improve = 10
        if no_improve > 80:
            # Perturbation restart
            rng = np.random.default_rng((it * 104729 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            K = K_init
            eta = eta_init
            no_improve = 0
        elif no_improve == 0 and it % 200 == 0:
            K = min(K + 2, 50)

    return best_a, best_obj


def lse_gradient_step(a, T):
    """
    Compute gradient of smooth approximation sum_k exp(conv_k/T) w.r.t. a.
    Uses FFT for O(n log n) computation.
    Returns (gradient, smooth_value).
    """
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-30:
        return np.zeros(n), 0.0
    p = p / p_sum
    smooth_val = T * np.log(p_sum) + T * scaled.max()

    # grad[j] = 2 * sum_m a[m] * p[j+m]
    # Use FFT: conv_full = flip(a_padded) * p, take positions [n-1 : 2n-1]
    size = 2 * n - 1
    fft_size = 1 << (2 * size - 1).bit_length()

    A = np.zeros(size)
    A[:n] = a
    A_flip = A[::-1]

    FA = np.fft.rfft(A_flip, fft_size)
    FP = np.fft.rfft(p, fft_size)
    conv_full = np.fft.irfft(FA * FP, fft_size)

    grad = 2.0 * conv_full[n - 1:2 * n - 1]
    return grad, smooth_val


def lse_projected_gradient(a_init, time_limit, T_init=0.0008, lr_init=0.03):
    """Projected gradient descent on log-sum-exp smoothed objective."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    T = T_init
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        grad, _ = lse_gradient_step(a, T)

        # Backtracking line search
        best_candidate = a
        best_val = fast_convolve(a).max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            T *= 1.08
            lr *= 0.92
        if no_improve > 50:
            T *= 0.85
        if no_improve > 150:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def coordinate_descent_peaks(a_init, time_limit):
    """
    Targeted coordinate descent: move mass between pairs to reduce top peaks.
    Uses responsibility analysis to choose efficient pairs.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    stall = 0
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        # Get top 3 peaks
        top_k = min(3, len(conv))
        peak_lags = np.argpartition(conv, -top_k)[-top_k:]
        peak_lags = peak_lags[np.argsort(conv[peak_lags])[::-1]]

        improved = False

        for k_peak in peak_lags:
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            # Find affected indices
            js = max(0, int(k_peak) - n + 1)
            je = min(n - 1, int(k_peak))

            if je - js + 1 < 4:
                continue

            # Compute responsibility for this peak
            resp = np.zeros(n)
            partners = int(k_peak) - np.arange(js, je + 1)
            resp[js:je + 1] = a[partners]

            # Also compute total responsibility across all top peaks
            total_resp = np.zeros(n)
            for kp in peak_lags:
                js2 = max(0, int(kp) - n + 1)
                je2 = min(n - 1, int(kp))
                p2 = int(kp) - np.arange(js2, je2 + 1)
                total_resp[js2:je2 + 1] += a[p2]

            # Sample pairs: high-resp source, low-resp destination
            affected = np.arange(js, je + 1)
            if len(affected) > 40:
                sampled = rng.choice(affected, size=40, replace=False)
            else:
                sampled = affected

            # Sort by responsibility
            order = np.argsort(resp[sampled])[::-1]
            high_resp = sampled[order[:10]]
            low_resp = sampled[order[-10:]]

            best_gain = 0
            best_move = None

            for src in high_resp:
                for dst in low_resp:
                    if src == dst:
                        continue
                    if a[src] < 1e-8 or total_resp[src] <= total_resp[dst]:
                        continue
                    # Try moving a fraction of a[src] to dst
                    for frac in [0.05, 0.1, 0.2, 0.35]:
                        delta = a[src] * frac
                        if delta < 1e-10:
                            continue
                        a_test = a.copy()
                        a_test[src] -= delta
                        a_test[dst] += delta
                        a_test = np.maximum(a_test, 0.0)
                        st = a_test.sum()
                        if st < 1e-15:
                            continue
                        a_test = a_test / st
                        c_new = fast_convolve(a_test)
                        new_max = c_new.max()
                        gain = best_obj - new_max
                        if gain > best_gain:
                            best_gain = gain
                            best_move = (src, dst, delta)

            if best_move is not None and best_gain > 1e-12:
                src, dst, delta = best_move
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                c_new = fast_convolve(a)
                if c_new.max() < best_obj:
                    best_obj = c_new.max()
                    best_a = a.copy()
                    improved = True
                    stall = 0

        if not improved:
            stall += 1
            if stall > 15:
                noise_scale = 0.001 + 0.002 * min(1.0, stall / 40.0)
                a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
                stall = 0
            if stall > 200:
                break

    return best_a, best_obj


def structured_initialization(n, seed=42):
    """Generate structured initializations that tend to have lower peak autocorrelation."""
    rng = np.random.default_rng(seed)

    # Option 1: Cosine-squared window
    x = np.linspace(0, 1, n)
    a1 = np.cos(np.pi * x) ** 2
    a1 = np.maximum(a1, 0)
    a1 = a1 / a1.sum()

    # Option 2: Triangular (Bartlett)
    a2 = np.maximum(0, 1 - np.abs(2 * x - 1))
    a2 = a2 / a2.sum()

    # Option 3: Gaussian window
    a3 = np.exp(-4 * (x - 0.5) ** 2)
    a3 = a3 / a3.sum()

    # Option 4: Random smoothed
    a4 = rng.standard_normal(n)
    a4 = np.abs(a4)
    # Apply a few smoothing iterations
    for _ in range(3):
        a4 = np.convolve(a4, [0.25, 0.5, 0.25], mode='same')
    a4 = np.maximum(a4, 0)
    a4 = a4 / a4.sum()

    # Option 5: Alternating pattern (spread out)
    a5 = np.ones(n)
    # Add periodic structure
    for i in range(n):
        a5[i] = 1.0 + 0.5 * np.cos(2 * np.pi * 3 * i / n)
    a5 = np.maximum(a5, 0.01)
    a5 = a5 / a5.sum()

    candidates = [a1, a2, a3, a4, a5]
    # Evaluate and return the best
    best_a = candidates[0]
    best_val = fast_convolve(best_a).max()
    for c in candidates[1:]:
        v = fast_convolve(c).max()
        if v < best_val:
            best_val = v
            best_a = c
    return best_a


def interpolate_sequence(a, target_n):
    """Interpolate a sequence to a target length."""
    n = len(a)
    if n == target_n:
        return a.copy()
    if target_n == 1:
        return np.array([a.sum()])
    x_orig = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, target_n)
    result = np.interp(x_new, x_orig, a)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def annealing_polish(a_init, time_limit):
    """Simulated annealing style: accept moves that don't increase objective too much early on."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    current_obj = best_obj

    t0 = time.monotonic()
    total_time = time_limit
    rng = np.random.default_rng(999)

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # Temperature schedule (exponential cooling)
        progress = (time.monotonic() - t0) / total_time
        T = 0.01 * (1.0 - progress) ** 2 + 0.0001

        # Move: pick two random indices, transfer mass
        if n < 2:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.uniform(0.05, 0.3)
        delta = a[i] * frac
        if delta < 1e-10:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        new_obj = fast_convolve(a_new).max()
        acceptance = (new_obj - current_obj) / T if T > 1e-10 else float('inf')
        if new_obj < current_obj or (T > 0.0005 and rng.random() < np.exp(-acceptance)):
            a = a_new
            current_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search combining multiplicative updates,
    smoothed gradient descent, coordinate descent, and annealing.
    """
    rng = np.random.default_rng(seed)
    t0_total = time.monotonic()
    deadline = t0_total + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t0_total
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(384) / 384

    n_parent = len(parent)

    # === Phase 1: Multi-n exploration with multiplicative updates (~350s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent,
                                max(256, n_parent - 128),
                                n_parent + 128,
                                512, 640, 768, 896, 1024, 1152, 1280]))
    candidate_ns = [n for n in candidate_ns if 320 <= n <= 1400]

    # Deduplicate and sort
    candidate_ns = sorted(set(candidate_ns))

    for n_try in candidate_ns:
        if time.monotonic() > deadline - 1200:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = interpolate_sequence(parent, n_try)

        # Also try structured initialization for shorter sequences
        if n_try <= 800 and n_try != n_parent:
            start_struct = structured_initialization(n_try, seed=seed + n_try)
            # Quick compare
            v1 = fast_convolve(start).max()
            v2 = fast_convolve(start_struct).max()
            if v2 < v1:
                start = start_struct

        t_alloc = min(35, deadline - time.monotonic() - 1200)
        if t_alloc < 8:
            break
        print(f"  Multiplicative n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = multiplicative_peak_reduce(start, time_limit=t_alloc, K_init=8, eta_init=0.2)
        try_seq(a_opt, f"mult_n{n_try}")

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep multiplicative optimization (~450s) ===
    print("\n=== Phase 2: Deep Multiplicative Optimization ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Deep run with more careful parameters
        t_alloc = min(200, deadline - time.monotonic() - 700)
        if t_alloc > 20:
            print(f"  Deep multiplicative n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = multiplicative_peak_reduce(best_seq, time_limit=t_alloc, K_init=6, eta_init=0.15)
            try_seq(a_opt, "deep_mult")

        # Multiple perturbations + re-optimize
        for pi, scale in enumerate([0.003, 0.008, 0.015]):
            t_alloc = min(60, deadline - time.monotonic() - 550)
            if t_alloc < 10:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            a_opt, _ = multiplicative_peak_reduce(f_pert, time_limit=t_alloc, K_init=10, eta_init=0.12)
            try_seq(a_opt, f"mult_pert{pi}")

    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Log-sum-exp gradient (~350s) ===
    print("\n=== Phase 3: LSE Projected Gradient ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        configs = [
            {'T_init': 0.0005, 'lr_init': 0.025},
            {'T_init': 0.002, 'lr_init': 0.015},
            {'T_init': 0.001, 'lr_init': 0.03},
        ]
        for ci, cfg in enumerate(configs):
            t_alloc = min(110, deadline - time.monotonic() - 400)
            if t_alloc < 10:
                break
            print(f"  LSE grad T={cfg['T_init']}, lr={cfg['lr_init']}, t={t_alloc:.0f}s")
            a_opt, _ = lse_projected_gradient(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"lse_{ci}")

    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Coordinate descent (~250s) ===
    print("\n=== Phase 4: Coordinate Descent ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(220, deadline - time.monotonic() - 200)
        if t_alloc > 10:
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent_peaks(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord")

    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Annealing polish + final multiplicative (~200s) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Annealing
        t_alloc = min(80, deadline - time.monotonic() - 100)
        if t_alloc > 10:
            print(f"  Annealing, t={t_alloc:.0f}s")
            a_opt, _ = annealing_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "anneal")

        # Final multiplicative with very small steps
        t_alloc = min(60, deadline - time.monotonic() - 50)
        if t_alloc > 5:
            print(f"  Final multiplicative, t={t_alloc:.0f}s")
            a_opt, _ = multiplicative_peak_reduce(best_seq, time_limit=t_alloc,
                                                   K_init=20, eta_init=0.05)
            try_seq(a_opt, "final_mult")

        # One more LSE pass
        t_alloc = min(40, deadline - time.monotonic() - 20)
        if t_alloc > 5:
            a_opt, _ = lse_projected_gradient(best_seq, time_limit=t_alloc,
                                               T_init=0.0003, lr_init=0.01)
            try_seq(a_opt, "final_lse")

    print(f"  Phase 5 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Try different n near the best (if time remains) ===
    print("\n=== Phase 6: Neighbor n exploration ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        for dn in [-64, 64, -128, 128]:
            n_try = n_best + dn
            if n_try < 320 or n_try > 1500:
                continue
            t_alloc = min(30, deadline - time.monotonic() - 15)
            if t_alloc < 5:
                break
            start = interpolate_sequence(best_seq, n_try)
            a_opt, _ = multiplicative_peak_reduce(start, time_limit=t_alloc)
            try_seq(a_opt, f"neighbor_n{n_try}")

    print(f"  Phase 6 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, total time={time.monotonic()-t0_total:.0f}s)")
    return [float(x) for x in best_seq]
