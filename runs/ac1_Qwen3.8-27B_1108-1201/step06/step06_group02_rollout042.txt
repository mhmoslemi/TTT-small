def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_soft_grad(a, T):
    """Compute the gradient of the logsumexp-smoothed max of conv(a,a)."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()
    # grad_i = 2 * sum_{m=0}^{n-1} p[m+i] * a[m]
    corr = np.correlate(p, a, mode='full')
    grad = 2.0 * corr[n - 1:n - 1 + n]
    return grad, conv, p


def adam_smooth_optimize(a0, time_limit, T_start, T_end, lr_init, seed=0):
    """Adam optimizer with logsumexp smoothing and temperature annealing."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr_init
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Temperature annealing (geometric)
        T = T_start * (T_end / T_start) ** progress
        
        # Compute gradient
        grad, conv, p = compute_soft_grad(a, T)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Project gradient to simplex tangent (sum to 0)
        grad_step = m_hat / (np.sqrt(v_hat) + eps)
        grad_step -= np.mean(grad_step)
        
        candidate = a - lr * grad_step
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s > 1e-15:
            candidate /= s
        
        obj = fast_convolve(candidate).max()
        if obj < best_obj + 1e-13:
            a = candidate
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Adapt learning rate
        if no_improve > 30:
            lr *= 0.95
        if no_improve > 100:
            lr = max(lr * 0.8, 1e-6)
        if no_improve > 300:
            # Restart with perturbation
            scale = 0.001 + 0.003 * min(1.0, (no_improve - 300) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = lr_init * 0.7
            no_improve = 0
        if no_improve > 600:
            break
    
    return best_a, best_obj


def lbfgs_smooth_optimize(a0, T, maxiter=10000):

    n = len(a0)

        # smoothed_max = T * log(sum(exp(conv_k / T)))

        # = max(conv) + T * log(sum(exp((conv - max(conv))/T)))

        obj = cmax + T * np.log(np.exp((conv - cmax) / T).sum())
        
        corr = np.correlate(p, a, mode='full')

def self_consistent_field(a0, time_limit, eta_init=0.1, seed=0):
    """Self-consistent field iteration: a_i * exp(-eta * V_i) where V_i is the convolution field."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    eta = eta_init
    rng = np.random.default_rng(seed)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        cmax = conv.max()
        
        # Compute the "field" V_i = sum_k (conv_k / cmax) * a_{k-i}
        # This is the same structure as the gradient but with hard max weights
        # Actually let's use the softmax for smoothness:
        T = 0.01 * cmax
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p /= p.sum()
        
        corr = np.correlate(p, a, mode='full')
        V = corr[n - 1:n - 1 + n]
        
        # Update: a_new_i ∝ a_i * exp(-eta * V_i / V_max)
        V_max = V.max()
        if V_max < 1e-15:
            break
        logits = np.log(np.maximum(a, 1e-15)) - eta * V / V_max
        a_new = np.exp(logits - logits.max())
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj:
            a = a_new
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            # Partial update (under-relaxation)
            for alpha in [0.5, 0.3, 0.1]:
                a_try = (1 - alpha) * a + alpha * a_new
                a_try = np.maximum(a_try, 0)
                st = a_try.sum()
                if st < 1e-15:
                    continue
                a_try /= st
                obj_try = fast_convolve(a_try).max()
                if obj_try < best_obj:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    no_improve = 0
                    break
                elif obj_try < fast_convolve(a).max():
                    a = a_try
                    no_improve += 1
                    break
            else:
                no_improve += 1
        
        eta *= 0.999
        if no_improve > 50:
            eta = max(eta * 1.5, 0.01)
        if no_improve > 200:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            eta = eta_init
            no_improve = 0
        if no_improve > 500:
            break
    
    return best_a, best_obj


def multi_peak_subgrad(a0, time_limit, top_k=20, lr_init=0.01, seed=0):
    """Multi-peak subgradient with adaptive step size."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    lr = lr_init
    rng = np.random.default_rng(seed)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        cmax = conv.max()
        
        # Find top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peak_indices = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peak_indices]
            order = np.argsort(pv)[::-1]
            weights = np.zeros(len(peak_indices))
            for rank, idx in enumerate(order):
                weights[idx] = 1.0 / (rank + 1) ** 1.5
            ws = weights.sum()
            weights /= ws
            peaks = peak_indices
        
        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]
        
        # Normalize and project
        sg -= np.mean(sg)
        candidate = a - lr * sg
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s > 1e-15:
            candidate /= s
        
        obj = fast_convolve(candidate).max()
        if obj < best_obj - 1e-14:
            a = candidate
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            # Try smaller step
            for frac in [0.5, 0.25, 0.1]:
                c2 = a - lr * frac * sg
                c2 = np.maximum(c2, 0.0)
                s2 = c2.sum()
                if s2 > 1e-15:
                    c2 /= s2
                obj2 = fast_convolve(c2).max()
                if obj2 < best_obj - 1e-14:
                    a = c2
                    best_obj = obj2
                    best_a = a.copy()
                    no_improve = 0
                    break
            else:
                no_improve += 1
        
        lr *= 0.9995
        if no_improve > 100:
            lr = max(lr * 0.9, 1e-6)
        if no_improve > 300:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            lr = lr_init * 0.7
            no_improve = 0
        if no_improve > 600:
            break
    
    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit, seed=0):
    """Coordinate descent: move mass from peak-contributing to non-contributing positions."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    rng = np.random.default_rng(seed)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        
        # Find the coordinate that most contributes to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]
        
        # Find the best position to move mass to
        # We want a position that doesn't contribute much to any peak
        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            break
        
        # Score each outside position by how much it would increase peaks
        # A simple heuristic: move to the position with smallest a value
        j_min = outside[np.argmin(a[outside])]
        
        # Try different amounts
        improved = False
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            delta = frac * a[j_max]
            if delta < 1e-12:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min] += delta
            candidate = np.maximum(candidate, 0.0)
            candidate /= candidate.sum()
            
            obj = fast_convolve(candidate).max()
            if obj < best_obj - 1e-14:
                a = candidate
                best_obj = obj
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                # Random perturbation
                scale = 0.001 + 0.002 * min(1.0, no_improve / 500.0)
                candidate = project_simplex(a + rng.standard_normal(n) * scale)
                obj = fast_convolve(candidate).max()
                if obj < best_obj - 1e-14:
                    a = candidate
                    best_obj = obj
                    best_a = a.copy()
                    no_improve = 0
            if no_improve > 1000:
                break
    
    return best_a, best_obj


def symmetric_project(a):
    """Project onto the palindromic subspace: a_i = a_{n-1-i}."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines Adam with logsumexp smoothing, L-BFGS-B, self-consistent field iteration,
multi-peak subgradient, and coordinate descent polish. Explores multiple sequence
lengths and uses symmetric projection as a structural improvement.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_soft_grad(a, T):
    """Compute gradient of logsumexp-smoothed max of conv(a,a)."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()
    corr = np.correlate(p, a, mode='full')
    grad = 2.0 * corr[n - 1:n - 1 + n]
    return grad, conv, p


def adam_smooth_optimize(a0, time_limit, T_start, T_end, lr_init, seed=0):
    """Adam optimizer with logsumexp smoothing and geometric temperature annealing."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr_init
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1e-6))
        T = T_start * (T_end / T_start) ** progress

        grad, conv, p = compute_soft_grad(a, T)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        grad_step = m_hat / (np.sqrt(v_hat) + eps)
        grad_step -= np.mean(grad_step)

        candidate = a - lr * grad_step
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s > 1e-15:
            candidate /= s

        obj = fast_convolve(candidate).max()
        if obj <= best_obj + 1e-13:
            a = candidate
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 30:
            lr *= 0.95
        if no_improve > 100:
            lr = max(lr * 0.85, 1e-7)
        if no_improve > 250:
            scale = 0.001 + 0.004 * min(1.0, (no_improve - 250) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = lr_init * 0.7
            no_improve = 0
        if no_improve > 600:
            break

    return best_a, best_obj


def lbfgs_smooth_optimize(a0, T, maxiter=10000):
    """L-BFGS-B on smoothed objective with simplex constraint handled via projection."""
    n = len(a0)

    def obj_and_grad(x):
        a = np.maximum(x, 0.0)
        s = a.sum()
        if s < 1e-15:
            return 1e10, np.zeros(n)
        a = a / s
        conv = fast_convolve(a)
        cmax = conv.max()
        scaled = (conv - cmax) / T
        p = np.exp(scaled)
        p /= p.sum()
        obj = cmax + T * np.log(p.sum())
        corr = np.correlate(p, a, mode='full')
        grad = 2.0 * corr[n - 1:n - 1 + n]
        grad -= np.mean(grad)
        return obj, grad

    try:
        res = minimize(obj_and_grad, a0, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                                'maxls': 50})
        a = np.maximum(res.x, 0.0)
        s = a.sum()
        if s > 1e-15:
            a /= s
        return a
    except Exception:
        return a0


def self_consistent_field(a0, time_limit, eta_init=0.08, seed=0):
    """SCF iteration: a_i proportional to a_i * exp(-eta * V_i) where V is conv field."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    eta = eta_init
    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        cmax = conv.max()
        T = 0.02 * cmax
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p /= p.sum()

        corr = np.correlate(p, a, mode='full')
        V = corr[n - 1:n - 1 + n]
        V_max = V.max()
        if V_max < 1e-15:
            break

        logits = np.log(np.maximum(a, 1e-15)) - eta * V / V_max
        a_new = np.exp(logits - logits.max())
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            a = a_new
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            improved_sub = False
            for alpha in [0.5, 0.3, 0.1, 0.05]:
                a_try = (1.0 - alpha) * a + alpha * a_new
                a_try = np.maximum(a_try, 0.0)
                st = a_try.sum()
                if st < 1e-15:
                    continue
                a_try /= st
                obj_try = fast_convolve(a_try).max()
                if obj_try < best_obj - 1e-14:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    no_improve = 0
                    improved_sub = True
                    break
                elif obj_try < fast_convolve(a).max() - 1e-15:
                    a = a_try
                    no_improve += 1
                    improved_sub = True
                    break
            if not improved_sub:
                no_improve += 1

        eta *= 0.9995
        if no_improve > 40:
            eta = max(eta * 1.3, 0.01)
        if no_improve > 150:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            eta = eta_init
            no_improve = 0
        if no_improve > 400:
            break

    return best_a, best_obj


def multi_peak_subgrad(a0, time_limit, top_k=20, lr_init=0.008, seed=0):
    """Multi-peak subgradient with adaptive step and backtracking."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    lr = lr_init
    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peak_indices = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peak_indices]
            order = np.argsort(pv)[::-1]
            weights = np.zeros(len(peak_indices))
            for rank, idx in enumerate(order):
                weights[idx] = 1.0 / (rank + 1) ** 1.5
            ws = weights.sum()
            weights /= ws
            peaks = peak_indices

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]

        sg -= np.mean(sg)

        improved = False
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            candidate = a - lr * frac * sg
            candidate = np.maximum(candidate, 0.0)
            s = candidate.sum()
            if s < 1e-15:
                continue
            candidate /= s
            obj = fast_convolve(candidate).max()
            if obj < best_obj - 1e-14:
                a = candidate
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
                improved = True
                break

        if not improved:
            no_improve += 1

        lr *= 0.9997
        if no_improve > 80:
            lr = max(lr * 0.9, 1e-7)
        if no_improve > 250:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            lr = lr_init * 0.7
            no_improve = 0
        if no_improve > 500:
            break

    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit, seed=0):
    """Targeted coordinate descent: move mass from peak contributors to low positions."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]

        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            break

        # Try multiple target positions (not just the minimum)
        target_indices = outside[np.argsort(a[outside])[:5]]
        improved = False

        for j_min in target_indices:
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:
                delta = frac * a[j_max]
                if delta < 1e-12:
                    continue
                candidate = a.copy()
                candidate[j_max] -= delta
                candidate[j_min] += delta
                candidate = np.maximum(candidate, 0.0)
                candidate /= candidate.sum()
                obj = fast_convolve(candidate).max()
                if obj < best_obj - 1e-14:
                    a = candidate
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                scale = 0.001 + 0.003 * min(1.0, no_improve / 500.0)
                candidate = project_simplex(a + rng.standard_normal(n) * scale)
                obj = fast_convolve(candidate).max()
                if obj < best_obj - 1e-14:
                    a = candidate
                    best_obj = obj
                    best_a = a.copy()
                    no_improve = 0
            if no_improve > 800:
                break

    return best_a, best_obj


def symmetric_project(a):
    """Project onto palindromic subspace: a_i = a_{n-1-i}, then normalize."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1].astype(np.float64))
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle for the non-convex auto-convolution problem."""
    from scipy.optimize import linprog
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Solve LP oracle
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n
        g = None
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': min(20, remaining - 1)})
            if res.success:
                g = np.maximum(res.x, 0.0)
        except Exception:
            pass

        if g is None or g.sum() < 1e-15:
            m_exp = min(m_exp * 1.2, 4.0)
            continue

        g = g / g.sum()
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                rng = np.random.default_rng(it * 7919 + 13)
                scale = 0.002 + 0.004 * min(1.0, no_improve / 100.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in np.linspace(0.02, 1.0, 15):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f /= f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 100:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.004 * min(1.0, (no_improve - 100) / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def try_and_update(a, state, label):
    """Evaluate a sequence and update state if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a)))


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid search combining Adam, L-BFGS-B, FW-LP, SCF,
    multi-peak subgradient, and coordinate descent. Explores multiple n values.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    state = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.8f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384

    n_parent = len(parent)

    # === Phase 1: Multi-n exploration (~350s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    # Include parent n and several others
    candidate_ns = sorted(set([n_parent, 1024, 1280, 1536, 2048, 2560, 3072]))
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 3072]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1350
        if remaining < 15:
            break

        # Multiple initializations
        inits = []
        # Upscaled parent
        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss > 1e-10:
                start /= ss
            else:
                start = np.ones(n_try) / n_try
        inits.append(start)

        # Uniform
        inits.append(np.ones(n_try) / n_try)

        # U-shape (edge concentrated)
        x = np.arange(n_try, dtype=np.float64)
        edge_w = 1.0 / (np.minimum(x, n_try - 1 - x) + 1.0) ** 2
        inits.append(edge_w / edge_w.sum())

        # Cosine bump
        g = 1.0 + 0.5 * np.cos(2.0 * np.pi * x / n_try)
        g = np.maximum(g, 0.01)
        inits.append(g / g.sum())

        best_for_n = None
        best_obj_for_n = float('inf')

        for idx, start in enumerate(inits[:3]):  # Limit to 3 inits per n
            t_alloc = min(remaining / 3, 30)
            if t_alloc < 5:
                break
            a_opt, obj = adam_smooth_optimize(start, time_limit=t_alloc,
                                               T_start=0.005, T_end=0.00005,
                                               lr_init=0.04, seed=n_try + idx)
            if obj < best_obj_for_n:
                best_obj_for_n = obj
                best_for_n = a_opt

        if best_for_n is not None:
            try_and_update(best_for_n, state, "adam_n{}".format(n_try))
            # Try symmetric
            a_sym = symmetric_project(best_for_n)
            try_and_update(a_sym, state, "sym_n{}".format(n_try))

    print("  Phase 1 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 2: Deep FW-LP on best (~400s) ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])
        remaining = deadline - time.monotonic() - 850
        if remaining > 30:
            t_alloc = min(remaining, 300)
            print("  FW-LP n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = fw_lp_optimize(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "fw_deep")

            # Perturbed restarts
            for scale in [0.003, 0.01, 0.02]:
                remaining = deadline - time.monotonic() - 500
                if remaining < 15:
                    break
                f_pert = project_simplex(state[1] + rng.standard_normal(n_best) * scale)
                t_alloc = min(remaining, 80)
                a_opt2, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
                try_and_update(a_opt2, state, "fw_pert_{}".format(scale))

    print("  Phase 2 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 3: Adam deep + L-BFGS-B (~300s) ===
    print("\n=== Phase 3: Adam Deep + L-BFGS-B ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Deep Adam with different schedules
        schedules = [
            (0.003, 0.00003, 0.03),
            (0.001, 0.00001, 0.02),
            (0.0005, 0.000005, 0.01),
        ]
        for T_s, T_e, lr in schedules:
            remaining = deadline - time.monotonic() - 200
            if remaining < 15:
                break
            t_alloc = min(remaining, 80)
            a_opt, _ = adam_smooth_optimize(state[1], time_limit=t_alloc,
                                             T_start=T_s, T_end=T_e,
                                             lr_init=lr, seed=777)
            try_and_update(a_opt, state, "adam_deep_T{}".format(T_s))

        # L-BFGS-B
        for T in [0.0005, 0.0002, 0.0001]:
            remaining = deadline - time.monotonic() - 100
            if remaining < 10:
                break
            a_opt = lbfgs_smooth_optimize(state[1], T=T, maxiter=5000)
            try_and_update(a_opt, state, "lbfgs_T{}".format(T))

    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 4: SCF + Multi-peak SG (~200s) ===
    print("\n=== Phase 4: SCF + Multi-peak SG ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Self-consistent field
        remaining = deadline - time.monotonic() - 120
        if remaining > 15:
            t_alloc = min(remaining, 80)
            a_opt, _ = self_consistent_field(state[1], time_limit=t_alloc,
                                             eta_init=0.06, seed=42)
            try_and_update(a_opt, state, "scf")

        # Multi-peak subgradient
        for top_k, lr in [(10, 0.005), (30, 0.003), (50, 0.002), (100, 0.001)]:
            remaining = deadline - time.monotonic() - 60
            if remaining < 10:
                break
            t_alloc = min(remaining, 40)
            a_opt, _ = multi_peak_subgrad(state[1], time_limit=t_alloc,
                                           top_k=top_k, lr_init=lr, seed=99)
            try_and_update(a_opt, state, "mpsg_k{}".format(top_k))

    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 5: Coordinate descent + final polish (~150s) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Coordinate descent
        remaining = deadline - time.monotonic() - 60
        if remaining > 10:
            t_alloc = min(remaining, 70)
            a_opt, _ = coordinate_descent_polish(state[1], time_limit=t_alloc, seed=271)
            try_and_update(a_opt, state, "coord")

        # Fine Adam
        remaining = deadline - time.monotonic() - 30
        if remaining > 5:
            t_alloc = min(remaining, 25)
            a_opt, _ = adam_smooth_optimize(state[1], time_limit=t_alloc,
                                             T_start=0.00005, T_end=0.000001,
                                             lr_init=0.003, seed=555)
            try_and_update(a_opt, state, "fine_adam")

        # L-BFGS-B final
        remaining = deadline - time.monotonic() - 15
        if remaining > 5:
            a_opt = lbfgs_smooth_optimize(state[1], T=0.0001, maxiter=3000)
            try_and_update(a_opt, state, "lbfgs_final")

        # Tiny perturbations with quick polish
        remaining = deadline - time.monotonic() - 5
        if remaining > 3:
            for _ in range(3):
                remaining = deadline - time.monotonic() - 3
                if remaining < 2:
                    break
                noise = rng.standard_normal(len(state[1])) * 0.0002
                f_pert = project_simplex(state[1] + noise)
                a_opt, _ = adam_smooth_optimize(f_pert, time_limit=min(remaining, 3),
                                                 T_start=0.00002, T_end=0.000001,
                                                 lr_init=0.002, seed=888)
                try_and_update(a_opt, state, "perturb_tiny")

    print("  Phase 5 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 6: Symmetric refinement (if time) ===
    print("\n=== Phase 6: Symmetric Refinement ===")
    if state[1] is not None and time.monotonic() < deadline - 10:
        a_sym = symmetric_project(state[1])
        try_and_update(a_sym, state, "final_sym")
        if time.monotonic() < deadline - 5:
            a_opt, _ = adam_smooth_optimize(a_sym, time_limit=min(5, deadline - time.monotonic() - 2),
                                             T_start=0.0001, T_end=0.00001,
                                             lr_init=0.005, seed=111)
            try_and_update(a_opt, state, "sym_adam_final")

    # Fallback
    if state[1] is None:
        state[1] = np.ones(384) / 384
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
