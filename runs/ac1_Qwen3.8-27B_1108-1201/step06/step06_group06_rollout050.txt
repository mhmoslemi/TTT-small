```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
Uses focused softmax (top-K), adaptive temperature cooling, FFT-based

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, p):
    """Compute gradient of smoothed autocorrelation peak.
    grad[m] = 2 * sum_{j=0}^{n-1} a[j] * p[m+j]
    Uses sliding window for efficiency.
    """
    n = len(a)
    window = np.lib.stride_tricks.sliding_window_view(p, n)
    return 2.0 * (window @ a)


def topk_softmax(conv, T, K):
    """Softmax over top-K values of conv for focused gradient."""
    n_conv = len(conv)
    if K >= n_conv:
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        return p / p.sum()
    # Get top K indices
    top_idx = np.argpartition(conv, -K)[-K:]
    # Create masked conv
    masked = np.full(n_conv, -np.inf)
    masked[top_idx] = conv[top_idx]
    scaled = masked / T
    scaled -= np.max(scaled[scaled > -np.inf])
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(n_conv)
        p[top_idx] = 1.0 / K
        return p
    return p / p_sum

        p = topk_softmax(conv, T, K)
        grad = compute_gradient(a, p)

        best_smoothed = np.dot(p, conv)

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    # FFT size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        # Try to make power spectrum flatter
        # target: geometric mean of power
        target = np.exp(np.mean(np.log(power + 1e-30)))
        # Direction: reduce where power > target, increase where power < target
        # In time domain, this means: a_new = IFFT(sqrt(power_adjusted))
        # But we need to maintain non-negativity and sum constraint

        # Perturb in frequency domain
        for strength in [0.001, 0.005, 0.01, 0.02]:
            power_adj = power.copy()
            # Reduce peaks in spectrum
            mask = power_adj > target * 1.5
            power_adj[mask] *= (1 - strength)
            # Boost valleys
            mask2 = power_adj < target * 0.7
            power_adj[mask2] *= (1 + strength)

            # Reconstruct (maintain magnitude, keep phase)
            A_new = A * np.sqrt(power_adj / np.maximum(power, 1e-30))
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-10:
                continue
            a_new = a_new / ss
            c = fast_convolve(a_new)
            if c.max() < best_obj - 1e-15:
                a = a_new
                best_obj = c.max()
                best_a = a.copy()
                break

        # Also try random phase perturbations
        if rng.random() < 0.3:
            phase_noise = rng.standard_normal(len(A)) * 0.01
            A_new = A * np.exp(1j * phase_noise)
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss > 1e-10:
                a_new = a_new / ss
                c = fast_convolve(a_new)
                if c.max() < best_obj - 1e-15:
                    a = a_new
                    best_obj = c.max()
                    best_a = a.copy()

    return best_a, best_obj


def try_seq(a, best_seq, best_score, label=""):
    """Evaluate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return a_norm.copy(), score
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: multi-phase optimization combining focused game-theoretic
    method, FW-LP, spectral refinement, and local polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Using uniform init: score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Quick n exploration ===
    print("\n=== Phase 1: n Exploration ===")
    t_phase = time.monotonic()

    n_candidates = []
    for delta in [-128, -64, -32, 0, 32, 64, 128, 256]:
        n_try = n_parent + delta
        if 128 <= n_try <= 3000:
            n_candidates.append(n_try)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 200:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try

        # Quick game-optimize
        t_alloc = min(remaining * 0.06, 20)
        a_opt, _ = game_optimize_focused(start, time_limit=t_alloc,
                                         T_start=0.001, T_min=0.0001,
                                         K_start=20, K_min=10)
        best_seq, best_score = try_seq(a_opt, best_seq, best_score, f"game_n{n_try}")

    print(f"  Phase 1 best: {best_score:.8f} (t={time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep focused game optimization ===
    print("\n=== Phase 2: Deep Game Optimization ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 100:
        t_alloc = min(remaining * 0.30, 400)
        print(f"  Focused game n={n_best}, T=0.0005, K=25, t={t_alloc:.0f}s")
        a_opt, _ = game_optimize_focused(best_seq, time_limit=t_alloc,
                                         T_start=0.0005, T_min=0.00001,
                                         K_start=25, K_min=5)
        best_seq, best_score = try_seq(a_opt, best_seq, best_score, "deep_game")

    # Second pass with different params
    remaining = deadline - time.monotonic()
    if remaining > 80:
        n_best = len(best_seq)
        t_alloc = min(remaining * 0.20, 200)
        print(f"  Focused game v2 T=0.0003, K=15, t={t_alloc:.0f}s")
        a_opt, _ = game_optimize_focused(best_seq, time_limit=t_alloc,
                                         T_start=0.0003, T_min=0.000005,
                                         K_start=15, K_min=3)
        best_seq, best_score = try_seq(a_opt, best_seq, best_score, "game_v2")

    print(f"  Phase 2 best: {best_score:.8f} (t={time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: FW-LP refinement ===
    print("\n=== Phase 3: FW-LP ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()

    if remaining > 80:
        n_best = len(best_seq)
        t_alloc = min(remaining * 0.25, 250)
        print(f"  FW-LP n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize_v2(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, best_seq, best_score, "fw_lp")

    # Perturbed FW-LP restarts
    for scale in [0.002, 0.005, 0.01]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        n_best = len(best_seq)
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc = min(remaining * 0.10, 80)
        a_opt, _ = fw_lp_optimize_v2(f_pert, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, best_seq, best_score, f"fw_pert{scale}")

    print(f"  Phase 3 best: {best_score:.8f} (t={time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Spectral refinement ===
    print("\n=== Phase 4: Spectral Refinement ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()

    if remaining > 60:
        n_best = len(best_seq)
        t_alloc = min(remaining * 0.15, 120)
        print(f"  Spectral n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = spectral_refine(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, best_seq, best_score, "spectral")

    print(f"  Phase 4 best: {best_score:.8f} (t={time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Local polish ===
    print("\n=== Phase 5: Local Polish ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()

    if remaining > 30:
        n_best = len(best_seq)
        t_alloc = min(remaining * 0.35, 150)
        print(f"  Local polish t={t_alloc:.0f}s")
        a_opt, _ = local_polish(best_seq, time_limit=t_alloc, seed=777)
        best_seq, best_score = try_seq(a_opt, best_seq, best_score, "local1")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        n_best = len(best_seq)
        t_alloc = min(remaining * 0.25, 60)
        a_opt, _ = local_polish(best_seq, time_limit=t_alloc, seed=4242)
        best_seq, best_score = try_seq(a_opt, best_seq, best_score, "local2")

    # Final game pass with very low T
    remaining = deadline - time.monotonic()
    if remaining > 20:
        n_best = len(best_seq)
        t_alloc = min(remaining * 0.3, 50)
        a_opt, _ = game_optimize_focused(best_seq, time_limit=t_alloc,
                                         T_start=0.0001, T_min=0.000005,
                                         K_start=10, K_min=3)
        best_seq, best_score = try_seq(a_opt, best_seq, best_score, "final_game")

    # Tiny perturbations
    remaining = deadline - time.monotonic()
    if remaining > 10:
        n_best = len(best_seq)
        for _ in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 3:
                break
            noise = rng.standard_normal(n_best) * 0.0003
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = game_optimize_focused(f_pert, time_limit=min(remaining * 0.15, 5),
                                             T_start=0.0001, T_min=0.00001,
                                             K_start=10, K_min=5)
            best_seq, best_score = try_seq(a_opt, best_seq, best_score, "micro")

    print(f"  Phase 5 best: {best_score:.8f} (t={time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Try a different n for final push ===
    print("\n=== Phase 6: Alternative n ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()

    if remaining > 40:
        for n_try in [len(best_seq) + 32, len(best_seq) - 32, len(best_seq) + 64]:
            if n_try < 128 or n_try > 3000:
                continue
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, len(best_seq)), best_seq)
            start = np.maximum(start, 0)
            start = start / start.sum()
            t_alloc = min(remaining * 0.3, 30)
            a_opt, _ = game_optimize_focused(start, time_limit=t_alloc,
                                             T_start=0.0005, T_min=0.00005,
                                             K_start=20, K_min=5)
            best_seq, best_score = try_seq(a_opt, best_seq, best_score, f"alt_n{n_try}")

    print(f"  Phase 6 best: {best_score:.8f} (t={time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
```
