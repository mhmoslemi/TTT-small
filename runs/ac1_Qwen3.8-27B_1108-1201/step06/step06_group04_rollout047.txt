import numpy as np
import cvxpy as cp
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_qp_cvxpy(w, n, a_init=None, time_limit=5.0):
    """Solve min a^T H a s.t. sum(a)=1, a>=0 using CVXPY."""
    a = cp.Variable(n, nonneg=True)
    
    # Build Hankel matrix
    H = np.fromfunction(lambda i, j: w[i+j], (n, n))
    H = (H + H.T) / 2  # Ensure symmetry
    
    objective = cp.Minimize(cp.quad_form(a, H, assuming_PSD=True))
    constraints = [cp.sum(a) == 1]
    prob = cp.Problem(objective, constraints)
    
    # Try solvers
    for solver in [cp.GUROBI, cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                val = a.value
                if val is not None:
                    val = np.maximum(val, 0)
                    s = val.sum()
                    if s > 1e-15:
                        val = val / s
                    return val
        except Exception:
            continue
    
    # Fallback: return a_init or uniform
    if a_init is not None:
        return a_init
    return np.ones(n) / n

def primal_dual_search(a0, time_limit, solver='auto'):
    """Main primal-dual optimization loop."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    T = 0.01  # Temperature for softmax
    blend = 0.7  # Initial blend factor
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        cmax = conv.max()
        
        # Compute dual weights
        # Use softmax with adaptive temperature
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()
        
        # Solve QP
        a_qp = solve_qp_cvxpy(w, n, a_init=a)
        
        # Decreasing blend
        progress = min(1.0, elapsed / time_limit)
        alpha = blend * (1.0 - 0.5 * progress)  # Decrease blend over time
        
        a_new = (1.0 - alpha) * a + alpha * a_qp
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a_qp
        else:
            a_new = a_new / s_new
        
        obj = fast_convolve(a_new).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adapt temperature
        if no_improve > 5:
            T *= 1.2
        if no_improve > 20:
            T *= 0.5  # Re-focus
        if no_improve > 50:
            # Restart with perturbation
            rng = np.random.default_rng((it * 104729 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            T = 0.01
            no_improve = 0
        
        if it % 50 == 0:
            print(f"  PD it={it}, obj={best_obj:.10f}, T={T:.6f}, no_imp={no_improve}")
    
    return best_a, best_obj

"""

- Primal-dual iteration with adaptive softmax/top-k dual weights

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(w, a):
    """Compute H*a where H[i,j] = w[i+j]. w has length 2n-1, a has length n."""
    n = len(a)
    # H*a[i] = sum_j w[i+j]*a[j]
    # Use FFT-based approach for speed
    result = np.zeros(n)
    for i in range(n):
        result[i] = np.dot(w[i:i+n], a)
    return result

def solve_qp_active_set(w, n, a_init, max_iters=200):
    """
    Solve min a^T H a s.t. sum(a)=1, a>=0 where H[i,j]=w[i+j] (Hankel, PSD).
    Uses active set method with vectorized numpy operations.
    """
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    active = np.where(a > 1e-12)[0]
    if len(active) == 0:
        active = np.arange(n)

    for it in range(max_iters):
        k = len(active)
        if k == 0:
            return np.ones(n) / n
        if k == 1:
            a_out = np.zeros(n)
            a_out[active[0]] = 1.0
            return a_out

        # Build H_A: H_A[i,j] = w[active[i] + active[j]]
        H_A = w[active[:, None] + active[None, :]]
        H_A = (H_A + H_A.T) / 2.0
        H_A += 1e-12 * np.eye(k)

        # Solve H_A * x = ones, then normalize
        try:
            x = np.linalg.solve(H_A, np.ones(k))
        except np.linalg.LinAlgError:
            try:
                x = np.linalg.lstsq(H_A, np.ones(k), rcond=None)[0]
            except Exception:
                x = np.ones(k) / k

        x_sum = x.sum()
        if abs(x_sum) < 1e-14:
            x = np.ones(k) / k
            x_sum = 1.0
        x = x / x_sum

        # Check non-negativity
        min_val = x.min()
        if min_val < -1e-10:
            idx_rm = np.argmin(x)
            active = np.delete(active, idx_rm)
            # Update a
            a = np.zeros(n)
            mask = np.ones(k, dtype=bool)
            mask[idx_rm] = False
            a[active] = np.maximum(x[mask], 0)
            s2 = a.sum()
            if s2 > 1e-15:
                a = a / s2
            continue

        # Form full solution
        a = np.zeros(n)
        a[active] = x

        # Compute gradient: grad = 2 * H * a
        # (H*a)[i] = sum_j w[i+j]*a[j]
        grad = np.empty(n)
        for i in range(n):
            grad[i] = np.dot(w[i:i + n], a)
        grad *= 2.0

        # Lambda = mean gradient over active set
        lam = np.mean(grad[active])

        # Check KKT for inactive indices
        inactive_mask = np.ones(n, dtype=bool)
        inactive_mask[active] = False
        inactive_idx = np.where(inactive_mask)[0]

        if len(inactive_idx) > 0:
            violations = grad[inactive_idx] - lam
            min_viol = violations.min()
            if min_viol < -1e-10:
                idx_add = inactive_idx[np.argmin(violations)]
                active = np.append(active, idx_add)
                continue

        # Converged - all KKT conditions satisfied
        return a

    return a


def primal_dual_solve(a0, time_limit, T_init=0.01, topk=0, blend_init=0.7):
    """
    Primal-dual iteration:
    - Dual: compute weights from conv(a,a) (softmax or top-k)
    - Primal: solve QP min a^T H_w a s.t. sum(a)=1, a>=0
    - Blend and iterate
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        cmax = conv.max()

        # Compute dual weights
        if topk > 0:
            k_use = min(topk, len(conv))
            peaks = np.argpartition(conv, -k_use)[-k_use:]
            w = np.zeros(len(conv))
            w[peaks] = 1.0 / k_use
        else:
            scaled = conv / T
            scaled -= scaled.max()
            w = np.exp(scaled)
            w = w / w.sum()

        # Solve QP
        a_qp = solve_qp_active_set(w, n, a)

        # Decreasing blend factor
        progress = min(1.0, elapsed / time_limit)
        alpha = blend_init * (1.0 - 0.6 * progress)
        alpha = max(0.05, alpha)

        a_new = (1.0 - alpha) * a + alpha * a_qp
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a_qp
        else:
            a_new = a_new / s_new

        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adapt temperature
        if no_improve == 5:
            T *= 1.3
        elif no_improve == 15:
            T *= 0.6
        elif no_improve == 40:
            rng = np.random.default_rng((it * 104729 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            T = T_init
            no_improve = 0

        if it % 100 == 0:
            print(f"    PD it={it}, obj={best_obj:.10f}, T={T:.6f}, no_imp={no_improve}")

    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate, find the optimal value
    that minimizes the max autocorrelation, holding others fixed.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 0.1:
                break

            # Try different values for a[i]
            old_val = a[i]
            if old_val < 1e-15:
                # Try adding a small amount
                for frac in [0.001, 0.003, 0.01]:
                    candidate = a.copy()
                    # Take from the largest element
                    j_max = np.argmax(candidate)
                    if j_max == i:
                        continue
                    candidate[i] = frac / n
                    candidate[j_max] -= frac / n
                    if candidate[j_max] < 0:
                        continue
                    candidate = project_simplex(candidate)
                    obj = fast_convolve(candidate).max()
                    if obj < best_obj - 1e-15:
                        a = candidate
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                        break
                continue

            # Try reducing a[i] and redistributing
            for frac in [0.02, 0.05, 0.1, 0.2]:
                shift = old_val * frac
                if shift < 1e-15:
                    continue
                candidate = a.copy()
                candidate[i] = old_val - shift
                # Redistribute to random other positions
                others = np.setdiff1d(np.arange(n), [i])
                # Give to positions that don't contribute to the main peak
                conv = fast_convolve(a)
                k_peak = int(np.argmax(conv))
                # Positions contributing to k_peak: i + j = k_peak
                contributing = set()
                for j in range(max(0, k_peak - n + 1), min(n, k_peak + 1)):
                    contributing.add(j)
                    if i != j:
                        pass
                # Just redistribute uniformly for simplicity
                candidate[others] += shift / len(others)
                candidate = project_simplex(candidate)
                obj = fast_convolve(candidate).max()
                if obj < best_obj - 1e-15:
                    a = candidate
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break

            if not improved and old_val > 0:
                # Also try small perturbations
                for pert in [0.001, -0.001, 0.003, -0.003]:
                    candidate = a.copy()
                    candidate[i] += pert
                    candidate = project_simplex(candidate)
                    obj = fast_convolve(candidate).max()
                    if obj < best_obj - 1e-15:
                        a = candidate
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            no_improve += 1
            if no_improve > 5:
                break

        if verbose and sweeps % 10 == 0:
            print(f"    CD sweep={sweeps}, obj={best_obj:.10f}")

    return best_a, best_obj


def nesterov_subgrad(a0, time_limit, lr_init=0.005, momentum_init=0.9):
    """Nesterov-accelerated subgradient for min max_k conv(a,a) on simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = momentum_init

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        y = a + mom * v
        y = project_simplex(y)

        conv_y = fast_convolve(y)
        k_peak = int(np.argmax(conv_y))

        # Gradient of conv_y[k_peak] w.r.t. a
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad = np.zeros(n)
        grad[idx] = 2.0 * y[k_peak - idx]

        v_new = project_simplex(y - lr * grad) - y
        a = y + v_new
        v = v_new
        a = project_simplex(a)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9999
        mom = min(0.995, mom + 0.00003)

        if no_improve > 200:
            lr *= 0.7
            mom *= 0.95
        if no_improve > 500:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = lr_init * 0.5
            mom = momentum_init
            no_improve = 0

    return best_a, best_obj


def smoothed_gradient(a0, time_limit, T_init=0.001, lr_init=0.02):
    """Smoothed gradient descent with softmax surrogate for max."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient: sum_k p[k] * d(conv[k])/d(a)
        # d(conv[k])/d(a[i]) = a[k-i] + a[i+k-(2n-2)] (with boundary handling)
        # More precisely: d(conv[k])/d(a[i]) = sum_j 1[i+j=k] * a[j] + sum_j 1[j+i=k] * a[j]
        # = a[k-i] (if valid) + a[k-i] (if valid) -- actually for linear conv:
        # conv[k] = sum_{i+j=k} a[i]*a[j]
        # d(conv[k])/d(a[m]) = sum_j 1[m+j=k]a[j] + sum_i 1[i+m=k]a[i] = 2*a[k-m] (if 0<=k-m<n)
        # But we need to be careful at boundaries
        # Actually for the "wedge" structure:
        # d(conv[k])/d(a[m]) = a[k-m] if 0 <= k-m < n, else 0 (counting both i=m and j=m terms)
        # Wait: conv[k] = sum_{i=0}^{n-1} sum_{j=0}^{n-1} a[i]*a[j]*1[i+j=k]
        # d(conv[k])/d(a[m]) = sum_j a[j]*1[m+j=k] + sum_i a[i]*1[i+m=k]
        # = a[k-m] (if 0<=k-m<n) + a[k-m] (if 0<=k-m<n) = 2*a[k-m] (if valid)

        # So grad[m] = 2 * sum_k p[k] * a[k-m] (where 0 <= k-m < n)
        # This is: grad = 2 * (p_reversed * a) -- a convolution!
        # Actually: grad[m] = 2 * sum_{k} p[k] * a[k-m] for valid k-m
        # Let l = k-m, so k = l+m, l ranges over max(0,...) to min(2n-2, ...)
        # grad[m] = 2 * sum_l p[l+m] * a[l] for l where 0 <= l < n and 0 <= l+m < 2n-1
        # This is a correlation of p and a!
        # grad = 2 * np.correlate(p, a, mode='same')? Let me verify...
        # Actually np.correlate(p, a, mode='valid') gives sum_l p[l+m]*a[l] for m=0..(len(p)-len(a))
        # p has length 2n-1, a has length n, so valid gives n results
        # np.correlate(p, a, mode='valid')[m] = sum_l p[l+m]*a[l] for l=0..n-1
        # That's exactly what we want!
        grad_half = np.correlate(p, a, mode='valid')
        grad = 2.0 * grad_half

        # Line search
        best_candidate = a
        best_val = fast_convolve(a).max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 20:
            T *= 1.15
            lr *= 0.95
        if no_improve > 60:
            T *= 0.7
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    x = np.arange(n, dtype=np.float64)
    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge / edge.sum())
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g / g.sum())
    # Two peaks
    g2 = (np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.08 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.08 * n)) ** 2))
    inits.append(g2 / g2.sum())
    # Cosine
    c1 = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    c1 = np.maximum(c1, 0.01)
    inits.append(c1 / c1.sum())
    # Raised cosine variant
    rc = 0.5 + 0.5 * np.cos(np.pi * x / (n - 1))
    rc = np.maximum(rc, 0.001)
    inits.append(rc / rc.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Uses primal-dual QP with active set method,
    multi-n exploration, diverse starts, and multiple polishing stages.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(256) / 256

    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")

    # === Phase 1: Quick n-exploration with diverse inits (~300s) ===
    print("\n=== Phase 1: n-exploration ===")
    t1 = time.monotonic()

    candidate_ns = []
    for n_try in [128, 192, 256, 320, 384, 448, 512, 640, 768, 896, 1024, 1280, 1536]:
        if n_try not in candidate_ns:
            candidate_ns.append(n_try)
    if n_parent not in candidate_ns:
        candidate_ns.append(n_parent)
    candidate_ns.sort()

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 30:
            break

        # Quick primal-dual with uniform start
        start = np.ones(n_try) / n_try
        t_alloc = min(remaining, 25)
        a_opt, _ = primal_dual_solve(start, time_limit=t_alloc, T_init=0.01, topk=0)
        try_seq(a_opt, f"pd_uni_n{n_try}")

        # Also try from parent (interpolated)
        if parent is not None and n_try != n_parent:
            start_p = np.interp(np.linspace(0, 1, n_try),
                                np.linspace(0, 1, n_parent), parent)
            start_p = np.maximum(start_p, 0)
            start_p = start_p / start_p.sum()
            remaining = deadline - time.monotonic() - 1180
            if remaining > 20:
                t_alloc2 = min(remaining, 15)
                a_opt2, _ = primal_dual_solve(start_p, time_limit=t_alloc2, T_init=0.01)
                try_seq(a_opt2, f"pd_par_n{n_try}")

    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")

    # === Phase 2: Deep optimization on best n (~600s) ===
    print("\n=== Phase 2: Deep optimization ===")
    t2 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple configurations
        configs = [
            {'T_init': 0.005, 'topk': 0, 'blend_init': 0.7, 't_frac': 0.2},
            {'T_init': 0.01, 'topk': 0, 'blend_init': 0.6, 't_frac': 0.15},
            {'T_init': 0.02, 'topk': 10, 'blend_init': 0.7, 't_frac': 0.15},
            {'T_init': 0.003, 'topk': 5, 'blend_init': 0.5, 't_frac': 0.12},
            {'T_init': 0.01, 'topk': 20, 'blend_init': 0.65, 't_frac': 0.12},
        ]

        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 400
            if remaining < 20:
                break
            t_alloc = min(remaining, int(budget_s * cfg['t_frac']))
            print(f"  PD config {ci} (T={cfg['T_init']}, topk={cfg['topk']}), t={t_alloc:.0f}s")
            a_opt, _ = primal_dual_solve(best_seq, time_limit=t_alloc,
                                         T_init=cfg['T_init'], topk=cfg['topk'],
                                         blend_init=cfg['blend_init'])
            try_seq(a_opt, f"pd_deep_{ci}")

    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")

    # === Phase 3: Nesterov + Smoothed gradient (~300s) ===
    print("\n=== Phase 3: Nesterov & Smoothed Gradient ===")
    t3 = time.monotonic()

    if best_seq is not None:
        for lr_i in [0.005, 0.002, 0.01]:
            remaining = deadline - time.monotonic() - 200
            if remaining < 15:
                break
            t_alloc = min(remaining, 80)
            print(f"  Nesterov lr={lr_i}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=t_alloc, lr_init=lr_i)
            try_seq(a_opt, f"nest_lr{lr_i}")

        remaining = deadline - time.monotonic() - 150
        if remaining > 15:
            t_alloc = min(remaining, 80)
            print(f"  Smoothed grad T=0.0005, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_gradient(best_seq, time_limit=t_alloc, T_init=0.0005, lr_init=0.015)
            try_seq(a_opt, "smooth_0.0005")

        remaining = deadline - time.monotonic() - 120
        if remaining > 15:
            t_alloc = min(remaining, 60)
            a_opt, _ = smoothed_gradient(best_seq, time_limit=t_alloc, T_init=0.002, lr_init=0.01)
            try_seq(a_opt, "smooth_0.002")

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")

    # === Phase 4: Multi-start with diverse inits on best n (~250s) ===
    print("\n=== Phase 4: Multi-start ===")
    t4 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        inits = generate_inits(n_best, best_seq, rng)

        for ii, init in enumerate(inits):
            remaining = deadline - time.monotonic() - 150
            if remaining < 15:
                break
            t_alloc = min(remaining, 30)
            print(f"  Multi-start {ii}, t={t_alloc:.0f}s")
            a_opt, _ = primal_dual_solve(init, time_limit=t_alloc, T_init=0.01)
            try_seq(a_opt, f"multi_{ii}")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # === Phase 5: Final polish (~150s) ===
    print("\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()

    if best_seq is not None:
        # Coordinate descent
        remaining = deadline - time.monotonic() - 50
        if remaining > 10:
            t_alloc = min(remaining, 60)
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_desc")

        # Final Nesterov with very small lr
        remaining = deadline - time.monotonic() - 30
        if remaining > 10:
            t_alloc = min(remaining, 40)
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=t_alloc, lr_init=0.001, momentum_init=0.95)
            try_seq(a_opt, "final_nest")

        # Tiny perturbation + quick refine
        for _ in range(3):
            remaining = deadline - time.monotonic() - 10
            if remaining < 5:
                break
            noise = 0.0005 + 0.001 * _
            f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * noise)
            t_alloc = min(remaining, 15)
            a_opt, _ = smoothed_gradient(f_pert, time_limit=t_alloc, T_init=0.0003, lr_init=0.005)
            try_seq(a_opt, f"perturb_{_}")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
