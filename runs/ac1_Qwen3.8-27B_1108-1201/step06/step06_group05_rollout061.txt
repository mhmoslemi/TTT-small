"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated gradient descent on sum-of-squares surrogate (fast, smooth)
2. Subgradient max-minimization with multi-peak weighting
3. LP-based Frank-Wolfe for global improvements
4. Active-set KKT refinement with Newton iterations
5. Multi-n exploration and perturbation restarts
6. Final polishing with very fine local search

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_sos(a):
    """Compute gradient of sum(conv(a,a)^2) w.r.t. a. Returns (grad, conv)."""
    n = len(a)
    conv = fast_convolve(a)
    # grad[m] = 4 * sum_{l=0}^{n-1} conv[m+l] * a[l]
    # This is 4 * cross-correlation of conv with a (valid mode)
    h = np.correlate(conv, a, mode='valid')
    grad = 4.0 * h
    return grad, conv


def nesterov_sos(a0, time_limit, lr_init=0.01, max_iter=50000):
    """
    Nesterov-accelerated projected gradient descent on sum(conv^2).
    Minimize f(a) = sum_k (conv(a,a)[k])^2 s.t. a >= 0, sum(a) = 1.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    y = a.copy()  # Nesterov momentum variable
    v = np.zeros(n)  # previous y
    lr = lr_init
    best_a = a.copy()
    best_f = compute_grad_sos(a)[1]
    best_val = np.sum(best_f ** 2)
    best_max = best_f.max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        grad, conv = compute_grad_sos(y)

        # Nesterov step
        momentum = (1 + np.sqrt(1 + 4 * it * it)) / (1 + np.sqrt(1 + 4 * (it - 1) * (it - 1))) if it > 1 else 1.0
        momentum = min(momentum, 1.0)  # cap for stability
        # Actually use standard Nesterov: y_{k+1} = (k/(k+2)) * y_k + ((2k+1)/(k+2)) * x_{k+1}
        # Simpler: use heavy-ball style
        v = y.copy()
        y = project_simplex(y - lr * grad)

        # Evaluate
        conv_new = fast_convolve(y)
        f_val = np.sum(conv_new ** 2)
        max_val = conv_new.max()

        if f_val < best_val - 1e-16:
            best_val = f_val
            best_max = max_val
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt learning rate
        if no_improve > 20:
            lr *= 0.9
        if no_improve > 100:
            lr *= 0.8
            no_improve = 0  # reset to allow recovery

        # Periodic restart with perturbation

            rng = np.random.default_rng((it * 48271 + 17) % (2**31 - 1))

            no_improve = 0

            print(f"    Nesterov-SOS it={it}, sumsq={best_val:.10f}, max={best_max:.10f}, lr={lr:.6f}")

    return best_a, best_max


def subgrad_multi_peak(a, top_k=20):
    """Compute weighted subgradient from top-k peaks of the auto-convolution."""
    n = len(a)
    conv = fast_convolve(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    order = np.argsort(pv)[::-1]
    weights = np.zeros(len(peaks))
    for rank, idx in enumerate(order):
        weights[idx] = 1.0 / (rank + 1) ** 1.5
    ws = weights.sum()
    if ws > 0:
        weights /= ws
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx_arr = np.arange(js, je + 1)
        sg[idx_arr] += weights[i] * a[k - idx_arr]
    return sg, conv

def solve_lp_oracle(f, M, max_time=20):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'max_time': max_time})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def active_set_refine(a0, time_limit, tol=1e-13, max_iter=5000):
    """
    Active-set KKT refinement.
    At optimum: for support S and active peaks P:
      sum_{k in P} mu_k * a[k-j] = lambda for all j in S
      (a*a)[k] = M for all k in P
      sum(a) = 1, a >= 0
    Uses iterative linearization.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1
        if it % 200 == 0:
            print(f"    AS-refine it={it}, obj={best_obj:.12f}")

        conv = fast_convolve(a)
        M = conv.max()

        # Find active peaks (within tight tolerance of max)
        threshold = M * (1.0 - 0.0005)
        active_peaks = np.where(conv >= threshold)[0]

        if len(active_peaks) <= 1:
            # Use top few peaks
            top_k = min(10, len(conv))
            active_peaks = np.argpartition(conv, -top_k)[-top_k:]
            active_peaks = np.where(conv[active_peaks] >= M * 0.999)[0]
            if len(active_peaks) == 0:
                active_peaks = np.array([np.argmax(conv)])

        support = np.where(a > 1e-12)[0]
        if len(support) == 0:
            break

        # Solve KKT: find mu >= 0 such that sum_k mu_k * a[k-j] is constant on support
        # A_sys[i, k_idx] = a[active_peaks[k_idx] - support[i]] (if valid)
        m_s = len(support)
        p = len(active_peaks)

        A_sys = np.zeros((m_s, p))
        for i, j in enumerate(support):
            for k_idx, k in enumerate(active_peaks):
                idx = k - j
                if 0 <= idx < n:
                    A_sys[i, k_idx] = a[idx]

        # We want A_sys @ mu = lambda * 1 (constant on support)
        # Eliminate lambda: (A_sys[:, :-1] - A_sys[:, -1:]) @ mu = 0 (if p > 1)
        # Or use: minimize ||A_sys @ mu - mean(A_sys @ mu)||^2 subject to mu >= 0, sum(mu) = 1
        # Simpler: use least squares with constraint sum(mu) = 1

        if p == 1:
            mu_vals = np.array([1.0])
        else:
            # Solve: find mu that makes A_sys @ mu as constant as possible
            # Equivalent to: minimize ||(I - (1/m)11^T) A_sys mu||^2 s.t. mu >= 0, sum(mu)=1
            # Use projection: center A_sys, then find mu in direction of smallest residual
            centered = A_sys - A_sys.mean(axis=0, keepdims=True)
            # Want centered @ mu = 0
            # Use SVD to find null space direction
            try:
                _, S_vals, Vt = np.linalg.svd(centered, full_matrices=True)
                # Smallest singular vector
                sol = Vt[-1, :]
                mu_vals = np.maximum(sol, 0)
                mu_sum = mu_vals.sum()
                if mu_sum < 1e-15:
                    mu_vals = np.ones(p) / p
                else:
                    mu_vals /= mu_sum
            except np.linalg.LinAlgError:
                mu_vals = np.ones(p) / p

        # Compute gradient direction using mu weights
        grad = np.zeros(n)
        for k_idx, k in enumerate(active_peaks):
            if mu_vals[k_idx] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] -= mu_vals[k_idx] * a[k - idx_arr]

        # Also add uniform component to maintain sum normalization
        # Step
        best_candidate = a
        best_val = M
        for step in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            candidate = project_simplex(a + step * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()

        # Convergence check
        if abs(M - best_val) < 1e-15 and it > 10:
            # Check if we can still improve
            if best_obj < M * (1.0 + 1e-12):
                break

    return best_a, best_obj

def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s
