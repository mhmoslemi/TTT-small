Actually, let me just fix sum(a) = 1 and use the bound a >= 0. I can handle the equality constraint by projecting back to the simplex after each step (like in Frank-Wolfe), or by using the softmax parameterization.

Let me go with the softmax parameterization:
- a = softmax(z) = exp(z) / sum(exp(z))
- This automatically ensures a > 0 and sum(a) = 1
- Optimize over z using L-BFGS or Adam
- The objective becomes 2n * max(conv(softmax(z), softmax(z)))

This is a clean formulation. Let me implement it.

For the gradient:
- d a[i] / d z[j] = a[i] * (delta[i,j] - a[j])
- d objective / d z[j] = sum_i (d objective / d a[i]) * (d a[i] / d z[j])

This is standard backprop through softmax.

Let me code this up:

```python
def softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def lse_objective(z, beta):
    a = softmax(z)
    conv = fast_convolve(a)
    # LSE smoothing
    bc = beta * conv
    bc_max = bc.max()
    lse_val = (bc_max + np.log(np.exp(bc - bc_max).sum())) / beta
    return lse_val

def lse_gradient(z, beta):
    a = softmax(z)
    conv = fast_convolve(a)
    bc = beta * conv
    bc_max = bc.max()
    w = np.exp(bc - bc_max)
    w /= w.sum()
    # grad of LSE wrt a
    grad_a = np.zeros(len(a))
    for k in range(len(conv)):
        if w[k] < 1e-15:
            continue
        lo = max(0, k - len(a) + 1)
        hi = min(len(a) - 1, k)
        grad_a[lo:hi+1] += w[k] * 2.0 * a[k - np.arange(lo, hi+1)]
    # Chain rule through softmax
    n = len(a)
    grad_z = np.zeros(n)
    for j in range(n):
        grad_z[j] = sum_i grad_a[i] * a[i] * (1 if i==j else -1) * a[j]
        # = a[j] * (grad_a[j] - sum_i grad_a[i] * a[i])
    grad_z = a * (grad_a - np.dot(grad_a, a))
    return grad_z
```

Wait, let me redo the chain rule:
- a[i] = exp(z[i]) / sum_k exp(z[k]) = softmax(z)[i]
- d a[i] / d z[j] = a[i] * (delta_{ij} - a[j])

So:
- d L / d z[j] = sum_i (dL/da[i]) * (d a[i] / d z[j])
- = sum_i grad_a[i] * a[i] * (delta_{ij} - a[j])
- = a[j] * grad_a[j] - a[j] * sum_i grad_a[i] * a[i]
- = a[j] * (grad_a[j] - <grad_a, a>)

Yes, that's what I had. Good.

Now, for L-BFGS, I need to provide the objective and gradient. Let me use scipy.optimize.minimize with method='L-BFGS-B'.

But wait, with the softmax parameterization, there are no bound constraints on z (z can be any real number). So I can use 'L-BFGS' (without the -B) or 'BFGS'.

Actually, L-BFGS-B is fine even without bounds (it just doesn't use the bound feature).

Let me also think about the computational cost:
- n = 768, so each gradient evaluation requires one FFT convolution (O(n log n)) and some O(n) work
- L-BFGS typically needs ~100-1000 iterations for moderate-dimensional problems
- With n = 768, this should be fast enough

Let me also consider: should I optimize over z (unconstrained) or over a (with simplex constraint)?
- z-parameterization: unconstrained, smooth, but the softmax can be "short" (some a[i] very close to 0)
- a-parameterization with projection: handles the constraint directly, but projection can be expensive

I'll go with the z-parameterization for L-BFGS, and also try a direct approach with simplex projection for comparison.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS on softmax-parameterized LSE-smoothed objective
2. Adaptive multi-scale beta annealing
3. Peak shaving via coordinate descent
4. Multi-length exploration
5. Frequency-domain perturbations for diversity
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad_a(a, weights):
    """Compute gradient of weighted sum of conv[k] wrt a."""
    n = len(a)
    grad = np.zeros(n)
    for k, w in enumerate(weights):
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi+1] += w * 2.0 * a[k - np.arange(lo, hi+1)]
    return grad

def lse_weights(conv, beta):
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_bc / total

def objective_and_grad_z(z, beta):
    """LSE-smoothed objective and gradient wrt z (softmax parameterization)."""
    a = softmax(z)
    conv = fast_convolve(a)
    w = lse_weights(conv, beta)
    # LSE value
    bc = beta * conv
    lse_val = (bc.max() + np.log(np.exp(bc - bc.max()).sum())) / beta
    # Gradient wrt a
    grad_a = compute_grad_a(a, w)
    # Chain rule through softmax
    grad_z = a * (grad_a - np.dot(grad_a, a))
    return lse_val, grad_z

def adam_lse_optimize(a0, time_limit, beta_start=20.0, beta_end=10000.0, 
                       lr=0.01, verbose=False):
    """Adam optimization on the simplex using LSE smoothing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta = beta_start
    beta_growth = (beta_end / beta_start) ** (1.0 / max(1, int(time_limit * 10)))  # roughly
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        a_curr = a.copy()
        conv = fast_convolve(a_curr)
        w = lse_weights(conv, beta)
        grad_a = compute_grad_a(a_curr, w)
        
        # Project gradient onto simplex tangent space
        grad_proj = grad_a - np.mean(grad_a)  # or: grad_a - np.dot(grad_a, a) * something
        # Actually for simplex, the projection of gradient is: grad - mean(grad) is not quite right
        # For the constraint sum(a)=1, a>=0, the projected gradient is:
        # If a[i] > 0: grad[i] - lambda where lambda = sum_{j: a[j]>0} grad[j] / count
        # If a[i] = 0: max(0, grad[i] - lambda)
        # For simplicity, use: grad_proj = grad_a - np.dot(grad_a, a_curr)
        # (this is the Riemannian gradient for the simplex with Euclidean metric)
        
        # Adam update
        lr_t = lr * np.sqrt(1 - 0.999**t) / (1 - 0.9**t)
        m = 0.9 * m + 0.1 * grad_a
        v = 0.99 * v + 0.01 * grad_a**2
        m_hat = m / (1 - 0.9**t)
        v_hat = v / (0.999**t)
        step = lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        a_new = a_curr - step
        a_new = project_simplex(a_new - a_new.min())  # hmm, this isn't quite right
        
        # Actually, let's just do: a_new = project_simplex(a_curr - step)
        # But project_simplex expects values that should sum to 1
        # Let me just clip and renormalize
        a_new = np.maximum(a_curr - step, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a_curr.copy()
        else:
            a_new = a_new / sn
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # If stuck, try larger step or reset
            if no_improve > 50:
                a = best_a.copy()
                lr = max(lr * 0.5, 0.0001)
                no_improve = 0
        
        a = a_new
        
        # Adaptive beta
        beta = min(beta * (1 + 1.0/max(10, time_limit-t0) * 2), beta_end)
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    t={t}, beta={beta:.1f}, obj={obj_new:.8f}, best={best_obj:.8f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj

def lbfgs_lse_optimize(a0, time_limit, beta_start=50.0, beta_end=5000.0, verbose=False):
    """L-BFGS optimization using softmax parameterization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    # Initialize z such that softmax(z) = a
    z = np.log(np.maximum(a, 1e-300))
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    beta = beta_start
    total_time = time_limit
    n_phases = 5
    time_per_phase = total_time / n_phases
    
    t0 = time.monotonic()
    
    for phase in range(n_phases):
        phase_timeout = time_per_phase
        current_beta = beta_start * (beta_end / beta_start) ** (phase / n_phases)
        
        def func_grad(z):
            val, grad = objective_and_grad_z(z, current_beta)
            # Convert to actual objective: 2n * LSE / (sum a)^2 = 2n * LSE (since sum=1)
            # Actually we just minimize LSE which approximates max(conv)
            return val, grad
        
        t_start = time.monotonic()
        try:
            # Use a wrapper that checks time
            result = minimize(func_grad, z, method='L-BFGS-B',
                            options={'maxiter': 200, 'maxfun': 500,
                                    'ftol': 1e-12, 'gtol': 1e-8,
                                    'maxls': 50})
        except Exception:
            break
        
        # Evaluate result
        a_new = softmax(result.x)
        conv = fast_convolve(a_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    Phase {phase+1}/{n_phases}: obj={obj:.8f}, best={best_obj:.8f}, elapsed={elapsed:.1f}s")
        
        if time.monotonic() - t0 > total_time - 10:
            break
    
    return best_a, best_obj

def peak_shaving(a0, time_limit, n_peaks=10, verbose=False):
    """Coordinate descent to shave peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    iterations = 0
    
    while time.monotonic() - t0 < time_limit:
        iterations += 1
        conv = fast_convolve(a)
        
        # Find the peak
        k_peak = int(np.argmax(conv))
        
        # Find the coordinate that most contributes to this peak
        lo = max(0, k_peak - n + 1)
        hi = min(n - 1, k_peak)
        contribs = a[lo:hi+1] * a[k_peak - np.arange(lo, hi+1)]
        j_best = lo + int(np.argmax(contribs))
        
        # Try to reduce a[j_best] by moving mass to other coordinates
        delta = a[j_best] * 0.05  # move 5% of mass
        
        # Find good destination: a coordinate that doesn't contribute to the peak
        # and is "far" from the peak in the convolution sense
        best_dest = -1
        best_new_obj = best_obj
        
        # Try several random destinations
        rng = np.random.default_rng(iterations * 7 + 3)
        candidates = rng.choice(n, size=min(20, n), replace=False)
        
        for j_dest in candidates:
            if j_dest == j_best or ((j_dest >= lo and j_dest <= hi) and (k_peak - j_dest >= 0 and k_peak - j_dest < n and k_peak - j_dest >= lo and k_peak - j_dest <= hi)):
                # j_dest would also contribute to the peak, skip
                # Actually, let me just try all and check
                pass
            
            a_trial = a.copy()
            a_trial[j_best] -= delta
            a_trial[j_dest] += delta
            if a_trial[j_best] < 0:
                continue
            a_trial = np.maximum(a_trial, 0)
            sn = a_trial.sum()
            a_trial /= sn
            conv_trial = fast_convolve(a_trial)
            obj_trial = conv_trial.max()
            
            if obj_trial < best_new_obj:
                best_new_obj = obj_trial
                best_dest = j_dest
                best_trial = a_trial.copy()
        
        if best_dest >= 0 and best_new_obj < best_obj - 1e-15:
            a = best_trial.copy()
            best_obj = best_new_obj
            best_a = a.copy()
        else:
            # No improvement, try smaller delta
            delta = a[j_best] * 0.01
            found = False
            for j_dest in candidates:
                if j_dest == j_best:
                    continue
                a_trial = a.copy()
                a_trial[j_best] -= delta
                a_trial[j_dest] += delta
                if a_trial[j_best] < 0:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum()
                conv_trial = fast_convolve(a_trial)
                obj_trial = conv_trial.max()
                if obj_trial < best_obj - 1e-15:
                    a = a_trial.copy()
                    best_obj = obj_trial
                    best_a = a.copy()
                    found = True
                    break
            
            if not found:
                # Try a different peak
                top_peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
                for k_try in top_peaks:
                    k_try = int(k_try)
                    lo2 = max(0, k_try - n + 1)
                    hi2 = min(n - 1, k_try)
                    contribs2 = a[lo2:hi2+1] * a[k_try - np.arange(lo2, hi2+1)]
                    j_try = lo2 + int(np.argmax(contribs2))
                    delta2 = a[j_try] * 0.02
                    for j_dest in candidates:
                        if j_dest == j_try:
                            continue
                        a_trial = a.copy()
                        a_trial[j_try] -= delta2
                        a_trial[j_dest] += delta2
                        if a_trial[j_try] < 0:
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        a_trial /= a_trial.sum()
                        conv_trial = fast_convolve(a_trial)
                        if conv_trial.max() < best_obj - 1e-15:
                            a = a_trial.copy()
                            best_obj = conv_trial.max()
                            best_a = a.copy()
                            break
                    if best_obj < conv.max() - 1e-10:
                        break
    
    return best_a, best_obj

def freq_perturb(a, strength, rng):
    """Perturb in frequency domain."""
    n = len(a)
    F = np.fft.rfft(a)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(F_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new

def interpolate_to_length(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer with adaptive LSE smoothing on the simplex
2. L-BFGS with softmax parameterization for complementary optimization
3. Greedy peak shaving via coordinate redistribution
4. Multi-length exploration with interpolation and frequency perturbation
5. Alternating refinement with restart strategy
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_weights(conv, beta):
    """Compute softmax weights on convolution values (LSE dual)."""
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_bc / total


def compute_grad_a(a, weights):
    """Compute gradient of sum_k w[k]*conv[k] wrt a."""
    n = len(a)
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += w * 2.0 * a[k - idx]
    return grad


def objective_and_grad_z(z, beta):
    """LSE-smoothed objective and gradient wrt z (softmax parameterization)."""
    a = softmax_fn(z)
    conv = fast_convolve(a)
    w = lse_weights(conv, beta)
    bc = beta * conv
    bc_max = bc.max()
    lse_val = (bc_max + np.log(np.exp(bc - bc_max).sum())) / beta
    grad_a = compute_grad_a(a, w)
    grad_z = a * (grad_a - np.dot(grad_a, a))
    return lse_val, grad_z


def adam_lse_optimize(a0, time_limit, beta_start=30.0, beta_end=20000.0,
                      lr=0.005, verbose=False):
    """Adam optimization on the simplex with LSE smoothing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta = beta_start
    beta_increase = (beta_end / beta_start)
    total_steps_est = max(100, int(time_limit * 5))
    beta_rate = beta_increase ** (1.0 / total_steps_est)

    t0 = time.monotonic()
    no_improve = 0
    stuck_count = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 0.5:
            break

        conv = fast_convolve(a)
        w = lse_weights(conv, beta)
        grad_a = compute_grad_a(a, w)

        # Adam update
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * grad_a
        v = beta2 * v + (1 - beta2) * grad_a ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr_t = lr * np.sqrt(1 - beta2 ** t) / (1 - beta1 ** t)
        step = lr_t * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            stuck_count = 0
        else:
            no_improve += 1

        a = a_new

        # Adaptive beta increase
        beta = min(beta * beta_rate, beta_end)

        # Stuck detection and recovery
        if no_improve > 80:
            stuck_count += 1
            if stuck_count > 5:
                # Restart from best with perturbation
                a = best_a.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                no_improve = 0
                stuck_count = 0
                lr = max(lr * 0.7, 0.0001)
                # Small perturbation
                rng = np.random.default_rng(t + int(time.monotonic() * 1000) % 100000)
                a = project_simplex(a + rng.standard_normal(n) * 0.002)
                beta = max(beta_start, beta * 0.5)
            elif stuck_count == 1:
                lr = max(lr * 0.5, 0.0001)
            elif stuck_count == 3:
                beta = min(beta * 3.0, beta_end)

        if verbose and t % 200 == 0:
            print(f"    Adam t={t}, beta={beta:.1f}, obj={obj_new:.8f}, best={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a, best_obj


def lbfgs_lse_optimize(a0, time_limit, beta_start=50.0, beta_end=5000.0, verbose=False):
    """L-BFGS optimization using softmax parameterization with phased beta."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    z = np.log(np.maximum(a, 1e-300))

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    n_phases = 6
    time_per_phase = max(5.0, time_limit / n_phases)
    t0 = time.monotonic()

    for phase in range(n_phases):
        if time.monotonic() - t0 > time_limit - 10:
            break
        current_beta = beta_start * (beta_end / beta_start) ** (phase / max(1, n_phases - 1))

        def func_grad(z_in):
            val, grad = objective_and_grad_z(z_in, current_beta)
            return val, grad

        t_start = time.monotonic()
        try:
            result = minimize(func_grad, z, method='L-BFGS-B',
                              options={'maxiter': 300, 'maxfun': 1000,
                                       'ftol': 1e-14, 'gtol': 1e-10,
                                       'maxls': 100})
        except Exception:
            break

        a_new = softmax_fn(result.x)
        conv = fast_convolve(a_new)
        obj = conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            z = result.x.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    L-BFGS phase {phase+1}/{n_phases}: obj={obj:.8f}, best={best_obj:.8f}, t={elapsed:.1f}s")
        else:
            # Didn't improve, keep best z
            z = np.log(np.maximum(best_a, 1e-300))

    return best_a, best_obj


def peak_shaving(a0, time_limit, n_peaks=10, verbose=False):
    """Greedy peak shaving via coordinate redistribution."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    iterations = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iterations += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        conv = fast_convolve(a)
        top_peaks = np.argpartition(conv, -min(n_peaks, len(conv)))[-min(n_peaks, len(conv)):][::-1]

        improved_this_iter = False
        for k_peak in top_peaks:
            k_peak = int(k_peak)
            lo = max(0, k_peak - n + 1)
            hi = min(n - 1, k_peak)
            if lo > hi:
                continue

            idx = np.arange(lo, hi + 1)
            contribs = a[idx] * a[k_peak - idx]
            j_best = idx[np.argmax(contribs)]

            # Try reducing a[j_best] and redistributing
            for frac in [0.1, 0.05, 0.02]:
                delta = a[j_best] * frac
                if delta < 1e-15:
                    continue

                rng = np.random.default_rng(iterations * 31 + k_peak + int(frac * 1000))
                # Prefer destinations far from the peak's support
                peak_support = set(range(lo, hi + 1))
                non_support = np.array([j for j in range(n) if j not in peak_support])
                if len(non_support) == 0:
                    non_support = np.arange(n)

                n_try = min(15, len(non_support))
                candidates = rng.choice(non_support, size=n_try, replace=False)

                for j_dest in candidates:
                    j_dest = int(j_dest)
                    a_trial = a.copy()
                    a_trial[j_best] -= delta
                    a_trial[j_dest] += delta
                    if a_trial[j_best] < 0:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    sn = a_trial.sum()
                    a_trial /= sn
                    conv_trial = fast_convolve(a_trial)
                    obj_trial = conv_trial.max()
                    if obj_trial < best_obj - 1e-14:
                        best_obj = obj_trial
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        improved_this_iter = True
                        no_improve = 0
                        break
                if improved_this_iter:
                    break

        if not improved_this_iter:
            no_improve += 1
            if no_improve > 200:
                # Reset to best and try different strategy
                a = best_a.copy()
                no_improve = 0

        if verbose and iterations % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave it={iterations}, obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a, best_obj


def freq_perturb(a, strength, rng):
    """Perturb in frequency domain for diversity."""
    n = len(a)
    F = np.fft.rfft(a)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(F_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new


def interpolate_to_length(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new

                                              beta_start=500.0, beta_end=30000.0)
                try_sequence(f_opt, best_tracker)

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)

    # Very high beta Adam (essentially optimizing the max directly)
    t_rem = deadline - time.monotonic()
    if t_rem > 40:
        t_alloc = min(80, t_rem - 30)
        print(f"  High-beta Adam, beta: 5000->100000, t={t_alloc:.0f}s")
        f_opt, _ = adam_lse_optimize(current_best, time_limit=t_alloc,
                                      beta_start=5000.0, beta_end=100000.0, lr=0.001)
        try_sequence(f_opt, best_tracker)

    # Final peak shaving
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t_rem = deadline - time.monotonic()
    if t_rem > 25:
        t_alloc = min(50, t_rem - 15)
        print(f"  Final PeakShave, t={t_alloc:.0f}s")
        f_opt, _ = peak_shaving(current_best, time_limit=t_alloc, n_peaks=5)
        try_sequence(f_opt, best_tracker)

    # Try small perturbations of the best
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t_rem = deadline - time.monotonic()
    if t_rem > 15:
        print(f"  Final perturbations, t={t_rem:.0f}s")
        for i in range(20):
            if time.monotonic() > deadline - 5:
                break
            f_pert = freq_perturb(current_best, strength=0.003 + 0.002 * i, rng=rng)
            try_sequence(f_pert, best_tracker)
            f_pert2 = project_simplex(current_best + rng.standard_normal(n_cur) * 0.001)
            try_sequence(f_pert2, best_tracker)

    # Final verification
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.12f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using parent fallback")
    final_seq = np.maximum(parent, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    return [float(x) for x in final_seq]
```
