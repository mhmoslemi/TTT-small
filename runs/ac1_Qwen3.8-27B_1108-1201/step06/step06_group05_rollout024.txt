        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    return e / e.sum()


def smooth_obj_and_grad(v, epsilon):
    """
    Compute smoothed objective and gradient in log-space (softmax parameterization).
    v: raw parameter vector
    Returns (objective, gradient w.r.t. v)
    """
    a = softmax_fn(v)
    n = len(a)
    conv = fast_convolve(a)
    c = conv / epsilon
    c_shift = c - c.max()
    exp_c = np.exp(c_shift)
    exp_sum = exp_c.sum()
    obj = epsilon * (c.max() + np.log(exp_sum))
    p = exp_c / exp_sum

    # Gradient w.r.t. a: grad_a[m] = 2 * sum_k p[k] * a[k-m]
    # Using cross-correlation via FFT
    N = 1 << (3 * n - 2).bit_length()
    P = rfft(p, N)
    A = rfft(a, N)
    r = irfft(np.conj(P) * A, N).real
    grad_a = 2.0 * r[:n]

    # Chain rule for softmax: grad_v = (I - a*a^T) grad_a = grad_a - a*(grad_a . a)
    dot = np.dot(grad_a, a)
    grad_v = grad_a - a * dot
    return obj, grad_v


def lbfgs_smooth(v0, epsilon, time_limit, max_iter=500):
    """Run L-BFGS on the smoothed objective with softmax parameterization."""
    from scipy.optimize import minimize

    t0 = time.monotonic()
    best_v = v0.copy()
    best_obj, _ = smooth_obj_and_grad(v0, epsilon)

    it_count = [0]

    def fun(v):
        it_count[0] += 1
        if time.monotonic() - t0 > time_limit:
            return best_obj, np.zeros(len(v))
        return smooth_obj_and_grad(v, epsilon)

    try:
        res = minimize(fun, v0, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'maxfun': 5000,
                                'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 50})
        if res.fun < best_obj:
            best_obj = res.fun
            best_v = res.x.copy()
    except Exception:
        pass

    # Also try with the final x even if not strictly better in smooth obj
    if hasattr(res, 'x') and res.x is not None:
        final_obj, _ = smooth_obj_and_grad(res.x, epsilon)
        if final_obj < best_obj:
            best_obj = final_obj
            best_v = res.x.copy()

    return best_v, best_obj


def anneal_lbfgs(a0, time_limit, epsilon_schedule=None):
    """Multi-scale L-BFGS with annealed epsilon."""
    n = len(a0)
    a = a0.copy()
    a = a / a.sum()
    # Convert to log-space
    a_safe = np.maximum(a, 1e-15)
    v = np.log(a_safe / a_safe.sum())

    if epsilon_schedule is None:
        epsilon_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]

    t0 = time.monotonic()
    for eps in epsilon_schedule:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        t_alloc = min(remaining, max(10, 600.0 / len(epsilon_schedule)))
        v, _ = lbfgs_smooth(v, eps, time_limit=t_alloc,
                            max_iter=max(50, int(t_alloc * 5)))
        # Update a
        a = softmax_fn(v)
        score_obj = fast_convolve(a).max()
        print(f"    eps={eps:.6f}, obj={score_obj:.10f}, elapsed={time.monotonic()-t0:.0f}s")

    return a


def solve_lp_oracle(f, M, max_time=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: A_ub has shape (2n-1, n)
    # A_ub[k, j] = f[j] if 0 <= k-j < n, else 0
    # Actually: (f*g)_k = sum_j f[j] * g[k-j] (with valid indices)
    # So A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # (f*g)_k = sum_j f[j] g[k-j], so coefficient of g[m] in (f*g)_k is f[k-m]
        # A_ub[k, m] = f[k-m] if 0 <= k-m < n
        lo = max(0, -n + 1 + k)  # k - m >= 0 => m <= k
        hi = min(n, k + n)       # k - m < n => m > k - n
        # Actually let me redo: for row k, column m: A_ub[k,m] = f[k-m] if 0<=k-m<n
        # k-m >= 0 => m <= k
        # k-m < n => m > k-n => m >= k-n+1
        # m in [0, n-1]
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            A_ub[k, indices] = f[k - indices]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': max_time, 'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp_step(f, M, max_time=10):
    """One Frank-Wolfe step with LP oracle."""
    n = len(f)
    g = solve_lp_oracle(f, M, max_time=max_time)
    if g is None:
        return None
    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs


def fw_lp_optimize(f0, time_limit, max_lp_time=15):
    """Frank-Wolfe with LP oracle."""
    f = f0.copy()
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_f = f.copy()
    best_obj = M

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        g = fw_lp_step(f, M, max_time=max_lp_time)
        if g is None:
            # Try with relaxed M
            g = fw_lp_step(f, M * 1.1, max_time=max_lp_time)
            if g is None:
                no_improve += 1
                if no_improve > 20:
                    break
                continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            # Bisection refinement
            lo = max(0.0, best_alpha - 0.02)
            hi = min(1.0, best_alpha + 0.02)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 30:
            break

    return best_f, best_obj


def active_set_polish(a0, time_limit, tol=1e-10):
    """
    Active-set Newton method exploiting KKT conditions.
    At optimum: sum_{k in P} mu_k * 2*a[k-j] = nu for all j in support S.
    """
    a = a0.copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < 500:
        it += 1
        if it % 50 == 0:
            print(f"    AS it={it}, obj={best_obj:.10f}")

        conv = fast_convolve(a)
        M = conv.max()

        # Find active peaks (within tolerance of max)
        threshold = M * (1.0 - 0.002)
        active_peaks = np.where(conv >= threshold)[0]
        if len(active_peaks) < 2:
            break

        # Find support
        support = np.where(a > 1e-12)[0]
        if len(support) < 2:
            break

        # Build KKT system: sum_{k in P} mu_k * 2*a[k-j] = nu for j in S
        # Unknowns: mu (len P) and nu (1 scalar)
        # Matrix A_sys[i, k_idx] = 2*a[active_peaks[k_idx] - support[i]] (if valid)
        # A_sys[i, -1] = -1 (for nu)
        m = len(support)
        p = len(active_peaks)
        A_sys = np.zeros((m, p + 1))
        for i, j in enumerate(support):
            A_sys[i, -1] = -1.0
            for k_idx, k in enumerate(active_peaks):
                idx = k - j
                if 0 <= idx < n:
                    A_sys[i, k_idx] = 2.0 * a[idx]

        # Solve: find mu, nu such that A_sys @ [mu; nu] = 0 (non-trivial)
        # Use SVD for the null space
        try:
            U, S_vals, Vt = np.linalg.svd(A_sys)
            sol = Vt[-1, :]
            mu_vals = np.maximum(sol[:p], 0.0)
        except np.linalg.LinAlgError:
            break

        mu_sum = mu_vals.sum()
        if mu_sum < 1e-15:
            break
        mu_vals /= mu_sum

        # Compute gradient direction
        # grad = -sum_k mu_k * 2*a[k-j] (negative because we want to decrease)
        # Actually, the direction should decrease the objective.
        # The subgradient of max(a*a) at a is (1/|P|) sum_{k in P} grad(a*a)_k
        # We use weighted version with mu
        grad = np.zeros(n)
        for k_idx, k in enumerate(active_peaks):
            w = mu_vals[k_idx]
            if w < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += w * 2.0 * a[k - idx_arr]

        # Take a step in the negative gradient direction (projected)
        improved = False
        for step in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
            candidate = project_simplex(a - step * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_obj - 1e-14:
                a = candidate
                best_obj = val
                best_a = a.copy()
                improved = True
                break
            elif val < M - 1e-14:
                a = candidate
                improved = True
                break

        if not improved:
            # Try moving in the + direction too (sometimes helpful)
            for step in [0.005, 0.01, 0.02]:
                candidate = project_simplex(a + step * grad)
                c = fast_convolve(candidate)
                val = c.max()
                if val < best_obj - 1e-14:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
                    break

            if not improved:
                break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, max_rounds=200):
    """Cycle through coordinates, optimizing each one greedily."""
    a = a0.copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    order = np.arange(n)

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit - 1:
            break
        if rnd > 0:
            rng.shuffle(order)

        improved_round = False
        for i in order:
            if time.monotonic() - t0 > time_limit - 1:
                break
            a_save = a[i]
            # Optimize a_i: fix all others, find best a_i
            # (a*a)_k = a_i^2 * delta_{k,2i}... no, it's more complex
            # (a*a)_k includes terms with a_i:
            #   a_i * a[k-i] (from j=i in sum)
            #   a_i * a[k-i] (from k-j=i, i.e., j=k-i)
            #   a_i^2 (when k=2i, j=i)
            # Actually let's just do a 1D search
            # For fixed a[-i], the objective is max_k (a*a)_k where (a*a)_k is
            # quadratic in a_i (specifically: a_i^2 + 2*a_i*b_k for each k)
            # So (a*a)_k = c_k + 2*a_i*b_k + a_i^2*delta_{k,2i}... 
            # Let me just do a grid search
            a_no_i = a.copy()
            a_no_i[i] = 0
            conv_base = fast_convolve(a_no_i)
            # (a*a)_k = conv_base[k] + 2*a_i*a[k-i] + a_i^2*(1 if 0<=2i<2n-1 else 0 + cross terms)
            # Actually: (a*a)_k = (a_no_i + a_i*e_i)*(a_no_i + a_i*e_i)
            #            = a_no_i*a_no_i + a_i*(a_no_i*e_i + e_i*a_no_i) + a_i^2*(e_i*e_i)
            # (a_no_i*e_i)_k = a_no_i[k-i] (if valid)
            # (e_i*e_i)_k = delta_{k,2i}
            # So (a*a)_k = conv_base[k] + 2*a_i*a_no_i[k-i] + a_i^2*delta_{k,2i}

            a_no_i_conv = conv_base  # already computed
            cross = np.zeros(2 * n - 1)
            lo = max(0, i - (n - 1))
            hi = min(2 * n - 2, i + (n - 1))
            # cross[k] = a_no_i[k-i] if 0 <= k-i < n
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            if k_lo <= k_hi:
                cross[k_lo:k_hi + 1] = a_no_i[k_lo - i:k_hi - i + 1]

            # (a*a)_k = conv_base[k] + 2*a_i*cross[k] + a_i^2*delta_{k,2i}
            # We want to minimize max_k of this
            # Try a few values of a_i
            a_i_range = np.linspace(0, a_save * 2.5 + 0.02, 30)
            conv_2i = 2 * i if 2 * i < 2 * n - 1 else -1

            best_a_i = a_save
            best_max = np.inf
            for ai in a_i_range:
                conv_test = a_no_i_conv + 2.0 * ai * cross
                if conv_2i >= 0:
                    conv_test[conv_2i] += ai * ai
                # Also need to normalize: sum(a) = 1
                # a_i is part of the sum, so sum = (1 - a_save) + ai
                # The actual sequence after setting a_i = ai needs renormalization
                # But for the local search, let's just keep the sum fixed
                # by adjusting one other element... this gets complicated.
                # Simpler: just varying a_i and not renormalizing (approximate)
                val = conv_test.max()
                if val < best_max:
                    best_max = val
                    best_a_i = ai

            # Renormalize
            a[i] = best_a_i
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a[i] = a_save
                continue

            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = a.copy()
                improved_round = True

        if not improved_round:
            break

    return best_a, best_obj


def pgd_fast(a0, time_limit, top_k=20, lr=0.02, mom=0.9):
    """Fast projected gradient descent with multi-peak subgradient."""
    a = a0.copy()
    a = a / a.sum()
    n = len(a)
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        obj = conv.max()

        # Multi-peak subgradient
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            weights = np.ones(len(peaks))
            # Weight by how close to max
            weights = (pv / obj) ** 4
            wsum = weights.sum()
            if wsum > 0:
                weights /= wsum

        sg = np.zeros(n)
        for i, k in enumerate(peaks):
            w = weights[i]
            if w < 1e-14:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += 2.0 * w * a[k - idx_arr]

        v = mom * v - lr * sg
        candidate = project_simplex(a + v)
        c = fast_convolve(candidate)
        val = c.max()

        if val < obj - 1e-15:
            a = candidate
            if val < best_obj - 1e-15:
                best_obj = val
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Backtrack
            v *= 0.5
            no_improve += 1
            a = project_simplex(a + v)
            c = fast_convolve(a)
            if c.max() < best_obj:
                best_obj = c.max()
                best_a = a.copy()

        if no_improve > 50:
            lr *= 0.8
            if lr < 0.001:
                break
        if no_improve > 200:
            rng = np.random.default_rng(no_improve * 31)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = 0.02
            mom = 0.9
            no_improve = 0

    return best_a, best_obj


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent) + 1)
    x_new = np.linspace(0, 1, n_target + 1)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def try_and_update(a, best_state, label=""):
    """Evaluate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_state
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_state
    best_seq, best_score = best_state
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return (best_seq, best_score)


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search:
    1. L-BFGS on smoothed objective with annealed epsilon
    2. Frank-Wolfe with LP oracle
    3. Active-set Newton refinement
    4. Coordinate descent polishing
    5. Diverse restarts and multi-n exploration
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Quick L-BFGS with moderate epsilon (~80s) ===
    print("\n=== Phase 1: Quick L-BFGS (moderate eps) ===")
    t_start = time.monotonic()
    remaining = deadline - time.monotonic() - 1400
    if remaining > 10:
        a_opt = anneal_lbfgs(best_seq, time_limit=min(remaining, 80),
                             epsilon_schedule=[0.01, 0.005, 0.002])
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "lbfgs_quick")

    print(f"  Phase 1: score={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 2: Full annealed L-BFGS (~400s) ===
    print("\n=== Phase 2: Full Annealed L-BFGS ===")
    t_start = time.monotonic()
    remaining = deadline - time.monotonic() - 1000
    if remaining > 30:
        eps_sched = [0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
        a_opt = anneal_lbfgs(best_seq, time_limit=min(remaining, 400),
                             epsilon_schedule=eps_sched)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "lbfgs_full")

    print(f"  Phase 2: score={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 3: Frank-Wolfe with LP oracle (~350s) ===
    print("\n=== Phase 3: Frank-Wolfe LP ===")
    t_start = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 600
    if remaining > 30:
        t_alloc = min(remaining, 200)
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, max_lp_time=15)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "fw_main")

    # Another FW pass
    remaining = deadline - time.monotonic() - 400
    if remaining > 20:
        t_alloc = min(remaining, 100)
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, max_lp_time=10)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "fw_2nd")

    print(f"  Phase 3: score={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 4: Active-set Newton + Coordinate Descent (~300s) ===
    print("\n=== Phase 4: Active-Set + Coord Descent ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    remaining = deadline - time.monotonic() - 350
    if remaining > 15:
        t_alloc = min(remaining, 120)
        a_opt, _ = active_set_polish(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "as_newton")

    remaining = deadline - time.monotonic() - 200
    if remaining > 15:
        t_alloc = min(remaining, 80)
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "coord_desc")

    # Another active-set pass
    remaining = deadline - time.monotonic() - 120
    if remaining > 10:
        t_alloc = min(remaining, 60)
        a_opt, _ = active_set_polish(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "as_2nd")

    print(f"  Phase 4: score={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Diverse restarts (~250s) ===
    print("\n=== Phase 5: Diverse Restarts ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    for trial in range(5):
        remaining = deadline - time.monotonic() - 150
        if remaining < 30:
            break

        # Create perturbed initialization
        scale = 0.002 + 0.003 * trial
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)

        # Quick L-BFGS
        t_alloc = min(remaining, 60)
        a_opt = anneal_lbfgs(f_pert, time_limit=t_alloc * 0.5,
                             epsilon_schedule=[0.002, 0.001, 0.0005])
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"restart_{trial}")

        # Quick FW
        remaining = deadline - time.monotonic() - 100
        if remaining > 15:
            a_opt2, _ = fw_lp_optimize(a_opt, time_limit=min(remaining, 30), max_lp_time=5)
            (best_seq, best_score) = try_and_update(a_opt2, (best_seq, best_score), f"restart_fw_{trial}")

    print(f"  Phase 5: score={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 6: Multi-n exploration (~150s) ===
    print("\n=== Phase 6: Multi-n Exploration ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    n_candidates = set()
    for n_try in [n_best // 2, n_best, n_best + 64, n_best + 128,
                  384, 512, 640, 768, 896, 1024, 1152]:
        if 128 <= n_try <= 2048:
            n_candidates.add(n_try)

    for n_try in sorted(n_candidates):
        remaining = deadline - time.monotonic() - 80
        if remaining < 20:
            break

        if n_try == n_best:
            start = best_seq.copy()
        else:
            start = interp_sequence(best_seq, n_try)

        t_alloc = min(remaining, 40)
        a_opt = anneal_lbfgs(start, time_limit=t_alloc,
                             epsilon_schedule=[0.005, 0.002, 0.001])
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"n_{n_try}")

    print(f"  Phase 6: score={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 7: Final polishing (~100s) ===
    print("\n=== Phase 7: Final Polish ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    # Fine L-BFGS
    remaining = deadline - time.monotonic() - 40
    if remaining > 10:
        a_opt = anneal_lbfgs(best_seq, time_limit=min(remaining, 40),
                             epsilon_schedule=[0.0001, 0.00005, 0.00002, 0.00001])
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "final_lbfgs")

    # Coordinate descent
    remaining = deadline - time.monotonic() - 20
    if remaining > 10:
        a_opt, _ = coordinate_descent(best_seq, time_limit=min(remaining, 30))
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "final_cd")

    # Symmetry check
    sym_final = (best_seq + best_seq[::-1]) / 2.0
    sym_final = sym_final / sym_final.sum()
    (best_seq, best_score) = try_and_update(sym_final, (best_seq, best_score), "final_sym")

    # Quick PGD polish
    remaining = deadline - time.monotonic() - 10
    if remaining > 5:
        a_opt = pgd_fast(best_seq, time_limit=min(remaining, 15), top_k=30, lr=0.005)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "final_pgd")

    print(f"  Phase 7: score={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
