def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]

    return np.maximum(v - theta, 0.0)

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def objective_val(a):
    """Compute the objective: max of auto-convolution (for normalized a)."""
    conv = fast_convolve(a)
    return conv.max()

def qp_polish(a0, time_limit, M_target=None):
    """
    QP-based polishing: iteratively linearize the active constraints
    and solve a QP to find a better point.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = float(conv.max())
    if M_target is None:
        M_target = M * 0.999  # Try to reduce slightly
    best_a = a.copy()
    best_M = M
    
    t0 = time.monotonic()
    it = 0
    n_active = min(5, 2 * n - 1)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        M = float(conv.max())
        
        if M < best_M:
            best_M = M
            best_a = a.copy()
        
        # Find active peaks (top n_active)
        peak_indices = np.argsort(conv)[::-1][:n_active]
        
        # Build linearized constraint matrix
        # For peak k: 2 * (Q_k @ a)^T @ a_new <= M + (a*a)_k
        # where (Q_k @ a)_i = a_{k-i} if 0 <= k-i < n
        A_qp = []
        b_qp = []
        for k in peak_indices:
            k = int(k)
            # (Q_k @ a)_i = a[k-i] if 0 <= k-i <= n-1
            row = np.zeros(n)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                row[j] = a[k - j]
            A_qp.append(2.0 * row)
            b_qp.append(M + conv[k])  # This is always > M, so constraint is 2*row@a_new <= M + conv[k]
        
        A_qp = np.array(A_qp)
        b_qp = np.array(b_qp)
        
        # Solve QP: min ||x - a||^2 s.t. A_qp @ x <= b_qp, sum(x) = 1, x >= 0
        # Use scipy minimize with SLSQP
        def qp_obj(x):
            return np.sum((x - a) ** 2)
        
        def qp_grad(x):
            return 2.0 * (x - a)
        
        constraints = []
        for i in range(len(A_qp)):
            # A_qp[i] @ x <= b_qp[i]
            # In scipy form: fun(x) >= 0, so: b_qp[i] - A_qp[i] @ x >= 0
            Ai = A_qp[i].copy()
            bi = b_qp[i]
            constraints.append({'type': 'ineq',
                                'fun': lambda x, Ai=Ai, bi=bi: bi - Ai @ x,
                                'jac': lambda x, Ai=Ai: -Ai})
        # sum(x) = 1
        constraints.append({'type': 'eq',
                            'fun': lambda x: x.sum() - 1.0,
                            'jac': lambda x: np.ones(n)})
        
        bounds = [(0, None)] * n
        
        x0 = a.copy()
        try:
            res = minimize(qp_obj, x0, jac=qp_grad, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 200, 'ftol': 1e-12})
            if res.success:
                x_new = np.maximum(res.x, 0.0)
                x_new = x_new / x_new.sum()
                val = objective_val(x_new)
                if val < M - 1e-15:
                    a = x_new
        except Exception:
            pass
        
        # If no improvement, try with a slightly tighter M
        conv_check = fast_convolve(a)
        if conv_check.max() >= best_M:
            M_target = best_M * (1.0 - 0.0005 * min(it, 20))
            # Perturb
            if it % 20 == 0:
                rng = np.random.default_rng((it * 9973) % (2**31))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.001)

    return best_a, best_M


def spectral_init(n, parent=None, rng=None):
    """Generate diverse initializations using spectral methods."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    starts = []
    
    # 1. Resampled parent
    if parent is not None and len(parent) > 1:
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        p = np.interp(x_new, x_old, parent)
        p = np.maximum(p, 0)
        p = p / p.sum()
        starts.append(p)
    
    # 2. Uniform
    starts.append(np.ones(n) / n)
    
    # 3. Cosine taper (Bell-shaped)
    x = np.arange(n, dtype=np.float64)
    cos_t = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    cos_t = np.maximum(cos_t, 0.01)
    starts.append(cos_t / cos_t.sum())
    
    # 4. Gaussian center
    sigma = 0.25 * n
    g = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    starts.append(g / g.sum())
    
    # 5. Two Gaussians
    g1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.1 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())
    
    # 6. Spectral flattening: take parent (or random), flatten its spectrum
    base = parent if parent is not None else rng.dirichlet(np.ones(n) * 0.5)
    base = base / base.sum()
    # DFT
    F = np.fft.fft(base)
    # Flatten magnitudes (keep phase)
    mags = np.abs(F)
    phases = np.angle(F)
    # Target: make mags more uniform
    target_mag = np.ones(n)
    target_mag[0] = 1.0  # DC component
    # Blend: new_mag = (1-t)*old_mag + t*target_mag
    t_flat = 0.5
    new_mags = (1 - t_flat) * mags + t_flat * target_mag
    new_mags = np.maximum(new_mags, 1e-10)
    # Inverse FFT
    F_new = new_mags * np.exp(1j * phases)
    a_new = np.real(np.fft.ifft(F_new))
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() > 1e-10:
        a_new = a_new / a_new.sum()
        starts.append(a_new)
    
    # 7. Random
    starts.append(rng.dirichlet(np.ones(n) * 0.3))
    starts.append(rng.dirichlet(np.ones(n) * 1.0))
    
    return starts


def micro_polish(a0, time_limit):

    a = np.maximum(a0.astype(np.float64), 0.0)
