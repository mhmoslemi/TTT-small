"""

def fast_autocorr(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def structured_subgrad(a, weights):
    """Compute 2 * sum_k weights[k] * a[k-j] for all j using FFT.
    weights has length 2n-1, a has length n.
    Result g[j] = 2 * sum_m a[m] * weights[m+j] for m=0..n-1.
    This is 2 * cross_correlation(weights, a)[j] for j=0..n-1.
    """
    n = len(a)
    # Use FFT for cross-correlation: corr[j] = sum_m a[m] * w[m+j]
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.fft(a, fft_size)
    W = np.fft.fft(weights, fft_size)
    # Cross-correlation: ifft(conj(A) * W) gives sum_m a[m] * conj(w[m-j])... 
    # Let me use the direct definition with np.correlate for correctness
    # For n <= 4096, this is fast enough
    corr = np.correlate(weights, a, mode='valid')
    return 2.0 * corr


def softmax_weights(v, T):
    """Compute softmax weights with temperature T."""
    v = v - v.max()
    e = np.exp(v / T)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def kl_project(log_a):
    """Project from log-space: normalize exponential to get probability vector."""
    log_a = np.maximum(log_a, -60.0)  # prevent underflow
    log_a_max = log_a.max()
    a = np.exp(log_a - log_a_max)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s

def fw_structured(a0, time_limit, top_k=20, seed=0):
    """Frank-Wolfe using the structured subgradient (no LP needed).
    The linearization minimum over the simplex is at a vertex e_j
    where j minimizes the linearized objective.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    t0 = time.monotonic()

    no_improve = 0

        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_autocorr(a)
        M = conv.max()
        
        # Use top-k peaks for the subgradient
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        # Softmax weights on active peaks
        peak_vals = conv[peaks]
        T_peak = max((peak_vals.max() - peak_vals.min()) / 5.0, 1e-10)
        w_peaks = softmax_weights(peak_vals, T_peak)
        
        # Build full weight vector
        w = np.zeros(len(conv))
        w[peaks] = w_peaks
        
        # Compute subgradient: g_j = 2 * sum_k w_k * a_{k-j}
        g = structured_subgrad(a, w)
        
        # FW step: move towards e_{j*} where j* = argmin g
        j_star = np.argmin(g)
        e_j = np.zeros(n)
        e_j[j_star] = 1.0
        
        # Line search
        alpha = 2.0 / (it + 2.0)
        best_alpha = 0.0
        best_val = M
        for a_try in [alpha, alpha * 0.5, alpha * 2.0, 0.1, 0.3, 0.5]:
            a_try = min(max(a_try, 0.01), 0.99)
            candidate = (1 - a_try) * a + a_try * e_j
            candidate = np.maximum(candidate, 0)
            candidate = candidate / candidate.sum()
            val = fast_autocorr(candidate).max()
            if val < best_val:
                best_val = val
                best_alpha = a_try

            a = np.maximum(a, 0)

def active_set_refine(a0, time_limit, tol=1e-10, max_iter=5000):
    """Active-set Newton: at optimum, identify active peaks and solve KKT."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()

        it += 1
        
        conv = fast_autocorr(a)
        M = conv.max()
        
        # Identify active peaks (within tolerance of max)
        thresh = M * (1 - 1e-6)
        active = np.where(conv >= thresh)[0]
        
        if len(active) == 0:
            break
        
        # Compute uniform weights on active set
        w = np.zeros(len(conv))
        w[active] = 1.0 / len(active)
        
        # Subgradient
        g = structured_subgrad(a, w)
        
        # Try to find a direction that reduces the objective
        # Project gradient onto the tangent space of the simplex
        g_tan = g - g.mean()
        
        # Check if we're at a stationary point
        if np.max(np.abs(g_tan)) < tol:
            break
        
        # Line search along -g_tan
        best_val = M
        best_a_new = a
        for step in [0.01, 0.005, 0.002, 0.001, 0.02, 0.05]:
            candidate = a - step * g_tan
            candidate = np.maximum(candidate, 0)
            cs = candidate.sum()
            if cs < 1e-15:
                continue
            candidate = candidate / cs
            val = fast_autocorr(candidate).max()
            if val < best_val - 1e-18:
                best_val = val
                best_a_new = candidate
        
        if best_a_new is not a:
            a = best_a_new
            if best_val < best_M - 1e-15:
                best_M = best_val
                best_a = a.copy()
    
    return best_a, best_M


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def solve_lp_oracle(f, M, max_time=20):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'max_time': max_time})
        if res.success:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None

        if remaining > 15:
            conv = fast_autocorr(best_seq)
            T_init = conv.max() / 100.0
            t_alloc2 = min(remaining, 80)
            a_opt2, _ = mirror_descent(best_seq, time_limit=t_alloc2, T_init=T_init,
                                       T_min=1e-10, eta_init=0.01, decay=0.9998,
                                       sym_every=20, fw_every=0, seed=seed+300)
            try_update(a_opt2, "md_fine")
            try_update((a_opt2 + a_opt2[::-1]) / 2.0, "md_fine_sym")
    
    print(f"  Phase 4 done: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")
    
    # === Phase 5: Perturbation restarts (~150s) ===
    print("\n=== Phase 5: Perturbation Restarts ===")
    t_start = time.monotonic()
    n_best = len(best_seq)
    
    for restart in range(8):
        remaining = deadline - time.monotonic() - 60
        if remaining < 15:
            break
        
        # Different perturbation scales
        scales = [0.001, 0.003, 0.007, 0.015, 0.002, 0.005, 0.01, 0.02]
        scale = scales[restart % len(scales)]
        perturbed = best_seq + rng.standard_normal(n_best) * scale
        perturbed = np.maximum(perturbed, 0)
        perturbed = perturbed / perturbed.sum()
        # Enforce symmetry
        perturbed = (perturbed + perturbed[::-1]) / 2.0
        perturbed = perturbed / perturbed.sum()
        
        t_alloc = min(remaining, 20)
        conv = fast_autocorr(perturbed)
        T_init = conv.max() / 30.0
        a_opt, _ = mirror_descent(perturbed, time_limit=t_alloc, T_init=T_init,
                                  eta_init=0.02, decay=0.9999, sym_every=20, 
                                  fw_every=0, seed=seed + 400 + restart)
        try_update(a_opt, f"pert_{restart}")
        
        # Active-set on the perturbed result
        remaining = deadline - time.monotonic() - 40
        if remaining > 8:
            a_opt2, _ = active_set_refine(a_opt, time_limit=min(remaining, 10), 
                                           tol=1e-12, max_iter=1000)
            try_update(a_opt2, f"pert_as_{restart}")
    
    print(f"  Phase 5 done: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")
    
    # === Phase 6: Final polishing (~100s) ===
    print("\n=== Phase 6: Final Polish ===")
    t_start = time.monotonic()
    n_best = len(best_seq)
    
    # Very fine mirror descent
    remaining = deadline - time.monotonic() - 30
    if remaining > 10:
        conv = fast_autocorr(best_seq)
        T_init = conv.max() / 200.0
        a_opt, _ = mirror_descent(best_seq, time_limit=min(remaining, 40), T_init=T_init,
                                  T_min=1e-10, eta_init=0.005, decay=0.9999,
                                  sym_every=10, fw_every=0, seed=seed+500)
        try_update(a_opt, "final_md")
    
    # Active-set Newton final
    remaining = deadline - time.monotonic() - 20
    if remaining > 8:
        a_opt2, _ = active_set_refine(best_seq, time_limit=min(remaining, 20),
                                       tol=1e-13, max_iter=5000)
        try_update(a_opt2, "final_as")
    
    # Final symmetry
    a_sym = (best_seq + best_seq[::-1]) / 2.0
    a_sym = a_sym / a_sym.sum()
    try_update(a_sym, "final_sym")
    
    # One more active-set on symmetrized
    remaining = deadline - time.monotonic() - 10
    if remaining > 5:
        a_opt3, _ = active_set_refine(a_sym, time_limit=min(remaining, 8),
                                       tol=1e-13, max_iter=2000)
        try_update(a_opt3, "final_sym_as")
    
    print(f"  Phase 6 done: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = eval_score(best_seq)
    
    # Final verification using the actual evaluate_sequence
    fs = float(evaluate_sequence([float(x) for x in best_seq]))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
