"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Low-dimensional cosine parameterization with L-BFGS for global shape
2. Nesterov-accelerated projected gradient on smoothed max
3. Peak-equalizing direction search (convex feasibility in gradient space)
4. Targeted pairwise swap local search
5. Simulated annealing polish
Key insight: at optimality, active peaks should be equalized; use dual structure
to guide search directions.

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_value(a):
    """Compute the actual score 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_n = a / s
    conv = fast_convolve(a_n)
    return 2.0 * n * conv.max()


def smoothed_obj(a, T):
    """Smoothed max: 2n * logsumexp(T*conv)/T (a assumed sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    shifted = conv * T
    shifted -= shifted.max()
    log_sum_exp = np.log(np.sum(np.exp(shifted))) / T
    return 2.0 * n * log_sum_exp


def gradient_smoothed(a, T):
    """Gradient of smoothed max w.r.t. a (before simplex projection)."""
    n = len(a)
    conv = fast_convolve(a)
    shifted = conv * T
    shifted -= shifted.max()
    p = np.exp(shifted)
    p /= p.sum()
    grad_a = 4.0 * n * np.correlate(p, a, mode='valid')
    return grad_a


def cosine_sequence(params, n, K):
    """Build sequence from Fourier coefficients. params = [c1,d1,c2,d2,...]."""
    j = np.arange(n)
    a = np.ones(n)
    for k in range(1, K + 1):
        ck = params[2 * (k - 1)]
        dk = params[2 * (k - 1) + 1]
        a = a + ck * np.cos(2.0 * np.pi * k * j / n) + dk * np.sin(2.0 * np.pi * k * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def cosine_obj(params, n, K, T):
    """Objective for cosine params."""
    a = cosine_sequence(params, n, K)
    return smoothed_obj(a, T)


def cosine_optimize(n, K, time_limit, T=15.0, x0=None):
    """Optimize K Fourier coefficients via L-BFGS + Nelder-Mead."""
    if x0 is None:
        x0 = np.zeros(2 * K)

    def obj_fn(p):
        return cosine_obj(p, n, K, T)

    t0 = time.monotonic()
    best_p = x0.copy()
    best_v = obj_fn(x0)

    # L-BFGS
    try:
        res = minimize(obj_fn, best_p, method='L-BFGS-B',
                       options={'maxiter': 300, 'maxfun': 1000, 'gtol': 1e-9})
        if res.fun < best_v:
            best_v = res.fun
            best_p = res.x
    except Exception:
        pass

    # Nelder-Mead polish
    if time.monotonic() - t0 < time_limit * 0.6:
        try:
            res2 = minimize(obj_fn, best_p, method='Nelder-Mead',
                            options={'maxiter': 500, 'xatol': 1e-11, 'fatol': 1e-12})
            if res2.fun < best_v:
                best_v = res2.fun
                best_p = res2.x
        except Exception:
            pass

    return cosine_sequence(best_p, n, K)


def nesterov_smoothed(a0, time_limit, T_init=5.0, T_final=80.0, lr_init=0.008):
    """Nesterov accelerated projected gradient on smoothed max with adaptive T."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    y = a.copy()
    v = np.zeros(n)
    t_nest = 1.0
    lr = lr_init
    best_a = a.copy()
    best_val = objective_value(a)

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        progress = (time.monotonic() - t0) / time_limit
        T_now = T_init + (T_final - T_init) * progress

        # Nesterov lookahead
        alpha_n = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / alpha_n
        y = (1.0 + beta) * a - beta * y
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y / ys
        t_nest += 1.0

        grad = gradient_smoothed(y, T_now)
        v = beta * v - lr * grad
        a_new = project_simplex(a + v)

        val = objective_value(a_new)
        if val < best_val - 1e-13:
            best_val = val
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            a_new = a

        a = a_new
        lr *= 0.9995

        if no_improve > 150:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.001 * min(1.0, no_improve / 100.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            y = a.copy()
            v = np.zeros(n)
            t_nest = 1.0
            no_improve = 0

    return best_a


def peak_equalize_step(a, n_peaks=5, lr=0.01):
    """
    Find a direction that reduces the top peak while not increasing other top peaks.
    Uses a simple projected approach: d = -g_top projected to satisfy constraints.
    """
    n = len(a)
    conv = fast_convolve(a)
    peak_lags = np.argpartition(conv, -n_peaks)[-n_peaks:]
    peak_lags = peak_lags[np.argsort(conv[peak_lags])[::-1]]

    # Compute gradients for each peak
    grads = []
    for k in peak_lags:
        g = np.zeros(n)
        js = max(0, int(k) - n + 1)
        je = min(n - 1, int(k))
        for j in range(js, je + 1):
            g[j] += 2.0 * a[int(k) - j]
        grads.append(g)

    # Target: reduce the top peak
    target_grad = grads[0]
    d = -target_grad / max(np.linalg.norm(target_grad), 1e-15)

    # Project against other peaks (simple: if d would increase them, reduce component)
    for i in range(1, len(peak_lags)):
        dot = np.dot(d, grads[i])
        if dot > 0:
            # Remove the component along grads[i]
            g_norm = np.dot(grads[i], grads[i])
            if g_norm > 1e-15:
                d -= dot / g_norm * grads[i]

    d_norm = np.linalg.norm(d)
    if d_norm < 1e-10:
        return a

    d = d / d_norm

    # Line search
    best_alpha = 0.0
    best_val = conv.max()
    for trial_alpha in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
        a_test = project_simplex(a + trial_alpha * d)
        c = fast_convolve(a_test)
        if c.max() < best_val:
            best_val = c.max()
            best_alpha = trial_alpha

    if best_alpha > 0:
        return project_simplex(a + best_alpha * d)
    return a


def local_search_targeted(a0, time_limit):
    """Targeted local search: always attack the global peak with optimal transfers."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_val = objective_value(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)

        # Find best single transfer to reduce peak
        best_gain = 0.0
        best_j_src = -1
        best_j_dst = -1
        best_delta = 0.0

        # Source candidates: indices contributing most to the peak
        contribs = []
        for j in range(js, je + 1):
            partner = peak_k - j
            if 0 <= partner < n:
                contribs.append((a[j] * a[partner], j))
        contribs.sort(reverse=True)

        rng = np.random.default_rng(stall * 7919 + 13)

        for c_val, j_src in contribs[:20]:
            if a[j_src] < 1e-8:
                continue
            # Destination: random index that contributes little to this peak
            for _ in range(8):
                j_dst = rng.integers(0, n)
                if j_dst == j_src:
                    continue
                partner_dst = peak_k - j_dst
                contrib_dst = a[j_dst] * a[partner_dst] if 0 <= partner_dst < n else 0.0
                if contrib_dst > c_val * 0.5:
                    continue

                for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    delta = a[j_src] * frac
                    if delta < 1e-12:
                        continue
                    a_test = a.copy()
                    a_test[j_src] -= delta
                    a_test[j_dst] += delta
                    a_test = np.maximum(a_test, 0.0)
                    st = a_test.sum()
                    if st < 1e-15:
                        continue
                    a_test = a_test / st
                    c_new = fast_convolve(a_test)
                    new_max = c_new.max()
                    gain = best_val - 2.0 * n * new_max
                    if gain > best_gain:
                        best_gain = gain
                        best_j_src = j_src
                        best_j_dst = j_dst
                        best_delta = delta

        if best_j_src >= 0 and best_gain > 1e-12:
            a[best_j_src] -= best_delta
            a[best_j_dst] += best_delta
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            best_val = objective_value(a)
            best_a = a.copy()
            stall = 0
        else:
            # Try peak-equalize step
            a_try = peak_equalize_step(a, n_peaks=4, lr=0.005)
            val_try = objective_value(a_try)
            if val_try < best_val - 1e-13:
                a = a_try
                best_val = val_try
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall % 50 == 0:
                    rng2 = np.random.default_rng(stall * 48271)
                    a = project_simplex(best_a + rng2.standard_normal(n) * 0.002)

        if stall > 500:
            break

    return best_a


def simulated_annealing(a0, time_limit, T_start=0.003, T_end=0.0001):
    """Simulated annealing with structured perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_val = objective_value(a)
    curr_val = best_val

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        progress = (time.monotonic() - t0) / time_limit
        T = T_start * (T_end / T_start) ** progress

        rng = np.random.default_rng(it * 2654435761)

        # Structured perturbation: bias toward reducing peaks
        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)

        noise = rng.standard_normal(n) * T
        # Reduce contributors to peak
        for j in range(js, je + 1):
            partner = peak_k - j
            if 0 <= partner < n:
                noise[j] -= T * a[partner] * 2.0

        a_new = project_simplex(a + noise)
        val_new = objective_value(a_new)
        delta = val_new - curr_val

        if delta < 0 or rng.random() < np.exp(-delta / max(T * 0.005, 1e-12)):
            a = a_new
            curr_val = val_new
            if val_new < best_val:
                best_val = val_new
                best_a = a_new.copy()

    return best_a

    print("\n=== Phase 4: SA + Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(100, max(10, deadline - time.monotonic() - 80))
        if t_alloc > 5:
            print(f"  SA t={t_alloc:.0f}s")
            a_opt = simulated_annealing(best_seq, t_alloc, T_start=0.002, T_end=0.00005)
            try_seq(a_opt, "SA")

        # Final targeted local search with very small steps
        if best_seq is not None:
            t_alloc = min(60, max(5, deadline - time.monotonic() - 40))
            if t_alloc > 5:
                a_opt = local_search_targeted(best_seq, t_alloc)
                try_seq(a_opt, "final_ls")

        # One more Nesterov with very high T (sharp max)
        if best_seq is not None:
            t_alloc = min(30, max(5, deadline - time.monotonic() - 20))
            if t_alloc > 3:
                a_opt = nesterov_smoothed(best_seq, t_alloc, T_init=50.0, T_final=300.0, lr_init=0.002)
                try_seq(a_opt, "final_nest")

    print(f"Phase 4 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final check
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)}, time={time.monotonic()-t0_total:.0f}s)")
    return [float(x) for x in best_seq]
