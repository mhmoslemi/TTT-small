import numpy as np
from scipy.optimize import minimize, linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_objective(a):
    """Compute 2*n*max(auto_conv) for normalized a (sum=1)."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)

            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:

            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)

"""
Hybrid optimization for minimum peak autocorrelation step function.
Uses coordinate descent with pair-wise golden section search,
smoothed gradient with annealing, projected gradient with momentum,
and multi-scale local search. Focuses on precise 1D sub-problem
solutions to squeeze out small improvements over the state of the art.
"""

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_val(a):
    """Compute 2*n*max(auto_conv) assuming sum(a)~1."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * float(c.max())


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def pair_eval(a, i, j, t):
    """Objective when a[i]-=t, a[j]+=t (no renormalization needed since sum preserved)."""
    b = a.copy()
    b[i] -= t
    b[j] += t
    if b[i] < -1e-14 or b[j] < -1e-14:
        return 1e10
    c = fast_convolve(b)
    return 2.0 * len(b) * float(c.max())


def coord_descent_pair(a, i, j, time_budget=None):
    """Golden section search over t for pair (i, j). Returns (best_t, best_obj)."""
    ai = a[i]
    aj = a[j]
    lo = -aj + 1e-15
    hi = ai - 1e-15
    if lo >= hi:
        return 0.0, obj_val(a)

    def f(t):
        return pair_eval(a, i, j, t)

    res = minimize_scalar(f, bounds=(lo, hi), method='bounded',
                          options={'xatol': 1e-14, 'maxiter': 80})
    if res.success:
        t_best = float(res.x)
        obj_best = float(res.fun)
    else:
        t_best = 0.0
        obj_best = obj_val(a)
    return t_best, obj_best


def coord_descent_pass(a, n_pairs, rng, verbose=False):

def smoothed_grad_descent(a0, time_limit, beta_start=50.0, beta_end=500.0,
                          lr_start=0.01, momentum=0.9):
    """Smoothed gradient descent with annealing on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = obj_val(a)

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    v = np.zeros(n)  # momentum

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Anneal beta
        progress = min(1.0, it / 2000.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start / (1.0 + 0.001 * it)

        # Compute smoothed gradient
        c = fast_convolve(a)
        beta_c = beta * c
        beta_c -= beta_c.max()
        p = np.exp(beta_c)
        p = p / p.sum()

        # Gradient: grad[m] = (2/beta) * sum_j a[j] * p[m+j]
        grad = np.zeros(n)
        for m in range(n):
            grad[m] = np.dot(a, p[m:m + n])
        grad *= 2.0 / beta
        grad *= 2.0 * n  # full objective gradient

        # Project gradient onto tangent space of simplex
        grad_t = grad - np.mean(grad)

        # Momentum update
        v = momentum * v - lr * grad_t
        a_new = a + v
        a_new = project_simplex(a_new)

        obj_new = obj_val(a_new)
        if obj_new < best_obj - 1e-15:
            a = a_new
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Revert momentum partially
            v *= 0.5

        if no_improve > 100:
            v *= 0.1
            lr *= 0.9
            if no_improve > 500:
                rng = np.random.default_rng(it * 7919)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                v = np.zeros(n)
                no_improve = 0

    return best_a, best_obj


def direct_lbfgs(a0, time_limit, beta=200.0):
    """L-BFGS optimization using softmax parameterization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    # Initialize z from a (log of a, with small epsilon)
    z = np.log(a + 1e-30)

    best_a = a.copy()
    best_obj = obj_val(a)

    def func_and_grad(z):
        # Softmax
        z_shift = z - z.max()
        exp_z = np.exp(z_shift)
        a_curr = exp_z / exp_z.sum()

        # Objective
        c = fast_convolve(a_curr)
        beta_c = beta * c
        beta_c -= beta_c.max()
        obj_val_curr = (2.0 * n / beta) * (beta_c.max() + np.log(np.sum(np.exp(beta_c))))

        # Gradient with respect to a
        p = np.exp(beta_c)
        p = p / p.sum()
        grad_a = np.zeros(n)
        for m in range(n):
            grad_a[m] = np.dot(a_curr, p[m:m + n])
        grad_a *= (2.0 * n) / beta

        # Gradient with respect to z: grad_z[j] = a[j] * (grad_a[j] - dot(a, grad_a))
        dot_ag = np.dot(a_curr, grad_a)
        grad_z = a_curr * (grad_a - dot_ag)

        return obj_val_curr, grad_z

    t0 = time.monotonic()
    try:
        res = minimize(func_and_grad, z, method='L-BFGS-B',
                       options={'maxiter': 200, 'maxfun': 500,
                                'ftol': 1e-15, 'gtol': 1e-10})
    except Exception:
        pass

    # Extract result
    try:
        z_opt = res.x
        z_shift = z_opt - z_opt.max()
        a_opt = np.exp(z_shift)
        a_opt = a_opt / a_opt.sum()
        a_opt = np.maximum(a_opt, 0)
        obj_opt = obj_val(a_opt)
        if obj_opt < best_obj:
            best_a = a_opt
            best_obj = obj_opt
    except Exception:
        pass

    elapsed = time.monotonic() - t0
    print(f"    L-BFGS: {elapsed:.1f}s, obj={best_obj:.8f}")

    return best_a, best_obj

def interp_sequence(source, n_target):
    """Interpolate a sequence to a different length."""
    if len(source) == n_target:
        return source.copy()
    x_source = np.linspace(0, 1, len(source))
    x_target = np.linspace(0, 1, n_target)
    result = np.interp(x_target, x_source, source)
    result = np.maximum(result, 0)
    ss = result.sum()
    return result / ss if ss > 1e-10 else np.ones(n_target) / n_target

                t_best, obj_c = coord_descent_pair(f_micro, i, j)
                if obj_c < obj_val(f_micro) - 1e-15:
                    f_micro[i] -= t_best
                    f_micro[j] += t_best
                    f_micro = np.maximum(f_micro, 0)
                    f_micro = f_micro / f_micro.sum()
                else:
                    break
            score = float(evaluate_sequence(f_micro.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = f_micro.copy()
                print(f"  Micro {attempt}: {best_score:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
