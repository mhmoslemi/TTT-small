    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = compute_grad(a, p)

        log_update = -eta * grad
        log_update -= log_update.max()
        update = np.exp(log_update)
        a_new = a * update
        a_new = np.maximum(a_new, 1e-30)
        sn = a_new.sum()
        a_new = a_new / sn

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            T *= 1.03
            eta *= 0.97
        if no_improve > 150:
            T *= 0.85
            eta = eta_init * 0.8
        if no_improve > 400:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            best_obj = fast_convolve(a).max()
            T = T_init
            eta = eta_init
            no_improve = 0

    return best_a, best_obj


def accelerate_smoothed_grad(a0, time_limit, T_init=0.0005, lr_init=0.04):
    """Accelerated (Nesterov) smoothed gradient descent on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    T = T_init
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    y = a.copy()  # Nesterov momentum variable
    t_mom = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(y)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = compute_grad(y, p)

        # Line search
        best_candidate = a
        best_val = fast_convolve(a).max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(y - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a_new = best_candidate

        # Nesterov momentum update
        t_mom_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_mom_new
        y_new = a_new + beta * (a_new - a)
        y_new = project_simplex(y_new)

        a = a_new
        y = y_new
        t_mom = t_mom_new

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 20:
            T *= 1.02
            lr *= 0.995
        if no_improve > 100:
            T *= 0.8
            lr = lr_init * 0.7
            t_mom = 1.0
            y = a.copy()
        if no_improve > 300:
            rng = np.random.default_rng((it * 31 + 7) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            y = a.copy()
            T = T_init
            lr = lr_init
            t_mom = 1.0
            no_improve = 0

    return best_a, best_obj


def build_lp_indices(n):
    """Build sparse index arrays for LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def fw_lp_optimize(f0, time_limit, row_idx=None, col_idx=None):
    """Frank-Wolfe with LP oracle and adaptive M relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if row_idx is None:
        row_idx, col_idx = build_lp_indices(n)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        g = solve_lp_oracle(f, M * m_exp, n, row_idx, col_idx)
        if g is None:
            m_exp = min(m_exp * 1.4, 8.0)
            if m_exp >= 8.0:
                rng = np.random.default_rng(it * 31 + 7)
                g = project_simplex(f + rng.standard_normal(n) * 0.01)
                m_exp = 1.0
            else:
                continue

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 50:
                rng = np.random.default_rng(it * 7919 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        best_alpha, best_val = fw_line_search(f, g, M, n)

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.15, 8.0)

    return best_f, best_obj


def solve_lp_oracle(f, M, n, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    vals = np.empty(len(row_idx))
    for idx in range(len(row_idx)):
        k = row_idx[idx]
        i = col_idx[idx]
        vals[idx] = f[k - i]

    A_mat = lil_matrix((2 * n - 1, n))
    A_mat[row_idx, col_idx] = vals
    A_mat = A_mat.tocsr()

    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.fun is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def fw_line_search(f, g, M, n):
    """Line search between f and g with ternary refinement."""
    best_alpha = 0.0
    best_val = M
    alphas = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]
    for alpha in alphas:
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha

    if 0 < best_alpha < 1:
        lo = max(0.0, best_alpha - 0.15)
        hi = min(1.0, best_alpha + 0.15)
        for _ in range(14):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = np.maximum((1 - m1) * f + m1 * g, 0)
            f1 = f1 / f1.sum()
            v1 = fast_convolve(f1).max()
            f2 = np.maximum((1 - m2) * f + m2 * g, 0)
            f2 = f2 / f2.sum()
            v2 = fast_convolve(f2).max()
            if v1 < v2:
                hi = m2
            else:
                lo = m1
        mid = (lo + hi) / 2.0
        ft = np.maximum((1 - mid) * f + mid * g, 0)
        ft = ft / ft.sum()
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = mid

    return best_alpha, best_val


def local_search_polish(a0, time_limit, seed=42):
    """Aggressive local search with multiple perturbation strategies."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        improved = False

        # Multi-scale random perturbations
        for scale in [0.0002, 0.0005, 0.001, 0.003, 0.008, 0.015]:
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(a + pert)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        # Coordinate moves (transfer mass between two positions)
        if not improved and it % 2 == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.4]:
                    candidate = a.copy()
                    amount = candidate[i] * frac
                    candidate[i] -= amount
                    candidate[j] += amount
                    candidate = np.maximum(candidate, 0)
                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break

        # Structured: sinusoidal perturbation
        if not improved and it % 5 == 0:
            freq = rng.uniform(1, 12)
            phase = rng.uniform(0, 2 * np.pi)
            pattern = np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            for amp in [0.001, 0.003, 0.006]:
                candidate = project_simplex(a + pattern * amp)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break

        # Block transfer: move mass from a block of high values to low values
        if not improved and it % 3 == 0:
            block = max(1, n // 32)
            idx_i = int(rng.integers(0, n - block))
            idx_j = int(rng.integers(0, n - block))
            if idx_i != idx_j:
                candidate = a.copy()
                src = candidate[idx_i:idx_i + block]
                dst = candidate[idx_j:idx_j + block]
                transfer = src.sum() * 0.1
                candidate[idx_i:idx_i + block] = src - transfer / block
                candidate[idx_j:idx_j + block] = dst + transfer / block
                candidate = np.maximum(candidate, 0)
                ss = candidate.sum()
                if ss > 1e-10:
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True

        if not improved and it > 2000:
            break

    return best_a, best_obj


def constraint_focused_opt(a0, time_limit):
    """Optimization that focuses on the most violated constraints."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    T = 0.0003
    lr = 0.03

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        # Focus on top few constraints
        top_k = max(3, n // 4)
        top_idx = np.argsort(conv)[::-1][:top_k]
        p = np.zeros(2 * n - 1)
        p[top_idx] = 1.0 / top_k
        # Add small weight to all
        p += 0.01 / (2 * n - 1)
        p = p / p.sum()

        grad = compute_grad(a, p)

        best_candidate = a
        best_val = conv.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        # Adapt
        if it % 50 == 0:
            T *= 1.01
            lr *= 0.999
        if it % 500 == 0:
            rng = np.random.default_rng(it)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)

    return best_a, best_obj


def try_seq(a, label, best_ref):
    """Evaluate a sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_ref
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_ref[1]:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return (a_norm.copy(), score)
    return best_ref

    if remaining > 20:
        t_alloc = min(remaining * 0.20, 40)
        n_best = len(best_seq)
        row_idx, col_idx = build_lp_indices(n_best)
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc,
                                  row_idx=row_idx, col_idx=col_idx)
        best_seq, best_score = try_seq(a_opt, "final_fw", (best_seq, best_score))

    # Micro perturbations + quick gradient
    remaining = deadline - time.monotonic()
    if remaining > 10:
        for _ in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 3:
                break
            noise = rng.standard_normal(len(best_seq)) * 0.0003
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(remaining * 0.25, 5)
            a_opt, _ = accelerate_smoothed_grad(f_pert, time_limit=t_alloc,
                                                T_init=0.0002, lr_init=0.01)
            best_seq, best_score = try_seq(a_opt, "micro", (best_seq, best_score))

    # Final local search
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_opt, _ = local_search_polish(best_seq, time_limit=min(remaining * 0.3, 20), seed=424242)
        best_seq, best_score = try_seq(a_opt, "local2", (best_seq, best_score))

    print(f"  Phase 5 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
