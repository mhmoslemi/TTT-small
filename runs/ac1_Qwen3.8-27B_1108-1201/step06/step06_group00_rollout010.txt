"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral shaping (iterative magnitude flattening + non-negativity projection)
2. Frank-Wolfe with delayed constraints (adaptive LP constraint growth)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def build_binding_matrix(f, constraint_indices):
    """Build sparse constraint matrix for LP. Shape: (num_constr, n)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([idx] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    if not rows:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(len(constraint_indices), n))


def solve_lp_direction(f, M, constraint_indices, time_limit=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for given constraints, g >= 0."""
    n = len(f)
    A = build_binding_matrix(f, constraint_indices)
    if A.shape[0] == 0:
        return None
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None


def spectral_shaping(f, n_iter=50, flatten_alpha=0.3, rng=None):
    """Iterative spectral flattening + non-negativity projection."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = np.maximum(f.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    fft_size = 1 << (2 * n - 1).bit_length()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    for it in range(n_iter):
        A = np.fft.rfft(f, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        # Shrink dynamic range: mag_new = mag^alpha * mean_mag^(1-alpha)
        mag_mean = mag.mean() + 1e-30
        log_mag = np.log(mag + 1e-30)
        log_mean = np.log(mag_mean)
        log_mag_new = log_mag * flatten_alpha + log_mean * (1.0 - flatten_alpha)
        mag_new = np.exp(log_mag_new)
        # Keep DC component fixed
        mag_new[0] = mag[0]
        A_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(A_new, fft_size)[:n]
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new = f_new / s_new
        obj = fast_convolve(f_new).max()
        if obj < best_obj:
            best_obj = obj
            best_f = f_new.copy()
        f = f_new

    return best_f, best_obj

def lse_subgradient(f0, time_limit, beta_start=5.0, beta_end=3000.0, top_k=50):
    """LSE subgradient with adaptive step size and backtracking."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    fft_size = 1 << (2 * n - 1).bit_length()

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 1.5 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        conv = fast_convolve(f)
        # LSE weights
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        total = exp_bc.sum()
        if total < 1e-300:
            w = np.ones(len(conv)) / len(conv)
        else:
            w = exp_bc / total

        k_top = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        w_peaks = w[peak_idx]
        w_sum = w_peaks.sum()
        if w_sum > 1e-15:
            w_peaks = w_peaks / w_sum
        else:
            w_peaks = np.ones(k_top) / k_top

        # Compute subgradient
        grad = np.zeros(n)
        for idx, k in enumerate(peak_idx):
            k = int(k)
            wv = w_peaks[idx]
            if wv < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            i_range = np.arange(lo, hi + 1)
            grad[i_range] += wv * 2.0 * f[k - i_range]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            no_improve += 100
            break
        grad_normed = grad / gnorm

        # Backtracking line search
        step = lr
        improved = False
        for _ in range(12):
            f_new = project_simplex(f - step * grad_normed)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                improved = True
                no_improve = 0
                lr = min(lr * 1.003, 5.0 / np.sqrt(n))
                break
            step *= 0.5

        if not improved:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.85
            if no_improve > 60:
                f = best_f.copy()
                lr = max(1.5 / np.sqrt(n), 0.003)
                beta = min(beta * 2.5, beta_end)
                no_improve = 0
                rng = np.random.default_rng(it * 31 + 17)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)

        if beta < beta_end:
            beta *= 1.003

    return best_f, best_obj

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    sweep = 0
    while time.monotonic() - t0 < time_limit and sweep < n_sweeps:
        sweep += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        conv = fast_convolve(f)
        # Focus on top peaks
        n_peaks = min(15, len(conv))
        top_peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]

        # For each peak, try to reduce it by moving mass
        for k in top_peaks:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo >= hi:
                continue

            # Find the pair (i, j) with i+j=k that maximizes f[i]*f[j]
            i_vals = np.arange(lo, hi + 1)
            j_vals = k - i_vals
            products = f[i_vals] * f[j_vals]
            max_idx = np.argmax(products)
            i_max = int(i_vals[max_idx])
            j_max = int(j_vals[max_idx])

            # Try moving small amount from i_max to a random position
            n_tries = min(5, max(1, int(remaining / 10)))
            for _ in range(n_tries):
                if time.monotonic() - t0 > time_limit:
                    break
                # Pick destination far from both i_max and j_max
                dist = np.abs(np.arange(n) - i_max) + np.abs(np.arange(n) - j_max)
                # Weight toward larger distances
                probs = dist ** 2
                probs = probs / probs.sum()
                dest = rng.choice(n, p=probs)
                if dest == i_max or dest == j_max:
                    continue

                # Determine amount to move
                frac = rng.uniform(0.02, 0.15)
                delta = f[i_max] * frac
                if delta < 1e-15:
                    continue

                f_trial = f.copy()
                f_trial[i_max] -= delta
                f_trial[dest] += delta
                f_trial = np.maximum(f_trial, 0.0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                conv_trial = fast_convolve(f_trial)
                if conv_trial.max() < best_obj - 1e-15:
                    best_obj = conv_trial.max()
                    best_f = f_trial.copy()
                    f = f_trial.copy()
                    break

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def spectral_perturb(f, rng, strength=0.03, mode='random'):
    """Perturb f in frequency domain with different modes."""
    n = len(f)
    fft_size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    mag = np.abs(F)
    phase = np.angle(F)

    if mode == 'random':
        mag_new = np.maximum(mag * (1.0 + rng.standard_normal(len(mag)) * strength), 0)
    elif mode == 'flatten':
        log_mag = np.log(mag + 1e-30)
        log_mean = np.log(mag.mean() + 1e-30)
        alpha = 1.0 - strength
        log_mag_new = log_mag * alpha + log_mean * (1.0 - alpha)
        mag_new = np.exp(log_mag_new)
        mag_new[0] = mag[0]
    elif mode == 'taper':
        # Apply a random sinusoidal taper
        k = np.arange(len(mag))
        freq = rng.uniform(1, len(mag) // 2)
        amp = rng.uniform(0.1, 0.4) * strength * 10
        mag_new = np.maximum(mag * (1.0 + amp * np.sin(2 * np.pi * freq * k / len(mag))), 0)
    else:
        mag_new = mag

    mag_new[0] = mag[0]  # Keep DC
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, fft_size)[:n]
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Spectral shaping initialization
    Phase 2: FW with delayed constraints (main workhorse)
    Phase 3: LSE subgradient refinement
    Phase 4: Coordinate descent polish
    Phase 5: Multi-length exploration
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        try_sequence(parent, best_tracker)

    n_parent = len(parent)

    # === Phase 1: Spectral shaping (quick, ~60s) ===
    print("\n=== Phase 1: Spectral shaping ===")
    t1_end = time.monotonic() + 60.0
    f_shaped, obj_shaped = spectral_shaping(parent, n_iter=30, flatten_alpha=0.2,
                                             rng=np.random.default_rng(seed + 1))
    try_sequence(f_shaped, best_tracker)

    # Also try different flatten strengths
    for alpha in [0.1, 0.4, 0.6]:
        if time.monotonic() > t1_end:
            break
        f_s, _ = spectral_shaping(parent, n_iter=20, flatten_alpha=alpha,
                                   rng=np.random.default_rng(seed + 100 + int(alpha * 10)))
        try_sequence(f_s, best_tracker)

    current_best = best_tracker[1] if best_tracker[1] is not None else parent

    # === Phase 2: FW with delayed constraints (main, ~500s) ===
    print(f"\n=== Phase 2: FW delayed constraints, n={len(current_best)} ===")
    t2_alloc = min(500, (deadline - time.monotonic()) * 0.35)
    if t2_alloc > 30:
        print(f"  FW n={len(current_best)}, t={t2_alloc:.0f}s")
        f_opt, obj = fw_delayed_constraints(current_best, time_limit=t2_alloc,
                                             top_k_start=15)
        try_sequence(f_opt, best_tracker)

    # Run FW again from the best found
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t2b_alloc = min(200, (deadline - time.monotonic()) * 0.15)
    if t2b_alloc > 20:
        print(f"  FW round 2 n={len(current_best)}, t={t2b_alloc:.0f}s")
        f_opt, obj = fw_delayed_constraints(current_best, time_limit=t2b_alloc,
                                             top_k_start=20)
        try_sequence(f_opt, best_tracker)

    # === Phase 3: LSE subgradient (refinement, ~300s) ===
    print(f"\n=== Phase 3: LSE subgradient ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t3_alloc = min(300, (deadline - time.monotonic()) * 0.25)
    if t3_alloc > 20:
        print(f"  Subgrad n={len(current_best)}, t={t3_alloc:.0f}s")
        f_opt, _ = lse_subgradient(current_best, time_limit=t3_alloc,
                                    beta_start=5.0, beta_end=2000.0, top_k=40)
        try_sequence(f_opt, best_tracker)

    # Second subgradient with higher beta
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t3b_alloc = min(150, (deadline - time.monotonic()) * 0.12)
    if t3b_alloc > 15:
        print(f"  Subgrad high-beta n={len(current_best)}, t={t3b_alloc:.0f}s")
        f_opt, _ = lse_subgradient(current_best, time_limit=t3b_alloc,
                                    beta_start=50.0, beta_end=10000.0, top_k=20)
        try_sequence(f_opt, best_tracker)

    # === Phase 4: Coordinate descent polish (~100s) ===
    print(f"\n=== Phase 4: Coordinate descent ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t4_alloc = min(100, (deadline - time.monotonic()) * 0.1)
    if t4_alloc > 10:
        print(f"  CoordDesc n={len(current_best)}, t={t4_alloc:.0f}s")
        f_opt, _ = coordinate_descent(current_best, time_limit=t4_alloc, n_sweeps=30)
        try_sequence(f_opt, best_tracker)

    # === Phase 5: Multi-length exploration + spectral diversity (~remaining) ===
    print(f"\n=== Phase 5: Multi-length + diversity ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)

    # Try a few different lengths
    lengths_to_try = sorted(set([
        n_cur,
        max(256, n_cur - 64),
        min(1500, n_cur + 64),
        max(256, int(n_cur * 0.8)),
        min(1200, int(n_cur * 1.2)),
    ]), key=lambda x: -abs(x - n_cur))

    for nc in lengths_to_try:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic() - 30
        if remaining < 30:
            break

        if nc == n_cur:
            continue  # Already optimized at this length

        f_interp = interpolate_to_length(current_best, nc)
        # Quick spectral shaping at new length
        f_interp, _ = spectral_shaping(f_interp, n_iter=10, flatten_alpha=0.2,
                                        rng=np.random.default_rng(seed + nc))
        try_sequence(f_interp, best_tracker)

        # Brief FW at new length
        t_alloc = min(80, remaining * 0.4)
        if t_alloc > 15:
            f_opt, _ = fw_delayed_constraints(f_interp, time_limit=t_alloc, top_k_start=10)
            try_sequence(f_opt, best_tracker)

    # Spectral diversity: perturb best and re-optimize briefly
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)

    for attempt in range(5):
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic() - 15
        if remaining < 20:
            break

        mode = ['flatten', 'random', 'taper', 'flatten', 'random'][attempt]
        strength = 0.02 + 0.01 * attempt
        f_pert = spectral_perturb(current_best, rng, strength=strength, mode=mode)
        try_sequence(f_pert, best_tracker)

        # Brief optimization from perturbed
        t_alloc = min(40, remaining * 0.5)
        if t_alloc > 10:
            f_opt, _ = fw_delayed_constraints(f_pert, time_limit=t_alloc, top_k_start=10)
            try_sequence(f_opt, best_tracker)

    # === Final polish ===
    print(f"\n=== Final polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)

    # One more subgradient pass with very high beta
    t_alloc = min(60, deadline - time.monotonic() - 10)
    if t_alloc > 10:
        f_opt, _ = lse_subgradient(current_best, time_limit=t_alloc,
                                    beta_start=200.0, beta_end=20000.0, top_k=10)
        try_sequence(f_opt, best_tracker)

    # Small random perturbations
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    if time.monotonic() < deadline - 5:
        for _ in range(8):
            if time.monotonic() > deadline - 3:
                break
            f_pert = project_simplex(current_best + rng.standard_normal(len(current_best)) * 0.001)
            try_sequence(f_pert, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using uniform fallback")
    fallback = np.ones(448) / 448.0
    return [float(x) for x in fallback]
