            exps = np.exp(scaled)
            p = exps / exps.sum()
            obj = np.dot(p, conv)
            grad_a = compute_conv_gradient(a_x, p)
            w_grad = a_x * (grad_a - np.dot(grad_a, a_x))
            return obj, w_grad
        
        try:
            result = minimize(fun_and_grad, current_w, method='L-BFGS-B',
                           jac=True, options={'maxiter': 100, 'gtol': 1e-9})
        except Exception as e:
            no_improve += 5
            T *= T_decay
            continue
        
        if result.status == 0 or (result.fun is not None and np.isfinite(result.fun)):
            current_w = result.x
            a_new = np.exp(current_w - current_w.max())
            a_new = a_new / a_new.sum()
            conv_max = fast_convolve(a_new).max()
            if conv_max < best_conv_max - 1e-15:
                best_conv_max = conv_max
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Adaptive temperature
        if no_improve <= 10:
            T *= T_decay
        elif no_improve <= 30:
            T *= T_decay * T_decay  # decrease faster
        else:
            T = max(T * T_decay, T_min)
        
        T = max(T, T_min)
        
        # Reset if stuck
        if no_improve > 50:
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            pert = rng.standard_normal(n) * 0.005
            current_w = np.log(np.maximum(best_a + pert, 1e-15))
            current_w -= current_w.mean()
            T = T_init * 0.5
            no_improve = 0
    
    return best_a, best_conv_max


def game_omd(a0, time_limit, T_start=0.005, lr_start=0.05, inner_steps=3):
    """Game-theoretic Online Mirror Descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_conv_max = fast_convolve(a).max()
    
    t0 = time.monotonic()
    T = T_start
    lr = lr_start
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit - 0.5:
        # Dual update: p = softmax(conv / T)
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Primal update: projected gradient on min a^T M(p) a
        for _ in range(inner_steps):
            grad = compute_conv_gradient(a, p)
            candidate = project_simplex(a - lr * grad)
            new_obj = np.dot(p, fast_convolve(candidate))
            old_obj = np.dot(p, fast_convolve(a))
            if new_obj < old_obj - 1e-15:
                a = candidate
            else:
                # Try smaller step
                candidate2 = project_simplex(a - lr * 0.5 * grad)
                new_obj2 = np.dot(p, fast_convolve(candidate2))
                if new_obj2 < old_obj - 1e-15:
                    a = candidate2
                else:
                    break
        
        conv_max = fast_convolve(a).max()
        if conv_max < best_conv_max - 1e-14:
            best_conv_max = conv_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt
        if no_improve > 20:
            T *= 1.05
            lr *= 0.97
        if no_improve > 80:
            T *= 0.7
            lr = lr_start * 0.6
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            T = T_start
            lr = lr_start
            no_improve = 0
    
    return best_a, best_conv_max


def local_search(a0, time_limit, seed=42):
    """Local search with various move types."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_conv_max = fast_convolve(a).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit - 0.2:
        improved = False
        
        # Try different perturbation scales
        for scale in [0.0002, 0.0005, 0.001, 0.003, 0.008]:
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(a + pert)
            c = fast_convolve(candidate)
            if c.max() < best_conv_max - 1e-15:
                a = candidate
                best_conv_max = c.max()
                best_a = a.copy()
                improved = True
                break
        
        # Coordinate moves
        if not improved:
            for _ in range(3):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    candidate = a.copy()
                    amount = candidate[i] * frac
                    if amount < 1e-15:
                        continue
                    candidate[i] -= amount
                    candidate[j] += amount
                    candidate = np.maximum(candidate, 0)
                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_conv_max - 1e-15:
                        a = candidate
                        best_conv_max = c.max()
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        # Peak-targeted move: reduce mass at positions contributing to the peak
        if not improved:
            conv = fast_convolve(a)
            peak_k = int(np.argmax(conv))
            # Positions contributing to peak: j and peak_k-j for j in valid range
            j_start = max(0, peak_k - n + 1)
            j_end = min(peak_k, n - 1)
            if j_start <= j_end:
                # Find the pair (j, peak_k-j) with maximum product
                contributions = a[j_start:j_end+1] * a[peak_k - j_start:peak_k - j_end+1][::-1]
                max_idx = int(np.argmax(contributions))
                peak_j = j_start + max_idx
                peak_j2 = peak_k - peak_j
                # Move mass from the larger of the two to a random other position
                src = peak_j if a[peak_j] >= a[peak_j2] else peak_j2
                dst = int(rng.integers(0, n))
                if dst != src:
                    for frac in [0.05, 0.1, 0.2, 0.3]:
                        candidate = a.copy()
                        amount = candidate[src] * frac
                        candidate[src] -= amount
                        candidate[dst] += amount
                        candidate = np.maximum(candidate, 0)
                        ss = candidate.sum()
                        if ss < 1e-10:
                            continue
                        candidate = candidate / ss
                        c = fast_convolve(candidate)
                        if c.max() < best_conv_max - 1e-15:
                            a = candidate
                            best_conv_max = c.max()
                            best_a = a.copy()
                            improved = True
                            break
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        
        if no_improve > 200:
            break
    
    return best_a, best_conv_max


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    from scipy.optimize import linprog
    from scipy.sparse import lil_matrix
    
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    # Build LP constraint structure
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    row_idx = np.array(rows)
    col_idx = np.array(cols)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit - 0.5:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        # Solve LP oracle
        c = -np.ones(n)
        vals = np.empty(len(row_idx))
        for idx in range(len(row_idx)):
            k = row_idx[idx]
            i = col_idx[idx]
            vals[idx] = f[k - i]
        
        A_mat = lil_matrix((2 * n - 1, n))
        A_mat[row_idx, col_idx] = vals
        A_mat = A_mat.tocsr()
        b = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n
        
        g = None
        try:
            res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(1, remaining - 5)})
            if res.success and res.x is not None:
                g = np.maximum(res.x, 0.0)
                s_g = g.sum()
                if s_g > 1e-15:
                    g = g / s_g
                else:
                    g = None
        except Exception:
            g = None
        
        if g is None:
            m_exp = min(m_exp * 1.3, 8.0)
            if m_exp >= 8.0:
                rng = np.random.default_rng(it * 31 + 7)
                g = project_simplex(f + rng.standard_normal(n) * 0.005)
                m_exp = 1.0
            else:
                continue
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 7919 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 8.0)
    
    return best_f, best_obj
