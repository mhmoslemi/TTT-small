def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def mirror_project(v, a):
    """KL projection: argmax <v, g> - KL(g||a) s.t. sum(g)=1, g>=0.
    This gives g[i] = a[i] * exp(v[i]) / Z."""
    # v is the "dual variable" (negative gradient)
    # g[i] = a[i] * exp(v[i]) / sum_j a[j] * exp(v[j])
    # Use log-sum-exp for numerical stability
    log_a = np.log(np.maximum(a, 1e-300))
    log_weights = log_a + v
    log_weights -= log_weights.max()
    weights = np.exp(log_weights)
    Z = weights.sum()
    if Z < 1e-300:
        return a.copy()
    return weights / Z

def build_lp_matrix(f, constraint_indices):
    """Build sparse constraint matrix for LP."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([idx] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    if not rows:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(len(constraint_indices), n))

def solve_lp_direction(f, M, peak_indices, time_limit=15):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in peak_indices, g >= 0."""
    n = len(f)
    A = build_lp_matrix(f, peak_indices)
    if A.shape[0] == 0:
        return None
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None

def compute_gradient(f, active_peaks, weights):
    """Compute weighted sum of gradients of (f*f)[k] for k in active_peaks."""
    n = len(f)
    grad = np.zeros(n)
    for idx, k in enumerate(active_peaks):
        k = int(k)
        w = weights[idx]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += w * 2.0 * f[k - i_range]
    return grad

def lse_weights(conv, beta):
    """LSE softmax weights."""
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_bc / total

def get_active_peaks(conv, n, method='top_k', k=40, threshold=0.01):
    """Get active peak indices based on method."""
    if method == 'top_k':

    elif method == 'threshold':

        return indices
    else:
        k_actual = min(k, len(conv))
        return np.argpartition(conv, -k_actual)[-k_actual:]

def exact_line_search(f, g, n_evals=30):
    """Exact line search: minimize max_k ((1-t)*f + t*g)^T Q_k ((1-t)*f + t*g) for t in [0,1]."""
    def obj(t):
        ft = (1.0 - t) * f + t * g
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            return 1e10
        ft /= s

        return ft, result.fun

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-strategy initialization (parent interpolation, structured candidates, spectral)
2. Frank-Wolfe with violation-based LP constraint selection + exact line search
3. Mirror descent (KL projection) with LSE-smoothed subgradients
4. Peak-shaving coordinate refinement
5. Adaptive multi-length exploration and ensemble-based final polish
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def mirror_project(a, log_update):
    """KL projection: g[i] = a[i] * exp(log_update[i]) / Z."""
    log_weights = np.log(np.maximum(a, 1e-300)) + log_update
    mx = log_weights.max()
    weights = np.exp(log_weights - mx)
    Z = weights.sum()
    if Z < 1e-300:
        return a.copy()
    return weights / Z


def build_lp_matrix(f, constraint_indices):
    """Build sparse constraint matrix for LP."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([idx] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    if not rows:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(len(constraint_indices), n))


def solve_lp_direction(f, M, peak_indices, time_limit=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in peak_indices, g >= 0."""
    n = len(f)
    A = build_lp_matrix(f, peak_indices)
    if A.shape[0] == 0:
        return None
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None


def compute_subgradient(f, active_peaks, weights):
    """Compute weighted sum of gradients of (f*f)[k] for k in active_peaks."""
    n = len(f)
    grad = np.zeros(n)
    for idx, k in enumerate(active_peaks):
        k = int(k)
        w = weights[idx]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += w * 2.0 * f[k - i_range]
    return grad


def lse_weights(conv, beta):
    """LSE softmax weights on the convolution peaks."""
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_bc / total


def lse_objective(conv, beta):
    """Smooth approximation of max: (1/beta)*log(sum(exp(beta*conv)))."""
    bc = beta * conv
    mx = bc.max()
    return (mx + np.log(np.exp(bc - mx).sum())) / beta


def get_violation_peaks(conv, n, k_max=60, frac=0.02):
    """Get peaks within 'frac' of the maximum, capped at k_max."""
    peak_val = conv.max()
    threshold = peak_val * (1.0 - frac)
    indices = np.where(conv >= threshold)[0]
    if len(indices) > k_max:
        k_actual = min(k_max, len(conv))
        indices = np.argpartition(conv, -k_actual)[-k_actual:]
    center = n - 1
    if center not in indices:
        indices = np.append(indices, center)
    return indices


def exact_line_search(f, g):
    """Minimize max_k ((1-t)f + tg)^T Q_k ((1-t)f + tg) for t in [0,1]."""
    def obj(t):
        ft = (1.0 - t) * f + t * g
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            return 1e10
        ft /= s
        return fast_convolve(ft).max()

    try:
        result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                 options={'xatol': 5e-7, 'maxiter': 40})
        t_opt = float(result.x)
        ft = (1.0 - t_opt) * f + t_opt * g
        ft = np.maximum(ft, 0.0)
        ft /= ft.sum()
        return ft, float(result.fun)
    except Exception:
        return f, obj(0.0)


def fw_optimize(f0, time_limit, k_constraints=50, verbose=False):
    """Frank-Wolfe with LP oracle, violation-based constraints, exact line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Adaptive constraint selection
        frac = 0.01 + 0.015 * min(1.0, no_improve / 40.0)
        k_cap = min(k_constraints + 10 * (no_improve // 20), n)
        peaks = get_violation_peaks(conv, n, k_max=k_cap, frac=frac)

        lp_time = min(12, max(3, remaining * 0.05))
        g = solve_lp_direction(f, M, peaks, time_limit=lp_time)

        if g is None:
            # Subgradient fallback
            k_top = min(40, 2 * n - 1)
            peak_idx = np.argpartition(conv, -k_top)[-k_top:]
            w = lse_weights(conv[peak_idx], 50.0)
            grad = compute_subgradient(f, peak_idx, w)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-15:
                g = project_simplex(f - 0.05 * grad / gnorm)
            else:
                g = f.copy()
            no_improve += 3
        else:
            f_new, val = exact_line_search(f, g)
            if val < best_obj - 1e-13:
                f = f_new
                M = val
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

        if no_improve > 35:
            rng = np.random.default_rng(it * 8273 + 17)
            F = np.fft.rfft(f)
            mag = np.abs(F)
            phase = np.angle(F)
            scale = 0.02 + 0.01 * min(1.0, no_improve / 70.0)
            mag_pert = np.maximum(mag * (1.0 + rng.standard_normal(len(mag)) * scale), 0)
            F_new = mag_pert * np.exp(1j * phase)
            f_new = np.fft.irfft(F_new, n)
            f_new = np.maximum(f_new, 0)
            s2 = f_new.sum()
            if s2 > 1e-15:
                f_new /= s2
                f = f_new
                conv = fast_convolve(f)
                M = conv.max()
                if M < best_obj:
                    best_obj = M
                    best_f = f.copy()
            no_improve = 0

    return best_f, best_obj


def mirror_optimize(f0, time_limit, beta_start=10.0, beta_end=5000.0, top_k=60):
    """Mirror descent (KL projection) with LSE-smoothed subgradients."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 1.5 / np.sqrt(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        conv = fast_convolve(f)
        progress = (time.monotonic() - t0) / time_limit
        beta = beta_start * (beta_end / beta_start) ** progress

        k_top = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        w = lse_weights(conv[peak_idx], beta)
        grad = compute_subgradient(f, peak_idx, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            no_improve += 100
            break

        # Mirror descent step with backtracking
        step = lr
        improved = False
        for _ in range(12):
            log_update = -step * grad / gnorm
            g = mirror_project(f, log_update)
            conv_new = fast_convolve(g)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-14:
                f = g
                best_obj = obj_new
                best_f = f.copy()
                improved = True
                no_improve = 0
                lr = min(lr * 1.01, 5.0 / np.sqrt(n))
                break
            step *= 0.5

        if not improved:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.85
            if no_improve > 60:
                f = best_f.copy()
                lr = max(1.5 / np.sqrt(n), 0.003)
                no_improve = 0
                rng = np.random.default_rng(it * 47 + 23)
                F = np.fft.rfft(f)
                mag = np.abs(F)
                phase = np.angle(F)
                mag_pert = np.maximum(mag * (1.0 + rng.standard_normal(len(mag)) * 0.015), 0)
                F_new = mag_pert * np.exp(1j * phase)
                f_new = np.fft.irfft(F_new, n)
                f_new = np.maximum(f_new, 0)
                s2 = f_new.sum()
                if s2 > 1e-15:
                    f = f_new / s2
                else:
                    f = best_f.copy()

    return best_f, best_obj


def peak_shaving(f0, n_rounds=200, time_limit=60.0):
    """Coordinate descent: reduce peaks by adjusting individual coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, 1e10
    f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        # Pick a random peak from the top 10
        k_top = min(10, len(conv))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        k = int(peaks[rng.integers(len(peaks))])

        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo >= hi:
            continue

        # Find the pair (i, k-i) with maximum product
        i_vals = np.arange(lo, hi + 1)
        products = f[i_vals] * f[k - i_vals]
        j = int(i_vals[np.argmax(products)])
        j_other = k - j
        if j_other < 0 or j_other >= n:
            continue

        # Try moving small amount from j to a random other position
        delta = f[j] * 0.05
        if delta < 1e-15:
            continue
        for _ in range(8):
            j_dest = int(rng.integers(n))
            if j_dest == j or j_dest == j_other:
                continue
            f_trial = f.copy()
            f_trial[j] -= delta
            f_trial[j_dest] += delta
            if f_trial[j] < 0:
                continue
            f_trial = np.maximum(f_trial, 0)
            f_trial /= f_trial.sum()
            conv_trial = fast_convolve(f_trial)
            if conv_trial.max() < best_obj - 1e-14:
                f = f_trial.copy()
                best_obj = conv_trial.max()
                best_f = f.copy()
                break

    return best_f, best_obj


def freq_perturb(f, rng, strength=0.03):
    """Perturb f in the frequency domain."""
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})", flush=True)
    return score


def generate_structured_candidates(n, rng):
    """Generate diverse structured initializations."""
    candidates = []
    x = np.arange(n)

    # Cosine taper
    c = np.cos(np.pi * x / (n - 1 + 1e-10)) ** 2
    c = np.maximum(c, 0)
    if c.sum() > 1e-15:
        candidates.append(c / c.sum())

    # Inverted cosine (peaked in middle)
    c2 = 1.0 - np.cos(np.pi * x / (n - 1 + 1e-10)) ** 2
    c2 = np.maximum(c2, 0)
    if c2.sum() > 1e-15:
        candidates.append(c2 / c2.sum())

    # Gaussian
    center = (n - 1) / 2.0
    sigma = n / 4.0
    g = np.exp(-((x - center) ** 2) / (2 * sigma ** 2))
    if g.sum() > 1e-15:
        candidates.append(g / g.sum())

    # Low-pass noise
    noise = rng.standard_normal(n)
    F = np.fft.rfft(noise)
    cutoff = n // 4
    F[cutoff:] = 0
    lp = np.fft.irfft(F, n)
    lp = np.maximum(lp, 0)
    if lp.sum() > 1e-15:
        candidates.append(lp / lp.sum())

    # Two-hump
    c3 = np.exp(-((x - n * 0.25) ** 2) / (n * 0.02)) + np.exp(-((x - n * 0.75) ** 2) / (n * 0.02))
    c3 = np.maximum(c3, 0)
    if c3.sum() > 1e-15:
        candidates.append(c3 / c3.sum())

    # Random Fourier (flat magnitude)
    theta = rng.uniform(0, 2 * np.pi, n // 2 + 1)
    mag = np.ones(n // 2 + 1)
    A = mag * np.exp(1j * theta)
    A[0] = 1.0  # DC = 1 for sum=1 normalization later
    if n % 2 == 0:
        A[-1] = float(A[-1].real)
    a_four = np.fft.irfft(A * n, n)
    a_four = np.maximum(a_four, 0)
    if a_four.sum() > 1e-15:
        candidates.append(a_four / a_four.sum())

    return candidates


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Multi-length, multi-init FW exploration
    Phase 2: Mirror descent refinement
    Phase 3: Alternating FW + Mirror + Peak-shaving
    Phase 4: Final high-precision polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s) - 30.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}", flush=True)
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        try_sequence(parent, best_tracker)

    n_parent = len(parent)

    # === Phase 1: Multi-length, multi-init exploration ===
    print("\n=== Phase 1: Multi-init exploration ===", flush=True)
    t1_end = time.monotonic() + total_budget * 0.30

    # Try several lengths
    n_options = sorted(set([n_parent,
                            n_parent - 32,
                            n_parent + 32,
                            max(256, n_parent - 64),
                            min(1200, n_parent + 64),
                            max(256, int(n_parent * 0.85)),
                            min(1200, int(n_parent * 1.15))]), reverse=True)
    # Deduplicate and limit
    seen = set()
    n_options = [x for x in n_options if not (x in seen or seen.add(x))]
    n_options = n_options[:5]

    results = []  # (score, sequence, length)
    for nc in n_options:
        t_rem = t1_end - time.monotonic()
        if t_rem < 30:
            break
        t_alloc_each = min(60, t_rem / max(1, len(n_options)))
        if t_alloc_each < 15:
            t_alloc_each = max(15, t_rem / max(1, len(n_options)))

        # Get initializations for this length
        if nc == n_parent and parent is not None:
            inits = [parent.copy()]
        else:
            inits = [interpolate_to_length(parent, nc)]

        # Add structured candidates
        struct = generate_structured_candidates(nc, rng)
        inits.extend(struct[:3])

        # Evaluate all inits quickly
        best_init = inits[0]
        best_init_score = float('inf')
        for ini in inits:
            score = try_sequence(ini, best_tracker)
            if np.isfinite(score) and score < best_init_score:
                best_init_score = score
                best_init = ini

        # Run FW on the best init
        t_fw = min(45, max(10, t_rem * 0.4 / len(n_options)))
        print(f"  FW n={nc}, t={t_fw:.0f}s", flush=True)
        f_opt, obj = fw_optimize(best_init, time_limit=t_fw, k_constraints=50)
        try_sequence(f_opt, best_tracker)
        results.append((obj, f_opt, nc))

    # === Phase 2: Mirror descent refinement ===
    print("\n=== Phase 2: Mirror descent refinement ===", flush=True)
    t2_end = t1_end + total_budget * 0.20
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)

    t_alloc = min(150, t2_end - time.monotonic())
    if t_alloc > 20:
        print(f"  MirrorDescent n={n_cur}, t={t_alloc:.0f}s", flush=True)
        f_opt, _ = mirror_optimize(current_best, time_limit=t_alloc,
                                   beta_start=10.0, beta_end=3000.0, top_k=60)
        try_sequence(f_opt, best_tracker)

    # Also try mirror on a different length if we have time
    current_best2 = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur2 = len(current_best2)
    if abs(n_cur2 - n_cur) > 0:
        t_alloc2 = min(80, t2_end - time.monotonic())
        if t_alloc2 > 15:
            n_alt = n_cur if n_cur != n_cur2 else (n_cur2 + 16)
            f_init_alt = interpolate_to_length(current_best2, n_alt)
            print(f"  MirrorDescent alt n={n_alt}, t={t_alloc2:.0f}s", flush=True)
            f_opt, _ = mirror_optimize(f_init_alt, time_limit=t_alloc2,
                                       beta_start=20.0, beta_end=5000.0, top_k=50)
            try_sequence(f_opt, best_tracker)

    # === Phase 3: Alternating optimization ===
    print("\n=== Phase 3: Alternating optimization ===", flush=True)
    t3_end = deadline - 180
    cycles = 0
    while time.monotonic() < t3_end and cycles < 20:
        cycles += 1
        current_best = best_tracker[1] if best_tracker[1] is not None else parent
        n_cur = len(current_best)
        remaining = t3_end - time.monotonic()
        if remaining < 35:
            break

        phase = cycles % 5
        if phase == 0:
            # FW with more constraints
            t_alloc = min(80, remaining * 0.35)
            if t_alloc > 12:
                print(f"  Cycle {cycles}: FW(k=70) n={n_cur}, t={t_alloc:.0f}s", flush=True)
                f_opt, _ = fw_optimize(current_best, time_limit=t_alloc, k_constraints=70)
                try_sequence(f_opt, best_tracker)

        elif phase == 1:
            # Mirror descent with high beta
            t_alloc = min(90, remaining * 0.35)
            if t_alloc > 12:
                beta_s = 50.0 + 60.0 * cycles
                print(f"  Cycle {cycles}: Mirror(beta={beta_s:.0f}+) n={n_cur}, t={t_alloc:.0f}s", flush=True)
                f_opt, _ = mirror_optimize(current_best, time_limit=t_alloc,
                                           beta_start=beta_s, beta_end=15000.0, top_k=40)
                try_sequence(f_opt, best_tracker)

        elif phase == 2:
            # Peak shaving
            t_alloc = min(50, remaining * 0.2)
            if t_alloc > 10:
                print(f"  Cycle {cycles}: PeakShave n={n_cur}, t={t_alloc:.0f}s", flush=True)
                f_opt, _ = peak_shaving(current_best, n_rounds=300, time_limit=t_alloc)
                try_sequence(f_opt, best_tracker)

        elif phase == 3:
            # Freq perturb + brief FW
            t_alloc = min(70, remaining * 0.3)
            if t_alloc > 12:
                print(f"  Cycle {cycles}: FreqPerturb+FW n={n_cur}, t={t_alloc:.0f}s", flush=True)
                for attempt in range(3):
                    t_rem = t3_end - time.monotonic() - 12
                    if t_rem < 15:
                        break
                    strength = 0.01 + 0.008 * attempt
                    f_pert = freq_perturb(current_best, rng, strength=strength)
                    t_sub = max(8, (t_rem - 8) / 2.0)
                    f_opt, _ = fw_optimize(f_pert, time_limit=t_sub, k_constraints=40)
                    try_sequence(f_opt, best_tracker)

        else:
            # FW with relaxed constraints (wider search)
            t_alloc = min(60, remaining * 0.3)
            if t_alloc > 12:
                print(f"  Cycle {cycles}: FW( relaxed ) n={n_cur}, t={t_alloc:.0f}s", flush=True)
                f_opt, _ = fw_optimize(current_best, time_limit=t_alloc, k_constraints=100)
                try_sequence(f_opt, best_tracker)

    # === Phase 4: Final high-precision polish ===
    print("\n=== Phase 4: Final polish ===", flush=True)
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)

    # Very high beta mirror descent
    t_alloc = min(80, deadline - time.monotonic() - 80)
    if t_alloc > 15:
        print(f"  High-beta mirror n={n_cur}, t={t_alloc:.0f}s", flush=True)
        f_opt, _ = mirror_optimize(current_best, time_limit=t_alloc,
                                   beta_start=500.0, beta_end=50000.0, top_k=25)
        try_sequence(f_opt, best_tracker)

    # Final FW with many constraints
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t_alloc = min(60, deadline - time.monotonic() - 50)
    if t_alloc > 12:
        print(f"  Final FW(k=100) n={len(current_best)}, t={t_alloc:.0f}s", flush=True)
        f_opt, _ = fw_optimize(current_best, time_limit=t_alloc, k_constraints=100)
        try_sequence(f_opt, best_tracker)

    # Peak shaving final pass
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t_alloc = min(40, deadline - time.monotonic() - 25)
    if t_alloc > 10:
        print(f"  Final PeakShave n={len(current_best)}, t={t_alloc:.0f}s", flush=True)
        f_opt, _ = peak_shaving(current_best, n_rounds=500, time_limit=t_alloc)
        try_sequence(f_opt, best_tracker)

    # Tiny perturbations of the best
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    n_cur = len(current_best)
    t_rem = deadline - time.monotonic()
    if t_rem > 15:
        print(f"  Final micro-perturbations, t={t_rem:.0f}s", flush=True)
        for i in range(20):
            if time.monotonic() > deadline - 5:
                break
            # Time-domain micro perturbation
            f_pert = project_simplex(current_best + rng.standard_normal(n_cur) * 0.001)
            try_sequence(f_pert, best_tracker)
            # Freq micro perturbation
            f_pert2 = freq_perturb(current_best, rng, strength=0.003)
            try_sequence(f_pert2, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}", flush=True)
            return [float(x) for x in final_seq]

    print("WARNING: Using uniform fallback", flush=True)
    fallback = np.ones(448) / 448.0
    return [float(x) for x in fallback]
