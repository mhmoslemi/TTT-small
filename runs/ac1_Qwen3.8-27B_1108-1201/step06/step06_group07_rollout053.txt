Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- ADMM with QP subproblems (CVXPY)
- 1D coordinate descent polish
- Multi-n exploration
- Randomized diversification

import numpy as np
import time
import cvxpy as cp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def admm_optimize(a_init, time_limit, rho=10.0, verbose=True):
    """ADMM for minimizing max(a*a) subject to a>=0, sum(a)=1."""
    n = len(a_init)
    a = a_init.copy()
    b = a_init.copy()
    u = np.zeros(n)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # a-update: QP
        # min t + (rho/2)*||a - b + u||^2
        # s.t. (a*b)[k] <= t for all k, sum(a)=1, a>=0
        a_target = b - u
        
        a_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [a_var >= 0, cp.sum(a_var) == 1]
        
        # Convolution constraints: (a*b)[k] <= t
        for k in range(2 * n - 1):
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # (a*b)[k] = sum_{i=i_start}^{i_end} a[i] * b[k-i]
            conv_k = a_var[i_start:i_end+1] @ b[k-i_end:k-i_start+1][::-1]
            constraints.append(conv_k <= t_var)
        
        obj = cp.Minimize(t_var + (rho / 2.0) * cp.sum_squares(a_var - a_target))
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, warm_start=True)
            if a_var.value is not None:
                a = np.maximum(a_var.value, 0)
                a = a / a.sum()
        except Exception:
            pass
        
        # b-update: QP
        # min t + (rho/2)*||a - b + u||^2
        # s.t. (a*b)[k] <= t for all k, b>=0
        b_target = a + u
        
        b_var = cp.Variable(n)
        t_var2 = cp.Variable()
        constraints2 = [b_var >= 0]
        
        for k in range(2 * n - 1):
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            conv_k = a[i_start:i_end+1] @ b_var[k-i_end:k-i_start+1][::-1]
            # Wait, this is (a * b)[k] = sum_i a[i] * b[k-i]
            # But we need to express it in terms of b_var
            # (a * b)[k] = sum_{i} a[i] * b[k-i]
            # For b_var: the coefficient of b_var[j] in (a*b)[k] is a[k-j]
            # So (a*b)[k] = sum_j b_var[j] * a[k-j]
            # j ranges over max(0, k-n+1) to min(n-1, k)
            coeffs = a[k-i_end:k-i_start+1][::-1]  # a[k-j] for j = i_start..i_end
            conv_k = b_var[i_start:i_end+1] @ coeffs
            constraints2.append(conv_k <= t_var2)
        
        obj2 = cp.Minimize(t_var2 + (rho / 2.0) * cp.sum_squares(b_var - b_target))
        prob2 = cp.Problem(obj2, constraints2)
        
        try:
            prob2.solve(solver=cp.ECOS, warm_start=True)
            if b_var.value is not None:
                b = np.maximum(b_var.value, 0)
        except Exception:
            pass
        
        # u-update
        u = u + a - b
        
        # Check convergence
        conv = fast_convolve(a)
        obj_val = conv.max()
        if obj_val < best_obj:
            best_obj = obj_val
            best_a = a.copy()
        
        if verbose and it % 10 == 0:
            print(f"  ADMM it={it}, obj={best_obj:.10f}, ||a-b||={np.linalg.norm(a-b):.6f}")
    
    return best_a, best_obj

import numpy as np

from scipy.optimize import minimize_scalar

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def admm_step_a(a_prev, b, u, rho, n):
    """Solve the a-subproblem: min t + (rho/2)||a - (b-u)||^2 s.t. (a*b)[k]<=t, 1^T a=1, a>=0."""
    target = b - u
    a_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [a_var >= 0, cp.sum(a_var) == 1]
    
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # (a*b)[k] = sum_{i=i_start}^{i_end} a[i] * b[k-i]
        coeffs = b[k - i_start : k - i_end + 1][::-1]
        constraints.append(a_var[i_start:i_end+1] @ coeffs <= t_var)
    
    obj = cp.Minimize(t_var + (rho / 2.0) * cp.sum_squares(a_var - target))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.SCS, warm_start=True, verbose=False)
        if a_var.value is not None:
            return np.maximum(a_var.value, 0.0)
    except Exception:
        pass
    return a_prev
