def solve_inner_qp(W_flat, n):
    """Solve min a^T W a s.t. sum(a)=1, a>=0. W is Hankel with W_flat of length 2n-1."""
    # Build P = 2W (since cvxopt minimizes 0.5*x^T P x)
    P = cvxopt.matrix(np.array([[W_flat[i+j] for j in range(n)] for i in range(n)]) * 2)
    q = cvxopt.matrix(np.zeros(n))
    A = cvxopt.matrix(np.ones((1, n)))
    b = cvxopt.matrix(np.ones(1))
    G = cvxopt.matrix(-np.eye(n))
    h = cvxopt.matrix(np.zeros(n))
    
    sol = cvxopt.solvers.qp(P, q, G, h, A, b)
    if sol['status'] == 'optimal':
        return np.array(sol['x']).flatten()
    else:
        return np.ones(n) / n

            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_curr + (t - 1.0) / t_new * (x_curr - x_prev)
        t = t_new

        grad = 2.0 * hankel_mv(w, y)
        x_next = project_simplex(y - alpha * grad)

        obj = float(x_next @ hankel_mv(w, x_next))
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_x = x_next.copy()
            last_improve = it

        x_prev = x_curr
        x_curr = x_next

        if it - last_improve > 20000 and it > 5000:
            break

    return best_x, best_obj


def solve_qp_active(w, n, a_init, max_iters=300, time_limit=5.0):
    """Solve min a^T W a s.t. sum(a)=1, a>=0 using active set with warm start."""
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    active = set(int(i) for i in range(n) if a[i] > 1e-10)
    if not active:
        active = set(range(n))

    t0 = time.monotonic()
    Wa = hankel_mv(w, a)  # Cache

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        act = sorted(active)
        k = len(act)

        if k == 0:
            return np.ones(n) / n
        if k == 1:
            result = np.zeros(n)
            result[act[0]] = 1.0
            return result

        idx_arr = np.array(act)
        W_AA = np.array([[w[act[i] + act[j]] for j in range(k)] for i in range(k)])
        W_AA += 1e-14 * np.eye(k)

        try:
            W_inv_1 = np.linalg.solve(W_AA, np.ones(k))
        except np.linalg.LinAlgError:
            W_inv_1 = np.linalg.lstsq(W_AA, np.ones(k), rcond=None)[0]

        ss = W_inv_1.sum()
        if abs(ss) < 1e-14:
            a_S = np.ones(k) / k
        else:
            mu = 1.0 / ss
            a_S = mu * W_inv_1

        neg = a_S < -1e-12
        if neg.any():
            min_pos = int(np.argmin(a_S))
            active.discard(act[min_pos])
            continue

        a_full = np.zeros(n)
        a_full[act] = np.maximum(a_S, 0.0)
        sf = a_full.sum()
        if sf < 1e-15:
            a_full = np.ones(n) / n
        else:
            a_full = a_full / sf

        grad = 2.0 * hankel_mv(w, a_full)
        nu = float(np.mean(grad[act]))

        inactive = set(range(n)) - active
        if inactive:
            min_viol = 0.0
            min_j = -1
            for j in inactive:
                v = grad[j] - nu
                if v < min_viol:
                    min_viol = v
                    min_j = j
            if min_viol < -1e-10:
                active.add(min_j)
                continue

        return a_full

    a_full = np.zeros(n)
    act = sorted(active)
    if act:
        a_full[act] = np.maximum(a_init[act], 0)
        sf = a_full.sum()
        if sf > 1e-15:
            a_full /= sf
        else:
            a_full = np.ones(n) / n
    else:
        a_full = np.ones(n) / n
    return a_full


def primal_dual_fista(a0, time_limit, T_init=0.01, use_topk=False, topk=30,
                      blend=0.5, inner_time=2.0, verbose=False):
    """Primal-dual with FISTA inner solver."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    dual_lr = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < inner_time + 0.5:
            break

        conv = fast_convolve(a)
        cmax = float(conv.max())

        if use_topk:
            k_use = min(topk, len(conv))
            peaks = np.argpartition(conv, -k_use)[-k_use:]
            w = np.zeros(len(conv))
            w[peaks] = 1.0 / k_use
        else:
            scaled = conv / T
            scaled -= scaled.max()
            w = np.exp(scaled)
            ws = w.sum()
            if ws < 1e-30:
                w = np.ones(len(conv)) / len(conv)
            else:
                w = w / ws

        inner_tl = min(inner_time, remaining * 0.4)
        a_qp, _ = solve_qp_fista(w, n, a, max_iters=50000, time_limit=inner_tl)

        alpha = blend
        a_new = (1.0 - alpha) * a + alpha * a_qp
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a_qp
        else:
            a_new = a_new / sn

        obj = float(fast_convolve(a_new).max())

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            dual_lr = min(dual_lr * 1.05, 3.0)
        else:
            no_improve += 1

        a = a_new

        if no_improve > 5:
            T *= 1.1
            dual_lr *= 0.9
        if no_improve > 15:
            T *= 0.8
        if no_improve > 40:
            rng = np.random.default_rng((it * 104729 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            T = T_init
            dual_lr = 1.0
            no_improve = 0

        if verbose and it % 20 == 0:
            print(f"    PD-FISTA it={it}, obj={best_obj:.10f}, T={T:.6f}, no_imp={no_improve}")

    return best_a, best_obj

def nesterov_polish(a0, time_limit, lr_init=0.005, momentum_init=0.9, verbose=False):
    """Nesterov-accelerated subgradient for min max_k conv(a,a) on simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = momentum_init
    y = a.copy()

    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.05:
            break

        y = a + mom * v
        y = project_simplex(y)

        conv_y = fast_convolve(y)
        k_peak = int(np.argmax(conv_y))
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * y[k_peak - idx]

        v_new = project_simplex(y - lr * grad) - y
        a = y + v_new
        a = project_simplex(a)
        v = v_new

        obj = float(fast_convolve(a).max())
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9999
        mom = min(0.995, mom + 0.00003)

        if no_improve > 200:
            lr *= 0.7
            mom = momentum_init
        if no_improve > 500:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = lr_init * 0.5
            mom = momentum_init
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    Nesterov it={it}, obj={best_obj:.10f}")

    return best_a, best_obj


def smoothed_grad_descent(a0, time_limit, T_init=0.001, lr_init=0.02, verbose=False):
    """Gradient descent on log-sum-exp smooth surrogate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = 2.0 * hankel_mv(p, a)

        best_candidate = a
        best_val = float(conv.max())
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 2.0]:
            candidate = project_simplex(a - trial_lr * grad)
            val = float(fast_convolve(candidate).max())
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            if no_improve == 0:
                lr = min(lr * 1.02, 0.5)
        else:
            no_improve += 1

        if no_improve > 10:
            lr *= 0.9
        if no_improve > 30:
            T *= 0.85
        if no_improve > 100:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            T = T_init
            lr = lr_init
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    SmoothedGrad it={it}, obj={best_obj:.10f}, T={T:.6f}")

    return best_a, best_obj


def trust_constr_polish(a0, time_limit, T=0.001, verbose=False):
    """Use scipy trust-constr on smooth surrogate for final polish."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    def obj_func(x):
        ax = np.maximum(x, 0)
        sx = ax.sum()
        if sx < 1e-15:
            return 1e10
        ax = ax / sx
        conv = fast_convolve(ax)
        return T * float(np.log(np.sum(np.exp(conv / T - conv.max()))))

    def grad_func(x):
        ax = np.maximum(x, 0)
        sx = ax.sum()
        if sx < 1e-15:
            return np.zeros(n)
        ax = ax / sx
        conv = fast_convolve(ax)
        scaled = conv / T - conv.max()
        p = np.exp(scaled)
        p = p / p.sum()
        return 2.0 * hankel_mv(p, ax) / sx

    t0 = time.monotonic()
    try:
        constraint = LinearConstraint(np.ones((1, n)), 1.0, 1.0)
        bounds = [(0.0, 1.0)] * n

        result = minimize(obj_func, a, jac=grad_func, method='trust-constr',
                          constraints=[constraint], bounds=bounds,
                          options={'maxiter': 1000, 'gtol': 1e-8,
                                   'xtol': 1e-12, 'verbose': 0})

        if result.x is not None:
            a_res = np.maximum(result.x, 0)
            sr = a_res.sum()
            if sr > 1e-15:
                a_res = a_res / sr
                obj = float(fast_convolve(a_res).max())
                if verbose:
                    print(f"    TrustConstr: obj={obj:.10f}, success={result.success}")
                return a_res, obj
    except Exception as e:
        if verbose:
            print(f"    TrustConstr failed: {e}")

    return a0, float(fast_convolve(a0).max())


def resample_sequence(parent, n_target):
    """Resample a sequence to a different length."""
    n_src = len(parent)
    if n_src == n_target:
        return parent.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_tgt, x_src, parent)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def generate_diverse_inits(n, parent, rng):
    """Generate several diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    x = np.arange(n, dtype=np.float64)
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge / edge.sum())
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())
    sym = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    sym = np.maximum(sym, 0.01)
    inits.append(sym / sym.sum())
    if parent is not None:
        p_res = resample_sequence(parent, n)
        inits.append(p_res)
    # Perturbed parent
    if parent is not None:
        p_res2 = resample_sequence(parent, n)
        pert = p_res2 + rng.standard_normal(n) * 0.01
        pert = np.maximum(pert, 0)
        if pert.sum() > 1e-10:
            inits.append(pert / pert.sum())
    return inits

    best_seq = None

            return
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)

                                              T_init=0.001, lr_init=0.01)
            try_seq(a_opt, "smooth_T0.001")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # === Phase 5: Trust-constr final polish (~150s) ===
    print("\n=== Phase 5: Trust-Constr Polish ===")
    t5 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 30
        if remaining > 30:
            t_alloc = min(remaining, 100)
            print(f"  Trust-constr T=0.001, t={t_alloc:.0f}s")
            a_opt, obj = trust_constr_polish(best_seq, time_limit=t_alloc, T=0.001, verbose=True)
            try_seq(a_opt, "trust_constr")

            remaining = deadline - time.monotonic() - 20
            if remaining > 20:
                t_alloc = min(remaining, 60)
                print(f"  Trust-constr T=0.0005, t={t_alloc:.0f}s")
                a_opt2, obj2 = trust_constr_polish(a_opt, time_limit=t_alloc, T=0.0005, verbose=True)
                try_seq(a_opt2, "trust_constr_fine")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # === Phase 6: Final micro-perturbations (~100s) ===
    print("\n=== Phase 6: Final Polish ===")
    t6 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Tiny random perturbations + Nesterov
        for i in range(8):
            remaining = deadline - time.monotonic() - 5
            if remaining < 8:
                break
            noise = 0.0003 + 0.0002 * i
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise)
            t_alloc = min(remaining, 12)
            a_opt, _ = nesterov_polish(f_pert, time_limit=t_alloc, lr_init=0.001)
            try_seq(a_opt, f"final_perturb_{i}")

        # One final smoothed gradient pass with very low T
        remaining = deadline - time.monotonic() - 10
        if remaining > 15:
            t_alloc = min(remaining, 30)
            a_opt, _ = smoothed_grad_descent(best_seq, time_limit=t_alloc,
                                              T_init=0.0001, lr_init=0.005)
            try_seq(a_opt, "final_smooth")

        # Final Nesterov
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            t_alloc = min(remaining, 20)
            a_opt, _ = nesterov_polish(best_seq, time_limit=t_alloc, lr_init=0.0005)
            try_seq(a_opt, "final_nesterov")

    print(f"  Phase 6 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t6:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
