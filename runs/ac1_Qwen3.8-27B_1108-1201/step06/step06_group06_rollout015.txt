```python
"""
FFT-based hybrid optimization for minimum peak autocorrelation of
non-negative step functions. Uses log-sum-exp smoothing with O(n log n)
FFT gradients, peak equalization via active-set methods, and scipy BFGS
for high-precision fine-tuning. Explores multiple sequence lengths and
leverages circular autocorrelation structure for fast gradient computation.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def circular_autocorr(a):
    """Circular autocorrelation via FFT. Returns real array of length n."""
    n = len(a)
    A = np.fft.fft(a)
    return np.real(np.fft.ifft(A * np.conj(A)))


def fft_grad_lse(a, T):
    """
    Compute gradient of log(sum exp(R[k]/T)) where R = circular autocorrelation.
    Returns gradient of length n.
    """
    n = len(a)
    R = circular_autocorr(a)
    R_max = R.max()
    log_weights = (R - R_max) / T
    p = np.exp(log_weights)
    p = p / p.sum()
    # grad[j] = (2/T) * sum_k p[k] * a[(k-j) % n]
    # = (2/T) * circular_cross_correlation(p, a)[j]
    # FFT: IFFT(FFT(p) * conj(FFT(a)))
    P = np.fft.fft(p)
    A = np.fft.fft(a)
    return (2.0 / T) * np.real(np.fft.ifft(P * np.conj(A)))


def fft_grad_sumsq(a):
    """
    Compute gradient of sum_k R[k]^2 where R = circular autocorrelation.
    grad[j] = 4 * (R cross-correlate a)[j]
    """
    n = len(a)
    R = circular_autocorr(a)
    # (R ⊛ a)[j] = sum_k R[k] * a[(k-j) % n]
    # FFT: IFFT(FFT(R) * conj(FFT(a)))
    Rf = np.fft.fft(R)
    A = np.fft.fft(a)
    return 4.0 * np.real(np.fft.ifft(Rf * np.conj(A)))


def lse_value(a, T):
    """Smoothed max approximation: T * log(sum(exp(R/T)))."""
    R = circular_autocorr(a)
    R_max = R.max()
    return R_max + T * np.log(np.sum(np.exp((R - R_max) / T)))
```
