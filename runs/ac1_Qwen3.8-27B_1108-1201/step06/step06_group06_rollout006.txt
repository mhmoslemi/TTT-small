"""
Hybrid FW-LP with active set management, deflation, and diverse restarts
for minimizing peak auto-convolution of non-negative step functions.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_pattern(n):
    """Precompute row/col indices for all lags in the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_active(f, M, active_lags, n, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_lags, g >= 0."""
    # Find indices in row_idx that correspond to active lags
    lag_set = set(active_lags)
    mask = np.zeros(len(row_idx), dtype=bool)
    for idx in range(len(row_idx)):
        if row_idx[idx] in lag_set:
            mask[idx] = True

    r_idx = row_idx[mask]
    c_idx = col_idx[mask]

    if len(r_idx) == 0:
        return np.ones(n) / n

    # Build values
    vals = np.empty(len(r_idx))
    for idx in range(len(r_idx)):
        k = r_idx[idx]
        i = c_idx[idx]
        vals[idx] = f[k - i]

    A_mat = lil_matrix((len(active_lags), n))
    # Map lag indices to row indices
    lag_to_row = {lag: r for r, lag in enumerate(active_lags)}
    for idx in range(len(r_idx)):
        r = lag_to_row[r_idx[idx]]
        A_mat[r, c_idx[idx]] = vals[idx]
    A_mat = csr_matrix(A_mat)

    c_obj = -np.ones(n)
    b_vec = np.full(len(active_lags), M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_vec, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 5})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None

    best_obj = M
    best_f = f.copy()

    no_improve = 0
    it = 0

                no_improve = 0

        # Line search

        best_val = M
        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:

            st = ft.sum()
            if st < 1e-15:

            if val < best_val:

            lo = max(0.0, best_alpha - 0.15)
            hi = min(1.0, best_alpha + 0.15)
            for _ in range(10):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.maximum((1 - m1) * f + m1 * g, 0)
                f1 = f1 / f1.sum()
                v1 = fast_convolve(f1).max()
                f2 = np.maximum((1 - m2) * f + m2 * g, 0)
                f2 = f2 / f2.sum()
                v2 = fast_convolve(f2).max()
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            mid = (lo + hi) / 2.0
            ft = np.maximum((1 - mid) * f + mid * g, 0)
            ft = ft / ft.sum()
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = mid

        if best_alpha > 1e-10:

            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

                best_obj = M
                best_f = f.copy()
                no_improve = 0
                pert_scale = max(pert_scale * 0.95, 0.001)

        else:
            no_improve += 1

def deflation_step(f, time_limit, iterations=500):
    """Targeted deflation of the peak lag."""
    f = np.maximum(f.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    lr = 0.01
    it = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit and it < iterations:
        it += 1
        conv = fast_convolve(f)
        k_star = conv.argmax()
        M = conv.max()

        if M < best_obj - 1e-12:
            best_obj = M
            best_f = f.copy()

        # Gradient of conv[k_star] w.r.t. a
        grad = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(k_star, n - 1)
        for i in range(i_start, i_end + 1):
            grad[i] += f[k_star - i]

        # Also include contribution from the symmetric side
        # (conv[k] = sum_i a[i]*a[k-i], so grad[j] = a[k-j] if valid)
        # This is already handled above

        # Try to reduce: move against gradient
        candidate = project_simplex(f - lr * grad)
        c = fast_convolve(candidate)
        if c.max() < M - 1e-15:
            f = candidate
        else:
            # Try smaller step
            candidate2 = project_simplex(f - lr * 0.1 * grad)
            c2 = fast_convolve(candidate2)
            if c2.max() < M - 1e-15:
                f = candidate2
                lr *= 0.5
            else:
                lr *= 0.7
                if lr < 1e-6:
                    # Perturb and restart
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                    lr = 0.01

    return best_f, best_obj


def smoothed_grad_opt(a0, time_limit, T_init=0.001, lr_init=0.02):
    """Smoothed gradient descent with softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient of sum_k p[k]*conv[a][k]
        # grad[j] = 2 * sum_k p[k] * a[k-j] (for valid k-j)
        grad = np.zeros(n)
        # Efficient computation using convolution
        # grad[j] = 2 * sum_k p[k] * a[k-j] = 2 * (p * reverse(a))[j + n - 1]... 
        # Actually: grad[j] = 2 * sum_k p[k] * a[k-j] where 0 <= k-j < n
        # = 2 * sum_m p[m+j] * a[m] where m = k-j, 0 <= m < n
        # So grad[j] = 2 * dot(a, p[j:j+n])
        for j in range(n):
            grad[j] = 2.0 * np.dot(a, p[j:j + n])

        # Try multiple step sizes
        best_candidate = a
        best_val = conv.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate

        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            T *= 1.03
            lr *= 0.98
        if no_improve > 150:
            T *= 0.7
            lr = lr_init * 0.6
        if no_improve > 400:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def local_search(a0, time_limit, seed=42):
    """Local search with coordinate moves and perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        improved = False

        # Random perturbations
        for scale in [0.0002, 0.0005, 0.001, 0.003, 0.008]:
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(a + pert)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        # Coordinate transfer
        if not improved:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j and a[i] > 1e-10:
                for frac in [0.05, 0.1, 0.2, 0.4]:
                    amount = a[i] * frac
                    candidate = a.copy()
                    candidate[i] -= amount
                    candidate[j] += amount
                    candidate = project_simplex(candidate)
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break

        # Pair swap (transfer from high to low region)
        if not improved and it % 3 == 0:
            # Find positions where reducing would help
            conv = fast_convolve(a)
            k_star = conv.argmax()
            # Move mass from positions that contribute to k_star
            i_start = max(0, k_star - n + 1)
            i_end = min(k_star, n - 1)
            if i_end > i_start:
                # Pick a high contributor and move to a low one
                contribs = a[i_start:i_end + 1] * a[k_star - (i_end - i_start):k_star - i_start:-1] if i_end - i_start < n else a[i_start:i_end+1]
                # Simplified: just pick random i, j
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j and a[i] > 1e-8:
                    for frac in [0.02, 0.05, 0.1]:
                        amount = a[i] * frac
                        candidate = a.copy()
                        candidate[i] -= amount
                        candidate[j] += amount
                        candidate = np.maximum(candidate, 0)
                        candidate = candidate / candidate.sum()
                        c = fast_convolve(candidate)
                        if c.max() < best_obj - 1e-15:
                            a = candidate
                            best_obj = c.max()
                            best_a = a.copy()
                            improved = True
                            break

        if not improved:
            stuck += 1
        else:
            stuck = 0

        if stuck > 500:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            stuck = 0

    return best_a, best_obj


def evaluate_and_update(a, label, best_seq, best_score):
    """Evaluate sequence and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15 or len(a) == 0:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return a_norm.copy(), score
    return best_seq, best_score


def generate_diverse_init(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Linear ramp (asymmetric)
    ramp = x + 1.0
    inits.append(ramp / ramp.sum())

    # Inverse ramp
    inv_ramp = n - x
    inits.append(inv_ramp / inv_ramp.sum())

    # Exponential (right-heavy)
    for rate in [0.005, 0.01, 0.02, 0.05]:
        exp_r = np.exp(rate * x)
        inits.append(exp_r / exp_r.sum())
        exp_l = np.exp(rate * (n - 1 - x))
        inits.append(exp_l / exp_l.sum())

    # Two-hump asymmetric
    for center1, center2, w1, w2 in [(0.3, 0.7, 0.08, 0.12), (0.2, 0.6, 0.1, 0.1), (0.4, 0.8, 0.05, 0.15)]:
        g1 = np.exp(-0.5 * ((x - center1 * n) / w1 / n) ** 2)
        g2 = np.exp(-0.5 * ((x - center2 * n) / w2 / n) ** 2)
        inits.append((g1 + g2) / (g1 + g2).sum())

    # Sinusoidal modulation
    for freq in [1, 2, 3]:
        for phase in [0, np.pi / 2, np.pi]:
            mod = 1.0 + 0.3 * np.sin(2 * np.pi * freq * x / n + phase)
            mod = np.maximum(mod, 0.1)
            inits.append(mod / mod.sum())

    # Cosine window
    c = np.cos(np.pi * x / (n - 1)) ** 2
    inits.append(c / c.sum())

    # Power law
    pw = (x + 1) ** 0.8
    inits.append(pw / pw.sum())
    pw2 = (n - x + 1) ** 0.8
    inits.append(pw2 / pw2.sum())

    # Random
    for _ in range(5):
        r = rng.random(n)
        inits.append(r / r.sum())

    # Interpolate from parent if available
    if parent is not None and len(parent) != n:
        p_interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        p_interp = np.maximum(p_interp, 0)
        p_interp = p_interp / p_interp.sum()
        inits.append(p_interp)
        # Perturbed parent
        for sc in [0.01, 0.03, 0.05]:
            pert = project_simplex(p_interp + rng.standard_normal(n) * sc)
            inits.append(pert)

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry: multi-phase search using FW-LP with active set, deflation,
    smoothed gradient, and local search. Explores diverse initializations
    and multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Uniform init: score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Quick multi-n exploration ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    n_candidates = [192, 256, 320, 384, 448, 512, 576, 640, 768, 896, 1024, 1280, 1536, 2048]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
        n_candidates.sort()

    quick_results = {}
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 300:
            break

        if n_try == n_parent and parent is not None:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try

        # Quick FW with active set
        t_alloc = min(remaining * 0.03, 20)
        a_opt, obj = fw_with_active_set(start, time_limit=t_alloc, K_frac=0.3)
        best_seq, best_score = evaluate_and_update(a_opt, f"fw_n{n_try}", best_seq, best_score)
        quick_results[n_try] = (a_opt, best_score)

        # Quick smoothed grad
        remaining2 = deadline - time.monotonic()
        if remaining2 > 200:
            a_opt2, _ = smoothed_grad_opt(a_opt, time_limit=min(remaining2 * 0.015, 8))
            best_seq, best_score = evaluate_and_update(a_opt2, f"sg_n{n_try}", best_seq, best_score)
            quick_results[n_try] = (a_opt2, best_score)

    print(f"  Phase 1 done: best={best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW-LP on best n ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 150:
        t_alloc = min(remaining * 0.35, 500)
        print(f"  Deep FW-LP n={n_best}, t={t_alloc:.0f}s")
        a_opt, obj = fw_with_active_set(best_seq, time_limit=t_alloc, K_frac=0.2)
        best_seq, best_score = evaluate_and_update(a_opt, "deep_fw", best_seq, best_score)

    # Deflation on the result
    remaining = deadline - time.monotonic()
    if remaining > 80:
        t_alloc = min(remaining * 0.15, 150)
        print(f"  Deflation, t={t_alloc:.0f}s")
        a_opt2, _ = deflation_step(best_seq, time_limit=t_alloc)
        best_seq, best_score = evaluate_and_update(a_opt2, "deflation", best_seq, best_score)

    # More FW
    remaining = deadline - time.monotonic()
    if remaining > 80:
        t_alloc = min(remaining * 0.2, 200)
        print(f"  More FW, t={t_alloc:.0f}s")
        a_opt3, _ = fw_with_active_set(best_seq, time_limit=t_alloc, K_frac=0.15, pert_scale=0.002)
        best_seq, best_score = evaluate_and_update(a_opt3, "fw2", best_seq, best_score)

    print(f"  Phase 2 done: best={best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Diverse restarts ===
    print("\n=== Phase 3: Diverse Restarts ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    if remaining > 200:
        inits = generate_diverse_init(n_best, rng, parent)
        # Also try at different n
        for n_alt in [n_best // 2, n_best * 2]:
            if 128 <= n_alt <= 3072:
                inits_alt = generate_diverse_init(n_alt, rng, best_seq)
                inits.extend(inits_alt)

        # Shuffle and try top diverse ones
        rng.shuffle(inits)
        tried = 0
        for idx, init in enumerate(inits):
            remaining = deadline - time.monotonic()
            if remaining < 60 or tried >= 8:
                break
            init = np.maximum(init, 0)
            ss = init.sum()
            if ss < 1e-10:
                continue
            init = init / ss
            n_init = len(init)

            # Quick FW
            t_alloc = min(remaining * 0.08, 40)
            a_opt, _ = fw_with_active_set(init, time_limit=t_alloc, K_frac=0.25)
            best_seq, best_score = evaluate_and_update(a_opt, f"div_{idx}_n{n_init}", best_seq, best_score)
            tried += 1

            # Quick smooth
            remaining2 = deadline - time.monotonic()
            if remaining2 > 40:
                a_opt2, _ = smoothed_grad_opt(a_opt, time_limit=min(remaining2 * 0.04, 15))
                best_seq, best_score = evaluate_and_update(a_opt2, f"divsg_{idx}", best_seq, best_score)

    print(f"  Phase 3 done: best={best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Perturbed restarts of best ===
    print("\n=== Phase 4: Perturbed Restarts ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for scale in [0.002, 0.005, 0.01, 0.02, 0.04]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc = min(remaining * 0.12, 100)
        a_opt, _ = fw_with_active_set(f_pert, time_limit=t_alloc, K_frac=0.2, pert_scale=scale * 0.5)
        best_seq, best_score = evaluate_and_update(a_opt, f"pert_{scale}", best_seq, best_score)

    # Smoothed gradient with different T
    remaining = deadline - time.monotonic()
    if remaining > 60:
        for T_init in [0.0003, 0.001, 0.003]:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            t_alloc = min(remaining * 0.15, 80)
            a_opt, _ = smoothed_grad_opt(best_seq, time_limit=t_alloc, T_init=T_init, lr_init=0.01)
            best_seq, best_score = evaluate_and_update(a_opt, f"smooth_T{T_init}", best_seq, best_score)

    print(f"  Phase 4 done: best={best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final polish ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Deflation
    if remaining > 40:
        t_alloc = min(remaining * 0.2, 80)
        a_opt, _ = deflation_step(best_seq, time_limit=t_alloc, iterations=300)
        best_seq, best_score = evaluate_and_update(a_opt, "def_final", best_seq, best_score)

    # Very fine smoothed gradient
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_alloc = min(remaining * 0.15, 50)
        a_opt, _ = smoothed_grad_opt(best_seq, time_limit=t_alloc, T_init=0.0001, lr_init=0.003)
        best_seq, best_score = evaluate_and_update(a_opt, "fine_smooth", best_seq, best_score)

    # Local search
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_alloc = min(remaining * 0.2, 60)
        a_opt, _ = local_search(best_seq, time_limit=t_alloc, seed=42)
        best_seq, best_score = evaluate_and_update(a_opt, "local1", best_seq, best_score)

    # Another FW pass
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t_alloc = min(remaining * 0.2, 40)
        a_opt, _ = fw_with_active_set(best_seq, time_limit=t_alloc, K_frac=0.1, pert_scale=0.001)
        best_seq, best_score = evaluate_and_update(a_opt, "fw_final", best_seq, best_score)

    # Tiny perturbations + smooth
    remaining = deadline - time.monotonic()
    if remaining > 15:
        for i in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            noise = rng.standard_normal(n_best) * 0.0003
            f_p = project_simplex(best_seq + noise)
            t_alloc = min(remaining * 0.25, 8)
            a_opt, _ = smoothed_grad_opt(f_p, time_limit=t_alloc, T_init=0.0002, lr_init=0.002)
            best_seq, best_score = evaluate_and_update(a_opt, f"micro_{i}", best_seq, best_score)

    # Final local search
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_opt, _ = local_search(best_seq, time_limit=min(remaining * 0.4, 20), seed=99999)
        best_seq, best_score = evaluate_and_update(a_opt, "local2", best_seq, best_score)

    print(f"  Phase 5 done: best={best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Fallback and verification ===
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final check
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={total_time:.0f}s")
    return [float(x) for x in best_seq]
