Uses mirror descent (multiplicative weights) with adaptive subgradient,
top-K focus, diverse initializations, and fine local search.

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient(a, w):
    """Compute weighted subgradient of sum_k w[k]*b[k] where b = convolve(a,a).
    
    w has length 2n-1 (the number of convolution lags).
    Returns gradient with respect to a (length n).

    a_rev = a[::-1]

    c_full = np.convolve(w, a_rev)
    c = c_full[n-1:n-1+n]

    diag = np.zeros(n)
    for j in range(n):
        k = 2 * j
        if k < len(w):

    return 2 * c - diag

def subgradient_descent(a0, time_limit, lr_init=0.01):

            break

        b_max = b.max()

        w[k_star] = 1.0

Hybrid optimization for minimum peak auto-convolution step function.
Uses mirror descent (multiplicative weights) with subgradient, exact subgradient
descent, parametric initialization, and fine local search.
Key differences from previous: mirror descent updates, top-K adaptive focus,
parametric family exploration, and more aggressive restart strategies.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, w):
    """Compute weighted gradient of sum_k w[k]*b[k] where b = convolve(a,a).
    
    w has length 2n-1. Returns gradient vector of length n.
    grad[j] = 2*sum_m w[j+m]*a[m] - w[2j]*a[j] (diagonal correction)
    """
    n = len(a)
    a_rev = a[::-1]
    # c[j] = sum_{m=0}^{n-1} w[j+m] * a[m] = convolve(w, a_rev)[n-1 + j]
    c_full = np.convolve(w, a_rev)
    c = c_full[n - 1:n - 1 + n]
    # Diagonal correction
    diag = np.zeros(n)
    for j in range(n):
        k = 2 * j
        if k < len(w):
            diag[j] = w[k] * a[j]
    return 2.0 * c - diag

        b = fast_convolve(a)
        b_max = b.max()

        # Top-K weighted subgradient
        if K < len(b):
            top_idx = np.argsort(b)[-K:]
        else:
            top_idx = np.arange(len(b))
        top_b = b[top_idx]
        scaled = top_b / max(T, 1e-12)
        scaled -= scaled.max()
        w_top = np.exp(scaled)
        w_top /= w_top.sum()

        w = np.zeros(2 * n - 1)
        w[top_idx] = w_top

        grad = compute_subgradient(a, w)

        # Multiplicative weights update with backtracking
        improved = False
        for lr_factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
            lr_try = lr * lr_factor
            log_update = -lr_try * grad
            log_update -= log_update.max()
            update = np.exp(log_update)
            a_new = a * update
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            b_new = fast_convolve(a_new)
            if b_new.max() < b_max - 1e-15:
                a = a_new
                lr = lr_try
                improved = True
                if b_new.max() < best_conv - 1e-14:
                    best_conv = b_new.max()
                    best_a = a.copy()
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            lr *= 0.9

        # Adapt temperature
        if no_improve > 30:
            T = min(T * 1.05, T_init * 10)
        if no_improve > 150:
            # Restart
            rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0

        # Slowly decrease T
        if it % 300 == 0:
            T = max(T * 0.9, T_init * 0.05)

    return best_a, best_conv


def subgradient_opt(a0, time_limit, lr_init=0.008):
    """Exact subgradient descent (use only the argmax lag) with line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    lr = lr_init
    best_a = a.copy()
    best_conv = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.03:
            break

        b = fast_convolve(a)
        b_max = b.max()
        k_star = int(np.argmax(b))

        # Exact subgradient: only the argmax lag
        w = np.zeros(2 * n - 1)
        w[k_star] = 1.0
        grad = compute_subgradient(a, w)

        # Line search with multiple step sizes
        improved = False
        for lr_try in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            a_try = project_simplex(a - lr_try * grad)
            b_try = fast_convolve(a_try)
            if b_try.max() < b_max - 1e-15:
                a = a_try
                lr = max(lr_try, 1e-6)
                improved = True
                if b_try.max() < best_conv - 1e-14:
                    best_conv = b_try.max()
                    best_a = a.copy()
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            lr *= 0.7

        if no_improve > 50:
            rng = np.random.default_rng((it * 31 + 7) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            lr = lr_init
            no_improve = 0

    return best_a, best_conv


def topK_blend_opt(a0, time_limit, K=5, T_init=0.001, lr_init=0.015):
    """Subgradient descent with top-K blend (smoothed max approximation)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_conv = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.03:
            break

        b = fast_convolve(a)
        b_max = b.max()

        top_idx = np.argsort(b)[-K:]
        top_b = b[top_idx]
        scaled = top_b / max(T, 1e-12)
        scaled -= scaled.max()
        w_top = np.exp(scaled)
        w_top /= w_top.sum()

        w = np.zeros(2 * n - 1)
        w[top_idx] = w_top

        grad = compute_subgradient(a, w)

        improved = False
        for lr_try in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            a_try = project_simplex(a - lr_try * grad)
            b_try = fast_convolve(a_try)
            if b_try.max() < b_max - 1e-15:
                a = a_try
                lr = lr_try
                improved = True
                if b_try.max() < best_conv - 1e-14:
                    best_conv = b_try.max()
                    best_a = a.copy()
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            lr *= 0.85

        # Adapt T
        if no_improve > 20:
            T *= 1.03
        if no_improve > 100:
            T = T_init
            lr = lr_init
            no_improve = 0
            rng = np.random.default_rng((it * 104729 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)

        if it % 500 == 0:
            T = max(T * 0.95, T_init * 0.1)

    return best_a, best_conv


def local_search_fine(a0, time_limit, seed=42):
    """Fine local search with diverse move types."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break

        improved = False

        # Move 1: Gaussian perturbation at multiple scales
        for scale in [1e-5, 3e-5, 1e-4, 3e-4, 1e-3]:
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(a + pert)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                stuck = 0
                break

        # Move 2: Coordinate transfer
        if not improved:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                for frac in [0.005, 0.01, 0.03, 0.05, 0.1, 0.2, 0.5]:
                    candidate = a.copy()
                    amount = candidate[i] * frac
                    candidate[i] -= amount
                    candidate[j] += amount
                    candidate = np.maximum(candidate, 0)
                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        stuck = 0
                        break

        # Move 3: Sinusoidal perturbation
        if not improved and it % 3 == 0:
            freq = rng.uniform(0.5, 12.0)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(1e-5, 5e-4)
            pattern = np.sin(2 * np.pi * freq * np.arange(n) / n + phase) * amp
            candidate = project_simplex(a + pattern)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                stuck = 0

        # Move 4: Block transfer
        if not improved and it % 7 == 0:
            block_size = max(2, int(rng.integers(2, max(3, n // 16))))
            i = int(rng.integers(0, n - block_size))
            j = int(rng.integers(0, n - block_size))
            if abs(i - j) >= block_size:
                candidate = a.copy()
                amt = 0.1 * candidate[i:i + block_size].sum()
                candidate[i:i + block_size] -= amt / block_size
                candidate[j:j + block_size] += amt / block_size
                candidate = np.maximum(candidate, 0)
                ss = candidate.sum()
                if ss > 1e-10:
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        stuck = 0

        if not improved:
            stuck += 1
            if stuck > 300:
                a = project_simplex(best_a + rng.standard_normal(n) * 1e-4)
                stuck = 0

        if stuck > 5000:
            break

    return best_a, best_obj


def parametric_init(n, rng):
    """Generate a diverse parametric initialization."""
    x = np.arange(n, dtype=np.float64)
    kind = int(rng.integers(0, 8))

    if kind == 0:
        # Ramp (asymmetric)
        a = x + 1.0
    elif kind == 1:
        # Inverted ramp
        a = n - x
    elif kind == 2:
        # Gaussian bump (asymmetric position)
        center = rng.uniform(0.2, 0.8) * n
        width = rng.uniform(0.1, 0.3) * n
        a = np.exp(-0.5 * ((x - center) / width) ** 2)
    elif kind == 3:
        # Two bumps
        c1 = rng.uniform(0.1, 0.4) * n
        c2 = rng.uniform(0.6, 0.9) * n
        w = rng.uniform(0.05, 0.15) * n
        a = np.exp(-0.5 * ((x - c1) / w) ** 2) + np.exp(-0.5 * ((x - c2) / w) ** 2)
    elif kind == 4:
        # Cosine modulation
        freq = rng.integers(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * freq * x / n + phase)
    elif kind == 5:
        # Power law
        gamma = rng.uniform(0.3, 2.0)
        a = (x + 1.0) ** gamma
    elif kind == 6:
        # Step function
        breakpoint = int(rng.integers(n // 4, 3 * n // 4))
        ratio = rng.uniform(0.3, 3.0)
        a = np.ones(n)
        a[:breakpoint] = ratio
    else:
        # Random smooth
        a = np.ones(n)
        for _ in range(rng.integers(2, 6)):
            center = rng.uniform(0, n)
            width = rng.uniform(0.05, 0.3) * n
            amp = rng.uniform(-0.3, 0.3)
            a += amp * np.exp(-0.5 * ((x - center) / width) ** 2)

    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point: Hybrid optimization for minimum peak auto-convolution.
    Combines mirror descent, subgradient methods, parametric initialization,
    and fine local search with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Using uniform init: score={best_score:.8f}")

    n_parent = len(parent)

    def record(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
            best_seq = a_norm.copy()
            best_score = score

    # === Phase 1: Multi-n exploration ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    n_candidates = [256, 384, 512, 640, 768, 896, 1024, 1280, 1536, 2048]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n for n in n_candidates if 128 <= n <= 2560))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 350:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try

        # Quick mirror descent
        t_alloc = min(remaining * 0.025, 18)
        a_opt, _ = mirror_descent_opt(start, time_limit=t_alloc, T_init=0.005, lr_init=0.03, K=5)
        record(a_opt, f"md_n{n_try}")

        # Quick subgradient
        remaining2 = deadline - time.monotonic()
        if remaining2 > 300:
            a_opt2, _ = subgradient_opt(a_opt, time_limit=min(remaining2 * 0.012, 8), lr_init=0.005)
            record(a_opt2, f"sgd_n{n_try}")

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep optimization ===
    print("\n=== Phase 2: Deep Optimization ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 200:
        t_alloc = min(remaining * 0.30, 450)
        print(f"  Mirror descent deep, n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = mirror_descent_opt(best_seq, time_limit=t_alloc, T_init=0.003, lr_init=0.02, K=5)
        record(a_opt, "md_deep")

    remaining = deadline - time.monotonic()
    if remaining > 150:
        t_alloc2 = min(remaining * 0.20, 250)
        print(f"  Subgradient deep, t={t_alloc2:.0f}s")
        a_opt2, _ = subgradient_opt(a_opt, time_limit=t_alloc2, lr_init=0.004)
        record(a_opt2, "sgd_deep")

    remaining = deadline - time.monotonic()
    if remaining > 100:
        t_alloc3 = min(remaining * 0.15, 150)
        print(f"  TopK blend opt, t={t_alloc3:.0f}s")
        a_opt3, _ = topK_blend_opt(a_opt2, time_limit=t_alloc3, K=3, T_init=0.0005, lr_init=0.008)
        record(a_opt3, "topk_blend")

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Perturbed restarts ===
    print("\n=== Phase 3: Perturbed Restarts ===")
    t_phase = time.monotonic()

    for scale in [0.003, 0.008, 0.015, 0.03, 0.05]:
        remaining = deadline - time.monotonic()
        if remaining < 120:
            break
        n_best = len(best_seq)
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc = min(remaining * 0.10, 80)
        a_opt, _ = mirror_descent_opt(f_pert, time_limit=t_alloc, T_init=0.002, lr_init=0.01, K=3)
        record(a_opt, f"md_pert_{scale}")

        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        a_opt2, _ = subgradient_opt(a_opt, time_limit=min(remaining * 0.05, 30), lr_init=0.003)

    # === Phase 4: Parametric initialization exploration ===

        a_param = parametric_init(n_try, rng)

        # Blend with best
        if n_try == n_best:
            for alpha in [0.3, 0.5, 0.7]:
                blend = (1 - alpha) * best_seq + alpha * a_param
                blend = project_simplex(blend)
            else:
                blend = a_param
        else:
            # Resample best to n_try and blend
            best_resamp = np.interp(np.linspace(0, 1, n_try),
                                    np.linspace(0, 1, n_best), best_seq)
            best_resamp = np.maximum(best_resamp, 0)
            best_resamp = best_resamp / best_resamp.sum()
            blend = 0.3 * best_resamp + 0.7 * a_param
            blend = project_simplex(blend)

        t_alloc = min(remaining * 0.04, 30)
        a_opt, _ = mirror_descent_opt(blend, time_limit=t_alloc, T_init=0.005, lr_init=0.03, K=5)
        record(a_opt, f"param_{trial}_n{n_try}")

    # === Phase 5: Local search ===
    print("\n=== Phase 5: Local Search ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()

    if remaining > 60:
        t_alloc = min(remaining * 0.35, 250)

    if remaining > 40:
        t_alloc2 = min(remaining * 0.30, 150)
        print(f"  Local search 2, t={t_alloc2:.0f}s")
        a_opt2, _ = local_search_fine(best_seq, time_limit=t_alloc2, seed=99999)
        record(a_opt2, "local2")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_alloc3 = min(remaining * 0.25, 100)
        a_opt3, _ = local_search_fine(best_seq, time_limit=t_alloc3, seed=777777)

    # === Phase 6: Alternating fine-tuning ===
    print("\n=== Phase 6: Final Polish ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)

    for round_num in range(6):

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
