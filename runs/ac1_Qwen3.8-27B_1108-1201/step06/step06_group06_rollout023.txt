"""
Hybrid optimization for minimum peak auto-convolution step function.
Uses Adam-smoothed gradient, LP-FW oracle, zoom-in strategy,
and ultra-fine local search. Multi-n exploration with adaptive budget.
Key difference from previous: Adam optimizer, zoom-in alternation,
sharper temperature schedules, and more n-value exploration.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_grad(a, p):
    """Compute gradient sum_k p[k]*conv[k] w.r.t. a. grad[j] = 2*sum_m a[m]*p[j+m]."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = 2.0 * np.dot(a, p[j:j + n])
    return grad


def hankel_grad_fft(a, p):
    """Faster hankel gradient using FFT-based cross-correlation."""
    n = len(a)
    # grad[j] = 2 * sum_{m=0}^{n-1} a[m] * p[j+m]
    # This is 2 * cross_correlation(a, p) for lags 0..n-1
    size = 4 * n
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    # cross-correlation: IFFT(conj(A) * P) gives sum_m a[m]*p[m+k]
    corr = np.fft.irfft(np.conj(A) * P, fft_size)
    # corr[k] = sum_m a[m] * p[m+k] for k = 0, 1, ...
    return 2.0 * corr[:n]


class AdamState:
    """Adam optimizer state."""
    def __init__(self, n, beta1=0.8, beta2=0.999, eps=1e-8):
        self.m = np.zeros(n)
        self.v = np.zeros(n)
        self.t = 0
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps

    def step(self, grad):
        self.t += 1
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * (grad * grad)
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)
        return m_hat / (np.sqrt(v_hat) + self.eps)


def build_lp_matrix(n):
    """Build sparse constraint matrix for LP oracle."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows), np.array(cols)


def solve_lp_oracle(f, M, n, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0. Returns normalized g or None."""
    c = -np.ones(n)
    vals = np.empty(len(row_idx))
    for idx in range(len(row_idx)):
        k = row_idx[idx]
        i = col_idx[idx]
        vals[idx] = f[k - i]

    A_mat = lil_matrix((2 * n - 1, n))
    A_mat[row_idx, col_idx] = vals
    A_mat = A_mat.tocsr()

    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success and res.fun is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None

        # Projected step with backtracking

        best_val = conv.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:

        f = best_candidate

                else:
                    lo = m1

            ft = np.maximum((1 - mid) * f + mid * g, 0)

            if val < best_val:

                best_f = f.copy()

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        if phase == 0:
            # Large move: FW-LP
            t_alloc = min(large_time, remaining * 0.5)
            f_opt, _ = fw_lp_optimize(f, time_limit=t_alloc)
            f = f_opt
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
            phase = 1
        else:
            # Small move: Adam smoothed gradient
            t_alloc = min(small_time, remaining * 0.5)
            T = max(1e-4, best_obj * 0.001)
            f_opt, _ = adam_smooth_grad(f, time_limit=t_alloc, T_init=T, lr_init=0.005, T_min=1e-6)
            f = f_opt
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
            phase = 0

    return best_f, best_obj


def fine_local_search(f0, time_limit, seed=42):
    """Ultra-fine local search with multi-scale perturbations."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        improved = False

        # Multi-scale random perturbations
        scales = [1e-5, 2e-5, 5e-5, 1e-4, 2e-4, 5e-4, 1e-3]
        for scale in scales:
            if not improved:
                pert = rng.standard_normal(n) * scale
                candidate = project_simplex(f + pert)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-16:
                    f = candidate
                    best_obj = c.max()
                    best_f = f.copy()
                    improved = True

        # Targeted peak reduction: reduce contribution to top peaks
        if not improved and it % 3 == 0:
            conv = fast_convolve(f)
            top_k = np.argsort(conv)[-5:]
            # For each top peak, find which coordinates contribute most
            for k in top_k:
                # Contribution of f[j] to conv[k] is ~f[k-j] (for valid indices)
                contrib = np.zeros(n)
                for j in range(n):
                    idx = k - j
                    if 0 <= idx < n:
                        contrib[j] = f[idx]
                # Move mass from high-contributing to low-contributing
                top_contrib = np.argsort(contrib)[-3:]
                low_contrib = np.argsort(contrib)[:3]
                for tc in top_contrib:
                    for lc in low_contrib:
                        if tc == lc or f[tc] < 1e-10:
                            continue
                        for frac in [0.01, 0.02, 0.05]:
                            amount = f[tc] * frac
                            candidate = f.copy()
                            candidate[tc] -= amount
                            candidate[lc] += amount
                            candidate = np.maximum(candidate, 0)
                            ss = candidate.sum()
                            if ss < 1e-10:
                                continue
                            candidate = candidate / ss
                            c = fast_convolve(candidate)
                            if c.max() < best_obj - 1e-16:
                                f = candidate
                                best_obj = c.max()
                                best_f = f.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break

        # Sinusoidal perturbation
        if not improved and it % 5 == 0:
            freq = rng.uniform(0.5, 10)
            phase = rng.uniform(0, 2 * np.pi)
            pattern = np.sin(2 * np.pi * freq * np.arange(n) / n + phase) * 5e-5
            candidate = project_simplex(f + pattern)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-16:
                f = candidate
                best_obj = c.max()
                best_f = f.copy()
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Perturb and restart
                f = project_simplex(best_f + rng.standard_normal(n) * 1e-4)
                stall = 0

    return best_f, best_obj


def try_seq(a, label, best_ref):
    """Evaluate a sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_ref
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_ref[1]:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return (a_norm.copy(), score)
    return best_ref


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point: Hybrid zoom-in search with Adam-optimized gradient,
    LP-FW, and ultra-fine local search. Multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Using uniform init: score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Multi-n exploration ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = [384, 512, 640, 700, 768, 832, 896, 960, 1024, 1152, 1280, 1536, 1920, 2048]
    if n_parent not in candidate_ns:
        candidate_ns.append(n_parent)
    candidate_ns = sorted(set(candidate_ns))

    n_results = {}
    for n_try in candidate_ns:
        remaining = deadline - time.monotonic()
        if remaining < 200:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try

        # Quick Adam gradient
        t_alloc = min(remaining * 0.03, 20)
        a_opt, obj = adam_smooth_grad(start, time_limit=t_alloc, T_init=0.002, lr_init=0.01)
        n_results[n_try] = obj
        best_seq, best_score = try_seq(a_opt, f"adam_n{n_try}", (best_seq, best_score))

        # Quick FW
        remaining2 = deadline - time.monotonic()
        if remaining2 > 150:
            t_alloc2 = min(remaining2 * 0.02, 12)
            a_opt2, _ = fw_lp_optimize(a_opt, time_limit=t_alloc2)
            best_seq, best_score = try_seq(a_opt2, f"fw_n{n_try}", (best_seq, best_score))
            n_results[n_try] = min(n_results[n_try], fast_convolve(a_opt2).max())

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")
    print(f"  n results: {', '.join(f'n={k}:{v:.6f}' for k,v in sorted(n_results.items()))}")

    # Select best n
    n_best = len(best_seq)
    print(f"  Selected n={n_best} for deep optimization")

    # === Phase 2: Deep zoom-in optimization ===
    print("\n=== Phase 2: Deep Zoom-In ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()

    if remaining > 200:
        t_alloc = min(remaining * 0.35, 500)
        print(f"  Zoom-in n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = zoom_in_optimize(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, "zoom_in", (best_seq, best_score))

    # Perturbed restarts
    for scale in [0.002, 0.005, 0.01, 0.02]:
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break
        n_best = len(best_seq)
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc2 = min(remaining * 0.1, 80)
        a_opt2, _ = zoom_in_optimize(f_pert, time_limit=t_alloc2)
        best_seq, best_score = try_seq(a_opt2, f"zoom_pert_{scale}", (best_seq, best_score))

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Sharp optimization ===
    print("\n=== Phase 3: Sharp Optimization ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    # Very sharp temperature
    if remaining > 100:
        t_alloc = min(remaining * 0.25, 250)
        T_sharp = max(1e-4, best_score / (2 * n_best) * 0.01)
        print(f"  Sharp Adam T={T_sharp:.6f}, t={t_alloc:.0f}s")
        a_opt, _ = adam_smooth_grad(best_seq, time_limit=t_alloc,
                                     T_init=T_sharp, lr_init=0.003, T_min=1e-7)
        best_seq, best_score = try_seq(a_opt, "sharp_adam", (best_seq, best_score))

    # Multiple sharp runs with different T
    for T_try in [5e-5, 1e-4, 2e-4, 5e-4]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        n_best = len(best_seq)
        a_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 1e-4)
        a_opt, _ = adam_smooth_grad(a_pert, time_limit=min(remaining * 0.15, 60),
                                     T_init=T_try, lr_init=0.002, T_min=1e-7)
        best_seq, best_score = try_seq(a_opt, f"sharp_T{T_try}", (best_seq, best_score))

    # Deep FW-LP
    remaining = deadline - time.monotonic()
    if remaining > 80:
        t_alloc = min(remaining * 0.2, 150)
        print(f"  Deep FW-LP, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, "deep_fw", (best_seq, best_score))

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Ultra-fine local search ===
    print("\n=== Phase 4: Ultra-fine Local Search ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()

    if remaining > 50:
        t_alloc = min(remaining * 0.35, 300)
        print(f"  Fine local search, t={t_alloc:.0f}s")
        a_opt, _ = fine_local_search(best_seq, time_limit=t_alloc, seed=12345)
        best_seq, best_score = try_seq(a_opt, "fine_local", (best_seq, best_score))

    # Another local search with different seed
    remaining = deadline - time.monotonic()
    if remaining > 40:
        t_alloc = min(remaining * 0.25, 150)
        a_opt, _ = fine_local_search(best_seq, time_limit=t_alloc, seed=99999)
        best_seq, best_score = try_seq(a_opt, "fine_local2", (best_seq, best_score))

    # Sharp Adam again on the locally-searched result
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_best = len(best_seq)
        T_sharp2 = max(1e-5, 1e-4)
        a_opt, _ = adam_smooth_grad(best_seq, time_limit=min(remaining * 0.2, 50),
                                     T_init=T_sharp2, lr_init=0.001, T_min=1e-8)
        best_seq, best_score = try_seq(a_opt, "final_sharp", (best_seq, best_score))

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Alternative n exploration ===
    print("\n=== Phase 5: Alternative n ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()

    # Try scaling the best sequence to nearby n values
    if remaining > 60:
        for n_try in [len(best_seq) + 32, len(best_seq) - 32, len(best_seq) * 2, len(best_seq) // 2]:
            if n_try < 128 or n_try > 3000:
                continue
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, len(best_seq)), best_seq)
            start = np.maximum(start, 0)
            start = start / start.sum()
            t_alloc = min(remaining * 0.15, 40)
            a_opt, _ = zoom_in_optimize(start, time_limit=t_alloc)
            best_seq, best_score = try_seq(a_opt, f"alt_n{n_try}", (best_seq, best_score))

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Final polish ===
    print("\n=== Phase 6: Final Polish ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Tiny perturbations with sharp optimization
    if remaining > 20:
        for i in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            noise = rng.standard_normal(n_best) * 5e-6
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = adam_smooth_grad(f_pert, time_limit=min(remaining * 0.15, 8),
                                         T_init=1e-5, lr_init=0.0005, T_min=1e-8)
            best_seq, best_score = try_seq(a_opt, f"micro_{i}", (best_seq, best_score))

    # Final local search
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_opt, _ = fine_local_search(best_seq, time_limit=min(remaining * 0.4, 20), seed=77777)
        best_seq, best_score = try_seq(a_opt, "final_local", (best_seq, best_score))

    # Final FW micro-step
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=min(remaining * 0.3, 10))
        best_seq, best_score = try_seq(a_opt, "final_fw", (best_seq, best_score))

    print(f"  Phase 6 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
