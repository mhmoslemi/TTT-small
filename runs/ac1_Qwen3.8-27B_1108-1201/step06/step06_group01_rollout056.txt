def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_topk_subgrad(a, K=30):
    """Compute subgradient from top-K peaks with inverse-rank weighting."""
    n = len(a)
    conv = fast_convolve(a)
    size = len(conv)
    K_eff = min(K, size)
    peaks = np.argpartition(conv, -K_eff)[-K_eff:]
    peak_vals = conv[peaks]
    order = np.argsort(peak_vals)[::-1]

    # Inverse-rank weighting
    weights = np.zeros(K_eff)
    for rank, idx in enumerate(order):
        weights[idx] = 1.0 / (rank + 1) ** 1.2
    ws = weights.sum()
    weights /= ws

    # Compute subgradient: grad_i = 2 * sum_k w_k * a_{k-i}
    grad = np.zeros(n)
    for i in range(K_eff):
        if weights[i] < 1e-15:
            continue
        k = int(peaks[i])
        w = weights[i] * 2.0
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad[j] += w * a[k - j]

    return grad, conv, peaks, peak_vals

def peak_equalize(a0, time_limit, n_peaks=5):
    """
    Try to equalize the top n_peaks by solving a linear system.
    Finds direction d such that (a+d*a)*a has equal values at those peaks.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        K = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -K)[-K:]
        peaks = peaks[np.argsort(conv[peaks])[::-1]]
        peak_vals = conv[peaks]

        # Only try to equalize peaks that are close to the max
        if peak_vals[-1] < peak_vals[0] * 0.95:
            # Too spread out, just reduce the max
            peaks = peaks[:1]
            K = 1

        # Build linear system: find d such that (d*a)[k_i] = -c for all i
        # (d*a)[k] = sum_j d_j * a_{k-j}
        # This is a Hankel matrix applied to d

        # Use least squares to find d
        A_eq = np.zeros((K, n))
        for i in range(K):
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_eq[i, j] = a[k - j]

        target = -peak_vals[0]  # Want to reduce peak by this amount
        b_eq = np.full(K, target * 0.1)  # Small reduction per step

        # Least squares solve with non-negativity-ish constraint
        try:
            d, _, _, _ = np.linalg.lstsq(A_eq, b_eq, rcond=None)
        except Exception:
            break

        # Project: a_new = a + alpha * d, maintain non-negativity and sum=1
        # Find max alpha such that a + alpha*d >= 0
        d_sum = d.sum()
        if abs(d_sum) < 1e-15:

            max_alpha = 1.0

            max_alpha = 1.0

            alpha = min(0.05, max_alpha * 0.9)

        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        obj = fast_convolve(a_new).max()

            a = a_new

                    a = a_new

                    best_obj = new_max
                    best_a = a_test.copy()
                    a = a_test
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1

        if stall > 30:
            rng2 = np.random.default_rng(stall * 7919)
            a = project_simplex(best_a + rng2.standard_normal(n) * 0.002)
            stall = 0
            if fast_convolve(a).max() > best_obj * 1.001:
                a = best_a.copy()

        if stall > 300:
            break

    return best_a, best_obj


def make_initial(n, shape_type="hann"):
    """Create initial sequence with given shape."""
    if shape_type == "uniform":
        return np.ones(n) / n
    elif shape_type == "hann":
        j = np.arange(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * j / (n - 1)))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif shape_type == "gaussian":
        j = np.arange(n)
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
        return a / a.sum()
    elif shape_type == "raised_cosine":
        j = np.linspace(0, np.pi, n)
        a = np.sin(j) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif shape_type == "power":
        j = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(j - center) / center, 0.0)
        a = a ** 2 + 1e-10
        return a / a.sum()
    elif shape_type == "binomial":
        # Based on the theoretical insight: a_k proportional to binom(2k,k)/4^k
        # Truncated and made symmetric
        m = n // 2
        c = np.array([1.0])
        for k in range(1, m + 1):
            c = np.append(c, c[-1] * (2 * k) * (2 * k - 1) / (4 * k * k))
        if n % 2 == 0:
            a = np.concatenate([c[m - 1:0:-1], c])
        else:
            a = np.concatenate([c[m:0:-1], c])
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search combining accelerated subgradient,
    softmax optimization, peak equalization, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t0_total = time.monotonic()
    deadline = t0_total + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t0_total
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = make_initial(768, "hann")

    n_parent = len(parent)

    # === Phase 1: Multi-shape, multi-n exploration (~250s) ===
    print("\n=== Phase 1: Multi-shape exploration ===")
    t_phase = time.monotonic()

    n_candidates = sorted(set([
        n_parent, n_parent - 64, n_parent + 64,
        512, 640, 768, 896, 1024, 1280
    ]))
    n_candidates = [n for n in n_candidates if 256 <= n <= 1400]

    shapes = ["hann", "gaussian", "raised_cosine", "power", "binomial"]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 1200:
            break

        t_alloc_per_shape = max(30, min(60, (deadline - time.monotonic() - 1200) / max(1, len(shapes))))

        for shape in shapes:
            if time.monotonic() > deadline - 1200:
                break

            if n_try == n_parent and shape == "hann":
                start = parent.copy()
            else:
                start = make_initial(n_try, shape)
                # Interpolate parent to this size as another option
                if parent is not None and rng.random() < 0.5:
                    start = np.interp(np.linspace(0, 1, n_try),
                                      np.linspace(0, 1, n_parent), parent)
                    start = np.maximum(start, 1e-10)
                    start = start / start.sum()

            t_alloc = min(t_alloc_per_shape, deadline - time.monotonic() - 1200)
            if t_alloc < 5:
                break

            a_opt, _ = opt_accel_subgrad(start, time_limit=t_alloc, K_init=20, lr_init=0.03)
            try_seq(a_opt, f"sub_n{n_try}_{shape[:4]}")

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep optimization on best ~700s ===
    print("\n=== Phase 2: Deep optimization ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Long accelerated subgradient run
        t_alloc = min(300, deadline - time.monotonic() - 700)
        if t_alloc > 20:
            print(f"  Deep subgrad n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = opt_accel_subgrad(best_seq, time_limit=t_alloc, K_init=30, lr_init=0.02)
            try_seq(a_opt, "deep_sub")

        # Softmax optimization
        t_alloc = min(200, deadline - time.monotonic() - 500)
        if t_alloc > 20:
            print(f"  Softmax opt n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = opt_softmax(best_seq, time_limit=t_alloc, lr_init=1.0, K=40)
            try_seq(a_opt, "softmax")

        # Peak equalization
        t_alloc = min(100, deadline - time.monotonic() - 400)
        if t_alloc > 10:
            print(f"  Peak equalize n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = peak_equalize(best_seq, time_limit=t_alloc, n_peaks=5)
            try_seq(a_opt, "peak_eq")

        # Second round softmax from best
        t_alloc = min(100, deadline - time.monotonic() - 300)
        if t_alloc > 10:
            print(f"  Softmax round 2, t={t_alloc:.0f}s")
            a_opt, _ = opt_softmax(best_seq, time_limit=t_alloc, lr_init=0.5, K=20)
            try_seq(a_opt, "softmax2")

    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Perturbed restarts and alternative n ~300s ===
    print("\n=== Phase 3: Perturbed restarts ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        for scale in [0.001, 0.003, 0.005, 0.01, 0.02]:
            t_alloc = min(50, deadline - time.monotonic() - 200)
            if t_alloc < 10:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            a_opt, _ = opt_accel_subgrad(f_pert, time_limit=t_alloc, K_init=30, lr_init=0.02)
            try_seq(a_opt, f"pert_{scale}")

        # Try nearby n values
        for n_delta in [-32, 32, -64, 64, 128]:
            n_try = n_best + n_delta
            if n_try < 200 or n_try > 2000:
                continue
            t_alloc = min(40, deadline - time.monotonic() - 200)
            if t_alloc < 10:
                break
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_best), best_seq)
            start = np.maximum(start, 1e-10)
            start = start / start.sum()
            a_opt, _ = opt_accel_subgrad(start, time_limit=t_alloc, K_init=30, lr_init=0.02)
            try_seq(a_opt, f"n{n_try}")

    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Coordinate descent polish ~200s ===
    print("\n=== Phase 4: Coordinate descent polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(150, deadline - time.monotonic() - 100)
        if t_alloc > 10:
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_opt, _ = coord_descent(best_seq, time_limit=t_alloc, max_pairs=30)
            try_seq(a_opt, "coord")

        # Another peak equalization pass
        t_alloc = min(50, deadline - time.monotonic() - 50)
        if t_alloc > 5:
            a_opt, _ = peak_equalize(best_seq, time_limit=t_alloc, n_peaks=4)
            try_seq(a_opt, "peak_eq2")

    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final fine polishing ~100s ===
    print("\n=== Phase 5: Final polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Very fine subgradient
        t_alloc = min(50, deadline - time.monotonic() - 30)
        if t_alloc > 5:
            a_opt, _ = opt_accel_subgrad(best_seq, time_limit=t_alloc, K_init=50, lr_init=0.005)
            try_seq(a_opt, "final_sub")

        # Softmax with small lr
        t_alloc = min(30, deadline - time.monotonic() - 15)
        if t_alloc > 5:
            a_opt, _ = opt_softmax(best_seq, time_limit=t_alloc, lr_init=0.1, K=50)
            try_seq(a_opt, "final_soft")

        # One more coord descent
        t_alloc = min(15, deadline - time.monotonic() - 5)
        if t_alloc > 3:
            a_opt, _ = coord_descent(best_seq, time_limit=t_alloc, max_pairs=40)
            try_seq(a_opt, "final_coord")

    print(f"  Phase 5 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = make_initial(768, "hann")
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, total time={time.monotonic()-t0_total:.0f}s)")
    return [float(x) for x in best_seq]
