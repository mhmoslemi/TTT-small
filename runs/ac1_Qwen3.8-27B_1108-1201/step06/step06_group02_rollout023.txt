"""
Optimized multi-phase search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses IRLS (majorization-minimization) with adaptive power, Adam optimizer,
FW-LP refinement, and targeted peak reduction.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def hankel_matvec(w, a):
    """Compute M(w) @ a where M(w)[i,j] = w[i+j] for i+j < 2n-1."""
    n = len(a)
    # M(a)[i] = sum_j w[i+j] * a[j]
    # = convolve(a_flipped, w)[n-1 : 2n-1]
    b = a[::-1]
    fft_size = 1 << (len(b) + len(w) - 1).bit_length()
    B = np.fft.rfft(b, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(B * W, fft_size)[:len(b) + len(w) - 1]
    return conv[n-1 : 2*n-1]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(a):
    """Compute 2*n*max(c)/S^2 where c=conv(a,a), S=sum(a)."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (S * S)

    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def irls_optimize(a0, time_limit, p_start=4, p_end=32, max_iters=5000):
    """IRLS/MM optimization: iteratively minimize a^T H(w) a where w = c^(p-1)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / S
    n = len(a)
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    t0 = time.monotonic()
    it = 0
    p = p_start
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        c = fast_convolve(a)
        c_max = c.max()
        if c_max < 1e-30:
            break
        
        # Normalize c for numerical stability
        w = (c / c_max) ** (p - 1)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(2*n-1) / (2*n-1)
        else:
            w = w / w_sum
        
        # Minimize a^T H(w) a s.t. a >= 0, sum(a) = 1
        # Using projected gradient with adaptive step
        inner_a = a.copy()
        for inner_it in range(200):
            grad = 2.0 * hankel_matvec(w, inner_a)
            # Adaptive step size
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            # Line search with backtracking
            lr = 1.0 / (np.max(np.abs(grad)) + 1e-10)
            for trial in range(10):
                trial_lr = lr * (0.5 ** trial)
                candidate = project_simplex(inner_a - trial_lr * grad)
                # Check if objective decreased
                obj_new = candidate @ hankel_matvec(w, candidate)
                obj_old = inner_a @ hankel_matvec(w, inner_a)
                if obj_new < obj_old:
                    inner_a = candidate
                    break
            else:
                break
        
        a = inner_a
        score = compute_score(a)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if it % 100 == 0:
                print("    IRLS it={}: score={:.10f}".format(it, best_score))
        else:
            no_improve += 1
        
        # Adapt p
        if no_improve > 50:
            p = min(p * 1.5, p_end)
            no_improve = 0
        
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.005 + 0.005 * min(1.0, (no_improve - 200) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            p = p_start
            no_improve = 0
    
    return best_a, best_score


def adam_optimize(a0, time_limit, p=8, lr_init=0.01):
    """Adam optimizer on the simplex for minimizing (sum c^p)^(1/p)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / S
    n = len(a)
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        c = fast_convolve(a)
        c_max = c.max()
        if c_max < 1e-30:
            break
        
        # Gradient of sum(c^p) w.r.t. a
        w = (c / c_max) ** (p - 1)
        grad_a = 2.0 * p * hankel_matvec(w, a)
        
        # Subtract mean for simplex compatibility
        grad_a = grad_a - np.mean(grad_a)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_a
        v = beta2 * v + (1 - beta2) * grad_a ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        
        score = compute_score(a_new)
        a = a_new
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if t % 200 == 0:
                print("    Adam t={}: score={:.10f}".format(t, best_score))
        else:
            no_improve += 1
        
        # Adapt learning rate
        if no_improve > 50:
            lr *= 0.95
        if no_improve > 200:
            # Restart
            rng = np.random.default_rng(t * 17 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            m = np.zeros(n)
            v = np.zeros(n)
            lr = lr_init
            no_improve = 0
    
    return best_a, best_score


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        g = solve_lp_oracle(f, M * m_exp, time_limit=min(20, remaining * 0.3))
        if g is None:
            m_exp = min(m_exp * 1.3, 5.0)
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 7919 + 13)
                scale = 0.002 + 0.003 * min(1.0, no_improve / 100.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
                continue
            m_exp = min(m_exp * 1.05, 3.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in np.linspace(0.01, 1.0, 15):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if 0.01 < best_alpha < 0.99:
            lo = max(0.0, best_alpha - 0.04)
            hi = min(1.0, best_alpha + 0.04)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if it % 20 == 0:
                    print("    FW-LP it={}: obj={:.10f}".format(it, best_obj))
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)
        
        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.004 * min(1.0, (no_improve - 60) / 150.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj


def targeted_peak_reduction(a0, time_limit):
    """Reduce the highest peak by targeted coordinate perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    t0 = time.monotonic()
    it = 0
    stale = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        c = fast_convolve(a)
        k_peak = int(np.argmax(c))
        
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        
        if len(affected) == 0:
            break
        
        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]
        
        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            break
        j_min_cand = outside[np.argmin(a[outside])]
        
        improved = False
        for delta in [0.0003, 0.0005, 0.001, 0.002, 0.005]:
            if a[j_max] < delta:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min_cand] += delta
            candidate = np.maximum(candidate, 0.0)
            candidate = candidate / candidate.sum()
            s = compute_score(candidate)
            if s < best_score - 1e-14:
                a = candidate
                best_score = s
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            stale = 0
        else:
            stale += 1
            if stale % 50 == 0:
                rng = np.random.default_rng(it * 4567 + 3)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
                s = compute_score(candidate)
                if s < best_score - 1e-14:
                    a = candidate
                    best_score = s
                    best_a = a.copy()
                    stale = 0
    
    return best_a, best_score


def try_and_update(a, state, label):
    """Evaluate a sequence and update state if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.10f} (n={})".format(label, score, len(a)))
