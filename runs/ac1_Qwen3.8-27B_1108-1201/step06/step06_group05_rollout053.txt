def build_lp_matrix(n):
    """Build the banded constraint matrix for the LP oracle.
    A_ub has shape (2n-1, n) where A_ub[k, j] = f[j] if |k - (n-1-j)| < n else 0.

    return A

def coordinate_descent(a0, time_limit):
    """
    Coordinate descent: optimize one element at a time.
    Since for non-negative a, max(conv) = sum(a^2), the objective with
    normalization sum(a)=1 is 2*n*sum(a^2). The minimum is at a=uniform.
    But if the actual objective is different, this still works.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    sweep = 0
    max_sweeps = 50

    while time.monotonic() - t0 < time_limit and sweep < max_sweeps:
        sweep += 1
        improved = False

        # Random order for diversity
        order = np.random.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit - 0.05:
                break

            old_val = a[j]
            if old_val < 1e-15:
                # Try setting to small positive
                candidates = [1.0 / n, 1.0 / (2 * n), 1.0 / (4 * n)]
            else:
                # Try geometric spacing around current value
                candidates = [0.0, old_val * 0.3, old_val * 0.6, old_val * 0.8,
                              old_val * 1.2, old_val * 1.5, old_val * 2.0, 1.0 / n]

            best_candidate = old_val
            best_obj_j = compute_score(a)

            for cv in candidates:
                a[j] = cv
                obj = compute_score(a)
                if obj < best_obj_j - 1e-15:
                    best_obj_j = obj
                    best_candidate = cv

            a[j] = best_candidate

            if best_obj_j < best_obj - 1e-14:
                best_obj = best_obj_j
                best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_obj


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase optimization exploiting problem structure.
    Uses symmetry, LP-based Frank-Wolfe, coordinate descent, and parametric search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return
        a_norm = a / s
        try:
            sc = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            return
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    n_parent = len(parent)

    # === Phase 0: Quick evaluations (~10s) ===
    print("\n=== Phase 0: Quick Baselines ===")
    t0 = time.monotonic()

    # Symmetrized parent
    sym_p = enforce_symmetry(parent)
    try_update(sym_p, "sym_parent")

    # Uniform at parent length
    try_update(np.ones(n_parent), "uniform_parent")

    # Parametric at parent length
    if n_parent <= 2048:
        param_a, _ = parametric_search(n_parent, time_limit=5)
        if param_a is not None:
            try_update(param_a, "parametric")

    print(f"  Phase 0 best: {best_score:.8f} ({time.monotonic()-t0:.1f}s)")

    # === Phase 1: Primary optimization on best sequence (~600s) ===
    print("\n=== Phase 1: Primary FW-LP + Subgrad ===")
    t0 = time.monotonic()
    n_best = len(best_seq)

    # FW-LP (most promising method)
    remaining = deadline - time.monotonic() - 800
    if remaining > 20:
        t_alloc = min(remaining, 300)
        print(f"  FW-LP n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, use_symmetry=True)
        try_update(a_opt, "fw_lp_1")

    # Subgradient
    remaining = deadline - time.monotonic() - 500
    if remaining > 15:
        t_alloc = min(remaining, 200)
        print(f"  Subgrad n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = subgrad_optimize(best_seq, time_limit=t_alloc, top_k=20, lr_init=0.02)
        try_update(a_opt, "subgrad_1")

    # FW-LP again on refined
    remaining = deadline - time.monotonic() - 400
    if remaining > 15:
        t_alloc = min(remaining, 150)
        print(f"  FW-LP refined n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, use_symmetry=True)
        try_update(a_opt, "fw_lp_2")

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t0:.1f}s)")

    # === Phase 2: Length exploration (~400s) ===
    print("\n=== Phase 2: Length Exploration ===")
    t0 = time.monotonic()

    n_best = len(best_seq)
    # Try various lengths
    candidate_ns = set()
    for n_try in [n_best, n_best // 2, n_best * 2, 256, 384, 512, 640, 768, 896, 1024, 1280, 1536]:
        if 64 <= n_try <= 4096:
            candidate_ns.add(n_try)

    candidate_ns = sorted(candidate_ns)

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 300
        if remaining < 30:
            break

        if n_try == n_best:
            start = best_seq.copy()
        else:
            start = interp_sequence(best_seq, n_try)

        # Quick FW-LP
        t_alloc = min(remaining, 60)
        if t_alloc < 10:
            continue
        print(f"  n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(start, time_limit=t_alloc * 0.6, use_symmetry=True)
        try_update(a_opt, f"fw_n{n_try}")

        # Quick subgrad
        remaining = deadline - time.monotonic() - 250
        if remaining > 10:
            a_opt2, _ = subgrad_optimize(a_opt, time_limit=min(remaining, t_alloc * 0.3),
                                         top_k=15, lr_init=0.015)
            try_update(a_opt2, f"sg_n{n_try}")

    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: Coordinate descent + perturbation (~250s) ===
    print("\n=== Phase 3: Coordinate Descent + Perturbation ===")
    t0 = time.monotonic()
    n_best = len(best_seq)

    # Coordinate descent
    remaining = deadline - time.monotonic() - 200
    if remaining > 15:
        t_alloc = min(remaining, 100)
        print(f"  Coord descent n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
        try_update(a_opt, "coord_desc")

    # Perturbation + re-optimization
    for scale in [0.005, 0.01, 0.02, 0.04]:
        remaining = deadline - time.monotonic() - 100
        if remaining < 15:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        f_pert = enforce_symmetry(f_pert)
        t_alloc = min(remaining, 60)
        a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc, use_symmetry=True)
        try_update(a_opt, f"perturb_{scale}")

    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Intensive refinement (~150s) ===
    print("\n=== Phase 4: Intensive Refinement ===")
    t0 = time.monotonic()
    n_best = len(best_seq)

    # Multiple FW-LP runs with different relaxations
    remaining = deadline - time.monotonic() - 60
    if remaining > 20:
        t_alloc = min(remaining, 80)
        print(f"  Deep FW-LP n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, use_symmetry=True)
        try_update(a_opt, "deep_fw")

    # Subgradient with different parameters
    for top_k, lr in [(10, 0.01), (30, 0.008), (50, 0.005)]:
        remaining = deadline - time.monotonic() - 30
        if remaining < 10:
            break
        t_alloc = min(remaining, 30)
        a_opt, _ = subgrad_optimize(best_seq, time_limit=t_alloc, top_k=top_k, lr_init=lr)
        try_update(a_opt, f"sg_k{top_k}")

    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: Final polishing (~50s) ===
    print("\n=== Phase 5: Final Polish ===")
    t0 = time.monotonic()
    n_best = len(best_seq)

    # Tiny perturbations
    for _ in range(3):
        remaining = deadline - time.monotonic() - 10
        if remaining < 10:
            break
        noise = rng.standard_normal(n_best) * 0.002
        f_pert = project_simplex(best_seq + noise)
        f_pert = enforce_symmetry(f_pert)
        t_alloc = min(remaining, 15)
        a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc, use_symmetry=True)
        try_update(a_opt, "final_perturb")

    # Final symmetry check
    sym_final = enforce_symmetry(best_seq)
    try_update(sym_final, "final_sym")

    # Final uniform check
    try_update(np.ones(n_best), "final_uniform")

    print(f"  Phase 5 best: {best_score:.8f} ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
