If we parameterize `a` as `a_j = max(0, |F^{-1}[s]|_j)` for some spectral function `s`, we could optimize over `s` directly. But this adds a non-differentiable `max(0, ...)` and is complex.

IRLS-QP + Smoothed Gradient + LP-FW hybrid optimizer for minimum peak autocorrelation.
Uses FFT-based operations, iterative reweighted least squares with QP subproblems,
peak-targeted smoothed gradient descent, and sparse LP-based Frank-Wolfe refinement.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_matvec(a, w):
    """Compute (M_w * a)_j = sum_i a_i * w_{j+i} where w has length 2n-1.
    Uses FFT for O(n log n) complexity."""
    n = len(a)
    size = 2 * n - 1
    a_rev = a[::-1]
    A = np.fft.rfft(a_rev, size)
    W = np.fft.rfft(w, size)
    conv = np.fft.irfft(A * W, size)
    return conv[n - 1:2 * n - 1]


def compute_grad(a, p):
    """Gradient of sum_k p_k * (a*a)[k] w.r.t. a.
    grad_j = 2 * sum_m a_m * p[j+m]. Uses FFT."""
    return 2.0 * hankel_matvec(a, p)


def irls_qp_solve(a, w, n_inner=40, lr_init=0.03):
    """Solve min sum_k w_k*(a*a)[k] s.t. a>=0, sum(a)=1 via projected gradient."""
    a_curr = a.copy()
    lr = lr_init
    c = fast_convolve(a_curr)
    obj = np.dot(w, c)
    
    for it in range(n_inner):
        grad = compute_grad(a_curr, w)
        # Backtracking line search
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
            trial_lr = lr * factor
            candidate = project_simplex(a_curr - trial_lr * grad)
            c_new = fast_convolve(candidate)
            obj_new = np.dot(w, c_new)
            if obj_new < obj - 1e-15:
                a_curr = candidate
                obj = obj_new
                improved = True
                break
        if not improved:
            # Try a smaller base lr
            lr *= 0.5
            if lr < 1e-6:
                break
    
    return a_curr, obj

def build_lp_components(n):
    """Build sparse constraint structure for LP oracle."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows), np.array(cols)


def solve_lp_oracle(f, M, n, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    A_mat = lil_matrix((2 * n - 1, n))
    for idx in range(len(row_idx)):
        k = row_idx[idx]
        i = col_idx[idx]
        A_mat[k, i] = f[k - i]
    A_mat = A_mat.tocsr()
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.fun is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None

        best_seq, best_score = try_seq(a_opt, "lpfw", (best_seq, best_score))
    
    # Perturbed FW restarts
    for scale in [0.003, 0.008, 0.02]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        n_best = len(best_seq)
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc2 = min(remaining * 0.2, 100)
        max_lp2 = max(5, int(t_alloc2 / 8))
        a_opt2, _ = fw_lp_optimize(f_pert, time_limit=t_alloc2, max_lp_calls=max_lp2)
        best_seq, best_score = try_seq(a_opt2, f"fw_pert_{scale}", (best_seq, best_score))
    
    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 4: Local Search + Final Polish ===
    print("\n=== Phase 4: Local Search ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    
    # Local search
    if remaining > 40:
        t_alloc = min(remaining * 0.3, 100)
        print(f"  Local search 1, t={t_alloc:.0f}s")
        a_opt, _ = local_search(best_seq, time_limit=t_alloc, seed=12345)
        best_seq, best_score = try_seq(a_opt, "local1", (best_seq, best_score))
    
    # Fine smoothed gradient
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_alloc = min(remaining * 0.2, 60)
        print(f"  Fine smooth T=0.0003, t={t_alloc:.0f}s")
        a_opt, _ = smoothed_grad_optimize(best_seq, time_limit=t_alloc,
                                          T_init=0.0003, lr_init=0.005)
        best_seq, best_score = try_seq(a_opt, "fine_smooth", (best_seq, best_score))
    
    # Another IRLS pass with low alpha (polish)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_alloc = min(remaining * 0.2, 50)
        a_opt, _ = irls_optimize(best_seq, time_limit=t_alloc, alpha_start=0.8, n_inner=50)
        best_seq, best_score = try_seq(a_opt, "irls_polish", (best_seq, best_score))
    
    # Local search with different seed
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t_alloc = min(remaining * 0.25, 50)
        a_opt, _ = local_search(best_seq, time_limit=t_alloc, seed=99999)
        best_seq, best_score = try_seq(a_opt, "local2", (best_seq, best_score))
    
    # Tiny perturbations + quick IRLS
    remaining = deadline - time.monotonic()
    if remaining > 15:
        for _ in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 4:
                break
            noise = rng.standard_normal(n_best) * 0.0004
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = smoothed_grad_optimize(f_pert, time_limit=min(remaining * 0.15, 5),
                                              T_init=0.0002, lr_init=0.003)
            best_seq, best_score = try_seq(a_opt, "micro", (best_seq, best_score))
    
    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")
    
    # === Final: Try slightly different n around best ===
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    if remaining > 30:
        for delta in [-32, -16, 16, 32]:
            n_try = n_best + delta
            if n_try < 256 or n_try > 2000:
                continue
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_best), best_seq)
            start = np.maximum(start, 0)
            start = start / start.sum()
            t_alloc = min(remaining * 0.2, 20)
            a_opt, _ = irls_optimize(start, time_limit=t_alloc, alpha_start=1.5, n_inner=25)
            best_seq, best_score = try_seq(a_opt, f"alt_n{n_try}", (best_seq, best_score))
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(640) / 640
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
