def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad_at_peaks(a, peak_indices, weights):
    """Compute weighted sum of gradients of (a*a)[k] at given peaks."""
    n = len(a)
    grad = np.zeros(n)
    for idx, k in enumerate(peak_indices):
        w = weights[idx]
        if w < 1e-15:
            continue
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        # For i in [js, je]: grad[i] += 2*w*a[k-i]
        # k-i for i=js: k-js, for i=je: k-je
        # So we need a[k-je : k-js+1] reversed
        rev = a[k - je : k - js + 1][::-1]
        grad[js:je + 1] += 2.0 * w * rev
    return grad

def smoothed_objective_and_grad(a, sigma, top_k=50):

                    y = a_new.copy()
                a = a_new
                # Increase lr slightly

                stagnant = 0
            else:

                lr *= 0.5
                stagnant += 1
                y = a  # Reset y
                if stagnant > 50:

            actual_max = conv_a.max()
            if actual_max < best_obj - 1e-15:

        if verbose:

def fw_with_grad_oracle(a0, time_limit, top_k=20, verbose=False):
    """
    Frank-Wolfe with gradient-based oracle.
    Oracle: find the coordinate with minimum gradient component.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit - 0.5:
        it += 1
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Find active peaks (within 5% of max)
        threshold = M * 0.95
        active = np.where(conv >= threshold)[0]
        
        # Weight by proximity to max
        w = (conv[active] - threshold) / (M - threshold + 1e-15)
        w = w / w.sum()
        
        # Compute gradient
        grad = compute_grad_at_peaks(a, active, w)
        
        # FW oracle: minimum gradient component
        min_idx = np.argmin(grad)
        g = np.zeros(n)
        g[min_idx] = 1.0
        
        # Also try top-5 minimum components
        top_min = np.argsort(grad)[:5]
        g_multi = np.zeros(n)
        g_multi[top_min] = 1.0 / len(top_min)
        
        # Line search on both directions
        best_alpha = 0.0
        best_val = M

                ft = (1 - alpha) * a + alpha * direction

                if val < best_val:

def active_set_refine(a0, time_limit, tol=0.02, verbose=False):
    """
    Active set refinement: identify active peaks, solve KKT system.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit - 0.5:
        it += 1
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Active set: peaks within tol*M of the max
        active = np.where(conv >= M * (1 - tol))[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(conv)])
        
        # Try to equalize active peaks by adjusting a
        # Gradient-based: move in direction that reduces all active peaks
        w = np.ones(len(active)) / len(active)
        grad = compute_grad_at_peaks(a, active, w)
        
        # Move opposite to gradient, projected
        step_size = 0.01
        a_new = project_simplex(a - step_size * grad / (np.max(np.abs(grad)) + 1e-15))
        
        conv_new = fast_convolve(a_new)
        M_new = conv_new.max()
        
        if M_new < M - 1e-15:
            a = a_new
            if M_new < best_obj - 1e-14:
                best_obj = M_new
                best_a = a.copy()
        else:
            # Try smaller step
            step_size *= 0.5
            if step_size < 1e-8:
                # Try perturbation
                rng = np.random.default_rng(it * 31)
                a = project_simplex(a + rng.standard_normal(n) * 0.002)
                step_size = 0.01
    
    return best_a, best_obj

def try_resample_and_optimize(parent, target_n, time_limit, verbose=False):
    """Resample parent to target_n and optimize."""
    if len(parent) == target_n:
        a0 = parent.copy()
    else:
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, target_n)
        a0 = np.interp(x_new, x_old, parent)
        a0 = np.maximum(a0, 0)
        a0 = a0 / a0.sum()
    
    # Run Nesterov smoothed descent

    # Run FW with gradient oracle
    a_opt2, obj2 = fw_with_grad_oracle(a_opt, time_limit=time_limit * 0.3, verbose=verbose)
    # Active set refinement
    a_opt3, obj3 = active_set_refine(a_opt2, time_limit=time_limit * 0.2, verbose=verbose)
    
    return a_opt3, obj3

"""

2. Nesterov-accelerated projected subgradient with adaptive smoothing

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_at_peaks(a, peak_indices, weights):
    """
    Compute weighted sum of gradients of (a*a)[k] at given peaks.
    grad[i] = sum_k weights[k] * 2 * a[k-i] (for valid k-i)
    """
    n = len(a)
    grad = np.zeros(n)
    for idx in range(len(peak_indices)):
        w = weights[idx]
        if w < 1e-16:
            continue
        k = int(peak_indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        # For i in [js, je]: grad[i] += 2*w*a[k-i]
        # k-i for i=js is k-js, for i=je is k-je
        rev = a[k - je : k - js + 1][::-1]
        grad[js:je + 1] += 2.0 * w * rev
    return grad


def get_gradients_smoothed(a, sigma, top_k=50):
    """
    Compute smoothed max and its gradient using top-k peaks.
    Returns (smooth_value, gradient, full_convolution).
    """
    conv = fast_convolve(a)
    n = len(a)

    if top_k < len(conv):
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
    else:
        peak_idx = np.arange(len(conv))

    c = conv[peak_idx]
    max_c = c.max()
    exp_c = np.exp(sigma * (c - max_c))
    exp_sum = exp_c.sum()
    p = exp_c / exp_sum
    smooth_val = max_c + np.log(exp_sum) / sigma

    grad = compute_grad_at_peaks(a, peak_idx, p)
    return smooth_val, grad, conv


def get_gradients_exact(a, tol_frac=0.03):
    """
    Get gradient at the exact max level with tolerance.
    Uses peaks within tol_frac of the maximum.
    Returns (max_val, gradient, full_convolution, active_indices).
    """
    conv = fast_convolve(a)
    M = conv.max()
    threshold = M * (1.0 - tol_frac)
    active = np.where(conv >= threshold)[0]

    # Weight proportional to excess over threshold
    excess = conv[active] - threshold
    if excess.max() > 0:
        w = excess / excess.sum()
    else:
        w = np.ones(len(active)) / len(active)

    grad = compute_grad_at_peaks(a, active, w)
    return M, grad, conv, active

            active = np.array([np.argmax(conv)])

        # Weights
        excess = conv[active] - threshold
        if excess.max() > 0:
            nu = excess / excess.sum()
        else:
            nu = np.ones(len(active)) / len(active)

        # Solve B @ x = 1 using CG
        # B[v] = compute_grad_at_peaks(v, active, nu)
        try:
            b_vec = np.ones(n)
            x = np.ones(n) / n  # Initial guess
            # CG iteration
            r = b_vec - hankel_matvec(x, active, nu)
            p_vec = r.copy()
            rr = np.dot(r, r)
            converged = False
            for cg_it in range(min(100, n)):
                Ap = hankel_matvec(p_vec, active, nu)
                pAp = np.dot(p_vec, Ap)
                if abs(pAp) < 1e-15:
                    break
                alpha_cg = rr / pAp
                x += alpha_cg * p_vec
                r_new = r - alpha_cg * Ap
                rr_new = np.dot(r_new, r_new)
                if rr_new < 1e-20 * rr:
                    converged = True
                    break
                p_vec = r_new + (rr_new / rr) * p_vec
                r = r_new
                rr = rr_new
        except Exception:
            x = a.copy()

        # Check and project
        x = np.maximum(x, 0.0)
        sx = x.sum()
        if sx < 1e-15:
            x = a.copy()
        else:
            x = x / sx

        # Check if this is better
        conv_x = fast_convolve(x)
        M_x = conv_x.max()

        if M_x < best_obj - 1e-14:
            best_obj = M_x
            best_a = x.copy()
            a = x
        else:
            # Blend
            a = 0.7 * a + 0.3 * x
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            conv_a = fast_convolve(a)
            if conv_a.max() < best_obj:
                best_obj = conv_a.max()
                best_a = a.copy()

        if verbose and outer_it % 20 == 0:
            print(f"    ActiveSet outer={outer_it}, obj={best_obj:.10f}, |A|={len(active)}")

    return best_a, best_obj


def fw_greedy(a0, time_limit, tol_frac=0.05, verbose=False):
    """
    Frank-Wolfe with greedy oracle (minimum gradient component).
    Fast iteration, good for quick improvements.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit - 0.3:
        it += 1

        M, grad, conv, active = get_gradients_exact(a, tol_frac=tol_frac)

        # Greedy oracle: put mass on coordinate with min gradient
        min_idx = int(np.argmin(grad))
        g = np.zeros(n)
        g[min_idx] = 1.0

        # Also try top-3 min
        top3 = np.argsort(grad)[:3]
        g3 = np.zeros(n)
        g3[top3] = 1.0 / 3.0

        # Line search
        best_alpha = 0.0
        best_val = M
        best_dir = g

        for direction in [g, g3]:
            for alpha in np.linspace(0.02, 0.8, 15):
                ft = (1.0 - alpha) * a + alpha * direction
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
                    best_dir = direction

        if best_alpha > 1e-6:
            a = (1.0 - best_alpha) * a + best_alpha * best_dir
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            val = fast_convolve(a).max()
            if val < best_obj - 1e-14:
                best_obj = val
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        if stagnant > 50:
            rng = np.random.default_rng(it * 48271)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            best_obj = fast_convolve(a).max()
            best_a = a.copy()
            stagnant = 0

        if verbose and it % 100 == 0:
            print(f"    FW-greedy it={it}, obj={best_obj:.10f}")

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_diverse_starts(n, parent, rng):
    """Generate diverse initial sequences for a given length."""
    starts = []
    x = np.arange(n, dtype=np.float64)

    # Resampled parent
    if parent is not None:
        starts.append(resample_sequence(parent, n))

    # Uniform
    starts.append(np.ones(n) / n)

    # Raised cosine
    w = 0.5 * (1.0 + np.cos(np.pi * 2.0 * x / (n - 1)))
    starts.append(w / w.sum())

    # Triangular (peak in center)
    w = 1.0 - np.abs(x - (n - 1) / 2.0) / (n / 2.0)
    w = np.maximum(w, 0.01)
    starts.append(w / w.sum())

    # Edge-weighted
    w = 1.0 / (np.minimum(x, n - 1.0 - x) + 2.0)
    starts.append(w / w.sum())

    # Two-sided (peaks at edges)
    w = np.exp(-x / (0.15 * n)) + np.exp(-(n - 1.0 - x) / (0.15 * n))
    starts.append(w / w.sum())

    # Random smooth
    for _ in range(2):
        raw = np.abs(rng.standard_normal(n))
        # Smooth it
        kernel = np.exp(-0.5 * (np.arange(-5, 6) / 2.0) ** 2)
        raw = np.convolve(raw, kernel, mode='same')
        raw = np.maximum(raw, 0.0)
        if raw.sum() > 0:
            starts.append(raw / raw.sum())

    return starts


def score_sequence(a):
    """Compute the evaluation score for a sequence."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return float(evaluate_sequence(a.tolist()))


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search for minimizing
    2*n*max(conv(a,a))/(sum(a))^2.

    Strategy:
    Phase 1: Multi-n, multi-start exploration (25% time)
    Phase 2: Deep Nesterov smoothing on best (30% time)
    Phase 3: Active-set linear refinement (20% time)
    Phase 4: FW + subgradient polishing (15% time)
    Phase 5: Final micro-optimization (10% time)
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + max(1.0, total_budget - 45.0)

    best_seq = None
    best_score = float('inf')

    # --- Load parent ---
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = score_sequence(parent)
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = score_sequence(parent)
        print(f"Using uniform n=768, score={best_score:.10f}")

    n_parent = len(parent)

    # --- Phase 1: Multi-n exploration ---
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()
    phase_time = total_budget * 0.20

    lengths = sorted(set([n_parent, max(256, n_parent - 64), min(2048, n_parent + 64),
                          512, 640, 768, 896, 1024]))
    lengths = [l for l in lengths if 128 <= l <= 2048]

    per_n_time = phase_time / max(1, len(lengths))

    for n_try in lengths:
        if time.monotonic() > deadline - total_budget * 0.7:
            break
        t_alloc = min(per_n_time, deadline - time.monotonic() - total_budget * 0.7)
        if t_alloc < 5:
            break

        starts = generate_diverse_starts(n_try, parent, rng)
        # Use top 2-3 starts per n
        for si, start in enumerate(starts[:3]):
            if time.monotonic() > deadline - total_budget * 0.7:
                break
            remaining = min(t_alloc / (3 - si), deadline - time.monotonic() - total_budget * 0.7)
            if remaining < 3:
                break
            # Quick Nesterov
            a_opt, _ = nesterov_smoothed(start, time_limit=remaining,
                                         sigma_start=20, sigma_end=5000,
                                         n_sigma_steps=5, top_k=30)
            sc = score_sequence(a_opt)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [explore n={n_try} s={si}]: {sc:.10f}")

    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # --- Phase 2: Deep Nesterov on best ---
    print("\n=== Phase 2: Deep Nesterov ===")
    t_phase = time.monotonic()
    phase_time = total_budget * 0.25

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = min(phase_time, deadline - time.monotonic() - total_budget * 0.4)
        if remaining > 10:
            print(f"  Deep Nesterov n={n_best}, time={remaining:.0f}s")
            a_opt, _ = nesterov_smoothed(best_seq, time_limit=remaining,
                                         sigma_start=20.0, sigma_end=50000.0,
                                         n_sigma_steps=15, top_k=50, verbose=True)
            sc = score_sequence(a_opt)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [deep_nesterov]: {sc:.10f}")

        # Also try on second-best n
        alt_n = n_best + 64 if n_best < 1500 else n_best - 64
        if alt_n >= 128:
            remaining = min(30, deadline - time.monotonic() - total_budget * 0.35)
            if remaining > 10:
                a_alt = resample_sequence(best_seq, alt_n)
                a_opt, _ = nesterov_smoothed(a_alt, time_limit=remaining,
                                             sigma_start=50.0, sigma_end=10000.0,
                                             n_sigma_steps=6, top_k=40)
                sc = score_sequence(a_opt)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_opt.copy()
                    print(f"  NEW BEST [alt_n={alt_n}]: {sc:.10f}")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # --- Phase 3: Active-set refinement ---
    print("\n=== Phase 3: Active-Set Refinement ===")
    t_phase = time.monotonic()
    phase_time = total_budget * 0.15

    if best_seq is not None:
        remaining = min(phase_time, deadline - time.monotonic() - total_budget * 0.25)
        if remaining > 10:
            print(f"  Active-set n={len(best_seq)}, time={remaining:.0f}s")
            a_opt, _ = active_set_linear_solve(best_seq, time_limit=remaining,
                                               tol_frac=0.02, max_iter=100, verbose=True)
            sc = score_sequence(a_opt)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [active_set]: {sc:.10f}")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # --- Phase 4: Subgradient + FW polishing ---
    print("\n=== Phase 4: Subgradient + FW ===")
    t_phase = time.monotonic()
    phase_time = total_budget * 0.15

    if best_seq is not None:
        # Exact subgradient
        remaining = min(phase_time * 0.5, deadline - time.monotonic() - total_budget * 0.15)
        if remaining > 5:
            a_opt, _ = subgradient_exact(best_seq, time_limit=remaining,
                                         tol_frac=0.02, verbose=True)
            sc = score_sequence(a_opt)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [subgrad]: {sc:.10f}")

        # FW greedy
        remaining = min(phase_time * 0.5, deadline - time.monotonic() - total_budget * 0.15)
        if remaining > 5:
            a_opt, _ = fw_greedy(best_seq, time_limit=remaining,
                                 tol_frac=0.03, verbose=True)
            sc = score_sequence(a_opt)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [fw_greedy]: {sc:.10f}")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # --- Phase 5: Final micro-optimization ---
    print("\n=== Phase 5: Final Micro-Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        polish_rounds = 0
        while time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic() - 10
            if remaining < 5:
                break
            polish_rounds += 1

            # Tiny perturbation + short Nesterov
            noise_scale = 0.0003 / (1.0 + 0.1 * polish_rounds)
            noise = rng.standard_normal(n_best) * noise_scale
            start = project_simplex(best_seq + noise)
            t_alloc = min(20.0, remaining / 2.0)
            a_opt, _ = nesterov_smoothed(start, time_limit=t_alloc,
                                         sigma_start=1000.0, sigma_end=50000.0,
                                         n_sigma_steps=4, top_k=30)
            sc = score_sequence(a_opt)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [polish_{polish_rounds}]: {sc:.10f}")

            # Quick active set
            t_alloc = min(15.0, remaining / 2.0)
            if t_alloc > 3:
                a_opt, _ = active_set_linear_solve(best_seq, time_limit=t_alloc,
                                                   tol_frac=0.01, max_iter=30)
                sc = score_sequence(a_opt)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_opt.copy()
                    print(f"  NEW BEST [polish_active_{polish_rounds}]: {sc:.10f}")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # --- Final: try a few different n values with the best shape ---
    print("\n=== Final: Cross-n Check ===")
    if best_seq is not None:
        for n_check in [len(best_seq) - 32, len(best_seq) + 32]:
            if n_check < 64 or time.monotonic() > deadline - 15:
                continue
            a_check = resample_sequence(best_seq, n_check)
            # Very short optimization
            remaining = min(15.0, deadline - time.monotonic() - 5)
            if remaining > 5:
                a_opt, _ = nesterov_smoothed(a_check, time_limit=remaining,
                                             sigma_start=200.0, sigma_end=20000.0,
                                             n_sigma_steps=4, top_k=30)
                sc = score_sequence(a_opt)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_opt.copy()
                    print(f"  NEW BEST [cross_n={n_check}]: {sc:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = score_sequence(best_seq)

    # Final verification
    fs = score_sequence(best_seq)
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
