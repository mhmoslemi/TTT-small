"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, conv, top_k, temperature):
    """
    Compute a weighted subgradient of max(conv) using top-k peaks with softmax weights.
    Returns the gradient vector.
    """
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    pv_max = pv.max()
    
    if temperature > 0:
        w = np.exp((pv - pv_max) / temperature)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / w_sum
    else:
        # Hard max
        w = np.zeros(len(peaks))
        w[pv.argmax()] = 1.0
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad[j] += w[i] * 2.0 * a[k - j]
    
    return grad

def adam_subgradient(a0, time_limit, lr=0.01, beta1=0.9, beta2=0.999,
                     top_k_start=3, top_k_end=20, temperature_init=0.001,
                     temperature_min=1e-6, sa_init_temp=0.005, sa_decay=0.999,
                     jolt_interval=500, jolt_scale=0.01):
    """
    Adam-based subgradient descent for minimizing max(auto_conv(a)).
    a is assumed to be on the simplex (sum=1, a>=0).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    temperature = temperature_init
    sa_temp = sa_init_temp
    stagnant = 0
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Adaptive top_k
        progress = min(1.0, step / 5000)
        top_k = max(top_k_start, int(top_k_start + (top_k_end - top_k_start) * progress))
        
        conv = fast_convolve(a)
        current_obj = conv.max()
        
        # Compute subgradient
        grad = compute_subgradient(a, conv, top_k, temperature)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        
        # Adaptive learning rate (decay over time)
        lr_t = lr / (1 + step * 0.0001)
        
        candidate = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        candidate = project_simplex(candidate)
        
        # Evaluate candidate
        cand_conv = fast_convolve(candidate)
        cand_obj = cand_conv.max()
        
        # Simulated annealing acceptance
        if cand_obj <= current_obj:
            a = candidate
            if cand_obj < best_obj - 1e-14:
                best_obj = cand_obj
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            # Accept with probability exp(-delta / sa_temp)
            delta = cand_obj - current_obj
            if sa_temp > 1e-8:
                accept_prob = np.exp(-delta / sa_temp)
                if np.random.random() < accept_prob:
                    a = candidate
                    stagnant += 1
                else:
                    stagnant += 1
            else:
                stagnant += 1
        
        # Anneal
        temperature *= 0.9995
        if temperature < temperature_min:
            temperature = temperature_min
        sa_temp *= sa_decay
        if sa_temp < 1e-8:
            sa_temp = 1e-8
        
        # Periodic jolt to escape local minima
        if stagnant > jolt_interval:
            jolt = np.random.standard_normal(n) * jolt_scale * min(1.0, stagnant / (2 * jolt_interval))
            a = project_simplex(best_a + jolt)
            m = np.zeros(n)
            v = np.zeros(n)
            step = 0  # Reset Adam (approximately)
            temperature = max(temperature_init * 0.5, temperature)
            sa_temp = max(sa_init_temp * 0.1, sa_temp)
            stagnant = 0
    
    return best_a, best_obj

def multi_scale_perturb(a, n_scales=5):
    """
    Generate perturbed versions of a at multiple scales using
    smoothed (low-pass filtered) random noise.
    """
    n = len(a)
    perturbations = []
    rng = np.random.default_rng()
    
    for scale in range(n_scales):
        # Create noise at different frequencies
        noise = rng.standard_normal(n)
        
        # Apply low-pass filter (more smoothing for larger scale)
        if scale > 0:
            freq = np.fft.rfftfreq(n)
            cutoff = 1.0 / (1.0 + scale * 2.0)
            mask = np.exp(-freq / cutoff)
            noise = np.fft.irfft(np.fft.rfft(noise) * mask, n)
        
        # Normalize the perturbation
        noise_norm = noise / (np.linalg.norm(noise) + 1e-15)
        perturbations.append(noise_norm)
    
    return perturbations

def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s

        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad[j] += w[i] * a[k - j]

    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k, temperature):
    """
    Compute weighted subgradient of max(conv) using top-k peaks with softmax weights.
    Returns the gradient vector with respect to a.
    """
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    pv_max = pv.max()

    if temperature > 1e-15:
        w = np.exp((pv - pv_max) / temperature)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / w_sum
    else:
        w = np.zeros(len(peaks))
        w[pv.argmax()] = 1.0

    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad[j] += w[i] * a[k - j]
            grad[k - j] += w[i] * a[j]

    return grad


def lowpass_noise(n, rng, cutoff_scale):
    """Generate low-pass filtered random noise of length n."""
    noise = rng.standard_normal(n)
    freq = np.fft.rfftfreq(n)
    mask = np.exp(-freq * cutoff_scale)
    noise = np.fft.irfft(np.fft.rfft(noise) * mask, n)
    norm = np.linalg.norm(noise)
    if norm > 1e-15:
        noise = noise / norm
    return noise


def adam_sg(a0, time_limit, lr=0.01, beta1=0.9, beta2=0.999,
            top_k_start=3, top_k_end=20, temp_init=0.001,
            temp_min=1e-7, sa_init=0.003, sa_decay=0.9995,
            jolt_interval=400, jolt_scale=0.01, verbose=False):
    """
    Adam subgradient descent minimizing max(auto_conv(a)) with SA acceptance.
    a0 should be on the simplex.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    temp = temp_init
    sa_temp = sa_init
    stagnant = 0
    rng = np.random.default_rng(step + 12345)

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        step += 1
        if time.monotonic() - t0 > time_limit - 0.03:
            break

        progress = min(1.0, step / 8000)
        tk = max(top_k_start, int(top_k_start + (top_k_end - top_k_start) * progress))

        conv = fast_convolve(a)
        current_obj = conv.max()

        grad = compute_subgradient(a, conv, tk, temp)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)

        lr_t = lr / (1.0 + 0.0002 * step)
        direction = -m_hat / (np.sqrt(v_hat) + 1e-8)

        # Backtracking line search
        best_cand = a
        best_val = current_obj
        for alpha in [lr_t, lr_t * 0.5, lr_t * 0.25, lr_t * 1.5]:
            cand = project_simplex(a + alpha * direction)
            val = fast_convolve(cand).max()
            if val < best_val - 1e-15:
                best_val = val
                best_cand = cand

        # Acceptance
        if best_val <= current_obj:
            a = best_cand
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            delta = best_val - current_obj
            if sa_temp > 1e-9 and rng.random() < np.exp(-delta / sa_temp):
                a = best_cand
                stagnant += 1
            else:
                stagnant += 1

        # Anneal
        temp *= 0.9997
        if temp < temp_min:
            temp = temp_min
        sa_temp *= sa_decay
        if sa_temp < 1e-9:
            sa_temp = 1e-9

        # Jolt to escape local minima
        if stagnant > jolt_interval:
            scale = jolt_scale * min(2.0, stagnant / jolt_interval)
            noise = lowpass_noise(n, rng, 2.0)
            a = project_simplex(best_a + noise * scale)
            m = np.zeros(n)
            v = np.zeros(n)
            temp = max(temp_init * 0.5, temp)
            sa_temp = max(sa_init * 0.2, sa_temp)
            stagnant = 0

        if verbose and step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam it={step}, obj={best_obj:.10f}, stag={stagnant}, t={elapsed:.0f}s")

    return best_a, best_obj


def peak_directed_shift(a, n_shifts=10, shift_frac=0.005, rng=None):
    """
    Directly shift mass away from peak-contributing elements.
    Returns a list of shifted candidates.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    conv = fast_convolve(a)
    candidates = []

    top_peaks = np.argsort(conv)[::-1][:5]
    for k in top_peaks:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        contributions = np.array([a[j] * a[k - j] for j in range(js, je + 1)])
        top_j = js + int(np.argmax(contributions))

        for _ in range(n_shifts // 5):
            target = int(rng.integers(0, n))
            if target == top_j or target == k - top_j:
                continue
            delta = a[top_j] * shift_frac
            cand = a.copy()
            cand[top_j] -= delta
            cand[target] += delta
            cand = np.maximum(cand, 0)
            ss = cand.sum()
            if ss > 1e-15:
                cand = cand / ss
                candidates.append(cand)

    return candidates


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search using Adam subgradient with
    simulated annealing, peak-directed shifts, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")

    n_parent = len(parent)
    print(f"Starting: n={n_parent}, best={best_score:.10f}")

    # === Phase 1: Adam SG on parent with multiple settings (350s) ===
    print("\n=== Phase 1: Adam Subgradient on Parent ===")
    t_phase = time.monotonic()

    configs = [
        {'lr': 0.01, 'top_k_start': 3, 'top_k_end': 15, 'temp_init': 0.0005,
         'sa_init': 0.003, 'jolt_interval': 300, 'jolt_scale': 0.01},
        {'lr': 0.02, 'top_k_start': 5, 'top_k_end': 30, 'temp_init': 0.001,
         'sa_init': 0.01, 'jolt_interval': 200, 'jolt_scale': 0.015},
        {'lr': 0.005, 'top_k_start': 2, 'top_k_end': 8, 'temp_init': 0.0002,
         'sa_init': 0.001, 'jolt_interval': 500, 'jolt_scale': 0.005},
    ]

    for ci, cfg in enumerate(configs):
        remaining = deadline - time.monotonic() - 800
        if remaining < 20:
            break
        t_alloc = min(remaining, 120)
        print(f"  Config {ci}: lr={cfg['lr']}, top_k={cfg['top_k_start']}-{cfg['top_k_end']}, t={t_alloc:.0f}s")
        a_opt, _ = adam_sg(parent, time_limit=t_alloc, **cfg)
        try_seq(a_opt, f"adam_c{ci}")

    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Multi-length exploration (150s) ===
    print("\n=== Phase 2: Multi-Length Exploration ===")
    t_phase = time.monotonic()

    lengths_to_try = sorted(set([
        n_parent, n_parent - 32, n_parent + 32,
        n_parent - 64, n_parent + 64, 640, 832, 896, 1024
    ]))
    lengths_to_try = [l for l in lengths_to_try if 384 <= l <= 1200]

    for n_try in lengths_to_try[:6]:
        if n_try == n_parent:
            continue
        remaining = deadline - time.monotonic() - 500
        if remaining < 15:
            break
        a_res = resample_sequence(parent, n_try)
        t_alloc = min(remaining, 40)
        print(f"  Length n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = adam_sg(a_res, time_limit=t_alloc, lr=0.01,
                           top_k_start=3, top_k_end=12,
                           temp_init=0.001, sa_init=0.005)
        try_seq(a_opt, f"mullen_{n_try}")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Deep refinement with long runs (500s) ===
    print("\n=== Phase 3: Deep Refinement ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Long slow Adam run
        remaining = deadline - time.monotonic() - 300
        if remaining > 30:
            t_alloc = min(remaining, 250)
            print(f"  Deep slow Adam n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = adam_sg(best_seq, time_limit=t_alloc,
                               lr=0.003, top_k_start=1, top_k_end=5,
                               temp_init=0.0001, temp_min=1e-9,
                               sa_init=0.0005, sa_decay=0.99995,
                               jolt_interval=1000, jolt_scale=0.003,
                               verbose=True)
            try_seq(a_opt, "deep_slow")

        # Peak-directed shifts + short Adam
        for shift_round in range(4):
            remaining = deadline - time.monotonic() - 150
            if remaining < 15:
                break
            cands = peak_directed_shift(best_seq, n_shifts=20,
                                        shift_frac=0.003 + 0.002 * shift_round, rng=rng)
            best_cand_score = best_score
            best_cand = best_seq
            for ci, cand in enumerate(cands[:5]):
                remaining = deadline - time.monotonic() - 100
                if remaining < 10:
                    break
                t_alloc = min(remaining, 25)
                a_opt, _ = adam_sg(cand, time_limit=t_alloc,
                                   lr=0.008, top_k_start=2, top_k_end=6,
                                   temp_init=0.0003, sa_init=0.001)
                sc = try_seq(a_opt, f"pdshift_{shift_round}_{ci}")
                if sc < best_cand_score:
                    best_cand_score = sc
                    best_cand = a_opt.copy()

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Alternating micro-polish (remaining) ===
    print("\n=== Phase 4: Micro-Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        it = 0
        no_improve_rounds = 0

        while time.monotonic() < deadline - 15:
            it += 1
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            score_before = best_score

            if it % 4 == 1:
                # Tiny random perturbation
                noise = rng.standard_normal(n_best) * 0.001
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 40)
                a_opt, _ = adam_sg(start, time_limit=t_alloc,
                                   lr=0.005, top_k_start=1, top_k_end=4,
                                   temp_init=0.0002, sa_init=0.0005,
                                   jolt_interval=500, jolt_scale=0.002)
                try_seq(a_opt, f"mp_rand_{it}")

            elif it % 4 == 2:
                # Low-frequency perturbation
                noise = lowpass_noise(n_best, rng, 5.0) * 0.005
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 40)
                a_opt, _ = adam_sg(start, time_limit=t_alloc,
                                   lr=0.005, top_k_start=1, top_k_end=4,
                                   temp_init=0.0002, sa_init=0.0005,
                                   jolt_interval=500, jolt_scale=0.002)
                try_seq(a_opt, f"mp_lf_{it}")

            elif it % 4 == 3:
                # Peak-directed micro-shift
                cands = peak_directed_shift(best_seq, n_shifts=5,
                                            shift_frac=0.002, rng=rng)
                if cands:
                    cand = cands[0]
                    t_alloc = min(remaining, 30)
                    a_opt, _ = adam_sg(cand, time_limit=t_alloc,
                                       lr=0.004, top_k_start=1, top_k_end=3,
                                       temp_init=0.0001, sa_init=0.0003,
                                       jolt_interval=400, jolt_scale=0.001)
                    try_seq(a_opt, f"mp_peak_{it}")

            else:
                # Slightly larger perturbation with more peaks
                noise = rng.standard_normal(n_best) * 0.003
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 35)
                a_opt, _ = adam_sg(start, time_limit=t_alloc,
                                   lr=0.008, top_k_start=2, top_k_end=8,
                                   temp_init=0.0005, sa_init=0.002,
                                   jolt_interval=300, jolt_scale=0.004)
                try_seq(a_opt, f"mp_med_{it}")

            if best_score < score_before - 1e-12:
                no_improve_rounds = 0
            else:
                no_improve_rounds += 1

            if no_improve_rounds > 20 and it > 30:
                break

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
