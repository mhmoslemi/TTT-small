Uses Adam optimizer with adaptive temperature, subgradient methods,

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def peak_info(a):
    """Return (max_value, peak_lag) of autocorrelation."""
    conv = fast_convolve(a)
    max_val = conv.max()
    peak_lag = int(np.argmax(conv))
    return max_val, peak_lag


def subgradient_at_peak(a, peak_lag):
    """Compute subgradient of max autocorrelation at the given peak lag."""
    n = len(a)
    k = peak_lag
    # grad[j] = 2 * a[k-j] if 0 <= k-j <= n-1, else 0
    grad = np.zeros(n)
    # k-j is valid when 0 <= k-j <= n-1, i.e., k-n+1 <= j <= k
    j_start = max(0, k - n + 1)
    j_end = min(k, n - 1)
    for j in range(j_start, j_end + 1):
        grad[j] = 2.0 * a[k - j]
    return grad


def smooth_weighted_grad(a, T):
    """Compute gradient of smoothed max (log-sum-exp) approximation."""
    n = len(a)
    conv = fast_convolve(a)
    shifted = conv / T
    shifted -= shifted.max()
    p = np.exp(shifted)
    p = p / p.sum()
    
    # grad[j] = 2 * sum_m a[m] * p[j+m] for valid j+m
    # Compute via cross-correlation
    ar = a[::-1]
    fft_size = 1 << (n + (2*n-1) - 1).bit_length()
    AR = np.fft.rfft(ar, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_result = np.fft.irfft(AR * P, fft_size)
    grad = 2.0 * conv_result[n-1:n-1+n]
    return grad, p


class Adam:
    def __init__(self, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.m = None
        self.v = None
        self.t = 0
    
    def step(self, params, grad):
        if self.m is None:
            self.m = np.zeros_like(params)
            self.v = np.zeros_like(params)
        self.t += 1
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * (grad ** 2)
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)
        return params - self.lr * m_hat / (np.sqrt(v_hat) + self.eps)

"""
Multi-strategy optimization for minimum peak autocorrelation step function.
Uses Adam optimizer with adaptive temperature annealing, game-theoretic dual ascent,
subgradient methods, coordinate descent, and multi-n exploration.
Key innovations: peak-equalization loss, geometric temperature cooling,
and efficient FFT-based gradient computation.
"""
import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(a, p):
    """Compute gradient of sum_k p[k]*(a*a)[k] w.r.t. a using FFT."""
    n = len(a)
    ar = a[::-1]
    fft_size = 1 << (n + (2 * n - 1) - 1).bit_length()
    AR = np.fft.rfft(ar, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_result = np.fft.irfft(AR * P, fft_size)
    return 2.0 * conv_result[n - 1:n - 1 + n]


def softmax_weights(conv, T):
    """Compute softmax weights over convolution values."""
    shifted = conv / T
    shifted -= shifted.max()
    p = np.exp(shifted)
    s = p.sum()
    if s < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return p / s


def adam_smooth_optimize(a0, time_limit, T_init=0.01, T_end=0.0001, lr_init=0.01,
                         use_peak_eq=False, peak_eq_alpha=0.1, top_K=20):
    """Adam optimizer with adaptive temperature and optional peak equalization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_count = 0
    lr = lr_init

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    T = T_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        p = softmax_weights(conv, T)

        # Base gradient
        grad = compute_smooth_grad(a, p)

        # Optional peak equalization: add penalty for spread of top peaks
        if use_peak_eq and it % 5 == 0:
            sorted_idx = np.argsort(conv)[::-1]
            top_peaks = conv[sorted_idx[:top_K]]
            peak_spread = top_peaks.max() - top_peaks.min()
            if peak_spread > 1e-15:
                # Gradient of peak spread: push top up, push bottom of top-K down
                eq_weight = np.zeros(len(conv))
                eq_weight[sorted_idx[0]] = peak_eq_alpha
                eq_weight[sorted_idx[min(top_K, len(conv) - 1)]] = -peak_eq_alpha
                eq_grad = compute_smooth_grad(a, eq_weight)
                grad = grad + eq_grad

        # Adam update
        t_count += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        # Evaluate
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adapt temperature (geometric cooling)
        T = max(T * 0.997, T_end)

        # Adapt learning rate
        if no_improve > 30:
            lr *= 0.97
        if no_improve > 100:
            lr *= 0.95
        if no_improve > 300:
            # Restart with perturbation
            rng = np.random.default_rng((it * 104729 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            m = np.zeros(n)
            v = np.zeros(n)
            t_count = 0
            lr = lr_init * 0.7
            T = T_init * 0.4
            no_improve = 0

        if no_improve > 800:
            break

    return best_a, best_obj


def game_theoretic_optimize(a0, time_limit, T_init=0.005, lr_start=0.02,
                            inner_steps=3, dual_lr=0.1):
    """
    Game-theoretic optimization: alternate between primal (Adam on QP subproblem)
    and dual (multiplicative weight updates on peak weights).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv_len = 2 * n - 1

    # Dual variables (weights on each autocorrelation lag)
    w = np.ones(conv_len) / conv_len

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    T = T_init
    lr = lr_start

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        # Update dual weights (multiplicative weights / exponentiated gradient)
        w = w * np.exp(dual_lr * conv / T)
        w = w / w.sum()

        # Inner loop: primal update (few Adam steps on weighted objective)
        for _ in range(inner_steps):
            grad = compute_smooth_grad(a, w)
            a_new = a - lr * grad
            a_new = project_simplex(a_new)
            new_weighted = np.dot(w, fast_convolve(a_new))
            old_weighted = np.dot(w, conv)
            if new_weighted < old_weighted:
                a = a_new
                conv = fast_convolve(a)
            else:
                break

        # Evaluate true objective
        obj = conv.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt
        if no_improve > 20:
            T *= 1.02
        if no_improve > 80:
            T *= 0.8
            lr *= 0.95
        if no_improve > 250:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            w = np.ones(conv_len) / conv_len
            T = T_init
            lr = lr_start
            no_improve = 0

        if no_improve > 600:
            break

    return best_a, best_obj


def subgradient_optimize(a0, time_limit, lr_init=0.01):
    """Subgradient method with Polyak step size and randomized peak selection."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lb = best_obj * 0.95  # crude lower bound

    t0 = time.monotonic()
    no_improve = 0
    lr = lr_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        max_val = conv.max()

        # Randomized subgradient: pick from top peaks with some probability
        top_threshold = max_val * 0.99
        top_lags = np.where(conv >= top_threshold)[0]
        if len(top_lags) > 0:
            rng = np.random.default_rng((it * 31 + 7) % (2**31 - 1))
            peak_lag = int(top_lags[rng.integers(0, len(top_lags))])
        else:
            peak_lag = int(np.argmax(conv))

        # Compute subgradient at this peak
        k = peak_lag
        grad = np.zeros(n)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            grad[j_start:j_end + 1] = 2.0 * a[k - j_end:k - j_start + 1][::-1]

        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-20:
            break

        # Polyak step size
        polyak_lr = (max_val - lb) / (grad_norm_sq + 1e-15)
        step = min(lr, max(polyak_lr, 1e-8))

        a_new = a - step * grad
        a_new = project_simplex(a_new)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            lb = best_obj * 0.98
            no_improve = 0
            lr = min(lr * 1.05, 0.1)
        else:
            no_improve += 1
            lr *= 0.98

        a = a_new

        if no_improve > 300:
            rng = np.random.default_rng((it * 104729 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            lr = lr_init
            no_improve = 0

        if no_improve > 800:
            break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, seed=42):
    """Coordinate descent with pair moves and triple moves."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        improved = False

        # Pair moves
        for _ in range(30):
            if time.monotonic() - t0 > time_limit - 0.02:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue

            for frac in [0.02, 0.05, 0.1, 0.2, 0.3]:
                # Move i -> j
                amt = a[i] * frac
                if amt < 1e-15:
                    continue
                cand = a.copy()
                cand[i] -= amt
                cand[j] += amt
                c = fast_convolve(cand)
                if c.max() < best_obj - 1e-15:
                    a = cand
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break

                # Move j -> i
                amt = a[j] * frac
                if amt < 1e-15:
                    continue
                cand = a.copy()
                cand[j] -= amt
                cand[i] += amt
                c = fast_convolve(cand)
                if c.max() < best_obj - 1e-15:
                    a = cand
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                stall = 0
                break

        # Triple moves (spread mass from one to two)
        if not improved and it % 3 == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            k = int(rng.integers(0, n))
            if len(set([i, j, k])) == 3 and a[i] > 1e-10:
                for frac1 in [0.1, 0.2, 0.3]:
                    amt1 = a[i] * frac1
                    amt2 = amt1 * 0.5
                    cand = a.copy()
                    cand[i] -= amt1
                    cand[j] += amt2
                    cand[k] += amt1 - amt2
                    if cand[i] >= -1e-15:
                        cand = np.maximum(cand, 0)
                        cand /= cand.sum()
                        c = fast_convolve(cand)
                        if c.max() < best_obj - 1e-15:
                            a = cand
                            best_obj = c.max()
                            best_a = a.copy()
                            improved = True
                            break

        if not improved:
            stall += 1
        if stall > 3000:
            break

    return best_a, best_obj


def multi_scale_optimize(a0, time_limit):
    """Progressive refinement: start coarse (high T), end fine (low T)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()

    # Temperature levels (log-spaced)
    temps = np.logspace(np.log10(0.1), np.log10(0.00005), 12)

    for T in temps:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        t_alloc = min(remaining * 0.45, 120)
        a_opt, obj = adam_smooth_optimize(a, time_limit=t_alloc,
                                          T_init=float(T), T_end=float(T * 0.3),
                                          lr_init=0.008, use_peak_eq=True)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_opt.copy()
            a = a_opt.copy()

    return best_a, best_obj


def try_seq(a, label, best_ref):
    """Evaluate sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_ref
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_ref[1]:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return (a_norm.copy(), score)
    return best_ref

        print(f"Using uniform init: score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Multi-n Exploration ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    n_candidates = [256, 384, 512, 768, 1024, 1280, 1536, 2048, 3072]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates.sort()

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 150:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try

        # Quick multi-scale
        t_alloc = min(remaining * 0.05, 30)
        a_opt, _ = multi_scale_optimize(start, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, f"ms_n{n_try}", (best_seq, best_score))

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep Multi-scale + Game ===
    print("\n=== Phase 2: Deep Optimization ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    # Multi-scale Adam
    if remaining > 200:
        t_alloc = min(remaining * 0.35, 500)
        print(f"  Multi-scale Adam n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = multi_scale_optimize(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, "deep_ms", (best_seq, best_score))

    # Game-theoretic
    remaining = deadline - time.monotonic()
    if remaining > 150:
        t_alloc = min(remaining * 0.2, 250)
        print(f"  Game theoretic n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = game_theoretic_optimize(best_seq, time_limit=t_alloc,
                                           T_init=0.005, lr_start=0.015,
                                           inner_steps=3, dual_lr=0.08)
        best_seq, best_score = try_seq(a_opt, "game", (best_seq, best_score))

    # Adam with different hyperparameters
    for lr, T_init in [(0.003, 0.002), (0.02, 0.005), (0.005, 0.001)]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        n_best = len(best_seq)
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.004)
        t_alloc = min(remaining * 0.15, 130)
        a_opt, _ = adam_smooth_optimize(f_pert, time_limit=t_alloc,
                                        T_init=T_init, T_end=T_init * 0.1,
                                        lr_init=lr, use_peak_eq=True)
        best_seq, best_score = try_seq(a_opt, f"adam_lr{lr}_T{T_init}", (best_seq, best_score))

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Subgradient + Coordinate Descent ===
    print("\n=== Phase 3: Subgradient + Coord ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()

    if remaining > 150:
        t_alloc = min(remaining * 0.2, 200)
        print(f"  Subgradient, t={t_alloc:.0f}s")
        a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc, lr_init=0.005)
        best_seq, best_score = try_seq(a_opt, "subgrad", (best_seq, best_score))

    remaining = deadline - time.monotonic()
    if remaining > 100:
        t_alloc = min(remaining * 0.3, 250)
        print(f"  Coordinate descent, t={t_alloc:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc, seed=42)
        best_seq, best_score = try_seq(a_opt, "coord1", (best_seq, best_score))

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Alternative n ===
    print("\n=== Phase 4: Alt n Exploration ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    alt_ns = set()
    for mult in [0.75, 1.25, 1.5, 2.0]:
        alt_n = int(n_best * mult)
        if 128 <= alt_n <= 5000:
            alt_ns.add(alt_n)
    for delta in [64, 128, 256]:
        if 128 <= n_best + delta <= 5000:
            alt_ns.add(n_best + delta)
        if 128 <= n_best - delta <= 5000:
            alt_ns.add(n_best - delta)
    alt_ns.discard(n_best)
    alt_ns = sorted(alt_ns)

    for n_try in alt_ns[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        start = np.interp(np.linspace(0, 1, n_try),
                          np.linspace(0, 1, n_best), best_seq)
        start = np.maximum(start, 0)
        ss = start.sum()
        start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try

        t_alloc = min(remaining * 0.2, 120)
        a_opt, _ = multi_scale_optimize(start, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, f"alt_n{n_try}", (best_seq, best_score))

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final Polish ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    # Very fine Adam
    if remaining > 30:
        t_alloc = min(remaining * 0.3, 100)
        a_opt, _ = adam_smooth_optimize(best_seq, time_limit=t_alloc,
                                        T_init=0.0005, T_end=0.00003,
                                        lr_init=0.003, use_peak_eq=True)
        best_seq, best_score = try_seq(a_opt, "fine_adam", (best_seq, best_score))

    # Fine coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_alloc = min(remaining * 0.25, 80)
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc, seed=777)
        best_seq, best_score = try_seq(a_opt, "fine_coord", (best_seq, best_score))

    # Game with very small T
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_alloc = min(remaining * 0.2, 60)
        a_opt, _ = game_theoretic_optimize(best_seq, time_limit=t_alloc,
                                           T_init=0.001, lr_start=0.005,
                                           inner_steps=5, dual_lr=0.05)
        best_seq, best_score = try_seq(a_opt, "game_fine", (best_seq, best_score))

    # Micro perturbations
    remaining = deadline - time.monotonic()
    for i in range(8):
        if time.monotonic() > deadline - 5:
            break
        n_best = len(best_seq)
        noise = rng.standard_normal(n_best) * (0.0005 + 0.001 * i)
        f_pert = project_simplex(best_seq + noise)
        t_alloc = max(2.0, min(deadline - time.monotonic() - 3, 8))
        if t_alloc > 2:
            a_opt, _ = adam_smooth_optimize(f_pert, time_limit=t_alloc,
                                            T_init=0.0003, T_end=0.0001,
                                            lr_init=0.002)
            best_seq, best_score = try_seq(a_opt, f"micro_{i}", (best_seq, best_score))

    # Final coordinate pass
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_opt, _ = coordinate_descent(best_seq, time_limit=min(remaining - 2, 25), seed=31415)
        best_seq, best_score = try_seq(a_opt, "final_coord", (best_seq, best_score))

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
