3. Solve LP: max 1^T g s.t. Cg ≤ M_0 * 1, g ≥ 0. Get g*.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using full-step Frank-Wolfe with LP oracle,
Nesterov-accelerated subgradient, and peak-shaving coordinate descent.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (f*g)[k] = sum_j A[k,j]*g[j]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A


def solve_lp_oracle(f, M, n):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def full_step_fw(f0, time_limit, verbose=False, m_relax=1.0):
    """Frank-Wolfe with full steps (no line search)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    stagnant = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        g = solve_lp_oracle(f, M * m_relax, n)

        if g is None or g.sum() < 1e-15:
            g = solve_lp_oracle(f, M * 1.01, n)

        if g is None or g.sum() < 1e-15:
            stagnant += 1
            if stagnant > 30:
                return best_f, best_obj
            continue

        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-12:
            stagnant += 1
            if stagnant > 50:
                return best_f, best_obj
            continue

        # Full step
        f = g
        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-15:
            best_obj = M
            best_f = f.copy()
            stagnant = 0
        else:
            stagnant += 1
            # If stagnant, try a blend
            if stagnant % 10 == 0:
                f = 0.9 * best_f + 0.1 * g
                f = np.maximum(f, 0)
                f = f / f.sum()
                conv = fast_convolve(f)
                M = conv.max()

        if verbose and it % 200 == 0:
            print(f"    FW-FS it={it}, obj={best_obj:.10f}, stagnant={stagnant}")

    return best_f, best_obj


def nesterov_subgradient(f0, time_limit, T_init=0.0003, lr_init=0.05, top_k=10):
    """Nesterov-accelerated subgradient descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    # Nesterov momentum
    y = f.copy()  # extrapolated point
    v = np.zeros(n)
    beta = 1.0  # will use (1+t)/(2+t) schedule

    t0 = time.monotonic()
    it = 0
    T = T_init
    lr = lr_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv_y = fast_convolve(y)

        # Top-k peaks
        if top_k < len(conv_y):
            peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv_y))

        pv = conv_y[peaks]
        pv_max = pv.max()
        w = np.exp((pv - pv_max) / (T * 0.1))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / w_sum

        # Subgradient
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += w[i] * 2.0 * y[k - idx_arr]

        # Nesterov step
        lr_t = lr / (1.0 + 0.01 * it)
        v = 0.9 * v - lr_t * grad
        y_new = f + v
        y_new = np.maximum(y_new, 0)
        y_new = y_new / y_new.sum()

        # Project f
        f_new = np.maximum(f + 0.5 * (y_new - f), 0)
        f_new = f_new / f_new.sum()

        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()

        f = f_new
        y = y_new
        T = T_init * (1.0 + it / 1000)

    return best_f, best_obj
