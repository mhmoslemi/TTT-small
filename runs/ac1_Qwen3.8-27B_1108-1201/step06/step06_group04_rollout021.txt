        # This is 2 * (p * a)[j] where * is cross-correlation (not convolution)
        # cross-corr(p, a)[j] = sum_m p[j+m] * a[m]

        # But p has length 2n-1 and a has length n
        # Let's pad a to length 2n-1
        a_pad = np.zeros(len(p))
        a_pad[:n] = a
        fft_size = 1 << (len(p) + n - 1).bit_length()
        P = np.fft.rfft(p, fft_size)
        A = np.fft.rfft(a_pad, fft_size)
        # cross-correlation: IFFT(conj(P) * A) gives sum_m p[m] * a_pad[m+j]
        # Hmm, let me think...
        # np.correlate(p, a_pad, 'valid') gives sum_m p[m] * a_pad[m+j] for j=0..len(p)-n
        # We want sum_m p[j+m] * a[m] for j=0..n-1
        # = sum_m p[j+m] * a_pad[m]
        # = cross-correlation of p and a_pad at lag j

        # np.correlate(a, v, 'valid')[k] = sum_i a[i+k] * v[i]

        # np.correlate(p, a, 'valid')[j] = sum_i p[i+j] * a[i] = sum_i a[i] * p[i+j]
        # Yes! That's exactly what we want.
        
        grad_raw = np.correlate(p, a, mode='valid')  # length n
        grad = 2.0 * grad_raw
        
        # Mirror descent update (exponentiated gradient)
        a_new = a * np.exp(-lr * grad)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Also clip (mirror descent should keep positive, but numerically)
        a_new = np.maximum(a_new, 0.0)
        s_new2 = a_new.sum()
        if s_new2 > 1e-15:
            a_new = a_new / s_new2
        
        obj = compute_objective(a_new)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            lr = min(lr * 1.01, 0.1)  # slightly increase step

            no_improve += 1
            lr *= 0.95
            if no_improve > 20:
                lr *= 0.8
            if no_improve > 100:
                # Restart with perturbation
                rng = np.random.default_rng((it * 7919 + 13) % (2**31))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                lr = lr_init
                T = T_init
                no_improve = 0
                continue
        
        # Adapt temperature
        if no_improve > 30:
            T *= 1.2
        if no_improve > 100:
            T *= 0.7
        
        a = a_new
        
        if verbose and it % 100 == 0:
            print(f"    MD it={it}, obj={best_obj:.10f}, T={T:.6f}, lr={lr:.6f}")
    
    return best_a

def coordinate_descent(a0, time_limit, verbose=False):
    """
    Cyclic coordinate descent. For each coordinate, try several values
    and pick the one that minimizes the objective.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = compute_objective(a)
    
    t0 = time.monotonic()
    full_pass = 0
    no_improve_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        
        full_pass += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit - 1.0:
                break
            
            old_val = a[j]
            if old_val < 1e-15:
                # Try setting it to a small positive value
                candidates = [old_val, max(0.001 / n, old_val * 2)]
            else:
                # Try several multipliers
                candidates = [0.0, old_val * 0.25, old_val * 0.5, old_val * 0.75,
                              old_val, old_val * 1.25, old_val * 1.5, old_val * 2.0]
                # Also try absolute values based on the average
                avg = 1.0 / n
                candidates.extend([avg * 0.5, avg * 2.0])
            
            best_candidate_val = old_val
            best_candidate_obj = best_obj  # current objective
            # Actually, compute current objective with old_val
            # (it should be best_obj if a hasn't changed, but let's be safe)
            
            for cand in candidates:
                cand = max(0.0, cand)
                a[j] = cand
                # Compute new sum
                s_new = a.sum()
                if s_new < 1e-15:
                    a[j] = old_val
                    continue
                # Compute new convolution efficiently
                # Only lags k in [j, j+n-1] are affected
                conv = fast_convolve(a)
                obj = 2.0 * n * conv.max() / (s_new * s_new)
                
                if obj < best_candidate_obj - 1e-14:
                    best_candidate_obj = obj
                    best_candidate_val = cand
            
            a[j] = best_candidate_val
            
            if best_candidate_val != old_val and best_candidate_obj < best_obj - 1e-14:
                improved = True
                best_obj = best_candidate_obj
                best_a = a.copy()
        
        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                # Try perturbation
                rng = np.random.default_rng((full_pass * 104729 + 7) % (2**31))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                best_obj = compute_objective(a)
                if best_obj < best_obj:  # never true, just for clarity
                    pass
                no_improve_passes = 0
        else:
            no_improve_passes = 0
        
        if verbose and full_pass % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={full_pass}, obj={best_obj:.10f}, elapsed={elapsed:.0f}s")
    
    return best_a

def spectral_peak_reduction(a0, time_limit, verbose=False):
    """
    Frequency-domain method: iteratively flatten the power spectrum
    to reduce auto-convolution peaks.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = compute_objective(a)
    
    t0 = time.monotonic()
    it = 0
    alpha = 0.5  # blending parameter
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        # Compute current auto-convolution
        conv = fast_convolve(a)
        cmax = conv.max()
        cmin = conv[conv > 0].min() if (conv > 0).any() else 0
        
        # Compute power spectrum
        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        P = A.real**2 + A.imag**2  # |A|^2
        
        # Compute target flat spectrum
        # If conv were perfectly flat, P would be flat
        # Target: P_flat = (sum(a))^2 * fft_size / (2n-1) / fft_size = (sum(a))^2 / (2n-1)
        # Actually: sum(conv) = (sum(a))^2, and sum(P) = fft_size * (sum(a))^2 / 2 (by Parseval for real)
        # So flat P would be: P_flat = (sum(a))^2 / (2n-1) at each of the 2n-1 "effective" positions
        # In FFT domain: P_flat[k] = (sum(a))^2 * fft_size / (2n-1) for all k? No...
        # By Parseval: sum_k P[k] / fft_size = sum_t x[t]^2 = ... hmm this is for energy
        # Let me just use: flat target = mean of current P
        P_mean = P.mean()
        
        # Blend current spectrum toward flat
        P_new = (1 - alpha) * P + alpha * P_mean
        
        # Take "square root" to get new Fourier coefficients (preserve phase)
        A_abs = A.real**2 + A.imag**2
        A_abs = np.sqrt(A_abs)
        phase = A / (A_abs + 1e-30)
        
        # New amplitude: sqrt of new power spectrum
        A_new_mag = np.sqrt(P_new)
        A_new = A_new_mag * phase
        
        # Inverse FFT
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Project to non-negative, normalize
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        obj = compute_objective(a_new)
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            alpha = min(alpha * 1.02, 0.9)
        else:
            # Revert and try smaller alpha
            alpha = max(alpha * 0.8, 0.05)
        
        if it % 50 == 0 and verbose:
            print(f"    SPKR it={it}, obj={best_obj:.10f}, alpha={alpha:.3f}")
        
        if it > 5000:
            break
    
    return best_a

def hankel_correlate(p, a):
    """Compute grad[j] = sum_m p[j+m] * a[m] for j=0..n-1."""
    return np.correlate(p, a, mode='valid')

def subgradient_step(a, k_peak):
    """
    Compute the subgradient at a specific peak lag.
    grad[j] = 2 * a[k_peak - j] if 0 <= k_peak - j <= n-1, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    # k_peak - j in [0, n-1] means j in [k_peak - (n-1), k_peak]
    js = max(0, k_peak - (n - 1))
    je = min(n - 1, k_peak)
    for j in range(js, je + 1):
        grad[j] = 2.0 * a[k_peak - j]
    return grad

def multi_peak_subgradient(a, peaks):
    """Average subgradient over multiple peak lags."""
    n = len(a)
    grad = np.zeros(n)
    for k in peaks:
        grad += subgradient_step(a, k)
    grad /= len(peaks)
    return grad

def accelerated_subgrad(a0, time_limit, verbose=False):
    """
    Nesterov-accelerated subgradient method with adaptive step size.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    v = np.zeros(n)
    t = 1.0
    lr = 0.05
    best_a = a.copy()
    best_obj = compute_objective(a)
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Nesterov lookahead
        y = a + (t - 1) / (t + 1) * v  # Actually, standard Nesterov: y = a + momentum * v
        # Let me use the simpler form: y = a + beta * v
        beta = (t - 1) / (t + 1)
        y = a + beta * v
        
        # Find peak lags
        conv_y = fast_convolve(y)
        cmax = conv_y.max()
        # Get all lags within 10% of max
        threshold = cmax * 0.9
        peaks = np.where(conv_y >= threshold)[0]
        
        # Subgradient
        grad = multi_peak_subgradient(y, peaks)
        
        # Step
        step = lr / (1.0 + 0.1 * t)
        y_new = project_simplex(y - step * grad)
        v = y_new - a
        a = y_new
        t += 1
        
        obj = compute_objective(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            lr *= 0.9
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            t = 1
            lr = 0.05
            no_improve = 0
        
        if verbose and it % 200 == 0:
            print(f"    AccSG it={it}, obj={best_obj:.10f}, t={t:.0f}")
    
    return best_a
