import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]."""
    return np.correlate(p, a, mode='valid')


def solve_lp_oracle(f, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-7,
                               'primal_feasibility_tolerance': 1e-7})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
        elif res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def solve_lp_oracle_greedy(f, M, n_iter=200):
    """Approximate LP oracle using a greedy coordinate ascent approach."""
    n = len(f)
    g = np.zeros(n)
    conv_f = fast_convolve(f)
    
    for _ in range(n_iter):
        # Compute current f*g
        fg = np.convolve(f, g, mode='full')
        # Find slack at each position
        slack = M - fg
        # For each j, max increase is min over k of slack[k] / f[k-j] where k-j in [0,n-1]
        best_j = -1
        best_delta = 0
        for j in range(n):
            # Positions k where f[k-j] appears in the convolution
            # fg[k] = sum_i f[i] * g[k-i], so changing g[j] affects fg[k] by f[k-j]*delta
            # k-j must be in [0, n-1], so k in [j, j+n-1]
            k_lo = j
            k_hi = min(j + n - 1, 2 * n - 2)
            f_slice = f[k_lo - j:k_hi - j + 1]  # f[0:n] essentially
            # Actually f[k-j] for k in [j, j+n-1] is f[0], f[1], ..., f[n-1]
            f_slice = f[0:n]
            positive = f_slice > 1e-15
            if not positive.any():
                continue
            slack_slice = slack[k_lo:k_hi + 1]
            ratios = np.full_like(f_slice, np.inf)
            ratios[positive] = slack_slice[positive] / f_slice[positive]
            max_delta = max(ratios.min(), 0.0)
            if max_delta > best_delta:
                best_delta = max_delta
                best_j = j
        if best_j < 0 or best_delta < 1e-12:
            break
        g[best_j] += best_delta
    
    return g

def coordinate_descent_polish(a0, time_limit):
    """Coordinate descent: move mass from high-contribution pairs to reduce peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)

    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    it = 0

        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Find the pair contributing most to this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]

        # Find where to move mass: minimize impact on other peaks
        # Try moving to a position that's far from the peak
        center = n // 2
        dist = np.abs(np.arange(n) - center)
        # Prefer positions far from current high values
        scores = dist - 10.0 * a
        j_target = int(np.argmin(a[j_max * scores + np.minimum(a, 0.001) * dist]))
        # Simpler: just pick the minimum a outside the affected range
        outside_mask = np.ones(n, dtype=bool)
        outside_mask[affected] = False
        if outside_mask.sum() > 0:
            outside_idx = np.where(outside_mask)[0]
            j_min = outside_idx[np.argmin(a[outside_idx])]
        else:
            j_min = int(np.argmin(a))

        improved = False
        for delta in [0.0001, 0.0003, 0.001, 0.003, 0.01]:
            if a[j_max] < delta:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min] += delta
            candidate = np.maximum(candidate, 0.0)
            s = candidate.sum()
            candidate = candidate / s
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:

                best_obj = c.max()
                best_a = a.copy()

                delta = a[j_src] * 0.1
                candidate = a.copy()
                candidate[j_src] -= delta
                candidate[j_dst] += delta
                candidate = np.maximum(candidate, 0.0)
                candidate = candidate / candidate.sum()
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stale = 0
        else:

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = 2.0 * hankel_mv(p, a)

        best_candidate = a
        best_val = conv.max()

        for trial_lr in [lr, lr * 0.5, lr * 0.2, lr * 0.1, lr * 2.0]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            T *= 1.03
        if no_improve > 40:
            T *= 0.7
        if no_improve > 100:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0

        lr *= 0.999

    return best_a, best_obj


def multi_peak_subgradient(a0, time_limit, top_k=20, lr_init=0.002, mom_init=0.93):
    """Multi-peak subgradient with momentum and adaptive learning rate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = mom_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        # Weighted peaks
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]

        v = mom * v - lr * sg
        a = project_simplex(a + v)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        mom = min(0.98, mom + 0.00005)

        if no_improve > 150:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.002 + 0.004 * min(1.0, (no_improve - 150) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init * 0.7
            no_improve = 0

    return best_a, best_obj


def generate_structured_init(n, mode, rng):
    """Generate diverse structured initializations."""
    x = np.arange(n, dtype=np.float64)
    
    if mode == 'uniform':
        return np.ones(n) / n
    
    elif mode == 'edge':
        d = np.minimum(x, n - 1 - x) + 1.0
        w = 1.0 / d
        return w / w.sum()
    
    elif mode == 'gaussian':
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.1 * n)) ** 2)
        return g / g.sum()
    
    elif mode == 'cosine':
        for freq in [2, 3, 5]:
            h = 1.0 + 0.2 * np.cos(2.0 * np.pi * freq * x / n)
            h = np.maximum(h, 0.01)
            return h / h.sum()
        return np.ones(n) / n
    
    elif mode == 'sawtooth':
        h = (x % (n // 4 + 1)) / (n // 4 + 1)
        h = 1.0 - 0.3 * h
        return h / h.sum()
    
    elif mode == 'random':
        return rng.dirichlet(np.ones(n) * 0.5)
    
    elif mode == 'bipsplit':
        # Two equal blocks
        a = np.zeros(n)
        a[:n//2] = 1.0
        a[n//2:] = 0.3
        return a / a.sum()
    
    elif mode == 'periodic':
        period = max(3, n // 8)
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if (i % period) < period // 2 else 0.3
        return h / h.sum()
    
    else:
        return np.ones(n) / n


def evaluate_obj(a):
    """Compute the objective value directly (faster than calling evaluate_sequence)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def try_and_update(a, state, label):
    """Evaluate and update state if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = evaluate_obj(a_norm)
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a_norm)))
        # Double check with actual evaluator
        real_score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(real_score) and real_score < state[0]:
            state[0] = real_score
        elif np.isfinite(real_score):
            state[0] = max(state[0], real_score)  # Ensure consistency
            # Actually use the real score
            if real_score < state[0] + 1e-10:
                state[0] = real_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid search with Nesterov FW, coordinate descent,
    and adaptive n selection.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    state = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.8f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    n_parent = len(parent)

    # === Phase 1: Quick n exploration (~300s) ===
    print("\n=== Phase 1: Quick n Exploration ===")
    t_phase = time.monotonic()

    # Try a range of n values, including those near the parent
    n_candidates = sorted(set([
        n_parent,
        384, 512, 640, 768, 896, 1024, 1152, 1280, 1408, 1536, 1664, 1792, 2048
    ]))
    n_candidates = [n for n in n_candidates if 256 <= n <= 2560]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 20:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss > 1e-10:
                start = start / ss
            else:
                start = np.ones(n_try) / n_try

        # Quick optimization
        t_alloc = min(remaining, 25)
        a_opt, _ = standard_fw_lp(start, time_limit=t_alloc)
        try_and_update(a_opt, state, "fw_n{}".format(n_try))

        # Quick gradient polish
        remaining = deadline - time.monotonic() - 1150
        if remaining > 8:
            a_opt2, _ = smoothed_gradient_descent(a_opt, time_limit=min(remaining, 8),
                                                   T_init=0.0005, lr_init=0.02)
            try_and_update(a_opt2, state, "sg_n{}".format(n_try))

    print("  Phase 1 best: {:.8f} (n={}) elapsed {:.0f}s".format(
        state[0], len(state[1]) if state[1] is not None else 0, time.monotonic() - t_phase))

    # === Phase 2: Deep optimization on best n (~600s) ===
    print("\n=== Phase 2: Deep FW on best n ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])
        
        # Nesterov FW (main workhorse)
        remaining = deadline - time.monotonic() - 500
        if remaining > 30:
            t_alloc = min(remaining, 250)
            print("  Nesterov FW n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = nesterov_fw_lp(state[1], time_limit=t_alloc, use_greedy=False)
            try_and_update(a_opt, state, "nest_fw")

        # Standard FW
        remaining = deadline - time.monotonic() - 400
        if remaining > 20:
            t_alloc = min(remaining, 100)
            print("  Standard FW n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = standard_fw_lp(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "std_fw")

        # Perturbed restarts
        for scale in [0.002, 0.005, 0.01, 0.02]:
            remaining = deadline - time.monotonic() - 300
            if remaining < 15:
                break
            f_pert = project_simplex(state[1] + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 40)
            a_opt, _ = nesterov_fw_lp(f_pert, time_limit=t_alloc)
            try_and_update(a_opt, state, "pert_{}".format(scale))

    print("  Phase 2 best: {:.8f} (n={}) elapsed {:.0f}s".format(
        state[0], len(state[1]) if state[1] is not None else 0, time.monotonic() - t_phase))

    # === Phase 3: Alternative n values with deep optimization (~300s) ===
    print("\n=== Phase 3: Alternative n deep search ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])
        # Try neighbors of best n
        alt_ns = [n for n in [n_best - 128, n_best - 64, n_best + 64, n_best + 128,
                              896, 1024, 1280, 1536]
                  if n != n_best and 256 <= n <= 2560]
        alt_ns = sorted(set(alt_ns))[:4]

        for n_try in alt_ns:
            remaining = deadline - time.monotonic() - 200
            if remaining < 20:
                break
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_best), state[1])
            start = np.maximum(start, 0)
            start = start / start.sum()
            t_alloc = min(remaining, 40)
            a_opt, _ = nesterov_fw_lp(start, time_limit=t_alloc)
            try_and_update(a_opt, state, "alt_n{}".format(n_try))

    print("  Phase 3 best: {:.8f} (n={}) elapsed {:.0f}s".format(
        state[0], len(state[1]) if state[1] is not None else 0, time.monotonic() - t_phase))

    # === Phase 4: Gradient methods (~250s) ===
    print("\n=== Phase 4: Gradient Methods ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Smoothed gradient with different temperatures
        for T_init, lr_init in [(0.0001, 0.01), (0.0005, 0.02), (0.002, 0.015), (0.01, 0.01)]:
            remaining = deadline - time.monotonic() - 100
            if remaining < 10:
                break
            t_alloc = min(remaining, 50)
            a_opt, _ = smoothed_gradient_descent(state[1], time_limit=t_alloc,
                                                  T_init=T_init, lr_init=lr_init)
            try_and_update(a_opt, state, "smooth_T{}".format(T_init))

        # Multi-peak subgradient
        for top_k, lr, mom in [(10, 0.001, 0.94), (30, 0.002, 0.93), (60, 0.001, 0.95)]:
            remaining = deadline - time.monotonic() - 50
            if remaining < 10:
                break
            t_alloc = min(remaining, 50)
            a_opt, _ = multi_peak_subgradient(state[1], time_limit=t_alloc,
                                               top_k=top_k, lr_init=lr, mom_init=mom)
            try_and_update(a_opt, state, "mpsg_k{}".format(top_k))

    print("  Phase 4 best: {:.8f} (n={}) elapsed {:.0f}s".format(
        state[0], len(state[1]) if state[1] is not None else 0, time.monotonic() - t_phase))

    # === Phase 5: Coordinate descent polish (~150s) ===
    print("\n=== Phase 5: Coordinate Descent Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        remaining = deadline - time.monotonic() - 50
        if remaining > 10:
            t_alloc = min(remaining, 80)
            a_opt, _ = coordinate_descent_polish(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "coord_cd")

        # Another FW pass after CD
        remaining = deadline - time.monotonic() - 30
        if remaining > 10:
            t_alloc = min(remaining, 40)
            a_opt, _ = nesterov_fw_lp(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "final_fw")

        # Fine gradient
        remaining = deadline - time.monotonic() - 15
        if remaining > 5:
            t_alloc = min(remaining, 20)
            a_opt, _ = smoothed_gradient_descent(state[1], time_limit=t_alloc,
                                                  T_init=0.00005, lr_init=0.005)
            try_and_update(a_opt, state, "fine_grad")

    print("  Phase 5 best: {:.8f} (n={}) elapsed {:.0f}s".format(
        state[0], len(state[1]) if state[1] is not None else 0, time.monotonic() - t_phase))

    # === Phase 6: Final micro-perturbations ===
    print("\n=== Phase 6: Final Micro-Perturbations ===")
    if state[1] is not None:
        remaining = deadline - time.monotonic()
        if remaining > 5:
            n_final = len(state[1])
            n_tries = min(10, int(remaining // 3))
            for i in range(n_tries):
                rem = deadline - time.monotonic() - 2
                if rem < 2:
                    break
                for noise_scale in [0.0003, 0.0005, 0.001]:
                    rem = deadline - time.monotonic() - 1
                    if rem < 1:
                        break
                    noise = rng.standard_normal(n_final) * noise_scale
                    f_pert = project_simplex(state[1] + noise)
                    # Quick coordinate descent
                    a_opt, _ = coordinate_descent_polish(f_pert, time_limit=min(rem, 3))
                    try_and_update(a_opt, state, "micro_{}_{}".format(i, noise_scale))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(512) / 512
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification with actual evaluator
    final_score = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(final_score):
        state[0] = final_score
        print("\nFINAL VERIFIED SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    else:
        print("\nWARNING: Final score not finite, using computed: {:.8f}".format(
            evaluate_obj(state[1])))

    return [float(x) for x in state[1]]
