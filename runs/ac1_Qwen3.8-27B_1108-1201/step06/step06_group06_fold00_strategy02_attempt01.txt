def fast_convolve(a):
    """Compute the linear convolution of a with itself quickly using FFT."""
    n = len(a)
    if n==0:
        return np.array([0.0])
    size = 2*n-1
    fft_size = int(1 << (size-1).bit_length())
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)
    return conv[:size]
