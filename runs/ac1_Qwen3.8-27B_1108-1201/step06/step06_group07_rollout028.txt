Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Log-sum-exp smoothing of the max objective
- Softmax parameterization (ensures a > 0, sum(a) = 1)
- L-BFGS optimization with temperature annealing

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_corr_for_grad(w, a):
    """
    Compute g[m] = sum_k w[k] * a[k-m] for m in [0, n-1].
    w has length 2n-1, a has length n.
    """
    n = len(a)
    a_flipped = a[::-1]
    size = (2 * n - 1) + n - 1  # = 3n - 2
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    Af = np.fft.rfft(a_flipped, fft_size)
    conv = np.fft.irfft(W * Af, fft_size)
    return conv[n - 1: 2 * n - 1]


def smooth_objective_and_grad(x, T, cache):
    """
    Objective and gradient for f(x) = T * logsumexp(conv(softmax(x), softmax(x)) / T)
    """
    a = np.exp(x - x.max())
    a = a / a.sum()
    
    z = fast_convolve(a)
    
    # Stable log-sum-exp
    z_max = z.max()
    log_sum = z_max / T + np.log(np.sum(np.exp((z - z_max) / T)))
    obj = T * log_sum
    
    # softmax(z / T)
    w = np.exp((z - z_max) / T)
    w = w / w.sum()

    return obj, grad

        }

    return result.x, result.fun, elapsed


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolInit(parent, n):
    """Interpolate parent to length n."""
    if parent is None:
        return np.ones(n) / n
    px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
    px = np.maximum(px, 0)
    ps = px.sum()
    if ps < 1e-10:
        return np.ones(n) / n
    return px / ps

    Projected gradient descent on the simplex with smoothed objective.

        lr *= 0.999
        if lr < 0.001:
            lr = 0.01
    
    return best_a, best_obj

    return obj, grad


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lbfgs_optimize(x0, T, max_iter, ftol=1e-12, gtol=1e-8):
    """Run L-BFGS-B on the smoothed objective."""
    result = minimize(
        smooth_obj_grad,
        x0,
        args=(T,),
        method='L-BFGS-B',
        jac=True,
        options={
            'maxiter': max_iter,
            'maxfun': max_iter * 20,
            'ftol': ftol,
            'gtol': gtol,
            'maxls': 40,
            'max_correction': 25,
        }
    )
    return result.x, result.fun


def temperature_anneal(a0, time_limit, n_temps=8, rng=None):
    """
    Temperature annealing with L-BFGS at each level.
    a0: initial sequence (sum=1, non-negative)
    Returns optimized sequence.
    """
    if rng is None:
        rng = np.random.default_rng(42)

    n = len(a0)
    a = a0.copy()
    a = a / a.sum()

    z = fast_convolve(a)
    max_z = z.max()

    best_a = a.copy()
    best_obj = 2.0 * n * max_z

    # Temperature schedule
    T_start = max_z * 0.4
    T_end = max_z * 0.01
    if T_end <= 0:
        T_end = 1e-10
    temps = np.geomspace(T_start, T_end, n_temps)

    # Initial x (log of a, which is approximately the softmax pre-image)
    x = np.log(np.maximum(a, 1e-30))

    t0 = time.monotonic()

    for i, T in enumerate(temps):
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 3:
            break

        # Allocate time: more time at lower temperatures
        weight = (i + 1) / (n_temps + 1)
        t_alloc = max(2.0, remaining * weight * 0.6)
        t_alloc = min(t_alloc, remaining - 2.0)

        # Determine max iterations based on n and time
        max_iter = max(30, min(500, int(t_alloc * 80 / n * 768)))

        try:
            x_new, obj_smooth = lbfgs_optimize(x, T, max_iter=max_iter)
            x = x_new
        except Exception:
            # If L-BFGS fails, keep current x
            pass

        # Convert back to a
        x_shifted = x - x.max()
        a_new = np.exp(x_shifted)
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        # Evaluate true (unsmoothed) objective
        true_obj = 2.0 * n * fast_convolve(a_new).max()

        if true_obj < best_obj:
            best_obj = true_obj
            best_a = a_new.copy()
            x = np.log(np.maximum(best_a, 1e-30))
        else:
            # Revert to best
            x = np.log(np.maximum(best_a, 1e-30))
            a_new = best_a.copy()

        a = a_new

        print(f"      T={T:.8f}, true_obj={best_obj:.10f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def make_init(n, kind, rng):
    """Generate a diverse initialization of length n."""
    x = np.arange(n, dtype=np.float64)
    t = x / max(n - 1, 1)

    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'asym_sin':
        a = (1.0 + 0.8 * np.sin(2 * np.pi * t)) ** 2 + 0.01
    elif kind == 'asym_cos':
        a = (1.0 + 0.7 * np.cos(2 * np.pi * t - np.pi / 2)) ** 2 + 0.01
    elif kind == 'two_peak':
        a = (np.exp(-((t - 0.25) ** 2) / (2 * 0.06 ** 2)) +
             np.exp(-((t - 0.75) ** 2) / (2 * 0.06 ** 2))) + 0.001
    elif kind == 'three_peak':
        a = (np.exp(-((t - 0.15) ** 2) / (2 * 0.04 ** 2)) +
             np.exp(-((t - 0.5) ** 2) / (2 * 0.04 ** 2)) +
             np.exp(-((t - 0.85) ** 2) / (2 * 0.04 ** 2))) + 0.001
    elif kind == 'edge_heavy':
        a = 1.0 / (np.minimum(t, 1.0 - t) + 0.05) ** 0.8
    elif kind == 'center_heavy':
        a = 1.0 / (np.abs(t - 0.5) * 2 + 0.05) ** 0.8
    elif kind == 'dirichlet_conc':
        a = rng.dirichlet(np.ones(n) * 0.15)
    elif kind == 'dirichlet_flat':
        a = rng.dirichlet(np.ones(n) * 1.0)
    elif kind == 'dirichlet_spiky':
        a = rng.dirichlet(np.ones(n) * 0.05)
    elif kind == 'alternating':
        a = np.where(x % 2 == 0, 1.5, 0.5) + 0.1
    elif kind == 'linear_ramp':
        a = np.linspace(0.1, 2.0, n)
    elif kind == 'sawtooth':
        a = (t * 4.0) % 1.0
        a = a + (1.0 - a) * 0.3 + 0.05
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_to(parent, n):
    """Interpolate a parent sequence to length n."""
    if parent is None or len(parent) == 0:
        return np.ones(n) / n
    if len(parent) == n:
        return parent.copy()
    px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
    px = np.maximum(px, 0)
    s = px.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return px / s


def local_polish(a0, time_limit, rng, n_perturbs=200):
    """
    Simple local search: random perturbations on the simplex.
    """
    n = len(a0)
    best_a = a0.copy()
    best_obj = 2.0 * n * fast_convolve(best_a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < n_perturbs:
        it += 1
        # Random perturbation with adaptive scale
        scale = 0.005 * (1.0 + 0.5 * np.random.default_rng(it * 31).standard_normal())
        scale = max(0.001, min(0.05, abs(scale)))
        perturbed = best_a + rng.standard_normal(n) * scale
        perturbed = project_simplex(perturbed)

        obj = 2.0 * n * fast_convolve(perturbed).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = perturbed.copy()

    return best_a, best_obj


def propose_candidate(budget_s=1800):
    """
    Main search: multi-phase optimization using log-sum-exp smoothing + L-BFGS.
    """
    rng = np.random.default_rng(12345)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        try:
            score = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(768)

    n_parent = len(parent)

    # ============================================================
    # Phase 1: Quick multi-n exploration (250s)
    # ============================================================
    print("\n=== Phase 1: Quick multi-n exploration ===")
    t1 = time.monotonic()
    p1_budget = 250

    n_candidates = sorted(set([
        n_parent, 512, 640, 768, 896, 1024, 1280, 1536
    ]))
    n_candidates = [n for n in n_candidates if 256 <= n <= 2048]

    inits_kinds = ['uniform', 'asym_sin', 'asym_cos', 'two_peak', 'edge_heavy']

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < p1_budget + 100:
            break

        init = interpolate_to(parent, n_try)
        t_alloc = min(40, remaining - p1_budget - 50)
        if t_alloc < 10:
            t_alloc = 10

        print(f"  n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = temperature_anneal(init, time_limit=t_alloc, n_temps=5, rng=rng)
        try_seq(a_opt, f"p1_n{n_try}")

        # Also try a diverse init
        init2 = make_init(n_try, 'asym_sin', rng)
        t_alloc2 = min(25, remaining - p1_budget - 30)
        if t_alloc2 < 5:
            t_alloc2 = 5
        a_opt2, _ = temperature_anneal(init2, time_limit=t_alloc2, n_temps=4, rng=rng)
        try_seq(a_opt2, f"p1div_n{n_try}")

    print(f"  P1 done: best={best_score:.8f}, t={time.monotonic()-t1:.0f}s")

    # ============================================================
    # Phase 2: Deep optimization at best n (600s)
    # ============================================================
    print("\n=== Phase 2: Deep L-BFGS annealing ===")
    t2 = time.monotonic()
    p2_budget = 600

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()
        t_alloc = min(p2_budget, remaining - 200)
        if t_alloc > 30:
            print(f"  Deep anneal n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = temperature_anneal(best_seq, time_limit=t_alloc, n_temps=10, rng=rng)
            try_seq(a_opt, "deep_anneal")

            # Second round from the result
            remaining = deadline - time.monotonic()
            t_alloc2 = min(200, remaining - 150)
            if t_alloc2 > 20:
                print(f"  Deep anneal 2 n={n_best}, t={t_alloc2:.0f}s")
                a_opt2, _ = temperature_anneal(a_opt, time_limit=t_alloc2, n_temps=10, rng=rng)
                try_seq(a_opt2, "deep_anneal2")

    print(f"  P2 done: best={best_score:.8f}, t={time.monotonic()-t2:.0f}s")

    # ============================================================
    # Phase 3: Multi-restart with diverse inits (300s)
    # ============================================================
    print("\n=== Phase 3: Multi-restart ===")
    t3 = time.monotonic()
    p3_budget = 300

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()

        restart_kinds = ['asym_sin', 'asym_cos', 'two_peak', 'three_peak',
                         'edge_heavy', 'dirichlet_conc', 'alternating', 'linear_ramp']

        for kind in restart_kinds:
            remaining = deadline - time.monotonic()
            if remaining < p3_budget + 100:
                break
            t_alloc = min(40, remaining - p3_budget - 50)
            if t_alloc < 10:
                t_alloc = 10

            # Blend: 50% best_seq interpolated + 50% new init
            init_base = make_init(n_best, kind, rng)
            init_blend = 0.4 * best_seq + 0.6 * init_base
            init_blend = np.maximum(init_blend, 0)
            init_blend = init_blend / init_blend.sum()

            print(f"  Restart {kind}, t={t_alloc:.0f}s")
            a_opt, _ = temperature_anneal(init_blend, time_limit=t_alloc, n_temps=6, rng=rng)
            try_seq(a_opt, f"r_{kind}")

        # Also try pure diverse inits (no blend with best)
        for kind in ['asym_cos', 'edge_heavy', 'three_peak']:
            remaining = deadline - time.monotonic()
            if remaining < 150:
                break
            t_alloc = min(30, remaining - 100)
            if t_alloc < 10:
                break
            init_pure = make_init(n_best, kind, rng)
            a_opt, _ = temperature_anneal(init_pure, time_limit=t_alloc, n_temps=5, rng=rng)
            try_seq(a_opt, f"pure_{kind}")

    print(f"  P3 done: best={best_score:.8f}, t={time.monotonic()-t3:.0f}s")

    # ============================================================
    # Phase 4: Try different n with best shape (200s)
    # ============================================================
    print("\n=== Phase 4: Multi-n with best shape ===")
    t4 = time.monotonic()
    p4_budget = 200

    if best_seq is not None:
        n_best = len(best_seq)
        # Try nearby n values
        for n_try in sorted(set([n_best - 64, n_best + 64, n_best - 128,
                                  n_best + 128, n_best * 2])):
            if n_try < 256 or n_try > 2500:
                continue
            remaining = deadline - time.monotonic()
            if remaining < 150:
                break
            t_alloc = min(35, remaining - 100)
            if t_alloc < 10:
                break
            init = interpolate_to(best_seq, n_try)
            print(f"  n={n_try}, t={t_alloc:.0f}s")
            a_opt, _ = temperature_anneal(init, time_limit=t_alloc, n_temps=6, rng=rng)
            try_seq(a_opt, f"p4_n{n_try}")

    print(f"  P4 done: best={best_score:.8f}, t={time.monotonic()-t4:.0f}s")

    # ============================================================
    # Phase 5: Local polish (remaining time)
    # ============================================================
    print("\n=== Phase 5: Local polish ===")
    t5 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t_alloc = min(100, remaining - 20)
            print(f"  Polish t={t_alloc:.0f}s")
            a_opt, _ = local_polish(best_seq, time_limit=t_alloc, rng=rng, n_perturbs=500)
            try_seq(a_opt, "polish1")

            # Final deep anneal with very low temps
            remaining = deadline - time.monotonic()
            if remaining > 30:
                t_alloc = min(80, remaining - 15)
                print(f"  Final anneal t={t_alloc:.0f}s")
                a_opt2, _ = temperature_anneal(a_opt, time_limit=t_alloc, n_temps=8, rng=rng)
                try_seq(a_opt2, "final_anneal")

            # One more polish
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_opt3, _ = local_polish(a_opt2, time_limit=min(30, remaining - 10),
                                         rng=rng, n_perturbs=200)
                try_seq(a_opt3, "polish2")

    print(f"  P5 done: best={best_score:.8f}, t={time.monotonic()-t5:.0f}s")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
