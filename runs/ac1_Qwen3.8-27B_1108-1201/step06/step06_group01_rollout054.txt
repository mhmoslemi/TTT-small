    """Solve T x = rhs where T is symmetric Toeplitz with first row q.
    Uses conjugate gradient with FFT-based matrix-vector product."""
    n = len(q)
    if max_iter is None:
        max_iter = n
    
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs_old = r @ r
    
    for iter in range(max_iter):
        if np.sqrt(rs_old) < tol * np.linalg.norm(rhs):
            break
        Ap = toeplitz_mv(q, p)
        pAp = p @ Ap
        if abs(pAp) < 1e-30:
            break
        alpha = rs_old / pAp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Build n×n Hankel matrix H where H[j,m] = p[j+m].
    p has length 2n-1."""
    H = np.zeros((n, n))
    for j in range(n):
        H[j, :] = p[j:j+n]
    return H

    where H(p)_{j,m} = p[j+m].

    If any component is negative, clips and returns.

    s = a.sum()

with Hankel-structured CG solves, adaptive temperature annealing, Nesterov
subgradient descent, and structured local polish. Key innovation: the QP oracle
exploits the Hankel structure of the quadratic form for fast matrix-vector
products via np.correlate, enabling many more outer-loop iterations than the
LP-oracle approach.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, v):
    """Compute H @ v where H_{j,m} = p[j+m]. p has length 2n-1, v has length n."""
    return np.correlate(p, v, mode='valid')


def cg_solve_hankel(p, rhs, max_iter=300, tol=1e-12):
    """Solve H x = rhs where H_{j,m} = p[j+m] using conjugate gradient."""
    n = len(rhs)
    x = np.zeros(n)
    r = rhs.copy()
    p_vec = r.copy()
    rs_old = np.dot(r, r)
    norm_rhs = np.linalg.norm(rhs)
    if norm_rhs < 1e-30:
        return x

    for _ in range(max_iter):
        if np.sqrt(rs_old) < tol * norm_rhs:
            break
        Aw = hankel_mv(p, p_vec)
        pAw = np.dot(p_vec, Aw)
        if abs(pAw) < 1e-30:
            break
        alpha = rs_old / pAw
        if alpha < 0:
            # Matrix not positive definite; fall back
            break
        x += alpha * p_vec
        r -= alpha * Aw
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * norm_rhs:
            break
        p_vec = r + (rs_new / rs_old) * p_vec
        rs_old = rs_new
    return x


def direct_solve_hankel(p, n):
    """Direct solve H x = 1 where H_{j,m} = p[j+m]."""
    H = np.zeros((n, n))
    for j in range(n):
        H[j, :] = p[j:j + n]
    ones = np.ones(n)
    try:
        return np.linalg.solve(H, ones)
    except np.linalg.LinAlgError:
        return np.linalg.lstsq(H, ones, rcond=None)[0]


def qp_oracle_hankel(peaks, n, method='auto'):
    """
    QP oracle: min a^T H a s.t. sum(a)=1, a>=0
    where H_{j,m} = peaks[j+m].
    Returns the (clipped) solution.
    """
    ones = np.ones(n)
    if method == 'direct' or n <= 400:
        x = direct_solve_hankel(peaks, n)
    else:
        x = cg_solve_hankel(peaks, ones, max_iter=200)
        # Check CG quality
        if np.any(np.abs(x) > 1e6) or not np.all(np.isfinite(x)):
            x = direct_solve_hankel(peaks, n)

    denom = np.dot(ones, x)
    if denom <= 1e-15:
        return np.ones(n) / n

    a = x / denom
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def fw_qp_optimize(f0, time_limit, n_peaks_weight_power=2.0):
    """
    Frank-Wolfe with QP oracle (Hankel-structured).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    temperature = 1.0
    momentum = 0.0
    v_mom = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break

        # Compute peak weights with temperature
        conv = fast_convolve(f)
        M = conv.max()
        scaled = conv / (M * temperature)
        p = np.exp(-scaled)
        p = p / p.sum()

        # QP oracle
        g = qp_oracle_hankel(p, n)

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 20:
                temperature *= 0.85
                if temperature < 0.1:
                    temperature = 1.0
                    # Informed perturbation
                    rng = np.random.default_rng(it * 7919 + 13)
                    peak_k = int(np.argmax(conv))
                    noise = rng.standard_normal(n) * 0.003
                    js = max(0, peak_k - n + 1)
                    je = min(n - 1, peak_k)
                    noise[js:je + 1] -= 0.002
                    f = project_simplex(best_f + noise)
                    no_improve = 0
                    continue
            else:
                temperature *= 0.9
            continue

        # Line search with momentum
        best_alpha = 0.0
        best_val = M

        # Coarse search
        for alpha in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.02)
            hi = min(1.0, best_alpha + 0.02)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            # Apply with momentum
            direction = g - f
            v_mom = 0.3 * v_mom + (1.0 - 0.3) * direction
            f_new = f + best_alpha * v_mom
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = f + best_alpha * direction
                f_new = np.maximum(f_new, 0.0)
                sn = f_new.sum()
            f_new = f_new / sn

            val_new = fast_convolve(f_new).max()

            if val_new < best_obj:
                f = f_new
                best_obj = val_new
                best_f = f.copy()
                no_improve = 0
            else:
                # Revert and try without momentum
                f_try = (1.0 - best_alpha) * f + best_alpha * g
                f_try = np.maximum(f_try, 0.0)
                f_try = f_try / f_try.sum()
                val_try = fast_convolve(f_try).max()
                if val_try < best_obj:
                    f = f_try
                    best_obj = val_try
                    best_f = f.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                    v_mom *= 0.5
                temperature *= 0.95
        else:
            no_improve += 1
            temperature *= 0.9

        # Adaptive temperature reset
        if no_improve % 30 == 0 and no_improve > 0:
            temperature = max(0.3, temperature * 0.8)

        if no_improve > 50:
            # Structured restart
            rng = np.random.default_rng((it * 104729 + 7) % (2**31))
            scale = 0.001 + 0.002 * min(1.0, no_improve / 100.0)
            conv = fast_convolve(best_f)
            peak_k = int(np.argmax(conv))
            noise = rng.standard_normal(n) * scale
            # Targeted reduction at peak
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            for j in range(js, je + 1):
                partner = peak_k - j
                if 0 <= partner < n:
                    noise[j] -= scale * 1.5 * best_f[partner]
            f = project_simplex(best_f + noise)
            v_mom = np.zeros(n)
            temperature = 1.0
            no_improve = 0

    return best_f, best_obj


def nesterov_subgradient(a0, time_limit, lr_init=0.005, max_k=40):
    """Nesterov-accelerated subgradient with top-k peak tracking."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    y = a.copy()
    v = np.zeros(n)
    t_nest = 1.0
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        alpha_n = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / alpha_n
        y = (1.0 + beta) * a - beta * y
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y / ys
        t_nest += 1.0

        conv = fast_convolve(y)
        kk = min(max_k, len(conv))
        peaks = np.argpartition(conv, -kk)[-kk:]
        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(kk)
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.3
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        sg = np.zeros(n)
        for i in range(kk):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * y[k - idx_arr]

        v = beta * v - lr * sg
        a_new = project_simplex(a + v)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            a_new = a

        a = a_new
        lr *= (1.0 - 0.0003)

        if no_improve > 80:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.001 + 0.001 * min(1.0, no_improve / 150.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            y = a.copy()
            v = np.zeros(n)
            t_nest = 1.0
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def smoothed_mirror_descent(a0, time_limit, T_init=0.001):
    """Mirror descent with exponential weighting on peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    eta = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        M = conv.max()
        # Softmax weights on peaks
        scaled = -conv / T
        scaled -= scaled.min()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient: grad_j = sum_k p_k * (a_{j+k} + a_{j-k})
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                partner = k - j
                if 0 <= partner < n:
                    grad[j] += p[k] * a[partner]

        # Mirror descent update (multiplicative)
        a_new = a * np.exp(-eta * grad)
        # Enforce minimum mass to avoid underflow
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            eta *= 1.01
        else:
            no_improve += 1
            eta *= 0.95

        # Adaptive temperature
        if no_improve > 20:
            T *= 0.9
        if no_improve > 80:
            T = max(T_init * 0.5, T * 0.7)
        if no_improve > 200:
            rng = np.random.default_rng((it * 48291 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            eta = 1.0
            no_improve = 0

    return best_a, best_obj


def local_polish(a0, time_limit):
    """Structured local optimization: try 2-swap and 3-swap moves targeting peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        top_k = min(8, len(conv))
        peak_lags = np.argpartition(conv, -top_k)[-top_k:]
        peak_lags = peak_lags[np.argsort(conv[peak_lags])[::-1]]

        improved = False

        for k_peak in peak_lags:
            if time.monotonic() - t0 > time_limit - 0.3:
                break

            js = max(0, int(k_peak) - n + 1)
            je = min(n - 1, int(k_peak))
            affected = np.arange(js, je + 1)

            if len(affected) < 2:
                continue

            # Sample a subset if too large
            if len(affected) > 40:
                rng_s = np.random.default_rng(it * 7919 + int(k_peak))
                affected = rng_s.choice(affected, size=40, replace=False)

            best_gain = 0.0
            best_move = None

            # Try pairwise transfers
            for ii in range(len(affected)):
                j1 = affected[ii]
                if a[j1] < 1e-10:
                    continue
                contrib1 = a[int(k_peak) - j1] if 0 <= int(k_peak) - j1 < n else 0.0
                for jj in range(ii + 1, len(affected)):
                    j2 = affected[jj]
                    contrib2 = a[int(k_peak) - j2] if 0 <= int(k_peak) - j2 < n else 0.0

                    # Determine direction: move from higher contributor to lower
                    if contrib1 > contrib2 and a[j1] > 1e-8:
                        src, dst = j1, j2
                    elif contrib2 > contrib1 and a[j2] > 1e-8:
                        src, dst = j2, j1
                    else:
                        continue

                    for frac in [0.05, 0.1, 0.2, 0.4]:
                        delta = a[src] * frac
                        if delta < 1e-12:
                            continue
                        a_t = a.copy()
                        a_t[src] -= delta
                        a_t[dst] += delta
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:
                            continue
                        a_t = a_t / st
                        c_t = fast_convolve(a_t)
                        gain = best_obj - c_t.max()
                        if gain > best_gain:
                            best_gain = gain
                            best_move = (src, dst, frac)

            if best_move is not None and best_gain > 1e-13:
                src, dst, frac = best_move
                delta = a[src] * frac
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                c_new = fast_convolve(a)
                best_obj = c_new.max()
                best_a = a.copy()
                improved = True
                stall = 0
            else:
                stall += 1

        if not improved:
            stall += 1

        if stall > 15:
            rng = np.random.default_rng((it * 62711 + 1) % (2**31))
            scale = 0.0003 + 0.0007 * min(1.0, stall / 40.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            stall = 0

        if stall > 100:
            break

    return best_a, best_obj


def micro_refine(a0, time_limit):
    """Ultra-fine perturbations for the last bit of improvement."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    scales = [5e-5, 1e-4, 2e-4, 4e-4, 8e-4]
    si = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        scale = scales[si % len(scales)]
        rng = np.random.default_rng((it * 2654435761 + 1) % (2**32))

        tries = 0
        while tries < 8 and time.monotonic() - t0 < time_limit - 0.05:
            tries += 1
            conv = fast_convolve(a)
            peak_k = int(np.argmax(conv))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            # Biased perturbation
            noise = rng.standard_normal(n) * scale
            for j in range(js, je + 1):
                partner = peak_k - j
                if 0 <= partner < n:
                    noise[j] -= scale * 2.0 * a[partner]

            candidate = project_simplex(a + noise)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                break

        if it % 200 == 0:
            si += 1

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):

    Main entry point: multi-phase hybrid search combining QP-FW, Nesterov,
    mirror descent, and local polish.

        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()

            return float('inf')

        score = float(evaluate_sequence(a_norm.tolist()))

            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t0_total
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return score

    t0_total = time.monotonic()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")

    print("\n=== Phase 2: Deep QP-FW ===")
    t_phase = time.monotonic()

        # Multiple runs with different parameters

            print(f"  Nesterov cfg{ci}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_subgradient(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"nest_{ci}")

    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Mirror Descent ===
    print("\n=== Phase 4: Mirror Descent ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        for T_init in [0.0005, 0.002, 0.005]:
            t_alloc = min(100, deadline - time.monotonic() - 250)
            if t_alloc < 10:
                break
            print(f"  Mirror T={T_init}, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_mirror_descent(best_seq, time_limit=t_alloc, T_init=T_init)
            try_seq(a_opt, f"mirror_T{T_init}")

    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Local Polish ===
    print("\n=== Phase 5: Local Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(180, deadline - time.monotonic() - 120)
        if t_alloc > 10:
            print(f"  Local polish, t={t_alloc:.0f}s")
            a_opt, _ = local_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "polish")

    print(f"  Phase 5 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Final Micro-Refine + QP-FW ===
    print("\n=== Phase 6: Final Refinement ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Micro refine
        t_alloc = min(60, deadline - time.monotonic() - 60)
        if t_alloc > 5:
            print(f"  Micro refine, t={t_alloc:.0f}s")
            a_opt, _ = micro_refine(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "micro")

        # Another QP-FW pass
        t_alloc = min(40, deadline - time.monotonic() - 30)
        if t_alloc > 5:
            print(f"  Final QP-FW, t={t_alloc:.0f}s")
            a_opt, _ = fw_qp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_qp")

        # Final micro on the result
        t_alloc = min(30, deadline - time.monotonic() - 15)
        if t_alloc > 5 and best_seq is not None:
            a_opt, _ = micro_refine(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_micro")

    print(f"  Phase 6 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, total_time={time.monotonic()-t0_total:.0f}s)")
    return [float(x) for x in best_seq]
