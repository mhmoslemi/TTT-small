"""

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_smooth_objective(a, T):
    """Compute the softmax-smoothed max of conv(a,a) and its gradient."""
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    p = exp_vals / sum_exp
    f_val = T * (np.log(sum_exp) + scaled.max())
    
    # Gradient: 2 * sum_k p_k * a[k-j] (approximate, ignoring the factor-1/2 diagonal)
    n = len(a)
    conv_len = len(conv)
    # Build gradient via correlation
    # grad[j] = 2 * sum_k p[k] * a[k-j] for valid k-j
    # This is: 2 * convolve(p_sorted, a) taken at appropriate indices
    # More precisely: grad[j] = 2 * sum_{k: 0<=k-j<n} p[k] * a[k-j]
    #              = 2 * (p * a_flipped)[j + n - 1] ... let me just do it directly
    
    # p has length 2n-1, a has length n
    # grad[j] = 2 * sum_k p[k] * a[k-j] where 0<=k-j<n
    # For each j: k ranges from j to j+n-1 (clamped to [0, 2n-2])
    grad = np.zeros(n)
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        if k_start <= k_end:
            # a[k-j] for k in [k_start, k_end] means a[0] to a[k_end-j]
            idx = np.arange(k_start, k_end + 1)
            a_idx = idx - j  # ranges from 0 to k_end - j
            grad[j] = 2.0 * np.dot(p[k_start:k_end + 1], a[a_idx])
    
    # Correct the diagonal (where k = 2j, the self-term)
    for j in range(n):
        k = 2 * j
        if 0 <= k < 2 * n - 1:
            grad[j] -= p[k] * a[j]
    
    return f_val, grad


def spectral_iteration(a0, time_limit, n_iter_max=2000):
    """
    Griffin-Lim style iteration for the auto-convolution problem.
    Iteratively:
    1. Compute current convolution
    2. Blend with flat target
    3. Update via spectral method (modify |DFT|)
    4. Project onto a >= 0
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    blend = 0.3  # How much to blend toward flat
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        conv = fast_convolve(a)
        current_max = conv.max()
        
        # Compute target: blend current with flat
        target = conv * (1 - blend) + (conv.sum() / len(conv)) * blend
        target = np.maximum(target, 0)
        
        # Spectral update: we want a new a' such that conv(a',a') ≈ target
        # Using the DFT relationship: DFT(conv(a,a)) = |DFT(a)|^2 (for circular)
        # For linear conv, we zero-pad a to length >= 2n-1
        A = np.fft.rfft(a, fft_size)
        C = np.fft.rfft(target, fft_size)
        
        # |A_new|^2 should be C (in FFT domain)
        # Keep the phase of A, update the magnitude
        A_new = np.sqrt(np.maximum(C, 0).real) * np.exp(1j * np.angle(A))
        
        # Handle the DC component and Nyquist
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn
        
        # Check improvement
        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive blend: increase if not improving (more aggressive flattening)
        if no_improve > 20:
            blend = min(0.7, blend * 1.05)
            if no_improve > 100:
                blend = 0.3
                # Perturb
                rng = np.random.default_rng(it * 31 + 7)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                no_improve = 0
        
        a = a_new
    
    return best_a, best_obj


def smooth_gradient_descent(a0, time_limit, T_init=0.005, T_min=0.0005, lr_init=0.02):
    """
    Gradient descent on the softmax-smoothed objective with adaptive T and LR.
    Works on the simplex (sum a = 1, a >= 0).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    T = T_init
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        f_val, grad = softmax_smooth_objective(a, T)
        
        # Project gradient onto simplex tangent space
        grad_tangent = grad - np.mean(grad)
        
        # Backtracking line search
        best_candidate = a
        best_val = fast_convolve(a).max()
        step = lr
        for _ in range(8):
            candidate = a - step * grad_tangent
            candidate = project_simplex(candidate)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
            step *= 0.5
            if step < 1e-10:
                break
        
        a = best_candidate
        obj = fast_convolve(a).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive T and LR
        if no_improve > 20:
            T = max(T_min, T * 0.9)
            lr *= 0.95
        if no_improve > 80:
            T = min(T_init, T * 1.3)
            lr = lr_init * 0.8
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj


def solve_lp_oracle(f, M, time_limit=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            g = np.maximum(res.x, 0.0)
            return g
    except Exception:
        pass
    return None

def coordinate_descent_polish(a0, time_limit):
    """
    Fine coordinate descent: try moving mass from peak-contributing
    positions to non-contributing positions.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]
        
        # Find the best single coordinate move
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        
        best_gain = 0
        best_move = None
        
        # Try moving mass from high-contributing to low-contributing positions
        for j in range(js, je + 1):
            if a[j] < 1e-10:
                continue
            partner = peak_k - j
            contrib = a[j] * a[partner] if 0 <= partner < n else 0
            
            # Try moving to positions that don't affect this peak
            for dst in range(n):
                if dst == j:
                    continue
                dst_partner = peak_k - dst
                dst_contrib = a[dst] * a[dst_partner] if 0 <= dst_partner < n else 0
                
                # Only consider moves that reduce this peak
                if contrib <= dst_contrib * 0.5:
                    continue
                
                for frac in [0.1, 0.3, 0.5]:
                    delta = a[j] * frac
                    if delta < 1e-12:
                        continue
                    a_test = a.copy()
                    a_test[j] -= delta
                    a_test[dst] += delta
                    a_test = np.maximum(a_test, 0)
                    st = a_test.sum()
                    if st < 1e-15:
                        continue
                    a_test = a_test / st
                    c_new = fast_convolve(a_test)
                    new_max = c_new.max()
                    gain = best_obj - new_max
                    if gain > best_gain:
                        best_gain = gain
                        best_move = (j, dst, frac)
        
        if best_move is not None and best_gain > 1e-12:
            j, dst, frac = best_move
            delta = a[j] * frac
            a[j] -= delta
            a[dst] += delta
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            obj = fast_convolve(a).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        if stall > 30:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
            scale = 0.001 + 0.002 * min(1.0, stall / 100.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)

            if t_alloc < 15:
                break
            
            # Resample to new length
            a_resampled = np.interp(np.linspace(0, 1, n_try),
                                    np.linspace(0, 1, n_best), best_seq)
            a_resampled = np.maximum(a_resampled, 0)
            sr = a_resampled.sum()
            if sr > 1e-10:
                a_resampled = a_resampled / sr
            else:
                a_resampled = np.ones(n_try) / n_try
            
            print(f"  Resample+FW n={n_try}, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(a_resampled, time_limit=t_alloc, lp_time=15)
            try_seq(a_opt, f"n_{n_try}")
    
    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 5: Coordinate Descent Polish (~200s) ===
    print("\n=== Phase 5: Coordinate Descent ===")
    t_phase = time.monotonic()
    
    if best_seq is not None:
        t_alloc = min(200, deadline - time.monotonic() - 80)
        if t_alloc > 10:
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord")
    
    print(f"  Phase 5 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 6: Final refinement loop ===
    print("\n=== Phase 6: Final Refinement ===")
    t_phase = time.monotonic()
    
    if best_seq is not None:
        # Alternating short passes
        methods = [
            ("fw", lambda a, t: fw_lp_optimize(a, time_limit=t, lp_time=min(15, t * 0.3))[0]),
            ("smooth", lambda a, t: smooth_gradient_descent(a, time_limit=t, T_init=0.002, T_min=0.0005, lr_init=0.015)[0]),
            ("spectral", lambda a, t: spectral_iteration(a, time_limit=t)[0]),
            ("coord", lambda a, t: coordinate_descent_polish(a, time_limit=t)[0]),
        ]
        
        mi = 0
        while time.monotonic() < deadline - 30:
            t_alloc = min(40, deadline - time.monotonic() - 20)
            if t_alloc < 8:
                break
            
            name, func = methods[mi % len(methods)]
            print(f"  Final {name}, t={t_alloc:.0f}s")
            a_opt = func(best_seq, t_alloc)
            try_seq(a_opt, f"final_{name}_{mi}")
            mi += 1
    
    print(f"  Phase 6 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, total time={time.monotonic()-t0_total:.0f}s)")
    return [float(x) for x in best_seq]
