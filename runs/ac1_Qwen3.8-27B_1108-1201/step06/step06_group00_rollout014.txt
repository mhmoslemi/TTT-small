from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size-1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A*A, fft_size)[:size]

def symmetrize(p):
    return (p + p[::-1]) / 2.0

def normalize(p):
    s = p.sum()
    if s > 1e-15:
        return p / s
    return np.ones(len(p)) / len(p)

def project_simplex(v):
    # Project onto {x : sum x = 1, x >= 0}
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def build_constraint_matrix(p, active_indices):
    n = len(p)
    rows, cols, vals = [], [], []
    for idx, k in enumerate(active_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        rows.extend([idx] * len(js))
        cols.extend(js.tolist())
        vals.extend(p[k - js].tolist())
    if not rows:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))

def solve_lp(p, M, active_indices, time_limit=10):
    n = len(p)
    A = build_constraint_matrix(p, active_indices)
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except:
        pass
    return None

    return best_p, best_obj

def coordinate_descent(p0, time_limit, n_steps=10000):
    p = symmetrize(p0.copy())
    p = normalize(p)
    n = len(p)
    best_obj = fast_convolve(p).max()
    best_p = p.copy()
    t0 = time.monotonic()
    step = 0.01
    rng = np.random.default_rng(42)
    
    for _ in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        # Pick a random index
        i = rng.integers(n)
        # Try to adjust p[i]
        c = fast_convolve(p)
        # Find the peak that involves p[i]
        # The peaks affected by changing p[i] are at positions k where k-i is in [0, n-1]
        # i.e., k in [i, i+n-1]
        k_lo = i
        k_hi = min(i + n - 1, 2*n - 2)
        # Current peak value in this range
        current_max = c[k_lo:k_hi+1].max()
        
        # Try decreasing p[i]
        for delta_frac in [0.1, 0.3, 0.5, 0.7]:
            delta = p[i] * delta_frac
            if delta < 1e-10:
                continue
            p_trial = p.copy()
            p_trial[i] -= delta
            # Redistribute to a random other index
            j = rng.integers(n)
            if j == i:
                j = (i + n//2) % n
            p_trial[j] += delta
            p_trial = np.maximum(p_trial, 0)
            p_trial = symmetrize(p_trial)
            p_trial = normalize(p_trial)
            val = fast_convolve(p_trial).max()
            if val < best_obj - 1e-12:
                best_obj = val
                best_p = p_trial.copy()
                p = p_trial.copy()
                break
    
    return best_p, best_obj

**Approach: Direct minimization of max c_k via bisection + feasibility**

For a target value T, check if there exists p with sum p = 1, p >= 0, and c_k(p) <= T for all k.

This is a feasibility problem. We can solve it using:
- Sequential Quadratic Programming (SQP)
- Penalty method: minimize sum(c_k - T)^2 for c_k > T, subject to sum p = 1, p >= 0
- Or: minimize max(c_k - T)^2 + penalty*(violation of constraints)

The penalty method is simpler. We minimize:
F(p) = sum_k max(0, c_k(p) - T)^2 + lambda * (sum p - 1)^2 + mu * sum max(0, -p_i)^2

As T decreases, the feasibility becomes harder. We binary search on T.

But this is essentially what the subgradient method is doing (minimizing a smoothed version of max c_k). So it's not really different.
