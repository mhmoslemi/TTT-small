"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Cyclic Coordinate Descent (CCD)
combined with Projected Gradient Descent (PGD) for escape from local minima.
The CCD exploits the fact that fixing all coordinates except one reduces the
problem to a 1D optimization, which can be solved precisely.
"""
import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            ks = np.arange(k_start, k_end)
            offsets = ks - j
            mask_quad = (ks == 2 * j)
            if (~mask_quad).any():
                conv[ks[~mask_quad]] += 2.0 * (new_val - old_val) * a[offsets[~mask_quad]]
            if mask_quad.any():
                conv[ks[mask_quad]] += new_val * new_val - old_val * old_val
            a[j] = new_val
            sum_a += (new_val - old_val)
    return sum_a


def pgd_momentum(a0, time_limit, top_k=20, lr_init=0.02, mom_init=0.9):
    """Projected gradient descent with momentum and adaptive step size."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = mom_init
    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        obj = conv.max()

        # Subgradient from top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            order = np.argsort(pv)[::-1]
            weights = np.zeros(len(peaks))
            for rank, idx in enumerate(order):
                weights[idx] = 1.0 / (rank + 1) ** 1.5
            ws = weights.sum()
            if ws > 0:
                weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * a[k - idx_arr]

        # Backtracking line search
        best_candidate = a
        best_val = obj
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.125]:
            v_new = mom * v - trial_lr * sg
            candidate = project_simplex(a + v_new)
            c_new = fast_convolve(candidate)
            val = c_new.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                v = v_new
                conv = c_new
                break

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.9

        if no_improve > 50:
            lr *= 0.93
            mom = min(0.97, mom + 0.003)
        if no_improve > 200:
            rng = np.random.default_rng((it * 48271 + 17) % (2**31 - 1))
            noise = 0.003 + 0.005 * min(1.0, (no_improve - 200) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init
            mom = mom_init
            no_improve = 0
            conv = fast_convolve(a)

    return best_a, best_obj


def fw_lp_optimize(f0, time_limit, m_tolerance=1.0):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M*m_exp, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'max_time': 15})
            if res.success:
                g = np.maximum(res.x, 0.0)
            else:
                g = None
        except Exception:
            g = None

        if g is None:
            m_exp *= 1.3
            if m_exp > 5.0:
                m_exp = 1.0
            no_improve += 1
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.03, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 40:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.003 + 0.005 * min(1.0, no_improve / 80.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def try_and_update(a, best_state, label=""):
    """Evaluate sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_state
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    best_seq, best_score = best_state
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return (best_seq, best_score)


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def symmetric_ccd_sweep(a, conv, sum_a):
    """CCD sweep enforcing symmetry a[j] = a[n-1-j]."""
    n = len(a)
    half = (n + 1) // 2
    for j in range(half):
        old_val = a[j]
        new_val = ccd_one_coordinate(a, conv, j, sum_a)
        if abs(new_val - old_val) > 1e-15:
            # Update for coordinate j
            k_start = j
            k_end = min(j + n, 2 * n - 1)
            ks = np.arange(k_start, k_end)
            offsets = ks - j
            mask_quad = (ks == 2 * j)
            if (~mask_quad).any():
                conv[ks[~mask_quad]] += 2.0 * (new_val - old_val) * a[offsets[~mask_quad]]
            if mask_quad.any():
                conv[ks[mask_quad]] += new_val * new_val - old_val * old_val

            # Also update the symmetric coordinate j_sym = n-1-j if different
            j_sym = n - 1 - j
            if j_sym != j:
                a[j] = new_val
                sum_a += (new_val - old_val)
                # Now update j_sym similarly (it had old_val too)
                # Update conv for j_sym with old_val -> new_val
                k_start2 = j_sym
                k_end2 = min(j_sym + n, 2 * n - 1)
                ks2 = np.arange(k_start2, k_end2)
                offsets2 = ks2 - j_sym
                mask_quad2 = (ks2 == 2 * j_sym)
                if (~mask_quad2).any():
                    conv[ks2[~mask_quad2]] += 2.0 * (new_val - old_val) * a[offsets2[~mask_quad2]]
                if mask_quad2.any():
                    conv[ks2[mask_quad2]] += new_val * new_val - old_val * old_val
                a[j_sym] = new_val
                sum_a += (new_val - old_val)
            else:
                a[j] = new_val
                sum_a += (new_val - old_val)
    return sum_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Uses Cyclic Coordinate Descent (CCD) as the primary
    optimization engine, combined with PGD for escape from local minima
    and FW-LP for global improvements.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Quick PGD warmup ===
    print("\n=== Phase 1: PGD Warmup ===")
    t_start = time.monotonic()

    # Try symmetrized version
    sym_parent = (parent + parent[::-1]) / 2.0
    sym_parent = sym_parent / sym_parent.sum()
    (best_seq, best_score) = try_and_update(sym_parent, (best_seq, best_score), "sym_parent")

    remaining = deadline - time.monotonic()
    t_alloc = min(remaining, 80)
    a_opt, _ = pgd_momentum(best_seq, time_limit=t_alloc, top_k=20, lr_init=0.03)
    (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "pgd_warmup")

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 2: CCD main optimization ===
    print("\n=== Phase 2: CCD Main Loop ===")
    t_start = time.monotonic()

    a = best_seq.copy()
    conv = fast_convolve(a)
    sum_a = a.sum()

    ccd_best = a.copy()
    ccd_best_score = best_score
    no_improve = 0
    sweep_count = 0

    while time.monotonic() < deadline - 400:
        # CCD sweep
        sum_a = ccd_sweep(a, conv, sum_a)
        sweep_count += 1

        # Evaluate
        conv = fast_convolve(a)
        max_b = conv.max()
        score = 2.0 * len(a) * max_b / (sum_a ** 2)

        if score < ccd_best_score - 1e-12:
            ccd_best_score = score
            ccd_best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if sweep_count % 20 == 0:
            elapsed = time.monotonic() - t_start
            print(f"  CCD sweep={sweep_count}, score={score:.8f}, best={ccd_best_score:.8f}, elapsed={elapsed:.0f}s")

        # Periodic PGD restart to escape local minima
        if no_improve > 100:
            remaining = deadline - time.monotonic() - 400
            if remaining > 30:
                t_pgd = min(remaining, 40)
                a_pgd, _ = pgd_momentum(ccd_best, time_limit=t_pgd, top_k=15, lr_init=0.02)
                score_pgd = float(evaluate_sequence(a_pgd.tolist()))
                if np.isfinite(score_pgd) and score_pgd < ccd_best_score:
                    ccd_best_score = score_pgd
                    ccd_best = a_pgd.copy()
                    print(f"  CCD+PGD improvement: {score_pgd:.8f}")
                # Restart CCD from the better solution
                a = ccd_best.copy()
                conv = fast_convolve(a)
                sum_a = a.sum()
                no_improve = 0
            else:
                break

    a_final = ccd_best.copy()
    (best_seq, best_score) = try_and_update(a_final, (best_seq, best_score), "ccd_main")
    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 3: FW-LP for global improvement ===
    print("\n=== Phase 3: FW-LP ===")
    t_start = time.monotonic()

    remaining = deadline - time.monotonic()
    t_alloc = min(remaining * 0.3, 200)
    a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
    (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "fw_lp")

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 4: CCD refinement on best ===
    print("\n=== Phase 4: CCD Refinement ===")
    t_start = time.monotonic()

    a = best_seq.copy()
    conv = fast_convolve(a)
    sum_a = a.sum()

    ccd_best = a.copy()
    ccd_best_score = best_score
    no_improve = 0
    sweep_count = 0

    while time.monotonic() < deadline - 200:
        sum_a = ccd_sweep(a, conv, sum_a)
        sweep_count += 1

        conv = fast_convolve(a)
        max_b = conv.max()
        score = 2.0 * len(a) * max_b / (sum_a ** 2)

        if score < ccd_best_score - 1e-12:
            ccd_best_score = score
            ccd_best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if sweep_count % 20 == 0:
            elapsed = time.monotonic() - t_start
            print(f"  CCD ref sweep={sweep_count}, score={score:.8f}, best={ccd_best_score:.8f}, elapsed={elapsed:.0f}s")

        if no_improve > 150:
            # Try a perturbation
            rng2 = np.random.default_rng(sweep_count * 31 + 7)
            noise = 0.002 + 0.003 * min(1.0, no_improve / 200.0)
            a_pert = project_simplex(ccd_best + rng2.standard_normal(len(ccd_best)) * noise)
            a = a_pert
            conv = fast_convolve(a)
            sum_a = a.sum()
            no_improve = 0

    a_final = ccd_best.copy()
    (best_seq, best_score) = try_and_update(a_final, (best_seq, best_score), "ccd_refine")
    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Try different n values ===
    print("\n=== Phase 5: Multi-n Exploration ===")
    t_start = time.monotonic()

    n_best = len(best_seq)
    n_candidates = set()
    for n_try in [n_best // 2, n_best, n_best * 2, 384, 512, 640, 768, 896, 1024, 1280, 1536]:
        if 128 <= n_try <= 3072:
            n_candidates.add(n_try)

    for n_try in sorted(n_candidates):
        remaining = deadline - time.monotonic() - 60
        if remaining < 30:
            break

        if n_try == n_best:
            start = best_seq.copy()
        else:
            start = interp_sequence(best_seq, n_try)

        # Quick CCD on this n
        a = start.copy()
        conv = fast_convolve(a)
        sum_a = a.sum()

        # Do a few CCD sweeps
        max_sweeps = 30
        sweeps = 0
        while sweeps < max_sweeps and time.monotonic() < deadline - 60:
            remaining_inner = deadline - time.monotonic() - 60
            if remaining_inner < 5:
                break
            sum_a = ccd_sweep(a, conv, sum_a)
            sweeps += 1

        conv = fast_convolve(a)
        (best_seq, best_score) = try_and_update(a, (best_seq, best_score), f"ccd_n{n_try}")

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 6: Final CCD polish ===
    print("\n=== Phase 6: Final Polish ===")
    t_start = time.monotonic()

    a = best_seq.copy()
    conv = fast_convolve(a)
    sum_a = a.sum()

    ccd_best = a.copy()
    ccd_best_score = best_score
    sweep_count = 0

    while time.monotonic() < deadline - 10:
        sum_a = ccd_sweep(a, conv, sum_a)
        sweep_count += 1

        conv = fast_convolve(a)
        max_b = conv.max()
        score = 2.0 * len(a) * max_b / (sum_a ** 2)

        if score < ccd_best_score - 1e-14:
            ccd_best_score = score
            ccd_best = a.copy()

        if sweep_count % 10 == 0:
            print(f"  Final CCD sweep={sweep_count}, score={score:.10f}")

    a_final = ccd_best.copy()
    (best_seq, best_score) = try_and_update(a_final, (best_seq, best_score), "final_ccd")

    # Symmetry check
    sym_final = (best_seq + best_seq[::-1]) / 2.0
    sym_final = sym_final / sym_final.sum()
    (best_seq, best_score) = try_and_update(sym_final, (best_seq, best_score), "final_sym")

    # One more CCD on the symmetric version
    a = best_seq.copy()
    conv = fast_convolve(a)
    sum_a = a.sum()
    for _ in range(20):
        if time.monotonic() > deadline - 5:
            break
        sum_a = ccd_sweep(a, conv, sum_a)
    conv = fast_convolve(a)
    (best_seq, best_score) = try_and_update(a, (best_seq, best_score), "final_ccd_sym")

    print(f"  Phase 6 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
