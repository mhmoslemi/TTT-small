def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]."""
    return np.correlate(p, a, mode='valid')


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def multi_peak_sg(a0, time_limit, top_k, lr_init, mom):

def fourier_to_seq(coeffs, n, M):
    """Convert Fourier coefficients to a sequence."""
    x = np.arange(n, dtype=np.float64)
    base = np.ones(n) / n
    for m in range(1, M + 1):
        c = coeffs[2 * (m - 1)]
        d = coeffs[2 * (m - 1) + 1]
        base += (c * np.cos(2.0 * np.pi * m * x / n) +
                 d * np.sin(2.0 * np.pi * m * x / n)) / n
    base = np.maximum(base, 0.0)
    s = base.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return base / s


def fourier_objective(coeffs, n, M):
    """Objective for Fourier parametrization."""
    seq = fourier_to_seq(coeffs, n, M)
    conv = fast_convolve(seq)
    return conv.max()

def resample_sequence(parent, target_n):
    """Resample a sequence to a different length using linear interpolation."""
    n_parent = len(parent)
    if n_parent == target_n:
        return parent.copy()
    x_old = np.linspace(0, 1, n_parent)
    x_new = np.linspace(0, 1, target_n)
    resampled = np.interp(x_new, x_old, parent)
    resampled = np.maximum(resampled, 0.0)
    s = resampled.sum()
    if s > 1e-15:
        resampled = resampled / s
    else:
        resampled = np.ones(target_n) / target_n
    return resampled

            a_opt, _ = fourier_refine(state[1], time_limit=t_alloc, n_fourier=12)
            try_and_update(a_opt, state, "fourier_12")

        remaining = deadline - time.monotonic() - 300
        if remaining > 12:
            t_alloc = min(remaining, 80)
            print("  Fourier refine (20 modes), t={:.0f}s".format(t_alloc))
            a_opt, _ = fourier_refine(state[1], time_limit=t_alloc, n_fourier=20)
            try_and_update(a_opt, state, "fourier_20")

    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 4: Intensive Gradient Methods (~450s) ===
    print("\n=== Phase 4: Intensive Gradient Methods ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])

        # Multiple smoothed gradient configs
        sg_configs = [
            (0.0003, 0.03, 1.01, 0.995),
            (0.001, 0.025, 1.02, 0.993),
            (0.003, 0.02, 1.03, 0.99),
            (0.0001, 0.04, 1.01, 0.997),
        ]
        for T_init, lr_init, T_dec, lr_dec in sg_configs:
            remaining = deadline - time.monotonic() - 200
            if remaining < 10:
                break
            t_alloc = min(remaining, 80)
            print("  Smoothed grad T={}, lr={}, t={:.0f}s".format(T_init, lr_init, t_alloc))
            a_opt, _ = smoothed_grad_descent(state[1], time_limit=t_alloc,
                                             T_init=T_init, lr_init=lr_init,
                                             T_decay=T_dec, lr_decay=lr_dec)
            try_and_update(a_opt, state, "smooth_T{}".format(T_init))

        # Multi-peak subgradient
        mpsg_configs = [
            (5, 0.0015, 0.94),
            (10, 0.002, 0.93),
            (20, 0.0025, 0.92),
            (50, 0.001, 0.95),
        ]
        for top_k, lr, mom in mpsg_configs:
            remaining = deadline - time.monotonic() - 100
            if remaining < 8:
                break
            t_alloc = min(remaining, 50)
            print("  Multi-peak SG k={}, t={:.0f}s".format(top_k, t_alloc))
            a_opt, _ = multi_peak_sg(state[1], time_limit=t_alloc,
                                     top_k=top_k, lr_init=lr, mom=mom)
            try_and_update(a_opt, state, "mpsg_k{}".format(top_k))

    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 5: Final Polish (~350s) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])
        rng_polish = np.random.default_rng(seed + 777)

        # 5a: Coordinate descent
        remaining = deadline - time.monotonic() - 150
        if remaining > 10:
            t_alloc = min(remaining, 100)
            print("  Coordinate descent, t={:.0f}s".format(t_alloc))
            a_opt, _ = coordinate_descent_polish(state[1], time_limit=t_alloc, rng=rng_polish)
            try_and_update(a_opt, state, "coord_descent")

        # 5b: Simulated annealing
        remaining = deadline - time.monotonic() - 100
        if remaining > 8:
            t_alloc = min(remaining, 80)
            print("  Simulated annealing, t={:.0f}s".format(t_alloc))
            a_opt, _ = simulated_annealing_polish(state[1], time_limit=t_alloc,
                                                  rng=np.random.default_rng(seed + 888))
            try_and_update(a_opt, state, "sa_polish")

        # 5c: Fine smoothed gradient
        remaining = deadline - time.monotonic() - 60
        if remaining > 5:
            t_alloc = min(remaining, 50)
            print("  Fine smooth grad, t={:.0f}s".format(t_alloc))
            a_opt, _ = smoothed_grad_descent(state[1], time_limit=t_alloc,
                                             T_init=0.0001, lr_init=0.005,
                                             T_decay=1.005, lr_decay=0.999)
            try_and_update(a_opt, state, "fine_smooth")

        # 5d: Tiny perturbations + quick refinement (repeat)
        remaining = deadline - time.monotonic() - 10
        while remaining > 5:
            noise = rng_polish.standard_normal(n_best) * 0.0003
            f_pert = project_simplex(state[1] + noise)
            t_alloc = min(remaining - 3, 8)
            a_opt, _ = smoothed_grad_descent(f_pert, time_limit=t_alloc,
                                             T_init=0.0002, lr_init=0.003,
                                             T_decay=1.01, lr_decay=0.995)
            try_and_update(a_opt, state, "perturb_fine")
            remaining = deadline - time.monotonic() - 5

        # 5e: Final coordinate descent pass
        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            t_alloc = min(remaining, 20)
            a_opt, _ = coordinate_descent_polish(state[1], time_limit=t_alloc,
                                                 rng=np.random.default_rng(seed + 999))
            try_and_update(a_opt, state, "final_coord")

    print("  Phase 5 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(384) / 384
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
