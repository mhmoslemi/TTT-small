1. Efficient subgradient computation via FFT/correlate
2. Proximal gradient with adaptive softmax smoothing and annealing
3. LP-based Frank-Wolfe with dual-informed perturbations
4. Multi-n exploration with coarse-to-fine optimization
5. Simulated annealing with structured (sinusoidal/polynomial) moves
6. Final coordinate descent and micro-polish
"""
import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def score_of(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_subgradient(a, T):
    """Compute smoothed subgradient of max(conv(a,a)) using softmax weights."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    weights = np.exp((conv - max_c) / T)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.zeros(len(conv))
        weights[np.argmax(conv)] = 1.0
    else:
        weights = weights / wsum
    # sg[m] = 2 * sum_j weights[m+j] * a[j] for m=0..n-1
    sg = 2.0 * np.correlate(weights, a, mode='valid')
    return sg, conv


def topk_subgradient(a, K):
    """Hard top-K subgradient."""
    n = len(a)
    conv = fast_convolve(a)
    K = min(K, len(conv))
    top_idx = np.argpartition(conv, -K)[-K:]
    weights = np.zeros(len(conv))
    vals = conv[top_idx]
    order = np.argsort(vals)[::-1]
    for rank, idx in enumerate(order):
        weights[idx] = 1.0 / (rank + 1) ** 1.3
    wsum = weights.sum()
    if wsum > 0:
        weights /= wsum
    sg = 2.0 * np.correlate(weights, a, mode='valid')
    return sg, conv


def solve_lp_oracle(f, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            g = np.maximum(res.x, 0.0)
            return g
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle and adaptive relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.3:
            break

        g = solve_lp_oracle(f, M * m_exp, time_limit=min(15, remaining * 0.3))
        if g is None:
            m_exp = min(m_exp * 1.12, 4.0)
            if m_exp > 4.0:
                m_exp = 1.0
                g = np.ones(n) / n
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.03, 3.0)
            continue

        best_alpha = 0.0
        best_val = M

        # Coarse + fine line search
        alphas_coarse = np.linspace(0.01, 1.0, 15)
        for alpha in alphas_coarse:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0.01:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 4.0)

        if no_improve > 35:
            rng = np.random.default_rng(it * 2654435761 + 13)
            noise = rng.standard_normal(n) * 0.004
            # Structured: bias toward reducing the peak
            peak_k = int(np.argmax(conv))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            noise[js:je + 1] -= 0.002
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
            if verbose:
                print(f"    FW restart at it={it}, M={M:.8f}")

    return best_f, best_obj


def proximal_gradient_optimize(a0, time_limit, T_init=0.002, T_min=0.0001):
    """
    Proximal gradient with adaptive softmax smoothing.
    Anneals T from T_init to T_min for increasingly sharp max approximation.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    T = T_init
    lr = 0.02
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        sg, conv = softmax_subgradient(a, T)

        # Backtracking line search
        best_candidate = a
        best_val = conv.max()
        trial_lr = lr
        for _ in range(20):
            candidate = project_simplex(a - trial_lr * sg)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                break
            trial_lr *= 0.5

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive T and lr
        if no_improve > 20:
            T = max(T * 0.92, T_min)
        if no_improve > 50:
            lr *= 0.95
        if no_improve > 80:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = max(T_init * 0.5, T_min)
            lr = 0.02
            no_improve = 0

    return best_a, best_obj


def topk_proximal(a0, time_limit, K=10):
    """Proximal gradient with hard top-K subgradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    lr = 0.015
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        sg, conv = topk_subgradient(a, K)
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        sg = sg / sg_norm

        best_candidate = a
        best_val = conv.max()
        trial_lr = lr
        for _ in range(15):
            candidate = project_simplex(a - trial_lr * sg)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                break
            trial_lr *= 0.5

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                lr *= 0.97
            if no_improve > 100:
                rng = np.random.default_rng((it * 17 + 3) % (2**31))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                lr = 0.015
                no_improve = 0

    return best_a, best_obj


def structured_annealing(a0, time_limit):
    """
    Simulated annealing with structured perturbations.
    Uses sinusoidal, polynomial, and peak-targeted moves.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # Choose perturbation type
        rng = np.random.default_rng((it * 48271 + 1) % (2**32))
        pert_type = rng.integers(0, 5)

        scale = 0.002 + 0.003 * min(1.0, stall / 100.0)

        if pert_type == 0:
            # Random Gaussian
            noise = rng.standard_normal(n) * scale
        elif pert_type == 1:
            # Sinusoidal
            freq = rng.uniform(1, 5)
            phase = rng.uniform(0, 2 * np.pi)
            x = np.arange(n) / n
            noise = np.sin(2 * np.pi * freq * x + phase) * scale
            if rng.random() > 0.5:
                freq2 = rng.uniform(5, 15)
                phase2 = rng.uniform(0, 2 * np.pi)
                noise += np.sin(2 * np.pi * freq2 * x + phase2) * scale * 0.5
        elif pert_type == 2:
            # Polynomial (Legendre)
            order = rng.integers(2, 6)
            x = 2 * np.arange(n) / n - 1
            p = np.polynomial.legendre.legvander(x, order)
            coeffs = rng.standard_normal(order + 1)
            noise = (p @ coeffs) * scale / np.max(np.abs(p @ coeffs) + 1e-10)
        elif pert_type == 3:
            # Peak-targeted: reduce values contributing to the peak
            conv = fast_convolve(a)
            peak_k = int(np.argmax(conv))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            noise = rng.standard_normal(n) * scale * 0.3
            # Reduce contributors to peak
            for j in range(js, je + 1):
                partner = peak_k - j
                if 0 <= partner < n:
                    noise[j] -= scale * a[partner] * 2.0
        else:
            # Wavelet-like: block structure
            block_size = n // rng.integers(3, 8)
            noise = np.zeros(n)
            for start in range(0, n, block_size):
                end = min(start + block_size, n)
                noise[start:end] = rng.standard_normal() * scale

        candidate = project_simplex(a + noise)
        c = fast_convolve(candidate)
        obj = c.max()

        # Metropolis acceptance (simplified: always accept improvements,
        # accept worsened with small probability early on)
        if obj < best_obj:
            a = candidate
            best_obj = obj
            best_a = a.copy()
            stall = 0
        elif obj < best_obj * (1.0 + 0.001 * max(0, 1 - it / 500)):
            a = candidate
            stall += 1
        else:
            stall += 1

        if stall > 200:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            stall = 0

    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit):
    """Fine coordinate descent: try moving mass between pairs that reduce the peak."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]

        # Find the pair (src, dst) that gives the best improvement
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        affected = np.arange(js, je + 1)

        if len(affected) < 2:
            break

        # Sample a subset if too many
        rng = np.random.default_rng((it * 7919 + 3) % (2**31))
        if len(affected) > 20:
            sampled = rng.choice(affected, size=20, replace=False)
        else:
            sampled = affected

        best_gain = 0
        best_move = None

        for ii in range(len(sampled)):
            j1 = sampled[ii]
            contrib1 = a[peak_k - j1] if 0 <= peak_k - j1 < n else 0
            if contrib1 < 1e-10:
                continue
            for jj in range(len(sampled)):
                if jj == ii:
                    continue
                j2 = sampled[jj]
                contrib2 = a[peak_k - j2] if 0 <= peak_k - j2 < n else 0

                # Move mass from j1 (high contrib) to j2 (low contrib)
                if contrib1 <= contrib2:
                    continue

                for frac in [0.05, 0.15, 0.3, 0.5]:
                    delta = a[j1] * frac
                    if delta < 1e-10:
                        continue
                    a_test = a.copy()
                    a_test[j1] -= delta
                    a_test[j2] += delta
                    a_test = np.maximum(a_test, 0.0)
                    st = a_test.sum()
                    if st < 1e-15:
                        continue
                    a_test = a_test / st
                    c_new = fast_convolve(a_test)
                    new_max = c_new.max()
                    gain = best_obj - new_max
                    if gain > best_gain:
                        best_gain = gain
                        best_move = (j1, j2, delta)

        if best_move is not None and best_gain > 1e-13:
            j1, j2, delta = best_move
            a[j1] -= delta
            a[j2] += delta
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            c_new = fast_convolve(a)
            best_obj = c_new.max()
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 50:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            stall = 0
        if stall > 300:
            break

    return best_a, best_obj


def micro_polish(a0, time_limit):
    """Very fine perturbation polish."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        rng = np.random.default_rng((it * 2654435761 + 1) % (2**32))

        # Try multiple scales
        for scale in [0.0001, 0.0003, 0.0005, 0.001]:
            if time.monotonic() - t0 > time_limit - 0.2:
                break

            conv = fast_convolve(a)
            peak_k = int(np.argmax(conv))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            noise = rng.standard_normal(n) * scale
            for j in range(js, je + 1):
                partner = peak_k - j
                if 0 <= partner < n:
                    noise[j] -= scale * 1.5 * a[partner]

            candidate = project_simplex(a + noise)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                break

    return best_a, best_obj


def spectral_init(n, seed=42):
    """Create an initial sequence with a relatively flat power spectrum."""
    rng = np.random.default_rng(seed)
    # Start with a smooth bump function
    x = np.linspace(0, 1, n)
    # Super-Gaussian (flatter top)
    a = np.exp(-((x - 0.5) ** 2) / (2 * 0.15 ** 2))
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def interpolate_sequence(a, new_n):
    """Linearly interpolate a sequence to a new length."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search combining:
    - Multi-n exploration with FW-LP
    - Proximal gradient with adaptive smoothing
    - Structured simulated annealing
    - Coordinate descent polish
    - Micro-polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t0_total = time.monotonic()

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        sc = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t0_total
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return sc

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = spectral_init(768)
        n_parent = 768
    else:
        n_parent = len(parent)

    print(f"\nStarting search, parent n={n_parent}, best={best_score:.8f}")

    # === Phase 1: Multi-n exploration ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent, max(384, n_parent - 128), min(1200, n_parent + 128),
                               512, 640, 768, 896, 1024, 1152]))
    candidate_ns = [n for n in candidate_ns if 320 <= n <= 1200]

    for n_try in candidate_ns:
        if time.monotonic() > deadline - 1400:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = interpolate_sequence(parent, n_try)

        t_alloc = min(40, deadline - time.monotonic() - 1400)
        if t_alloc < 8:
            break
        print(f"  FW-LP n={n_try}, alloc={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(start, time_limit=t_alloc)
        try_seq(a_opt, f"fw_n{n_try}")

    print(f"  Phase 1 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep optimization on best n ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Extended FW-LP
        t_alloc = min(400, deadline - time.monotonic() - 800)
        if t_alloc > 30:
            print(f"  Deep FW n={n_best}, alloc={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_fw")

        # Perturbed restarts
        rng = np.random.default_rng(seed + 1)
        for scale in [0.003, 0.008, 0.015]:
            t_alloc = min(60, deadline - time.monotonic() - 700)
            if t_alloc < 10:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"fw_pert{scale}")

    print(f"  Phase 2 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Proximal gradient with adaptive smoothing ===
    print("\n=== Phase 3: Proximal Gradient ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Softmax-proximal with different T settings
        for T_init in [0.001, 0.003, 0.005]:
            t_alloc = min(100, deadline - time.monotonic() - 500)
            if t_alloc < 10:
                break
            print(f"  ProxGrad T_init={T_init}, alloc={t_alloc:.0f}s")
            a_opt, _ = proximal_gradient_optimize(best_seq, time_limit=t_alloc, T_init=T_init)
            try_seq(a_opt, f"proxT{T_init}")

        # Top-K proximal
        for K in [5, 15, 30]:
            t_alloc = min(80, deadline - time.monotonic() - 400)
            if t_alloc < 10:
                break
            print(f"  TopK-Prox K={K}, alloc={t_alloc:.0f}s")
            a_opt, _ = topk_proximal(best_seq, time_limit=t_alloc, K=K)
            try_seq(a_opt, f"topk{K}")

    print(f"  Phase 3 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Structured annealing ===
    print("\n=== Phase 4: Structured Annealing ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(250, deadline - time.monotonic() - 250)
        if t_alloc > 20:
            print(f"  Annealing, alloc={t_alloc:.0f}s")
            a_opt, _ = structured_annealing(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "anneal")

        # ANnealing from different starting points
        rng2 = np.random.default_rng(seed + 99)
        for scale in [0.01, 0.02]:
            t_alloc = min(80, deadline - time.monotonic() - 200)
            if t_alloc < 10:
                break
            start_pert = project_simplex(best_seq + rng2.standard_normal(len(best_seq)) * scale)
            a_opt, _ = structured_annealing(start_pert, time_limit=t_alloc)
            try_seq(a_opt, f"anneal_p{scale}")

    print(f"  Phase 4 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Coordinate descent polish ===
    print("\n=== Phase 5: Coordinate Descent ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(150, deadline - time.monotonic() - 100)
        if t_alloc > 10:
            print(f"  Coord descent, alloc={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord")

    print(f"  Phase 5 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Final micro-polish + one more FW ===
    print("\n=== Phase 6: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Micro polish
        t_alloc = min(60, deadline - time.monotonic() - 60)
        if t_alloc > 5:
            print(f"  Micro polish, alloc={t_alloc:.0f}s")
            a_opt, _ = micro_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "micro")

        # Final FW
        t_alloc = min(50, deadline - time.monotonic() - 30)
        if t_alloc > 5:
            print(f"  Final FW, alloc={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_fw")

        # Final proximal with small T
        t_alloc = min(30, deadline - time.monotonic() - 15)
        if t_alloc > 5:
            a_opt, _ = proximal_gradient_optimize(best_seq, time_limit=t_alloc,
                                                   T_init=0.0005, T_min=0.0001)
            try_seq(a_opt, "final_prox")

        # Another coordinate descent pass
        t_alloc = min(20, deadline - time.monotonic() - 10)
        if t_alloc > 3:
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_coord")

    print(f"  Phase 6 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy() if parent is not None else np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - t0_total
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={total_time:.0f}s")
    return [float(x) for x in best_seq]
