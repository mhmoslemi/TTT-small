"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-n exploration with L-BFGS on smoothed objective
2. Temperature annealing (from smooth to sharp)
3. Active-set LP refinement for peak reduction
4. Final coordinate descent polish

import numpy as np
from scipy.optimize import minimize, linprog
import time

def fft_autoconv(a):
    """Compute linear auto-convolution of a using FFT."""
    n = len(a)
    size = 1 << (2*n - 2).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n - 1]

def softmax(z):
    """Stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def obj_and_grad(z, T):
    """Compute smoothed objective and gradient w.r.t. z."""
    a = softmax(z)
    n = len(a)
    conv = fft_autoconv(a)
    
    # Objective: (1/T) * log(sum exp(T * conv))
    scaled = T * conv
    scaled_max = scaled.max()
    log_sum = scaled_max + np.log(np.sum(np.exp(scaled - scaled_max)))
    obj = log_sum / T
    
    # Gradient
    # q[k] = exp(T*conv[k]) / sum(exp(T*conv[k]))
    q = np.exp(scaled - scaled_max)
    q /= q.sum()
    
    # grad_a[j] = 2 * sum_m q[j+m] * a[m] for m=0..n-1
    # = 2 * correlate(q, a, 'valid'-style)
    full_corr = np.correlate(q, a, 'full')
    grad_a = 2.0 * full_corr[n-1:2*n-1]
    
    # Transform to grad_z
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    
    return obj, grad_z

def run_lbgfs(z0, T, maxiter, tol):
    """Run L-BFGS-B on the smoothed objective."""
    def fun(z):
        return obj_and_grad(z, T)
    
    result = minimize(fun, z0, method='L-BFGS-B', jac=True,
                      options={'maxiter': maxiter, 'ftol': tol, 'gtol': 1e-8,
                               'maxls': 50})
    return result.x, result.fun, result.nit, result.success

def active_set_lp(a, K, n):
    """Solve LP to find descent direction for active lags K."""
    # For each active lag k, the linearized gradient is g_k[i] = 2*a[k-i]*I(0<=k-i<n)
    # We want: min t s.t. g_k^T d <= t, sum(d)=0, -1<=d[i]<=1
    m = len(K)
    # A_ub: first m rows are g_k, then identity for d<=1, then -identity for -d<=1
    A_ub = np.zeros((m + 2*n, n + 1))
    b_ub = np.zeros(m + 2*n)
    
    for i, k in enumerate(K):
        # g_k[d] = sum_j 2*a[k-j]*d[j] for 0<=k-j<n
        for j in range(n):
            partner = k - j
            if 0 <= partner < n:
                A_ub[i, j] = 2.0 * a[partner]
        A_ub[i, n] = -1.0  # coefficient of t
        b_ub[i] = 0.0
    
    # d[i] <= 1: A_ub row has 1 at column i, b=1
    for j in range(n):
        A_ub[m + j, j] = 1.0
        b_ub[m + j] = 1.0
    # -d[i] <= 1: A_ub row has -1 at column i, b=1
    for j in range(n):
        A_ub[m + n + j, j] = -1.0
        b_ub[m + n + j] = 1.0
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Minimize t (the last variable)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      method='highs', options={'presolve': True})
        if res.success:
            d = res.x[:n]
            t = res.x[n]
            return d, t
    except Exception:
        pass
    return None, 0.0

def fft_autoconv(a):
    """Compute linear auto-convolution of a using FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def softmax_func(z):
    """Stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def obj_grad_full(z, T):
    """Smoothed log-sum-exp objective and gradient over all lags."""
    a = softmax_func(z)
    n = len(a)
    conv = fft_autoconv(a)
    scaled = T * conv
    shifted = scaled - scaled.max()
    log_sum = shifted.max() + np.log(np.sum(np.exp(shifted)))
    obj = (scaled.max() + np.log(np.sum(np.exp(shifted)))) / T
    q = np.exp(shifted)
    q_sum = q.sum()
    if q_sum < 1e-300:
        q = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        q /= q_sum
    full_corr = np.correlate(q, a, mode='full')
    grad_a = 2.0 * full_corr[n - 1:2 * n - 1]
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    return obj, grad_z


def obj_grad_topk(z, T, topk):
    """Smoothed objective using only top-k lags for focused gradient."""
    a = softmax_func(z)
    n = len(a)
    conv = fft_autoconv(a)
    if topk >= 2 * n - 1:
        return obj_grad_full(z, T)
    k_idx = np.argpartition(conv, -topk)[-topk:]
    conv_k = conv[k_idx]
    scaled = T * conv_k
    shifted = scaled - scaled.max()
    obj = (scaled.max() + np.log(np.sum(np.exp(shifted)))) / T
    q_k = np.exp(shifted)
    qk_sum = q_k.sum()
    if qk_sum < 1e-300:
        q_k = np.ones(topk) / topk
    else:
        q_k /= qk_sum
    q = np.zeros(2 * n - 1)
    q[k_idx] = q_k
    full_corr = np.correlate(q, a, mode='full')
    grad_a = 2.0 * full_corr[n - 1:2 * n - 1]
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    return obj, grad_z


def run_lbgfs(z0, T, maxiter, topk=None, deadline=None):
    """Run L-BFGS-B on the smoothed objective."""
    if topk is not None and topk < 2 * len(z0) - 1:
        def fun(z):
            return obj_grad_topk(z, T, topk)
    else:
        def fun(z):
            return obj_grad_full(z, T)
    opts = {'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-7, 'maxls': 40}
    try:
        res = minimize(fun, z0, method='L-BFGS-B', jac=True, options=opts)
        return res.x, res.fun, res.nit
    except Exception:
        return z0, None, 0


def active_set_direction(a, K):
    """Find descent direction for active lags using LP."""
    n = len(a)
    m = len(K)
    if m == 0 or n > 3000:
        return None
    A_ub = np.zeros((m + 2 * n, n + 1))
    b_ub = np.zeros(m + 2 * n)
    for i, k in enumerate(K):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[i, j] = 2.0 * a[k - j]
        A_ub[i, n] = -1.0
    for j in range(n):
        A_ub[m + j, j] = 1.0
        b_ub[m + j] = 1.0
    for j in range(n):
        A_ub[m + n + j, j] = -1.0
        b_ub[m + n + j] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    c = np.zeros(n + 1)
    c[n] = 1.0
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      method='highs', options={'presolve': True, 'time_limit': 10})
        if res.success:
            d = res.x[:n]
            t_val = res.x[n]
            if t_val < -1e-10:
                return d
    except Exception:
        pass
    return None

def coordinate_polish(a0, deadline, n_trials=2000):
