"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
- Fast smoothed gradient descent with adaptive temperature and Nesterov momentum
- Top-k peak gradient (bottleneck smoothing) for precision
- Periodic LP-based Frank-Wolfe for global direction
- Simulated annealing polish with structured perturbations
- Multi-n exploration with interpolation
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_correlate(a, b):
    """Valid cross-correlation: result[j] = sum_i a[i+j]*b[i] (aligned)."""
    n = min(len(a), len(b))
    result = np.zeros(n)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.rfft(a[:n], fft_size)
    B = np.fft.rfft(b[:n], fft_size)
    corr = np.fft.irfft(A * np.conj(B), fft_size)
    return corr[n - 1:2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(conv, a, T):
    """Compute gradient of smoothed max using softmax weights."""
    n = len(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()
    # grad_j = 2 * sum_k p_k * a_{k-j} for valid k-j
    # This is 2 * correlate(p, a) (valid)
    grad = 2.0 * fast_correlate(p, a)
    return grad, p


def compute_tophk_grad(conv, a, k):
    """Compute gradient of average of top-k peaks."""
    n = len(a)
    topk = np.argpartition(conv, -k)[-k:]
    p = np.zeros(len(conv))
    p[topk] = 1.0 / k
    grad = 2.0 * fast_correlate(p, a)
    return grad, p


def smoothed_grad_descent(a0, time_limit, T_init=0.001, T_min=0.0001,
                          lr_init=0.02, momentum=0.9, top_k_switch=500):
    """
    Adaptive smoothed gradient descent on the simplex with Nesterov momentum.
    Switches from softmax to top-k gradient when close to convergence.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)  # momentum
    t_nest = 1.0
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    decay = 0.00005

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Nesterov lookahead
        alpha_n = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta_n = (t_nest - 1.0) / alpha_n
        y = (1.0 + beta_n) * a - beta_n * y
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y / ys
        t_nest += 1.0

        conv = fast_convolve(y)
        peak = conv.max()

        # Choose gradient type based on how peaked the conv is
        ratio = peak / (conv.mean() + 1e-30)
        if ratio > 1.8 or it > top_k_switch:
            k = max(1, min(20, int(n * 0.02)))
            grad, _ = compute_tophk_grad(conv, y, k)
        else:
            grad, _ = compute_smoothed_grad(conv, y, T)

        # Project gradient to tangent space
        grad = grad - np.mean(grad)

        # Momentum update
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        # Evaluate
        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            # Adapt T: decrease when making progress
            T = max(T_min, T * 0.999)
        else:
            no_improve += 1
            # Don't accept the step
            lr *= 0.995
            T = max(T_min, T * 0.998)

        # Adapt learning rate
        if no_improve > 50:
            lr = max(0.0001, lr * 0.9)
            T = max(T * 1.02, T_min * 0.5)

        # Restart when stuck
        if no_improve > 200:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            noise_scale = 0.001 * min(1.0, no_improve / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            y = a.copy()
            v = np.zeros(n)
            t_nest = 1.0
            T = max(T_init * 0.5, T)
            lr = lr_init * 0.5
            no_improve = 0

    return best_a, best_obj


def fw_lp_optimize(f0, time_limit, m_expand=1.0):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Solve LP: max sum(g) s.t. conv(f,g) <= M*m_exp, g >= 0
        c = -np.ones(n)
        size = 2 * n - 1
        A_ub = np.zeros((size, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(size, M * m_exp)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': 20})
            if res.success:
                g = np.maximum(res.x, 0.0)
            else:
                g = None
        except Exception:
            g = None

        if g is None or g.sum() < 1e-15:
            m_exp *= 1.2
            if m_exp > 5.0:
                g = np.ones(n) / n
                m_exp = 1.0
            else:
                continue

        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.02)
            hi = min(1.0, best_alpha + 0.02)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 30:
            rng = np.random.default_rng(it * 104729 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def annealing_polish(a0, time_limit, n_trials_per_iter=5):
    """Simulated annealing with structured perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    current_obj = best_obj

    T_anneal = 0.002
    T_floor = 0.0001
    cooling = 0.9995

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))

        for trial in range(n_trials_per_iter):
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            rng = np.random.default_rng((it * 1000 + trial * 7 + 3) % (2**31))
            scale = T_anneal * (0.5 + 0.5 * rng.random())

            # Structured perturbation: bias against peak contributors
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            noise = rng.standard_normal(n) * scale
            for j in range(js, je + 1):
                partner = peak_k - j
                if 0 <= partner < n:
                    noise[j] -= scale * 1.5 * a[partner]

            candidate = project_simplex(a + noise)
            c_new = fast_convolve(candidate)
            obj_new = c_new.max()

            delta = obj_new - current_obj
            if delta < 0 or rng.random() < np.exp(-delta / max(T_anneal, 1e-10)):
                a = candidate
                current_obj = obj_new
                if obj_new < best_obj - 1e-15:
                    best_obj = obj_new
                    best_a = a.copy()

        T_anneal = max(T_floor, T_anneal * cooling)

    return best_a, best_obj


def rfgd_hybrid(a0, time_limit):
    """
    Riemannian Frank-Wolfe Gradient Descent hybrid:
    Alternates between gradient steps and LP-based direction finding.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    T = best_obj * 0.05
    lr = 0.03

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lp_interval = 50

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # LP step every lp_interval iterations
        if it % lp_interval == 0 and time.monotonic() - t0 < time_limit - 30:
            a_opt, obj_opt = fw_lp_optimize(a, time_limit=15)
            if obj_opt < best_obj - 1e-15:
                best_obj = obj_opt
                best_a = a_opt.copy()
                a = a_opt.copy()
                no_improve = 0
                T = best_obj * 0.05
                continue

        # Gradient step
        conv = fast_convolve(a)
        peak = conv.max()
        T = max(peak * 0.1, best_obj * 0.01)

        grad, p = compute_smoothed_grad(conv, a, T)
        grad = grad - np.mean(grad)

        # Backtracking line search
        best_candidate = a
        best_val = peak
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.125]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.01, 0.1)
        else:
            no_improve += 1
            lr *= 0.98

        if no_improve > 100:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            lr = 0.03
            T = best_obj * 0.05
            no_improve = 0

        if no_improve > 500:
            break

    return best_a, best_obj


def try_seq(a, best_state, label=""):
    """Try a sequence and update best state if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_state
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_state[1]:
        best_state = (a_norm.copy(), score)
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return best_state
