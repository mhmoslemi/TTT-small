def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_oracle(f, M, time_limit_lp=20):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit_lp})
        if res.success:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

            break

        M_try = M * m_exp
        lp_time = min(15, max(3, remaining * 0.1))
        g = solve_lp_oracle(f, M_try, time_limit_lp=lp_time)
        if g is None:
            m_exp *= 1.2
            if m_exp > 4.0:
                m_exp = 1.0
                g = np.ones(n) / n
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)

        if no_improve > 30:
            rng = np.random.default_rng(it * 7919 + 3)
            noise = rng.standard_normal(n) * 0.003
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(a0, time_limit):
    """Aggressive coordinate descent targeting top peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        top_k = min(5, len(conv))
        peak_lags = np.argpartition(conv, -top_k)[-top_k:]
        peak_lags = peak_lags[np.argsort(conv[peak_lags])[::-1]]

        improved = False

        for k_peak in peak_lags:
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            affected = np.arange(js, je + 1)

            if len(affected) < 2:
                continue

            # Sample pairs for efficiency
            if len(affected) > 40:
                rng_s = np.random.default_rng(it * 17 + k_peak)
                affected = rng_s.choice(affected, size=40, replace=False)

            for ii in range(len(affected)):
                for jj in range(ii + 1, len(affected)):
                    j1 = affected[ii]
                    j2 = affected[jj]

                    c1 = a[k_peak - j1] if 0 <= k_peak - j1 < n else 0
                    c2 = a[k_peak - j2] if 0 <= k_peak - j2 < n else 0

                    if c1 >= c2 and a[j1] > 1e-8:
                        src, dst = j1, j2
                    elif c2 > c1 and a[j2] > 1e-8:
                        src, dst = j2, j1
                    else:
                        continue

                    for frac in [0.05, 0.15, 0.3, 0.5]:
                        delta = a[src] * frac
                        if delta < 1e-10:
                            continue
                        a_test = a.copy()
                        a_test[src] -= delta
                        a_test[dst] += delta
                        a_test = np.maximum(a_test, 0.0)
                        s_t = a_test.sum()
                        if s_t < 1e-15:
                            continue
                        a_test = a_test / s_t
                        new_max = fast_convolve(a_test).max()
                        if new_max < best_obj - 1e-13:
                            a = a_test
                            best_obj = new_max
                            best_a = a.copy()
                            improved = True
                            break

        if not improved:
            stall += 1
            if stall > 15:
                rng = np.random.default_rng((it * 31 + 99) % (2**31))
                scale = 0.001 + 0.002 * min(1.0, stall / 30.0)
                a = project_simplex(best_a + rng.standard_normal(n) * scale)
                new_max = fast_convolve(a).max()
                if new_max < best_obj:
                    best_obj = new_max
                    best_a = a.copy()
                stall = 0

        if stall > 100:
            break

    return best_a, best_obj


def micro_polish(a0, time_limit):
    """Very fine perturbation-based polish."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    scales = [0.0002, 0.0005, 0.001, 0.002, 0.0008]
    si = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        scale = scales[si % len(scales)]
        if it % 50 == 0:
            si += 1

        rng = np.random.default_rng((it * 2654435761 + 42) % (2**32))

        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)

        for trial in range(8):
            noise = rng.standard_normal(n) * scale
            # Bias toward reducing peak contributors
            for j in range(js, je + 1):
                partner = peak_k - j
                if 0 <= partner < n:
                    noise[j] -= scale * 1.5 * a[partner]

            candidate = project_simplex(a + noise)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-14:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                break

    return best_a, best_obj


def asymmetric_init(n, seed=0):
    """Create an asymmetric initialization biased to one side."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    # Biased distribution (exponential-like)
    a = np.exp(-x * rng.uniform(1.0, 3.0))
    a = a + rng.uniform(0, 0.1) * rng.standard_normal(n)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Hybrid optimization combining FISTA, LP-guided FW,
    coordinate descent, and micro-polish with multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    t0_total = time.monotonic()
    deadline = t0_total + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t0_total
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(384) / 384

    n_parent = len(parent)

    # === Phase 1: Multi-n exploration with FISTA (~250s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent, n_parent - 64, n_parent + 64,
                                384, 512, 640, 768, 896, 1024, 1152]))
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 1300]

    for n_try in candidate_ns:
        if time.monotonic() > deadline - 1300:
            break

        t_alloc = min(25, deadline - time.monotonic() - 1300)
        if t_alloc < 5:
            break

        # Try multiple initializations
        inits = []
        if n_try == n_parent:
            inits.append(parent.copy())
        else:
            resampled = np.interp(np.linspace(0, 1, n_try),
                                   np.linspace(0, 1, n_parent), parent)
            resampled = np.maximum(resampled, 0)
            if resampled.sum() > 1e-10:
                resampled = resampled / resampled.sum()
            inits.append(resampled)

        # Asymmetric init
        inits.append(asymmetric_init(n_try, seed=n_try + seed))

        if n_try != n_parent and len(inits) < 3:
            inits.append(np.ones(n_try) / n_try)

        best_for_n = None
        best_obj_n = float('inf')
        for ii, init in enumerate(inits):
            t_sub = max(3, t_alloc / len(inits))
            a_opt, obj = fista_smooth(init, T_init=0.01, time_limit=t_sub, lr_init=0.015)
            if obj < best_obj_n:
                best_obj_n = obj
                best_for_n = a_opt

        if best_for_n is not None:
            try_seq(best_for_n, f"fista_n{n_try}")

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FISTA with decreasing T (~350s) ===
    print("\n=== Phase 2: Deep FISTA ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        configs = [
            {'T_init': 0.005, 'lr_init': 0.012},
            {'T_init': 0.002, 'lr_init': 0.008},
            {'T_init': 0.001, 'lr_init': 0.005},
        ]
        for ci, cfg in enumerate(configs):
            t_alloc = min(110, deadline - time.monotonic() - 700)
            if t_alloc < 10:
                break
            print(f"  FISTA T={cfg['T_init']}, lr={cfg['lr_init']}, t={t_alloc:.0f}s")
            a_opt, _ = fista_smooth(best_seq, T_init=cfg['T_init'],
                                     time_limit=t_alloc, lr_init=cfg['lr_init'])
            try_seq(a_opt, f"fista2_{ci}")

    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: LP-guided FW (~300s) ===
    print("\n=== Phase 3: LP-Guided FW ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(200, deadline - time.monotonic() - 500)
        if t_alloc > 15:
            print(f"  FW-LP n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "fw_lp")

        # Perturbed restart
        for scale in [0.003, 0.008]:
            t_alloc = min(80, deadline - time.monotonic() - 400)
            if t_alloc < 10:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * scale)
            a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"fw_pers{scale}")

    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Coordinate Descent (~300s) ===
    print("\n=== Phase 4: Coordinate Descent ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(250, deadline - time.monotonic() - 250)
        if t_alloc > 10:
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord")

    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: FISTA again from improved point (~150s) ===
    print("\n=== Phase 5: Second FISTA Pass ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(120, deadline - time.monotonic() - 150)
        if t_alloc > 10:
            print(f"  FISTA pass 2, t={t_alloc:.0f}s")
            a_opt, _ = fista_smooth(best_seq, T_init=0.001,
                                     time_limit=t_alloc, lr_init=0.004)
            try_seq(a_opt, "fista3")

    print(f"  Phase 5 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Micro-polish and final tweaks (~130s) ===
    print("\n=== Phase 6: Micro-Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Micro polish
        t_alloc = min(80, deadline - time.monotonic() - 50)
        if t_alloc > 5:
            print(f"  Micro polish, t={t_alloc:.0f}s")
            a_opt, _ = micro_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "micro")

        # Another coord descent on polished
        t_alloc = min(40, deadline - time.monotonic() - 20)
        if t_alloc > 5:
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord2")

        # Final tiny FISTA
        t_alloc = min(20, deadline - time.monotonic() - 5)
        if t_alloc > 3:
            a_opt, _ = fista_smooth(best_seq, T_init=0.0005,
                                     time_limit=t_alloc, lr_init=0.002)
            try_seq(a_opt, "fista_final")

    print(f"  Phase 6 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, total time={time.monotonic()-t0_total:.0f}s)")
    return [float(x) for x in best_seq]
