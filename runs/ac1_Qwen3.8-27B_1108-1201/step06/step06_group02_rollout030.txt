import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.ndimage import gaussian_filter1d
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def solve_lp_with_dual(f, M_array, time_limit=5):
    n = len(f)
    n_rows = 2 * n - 1
    n_cols = n
    
    # Build sparse A_ub
    # A_ub[k, m] = f[k-m] if 0 <= k-m < n
    # For column m: rows m to m+n-1, values f[0:n]
    rows = np.concatenate([np.arange(m, m + n) for m in range(n_cols)])
    cols = np.concatenate([np.full(n, m) for m in range(n_cols)])
    vals = np.tile(f, n_cols)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(n_rows, n_cols))
    c = -np.ones(n_cols)
    bounds = [(0, None)] * n_cols
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=M_array, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            dual = None
            if hasattr(res, 'ineqlin_dual'):
                dual = res.ineqlin_dual
            return np.maximum(res.x, 0.0), dual
    except Exception:
        pass
    return None, None

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_sparsity(n):
    """Precompute row/col indices for the LP constraint matrix."""
    rows = np.concatenate([np.arange(m, m + n) for m in range(n)])
    cols = np.concatenate([np.full(n, m) for m in range(n)])
    return rows, cols


def solve_lp(f, M_array, sparsity, time_limit=5):
    n = len(f)
    rows, cols = sparsity
    vals = np.tile(f, n)
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=M_array, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            dual = None
            if hasattr(res, 'ineqlin_dual') and res.ineqlin_dual is not None:
                dual = res.ineqlin_dual
            return np.maximum(res.x, 0.0), dual
    except Exception:
        pass
    return None, None

        print("  Deep FW n={}, t={:.0f}s".format(n_best, t_alloc))
        a_opt, _ = fw_momentum(state_seq, time_limit=t_alloc, sparsity=sparsity, beta=0.3)

        if np.isfinite(score) and score < state_score:

            state_seq = a_opt.copy()
            print("    Deep FW: {:.8f}".format(score))

        # Perturbed restarts with different momentum
        for pert_scale, beta in [(0.005, 0.2), (0.01, 0.4), (0.02, 0.3)]:
            remaining = (deadline - time.monotonic()) * 0.25 - 5
            if remaining < 15:

            f_pert = project_simplex(state_seq + rng.standard_normal(n_best) * pert_scale)

                print("    FW perturb(scale={}, beta={}): {:.8f}".format(pert_scale, beta, score2))

    print("  Phase 2 best: {:.8f} (elapsed {:.0f}s)".format(state_score, time.monotonic() - t_start))

    # === Phase 3: Diversified Local Optimization (~400s) ===
    print("\n=== Phase 3: Diversified Local Optimization ===")
    t_start = time.monotonic()

    if state_seq is not None:

        sparsity = build_sparsity(n_best)

        # Smoothed gradient with multiple temperature settings
        for T_init, lr_init in [(0.0005, 0.04), (0.001, 0.03), (0.003, 0.02), (0.0002, 0.05)]:

    # === Phase 4: Final Polish (~200s) ===
    print("\n=== Phase 4: Final Polish ===")
    t_start = time.monotonic()

    if state_seq is not None:
        n_best = len(state_seq)
        sparsity = build_sparsity(n_best)

        # Fine smoothed gradient
        for T_init, lr_init in [(0.0001, 0.01), (0.0003, 0.008), (0.00005, 0.015)]:
