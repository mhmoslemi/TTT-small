import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n <= 1:
        return np.array([max(float(v[0]) if n == 1 else 0.0, 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute 2*n*max(conv)/sum^2 for non-negative a."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max() / (s * s)


def lse_smooth_max(conv, beta):
    """Compute LSE approximation of max and its gradient weights."""
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        return conv.max(), np.ones(len(conv)) / len(conv)
    lse_val = (bc_max + np.log(total)) / beta
    weights = exp_bc / total
    return lse_val, weights


def subgradient_lse(f, weights):
    """Compute weighted subgradient of LSE(beta * max_k (f*f)[k]) wrt f."""
    n = len(f)
    conv_len = 2 * n - 1
    grad = np.zeros(n)
    for k in range(conv_len):
        w = weights[k]
        if w < 1e-18:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        i_idx = np.arange(lo, hi + 1)
        grad[i_idx] += w * 2.0 * f[k - i_idx]
    return grad

    n = len(f)

    for idx, k in enumerate(peak_indices):

def fw_lp_step(f, M, top_k=40, time_limit=10):
    """Single Frank-Wolfe step with LP oracle."""
    n = len(f)
    conv = fast_convolve(f)
    k_select = min(top_k, 2 * n - 1)
    peak_idx = np.argpartition(conv, -k_select)[-k_select:]
    center = n - 1
    if center not in peak_idx:
        peak_idx = np.append(peak_idx, center)

    rows = []
    cols = []
    vals = []

            gs = g.sum()

        pass
    return None

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n <= 1:
        return np.array([max(float(v[0]) if n == 1 else 0.0, 0.0)])

    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
