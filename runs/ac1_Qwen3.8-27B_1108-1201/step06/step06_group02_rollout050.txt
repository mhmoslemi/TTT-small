def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def conv_grad_at_k(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if js > je:
        return grad
    valid = np.arange(js, je + 1)
    grad[valid] = 2.0 * a[k - valid]
    return grad

def mirror_descent_optimize(a0, time_limit, T_init, lr_init):
    """Mirror descent (multiplicative weights) with adaptive temperature."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        
        # Softmax-weighted peak selection
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()
        
        # Aggregate gradient from weighted peaks
        # Only use peaks with significant weight for efficiency
        top_indices = np.where(w > w.max() * 0.01)[0]
        grad = np.zeros(n)
        for k in top_indices:
            if w[k] < 1e-12:
                continue
            grad += w[k] * conv_grad_at_k(a, k)
        
        # Multiplicative update
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn
        
        obj = fast_convolve(a_new).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive parameters
        if no_improve > 10:
            T *= 1.08
            lr *= 0.98
        if no_improve > 30:
            T *= 0.7
        if no_improve > 100:
            rng = np.random.default_rng((it * 48271) % (2**31))
            noise = 0.001 + 0.004 * min(1.0, (no_improve - 100) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj


def mass_redistribute(a0, time_limit):

def subgradient_momentum(a0, time_limit, top_k=20, lr_init=0.003, mom_init=0.93):

def generate_structural_init(n, strategy, rng):
    """Generate diverse structural initializations."""
    x = np.arange(n, dtype=np.float64)
    
    if strategy == 'uniform':
        return np.ones(n) / n
    
    elif strategy == 'edge_weighted':
        w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
        return w / w.sum()
    
    elif strategy == 'gaussian':
        sigma = rng.uniform(0.1, 0.3) * n
        center = rng.uniform(0.2, 0.8) * (n - 1)
        g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return g / g.sum()
    
    elif strategy == 'cosine_taper':
        h = 0.5 + 0.5 * np.cos(np.pi * x / (n - 1))
        h = np.maximum(h, 0.01)
        return h / h.sum()
    
    elif strategy == 'bimodal':
        c1 = rng.uniform(0.2, 0.4) * (n - 1)
        c2 = rng.uniform(0.6, 0.8) * (n - 1)
        sigma = rng.uniform(0.05, 0.15) * n
        g = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
        return g / g.sum()
    
    elif strategy == 'high_freq':
        freq = rng.choice([3, 5, 7, 11, 13])
        h = 1.0 + 0.3 * np.cos(2.0 * np.pi * freq * x / n)
        h = np.maximum(h, 0.05)
        return h / h.sum()
    
    elif strategy == 'plateau':
        # Flat region with tapered edges
        frac = rng.uniform(0.3, 0.7)
        edge = rng.uniform(0.05, 0.2)
        h = np.zeros(n)
        for i in range(n):
            t = i / (n - 1)
            if edge < t < 1 - edge:
                h[i] = 1.0
            elif t <= edge:
                h[i] = t / edge
            else:
                h[i] = (1 - t) / edge
        h = np.maximum(h, 0.01)
        return h / h.sum()
    
    elif strategy == 'quantized':
        # Few distinct levels
        base = rng.dirichlet(np.ones(n) * 0.5)
        levels = rng.integers(3, 8)
        q = np.digitize(base, np.quantile(base, np.linspace(0, 1, levels + 1)[1:-1]))
        q = q.astype(float) + 1.0
        return q / q.sum()
    
    else:
        return rng.dirichlet(np.ones(n) * 0.3)

def resample_sequence(parent, n_new):
    """Resample a sequence to a different length."""
    s = parent.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    parent_norm = parent / s
    x_old = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_new)
    new_seq = np.interp(x_new, x_old, parent_norm)
    new_seq = np.maximum(new_seq, 0)
    s_new = new_seq.sum()
    if s_new < 1e-15:
        return np.ones(n_new) / n_new
    return new_seq / s_new

    remaining = deadline - time.monotonic() - 1300
    for strat in strategies:
        if remaining < 15:
            break
        init = generate_structural_init(n_parent, strat, rng)
        t_alloc = min(remaining, 20)
        a_opt, _ = fw_lp_optimize(init, time_limit=t_alloc, max_constraints=80)
        try_and_update(a_opt, state, "init_{}".format(strat))
        remaining = deadline - time.monotonic() - 1300
    
    print("  Phase 1 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))
    
    # === Phase 2: Deep Optimization (~700s) ===
    print("\n=== Phase 2: Deep FW + Mirror Descent ===")
    t_phase = time.monotonic()
    
    if state[1] is not None:
        n_best = len(state[1])
        
        # Alternating FW and mirror descent
        current = state[1].copy()
        
        # Deep FW
        remaining = deadline - time.monotonic() - 600
        if remaining > 30:
            t_alloc = min(remaining, 200)
            print("  Deep FW n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = fw_lp_optimize(current, time_limit=t_alloc, max_constraints=200)
            try_and_update(a_opt, state, "deep_fw")
            current = a_opt
        
        # Mirror descent
        remaining = deadline - time.monotonic() - 500
        if remaining > 20:
            t_alloc = min(remaining, 150)
            print("  Mirror descent, t={:.0f}s".format(t_alloc))
            a_opt, _ = mirror_descent_optimize(current, time_limit=t_alloc, T_init=0.0005, lr_init=0.03)
            try_and_update(a_opt, state, "deep_md")
            current = a_opt
        
        # Another FW pass
        remaining = deadline - time.monotonic() - 400
        if remaining > 20:
            t_alloc = min(remaining, 100)
            print("  FW pass 2, t={:.0f}s".format(t_alloc))
            a_opt, _ = fw_lp_optimize(current, time_limit=t_alloc, max_constraints=200)
            try_and_update(a_opt, state, "fw2")
            current = a_opt
        
        # Perturbed restarts
        for scale in [0.002, 0.005, 0.01, 0.02]:
            remaining = deadline - time.monotonic() - 300
            if remaining < 15:
                break
            f_pert = project_simplex(state[1] + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 40)
            a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc, max_constraints=150)
            try_and_update(a_opt, state, "fw_pert_{}".format(scale))
    
    print("  Phase 2 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))
    
    # === Phase 3: Subgradient + Mass Redistribution (~400s) ===
    print("\n=== Phase 3: Subgradient + Redistribution ===")
    t_phase = time.monotonic()
    
    if state[1] is not None:
        # Multiple subgradient configs
        for tk, lr, mom in [(10, 0.002, 0.94), (30, 0.003, 0.92), (50, 0.001, 0.95), (100, 0.002, 0.93)]:
            remaining = deadline - time.monotonic() - 150
            if remaining < 10:
                break
            t_alloc = min(remaining, 60)
            print("  SubGrad k={}, lr={}, t={:.0f}s".format(tk, lr, t_alloc))
            a_opt, _ = subgradient_momentum(state[1], time_limit=t_alloc,
                                            top_k=tk, lr_init=lr, mom_init=mom)
            try_and_update(a_opt, state, "sg_k{}_lr{}".format(tk, int(lr*1000)))
        
        # Mass redistribution
        remaining = deadline - time.monotonic() - 80
        if remaining > 10:
            t_alloc = min(remaining, 80)
            print("  Mass redistribution, t={:.0f}s".format(t_alloc))
            a_opt, _ = mass_redistribute(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "mass_red")
        
        # Another round of mirror descent with different params
        remaining = deadline - time.monotonic() - 50
        if remaining > 10:
            t_alloc = min(remaining, 50)
            a_opt, _ = mirror_descent_optimize(state[1], time_limit=t_alloc, T_init=0.0003, lr_init=0.02)
            try_and_update(a_opt, state, "md_fine")
    
    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))
    
    # === Phase 4: Final Polish (~350s) ===
    print("\n=== Phase 4: Final Polish ===")
    t_phase = time.monotonic()
    
    if state[1] is not None:
        n_best = len(state[1])
        
        # Fine FW
        remaining = deadline - time.monotonic() - 100
        if remaining > 10:
            t_alloc = min(remaining, 80)
            print("  Final FW, t={:.0f}s".format(t_alloc))
            a_opt, _ = fw_lp_optimize(state[1], time_limit=t_alloc, max_constraints=300)
            try_and_update(a_opt, state, "final_fw")
        
        # Fine mirror descent
        remaining = deadline - time.monotonic() - 50
        if remaining > 5:
            t_alloc = min(remaining, 40)
            a_opt, _ = mirror_descent_optimize(state[1], time_limit=t_alloc, T_init=0.0001, lr_init=0.01)
            try_and_update(a_opt, state, "final_md")
        
        # Fine mass redistribution
        remaining = deadline - time.monotonic() - 20
        if remaining > 5:
            t_alloc = min(remaining, 30)
            a_opt, _ = mass_redistribute(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "final_mass")
        
        # Tiny perturbations + quick polish
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            for i in range(min(10, int(remaining // 3))):
                remaining = deadline - time.monotonic() - 5
                if remaining < 2:
                    break
                noise = rng.standard_normal(n_best) * 0.0003
                f_pert = project_simplex(state[1] + noise)
                t_alloc = min(remaining, 3)
                a_opt, _ = mirror_descent_optimize(f_pert, time_limit=t_alloc,
                                                   T_init=0.0002, lr_init=0.005)
                try_and_update(a_opt, state, "perturb_{}".format(i))
    
    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))
    
    # Fallback
    if state[1] is None:
        state[1] = np.ones(768)
        state[0] = float(evaluate_sequence(state[1].tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs
    
    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
