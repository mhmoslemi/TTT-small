import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    n = len(a)
    m = len(b)

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_active_set(f, M, K_max=80, max_iter=15):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k=0..2n-2, g >= 0
    using constraint generation (active set method).
    """
    n = len(f)
    num_k = 2 * n - 1
    
    # Initial active set: positions where f is largest
    # (these positions create the tightest constraints)
    top_positions = np.argsort(f)[::-1][:min(K_max // 2, n)]
    active = set(top_positions.tolist())
    
    # Add edge positions
    edge_size = min(5, n)
    for k in range(edge_size):
        active.add(k)
        active.add(num_k - 1 - k)
    active = [k for k in active if 0 <= k < num_k]
    
    g = np.zeros(n)
    
    for it in range(max_iter):
        active = sorted(list(set(active)))
        if len(active) > K_max:
            # Keep the most violated ones + top f positions
            active = active[:K_max]
        
        # Build constraint submatrix
        m_active = len(active)
        A_sub = np.zeros((m_active, n))
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_sub[i, j] = f[k - j]
        
        c = -np.ones(n)
        b_ub = np.full(m_active, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_sub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True})
            if not res.success or res.x is None:
                # Fallback: try with all constraints
                A_full = np.zeros((num_k, n))
                for j in range(n):
                    A_full[j:j+n, j] = f
                res = linprog(c, A_ub=A_full, b_ub=np.full(num_k, M), bounds=bounds,
                             method='highs', options={'presolve': True})
                if not res.success or res.x is None:
                    return None
                g = np.maximum(res.x, 0.0)
                # Check violations
                fg = fast_convolve_cross(f, g)
                if (fg - M).max() <= 1e-8:
                    return g
                return None
            g = np.maximum(res.x, 0.0)
        except Exception:
            return None
        
        # Check for constraint violations
        fg = fast_convolve_cross(f, g)
        violations = fg - M
        max_viol = violations.max()
        
        if max_viol <= 1e-8:
            return g  # Feasible, hence optimal (since LP with subset constraints)
        
        # Add worst violating constraint
        worst_k = int(np.argmax(violations))
        active.append(worst_k)
        active = list(set(active))
    
    # Return best solution found (may have small violations)
    fg = fast_convolve_cross(f, g)
    if (fg - M).max() <= 1e-6:
        return g
    # Scale down to be feasible
    scale = M / fg.max() if fg.max() > M else 1.0
    if scale < 1.0:
        return g * scale
    return g


def fw_away_steps(f0, time_limit, verbose=False, K_lp=60):
    """
    Frank-Wolfe with away steps and active-set LP oracle.
    f0: initial point on simplex
    """
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    stagnant = 0
    it = 0
    warm_g = None
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        # Get LP oracle direction
        g = solve_lp_active_set(f, M, K_max=K_lp)
        
        if g is None or g.sum() < 1e-15:
            stagnant += 1
            if stagnant > 30:
                # Restart with perturbation
                rng = np.random.default_rng(it * 7919 + 3)
                scale = 0.001 + 0.003 * min(1.0, stagnant / 100.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
                warm_g = None
            continue
        
        g = g / g.sum()
        
        # Check if direction is meaningful
        if np.max(np.abs(g - f)) < 1e-10:
            stagnant += 1
            if stagnant > 40:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
                warm_g = None
            continue
        
        # Away step: find coordinate to move away from
        # The away coordinate is the one where f is largest but g is smallest
        diff = f - g
        away_idx = int(np.argmax(diff))
        away_val = f[away_idx]
        
        # Line search: try steps in the FW direction and away direction
        best_val = M + 1e-10
        best_f_new = f.copy()
        best_gam = 0.0  # FW step
        best_beta = 0.0  # Away step
        
        # Standard FW step sizes
        for gam in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.02, 0.01]:
            if gam > 1.0:
                continue
            candidate = (1.0 - gam) * f + gam * g
            candidate = np.maximum(candidate, 0.0)
            cs = candidate.sum()
            if cs < 1e-15:
                continue
            candidate = candidate / cs
            val = fast_convolve(candidate).max()
            if val < best_val:
                best_val = val
                best_f_new = candidate
                best_gam = gam
                best_beta = 0.0
        
        # Away step variants
        if away_val > 0.01 / n:  # Only if away coordinate has significant mass
            for beta in [0.1, 0.2, 0.3, 0.5]:
                for gam in [best_gam, 0.5 * best_gam, 2.0 * best_gam]:
                    if gam <= 0 or gam > 1.0:
                        continue
                    # f_new = (1-gam-beta)*f + gam*g + beta*(1/n)*ones  (away from away_idx)
                    # Actually, the away step formulation:
                    # f_new = (1 - gam - beta) * f + gam * g + beta * (f[away_idx] / (1 - f[away_idx])) * (f - f[away_idx]e_away)
                    # Simplified: move mass from away_idx toward g direction
                    candidate = f.copy()
                    # Remove beta fraction of f[away_idx]
                    remove_amt = beta * f[away_idx]
                    candidate[away_idx] -= remove_amt
                    # Distribute removed mass according to g (disproportionately)
                    g_no_away = g.copy()
                    g_no_away[away_idx] = 0
                    g_no_away_sum = g_no_away.sum()
                    if g_no_away_sum > 1e-15:
                        candidate += remove_amt * g_no_away / g_no_away_sum * gam / max(gam, 0.01)
                    
                    candidate = np.maximum(candidate, 0.0)
                    cs = candidate.sum()
                    if cs < 1e-15:
                        continue
                    candidate = candidate / cs
                    val = fast_convolve(candidate).max()
                    if val < best_val - 1e-12:
                        best_val = val
                        best_f_new = candidate
                        best_gam = gam
                        best_beta = beta
        
        # Also try a "combined" step with parabolic interpolation around best
        if best_gam > 0.01:
            for extra_step in [best_gam * 0.8, best_gam * 1.2, best_gam * 0.5, best_gam * 2.0]:
                if extra_step <= 0 or extra_step > 1.0:
                    continue
                candidate = (1.0 - extra_step) * f + extra_step * g
                candidate = np.maximum(candidate, 0.0)
                cs = candidate.sum()
                if cs < 1e-15:
                    continue
                candidate = candidate / cs
                val = fast_convolve(candidate).max()
                if val < best_val - 1e-12:
                    best_val = val
                    best_f_new = candidate
                    best_gam = extra_step
        
        if best_gam > 1e-8 or best_beta > 1e-8:
            f = best_f_new
            conv = fast_convolve(f)
            M = conv.max()
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                stagnant = 0
                warm_g = g
            else:
                stagnant += 1
        else:
            stagnant += 1
        
        if verbose and it % 50 == 0:
            print(f"    FW-AS it={it}, obj={best_obj:.10f}, stagnant={stagnant}, M={M:.10f}")
    
    return best_f, best_obj


def peak_polish(a0, time_limit):
    """
    Peak-targeted coordinate descent for final polishing.
    Directly reduces the tallest peaks by moving mass from contributing coordinates.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step_frac = 0.005
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        # Get top peaks
        n_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]
        
        improvement = False
        
        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            if js > je:
                continue
            
            # Find top contributing pairs
            contribs = []
            for j in range(js, je + 1):
                c = a[j] * a[k - j]
                contribs.append((c, j, k - j))
            contribs.sort(reverse=True)
            
            # Try to reduce top contributions
            for (_, j1, j2) in contribs[:3]:
                if a[j1] < 1e-10 and a[j2] < 1e-10:
                    continue
                
                # Find a "safe" target: position where adding mass won't create new peaks
                # Heuristic: position far from all current peaks
                peak_set = set(int(p) for p in peak_indices)
                
                # Score each position by how "safe" it is
                best_target = -1
                best_score = -np.inf
                
                # Sample some candidate targets
                rng = np.random.default_rng((it * 9973 + k * 31 + j1 * 7 + j2 * 13) % (2**31))
                candidates = rng.integers(0, n, size=min(20, n))
                
                for target in candidates:
                    target = int(target)
                    if target == j1 or target == j2:
                        continue
                    # Check if target is "far" from peaks
                    # A position j contributes to peaks at positions j+j' for all j'
                    # So we want j such that j+j' is not near any peak for any significant j'
                    score = 0
                    for pk in peak_indices[:3]:
                        pk = int(pk)
                        # If we add mass at target, it contributes to peak pk via a[target]*a[pk-target]
                        # So we want a[pk-target] to be small
                        if 0 <= pk - target < n:
                            score += a[pk - target]
                        # Also contributes to peak at position 2*target (if in range)
                        if 0 <= 2 * target < 2 * n - 1:
                            pass  # Hard to check without full conv
                    # Lower score = safer
                    if score < best_score:
                        best_score = score
                        best_target = target
                
                if best_target < 0:
                    continue
                
                src = j1 if a[j1] >= a[j2] else j2
                if a[src] < 1e-10:
                    continue
                
                for frac in [step_frac, step_frac * 0.5, step_frac * 2]:
                    delta = a[src] * frac
                    if delta < 1e-14:
                        continue
                    candidate = a.copy()
                    candidate[src] -= delta
                    candidate[best_target] += delta
                    candidate = np.maximum(candidate, 0)
                    cs = candidate.sum()
                    if cs < 1e-15:
                        continue
                    candidate = candidate / cs
                    val = fast_convolve(candidate).max()
                    if val < best_obj - 1e-14:
                        a = candidate
                        best_obj = val
                        best_a = a.copy()
                        improvement = True
                        break
                if improvement:
                    break
            if improvement:
                break
        
        if not improvement:
            step_frac *= 0.9
            if step_frac < 1e-7:
                break
            # Periodic perturbation
            if it % 100 == 0:
                rng = np.random.default_rng(it * 48271)
                candidate = project_simplex(a + rng.standard_normal(n) * step_frac * 3)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-12:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
    
    return best_a, best_obj


def subgradient_optimize(a0, time_limit, T_init=0.0005, lr_init=0.015, top_k_frac=0.05):
    """Smoothed subgradient descent with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(5, int(n * top_k_frac))
    
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.9
    
    t0 = time.monotonic()
    stagnant = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        
        # Top-k peaks
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))
        
        pv = conv[peaks]
        pv_max = pv.max()
        w = np.exp((pv - pv_max) / (T * 0.1))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / w_sum
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += w[i] * 2.0 * a[k - idx_arr]
        
        v = mom * v - lr * grad
        
        best_candidate = a
        best_val = conv.max()
        
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            candidate = project_simplex(a + frac * v)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
        
        a = best_candidate
        obj = conv.max()
        
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 15:
            T *= 1.05
            lr *= 0.96
        if stagnant > 50:
            T *= 0.5
            lr *= 1.5
            mom = 0.93
        if stagnant > 150:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise = 0.001 + 0.003 * min(1.0, (stagnant - 150) / 150.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            v = np.zeros(n)
            mom = 0.9
            stagnant = 0
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_starts(n, parent, rng):
    """Generate diverse initializations."""
    starts = []
    
    if parent is not None:
        starts.append(resample_sequence(parent, n))
    
    starts.append(np.ones(n) / n)
    
    x = np.arange(n, dtype=np.float64)
    # Cosine taper
    cos_t = np.cos(np.pi * x / (n - 1))
    w = 0.5 * (1 + cos_t)
    w = np.maximum(w, 0.01)
    starts.append(w / w.sum())
    
    # Gaussian
    g = np.exp(-0.5 * ((x - (n - 1) / 2) / (0.2 * n)) ** 2)
    starts.append(g / g.sum())
    
    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    starts.append(edge / edge.sum())
    
    # Bimodal
    g1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.08 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.08 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())
    
    # Random Dirichlet
    for _ in range(2):
        starts.append(rng.dirichlet(np.ones(n) * 0.3))
    
    # "S-shaped" (sigmoid-like)
    sig = 1.0 / (1.0 + np.exp(-5.0 * (x / n - 0.5)))
    starts.append(sig / sig.sum())
    
    return starts


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search for minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    
    Uses FW with away steps + active-set LP oracle,
    subgradient refinement, and peak-polish coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")
    
    n_parent = len(parent)
    
    # === Phase 1: Quick exploration across n values (200s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_phase = time.monotonic()
    phase1_time = 200
    
    lengths = sorted(set([n_parent, max(256, n_parent - 128), n_parent + 128, 512, 640, 768, 896, 1024, 1280]))
    lengths = [l for l in lengths if 128 <= l <= 2048]
    
    for n_try in lengths:
        if time.monotonic() - t_phase > phase1_time:
            break
        if deadline - time.monotonic() < 1200:
            break
        
        starts = generate_starts(n_try, parent, rng)
        
        for si, start in enumerate(starts[:3]):
            if time.monotonic() - t_phase > phase1_time - 10:
                break
            if deadline - time.monotonic() < 1200:
                break
            t_alloc = min(deadline - time.monotonic() - 1200, 30)
            if t_alloc < 10:
                break
            a_opt, _ = fw_away_steps(start, time_limit=t_alloc)
            try_seq(a_opt, f"fw_n{n_try}_s{si}")
    
    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 2: Deep FW on best (500s) ===
    print("\n=== Phase 2: Deep FW with Away Steps ===")
    t_phase = time.monotonic()
    phase2_time = 500
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Main deep FW
        t_alloc = min(deadline - time.monotonic() - 700, phase2_time - 50)
        if t_alloc > 30:
            print(f"  Deep FW-AS n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_away_steps(best_seq, time_limit=t_alloc, verbose=True, K_lp=60)
            try_seq(a_opt, "deep_fw_as")
        
        # Try with different K_lp values
        for K in [40, 80, 120]:
            if time.monotonic() - t_phase > phase2_time - 30:
                break
            if deadline - time.monotonic() < 700:
                break
            t_alloc = min(deadline - time.monotonic() - 700, 40)
            if t_alloc < 15:
                break
            a_opt, _ = fw_away_steps(best_seq, time_limit=t_alloc, K_lp=K)
            try_seq(a_opt, f"fw_K{K}")
        
        # Perturbed restarts
        for scale in [0.001, 0.003, 0.008]:
            if time.monotonic() - t_phase > phase2_time - 20:
                break
            if deadline - time.monotonic() < 700:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(deadline - time.monotonic() - 700, 30)
            if t_alloc < 10:
                break
            a_opt, _ = fw_away_steps(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"fw_pert_{scale}")
    
    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 3: Subgradient refinement (250s) ===
    print("\n=== Phase 3: Subgradient ===")
    t_phase = time.monotonic()
    phase3_time = 250
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        configs = [
            {'T_init': 0.0003, 'lr_init': 0.015, 'top_k_frac': 0.03},
            {'T_init': 0.001, 'lr_init': 0.01, 'top_k_frac': 0.05},
            {'T_init': 0.003, 'lr_init': 0.008, 'top_k_frac': 0.08},
            {'T_init': 0.0001, 'lr_init': 0.02, 'top_k_frac': 0.02},
        ]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Frank-Wolfe with away steps and active-set LP oracle (constraint generation)
- Parabolic line search for efficient step selection
- Smoothed subgradient with adaptive parameters
- Peak-targeted coordinate descent for final polishing
- Multi-length exploration with parent warm start
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_active_set(f, M, K_max=80, max_iter=20):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k=0..2n-2, g >= 0
    using constraint generation (active set method).
    Returns the optimal g (not normalized).
    """
    n = len(f)
    num_k = 2 * n - 1

    # Initial active set: positions where f is largest
    top_positions = np.argsort(f)[::-1][:min(K_max // 2, n)]
    active = set(top_positions.tolist())

    # Add edge positions
    edge_size = min(5, n)
    for k in range(edge_size):
        active.add(k)
        active.add(num_k - 1 - k)
    active = [k for k in active if 0 <= k < num_k]

    g = np.ones(n) / n  # Default feasible point

    for it in range(max_iter):
        active = sorted(list(set(active)))
        if len(active) > K_max:
            active = active[:K_max]

        # Build constraint submatrix
        m_active = len(active)
        A_sub = np.zeros((m_active, n))
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_sub[i, j] = f[k - j]

        c = -np.ones(n)
        b_ub = np.full(m_active, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_sub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True})
            if not res.success or res.x is None:
                # Fallback: full constraint set
                A_full = np.zeros((num_k, n))
                for j in range(n):
                    A_full[j:j + n, j] = f
                res = linprog(c, A_ub=A_full, b_ub=np.full(num_k, M), bounds=bounds,
                              method='highs', options={'presolve': True})
                if not res.success or res.x is None:
                    return None
                g = np.maximum(res.x, 0.0)
                fg = fast_convolve_cross(f, g)
                if (fg - M).max() <= 1e-8:
                    return g
                return None
            g = np.maximum(res.x, 0.0)
        except Exception:
            return None

        # Check for constraint violations using FFT
        fg = fast_convolve_cross(f, g)
        violations = fg - M
        max_viol = violations.max()

        if max_viol <= 1e-8:
            return g

        # Add worst violating constraint
        worst_k = int(np.argmax(violations))
        if worst_k not in active:
            active.append(worst_k)
            active = list(set(active))
        else:
            # Already in active set but violated - LP may have numerical issues
            # Try with slightly larger M
            return None

    # Return best solution (may have tiny violations)
    fg = fast_convolve_cross(f, g)
    if fg.max() > M * (1 + 1e-6):
        scale = M / fg.max()
        g = g * scale
    return g

def subgradient_optimize(a0, time_limit, T_init=0.0005, lr_init=0.015, top_k_frac=0.05):
    """Smoothed subgradient descent with momentum and adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(5, int(n * top_k_frac))

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.9

    t0 = time.monotonic()
    stagnant = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))

        pv = conv[peaks]
        pv_max = pv.max()
        w = np.exp((pv - pv_max) / (T * 0.1))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += w[i] * 2.0 * a[k - idx_arr]

        v = mom * v - lr * grad

        best_candidate = a
        best_val = conv.max()

        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            candidate = project_simplex(a + frac * v)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 15:
            T *= 1.05
            lr *= 0.96
        if stagnant > 50:
            T *= 0.5
            lr *= 1.5
            mom = 0.93
        if stagnant > 150:
            rng = np.random.default_rng((it * 31 + 7) % (2**31 - 1))
            noise = 0.001 + 0.003 * min(1.0, (stagnant - 150) / 150.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            v = np.zeros(n)
            mom = 0.9
            stagnant = 0

    return best_a, best_obj


def peak_polish(a0, time_limit):

    Directly reduces the tallest peaks by moving mass from contributing coordinates
    to positions that don't significantly contribute to any peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step_frac = 0.004
    it = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        n_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        improvement = False

        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)

            if js > je:
                continue

            # Find top contributing pairs
            contribs = []
            for j in range(js, je + 1):
                c = a[j] * a[k - j]
                contribs.append((c, j, k - j))
            contribs.sort(reverse=True)

            for (_, j1, j2) in contribs[:4]:
                if a[j1] < 1e-10 and a[j2] < 1e-10:
                    continue

                # Find safe target positions
                rng = np.random.default_rng((it * 9973 + k * 31 + j1 * 7 + j2 * 13) % (2**31 - 1))
                n_candidates = min(30, n)
                candidates = rng.integers(0, n, size=n_candidates)

                best_target = -1
                best_target_score = np.inf

                for target in candidates:
                    target = int(target)
                    if target == j1 or target == j2:
                        continue
                    # Score: how much would adding mass here increase peak contributions?
                    score = 0.0
                    for pk in peak_indices[:4]:
                        pk = int(pk)
                        other_idx = pk - target
                        if 0 <= other_idx < n:
                            score += a[other_idx]
                    if score < best_target_score:
                        best_target_score = score
                        best_target = target

                if best_target < 0:
                    continue

                src = j1 if a[j1] >= a[j2] else j2
                if a[src] < 1e-10:
                    continue

                for frac in [step_frac, step_frac * 0.5, step_frac * 2.0]:
                    delta = a[src] * frac
                    if delta < 1e-14:
                        continue
                    candidate = a.copy()
                    candidate[src] -= delta
                    candidate[best_target] += delta
                    candidate = np.maximum(candidate, 0)
                    cs = candidate.sum()
                    if cs < 1e-15:
                        continue
                    candidate = candidate / cs
                    val = fast_convolve(candidate).max()
                    if val < best_obj - 1e-14:
                        a = candidate
                        best_obj = val
                        best_a = a.copy()
                        improvement = True
                        break
                if improvement:
                    break
            if improvement:
                break

        if not improvement:
            no_improve_count += 1
            step_frac *= 0.92
            if step_frac < 1e-7:
                break
            # Periodic perturbation to escape local stagnation
            if no_improve_count % 50 == 0:
                rng = np.random.default_rng(it * 48271 + 17)
                candidate = project_simplex(a + rng.standard_normal(n) * step_frac * 5)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-12:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
                    no_improve_count = 0
            if no_improve_count > 300:
                break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_starts(n, parent, rng):
    """Generate diverse initializations for FW."""
    starts = []

    if parent is not None:
        starts.append(resample_sequence(parent, n))

    starts.append(np.ones(n) / n)

    x = np.arange(n, dtype=np.float64)

    # Cosine taper
    cos_t = np.cos(np.pi * x / max(1, n - 1))
    w = 0.5 * (1 + cos_t)
    w = np.maximum(w, 0.01)
    starts.append(w / w.sum())

    # Gaussian
    g = np.exp(-0.5 * ((x - (n - 1) / 2) / (0.2 * n)) ** 2)
    starts.append(g / g.sum())

    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    starts.append(edge / edge.sum())

    # Bimodal
    g1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.08 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.08 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())

    # Sigmoid
    sig = 1.0 / (1.0 + np.exp(-5.0 * (x / n - 0.5)))
    starts.append(sig / sig.sum())

    # Random Dirichlet
    for _ in range(2):
        starts.append(rng.dirichlet(np.ones(n) * 0.3))

    return starts


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search for minimizing
    2*n*max(conv(a,a))/(sum(a))^2.

    Phases:
    1. Quick multi-n, multi-start exploration with FW-AS (150s)
    2. Deep FW with away steps on best candidate (500s)
    3. Subgradient refinement (200s)
    4. Peak-targeted coordinate descent (150s)
    5. Final alternating micro-polish (remaining time)
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + max(1.0, total_budget - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")

    n_parent = len(parent)

    # === Phase 1: Quick exploration across n values (150s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_phase = time.monotonic()
    phase1_budget = 150

    lengths = sorted(set([n_parent, max(256, n_parent - 128), n_parent + 128,
                          512, 640, 768, 896, 1024, 1280]))
    lengths = [l for l in lengths if 128 <= l <= 2048]

    for n_try in lengths:
        elapsed = time.monotonic() - t_phase
        if elapsed > phase1_budget:
            break
        if deadline - time.monotonic() < 1200:
            break

        starts = generate_starts(n_try, parent, rng)

        for si, start in enumerate(starts[:3]):
            elapsed = time.monotonic() - t_phase
            if elapsed > phase1_budget - 10:
                break
            if deadline - time.monotonic() < 1200:
                break
            t_alloc = min(deadline - time.monotonic() - 1200, 35)
            if t_alloc < 12:
                break
            a_opt, _ = fw_away_steps(start, time_limit=t_alloc, K_lp=50)
            try_seq(a_opt, f"fw_n{n_try}_s{si}")

    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW with away steps (500s) ===
    print("\n=== Phase 2: Deep FW with Away Steps ===")
    t_phase = time.monotonic()
    phase2_budget = 500

    if best_seq is not None:
        n_best = len(best_seq)

        # Main deep FW
        t_alloc = min(deadline - time.monotonic() - 600, phase2_budget - 80)
        if t_alloc > 30:
            print(f"  Deep FW-AS n={n_best}, t={t_alloc:.0f}s, K_lp=60")
            a_opt, _ = fw_away_steps(best_seq, time_limit=t_alloc, verbose=True, K_lp=60)
            try_seq(a_opt, "deep_fw_as")

        # Try different K_lp values
        for K in [40, 100]:
            elapsed = time.monotonic() - t_phase
            if elapsed > phase2_budget - 60:
                break
            if deadline - time.monotonic() < 600:
                break
            t_alloc = min(deadline - time.monotonic() - 600, 45)
            if t_alloc < 15:
                break
            print(f"  FW-AS K_lp={K}, t={t_alloc:.0f}s")
            a_opt, _ = fw_away_steps(best_seq, time_limit=t_alloc, K_lp=K)
            try_seq(a_opt, f"fw_K{K}")

        # Perturbed restarts
        for scale in [0.001, 0.003, 0.008, 0.015]:
            elapsed = time.monotonic() - t_phase
            if elapsed > phase2_budget - 30:
                break
            if deadline - time.monotonic() < 600:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(deadline - time.monotonic() - 600, 30)
            if t_alloc < 10:
                break
            a_opt, _ = fw_away_steps(f_pert, time_limit=t_alloc, K_lp=50)
            try_seq(a_opt, f"fw_pert_{scale}")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Subgradient refinement (200s) ===
    print("\n=== Phase 3: Subgradient ===")
    t_phase = time.monotonic()
    phase3_budget = 200

    if best_seq is not None:
        n_best = len(best_seq)

        configs = [
            {'T_init': 0.0003, 'lr_init': 0.015, 'top_k_frac': 0.03},
            {'T_init': 0.001, 'lr_init': 0.01, 'top_k_frac': 0.05},
            {'T_init': 0.003, 'lr_init': 0.007, 'top_k_frac': 0.08},
            {'T_init': 0.0001, 'lr_init': 0.02, 'top_k_frac': 0.02},
        ]

        for ci, cfg in enumerate(configs):
            elapsed = time.monotonic() - t_phase
            if elapsed > phase3_budget:
                break
            if deadline - time.monotonic() < 400:
                break
            t_alloc = min(deadline - time.monotonic() - 400, 50)
            if t_alloc < 12:
                break
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"subgrad_{ci}")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Peak polish (150s) ===
    print("\n=== Phase 4: Peak Polish ===")
    t_phase = time.monotonic()
    phase4_budget = 150

    if best_seq is not None:
        t_alloc = min(deadline - time.monotonic() - 150, phase4_budget)
        if t_alloc > 20:
            a_opt, _ = peak_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_polish")

        # Second round
        if best_seq is not None:
            elapsed = time.monotonic() - t_phase
            if elapsed < phase4_budget - 20:
                t_alloc = min(deadline - time.monotonic() - 150, 50)
                if t_alloc > 10:
                    a_opt, _ = peak_polish(best_seq, time_limit=t_alloc)
                    try_seq(a_opt, "peak_polish_2")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final alternating polish (remaining) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        rounds = 0

        while deadline - time.monotonic() > 25:
            rounds += 1
            t_alloc = min(deadline - time.monotonic() - 20, 35)
            if t_alloc < 12:
                break

            if rounds % 4 == 1:
                # Short FW with small perturbation
                noise = rng.standard_normal(n_best) * 0.0004
                start = project_simplex(best_seq + noise)
                a_opt, _ = fw_away_steps(start, time_limit=t_alloc, K_lp=50)
                try_seq(a_opt, f"final_fw_{rounds}")
            elif rounds % 4 == 2:
                # Subgradient
                noise = rng.standard_normal(n_best) * 0.0003
                start = project_simplex(best_seq + noise)
                a_opt, _ = subgradient_optimize(start, time_limit=t_alloc,
                                                T_init=0.0002, lr_init=0.005,
                                                top_k_frac=0.02)
                try_seq(a_opt, f"final_sg_{rounds}")
            elif rounds % 4 == 3:
                # Peak polish
                a_opt, _ = peak_polish(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"final_peak_{rounds}")
            else:
                # Longer FW from slightly perturbed best
                noise = rng.standard_normal(n_best) * 0.0008
                start = project_simplex(best_seq + noise)
                t_alloc2 = min(deadline - time.monotonic() - 20, 45)
                if t_alloc2 < 12:
                    break
                a_opt, _ = fw_away_steps(start, time_limit=t_alloc2, K_lp=60)
                try_seq(a_opt, f"final_fw_long_{rounds}")

            if rounds % 8 == 0:
                print(f"  Round {rounds}: best={best_score:.10f}, time_left={deadline-time.monotonic():.0f}s")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification and cleanup
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = best_seq / best_seq.sum()
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    print(f"Total time used: {time.monotonic() - (deadline - float(budget_s) + 30):.0f}s")
    return [float(x) for x in best_seq]
