"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-length diverse initialization
2. Improved Frank-Wolfe with expanded constraint sets and adaptive M
3. LSE subgradient descent with backtracking

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, constraint_indices):
    """Build sparse constraint matrix for LP."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([idx] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    if not rows:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(len(constraint_indices), n))


def solve_lp_direction(f, M, constraint_indices, time_limit=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for given constraints, g >= 0."""
    n = len(f)
    A = build_lp_matrix(f, constraint_indices)
    if A.shape[0] == 0:
        return np.ones(n) / n
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_optimize(f0, time_limit, top_k=40, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)

        f = f / s

    best_obj = M

                    best_f = f.copy()

        g = np.maximum(g, 0.0)

        if gs < 1e-15:

def subgradient_optimize(f0, time_limit, beta_start=20.0, beta_end=8000.0, top_k=50):
    """Subgradient method with LSE smoothing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    beta = beta_start
    lr = 3.0 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        conv = fast_convolve(f)

        # LSE weights
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        total = exp_bc.sum()
        if total < 1e-300:
            w = np.ones(len(conv)) / len(conv)
        else:
            w = exp_bc / total

        k_top = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        w_peaks = w[peak_idx]
        w_sum = w_peaks.sum()
        if w_sum > 1e-15:
            w_peaks = w_peaks / w_sum
        else:
            w_peaks = np.ones(k_top) / k_top

        # Compute gradient
        grad = np.zeros(n)
        for idx, k in enumerate(peak_idx):
            k = int(k)
            wt = w_peaks[idx]
            if wt < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            i_range = np.arange(lo, hi + 1)
            grad[i_range] += wt * 2.0 * f[k - i_range]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_normed = grad / gnorm

        # Backtracking
        step = lr
        improved = False
        for _ in range(8):
            f_new = project_simplex(f - step * grad_normed)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                improved = True
                no_improve = 0
                lr = min(lr * 1.01, 20.0 / np.sqrt(n))
                break
            step *= 0.5

        if not improved:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.85
            if no_improve > 50:
                f = best_f.copy()
                lr = max(3.0 / np.sqrt(n), 0.01)
                no_improve = 0
                rng = np.random.default_rng(it * 31 + 17)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)

        if beta < beta_end:
            beta *= 1.003

    return best_f, best_obj


def coordinate_descent(a, time_limit, verbose=False):
    """Fast coordinate descent: optimize one element at a time."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            t_old = a[i]
            if t_old < 1e-15:
                # Zero element: try small positive values
                candidates = [1e-6, 1e-5, 1e-4, a.max() * 0.01, a.max() * 0.05]
            else:
                # Generate candidates around current value
                candidates = [
                    0.0,
                    t_old * 0.3, t_old * 0.5, t_old * 0.7,
                    t_old,
                    t_old * 1.3, t_old * 1.5, t_old * 2.0,
                    t_old * 0.8, t_old * 1.2
                ]
                # Add value that would "equalize" with current peak
                # Approximate: t such that t^2 ~ peak (for k=2i case)
                if 2 * i < 2 * n - 1:
                    candidates.append(np.sqrt(conv[2 * i]) * 1.1)
                    candidates.append(np.sqrt(conv[2 * i]) * 0.9)

            best_t = t_old
            best_local_obj = best_obj

            S_rest = a.sum() - t_old  # sum of all other elements

            for t in candidates:
                if t < 0 or t > 1000:
                    continue
                new_S = S_rest + t
                if new_S < 1e-10:
                    continue

                # Compute new peak efficiently
                k_start = max(0, i)
                k_end = min(2 * n - 2, i + n - 1)

                # Changed entries: conv[k] + 2*(t-t_old)*a[k-i] for k in [k_start,k_end], k != 2i
                if k_end >= k_start:
                    ks = np.arange(k_start, k_end + 1)
                    ki_indices = ks - i  # indices into a
                    linear_part = conv[ks] + 2.0 * (t - t_old) * a[ki_indices]
                    # Mask out k=2i
                    if 2 * i >= k_start and 2 * i <= k_end:
                        mask = ks != 2 * i
                        peak_linear = linear_part[mask].max() if mask.any() else 0.0
                    else:
                        peak_linear = linear_part.max()
                else:
                    peak_linear = 0.0

                # Quadratic part (k = 2i)
                if 2 * i < 2 * n - 1:
                    peak_quad = conv[2 * i] - t_old * t_old + t * t
                else:
                    peak_quad = 0.0

                # Unchanged part
                peak_unchanged = 0.0
                if k_start > 0:
                    peak_unchanged = max(peak_unchanged, conv[:k_start].max())
                if k_end < 2 * n - 2:
                    peak_unchanged = max(peak_unchanged, conv[k_end + 1:].max())

                new_peak = max(peak_linear, peak_quad, peak_unchanged)
                new_obj = 2.0 * n * new_peak / (new_S * new_S)

                if new_obj < best_local_obj - 1e-12:
                    best_local_obj = new_obj
                    best_t = t

            if best_t != t_old:
                a[i] = best_t
                # Update conv incrementally
                delta_t = best_t - t_old
                k_start = max(0, i)
                k_end = min(2 * n - 2, i + n - 1)
                ks = np.arange(k_start, k_end + 1)
                ki_indices = ks - i
                if 2 * i >= k_start and 2 * i <= k_end:
                    mask = ks != 2 * i
                    conv[ks[mask]] += 2.0 * delta_t * a[ki_indices[mask]]
                    conv[2 * i] += (best_t * best_t - t_old * t_old)
                else:
                    conv[ks] += 2.0 * delta_t * a[ki_indices]
                conv = np.maximum(conv, 0)  # numerical safety

                current_obj = 2.0 * n * conv.max() / (a.sum() ** 2)
                if current_obj < best_obj - 1e-12:
                    best_obj = current_obj
                    best_a = a.copy()
                improved = True
                no_improve = 0

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break

    return best_a, best_obj


def peak_shave(a, time_limit, n_peaks=10, verbose=False):
    """Targeted peak reduction by adjusting responsible elements."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        current_peak = conv.max()
        peak_positions = np.argsort(conv)[-n_peaks:]

        made_progress = False
        for k in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo >= hi:
                continue

            # Find the dominant pair
            vals = a[lo:hi + 1] * a[k - np.arange(lo, hi + 1)]
            if vals.max() < 1e-15:
                continue
            j_max = lo + np.argmax(vals)
            j_other = k - j_max
            if j_other < 0 or j_other >= n or j_other == j_max:
                continue

            # Try to reduce the dominant pair
            reduction_frac = 0.05 + 0.05 * rng.random()
            delta = a[j_max] * reduction_frac
            if delta < 1e-12:
                continue

            # Find good redistribution targets (positions far from peaks)
            # Score each position by how much it would increase peaks
            n_targets = min(20, n)
            targets = rng.choice(n, size=n_targets, replace=False)

            best_target = None
            best_score = best_obj

            for jt in targets:
                if jt == j_max or jt == j_other:
                    continue
                a_trial = a.copy()
                a_trial[j_max] -= delta
                a_trial[jt] += delta
                if a_trial[j_max] < 0:
                    # Split reduction
                    delta = a[j_max] * 0.5
                    a_trial = a.copy()
                    a_trial[j_max] -= delta
                    a_trial[jt] += delta
                    if a_trial[j_max] < -1e-10:
                        continue
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum()
                conv_trial = fast_convolve(a_trial)
                if conv_trial.max() < best_score - 1e-12:
                    best_score = conv_trial.max()
                    best_target = (jt, a_trial.copy())

            if best_target is not None:
                a = best_target[1]
                if best_score < best_obj:
                    best_obj = best_score
                    best_a = a.copy()
                made_progress = True

        if not made_progress:
            # Try smaller reductions
            made_progress = False
            for k in peak_positions[:3]:
                if time.monotonic() - t0 > time_limit:
                    break
                k = int(k)
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                if lo >= hi:
                    continue
                vals = a[lo:hi + 1] * a[k - np.arange(lo, hi + 1)]
                j_max = lo + np.argmax(vals)
                j_other = k - j_max
                if j_other < 0 or j_other >= n or j_other == j_max:
                    continue
                delta = a[j_max] * 0.02
                if delta < 1e-12:
                    continue
                # Move to average position
                jt = rng.integers(0, n)
                while jt == j_max or jt == j_other:
                    jt = rng.integers(0, n)
                a_trial = a.copy()
                a_trial[j_max] -= delta
                a_trial[jt] += delta
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum()
                conv_trial = fast_convolve(a_trial)
                if conv_trial.max() < best_obj - 1e-12:
                    a = a_trial
                    best_obj = conv_trial.max()
                    best_a = a.copy()
                    made_progress = True

        if not made_progress:
            break

    return best_a, best_obj


def freq_perturb(f, rng, strength=0.03):
    """Perturb f in the frequency domain."""
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def evaluate_score(a):
    """Quick evaluation without calling the external function."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Diverse initialization (parent + perturbations + different lengths)
    2. FW + Subgradient optimization
    3. Coordinate descent polish
    4. Peak shaving
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s) - 15.0

    def try_seq(a):
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_tracker[0]:
            best_tracker[0] = score
            best_tracker[1] = a.copy()
            print(f"  NEW BEST: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_score(parent):.8f}")
            try_seq(parent)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768.0
        try_seq(parent)

    n_parent = len(parent)

    # === Phase 1: Multi-length FW exploration ===
    print("\n=== Phase 1: Multi-length FW exploration ===")
    t1_end = time.monotonic() + total_budget * 0.20
    n_candidates = sorted(set([
        n_parent,
        max(256, n_parent - 64),
        min(2048, n_parent + 64),
        max(256, int(n_parent * 0.75)),
        min(2048, int(n_parent * 1.25)),
        1024, 896, 640
    ]), key=lambda x: -abs(x - n_parent))
    seen = set()
    n_candidates = [x for x in n_candidates if not (x in seen or seen.add(x))]
    n_candidates = n_candidates[:6]

    for nc in n_candidates:
        t_rem = t1_end - time.monotonic()
        if t_rem < 20:
            break
        t_alloc = min(80, t_rem * 0.4)
        if t_alloc < 10:
            break

        if nc == n_parent:
            f_init = parent.copy()
        else:
            f_init = interpolate_to_length(parent, nc)

        print(f"  FW n={nc}, t={t_alloc:.0f}s")
        f_opt, _ = fw_optimize(f_init, time_limit=t_alloc, top_k=40)
        try_seq(f_opt)

    # === Phase 2: Subgradient refinement on best so far ===
    print(f"\n=== Phase 2: Subgradient refinement ===")
    t2_end = t1_end + total_budget * 0.20
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)

    t_alloc = min(150, t2_end - time.monotonic())
    if t_alloc > 20:
        print(f"  Subgrad n={n_cur}, t={t_alloc:.0f}s")
        f_opt, _ = subgradient_optimize(current_best, time_limit=t_alloc,
                                         beta_start=20.0, beta_end=8000.0, top_k=50)
        try_seq(f_opt)

    # === Phase 3: Alternating FW + Subgrad + perturbations ===
    print(f"\n=== Phase 3: Alternating optimization ===")
    t3_end = deadline - total_budget * 0.35
    cycles = 0
    while time.monotonic() < t3_end and cycles < 20:
        cycles += 1
        current_best = best_tracker[1] if best_tracker[1] is not None else parent
        n_cur = len(current_best)
        remaining = t3_end - time.monotonic()
        if remaining < 30:
            break

        if cycles % 5 == 1:
            t_alloc = min(70, remaining * 0.3)
            if t_alloc > 10:
                print(f"  Cycle {cycles}: FW n={n_cur}, t={t_alloc:.0f}s")
                f_opt, _ = fw_optimize(current_best, time_limit=t_alloc, top_k=45)
                try_seq(f_opt)

        elif cycles % 5 == 2:
            t_alloc = min(80, remaining * 0.3)
            if t_alloc > 10:
                beta_s = 30.0 + 50.0 * cycles
                print(f"  Cycle {cycles}: Subgrad n={n_cur}, t={t_alloc:.0f}s, beta={beta_s:.0f}")
                f_opt, _ = subgradient_optimize(current_best, time_limit=t_alloc,
                                                 beta_start=beta_s, beta_end=12000.0, top_k=60)
                try_seq(f_opt)

        elif cycles % 5 == 3:
            t_alloc = min(50, remaining * 0.25)
            if t_alloc > 8:
                print(f"  Cycle {cycles}: FreqPerturb+FW n={n_cur}, t={t_alloc:.0f}s")
                for attempt in range(3):
                    t_rem_inner = t3_end - time.monotonic() - 10
                    if t_rem_inner < 10:
                        break
                    strength = 0.02 + 0.01 * attempt
                    f_pert = freq_perturb(current_best, rng, strength=strength)
                    t_sub = (t_rem_inner - 5) / 3.0
                    if t_sub > 5:
                        f_opt, _ = fw_optimize(f_pert, time_limit=t_sub, top_k=30)
                        try_seq(f_opt)

        elif cycles % 5 == 4:
            # Try a different length
            t_alloc = min(40, remaining * 0.2)
            if t_alloc > 8:
                alt_n = max(256, int(n_cur * (1.0 + 0.1 * ((cycles // 5) % 3 - 1))))
                print(f"  Cycle {cycles}: Alt length n={alt_n}, t={t_alloc:.0f}s")
                f_interp = interpolate_to_length(current_best, alt_n)
                f_opt, _ = fw_optimize(f_interp, time_limit=t_alloc, top_k=35)
                try_seq(f_opt)

        else:
            # Peak shaving
            t_alloc = min(30, remaining * 0.15)
            if t_alloc > 5:
                print(f"  Cycle {cycles}: PeakShave n={n_cur}, t={t_alloc:.0f}s")
                f_opt, _ = peak_shave(current_best, time_limit=t_alloc, n_peaks=8)
                try_seq(f_opt)

    # === Phase 4: Coordinate descent polish (key new component) ===
    print(f"\n=== Phase 4: Coordinate descent polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)
    t4_end = min(deadline - 30, time.monotonic() + total_budget * 0.30)

    # Multiple passes of coordinate descent with increasing time
    cd_time_total = t4_end - time.monotonic()
    if cd_time_total > 10:
        # Split into chunks
        n_chunks = min(5, max(2, int(cd_time_total / 60)))
        chunk_time = cd_time_total / n_chunks
        for ci in range(n_chunks):
            t_rem = t4_end - time.monotonic()
            if t_rem < 5:
                break
            t_alloc = min(chunk_time, t_rem)
            print(f"  CD chunk {ci+1}/{n_chunks} n={n_cur}, t={t_alloc:.0f}s")
            f_opt, _ = coordinate_descent(current_best, time_limit=t_alloc)
            score = try_seq(f_opt)
            current_best = best_tracker[1] if best_tracker[1] is not None else current_best
            n_cur = len(current_best)

    # === Phase 5: Final peak shaving + short FW ===
    print(f"\n=== Phase 5: Final refinement ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_rem = deadline - time.monotonic()

    if t_rem > 20:
        t_alloc = min(40, t_rem * 0.4)
        if t_alloc > 8:
            print(f"  Final peak shave, t={t_alloc:.0f}s")
            f_opt, _ = peak_shave(current_best, time_limit=t_alloc, n_peaks=12)
            try_seq(f_opt)

    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_rem = deadline - time.monotonic()
    if t_rem > 15:
        t_alloc = min(30, t_rem * 0.5)
        if t_alloc > 5:
            print(f"  Final FW, t={t_alloc:.0f}s")
            f_opt, _ = fw_optimize(current_best, time_limit=t_alloc, top_k=50)
            try_seq(f_opt)

    # One more coordinate descent pass
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_rem = deadline - time.monotonic()
    if t_rem > 10:
        t_alloc = min(30, t_rem * 0.6)
        if t_alloc > 5:
            print(f"  Final CD, t={t_alloc:.0f}s")
            f_opt, _ = coordinate_descent(current_best, time_limit=t_alloc)
            try_seq(f_opt)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using fallback")
    fallback = np.ones(768) / 768.0
    return [float(x) for x in fallback]
