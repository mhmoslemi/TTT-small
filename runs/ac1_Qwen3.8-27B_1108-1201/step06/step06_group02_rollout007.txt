import numpy as np
from scipy.optimize import linprog, minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

        # We want sum_j a[j] * conv[i+j] = xcorr(conv, a)[i] (with signs adjusted)
        # corr = IFFT(FFT(conv) * conj(FFT(a)))
        # corr[i] = sum_j conv[j] * a[j-i] ... not quite

        # Let me just do it directly for clarity (n up to ~2048, this is fine):
        # grad[i] = 2 * sum_{j=0}^{n-1} conv[i+j] * a[j] (where i+j < 2n-1)
        # For efficiency, note this is a sliding dot product

        # Use the fact that this equals the (i-th element of) the convolution of
        # reversed a with conv, shifted appropriately
        a_rev = a[::-1]
        # convolve(a_rev, conv, mode='full')[i + n - 1] = sum_j a_rev[j] * conv[i+n-1-j]
        # = sum_j a[n-1-j] * conv[i+n-1-j]
        # Let k = n-1-j: = sum_k a[k] * conv[i+k]
        # Yes! That's what we want.
        temp = np.fft.rfft(a_rev, fft_len)
        corr = np.fft.irfft(temp * C_fft, fft_len)[:3*n-2]
        grad = 2.0 * corr[n-1:2*n-1]

        # Nesterov momentum step
        v = mom * v - lr * grad
        a_new = project_simplex(a + v)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            a = a  # Don't update if no improvement in max

        # Adaptive learning rate
        if no_improve > 15:
            lr *= 0.95
        if no_improve > 40:
            mom = min(0.95, mom + 0.005)
        if no_improve > 100:
            rng = np.random.default_rng(it * 12345 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = 0.02
            mom = 0.9
            no_improve = 0

        if no_improve > 200 and time.monotonic() - t0 > time_limit * 0.8:
            break

    return best_a, best_obj


def peak_equalize(f0, time_limit):
    """
    Iteratively reduce the highest peaks of the convolution by
    moving mass from peak contributors to low-convolution positions.
    Uses a lookahead to avoid creating new peaks.
    """
    a = np.maximum(f0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stale = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Find contributing pairs
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        if len(affected) == 0:
            break

        contributions = a[affected] * a[k_peak - affected]

        # Find the top contributor
        j_max = affected[np.argmax(contributions)]

        # Find a good target: a position that minimizes the maximum increase in conv
        # Strategy: look at positions where a is small AND that are far from the peak
        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            break

        # Score each outside position by the potential increase in conv
        # Moving mass to position p increases conv[k] by approximately 2*a[p]*a[k-p]
        # We want to minimize the max increase
        # Approximate: pick the position with smallest a value that's far from peak
        sorted_outside = outside[np.argsort(a[outside])]
        target_candidates = sorted_outside[:min(10, len(sorted_outside))]

        improved = False
        for j_min_cand in target_candidates:
            for delta in [0.0003, 0.0005, 0.001, 0.002, 0.005]:
                if a[j_max] < delta:
                    break
                candidate = a.copy()
                candidate[j_max] -= delta
                candidate[j_min_cand] += delta
                candidate = np.maximum(candidate, 0.0)
                ss = candidate.sum()
                if ss < 1e-15:
                    continue
                candidate = candidate / ss
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if improved:
            stale = 0
        else:
            stale += 1
            if stale % 20 == 0:
                # Try a random perturbation
                rng = np.random.default_rng(it * 4567 + 3)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    stale = 0
                elif stale > 300:
                    break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations including U-shaped patterns."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)

    # Uniform
    inits.append(np.ones(n) / n)

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # U-shaped (inverse Gaussian): high at edges, low in middle
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        u = 1.0 + alpha * ((x - mid) / (0.5 * n)) ** 2
        u = np.maximum(u, 0.01)
        inits.append(u / u.sum())

    # Cosine U-shape
    for beta in [0.3, 0.5, 0.7]:
        c = 1.0 - beta * np.cos(np.pi * x / (n - 1))
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())

    # Edge-weighted
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    inits.append(edge_w / edge_w.sum())

    # Gaussian (center-heavy)
    g = np.exp(-0.5 * ((x - mid) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Periodic
    for freq in [2, 3, 5]:
        h = 1.0 + 0.3 * np.cos(2.0 * np.pi * freq * x / n)
        h = np.maximum(h, 0.01)
        inits.append(h / h.sum())

    # Interpolated parent
    if parent is not None and len(parent) > 0:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Noisy parent
    if parent is not None and len(parent) > 0:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px += rng.standard_normal(n) * 0.01
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def try_and_update(a, state, label):
    """Evaluate a sequence and update state if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a)))


def nesterov_smooth_grad(a0, time_limit, T_init=0.001):
    """Nesterov-accelerated smoothed gradient descent on max-conv."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = 0.03
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    # Nesterov: maintain a "lookahead" point
    y = a.copy()
    beta = 0.9

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Evaluate at y (lookahead)
        conv = fast_convolve(y)
        M = conv.max()

        # Softmax weights
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient at y
        # grad[i] = 2 * sum_k p[k] * y[k-i] (with bounds)
        y_rev = y[::-1]
        fft_len = 1 << (4 * n - 1).bit_length()
        Y_fft = np.fft.rfft(y_rev, fft_len)
        P_fft = np.fft.rfft(conv * p, fft_len)  # Hmm, this isn't quite right
        # Actually: grad[i] = 2 * sum_k p[k] * y[k-i]
        # = 2 * (y_reversed * (p * conv))... no
        # sum_k p[k] * y[k-i] = cross-correlation of p*conv... 
        # Let me just do: grad[i] = 2 * sum_k p[k] * y[k-i] where 0<=k-i<n
        # = 2 * sum_j p[i+j] * y[j] (j=0..n-1, i+j < 2n-1)
        # = 2 * correlate(p_conv, y, mode='full')[n-1:i+n]

        # Use FFT:
        pc = p * conv  # weighted convolution values
        # We want h[i] = sum_j pc[i+j] * y[j] for i=0..n-1
        # = (y_rev * pc)[n-1+i] (linear convolution)
        temp = np.fft.rfft(y_rev, fft_len)
        PC_fft = np.fft.rfft(pc, fft_len)
        corr = np.fft.irfft(temp * PC_fft, fft_len)[:3*n-2]
        grad = 2.0 * corr[n-1:2*n-1]

        # Nesterov step
        a_new = project_simplex(y - lr * grad)
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = a_new + (beta_new - 1.0) / (beta + 1.0) * (a_new - a)
        a = a_new

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            T *= 1.03
            lr *= 0.97
        if no_improve > 50:
            rng = np.random.default_rng(it * 999 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            y = a.copy()
            T = T_init
            lr = 0.03
            no_improve = 0

        if no_improve > 200 and time.monotonic() - t0 > time_limit * 0.85:
            break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid search:
    Phase 1: Multi-n exploration with diverse initializations
    Phase 2: Deep FW-LP optimization
    Phase 3: Spectral flattening (novel: minimize L2-conv via gradient)
    Phase 4: Peak equalization + Nesterov smooth gradient
    Phase 5: Final polish with tiny perturbations
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    state = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.8f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384

    n_parent = len(parent)

    # === Phase 1: Multi-n Exploration with diverse inits (~150s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent, 512, 640, 768, 896, 1024, 1280]))
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 2048]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1400
        if remaining < 10:
            break

        # Try a few initializations for this n
        inits = generate_initializations(n_try, parent, rng)

        for idx, init in enumerate(inits[:4]):  # Limit to 4 inits per n
            remaining = deadline - time.monotonic() - 1350
            if remaining < 8:
                break

            # Quick FW-LP
            t_alloc = min(remaining, 15)
            a_opt, _ = fw_lp_optimize(init, time_limit=t_alloc)
            try_and_update(a_opt, state, "expl_n{}_i{}".format(n_try, idx))

    print("  Phase 1 done (elapsed {:.0f}s), best={:.8f}".format(
        time.monotonic() - t_phase, state[0]))

    # === Phase 2: Deep FW-LP (~650s) ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])
        remaining = deadline - time.monotonic() - 900
        if remaining > 30:
            t_alloc = min(remaining, 500)
            print("  Deep FW-LP n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = fw_lp_optimize(state[1], time_limit=t_alloc, verbose=True)
            try_and_update(a_opt, state, "deep_fw")

            # Spectral flatten between FW runs
            remaining = deadline - time.monotonic() - 800
            if remaining > 20:
                t_alloc = min(remaining, 80)
                print("  Spectral flatten t={:.0f}s".format(t_alloc))
                a_opt2, _ = spectral_flatten(a_opt, time_limit=t_alloc)
                try_and_update(a_opt2, state, "spectral_mid")

            # Another FW pass after spectral
            remaining = deadline - time.monotonic() - 700
            if remaining > 20:
                t_alloc = min(remaining, 100)
                a_opt3, _ = fw_lp_optimize(state[1], time_limit=t_alloc)
                try_and_update(a_opt3, state, "fw_after_spectral")

        # Perturbed restarts
        for scale in [0.002, 0.005, 0.01, 0.02]:
            remaining = deadline - time.monotonic() - 500
            if remaining < 10:
                break
            f_pert = project_simplex(state[1] + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 50)
            a_opt2, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
            try_and_update(a_opt2, state, "fw_pert_{}".format(scale))

    print("  Phase 2 done (elapsed {:.0f}s), best={:.8f}".format(
        time.monotonic() - t_phase, state[0]))

    # === Phase 3: Spectral Flattening (~350s) ===
    print("\n=== Phase 3: Spectral Flattening ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        remaining = deadline - time.monotonic() - 450
        if remaining > 20:
            t_alloc = min(remaining, 250)
            print("  Spectral flatten deep, t={:.0f}s".format(t_alloc))
            a_opt, _ = spectral_flatten(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "spectral_deep")

            # Follow up with FW
            remaining = deadline - time.monotonic() - 300
            if remaining > 15:
                t_alloc = min(remaining, 80)
                a_opt2, _ = fw_lp_optimize(a_opt, time_limit=t_alloc)
                try_and_update(a_opt2, state, "fw_after_spectral2")

    print("  Phase 3 done (elapsed {:.0f}s), best={:.8f}".format(
        time.monotonic() - t_phase, state[0]))

    # === Phase 4: Peak Equalization + Nesterov (~300s) ===
    print("\n=== Phase 4: Peak Equalization ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Peak equalization
        remaining = deadline - time.monotonic() - 250
        if remaining > 10:
            t_alloc = min(remaining, 120)
            print("  Peak equalization, t={:.0f}s".format(t_alloc))
            a_opt, _ = peak_equalize(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "peak_eq")

        # Nesterov smooth gradient
        remaining = deadline - time.monotonic() - 150
        if remaining > 10:
            t_alloc = min(remaining, 100)
            print("  Nesterov smooth grad, t={:.0f}s".format(t_alloc))
            a_opt2, _ = nesterov_smooth_grad(state[1], time_limit=t_alloc)
            try_and_update(a_opt2, state, "nesterov")

    print("  Phase 4 done (elapsed {:.0f}s), best={:.8f}".format(
        time.monotonic() - t_phase, state[0]))

    # === Phase 5: Final Polish (~150s) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])

        # Quick FW-LP final pass
        remaining = deadline - time.monotonic() - 100
        if remaining > 10:
            t_alloc = min(remaining, 50)
            a_opt, _ = fw_lp_optimize(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "final_fw")

        # Nesterov with fine T
        remaining = deadline - time.monotonic() - 60
        if remaining > 5:
            t_alloc = min(remaining, 40)
            a_opt2, _ = nesterov_smooth_grad(state[1], time_limit=t_alloc, T_init=0.0003)
            try_and_update(a_opt2, state, "nesterov_fine")

        # Tiny perturbations with quick polish
        for i in range(8):
            remaining = deadline - time.monotonic() - 10
            if remaining < 3:
                break
            noise = rng.standard_normal(n_best) * (0.0003 + 0.0002 * i)
            f_pert = project_simplex(state[1] + noise)
            t_alloc = min(remaining, 8)
            a_opt3, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
            try_and_update(a_opt3, state, "perturb_{}".format(i))

    print("  Phase 5 done (elapsed {:.0f}s), best={:.8f}".format(
        time.monotonic() - t_phase, state[0]))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(384) / 384
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
