Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a Hessian-informed adaptive
gradient method with peak equalization and LP oracle escapes.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()
    
    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i]
    # This is 2 * (p * a_reversed) but we need to be careful with indices
    # c[k] = sum_j a[j]*a[k-j], so dc[k]/da[i] = 2*a[k-i] (if in range)
    # grad[i] = sum_k p[k] * 2*a[k-i]
    
    # Equivalent to: grad = 2 * correlate(p, a) but with specific indexing
    # Let's compute it directly using the convolution structure
    # grad[i] = 2 * sum_k p[k] * a[k-i] for valid k-i
    # = 2 * sum_m p[i+m] * a[m] for valid i+m
    # This is like a cross-correlation
    
    # Actually, let's use the Hankel matrix structure:
    # grad[i] = 2 * sum_k p[k] * a[k-i]
    # Let m = k-i, so k = i+m:
    # grad[i] = 2 * sum_m p[i+m] * a[m]  (for m = 0..n-1, i+m = 0..2n-2)
    
    # This is the (i-th) element of the product of a Hankel matrix H (with H[i,m] = p[i+m])
    # with the vector a.
    
    # We can compute this as: for each m, add a[m] * p[m:] to grad[:2n-1-m]
    # But that's O(n^2). Let's use FFT.
    
    # Alternative: pad p to length 2n-1, compute convolution-like operation
    # grad[i] = 2 * sum_m a[m] * p[i+m]
    # This is the cross-correlation of a and p (reversed p):
    # If we define q[m] = p[2n-2-m] (reversed p), then:
    # sum_m a[m] * p[i+m] = sum_m a[m] * q[(2n-2)-(i+m)] 
    # Hmm, this is getting complicated. Let me just use the direct approach with FFT.
    
    # Actually, the simplest way: 
    # grad = 2 * H @ a where H[i,j] = p[i+j]
    # H is a Hankel matrix. We can apply H to a vector using:
    # (H @ v)[i] = sum_j p[i+j] * v[j]
    # = sum_j p[i+j] * v[j]
    
    # Let's compute using the "first row/column trick" for Hankel matrices:
    # H has first row [p[0], p[1], ..., p[n-1]]
    # and first column [p[0], p[1], ..., p[2n-2]]
    
    # (H @ v)[i] = sum_{j=0}^{n-1} p[i+j] * v[j]
    
    # We can compute this as a sum of shifted dot products.
    # For efficiency, note that if we define:
    # r[i] = sum_j p[i+j] * v[j] for i = 0, ..., 2n-2
    # This is a cross-correlation! r = correlate(p, v) (with appropriate definition)
    
    #Using numpy: correlate(p, v, mode='valid') won't work directly because of the asymmetry.
    # Let's use the definition: r[i] = sum_j p[i+j] * v[j]
    # Pad v to length 2n-1 with zeros: v_padded = [v[0], ..., v[n-1], 0, ..., 0]
    # Then r[i] = sum_j p[i+j] * v_padded[i+j] * delta(0, j-i)... no that's not right.
    
    # Let me just do it with a loop over the top p values (sparse approximation):
    
    # Find top contributors to p
    top_idx = np.argsort(p)[::-1][:top_k]
    
    grad = np.zeros(n)
    for idx in top_idx:
        if p[idx] < 1e-15:
            continue
        k = int(idx)
        # grad[i] += 2 * p[k] * a[k-i] for i where 0 <= k-i < n
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # For i in [js, je]: grad[i] += 2*p[k]*a[k-i]
        indices = np.arange(js, je + 1)
        grad[indices] += 2.0 * p[k] * a[k - indices]
    
    return grad


def solve_lp_oracle(f, M, time_limit_per_call=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[|k-j|]... 
    # Actually (f*g)[k] = sum_j f[j]*g[k-j] for valid indices
    # So (f*g)[k] = sum_j g[j]*f[k-j] = sum_j g[j]*f[|k-j|]... no
    # (f*g)[k] = sum_j f[j] * g[k-j]
    # For fixed k: (f*g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[j] * g[k-j]
    # Let i = k-j, so j = k-i:
    # (f*g)[k] = sum_i f[k-i] * g[i] for i in [max(0,k-n+1), min(n-1,k)]
    # So the constraint is: sum_i g[i] * f[k-i] <= M
    
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            A_ub[k, i] = f[k - i]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit_per_call})
        if res.success:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        g = solve_lp_oracle(f, M * 1.0, time_limit_per_call=min(15, remaining - 2))
        if g is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 30:
            break

    return best_f, best_obj


def adaptive_gradient_descent(a0, time_limit, T_init=0.002, lr_init=0.05, top_k=30):
    """
    Adaptive gradient descent with RMSProp and temperature annealing.
    Uses sparse gradient computation (top-k peaks only).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    # RMSProp state
    W = np.zeros(n)  # running average of squared gradients
    beta1 = 0.9

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        grad = compute_gradient(conv, a, T=T, top_k=top_k)
        
        if np.linalg.norm(grad) < 1e-15:
            break

        # RMSProp update
        W = beta1 * W + (1 - beta1) * (grad ** 2)
        W_hat = W / (1 - beta1 ** (it + 1))  # bias correction
        eps = 1e-8
        update = grad / (np.sqrt(W_hat) + eps)
        
        # Normalize update
        upd_norm = np.linalg.norm(update)
        if upd_norm > 0:
            update = update / upd_norm

        # Backtracking line search
        alpha = lr
        best_candidate = a
        best_val = best_obj
        for _ in range(8):
            candidate = project_simplex(a - alpha * update)
            c_new = fast_convolve(candidate)
            val = c_new.max()
            if val < best_val - 1e-15:
                best_candidate = candidate
                best_val = val
                break
            alpha *= 0.5

        if best_val < best_obj - 1e-14:
            a = best_candidate
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.5)
        else:
            a = best_candidate  # Still move to the best candidate found
            no_improve += 1
            lr *= 0.95

        # Temperature annealing
        if no_improve > 50:
            T *= 1.1
        if no_improve > 200:
            T = max(T_init * 0.5, T * 0.8)
            top_k = min(top_k + 10, n)

        # Restart if stuck
        if no_improve > 500:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            W = np.zeros(n)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def peak_equalization(a0, time_limit):
    """
    Targeted peak equalization: find the top peaks and try to equalize them
    by moving mass between coordinates.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        # Get top 8 peaks
        top_k = min(8, len(conv))
        peak_lags = np.argpartition(conv, -top_k)[-top_k:]
        peak_lags = peak_lags[np.argsort(conv[peak_lags])[::-1]]
        peak_vals = conv[peak_lags]
        
        # If peaks are already well-equalized, try reducing the max
        peak_range = peak_vals[0] - peak_vals[-1]
        
        improved = False
        
        # Strategy 1: Reduce the highest peak by moving mass away
        k_max = int(peak_lags[0])
        # Coordinates contributing to this peak
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        affected = np.arange(js, je + 1)
        
        if len(affected) >= 2:
            # Find coordinate with max contribution
            contributions = a[k_max - affected]
            max_contrib_idx = np.argmax(contributions)
            src = affected[max_contrib_idx]
            
            # Find destination: coordinate NOT contributing to this peak,
            # or contributing less to other peaks
            peak_set = set(peak_lags.tolist())
            
            best_move = None
            best_gain = 1e-15
            
            # Try moving mass from src to various destinations
            for dst in range(n):
                if dst == src:
                    continue
                # Check if dst contributes to the max peak
                if 0 <= k_max - dst < n and a[k_max - dst] > 0.5 * a[k_max - src]:
                    continue  # Would still contribute significantly
                
                # Try moving a small fraction
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = a[src] * frac
                    if delta < 1e-10:
                        continue
                    a_test = a.copy()
                    a_test[src] -= delta
                    a_test[dst] += delta
                    a_test = np.maximum(a_test, 0.0)
                    s = a_test.sum()
                    a_test = a_test / s
                    c_new = fast_convolve(a_test)
                    if c_new.max() < best_obj - 1e-14:
                        gain = best_obj - c_new.max()
                        if gain > best_gain:
                            best_gain = gain
                            best_move = (src, dst, delta)
            
            if best_move is not None:
                src, dst, delta = best_move
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                conv = fast_convolve(a)
                best_obj = conv.max()
                best_a = a.copy()
                improved = True
        
        if not improved:
            # Try random perturbation
            if it % 50 == 0:
                rng = np.random.default_rng((it * 7919 + 13) % (2**31))
                noise_scale = 0.001 + 0.002 * min(1.0, it / 500.0)
                a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
                conv = fast_convolve(a)
                if conv.max() < best_obj:
                    best_obj = conv.max()
                    best_a = a.copy()

        if it % 100 == 0 and not improved:
            # Check convergence
            pass

    return best_a, best_obj


def nesterov_subgradient(a0, time_limit, lr_init=0.005, max_k=30):
    """Nesterov-accelerated subgradient with top-k peak tracking."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    y = a.copy()
    v = np.zeros(n)
    t_nest = 1.0
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        alpha_nest = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / alpha_nest
        y = (1.0 + beta) * a - beta * y
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y / ys
        t_nest += 1.0

        conv = fast_convolve(y)
        
        if max_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -max_k)[-max_k:]

        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * y[k - idx_arr]

        v = beta * v - lr * sg
        a_new = project_simplex(a + v)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            a_new = a

        a = a_new
        lr *= (1.0 - 0.0003)

        if no_improve > 100:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.001 * np.sqrt(no_improve / 100.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            y = a.copy()
            v = np.zeros(n)
            t_nest = 1.0
            no_improve = 0

    return best_a, best_obj


def hankel_preconditioned_step(a, conv, T, top_k, lr):
    """
    One step of preconditioned gradient descent using Hankel Hessian structure.
    The preconditioner is based on the diagonal of the (approximate) Hessian.
    """
    n = len(a)
    
    # Compute weights
    peak_lags = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peak_lags]
    scaled = pv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    w = w / w.sum()
    
    # Gradient using only top-k peaks
    grad = np.zeros(n)
    for i in range(len(peak_lags)):
        k = int(peak_lags[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx_arr = np.arange(js, je + 1)
        grad[idx_arr] += 2.0 * w[i] * a[k - idx_arr]
    
    # Preconditioner: use "importance" of each coordinate
    # A coordinate a[i] is "important" if it appears in many active peaks' computation
    # importance[i] = sum_k w[k] * 1[0 <= k-i < n]
    # This is (H @ 1)[i] where H[i,m] = w[m] * 1[i+m in range]
    # Actually H[i,j] = sum_k w[k] * delta(i+j, k) for the relevant w
    # Diagonal: H[i,i] = w[2i] (if 2i is in peak_lags)
    # 
    # Better: use the "row activity" = sum_j H[i,j] = sum_k w[k] * 1[k >= i] (roughly)
    # Let's just use a simpler heuristic: normalize gradient by the "spread" of contributions
    
    # Simple adaptive learning rate per coordinate
    grad_mag = np.abs(grad)
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return a, lr
    
    # Normalize and take step
    direction = -grad / grad_norm
    candidate = project_simplex(a + lr * direction)
    
    return candidate, lr


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Hybrid search combining:
    1. FW-LP for global moves
    2. Adaptive RMSProp gradient descent with temperature annealing
    3. Nesterov subgradient for accelerated convergence
    4. Peak equalization for fine-tuning
    5. Multi-n exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t0_total = time.monotonic()

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t0_total
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512
        try_seq(parent, "uniform")

    n_parent = len(parent)

    # === Phase 1: Multi-n exploration ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent, max(256, n_parent - 128), min(1500, n_parent + 128),
                                384, 512, 640, 768]))
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 1200]

    for n_try in candidate_ns:
        rem = deadline - time.monotonic()
        if rem < 20:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss > 1e-10:
                start = start / ss
            else:
                start = np.ones(n_try) / n_try

        t_alloc = min(20, rem - 1500)
        if t_alloc < 5:
            break
        print(f"  FW-LP n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(start, time_limit=t_alloc)
        try_seq(a_opt, f"fw_n{n_try}")

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW-LP ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t_alloc = min(200, rem - 1200)
        if t_alloc > 20:
            print(f"  Deep FW n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_fw")

        # Perturbed restarts
        for scale in [0.003, 0.008]:
            rem = deadline - time.monotonic()
            t_alloc = min(40, rem - 1100)
            if t_alloc < 10:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"fw_pert{scale}")

    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Adaptive Gradient (RMSProp) ===
    print("\n=== Phase 3: Adaptive Gradient ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        configs = [
            {'T_init': 0.001, 'lr_init': 0.08, 'top_k': 20},

        ]
        for ci, cfg in enumerate(configs):
            rem = deadline - time.monotonic()
            t_alloc = min(250, rem - 500)
            if t_alloc < 15:
                break
            print(f"  RMSProp T={cfg['T_init']}, lr={cfg['lr_init']}, k={cfg['top_k']}, t={t_alloc:.0f}s")
            a_opt, _ = adaptive_gradient_descent(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"rms_{ci}")
            # If improved, use as new base
            if best_score < 1.504:
                pass  # Good, keep going

    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Nesterov Subgradient ===
    print("\n=== Phase 4: Nesterov Subgradient ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        configs = [
            {'lr_init': 0.004, 'max_k': 25},
            {'lr_init': 0.007, 'max_k': 40},
        ]
        for ci, cfg in enumerate(configs):
            rem = deadline - time.monotonic()
            t_alloc = min(150, rem - 300)
            if t_alloc < 15:
