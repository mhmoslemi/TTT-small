def project_simplex(v):
    n=len(v)
    if n==1: return np.array([max(v[0],0.0)])
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    rho=np.arange(1,n+1)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond): return np.ones(n)/n
    r=int(np.nonzero(cond)[0][-1])
    theta=(cssv[r]-1.0)/(r+1.0)
    return np.maximum(v-theta,0.0)
```

def hankel_mv(p,a):

- Faster Hankel-vector products via FFT.
- LP-based dual weights that try to make the KKT multiplier flat on the support.
- Nonlinear primal-dual / KKT active-set iterations with dense solves on a
  controlled active set.
- Adam and FISTA-style smoothed-gradient polishers.
- Exact pair-move line search for final peak shaving.
"""

import time
import math
import numpy as np
from scipy import optimize

# Globals used by the L-BFGS wrapper so that all functions remain top-level.
_LBD_T = 1e-4
_LBD_DEADLINE = 0.0
_LBD_LAST_B = None


def normalize_nonneg(a):
    """Clip, flatten, normalize to sum 1. Return None if invalid."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = a.ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return None
    return a / s


def fast_convolve(a):
    """Full linear auto-convolution."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    if n <= 128:
        out = np.convolve(a, a, mode="full")
    else:
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        out = np.fft.irfft(A * A, fft_size, fft_size)[:size]
    return np.maximum(out, 0.0)


def direct_score(a):
    """
    Fast score for normalized non-negative a:
        2*n*max(conv(a,a))
    (scale invariant, so normalization is safe).
    """
    b = normalize_nonneg(a)
    if b is None:
        return float("inf")
    n = len(b)
    C = fast_convolve(b)
    return float(2.0 * n * np.max(C))


def try_candidate(a, best_seq, best_score):
    """Update best if a has a better direct score."""
    b = normalize_nonneg(a)
    if b is None:
        return best_seq, best_score
    sc = direct_score(b)
    if np.isfinite(sc) and sc < best_score:
        return b.copy(), float(sc)
    return best_seq, best_score


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64).ravel()
    v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    r = int(np.nonzero(cond)[0][-1])
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """
    Compute g_i = sum_j a_j p_{i+j}.
    p should have length 2n-1; a length n.
    Uses FFT for larger n.
    """
    n = len(a)
    if n == 0:
        return np.array([])
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size < 2 * n - 1:
        pp = np.zeros(2 * n - 1, dtype=np.float64)
        pp[:p.size] = p
        p = pp
    elif p.size > 2 * n - 1:
        p = p[:2 * n - 1]

    if n <= 250:
        return np.correlate(p, a, mode="valid")

    size_needed = 3 * n - 2
    if size_needed <= 1:
        return np.correlate(p, a, mode="valid")
    size = 1 << (size_needed - 1).bit_length()

    # out_i = (flip(a) * p)_{i+n-1}
    a_rev = a[::-1].astype(np.float64, copy=False)
    A = np.fft.rfft(a_rev, size)
    P = np.fft.rfft(p, size)
    out = np.fft.irfft(A * P, size)[: size]
    return out[n - 1 : 2 * n - 1]


def softmax_weights(C, T):
    """Normalized exp(C / T)."""
    C = np.asarray(C, dtype=np.float64)
    if C.size == 0:
        return np.array([])
    if T <= 1e-12:
        T = 1e-12
    s = C / T
    s -= np.max(s)
    e = np.exp(s)
    ssum = float(e.sum())
    if ssum <= 0.0:
        return np.ones(C.size, dtype=np.float64) / C.size
    return e / ssum


def power_weights(C, q):
    """Normalized C^q with non-negative floor by zeros."""
    C = np.asarray(C, dtype=np.float64)
    if C.size == 0:
        return np.array([])
    C = np.maximum(C, 0.0)
    w = np.power(C, float(q))
    s = float(w.sum())
    if s <= 0.0:
        return np.ones(C.size, dtype=np.float64) / C.size
    return w / s


def mix_weights(C, top_m=120, q=3.0, T=1e-4, top_w=0.4):
    """
    Heuristically useful dual distribution:
      - uniform over a few peaks,
      - power of peaks,
      - small softmax for smoothness.
    """
    C = np.asarray(C, dtype=np.float64)
    nC = C.size
    if nC == 0:
        return np.array([])

    pw = power_weights(C, q)
    w = pw.copy()

    if top_m > 0:
        m = min(int(top_m), nC)
        tw = np.zeros(nC, dtype=np.float64)
        if m >= nC:
            idx = np.arange(nC)
        else:
            idx = np.argpartition(C, -m)[-m:]
        if m > 0:
            tw[idx] = 1.0 / m
        if top_w >= 0.999:
            w = tw
        else:
            w = top_w * tw + (1.0 - top_w) * w

    sp = softmax_weights(C, T)
    w = 0.7 * w + 0.3 * sp
    ss = float(w.sum())
    if ss <= 0.0:
        return power_weights(C, 2.0)
    return w / ss


def kkt_solve(p, a, active_limit=350, max_iters=8, time_limit=1.0):
    """
    Given dual weights p, solve the KKT linear system on a controlled active set:
        H(p) a = 1 on active coordinates,
    then normalize a to the simplex.
    """
    a = normalize_nonneg(a)
    if a is None:
        return None
    n = len(a)
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size < 2 * n - 1:
        pp = np.zeros(2 * n - 1, dtype=np.float64)
        pp[:p.size] = p
        p = pp
    elif p.size > 2 * n - 1:
        p = p[:2 * n - 1]

    best_a = a.copy()
    if active_limit is None or active_limit >= n:
        active = set(range(n))
    else:
        active_limit = max(1, int(active_limit))
        active = set(np.argsort(a)[-active_limit:].tolist())

    t0 = time.monotonic()
    for _ in range(max(1, int(max_iters))):
        if time.monotonic() - t0 > max(0.0, float(time_limit)):
            break
        if len(active) == 0:
            return np.ones(n, dtype=np.float64) / n

        act = np.array(sorted(active), dtype=int)
        k = len(act)
        if k == 1:
            out = np.zeros(n, dtype=np.float64)
            out[act[0]] = 1.0
            return out

        W = np.empty((k, k), dtype=np.float64)
        for ii, i in enumerate(act):
            W[ii, :] = p[i + act]
        W = 0.5 * (W + W.T)
        W += 1e-12 * np.eye(k)

        rhs = np.ones(k, dtype=np.float64)
        try:
            x = np.linalg.solve(W, rhs)
        except Exception:
            try:
                x = np.linalg.lstsq(W, rhs, rcond=None)[0]
            except Exception:
                x = np.ones(k, dtype=np.float64)

        if not np.all(np.isfinite(x)):
            continue

        neg = np.where(x < -1e-10)[0]
        if neg.size:
            rem = act[int(neg[np.argmin(x)])]
            active.discard(int(rem))
            pos = np.maximum(x, 0.0)
            tmp = np.zeros(n, dtype=np.float64)
            tmp[act] = pos
            ss = float(tmp.sum())
            if ss > 1e-12:
                best_a = tmp / ss
            continue

        full = np.zeros(n, dtype=np.float64)
        full[act] = np.maximum(x, 0.0)
        ss = float(full.sum())
        if ss <= 1e-12:
            continue
        full /= ss
        best_a = full.copy()

        if k >= n:
            return full

        g = hankel_mv(p, full)
        inactive = np.setdiff1d(np.arange(n), act, assume_unique=False)
        if inactive.size:
            c = float(np.mean(g[act]))
            viol = c - g[inactive]
            maxv = float(np.max(viol))
            if maxv > 1e-7 and len(active) < int(active_limit):
                order = np.argsort(viol)[::-1]
                add_n = min(5, len(order))
                for add_idx in order[:add_n]:
                    active.add(int(inactive[int(add_idx)]))
                continue
        return full

    return best_a


def pd_refine(a0, time_limit, T_init=1e-4, q=3.0, top_m=120, top_w=0.4,
              blend=0.5, active_limit=350, max_kkt_iters=3, inner_time=1.0,
              seed=10):
    """
    Primal-dual refinement:
      - dual weights from current peaks,
      - active-set KKT solve,
      - blend back to objective.
    """
    a = normalize_nonneg(a0)
    if a is None or time_limit <= 0:
        return a, direct_score(a)

    n = len(a)
    C = fast_convolve(a)
    best_M = float(np.max(C))
    best_a = a.copy()
    T_start = max(1e-7, float(T_init))
    T = T_start
    blend0 = float(blend)
    top0 = max(1, int(top_m))
    kmax0 = max(50, int(active_limit))
    no_improve = 0
    it = 0
    end = time.monotonic() + max(0.0, float(time_limit))
    rng = np.random.default_rng(int(seed) if seed is not None else 10)

    while time.monotonic() < end:
        it += 1
        rem = end - time.monotonic()
        if rem < 0.05:
            break

        C = fast_convolve(a)
        M = float(np.max(C))
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()

        w = mix_weights(C, top_m=top0, q=q, T=T, top_w=top_w)
        inner = min(float(inner_time), 0.35 * rem)
        a_k = kkt_solve(
            w, a,
            active_limit=kmax0,
            max_iters=max_kkt_iters,
            time_limit=inner
        )
        if a_k is None:
            a_k = best_a

        cand = project_simplex((1.0 - blend0) * a + blend0 * a_k)
        cand_M = float(np.max(fast_convolve(cand)))
        if cand_M < best_M - 1e-15:
            best_M = cand_M
            best_a = cand.copy()
        a = cand

        if cand_M < M - 1e-14:
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 20:
            T = min(T_start * 5.0, T * 1.2)
            top0 = min(int(top0 * 1.2) + 1, 2 * n)
            blend0 = max(0.2, blend0 * 0.92)
            q = min(8.0, q * 1.05)

        if no_improve > 120:
            T = T_start
            blend0 = blend0 * 0.75

        if no_improve > 350:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            no_improve = 0
            T = T_start

        if no_improve > 3000:
            break

    return best_a, float(2.0 * n * best_M)


def lp_dual_weights_from_conv(C, a, K_max=150, min_K=20):
    """
    LP-based dual weights selected from near-max convolution lags.
    Minimizes deviation of H(p)a from a constant.
    """
    C = np.asarray(C, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0 or C.size == 0:
        return None

    M = float(np.max(C))
    if M <= 0.0:
        return power_weights(C, 2.0)

    thresh = M - max(1e-12, 1e-5 * M)
    idx = np.where(C >= thresh)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(C))], dtype=int)

    K_max = max(5, int(K_max))
    min_K = max(1, int(min_K))

    if idx.size > K_max:
        idx = np.argpartition(C, -K_max)[-K_max:]
    else:
        take = min(K_max, max(min_K, idx.size))
        idx = np.argsort(C)[-take:]

    m = len(idx)
    if m == 0:
        return power_weights(C, 2.0)

    A = np.zeros((n, m), dtype=np.float64)
    for ki, kk in enumerate(idx):
        kk = int(kk)
        imax = max(0, kk - n + 1)
        iex = min(n - 1, kk) + 1
        if iex <= imax:
            continue
        start = kk - imax
        j_last = kk - (iex - 1)
        if j_last <= 0:
            A[imax:iex, ki] = a[start::-1]
        else:
            A[imax:iex, ki] = a[start:j_last - 1:-1]

    if not np.any(A):
        return power_weights(C, 2.0)

    c_lp = np.zeros(m + 2, dtype=np.float64)
    c_lp[-1] = 1.0

    A_ub = np.zeros((2 * n, m + 2), dtype=np.float64)
    b_ub = np.zeros(2 * n, dtype=np.float64)

    A_ub[:n, :m] = A
    A_ub[:n, m] = -1.0
    A_ub[:n, m + 1] = -1.0

    A_ub[n:, :m] = -A
    A_ub[n:, m] = 1.0
    A_ub[n:, m + 1] = -1.0

    A_eq = np.zeros((1, m + 2), dtype=np.float64)
    A_eq[0, :m] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    bounds = [(0.0, None)] * m + [(None, None), (0.0, None)]

    try:
        res = optimize.linprog(
            c_lp,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs"
        )
    except Exception:
        return power_weights(C, 2.0)

    if res is None or int(res.status) != 0:
        return power_weights(C, 2.0)

    p_full = np.zeros(2 * n - 1, dtype=np.float64)
    p_full[idx] = np.maximum(res.x[:m], 0.0)
    s = float(p_full.sum())
    if s <= 0.0:
        return power_weights(C, 2.0)
    return p_full / s


def lp_kkt_refine(a0, time_limit, K_max=150, lp_every=12,
                  active_limit=350, blend=0.5, max_kkt_iters=4, seed=11):
    """
    KKT refinement using LP dual weights periodically.
    """
    a = normalize_nonneg(a0)
    if a is None or time_limit <= 0:
        return a, direct_score(a)

    n = len(a)
    C = fast_convolve(a)
    best_M = float(np.max(C))
    best_a = a.copy()
    p = None
    it = 0
    no_improve = 0
    K_cur = int(K_max)
    lp_every_cur = max(2, int(lp_every))
    blend_cur = float(blend)
    end = time.monotonic() + max(0.0, float(time_limit))
    rng = np.random.default_rng(int(seed) if seed is not None else 11)

    while time.monotonic() < end:
        it += 1
        rem = end - time.monotonic()
        if rem < 0.05:
            break

        C = fast_convolve(a)
        M = float(np.max(C))
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()

        if p is None or it % lp_every_cur == 0:
            p_lp = lp_dual_weights_from_conv(C, a, K_max=K_cur, min_K=16)
            if p_lp is not None:
                p = p_lp.copy()
            else:
                p = mix_weights(C, top_m=min(200, 2 * n), q=3.0,
                                T=max(1e-6, 0.05 * M), top_w=0.5)
        else:
            pw = power_weights(C, 2.5)
            p = 0.5 * p + 0.5 * pw
            ss = float(p.sum())
            if ss > 0:
                p /= ss

        inner = min(1.2, 0.35 * rem)
        a_k = kkt_solve(
            p, a,
            active_limit=active_limit,
            max_iters=max_kkt_iters,
            time_limit=inner
        )
        if a_k is None:
            a_k = best_a

        cand = project_simplex((1.0 - blend_cur) * a + blend_cur * a_k)
        cand_M = float(np.max(fast_convolve(cand)))
        if cand_M < best_M - 1e-15:
            best_M = cand_M
            best_a = cand.copy()
        a = cand

        if cand_M < M - 1e-14:
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            K_cur = min(300, int(K_cur * 1.25))
            lp_every_cur = max(3, int(lp_every_cur * 0.85))
            blend_cur = max(0.3, blend_cur * 0.92)

        if no_improve > 150:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            no_improve = 0

        if no_improve > 2000:
            break

    return best_a, float(2.0 * n * best_M)


def adam_refine(a0, time_limit, T_init=2e-4, lr_init=0.02,
                T_decay=0.999, T_min=1e-7, seed=12):
    """
    Adam polish on the simplex with a log-sum-exp smoothed peak objective.
    """
    a = normalize_nonneg(a0)
    if a is None or time_limit <= 0:
        return a, direct_score(a)

    n = len(a)
    C = fast_convolve(a)
    best_M = float(np.max(C))
    best_a = a.copy()

    T_start = max(1e-8, float(T_init))
    T = T_start
    T_min = max(1e-8, float(T_min), 0.001 * T_start)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    mom_b1 = 0.9
    mom_b2 = 0.999
    eps = 1e-8
    lr = float(lr_init)
    it = 0
    no_improve = 0
    end = time.monotonic() + max(0.0, float(time_limit))
    rng = np.random.default_rng(int(seed) if seed is not None else 12)

    while time.monotonic() < end:
        it += 1
        rem = end - time.monotonic()
        if rem < 0.02:
            break

        C = fast_convolve(a)
        M = float(np.max(C))
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()

        p = softmax_weights(C, T)
        g = 2.0 * hankel_mv(p, a)
        g = g - float(np.mean(g))

        m = mom_b1 * m + (1.0 - mom_b1) * g
        v = mom_b2 * v + (1.0 - mom_b1) * (g * g)
        corr1 = 1.0 - (mom_b1 ** it)
        corr2 = 1.0 - (mom_b2 ** it)
        step = lr * (m / corr1) / (np.sqrt(v / corr2) + eps)
        step = np.clip(step, -0.008, 0.008)

        scale = 1.0
        if no_improve > 20:
            scale = 0.5
        if no_improve > 80:
            scale = 0.25

        cand = project_simplex(a - scale * step)
        cand_M = float(np.max(fast_convolve(cand)))
        a = cand
        if cand_M < best_M - 1e-15:
            best_M = cand_M
            best_a = a.copy()

        if cand_M < M - 1e-14:
            no_improve = 0
        else:
            no_improve += 1

        T = max(T_min, T_start * (float(T_decay) ** it))

        if no_improve > 40:
            lr *= 0.9
        if no_improve > 80:
            T = min(T_start * 5.0, T * 1.25)

        if no_improve > 300:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            lr = float(lr_init) * 0.5
            T = T_start * 0.5
            no_improve = 0

        if no_improve > 3000:
            break

    return best_a, float(2.0 * n * best_M)


def nesterov_smooth(a0, time_limit, T_init=1e-4, lr_init=0.01,
                    T_decay=0.9995, T_min=1e-7, seed=13):
    """
    FISTA-style accelerated smoothed gradient polish.
    """
    a = normalize_nonneg(a0)
    if a is None or time_limit <= 0:
        return a, direct_score(a)

    n = len(a)
    C = fast_convolve(a)
    best_M = float(np.max(C))
    best_a = a.copy()

    T_start = max(1e-8, float(T_init))
    T = T_start
    T_min = max(1e-8, float(T_min), 0.002 * T_start)
    lr = float(lr_init)
    x = a.copy()
    y = a.copy()
    t = 1.0
    it = 0
    no_improve = 0
    end = time.monotonic() + max(0.0, float(time_limit))
    rng = np.random.default_rng(int(seed) if seed is not None else 13)

    while time.monotonic() < end:
        it += 1
        rem = end - time.monotonic()
        if rem < 0.02:
            break

        y = project_simplex(y)
        C = fast_convolve(y)
        p = softmax_weights(C, T)
        g = 2.0 * hankel_mv(p, y)
        g = g - float(np.mean(g))

        x_new = project_simplex(y - lr * g)
        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) * 0.5
        y_new = x_new + (t / t_new - 1.0) * (x_new - x)
        x = x_new
        y = project_simplex(y_new)
        t = t_new

        M = float(np.max(fast_convolve(x)))
        if M < best_M - 1e-15:
            best_M = M
            best_a = x.copy()

        if M < C.max() - 1e-14:
            no_improve = 0
        else:
            no_improve += 1

        T = max(T_min, T_start * (float(T_decay) ** it))

        if no_improve > 50:
            lr *= 0.92
        if no_improve > 100:
            T = min(T_start * 5.0, T * 1.2)
        if no_improve > 350:
            x = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            y = x.copy()
            lr = float(lr_init) * 0.5
            T = T_start * 0.5
            t = 1.0
            no_improve = 0

        if no_improve > 5000:
            break

    return best_a, float(2.0 * n * best_M)


def suggested_T(a, factor=0.1):
    """Suggest a smoothing temperature from current peak height."""
    b = normalize_nonneg(a)
    if b is None:
        return 1e-4
    C = fast_convolve(b)
    M = float(np.max(C)) if C.size else 0.0
    return float(np.clip(factor * M, 1e-7, 1e-3))


def logit_make_b(a):
    """Logits for first n-1 coordinates relative to last fixed logit 0."""
    a = normalize_nonneg(a)
    if a is None:
        return np.array([])
    eps = 1e-12
    l = np.log(np.maximum(a, eps))
    l = l - l[-1]
    b = l[:-1]
    return np.clip(b, -30.0, 30.0)


def logit_b_to_a(b):
    """Recover normalized a from logits (last logit fixed to 0)."""
    b = np.asarray(b, dtype=np.float64).ravel()
    n = len(b) + 1
    full = np.empty(n, dtype=np.float64)
    if len(b):
        full[:-1] = b
    full[-1] = 0.0
    m = float(np.max(full))
    e = np.exp(full - m)
    s = float(e.sum())
    if s <= 0.0:
        return np.ones(n, dtype=np.float64) / n
    return e / s


def logit_val_grad(b, T):
    """
    Value and gradient of log-sum-exp_T(conv(softmax(b,0), softmax(b,0))).
    """
    b = np.asarray(b, dtype=np.float64).ravel()
    n = len(b) + 1
    full = np.empty(n, dtype=np.float64)
    if len(b):
        full[:-1] = b
    full[-1] = 0.0

    m = float(np.max(full))
    e = np.exp(full - m)
    s = float(e.sum())
    if s <= 0.0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = e / s

    C = fast_convolve(a)
    T = max(1e-12, float(T))
    maxC = float(np.max(C))
    scaled = (C - maxC) / T
    ee = np.exp(scaled)
    ssum = float(ee.sum())
    if ssum <= 0.0:
        p = np.ones(C.size, dtype=np.float64) / C.size
        val = maxC
    else:
        p = ee / ssum
        val = maxC + T * math.log(ssum)

    g = 2.0 * hankel_mv(p, a)
    dot = float(np.dot(a, g))
    dfull = a * (g - dot)
    if len(b) == 0:
        return val, np.array([])
    return val, dfull[:-1]


def lbfgs_fun(b):
    """Time-guarded objective for scipy L-BFGS."""
    global _LBD_LAST_B
    if time.monotonic() > _LBD_DEADLINE:
        raise TimeoutError
    val, grad = logit_val_grad(b, _LBD_T)
    _LBD_LAST_B = b.copy()
    return val, grad


def run_lbfgs_logit(a, T, time_limit, deadline, maxiter=150):
    """Run a short L-BFGS-B step in log-parameterization."""
    global _LBD_T, _LBD_DEADLINE, _LBD_LAST_B

    a = normalize_nonneg(a)
    if a is None or len(a) <= 1 or time_limit <= 0:
        return a

    b = logit_make_b(a)
    if len(b) == 0:
        return a

    _LBD_T = max(1e-9, float(T))
    _LBD_DEADLINE = min(float(deadline), time.monotonic() + max(0.0, float(time_limit)))
    _LBD_LAST_B = b.copy()

    bounds = [(-40.0, 40.0)] * len(b)
    options = {
        "maxiter": int(maxiter),
        "ftol": 1e-15,
        "gtol": 1e-8,
        "maxls": 20,
        "disp": False
    }

    try:
        res = optimize.minimize(
            lbfgs_fun,
            b,
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options=options
        )
    except Exception:
        res = None

    if _LBD_LAST_B is None:
        _LBD_LAST_B = b
    return logit_b_to_a(_LBD_LAST_B)


def lbfgs_continuation(a0, time_limit, deadline):
    """
    L-BFGS continuation with decreasing smoothing temperature.
    """
    a = normalize_nonneg(a0)
    if a is None or time_limit <= 0:
        return a, direct_score(a)

    C = fast_convolve(a)
    M = float(np.max(C))
    T0 = max(1e-7, 0.1 * M if M > 0 else 1e-4)

    best_a = a.copy()
    best_score = direct_score(a)

    end = min(float(deadline), time.monotonic() + max(0.0, float(time_limit)))
    stages = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02]
    ratios = [0.15, 0.15, 0.20, 0.20, 0.15, 0.15]

    for i, frac in enumerate(stages):
        if time.monotonic() >= end:
            break
        t_alloc = min(ratios[i] * float(time_limit), end - time.monotonic())
        if t_alloc < 5.0:
            break
        T_stage = T0 * frac
        maxiter = 160 if len(a) <= 1000 else 70
        a_new = run_lbfgs_logit(a, T_stage, t_alloc, end, maxiter=maxiter)
        a_new = normalize_nonneg(a_new)
        if a_new is not None:
            sc = direct_score(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        a = best_a

    return best_a, float(best_score)


def interp_sequence(a, n):
    """Interpolate a normalized sequence to a new length."""
    a = normalize_nonneg(a)
    if a is None:
        return np.ones(int(n), dtype=np.float64) / n
    if len(a) == int(n):
        return a.copy()
    x_new = np.linspace(0.0, 1.0, int(n), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, len(a), dtype=np.float64)
    b = np.interp(x_new, x_old, a)
    return normalize_nonneg(b)


def quick_explore(parent, best_seq, best_score, deadline, time_limit):
    """
    Quickly explore nearby lengths / initializations.
    """
    parent = normalize_nonneg(parent)
    if parent is None or time_limit <= 0:
        return best_seq, best_score

    ns = {len(parent)}
    for x in (len(parent) + 256, len(parent) + 512,
              max(128, len(parent) // 2),
              2 * len(parent)):
        if 128 <= x <= 2048:
            ns.add(int(x))
    ns = sorted(ns)

    end = min(float(deadline), time.monotonic() + max(0.0, float(time_limit)))
    per_n = max(15.0, min(30.0, (end - time.monotonic()) / max(1, len(ns))))

    for n_try in ns:
        if time.monotonic() > end - 15.0:
            break
        a_init = interp_sequence(parent, int(n_try))
        t_alloc = min(per_n, end - time.monotonic() - 10.0)
        if t_alloc < 8.0:
            break

        T0 = suggested_T(a_init, 0.1)
        a_opt, _ = adam_refine(a_init, t_alloc, T_init=T0, lr_init=0.02, seed=100)
        best_seq, best_score = try_candidate(a_opt, best_seq, best_score)

        t_alloc2 = min(15.0, end - time.monotonic() - 10.0)
        if t_alloc2 > 5.0:
            a_opt2, _ = pd_refine(
                a_opt,
                t_alloc2,
                T_init=suggested_T(a_opt, 0.08),
                q=3.0,
                top_m=80,
                top_w=0.5,
                active_limit=min(300, int(n_try)),
                max_kkt_iters=2,
                inner_time=0.5,
                seed=101
            )
            best_seq, best_score = try_candidate(a_opt2, best_seq, best_score)

    return best_seq, best_score


def pair_descent(a0, time_limit, top_peaks=8, top_cand=10,
                 max_pairs=120, fracs=(0.03, 0.08, 0.15, 0.25, 0.40, 0.65)):
    """
    Exploits exact pairwise mass moves.
    Linearly predicts the new convolution and verifies exact improvement.
    """
    a = normalize_nonneg(a0)
    if a is None or time_limit <= 0:
        return a, direct_score(a)

    n = len(a)
    C = fast_convolve(a)
    best_M = float(np.max(C))
    best_a = a.copy()
    end = time.monotonic() + max(0.0, float(time_limit))
    rng = np.random.default_rng(17)

    while time.monotonic() < end:
        rem = end - time.monotonic()
        if rem < 0.05:
            break

        C = fast_convolve(a)
        M = float(np.max(C))
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()

        m = min(max(1, int(top_peaks)), C.size)
        peak_idx = np.argpartition(C, -m)[-m:]
        p = np.zeros(C.size, dtype=np.float64)
        p[peak_idx] = 1.0 / m
        g = 2.0 * hankel_mv(p, a)

        nc = min(max(1, int(top_cand)), n)
        if nc >= n:
            donor_scores = a * np.maximum(g, 0.0)
            donors = np.argsort(donor_scores)[::-1]
            recipients = np.argsort(g)
        else:
            donor_scores = a * np.maximum(g, 0.0)
            donors = np.argpartition(donor_scores, -nc)[-nc:]
            recipients = np.argpartition(g, nc)[:nc]

        pairs = []
        for d in donors:
            d = int(d)
            for r in recipients:
                r = int(r)
                if d != r:
                    pairs.append((d, r))

        if not pairs:
            break

        if len(pairs) > max_pairs:
            arr = np.array(pairs, dtype=int)
            perm = rng.permutation(arr.shape[0])[:max_pairs]
            pairs = [tuple(x) for x in arr[perm]]

        improved = False
        for d, r in pairs:
            max_shift = float(a[d])
            if max_shift <= 1e-13:
                continue

            dvec = np.zeros(C.size, dtype=np.float64)
            dvec[d : d + n] -= 2.0 * a
            dvec[r : r + n] += 2.0 * a

            for frac in fracs:
                frac = float(frac)
                if frac <= 0.0:
                    continue
                shift = frac * max_shift
                if shift <= 0.0:
                    continue
                # Linear prediction only; exact convolve for verification.
                pred_M = float(np.max(C + shift * dvec))
                if pred_M >= best_M:
                    continue
                cand = a.copy()
                cand[d] -= shift
                cand[r] += shift
                cand = normalize_nonneg(cand)
                if cand is None:
                    continue
                cand_M = float(np.max(fast_convolve(cand)))
                if cand_M < best_M - 1e-12:
                    best_M = cand_M
                    best_a = cand
                    a = cand
                    improved = True
                    break
            if improved:
                C = fast_convolve(a)
                break

        if not improved:
            if rng.random() < 0.05:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0006)
            else:
                # If many stalls, give up to avoid burning all time.
                if rem < float(time_limit) * 0.25:
                    break

    return best_a, float(2.0 * n * best_M)


def propose_candidate(seed=42, budget_s=1800):
    start = time.monotonic()
    deadline = start + max(10.0, float(budget_s) - 30.0)
    rng_master = np.random.default_rng(int(seed) if seed is not None else 42)

    # Load parent if present.
    parent = None
    best_seq = None
    best_score = float("inf")

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)  # noqa: F821
        p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
        p = np.maximum(p, 0.0)
        s = float(p.sum())
        if p.ndim == 1 and p.size > 0 and s > 1e-12 and np.all(np.isfinite(p)):
            parent = p / s
            best_seq = parent.copy()
            best_score = direct_score(parent)
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(768, dtype=np.float64) / 768.0
        best_score = direct_score(best_seq)
    print(f"[init] n={len(best_seq)} score={best_score:.12f}")

    # Phase 1: quick multi-n exploration.
    if parent is not None and time.monotonic() < deadline - 600.0:
        t_alloc = min(120.0, deadline - time.monotonic() - 600.0)
        if t_alloc > 20.0:
            print("[phase1] quick exploration")
            best_seq, best_score = quick_explore(
                parent, best_seq, best_score, deadline, t_alloc
            )
            print(f"[phase1] best n={len(best_seq)} score={best_score:.12f}")

    # Phase 2: deep refinement at current best length.
    if best_seq is not None and time.monotonic() < deadline - 300.0:
        n_best = len(best_seq)

        T0 = suggested_T(best_seq, 0.10)
        rem = deadline - time.monotonic() - 400.0
        t = max(0.0, min(250.0, rem))
        if t > 20.0:
            print(f"[phase2] pd smooth n={n_best} t={t:.0f}")
            a_opt, sc = pd_refine(
                best_seq,
                t,
                T_init=T0,
                q=2.5,
                top_m=min(220, 2 * n_best),
                top_w=0.35,
                blend=0.5,
                active_limit=min(380, n_best),
                max_kkt_iters=3,
                inner_time=1.0,
                seed=20
            )
            best_seq, best_score = try_candidate(a_opt, best_seq, best_score)
            print(f"[phase2] pd smooth -> score={best_score:.12f}")

        T0 = suggested_T(best_seq, 0.08)
        rem = deadline - time.monotonic() - 350.0
        t = max(0.0, min(200.0, rem))
        if t > 20.0:
            K_max = 160 if n_best <= 1000 else 90
            print(f"[phase2] lp-kkt n={n_best} t={t:.0f}")
            a_opt, sc = lp_kkt_refine(
                best_seq,
                t,
                K_max=K_max,
                lp_every=12,
                active_limit=min(380, n_best),
                blend=0.5,
                max_kkt_iters=4,
                seed=21
            )
            best_seq, best_score = try_candidate(a_opt, best_seq, best_score)
            print(f"[phase2] lp-kkt -> score={best_score:.12f}")

        T0 = suggested_T(best_seq, 0.07)
        rem = deadline - time.monotonic() - 250.0
        t = max(0.0, min(120.0, rem))
        if t > 20.0:
            print(f"[phase2] nesterov n={n_best} t={t:.0f}")
            a_opt, sc = nesterov_smooth(
                best_seq,
                t,
                T_init=T0,
                lr_init=0.01,
                T_decay=0.9995,
                seed=22
            )
            best_seq, best_score = try_candidate(a_opt, best_seq, best_score)
            print(f"[phase2] nesterov -> score={best_score:.12f}")

        T0 = suggested_T(best_seq, 0.05)
        rem = deadline - time.monotonic() - 180.0
        t = max(0.0, min(120.0, rem))
        if t > 20.0:
            print(f"[phase2] adam n={n_best} t={t:.0f}")
            a_opt, sc = adam_refine(
                best_seq,
                t,
                T_init=T0,
                lr_init=0.015,
                T_decay=0.999,
                seed=23
            )
            best_seq, best_score = try_candidate(a_opt, best_seq, best_score)
            print(f"[phase2] adam -> score={best_score:.12f}")

        rem = deadline - time.monotonic() - 120.0
        t = max(0.0, min(170.0, rem))
        if t > 30.0:
            print(f"[phase2] lbfgs continuation n={n_best} t={t:.0f}")
            a_opt, sc = lbfgs_continuation(best_seq, t, deadline)
            best_seq, best_score = try_candidate(a_opt, best_seq, best_score)
            print(f"[phase2] lbfgs -> score={best_score:.12f}")

    # Phase 3: try a larger length from the current best shape.
    if best_seq is not None and time.monotonic() < deadline - 220.0:
        n_best = len(best_seq)
        targets = []
        for x in (n_best + 256, 2 * n_best, 1024, 1536, 2048):
            if 192 <= int(x) <= 2048 and int(x) != n_best:
                targets.append(int(x))
        targets = sorted(set(targets))

        for n_try in targets[:2]:
            rem = deadline - time.monotonic() - 120.0
            t = max(0.0, min(150.0, rem))
            if t < 40.0:
                break
            print(f"[phase3] larger n={n_try} t={t:.0f}")
            a_init = interp_sequence(best_seq, int(n_try))
            T0 = suggested_T(a_init, 0.10)
            a_opt, sc = lbfgs_continuation(a_init, t, deadline)
            best_seq, best_score = try_candidate(a_opt, best_seq, best_score)

            rem2 = deadline - time.monotonic() - 80.0
            t2 = max(0.0, min(70.0, rem2))
            if t2 > 15.0:
                a_opt2, sc2 = adam_refine(
                    a_opt,
                    t2,
                    T_init=suggested_T(a_opt, 0.06),
                    lr_init=0.012,
                    T_decay=0.999,
                    seed=24
                )
                best_seq, best_score = try_candidate(a_opt2, best_seq, best_score)
            print(f"[phase3] n={n_try} -> best score={best_score:.12f}")

    # Phase 4: final polish.
    if best_seq is not None and time.monotonic() < deadline - 50.0:
        n_best = len(best_seq)

        rem = deadline - time.monotonic() - 30.0
        t = max(0.0, min(70.0, rem))
        if t > 5.0:
            print(f"[phase4] pair descent n={n_best} t={t:.0f}")
            a_opt, sc = pair_descent(
                best_seq,
                t,
                top_peaks=10,
                top_cand=12,
                max_pairs=160
            )
            best_seq, best_score = try_candidate(a_opt, best_seq, best_score)
            print(f"[phase4] pair -> score={best_score:.12f}")

        T0 = suggested_T(best_seq, 0.05)
        rem = deadline - time.monotonic() - 25.0
        t = max(0.0, min(60.0, rem))
        if t > 10.0:
            print(f"[phase4] pd focused n={n_best} t={t:.0f}")
            a_opt, sc = pd_refine(
                best_seq,
                t,
                T_init=T0,
                q=4.5,
                top_m=35,
                top_w=0.65,
                blend=0.35,
                active_limit=min(320, n_best),
                max_kkt_iters=4,
                inner_time=0.8,
                seed=25
            )
            best_seq, best_score = try_candidate(a_opt, best_seq, best_score)
            print(f"[phase4] pd focused -> score={best_score:.12f}")

        T0 = suggested_T(best_seq, 0.04)
        rem = deadline - time.monotonic() - 15.0
        t = max(0.0, min(30.0, rem))
        if t > 8.0:
            a_opt, sc = nesterov_smooth(
                best_seq,
                t,
                T_init=T0,
                lr_init=0.004,
                T_decay=0.9997,
                seed=26
            )
            best_seq, best_score = try_candidate(a_opt, best_seq, best_score)
            print(f"[phase4] final nesterov -> score={best_score:.12f}")

        # Tiny random restarts around best.
        for rep in range(3):
            rem = deadline - time.monotonic() - 5.0
            t = max(0.0, min(8.0, rem))
            if t < 2.0:
                break
            a_pert = project_simplex(
                best_seq + rng_master.standard_normal(len(best_seq)) * 0.0005
            )
            a_opt, sc = nesterov_smooth(
                a_pert,
                t,
                T_init=suggested_T(a_pert, 0.03),
                lr_init=0.003,
                T_decay=0.9998,
                seed=30 + rep
            )
            best_seq, best_score = try_candidate(a_opt, best_seq, best_score)
            print(f"[phase4] pertur {rep} -> score={best_score:.12f}")

    # Final official-score sanity check.
    final_list = [float(x) for x in best_seq]
    try:
        official = float(evaluate_sequence(final_list))  # noqa: F821
        if np.isfinite(official) and official < best_score:
            best_score = official
    except Exception:
        pass

    print(f"[final] n={len(final_list)} score={best_score:.12f} "
          f"elapsed={time.monotonic() - start:.1f}s")
    return final_list
