"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Smooth subgradient optimization with adaptive temperature
- Multi-scale n exploration
- Diverse initializations from window functions
- Momentum-based gradient descent with backtracking
- Periodic restarts and fine-tuning

import numpy as np
from scipy.optimize import linprog
from scipy.special import softmax
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_obj_normalized(a):
    """Compute the objective for a normalized sequence (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, conv, top_k=10, temperature=0.01):
    """Compute softmax-weighted subgradient of max(conv)."""
    max_val = conv.max()
    if max_val < 1e-15:
        return np.zeros(len(a))
    
    n = len(a)
    # Find top peaks
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    # Softmax weights
    pv = conv[peaks]
    weights = softmax((pv - max_val) / (temperature * max_val + 1e-15))
    
    grad = np.zeros(n)
    for w, k in zip(weights, peaks):
        if w < 1e-12:
            continue
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        grad[js:je + 1] += w * 2.0 * a[k - je:k - js + 1][::-1]
    
    return grad

    """
    Gradient-based optimization with adaptive temperature and learning rate.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    
    t0 = time.monotonic()
    lr = 0.1
    momentum = 0.85
    temperature = 0.02
    v = np.zeros(n)
    
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        max_val = conv.max()
        obj = 2.0 * n * max_val
        
        # Compute subgradient
        grad = compute_subgradient(a, conv, top_k=min(20, 2*n-1), temperature=temperature)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm
        
        # Momentum step
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn
        
        new_obj = compute_obj_normalized(a_new)
        
        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Slightly increase learning rate on improvement
            lr = min(lr * 1.02, 0.3)
        else:
            no_improve += 1
            lr *= 0.92
            v *= 0.5
        
        # Adapt temperature
        if no_improve > 30:

def diverse_inits(n, parent=None, rng=None):
    """Generate diverse initializations for a given n."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    inits = []
    x = np.arange(n, dtype=np.float64)
    t = x / (n - 1) if n > 1 else x
    
    # From parent (interpolated)
    if parent is not None:
        px = np.interp(x, np.arange(len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Raised cosine (Hann)
    w = 0.5 * (1 + np.cos(2 * np.pi * t))
    w = np.maximum(w, 0.01)
    inits.append(w / w.sum())
    
    # Flat-top-like
    w = 1.0 - 0.3 * np.cos(2 * np.pi * t)
    inits.append(w / w.sum())
    
    # Sine-modulated
    w = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * t)
    w = np.maximum(w, 0.01)
    inits.append(w / w.sum())
    
    # Edge-heavy
    w = 1.0 / (np.minimum(x, n - 1 - x).astype(float) + 1.0) ** 0.8
    inits.append(w / w.sum())
    
    # Two-peak Gaussian
    sig = 0.15 * n
    w = np.exp(-0.5 * ((x - 0.25 * (n-1)) / sig) ** 2) + np.exp(-0.5 * ((x - 0.75 * (n-1)) / sig) ** 2)
    inits.append(w / w.sum())
    
    # Center-heavy Gaussian
    sig = 0.3 * n
    w = np.exp(-0.5 * ((x - 0.5 * (n-1)) / sig) ** 2)
    inits.append(w / w.sum())
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))
    
    # Parent with perturbation
    if parent is not None:
        px = np.interp(x, np.arange(len(parent)), parent)
        px = np.maximum(px + rng.standard_normal(n) * 0.02, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits
