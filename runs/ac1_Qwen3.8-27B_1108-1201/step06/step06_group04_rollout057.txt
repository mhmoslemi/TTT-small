**New idea: "Spectral shaping" approach**

The key insight is that the auto-convolution in the time domain relates to the power spectrum. Specifically, if we define the "padded" sequence $\tilde{a}$ of length $2n-1$ (with $a$ in the first $n$ positions and zeros elsewhere), then the linear auto-convolution is:
$(a*a)[k] = [\tilde{a} * \tilde{a}][k]$ for $k = 0, \ldots, 2n-2$

The DFT of $\tilde{a}$ is $A(m) = \sum_{j=0}^{n-1} a_j e^{-2\pi i jm/(2n-1)}$ (DFT of length $2n-1$).
The DFT of $\tilde{a} * \tilde{a}$ is $|A(m)|^2$ (circular convolution equals linear since we padded to length $2n-1$... wait, no. The circular convolution of two length-$2n-1$ sequences equals the circular convolution, which equals the linear convolution only if the linear convolution has length at most $2n-1$. The linear convolution of two length-$n$ sequences has length $2n-1$, so yes, it equals the circular convolution of the padded sequences of length $2n-1$.)

So: $(a*a)[k] = \frac{1}{2n-1} \sum_{m=0}^{2n-2} |A(m)|^2 e^{2\pi i mk/(2n-1)}$

We want to minimize $\max_k (a*a)[k]$. In the frequency domain, this is related to making $|A(m)|^2$ as "flat" as possible (in a certain sense).

More precisely, by Parseval: $\sum_k ((a*a)[k])^2 = \frac{1}{2n-1} \sum_m |A(m)|^4$

And $\sum_k (a*a)[k] = (\sum a_j)^2 = 1$.

By Cauchy-Schwarz: $\sum_k ((a*a)[k])^2 \geq \frac{(\sum_k (a*a)[k])^2}{2n-1} = \frac{1}{2n-1}$

So $\max_k (a*a)[k] \geq \sqrt{\frac{\sum_k ((a*a)[k])^2}{2n-1}} \geq \frac{1}{2n-1}$

The first inequality is equality when all $(a*a)[k]$ are equal, which is the ideal case.

The second inequality is equality when $\sum_k ((a*a)[k])^2 = \frac{1}{2n-1}$, which means all peaks are equal (flat auto-convolution).

So the lower bound is $\frac{1}{2n-1}$, giving score $\frac{2n}{2n-1}$.

    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_full_symmetric(a_half, n):
    """Create full symmetric sequence from its half."""
    a = np.zeros(n)
    if n % 2 == 0:
        a[:n // 2] = a_half
        a[n // 2:] = a_half[::-1]
    else:
        a[:n // 2 + 1] = a_half
        a[n // 2 + 1:] = a_half[:n // 2][::-1]
    return a


def get_half(a, n):
    """Extract half of symmetric sequence."""
    if n % 2 == 0:
        return a[:n // 2].copy()
    else:
        return a[:n // 2 + 1].copy()


def symmetrize(a):
    """Enforce symmetry: a_j = a_{n-1-j}."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def score_direct(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max() / (s * s)


def mirror_descent_smooth(a0, time_limit, T_init=0.001, lr_init=0.05,
                          use_symmetry=True, plateau_tol=0.05, verbose=False):
    """
    Mirror descent (exponential weights) with log-sum-exp smoothed objective.
    Adaptive temperature and plateau-aware peak weighting.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    if use_symmetry:
        a = symmetrize(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_max = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit - 0.05:
        it += 1

        conv = fast_convolve(a)
        cmax = conv.max()

        # Plateau-aware weighting: all peaks within tol of max get equal weight
        threshold = cmax * (1.0 - plateau_tol)
        active_peaks = np.where(conv >= threshold)[0]
        if len(active_peaks) == 0:
            active_peaks = np.array([np.argmax(conv)])

        # Softmax weights over active peaks
        T_local = max(T, 1e-6)
        w_active = (conv[active_peaks] - cmax) / T_local
        p_active = np.exp(w_active)
        p_active = p_active / p_active.sum()

        # Full weight vector
        w = np.zeros(2 * n - 1)
        w[active_peaks] = p_active

        # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (for valid i)
        # This is 2 * correlate(w, a) in "valid" mode
        # grad[i] = 2 * sum_k w[k] * a[k-i] where a is defined on [0,n-1]
        # For k in [0, 2n-2], i in [0, n-1]: we need k-i in [0, n-1]
        # So i in [max(0, k-(n-1)), min(n-1, k)]
        # Equivalently: grad = 2 * (w valid a) where (valid) means:
        # grad[i] = sum_k w[k] * a[k-i] for k-i in [0,n-1]
        # = sum_j w[i+j] * a[j] for j in [0, n-1]
        # This is a "Hankel" operation

        # Efficient computation using the fact that w has only a few non-zero entries
        grad = np.zeros(n)
        for k_idx in active_peaks:
            pk = w[k_idx]
            if pk < 1e-12:
                continue
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * pk * a[k_idx - idx]

        # Mirror descent update: a_new[i] proportional to a[i] * exp(-lr * grad[i])
        log_a = np.full(n, -50.0)
        pos = a > 1e-30
        log_a[pos] = np.log(a[pos])
        log_a_new = log_a - lr * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        if use_symmetry:
            a_new = symmetrize(a_new)

        # Line search
        best_candidate = a_new
        best_val = fast_convolve(a_new).max()
        if best_val >= cmax:
            for frac in [0.5, 0.25, 0.1, 0.05]:
                trial = (1 - frac) * a + frac * a_new
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                if use_symmetry:
                    trial = symmetrize(trial)
                val = fast_convolve(trial).max()
                if val < cmax - 1e-15:
                    best_candidate = trial
                    best_val = val
                    break
            else:
                best_candidate = a
                best_val = cmax

        a = best_candidate
        obj = best_val

        if obj < best_max - 1e-14:
            best_max = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive temperature
        if no_improve % 20 == 0 and no_improve > 0:
            T *= 1.08
        if no_improve > 100:
            T = max(T_init * 0.5, T * 0.8)
            lr *= 0.9
        if no_improve > 300:
            rng = np.random.default_rng((it * 7919 + 3) % (2**31))
            perturb = rng.standard_normal(n) * 0.003
            a = project_simplex(best_a + perturb)
            if use_symmetry:
                a = symmetrize(a)
            T = T_init
            lr = lr_init
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    MD it={it}, obj={best_max:.12f}, T={T:.6f}, lr={lr:.6f}, no_imp={no_improve}")

    return best_a, best_max


def peak_shave(a0, time_limit, K=5, verbose=False):
    """
    Direct peak shaving: for the dominant peak, shift mass from 
    contributing positions to non-contributing ones.
    Uses bisection for step size.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit - 0.05 and no_improve < 500:
        it += 1

        conv = fast_convolve(a)
        cmax = conv.max()

        # Get top K peaks
        top_k = np.argpartition(conv, -min(K, 2*n-1))[-min(K, 2*n-1):]

        improved = False
        for k_peak in top_k:
            if time.monotonic() - t0 > time_limit - 0.1:
                break

            # Variables contributing to this peak
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            contrib = np.arange(js, je + 1)
            non_contrib = np.setdiff1d(np.arange(n), contrib)

            if len(contrib) == 0 or len(non_contrib) == 0:
                continue

            # Try to shift mass from the largest contributor to non-contributors
            # Minimize (a*a)[k_peak] by reducing a[contrib] and increasing a[non_contrib]
            
            # Direction: -a[contrib] for contributors, +uniform for non-contributors
            direction = np.zeros(n)
            direction[contrib] = -a[contrib]
            direction[non_contrib] = a[contrib].sum() / len(non_contrib)

            # Ensure direction preserves sum = 0
            direction -= direction.sum() / n

            # Line search
            best_frac = 0.0
            best_val = cmax
            for frac in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]:
                candidate = a + frac * a.sum() * direction
                candidate = project_simplex(candidate)
                c = fast_convolve(candidate)
                if c.max() < best_val - 1e-14:
                    best_val = c.max()
                    best_frac = frac

            if best_frac > 0:
                a = a + best_frac * a.sum() * direction
                a = project_simplex(a)
                new_obj = fast_convolve(a).max()
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    no_improve = 0
                    improved = True

        if not improved:
            no_improve += 1
            # Try random perturbations
            rng = np.random.default_rng((it * 104729 + k_peak) % (2**31))
            for _ in range(5):
                delta = rng.standard_normal(n) * 0.002
                candidate = project_simplex(a + delta)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-14:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    no_improve = 0
                    improved = True
                    break


    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate, find the optimal value
    given all others fixed.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit - 0.1:
        it += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit - 0.1:
                break

            # Fix all coordinates except j
            # Objective: max_k (a*a)[k]
            # (a*a)[k] as function of a[j]:
            #   For k = 2j: a[j]^2 + const
            #   For k != 2j (but k in [0, 2n-2]): a[j] * (sum of a[k-j] for valid k-j) + const
            #   For k where j not in [max(0,k-n+1), min(n-1,k)]: const

            # The peak as a function of a[j] is the max of several linear/quadratic functions
            # To minimize the max, we want a[j] such that the peak is minimized
            # This is a 1D minimization: find a[j] in [0, 1 - sum(others)] that minimizes the peak

            others_sum = a.sum() - a[j]
            a[j] = 0.0  # temporarily set to 0

            # Precompute the "base" convolution (with a[j]=0)
            conv_base = fast_convolve(a)

            # The change in conv when we set a[j] = x:
            # conv_new[k] = conv_base[k] + x * (sum of a[k-j] for valid k-j where k-j != j) 
            #                              + x^2 * [k == 2j]
            # More precisely: conv_new[k] = conv_base[k] + x * g[k] + x^2 * h[k]
            # where g[k] = 2*a[k-j] if k-j in [0,n-1] and k-j != j, else a[k-j] if k-j==j
            # and h[k] = 1 if k == 2j, else 0

            # Actually: (a*a)[k] with a[j] = x:
            # Original has a[j] = 0, so conv_base[k] = sum_{i != j} a[i] * a[k-i] (with a[j]=0)
            # New: sum_i a_new[i] * a_new[k-i] where a_new[j] = x
            # = conv_base[k] + x * a[k-j] + x * a[j] * 0 ... 
            # Wait, let me be more careful.
            # (a_new * a_new)[k] = sum_i a_new[i] * a_new[k-i]
            # = sum_{i != j} a[i] * a_new[k-i] + a_new[j] * a_new[k-j]
            # = sum_{i != j} a[i] * (a[k-i] + (x if k-i == j else 0)) + x * (a[k-j] + (x if k-j == j else 0))
            # = conv_base[k] + x * (sum_{i != j} a[i] * [k-i==j]) + x * a[k-j] + x * [k-j==j] * x + x * [k-j==j] * 0
            # Hmm this is getting complicated. Let me just do a numerical 1D search.

            # 1D golden section search for optimal a[j]
            a[j] = 0.0
            lo, hi = 0.0, others_sum * 0.999

            best_x = 0.0
            best_peak = fast_convolve(a).max()

            # Try a few values
            for trial_x in [0.0, 0.1, 0.25, 0.5, 0.75, 1.0]:
                x = trial_x * others_sum
                if x < 1e-15:
                    continue
                a[j] = x
                a_rest_sum = a.sum() - x
                if a_rest_sum < 1e-15:
                    a[j] = 0
                    continue
                # Normalize: we want sum = 1
                # Actually let's not normalize, just find the best x
                peak = fast_convolve(a).max()
                if peak < best_peak:
                    best_peak = peak
                    best_x = x

            if best_x != a[j]:
                a[j] = best_x
                # Renormalize
                a = a / a.sum()
                new_obj = fast_convolve(a).max()
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True

        if not improved and it > 20:
            break

    return best_a, best_obj


def hankel_gradient(w, a, n):
    """Compute grad[i] = 2 * sum_j w[i+j] * a[j] (Hankel matrix-vector product)."""
    # grad[i] = 2 * sum_j w[i+j] * a[j] for j = 0..n-1
    # This is 2 * (w * a) in a "valid" correlation sense
    # Efficient: for each j with a[j] > 0, add a[j] * w[j:j+n] to grad
    grad = np.zeros(n)
    nonzero = np.where(a > 1e-15)[0]
    for j in nonzero:
        grad += a[j] * w[j:j+n]
    return 2.0 * grad


def active_set_qp(w, a_init, n, max_iters=200, time_limit=3.0):
    """
    Active set QP: min a^T H a s.t. sum(a)=1, a>=0
    where H[i,j] = w[i+j] (Hankel matrix).
    Vectorized for speed.
    """
    t0 = time.monotonic()
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    active = set(np.where(a > 1e-10)[0])
    if not active:
        active = set(range(n))

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if len(active) <= 1:
            a = np.zeros(n)
            if active:
                a[list(active)[0]] = 1.0
            else:
                a[0] = 1.0
            return a

        act = np.array(sorted(active))
        k = len(act)

        # Build Hankel submatrix vectorized
        ii, jj = np.meshgrid(act, act, indexing='ij')
        W_AA = w[ii + jj]
        W_AA = W_AA + 1e-12 * np.eye(k)

        # Solve W_AA x = 1
        try:
            x = np.linalg.solve(W_AA, np.ones(k))
        except np.linalg.LinAlgError:
            try:
                x = np.linalg.lstsq(W_AA, np.ones(k), rcond=None)[0]
            except:
                return a

        x_sum = x.sum()
        if abs(x_sum) < 1e-14:
            return a
        x = x / x_sum

        # Check for negative components
        if x.min() < -1e-12:
            min_i = int(np.argmin(x))
            active.discard(act[min_i])
            a_new = np.zeros(n)
            a_new[act] = np.maximum(x, 0)
            s2 = a_new.sum()
            if s2 > 1e-15:
                a = a_new / s2
            continue

        # Form current solution
        a = np.zeros(n)
        a[act] = np.maximum(x, 0)

        # Compute gradient: grad[i] = 2 * sum_j w[i+j] * a[j]
        grad = hankel_gradient(w, a, n)

        # Lagrange multiplier
        mu = np.mean(grad[act])

        # Check KKT for inactive variables
        inactive_mask = np.zeros(n, dtype=bool)
        inactive_mask[list(active)] = False
        inactive_idx = np.where(inactive_mask)[0]

        if len(inactive_idx) > 0:
            violations = grad[inactive_idx] - mu
            min_viol = violations.min()
            if min_viol < -1e-8:
                worst = inactive_idx[np.argmin(violations)]
                active.add(int(worst))
                continue

        # Converged
        return a

    return a


def qp_primal_dual_v2(a0, time_limit, T_init=0.005, use_symmetry=True,
                      plateau_frac=0.1, verbose=False):
    """
    Improved primal-dual iteration with plateau-aware weighting.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    if use_symmetry:
        a = symmetrize(a)
    n = len(a)

    T = T_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit - 0.5:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        conv = fast_convolve(a)
        cmax = conv.max()

        # Plateau-aware weights
        threshold = cmax * (1.0 - plateau_frac)
        active_peaks = np.where(conv >= threshold)[0]
        k_peaks = len(active_peaks)
        if k_peaks == 0:
            active_peaks = np.array([np.argmax(conv)])
            k_peaks = 1

        # Combine softmax and uniform over active peaks
        scaled = (conv[active_peaks] - cmax) / T
        p_sm = np.exp(scaled)
        p_sm = p_sm / p_sm.sum()
        p_uniform = np.ones(k_peaks) / k_peaks
        p = 0.5 * p_sm + 0.5 * p_uniform

        w = np.zeros(2 * n - 1)
        w[active_peaks] = p

        # Solve QP
        qp_time = min(2.0, remaining * 0.3)
        a_qp = active_set_qp(w, a, n, max_iters=150, time_limit=qp_time)

        # Blend
        a_new = 0.5 * a + 0.5 * a_qp
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = a_qp

        if use_symmetry:
            a_new = symmetrize(a_new)

        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller blend
            for alpha in [0.3, 0.1, 0.05]:
                a_try = (1 - alpha) * a + alpha * a_qp
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                if use_symmetry:
                    a_try = symmetrize(a_try)
                obj_try = fast_convolve(a_try).max()
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a_new = a_try
                    no_improve = 0
                    break
            else:
                a_new = a

        a = a_new

        # Adapt
        if no_improve > 15:
            T *= 1.12
        if no_improve > 50:
            T *= 0.7
        if no_improve > 150:
            rng = np.random.default_rng((it * 6151 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            if use_symmetry:
                a = symmetrize(a)
            T = T_init
            no_improve = 0

    return best_a, best_obj


def nesterov_smoothed(a0, time_limit, T_init=0.0005, lr_init=0.02,
                      use_symmetry=True, verbose=False):
    """
    Nesterov-accelerated gradient with log-sum-exp smoothing.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    if use_symmetry:
        a = symmetrize(a)
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = 0.9
    T = T_init
    y = a.copy()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit - 0.05:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Nesterov lookahead
        y = a + mom * v

        # Compute smoothed gradient at y
        conv_y = fast_convolve(y)
        scaled = conv_y / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient: grad[i] = 2 * sum_k p[k] * y[k-i]
        # = 2 * (p * y) in Hankel sense
        grad = hankel_gradient(p, y, n)

        # Update
        y_proj = project_simplex(y - lr * grad)
        v = y_proj - y
        a = y + v
        a = project_simplex(a)

        if use_symmetry:
            a = symmetrize(a)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt
        lr *= 0.9997
        mom = min(0.995, mom + 0.00003)
        T *= 1.001

        if no_improve > 200:
            lr *= 0.8
            T = T_init
        if no_improve > 500:
            rng = np.random.default_rng((it * 15485863 + 1) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            if use_symmetry:
                a = symmetrize(a)
            v = np.zeros(n)
            lr = lr_init * 0.5
            mom = 0.9
            T = T_init
            no_improve = 0

    return best_a, best_obj


def fine_perturb_search(a0, time_limit, n_trials=200, verbose=False):
    """
    Fine perturbation search: try many small random perturbations
    and keep improvements.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    scale = 0.003

    for it in range(n_trials):
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        rng = np.random.default_rng((it * 48271 + 11) % (2**31))

        for trial in range(3):
            delta = rng.standard_normal(n) * scale
            candidate = project_simplex(a + delta)
            candidate = symmetrize(candidate)
            obj = fast_convolve(candidate).max()

            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = candidate.copy()
                a = candidate.copy()
                break
        else:
            # No improvement, reduce scale sometimes
            if it % 50 == 49:
                scale *= 0.8

    return best_a, best_obj


def multi_scale_refine(a0, time_limit, use_symmetry=True, verbose=False):
    """
    Multi-scale: downsample, optimize, upsample, refine.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()

    # Downsample by factor 2
    n_small = n // 2
    if n_small >= 32:
        # Average pairs
        a_small = a[:n_small * 2].reshape(n_small, 2).mean(axis=1)
        a_small = a_small / a_small.sum()

        t_alloc = min(time_limit * 0.3, time.monotonic() - t0 + time_limit * 0.3)
        a_small_opt, _ = mirror_descent_smooth(
            a_small, time_limit=max(0, t_alloc - (time.monotonic() - t0)),
            T_init=0.01, lr_init=0.03, use_symmetry=use_symmetry
        )

        # Upsample
        a_up = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_small), a_small_opt)
        a_up = np.maximum(a_up, 0)
        a_up = a_up / a_up.sum()
        if use_symmetry:
            a_up = symmetrize(a_up)

        obj_up = fast_convolve(a_up).max()
        if obj_up < best_obj - 1e-14:
            best_obj = obj_up
            best_a = a_up.copy()
            a = a_up.copy()

    # Refine at full scale
    remaining = time_limit - (time.monotonic() - t0) - 0.5
    if remaining > 1:
        a_opt, obj_opt = mirror_descent_smooth(
            a, time_limit=remaining, T_init=0.001, lr_init=0.02, use_symmetry=use_symmetry
        )
        if obj_opt < best_obj - 1e-14:
            best_obj = obj_opt
            best_a = a_opt.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search exploiting symmetry and 
    using diverse optimization strategies for fine-grained improvement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")

    n_parent = len(parent)
    n_sym_parent = symmetrize(parent)
    try_seq(n_sym_parent, "sym_parent")

    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")

    # === Phase 1: Quick exploration of n values (~200s) ===
    print("\n=== Phase 1: Quick n-exploration ===")
    t1 = time.monotonic()

    candidate_ns = sorted(set([
        n_parent,
        n_parent - 32,
        n_parent + 32,
        n_parent - 64,
        n_parent + 64,
        n_parent // 2 if n_parent >= 256 else n_parent,
        n_parent * 2 if n_parent * 2 <= 1536 else n_parent,
        512,
        1024,
    ]))
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 2048]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - (budget_s - 200)
        if remaining < 20:
            break

        # Interpolate parent to new n
        start = np.interp(np.linspace(0, 1, n_try),
                          np.linspace(0, 1, n_parent), parent)
        start = np.maximum(start, 0)
        start = start / start.sum()
        start = symmetrize(start)

        t_alloc = min(remaining, 25)
        print(f"  MD n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = mirror_descent_smooth(start, time_limit=t_alloc,
                                         T_init=0.01, lr_init=0.03, use_symmetry=True)
        try_seq(a_opt, f"md_n{n_try}")

    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")

    # === Phase 2: Deep optimization on best n (~700s) ===
    print("\n=== Phase 2: Deep QP Primal-Dual ===")
    t2 = time.monotonic()

    if best_seq is not None:
        configs = [
            {'T_init': 0.003, 'plateau_frac': 0.05, 't_frac': 0.25},
            {'T_init': 0.005, 'plateau_frac': 0.10, 't_frac': 0.25},
            {'T_init': 0.008, 'plateau_frac': 0.15, 't_frac': 0.20},
            {'T_init': 0.002, 'plateau_frac': 0.03, 't_frac': 0.15},
        ]

        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - (budget_s - 200 - 700)
            if remaining < 15:
                break
            t_alloc = min(remaining, int(budget_s * cfg['t_frac']))
            print(f"  QP-PD config {ci} (T={cfg['T_init']}, pf={cfg['plateau_frac']}), t={t_alloc:.0f}s")
            a_opt, _ = qp_primal_dual_v2(best_seq, time_limit=t_alloc,
                                         T_init=cfg['T_init'],
                                         use_symmetry=True,
                                         plateau_frac=cfg['plateau_frac'])
            try_seq(a_opt, f"qp_pd_{ci}")

    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")

    # === Phase 3: Nesterov + Mirror Descent refinement (~400s) ===
    print("\n=== Phase 3: Nesterov & Mirror Descent ===")
    t3 = time.monotonic()

    if best_seq is not None:
        # Nesterov with different settings
        for T_i, lr_i in [(0.0003, 0.01), (0.001, 0.02), (0.0001, 0.005)]:
            remaining = deadline - time.monotonic() - (budget_s - 200 - 700 - 400)
            if remaining < 10:
                break
            t_alloc = min(remaining, 80)
            print(f"  Nesterov T={T_i}, lr={lr_i}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_smoothed(best_seq, time_limit=t_alloc,
                                         T_init=T_i, lr_init=lr_i, use_symmetry=True)
            try_seq(a_opt, f"nest_T{T_i}")

        # Mirror descent
        remaining = deadline - time.monotonic() - (budget_s - 200 - 700 - 400)
        if remaining > 15:
            t_alloc = min(remaining, 100)
            print(f"  Mirror descent T=0.0005, t={t_alloc:.0f}s")
            a_opt, _ = mirror_descent_smooth(best_seq, time_limit=t_alloc,
                                             T_init=0.0005, lr_init=0.02, use_symmetry=True)
            try_seq(a_opt, "md_fine")

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")

    # === Phase 4: Peak shaving + coordinate descent (~250s) ===
    print("\n=== Phase 4: Peak Shaving & Coord Descent ===")
    t4 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - (budget_s - 200 - 700 - 400 - 250)
        if remaining > 10:
            t_alloc = min(remaining, 100)
            print(f"  Peak shave, t={t_alloc:.0f}s")
            a_opt, _ = peak_shave(best_seq, time_limit=t_alloc, K=5)
            a_opt = symmetrize(a_opt)
            try_seq(a_opt, "peak_shave")

        remaining = deadline - time.monotonic() - (budget_s - 200 - 700 - 400 - 250)
        if remaining > 10:
            t_alloc = min(remaining, 80)
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            a_opt = symmetrize(a_opt)
            try_seq(a_opt, "coord_descent")

        remaining = deadline - time.monotonic() - (budget_s - 200 - 700 - 400 - 250)
        if remaining > 10:
            t_alloc = min(remaining, 50)
            print(f"  Multi-scale, t={t_alloc:.0f}s")
            a_opt, _ = multi_scale_refine(best_seq, time_limit=t_alloc, use_symmetry=True)
            try_seq(a_opt, "multi_scale")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # === Phase 5: Final fine polish (~200s) ===
    print("\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()

    if best_seq is not None:
        # Many rounds of fine perturbation + quick refinement
        for round_i in range(6):
            remaining = deadline - time.monotonic() - 20
            if remaining < 5:
                break

            # Fine perturb
            t_alloc = min(remaining, 30)
            a_opt, _ = fine_perturb_search(best_seq, time_limit=t_alloc,
                                           n_trials=200)
            try_seq(a_opt, f"perturb_r{round_i}")

            # Quick MD refinement
            remaining = deadline - time.monotonic() - 15
            if remaining > 5:
                t_alloc = min(remaining, 20)
                a_opt, _ = mirror_descent_smooth(best_seq, time_limit=t_alloc,
                                                 T_init=0.0002, lr_init=0.01,
                                                 use_symmetry=True, plateau_tol=0.02)
                try_seq(a_opt, f"md_polish_r{round_i}")

            # Peak shave
            remaining = deadline - time.monotonic() - 10
            if remaining > 5:
                t_alloc = min(remaining, 20)
                a_opt, _ = peak_shave(best_seq, time_limit=t_alloc, K=3)
                a_opt = symmetrize(a_opt)
                try_seq(a_opt, f"shave_polish_r{round_i}")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # === Final: One more QP-PD round if time remains ===
    remaining = deadline - time.monotonic() - 10
    if remaining > 10 and best_seq is not None:
        print(f"\n=== Final QP-PD round (t={remaining:.0f}s) ===")
        a_opt, _ = qp_primal_dual_v2(best_seq, time_limit=remaining - 5,
                                     T_init=0.001, use_symmetry=True,
                                     plateau_frac=0.03)
        try_seq(a_opt, "final_qp")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
