"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exponentiated gradient (multiplicative updates) with LSE smoothing
2. Pairwise coordinate descent with peak-targeting
3. Simulated annealing with structured perturbations
4. Multi-length exploration and spectral diversification
5. Adaptive learning rate scheduling
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_obj(a):
    """Compute the objective value for a normalized non-negative sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    c = fast_convolve(a_norm)
    return 2.0 * n * c.max()

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_max(c, beta):
    """Compute LSE approximation of max and its weights."""
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        w = np.zeros(len(c))
        w[c.argmax()] = 1.0
        return c.max(), w
    w = exp_bc / total
    lse_val = bc_max / beta + np.log(total) / beta
    return lse_val, w

def multiplicative_optimize(f0, time_limit, beta_start=50.0, beta_end=50000.0, verbose=False):
    """
    Exponentiated gradient (multiplicative) method with LSE smoothing.
    a[i] <- a[i] * exp(-lr * grad[i]), then renormalize.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = 1.0 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        c = fast_convolve(f)
        _, w = lse_max(c, beta)
        
        # Compute gradient: g[i] = sum_k w[k] * a[k-i] (with bounds)
        g = np.zeros(n)
        for k in range(len(c)):
            if w[k] < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # g[lo:hi+1] += w[k] * f[k - lo : k - hi]
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * f[k - idx]
        
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        
        # Multiplicative update
        step = lr * gn / max(g.max(), 1e-15)  # Normalize step
        f_new = f * np.exp(-step * g / gn * np.sqrt(n))
        f_new_s = f_new.sum()
        if f_new_s < 1e-15:
            # Fallback
            f_new = project_simplex(f - step * g / gn)
        else:
            f_new = f_new / f_new_s
        
        c_new = fast_convolve(f_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.002, 5.0 / np.sqrt(n))
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.9
            if no_improve > 100:
                f = best_f.copy()
                no_improve = 0
                lr = max(lr * 0.5, 0.1 / np.sqrt(n))
                if beta < beta_end:
                    beta = min(beta * 1.5, beta_end)
        
        # Gradually increase beta
        if beta < beta_end:
            beta = min(beta * 1.001, beta_end)
    
    return best_f, best_obj

def coord_descent_optimize(f0, time_limit, verbose=False):

    Pairwise coordinate descent targeting the maximum peaks.
    For each peak, try to reduce it by adjusting the contributing pairs.
    """

    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    # Get top peaks
    n_peaks = min(10, 2*n-1)
    
    while time.monotonic() - t0 < time_limit:

def simulated_annealing(f0, time_limit, T_start=0.01, T_end=0.0001, verbose=False):
    """Simulated annealing with structured perturbations."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    cur_obj = best_obj
    cur_f = f.copy()
    
    t0 = time.monotonic()
    total_iters = int(time_limit * 500)  # estimate ~500 iters/sec
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        T = T_start * (T_end / T_start) ** (it / max(1, total_iters))
        
        # Generate perturbation
        rng = np.random.default_rng(it * 1000003 + 7)
        
        # Random choice of perturbation type
        ptype = rng.integers(0, 4)
        
        if ptype == 0:
            # Single coordinate perturbation
            i = rng.integers(0, n)
            delta = rng.standard_normal() * 0.02 * cur_f.max()
            f_new = cur_f.copy()
            f_new[i] = max(0, cur_f[i] + delta)
            s_new = f_new.sum()
            if s_new > 1e-15:
                f_new /= s_new
        elif ptype == 1:
            # Pair perturbation
            i, j = rng.integers(0, n, size=2)
            if i == j:
                i = (j + 1) % n
            delta = rng.standard_normal() * 0.01 * cur_f.max()
            f_new = cur_f.copy()
            f_new[i] = max(0, cur_f[i] + delta)
            f_new[j] = max(0, cur_f[j] - delta)
            s_new = f_new.sum()
            if s_new > 1e-15:
                f_new /= s_new
        elif ptype == 2:
            # Block perturbation
            block_size = rng.integers(2, min(20, n//4))
            start = rng.integers(0, n - block_size)
            delta = rng.standard_normal(block_size) * 0.01 * cur_f.max()
            f_new = cur_f.copy()
            f_new[start:start+block_size] = np.maximum(0, cur_f[start:start+block_size] + delta)
            s_new = f_new.sum()
            if s_new > 1e-15:
                f_new /= s_new
        else:
            # Spectral perturbation
            F = np.fft.rfft(cur_f)
            mag = np.abs(F)
            phase = np.angle(F)
            pert = rng.standard_normal(len(mag)) * 0.01
            mag_new = np.maximum(mag * (1 + pert), 0)
            F_new = mag_new * np.exp(1j * phase)
            f_new = np.fft.irfft(F_new, n)
            f_new = np.maximum(f_new, 0)
            s_new = f_new.sum()
            if s_new > 1e-15:
                f_new /= s_new
            else:
                f_new = cur_f.copy()
        
        c_new = fast_convolve(f_new)
        obj_new = c_new.max()
        
        delta_obj = obj_new - cur_obj
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / T):
            cur_f = f_new
            cur_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_f = f_new.copy()
    
    return best_f, best_obj

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

def smoothen_and_sharpen(f, sigma):
    """Gaussian smooth then re-sharpen to maintain peakiness."""
    n = len(f)
    kernel_size = max(3, int(6 * sigma) | 1)  # odd
    kernel = np.exp(-0.5 * (np.arange(-kernel_size//2, kernel_size//2+1) / sigma)**2)
    kernel /= kernel.sum()
    f_smooth = np.convolve(f, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0)
    s = f_smooth.sum()
    if s > 1e-15:
        f_smooth /= s
    return f_smooth

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Flat-convolution recursive initialization (theoretical construction)
2. Fast multiplicative (exponentiated gradient) updates with LSE smoothing
3. Targeted pairwise coordinate descent on peak-contributing pairs
4. Simulated annealing with multi-type structured perturbations
5. Multi-length exploration and spectral diversification
6. Adaptive hyperparameter scheduling

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE approximation of max."""
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        w = np.zeros(len(c))
        w[c.argmax()] = 1.0
        return w
    return exp_bc / total


def compute_gradient(f, w, top_k=50):
    """Compute gradient g[i] = sum_k w[k] * f[k-i] using top-k peaks."""
    n = len(f)
    c = fast_convolve(f)
    k_sel = np.argpartition(c, -min(top_k, len(c)))[-min(top_k, len(c)):]
    g = np.zeros(n)
    for k in k_sel:
        k = int(k)
        wk = w[k]
        if wk < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += wk * f[k - idx]
    return g


def flat_conv_init(n):
    """Construct sequence from flat-convolution recursion."""
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):

        a[k] = val / 2.0
    if np.any(a < -1e-8):
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:

        a /= a.sum()
        return a
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    a /= s
    return a


def multiplicative_optimize(f0, time_limit, beta_start=50.0, beta_end=50000.0, top_k=50):
    """Exponentiated gradient method with LSE smoothing and adaptive step."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()

    beta = beta_start
    lr = 2.0 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        c = fast_convolve(f)
        w = lse_weights(c, beta)
        g = compute_gradient(f, w, top_k=top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break

        # Adaptive step size
        step = lr * np.sqrt(n) / gn
        f_new = f * np.exp(-step * g)
        f_new_s = f_new.sum()
        if f_new_s < 1e-15:
            f_new = project_simplex(f - lr * g / gn)
        else:
            f_new = f_new / f_new_s

        c_new = fast_convolve(f_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-16:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 10.0 / np.sqrt(n))
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.92
            if no_improve > 150:
                f = best_f.copy()
                no_improve = 0
                lr = max(lr * 0.6, 0.5 / np.sqrt(n))
                beta = min(beta * 2.0, beta_end)

        if beta < beta_end:
            beta = min(beta * 1.002, beta_end)

    return best_f, best_obj

    n = len(f)

    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()

    beta = beta_start
    lr = 2.0 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        c = fast_convolve(f)
        w = lse_weights(c, beta)
        g = compute_gradient(f, w, top_k=top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g_norm = g / gn

        step = lr
        improved = False
        for _ in range(8):
            f_new = project_simplex(f - step * g_norm)
            c_new = fast_convolve(f_new)
            obj_new = c_new.max()
            if obj_new < best_obj - 1e-16:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                improved = True
                no_improve = 0
                lr = min(lr * 1.002, 5.0 / np.sqrt(n))
                break
            step *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 15 == 0:
                lr *= 0.9
            if no_improve > 80:
                f = best_f.copy()
                lr = max(2.0 / np.sqrt(n), 0.01)
                beta = min(beta * 2.0, beta_end)
                no_improve = 0

        if beta < beta_end:
            beta = min(beta * 1.003, beta_end)

    return best_f, best_obj


def coord_descent_optimize(f0, time_limit, n_peaks=8, n_trials=30):
    """Pairwise coordinate descent targeting maximum peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0u
    else:
        f = f / s
    n = len(f)

    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(f)
        peak_idx = np.argpartition(c, -n_peaks)[-n_peaks:]

        for k in peak_idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo >= hi:
                continue

            js = np.arange(lo, hi + 1)
            products = f[js] * f[k - js]
            j_max = js[np.argmax(products)]
            j_partner = k - j_max

            # Try reducing the dominant contributor
            for j_reduce in [j_max, j_partner]:
                if j_reduce == j_partner:
                    break
                base = f[j_reduce]
                if base < 1e-12:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.15]:
                    delta = frac * base
                    if delta < 1e-14:
                        continue
                    # Try moving to several candidate positions
                    candidates = [0, n // 6, n // 3, n // 2, 2 * n // 3, 5 * n // 6, n - 1]
                    best_dest = -1
                    best_val = best_obj
                    for d in candidates:
                        if d == j_reduce or d == (k - j_reduce):
                            continue
                        f_trial = f.copy()
                        f_trial[j_reduce] -= delta
                        f_trial[d] += delta
                        f_trial = np.maximum(f_trial, 0)
                        st = f_trial.sum()
                        if st < 1e-15:
                            continue
                        f_trial /= st
                        c_trial = fast_convolve(f_trial)
                        if c_trial.max() < best_val:
                            best_val = c_trial.max()
                            best_dest = d
                    if best_dest >= 0:
                        f[j_reduce] -= delta
                        f[best_dest] += delta
                        f = np.maximum(f, 0)
                        f /= f.sum()
                        if best_val < best_obj - 1e-16:
                            best_obj = best_val
                            best_f = f.copy()
                        break

    return best_f, best_obj


def simulated_annealing(f0, time_limit, T_start=0.005, T_end=0.00002, seed=0):
    """Simulated annealing with multi-type perturbations."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    cur_f = f.copy()
    cur_obj = best_obj

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        T = T_start * (T_end / T_start) ** progress

        ptype = rng.integers(0, 5)

        if ptype == 0:
            i = rng.integers(0, n)
            delta = rng.standard_normal() * 0.03 * cur_f.max()
            f_new = cur_f.copy()
            f_new[i] = max(0.0, cur_f[i] + delta)
            s_new = f_new.sum()
            if s_new > 1e-15:
                f_new /= s_new
            else:
                f_new = cur_f.copy()
        elif ptype == 1:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                j = (j + 1) % n
            delta = rng.standard_normal() * 0.02 * cur_f.max()
            f_new = cur_f.copy()
            f_new[i] = max(0.0, cur_f[i] + delta)
            f_new[j] = max(0.0, cur_f[j] - delta)
            s_new = f_new.sum()
            if s_new > 1e-15:
                f_new /= s_new
            else:
                f_new = cur_f.copy()
        elif ptype == 2:
            blk = rng.integers(3, min(30, n // 5))
            start = rng.integers(0, max(1, n - blk))
            delta = rng.standard_normal(blk) * 0.015 * cur_f.max()
            f_new = cur_f.copy()
            f_new[start:start + blk] = np.maximum(0.0, cur_f[start:start + blk] + delta)
            s_new = f_new.sum()
            if s_new > 1e-15:
                f_new /= s_new
            else:
                f_new = cur_f.copy()
        elif ptype == 3:
            F = np.fft.rfft(cur_f)
            mag = np.abs(F)
            phase = np.angle(F)
            pert = rng.standard_normal(len(mag)) * 0.015
            mag_new = np.maximum(mag * (1.0 + pert), 0)
            F_new = mag_new * np.exp(1j * phase)
            f_new = np.fft.irfft(F_new, n)
            f_new = np.maximum(f_new, 0)
            s_new = f_new.sum()
            if s_new > 1e-15:
                f_new /= s_new
            else:
                f_new = cur_f.copy()
        else:
            # Multi-scale: smooth then perturb
            ks = max(3, int(4 + 2 * rng.integers(0, 4)) | 1)
            kernel = np.ones(ks) / ks
            f_smooth = np.convolve(cur_f, kernel, mode='same')
            f_new = f_smooth + rng.standard_normal(n) * 0.005 * cur_f.max()
            f_new = np.maximum(f_new, 0)
            s_new = f_new.sum()
            if s_new > 1e-15:
                f_new /= s_new
            else:
                f_new = cur_f.copy()

        c_new = fast_convolve(f_new)
        obj_new = c_new.max()

        delta_obj = obj_new - cur_obj
        if delta_obj < 0 or (T > 1e-10 and rng.random() < np.exp(-delta_obj / T)):
            cur_f = f_new
            cur_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_f = f_new.copy()

    return best_f, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Generate diverse initializations (flat-conv recursion, parent, interpolated)
    Phase 2: Multiplicative optimization with multiple beta schedules
    Phase 3: Subgradient refinement
    Phase 4: Simulated annealing for diversification
    Phase 5: Coordinate descent for fine-tuning
    Phase 6: Final high-precision polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = flat_conv_init(768)
        print("Using flat-conv initialization")
        try_sequence(parent, best_tracker)

    n_parent = len(parent)

    # === Phase 1: Diverse initializations ===
    print("\n=== Phase 1: Diverse initializations ===")
    t1_end = time.monotonic() + 120.0

    # Try flat-conv initialization for the parent length
    f_flat = flat_conv_init(n_parent)
    try_sequence(f_flat, best_tracker)

    # Try nearby lengths
    for dn in [-16, -8, -4, -2, 2, 4, 8, 16]:
        if time.monotonic() > t1_end:
            break
        nl = n_parent + dn
        if nl < 128 or nl > 2000:
            continue
        f_interp = interpolate_to_length(parent, nl)
        try_sequence(f_interp, best_tracker)
        # Quick multiplicative pass
        t_rem = t1_end - time.monotonic()
        if t_rem > 15:
            f_opt, _ = multiplicative_optimize(f_interp, time_limit=min(20, t_rem),
                                               beta_start=100.0, beta_end=5000.0, top_k=40)
            try_sequence(f_opt, best_tracker)

    # === Phase 2: Main multiplicative optimization ===
    print("\n=== Phase 2: Multiplicative optimization ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t2_end = time.monotonic() + 400.0

    beta_schedules = [
        (50, 5000, 40),
        (100, 10000, 50),
        (200, 20000, 60),
        (300, 30000, 50),
        (500, 50000, 40),
    ]
    for bs, be, tk in beta_schedules:
        if time.monotonic() > t2_end:
            break
        t_rem = t2_end - time.monotonic()
        if t_rem < 20:
            break
        t_alloc = min(100, t_rem * 0.3)
        if t_alloc < 10:
            break
        current_best = best_tracker[1] if best_tracker[1] is not None else current_best
        print(f"  Mult: n={len(current_best)}, beta={bs}->{be}, top_k={tk}, t={t_alloc:.0f}s")
        f_opt, _ = multiplicative_optimize(current_best, time_limit=t_alloc,
                                           beta_start=float(bs), beta_end=float(be), top_k=tk)
        try_sequence(f_opt, best_tracker)

    # === Phase 3: Subgradient refinement ===
    print("\n=== Phase 3: Subgradient refinement ===")
    t3_end = time.monotonic() + 250.0
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    for bs, be, tk in [(20, 2000, 60), (100, 10000, 50), (500, 30000, 30)]:
        if time.monotonic() > t3_end:
            break
        t_rem = t3_end - time.monotonic()
        if t_rem < 20:
            break
        t_alloc = min(80, t_rem * 0.35)
        if t_alloc < 10:
            break
        current_best = best_tracker[1] if best_tracker[1] is not None else current_best
        print(f"  Subgrad: n={len(current_best)}, beta={bs}->{be}, t={t_alloc:.0f}s")
        f_opt, _ = subgradient_optimize(current_best, time_limit=t_alloc,
                                        beta_start=float(bs), beta_end=float(be), top_k=tk)
        try_sequence(f_opt, best_tracker)

    # === Phase 4: Simulated annealing ===
    print("\n=== Phase 4: Simulated annealing ===")
    t4_end = time.monotonic() + 350.0
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)

    for sa_i in range(6):
        if time.monotonic() > t4_end:
            break
        t_rem = t4_end - time.monotonic()
        if t_rem < 25:
            break
        t_alloc = min(60, t_rem * 0.25)
        if t_alloc < 10:
            break

        # Create perturbed starting point for diversity
        if sa_i == 0:
            f_sa = current_best.copy()
        elif sa_i == 1:
            F = np.fft.rfft(current_best)
            mag = np.abs(F)
            phase = np.angle(F)
            pert = rng.standard_normal(len(mag)) * 0.02
            mag_new = np.maximum(mag * (1.0 + pert), 0)
            f_sa = np.fft.irfft(mag_new * np.exp(1j * phase), n_cur)
            f_sa = np.maximum(f_sa, 0)
            f_sa /= f_sa.sum()
        elif sa_i == 2:
            f_sa = current_best + rng.standard_normal(n_cur) * 0.01
            f_sa = project_simplex(f_sa)
        else:
            F = np.fft.rfft(current_best)
            mag = np.abs(F)
            phase = np.angle(F)
            pert = rng.standard_normal(len(mag)) * (0.03 + 0.01 * sa_i)
            mag_new = np.maximum(mag * (1.0 + pert), 0)
            f_sa = np.fft.irfft(mag_new * np.exp(1j * phase), n_cur)
            f_sa = np.maximum(f_sa, 0)
            f_sa /= f_sa.sum()

        T_s = 0.003 * (1.0 + 0.3 * sa_i)
        T_e = 0.00001 * (0.8 ** sa_i)
        print(f"  SA {sa_i}: n={n_cur}, T={T_s:.5f}->{T_e:.7f}, t={t_alloc:.0f}s")
        f_opt, _ = simulated_annealing(f_sa, time_limit=t_alloc,
                                        T_start=T_s, T_end=T_e, seed=seed + sa_i * 1000)
        try_sequence(f_opt, best_tracker)
        current_best = best_tracker[1] if best_tracker[1] is not None else current_best

    # === Phase 5: Coordinate descent ===
    print("\n=== Phase 5: Coordinate descent ===")
    t5_end = time.monotonic() + 200.0
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(150, t5_end - time.monotonic())
    if t_alloc > 15:
        print(f"  CoordDesc: n={len(current_best)}, t={t_alloc:.0f}s")
        f_opt, _ = coord_descent_optimize(current_best, time_limit=t_alloc,
                                          n_peaks=8, n_trials=30)
        try_sequence(f_opt, best_tracker)

    # === Phase 6: Final high-precision polish ===
    print("\n=== Phase 6: Final polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_final = len(current_best)

    # Very high beta multiplicative (target top few peaks)
    t_rem = deadline - time.monotonic()
    if t_rem > 40:
        t_alloc = min(120, t_rem * 0.35)
        print(f"  High-beta mult: n={n_final}, t={t_alloc:.0f}s")
        f_opt, _ = multiplicative_optimize(current_best, time_limit=t_alloc,
                                           beta_start=1000.0, beta_end=80000.0, top_k=25)
        try_sequence(f_opt, best_tracker)

    # Another SA pass at low temperature
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t_rem = deadline - time.monotonic()
    if t_rem > 30:
        t_alloc = min(80, t_rem * 0.4)
        print(f"  Final SA: n={len(current_best)}, t={t_alloc:.0f}s")
        f_opt, _ = simulated_annealing(current_best, time_limit=t_alloc,
                                        T_start=0.001, T_end=0.000005, seed=seed + 9999)
        try_sequence(f_opt, best_tracker)

    # Quick random perturbations
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t_rem = deadline - time.monotonic()
    if t_rem > 15:
        print(f"  Final perturbations: t={t_rem:.0f}s")
        for _ in range(30):
            if time.monotonic() > deadline - 5:
                break
            # Time-domain
            f_pert = current_best + rng.standard_normal(len(current_best)) * 0.0008
            f_pert = project_simplex(f_pert)
            try_sequence(f_pert, best_tracker)
            # Spectral
            F = np.fft.rfft(current_best)
            mag = np.abs(F)
            phase = np.angle(F)
            pert = rng.standard_normal(len(mag)) * 0.002
            mag_new = np.maximum(mag * (1.0 + pert), 0)
            f_pert2 = np.fft.irfft(mag_new * np.exp(1j * phase), len(current_best))
            f_pert2 = np.maximum(f_pert2, 0)
            s = f_pert2.sum()
            if s > 1e-15:
                f_pert2 /= s
            else:
                f_pert2 = current_best.copy()
            try_sequence(f_pert2, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    print("WARNING: Using parent or uniform fallback")
    if parent is not None:
        return [float(x) for x in parent]
    fallback = np.ones(768) / 768.0
    return [float(x) for x in fallback]
