def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_objective(a, T):
    conv = fast_convolve(a)
    # Log-sum-exp for numerical stability
    shifted = T * (conv - conv.max())
    return conv.max() + np.log(np.sum(np.exp(shifted))) / T

    shifted = T * (conv - conv.max())

import numpy as np

def fast_convolve(a):

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize(a):
    """Enforce symmetry a[j] = a[n-1-j]."""
    return (a + a[::-1]) / 2.0


def softmax(w):
    """Numerically stable softmax."""
    w_shifted = w - w.max()
    ew = np.exp(w_shifted)
    s = ew.sum()
    if s < 1e-300:
        return np.ones(len(w)) / len(w)
    return ew / s

    else:

    a = softmax(w)
    conv = fast_convolve(a)

            y = np.ones(n) / n
        else:
            y = y / ys

        # Adaptive T
        progress = min(1.0, it / 5000.0)
        T_curr = T_start * (T_end / T_start) ** progress

        # Gradient
        conv_y = fast_convolve(y)
        max_c = conv_y.max()
        shifted = T_curr * (conv_y - max_c)
        if shifted.max() > 700:
            shifted = shifted - shifted.max()
        p = np.exp(shifted)
        ps = p.sum()
        if ps < 1e-300:
            p = np.ones(len(p)) / len(p)
        else:
            p = p / ps
        grad = 2.0 * np.correlate(p, y, mode='valid')

        # Backtracking line search
        step = lr
        candidate = y
        best_val = max_c
        for _ in range(15):
            candidate = y - step * grad
            if use_symmetry:
                candidate = symmetrize(candidate)
            candidate = project_simplex(candidate)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                break
            step *= 0.5

        x_prev = a
        a = candidate

        true_obj = fast_convolve(a).max()
        if true_obj < best_obj - 1e-13:
            best_obj = true_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                lr *= 0.9
            if no_improve > 200:
                rng = np.random.default_rng(it * 31 + 7)
                noise = 0.003 * min(1.0, (no_improve - 200) / 200.0) + 0.001
                a = project_simplex(best_a + rng.standard_normal(n) * noise)
                if use_symmetry:
                    a = symmetrize(a)
                    a = a / a.sum()
                x_prev = a.copy()
                beta = 1.0
                lr = lr_init
                no_improve = 0

    return best_a, best_obj


def solve_lp_oracle(f, M, max_time=15):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix A_ub where A_ub[k, j] = f[j] for k-j in [0, n-1]
    # A_ub has shape (2n-1, n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max_time})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def frank_wolfe(f0, time_limit, use_symmetry=True):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0, 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    if use_symmetry:
        f = symmetrize(f)
        f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Solve LP oracle
        g = solve_lp_oracle(f, M, max_time=min(10, remaining - 2))
        if g is None:
            no_improve += 1
            if no_improve > 10:
                break
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        if use_symmetry:
            g = symmetrize(g)
            g = g / g.sum()

        # Check FW gap
        gap = M - np.dot(f, g)
        if gap < 1e-12:
            break

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val - 1e-16:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            # Bisection refinement
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            if use_symmetry:
                f = symmetrize(f)
                f = f / f.sum()

            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            if no_improve > 20:
                break

    return best_f, best_obj


def active_set_newton(a0, time_limit, use_symmetry=True, tol=1e-10):
    """
    Active-set Newton: identify active peaks, solve KKT conditions iteratively.
    """
    a = np.maximum(a0, 0.0).copy()
    a = a / a.sum()
    if use_symmetry:
        a = symmetrize(a)
        a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < 5000:
        it += 1
        if time.monotonic() - t0 > time_limit - 1:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Find active peaks (within 0.05% of max)
        threshold = M * (1.0 - 0.0005)
        active_peaks = np.where(conv >= threshold)[0]

        if len(active_peaks) <= 1:
            break

        # Support of a
        support = np.where(a > 1e-8)[0]
        if len(support) == 0:
            break

        # Build the KKT linear system
        # For j in support: sum_{k in active} mu_k * 2*a[k-j] = lambda
        # We solve for mu (non-negative) and lambda
        m = len(support)
        p = len(active_peaks)

        # A_sys[i, k_idx] = 2 * a[active_peaks[k_idx] - support[i]] (if valid)
        # A_sys[i, -1] = -1 (for lambda)
        A_sys = np.zeros((m, p + 1))
        for i, j in enumerate(support):
            A_sys[i, -1] = -1.0
            for k_idx, k in enumerate(active_peaks):
                idx = k - j
                if 0 <= idx < n:
                    A_sys[i, k_idx] = 2.0 * a[idx]

        # We want: A_sys @ [mu; lambda] = 0 (homogeneous)
        # Use SVD to find the smallest singular vector
        try:
            U, S_vals, Vt = np.linalg.svd(A_sys)
            sol = Vt[-1, :]
            mu_vals = sol[:p]
            lam_val = sol[p]
        except np.linalg.LinAlgError:
            break

        # Ensure mu >= 0
        mu_vals = np.maximum(mu_vals, 0)
        mu_sum = mu_vals.sum()
        if mu_sum < 1e-15:
            # Try flipping sign
            mu_vals = np.maximum(-mu_vals, 0)
            mu_sum = mu_vals.sum()
            if mu_sum < 1e-15:
                break
        mu_vals /= mu_sum

        # Compute gradient direction: -sum_k mu_k * 2*a[k-j]
        grad = np.zeros(n)
        for k_idx, k in enumerate(active_peaks):
            if mu_vals[k_idx] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] -= 2.0 * mu_vals[k_idx] * a[k - idx_arr]

        # Projected gradient step with line search
        best_candidate = a
        best_val = M
        for step in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            candidate = a + step * grad
            if use_symmetry:
                candidate = symmetrize(candidate)
            candidate = project_simplex(candidate)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate

        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
        elif it > 50:
            # Stuck - try different active set threshold
            threshold_factor = 1.0 - 0.002
            active_peaks2 = np.where(conv >= M * threshold_factor)[0]
            if len(active_peaks2) <= 1:
                break

    return best_a, best_obj


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def try_and_update(a, best_state, label=""):
    """Evaluate sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_state
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    best_seq, best_score = best_state
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return (best_seq, best_score)
