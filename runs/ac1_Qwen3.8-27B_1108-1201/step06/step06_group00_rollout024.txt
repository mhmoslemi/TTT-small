Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS on LSE-smoothed objective with softmax parameterization (quasi-Newton, fast convergence)
2. Geometric beta schedule for progressive sharpening

4. Gerchberg-Saxton spectrum flattening as secondary diversifier

import numpy as np

def fast_autoconv(a):
    """Fast linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:size]


def softmax_vec(t):
    """Numerically stable softmax."""
    t_shifted = t - t.max()
    e = np.exp(t_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(t)) / len(t)
    return e / s


def compute_grad_a(w, a):
    """Compute grad_a[j] = 2 * sum_k w[k] * a[k-j] for j=0..n-1 using FFT."""
    n = len(a)
    a_flipped = a[::-1]
    size = len(w) + n - 1
    N = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, N)
    AF = np.fft.rfft(a_flipped, N)
    conv = np.fft.irfft(W * AF, N)[:size]
    return 2.0 * conv[n - 1:2 * n - 1]


def lse_value_and_grad(t, beta):
    """LSE-smoothed objective and its gradient w.r.t. log-domain parameter t."""
    a = softmax_vec(t)
    c = fast_autoconv(a)
    bc = beta * c
    bc_max = bc.max()
    shifted = bc - bc_max
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    if sum_exp < 1e-300:
        return 0.0, np.zeros_like(t)
    log_Z = bc_max + np.log(sum_exp)
    L = log_Z / beta
    w = exp_shifted / sum_exp
    grad_a = compute_grad_a(w, a)
    S = np.dot(a, grad_a)
    grad_t = a * (grad_a - S)
    return L, grad_t

    print("WARNING: Using uniform fallback")
    fallback = np.ones(768) / 768.0
    return [float(x) for x in fallback]
