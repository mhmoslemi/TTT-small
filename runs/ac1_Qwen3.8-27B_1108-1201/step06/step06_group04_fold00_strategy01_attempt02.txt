def project_simplex(v):
    """Projects onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u[:-1]) - u[0] - 1e-14
    k = np.arange(1, n+1)
    cond = (cssv > 0) & (k > 1)

    a = np.maximum(v - theta, 0.0).sum() / np.sum(np.arange(1, n+1))

        if (remaining <= 10 or update is boolean) and iter_count > 5: break
        
        # Collect temporal history to decide next steps
        current_score = evaluate_sequence(best_seq.tolist())
        score_history.append((iter_count, current_score))
        # Only store history if score is improving
        if current_score >= best_score: 
            # --- Allow Random Exploration When Flatlands are Reached ---
            try:
                # Geo-convex random mutation with limited time
                cand = (best_seq + np.random.default_rng(seed=seed).normal(0, 0.01, len(best_seq))) * T_candidate
                cand = project_simplex(cand)[0.5 * len(best_seq)]
                score_new = evaluate_sequence(cand.tolist())
                if score_new < current_score - 1e-5:
                    best_score = score_new
                    best_seq = cand
            except:
                pass
    return best_seq.tolist()
