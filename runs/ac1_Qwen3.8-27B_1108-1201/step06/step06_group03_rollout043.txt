"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-sum-exp smoothed
gradient descent with Adam optimizer, annealing, and multi-length search.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Projected gradient step onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_objective_and_grad(a, beta):
    """Compute smooth objective value and gradient."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed gradient descent with Adam optimizer (core method)
2. Beta annealing for progressively sharper peak targeting
3. Multi-length exploration with resampling
4. Hard subgradient refinement and coordinate descent for final polishing
Key difference from prior approach: uses smooth log-sum-exp surrogate with Adam
rather than LP-based Frank-Wolfe, enabling per-coordinate adaptive step sizes
and smoother convergence landscape.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(a, beta):
    """
    Compute gradient of f_smooth(a) = (1/beta) * log(sum_k exp(beta * b_k))
    where b = a * a (auto-convolution).
    Returns (objective_value, gradient).
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = beta * (b - max_b)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    w_sum = w.sum()
    w = w / w_sum
    obj = max_b + np.log(w_sum) / beta

    # Gradient: grad[i] = 2 * sum_m a[m] * w[m+i] (cross-correlation)
    # Using FFT: (a cross w)[i] = (a * w_flipped)[m-1-i] where m = len(w)
    m = len(w)
    size = 1 << (n + m - 1).bit_length()
    A = np.fft.rfft(a, size)
    W_flipped = np.fft.rfft(w[::-1], size)
    conv = np.fft.irfft(A * W_flipped, size)
    # (a cross w)[i] = conv[m-1-i] for i = 0, ..., n-1
    # We need indices: m-1, m-2, ..., m-n
    # In array: conv[m-1], conv[m-2], ..., conv[m-n]
    start = m - n
    end = m  # exclusive
    cross = conv[start:end][::-1]  # cross[i] = (a cross w)[i]
    grad = 2.0 * cross
    return obj, grad, w, b


def adam_smooth_optimize(a0, time_limit, beta_init=5.0, beta_max=100.0,
                         lr=0.008, anneal_rate=1.008, restart_every=300):
    """
    Adam optimization of the log-sum-exp smoothed objective.
    Anneals beta to gradually sharpen peak targeting.
    """
    a = project_simplex(a0.astype(np.float64).copy())
    n = len(a)

    # Adam hyperparameters
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8

    # Initialize Adam state
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_iter = 0

    beta = beta_init
    best_a = a.copy()
    best_max = fast_convolve(a).max()

    t0 = time.monotonic()
    stagnant = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        t_iter += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.05:
            break

        # Compute smooth gradient
        obj, grad, w, b = compute_smooth_gradient(a, beta)
        true_max = b.max()

        # Anneal beta
        beta = min(beta * anneal_rate, beta_max)

        # Adam update
        t_iter += 1
        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * (grad ** 2)
        m_hat = m_adam / (1 - beta1 ** t_iter)
        v_hat = v_adam / (1 - beta2 ** t_iter)

        # Adaptive learning rate based on iteration
        effective_lr = lr * min(1.0, 10.0 / (1.0 + t_iter * 0.01))

        # Gradient step with line search
        step_size = effective_lr
        candidate = None
        for _attempt in range(5):
            raw_step = -step_size * m_hat / (np.sqrt(v_hat) + eps)
            candidate = project_simplex(a + raw_step)
            cand_max = fast_convolve(candidate).max()
            if cand_max < true_max - 1e-12:
                break
            step_size *= 0.5
            if step_size < 1e-10:
                candidate = None
                break

        if candidate is not None:
            a = candidate
            new_max = fast_convolve(a).max()
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                stagnant = 0
                no_improve_count = 0
            else:
                stagnant += 1
                no_improve_count += 1
        else:
            stagnant += 1
            no_improve_count += 1

        # Periodic restart to escape local minima
        if no_improve_count > restart_every:
            rng = np.random.default_rng(t_iter * 31 + 7)
            noise_scale = 0.002 + 0.005 * min(1.0, no_improve_count / (2 * restart_every))
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            t_iter = 0
            beta = max(beta_init, beta * 0.5)  # Reset beta somewhat
            stagnant = 0
            no_improve_count = 0

    return best_a, best_max


def hard_subgradient_optimize(a0, time_limit, top_k=3, lr_init=0.01):
    """
    Hard subgradient descent targeting the top-k peaks of the convolution.
    Used for final polishing after smooth optimization.
    """
    a = project_simplex(a0.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_max = fast_convolve(a).max()
    lr = lr_init
    v_mom = np.zeros(n)
    momentum = 0.9

    t0 = time.monotonic()
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.03:
            break

        b = fast_convolve(a)
        # Get top-k peak indices
        if top_k < len(b):
            peak_indices = np.argpartition(b, -top_k)[-top_k:]
        else:
            peak_indices = np.arange(len(b))

        # Compute subgradient: average of gradients at top peaks
        grad = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            # db_k/da[i] = 2*a[k-i] for 0 <= k-i < n
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            indices = np.arange(js, je + 1)
            grad[indices] += 2.0 * a[k - indices]

        grad /= len(peak_indices)

        # Momentum + step
        v_mom = momentum * v_mom - lr * grad

        # Line search
        best_candidate = None
        best_val = b.max()
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            candidate = project_simplex(a + frac * v_mom)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-13:
                best_val = val
                best_candidate = candidate

        if best_candidate is not None:
            a = best_candidate
            if best_val < best_max - 1e-12:
                best_max = best_val
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        # Adapt learning rate
        if stagnant > 30:
            lr *= 0.9
        if stagnant > 100:
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            v_mom = np.zeros(n)
            lr = lr_init
            stagnant = 0
        if lr < 1e-6:
            break

    return best_a, best_max


def peak_coordinate_descent(a0, time_limit):
    """
    Coordinate descent: for each top peak, try shifting mass from
    the largest contributing pair to a distant location.
    """
    a = project_simplex(a0.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_max = fast_convolve(a).max()

    t0 = time.monotonic()
    step_fracs = [0.05, 0.1, 0.2, 0.3, 0.5]
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.03:
            break

        b = fast_convolve(a)
        # Target top 3 peaks
        n_peaks = min(3, len(b))
        if n_peaks < len(b):
            peak_indices = np.argpartition(b, -n_peaks)[-n_peaks:]
        else:
            peak_indices = np.arange(len(b))

        improved = False
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue

            # Find the pair contributing most to this peak
            best_pair = None
            best_val_pair = 0
            for j in range(js, je + 1):
                val = a[j] * a[k - j]
                if val > best_val_pair:
                    best_val_pair = val
                    best_pair = (j, k - j)

            if best_pair is None:
                continue
            j1, j2 = best_pair
            src = j1 if a[j1] >= a[j2] else j2

            # Try moving some mass from src to random distant positions
            rng = np.random.default_rng(it * 9973 + k * 31 + src * 7)
            for _ in range(10):
                target = int(rng.integers(0, n))
                if abs(target - src) < n // 8:
                    continue
                for frac in step_fracs[:3]:
                    delta = a[src] * frac
                    if delta < 1e-10:
                        continue
                    candidate = a.copy()
                    candidate[src] -= delta
                    candidate[target] += delta
                    candidate = np.maximum(candidate, 0)
                    s = candidate.sum()
                    if s < 1e-15:
                        continue
                    candidate /= s
                    val = fast_convolve(candidate).max()
                    if val < best_max - 1e-12:
                        a = candidate
                        best_max = val
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved and it > 500:
            # Try increasing step size
            step_fracs = [f * 1.5 for f in step_fracs]

    return best_a, best_max


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_startup(n, parent, rng):
    """Generate a diverse set of initial sequences for length n."""
    starts = []
    x = np.arange(n, dtype=np.float64)

    # Resampled parent
    if parent is not None:
        starts.append(resample_sequence(parent, n))

    # Uniform
    starts.append(np.ones(n) / n)

    # Cosine taper (smooth, peaks at edges)
    cos_t = 0.5 * (1 + np.cos(np.pi * x / max(1, n - 1)))
    cos_t = np.maximum(cos_t, 0.01)
    starts.append(cos_t / cos_t.sum())

    # Gaussian centered
    g = np.exp(-0.5 * ((x - (n - 1) / 2) / (0.15 * n)) ** 2)
    starts.append(g / g.sum())

    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    starts.append(edge / edge.sum())

    # Two peaks
    g1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.06 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.06 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())

    # Random Dirichlet (concentrated)
    starts.append(rng.dirichlet(np.ones(n) * 0.5))

    return starts
