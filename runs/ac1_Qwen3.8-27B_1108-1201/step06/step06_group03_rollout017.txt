    Gerchberg-Saxton spectral flattening.
    Alternates between:
    - Time domain: enforce p >= 0, sum(p) = 1
    - Frequency domain: flatten |FFT(p)|

    if s < 1e-15:

        # Blend current and target
        new_mag = mag.copy()
        new_mag[1:] = overlap * mag[1:] + (1.0 - overlap) * target_mag

            new_mag[k2] = avg_mag

def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (f*g) = A @ g."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

            lr = lr_init * (1 + 0.1 * (t_step // 100))
            m = np.zeros(n)
            v = np.zeros(n)
            stagnant = 0
        
        if stagnant > 300:
            rng = np.random.default_rng((t_step * 997 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            m = np.zeros(n)
            v = np.zeros(n)

def peak_coordinate_descent(a0, time_limit):
    """Coordinate descent targeting the top peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    step_scale = 0.003
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        n_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]
        
        improved = False
        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            if js > je:
                continue
            
            # Find the coordinate that contributes most to this peak
            contributions = np.array([a[j] * a[k - j] for j in range(js, je + 1)])
            j_rel = np.argmax(contributions)
            j1 = js + j_rel
            j2 = k - j1
            
            # Try to reduce a[j1] (or a[j2]) and add to a random distant location
            for src in [j1, j2]:
                if a[src] < 1e-8:
                    continue
                
                # Try several targets
                for _attempt in range(8):
                    j_target = int(rng.integers(0, n))
                    if abs(j_target - src) < n // 10:
                        continue
                    
                    for delta_frac in [step_scale, step_scale * 0.5, step_scale * 2.0]:
                        delta = a[src] * delta_frac
                        candidate = a.copy()
                        candidate[src] -= delta
                        candidate[j_target] += delta
                        if candidate[src] < -1e-12:
                            continue
                        candidate = np.maximum(candidate, 0)
                        cst = candidate.sum()
                        if cst < 1e-15:
                            continue
                        candidate = candidate / cst
                        val = fast_convolve(candidate).max()
                        if val < best_obj - 1e-15:
                            a = candidate
                            best_obj = val
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            step_scale *= 0.95
            if step_scale < 1e-7:
                break
            if it % 100 == 0:
                candidate = project_simplex(a + rng.standard_normal(n) * step_scale * 3)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
            elif it > 3000:
                break
    
    return best_a, best_obj
