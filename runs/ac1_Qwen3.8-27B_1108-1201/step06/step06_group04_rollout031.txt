def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]

So: f(a) = f(a*) + 2*(H_k a*)^T (a-a*) + (a-a*)^T (2*H_k) (a-a*)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Fast pair coordinate descent with O(n) incremental convolution updates
- Smoothed gradient (log-sum-exp) with adaptive temperature
- Multi-n exploration with structured initializations
- Symmetry exploitation and random restarts
- LP-based refinement via cvxpy (if available)
"""

import numpy as np
import time
import sys

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def pair_cd(a, time_limit, rng=None, verbose=False):
    """
    Pair coordinate descent: minimize max(conv(a,a)) s.t. a>=0, sum(a)=1.
    At each step, find the best (i, j, delta) triple to shift mass
    from position i to position j, optimizing delta for that pair.
    Uses O(n) incremental convolution updates between FFT refreshes.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    if rng is None:
        rng = np.random.default_rng(42)
    
    t0 = time.monotonic()
    conv = fast_convolve(a)
    best_max = conv.max()
    best_a = a.copy()
    no_improve = 0
    it = 0
    moves_since_fft = 0
    fft_interval = 50  # recompute FFT every 50 moves
    
    delta_options = [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1]
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        cmax = conv.max()
        k_peak = int(np.argmax(conv))
        
        # Compute partner values for the peak
        # partner[i] = a[k_peak - i] if 0 <= k_peak-i < n, else 0
        partners = np.zeros(n)
        lo = max(0, k_peak - n + 1)
        hi = min(n - 1, k_peak)
        partners[lo:hi+1] = a[k_peak - hi : k_peak - lo + 1]
        
        # Identify source candidates (high partner, contribute to peak)
        # and target candidates (low partner, don't contribute much)
        nonzero_mask = a > 1e-12
        if not nonzero_mask.any():
            break
        
        # Score: contribution to peak = a[i] * partners[i]
        contrib = a * partners
        
        # Sources: top contributors
        n_src = min(20, n // 5)
        n_tgt = min(20, n // 5)
        
        sorted_contrib = np.argsort(contrib)[::-1]
        sources = [sorted_contrib[k] for k in range(min(n_src, len(sorted_contrib))) if nonzero_mask[sorted_contrib[k]]]
        
        sorted_partner_asc = np.argsort(partners)
        targets = [sorted_partner_asc[k] for k in range(min(n_tgt, len(sorted_partner_asc)))]
        # Ensure targets are different from sources
        target_set = set(targets)
        source_set = set(sources)
        targets = [t for t in targets if t not in source_set]
        if len(targets) < 5:
            # Add random targets
            all_idx = np.where(nonzero_mask)[0]
            for t in rng.choice(all_idx, size=min(10, len(all_idx)), replace=False):
                t = int(t)
                if t not in source_set and t not in target_set:
                    targets.append(t)
                    target_set.add(t)
        
        best_move = None
        best_new_max = cmax
        
        for i in sources:
            if a[i] < 1e-10:
                continue
            # Partner vector for source i: pi[k] = a[k-i] if valid
            # pi[i:i+n] = a
            max_d = min(a[i] * 0.95, 0.15)
            
            for j in targets:
                # Change in conv: -2d*a[k-i] + 2d*a[k-j] + d^2*(I[k=2i] - I[k=2j])
                # = 2d*(a[k-j] - a[k-i]) + d^2*(I[k=2i] - I[k=2j])
                
                # For the peak specifically:
                # Δconv[peak] = 2d*(partners[j] - partners[i]) + d^2*(I[peak=2i] - I[peak=2j])
                # Since partners[i] > partners[j], the linear term is negative (good)
                
                # Quick estimate of improvement at peak
                est_peak_change = 2 * max_d * (partners[j] - partners[i])
                if est_peak_change > -1e-12:
                    continue  # Won't help at peak
                
                # Try delta values
                for d in delta_options:
                    if d > max_d:
                        break
                    # Compute new max (approximate: check peak and a few neighbors)
                    # Full check:
                    # conv_new = conv + 2*d*(pj_vec - pi_vec) + d^2*(e_{2i} - e_{2j})
                    # For efficiency, compute the change vector
                    # delta_conv[k] = 2*d*(a[k-j] - a[k-i]) + d^2*(I[k=2i] - I[k=2j])
                    
                    # Compute a[k-i] and a[k-j] vectors
                    # a[k-i] is non-zero for k in [i, i+n-1]
                    # a[k-j] is non-zero for k in [j, j+n-1]
                    
                    # The change is non-zero only in [min(i,j), max(i,j)+n-1]
                    k_lo = max(0, min(i, j))
                    k_hi = min(2*n-2, max(i, j) + n - 1)
                    
                    # Build change vector (only for the relevant range)
                    # delta_conv[k] = 2d*(a[k-j] - a[k-i])
                    ak_i = np.zeros(k_hi - k_lo + 1)
                    ak_j = np.zeros(k_hi - k_lo + 1)
                    
                    # a[k-i] for k in [k_lo, k_hi]: valid when 0 <= k-i < n
                    si = max(0, k_lo - i)  # first valid a index
                    ei = min(n-1, k_hi - i)  # last valid a index
                    if si <= ei:
                        ak_i[k_lo-i+si : k_lo-i+ei+1] = a[si:ei+1]
                    
                    sj = max(0, k_lo - j)
                    ej = min(n-1, k_hi - j)
                    if sj <= ej:
                        ak_j[k_lo-j+sj : k_lo-j+ej+1] = a[sj:ej+1]
                    
                    delta_conv = 2*d*(ak_j - ak_i)
                    if 2*i >= k_lo and 2*i <= k_hi:
                        delta_conv[2*i - k_lo] += d*d
                    if 2*j >= k_lo and 2*j <= k_hi:
                        delta_conv[2*j - k_lo] -= d*d
                    
                    new_conv_seg = conv[k_lo:k_hi+1] + delta_conv
                    new_max = max(new_conv_seg.max(), 
                                  conv[:k_lo].max() if k_lo > 0 else -np.inf,
                                  conv[k_hi+1:].max() if k_hi < 2*n-2 else -np.inf)
                    
                    if new_max < best_new_max:
                        best_new_max = new_max
                        best_move = (i, j, d)
        
        if best_move is not None:
            i, j, d = best_move
            a[i] -= d
            a[j] += d
            a = np.maximum(a, 0.0)
            # Don't renormalize here (pair move preserves sum)
            no_improve = 0
            moves_since_fft += 1
            
            if moves_since_fft >= fft_interval:
                conv = fast_convolve(a)
                moves_since_fft = 0
                if conv.max() < best_max:
                    best_max = conv.max()
                    best_a = a.copy()
        else:
            no_improve += 1
            if no_improve > 300:
                # Random restart with small perturbation
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(a)
                no_improve = 0
            elif no_improve > 50:
                # Try random (i,j) pairs
                for _ in range(10):
                    idx = np.where(a > 1e-8)[0]
                    if len(idx) < 2:
                        break
                    i = int(rng.choice(idx))
                    j = int(rng.choice(idx))
                    if i == j:
                        continue
                    d = 0.001
                    if a[i] < d:
                        continue
                    # Quick check at peak
                    pi_peak = a[k_peak - i] if 0 <= k_peak - i < n else 0
                    pj_peak = a[k_peak - j] if 0 <= k_peak - j < n else 0
                    if pi_peak <= pj_peak:
                        continue
                    # Estimate improvement
                    est = 2 * d * (pj_peak - pi_peak)
                    if est > -0.01 * d:
                        continue
                    a[i] -= d
                    a[j] += d
                    a = np.maximum(a, 0.0)
                    moves_since_fft += 1
                    if moves_since_fft >= fft_interval:
                        conv = fast_convolve(a)
                        moves_since_fft = 0
                    break
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    pair_cd it={it}, max={best_max:.10f}, no_imp={no_improve}, elapsed={elapsed:.0f}s")
    
    # Final FFT
    conv = fast_convolve(a)
    if conv.max() < best_max:
        best_max = conv.max()
        best_a = a.copy()
    
    return best_a, best_max


def smoothed_grad_descent(a, time_limit, T_init=0.001, lr_init=0.02, rng=None, verbose=False):
    """
    Smoothed gradient descent using log-sum-exp surrogate for max.
    Minimizes T*log(sum(exp(conv/T))) as a smooth approximation of max(conv).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    
    if rng is None:
        rng = np.random.default_rng(123)
    
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_max = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        
        # Log-sum-exp gradient: p_k = exp(conv_k / T) / sum(exp(conv_j / T))
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Gradient of log-sum-exp: grad_logsumexp = 2 * hankel(p, a)
        # where hankel(p,a)[i] = sum_k p[k] * a[k-i] (with bounds)
        # This is 2 * correlate(p, a, 'valid')... let me compute it properly
        # grad[i] = 2 * sum_k p[k] * a[k-i] (if valid) - correction for self terms
        # Actually: d/d a_i [sum_k p_k * conv_k] = 2 * sum_k p_k * a[k-i] (if k-i valid)
        # But we also need to account for p depending on a... for the smoothed version,
        # we treat p as fixed (first-order approximation)
        
        # Compute gradient: grad[i] = 2 * sum_{k: k-i valid} p[k] * a[k-i]
        # = 2 * (p * a_flipped) at appropriate shifts
        # More precisely: grad[i] = 2 * sum_{j=0}^{n-1} a[j] * p[i+j] (if i+j < 2n-1)
        # This is a cross-correlation
        
        # Using the formula: grad[i] = 2 * sum_j a[j] * p[i+j]
        # where p[i+j] = 0 if i+j >= 2n-1
        
        # We can compute this as: for each j, add 2*a[j]*p[j:j+n] to grad
        # This is O(n^2) which might be slow for large n
        # Alternative: use FFT
        
        # Actually, grad[i] = 2 * sum_j a[j] * p[i+j] = 2 * (cross_correlate)(a, reversed_p)[i]
        # Let's use a simpler O(n^2) approach for moderate n, or FFT-based
        
        if n <= 2000:
            # O(n^2) but with numpy vectorization
            # grad[i] = 2 * sum_j a[j] * p[i+j]
            # Build matrix P where P[j, i] = p[i+j] (if i+j < 2n-1)
            # Then grad = 2 * P.T @ a
            # But this is O(n^2) memory. For n=2000, that's 4M * 8 = 32MB. OK.
            
            # Actually, let's use the structure: 
            # grad[i] = 2 * sum_j a[j] * p[i+j]
            # = 2 * sum_j a[j] * p[i+j]
            # This is a "partial convolution" or "Hankel" operation
            
            # We can write it as: grad = 2 * H @ a where H[i,j] = p[i+j]
            # H is a Hankel matrix. We can compute H @ a via:
            # (H @ a)[i] = sum_j H[i,j] * a[j] = sum_j p[i+j] * a[j]
            # = sum_{k=i}^{i+n-1} p[k] * a[k-i] (with bounds)
            # = (correlate)(p, a, mode='full')[n-1 : n-1+n]  ... hmm not exactly
            
            # Let me just do it directly with FFT:
            # grad[i] = sum_j a[j] * p[i+j]
            # Let q = reversed(a), then:
            # sum_j a[j] * p[i+j] = sum_j q[n-1-j] * p[i+j]
            # = (q * p)[n-1+i] (convolution)
            # where * denotes linear convolution
            
            q = a[::-1]
            size = n + len(p) - 1
            fft_size = 1 << (size - 1).bit_length()
            Q = np.fft.rfft(q, fft_size)
            P = np.fft.rfft(p, fft_size)
            conv_qp = np.fft.irfft(Q * P, fft_size)[:size]
            # (q * p)[n-1+i] for i = 0, ..., n-1
            grad = 2.0 * conv_qp[n-1:n-1+n]
        else:
            # Use sparse approach
            grad = np.zeros(n)
            for j in range(n):
                if a[j] > 1e-15:
                    grad += 2.0 * a[j] * p[j:j+n]
        
        # Projected gradient step
        a_new = project_simplex(a - lr * grad)
        
        new_max = fast_convolve(a_new).max()
        if new_max < best_max:
            best_max = new_max
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            # Try smaller step
            a_half = project_simplex(a - lr * 0.5 * grad)
            new_max_half = fast_convolve(a_half).max()
            if new_max_half < best_max:
                best_max = new_max_half
                best_a = a_half.copy()
                a = a_half
                no_improve = 0
            else:
                lr *= 0.9
        
        # Adapt temperature
        if no_improve > 20:
            T *= 1.2
            lr *= 0.95
        if no_improve > 100:
            T *= 0.8  # Re-focus
        if no_improve > 300:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            T = T_init
            lr = lr_init
            no_improve = 0
        
        if verbose and it % 200 == 0:
            print(f"    smooth_grad it={it}, max={best_max:.10f}, T={T:.6f}")
    
    return best_a, best_max


def lp_iteration(a, time_limit, K=30, verbose=False):
    """
    LP-based iteration: linearize the quadratic constraints and solve LP.
    Re-linearize at the new solution.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return a, fast_convolve(a).max()
    
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_max = fast_convolve(a).max()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        
        conv = fast_convolve(a)
        cmax = conv.max()
        
        # Select active constraints (top K by value)
        sorted_k = np.argsort(conv)[::-1]
        active_k = sorted_k[:K]
        
        # Build gradient for each active k
        # grad_k[m] = (2 - I[k==2*m]) * a[k-m] if 0 <= k-m < n
        grads = np.zeros((K, n))
        for ik, k in enumerate(active_k):
            for m in range(n):
                idx = k - m
                if 0 <= idx < n:
                    factor = 2.0
                    if k == 2 * m:
                        factor = 1.0
                    grads[ik, m] = factor * a[idx]
        
        # Solve LP:
        # min t
        # s.t. grads @ a_var <= t - conv[active_k] + grads @ a  (for each active k)
        #      sum(a_var) = 1
        #      a_var >= 0
        
        a_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1, a_var >= 0]
        
        for ik in range(K):
            k = active_k[ik]
            rhs_offset = -conv[k] + np.dot(grads[ik], a)
            constraints.append(np.dot(grads[ik], a_var) <= t_var + rhs_offset)
        
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=10000)
        except Exception:
            try:
                prob.solve(solver=cp.CLARABEL)
            except Exception:
                try:
                    prob.solve(solver=cp.SCS)
                except Exception:
                    break
        
        if a_var.value is None:
            break
        
        a_new = np.maximum(a_var.value, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        new_max = fast_convolve(a_new).max()
        
        if new_max < best_max:
            best_max = new_max
            best_a = a_new.copy()
            a = a_new
        else:
            # Try blend
            alpha = 0.5
            a_blend = (1-alpha) * best_a + alpha * a_new
            a_blend = a_blend / a_blend.sum()
            blend_max = fast_convolve(a_blend).max()
            if blend_max < best_max:
                best_max = blend_max
                best_a = a_blend.copy()
                a = a_blend
            else:
                # No improvement, try smaller step
                alpha = 0.1
                a_blend = (1-alpha) * best_a + alpha * a_new
                a_blend = a_blend / a_blend.sum()
                blend_max = fast_convolve(a_blend).max()
                if blend_max < best_max:
                    best_max = blend_max
                    best_a = a_blend.copy()
                    a = a_blend
                else:
                    break
        
        if verbose and it % 5 == 0:
            print(f"    lp_iter it={it}, max={best_max:.10f}")
    
    return best_a, best_max


def structured_init(n, rng=None, parent=None):
    """Generate structured initializations."""
    if rng is None:
        rng = np.random.default_rng()
    
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # If parent available, interpolate
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())
    
    # Symmetric bump (Gaussian)
    x = np.arange(n, dtype=np.float64)
    g = np.exp(-0.5 * ((x - 0.5*(n-1)) / (0.2*n))**2)
    inits.append(g / g.sum())
    
    # Two bumps
    g2 = (np.exp(-0.5*((x-0.25*(n-1))/(0.1*n))**2) + 
          np.exp(-0.5*((x-0.75*(n-1))/(0.1*n))**2))
    inits.append(g2 / g2.sum())
    
    # Cosine window
    c = 0.5 + 0.5*np.cos(2*np.pi*x/n)
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Symmetric random
    half = n // 2
    h = rng.dirichlet(np.ones(half) * 0.5)
    sym = np.zeros(n)
    sym[:half] = h
    sym[half:] = h[::-1] if n % 2 == 0 else h[:n-half][::-1]
    if n % 2 == 1:
        sym[half] = h[-1] * 0.5  # center
    sym = np.maximum(sym, 0)
    if sym.sum() > 1e-10:
        inits.append(sym / sym.sum())
    
    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search for minimizing
    2*n*max(conv(a,a))/(sum(a))^2 over nonnegative sequences.
    
    Uses pair coordinate descent (exact 1D subproblems), smoothed gradient
    (log-sum-exp), and LP refinement, with multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(512) / 512
    
    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")
    
    # === Phase 1: Multi-n exploration with pair_cd (~300s) ===
    print("\n=== Phase 1: Multi-n exploration ===")
    t1 = time.monotonic()
    
    # Choose n values to explore
    candidate_ns = sorted(set([
        n_parent,
        max(128, n_parent - 128),
        n_parent + 128,
        n_parent + 256,
        512, 1024, 1536, 2048,
    ]))
    # Filter to reasonable range
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 4096]
    
    phase1_time = 300
    time_per_n = max(15, phase1_time // len(candidate_ns))
    
    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 800
        if remaining < 20:
            break
        
        t_alloc = min(time_per_n, remaining * 0.8)
        
        # Generate a few initializations and pick the best
        inits = structured_init(n_try, rng, parent)
        best_init = None
        best_init_max = float('inf')
        for init in inits:
            c = fast_convolve(init).max()
            if c < best_init_max:
                best_init_max = c
                best_init = init
        
        print(f"  pair_cd n={n_try}, t={t_alloc:.0f}s, init_max={best_init_max:.6f}")
        a_opt, opt_max = pair_cd(best_init, time_limit=t_alloc, rng=rng)
        try_seq(a_opt, f"paircd_n{n_try}")
    
    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")
    
    # === Phase 2: Deep pair_cd on best (~500s) ===
    print("\n=== Phase 2: Deep pair coordinate descent ===")
    t2 = time.monotonic()
    
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 500
        if remaining > 20:
            t_alloc = min(remaining, 500)
            print(f"  Deep pair_cd, t={t_alloc:.0f}s")
            a_opt, _ = pair_cd(best_seq, time_limit=t_alloc, rng=rng, verbose=True)
            try_seq(a_opt, "deep_paircd")
