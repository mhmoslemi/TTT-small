import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]."""
    return np.correlate(p, a, mode='valid')

    # Get top n_constraints peaks

    return None

- Truncated Frank-Wolfe (top-K peak constraints) for efficient global moves

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]. Uses correlate for speed."""
    return np.correlate(p, a, mode='valid')


def compute_grad_smoothed(a, T):
    """Compute smoothed gradient: 2 * hankel_mv(p_T, a) where p_T = softmax(conv/T)."""
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()
    grad = 2.0 * hankel_mv(p, a)
    return grad, conv.max()


def nested_accelerated_smoothed(a0, time_limit, T_init=0.002, T_min=0.0001,
                                step_init=0.05, decay=0.9995):
    """Nesterov-accelerated smoothed gradient descent with temperature annealing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    y = a.copy()
    y_prev = a.copy()
    t_mom = 1.0
    T = T_init
    step = step_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        grad, _ = compute_grad_smoothed(y, T)

        momentum = (t_mom - 1.0) / (t_mom + 1.0)
        y_new = y + momentum * (y - y_prev) - step * grad
        y_prev = y
        y = project_simplex(y_new)
        y = np.maximum(y, 0.0)
        y = y / y.sum()
        t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0

        obj = fast_convolve(y).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        step *= decay
        if no_improve > 50:
            T = max(T * 0.7, T_min)
            step = step_init * 0.5
            if no_improve > 200:
                y = best_a.copy()
                y_prev = best_a.copy()
                t_mom = 1.0
                T = max(T_init * 0.3, T_min)
                step = step_init
                no_improve = 0

        if T < T_min * 1.01:
            T = T_min

    return best_a, best_obj


def truncated_fw(f0, time_limit, n_constraints=30, m_relax=1.0):
    """Frank-Wolfe with truncated LP oracle (top-K constraints only)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        remaining_per_lp = 3.0
        g = solve_truncated_lp(f, M * m_relax, n_constraints, remaining_per_lp)
        if g is None:
            m_relax = min(m_relax * 1.2, 4.0)
            if m_relax > 3.5:
                m_relax = 1.0
                n_constraints = min(n_constraints + 10, 100)
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            m_relax = min(m_relax * 1.1, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in np.linspace(0.05, 1.0, 15):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.15, 5.0)

        if no_improve > 60:
            break

    return best_f, best_obj


def solve_truncated_lp(f, M, n_constraints, time_limit):
    """Solve truncated LP: max sum(g) s.t. (f*g)[k] <= M for top-k peaks, g >= 0."""
    n = len(f)
    conv_f = fast_convolve(f)
    peaks = np.argsort(conv_f)[::-1][:n_constraints]

    A_ub = np.zeros((len(peaks), n))
    for i, k in enumerate(peaks):
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        indices = np.arange(m_start, m_end + 1)
        A_ub[i, indices] = f[k - indices]

    c = -np.ones(n)
    b_ub = np.full(len(peaks), M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def coordinate_descent_peak(a0, time_limit):
    """Coordinate descent: transfer mass from peak-contributing to low positions."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stale = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)

        if len(affected) == 0:
            break

        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]

        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            break
        j_min = outside[np.argmin(a[outside])]

        improved = False
        for delta in [0.0002, 0.0005, 0.001, 0.002, 0.005]:
            if a[j_max] < delta:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min] += delta
            candidate = np.maximum(candidate, 0.0)
            candidate = candidate / candidate.sum()
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        if improved:
            stale = 0
        else:
            stale += 1
            if stale % 20 == 0:
                rng = np.random.default_rng(it * 7919 + 13)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.0008)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    stale = 0
                elif stale > 300:
                    break

    return best_a, best_obj


def multi_subgradient(a0, time_limit, top_k=20, lr_init=0.003, mom_init=0.93):
    """Multi-peak subgradient with momentum and adaptive learning rate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = mom_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]

        v = mom * v - lr * sg
        a = project_simplex(a + v)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9993
        mom = min(0.98, mom + 0.00005)

        if no_improve > 150:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.001 + 0.002 * min(1.0, (no_improve - 150) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init * 0.7
            mom = mom_init
            no_improve = 0

    return best_a, best_obj


def generate_structured_init(n, n_parent, parent, rng, idx):
    """Generate diverse structured initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Resampled parent
    if parent is not None:
        p_resampled = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        p_resampled = np.maximum(p_resampled, 0)
        ps = p_resampled.sum()
        if ps > 1e-10:
            inits.append(p_resampled / ps)

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge-weighted (high at edges, low in middle)
    edge_w = 1.0 / (np.abs(x - 0.5 * (n - 1)) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Chirp-like (real part of quadratic phase, shifted to be positive)
    chirp = np.cos(np.pi * (x / n) ** 2)
    chirp = np.maximum(chirp, 0)
    cs = chirp.sum()
    if cs > 1e-10:
        inits.append(chirp / cs)

    # Cosine window
    cw = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cw = np.maximum(cw, 0.01)
    inits.append(cw / cw.sum())

    # Periodic pattern
    for freq in [3, 5, 8]:
        h = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * x / n)
        h = np.maximum(h, 0.01)
        inits.append(h / h.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Perturbed parent
    if parent is not None:
        p_resampled = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        p_resampled = np.maximum(p_resampled, 0)
        ps = p_resampled.sum()
        if ps > 1e-10:
            p_norm = p_resampled / ps
            perturbed = p_norm + rng.standard_normal(n) * 0.01
            perturbed = np.maximum(perturbed, 0)
            ps2 = perturbed.sum()
            if ps2 > 1e-10:
                inits.append(perturbed / ps2)

    return inits[idx % len(inits)]

                try_and_update(a_opt, state, "perturb_{}_{}".format(label, r))

                remaining = deadline - time.monotonic() - 150
                if remaining > 10 and state[0] < 1.505:
                    a_opt2, _ = coordinate_descent_peak(a_opt, time_limit=min(remaining, 15))
                    try_and_update(a_opt2, state, "cd_{}_{}".format(label, r))

        # Structured restarts (different initialization patterns)
        for idx in [2, 3, 4, 5, 6]:
            remaining = deadline - time.monotonic() - 100
            if remaining < 15:
                break

            inits = generate_structured_init(n_best, n_best, state[1], rng, idx)
            start = inits[-1]  # Use perturbed parent or structured
            t_alloc = min(remaining, 30)
            a_opt, _ = nested_accelerated_smoothed(start, time_limit=t_alloc,
                                                   T_init=0.003, step_init=0.04)
            try_and_update(a_opt, state, "struct_{}".format(idx))

    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(
        state[0], time.monotonic() - t_phase))

    # === Phase 4: Final polish (~250s) ===
    print("\n=== Phase 4: Final Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])

        # Coordinate descent for surgical peak reduction
        remaining = deadline - time.monotonic() - 80
        if remaining > 10:
            t_alloc = min(remaining, 60)
            print("  Coordinate descent, t={:.0f}s".format(t_alloc))
            a_opt, _ = coordinate_descent_peak(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "coord_desc")

        # Fine smoothed gradient (very low temperature)
        remaining = deadline - time.monotonic() - 50
        if remaining > 10:
            t_alloc = min(remaining, 40)
            print("  Fine smooth, t={:.0f}s".format(t_alloc))
            a_opt, _ = nested_accelerated_smoothed(state[1], time_limit=t_alloc,
                                                   T_init=0.0003, T_min=0.00005,
                                                   step_init=0.01, decay=0.9998)
            try_and_update(a_opt, state, "fine_smooth")

        # Truncated FW with many constraints (near-exact)
        remaining = deadline - time.monotonic() - 30
        if remaining > 10:
            t_alloc = min(remaining, 30)
            print("  Final TFK, t={:.0f}s".format(t_alloc))
            a_opt, _ = truncated_fw(state[1], time_limit=t_alloc, n_constraints=80)
            try_and_update(a_opt, state, "final_tfw")

        # Tiny perturbations with immediate coordinate descent
        remaining = deadline - time.monotonic() - 15
        if remaining > 5:
            for _ in range(5):
                remaining = deadline - time.monotonic() - 5
                if remaining < 4:
                    break
                noise = rng.standard_normal(n_best) * 0.0003
                f_pert = project_simplex(state[1] + noise)
                a_opt, _ = coordinate_descent_peak(f_pert, time_limit=min(remaining, 4))
                try_and_update(a_opt, state, "tiny_pert")

    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(
        state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(768) / 768
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
