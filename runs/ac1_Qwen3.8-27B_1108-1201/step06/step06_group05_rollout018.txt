def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_gradient(a, p):
    """Compute gradient of log-sum-exp objective.
    g_j = 2 * sum_k p_k * a_{k-j}

    n = len(a)

    P = np.fft.rfft(p, N)

    corr = np.fft.irfft(P * np.conj(A), N)

    return 2.0 * corr[:n]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_gradient(a, p):
    """Compute gradient: g_j = 2 * sum_k p_k * a_{k-j}."""
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(P * np.conj(A), N)
    return 2.0 * corr[:n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def enforce_symmetry(a):

def solve_lp_oracle(f, M, method='highs'):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method,
                      options={'presolve': True, 'max_time': 20})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def active_set_newton(a0, time_limit, tol=1e-10, max_iter=5000):

    a = np.maximum(a0.astype(np.float64), 0.0)

        M = conv.max()

        if len(support) == 0:
            break

        active_peaks = np.where(conv >= threshold)[0]

        if len(active_peaks) <= 1:
            # Single peak: use subgradient
            k = active_peaks[0]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)

            sg[idx_arr] = a[k - idx_arr]

                candidate = project_simplex(a - step * sg)
                c = fast_convolve(candidate)
                if c.max() < M - 1e-16:
                    a = candidate
                    if a.max() < best_obj - 1e-14:
                        best_obj = a.max()
                        best_a = a.copy()
                    break
            else:
                break
            continue

        # Build KKT system: for j in support, sum_k mu_k * 2*a[k-j] = lambda
        m = len(support)
        p = len(active_peaks)
        A_sys = np.zeros((m, p + 1))
        for i, j in enumerate(support):
            A_sys[i, -1] = 1.0
            for k_idx, k in enumerate(active_peaks):
                idx = k - j
                if 0 <= idx < n:
                    A_sys[i, k_idx] = 2.0 * a[idx]

        try:
            U, S_vals, Vt = np.linalg.svd(A_sys)
            sol = Vt[-1, :]
            mu_vals = np.maximum(sol[:p], 0)

            if mu_sum < 1e-15:

            mu_vals /= mu_sum

            break

            grad[idx_arr] -= mu_vals[k_idx] * 2.0 * a[k - idx_arr]

def coordinate_refine(a0, time_limit, max_sweeps=100):

                # Actually simpler: set a[j] = new_x, then scale others to make sum=1

def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s

    remaining = deadline - time.monotonic() - 1200
    if remaining > 20:
        t_alloc = min(remaining, 80)
        a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc, beta_init=5.0, beta_max=500.0, eta_init=0.1, use_symmetry=True)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "md_sym")

    remaining = deadline - time.monotonic() - 1100
    if remaining > 20:
        t_alloc = min(remaining, 60)
        a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc, beta_init=5.0, beta_max=500.0, eta_init=0.1, use_symmetry=False)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "md_asym")

    remaining = deadline - time.monotonic() - 1000
    if remaining > 15:
        t_alloc = min(remaining, 50)
        a_opt, _ = pgd_momentum(best_seq, time_limit=t_alloc, top_k=20)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "pgd_quick")

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # === Phase 2: Multi-n exploration (~400s) ===
    print("\n=== Phase 2: Multi-n Exploration ===")
    t_start = time.monotonic()

    n_best = len(best_seq)
    candidate_ns = sorted(set([n_best, n_best // 2, n_best * 2, 384, 512, 640, 768, 896, 1024, 1152]))
    candidate_ns = [n for n in candidate_ns if 192 <= n <= 2048]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 600
        if remaining < 40:
            break

        if n_try == n_best and (best_seq is not None):
            start = best_seq.copy()
        else:
            start = interp_sequence(best_seq, n_try)
            # Also try symmetric version
            start_sym = enforce_symmetry(start)

        t_alloc = min(remaining, 100)
        print(f"  Exploring n={n_try}, t={t_alloc:.0f}s")

        # Mirror descent symmetric
        t1 = min(t_alloc * 0.4, remaining - 100)
        if t1 > 5:
            a_opt, _ = mirror_descent(start, time_limit=t1, beta_init=3.0, beta_max=300.0, use_symmetry=True)
            (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"md_sym_n{n_try}")

        # Mirror descent asymmetric
        remaining = deadline - time.monotonic() - 500
        t2 = min(t_alloc * 0.3, remaining - 80)
        if t2 > 5:
            a_opt, _ = mirror_descent(a_opt if 'a_opt' in dir() else start, time_limit=t2, beta_init=3.0, beta_max=300.0, use_symmetry=False)
            (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"md_asym_n{n_try}")

        # PGD
        remaining = deadline - time.monotonic() - 450
        t3 = min(t_alloc * 0.2, remaining - 50)
        if t3 > 5:
            a_opt, _ = pgd_momentum(a_opt if 'a_opt' in dir() else start, time_limit=t3, top_k=15)
            (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"pgd_n{n_try}")

    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # === Phase 3: Deep mirror descent with increasing beta (~300s) ===
    print("\n=== Phase 3: Deep Mirror Descent ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    # High beta mirror descent
    remaining = deadline - time.monotonic() - 400
    if remaining > 30:
        t_alloc = min(remaining, 150)
        print(f"  High-beta MD, n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc, beta_init=50.0, beta_max=2000.0,
                                  eta_init=0.05, use_symmetry=True, top_k_smooth=30)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "md_high_beta")

    # Top-k focused
    remaining = deadline - time.monotonic() - 300
    if remaining > 20:
        t_alloc = min(remaining, 100)
        a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc, beta_init=100.0, beta_max=5000.0,
                                  eta_init=0.03, use_symmetry=False, top_k_smooth=20)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "md_topk")

    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # === Phase 4: FW-LP global moves (~300s) ===
    print("\n=== Phase 4: Frank-Wolfe LP ===")
    t_start = time.monotonic()

    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 300
    if remaining > 30:
        t_alloc = min(remaining, 120)
        print(f"  FW-LP, n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "fw_lp")

    # Perturbed FW restarts
    for scale in [0.005, 0.01, 0.02]:
        remaining = deadline - time.monotonic() - 200
        if remaining < 20:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc = min(remaining, 60)
        a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"fw_pert_{scale}")

    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Active-set Newton + Coordinate refinement (~250s) ===
    print("\n=== Phase 5: Newton + Coordinate ===")
    t_start = time.monotonic()

    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 150
    if remaining > 15:
        t_alloc = min(remaining, 100)
        print(f"  Active-set Newton, t={t_alloc:.0f}s")
        a_opt, _ = active_set_newton(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "as_newton")

    # Coordinate refinement
    remaining = deadline - time.monotonic() - 100
    if remaining > 10:
        t_alloc = min(remaining, 80)
        a_opt, _ = coordinate_refine(best_seq, time_limit=t_alloc, max_sweeps=50)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "coord_refine")

    # Another Newton pass
    remaining = deadline - time.monotonic() - 60
    if remaining > 10:
        t_alloc = min(remaining, 50)
        a_opt, _ = active_set_newton(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "as_newton2")

    print(f"  Phase 5 best: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # === Phase 6: Final intensive polishing (~150s) ===
    print("\n=== Phase 6: Final Polish ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    # Multiple fine mirror descent runs
    for beta_init, eta_init, sym in [(200, 0.02, True), (500, 0.01, True), (100, 0.03, False)]:
        remaining = deadline - time.monotonic() - 50
        if remaining < 10:
            break
        t_alloc = min(remaining, 40)
        a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc, beta_init=beta_init,
                                  beta_max=beta_init * 10, eta_init=eta_init, use_symmetry=sym)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"polish_b{beta_init}_s{sym}")

    # Tiny perturbations with quick Newton
    for _ in range(3):
        remaining = deadline - time.monotonic() - 30
        if remaining < 8:
            break
        noise = rng.standard_normal(n_best) * 0.0005
        f_pert = project_simplex(best_seq + noise)
        t_alloc = min(remaining, 20)
        a_opt, _ = active_set_newton(f_pert, time_limit=t_alloc, max_iter=1000)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "perturb_newton")

    # Symmetry final check
    sym_final = enforce_symmetry(best_seq)
    (best_seq, best_score) = try_and_update(sym_final, (best_seq, best_score), "final_sym")

    # One last mirror descent
    remaining = deadline - time.monotonic() - 10
    if remaining > 5:
        a_opt, _ = mirror_descent(best_seq, time_limit=min(remaining, 10),
                                  beta_init=500, beta_max=5000, eta_init=0.01, use_symmetry=True)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "final_md")

    print(f"  Phase 6 best: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
