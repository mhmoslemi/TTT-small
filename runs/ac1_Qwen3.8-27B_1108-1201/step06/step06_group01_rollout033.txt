"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS on smoothed objective
with continuation, combined with projected subgradient and coordinate descent.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad_a(a, p):
    """Compute grad_a = 2 * sum_k p_k * (a_{i+k} + a_{i-k}) for each i."""
    n = len(a)
    grad_a = np.zeros(n)
    for k in range(n):
        if p[k] < 1e-15:
            continue
        pk = p[k]
        grad_a[:n-k] += pk * a[k:]
        grad_a[k:] += pk * a[:n-k]
    grad_a *= 2.0
    return grad_a

def smoothed_obj_grad(x, T):
    """Compute F_T(softmax(x)) and its gradient w.r.t. x."""
    a = softmax(x)
    n = len(a)
    conv = fast_convolve(a)
    r = conv[:n]  # k = 0, ..., n-1 (by symmetry)
    
    # Log-sum-exp
    r_max = r.max()
    exp_T = np.exp(T * (r - r_max))
    log_sum = r_max + np.log(exp_T.sum())
    F = log_sum / T
    
    p = exp_T / exp_T.sum()
    
    grad_a = compute_grad_a(a, p)
    dot_ga_a = np.dot(a, grad_a)
    grad_x = a * grad_a - a * dot_ga_a
    
    return F, grad_x

def lbfgs_smoothed(a0, T_list, max_iter_per_T=200):
    """L-BFGS with continuation on T."""
    n = len(a0)
    x = np.log(a0 + 1e-30)  # Initialize log-parameters
    best_a = a0.copy()
    best_obj = fast_convolve(a0).max()
    
    for T in T_list:
        x_copy = x.copy()
        
        # scipy minimize with L-BFGS-B
        def fun(x_inner):
            val, _ = smoothed_obj_grad(x_inner, T)
            return val
        
        def grad(x_inner):
            _, g = smoothed_obj_grad(x_inner, T)
            return g
        
        try:
            res = minimize(fun, x_copy, jac=grad, method='L-BFGS-B',
                         options={'maxiter': max_iter_per_T, 'gtol': 1e-8})
            if res.success or res.fun < fun(x_copy):
                x = res.x
        except Exception:
            pass
        
        a = softmax(x)
        obj = fast_convolve(a).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj

def adam_subgradient(a0, time_limit, lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8, top_k=20):
    """Adam optimizer on the simplex with top-k subgradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(a)
        r = conv[:n]
        
        # Top-k subgradient
        if top_k >= n:
            peaks = np.arange(n)
            weights = np.ones(n) / n
        else:
            peaks = np.argpartition(r, -top_k)[-top_k:]
            w = 1.0 / (np.arange(top_k) + 1) ** 0.5
            weights_dict = dict(zip(peaks, w / w.sum()))
        
        grad = np.zeros(n)
        if top_k >= n:
            p = np.ones(n) / n
        else:
            p = np.zeros(n)
            for k, w in weights_dict.items():
                p[k] = w
        
        for k in range(n):
            if p[k] < 1e-15:
                continue
            pk = p[k]
            grad[:n-k] += pk * a[k:]
            grad[k:] += pk * a[:n-k]
        grad *= 2.0
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj

- Standard FW: the linear oracle minimizes $\nabla f(a^{(t)}) \cdot a$ over the simplex
- My approach: the LP minimizes $t$ subject to the linearized constraints

These are related but different. My approach should give a "smarter" direction because it considers all constraints simultaneously.

This is a minimax LP, which is exactly what I wrote above. The solution gives a point that balances all the linearized autocorrelation values.

def project_simplex(v):

    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n

def compute_autocorr_weights(a):
    """Compute g_{k,i} = a_{i+k}*1[i+k<n] + a_{i-k}*1[i>=k] for all k, i."""
    n = len(a)
    # g[k, i] = a[i+k] if i+k < n else 0, plus a[i-k] if i >= k else 0
    # Shape: (n, n)
    g = np.zeros((n, n))
    for k in range(n):
        g[k, :n-k] += a[k:]      # a[i+k] for i = 0, ..., n-1-k
        g[k, k:] += a[:n-k]      # a[i-k] for i = k, ..., n-1

            break

        conv = fast_convolve(a)
        r = conv[:n]  # k = 0, ..., n-1

        # g[k, i] = a[i+k]*1[i+k<n] + a[i-k]*1[i>=k]
        # This is the "Hankel" matrix
        # Actually, let's build the constraint matrix for the LP
        
        # LP: min t
        # s.t. sum_i g[k,i] * a_i - t <= r[k] for k = 0, ..., n-1
        #      sum_i a_i = 1
        #      a_i >= 0

        # For each k, the row is [g[k,0], g[k,1], ..., g[k,n-1], -1]
        # RHS is r[k]

        # g[k, i] for fixed k: first part is a[k:] shifted to positions 0..n-1-k
        #                     second part is a[:n-k] shifted to positions k..n-1

            A_ub[k, k:n] += a[:n-k]      # g[k, i] second part for i = k, ..., n-1
            A_ub[k, n] = -1.0
            b_ub[k] = r[k]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(None, None)]
        c = np.zeros(n + 1)
        c[n] = 1.0  # minimize t

            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': 5, 'presolve': True})
            if not res.success:
                break
            a_lp = np.maximum(res.x[:n], 0.0)
            s = a_lp.sum()
            if s < 1e-15:
                break
            a_lp = a_lp / s
        except Exception:
            break

        best_val = best_obj
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:

            a_try = np.maximum(a_try, 0.0)

            val = fast_convolve(a_try).max()

                best_alpha = alpha

                best_a = a.copy()

                # Convergence
                break
        else:
            # No improvement, try smaller step
            break
    
    return best_a, best_obj

def adam_subgradient(a0, time_limit, lr=0.002, top_k=15, momentum=0.9):
    """Adam-style subgradient on the simplex."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(a)
        r = conv[:n]

        # Top-k weighted subgradient
        k_sel = min(top_k, n)
        peaks = np.argpartition(r, -k_sel)[-k_sel:]
        p = np.zeros(n)
        for rank, k in enumerate(peaks):
            p[k] = 1.0 / (rank + 1) ** 1.0
        p /= p.sum()

        grad = np.zeros(n)
        for k in range(n):
            if p[k] < 1e-15:
                continue
            grad[:n - k] += p[k] * a[k:]
            grad[k:] += p[k] * a[:n - k]
        grad *= 2.0

        # Adam update
        b1, b2, eps = 0.9, 0.999, 1e-8
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        if no_improve > 100:
            rng = np.random.default_rng(t * 31 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0

    return best_a, best_obj

                    # Try moving mass from larger to smaller (or vice versa)
                    for frac in [0.05, 0.15, 0.3]:
                        if a[i1] > a[i2]:
                            delta = a[i1] * frac
                            a_test = a.copy()
                            a_test[i1] -= delta
                            a_test[i2] += delta
                        else:
                            delta = a[i2] * frac
                            a_test = a.copy()
                            a_test[i2] -= delta
                            a_test[i1] += delta
                        a_test = np.maximum(a_test, 0.0)
                        st = a_test.sum()
                        if st < 1e-15:
                            continue
                        a_test = a_test / st
                        c_new = fast_convolve(a_test)
                        gain = best_obj - c_new.max()
                        if gain > best_gain:
                            best_gain = gain
                            best_move = (a_test.copy(), i1, i2, frac)

            if best_move is not None and best_gain > 1e-13:
                a = best_move[0]
                best_obj = best_move[1] if False else fast_convolve(a).max()
                best_a = a.copy()
                improved = True
                stall = 0
            else:
                stall += 1

        if not improved:
            stall += 1
        if stall > 30:
            rng = np.random.default_rng((it * 48271 + 1) % (2 ** 31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            stall = 0
        if stall > 300:
            break

    return best_a, best_obj


def fine_polish(a0, time_limit):
    """Very fine random search polish."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    scales = [0.0001, 0.0003, 0.0005, 0.001, 0.0002]

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        scale = scales[it % len(scales)]
        rng = np.random.default_rng((it * 2654435761 + 1) % (2 ** 32))

        for _ in range(20):
            if time.monotonic() - t0 > time_limit - 0.05:
                break
            noise = rng.standard_normal(n) * scale
            # Biased: reduce values contributing to peak
            conv = fast_convolve(a)
            pk = int(np.argmax(conv))
            js = max(0, pk)
            je = min(n, pk + 1)
            for j in range(js, je):
                partner = pk - j
                if 0 <= partner < n:
                    noise[j] -= scale * a[partner] * 2.0

            candidate = project_simplex(a + noise)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid search combining L-BFGS, minimax LP,
    Adam subgradient, coordinate descent, and fine polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512
        try_seq(parent, "uniform")

    # === Phase 1: L-BFGS on parent ===
    print("\n=== Phase 1: L-BFGS Continuation ===")
    t_ph = time.monotonic()
    a_opt, _ = lbfgs_continuation(parent, time_limit=60)
    try_seq(a_opt, "lbfgs")
    print(f"  Phase 1 done ({time.monotonic()-t_ph:.0f}s), best={best_score:.8f}")

    # === Phase 2: Minimax LP (main workhorse) ===
    print("\n=== Phase 2: Minimax LP ===")
    t_ph = time.monotonic()
    if best_seq is not None:
        t_alloc = min(350, deadline - time.monotonic() - 900)
        if t_alloc > 10:
            a_opt, _ = minimax_lp_iterate(best_seq, time_limit=t_alloc, max_iters=1000)
            try_seq(a_opt, "lp")

        # Perturbed LP restarts
        for scale in [0.003, 0.01]:
            t_alloc = min(80, deadline - time.monotonic() - 700)
            if t_alloc < 10:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * scale)
            a_opt, _ = minimax_lp_iterate(f_pert, time_limit=t_alloc, max_iters=300)
            try_seq(a_opt, f"lp_pert{scale}")

    print(f"  Phase 2 done ({time.monotonic()-t_ph:.0f}s), best={best_score:.8f}")

    # === Phase 3: Multi-n exploration ===
    print("\n=== Phase 3: Multi-n Exploration ===")
    t_ph = time.monotonic()
    n_base = len(best_seq) if best_seq is not None else 512
    candidate_ns = sorted(set([n_base, n_base - 64, n_base + 64,
                                384, 512, 640, 768, 896, 1024]))
    candidate_ns = [n for n in candidate_ns if 200 <= n <= 1200 and n != n_base]

    for n_try in candidate_ns[:4]:
        t_alloc = min(60, deadline - time.monotonic() - 500)
        if t_alloc < 15:
            break
        # Resample parent to new n
        start = np.interp(np.linspace(0, 1, n_try),
                          np.linspace(0, 1, n_base), best_seq)
        start = np.maximum(start, 0)
        ss = start.sum()
        if ss > 1e-10:
            start = start / ss
        else:
            start = np.ones(n_try) / n_try

        # Quick L-BFGS
        a_opt, _ = lbfgs_continuation(start, time_limit=20)
        # LP
        t_lp = min(30, deadline - time.monotonic() - 450)
        if t_lp > 5:
            a_opt, _ = minimax_lp_iterate(a_opt, time_limit=t_lp, max_iters=100)
        try_seq(a_opt, f"n{n_try}")

    print(f"  Phase 3 done ({time.monotonic()-t_ph:.0f}s), best={best_score:.8f}")

    # === Phase 4: Adam Subgradient ===
    print("\n=== Phase 4: Adam Subgradient ===")
    t_ph = time.monotonic()
    if best_seq is not None:
        configs = [
            {'lr': 0.003, 'top_k': 10},
            {'lr': 0.001, 'top_k': 20},
            {'lr': 0.005, 'top_k': 5},
        ]
        for ci, cfg in enumerate(configs):
            t_alloc = min(80, deadline - time.monotonic() - 250)
            if t_alloc < 10:
                break
            a_opt, _ = adam_subgradient(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"adam_{ci}")

    print(f"  Phase 4 done ({time.monotonic()-t_ph:.0f}s), best={best_score:.8f}")

    # === Phase 5: Coordinate Descent ===
    print("\n=== Phase 5: Coordinate Descent ===")
    t_ph = time.monotonic()
    if best_seq is not None:
        t_alloc = min(150, deadline - time.monotonic() - 150)
        if t_alloc > 10:
            a_opt, _ = coordinate_descent_2swap(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord")

    print(f"  Phase 5 done ({time.monotonic()-t_ph:.0f}s), best={best_score:.8f}")

    # === Phase 6: LP + Polish ===
    print("\n=== Phase 6: Final LP + Polish ===")
    t_ph = time.monotonic()
    if best_seq is not None:
        # Another LP pass
        t_alloc = min(80, deadline - time.monotonic() - 80)
        if t_alloc > 10:
            a_opt, _ = minimax_lp_iterate(best_seq, time_limit=t_alloc, max_iters=300)
            try_seq(a_opt, "lp_final")

        # Fine polish
        t_alloc = min(60, deadline - time.monotonic() - 40)
        if t_alloc > 5:
            a_opt, _ = fine_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "polish")

        # Final L-BFGS
        t_alloc = min(30, deadline - time.monotonic() - 15)
        if t_alloc > 5:
            a_opt, _ = lbfgs_continuation(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "lbfgs_final")

    print(f"  Phase 6 done ({time.monotonic()-t_ph:.0f}s), best={best_score:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)}, total={time.monotonic()-t_start:.0f}s)")
    return [float(x) for x in best_seq]
