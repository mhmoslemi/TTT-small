def convolve_grad(f, peaks):
    """Compute gradient of max of conv at given peak locations.
    grad[k][i] = 2*f[k-i] for valid i, else 0 (symmetrized).
    """
    n = len(f)
    grad = np.zeros(n)
    for k in peaks:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += 2.0 * f[k - i_range]
    return grad


def project_simplex(v):
    n = len(v)

    """Build sparse matrix for LP constraints (f*g)[k] <= M."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([idx] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    if not rows:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(len(constraint_indices), n))


def solve_lp_direction(f, M, constraint_indices, time_limit=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for given constraints, g >= 0."""
    n = len(f)
    A = build_constraint_matrix(f, constraint_indices)
    if A.shape[0] == 0:
        return np.ones(n) / n
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

            lo = max(0.0, best_alpha - 0.06)
            hi = min(1.0, best_alpha + 0.06)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            for _ in range(8):
                ft1 = (1.0 - c1) * f + c1 * g
                ft1 = np.maximum(ft1, 0.0)
                if ft1.sum() > 1e-15:
                    ft1 /= ft1.sum()
                    v1 = fast_convolve(ft1).max()
                else:
                    v1 = best_val
                ft2 = (1.0 - c2) * f + c2 * g
                ft2 = np.maximum(ft2, 0.0)
                if ft2.sum() > 1e-15:
                    ft2 /= ft2.sum()
                    v2 = fast_convolve(ft2).max()
                else:
                    v2 = best_val
                if v1 < v2:
                    hi = c2
                    c2 = c1
                    v2 = v1
                    c1 = hi - gr * (hi - lo)
                else:
                    lo = c1
                    c1 = c2
                    v1 = v2
                    c2 = lo + gr * (hi - lo)
                mid = (lo + hi) / 2.0
                ftm = (1.0 - mid) * f + mid * g
                ftm = np.maximum(ftm, 0.0)
                if ftm.sum() > 1e-15:
                    ftm /= ftm.sum()
                    vm = fast_convolve(ftm).max()
                    if vm < v1:
                        best_val = vm
                        best_alpha = mid

            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f /= f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

    return best_f, best_obj


def nesterov_subgradient(f0, time_limit, beta_start=20.0, beta_end=8000.0, top_k=50):
    """Nesterov-accelerated subgradient with LSE smoothing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    x = f.copy()  # Nesterov auxiliary
    beta = beta_start
    lr = 3.0 / np.sqrt(n)
    momentum = 0.9
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # Compute subgradient at x
        conv_x = fast_convolve(x)
        # LSE smoothing weights
        bc = beta * conv_x
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        total = exp_bc.sum()
        if total > 1e-300:
            w = exp_bc / total
        else:
            w = np.ones(len(conv_x)) / len(conv_x)

        # Use top-k for efficiency
        k_top = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(conv_x, -k_top)[-k_top:]
        w_peaks = w[peak_idx]
        w_sum = w_peaks.sum()
        if w_sum > 1e-15:
            w_peaks = w_peaks / w_sum
        else:
            w_peaks = np.ones(k_top) / k_top

        # Weighted subgradient
        grad = np.zeros(n)
        for idx, k in enumerate(peak_idx):
            k = int(k)
            wi = w_peaks[idx]
            if wi < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            i_range = np.arange(lo, hi + 1)
            grad[i_range] += wi * 2.0 * x[k - i_range]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_normed = grad / gnorm

        # Nesterov step
        y = project_simplex(x - lr * grad_normed)

        # Update momentum
        mu = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        x_new = y + momentum * (y - f)
        x_new = np.maximum(x_new, 0)
        sn = x_new.sum()
        if sn > 1e-15:
            x_new /= sn
        else:
            x_new = f.copy()

        f = x_new
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.003, 8.0 / np.sqrt(n))
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.9
            if no_improve > 50:
                f = best_f.copy()
                x = f.copy()
                lr = max(3.0 / np.sqrt(n), 0.005)
                beta = min(beta * 2.0, beta_end)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.003

    return best_f, best_obj


def exact_subgradient_polish(f0, time_limit, top_k=5):
    """Polish using exact subgradient on only the top-k peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    lr = 1.0 / np.sqrt(n)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        conv = fast_convolve(f)
        M = conv.max()
        # Get exact top-k peaks
        k_top = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        # Sort by value
        peak_idx = peak_idx[np.argsort(-conv[peak_idx])]

        # Subgradient: equal weights on top peaks
        grad = np.zeros(n)
        for k in peak_idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            i_range = np.arange(lo, hi + 1)
            grad[i_range] += 2.0 * f[k - i_range]
        grad /= len(peak_idx)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_normed = grad / gnorm

        # Backtracking line search
        step = lr
        improved = False
        for _ in range(8):
            f_new = project_simplex(f - step * grad_normed)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-16:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                improved = True
                if no_improve > 5:
                    lr = min(lr * 1.05, 5.0 / np.sqrt(n))
                no_improve = 0
                break
            step *= 0.5

        if not improved:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.85
            if no_improve > 40:
                # Try to move mass away from peak-contributing positions
                k_main = int(peak_idx[0])
                lo = max(0, k_main - n + 1)
                hi = min(n - 1, k_main)
                i_range = np.arange(lo, hi + 1)
                contribs = f[i_range] * f[k_main - i_range]
                j_max = i_range[np.argmax(contribs)]
                # Move small amount from j_max to a random position
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                j_dest = rng.integers(0, n)
                if j_dest != j_max:
                    delta = f[j_max] * 0.05
                    f_trial = f.copy()
                    f_trial[j_max] -= delta
                    f_trial[j_dest] += delta
                    f_trial = np.maximum(f_trial, 0)
                    f_trial /= f_trial.sum()
                    conv_trial = fast_convolve(f_trial)
                    if conv_trial.max() < best_obj:
                        best_obj = conv_trial.max()
                        best_f = f_trial.copy()
                        f = f_trial.copy()
                        no_improve = 0

    return best_f, best_obj


def freq_perturb(f, rng, strength=0.03):
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def structured_init(n, method='sine', alpha=0.3):
    """Generate structured initializations."""
    if method == 'uniform':
        return np.ones(n) / n
    elif method == 'sine':
        i = np.arange(n)
        a = 1.0 + alpha * np.cos(2.0 * np.pi * i / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif method == 'gaussian':
        i = np.arange(n)
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif method == 'triangle':
        i = np.arange(n)
        a = 1.0 - np.abs(2.0 * i / (n - 1) - 1.0)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif method == 'bimodal':
        i = np.arange(n)
        c1 = n * 0.3
        c2 = n * 0.7
        sigma = n * 0.1
        a = np.exp(-0.5 * ((i - c1) / sigma) ** 2) + np.exp(-0.5 * ((i - c2) / sigma) ** 2)
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif method == 'comb':
        i = np.arange(n)
        period = max(4, n // 20)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * i / period)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def try_sequence(a, best_tracker):
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.10f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Multi-length, multi-structure initialization
    Phase 2: FW optimization on best candidates
    Phase 3: Nesterov subgradient refinement
    Phase 4: Exact subgradient polish
    Phase 5: Final micro-optimization
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.10f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    n_parent = len(parent) if parent is not None else 768

    # === Phase 1: Multi-length, multi-structure initialization (30% time) ===
    print("\n=== Phase 1: Multi-length initialization ===")
    total_budget = float(budget_s) - 30.0
    p1_end = time.monotonic() + total_budget * 0.30

    # Candidate lengths around parent length
    lengths = sorted(set([
        n_parent,
        max(128, n_parent - 64),
        min(2048, n_parent + 64),
        max(128, n_parent - 128),
        min(2048, n_parent + 128),
        max(128, int(n_parent * 0.8)),
        min(2048, int(n_parent * 1.2)),
        512, 768, 1024  # common powers of 2
    ]))
    lengths = [l for l in lengths if l >= 64 and l <= 2048]

    # Try structured initializations at a few lengths
    structures = ['sine', 'gaussian', 'bimodal']
    candidates = []

    for nc in lengths[:8]:
        if time.monotonic() > p1_end - 15:
            break
        # Interpolated parent
        if parent is not None and nc != n_parent:
            f_init = interpolate_to_length(parent, nc)
            score = try_sequence(f_init, best_tracker)
            if np.isfinite(score):
                candidates.append((score, f_init, nc, 'interp'))
        elif parent is not None:
            score = try_sequence(parent, best_tracker)
            candidates.append((score, parent.copy(), nc, 'parent'))

        # Structured inits
        for struct in structures:
            f_init = structured_init(nc, method=struct)
            score = try_sequence(f_init, best_tracker)
            if np.isfinite(score):
                candidates.append((score, f_init, nc, struct))

    # Sort by score
    candidates.sort(key=lambda x: x[0])
    print(f"  Top candidates: {[(f'{s:.6f}', n) for s, _, n, _ in candidates[:5]]}")

    # === Phase 2: FW optimization on top candidates (35% time) ===
    print("\n=== Phase 2: FW optimization ===")
    p2_end = p1_end + total_budget * 0.35

    # Optimize top 3-4 distinct (length, method) candidates
    seen_lengths = set()
    optimized = 0
    for score, f_init, nc, method in candidates:
        if time.monotonic() > p2_end - 20:
            break
        if nc in seen_lengths and optimized >= 2:
            continue
        seen_lengths.add(nc)
        t_alloc = min(120, (p2_end - time.monotonic()) * 0.4)
        if t_alloc < 15:
            break
        print(f"  FW: n={nc}, method={method}, t={t_alloc:.0f}s")
        f_opt, _ = fw_optimize(f_init, time_limit=t_alloc, top_k=40, near_frac=1.01)
        try_sequence(f_opt, best_tracker)
        optimized += 1
        if optimized >= 4:
            break

    # === Phase 3: Nesterov subgradient refinement (20% time) ===
    print("\n=== Phase 3: Nesterov subgradient ===")
    p3_end = p2_end + total_budget * 0.20

    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)

    t_alloc = min(180, p3_end - time.monotonic())
    if t_alloc > 20:
        print(f"  Nesterov: n={n_cur}, t={t_alloc:.0f}s")
        f_opt, _ = nesterov_subgradient(current_best, time_limit=t_alloc,
                                         beta_start=20.0, beta_end=10000.0, top_k=60)
        try_sequence(f_opt, best_tracker)

    # === Phase 4: Exact subgradient polish (10% time) ===
    print("\n=== Phase 4: Exact subgradient polish ===")
    p4_end = p3_end + total_budget * 0.10

    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t_alloc = min(150, p4_end - time.monotonic())
    if t_alloc > 15:
        print(f"  Exact subgrad: n={len(current_best)}, t={t_alloc:.0f}s")
        f_opt, _ = exact_subgradient_polish(current_best, time_limit=t_alloc, top_k=5)
        try_sequence(f_opt, best_tracker)

    # === Phase 5: Final micro-optimization (5% time) ===
    print("\n=== Phase 5: Final micro-optimization ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    n_cur = len(current_best)
    t_rem = deadline - time.monotonic()

    if t_rem > 30:
        # Alternating: brief FW + brief subgradient + perturb
        cycles = 0
        while time.monotonic() < deadline - 15 and cycles < 20:
            cycles += 1
            current_best = best_tracker[1] if best_tracker[1] is not None else current_best
            n_cur = len(current_best)
            remaining = deadline - time.monotonic() - 10
            if remaining < 10:
                break

            if cycles % 3 == 1:
                t_alloc = min(40, remaining * 0.4)
                if t_alloc > 8:
                    f_opt, _ = fw_optimize(current_best, time_limit=t_alloc, top_k=50, near_frac=1.005)
                    try_sequence(f_opt, best_tracker)
            elif cycles % 3 == 2:
                t_alloc = min(35, remaining * 0.4)
                if t_alloc > 8:
                    f_opt, _ = exact_subgradient_polish(current_best, time_limit=t_alloc, top_k=3)
                    try_sequence(f_opt, best_tracker)
            else:
                # Perturbation + quick FW
                for strength in [0.01, 0.015, 0.008]:
                    if time.monotonic() > deadline - 15:
                        break
                    f_pert = freq_perturb(current_best, rng, strength=strength)
                    t_alloc = min(15, (deadline - time.monotonic() - 5) * 0.3)
                    if t_alloc > 5:
                        f_opt, _ = fw_optimize(f_pert, time_limit=t_alloc, top_k=35, near_frac=1.01)
                        try_sequence(f_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using fallback")
    if parent is not None:
        return [float(x) for x in parent]
    fallback = np.ones(768) / 768.0
    return [float(x) for x in fallback]
