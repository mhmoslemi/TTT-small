import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_objective(a):
    """Compute 2n*max(a*a)/(sum(a))^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)

def solve_lp_oracle(f, M):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using asymmetric starting points (ramp sequences)
and Frank-Wolfe with LP oracle. Key insight: the ramp (linearly increasing) sequence gives
score ≈ 4/3 for large n, far below the symmetric uniform solution (score = 2).
The previous approach's symmetrization destroyed this asymmetry.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """FFT-based full auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute 2n*max(a*a)/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def build_constraint_matrix(f):
    """Build the constraint matrix for (f*g)[k] = sum_i A[k,i]*g[i]."""
    n = len(f)
    k = np.arange(2 * n - 1)[:, None]
    i = np.arange(n)[None, :]
    diff = k - i
    A = np.zeros((2 * n - 1, n))
    valid = (diff >= 0) & (diff < n)
    A[valid] = f[diff[valid]]
    return A


def solve_lp_oracle(f, M, time_limit_per_solve=20):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = build_constraint_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit_per_solve})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def pgd_multi_peak(a0, time_limit, top_k=20, lr_init=0.02):

    a = np.maximum(a0.astype(np.float64), 0.0)

        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:

            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * a[k - idx_arr]

        obj = conv.max()
        v = mom * v - lr * sg

        best_candidate = a
        best_val = obj

            candidate = project_simplex(a + scale * v)

            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.92

        if no_improve > 30:

        if no_improve > 100:

def make_half_ramp(n):
