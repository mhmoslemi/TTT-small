import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad(a, p):
    """Compute gradient of sum_k p[k]*(a*a)[k] w.r.t. a.
    Uses the identity: d(a*a)[k]/da[j] = a[j+k] + a[k-j] (with bounds).
    """
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        # First term: sum_{d=0}^{n-1-j} p[d] * a[j+d]
        grad[j] += np.dot(p[:n - j], a[j:])
        # Second term: sum_{d=0}^{j} p[d] * a[j-d]
        grad[j] += np.dot(p[:j + 1], a[j::-1])
    return 2.0 * grad


def compute_grad_fft(a, p):
    """Faster gradient computation using FFT for large n."""
    n = len(a)
    if n < 512:
        return compute_grad(a, p)
    # forward[j] = sum_{d=0}^{n-1-j} p[d]*a[j+d] = cross-correlation
    # Use FFT: correlate a with reversed p
    L = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, L)
    P = np.fft.rfft(p[:n], L)
    # forward[j] = sum_d p[d]*a[j+d] = (a * conj(p_rev))[j] in frequency
    p_rev = p[:n][::-1].copy()
    P_rev = np.fft.rfft(p_rev, L)
    forward = np.fft.irfft(A * P_rev, L)[:n]
    # backward[j] = sum_d p[d]*a[j-d] = standard cross-correlation
    P_fwd = np.fft.rfft(p[:n], L)
    backward = np.fft.irfft(A * np.conj(P_fwd), L)[:n]
    return 2.0 * (forward + backward)


def softened_objective(a, T):
    """Compute softmax-weighted p vector for smoothing."""
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()
    return p, conv


def nesterov_pgd(a0, time_limit, T_init=0.001, lr_init=0.02, momentum_init=1.0):
    """Nesterov accelerated projected gradient descent on smoothed objective."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    T = T_init
    lr = lr_init
    t_ks = 1.0  # Nesterov step counter
    a_prev = a.copy()
    y = a.copy()  # Nesterov lookahead point
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Compute gradient at y (Nesterov lookahead)
        p, conv_y = softened_objective(y, T)
        grad = compute_grad(y, p)

        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_ks * t_ks)) / 2.0
        beta = (t_ks - 1.0) / t_new

        # Gradient step from y
        candidate = project_simplex(y - lr * grad)

        # Line search: try multiple step sizes
        best_cand = y
        best_val = fast_convolve(y).max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25]:
            test = project_simplex(y - trial_lr * grad)
            val = fast_convolve(test).max()
            if val < best_val:
                best_val = val
                best_cand = test

        a_new = (1.0 - beta) * best_cand + beta * a_prev

        # Check if we improved
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a_prev = a
        a = a_new
        t_ks = t_new

        # Adapt parameters
        if no_improve > 30:
            T *= 1.02
            lr *= 0.99
        if no_improve > 100:
            T = max(T * 0.7, 1e-5)
            lr = lr_init * 0.8
            t_ks = 1.0
        if no_improve > 250:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            a_prev = a.copy()
            T = T_init
            lr = lr_init
            t_ks = 1.0
            no_improve = 0

    return best_a, best_obj


def topk_lp_direction(f, M, K, n):
    """Get Frank-Wolfe direction using LP with only top-K binding constraints."""
    conv = fast_convolve(f)
    # Get top K lags (most binding constraints)
    top_k_indices = np.argsort(conv)[::-1][:K]

    # Build constraint matrix
    A_rows = []
    A_cols = []
    A_vals = []
    b = np.full(K, M)

    for idx, k in enumerate(top_k_indices):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            A_rows.append(idx)
            A_cols.append(i)
            A_vals.append(f[k - i])

    if len(A_rows) == 0:
        return None

    A_mat = csr_matrix((A_vals, (A_rows, A_cols)), shape=(K, n))
    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            sg = g.sum()
            if sg > 1e-15:
                return g / sg
    except Exception:
        pass
    return None


def fw_topk(a0, time_limit, K=50):
    """Frank-Wolfe with top-K LP oracle and line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Get LP direction
        g = topk_lp_direction(a, M, K, n)
        if g is None:
            # Fallback: use simplex vertex
            p, _ = softened_objective(a, 0.001)
            grad = compute_grad(a, p)
            j_star = np.argmin(grad)
            g = np.zeros(n)
            g[j_star] = 1.0

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7]:
            f_t = (1.0 - alpha) * a + alpha * g
            f_t = np.maximum(f_t, 0.0)
            st = f_t.sum()
            if st < 1e-15:
                continue
            f_t = f_t / st
            val = fast_convolve(f_t).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best
