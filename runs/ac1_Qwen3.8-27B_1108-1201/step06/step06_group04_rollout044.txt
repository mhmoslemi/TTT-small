def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_mv(w, a):
    """Compute W @ a where W_{i,j} = w[i+j]. w has length 2n-1, a has length n."""
    return np.correlate(w, a, mode='valid')

def conv_gradient(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    idx = np.arange(js, je + 1)
    grad[idx] = a[k - idx]
    return grad

def sca_loop(a0, time_limit, topk=5, verbose=False, use_gurobi=True):
    """SCA main loop."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_conv_max = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        
        conv = fast_convolve(a)
        cmax = conv.max()
        
        # Find top-k peaks
        if topk >= len(conv):
            topk_idx = np.arange(len(conv))
        else:
            topk_idx = np.argpartition(conv, -topk)[-topk:]
        
        peak_k = int(np.argmax(conv))
        
        # SCA step
        if use_gurobi:
            a_new = sca_iteration(a, conv, n, peak_k, topk_idx, use_gurobi=True)
        else:
            # Gradient-based fallback
            grad_sum = np.zeros(n)
            for k in topk_idx:
                grad_sum += conv_gradient(a, k) * conv[k]
            a_new = project_simplex(a - 0.01 * grad_sum)
        
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a
        else:
            a_new = a_new / s_new
        
        new_cmax = fast_convolve(a_new).max()
        
        if new_cmax < best_conv_max - 1e-14:
            best_conv_max = new_cmax
            best_a = a_new.copy()
            no_improve = 0
        else:
            # Try a blend
            for alpha in [0.5, 0.3, 0.1]:
                a_blend = (1 - alpha) * a + alpha * a_new
                a_blend = project_simplex(a_blend)
                blend_cmax = fast_convolve(a_blend).max()
                if blend_cmax < best_conv_max - 1e-14:
                    best_conv_max = blend_cmax

                    break
            else:
                no_improve += 1

            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)

def nesterov_subgrad(a0, time_limit, lr_init=0.002, momentum_init=0.9):

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:

        y = a + mom * v
        y = np.maximum(y, 0)
        y_sum = y.sum()

        grad = conv_gradient(y, k_peak)
        
        v_new = project_simplex(y - lr * 2 * grad) - y

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

                v = np.zeros(n)
                lr = lr_init * 0.5
                mom = momentum_init
                no_improve = 0
    
    return best_a, best_obj

                i = contrib[np.argmax(a[contrib])]
                j = non_contrib[rng.integers(0, len(non_contrib))]
                for frac in [0.01, 0.05, 0.1]:
                    shift = a[i] * frac
                    a[i] -= shift
                    a[j] += shift
                    a = project_simplex(a)
                    c = fast_convolve(a)
                    if c.max() < best_cmax - 1e-14:
                        best_cmax = c.max()
                        best_a = a.copy()
                        improved = True
                        break
    
    return best_a, best_cmax

def geometric_init(n, delta):
    """Geometric (asymmetric) initialization with parameter delta."""
    r = 1.0 - delta / n
    a = r ** np.arange(n)
    a = a / a.sum()
    return a

def exp_decay_init(n, beta):
    """Exponential decay initialization."""
    a = np.exp(-beta * np.arange(n) / n)
    a = a / a.sum()
    return a

def two_sided_exp_init(n, beta):
    """Two-sided exponential (but asymmetric)."""
    a = np.exp(-beta * np.abs(np.arange(n) - 0.3 * n) / n)
    a = a / a.sum()
    return a

def interpolate_to_n(parent, n):
    """Interpolate a sequence to a different length."""
    if len(parent) == n:
        return parent.copy()
    x_old = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n)
    result = np.interp(x_new, x_old, parent)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    return result
