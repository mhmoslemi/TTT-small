    n = len(f0)

        # This is: min t s.t. H_k @ f <= t for all k, sum(f)=1, f>=0
        # where H_k[i] = g[k-i] (with zero padding)

        # Subject to: for each k, sum_i f_i * g[k-i] - t <= 0
        #             sum f_i = 1
        #             f_i >= 0, t free (but will be >= 0)

        # min t

        #      1^T f = 1
        #      f >= 0

        # x = [f_0, ..., f_{n-1}, t]
        # c = [0, 0, ..., 0, 1]
        # A_ub x <= b_ub:

        # A_eq x = b_eq: [1, 1, ..., 1, 0] @ x = 1
        # bounds: f_i in [0, inf], t in [-inf, inf]
        
        # This is a standard LP that scipy can solve.
        
        c = np.zeros(n + 1)
        c[-1] = 1.0  # minimize t

        for k in range(2*n-1):
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    A_ub[k, i] = g[j]
            A_ub[k, -1] = -1.0
        
        b_ub = np.zeros(2*n-1)
        
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(None, None)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')

            f = res.x[:n]
            f = np.maximum(f, 0)
            f = f / f.sum()
        
        # Similarly for g given f

Alternating LP optimization for minimum peak autocorrelation step function.
Key insight: asymmetric sequences can beat the symmetric bound of 2.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_for_f(g, n):
    """
    Given g (fixed), find f that minimizes t such that:
    (f*g)[k] <= t for all k
    sum(f) = 1
    f >= 0
    
    Returns optimal f and t.
    """
    # Variables: x = [f_0, ..., f_{n-1}, t]
    # Minimize: t → c = [0,...,0, 1]
    # Constraints:
    #   (f*g)[k] - t <= 0 for k = 0,...,2n-2
    #   sum(f) = 1
    #   f_i >= 0, t free
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Build A_ub sparsely
    # Row k: for each i where 0 <= k-i < n, A[k,i] = g[k-i]; A[k, n] = -1
    row_ind = []
    col_ind = []
    data = []
    
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            row_ind.append(k)
            col_ind.append(i)
            data.append(g[k - i])
        # The -1 for t variable
        row_ind.append(k)
        col_ind.append(n)
        data.append(-1.0)
    
    A_ub = csr_matrix((data, (row_ind, col_ind)), shape=(2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    
    A_eq = [[1.0] * n + [0.0]]
    b_eq = [1.0]
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if res.success:
        f = np.maximum(res.x[:n], 0)
        s = f.sum()
        if s > 1e-15:
            f = f / s
        t = res.x[n]
        return f, t
    return None, None

def alternating_optimize(f0, time_limit, over_relax=1.0, tol=1e-12):
    """
    Alternating LP minimization with over-relaxation.
    """
    n = len(f0)
    f = np.maximum(f0, 0).astype(np.float64)
    f = f / f.sum()
    g = f.copy()
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Solve for optimal f given g
        f_new, t_f = solve_lp_for_f(g, n)
        if f_new is None:
            break
        
        # Solve for optimal g given f_new
        g_new, t_g = solve_lp_for_f(f_new, n)
        if g_new is None:
            break
        
        # Over-relaxation
        f = f_new * over_relax + f * (1 - over_relax)
        f = np.maximum(f, 0)
        s = f.sum()
        if s > 1e-15:
            f = f / s
        else:
            f = f_new
        
        g = g_new * over_relax + g * (1 - over_relax)
        g = np.maximum(g, 0)
        s = g.sum()
        if s > 1e-15:
            g = g / s
        else:
            g = g_new
        
        # Symmetrize (average f and g since at optimum they should be equal)
        f_avg = (f + g) / 2
        f_avg = f_avg / f_avg.sum()
        
        obj = fast_convolve(f_avg).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_avg.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt over-relaxation
        if no_improve > 50:
            over_relax = max(0.5, over_relax * 0.9)
        if no_improve > 200:
            # Restart with perturbation
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            g = f.copy()
            over_relax = 1.0
            no_improve = 0
        
        # Convergence check
        if abs(f_avg.max() - f.max()) < tol and no_improve > 100:
            break
    
    return best_f, best_obj

    k_max = int(np.argmax(conv))

    cuts_a = [a_curr.copy()]
    cuts_k = [k_max]

"""
Cutting-plane optimization for minimum peak autocorrelation step function.
Key insight: asymmetric sequences can beat the symmetric bound of 2.
Uses a cutting-plane method that builds up linear approximations of the
quadratic constraints, converging to the true optimum of the non-convex

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_cuts(cuts_a, cuts_k, n):
    """
    Solve LP: min t s.t. (cuts_a[i] * a)[cuts_k[i]] <= t, sum(a)=1, a>=0.
    Returns (a, t) or (None, None) on failure.
    """
    m = len(cuts_a)
    c = np.zeros(n + 1)
    c[-1] = 1.0

    row_ind = []
    col_ind = []
    data = []

    for i in range(m):
        a_cut = cuts_a[i]
        k_cut = cuts_k[i]
        i_start = max(0, k_cut - n + 1)
        i_end = min(k_cut, n - 1)
        for j in range(i_start, i_end + 1):
            row_ind.append(i)
            col_ind.append(j)
            data.append(a_cut[k_cut - j])
        row_ind.append(i)
        col_ind.append(n)
        data.append(-1.0)

    A_ub = csr_matrix((data, (row_ind, col_ind)), shape=(m, n + 1))
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            a = np.maximum(res.x[:n], 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            return a, res.x[n]
    except Exception:
        pass
    return None, None

def solve_lp_direct(g, n):
    """
    Solve LP: min t s.t. (f*g)[k] <= t for all k, sum(f)=1, f>=0.
    This is the alternating step (exact LP, not FW direction).
    """
    c = np.zeros(n + 1)
    c[-1] = 1.0

    row_ind = []
    col_ind = []
    data = []

    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            row_ind.append(k)
            col_ind.append(i)
            data.append(g[k - i])
        row_ind.append(k)
        col_ind.append(n)
        data.append(-1.0)

    A_ub = csr_matrix((data, (row_ind, col_ind)), shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            f = np.maximum(res.x[:n], 0)
            s = f.sum()
            if s > 1e-15:
                f = f / s
            return f, res.x[n]
    except Exception:
        pass
    return None, None


def alternating_lp(f0, time_limit, max_iter=5000):
    """
    Alternating exact-LP minimization.
    Alternates between solving min t s.t. (f*g)[k]<=t and min t s.t. (g*f)[k]<=t.
    """
    n = len(f0)
    f = np.maximum(f0, 0).astype(np.float64)
    f = f / f.sum()
    g = f.copy()

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.3:
            break

        # Solve for f given g
        f_new, t1 = solve_lp_direct(g, n)
        if f_new is None:
            break

        # Solve for g given f_new
        g_new, t2 = solve_lp_direct(f_new, n)
        if g_new is None:
            break

        # Take the average (at optimum, f = g)
        f_avg = (f_new + g_new) / 2.0
        f_avg = f_avg / f_avg.sum()

        obj = fast_convolve(f_avg).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_avg.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_avg
        g = f_avg

        # Gap check: if t1 and t2 are close to obj, we've converged
        if abs(obj - t1) < 1e-10 and abs(obj - t2) < 1e-10:
            break

        if no_improve > 100:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            g = f.copy()
            no_improve = 0

    return best_f, best_obj


def game_iterations(f0, time_limit, T_start=0.001, lr_start=0.02):
    """
    Game-theoretic iterations: dual is softmax over peaks,
    primal is projected gradient on the weighted quadratic.
    """
    n = len(f0)
    a = np.maximum(f0, 0).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    T = T_start
    lr = lr_start
    no_improve = 0
    it = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Primal step: projected gradient on min a^T H(p) a
        for _ in range(3):
            # Gradient of sum_k p[k] * (a*a)[k] = 2 * sum_k p[k] * H_k @ a
            # = 2 * (p_hankel @ a) where p_hankel is built from p
            # grad[j] = 2 * sum_m a[m] * p[j+m] (with appropriate bounds)
            grad = np.zeros(n)
            for j in range(n):
                m_start = 0
                m_end = n
                k_start = j
                k_end = j + n - 1
                # p indices: j+m for m in [0, n-1], valid when j+m in [0, 2n-2]
                valid_m = np.arange(n)
                k_vals = j + valid_m
                mask = (k_vals >= 0) & (k_vals < 2 * n - 1)
                if mask.any():
                    grad[j] = 2.0 * np.dot(a[valid_m[mask]], p[k_vals[mask]])

            candidate = project_simplex(a - lr * grad)
            new_obj = np.dot(p, fast_convolve(candidate))
            old_obj = np.dot(p, fast_convolve(a))
            if new_obj < old_obj:
                a = candidate
            else:
                break

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt
        if it % 50 == 0:
            T *= 1.05
            lr *= 0.99
        if no_improve > 150:
            T = max(0.0001, T * 0.5)
            lr = lr_start * 0.5
        if no_improve > 400:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_start
            lr = lr_start
            no_improve = 0

    return best_a, best_obj


def local_search(a0, time_limit, seed=42):
    """Local search with multiple perturbation scales."""
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        improved = False

        # Random perturbations
        for scale in [0.0002, 0.0005, 0.001, 0.003, 0.008, 0.02]:
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(a + pert)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        # Pairwise transfers
        if not improved:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                for frac in [0.01, 0.05, 0.1, 0.2, 0.5]:
                    candidate = a.copy()
                    amount = candidate[i] * frac
                    candidate[i] -= amount
                    candidate[j] += amount
                    candidate = np.maximum(candidate, 0)
                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break

        # Structured perturbation
        if not improved and it % 5 == 0:
            freq = rng.uniform(0.5, 10)
            phase = rng.uniform(0, 2 * np.pi)
            pattern = np.sin(np.pi * freq * np.arange(n) / n + phase) * 0.003
            candidate = project_simplex(a + pattern)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True

        if not improved and it > 5000:
            break

    return best_a, best_obj

    remaining = deadline - time.monotonic()

    if remaining > 30:
        t_alloc = min(remaining * 0.3, 120)
        print(f"  Local search, t={t_alloc:.0f}s")
        a_opt, obj = local_search(best_seq, time_limit=t_alloc, seed=31415)
        new_s, best_score = try_seq(a_opt, "local1", best_score)
        if new_s is not None:
            best_seq = new_s

    # === Phase 8: Final alternating LP ===
    print("\n=== Phase 8: Final Alt-LP ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()

    if remaining > 30:
        t_alloc = min(remaining * 0.3, 120)
        print(f"  Final alt-LP, t={t_alloc:.0f}s")
        a_opt, obj = alternating_lp(best_seq, time_limit=t_alloc)
        new_s, best_score = try_seq(a_opt, "final_alt", best_score)
        if new_s is not None:
            best_seq = new_s

    # === Phase 9: Tiny perturbations ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        n_best = len(best_seq)
        for scale in [0.0001, 0.0003, 0.001]:
            remaining = deadline - time.monotonic()
            if remaining < 3:
                break
            noise = rng.standard_normal(n_best) * scale
            f_pert = project_simplex(best_seq + noise)
            new_s, best_score = try_seq(f_pert, f"micro_{scale}", best_score)
            if new_s is not None:
                best_seq = new_s
                # Quick local search
                remaining = deadline - time.monotonic()
                if remaining > 10:
                    a_opt, _ = local_search(best_seq, time_limit=min(remaining * 0.3, 20),
                                            seed=int(rng.integers(0, 100000)))
                    new_s2, best_score = try_seq(a_opt, "micro_local", best_score)
                    if new_s2 is not None:
                        best_seq = new_s2

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
