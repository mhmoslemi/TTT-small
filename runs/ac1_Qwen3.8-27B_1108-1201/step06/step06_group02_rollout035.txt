"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
- Cyclic coordinate descent with efficient 1D subproblem solving
- LP-based Frank-Wolfe for global moves

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_obj(p):
    """Compute objective 2n*max(conv)/(sum)^2 assuming p is normalized."""
    n = len(p)
    conv = fast_convolve(p)
    return 2.0 * n * conv.max()


def coord_descent_1d(p, i, s, conv_current):

    Returns optimal t for p[i].
    p is currently normalized (sum=1), s=1.

    old_t = p[i]
    q = p.copy()
    q[i] = 0.0
    s_prime = 1.0 - old_t

    if s_prime < 1e-15:
        return old_t

    # Compute q*q convolution
    q_conv = fast_convolve(q)

    # q_shifted[k] = q[k-i] if 0 <= k-i < n else 0

    # where q_shifted[k] = q[k-i] if valid, else 0

    ks = np.arange(0, 2 * n - 1)

    q_shifted = np.zeros(2 * n - 1)

    delta_k = np.zeros(2 * n - 1)

        return num.max() / den

        grid = np.sort(grid)

def coordinate_descent(p0, time_limit, rng, verbose=False):
    """Cyclic coordinate descent with 1D optimization per coordinate."""
    p = np.maximum(p0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p)) / len(p)
    else:
        p = p / s
    n = len(p)

    best_p = p.copy()
    best_obj = compute_obj(p)

    t0 = time.monotonic()
    sweep = 0
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        sweep_start = time.monotonic()
        sweep_improved = 0

        # Randomized cyclic order
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit - 0.01:
                break

            new_t = coord_descent_1d(p, i, 1.0, None)

            if abs(new_t - p[i]) > 1e-10:
                p[i] = new_t
                s_new = p.sum()
                if s_new > 1e-15:
                    p = p / s_new
                sweep_improved += 1

        if sweep_improved > 0:
            cur_obj = compute_obj(p)
            total_improvements += sweep_improved
            if cur_obj < best_obj - 1e-12:
                if verbose and (sweep % 10 == 1):
                    print("    CD sweep {}: obj={:.10f}".format(sweep, cur_obj))
                best_obj = cur_obj
                best_p = p.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Stochastic restart if stuck
        if no_improve >= 3:
            scale = 0.002 * min(no_improve, 10)
            p = project_simplex(best_p + rng.standard_normal(n) * scale)
            no_improve = 0

        # Adaptive: if sweep is too slow, reduce time per evaluation
        sweep_time = time.monotonic() - sweep_start
        remaining = time_limit - (time.monotonic() - t0)
        if sweep_time > remaining / 3:
            # Too slow, break
            break

    return best_p, best_obj

        f = f / s
    n = len(f)

    best_obj = M
    best_f = f.copy()

    no_improve = 0
    it = 0
    m_exp = 1.0

        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:

        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(1, remaining - 1)})
            if not res.success:
                m_exp = min(m_exp * 1.5, 5.0)
                continue
            g = np.maximum(res.x, 0.0)
        except Exception:
            m_exp = min(m_exp * 1.5, 5.0)
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                rng = np.random.default_rng(it * 7919 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
                continue
            m_exp = min(m_exp * 1.1, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in np.linspace(0.01, 1.0, 15):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.2, 5.0)

        if no_improve > 40:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations for a given n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian (centered)
    sigma = 0.2 * n
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
    inits.append(g / g.sum())

    # Triangle (bell)
    t = np.minimum(x + 1, n - x)
    inits.append(t / t.sum())

    # Edge-weighted
    e = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(e / e.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 0.1))

    # Interpolated parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Cosine-modulated
    for freq in [1, 2, 3]:
        h = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * x / n)
        h = np.maximum(h, 0.01)
        inits.append(h / h.sum())

    # Two-peak
    peak1 = 0.3 * (n - 1)
    peak2 = 0.7 * (n - 1)
    p1 = np.exp(-0.5 * ((x - peak1) / (0.1 * n)) ** 2)
    p2 = np.exp(-0.5 * ((x - peak2) / (0.1 * n)) ** 2)
    two_peak = p1 + p2
    inits.append(two_peak / two_peak.sum())

    return inits


def anneal_search(best_p, rng, time_limit, polish_time=5.0):
    """
    Simulated annealing style: large perturbations + coordinate descent polish.
    """
    p = best_p.copy()
    n = len(p)
    best_obj = compute_obj(p)
    t0 = time.monotonic()
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < polish_time + 1.0:
            break

        # Annealing schedule
        progress = (time.monotonic() - t0) / time_limit
        # Scale decreases over time
        if progress < 0.3:
            scale = 0.05
        elif progress < 0.5:
            scale = 0.03
        elif progress < 0.7:
            scale = 0.015
        elif progress < 0.9:
            scale = 0.005
        else:
            scale = 0.002

        # Generate perturbation
        perturbation_type = rng.integers(0, 4)
        candidate = p.copy()

        if perturbation_type == 0:
            # Gaussian noise
            candidate = project_simplex(p + rng.standard_normal(n) * scale)
        elif perturbation_type == 1:
            # Block perturbation
            block_size = max(1, n // 10)
            start = rng.integers(0, n - block_size)
            new_block = rng.dirichlet(np.ones(block_size) * 0.3)
            candidate[start:start + block_size] = new_block * candidate[start:start + block_size].sum()
            candidate = project_simplex(candidate)
        elif perturbation_type == 2:
            # Cosine modulation perturbation
            freq = rng.integers(1, max(2, n // 50))
            phase = rng.uniform(0, 2 * np.pi)
            mod = 1.0 + scale * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
            candidate = p * mod
            candidate = np.maximum(candidate, 0)
            cs = candidate.sum()
            if cs > 1e-10:
                candidate = candidate / cs
            else:
                candidate = p.copy()
        else:
            # Sparse perturbation (change a few coordinates)
            n_changes = max(1, n // 20)
            idx = rng.choice(n, size=n_changes, replace=False)
            candidate[idx] += rng.standard_normal(n_changes) * scale
            candidate = project_simplex(candidate)

        # Quick polish with coordinate descent
        cand_obj = compute_obj(candidate)
        if cand_obj < best_obj:
            # Accept and polish
            time_left = time.monotonic() - t0
            polish_budget = min(polish_time, time_limit - (time.monotonic() - t0) - 1.0)
            if polish_budget > 0.5:
                polished, pol_obj = coordinate_descent(candidate, polish_budget, rng)
                if pol_obj < best_obj:
                    best_obj = pol_obj
                    p = polished
            else:
                p = candidate
        elif cand_obj < best_obj * (1.0 + 0.05 * (1.0 - progress)):
            # Accept with probability (annealing)
            # Polish even if slightly worse, in hopes of finding a better basin
            remaining = time_limit - (time.monotonic() - t0)
            if remaining > polish_time + 1.0:
                polish_budget = min(polish_time, remaining / 3)
                polished, pol_obj = coordinate_descent(candidate, polish_budget, rng)
                if pol_obj < best_obj:
                    best_obj = pol_obj
                    p = polished

    return p, best_obj


def subgradient_descent(p0, time_limit, rng, top_k=20, lr_init=0.01):
    """Subgradient descent focusing on top peaks."""
    p = np.maximum(p0.astype(np.float64), 0.0)
    p = p / p.sum()
    n = len(p)

    v = np.zeros(n)
    lr = lr_init
    mom = 0.9
    best_p = p.copy()
    best_obj = compute_obj(p)

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break

        conv = fast_convolve(p)
        # Top k peaks
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))

        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * p[k - idx_arr]

        v = mom * v - lr * sg
        p_new = project_simplex(p + v)

        obj = compute_obj(p_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_p = p_new.copy()
            p = p_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.999
        mom = min(0.98, mom + 0.00005)

        if no_improve > 100:
            scale = 0.003 * min(1.0, no_improve / 300.0)
            p = project_simplex(best_p + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            lr = lr_init * 0.5
            no_improve = 0

    return best_p, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid search:
    Phase 1: Quick exploration of multiple (n, init) combinations
    Phase 2: Deep coordinate descent on best candidates
    Phase 3: Annealing + FW-LP for global refinement
    Phase 4: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)

    state = [float('inf'), None]

    def try_update(a, label):
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < state[0]:
            state[0] = score
            state[1] = a_norm.copy()
            print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a)))

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.8f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        n = 512
        x = np.arange(n, dtype=np.float64)
        parent = np.exp(-0.5 * ((x - 0.5*(n-1)) / (0.2*n))**2)
        parent = parent / parent.sum()
        state = [float(evaluate_sequence(parent.tolist())), parent.copy()]
        print("Generated default parent: n={}".format(n))

    n_parent = len(parent)
    print("Starting score: {:.8f}".format(state[0]))

    # === Phase 1: Quick Exploration (~400s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent, 384, 512, 640, 768, 896, 1024, 1280]))
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 2048]

    # Process smaller n first (faster), then larger
    candidate_ns.sort(key=lambda n: abs(n - n_parent))

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1000
        if remaining < 20:
            break

        inits = generate_initializations(n_try, parent, rng)

        for init_idx, init in enumerate(inits):
            remaining = deadline - time.monotonic() - 900
            if remaining < 5:
                break

            quick_cd_time = min(8.0, remaining * 0.4)
            a_opt, _ = coordinate_descent(init, quick_cd_time, rng)
            try_update(a_opt, "cd_n{}_i{}".format(n_try, init_idx))

        # Best init from this n gets more time
        if state[1] is not None and len(state[1]) == n_try:
            remaining = deadline - time.monotonic() - 800
            if remaining > 10:
                a_opt, _ = coordinate_descent(state[1], min(remaining, 20.0), rng)
                try_update(a_opt, "cd2_n{}".format(n_try))

    print("  Phase 1 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 2: Deep Coordinate Descent (~500s) ===
    print("\n=== Phase 2: Deep Coordinate Descent ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])
        remaining = deadline - time.monotonic() - 500
        if remaining > 30:
            t_alloc = min(remaining, 300)
            print("  Deep CD n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = coordinate_descent(state[1], t_alloc, rng, verbose=True)
            try_update(a_opt, "deep_cd")

        # Subgradient
        remaining = deadline - time.monotonic() - 300
        if remaining > 20:
            for top_k, lr in [(10, 0.01), (30, 0.005), (50, 0.003)]:
                remaining = deadline - time.monotonic() - 200
                if remaining < 10:
                    break
                t_alloc = min(remaining, 60)
                a_opt, _ = subgradient_descent(state[1], t_alloc, rng, top_k=top_k, lr_init=lr)
                try_update(a_opt, "sg_k{}_lr{}".format(top_k, lr))

        # Another CD round
        remaining = deadline - time.monotonic() - 150
        if remaining > 15:
            a_opt, _ = coordinate_descent(state[1], min(remaining, 100), rng)
            try_update(a_opt, "cd_round2")

    print("  Phase 2 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 3: Annealing + FW-LP (~500s) ===
    print("\n=== Phase 3: Annealing + FW-LP ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Simulated annealing
        remaining = deadline - time.monotonic() - 200
        if remaining > 30:
            t_alloc = min(remaining, 250)
            print("  Annealing search, t={:.0f}s".format(t_alloc))
            a_opt, _ = anneal_search(state[1], rng, t_alloc, polish_time=3.0)
            try_update(a_opt, "anneal")

        # FW-LP
        remaining = deadline - time.monotonic() - 150
        if remaining > 20:
            t_alloc = min(remaining, 150)
            print("  FW-LP, t={:.0f}s".format(t_alloc))
            a_opt, _ = fw_lp_optimize(state[1], t_alloc)
            try_update(a_opt, "fw_lp")

        # CD after FW-LP
        remaining = deadline - time.monotonic() - 80
        if remaining > 10:
            a_opt, _ = coordinate_descent(state[1], min(remaining, 80), rng)
            try_update(a_opt, "cd_after_fw")

        # More annealing with different parameters
        remaining = deadline - time.monotonic() - 50
        if remaining > 20:
            t_alloc = min(remaining, 60)
            a_opt, _ = anneal_search(state[1], rng, t_alloc, polish_time=2.0)
            try_update(a_opt, "anneal2")

    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 4: Try other n values with best-informed init (~200s) ===
    print("\n=== Phase 4: Cross-n Refinement ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])
        # Try nearby n values
        nearby_ns = [n_best + d for d in [-128, -64, 64, 128, 256] if 128 <= n_best + d <= 2048]
        nearby_ns = [n for n in nearby_ns if n != n_best]

        for n_try in nearby_ns[:3]:
            remaining = deadline - time.monotonic() - 50
            if remaining < 15:
                break

            # Interpolate best to new n
            init = np.interp(np.linspace(0, 1, n_try),
                             np.linspace(0, 1, n_best), state[1])
            init = np.maximum(init, 0)
            iss = init.sum()
            if iss > 1e-10:
                init = init / iss

            t_alloc = min(remaining, 30)
            a_opt, _ = coordinate_descent(init, t_alloc, rng)
            try_update(a_opt, "cross_n{}".format(n_try))

    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 5: Final Polish (~100s) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Multiple short CD passes
        for i in range(3):
            remaining = deadline - time.monotonic() - 10
            if remaining < 3:
                break
            t_alloc = min(remaining, 30)
            a_opt, _ = coordinate_descent(state[1], t_alloc, rng)
            try_update(a_opt, "polish_{}".format(i))

        # Tiny perturbations + quick polish
        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            for _ in range(8):
                remaining = deadline - time.monotonic() - 2
                if remaining < 1:
                    break
                noise = rng.standard_normal(len(state[1])) * 0.001
                perturbed = project_simplex(state[1] + noise)
                a_opt, _ = coordinate_descent(perturbed, min(remaining, 2.0), rng)
                try_update(a_opt, "micro_perturb")

    print("  Phase 5 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(384) / 384
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
