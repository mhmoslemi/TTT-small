"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def conv_max_and_peak(a):
    """Return (max_conv, peak_index, full_conv) for a sequence."""
    c = fast_convolve(a)
    idx = np.argmax(c)
    return c[idx], idx, c


def project_simplex(v):
    """Euclidean projection onto {x : x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def build_sparse_constraint(f, indices):
    """Build sparse matrix for constraints (f*g)[k] <= M for k in indices."""
    n = len(f)
    rows, cols, vals = [], [], []
    for idx, k in enumerate(indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([idx] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    if not rows:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(len(indices), n))


def lp_direction(f, M, top_k=40, time_limit=10):
    """LP: max sum(g) s.t. (f*g)[k] <= M for top peaks, g >= 0."""
    n = len(f)
    conv = fast_convolve(f)
    k_use = min(top_k, 2 * n - 1)
    peak_idx = np.argpartition(conv, -k_use)[-k_use:]
    center = n - 1
    if center not in peak_idx:
        peak_idx = np.append(peak_idx, center)
    A = build_sparse_constraint(f, peak_idx)
    if A.shape[0] == 0:
        return np.ones(n) / n
    c_obj = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None


def lse_grad_weights(conv, beta):
    """Softmax weights over convolution values for subgradient computation."""
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    s = exp_bc.sum()
    if s < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_bc / s


def compute_gradient(f, peaks, weights):
    """Weighted sum of gradients of (f*f)[k] for k in peaks."""
    n = len(f)
    grad = np.zeros(n)
    for idx, k in enumerate(peaks):
        k = int(k)
        w = weights[idx]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += w * 2.0 * f[k - i_range]
    return grad


def nesterov_subgrad(f0, time_limit, beta_start=15.0, beta_end=8000.0,
                     top_k=50, momentum=0.9):
    """Nesterov accelerated subgradient with simplex projection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    y = f.copy()  # Nesterov lookahead
    v = np.zeros(n)  # velocity
    beta = beta_start
    lr = 3.0 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        conv = fast_convolve(y)
        w_full = lse_grad_weights(conv, beta)
        k_top = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        w_peaks = w_full[peak_idx]
        ws = w_peaks.sum()
        if ws > 1e-15:
            w_peaks = w_peaks / ws
        else:
            w_peaks = np.ones(k_top) / k_top

        grad = compute_gradient(y, peak_idx, w_peaks)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_n = grad / gnorm

        # Nesterov step
        z = project_simplex(y - lr * grad_n)
        v = momentum * v + (z - y)
        y_new = project_simplex(z + momentum * v)

        # Evaluate and accept if improved
        conv_new = fast_convolve(y_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = y_new.copy()
            no_improve = 0
            lr = min(lr * 1.008, 15.0 / np.sqrt(n))
        else:
            no_improve += 1
            if no_improve >= 20:
                lr *= 0.85
            if no_improve > 80:
                # Restart from best with fresh momentum
                y = best_f.copy()
                v = np.zeros(n)
                lr = max(3.0 / np.sqrt(n), 0.01)
                no_improve = 0
                # Small random perturbation
                rng = np.random.default_rng(it * 104729 + 3)
                y = project_simplex(best_f + rng.standard_normal(n) * 0.002)

        y = y_new
        beta = min(beta * 1.006, beta_end)

    return best_f, best_obj

def coordinate_descent_polish(f0, time_limit=60.0):
    """Coordinate descent: move mass from peak contributors to non-peak positions."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, float('inf')
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        # Top 10 peaks
        k_peaks = min(10, 2 * n - 1)
        peak_list = np.argpartition(conv, -k_peaks)[-k_peaks:]
        peak_list = peak_list[np.argsort(conv[peak_list])[::-1]]

        improved = False
        for k in peak_list:
            k = int(k)
            if time.monotonic() - t0 > time_limit:
                break
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo >= hi:
                continue
            i_range = np.arange(lo, hi + 1)
            products = f[i_range] * f[k - i_range]
            j_max = i_range[np.argmax(products)]
            val_max = products.max()
            if val_max < 1e-15:
                continue

            # Try moving mass from j_max to random positions
            n_trials = min(20, n)
            candidates = rng.choice(n, size=n_trials, replace=False)
            best_move = None
            best_score = best_obj

            for j_dest in candidates:
                j_dest = int(j_dest)
                if j_dest == j_max or j_dest == k - j_max:
                    continue
                # Try different fractions
                for frac in [0.05, 0.1, 0.2, 0.3]:
                    delta = f[j_max] * frac
                    if delta < 1e-12:
                        continue
                    f_trial = f.copy()
                    f_trial[j_max] -= delta
                    f_trial[j_dest] += delta
                    f_trial = np.maximum(f_trial, 0)
                    ts = f_trial.sum()
                    if ts < 1e-15:
                        continue
                    f_trial /= ts
                    c_trial = fast_convolve(f_trial)
                    obj_trial = c_trial.max()
                    if obj_trial < best_score - 1e-14:
                        best_score = obj_trial
                        best_move = (f_trial, j_max, j_dest, frac)

            if best_move is not None:
                f = best_move[0]
                best_obj = best_score
                best_f = f.copy()
                improved = True

        if not improved:
            # Try a broader random move
            i_src = rng.integers(n)
            i_dst = rng.integers(n)
            if i_src == i_dst:
                continue
            for frac in [0.02, 0.05, 0.1]:
                delta = f[i_src] * frac
                if delta < 1e-12:
                    break
                f_trial = f.copy()
                f_trial[i_src] -= delta
                f_trial[i_dst] += delta
                f_trial = np.maximum(f_trial, 0)
                ts = f_trial.sum()
                if ts < 1e-15:
                    break
                f_trial /= ts
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < best_obj - 1e-14:
                    f = f_trial
                    best_obj = obj_trial
                    best_f = f.copy()
                    improved = True
                    break

    return best_f, best_obj


def freq_perturb(f, rng, strength=0.04):
    """Perturb f in frequency domain for diversity."""
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def interpolate_length(f, new_n):
    """Interpolate normalized sequence to new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def try_eval(a, best_tracker, verbose=True):
    """Evaluate and update best tracker."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        if verbose:
            print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Multi-length FW-LP exploration
    Phase 2: Nesterov subgradient refinement
    Phase 3: Alternating FW + Nesterov + FreqPerturb
    Phase 4: Coordinate descent polish
    Phase 5: Final high-beta squeeze
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_eval(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        try_eval(parent, best_tracker)

    n_parent = len(parent)
    total_budget = float(budget_s) - 30.0

    # === Phase 1: Multi-length FW exploration ===
    print("\n=== Phase 1: Multi-length FW exploration ===")
    t1_end = time.monotonic() + total_budget * 0.20

    lengths = sorted(set([
        n_parent,
        max(128, n_parent - 64),
        min(2000, n_parent + 64),
        max(128, int(n_parent * 0.75)),
        min(2000, int(n_parent * 1.25)),
    ]), reverse=True)
    seen = set()
    lengths = [x for x in lengths if not (x in seen or seen.add(x))]

    for nc in lengths[:5]:
        t_rem = t1_end - time.monotonic()
        if t_rem < 20:
            break
        t_alloc = min(90, t_rem * 0.3)
        if t_alloc < 10:
            break
        if nc == n_parent:
            f_init = parent.copy()
        else:
            f_init = interpolate_length(parent, nc)
        print(f"  FW n={nc}, t={t_alloc:.0f}s")
        f_opt, _ = fw_optimize(f_init, time_limit=t_alloc, top_k=35)
        try_eval(f_opt, best_tracker)

    # === Phase 2: Nesterov subgradient ===
    print(f"\n=== Phase 2: Nesterov subgradient ===")
    t2_end = t1_end + total_budget * 0.22
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)

    t_alloc = min(180, t2_end - time.monotonic())
    if t_alloc > 15:
        print(f"  Nesterov n={n_cur}, t={t_alloc:.0f}s")
        f_opt, _ = nesterov_subgrad(current_best, time_limit=t_alloc,
                                     beta_start=15.0, beta_end=8000.0, top_k=50)
        try_eval(f_opt, best_tracker)

    # === Phase 3: Alternating optimization ===
    print(f"\n=== Phase 3: Alternating optimization ===")
    t3_end = deadline - 180
    cycles = 0
    while time.monotonic() < t3_end and cycles < 20:
        cycles += 1
        current_best = best_tracker[1] if best_tracker[1] is not None else parent
        n_cur = len(current_best)
        remaining = t3_end - time.monotonic()
        if remaining < 30:
            break

        phase = cycles % 5
        if phase == 0:
            # FW step
            t_alloc = min(80, remaining * 0.3)
            if t_alloc > 10:
                print(f"  C{cycles}: FW n={n_cur}, t={t_alloc:.0f}s")
                f_opt, _ = fw_optimize(current_best, time_limit=t_alloc, top_k=40)
                try_eval(f_opt, best_tracker)

        elif phase == 1:
            # Nesterov with higher beta
            t_alloc = min(90, remaining * 0.35)
            if t_alloc > 10:
                beta_s = 50.0 + 60.0 * cycles
                print(f"  C{cycles}: Nesterov n={n_cur}, t={t_alloc:.0f}s, beta={beta_s:.0f}")
                f_opt, _ = nesterov_subgrad(current_best, time_limit=t_alloc,
                                             beta_start=beta_s, beta_end=15000.0, top_k=60)
                try_eval(f_opt, best_tracker)

        elif phase == 2:
            # Freq perturb + FW
            t_alloc = min(70, remaining * 0.3)
            if t_alloc > 10:
                print(f"  C{cycles}: Freq+FW n={n_cur}, t={t_alloc:.0f}s")
                for att in range(2):
                    t_rem2 = t3_end - time.monotonic() - 10
                    if t_rem2 < 12:
                        break
                    strength = 0.015 + 0.01 * att
                    f_pert = freq_perturb(current_best, rng, strength=strength)
                    t_sub = max(6.0, (t_rem2 - 5) / 2.0)
                    f_opt, _ = fw_optimize(f_pert, time_limit=t_sub, top_k=30)
                    try_eval(f_opt, best_tracker)

        elif phase == 3:
            # Coordinate descent
            t_alloc = min(50, remaining * 0.2)
            if t_alloc > 8:
                print(f"  C{cycles}: CoordDesc n={n_cur}, t={t_alloc:.0f}s")
                f_opt, _ = coordinate_descent_polish(current_best, time_limit=t_alloc)
                try_eval(f_opt, best_tracker)

        else:
            # Try different length (interpolate + quick optimize)
            t_alloc = min(60, remaining * 0.25)
            if t_alloc > 10:
                n_new = int(n_cur * rng.uniform(0.9, 1.1))
                n_new = max(128, min(2000, n_new))
                if n_new != n_cur:
                    print(f"  C{cycles}: Reinterp n={n_new}, t={t_alloc:.0f}s")
                    f_interp = interpolate_length(current_best, n_new)
                    t_half = t_alloc / 2.0
                    f_opt, _ = fw_optimize(f_interp, time_limit=t_half, top_k=30)
                    try_eval(f_opt, best_tracker)
                    f_opt2, _ = nesterov_subgrad(f_opt, time_limit=t_half,
                                                  beta_start=30.0, beta_end=5000.0, top_k=40)
                    try_eval(f_opt2, best_tracker)

    # === Phase 4: Final coordinate descent polish ===
    print(f"\n=== Phase 4: Final coordinate descent polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(120, deadline - time.monotonic() - 60)
    if t_alloc > 10:
        print(f"  CoordDesc final n={len(current_best)}, t={t_alloc:.0f}s")
        f_opt, _ = coordinate_descent_polish(current_best, time_limit=t_alloc)
        try_eval(f_opt, best_tracker)

    # === Phase 5: High-beta squeeze ===
    print(f"\n=== Phase 5: High-beta squeeze ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(80, deadline - time.monotonic() - 30)
    if t_alloc > 10:
        print(f"  High-beta Nesterov n={len(current_best)}, t={t_alloc:.0f}s")
        f_opt, _ = nesterov_subgrad(current_best, time_limit=t_alloc,
                                     beta_start=500.0, beta_end=50000.0, top_k=25,
                                     momentum=0.95)
        try_eval(f_opt, best_tracker)

    # One more FW with many constraints
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(50, deadline - time.monotonic() - 10)
    if t_alloc > 10:
        print(f"  Final FW (top_k=80) n={len(current_best)}, t={t_alloc:.0f}s")
        f_opt, _ = fw_optimize(current_best, time_limit=t_alloc, top_k=80)
        try_eval(f_opt, best_tracker)

    # Final tiny perturbations
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_rem = deadline - time.monotonic()
    if t_rem > 15:
        print(f"  Final perturbations, t={t_rem:.0f}s")
        for _ in range(15):
            if time.monotonic() > deadline - 3:
                break
            f_p = freq_perturb(current_best, rng, strength=0.003)
            try_eval(f_p, best_tracker, verbose=False)
            f_p2 = project_simplex(current_best + rng.standard_normal(len(current_best)) * 0.001)
            try_eval(f_p2, best_tracker, verbose=False)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using uniform fallback")
    fallback = np.ones(512) / 512.0
    return [float(x) for x in fallback]
