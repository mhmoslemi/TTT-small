def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, use_all_constraints=True, time_limit=20):
    """

    # Build constraint matrix: (f*g)[k] = sum_j f[k-j]*g[j] for valid j

    rows = []
    cols = []
    vals = []
    
    for k in range(num_k):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    
    if not rows:
        return np.ones(n) / n, M
    
    A = csr_matrix((vals, (rows, cols)), shape=(num_k, n))

    c = np.zeros(n + 1)

    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(None, None)]  # g >= 0, t free
    
    try:

                      options={'presolve': True, 'time_limit': time_limit})

            if gs > 1e-15:

            else:
                g = np.ones(n) / n
            t_opt = res.x[-1]
            return g, t_opt

        pass
    
    # Fallback: return uniform
    return np.ones(n) / n, M

3. Subgradient refinement with top-K exact subgradient
4. Multi-length exploration with intelligent time allocation
5. Adaptive perturbation for escaping local minima
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build the (2n-1) x n matrix A where (A*g)[k] = (f*g)[k]."""
    n = len(f)
    num_k = 2 * n - 1
    A = np.zeros((num_k, n))
    for k in range(num_k):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            A[k, js] = f[k - js]
    return A


def solve_lp_direction(f, M, time_limit=30.0):
    """
    Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1.
    Returns (g, t_opt). If LP fails, returns (f, M).
    """
    n = len(f)
    num_k = 2 * n - 1
    
    # Build constraint matrix
    A = build_conv_matrix(f)
    
    # Full constraint matrix: A * g - t <= 0
    A_full = np.hstack([A, -np.ones((num_k, 1))])
    b_full = np.zeros(num_k)
    
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_full, b_ub=b_full, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.maximum(res.x[:n], 0.0)
            gs = g.sum()
            if gs > 1e-15:
                g = g / gs
            else:
                g = f.copy()
            t_opt = float(res.x[-1])
            return g, t_opt
    except Exception:
        pass
    
    return f.copy(), M


def solve_lp_topk(f, M, top_k=50, time_limit=15.0):
    """Solve LP using only top-K peaks as constraints (faster for large n)."""
    n = len(f)
    conv = fast_convolve(f)
    peak_idx = np.argpartition(conv, -top_k)[-top_k:]
    # Ensure center is included
    center = n - 1
    if center not in peak_idx:
        peak_idx = np.append(peak_idx, center)
    
    A = build_conv_matrix(f)
    A_sel = A[peak_idx, :]
    
    A_full = np.hstack([A_sel, -np.ones((len(peak_idx), 1))])
    b_full = np.zeros(len(peak_idx))
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_full, b_ub=b_full, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.maximum(res.x[:n], 0.0)
            gs = g.sum()
            if gs > 1e-15:
                g = g / gs
            else:
                g = f.copy()
            t_opt = float(res.x[-1])
            return g, t_opt
    except Exception:
        pass
    
    return f.copy(), M


def golden_section_line_search(f, g, M, max_iter=60):
    """
    Find alpha in [0,1] minimizing h(alpha) = max_k (((1-a)*f + a*g) * ((1-a)*f + a*g))[k].
    f and g are both on the simplex, so interpolation stays on the simplex.
    """
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_lo, a_hi = 0.0, 1.0
    
    def h(alpha):
        ft = (1.0 - alpha) * f + alpha * g
        return fast_convolve(ft).max()
    
    a1 = a_hi - gr * (a_hi - a_lo)
    a2 = a_lo + gr * (a_hi - a_lo)
    h1 = h(a1)
    h2 = h(a2)
    
    for _ in range(max_iter):
        if h1 < h2:
            a_hi = a2
            a2 = a1
            h2 = h1
            a1 = a_hi - gr * (a_hi - a_lo)
            h1 = h(a1)
        else:
            a_lo = a1
            a1 = a2
            h1 = h2
            a2 = a_lo + gr * (a_hi - a_lo)
            h2 = h(a2)
    
    alpha_opt = (a_lo + a_hi) / 2.0
    return alpha_opt


def subgradient_step(f, beta=100.0, k_top=30):
    """Compute subgradient direction for min max_k (a*a)[k] using LSE weights."""
    n = len(f)
    conv = fast_convolve(f)
    
    # LSE weights
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        return None
    
    # Use only top-k peaks for efficiency
    k_use = min(k_top, 2 * n - 1)
    peak_idx = np.argpartition(conv, -k_use)[-k_use:]
    w = exp_bc[peak_idx]
    w_sum = w.sum()
    if w_sum < 1e-300:
        return None
    w = w / w_sum
    
    # Subgradient: sum_k w_k * 2 * a[k-j] (reversed and zero-padded)
    grad = np.zeros(n)
    for idx, k in enumerate(peak_idx):
        k = int(k)
        wi = w[idx]
        if wi < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        grad[js] += wi * 2.0 * f[k - js]
    
    return grad


def optimize_single_length(f_init, time_limit, verbose=False):
    """
    Optimize a normalized non-negative sequence using LP-based FW + subgradient.
    f_init should be normalized (sum=1, all >= 0).
    Returns (best_f, best_obj).
    """
    f = np.maximum(f_init.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    use_topk = n > 1200  # Use top-K for very large n
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 8:
            break
        
        # Determine LP time budget
        lp_time = min(remaining * 0.12, 45.0)
        if lp_time < 2.0:
            lp_time = remaining * 0.5
        
        # Solve LP for direction
        if use_topk:
            g, t_opt = solve_lp_topk(f, M, top_k=min(80, 2 * n - 1), time_limit=lp_time)
        else:
            g, t_opt = solve_lp_direction(f, M, time_limit=lp_time)
        
        # Check if LP gives useful direction
        if t_opt >= M * (1.0 - 1e-9):
            no_improve += 1
            if no_improve <= 15:
                # Try subgradient step
                grad = subgradient_step(f, beta=50.0 + 20.0 * no_improve, k_top=40)
                if grad is not None:
                    gnorm = np.linalg.norm(grad)
                    if gnorm > 1e-15:
                        # Projected gradient step with backtracking
                        for step in [0.05, 0.03, 0.015, 0.007, 0.003, 0.001]:
                            f_new = project_simplex(f - step * grad / gnorm)
                            conv_new = fast_convolve(f_new)
                            if conv_new.max() < M - 1e-14:
                                f = f_new
                                M = conv_new.max()
                                if M < best_obj - 1e-14:
                                    best_obj = M
                                    best_f = f.copy()
                                    no_improve = 0
                                break
            elif no_improve <= 40:
                # Small perturbation
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                if M < best_obj:
                    best_obj = M
                    best_f = f.copy()
            else:
                # Larger perturbation and reset
                rng = np.random.default_rng(it * 7919 + 13)
                # Frequency domain perturbation
                F = np.fft.rfft(f)
                mag = np.abs(F)
                phase = np.angle(F)
                scale = 0.02 + 0.01 * min(1.0, (no_improve - 40) / 30.0)
                mag_pert = np.maximum(mag * (1.0 + rng.standard_normal(len(mag)) * scale), 0)
                F_new = mag_pert * np.exp(1j * phase)
                f_new = np.fft.irfft(F_new, n)
                f_new = np.maximum(f_new, 0)
                s_new = f_new.sum()
                if s_new > 1e-15:
                    f_new /= s_new
                else:
                    f_new = best_f.copy()
                f = f_new
                conv = fast_convolve(f)
                M = conv.max()
                if M < best_obj:
                    best_obj = M
                    best_f = f.copy()
                no_improve = 0
            continue
        
        # LP gave a good direction, do line search
        alpha_opt = golden_section_line_search(f, g, M, max_iter=50)
        
        if alpha_opt < 1e-6:
            no_improve += 5
            continue
        
        # Update
        f = (1.0 - alpha_opt) * f + alpha_opt * g
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-15:
            f /= s
        else:
            f = best_f.copy()
            M = best_obj
            continue
        
        conv = fast_convolve(f)
        M = conv.max()
        
        if M < best_obj - 1e-13:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 20 == 0:
                elapsed = time.monotonic() - t0
                print(f"      it={it}, M={M:.10f}, best_obj={best_obj:.10f}, t={elapsed:.0f}s")
        else:
            no_improve += 1
            if no_improve >= 10:
                # Revert to best and try again with perturbation
                f = best_f.copy()
                M = best_obj
                no_improve = 5
    
    return best_f, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length using spline-like method."""
    old_n = len(f)
    if new_n == old_n:
        return f.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def try_sequence(a, best_tracker):

        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score
