"""
Hybrid optimization for minimum peak auto-convolution of non-negative sequences.
Uses FFT-based gradient computation, adaptive log-sum-exp smoothing,
LP oracle for direction finding, frequency-domain shaping, and local search.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_fast(a, p, fft_size):
    """Compute gradient of sum_k p[k]*(a*a)[k] with respect to a using FFT.
    
    grad[j] = 2 * sum_l p[j+l] * a[l] (cross-correlation of p and a)
    where the sum is over valid l.
    """
    n = len(a)
    # We need: grad[j] = 2 * sum_{l=0}^{n-1} p[j+l] * a[l] for j = 0, ..., n-1
    # This is the cross-correlation of p (length 2n-1) and a (length n), evaluated at lags 0 to n-1.
    # cross[j] = sum_l p[j+l] * a[l]
    # This equals the correlation, which can be computed as:
    # FFT-based: IDFT(conj(FFT(p_padded)) * FFT(a_padded))[0:n]
    # Or equivalently, it's the convolution of a with reverse(p_padded):
    # conv(a, reverse(p))[j + (2n-2)] ... let me think.
    
    # Actually, cross[j] = sum_{l=0}^{n-1} a[l] * p[j+l]
    # = sum_{l=0}^{n-1} a[l] * p[j+l]
    # Let m = j+l, then l = m-j, and m ranges from j to j+n-1.
    # cross[j] = sum_{m=j}^{j+n-1} a[m-j] * p[m]
    
    # This is the cross-correlation of a and p at lag j.
    # In terms of convolution: cross[j] = (a * reverse(p))[j + (2n-2)] ... no.
    
    # Let's use: cross[j] = sum_l a[l] * p[j+l]
    # If we define q = reverse(p) (length 2n-1), then q[k] = p[2n-2-k].
    # (a * q)[k] = sum_l a[l] * q[k-l] = sum_l a[l] * p[2n-2-k+l]
    # We want sum_l a[l] * p[j+l]. So 2n-2-k+l = j+l → 2n-2-k = j → k = 2n-2-j.
    # So cross[j] = (a * q)[2n-2-j] = (a * reverse(p))[2n-2-j]
    
    # Alternatively, cross[j] = (reverse(a) * p)[2n-2-j] ... let me verify.
    # (reverse(a) * p)[k] = sum_l reverse(a)[l] * p[k-l] = sum_l a[n-1-l] * p[k-l]
    # With m = n-1-l: = sum_m a[m] * p[k-n+1+m]
    # We want this to equal sum_m a[m] * p[j+m], so k-n+1 = j, k = j+n-1.
    # So cross[j] = (reverse(a) * p)[j+n-1]
    
    # Let's use the second form. Compute conv(reverse(a), p) and take indices n-1 to 2n-2.
    # conv(reverse(a), p) has length n + (2n-1) - 1 = 3n-1.
    # cross[j] = conv(reverse(a), p)[j + n - 1] for j = 0, ..., n-1.
    
    ra = a[::-1]
    # Use FFT for the convolution
    conv_size = n + (2*n - 1) - 1  # = 3n - 1
    fft_sz = 1 << (conv_size - 1).bit_length()
    RA = np.fft.rfft(ra, fft_sz)
    P = np.fft.rfft(p, fft_sz)
    conv = np.fft.irfft(RA * P, fft_sz)
    # cross[j] = conv[j + n - 1]
    cross = conv[n-1:n-1+n]
    return 2.0 * cross


def compute_gradient_direct(a, p):
    """Direct computation of gradient (for verification or small n)."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        # grad[j] = 2 * sum_l p[j+l] * a[l] for l = 0, ..., n-1 (with j+l < 2n-1)
        l_max = min(n-1, 2*n-2-j)
        if l_max >= 0:
            grad[j] = 2.0 * np.dot(a[:l_max+1], p[j:j+l_max+1])
    return grad
