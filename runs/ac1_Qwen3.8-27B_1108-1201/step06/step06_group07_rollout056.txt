       n = len(v)

"""
Minimizes 2*n*max(auto_conv(a))/(sum(a))^2 using:
- SCA (Successive Convex Approximation) with LP subproblems
- Log-domain Adam gradient optimization
- Multi-n exploration with diverse initializations
- Adaptive active set management and diversified restarts

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_sca_lp(a_current, peaks, conv_vals):
    """
    Solve the SCA LP subproblem:
      min t
      s.t. 2*sum_i a_current[k-i]*a[i] - t <= conv[k]  for k in peaks
           sum(a) = 1
           a >= 0
    Returns (a_new, t_new) or (None, None) on failure.
    """
    n = len(a_current)
    m = len(peaks)

    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)

    for r, k in enumerate(peaks):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            # a_current[k - i] for i in [i_start, i_end]
            # = a_current[k-i_end : k-i_start+1][::-1]
            A_ub[r, i_start:i_end + 1] = 2.0 * a_current[k - i_end:k - i_start + 1][::-1]
        A_ub[r, n] = -1.0
        b_ub[r] = conv_vals[k]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            a_new = np.maximum(res.x[:n], 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
            return a_new, float(res.x[n])
    except Exception:
        pass
    return None, None


def run_sca(a_init, time_limit, K=10, verbose=False):
    """Run SCA to minimize max(auto_conv(a)) over the simplex."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.05:
            break

        conv = fast_convolve(a)
        max_val = conv.max()

        # Adaptive K
        K_curr = min(K + no_improve // 15, 2 * n - 1)
        K_curr = max(K_curr, 3)

        if K_curr >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -K_curr)[-K_curr:]

        a_new, t_new = solve_sca_lp(a, peaks, conv)

        if a_new is not None:
            accepted = False
            for alpha in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
                a_blend = (1.0 - alpha) * a + alpha * a_new
                a_blend = np.maximum(a_blend, 0.0)
                sb = a_blend.sum()
                if sb < 1e-15:
                    continue
                a_blend = a_blend / sb
                val = fast_convolve(a_blend).max()
                if val < best_obj - 1e-15:
                    best_obj = val
                    best_a = a_blend.copy()
                    a = a_blend
                    no_improve = 0
                    accepted = True
                    break

            if not accepted:
                no_improve += 1
                if no_improve % 3 == 0:
                    a = a_new.copy()
        else:
            no_improve += 1

        if no_improve > 40:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
            scale = 0.002 + 0.004 * min(1.0, no_improve / 80.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    SCA it={it}, obj={best_obj:.10f}, no_improve={no_improve}")

    return best_a, best_obj


def run_adam_log(a_init, time_limit, lr=0.005, top_k=5, verbose=False):
    """Log-domain Adam optimization minimizing max(auto_conv)."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 1e-12)
    a = a / a.sum()
    w = np.log(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        max_val = conv.max()

        # Use top-k peaks with softmax weights for smoother gradient
        k_sel = min(top_k, len(conv))
        if k_sel < len(conv):
            peak_idx = np.argpartition(conv, -k_sel)[-k_sel:]
        else:
            peak_idx = np.arange(len(conv))
        pv = conv[peak_idx]
        pw = pv / pv.sum()

        # Compute gradient
        grad = np.zeros(n)
        for idx in range(len(peak_idx)):
            wt = pw[idx]
            if wt < 1e-12:
                continue
            k = int(peak_idx[idx])
            conv_k = conv[k]
            for j in range(n):
                k_j = k - j
                if 0 <= k_j < n:
                    grad[j] += wt * 2.0 * a[j] * (a[k_j] - conv_k)
                else:
                    grad[j] += wt * (-2.0 * a[j] * conv_k)

        # Center gradient (maintain sum constraint approximately)
        grad = grad - grad.mean()

        # Adam update
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad * grad
        m_hat = m / (1.0 - 0.9**t)
        v_hat = v / (1.0 - 0.999**t)
        w = w - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        a_new = np.exp(w - np.max(w))
        a_new = a_new / a_new.sum()

        new_obj = fast_convolve(a_new).max()
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a_new
            no_improve += 1

        if no_improve > 30:
            lr *= 0.7
            if lr < 1e-5:
                lr = 0.005
                no_improve = 0
                m[:] = 0
                v[:] = 0

        if t % 200 == 0:
            w = w - np.max(w)

        if verbose and it % 100 == 0:
            pass

    return best_a, best_obj


def run_hybrid(a_init, time_limit, verbose=False):
    """Alternate between SCA and Adam for complementary progress."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()

    t0 = time.monotonic()

    phase_time = min(30.0, time_limit / 4.0)

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < phase_time:
            phase_time = remaining
        if phase_time < 1.0:
            break

        if phase % 2 == 0:
            a_opt, obj = run_sca(a, time_limit=phase_time, K=12)
        else:

        if obj < best_obj:

            best_a = a_opt.copy()
            a = a_opt
        else:

            a_blend = 0.5 * a + 0.5 * a_opt

            a_blend = a_blend / a_blend.sum()
            blend_obj = fast_convolve(a_blend).max()
            if blend_obj < best_obj:
                best_obj = blend_obj
                best_a = a_blend.copy()
            a = a_opt

        phase += 1
        phase_time = min(40.0, remaining / 3.0)

def generate_inits(n, parent, rng):
    """Generate diverse initializations for given n."""
    inits = []

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    # Edge-heavy
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)**1.5
    inits.append(edge_w / edge_w.sum())

    # Two-peak
    g2 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n))**2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n))**2)
    inits.append(g2 / g2.sum())

    # Sinusoidal modulation
    t = x / max(n - 1, 1)
    asym = (1.0 + 0.3 * np.sin(4 * np.pi * t))**2 + 0.05
    inits.append(asym / asym.sum())

    return inits

            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(15, remaining_after - 25)
            if t_alloc < 3:
                break
            print(f"  Restart scale={scale}, t={t_alloc:.0f}s")
            a_opt, _ = run_sca(f_pert, time_limit=t_alloc, K=12)
            best_seq, best_score, _ = try_and_update(a_opt, best_seq, best_score, f"restart_{scale}")

    print(f"  P6 done: best={best_score:.8f}, t={time.monotonic()-t6:.0f}s")

    # === Phase 7: Final polish ===
    print("\n=== Phase 7: Final Polish ===")
    t7 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            t_alloc = min(remaining, 40)
            print(f"  Final SCA K=30 t={t_alloc:.0f}s")
            a_opt, _ = run_sca(best_seq, time_limit=t_alloc, K=30)
            best_seq, best_score, _ = try_and_update(a_opt, best_seq, best_score, "final_sca")

        remaining = deadline - time.monotonic() - 3
        if remaining > 5:
            a_opt2, _ = run_adam_log(best_seq, time_limit=min(remaining, 15), lr=0.0005, top_k=10)
            best_seq, best_score, _ = try_and_update(a_opt2, best_seq, best_score, "final_adam")

    print(f"  P7 done: best={best_score:.8f}, t={time.monotonic()-t7:.0f}s")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
