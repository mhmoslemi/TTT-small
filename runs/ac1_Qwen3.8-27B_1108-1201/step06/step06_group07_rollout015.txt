"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Smooth optimization (log-sum-exp) with projected gradient
- LP-based SCA for refinement
- Multi-n exploration with diverse initializations
- Chebyshev/Fourier structured starting points

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_correlate(w, a):
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.fft(w, fft_size)
    A = np.fft.fft(a, fft_size)
    R = np.fft.ifft(W * np.conj(A)).real
    return R[:n]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_objective(a, tau):
    conv = fast_convolve(a)
    m = conv.max()
    return 2.0 * len(a) * (m + tau * np.log(np.sum(np.exp((conv - m) / tau))))

def smooth_gradient(a, tau):
    conv = fast_convolve(a)
    m = conv.max()
    w = np.exp((conv - m) / tau)
    w /= w.sum()
    R = cross_correlate(w, a)
    return 4.0 * len(a) * R

def projected_gradient_smooth(a0, time_limit, tau_init=0.01, verbose=False):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    tau = tau_init
    lr = 1.0 / (4.0 * n * n)  # Initial step size estimate
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        grad = smooth_gradient(a, tau)
        
        # Projected gradient step with backtracking
        a_try = a
        obj_try = fast_convolve(a_try).max()
        alpha = lr
        for _ in range(20):
            candidate = project_simplex(a - alpha * grad)
            obj = fast_convolve(candidate).max()
            if obj < obj_try:
                a_try = candidate
                obj_try = obj
                break
            alpha *= 0.5
        
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_try
        
        # Adapt tau and lr
        if no_improve > 50:
            tau *= 0.9
            lr = 1.0 / (4.0 * n * n)
            no_improve = 0
        
        if it % 100 == 0 and verbose:
            print(f"    PG it={it}, obj={best_obj:.10f}, tau={tau:.6f}, lr={lr:.8f}")
    
    return best_a, best_obj

def sca_optimize(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    # Precompute LP matrix
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    row_idx = np.array(rows, dtype=np.int32)
    col_idx = np.array(cols, dtype=np.int32)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        # Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0
        c = -np.ones(n)
        k_arr = row_idx
        i_arr = col_idx
        vals = f[k_arr - i_arr]
        A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        g = None
        try:
            res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True})
            if res.success and res.x is not None:
                g = np.maximum(res.x, 0.0)
                if g.sum() < 1e-15:
                    g = None
        except Exception:
            pass
        
        if g is None:
            M *= 1.02
            no_improve += 1
            if no_improve > 30:
                M = fast_convolve(f).max() * 1.005
                no_improve = 0
            continue
        
        g = g / g.sum()
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 20:
                rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
                f = project_simplex(f + rng.standard_normal(n) * 0.005)
                M = fast_convolve(f).max() * 1.005
                no_improve = 0
            continue
        
        # Line search: find best blend
        best_val = fast_convolve(f).max()
        best_ft = f.copy()
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_ft = ft.copy()
        
        if best_val < best_obj - 1e-14:
            f = best_ft
            best_obj = best_val
            best_f = f.copy()
            M = best_val * 0.999
            no_improve = 0
        else:
            no_improve += 1
            M *= 0.998
        
        if no_improve > 50:
            rng = np.random.default_rng((it * 31831 + 7) % (2**31 - 1))
            scale = 0.003 + 0.005 * min(1.0, (no_improve - 50) / 50.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max() * 1.005
            no_improve = 0
        
        if verbose and it % 50 == 0:
            print(f"    SCA it={it}, obj={best_obj:.10f}, M={M:.10f}")
    
    return best_f, best_obj

def mirror_descent(a0, time_limit, eta_init=0.05):
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    eta = eta_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        conv = fast_convolve(a)
        max_val = conv.max()
        
        # Softmax over top peaks
        top_k = min(20, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        pw = np.exp(pv / (0.01 * max_val + 1e-10))
        pw /= pw.sum()
        
        # Subgradient using all peaks weighted
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            w = pw[idx]
            if w < 1e-12:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je + 1] += w * 2.0 * a[k - je:k - js + 1][::-1]
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        # Line search
        best_candidate = a
        best_val = max_val
        for alpha in [0.3, 0.5, 0.7, 1.0]:
            cand = (1 - alpha) * a + alpha * a_new
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_candidate = cand
        
        a = best_candidate
        obj = fast_convolve(a).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 30:
            eta *= 0.9
        if eta < 1e-4:
            eta = eta_init
            no_improve = 0
        
        if no_improve > 200:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            eta = eta_init
            no_improve = 0
    
    return best_a, best_obj

def generate_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    t = x / (n - 1) if n > 1 else np.array([0.5])
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Raised cosine
    rc = (1 + np.cos(2 * np.pi * t)) / 2
    rc = np.maximum(rc, 1e-6)
    inits.append(rc / rc.sum())
    
    # Sinusoidal
    for freq in [1, 2, 3, 4]:
        for amp in [0.1, 0.2, 0.3, 0.5]:
            s = 1.0 + amp * np.cos(2 * np.pi * freq * t)
            s = np.maximum(s, 1e-6)
            inits.append(s / s.sum())
    
    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        for width in [0.05, 0.1, 0.2, 0.3]:
            g = np.exp(-0.5 * ((t - center) / width) ** 2)
            inits.append(g / g.sum())
    
    # Dual峰 (two peaks)
    for c1, c2, w in [(0.25, 0.75, 0.08), (0.2, 0.8, 0.1), (0.3, 0.7, 0.15)]:
        dp = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
        inits.append(dp / dp.sum())
    
    # Edge-heavy
    edge = 1.0 / (0.1 + (t * (1 - t)) ** 0.5)
    inits.append(edge / edge.sum())
    
    # Polynomial
    for p in [0.5, 1.0, 1.5, 2.0]:
        poly = (t * (1 - t)) ** p + 0.01
        inits.append(poly / poly.sum())
    
    # Random Dirichlet
    for alpha in [0.1, 0.5, 1.0, 2.0, 5.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha))
    
    # Chebyshev-based
    for k in [2, 4, 6]:
        for amp in [0.1, 0.2, 0.3]:
            # T_k(2t-1)
            tk = np.cos(k * np.arccos(np.clip(2 * t - 1, -1, 1)))
            s = 1 + amp * tk
            s = np.maximum(s, 1e-6)
            inits.append(s / s.sum())
    
    return inits

def subgradient_ls(a0, time_limit, top_k=10):
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = 1.0 / (8.0 * n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        conv = fast_convolve(a)
        
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            weights = np.exp(pv / (0.01 * pv.max() + 1e-10))
            weights /= weights.sum()
        
        # Compute subgradient via FFT
        # grad[i] = 2 * sum_k w[k] * a[k-i] * 2 (factor of 2 from the quadratic)
        # Actually: d(conv[k])/da[i] = 2*a[k-i], so subgradient = sum_k w[k] * 2*a[k-i]
        # Using cross-correlation:
        R = cross_correlate(weights, a)
        grad = 2.0 * R
        
        # Projected gradient step
        direction = project_simplex(a - lr * grad) - a
        
        # Line search
        lo, hi = 0.0, 1.0
        best_alpha = 0.0
        best_val = conv.max()
        for _ in range(15):
            mid = (lo + hi) / 2.0
            cand = a + mid * direction
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                hi = mid
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
                lo = mid
            else:
                hi = mid
        
        if best_alpha > 1e-12:
            a = a + best_alpha * direction
            a = np.maximum(a, 0)
            a = a / a.sum()
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        lr *= 0.998
        if lr < 1e-5:
            lr = 1.0 / (8.0 * n)
        
        if no_improve > 100:
            rng = np.random.default_rng((it * 1299709 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0
            lr = 1.0 / (8.0 * n)
    
    return best_a, best_obj
