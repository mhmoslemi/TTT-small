import numpy as np
from scipy.optimize import minimize, linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lbgfs_func_and_grad(z, n, T):
    a = np.exp(z - z.max())
    a = a / a.sum()
    conv = fast_convolve(a)
    m = conv.max()
    shifted = (conv - m) / T
    amax = shifted.max()
    ex = np.exp(shifted - amax)
    s = ex.sum()
    sm = m + T * (amax + np.log(s))
    
    # Gradient of smooth max w.r.t. a
    p = ex / s  # softmax weights over conv values
    grad_a = np.zeros(n)
    # For efficiency, only use top peaks

        je = min(n - 1, int(j))
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w * a[int(j) - idx]
    
    # Chain rule through softmax
    grad_fa = np.dot(grad_a, a)
    grad_z = a * (grad_a - grad_fa)
    grad_z *= 2.0 * n
    obj = 2.0 * n * sm
    return obj, grad_z

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiple structural initializations (symmetric polynomials, Gaussians, semicircles)
2. L-BFGS-B with softmax parameterization and adaptive smooth max (log-sum-exp)
3. Active-set Newton KKT solver with automatic active set management
4. Symmetry-reduced optimization
5. Multi-n exploration with interpolation
6. Frank-Wolfe with LP oracle for verification

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_obj(z, n, T):
    a = np.exp(z - z.max())
    a = a / a.sum()
    conv = fast_convolve(a)
    m = conv.max()
    shifted = (conv - m) / T
    amax = shifted.max()
    ex = np.exp(shifted - amax)
    s = ex.sum()
    sm = m + T * (amax + np.log(s))
    return sm


def smooth_max_obj_grad(z, n, T):
    a = np.exp(z - z.max())
    a = a / a.sum()
    conv = fast_convolve(a)
    m = conv.max()
    shifted = (conv - m) / T
    amax = shifted.max()
    ex = np.exp(shifted - amax)
    s = ex.sum()
    sm = m + T * (amax + np.log(s))
    p = ex / s
    top_k = min(80, len(p))
    top_idx = np.argpartition(p, -top_k)[-top_k:]
    grad_a = np.zeros(n)
    for j in top_idx:
        w = p[j]
        if w < 1e-15:
            continue
        ji = int(j)
        js = max(0, ji - n + 1)
        je = min(n - 1, ji)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w * a[ji - idx]
    grad_fa = np.dot(grad_a, a)
    grad_z = a * (grad_a - grad_fa)
    obj = 2.0 * n * sm
    grad_z *= 2.0 * n
    return obj, grad_z


def lbgfs_optimize(a0, time_limit, T_init=0.005):

def make_symmetric_init(n, kind='cosine', p=2.0):
    x = np.linspace(0, 1, n)
    xc = x - 0.5
    if kind == 'cosine':
        a = 0.5 * (1.0 + np.cos(np.pi * x))
    elif kind == 'gaussian':
        a = np.exp(-p * (2.0 * xc) ** 2)
    elif kind == 'semicircle':
        a = np.sqrt(np.maximum(0.0, 1.0 - (2.0 * xc) ** 2))
    elif kind == 'poly':
        a = np.maximum(0.0, 1.0 - (2.0 * xc) ** p)
    elif kind == 'bump':
        a = np.exp(-p * (2.0 * xc) ** 2) * (1.0 - (2.0 * xc) ** 2) ** 2
        a = np.maximum(a, 0)
    elif kind == 'box':
        frac = min(0.4, 0.5 - 0.05 * p)
        a = np.where(np.abs(xc) < frac, 1.0, 0.0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s

def active_set_newton(a0, time_limit, peak_tol=1e-4, max_iter=200):

            break
        conv = fast_convolve(a)
        M = conv.max()

        support = np.where(a > 1e-10)[0]
        if len(active_peaks) <= 1 or len(support) == 0:
            break
        p = len(active_peaks)
        m = len(support)
        A_sys = np.zeros((m, p + 1))
        for i, j in enumerate(support):
            A_sys[i, -1] = 1.0
            for k_idx, k in enumerate(active_peaks):
                idx = k - j
                if 0 <= idx < n:
                    A_sys[i, k_idx] = a[idx]
        try:
            U, S_vals, Vt = np.linalg.svd(A_sys)
            sol = Vt[-1, :]

            break
        mu_vals = np.maximum(mu_vals, 0)

        if mu_sum < 1e-15:
            break
        mu_vals /= mu_sum
        grad = np.zeros(n)
        for k_idx, k in enumerate(active_peaks):
            w = mu_vals[k_idx]
            if w < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] -= w * a[k - idx_arr]
        best_candidate = a

            break
    return best_a, best_obj


def solve_lp_oracle(f, M):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'max_time': 20})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None
