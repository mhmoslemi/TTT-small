This can be solved by binary search on the target level T:
- For each m, find if there exists t such that `c_m + 2t*(d*a)_m + t^2*(d*d)_m <= T`
- The feasible t for each m is an interval (possibly empty)
- We need the intersection of all intervals to be non-empty

This gives us an exact (or near-exact) optimal step size!

2. Compute the "base" convolution c = a*a
3. Compute the cross term: d*a (convolution of d with a)
4. Compute the quadratic term: d*d (convolution of d with d)
5. For the top K peaks, find the optimal t using binary search on T
6. Update a = a + t*d (projected to simplex)

```python
def build_lp_matrix(f, constraint_indices):
    n = len(f)
    rows, cols, vals = [], [], []
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end+1)
            rows.extend([idx]*len(js))
            cols.extend(js.tolist())
            vals.extend(f[k-js].tolist())
    if not rows:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(len(constraint_indices), n))

    n = len(f)
    
    c_aa = fast_convolve(f)
    M = c_aa.max()
    best_M = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break
        
        lp_time = min(10, max(2, rem * 0.04))
        g = solve_lp_direction(f, M, top_k=top_k, time_limit=lp_time)
        if g is None:
            no_improve += 3
            if no_improve > 40:
                # Reset with perturbation
                rng = np.random.default_rng(it*7919)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                c_aa = fast_convolve(f)
                M = c_aa.max()
                no_improve = 0
            continue
        
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it*104729)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.008)
                c_aa = fast_convolve(f)
                M = c_aa.max()
                no_improve = 0
            continue
        
        # Compute line search ingredients
        c_ag = cross_convolve(f, g)
        c_gg = fast_convolve(g)
        
        t_opt = exact_line_search(f, g, c_aa, c_ag, c_gg)
        
        if t_opt < 1e-8:
            no_improve += 1
            continue
        
        f_new = (1-t_opt)*f + t_opt*g
        f_new = np.maximum(f_new, 0.0)
        f_new /= f_new.sum()
        
        c_aa = fast_convolve(f_new)
        M = c_aa.max()
        
        if M < best_M - 1e-14:
            best_M = M
            best_f = f_new.copy()
            f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            f = f_new.copy()  # Still move, even if not improving best
    
    return best_f, best_M

def pairwise_coordinate_descent(f0, time_limit, n_top_peaks=30):
    """Coordinate descent with pairwise updates for the top peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    c_aa = fast_convolve(f)
    best_M = c_aa.max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        rem = time.monotonic() and (time_limit - (time.monotonic() - t0))
        if rem < 1:
            break
        
        # Get top peaks
        k = min(n_top_peaks, 2*n-1)
        peaks = np.argpartition(c_aa, -k)[-k:]
        
        improved_this_round = False
        
        for m in peaks:
            if time.monotonic() - t0 > time_limit:
                break
            
            m = int(m)
            lo = max(0, m - n + 1)
            hi = min(n-1, m)
            if lo >= hi:
                continue
            
            # c[m] = sum_{i=lo}^{hi} f[i]*f[m-i]
            # Find the pair (i, j) with i+j=m that maximizes f[i]*f[j]
            i_vals = np.arange(lo, hi+1)
            j_vals = m - i_vals
            prods = f[i_vals] * f[j_vals]
            
            # Try to reduce the peak by moving mass from the largest contributor
            # to a position that doesn't contribute to this peak
            max_idx = np.argmax(prods)
            i_src = i_vals[max_idx]
            j_src = j_vals[max_idx]
            
            if i_src == j_src:
                # Can't split (it's a squared term), try moving from i_src to elsewhere
                # Find a destination that minimizes increase in other peaks
                # Simple heuristic: move to a random position
                rng = np.random.default_rng(hash((m, int(time.monotonic()*1000))) % 2**31)
                candidates = rng.choice(n, size=min(20, n), replace=False)
            else:
                # Move from the larger of f[i_src], f[j_src] to a position far from peak
                if f[i_src] >= f[j_src]:
                    src = i_src
                else:
                    src = j_src
                # Destination: positions that don't contribute to peak m
                # i.e., positions p such that p + (m-p) = m is not what we want
                # Actually, if we move from position src to position dest,
                # the peak m changes by: delta * (f[m-src] - f[m-dest]) + higher order
                # We want f[m-src] > f[m-dest] to reduce the peak
                f_at_m_minus_src = f[m-src] if 0 <= m-src < n else 0
                # Find dest that minimizes f[m-dest]
                valid_dest = []
                for p in range(n):
                    if p == src:
                        continue
                    f_at_m_minus_p = f[m-p] if 0 <= m-p < n else 0
                    valid_dest.append((f_at_m_minus_p, p))
                valid_dest.sort()
                candidates = [p for _, p in valid_dest[:10]]
            
            for dest in candidates:
                if dest == src:
                    continue
                # Try different fractions of transfer
                max_delta = min(f[src], 0.5 * f[src])  # transfer at most 50%
                for frac in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]:
                    delta = frac * f[src]
                    if delta < 1e-15 or delta > f[src]:
                        continue
                    f_trial = f.copy()
                    f_trial[src] -= delta
                    f_trial[dest] += delta
                    if f_trial[src] < -1e-15:
                        continue
                    f_trial = np.maximum(f_trial, 0)
                    f_trial /= f_trial.sum()
                    c_new = fast_convolve(f_trial)
                    M_new = c_new.max()
                    if M_new < best_M - 1e-14:
                        best_M = M_new
                        best_f = f_trial.copy()
                        f = f_trial.copy()
                        c_aa = c_new
                        improved_this_round = True
                        break
                if improved_this_round:
                    break
        
        if not improved_this_round:
            # Try random pairs
            rng = np.random.default_rng()
            for _ in range(50):
                if time.monotonic() - t0 > time_limit:
                    break
                src = rng.integers(n)
                dest = rng.integers(n)
                if src == dest:
                    continue
                delta = 0.02 * f[src]
                if delta < 1e-15:
                    continue
                f_trial = f.copy()
                f_trial[src] -= delta
                f_trial[dest] += delta
                f_trial = np.maximum(f_trial, 0)
                f_trial /= f_trial.sum()
                c_new = fast_convolve(f_trial)
                if c_new.max() < best_M - 1e-14:
                    best_M = c_new.max()
                    best_f = f_trial.copy()
                    f = f_trial.copy()
                    c_aa = c_new
    
    return best_f, best_M
```

`= f*f - δ*(e_src*f + f*e_src) + δ*(e_dest*f + f*e_dest) + δ^2*(e_src*e_src - e_src*e_dest - e_dest*e_src + e_dest*e_dest)`

`(e_src * f)[m] = f[m-src]` (if valid)
`(e_dest * f)[m] = f[m-dest]` (if valid)
`(e_src * e_src)[m] = [m = 2*src]`
`(e.dest * e_dest)[m] = [m = 2*dest]`

```python
def compute_delta_c(f, src, dest, delta):
    """Compute the change in convolution when f[src]-=delta, f[dest]+=delta."""
    n = len(f)
    size = 2*n - 1
    
    # -2δ*f convolved with e_src → -2δ * f shifted by src
    # +2δ*f convolved with e_dest → +2δ * f shifted by dest
    
    delta_c = np.zeros(size)
    
    # Contribution: +2δ * f[m-dest] for valid m-dest
    lo_d = dest
    hi_d = dest + n - 1
    delta_c[lo_d:hi_d+1] += 2*delta*f
    
    # Contribution: -2δ * f[m-src] for valid m-src
    lo_s = src
    hi_s = src + n - 1
    delta_c[lo_s:hi_s+1] -= 2*delta*f
    
    # Quadratic terms
    delta_c[2*src] += delta*delta
    delta_c[2*dest] += delta*delta
    delta_c[src+dest] -= 2*delta*delta
    
    return delta_c
```
