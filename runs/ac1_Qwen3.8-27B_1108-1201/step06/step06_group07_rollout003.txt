def sca_optimize_v2(f0, time_limit, M_init=None, verbose=False):
    """
    Improved SCA with:
    - All-peak constraints (safe)
    - Adaptive M with backtracking
    - Better perturbation strategy
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max() * 1.001 if M_init is None else max(M_init, conv.max())
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    M_history = []
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        # Build LP: max sum(g) s.t. 2*a[k-j]*g[j] summed over j <= M for all k, g >= 0
        # The constraint matrix A has A[k,j] = 2*a[k-j] (with a[x]=0 for x out of [0,n-1])
        
        # For efficiency, build sparse matrix
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                rows.append(k)
                cols.append(j)
                vals.append(2.0 * f[k - j])
        
        A_sparse = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        b_ub = np.full(2 * n - 1, M)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, 
                         method='highs', options={'presolve': True, 'time_limit': max(0.1, remaining*0.05)})
            if not (res.success and res.x is not None):
                M *= 1.01
                no_improve += 1
                if no_improve > 30:
                    M = fast_convolve(f).max() * 1.005
                    no_improve = 0
                continue
        except Exception:
            M *= 1.01
            no_improve += 1
            continue
        
        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        
        if gs < 1e-15:
            M *= 1.005
            no_improve += 1
            continue
        
        g = g / gs  # Normalize to sum=1
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                # Try decreasing M more aggressively
                M *= 0.98
                no_improve = 0
            if M < fast_convolve(f).max() * 0.9:
                # Perturb
                rng = np.random.default_rng((it * 31831 + 7) % (2**31 - 1))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                M = fast_convolve(f).max() * 1.005
                no_improve = 0
            continue
        
        # Line search along (1-alpha)*f + alpha*g
        best_val = fast_convolve(f).max()
        best_ft = f.copy()
        
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_ft = ft.copy()
        
        if best_val < best_obj - 1e-13:
            f = best_ft
            best_obj = best_val
            best_f = f.copy()
            M = best_val * 0.999
            no_improve = 0
        else:
            no_improve += 1
            M *= 0.995
        
        if no_improve > 50:
            rng = np.random.default_rng((it * 1299709 + 3) % (2**31 - 1))
            scale = 0.002 + 0.005 * min(1.0, (no_improve - 50) / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max() * 1.005
            no_improve = 0
        
        if verbose and it % 100 == 0:
            print(f"    SCA2 it={it}, obj={best_obj:.10f}, M={M:.10f}")
    
    return best_f, best_obj


def spectral_shaping(a0, time_limit):
    """
    Frequency-domain based shaping: identify frequencies that contribute
    to the autocorrelation peak and reduce them.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]
        
        # The peak at lag k is influenced by the "energy" of a at frequencies
        # that satisfy certain phase conditions. For the center peak (k ≈ n-1),
        # it's related to the DC component and low frequencies.
        
        # Strategy: reduce the contribution of the peak by modifying a in a
        # way that reduces a[j]*a[k-j] for the j's that contribute most.
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        
        # Contribution of pair (j, k-j) to the peak
        contrib = a[js:je+1] * a[peak_k-je:peak_k-js+1][::-1]
        
        # Identify the most contributing positions
        top_contrib = np.argsort(contrib)[::-1][:max(1, len(contrib)//4)]
        
        # Reduce these positions and increase others
        reduction = 0.02
        a_new = a.copy()
        for idx in top_contrib:
            j = js + idx
            a_new[j] *= (1 - reduction)
        
        # Redistribute the mass to positions far from the peak span
        outside = [j for j in range(n) if j < js or j > je]
        if outside:
            total_removed = a.sum() - a_new.sum()
            # Add to outside positions proportional to their current value
            ext_vals = a_new[outside]
            ext_sum = ext_vals.sum()
            if ext_sum > 1e-15:
                for i, j in enumerate(outside):
                    a_new[j] += total_removed * ext_vals[i] / ext_sum
            else:
                for j in outside:
                    a_new[j] += total_removed / len(outside)
        
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        new_obj = fast_convolve(a_new).max()
        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            rng = np.random.default_rng((it * 9973 + 5) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0
    
    return best_a, best_obj


def frequency_domain_optimize(a0, time_limit):
    """
    Optimize in the frequency domain: parameterize a as inverse FFT of
    a non-negative spectrum, and optimize the spectrum.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    # Work with the real FFT
    fft_size = 1 << (2*n - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    # Magnitude and phase
    mag = np.abs(A)
    phase = np.angle(A)
    
    eta = 0.1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Current a
        a_curr = np.fft.irfft(mag * np.exp(1j * phase), fft_size)[:n]
        a_curr = np.maximum(a_curr, 0)
        s = a_curr.sum()
        if s < 1e-15:
            break
        a_curr = a_curr / s
        
        conv = fast_convolve(a_curr)
        peak_val = conv.max()
        
        if peak_val < best_obj - 1e-12:
            best_obj = peak_val
            best_a = a_curr.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Gradient with respect to magnitude:
        # Reducing magnitude at frequency f reduces the power at all lags
        # proportional to |A[f]|^2. The peak is most sensitive to frequencies
        # where |A[f]| is large.
        
        # Simple heuristic: reduce the magnitude at high-power frequencies
        power = mag ** 2
        top_f = np.argsort(power)[::-1][:max(2, len(power)//10)]
        
        # Reduce top frequencies
        for f in top_f:
            mag[f] *= (1 - eta * 0.1)
        
        # Increase low frequencies (to maintain sum(a) ≈ 1)
        # Actually, we normalize anyway, so just do the reduction.
        
        if no_improve > 30:
            eta *= 0.9
        if eta < 0.01:
            # Reset
            A = np.fft.rfft(best_a, fft_size)
            mag = np.abs(A)
            phase = np.angle(A)
            eta = 0.1
            no_improve = 0
        
        if no_improve > 200:
            break
    
    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # From parent (interpolated)
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))
    inits.append(rng.dirichlet(np.ones(n) * 0.2))
    
    # Edge-heavy
    dist = np.minimum(x, n - 1 - x).astype(float) + 1.0
    edge_w = 1.0 / dist ** 2.0
    inits.append(edge_w / edge_w.sum())
    
    # Sine-modulated
    t = x / max(n - 1, 1)
    for freq in [1, 2, 3]:
        asym = (1.0 + 0.3 * np.cos(2 * np.pi * freq * t)) ** 2 + 0.05
        inits.append(asym / asym.sum())
    
    # Two-peak
    g2 = np.exp(-0.5 * ((x - 0.25 * (n-1)) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n-1)) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())
    
    # Three-peak
    g3 = np.exp(-0.5 * ((x - 0.2 * (n-1)) / (0.08 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.5 * (n-1)) / (0.08 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.8 * (n-1)) / (0.08 * n)) ** 2)
    inits.append(g3 / g3.sum())
    
    # Random perturbation of parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px + rng.standard_normal(n) * 0.05, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits
