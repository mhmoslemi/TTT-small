**Approach: Alternating Minimization with "Anti-Peak" Correction**

1. Compute conv = fast_convolve(a)
2. Find the peak position k*
3. The peak is caused by the "cross" sum: sum_j a[j]*a[k*-j]
4. To reduce this peak, we want to decrease the product a[j]*a[k*-j] for j in the active set
5. But we can't just decrease (need to maintain sum = 1)
6. So: decrease a[j] for j that contribute to the peak, and increase a[k*-j] (or vice versa)
7. This is a "see-saw" operation

More precisely:
- For the peak at k*, the subgradient direction is: grad[j] = a[k*-j] (for valid j)
- We want to move in the direction -grad (to decrease the peak)
- But we also need to maintain non-negativity and sum = 1
- The project_simplex(-grad) gives the best point in the simplex that's "opposite" to the peak direction

This is essentially what the subgradient method does. Let me make it more effective by:
- Using multiple peaks (not just the max)
- Using a "weighted" subgradient where weights are proportional to how close each peak is to the max
- Using momentum to accelerate convergence

def fast_convolve(a):

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_pattern(n):
    """Precompute sparsity pattern for LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):

        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp(f, M, n, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    vals = f[row_idx - col_idx]

    a = np.maximum(a0.astype(np.float64), 0.0)

        a = np.ones(len(a)) / len(a)

def spectral_perturb(a, rng, strength=0.03):
    """Perturb sequence in frequency domain for diversity."""
    n = len(a)
    fft_size = 2 * n - 1
    fft_size = 1 << (fft_size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    # Random phase perturbation (preserve DC)
    if len(phase) > 1:
        phase_pert = phase + rng.standard_normal(len(phase) - 1)[np.newaxis].T * 0.01
        phase_pert = np.concatenate([phase[:1].reshape(1, -1), phase_pert], axis=0).flatten()
    else:
        phase_pert = phase
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s
    # Blend with original
    blended = (1.0 - strength) * a + strength * a_new
    blended = np.maximum(blended, 0)
    bs = blended.sum()
    if bs < 1e-15:
        return a.copy()
    return blended / bs

        for alpha in [0.3, 0.5, 0.7, 1.0]:
            cand = (1 - alpha) * a + alpha * a_new
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_candidate = cand

        a = best_candidate
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-11:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            eta *= 0.93
        if eta < 1e-4:
            eta = eta_init * 0.5
            no_improve = 0

        if no_improve > 120:
            rng = np.random.default_rng((it * 48271 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            eta = eta_init
            no_improve = 0

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function: multi-phase optimization combining SCA (with warm-started LP),
    smart subgradient, mirror descent, spectral perturbation, and adaptive n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        try:
            score = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        p = p[np.isfinite(p)]
        if p.ndim == 1 and p.size > 0 and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = try_seq(parent, "parent")
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_init")

    n_parent = len(parent)

    # === Phase 1: Quick SCA across multiple n values ===
    print("\n=== Phase 1: Multi-n SCA Exploration ===")
    t1 = time.monotonic()

    ns_to_try = sorted(set([n_parent, 640, 704, 768, 832, 896, 960]))
    ns_to_try = [n for n in ns_to_try if 256 <= n <= 2048]

    for n_try in ns_to_try:
        remaining_after = deadline - time.monotonic()
        if remaining_after < 300:
            break
        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss < 1e-10:
                start = np.ones(n_try) / n_try
            start = start / ss

        t_alloc = min(20, remaining_after - 280)
        if t_alloc < 3:
            continue
        print(f"  SCA n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = sca_optimize(start, time_limit=t_alloc)
        try_seq(a_opt, f"sca_n{n_try}")

    print(f"  P1 done: best={best_score:.8f}, t={time.monotonic()-t1:.1f}s")

    # === Phase 2: Deep SCA on best n ===
    print("\n=== Phase 2: Deep SCA ===")
    t2 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 400
        if remaining > 100:
            t_alloc = min(remaining * 0.6, 300)
            print(f"  Deep SCA n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, _ = sca_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_sca_1")

            remaining = deadline - time.monotonic() - 350
            if remaining > 50:
                t_alloc2 = min(remaining * 0.5, 200)
                print(f"  Deep SCA cont, t={t_alloc2:.0f}s")
                a_opt2, _ = sca_optimize(a_opt, time_limit=t_alloc2)
                try_seq(a_opt2, "deep_sca_2")

    print(f"  P2 done: best={best_score:.8f}, t={time.monotonic()-t2:.1f}s")

    # === Phase 3: Smart Subgradient + Mirror Descent ===
    print("\n=== Phase 3: Subgradient + Mirror ===")
    t3 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 300
        if remaining > 30:
            t_alloc = min(remaining * 0.4, 100)
            print(f"  Smart subgrad t={t_alloc:.0f}s")
            a_opt, _ = smart_subgradient(best_seq, time_limit=t_alloc, top_k=10)
            try_seq(a_opt, "subgrad_1")
            if best_seq is not None and a_opt is not best_seq:
                # Try SCA from subgradient result
                remaining = deadline - time.monotonic() - 250
                if remaining > 20:
                    a_opt2, _ = sca_optimize(a_opt, time_limit=min(remaining * 0.5, 80))
                    try_seq(a_opt2, "sca_from_sg")

        remaining = deadline - time.monotonic() - 250
        if remaining > 20:
            t_alloc = min(remaining * 0.4, 80)
            print(f"  Mirror descent t={t_alloc:.0f}s")
            a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc, eta_init=0.02)
            try_seq(a_opt, "mirror_1")

    print(f"  P3 done: best={best_score:.8f}, t={time.monotonic()-t3:.1f}s")

    # === Phase 4: Diversified restarts with spectral perturbation ===
    print("\n=== Phase 4: Diversified Restarts ===")
    t4 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        restart_configs = [
            (0.003, 0.0),   # small perturb, tight M
            (0.01, 0.01),   # medium perturb, slightly loose M
            (0.03, 0.03),   # large perturb, looser M
            (0.06, 0.05),   # very large perturb
        ]
        for i, (scale, m_off) in enumerate(restart_configs):
            remaining = deadline - time.monotonic() - 150
            if remaining < 15:
                break
            # Generate perturbed starting point
            perturbed = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            # Also try spectral perturbation
            spec_pert = spectral_perturb(best_seq, rng, strength=scale * 5)
            # Pick the better of the two
            c1 = fast_convolve(perturbed).max()
            c2 = fast_convolve(spec_pert).max()
            start = perturbed if c1 <= c2 else spec_pert

            t_alloc = min(25, remaining - 20)
            if t_alloc < 5:
                break
            print(f"  Restart {i+1}: scale={scale}, M_off={m_off}, t={t_alloc:.0f}s")
            a_opt, _ = sca_optimize(start, time_limit=t_alloc, M_offset=m_off)
            try_seq(a_opt, f"restart_{i}")

    print(f"  P4 done: best={best_score:.8f}, t={time.monotonic()-t4:.1f}s")

    # === Phase 5: Try different n values from best solution ===
    print("\n=== Phase 5: N-value Refinement ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        # Try nearby n values
        for delta in [-64, -32, 32, 64]:
            n_try = n_best + delta
            if n_try < 256 or n_try > 2048:
                continue
            remaining = deadline - time.monotonic() - 100
            if remaining < 15:
                break
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_best), best_seq)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss < 1e-10:
                continue
            start = start / ss
            t_alloc = min(20, remaining - 10)
            if t_alloc < 5:
                continue
            print(f"  N-refine n={n_try}, t={t_alloc:.0f}s")
            a_opt, _ = sca_optimize(start, time_limit=t_alloc)
            try_seq(a_opt, f"nrefine_{n_try}")

    print(f"  P5 done: best={best_score:.8f}, t={time.monotonic()-t5:.1f}s")

    # === Phase 6: Final polish ===
    print("\n=== Phase 6: Final Polish ===")
    t6 = time.monotonic()

    if best_seq is not None:
        # Multiple short SCA rounds with tight M
        for round_i in range(3):
            remaining = deadline - time.monotonic() - 5
            if remaining < 5:
                break
            t_alloc = min(remaining * 0.6, 30)
            if t_alloc < 3:
                break
            print(f"  Polish round {round_i+1}, t={t_alloc:.0f}s")
            a_opt, _ = sca_optimize(best_seq, time_limit=t_alloc, M_offset=0.0)
            try_seq(a_opt, f"polish_{round_i}")

        # Final subgradient pass
        remaining = deadline - time.monotonic() - 3
        if remaining > 3:
            t_alloc = min(remaining, 10)
            a_opt, _ = smart_subgradient(best_seq, time_limit=t_alloc, top_k=5)
            try_seq(a_opt, "final_sg")

    print(f"  P6 done: best={best_score:.8f}, t={time.monotonic()-t6:.1f}s")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float('inf')

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = min(best_score, fs)
    except Exception:
        pass

    print(f"\nFINAL RESULT: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
