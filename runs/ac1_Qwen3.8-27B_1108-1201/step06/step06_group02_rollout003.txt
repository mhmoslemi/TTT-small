"""
Optimization of step function heights to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses multi-peak subgradient descent with Barzilai-Borwein step size,
L-BFGS-B polish with adaptive smoothing, FLAT method (spectral peak reduction),
and diversified search across different sequence lengths.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 1e-30)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_peak_single(a, k):
    """Compute gradient of c[k] = sum_j a[j]*a[k-j] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if js <= je:
        grad[js:je + 1] = 2.0 * a[k - je:k - js + 1][::-1]
    return grad


def multi_peak_subgrad(a, top_k, weight_power=2.0):
    """Compute weighted multi-peak subgradient. Returns (gradient, max_value)."""
    conv = fast_convolve(a)
    n = len(a)
    if top_k >= len(conv):
        idx = np.arange(len(conv))
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[idx]
    order = np.argsort(peak_vals)[::-1]
    raw_w = 1.0 / (np.arange(len(idx)) + 1) ** weight_power
    w_sum = raw_w.sum()
    if w_sum < 1e-30:
        return np.zeros(n), float(peak_vals[order[0]])
    w = raw_w / w_sum
    grad = np.zeros(n)
    for rank in range(len(idx)):
        wi = w[order[rank]]
        if wi < 1e-15:
            continue
        k = int(idx[order[rank]])
        grad += wi * grad_peak_single(a, k)
    return grad, float(peak_vals[order[0]])


def smoothed_objective_and_grad(a, T, top_k=50):
    """Compute smoothed objective (log-sum-exp of top-k peaks / T) and its gradient."""
    conv = fast_convolve(a)
    n = len(a)
    if top_k >= len(conv):
        idx = np.arange(len(conv))
        peak_vals = conv.copy()
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[idx].copy()
    # Normalize for stability
    peak_max = peak_vals.max()
    shifted = peak_vals - peak_max
    scaled = shifted / T
    exp_vals = np.exp(scaled)
    log_sum_exp = T * (np.log(exp_vals.sum()) + shifted.max() / T) + peak_max
    # Gradient: sum_k (exp(c[k]/T) / sum(exp(c[k']/T))) * grad c[k] / T * T
    # = sum_k p[k] * grad c[k] where p[k] = exp(c[k]/T) / sum exp(c[k']/T)
    p = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    # Sort by value descending for weighting
    order = np.argsort(peak_vals)[::-1]
    for i in range(len(idx)):
        pi = p[order[i]]
        if pi < 1e-15:
            continue
        k = int(idx[order[i]])
        grad += pi * grad_peak_single(a, k)
    return float(log_sum_exp), grad


def bb_step(a_prev, a_curr, g_prev, g_curr, f_prev, f_curr, use_bb=True):
    """Barzilai-Borwein or fixed step size."""
    if not use_bb or f_prev is None or g_prev is None:
        return 0.01
    s = a_curr - a_prev
    y = g_curr - g_prev
    sy = np.dot(s, y)
    yy = np.dot(y, y)
    ss = np.dot(s, s)
    if yy < 1e-30:
        return 0.5
    # ZBB
    alpha_z = sy / yy if abs(yy) > 1e-30 else 0.01
    # BBB
    alpha_b = ss / sy if abs(sy) > 1e-30 else 0.01
    # Average
    alpha = 0.5 * (alpha_z + alpha_b)
    alpha = max(1e-8, min(alpha, 1.0))
    return alpha


def subgradient_descent(a0, time_limit, top_k=20, lr_init=0.05, weight_power=2.0):
    """Multi-peak subgradient descent with BB step size and momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    v = np.zeros(n)  # momentum
    momentum = 0.9
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    a_prev = a.copy()
    g_prev = None
    f_prev = None

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break

        grad, obj = multi_peak_subgrad(a, top_k, weight_power)
        alpha = bb_step(a_prev, a, g_prev, grad, f_prev, obj)
        alpha = min(alpha, lr)
        lr *= 0.9995

        v = momentum * v - alpha * grad
        a_new = project_simplex(a + v)

        obj_new = fast_convolve(a_new).max()
        if obj_new < obj + 0.001:  # Allow small increases for exploration
            a = a_new
        else:
            # Reject, shrink step
            lr *= 0.5
            v *= 0.5
            a = a.copy()
            obj_new = obj

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            # Restart with perturbation
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.004 * min(1.0, (no_improve - 100) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            lr = lr_init * 0.5
            momentum = 0.9
            no_improve = 0

        a_prev = a.copy()
        g_prev = grad
        f_prev = obj

        # Adapt momentum
        momentum = min(0.98, 0.9 + it * 0.0001)

    return best_a, best_obj


def lbfgs_polish(a0, time_limit, T_init=0.001, top_k=50, max_iter_lbfgs=200):
    """L-BFGS-B polish with annealing temperature."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    T = T_init
    n_restarts = min(10, max(3, int(time_limit // 30)))
    seeds = [0, 1, 7, 42, 123, 999, 31415, 27182, 16180, 55555]

    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit - 5:
            break
        time_per = min((time_limit - (time.monotonic() - t0)) / max(1, n_restarts - r), 30)

        if r == 0:
            x0 = a.copy()
        else:
            rng = np.random.default_rng(seeds[r % len(seeds)])
            scale = 0.003 * (1 + 0.5 * r)
            x0 = project_simplex(best_a + rng.standard_normal(n) * scale)

        T_cur = T * (1 + 0.3 * r)

        def obj_func(x):
            x_clipped = np.maximum(x, 0.0)
            sx = x_clipped.sum()
            if sx < 1e-15:
                return 1e10
            x_norm = x_clipped / sx
            return smoothed_objective_and_grad(x_norm, T_cur, top_k)[0]

        def grad_func(x):
            x_clipped = np.maximum(x, 0.0)
            sx = x_clipped.sum()
            if sx < 1e-15:
                return np.zeros(len(x))
            x_norm = x_clipped / sx
            _, g = smoothed_objective_and_grad(x_norm, T_cur, top_k)
            return g / sx

        try:
            result = minimize(obj_func, x0, jac=grad_func, method='L-BFGS-B',
                              bounds=[(0, 1)] * n,
                              options={'maxiter': max_iter_lbfgs, 'gtol': 1e-10,
                                       'ftol': 1e-14})
            if result.success or result.fun < 1e10:
                x_opt = np.maximum(result.x, 0.0)
                sx = x_opt.sum()
                if sx > 1e-15:
                    x_opt = x_opt / sx
                    obj_val = fast_convolve(x_opt).max()
                    if obj_val < best_obj:
                        best_obj = obj_val
                        best_a = x_opt.copy()
                        print("    L-BFGS-B restart {}: obj={:.10f}".format(r, obj_val))
        except Exception:
            pass

        T *= 0.7  # Anneal

    return best_a, best_obj


def flat_method(a0, M_init=None, max_iter=500, time_limit=60):
    """Spectral peak reduction (FLAT-like method)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    if M_init is None:
        M_init = conv.max() * 0.99
    M_target = M_init

    best_a = a.copy()
    best_obj = conv.max()

    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit - 1:
            break

        A = np.fft.fft(a, fft_size)
        r = np.fft.ifft(np.abs(A) ** 2).real[:2 * n - 1]

        peaks = r > M_target
        if not peaks.any():
            # Try to reduce M_target
            M_target *= 0.99
            continue

        # Clip the peaks
        r_clipped = r.copy()
        r_clipped[peaks] = M_target
        # Zero out the rest to maintain energy
        r_clipped[2 * n - 1:fft_size] = 0 if fft_size > 2 * n - 1 else r_clipped[2 * n - 1:fft_size]

        R_clipped = np.fft.fft(r_clipped)
        magnitude = np.sqrt(np.maximum(R_clipped.real, 0))
        # R should be real and non-negative if r is symmetric
        # But due to numerical issues, take the real part
        R_clipped_real = np.real(R_clipped)
        magnitude = np.sqrt(np.maximum(R_clipped_real, 1e-30))
        phase = np.angle(A)
        A_new = magnitude * np.cos(phase) + 1j * magnitude * np.sin(phase)
        a_new = np.fft.ifft(A_new).real[:n]

        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            # Reduce target
            M_target = max(M_target * 0.995, obj_new * 0.95)
        else:
            # Slowly reduce target
            M_target = max(M_target * 0.999, fast_convolve(a).max() * 0.98)
            a = a_new  # Still update to explore

    return best_a, best_obj


def symmetric_optimize(a0, time_limit, top_k=20):
    """Optimize under the constraint a[i] = a[n-1-i]."""
    n = len(a0)
    # Work with half the sequence
    half_n = (n + 1) // 2
    a_half = np.maximum(a0[:half_n], 0.0).copy()

    def half_to_full(h):
        if n % 2 == 0:
            full = np.concatenate([h, h[::-1]])
        else:
            full = np.concatenate([h[:-1], [h[-1]], h[:-1][::-1]])
        s = full.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return full / s

    full = half_to_full(a_half)
    best_a = full.copy()
    best_obj = fast_convolve(full).max()

    t0 = time.monotonic()
    lr = 0.03
    v = np.zeros(half_n)
    momentum = 0.9
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        full = half_to_full(a_half)
        conv = fast_convolve(full)
        top_k_eff = min(top_k, len(conv))
        idx = np.argpartition(conv, -top_k_eff)[-top_k_eff:]
        peak_vals = conv[idx]
        order = np.argsort(peak_vals)[::-1]
        raw_w = 1.0 / (np.arange(len(idx)) + 1) ** 2.0
        w = raw_w / raw_w.sum()

        grad_full = np.zeros(n)
        for rank in range(len(idx)):
            wi = w[order[rank]]
            if wi < 1e-15:
                continue
            k = int(idx[order[rank]])
            grad_full += wi * grad_peak_single(full, k)

        # Project gradient onto symmetric subspace
        grad_sym = np.zeros(half_n)
        for i in range(half_n):
            j = n - 1 - i
            grad_sym[i] = grad_full[i] + grad_full[j]
            if i == j:
                grad_sym[i] = grad_full[i]

        v = momentum * v - lr * grad_sym
        a_half_new = a_half + v
        a_half_new = np.maximum(a_half_new, 0.0)
        if a_half_new.sum() < 1e-15:
            a_half_new = np.ones(half_n) / half_n

        a_half = a_half_new
        full_new = half_to_full(a_half)
        obj_new = fast_convolve(full_new).max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = full_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.98

        if no_improve > 200:
            rng = np.random.default_rng(it * 31 + 7)
            a_half = np.maximum(a_half + rng.standard_normal(half_n) * 0.003, 0)
            v = np.zeros(half_n)
            lr = 0.03
            no_improve = 0

    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    """Cyclic coordinate descent with pairwise swaps."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stale = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Find the pair that contributes most to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]

        # Find best location to move mass
        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            break

        # Try moving small amounts
        improved = False
        deltas = [0.0002, 0.0005, 0.001, 0.002, 0.004]
        # Sort outside by a[outside] ascending (prefer moving to low values)
        outside_sorted = outside[np.argsort(a[outside])[:20]]

        for j_min in outside_sorted[:5]:
            for delta in deltas:
                if a[j_max] < delta:
                    break
                candidate = a.copy()
                candidate[j_max] -= delta
                candidate[j_min] += delta
                candidate = np.maximum(candidate, 0.0)
                candidate = candidate / candidate.sum()
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if improved:
            stale = 0
        else:
            stale += 1
            if stale % 50 == 0:
                noise = rng.standard_normal(n) * 0.001
                candidate = project_simplex(a + noise)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    stale = 0
                elif stale > 1000:
                    break

    return best_a, best_obj


def generate_structured_init(n, parent=None, rng=None):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine windows
    for freq in [1, 2, 3]:
        h = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n)
        h = np.maximum(h, 0.01)
        inits.append(h / h.sum())

    # Interpolated parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Sinc-like
    t = (x - n / 2) / (n / 2)
    sinc = np.sinc(t) * np.exp(-0.3 * t ** 2)
    sinc = np.maximum(sinc, 0.01)
    inits.append(sinc / sinc.sum())

    # Random Dirichlet
    if rng is not None:
        inits.append(rng.dirichlet(np.ones(n) * 0.5))
        inits.append(rng.dirichlet(np.ones(n) * 0.1))

    return inits


def try_and_update(a, state, label=""):
    """Evaluate and update state if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a)))
        return True
    return False


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization:
    1. Diversified n exploration with subgradient + L-BFGS-B
    2. Deep optimization: subgradient, FLAT, symmetric, coordinate descent
    3. Final polish with fine-tuned methods
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    state = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.8f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384

    n_parent = len(parent)

    # === Phase 1: Diversified n exploration (~400s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent, 640, 700, 768, 832, 900, 1024, 1152, 1280]))
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 2048]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1100
        if remaining < 20:
            break

        # Generate initialization
        if n_try == n_parent:
            start = parent.copy()
        else:
            starts = generate_structured_init(n_try, parent, rng)
            # Quick eval to pick best start
            best_start = starts[0]
            best_start_obj = fast_convolve(best_start).max()
            for st in starts:
                obj = fast_convolve(st).max()
                if obj < best_start_obj:
                    best_start_obj = obj
                    best_start = st
            start = best_start

        # Subgradient
        t_alloc = min(remaining, 45)
        a_opt, _ = subgradient_descent(start, time_limit=t_alloc, top_k=20, lr_init=0.04)
        try_and_update(a_opt, state, "sub_n{}".format(n_try))

        # L-BFGS-B polish
        remaining = deadline - time.monotonic() - 1080
        if remaining > 10 and state[1] is not None and len(state[1]) == n_try:
            t_alloc = min(remaining, 30)
            a_opt2, _ = lbfgs_polish(a_opt, time_limit=t_alloc, T_init=0.002, top_k=40)
            try_and_update(a_opt2, state, "lbfgs_n{}".format(n_try))

    print("  Phase 1 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 2: Deep subgradient on best (~400s) ===
    print("\n=== Phase 2: Deep Subgradient ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])
        remaining = deadline - time.monotonic() - 700
        if remaining > 30:
            t_alloc = min(remaining, 300)
            print("  Deep subgradient n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = subgradient_descent(state[1], time_limit=t_alloc,
                                           top_k=30, lr_init=0.03, weight_power=1.5)
            try_and_update(a_opt, state, "deep_sub")

            # Try with different params
            remaining = deadline - time.monotonic() - 500
            if remaining > 20:
                t_alloc = min(remaining, 100)
                a_opt2, _ = subgradient_descent(state[1], time_limit=t_alloc,
                                                top_k=50, lr_init=0.02, weight_power=3.0)
                try_and_update(a_opt2, state, "deep_sub_k50")

    print("  Phase 2 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 3: FLAT method + Symmetric + L-BFGS-B (~350s) ===
    print("\n=== Phase 3: FLAT + Symmetric + L-BFGS-B ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # FLAT method
        remaining = deadline - time.monotonic() - 450
        if remaining > 15:
            t_alloc = min(remaining, 80)
            print("  FLAT method, t={:.0f}s".format(t_alloc))
            a_opt, _ = flat_method(state[1], time_limit=t_alloc, max_iter=1000)
            try_and_update(a_opt, state, "flat")
            # Polish FLAT result
            remaining = deadline - time.monotonic() - 420
            if remaining > 10:
                a_opt2, _ = subgradient_descent(a_opt, time_limit=min(remaining, 20),
                                                top_k=20, lr_init=0.02)
                try_and_update(a_opt2, state, "flat_polish")

        # Symmetric optimization
        remaining = deadline - time.monotonic() - 350
        if remaining > 15:
            t_alloc = min(remaining, 60)
            print("  Symmetric opt, t={:.0f}s".format(t_alloc))
            a_opt, _ = symmetric_optimize(state[1], time_limit=t_alloc, top_k=20)
            try_and_update(a_opt, state, "symmetric")

        # L-BFGS-B with various temps
        for T_init in [0.0005, 0.001, 0.003]:
            remaining = deadline - time.monotonic() - 250
            if remaining < 10:
                break
            t_alloc = min(remaining, 50)
            print("  L-BFGS-B T={}, t={:.0f}s".format(T_init, t_alloc))
            a_opt, _ = lbfgs_polish(state[1], time_limit=t_alloc, T_init=T_init, top_k=30)
            try_and_update(a_opt, state, "lbfgs_T{}".format(T_init))

    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 4: Coordinate descent + fine subgradient (~250s) ===
    print("\n=== Phase 4: Coordinate + Fine Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Coordinate descent
        remaining = deadline - time.monotonic() - 150
        if remaining > 10:
            t_alloc = min(remaining, 80)
            print("  Coordinate descent, t={:.0f}s".format(t_alloc))
            a_opt, _ = coordinate_descent(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "coord")

        # Fine subgradient (small top_k, small lr)
        for top_k, lr in [(5, 0.01), (10, 0.008), (3, 0.005)]:
            remaining = deadline - time.monotonic() - 100
            if remaining < 10:
                break
            t_alloc = min(remaining, 40)
            print("  Fine sub k={}, lr={}, t={:.0f}s".format(top_k, lr, t_alloc))
            a_opt, _ = subgradient_descent(state[1], time_limit=t_alloc,
                                           top_k=top_k, lr_init=lr, weight_power=1.0)
            try_and_update(a_opt, state, "fine_{}_{}".format(top_k, lr))

    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 5: Random restarts + quick polish (~200s) ===
    print("\n=== Phase 5: Random Restarts ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])
        n_restarts = 8
        for r in range(n_restarts):
            remaining = deadline - time.monotonic() - 60
            if remaining < 10:
                break
            t_per = min((remaining) / max(1, n_restarts - r), 25)

            rng_r = np.random.default_rng(seed + 1000 + r * 7)
            scale = 0.002 + 0.003 * r
            randomized = project_simplex(state[1] + rng_r.standard_normal(n_best) * scale)

            a_opt, _ = subgradient_descent(randomized, time_limit=t_per * 0.6,
                                           top_k=20, lr_init=0.03)
            try_and_update(a_opt, state, "restart_{}_sub".format(r))

            remaining = deadline - time.monotonic() - 40
            if remaining > 5:
                a_opt2, _ = lbfgs_polish(a_opt, time_limit=min(remaining, t_per * 0.4),
                                         T_init=0.0005, top_k=30)
                try_and_update(a_opt2, state, "restart_{}_lbfgs".format(r))

    print("  Phase 5 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 6: Final ultra-fine polish (~100s) ===
    print("\n=== Phase 6: Final Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Very fine subgradient
        remaining = deadline - time.monotonic() - 30
        if remaining > 5:
            a_opt, _ = subgradient_descent(state[1], time_limit=min(remaining, 40),
                                           top_k=10, lr_init=0.005, weight_power=1.0)
            try_and_update(a_opt, state, "final_sub")

        # Tiny perturbations
        remaining = deadline - time.monotonic() - 10
        for _ in range(10):
            remaining = deadline - time.monotonic() - 5
            if remaining < 3:
                break
            noise = rng.standard_normal(len(state[1])) * 0.0005
            perturbed = project_simplex(state[1] + noise)
            # Quick coordinate descent
            a_opt, _ = coordinate_descent(perturbed, time_limit=min(remaining, 3))
            try_and_update(a_opt, state, "tiny_pert")

    print("  Phase 6 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(384) / 384
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
