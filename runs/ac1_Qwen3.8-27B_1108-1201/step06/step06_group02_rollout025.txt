"""
Optimization of non-negative step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max smoothed gradient descent with momentum, combined with
peak-targeting subgradient steps, periodic random perturbations,
and multi-n exploration.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient(a, w):
    """Compute gradient of soft-max weighted peak reduction.
    grad[i] = 2 * sum_k w[k] * a[k-i] (for valid indices)
    Computed via FFT cross-correlation."""
    n = len(a)
    size = len(w) + n - 1  # >= 3n-2
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    R = np.fft.irfft(W * np.conj(A), fft_size)
    return 2.0 * R[:n]

def soft_max_weighted(a, T):
    """Compute soft-max weights over convolution values."""
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp(T * (c - c_max))
    w = exp_c / exp_c.sum()
    return w, c, c_max

def gradient_descent_smooth(a0, time_limit, T0, T1, eta0):
    """Main gradient descent with soft-max smoothing and momentum."""
    a = project_simplex(a0.copy())
    n = len(a)
    
    best_a = a.copy()
    best_maxc = fast_convolve(a).max()
    
    t0 = time.monotonic()
    v = np.zeros(n)
    mu = 0.85  # momentum
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Schedule
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T0 + (T1 - T0) * frac  # increase temperature (sharper peak focus)
        eta = eta0 / (1.0 + 0.01 * it)  # decaying learning rate
        
        # Compute soft-max weights and gradient
        w, c, c_max = soft_max_weighted(a, T)
        grad = compute_gradient(a, w)
        
        # Momentum step
        v = mu * v - eta * grad
        a_new = a + v
        a_new = project_simplex(a_new)
        
        # Evaluate
        c_new = fast_convolve(a_new)
        maxc_new = c_new.max()
        
        if maxc_new < best_maxc - 1e-11:
            best_maxc = maxc_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

"""
Optimization of non-negative step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max smoothed gradient descent with FFT-based gradient computation,
momentum, adaptive learning rate, simulated annealing acceptance,
periodic peak-targeting subgradient steps, and multi-n exploration.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_fft(a, w):
    """Compute gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (valid indices).
    Uses FFT cross-correlation for O(n log n) efficiency."""
    n = len(a)
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    R = np.fft.irfft(W * np.conj(A), fft_size)
    return 2.0 * R[:n]


def smooth_grad_descent(a0, time_limit, T_start, T_end, eta0, eta_decay, mu, rng):
    """Core soft-max gradient descent with momentum and annealing acceptance."""
    a = project_simplex(a0.astype(np.float64).copy())
    n = len(a)

    c = fast_convolve(a)
    best_maxc = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    v = np.zeros(n)
    it = 0
    no_improve = 0
    sa_temp = 0.005
    sa_min = 0.0001

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)

        T = T_start + (T_end - T_start) * frac
        eta = eta0 / (1.0 + eta_decay * it)
        mu_cur = min(0.95, mu + 0.00005 * it)

        # Compute soft-max weights
        c = fast_convolve(a)
        c_max = c.max()
        exp_c = np.exp(T * (c - c_max))
        w = exp_c / exp_c.sum()

        # Gradient via FFT
        grad = compute_grad_fft(a, w)

        # Momentum step
        v = mu_cur * v - eta * grad
        a_new = a + v
        a_new = project_simplex(a_new)

        # Evaluate
        c_new = fast_convolve(a_new)
        maxc_new = c_new.max()

        # Annealing acceptance
        if maxc_new < best_maxc - 1e-12:
            best_maxc = maxc_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Simulated annealing: accept worse with some probability
            delta = maxc_new - best_maxc
            if delta < sa_temp and rng.random() < np.exp(-delta / max(sa_temp, 1e-10)):
                a = a_new
            else:
                a = a_new  # Still move forward (gradient direction is useful)
                # But keep best_a unchanged

        # Decay annealing temperature
        sa_temp = max(sa_min, sa_temp * 0.9995)

        # Periodic restart from best with perturbation
        if no_improve > 300:
            noise_scale = 0.001 + 0.005 * min(1.0, (no_improve - 300) / 500.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            no_improve = 0
            sa_temp = 0.003

    return best_a, best_maxc


def peak_targeting(a0, time_limit, rng):

    a = project_simplex(a0.astype(np.float64).copy())

    eta = 0.01
    it = 0

        it += 1

        # Find top peaks (use top-3 for smoother subgradient)
        top_k = min(3, len(c))
        peaks = np.argpartition(c, -top_k)[-top_k:]
        peak_vals = c[peaks]

        # Weighted subgradient
        weights = np.exp(peak_vals * 100.0)
        weights /= weights.sum()

        grad = np.zeros(n)
        for idx, k_p in enumerate(peaks):
            k_p = int(k_p)

            for i in range(n):

                    grad[i] += 2.0 * weights[idx] * a[j]
