- In the circular convolution sense (with period N ≥ 2n-1), (f*f)(k) = ifft_N(|F_N(ω)|²)(k) where F_N = fft_N(f).

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def symmetrize(f):
    """Enforce palindrome symmetry."""
    n = len(f)
    return (f + f[::-1]) / 2.0

def project_simplex_nonneg(v, target_sum=None):
    """Project onto {x >= 0, sum(x) = target_sum} (or sum=1 if None)."""
    n = len(v)
    if target_sum is None:
        target_sum = 1.0
    # Project onto simplex scaled by target_sum
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def score_of(f):
    """Compute the score for sequence f."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return float('inf')
    f = f / s
    conv = fast_convolve(f)
    n = len(f)
    return 2.0 * n * conv.max()

def subgradient_step(f, top_k, T, lr, momentum, v):
    """One step of the subgradient method with symmetry."""
    n = len(f)
    conv = fast_convolve(f)
    
    # Get top-k peaks
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    # Weight by softmax
    peak_vals = conv[peaks]
    weights = np.exp((peak_vals - peak_vals.max()) / T)
    weights /= weights.sum()
    
    # Compute subgradient
    sg = np.zeros(n)
    for w, k in zip(weights, peaks):
        k = int(k)
        # ∂(f*f)(k)/∂f[j] = 2*f[k-j] for valid j
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += w * 2.0 * f[k - idx]
    
    # Momentum
    v = momentum * v - lr * sg
    
    # Apply step with symmetry
    f_new = symmetrize(f + v)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        f_new = np.ones(n) / n
    else:
        f_new = f_new / s
    
    return f_new, v, conv.max()
