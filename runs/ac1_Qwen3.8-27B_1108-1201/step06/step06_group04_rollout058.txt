"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Ideal magnitude initialization from flat auto-correlation target
- SCA (Successive Convex Approximation) with LP subproblems on top-K peaks
- Gerchberg-Saxton phase retrieval with target power spectrum
- Smoothed projected gradient (log-sum-exp) for polish
- Multi-n exploration with adaptive time allocation
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def convolve_grad_multi(a, peak_positions):
    """Compute gradients of b[k] for multiple peak positions. Returns (K, n) array."""
    n = len(a)
    K = len(peak_positions)
    grads = np.zeros((K, n))
    for idx, k in enumerate(peak_positions):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        i_range = np.arange(js, je + 1)
        j_range = k - i_range
        vals = 2.0 * a[j_range]
        if k % 2 == 0:
            diag_i = k // 2
            if js <= diag_i <= je:
                mask = (i_range == diag_i)
                vals[mask] = a[diag_i]
        grads[idx, i_range] = vals
    return grads


def ideal_init(n):
    """
    Compute initialization from the 'ideal' flat auto-correlation target.
    Target: |A[f]|^2 proportional to |Dirichlet_kernel(2n-1)[f]|
    So |A[f]| proportional to sqrt(|D[f]|)
    """
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    # Dirichlet kernel D[f] = sum_{k=0}^{2n-2} exp(-2*pi*i*k*f/fft_size)
    f = np.arange(fft_size // 2 + 1)
    k = np.arange(2 * n - 1)
    # D[f] = exp(-pi*i*(2n-2)*f/N) * sin(pi*(2n-1)*f/N) / sin(pi*f/N)
    # |D[f]| = |sin(pi*(2n-1)*f/N) / sin(pi*f/N)|
    num = np.sin(np.pi * (2 * n - 1) * f / fft_size)
    den = np.sin(np.pi * f / fft_size)
    den[f == 0] = 1.0
    num[f == 0] = 2 * n - 1
    D_abs = np.abs(num / den)
    
    # Target magnitude: sqrt(D_abs), normalized so sum of a = 1 after IFFT
    target_mag = np.sqrt(D_abs)
    
    # Use zero phase (gives real, symmetric result)
    A_target = np.zeros(fft_size, dtype=np.complex128)
    A_target[:fft_size // 2 + 1] = target_mag
    # Hermitian symmetry
    for i in range(1, fft_size // 2):
        A_target[fft_size - i] = A_target[i]
    
    a_ideal = np.fft.irfft(A_target, fft_size)[:n]
    a_ideal = np.maximum(a_ideal, 0.0)
    s = a_ideal.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_ideal / s


def cosine_init(n, alpha=0.3):
    """Raised cosine initialization."""
    j = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def gaussian_init(n, sigma_frac=0.25):
    """Gaussian initialization."""
    j = np.arange(n, dtype=np.float64)
    center = 0.5 * (n - 1)
    a = np.exp(-0.5 * ((j - center) / (sigma_frac * n)) ** 2)
    s = a.sum()
    return a / s


def sca_one_step(a, K=20):
    """
    One SCA step: linearize top-K peak constraints, solve LP.
    Returns new a (normalized to sum=1).
    """
    n = len(a)
    b = fast_convolve(a)
    
    if K >= len(b):
        peak_idx = np.argsort(b)[::-1]
    else:
        peak_idx = np.argpartition(b, -K)[-K:]
    
    # Sort peaks by value descending
    pv = b[peak_idx]
    order = np.argsort(pv)[::-1]
    peak_idx = peak_idx[order]
    peak_vals = pv[order]
    
    # Compute gradients at current point
    grads = convolve_grad_multi(a, peak_idx)
    
    # LP: min t
    # s.t. grads[i] @ a_new - t <= c_i  (linearized constraint)
    #      sum a_new = 1
    #      a_new >= 0, t >= 0
    # where c_i = grads[i] @ a - b[peak_idx[i]]
    
    c_vals = grads @ a - peak_vals
    
    # Build LP
    nvars = n + 1
    A_ub = np.zeros((len(peak_idx), nvars))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = c_vals
    
    A_eq = np.zeros((1, nvars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    c_obj = np.zeros(nvars)
    c_obj[n] = 1.0
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.status != 0:
            return a.copy()
        a_new = result.x[:n].copy()
    except Exception:
        return a.copy()
    
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s
    return a_new


def sca_iterate(a0, time_limit, K=20, schedule='const'):
    """

    """
    a = a0.copy()
    n = len(a)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    
    # Precompute target magnitude
    f = np.arange(fft_size // 2 + 1)
    num = np.sin(np.pi * (2 * n - 1) * f / fft_size)
    den = np.sin(np.pi * f / fft_size)
    den[f == 0] = 1.0
    num[f == 0] = 2 * n - 1
    target_mag = np.sqrt(np.abs(num / den))
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit - 0.1 and it < n_inner:
        it += 1
        # Forward transform
        A = np.fft.rfft(a, fft_size)
        
        # Enforce target magnitude, keep phase
        phase = np.angle(A)
        mag = np.abs(A)
        # Blend: use target where current mag is small, keep where large
        new_mag = 0.5 * target_mag + 0.5 * mag
        A_new = new_mag * np.exp(1j * phase)
        
        # Inverse transform
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Project to non-negative simplex
        a_new = project_simplex(a_new)
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a


def weighted_qp_grad(a0, time_limit, T_init=0.005, lr_init=0.02):
    """
    Projected gradient on weighted sum of peaks (convex QP approximation).
    W = sum_k p_k * H_k, minimize a^T W a subject to simplex.
    """
    a = a0.copy()
    n = len(a)
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit - 0.1:
        it += 1
        b = fast_convolve(a)
        
        # Softmax weights
        scaled = b / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p /= p.sum()
        
        # Gradient of weighted sum: sum_k p[k] * grad b[k]
        # = 2 * cross-correlation of (weighted p) with a
        # grad[i] = 2 * sum_k p[k] * a[k-i] (valid k-i)
        # Use top peaks for efficiency
        K_use = min(100, len(p))
        top_k = np.argpartition(b, -K_use)[-K_use:]
        # Renormalize p for top peaks
        p_top = p[top_k]
        p_top = p_top / p_top.sum()
        
        grads = convolve_grad_multi(a, top_k)
        grad = np.zeros(n)
        for idx in range(len(top_k)):
            grad += p_top[idx] * grads[idx]
        
        # Projected gradient step
        best_candidate = a
        best_val = best_obj
        for trial_lr in [lr, lr * 0.5, lr * 0.25]:
            candidate = project_simplex(a - trial_lr * grad)
            obj = fast_convolve(candidate).max()
            if obj < best_val - 1e-16:
                best_val = obj
                best_candidate = candidate
        
        a = best_candidate
        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 30:
            T *= 1.15
            lr *= 0.93
        if no_improve > 100:
            T *= 0.7
        if no_improve > 400:
            rng = np.random.default_rng(it * 104729 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0
    
    return best_a


def peak_shave(a0, time_limit):
    """
    Directly shave peaks by reducing contributing variables and
    distributing mass to non-contributing positions.
    """
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit - 0.1:
        it += 1
        b = fast_convolve(a)
        k_peak = int(np.argmax(b))
        peak_val = b[k_peak]
        
        # Find contributing indices
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = np.arange(js, je + 1)
        non_contrib = np.setdiff1d(np.arange(n), contrib)
        
        if len(contrib) == 0 or len(non_contrib) == 0:
            break
        
        # Find the largest contributor
        idx_max = contrib[np.argmax(a[contrib])]
        val = a[idx_max]
        if val < 1e-10:
            break
        
        # Try different shift fractions
        improved = False
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
            shift = val * frac
            if shift < 1e-12:
                continue
            
            delta = np.zeros(n)
            delta[idx_max] -= shift
            # Distribute to non-contrib positions
            nc_vals = a[non_contrib]
            nc_sum = nc_vals.sum()
            if nc_sum > 1e-15:
                for j in non_contrib:
                    delta[j] += shift * nc_vals[j] / nc_sum
            else:
                for j in non_contrib:
                    delta[j] += shift / len(non_contrib)
            
            candidate = project_simplex(a + delta)
            obj = fast_convolve(candidate).max()
            if obj < best_obj - 1e-14:
                a = candidate
                best_obj = obj
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try random perturbation
            if it % 50 == 0:
                rng = np.random.default_rng(it * 31 + k_peak)
                for _ in range(5):
                    delta = np.zeros(n)
                    delta[contrib] = -rng.random(len(contrib)) * 0.01
                    delta[non_contrib] += rng.random(len(non_contrib)) * 0.01
                    candidate = project_simplex(a + delta)
                    obj = fast_convolve(candidate).max()
                    if obj < best_obj - 1e-14:
                        a = candidate
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved and it > 500:
            break
    
    return best_a


def objective_from_b(a, b):
    """Compute objective from sequence and its convolution."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return float(2 * n * b.max() / (s * s))


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search:
    Phase 1: Multi-n initialization with ideal/cosine/parent
    Phase 2: SCA with LP (major optimization)
    Phase 3: Phase retrieval + weighted QP
    Phase 4: Smoothed gradient polish
    Phase 5: Peak shaving + final refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(384) / 384
    
    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")
    
    # === Phase 1: Multi-n initialization (~400s) ===
    print("\n=== Phase 1: Multi-n Initialization ===")
    t1 = time.monotonic()
    
    n_candidates = sorted(set([
        n_parent,
        512, 768, 1024, 1536, 2048,
        max(256, n_parent // 2),
        min(4096, n_parent * 2) if n_parent * 2 <= 4096 else n_parent,
    ]))
    
    inits_per_n = {}
    
    for n_try in n_candidates:
        remaining = deadline - time.monotonic() - 1000
        if remaining < 20:
            break
        
        # Generate initializations
        inits = []
        # Ideal magnitude init
        inits.append(ideal_init(n_try))
        # Cosine
        inits.append(cosine_init(n_try, 0.3))
        inits.append(cosine_init(n_try, 0.5))
        inits.append(cosine_init(n_try, 0.1))
        # Gaussian
        inits.append(gaussian_init(n_try, 0.2))
        inits.append(gaussian_init(n_try, 0.35))
        # Parent interpolation
        if parent is not None:
            pi = np.interp(np.linspace(0, 1, n_try),
                           np.linspace(0, 1, n_parent), parent)
            pi = np.maximum(pi, 0)
            if pi.sum() > 1e-10:
                inits.append(pi / pi.sum())
        # Uniform
        inits.append(np.ones(n_try) / n_try)
        # Two-peak
        j = np.arange(n_try, dtype=np.float64)
        two_peak = np.exp(-0.5 * ((j - 0.3 * (n_try - 1)) / (0.15 * n_try))**2) + \
                    np.exp(-0.5 * ((j - 0.7 * (n_try - 1)) / (0.15 * n_try))**2)
        inits.append(two_peak / two_peak.sum())
        
        # Quick SCA on each init (10-20s each)
        t_per_init = min(15.0, remaining / len(inits))
        best_for_n = None
        best_obj_n = float('inf')
        
        for ii, init in enumerate(inits):
            if time.monotonic() > deadline - 1000:
                break
            a_opt = sca_iterate(init, time_limit=t_per_init, K=15, schedule='increase')
            obj = fast_convolve(a_opt).max()
            if obj < best_obj_n:
                best_obj_n = obj
                best_for_n = a_opt
        
        if best_for_n is not None:
            inits_per_n[n_try] = best_for_n
            try_seq(best_for_n, f"init_n{n_try}")
    
    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")
    
    # Determine best n from Phase 1
    if best_seq is not None:
        n_best = len(best_seq)
    else:
        n_best = n_parent
        best_seq = parent.copy()
    
    # === Phase 2: Deep SCA with LP (~600s) ===
    print("\n=== Phase 2: Deep SCA ===")
    t2 = time.monotonic()
    
    if best_seq is not None:
        # Multiple SCA configurations
        configs = [
            {'K': 15, 'schedule': 'increase', 't_frac': 0.35},
            {'K': 30, 'schedule': 'const', 't_frac': 0.25},
            {'K': 50, 'schedule': 'const', 't_frac': 0.2},
            {'K': 10, 'schedule': 'increase', 't_frac': 0.2},
        ]
        
        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 400
            if remaining < 15:
                break
            t_alloc = min(remaining, int(budget_s * cfg['t_frac']))
            print(f"  SCA K={cfg['K']}, schedule={cfg['schedule']}, t={t_alloc:.0f}s")
            a_opt = sca_iterate(best_seq, time_limit=t_alloc, **{k: v for k, v in cfg.items() if k != 't_frac'})
            try_seq(a_opt, f"sca_deep_{ci}")
    
    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")
    
    # === Phase 3: Phase retrieval + Weighted QP (~350s) ===
    print("\n=== Phase 3: Phase Retrieval & Weighted QP ===")
    t3 = time.monotonic()
    
    if best_seq is not None:
        # Phase retrieval
        remaining = deadline - time.monotonic() - 350
        if remaining > 20:
            t_alloc = min(remaining, 80)
            print(f"  Phase retrieval, t={t_alloc:.0f}s")
            a_opt = phase_retrieve(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "phase_ret")
            if best_seq is not None:
                # Polish after phase retrieval
                remaining2 = deadline - time.monotonic() - 300
                if remaining2 > 10:
                    a_opt2 = sca_iterate(a_opt, time_limit=min(remaining2, 40), K=20)
                    try_seq(a_opt2, "phase_ret_polish")
        
        # Weighted QP gradient
        for T_val in [0.003, 0.01, 0.03]:
            remaining = deadline - time.monotonic() - 250
            if remaining < 10:
                break
            t_alloc = min(remaining, 60)
            print(f"  Weighted QP T={T_val}, t={t_alloc:.0f}s")
            a_opt = weighted_qp_grad(best_seq, time_limit=t_alloc, T_init=T_val, lr_init=0.03)
            try_seq(a_opt, f"wqp_T{T_val}")
    
    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")
    
    # === Phase 4: Smoothed gradient polish (~250s) ===
    print("\n=== Phase 4: Smoothed Gradient ===")
    t4 = time.monotonic()
    
    if best_seq is not None:
        for T_val, lr_val in [(0.002, 0.03), (0.005, 0.05), (0.01, 0.02), (0.001, 0.01)]:
            remaining = deadline - time.monotonic() - 150
            if remaining < 10:
                break
            t_alloc = min(remaining, 50)
            print(f"  Smoothed grad T={T_val}, lr={lr_val}, t={t_alloc:.0f}s")
            a_opt = smoothed_projected_grad(best_seq, time_limit=t_alloc, T_init=T_val, lr_init=lr_val)
            try_seq(a_opt, f"smooth_T{T_val}")
    
    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")
    
    # === Phase 5: Peak shaving + final SCA (~150s) ===
    print("\n=== Phase 5: Peak Shaving & Final ===")
    t5 = time.monotonic()
    
    if best_seq is not None:
        # Peak shaving
        remaining = deadline - time.monotonic() - 80
        if remaining > 10:
            t_alloc = min(remaining, 50)
            print(f"  Peak shave, t={t_alloc:.0f}s")
            a_opt = peak_shave(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_shave")
        
        # Final SCA with small K
        remaining = deadline - time.monotonic() - 40
        if remaining > 10:
            t_alloc = min(remaining, 40)
            print(f"  Final SCA K=8, t={t_alloc:.0f}s")
            a_opt = sca_iterate(best_seq, time_limit=t_alloc, K=8, schedule='const')
            try_seq(a_opt, "final_sca")
        
        # Tiny perturbations
        n_best = len(best_seq)
        for _ in range(3):
            remaining = deadline - time.monotonic() - 10
            if remaining < 5:
                break
            noise = 0.001 + 0.001 * _
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise)
            a_opt = smoothed_projected_grad(f_pert, time_limit=min(remaining, 15), T_init=0.001, lr_init=0.02)
            try_seq(a_opt, f"perturb_{_}")
    
    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
