def project_simplex(v):

        return np.array([max(float(v[0]), 0.0)])

"""
Multi-strategy optimization for minimum peak autocorrelation.
Uses Nesterov-accelerated gradient on softmax-smoothed objective,
iterative peaking QP, and local search. Different from previous FW-LP approach.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(a, T):
    """Compute gradient of smoothed objective (1/T)log(sum exp(T*b[k])).
    Assumes sum(a) = 1. Returns (grad, smoothed_val, b)."""
    n = len(a)
    b = fast_convolve(a)  # length 2n-1
    # Softmax weights
    Tb = T * b
    Tb_shifted = Tb - Tb.max()
    w = np.exp(Tb_shifted)
    w_sum = w.sum()
    w = w / w_sum
    smoothed_val = (Tb.max() + np.log(w_sum)) / T
    
    # Gradient: grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    # This is cross-correlation of w and a
    cross = np.correlate(w, a, mode='valid')  # length n
    grad = 2.0 * cross
    
    return grad, smoothed_val, b, w


def nesterov_smoothed_descent(a0, time_limit, T_init=0.01, lr_init=0.1, momentum_init=0.9):
    """Nesterov-accelerated gradient descent on smoothed objective."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    T = T_init
    lr = lr_init
    momentum = momentum_init
    v = np.zeros(n)  # velocity for Nesterov
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Nesterov lookahead
        y = project_simplex(a - momentum * v)
        
        # Compute gradient at y
        grad, smooth_val, b, w = compute_smoothed_grad(y, T)
        
        # Update velocity
        v = momentum * v + lr * grad
        
        # Update position
        a_new = project_simplex(y - v)
        
        # Check actual objective (not smoothed)
        obj = fast_convolve(a_new).max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive parameters
        if no_improve > 50:
            T *= 1.05
            lr *= 0.95
            momentum = min(momentum * 1.001, 0.999)
        if no_improve > 200:
            # Restart from best with perturbation
            rng = np.random.default_rng(it * 7919)

            v = np.zeros(n)
            T = T_init
            lr = lr_init
            momentum = momentum_init
            no_improve = 0
        
        # Periodic annealing
        if it % 500 == 0:
            T *= 0.8
            lr = lr_init * (0.9 ** (it // 500))
    
    return best_a, best_obj


def peaking_qp(a0, time_limit, max_iters=100):
    """Iteratively reduce the peak autocorrelation by solving QP subproblems.
    At each step, find the peak lag k*, then solve:
    min a^T C_{k*} a s.t. a >= 0, sum(a) = 1, ||a - a0||_2 <= eps
    """

    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    eps = 0.1  # ball radius
    
    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        # Find peak lag
        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        peak_val = b[k_star]
        
        # Build the QP: min a^T C_k a s.t. a >= 0, sum(a)=1, (a-a0)^T(a-a0) <= eps^2
        # The objective is a^T C_k a where C_k[j,l] = 1 if j+l = k (with bounds), else 0
        # Actually C_k[j,l] = 1/2 if j+l=k, j!=l, and C_k[j,j] = 1 if 2j=k
        # For the QP solver, we need the Hessian H where objective = a^T H a + f^T a + c
        # Here objective = a^T C_k a (no linear term, no constant)
        
        # Build C_k (sparse)
        C = np.zeros((n, n))
        for j in range(n):
            l = k_star - j
            if 0 <= l < n:
                if j == l:
                    C[j, l] = 1.0
                else:
                    C[j, l] = 0.5
                    C[l, j] = 0.5
        
        # Solve QP using scipy minimize
        # min a^T C a s.t. a >= 0, sum(a) = 1, ||a - a0|| <= eps
        # Use SLSQP
        
        def qf(x):
            return x @ C @ x
        
        def qf_grad(x):
            return 2.0 * C @ x
        
        constraints = [
            {'type': 'eq', 'fun': lambda x: np.sum(x) - 1.0, 'jac': lambda x: np.ones(n)},
            {'type': 'ineq', 'fun': lambda x: eps**2 - np.sum((x - a)**2)}
        ]
        bounds = [(0, None)] * n
        
        try:
            res = minimize(qf, a, jac=qf_grad, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-10})
            if res.success:
                a_new = np.maximum(res.x, 0)
                a_new = a_new / a_new.sum()
                new_obj = fast_convolve(a_new).max()
                if new_obj < best_obj - 1e-15:
                    best_obj = new_obj
                    best_a = a_new.copy()
                    a = a_new
                    # Gradually reduce eps
                    eps *= 0.95
                else:
                    # Try direct projection
                    a = a_new
            else:
                break
        except Exception:
            break
        
        if eps < 0.01:
            # Reset eps for exploration
            rng = np.random.default_rng(it * 31)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            eps = 0.05
    
    return best_a, best_obj


def iterative_flatten(a0, time_limit, T_schedule=None):
    """Iteratively flatten the autocorrelation using weighted gradient steps.
    At each step, weight the lags proportionally to how much they exceed the average.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    T = 0.01
    lr = 0.05
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        b = fast_convolve(a)
        
        # Compute "excess" weights: how much each lag exceeds the average
        avg = 1.0 / len(b)  # since sum(b) = (sum a)^2 = 1
        excess = np.maximum(b - avg, 0)
        total_excess = excess.sum()
        
        if total_excess < 1e-15:
            # Already flat (within numerical precision)
            break
        
        # Normalize excess to get weights
        w = excess / total_excess
        # Blend with uniform for stability
        w = 0.5 * w + 0.5 / len(b)
        w = w / w.sum()
        
        # Compute gradient
        cross = np.correlate(w, a, mode='valid')
        grad = 2.0 * cross
        
        # Line search
        best_a_trial = a
        best_obj_trial = best_obj
        for trial_lr in [lr, lr*0.5, lr*0.25, lr*2]:
            candidate = project_simplex(a - trial_lr * grad)
            obj = fast_convolve(candidate).max()
            if obj < best_obj_trial:
                best_obj_trial = obj
                best_a_trial = candidate
        
        a = best_a_trial
        if best_obj_trial < best_obj - 1e-15:
            best_obj = best_obj_trial
            best_a = a.copy()
        else:
            # No improvement, adjust
            lr *= 0.9
            if lr < 0.001:
                # Perturb and restart
                rng = np.random.default_rng(it * 104729)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                lr = 0.05
    
    return best_a, best_obj


def try_eval(a, label, best_ref):
    """Evaluate a sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_ref
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_ref
    if np.isfinite(score) and score < best_ref[1]:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return (a_norm.copy(), score)
    return best_ref
