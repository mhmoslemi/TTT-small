```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Multi-n exploration with structured initializations
- L-BFGS-B on log-sum-exp smoothed objective with softmax parameterization (scipy)
- Fourier low-dimensional parameterization for basin-hopping
- Nesterov-accelerated subgradient on true max
- Block coordinate descent for final polish
- Adaptive temperature scheduling
"""
import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_smoothed(a, p):
    """Compute gradient of smoothed objective w.r.t. a, given weights p on convolution peaks."""
    # grad[j] = 2 * sum_m p[j+m] * a[m] = 2 * np.correlate(p, a, mode='valid')[j]
    corr = np.correlate(p, a, mode='valid')
    return 2.0 * corr


def lse_val_grad(x, T):
    """Value and gradient of log-sum-exp smoothed objective using softmax parameterization.
    x is the raw (unnormalized) parameter vector.
    Returns (objective_value, gradient_wrt_x).
    """
    # Softmax
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()
    n = len(a)
    
    # Convolution
    c = fast_convolve(a)
    
    # Log-sum-exp
    scaled = c / T
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    
    val = T * np.log(np.sum(exp_scaled))
    
    # Gradient w.r.t. a
    g = compute_grad_smoothed(a, p)
    
    # Chain rule: dval/dx_j = a_j * (g_j - sum(g*a))
    sum_ga = np.dot(g, a)
    grad_x = a * (g - sum_ga)
    
    return val, grad_x


def scipy_smoothed_optimize(a0, time_limit, T_init=0.05, T_min=0.001, verbose=False):
    """Run scipy L-BFGS-B on the smoothed objective with decreasing temperature.
    Returns the optimized probability vector a.
    """
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    # Initialize x from a (inverse softmax)
    x = np.log(a + 1e-30)
    x -= x.max()  # center for numerical stability
    
    T = T_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit - 1.0:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break
        
        # Run L-BFGS-B for this temperature
        max_iter = 300
        if remaining < 5.0:
            max_iter = 50
        
        res = minimize(
            lse_val_grad, x, method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-12,
                     'maxls': 20, 'maxfun': 2000}
        )
        
        x = res.x
        a_new = np.exp(x - x.max())
        a_new = a_new / a_new.sum()
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            # Update x for next round
            x = np.log(a + 1e-30)
            x -= x.max()
        
        if verbose:

        # Decrease temperature
        T *= 0.65
        if T < T_min:

            break
    
    return best_a, best_obj


def nesterov_optimize(a0, time_limit, lr_init=0.005, momentum_init=0.9, verbose=False):
    """Nesterov-accelerated subgradient for min max_k conv[a,a]_k subject to a in simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    v = np.zeros(n)
    lr = lr_init
    mom = momentum_init
    y = a.copy()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit - 0.5:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        # Nesterov lookahead
        y = a + mom * v
        y = project_simplex(y)
        
        # Subgradient at y
        conv_y = fast_convolve(y)
        k_peak = int(np.argmax(conv_y))
        
        # Gradient of c[k_peak] w.r.t. a
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * y[k_peak - idx]
        
        # Update
        y_new = project_simplex(y - lr * grad)
        v = y_new - y
        a = y_new
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        lr *= 0.9997
        mom = min(0.995, mom + 0.00003)
        
        if no_improve > 150:
            lr = max(lr * 0.85, 1e-5)
            mom *= 0.98
        if no_improve > 400:
            rng = np.random.default_rng((it * 48271 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = lr_init * 0.7
            mom = momentum_init
            no_improve = 0
        
        if verbose and it % 500 == 0:
            print(f"        nesterov it={it}, obj={best_obj:.12f}, lr={lr:.7f}")
    
    return best_a, best_obj


def block_coordinate_descent(a0, time_limit, block_size=32, verbose=False):

    n = len(a)

        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        # Pick a random block
        start = rng.integers(0, max(1, n - block_size))
        block = np.arange(start, start + min(block_size, n - start))
        rest = np.setdiff1d(np.arange(n), block)

            continue
        
        rest_sum = a[rest].sum()
        if rest_sum < 1e-15:

            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            grad[idx] = 2.0 * a[k_peak - idx]
            
            # Only update block coordinates (keep rest fixed)
            # Transfer from block to rest (or vice versa) to maintain sum

            rest_grad_mean = grad[rest].mean() if len(rest) > 0 else 0.0
            
            lr_block = 0.002
            a[block] -= lr_block * block_grad
            # Redistribute to maintain sum
            a[rest] += lr_block * block_grad.sum() / len(rest)
            
            # Project block to be non-negative
            neg_mask = a[block] < 0
            if neg_mask.any():
                deficit = (-a[block][neg_mask]).sum()
                a[block] = np.maximum(a[block], 0)
                # Take from rest
                if rest_sum > 0:
                    reduce = deficit / len(rest)
                    a[rest] = np.maximum(a[rest] - reduce, 0)
                    # Re-normalize
                    a = a / a.sum()
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
        else:
            a = best_a.copy()  # Revert if no improvement
    
    return best_a, best_obj


def fourier_param_optimize(n, time_limit, num_harmonics=15, verbose=False):
    """Optimize a low-dimensional Fourier parameterization.
    a[i] = exp(sum_m x_m * cos(2*pi*m*i/n)) / Z
    """
    def fourier_obj(x):
        i = np.arange(n, dtype=np.float64)
        log_a = np.zeros(n)
        for m in range(1, len(x)):
            log_a += x[m] * np.cos(2 * np.pi * m * i / n)
        # x[0] is the offset
        log_a += x[0]
        log_a -= log_a.max()
        a = np.exp(log_a)
        a = a / a.sum()
        
        c = fast_convolve(a)
        scaled = c / 0.01
        scaled -= scaled.max()
        val = 0.01 * np.log(np.sum(np.exp(scaled)))
        return val
    
    # Initial parameters: flat (all zeros except x[0]=0)
    x0 = np.zeros(num_harmonics)
    
    t0 = time.monotonic()
    best_a = np.ones(n) / n
    best_val = fourier_obj(x0)
    
    # Try multiple random starts
    rng = np.random.default_rng(999)
    for trial in range(10):
        if time.monotonic() - t0 > time_limit - 2.0:
            break
        
        if trial == 0:
            x_start = x0.copy()
        else:
            x_start = rng.standard_normal(num_harmonics) * 0.3
            x_start[0] = 0.0  # Fix offset at 0 (doesn't matter after normalization)
        
        res = minimize(fourier_obj, x_start, method='L-BFGS-B',
                       options={'maxiter': 200, 'ftol': 1e-14, 'maxls': 30})
        
        # Convert back to a
        i = np.arange(n, dtype=np.float64)
        log_a = np.zeros(n)
        for m in range(1, len(res.x)):
            log_a += res.x[m] * np.cos(2 * np.pi * m * i / n)
        log_a += res.x[0]
        log_a -= log_a.max()
        a = np.exp(log_a)
        a = a / a.sum()
        
        obj = fast_convolve(a).max()
        if obj < fast_convolve(best_a).max():
            best_a = a.copy()
    
    return best_a

def generate_structured_init(n, style=0):
    """Generate different structured initializations."""
    x = np.arange(n, dtype=np.float64)
    if style == 0:
        return np.ones(n) / n
    elif style == 1:
        # Slight U-shape
        u = 1.0 + 0.3 * np.cos(np.pi * x / (n - 1))
        return u / u.sum()
    elif style == 2:
        # Slight inverted U (peak in center)
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
        return g / g.sum()
    elif style == 3:
        # Two peaks
        g = (np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.12 * n)) ** 2) +
             np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.12 * n)) ** 2))
        return g / g.sum()
    elif style == 4:
        # Cosine window
        c = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
        c = np.maximum(c, 0.001)
        return c / c.sum()
    elif style == 5:
        # Sinc-like (abrupt edges, flat center)
        t = x / (n - 1) * 2.0 - 1.0
        s = np.sinc(3 * t)
        s = np.abs(s)
        s = np.maximum(s, 0.001)
        return s / s.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Multi-phase hybrid optimization combining scipy L-BFGS-B
    on smoothed objectives, Nesterov subgradient, Fourier parameterization, and
    block coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")
    
    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")
    
    # === Phase 1: Quick multi-n exploration (~250s) ===
    print("\n=== Phase 1: Multi-n exploration ===")
    t1 = time.monotonic()
    
    candidate_ns = sorted(set([
        n_parent,
        max(256, n_parent - 128),
        min(2048, n_parent + 128),
        512, 768, 1024, 1536,
    ]))
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 2048]
    
    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 20:
            break
        
        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            start = start / start.sum()
        
        t_alloc = min(remaining, 35)
        print(f"  scipy L-BFGS-B n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = scipy_smoothed_optimize(start, time_limit=t_alloc, T_init=0.05, T_min=0.005)
        try_seq(a_opt, f"scipy_n{n_try}")
        
        # Also try structured inits for non-parent n
        if n_try != n_parent:
            remaining = deadline - time.monotonic() - 1150
            if remaining > 15:
                for style in [0, 1, 4]:
                    if remaining < 5:
                        break
                    init_s = generate_structured_init(n_try, style)
                    a_opt2, _ = scipy_smoothed_optimize(init_s, time_limit=min(remaining, 15),
                                                        T_init=0.05, T_min=0.01)
                    try_seq(a_opt2, f"struct{style}_n{n_try}")
                    remaining = deadline - time.monotonic() - 1150
    
    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")
    
    # === Phase 2: Fourier parameterization (~200s) ===
    print("\n=== Phase 2: Fourier parameterization ===")
    t2 = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        fourier_ns = [n_best, min(2048, n_best * 2), max(256, n_best // 2)]
        for nf in fourier_ns:
            remaining = deadline - time.monotonic() - 800
            if remaining < 15:
                break
            t_alloc = min(remaining, 40)
            print(f"  Fourier n={nf}, harmonics=20, t={t_alloc:.0f}s")
            a_f = fourier_param_optimize(nf, time_limit=t_alloc, num_harmonics=20)
            try_seq(a_f, f"fourier_n{nf}")
        
        # Refine Fourier result with full-dim scipy
        remaining = deadline - time.monotonic() - 750
        if remaining > 15 and best_seq is not None:
            t_alloc = min(remaining, 40)
            print(f"  scipy refine after Fourier, t={t_alloc:.0f}s")
            a_opt, _ = scipy_smoothed_optimize(best_seq, time_limit=t_alloc, T_init=0.03, T_min=0.005)
            try_seq(a_opt, "scipy_after_fourier")
    
    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")
    
    # === Phase 3: Deep scipy optimization with temperature annealing (~700s) ===
    print("\n=== Phase 3: Deep scipy optimization ===")
    t3 = time.monotonic()
    
    if best_seq is not None:
        # Multiple temperature schedules
        schedules = [
            (0.08, 0.002),  # Wide to narrow
            (0.03, 0.001),  # Medium start
            (0.01, 0.0005), # Already narrow
            (0.15, 0.005),  # Very wide start
        ]
        
        for si, (T_init, T_min) in enumerate(schedules):
            remaining = deadline - time.monotonic() - 300
            if remaining < 20:
                break
            t_alloc = min(remaining, 150)
            print(f"  scipy deep T=[{T_init}→{T_min}], t={t_alloc:.0f}s")
            a_opt, obj_local = scipy_smoothed_optimize(best_seq, time_limit=t_alloc,
                                                       T_init=T_init, T_min=T_min, verbose=True)
            try_seq(a_opt, f"scipy_deep_{si}")
            
            # Random restart from best
            if remaining > 30:
                a_restart = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.005)
                t_alloc2 = min(remaining - 30, 40)
                if t_alloc2 > 5:
                    a_opt2, _ = scipy_smoothed_optimize(a_restart, time_limit=t_alloc2,
                                                        T_init=T_init * 0.5, T_min=T_min)
                    try_seq(a_opt2, f"scipy_restart_{si}")
    
    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")
    
    # === Phase 4: Nesterov subgradient (~300s) ===
    print("\n=== Phase 4: Nesterov subgradient ===")
    t4 = time.monotonic()
    
    if best_seq is not None:
        nesterov_configs = [
            (0.008, 0.92),
            (0.004, 0.95),
            (0.002, 0.97),
            (0.01, 0.90),
        ]
        for ci, (lr_i, mom_i) in enumerate(nesterov_configs):
            remaining = deadline - time.monotonic() - 200
            if remaining < 10:
                break
            t_alloc = min(remaining, 60)
            print(f"  Nesterov lr={lr_i}, mom={mom_i}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_optimize(best_seq, time_limit=t_alloc,
                                         lr_init=lr_i, momentum_init=mom_i)
            try_seq(a_opt, f"nesterov_{ci}")
    
    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")
    
    # === Phase 5: Block coordinate descent + fine polish (~200s) ===
    print("\n=== Phase 5: Block CD and fine polish ===")
    t5 = time.monotonic()
    
    if best_seq is not None:
        # Block CD
        remaining = deadline - time.monotonic() - 100
        if remaining > 10:
            t_alloc = min(remaining, 50)
            print(f"  Block CD, t={t_alloc:.0f}s")
            a_opt, _ = block_coordinate_descent(best_seq, time_limit=t_alloc, block_size=16)
            try_seq(a_opt, "block_cd")
        
        # Fine scipy with very low temperature
        remaining = deadline - time.monotonic() - 80
        if remaining > 10:
            t_alloc = min(remaining, 50)
            print(f"  Fine scipy T=0.003→0.0003, t={t_alloc:.0f}s")
            a_opt, _ = scipy_smoothed_optimize(best_seq, time_limit=t_alloc,
                                               T_init=0.003, T_min=0.0003)
            try_seq(a_opt, "fine_scipy")
        
        # Interleave: Nesterov then scipy
        remaining = deadline - time.monotonic() - 60
        if remaining > 15 and best_seq is not None:
            a_n = nesterov_optimize(best_seq, time_limit=min(remaining, 30), lr_init=0.003, momentum_init=0.95)
            try_seq(a_n, "interleave_nest")
            if remaining > 30:
                a_s = scipy_smoothed_optimize(a_n, time_limit=min(remaining - 30, 25),
                                              T_init=0.002, T_min=0.0003)
                try_seq(a_s, "interleave_scipy")
        
        # Coordinate-level micro-optimization
        remaining = deadline - time.monotonic() - 40
        if remaining > 10 and best_seq is not None:
            a_micro = best_seq.copy()
            n_b = len(a_micro)
            for _ in range(50):
                if time.monotonic() > deadline - 10:
                    break
                # Pick random pair and try transfer
                i, j = rng.integers(0, n_b), rng.integers(0, n_b)
                if i == j:
                    continue
                for frac in [0.001, 0.005, 0.01, 0.02]:
                    candidate = a_micro.copy()
                    transfer = a_micro[i] * frac
                    candidate[i] -= transfer
                    candidate[j] += transfer
                    if candidate[i] < 0:
                        continue
                    candidate = candidate / candidate.sum()
                    c = fast_convolve(candidate)
                    if c.max() < fast_convolve(a_micro).max() - 1e-14:
                        a_micro = candidate
                        break
            try_seq(a_micro, "micro_coord")
    
    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")
    
    # === Phase 6: Final attempts with different n (~80s) ===
    print("\n=== Phase 6: Final n-exploration ===")
    t6 = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        final_ns = [max(256, n_best // 2), min(2048, n_best * 2)]
        for nf in final_ns:
            remaining = deadline - time.monotonic() - 20
            if remaining < 10 or nf == n_best:
                continue
            start_f = np.interp(np.linspace(0, 1, nf), np.linspace(0, 1, n_best), best_seq)
            start_f = np.maximum(start_f, 0)
            start_f = start_f / start_f.sum()
            t_alloc = min(remaining, 25)
            a_f = scipy_smoothed_optimize(start_f, time_limit=t_alloc, T_init=0.03, T_min=0.003)
            try_seq(a_f, f"final_n{nf}")
    
    print(f"  Phase 6 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t6:.0f}s)")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
