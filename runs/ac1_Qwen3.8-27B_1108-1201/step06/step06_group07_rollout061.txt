import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 for non-negative a."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max() / (s * s)


def sca_optimize(f0, time_limit, verbose=False):
    """SCA with LP oracle, improved M tracking."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    # Precompute LP matrix
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    row_idx = np.array(rows, dtype=np.int32)
    col_idx = np.array(cols, dtype=np.int32)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0
        k_arr = row_idx
        i_arr = col_idx
        vals = f[k_arr - i_arr]
        A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True,
                                                   'dual_feasibility_tolerance': 1e-9,
                                                   'primal_feasibility_tolerance': 1e-9})
            if not res.success or res.x is None:
                M *= 1.05
                no_improve += 1
                if no_improve > 30:
                    M = fast_convolve(f).max() * 1.01
                    no_improve = 0
                continue
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs < 1e-15:
                M *= 1.05
                no_improve += 1
                continue
            g = g / gs
        except Exception:
            M *= 1.05
            no_improve += 1
            continue

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
                f = project_simplex(f + rng.standard_normal(n) * 0.002)
                M = fast_convolve(f).max() * 1.01
                no_improve = 0
            continue

        # Line search
        best_val = fast_convolve(f).max()
        best_ft = f.copy()
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_ft = ft.copy()

        if best_val < best_obj - 1e-15:
            f = best_ft
            best_obj = best_val
            best_f = f.copy()
            M = best_val * 0.999
            no_improve = 0
        else:
            no_improve += 1
            M *= 0.998

        if no_improve > 50:
            rng = np.random.default_rng((it * 31831 + 7) % (2**31 - 1))
            scale = 0.001 + 0.002 * min(1.0, (no_improve - 50) / 50.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max() * 1.01
            no_improve = 0

    return best_f, best_obj


def freq_polish(a0, time_limit, verbose=False):
    """
    Frequency-domain iterative polishing.
    Uses the relationship conv(a,a) = IDFT(|DFT(a)|^2) to
    directly attack the peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    gamma = 0.5  # Step size for frequency modification

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # Compute FFT
        A = np.fft.rfft(a, fft_size)
        B = np.abs(A) ** 2
        phase = np.angle(A)

        # Compute current convolution
        conv = np.fft.irfft(B, fft_size)[:2*n-1]
        peak = conv.max()

        if peak < best_obj - 1e-15:
            best_obj = peak
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Identify peak positions
        threshold = peak * 0.95
        peak_positions = np.where(conv >= threshold)[0]

        # For each peak position m, the contribution from frequency k is:
        # B[k] * cos(2*pi*k*m/N) (real part)
        # To reduce the peak, we want to decrease B[k] where it positively
        # contributes to the peak.

        # Compute the "peak influence" for each frequency bin
        m_star = int(np.argmax(conv))
        # Influence of B[k] on conv[m_star]:
        # conv[m] = (1/N) * sum_k B[k] * cos(2*pi*k*m/N)  (real, for real a)
        # Wait, this isn't quite right for the auto-convolution.
        # Let me use a different approach.

        # Subgradient approach in frequency domain:
        # grad_a[j] = sum_k w_k * (a[k-j] + a[j-k]) for active peaks k
        # This is the same as the time-domain subgradient.

        # Instead, let's use a direct approach:
        # Modify B to reduce the peak, then reconstruct a.

        # Method: Reduce B at frequencies that most increase the peak
        m_max = int(np.argmax(conv))
        # The value at m_max is:
        # conv[m_max] = (1/N) * Re(sum_k B[k] * exp(2*pi*i*k*m_max/N))
        # = (1/N) * sum_k B[k] * cos(2*pi*k*m_max/N)
        # (for the full circular convolution; for linear it's slightly different)

        # So to reduce conv[m_max], we should decrease B[k] where cos(2*pi*k*m_max/N) > 0

        # But we need to be careful: decreasing B[k] changes a, which changes B.
        # Use a small step.

        cos_val = np.cos(2 * np.pi * np.arange(len(B)) * m_max / fft_size)
        # Only reduce where it helps
        reduce_mask = cos_val > 0
        # Amount to reduce: proportional to B[k] * cos_val[k]
        reduction = gamma * B * reduce_mask * cos_val / (B.sum() + 1e-15)

        B_new = np.maximum(B - reduction, 1e-10)

        # Reconstruct A_new with modified amplitude
        A_new = np.sqrt(B_new) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]

        # Project to feasible set
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Check improvement
        new_conv = fast_convolve(a_new)
        new_peak = new_conv.max()

        if new_peak < peak - 1e-15:
            a = a_new
            if new_peak < best_obj:
                best_obj = new_peak
                best_a = a.copy()
                no_improve = 0
        else:
            # Blend
            for alpha in [0.1, 0.3, 0.5]:
                a_blend = (1 - alpha) * a + alpha * a_new
                a_blend = np.maximum(a_blend, 0)
                sb = a_blend.sum()
                if sb < 1e-15:
                    continue
                a_blend = a_blend / sb
                cb = fast_convolve(a_blend).max()
                if cb < peak:
                    a = a_blend
                    if cb < best_obj:
                        best_obj = cb
                        best_a = a.copy()
                        no_improve = 0
                    break

        if no_improve > 30:
            gamma *= 0.8
            if gamma < 0.01:
                gamma = 0.5
                rng = np.random.default_rng((it * 12345 + 7) % (2**31 - 1))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                best_obj = fast_convolve(a).max()
                best_a = a.copy()
            no_improve = 0

    return best_a, best_obj


def local_polish(a0, time_limit, verbose=False):
    """
    Aggressive local polishing using coordinate descent and pair swaps.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_peak = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        peak_pos = int(np.argmax(conv))
        peak_val = conv[peak_pos]

        # Coordinate descent: try adjusting each element
        improved = False
        step = 0.001

        for _ in range(3):  # Passes
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            conv = fast_convolve(a)
            peak_pos = int(np.argmax(conv))
            peak_val = conv[peak_pos]

            # Elements that contribute to the peak
            js = max(0, peak_pos - n + 1)
            je = min(n - 1, peak_pos)

            for j in range(n):
                if time.monotonic() - t0 > time_limit - 0.5:
                    break
                # Try increasing and decreasing a[j]
                for delta in [step, -step, 2*step, -2*step]:
                    a_trial = a.copy()
                    a_trial[j] += delta
                    if a_trial[j] < 0:
                        continue
                    # Renormalize (subtract delta from others proportionally)
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    val = fast_convolve(a_trial).max()
                    if val < peak_val - 1e-15:
                        a = a_trial
                        peak_val = val
                        improved = True
                        break

            if time.monotonic() - t0 > time_limit - 0.5:
                break

        if peak_val < best_peak - 1e-15:
            best_peak = peak_val
            best_a = a.copy()
        elif not improved:
            # Random perturbation
            rng = np.random.default_rng((it * 999983 + 1) % (2**31 - 1))
            pert = rng.standard_normal(n) * 0.001
            a_pert = project_simplex(a + pert)
            val = fast_convolve(a_pert).max()
            if val < best_peak:
                a = a_pert
                best_peak = val
                best_a = a.copy()
            else:
                # Try from best_a
                a_pert2 = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                val2 = fast_convolve(a_pert2).max()
                if val2 < best_peak:
                    a = a_pert2
                    best_peak = val2
                    best_a = a.copy()
                else:
                    a = best_a.copy()

    return best_a, best_peak


def quadratic_residue_init(n, rng):
    """Create a sequence based on quadratic residues for a prime near n."""
    # Find the smallest prime >= n
    def is_prime(x):
        if x < 2:
            return False
        if x % 2 == 0:
            return x == 2
        for i in range(3, int(x**0.5) + 1, 2):
            if x % i == 0:
                return False
        return True

    p = n
    while not is_prime(p):
        p += 1

    # Create quadratic residue sequence of length p
    residues = set()
    for x in range(1, p):
        residues.add((x * x) % p)

    a = np.zeros(p)
    for r in residues:
        a[r] = 1.0

    # Truncate or pad to length n
    if p >= n:
        a = a[:n]
    else:
        a = np.pad(a, (0, n - p))

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- SCA with LP oracle (improved M tracking and dual exploitation)
- Frequency-domain iterative peak reduction
- Aggressive local polish (coordinate descent on peak contributors)
- Multi-n exploration with structured initializations (quadratic residues)
- Simulated annealing for global escape

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 for non-negative a with sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max() / (s * s)


def sca_optimize(f0, time_limit, verbose=False):
    """SCA with LP oracle and adaptive M tracking."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    # Precompute LP matrix structure
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    row_idx = np.array(rows, dtype=np.int32)
    col_idx = np.array(cols, dtype=np.int32)
    k_arr = row_idx
    i_arr = col_idx

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
        try:
            vals = f[k_arr - i_arr]
            A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
            c_vec = -np.ones(n)
            b_ub = np.full(2 * n - 1, M)
            bounds = [(0, None)] * n
            res = linprog(c_vec, A_ub=A_sparse, b_ub=b_ub, bounds=bounds,
                          method='highs',
                          options={'presolve': True,
                                   'dual_feasibility_tolerance': 1e-10,
                                   'primal_feasibility_tolerance': 1e-10})
            if not res.success or res.x is None:
                raise ValueError("LP failed")
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs < 1e-15:
                raise ValueError("Zero solution")
            g = g / gs
        except Exception:
            M *= 1.05
            no_improve += 1
            if no_improve > 40:
                M = fast_convolve(f).max() * 1.01
                no_improve = 0
            continue

            no_improve += 1

                rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
                f = project_simplex(f + rng.standard_normal(n) * 0.002)
                M = fast_convolve(f).max() * 1.01
                no_improve = 0
            continue

        best_val = fast_convolve(f).max()
        best_ft = f.copy()
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_ft = ft.copy()

        if best_val < best_obj - 1e-15:
            f = best_ft
            best_obj = best_val
            best_f = f.copy()
            M = best_val * 0.9995
            no_improve = 0
        else:
            no_improve += 1
            M *= 0.9985

        if no_improve > 60:
            rng = np.random.default_rng((it * 31831 + 7) % (2**31 - 1))

            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max() * 1.01
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    SCA it={it}, obj={best_obj:.10f}, M={M:.10f}")

    return best_f, best_obj


def freq_domain_polish(a0, time_limit, verbose=False):
    """
    Frequency-domain iterative method:
    Modifies the power spectrum |DFT(a)|^2 to reduce peak of inverse transform.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_peak = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    gamma = 0.3

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # Current state
        conv = fast_convolve(a)
        peak = conv.max()
        m_max = int(np.argmax(conv))

        if peak < best_peak - 1e-15:
            best_peak = peak
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # FFT of current a
        A = np.fft.rfft(a, fft_size)
        B = np.abs(A) ** 2
        phase = np.angle(A)
        freq_idx = np.arange(len(B))

        # The peak at m_max gets contribution from each frequency k proportional to
        # B[k] * cos(2*pi*k*m_max/fft_size)
        # We want to reduce B[k] where this contribution is positive
        cos_peak = np.cos(2 * np.pi * freq_idx * m_max / fft_size)

        # Also consider the second-highest peak
        sorted_conv = np.sort(conv)[::-1]
        if len(sorted_conv) > 1:
            m_2nd = int(np.argsort(conv)[::-1][1])
            cos_2nd = np.cos(2 * np.pi * freq_idx * m_2nd / fft_size)
            combined = 0.7 * cos_peak + 0.3 * cos_2nd
        else:
            combined = cos_peak

        # Reduce B where it positively contributes to peaks
        positive_mask = combined > 0
        # Proportional reduction
        B_reduction = gamma * B * positive_mask * combined / (np.sum(B * positive_mask * combined) + 1e-15)
        B_new = np.maximum(B - B_reduction, 1e-12)

        # Reconstruct
        A_new = np.sqrt(B_new) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]

        # Project to feasible set
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Evaluate and possibly accept
        new_conv = fast_convolve(a_new)
        new_peak = new_conv.max()

        accepted = False
        if new_peak < peak - 1e-15:
            a = a_new
            accepted = True
        else:
            # Try blending
            for alpha in [0.1, 0.2, 0.3, 0.5]:
                a_blend = (1 - alpha) * a + alpha * a_new
                a_blend = np.maximum(a_blend, 0)
                sb = a_blend.sum()
                if sb < 1e-15:
                    continue
                a_blend = a_blend / sb
                cb = fast_convolve(a_blend).max()
                if cb < peak - 1e-15:
                    a = a_blend
                    new_peak = cb
                    accepted = True
                    break

        if not accepted:
            # Geometric random walk in frequency domain
            rng = np.random.default_rng((it * 48271 + 3) % (2**31 - 1))
            B_noise = B * (1.0 + rng.standard_normal(len(B)) * gamma * 0.5)
            B_noise = np.maximum(B_noise, 1e-12)
            A_noisy = np.sqrt(B_noise) * np.exp(1j * phase)
            a_noisy = np.fft.irfft(A_noisy, fft_size)[:n]
            a_noisy = np.maximum(a_noisy, 0)
            sn = a_noisy.sum()
            if sn > 1e-15:
                a_noisy = a_noisy / sn
                cn = fast_convolve(a_noisy).max()
                if cn < best_peak:
                    a = a_noisy
                    best_peak = cn
                    best_a = a.copy()
                elif cn < peak * 1.01:
                    # Accept with probability (simulated annealing style)
                    if rng.random() < 0.1:
                        a = a_noisy

        if no_improve > 40:
            gamma *= 0.85
            if gamma < 0.02:
                gamma = 0.5
                rng = np.random.default_rng((it * 1234567 + 11) % (2**31 - 1))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                best_peak = fast_convolve(a).max()
                best_a = a.copy()
            no_improve = 0

    return best_a, best_peak


def local_polish(a0, time_limit, verbose=False):
    """
    Coordinate descent on peak-contributing elements with adaptive step size.
    Focuses on the top contributors to the maximum peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_peak = fast_convolve(a).max()

    t0 = time.monotonic()
    step = 0.002
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.3:
            break

        conv = fast_convolve(a)
        peak = conv.max()
        m_max = int(np.argmax(conv))

        if peak < best_peak - 1e-15:
            best_peak = peak
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Find contributors to the peak
        js = max(0, m_max - n + 1)
        je = min(n - 1, m_max)

        # Contribution of each j to the peak: a[j] * a[m_max - j]
        contrib = np.zeros(n)
        for j in range(js, je + 1):
            other = m_max - j
            if 0 <= other < n:
                contrib[j] = a[j] * a[other]

        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break

        # Try modifying top contributors
        top_indices = np.argsort(contrib)[::-1][:min(50, n)]

        improved_this_round = False
        for idx in top_indices:
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            j = int(idx)
            if contrib[j] < 1e-10:
                continue

            # Try reducing a[j] (move mass to non-contributing positions)
            delta = step * a[j]
            if delta < 1e-8:
                delta = 1e-6

            # Move mass from j to a random non-contributing position
            non_contrib = np.where(contrib < 0.01 * max(contrib))[0]
            if len(non_contrib) == 0:
                non_contrib = np.array([0, n - 1])

            for target in non_contrib[:3]:
                a_trial = a.copy()
                a_trial[j] -= delta
                a_trial[int(target)] += delta
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                val = fast_convolve(a_trial).max()
                if val < peak - 1e-15:
                    a = a_trial
                    peak = val
                    improved_this_round = True
                    break

            if improved_this_round:
                # Recompute contributions
                conv = fast_convolve(a)
                m_max = int(np.argmax(conv))
                js = max(0, m_max - n + 1)
                je = min(n - 1, m_max)
                contrib = np.zeros(n)
                for jj in range(js, je + 1):
                    other = m_max - jj
                    if 0 <= other < n:
                        contrib[jj] = a[jj] * a[other]

        if not improved_this_round:
            # Try small random perturbations
            rng = np.random.default_rng((int(time.monotonic() * 1000) + n) % (2**31 - 1))
            pert_scale = 0.001 * (1 + stagnation * 0.01)
            a_pert = project_simplex(a + rng.standard_normal(n) * pert_scale)
            val = fast_convolve(a_pert).max()
            if val < best_peak:
                a = a_pert
                best_peak = val
                best_a = a.copy()
                stagnation = 0
            else:
                # Occasionally jump from best
                if stagnation > 50 and rng.random() < 0.1:
                    a = project_simplex(best_a + rng.standard_normal(n) * 0.005)

        # Adaptive step
        if stagnation > 20:
            step *= 0.9
        if step < 1e-5:
            step = 0.002
            if stagnation > 100:
                rng = np.random.default_rng((stagnation * 7 + 1) % (2**31 - 1))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
                best_peak = fast_convolve(a).max()
                best_a = a.copy()
                step = 0.002
                stagnation = 0

    return best_a, best_peak

        for k in peak_constraints:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                rows.append(k)
                cols.append(i)

        if len(rows) == 0:
            M *= 1.01
            no_improve += 1
            continue

        row_idx = np.array(rows, dtype=np.int32)
        col_idx = np.array(cols, dtype=np.int32)

        try:
            vals = f[row_idx - col_idx]
            A_sparse = csr_matrix((vals, (row_idx, col_idx)), shape=(len(peak_constraints), n))
            c_vec = -np.ones(n)
            b_ub = np.full(len(peak_constraints), M)
            bounds = [(0, None)] * n
            res = linprog(c_vec, A_ub=A_sparse, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
            if not res.success or res.x is None:
                raise ValueError("LP failed")
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs < 1e-15:
                raise ValueError("Zero solution")
            g = g / gs
        except Exception:
            M *= 1.03
            no_improve += 1
            if no_improve > 30:
                M = fast_convolve(f).max() * 1.01
                no_improve = 0
            continue

        # Check against ALL constraints (not just active ones)
        conv_g = np.array([np.dot(f, g[max(0, k-n+1):min(n, k+1)][::-1][:min(n, k+1)-max(0,k-n+1)])
                          for k in range(2*n-1)])
        # Actually just use fast_convolve for the blended solution
        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 20:
                rng = np.random.default_rng((it * 99991 + 7) % (2**31 - 1))
                f = project_simplex(f + rng.standard_normal(n) * 0.002)
                M = fast_convolve(f).max() * 1.01
                no_improve = 0
            continue

        # Line search
        best_val = fast_convolve(f).max()
        best_ft = f.copy()
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_ft = ft.copy()

        if best_val < best_obj - 1e-15:
            f = best_ft
            best_obj = best_val
            best_f = f.copy()
            M = best_val * 0.999
            no_improve = 0
        else:
            no_improve += 1
            M *= 0.998

        if no_improve > 50:
            rng = np.random.default_rng((it * 6151 + 3) % (2**31 - 1))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            M = fast_convolve(f).max() * 1.01
            no_improve = 0

    return best_f, best_obj


def quadratic_residue_seq(n):
    """Create a sequence based on quadratic residues mod p (p prime >= n)."""
    def is_prime(x):
        if x < 2:
            return False
        if x < 4:
            return True
        if x % 2 == 0 or x % 3 == 0:
            return False
        i = 5
        while i * i <= x:
            if x % i == 0 or x % (i + 2) == 0:
                return False
            i += 6
        return True

    p = n
    while not is_prime(p):
        p += 1

    a = np.zeros(p)
    for x in range(p):
        if (x * x) % p < p // 2:
            a[x] = 1.0
        else:
            a[x] = 0.0
    # Include 0 as a quadratic residue
    a[0] = 1.0

    if p >= n:
        a = a[:n]
    else:
        a = np.pad(a, (0, n - p))

    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def simulated_annealing(a0, time_limit, verbose=False):
    """Simulated annealing for global escape from local minima."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_peak = fast_convolve(a).max()
    current_peak = best_peak

    T = 0.01  # Initial temperature
    T_min = 0.0001
    cooling = 0.995

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # Generate perturbation
        scale = T * 10
        a_new = project_simplex(a + rng.standard_normal(n) * scale)

        new_peak = fast_convolve(a_new).max()
        delta = new_peak - current_peak

        # Accept or reject
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_new
            current_peak = new_peak
            if new_peak < best_peak:
                best_peak = new_peak
                best_a = a.copy()

        T = max(T * cooling, T_min)

        if T <= T_min * 1.01:
            # Restart with higher temperature occasionally
            if rng.random() < 0.1:
                T = 0.005
                a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
                current_peak = fast_convolve(a).max()

    return best_a, best_peak


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: multi-phase hybrid combining SCA, frequency-domain methods,
    local polish, simulated annealing, and structured initialization.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Helper to try a sequence (inlined to avoid closure)
    _try_calls = []

    def _try(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            _try(parent, "parent")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        _try(parent, "uniform")

    n_parent = len(parent)

    # === Phase 1: Deep SCA on parent (600s) ===
    print("\n=== Phase 1: Deep SCA on parent ===")
    t1 = time.monotonic()
    t_alloc = min(550, (deadline - time.monotonic()) - 400)
    if t_alloc > 30:
        print(f"  SCA t={t_alloc:.0f}s")
        a_opt, _ = sca_optimize(parent, time_limit=t_alloc)
        _try(a_opt, "sca1")

        # Continue SCA from result
        remaining = deadline - time.monotonic() - 350
        if remaining > 30:
            t_alloc2 = min(100, remaining)
            a_opt2, _ = sca_optimize(a_opt, time_limit=t_alloc2)
            _try(a_opt2, "sca2")

    print(f"  P1 done: best={best_score:.8f}, t={time.monotonic()-t1:.0f}s")

    # === Phase 2: Peak-directed SCA (200s) ===
    print("\n=== Phase 2: Peak-Directed SCA ===")
    t2 = time.monotonic()
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 300
        if remaining > 30:
            t_alloc = min(150, remaining - 100)
            print(f"  PD-SCA t={t_alloc:.0f}s")
            a_opt, _ = peak_directed_sca(best_seq, time_limit=t_alloc)
            _try(a_opt, "pd_sca")

    print(f"  P2 done: best={best_score:.8f}, t={time.monotonic()-t2:.0f}s")

    # === Phase 3: Frequency domain polish (150s) ===
    print("\n=== Phase 3: Frequency Domain Polish ===")
    t3 = time.monotonic()
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 200
        if remaining > 20:
            t_alloc = min(120, remaining - 80)
            print(f"  Freq t={t_alloc:.0f}s")
            a_opt, _ = freq_domain_polish(best_seq, time_limit=t_alloc)
            _try(a_opt, "freq")

            # Second pass
            remaining = deadline - time.monotonic() - 150
            if remaining > 20:
                t_alloc = min(60, remaining - 60)
                a_opt2, _ = freq_domain_polish(a_opt, time_limit=t_alloc)
                _try(a_opt2, "freq2")

    print(f"  P3 done: best={best_score:.8f}, t={time.monotonic()-t3:.0f}s")

    # === Phase 4: Local polish (150s) ===
    print("\n=== Phase 4: Local Polish ===")
    t4 = time.monotonic()
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 150
        if remaining > 20:
            t_alloc = min(120, remaining - 60)
            print(f"  Local t={t_alloc:.0f}s")
            a_opt, _ = local_polish(best_seq, time_limit=t_alloc)
            _try(a_opt, "local")

    print(f"  P4 done: best={best_score:.8f}, t={time.monotonic()-t4:.0f}s")

    # === Phase 5: Multi-n exploration (200s) ===
    print("\n=== Phase 5: Multi-n Exploration ===")
    t5 = time.monotonic()

    # Try n values near the current best
    n_best = len(best_seq) if best_seq is not None else n_parent
    ns_near = []
    for delta in [-64, -32, -16, -8, -4, 4, 8, 16, 32, 64, 128]:
        nn = n_best + delta
        if 200 <= nn <= 3000:
            ns_near.append(nn)
    # Also try some structured n values
    for nn in [512, 640, 768, 900, 1024, 1200, 1536, 2048]:
        if nn not in ns_near and 200 <= nn <= 3000:
            ns_near.append(nn)
    ns_near = sorted(set(ns_near))

    for n_try in ns_near:
        remaining = deadline - time.monotonic() - 100
        if remaining < 20:
            break
        if best_seq is not None:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_best), best_seq)
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
        start = np.maximum(start, 0)
        ss = start.sum()
        if ss < 1e-10:
            continue
        start = start / ss

        t_alloc = min(20, remaining - 50)
        if t_alloc < 8:
            break
        print(f"  n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = sca_optimize(start, time_limit=t_alloc)
        _try(a_opt, f"n{n_try}")

    print(f"  P5 done: best={best_score:.8f}, t={time.monotonic()-t5:.0f}s")

    # === Phase 6: Simulated annealing + structured inits (100s) ===
    print("\n=== Phase 6: SA + Structured ===")
    t6 = time.monotonic()

    if best_seq is not None:
        n_b = len(best_seq)
        # Simulated annealing
        remaining = deadline - time.monotonic() - 50
        if remaining > 20:
            t_alloc = min(50, remaining - 20)
            print(f"  SA t={t_alloc:.0f}s")
            a_opt, _ = simulated_annealing(best_seq, time_limit=t_alloc)
            _try(a_opt, "sa")

        # Quadratic residue blend
        remaining = deadline - time.monotonic() - 20
        if remaining > 15:
            a_qr = quadratic_residue_seq(n_b)
            for alpha in [0.2, 0.4, 0.6, 0.8]:
                blend = (1 - alpha) * best_seq + alpha * a_qr
                blend = np.maximum(blend, 0)
                sb = blend.sum()
                if sb > 1e-10:
                    _try(blend / sb, f"qr_{alpha}")

            # SCA from QR
            if deadline - time.monotonic() > 20:
                t_alloc = min(25, deadline - time.monotonic() - 10)
                a_opt, _ = sca_optimize(a_qr, time_limit=t_alloc)
                _try(a_opt, "sca_qr")

    print(f"  P6 done: best={best_score:.8f}, t={time.monotonic()-t6:.0f}s")

    # === Phase 7: Final extended polish (remaining) ===
    print("\n=== Phase 7: Final Polish ===")
    t7 = time.monotonic()

    if best_seq is not None:
        # SCA final
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            t_alloc = min(40, remaining - 5)
            a_opt, _ = sca_optimize(best_seq, time_limit=t_alloc)
            _try(a_opt, "final_sca")

        # Local polish final
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            t_alloc = min(20, remaining - 2)
            a_opt, _ = local_polish(best_seq, time_limit=t_alloc)
            _try(a_opt, "final_local")

        # Freq final
        remaining = deadline - time.monotonic() - 2
        if remaining > 5:
            t_alloc = min(10, remaining - 1)
            a_opt, _ = freq_domain_polish(best_seq, time_limit=t_alloc)
            _try(a_opt, "final_freq")

    print(f"  P7 done: best={best_score:.8f}, t={time.monotonic()-t7:.0f}s")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs

    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
