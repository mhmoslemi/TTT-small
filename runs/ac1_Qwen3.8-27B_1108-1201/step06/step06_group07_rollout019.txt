import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gc_flatten(a, n_iter=200, blend=0.5):
    """Gerchberg-Saxton to flatten auto-correlation spectrum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    L = 1 << ((2 * n - 1) - 1).bit_length()
    if L < 2 * n - 1:
        L *= 2

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    for it in range(n_iter):
        F = np.fft.rfft(a, L)
        mag = np.abs(F)
        phase = np.angle(F)

        # Target: flatten |F|^2 proportional to 1 (giving flat autocorrelation)
        # Keep DC component fixed (F[0] = sum(a) = 1)
        target_power = np.sum(mag ** 2) / len(mag)
        target_mag = np.full_like(mag, np.sqrt(target_power))
        target_mag[0] = mag[0]  # Fix DC

        # Blend current and target
        new_mag = (1.0 - blend) * mag + blend * target_mag
        F_new = new_mag * np.exp(1j * phase)

        a_new = np.fft.irfft(F_new, L)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        obj = fast_convolve(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new

    return best_a

            if T > 200:
                T = T_init
        if no_improve > 80:
            # Perturb and restart
            rng = np.random.default_rng((it * 48271 + 17) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            T = T_init
            lr = 0.02
            no_improve = 0

        if lr < 0.001:
            lr = 0.02

    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    """Coordinate descent: optimize one element at a time."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        if sweeps % 5 == 0:
            pass
        improved_this_sweep = False
        conv = fast_convolve(a)

        for j in range(n):
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            t_old = a[j]
            if t_old < 1e-15:
                # Try setting it to a small positive value
                candidates = [0.001, 0.005, t_old + 0.001]
            else:
                # Try values around current
                spread = max(0.005, 0.3 * t_old)
                candidates = [
                    max(0.0, t_old - spread),
                    max(0.0, t_old - 0.5 * spread),
                    t_old,
                    t_old + 0.5 * spread,
                    t_old + spread,
                    max(0.0, 2.0 * t_old),
                ]

            best_t = t_old
            best_val = conv.max()

            for t_new in candidates:
                if abs(t_new - t_old) < 1e-15:
                    continue
                a[j] = t_new
                s = a.sum()
                if s < 1e-15:
                    a[j] = t_old
                    continue
                a_norm = a / s
                val = fast_convolve(a_norm).max()
                if val < best_val - 1e-13:
                    best_val = val
                    best_t = t_new

            a[j] = best_t
            s = a.sum()
            if s > 1e-15:
                a = a / s
            if best_t != t_old:
                improved_this_sweep = True
                conv = fast_convolve(a)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1

        if no_improve_sweeps > 5:
            rng = np.random.default_rng((sweeps * 65537 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve_sweeps = 0

    return best_a, best_obj


def build_lp_constraints(n, max_constraints=None):
    """Build sparse constraint matrix for LP auto-correlation bound."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    return rows, cols


def solve_lp_oracle(f, M, n, row_idx=None, col_idx=None, max_constr=None):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g or None.
    """
    if row_idx is None or col_idx is None:
        row_idx, col_idx = build_lp_constraints(n)

    k_arr = row_idx
    i_arr = col_idx
    vals = f[k_arr - i_arr]

    # Optionally reduce constraints to most critical ones
    if max_constr is not None and len(k_arr) > max_constr:
        # Compute current f*g for each constraint to find the tightest
        # For now, use all constraints (simpler)
        pass

    A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def sca_optimize(f0, time_limit):
    """SCA optimization with adaptive M and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, fast_convolve(f).max()
    f = f / s
    n = len(f)

    row_idx, col_idx = build_lp_constraints(n)

    conv = fast_convolve(f)
    M = conv.max() * 1.005
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        g_raw = solve_lp_oracle(f, M, n, row_idx, col_idx)

        if g_raw is None:
            M *= 1.02
            no_improve += 1
            if no_improve > 40:
                M = fast_convolve(f).max() * 1.005
                no_improve = 10
            continue

        g = g_raw / g_raw.sum()

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 25:
                rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
                f = project_simplex(f + rng.standard_normal(n) * 0.003)
                M = fast_convolve(f).max() * 1.005
                no_improve = 0
            continue

        # Line search over blending alphas
        best_val = fast_convolve(f).max()
        best_ft = f.copy()
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_ft = ft.copy()

        if best_val < best_obj - 1e-14:
            f = best_ft
            best_obj = best_val
            best_f = f.copy()
            M = best_val * 0.999
            no_improve = 0
        else:
            no_improve += 1
            M *= 0.998

        if no_improve > 50:
            rng = np.random.default_rng((it * 31831 + 7) % (2**31 - 1))
            scale = 0.002 + 0.002 * min(1.0, (no_improve - 50) / 50.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max() * 1.005
            no_improve = 0

    return best_f, best_obj


def spectral_redistribute(a0, time_limit):
    """
    Spectral redistribution: in the frequency domain, reduce peaks
    in |F|^2 that correspond to auto-correlation peaks.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()

    L = 1 << ((2 * n - 1) - 1).bit_length()
    if L < 2 * n - 1:
        L *= 2

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]

        # The peak at lag k is influenced by all frequency components
        # acorr[k] = (1/L) * sum_m |F[m]|^2 * exp(2*pi*i*m*k/L)
        # To reduce acorr[peak_idx], reduce |F[m]|^2 where exp(2pi*i*m*k/L) > 0
        # i.e., where cos(2*pi*m*k/L) > 0

        F = np.fft.rfft(a, L)
        mag = np.abs(F).copy()
        phase = np.angle(F)

        # Compute which frequencies positively contribute to the peak
        freqs = np.arange(L // 2 + 1)
        alignment = np.cos(2.0 * np.pi * freqs * peak_idx / L)

        # Reduce magnitude at aligned frequencies
        n_reduce = max(3, L // 20)
        # Pick top aligned frequencies
        aligned_idx = np.where(alignment > 0.3)[0]
        if len(aligned_idx) > n_reduce:
            # Pick the ones with highest current magnitude
            ranked = aligned_idx[np.argsort(mag[aligned_idx])[::-1][:n_reduce]]
        else:
            ranked = aligned_idx[:n_reduce]

        reduction = 0.02 + 0.01 * (no_improve % 10)
        mag[ranked] *= (1.0 - reduction)
        mag[0] = np.sum(a)  # Fix DC

        F_new = mag * np.exp(1j * phase)
        a_new = np.fft.irfft(F_new, L)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            no_improve += 1
            continue
        a_new = a_new / s_new

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            # Try opposite: boost anti-aligned frequencies
            anti_aligned = np.where(alignment < -0.3)[0]
            if len(anti_aligned) > 0:
                boost_idx = anti_aligned[np.argsort(mag[anti_aligned])[::-1][:n_reduce]]
                mag2 = mag.copy()
                mag2[boost_idx] *= 1.0 + 0.03
                mag2[0] = np.sum(a)
                F2 = mag2 * np.exp(1j * phase)
                a2 = np.fft.irfft(F2, L)[:n]
                a2 = np.maximum(a2, 0.0)
                s2 = a2.sum()
                if s2 > 1e-15:
                    a2 = a2 / s2
                    obj2 = fast_convolve(a2).max()
                    if obj2 < best_obj:
                        best_obj = obj2
                        best_a = a2.copy()
                        a = a2
                        no_improve = 0
                        continue
            # Reset
            a = best_a.copy()
            no_improve = 0

    return best_a, best_obj


def diverse_inits(n, parent, rng):
    """Generate diverse initial sequences."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    t = x / (n - 1)

    # Parent (resampled)
    if parent is not None:
        if len(parent) != n:
            px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        else:
            px = parent.copy()
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge-heavy
    edge_w = 1.0 / (np.minimum(x, n - 1 - x).astype(float) + 1.0) ** 1.2
    inits.append(edge_w / edge_w.sum())

    # Two-peak
    g2 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.06 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.06 * n)) ** 2)
    inits.append(g2 / g2.sum())

    # Cosine-modulated
    cos_mod = (1.0 + 0.4 * np.cos(4 * np.pi * t)) ** 2 + 0.05
    inits.append(cos_mod / cos_mod.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Sinusoidal
    sin_seq = (1.0 + 0.3 * np.sin(2 * np.pi * 3 * t)) ** 2 + 0.1
    inits.append(sin_seq / sin_seq.sum())

    # Triangular (peak in center)
    tri = 1.0 - np.abs(2.0 * t - 1.0)
    tri = tri ** 2 + 0.01
    inits.append(tri / tri.sum())

    # Parent perturbed
    if parent is not None:
        if len(parent) != n:
            px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        else:
            px = parent.copy()
        px = np.maximum(px + rng.standard_normal(n) * 0.05, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Combines GC, SCA, smoothed-max GD,
    coordinate descent, and spectral redistribution.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)
    print(f"Starting with n={n_parent}, best_score={best_score:.10f}")

    # === Phase 1: GC + diverse inits (150s) ===
    print("\n=== Phase 1: GC + Diverse Initialization ===")
    t1 = time.monotonic()

    ns_to_try = [n_parent]
    if n_parent != 768:
        ns_to_try.append(768)
    ns_to_try.append(n_parent + 128)
    ns_to_try.append(max(256, n_parent - 128))
    ns_to_try = [n for n in ns_to_try if 256 <= n <= 2048]
    ns_to_try = sorted(set(ns_to_try))

    for n_try in ns_to_try:
        remaining = deadline - time.monotonic()
        if remaining < 120:
            break
        inits = diverse_inits(n_try, parent, rng)
        for idx, init in enumerate(inits):
            remaining = deadline - time.monotonic()
            if remaining < 110:
                break
            t_alloc = min(8, remaining - 100)
            if t_alloc < 2:
                break
            # GC flatten
            a_gc = gc_flatten(init, n_iter=80, blend=0.7)
            try_seq(a_gc, f"gc_n{n_try}_{idx}")

    print(f"  P1 done: best={best_score:.10f}, t={time.monotonic()-t1:.0f}s")

    # === Phase 2: SCA from best (500s) ===
    print("\n=== Phase 2: Deep SCA ===")
    t2 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 400
        if remaining > 50:
            t_alloc = min(remaining, 350)
            print(f"  SCA n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = sca_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "sca_deep")

            remaining = deadline - time.monotonic() - 300
            if remaining > 30:
                t_alloc2 = min(remaining, 150)
                a_opt2, _ = sca_optimize(a_opt, time_limit=t_alloc2)
                try_seq(a_opt2, "sca_deep2")

    print(f"  P2 done: best={best_score:.10f}, t={time.monotonic()-t2:.0f}s")

    # === Phase 3: Smoothed-max gradient descent (300s) ===
    print("\n=== Phase 3: Smoothed-Max Gradient Descent ===")
    t3 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 350
        if remaining > 30:
            t_alloc = min(remaining, 200)
            print(f"  SmoothedMax t={t_alloc:.0f}s")
            a_opt, _ = smoothed_max_gd(best_seq, time_limit=t_alloc, T_init=40.0)
            try_seq(a_opt, "smoothed_max")

            remaining = deadline - time.monotonic() - 250
            if remaining > 20:
                t_alloc2 = min(remaining, 100)
                a_opt2, _ = smoothed_max_gd(a_opt, time_limit=t_alloc2, T_init=80.0)
                try_seq(a_opt2, "smoothed_max2")

    print(f"  P3 done: best={best_score:.10f}, t={time.monotonic()-t3:.0f}s")

    # === Phase 4: Spectral redistribution (200s) ===
    print("\n=== Phase 4: Spectral Redistribution ===")
    t4 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 300
        if remaining > 20:
            t_alloc = min(remaining, 120)
            print(f"  Spectral t={t_alloc:.0f}s")
            a_opt, _ = spectral_redistribute(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "spectral")

    print(f"  P4 done: best={best_score:.10f}, t={time.monotonic()-t4:.0f}s")

    # === Phase 5: Coordinate descent (200s) ===
    print("\n=== Phase 5: Coordinate Descent ===")
    t5 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 250
        if remaining > 20:
            t_alloc = min(remaining, 120)
            print(f"  CoordDesc t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_desc")

            remaining = deadline - time.monotonic() - 150
            if remaining > 15:
                t_alloc2 = min(remaining, 60)
                a_opt2, _ = coordinate_descent(a_opt, time_limit=t_alloc2)
                try_seq(a_opt2, "coord_desc2")

    print(f"  P5 done: best={best_score:.10f}, t={time.monotonic()-t5:.0f}s")

    # === Phase 6: Multi-start SCA with structured perturbations (300s) ===
    print("\n=== Phase 6: Multi-Start SCA ===")
    t6 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        # Structured perturbations
        x = np.arange(n_best, dtype=np.float64)
        t = x / (n_best - 1)

        perturbations = []
        # Sinusoidal perturbations at different frequencies
        for freq in [1, 2, 3, 5]:
            pert = best_seq * (1.0 + 0.02 * np.sin(2 * np.pi * freq * t))
            perturbations.append(pert)
        # Edge boost
        edge_boost = best_seq * (1.0 + 0.03 * np.exp(-x.astype(float) / (n_best * 0.1)) +
                                  0.03 * np.exp(-(n_best - 1 - x).astype(float) / (n_best * 0.1)))
        perturbations.append(edge_boost)
        # Center boost
        center_boost = best_seq * (1.0 + 0.02 * np.exp(-0.5 * ((x - n_best / 2) / (n_best * 0.15)) ** 2))
        perturbations.append(center_boost)
        # Random
        perturbations.append(best_seq + rng.standard_normal(n_best) * 0.005)
        perturbations.append(best_seq + rng.standard_normal(n_best) * 0.01)

        for idx, pert in enumerate(perturbations):
            remaining = deadline - time.monotonic() - 100
            if remaining < 20:
                break
            a_pert = project_simplex(np.maximum(pert, 0))
            t_alloc = min(30, remaining - 80)
            if t_alloc < 5:
                break
            print(f"  MultiStart {idx}, t={t_alloc:.0f}s")
            a_opt, _ = sca_optimize(a_pert, time_limit=t_alloc)
            try_seq(a_opt, f"multi_{idx}")

    print(f"  P6 done: best={best_score:.10f}, t={time.monotonic()-t6:.0f}s")

    # === Phase 7: Final polish with SCA + GD (remaining - 30s) ===
    print("\n=== Phase 7: Final Polish ===")
    t7 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 25
        if remaining > 20:
            t_alloc = min(remaining, 40)
            a_opt, _ = sca_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_sca")

        remaining = deadline - time.monotonic() - 15
        if remaining > 10:
            a_opt, _ = smoothed_max_gd(best_seq, time_limit=min(remaining, 20), T_init=60.0)
            try_seq(a_opt, "final_gd")

        remaining = deadline - time.monotonic() - 8
        if remaining > 5:
            a_opt, _ = spectral_redistribute(best_seq, time_limit=min(remaining, 8))
            try_seq(a_opt, "final_spec")

    print(f"  P7 done: best={best_score:.10f}, t={time.monotonic()-t7:.0f}s")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final validation
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
