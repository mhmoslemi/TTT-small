def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def pair_descent(a0, time_limit, n_pairs_per_iter=500, verbose=False):
    """
    Random pair coordinate descent: pick random pairs (i,j) and optimize
    the transfer delta from i to j.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(12345)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        # Pick random pairs
        for _ in range(min(n_pairs_per_iter, 3 * n)):
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            if a[i] < 1e-12:
                continue
            
            # Find optimal delta (amount to move from i to j)
            # a[i] -> a[i] - delta, a[j] -> a[j] + delta
            # delta in [0, a[i]]
            # Binary search for the best delta
            
            best_delta = 0.0
            best_val = best_obj
            lo, hi = 0.0, a[i]
            
            for _ in range(10):  # 10 bisection steps
                mid = (lo + hi) / 2
                a_trial = a.copy()
                a_trial[i] -= mid
                a_trial[j] += mid
                # Only need to check affected convolution values
                # But for simplicity, compute full conv (might be slow)
                # Actually for n=768, FFT conv is fast
                val = fast_convolve(a_trial).max()
                if val < best_val:
                    best_val = val
                    best_delta = mid
                    lo = mid
                else:
                    hi = mid
            
            if best_delta > 1e-10 and best_val < best_obj - 1e-14:
                a[i] -= best_delta
                a[j] += best_delta
                best_obj = best_val
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve > 5000:
            break
        
        if verbose and it % 100 == 0:
            print(f"    pair_cd it={it}, obj={best_obj:.10f}, no_imp={no_improve}")
    
    return best_a, best_obj

def simulated_annealing(a0, time_limit, T_init=0.01, T_end=0.0001, verbose=False):
    """
    Simulated annealing with pair moves.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    current_obj = best_obj
    
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(999)
    
    total_iters = 50000  # estimate
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Temperature schedule
        progress = min(1.0, it / total_iters)
        T = T_init * (T_end / T_init) ** progress
        
        # Random pair move
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        delta_mag = rng.uniform(0.001, 0.01)
        if rng.random() < 0.5:
            i, j = j, i
        if a[i] < delta_mag:
            delta_mag = a[i] * 0.5
        if delta_mag < 1e-12:
            continue
        
        a[i] -= delta_mag
        a[j] += delta_mag
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        new_obj = fast_convolve(a).max()
        delta_obj = new_obj - current_obj
        
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-15)):
            current_obj = new_obj
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_a = a.copy()
        else:
            # Revert
            a[j] -= delta_mag
            a[i] += delta_mag
            a = a / a.sum()
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- LP-based primal-dual iteration (SoTA-inspired with bisection on target peak)
- Nesterov accelerated smoothed gradient (log-sum-exp surrogate)
- Exact pair-move coordinate descent using analytic delta_conv formula
- Multi-n exploration with adaptive time allocation
- Simulated annealing for escaping local minima

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n matrix A where (A@g)[k] = (f*g)[k]."""
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        lo = max(0, i)
        hi = min(2 * n - 1, i + n)
        A[lo:hi, i] = f[:hi - lo]
    return A


def solve_lp_for_g(f, M, n, timeout=5.0):
    """
    Solve LP: maximize sum(b) s.t. (f*b)[k] <= M for all k, b >= 0.
    Returns (b, optimal_value) or (None, -inf) on failure.
    """
    A = build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return None, -np.inf


def lp_primal_dual(a0, time_limit, t_init=0.3, m_factor=0.995, verbose=False):
    """
    LP-based primal-dual iteration.
    At each step: solve LP to find g maximizing sum(g) s.t. f*g <= M_target, g>=0.
    Blend f with rescaled g.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    t = t_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3.0:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Try progressively looser targets
        g = None
        for mf in [m_factor, 0.99, 0.998, 1.0]:
            M_target = M * mf
            g, g_sum = solve_lp_for_g(a, M_target, n, timeout=min(3.0, remaining * 0.3))
            if g is not None and g_sum > 1e-12:
                break

        if g is None or g_sum < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        g = g / g.sum()

        # Try multiple blend factors
        best_candidate = a
        best_candidate_obj = best_obj
        for alpha in [t, t * 0.5, t * 1.5, 0.1, 0.7]:
            alpha = min(1.0, max(0.01, alpha))
            a_new = (1.0 - alpha) * a + alpha * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:

            a_new = a_new / s_new
            obj = fast_convolve(a_new).max()
            if obj < best_candidate_obj - 1e-14:

            best_a = best_candidate.copy()
            a = best_candidate
            no_improve = 0

        else:

            t = max(0.05, t * 0.7)

def nesterov_smoothed(a0, time_limit, T_init=0.001, lr_init=0.005, verbose=False):
    """
    Nesterov-accelerated gradient for log-sum-exp smoothed objective.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = 0.9
    y = a.copy()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        y = a + mom * v
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = a.copy()
        else:
            y = y / y_sum

        conv_y = fast_convolve(y)
        scaled = conv_y / T_init
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient: d/dy[i] sum_k p[k]*(y*y)[k] = 2 * sum_k p[k]*y[k-i] (valid k-i)
        # = 2 * correlate(p, y, 'valid')[i]
        grad = 2.0 * np.correlate(p, y, mode='valid')

        y_new = project_simplex(y - lr * grad)
        v_new = y_new - y
        a = y_new
        v = v_new

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.99995
        mom = min(0.998, mom + 0.00005)

        if no_improve > 200:
            lr *= 0.7
            T_init *= 0.8

                rng2 = np.random.default_rng((it * 7 + 3) % (2**31))

                v = np.zeros(n)

                no_improve = 0

            print(f"    nesterov it={it}, obj={best_obj:.10f}, lr={lr:.6f}")

        T = T_init * (T_end / T_init) ** progress

        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        frac = rng.uniform(0.005, 0.05)
        delta = a[i] * frac
        if delta < 1e-14:
            continue
        direction = 1 if rng.random() < 0.5 else -1
        if direction < 0:
            i, j = j, i
        if a[i] < delta:
            delta = a[i] * 0.5
        if delta < 1e-14:
            continue

        a[i] -= delta
        a[j] += delta
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        new_obj = fast_convolve(a).max()
        d_obj = new_obj - current_obj

        if d_obj < 0 or (T > 1e-15 and rng.random() < np.exp(-min(d_obj, 100) / T)):
            current_obj = new_obj
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_a = a.copy()
        else:
            a[j] -= delta
            a[i] += delta
            a = a / a.sum()

        if verbose and it % 10000 == 0:
            print(f"    SA it={it}, obj={best_obj:.10f}, T={T:.6f}")

    return best_a, best_obj


def fourier_smoothed(a0, time_limit, lr_init=0.01, verbose=False):
    """
    Gradient descent in Fourier domain: optimize |A|^2 to be flat.
    Work with zero-padded circular convolution.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()  # FFT size

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # Zero-pad and FFT
        a_padded = np.zeros(N)
        a_padded[:n] = a
        A = np.fft.fft(a_padded)

        # Circular autocorrelation
        R = np.fft.ifft(np.abs(A) ** 2).real

        # Gradient of max_sidelobe w.r.t. a
        # Use smoothed version: weight by softmax of R
        T = 0.01
        R_sidelobe = R.copy()
        R_sidelobe[0] = 0  # Ignore main lobe
        scaled = R_sidelobe / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()

        # Gradient: dR[k]/da[i] = 2*a[(k-i) mod N] (for circular)
        # grad[i] = 2 * sum_k w[k] * a[(k-i) mod N]
        # = 2 * circular_correlate(w, a)[i]
        W = np.fft.fft(w)
        grad_F = np.fft.ifft(W * np.conj(A)).real * 2.0
        grad = grad_F[:n]

        # Update
        a_new = project_simplex(a - lr * grad)
        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            a = best_a.copy()

        lr *= 0.9999
        if no_improve > 100:
            lr *= 0.8
        if no_improve > 500:
            rng2 = np.random.default_rng((it * 13 + 1) % (2**31))
            a = project_simplex(best_a + rng2.standard_normal(n) * 0.002)
            lr = lr_init * 0.5
            no_improve = 0

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: hybrid of LP primal-dual, Nesterov smoothed gradient,
    pair descent, and simulated annealing with multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={ps:.10f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.10f}")

    def record(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return
        a_n = a / s
        sc = float(evaluate_sequence(a_n.tolist()))
        if np.isfinite(sc) and sc < best_score:
            old = best_score
            best_score = sc
            best_seq = a_n.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a_n)}, prev={old:.10f})")

    n_parent = len(parent)
    print(f"Starting search. Parent n={n_parent}, best={best_score:.10f}")

    # === Phase 1: LP-based optimization on parent and nearby n (~600s) ===
    print("\n=== Phase 1: LP Primal-Dual ===")
    t1 = time.monotonic()

    n_list = sorted(set([n_parent, max(128, n_parent - 128), n_parent + 64,
                         max(256, n_parent + 128), max(256, n_parent // 2)]))

    for n_try in n_list:
        remaining = deadline - time.monotonic() - 700
        if remaining < 60:
            break
        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0.0)
            ss = start.sum()
            if ss < 1e-10:
                start = np.ones(n_try) / n_try
            else:
                start = start / ss

        t_alloc = min(remaining, 180)
        print(f"  LP-PD n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = lp_primal_dual(start, time_limit=t_alloc, t_init=0.3, m_factor=0.995)
        record(a_opt, f"lp_n{n_try}")

        # Second pass with different params
        remaining = deadline - time.monotonic() - 650
        if remaining > 30:
            t_alloc2 = min(remaining, 100)
            a_opt2, _ = lp_primal_dual(a_opt, time_limit=t_alloc2, t_init=0.2, m_factor=0.99)
            record(a_opt2, f"lp2_n{n_try}")

    print(f"  Phase 1 done ({time.monotonic()-t1:.0f}s). Best={best_score:.10f}")

    # === Phase 2: Nesterov smoothed gradient (~400s) ===
    print("\n=== Phase 2: Nesterov Smoothed ===")
    t2 = time.monotonic()

    if best_seq is not None:
        for T_val, lr_val in [(0.001, 0.004), (0.005, 0.006), (0.0005, 0.003), (0.01, 0.008)]:
            remaining = deadline - time.monotonic() - 350
            if remaining < 30:
                break
            t_alloc = min(remaining, 120)
            print(f"  Nesterov T={T_val}, lr={lr_val}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_smoothed(best_seq, time_limit=t_alloc, T_init=T_val, lr_init=lr_val)
            record(a_opt, f"nest_T{T_val}")

    print(f"  Phase 2 done ({time.monotonic()-t2:.0f}s). Best={best_score:.10f}")

    # === Phase 3: Fourier smoothed + pair descent (~350s) ===
    print("\n=== Phase 3: Fourier + Pair Descent ===")
    t3 = time.monotonic()

    if best_seq is not None:
        # Fourier
        remaining = deadline - time.monotonic() - 250
        if remaining > 30:
            t_alloc = min(remaining, 100)
            print(f"  Fourier t={t_alloc:.0f}s")
            a_opt, _ = fourier_smoothed(best_seq, time_limit=t_alloc, lr_init=0.005)
            record(a_opt, "fourier")

        # Pair descent
        remaining = deadline - time.monotonic() - 200
        if remaining > 30:
            t_alloc = min(remaining, 150)
            print(f"  Pair descent t={t_alloc:.0f}s")
            a_opt, _ = pair_descent(best_seq, time_limit=t_alloc, n_pairs=200)
            record(a_opt, "pair1")

        remaining = deadline - time.monotonic() - 150
        if remaining > 20:
            t_alloc = min(remaining, 80)
            a_opt, _ = pair_descent(best_seq, time_limit=t_alloc, n_pairs=100)
            record(a_opt, "pair2")

    print(f"  Phase 3 done ({time.monotonic()-t3:.0f}s). Best={best_score:.10f}")

    # === Phase 4: Simulated annealing + diverse restarts (~300s) ===
    print("\n=== Phase 4: Sim Anneal + Restarts ===")
    t4 = time.monotonic()

    if best_seq is not None:
        # Simulated annealing
        remaining = deadline - time.monotonic() - 100
        if remaining > 30:
            t_alloc = min(remaining, 120)
            print(f"  Sim Anneal t={t_alloc:.0f}s")
            a_opt, _ = sim_anneal(best_seq, time_limit=t_alloc, T_init=0.002, T_end=0.00003)
            record(a_opt, "sim_anneal")

        # Diverse restarts: perturb best and re-optimize
        n_best = len(best_seq)
        for restart in range(3):
            remaining = deadline - time.monotonic() - 60
            if remaining < 20:
                break
            noise = [0.005, 0.01, 0.02][restart]
            perturbed = project_simplex(best_seq + rng.standard_normal(n_best) * noise)

            t_alloc = min(remaining, 60)
            # Quick LP pass
            a_opt, _ = lp_primal_dual(perturbed, time_limit=t_alloc, t_init=0.3)
            record(a_opt, f"restart_{restart}")

    print(f"  Phase 4 done ({time.monotonic()-t4:.0f}s). Best={best_score:.10f}")

    # === Phase 5: Final polish (~remaining) ===
    print("\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()

    if best_seq is not None:
        # Very fine Nesterov
        remaining = deadline - time.monotonic() - 20
        if remaining > 15:
            t_alloc = min(remaining, 50)
            a_opt, _ = nesterov_smoothed(best_seq, time_limit=t_alloc, T_init=0.0003, lr_init=0.002)
            record(a_opt, "final_nest")

        # Fine pair descent
        remaining = deadline - time.monotonic() - 15
        if remaining > 10:
            t_alloc = min(remaining, 40)
            a_opt, _ = pair_descent(best_seq, time_limit=t_alloc, n_pairs=100)
            record(a_opt, "final_pair")

        # Tiny LP
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            t_alloc = min(remaining, 30)
            a_opt, _ = lp_primal_dual(best_seq, time_limit=t_alloc, t_init=0.1, m_factor=0.999)
            record(a_opt, "final_lp")

    print(f"  Phase 5 done ({time.monotonic()-t5:.0f}s). Best={best_score:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
