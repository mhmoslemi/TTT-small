def peak_target_step(a, k_peak, n):
    # Try to reduce conv(a,a)[k_peak]
    # Find the largest contributor
    js = max(0, k_peak - n + 1)
    je = min(n - 1, k_peak)
    contrib = np.arange(js, je + 1)
    # Contribution of a[i] to c[k_peak] is a[i] * a[k_peak - i]
    contrib_vals = a[contrib] * a[k_peak - contrib]
    # Find the index with largest contribution
    i_max = contrib[np.argmax(contrib_vals)]
    
    # Try to shift mass from i_max to a non-contributing position
    non_contrib = np.setdiff1d(np.arange(n), contrib)
    if len(non_contrib) == 0:
        return a
    
    # Find the position in non_contrib with smallest "impact"
    # Approximate impact: sum of |a[non_contrib[j] + i] * a[non_contrib[j] - i]| for all i
    # This is too expensive; just use a simple heuristic

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_mv(w, a):
    """Compute W @ a where W_{i,j} = w[i+j]."""
    return np.correlate(w, a, mode='valid')

def solve_qp_active_set(w, a_init, n, max_iters=200, time_limit=3.0):
    """
    Solve min a^T W a s.t. sum(a)=1, a>=0 using active set method.
    W is Hankel: W_{i,j} = w[i+j].
    """
    t0 = time.monotonic()
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    active = np.zeros(n, dtype=bool)
    active[a > 1e-10] = True
    if not active.any():
        active[:] = True

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        act_idx = np.where(active)[0]
        k = len(act_idx)
        if k == 0:
            return np.ones(n) / n
        if k == 1:
            a = np.zeros(n)
            a[act_idx[0]] = 1.0
            return a

        # Build submatrix
        W_AA = np.array([[w[act_idx[i] + act_idx[j]] for j in range(k)] for i in range(k)])
        W_AA_reg = W_AA + 1e-12 * np.eye(k)

        try:
            x = np.linalg.solve(W_AA_reg, np.ones(k))
        except np.linalg.LinAlgError:
            x = np.linalg.lstsq(W_AA_reg, np.ones(k), rcond=None)[0]

        x_sum = x.sum()
        if abs(x_sum) < 1e-14:
            x = np.ones(k) / k
            x_sum = 1.0

        a_act = x / x_sum

        # Check for negative components
        neg_mask = a_act < 0
        if neg_mask.any():
            min_val = a_act[neg_mask].min()
            min_pos = np.where(neg_mask)[0][np.argmin(a_act[neg_mask])]
            active[act_idx[min_pos]] = False
            # Update a
            a_new = np.zeros(n)
            for i_idx in np.where(~neg_mask)[0]:
                a_new[act_idx[i_idx]] = a_act[i_idx]
            s2 = a_new.sum()
            if s2 > 1e-15:
                a = a_new / s2
            continue

        # All non-negative, form solution
        a = np.zeros(n)
        a[act_idx] = a_act

        # Compute gradient
        Wa = hankel_mv(w, a)
        grad = 2.0 * Wa

        # Lagrange multiplier
        mu = np.mean(grad[act_idx])

        # Check KKT for inactive
        inactive_mask = ~active
        if inactive_mask.any():
            inactive_idx = np.where(inactive_mask)[0]
            viol = grad[inactive_idx] - mu
            min_viol = viol.min()
            if min_viol < -1e-8:
                active[inactive_idx[np.argmin(viol)]] = True
                continue

        # Converged
        return a

    return a

def nesterov_polish(a0, time_limit, lr_init=0.003):
    """Nesterov-accelerated subgradient polish."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = 0.9
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit - 0.05:
        # Nesterov lookahead
        y = a + mom * v

        # Compute subgradient at y
        conv_y = fast_convolve(y)
        k_peak = int(np.argmax(conv_y))
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * y[k_peak - idx]

        # Update
        y_proj = project_simplex(y - lr * grad)
        v_new = y_proj - y
        a = y_proj
        v = v_new

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9999
        mom = min(0.995, mom + 0.00002)

        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = lr_init
            mom = 0.9
            no_improve = 0

    return best_a, best_obj


def coordinate_refine(a0, time_limit):
    """
    Coordinate descent refinement: try small perturbations of individual
    coordinates and pairs of coordinates.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit - 0.1:
        # Random coordinate or pair perturbation
        if rng.random() < 0.3:
            # Single coordinate
            i = rng.integers(0, n)
            for delta_frac in [0.01, -0.01, 0.05, -0.05, 0.1, -0.1]:
                shift = a[i] * delta_frac
                if shift == 0:
                    continue
                # Redistribute to random position
                j = rng.integers(0, n)
                if j == i:
                    continue
                candidate = a.copy()
                candidate[i] -= shift
                candidate[j] += shift
                if candidate[i] < 0 or candidate[j] < 0:
                    continue
                candidate = project_simplex(candidate)
                obj = fast_convolve(candidate).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = candidate.copy()
                    a = candidate.copy()
                    break
        else:
            # Pair perturbation
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for delta_frac in [0.01, -0.01, 0.05, -0.05]:
                shift = (a[i] + a[j]) * delta_frac / 2
                candidate = a.copy()
                candidate[i] -= shift
                candidate[j] += shift
                if candidate[i] < 0 or candidate[j] < 0:
                    continue
                candidate = project_simplex(candidate)
                obj = fast_convolve(candidate).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = candidate.copy()
                    a = candidate.copy()
                    break

    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Primal-dual QP iteration with adaptive temperature
- Peak-targeted coordinate steps integrated into main loop
- Nesterov-accelerated subgradient polish
- Multi-n exploration with careful time allocation
- Random restarts to escape local minima

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(w, a):
    """Compute W @ a where W_{i,j} = w[i+j]. w has length 2n-1, a has length n."""
    return np.correlate(w, a, mode='valid')


def solve_qp_hankel(w, a_init, n, max_iters=300, time_limit=3.0):
    """
    Solve min a^T W a s.t. sum(a)=1, a>=0 using active set method.
    W is Hankel: W_{i,j} = w[i+j] where w has length 2n-1.
    """
    t0 = time.monotonic()
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    active = np.zeros(n, dtype=bool)
    active[a > 1e-10] = True
    if not active.any():
        active[:] = True

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        act_idx = np.where(active)[0]
        k = len(act_idx)
        if k == 0:
            return np.ones(n) / n
        if k == 1:
            out = np.zeros(n)
            out[act_idx[0]] = 1.0
            return out

        # Build submatrix W_A
        W_AA = np.empty((k, k))
        for i in range(k):
            for j in range(k):
                W_AA[i, j] = w[act_idx[i] + act_idx[j]]

        W_AA_reg = W_AA + 1e-12 * np.eye(k)

        try:
            x = np.linalg.solve(W_AA_reg, np.ones(k))
        except np.linalg.LinAlgError:
            try:
                x = np.linalg.lstsq(W_AA_reg, np.ones(k), rcond=None)[0]
            except Exception:
                x = np.ones(k) / k

        x_sum = x.sum()
        if abs(x_sum) < 1e-14:
            x = np.ones(k) / k
            x_sum = 1.0

        a_act = x / x_sum

        # Remove most negative component
        neg_idx = np.where(a_act < 0)[0]
        if len(neg_idx) > 0:
            min_pos = neg_idx[np.argmin(a_act[neg_idx])]
            active[act_idx[min_pos]] = False
            new_a = np.zeros(n)
            for i in range(k):
                if i != min_pos:
                    new_a[act_idx[i]] = max(0.0, a_act[i])
            s2 = new_a.sum()
            if s2 > 1e-15:
                a = new_a / s2
            continue

        # All non-negative
        a = np.zeros(n)
        a[act_idx] = a_act

        # Gradient check
        Wa = hankel_mv(w, a)
        grad = 2.0 * Wa
        mu = np.mean(grad[act_idx])

        inactive_idx = np.where(~active)[0]
        if len(inactive_idx) > 0:
            viol = grad[inactive_idx] - mu
            if viol.min() < -1e-8:
                active[inactive_idx[np.argmin(viol)]] = True
                continue

        return a

    return a


def topk_weights(conv, k):
    """Create uniform weights on top-k peaks."""
    kk = min(k, len(conv))
    peaks = np.argpartition(conv, -kk)[-kk:]
    w = np.zeros(len(conv))
    w[peaks] = 1.0 / kk
    return w


def softmax_weights(conv, T):
    """Create softmax weights."""
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    s = w.sum()
    if s < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return w / s


def main_optimize(a0, time_limit, T_init=0.01, blend_init=0.5, use_topk=False, topk=15, verbose=False):
    """
    Primal-dual QP iteration with adaptive parameters and peak targeting.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    blend = blend_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.3:
            break

        conv = fast_convolve(a)

        # Compute weights
        if use_topk:
            w = topk_weights(conv, topk)
        else:
            w = softmax_weights(conv, T)

        # Solve QP
        qp_time = min(2.0, remaining * 0.25)
        a_qp = solve_qp_hankel(w, a, n, max_iters=150, time_limit=qp_time)

        # Blend
        a_new = (1.0 - blend) * a + blend * a_qp
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a_qp
        else:
            a_new = a_new / s_new

        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adapt parameters
        if no_improve > 5:
            T *= 1.15
            blend = max(0.1, blend * 0.97)
        if no_improve > 20:
            T *= 0.7
        if no_improve > 50:
            rng = np.random.default_rng((it * 104729 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            T = T_init
            blend = blend_init
            no_improve = 0

        # Peak targeting step (every 15 iterations)
        if it % 15 == 0 and remaining > 1.0:
            conv_cur = fast_convolve(a)
            k_peak = int(np.argmax(conv_cur))
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            contrib = np.arange(js, je + 1)
            non_contrib = np.setdiff1d(np.arange(n), contrib)
            if len(non_contrib) > 2 and len(contrib) > 0:
                # Try shifting from largest contributor
                contrib_vals = a[contrib] * a[k_peak - contrib]
                top_contribs = contrib[np.argsort(contrib_vals)[-3:]]
                non_contrib_sorted = non_contrib[np.argsort(a[non_contrib])][-max(3, len(non_contrib)//10):]
                for i_max in top_contribs:
                    for j_target in non_contrib_sorted:
                        for frac in [0.05, 0.1, 0.15]:
                            shift = a[i_max] * frac
                            if shift < 1e-12:
                                continue
                            delta = np.zeros(n)
                            delta[i_max] -= shift
                            delta[j_target] += shift
                            candidate = project_simplex(a + delta)
                            c_new = fast_convolve(candidate).max()
                            if c_new < best_obj - 1e-14:
                                a = candidate
                                best_obj = c_new
                                best_a = a.copy()
                                no_improve = 0
                                break
                        if no_improve == 0:
                            break

        if verbose and it % 50 == 0:
            print(f"    main_opt it={it}, obj={best_obj:.10f}, T={T:.6f}, "
                  f"blend={blend:.3f}, no_imp={no_improve}")

    return best_a, best_obj


def nesterov_polish(a0, time_limit, lr_init=0.003, verbose=False):
    """Nesterov-accelerated subgradient polish for min max_k conv(a,a)[k]."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = 0.9
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit - 0.05:
        it += 1
        # Nesterov lookahead
        y = a + mom * v

        # Compute subgradient at y
        conv_y = fast_convolve(y)
        k_peak = int(np.argmax(conv_y))
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * y[k_peak - idx]

        # Projection step
        y_proj = project_simplex(y - lr * grad)
        v = y_proj - y
        a = y_proj

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.99995
        mom = min(0.995, mom + 0.00001)

        if no_improve > 150:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = lr_init * 0.7
            mom = 0.9
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    nesterov it={it}, obj={best_obj:.10f}, lr={lr:.6f}")

    return best_a, best_obj


def coordinate_refine(a0, time_limit, verbose=False):
    """
    Coordinate descent: try pairwise shifts between contributing and
    non-contributing positions for the current peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    it = 0

    while time.monotonic() - t0 < time_limit - 0.1:
        it += 1
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = np.arange(js, je + 1)
        non_contrib = np.setdiff1d(np.arange(n), contrib)

        if len(contrib) == 0 or len(non_contrib) == 0:
            break

        improved = False
        # Try top contributors
        contrib_vals = a[contrib] * a[k_peak - contrib]
        top_c = contrib[np.argsort(contrib_vals)[-5:]]

        for i_max in top_c:
            # Try shifting to random non-contributing positions
            targets = rng.choice(non_contrib, size=min(5, len(non_contrib)), replace=False)
            for j_target in targets:
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    shift = a[i_max] * frac
                    if shift < 1e-12:
                        continue
                    candidate = a.copy()
                    candidate[i_max] -= shift
                    candidate[j_target] += shift
                    if candidate[i_max] < -1e-12 or candidate[j_target] < 0:
                        continue
                    candidate = project_simplex(candidate)
                    obj = fast_convolve(candidate).max()
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_a = candidate.copy()
                        a = candidate.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved and it % 500 == 0:
            # Random perturbation
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)

        if not improved and it > 2000:
            break

    return best_a, best_obj


def smoothed_gradient(a0, time_limit, T_init=0.001, lr_init=0.02, verbose=False):
    """Smoothed gradient descent with log-sum-exp surrogate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit - 0.05:
        it += 1
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient of smoothed surrogate
        grad = 2.0 * hankel_mv(p, a)

        # Backtracking line search
        best_candidate = a
        best_val = conv.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 20:
            T *= 1.1
            lr *= 0.95
        if no_improve > 60:
            T *= 0.7
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj

        for _ in range(4):
            remaining = deadline - time.monotonic() - 8
            if remaining < 8:
                break
            noise_scale = 0.0008 * (_ + 1)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            t_alloc = min(remaining, 20)
            a_opt, _ = main_optimize(f_pert, time_limit=t_alloc, T_init=0.003,
                                     blend_init=0.3, use_topk=False, topk=10)
            try_seq(a_opt, f"perturb_{_}")

        # Final Nesterov
        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            t_alloc = min(remaining, 15)
            a_opt, _ = nesterov_polish(best_seq, time_limit=t_alloc, lr_init=0.001)
            try_seq(a_opt, "final_nest")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
