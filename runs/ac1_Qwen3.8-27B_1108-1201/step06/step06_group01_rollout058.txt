def fw_subgradient(a0, time_limit):

        a = np.ones(len(a)) / len(a)

        a = a / s
    n = len(a)

    no_improve = 0
    it = 0
    alpha = 2.0 / 3.0

        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        c = fast_convolve(a)
        M = c.max()
        
        # Subgradient: find the coordinate that, if we move mass to it, most reduces max(c)
        # The subgradient of max(c) is 2*a[k*-i] where k* = argmax(c)
        k_peak = int(np.argmax(c))

        # FW: find s in simplex minimizing <grad, s>

        for i in range(n):

            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        j_star = int(np.argmin(grad))
        s_vec = np.zeros(n)
        s_vec[j_star] = 1.0
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for trial_alpha in [alpha / (it + 1), 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3]:
            candidate = (1.0 - trial_alpha) * a + trial_alpha * s_vec
            candidate = np.maximum(candidate, 0)

            if cs < 1e-15:
                continue
            candidate = candidate / cs
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-14:
                best_val = val
                best_alpha = trial_alpha

            a = (1.0 - best_alpha) * a + best_alpha * s_vec
            a = np.maximum(a, 0)
            a = a / a.sum()
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Periodic: also try the "opposite" direction (move mass TO peak contributors)
        if it % 10 == 0:
            # Try moving mass to distribute more evenly
            rng = np.random.default_rng(it * 13 + 7)
            j_noise = rng.integers(0, n)
            s_noise = np.zeros(n)
            s_noise[j_noise] = 1.0
            for trial_alpha in [0.01, 0.02, 0.05]:
                candidate = (1.0 - trial_alpha) * best_a + trial_alpha * s_noise
                candidate = np.maximum(candidate, 0)
                cs = candidate.sum()
                if cs > 1e-15:
                    candidate = candidate / cs
                    if fast_convolve(candidate).max() < best_obj - 1e-12:
                        best_a = candidate.copy()
                        best_obj = fast_convolve(best_a).max()
                        a = best_a.copy()
                        no_improve = 0
                        break
        
        if no_improve > 200:
            break
    
    return best_a, best_obj


def coord_descent_peak(a0, time_limit):
    """
    Coordinate descent: try to reduce each peak by adjusting contributing variables.
    Uses golden section search on 2D subproblems.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    stall = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        c = fast_convolve(a)
        # Top 3 peaks
        top_k = min(3, len(c))
        peak_indices = np.argpartition(c, -top_k)[-top_k:]
        peak_indices = peak_indices[np.argsort(c[peak_indices])[::-1]]
        
        improved = False
        
        for k_peak in peak_indices:
            if time.monotonic() - t0 > time_limit - 1.0:
                break
            
            k = int(k_peak)
            # Variables that affect this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            affected = list(range(js, je + 1))
            
            if len(affected) < 2:
                continue
            
            # Try pairwise mass transfers
            # Source: variable that contributes a lot to this peak
            # Destination: variable that contributes less to this peak (or to other peaks)
            peak_val = c[k]
            
            # Find best source and destination
            src_scores = []
            for j in affected:
                partner = k - j
                if 0 <= partner < n:
                    contrib = a[j] * a[partner]
                else:
                    contrib = 0
                src_scores.append((contrib, j))
            src_scores.sort(reverse=True)
            
            # Sample destinations (not in affected set, or in affected but low contribution)
            rng = np.random.default_rng(it * 1009 + k)
            n_dest = min(20, n)
            dest_candidates = rng.choice(n, size=n_dest * 3, replace=False)
            
            best_gain = 0
            best_move = None
            
            for contrib_src, j_src in src_scores[:5]:
                if a[j_src] < 1e-10:
                    continue
                for j_dst in dest_candidates:
                    if j_dst == j_src:
                        continue
                    # Check if j_dst contributes less to this peak
                    partner_dst = k - j_dst
                    contrib_dst = a[j_dst] * a[partner_dst] if 0 <= partner_dst < n else 0
                    if contrib_dst >= contrib_src * 0.5:
                        continue
                    
                    # Try transferring a fraction of a[j_src] to a[j_dst]
                    for frac in [0.05, 0.1, 0.2, 0.3]:
                        delta = a[j_src] * frac
                        a_test = a.copy()
                        a_test[j_src] -= delta
                        a_test[j_dst] += delta
                        a_test = np.maximum(a_test, 0)
                        st = a_test.sum()
                        if st < 1e-15:
                            continue
                        a_test = a_test / st
                        c_new = fast_convolve(a_test)
                        new_max = c_new.max()
                        gain = best_obj - new_max
                        if gain > best_gain:
                            best_gain = gain
                            best_move = (j_src, j_dst, delta)
            
            if best_move is not None and best_gain > 1e-10:
                j_src, j_dst, delta = best_move
                a[j_src] -= delta
                a[j_dst] += delta
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_obj = fast_convolve(a).max()
                best_a = a.copy()
                improved = True
                stall = 0
        
        if not improved:
            stall += 1
            if stall > 50:
                # Random perturbation
                rng = np.random.default_rng((it * 48271 + 1) % (2**31))
                scale = 0.001 + 0.002 * min(1.0, stall / 100.0)
                a = project_simplex(best_a + rng.standard_normal(n) * scale)
                stall = 0
            if stall > 300:
                break
    
    return best_a, best_obj


def micro_polish(a0, time_limit):
    """Very fine perturbation polish."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        rng = np.random.default_rng((it * 2654435761 + 1) % (2**32))
        
        # Structured perturbation targeting the peak
        c = fast_convolve(a)
        k_peak = int(np.argmax(c))
        
        # Compute "pressure" on each variable
        pressure = np.zeros(n)
        for i in range(n):
            j = k_peak - i
            if 0 <= j < n:
                pressure[i] = a[j]
        
        # Perturb: reduce high-pressure variables, increase low-pressure
        noise = rng.standard_normal(n) * 0.001
        noise -= pressure * 0.002  # bias toward reducing peak contributors
        
        candidate = project_simplex(a + noise)
        c_new = fast_convolve(candidate)
        if c_new.max() < best_obj - 1e-13:
            best_obj = c_new.max()
            best_a = candidate.copy()
            a = candidate
        
        # Every 1000 iterations, try a slightly larger perturbation
        if it % 1000 == 0:
            noise2 = rng.standard_normal(n) * 0.005
            candidate2 = project_simplex(best_a + noise2)
            c2 = fast_convolve(candidate2)
            if c2.max() < best_obj - 1e-13:
                best_obj = c2.max()
                best_a = candidate2.copy()
                a = candidate2
    
    return best_a, best_obj


def solve_lp_oracle(f, M, time_limit_lp=5):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c_obj = -np.ones(n)
    
    # Build constraint matrix: (f*g)[k] = sum_j f[k-j]*g[j] <= M
    # For efficiency, use sparse structure
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit_lp})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, time_per_lp=3):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_scale = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        g = solve_lp_oracle(f, M * m_scale, time_limit_lp=time_per_lp)
        if g is None:
            m_scale *= 1.2
            if m_scale > 5.0:
                m_scale = 1.0
                g = np.ones(n) / n
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                break
            m_scale = min(m_scale * 1.05, 3.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-12:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_scale = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_scale = min(m_scale * 1.1, 5.0)
        
        if no_improve > 30:
            rng = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            m_scale = 1.0
            no_improve = 0
    
    return best_f, best_obj


def interpolate_sequence(src, n_target):
    """Interpolate a sequence to a different length."""
    n_src = len(src)
    t_src = np.linspace(0, 1, n_src)
    t_tgt = np.linspace(0, 1, n_target)
    a = np.interp(t_tgt, t_src, src)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return a / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search combining spectral init,
    adaptive smoothed gradient, LP-free FW, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t0_total = time.monotonic()
    deadline = t0_total + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t0_total
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        parent = np.ones(512) / 512
        try_seq(parent, "uniform_512")
    
    n_parent = len(parent)
    
    # === Phase 1: Diverse Initialization (~200s) ===
    print("\n=== Phase 1: Diverse Initialization ===")
    t_phase = time.monotonic()
    
    # Try spectral inits at various n
    for n_try in [n_parent, min(n_parent + 128, 1200), max(n_parent - 128, 384)]:
        if time.monotonic() > deadline - 1500:
            break
        if time.monotonic() - t_phase > 40:
            break
        try:
            a_spec = spectral_init(n_try, flat_width_frac=0.85, iterations=30)
            try_seq(a_spec, f"spectral_flat_n{n_try}")
        except Exception as e:
            print(f"    spectral_flat failed: {e}")
        
        try:
            a_cos = spectral_init_cosine(n_try, iterations=30)
            try_seq(a_cos, f"spectral_cos_n{n_try}")
        except Exception as e:
            print(f"    spectral_cos failed: {e}")
    
    # Two-hump initialization
    for n_try in [n_parent, min(n_parent + 64, 1000)]:
        if time.monotonic() > deadline - 1500:
            break
        for p1, p2 in [(0.25, 0.75), (0.3, 0.7), (0.2, 0.8)]:
            a_2h = two_hump_init(n_try, pos1_frac=p1, pos2_frac=p2, width_frac=0.3)
            try_seq(a_2h, f"2hump_{p1}_{p2}_n{n_try}")
    
    # Asymmetric Gaussian
    for center in [0.4, 0.6]:
        a_g = gaussian_init(n_parent, center_frac=center, width_frac=0.45)
        try_seq(a_g, f"gauss_c{center}")
    
    # Asymmetric perturbation of parent
    for scale in [0.01, 0.02, 0.05]:
        # Break symmetry: shift mass from left to right (or vice versa)
        n = n_parent
        ramp = np.linspace(-1, 1, n)
        a_pert = parent + scale * ramp / np.max(np.abs(ramp))
        a_pert = project_simplex(a_pert)
        try_seq(a_pert, f"asym_ramp_{scale}")
        
        # Random asymmetric perturbation
        a_rand = project_simplex(parent + rng.standard_normal(n) * scale)
        try_seq(a_rand, f"asym_rand_{scale}")
    
    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")
    
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # === Phase 2: Smoothed Gradient Descent (~500s) ===
    print("\n=== Phase 2: Smoothed Gradient Descent ===")
    t_phase = time.monotonic()
    
    n_best = len(best_seq)
    
    # Multiple T configurations
    for T_init in [0.002, 0.001, 0.0005]:
        t_alloc = min(150, deadline - time.monotonic() - 700)
        if t_alloc < 15:
            break
        print(f"  SGD T_init={T_init}, t={t_alloc:.0f}s")
        a_opt, obj = smoothed_grad_descent(best_seq, time_limit=t_alloc, T_init=T_init, T_min=0.0001)
        try_seq(a_opt, f"sgrad_T{T_init}")
    
    # L2 peak descent
    t_alloc = min(100, deadline - time.monotonic() - 600)
    if t_alloc > 15:
        print(f"  L2-peak, t={t_alloc:.0f}s")
        a_opt, obj = l2_peak_descent(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "l2peak")
    
    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 3: FW-LP on best (~300s) ===
    print("\n=== Phase 3: FW-LP ===")
    t_phase = time.monotonic()
    
    t_alloc = min(200, deadline - time.monotonic() - 400)
    if t_alloc > 20:
        print(f"  FW-LP n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, obj = fw_lp_optimize(best_seq, time_limit=t_alloc, time_per_lp=3)
        try_seq(a_opt, "fwlp_main")
    
    # FW with perturbed start
    t_alloc = min(80, deadline - time.monotonic() - 300)
    if t_alloc > 15:
        perturbed = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.01)
        print(f"  FW-LP perturbed, t={t_alloc:.0f}s")
        a_opt, obj = fw_lp_optimize(perturbed, time_limit=t_alloc, time_per_lp=3)
        try_seq(a_opt, "fwlp_pert")
    
    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 4: FW Subgradient + Coord Descent (~300s) ===
    print("\n=== Phase 4: FW-SG + Coord Descent ===")
    t_phase = time.monotonic()
    
    t_alloc = min(120, deadline - time.monotonic() - 250)
    if t_alloc > 15:
        print(f"  FW-SG, t={t_alloc:.0f}s")
        a_opt, obj = fw_subgradient(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "fwsg")
    
    t_alloc = min(150, deadline - time.monotonic() - 200)
    if t_alloc > 15:
        print(f"  Coord descent, t={t_alloc:.0f}s")
        a_opt, obj = coord_descent_peak(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "coord")
    
    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 5: N-sweep with optimization (~250s) ===
    print("\n=== Phase 5: N-Sweep ===")
    t_phase = time.monotonic()
    
    base_seq = best_seq.copy()
    base_n = len(base_seq)
    
    n_candidates = sorted(set([
        base_n - 64, base_n - 32, base_n, base_n + 32, base_n + 64,
        base_n + 128, base_n - 128,
    ]))
    n_candidates = [n for n in n_candidates if 256 <= n <= 1500]
    
    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        t_alloc = min(30, deadline - time.monotonic() - 150)
        if t_alloc < 10:
            break
        
        # Interpolate to new n
        a_interp = interpolate_sequence(base_seq, n_try)
        
        # Quick optimization
        a_opt, obj = smoothed_grad_descent(a_interp, time_limit=t_alloc, T_init=0.001, T_min=0.0002)
        try_seq(a_opt, f"nsweep_n{n_try}")
    
    print(f"  Phase 5 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 6: Final Polish (~150s) ===
    print("\n=== Phase 6: Final Polish ===")
    t_phase = time.monotonic()
    
    # Micro polish
    t_alloc = min(60, deadline - time.monotonic() - 60)
    if t_alloc > 10:
        print(f"  Micro polish, t={t_alloc:.0f}s")
        a_opt, obj = micro_polish(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "micro")
    
    # Another coord descent pass
    t_alloc = min(50, deadline - time.monotonic() - 40)
    if t_alloc > 10:
        print(f"  Coord polish, t={t_alloc:.0f}s")
        a_opt, obj = coord_descent_peak(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "coord_polish")
    
    # Final SGD with very low T
    t_alloc = min(40, deadline - time.monotonic() - 20)
    if t_alloc > 10:
        print(f"  Final SGD (low T), t={t_alloc:.0f}s")
        a_opt, obj = smoothed_grad_descent(best_seq, time_limit=t_alloc, T_init=0.0003, T_min=0.0001, lr_init=0.01)
        try_seq(a_opt, "sgrad_final")
    
    print(f"  Phase 6 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, total time={time.monotonic()-t0_total:.0f}s)")
    return [float(x) for x in best_seq]
