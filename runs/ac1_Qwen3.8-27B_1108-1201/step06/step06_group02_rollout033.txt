Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive smoothed gradient descent with frequency-domain insights,
multi-scale exploration, and coordinate refinement.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():

    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad_fft(a, p):
    """Compute gradient g_j = 2 * sum_m a_m * p[j+m] using FFT.
    a has length n, p has length 2n-1.
    """
    n = len(a)
    fft_size = 1 << (4 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    cc = np.fft.irfft(A * np.conj(P), fft_size)
    return 2.0 * cc[:n]

def frequency_equalize(a0, time_limit):
    """Try to flatten the power spectrum |fft(a)|^2."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        A = np.fft.rfft(a)
        power = np.abs(A) ** 2
        power_norm = power / power.sum()
        
        # We want to reduce the sidelobes of power
        # Strategy: reduce high-power non-DC bin, add to low-power bins
        # But we need to maintain that ifft(|A|) is non-negative... 
        # This is tricky. Instead, let's work with phase adjustments.
        
        # Alternative: compute the "ideal" flat power spectrum
        # and try to move towards it
        # flat_power = 1 / n everywhere
        # But we need |A[0]|^2 = 1 (DC component)
        
        # Actually, let's use a different approach:
        # Compute the target: we want conv to be flat = 1/(2n-1)
        # In frequency: |A|^2 should be (2n-1) * delta_0 (concentrated at DC)
        # So we want to minimize sum_{k=1}^{n-1} |A[k]|^2
        # subject to a >= 0, sum(a) = 1
        # equivalently: minimize sum_{k=1}^{n-1} |A[k]|^2 = sum a_i^2 - 1/n
        # Wait no: sum |A[k]|^2 = n * sum a_i^2 (Parseval)
        # sum_{k=1}^{n-1} |A[k]|^2 = n * sum a_i^2 - |A[0]|^2 = n * sum a_i^2 - 1
        
        # So minimizing sidelobes = minimizing sum a_i^2 = making a as uniform as possible
        # But uniform gives max conv = 1/n, score = 2. That's not helpful.
        
        # The issue is that minimizing total sidelobe energy is not the same as minimizing peak sidelobe.
        # Let's try a different approach: iteratively reduce the peak of conv.
        
        conv = fast_convolve(a)
        peak_idx = np.argmax(conv)
        peak_val = conv[peak_idx]
        
        # Find which a_j most contribute to the peak
        # conv[peak_idx] = sum_i a_i * a_{peak_idx - i}
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        contributions = np.zeros(n)
        for j in range(js, je + 1):
            contributions[j] = a[j] * a[peak_idx - j]
        
        # Reduce the largest contributor, increase the smallest non-contributor
        j_max = js + np.argmax(contributions[js:je+1])
        outside = np.setdiff1d(np.arange(n), range(js, je+1))
        if len(outside) == 0:
            outside = np.setdiff1d(np.arange(n), range(js, je+1))
            if len(outside) == 0:
                outside = np.array([0])
        j_min = outside[np.argmin(a[outside])]
        
        # Try multiple transfer amounts
        improved = False
        for frac in [0.01, 0.02, 0.03, 0.05, 0.08]:
            delta = a[j_max] * frac
            if delta < 1e-10:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min] += delta
            candidate = np.maximum(candidate, 0.0)
            candidate /= candidate.sum()
            new_peak = fast_convolve(candidate).max()
            if new_peak < peak_val - 1e-15:
                a = candidate
                best_obj = new_peak
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try random perturbation
            rng = np.random.default_rng(it * 31 + 7)
            for scale in [0.001, 0.005, 0.01]:
                candidate = project_simplex(a + rng.standard_normal(n) * scale)
                new_peak = fast_convolve(candidate).max()
                if new_peak < best_obj - 1e-15:
                    a = candidate
                    best_a = a.copy()
                    best_obj = new_peak
                    improved = True
                    break
            if not improved and it % 100 == 0:
                break
    
    return best_a, best_obj

def coordinate_refine(a0, time_limit):
    """Coordinate-wise refinement: try moving mass from peak contributors to non-contributors."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    stale = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        peak_idx = int(np.argmax(conv))
        
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        affected = np.arange(js, je + 1)
        
        contributions = a[affected] * a[peak_idx - affected]
        j_max = affected[np.argmax(contributions)]
        
        # Try moving mass from j_max to each non-affected position
        outside_mask = np.ones(n, dtype=bool)
        outside_mask[affected] = False
        outside_indices = np.where(outside_mask)[0]
        
        if len(outside_indices) == 0:
            break
        
        best_delta = 0
        best_target = -1
        best_new_val = best_obj + 1
        
        # Only try a subset for speed
        step = max(1, len(outside_indices) // 50)
        for idx_i in range(0, len(outside_indices), step):
            j_min = outside_indices[idx_i]
            for frac in [0.02, 0.05, 0.1]:
                delta = a[j_max] * frac
                if delta < 1e-12:
                    continue
                candidate = a.copy()
                candidate[j_max] -= delta
                candidate[j_min] += delta
                candidate = np.maximum(candidate, 0.0)
                s = candidate.sum()
                if s < 1e-15:
                    continue
                candidate /= s
                new_val = fast_convolve(candidate).max()
                if new_val < best_new_val:
                    best_new_val = new_val
                    best_delta = delta
                    best_target = j_min
        
        if best_target >= 0 and best_new_val < best_obj - 1e-15:
            a[j_max] -= best_delta
            a[best_target] += best_delta
            a = np.maximum(a, 0.0)
            a /= a.sum()
            best_obj = best_new_val
            best_a = a.copy()
            stale = 0
        else:
            stale += 1
            if stale > 50:
                # Random perturbation
                rng = np.random.default_rng(it * 4567)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.003)
                new_val = fast_convolve(candidate).max()
                if new_val < best_obj - 1e-15:
                    a = candidate
                    best_a = a.copy()
                    best_obj = new_val
                    stale = 0
                elif stale > 200:
                    break
    
    return best_a, best_obj

def generate_initializations(n, parent, rng):
    """Generate diverse initializations for a given n."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Raised cosine (Hann)
    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 + np.cos(2 * np.pi * x))
    # Actually use sin^2 which is a better Hann on [0,1]
    sin2 = np.sin(np.pi * x) ** 2
    s = sin2.sum()
    if s > 0:
        inits.append(sin2 / s)
    
    # Super-Gaussian
    center = (n - 1) / 2.0
    for p in [3, 4, 6]:
        sg = np.exp(-np.abs(np.arange(n) - center) / (n * 0.3)) ** p
        s = sg.sum()
        if s > 0:
            inits.append(sg / s)
    
    # Edge-weighted
    edge = 1.0 / (np.minimum(np.arange(n), n - 1 - np.arange(n)) + 1.0)
    inits.append(edge / edge.sum())
    
    # Random Dirichlet
    for alpha in [0.3, 0.5, 1.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha))
    
    # Cosine modulation
    for freq in [2, 3, 5]:
        c = 1.0 + 0.3 * np.cos(2.0 * np.pi * freq * x)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())
    
    # From parent (interpolated)
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        s = px.sum()
        if s > 1e-10:
            inits.append(px / s)
    
    return inits

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive smoothed gradient descent with FFT-based gradient computation,
frequency-domain insights, multi-scale exploration, and coordinate refinement.
Key differences from previous approach: heavier reliance on gradient methods,
better temperature scheduling, FFT-accelerated gradients, and simulated annealing.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_fft(a, p):
    """Compute gradient g_j = 2 * sum_m a_m * p[j+m] using FFT cross-correlation.
    a has length n, p has length 2n-1.
    """
    n = len(a)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    cc = np.fft.irfft(A * np.conj(P), fft_size)
    return 2.0 * cc[:n]


def gradient_descent(a0, time_limit, T_init=0.001, lr_init=0.04, restart_every=300):
    """Smoothed gradient descent with adaptive temperature and periodic restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_conv_max = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.05:
            break

        conv = fast_convolve(a)
        conv_max = conv.max()

        # Softmax weights (temperature controls peak focus vs global)
        scaled = (conv - conv_max) / T
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p /= p_sum

        # Gradient via FFT
        g = compute_grad_fft(a, p)

        # Line search with multiple step sizes
        best_alpha = 0.0
        best_val = conv_max

        for frac in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1]:
            alpha = lr * frac
            candidate = a - alpha * g
            candidate = project_simplex(candidate)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-16:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            a = project_simplex(a - best_alpha * g)

        # Update best
        curr_max = fast_convolve(a).max()
        if curr_max < best_conv_max - 1e-14:
            best_conv_max = curr_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt temperature: decrease when improving, oscillate when stuck
        if no_improve > 5:
            T *= 0.98
        if no_improve > 30:
            T = max(T * 1.1, T_init * 0.1)
        if no_improve > 100:
            lr = max(lr * 0.95, 0.0001)

        # Periodic restart to escape local minima
        if no_improve >= restart_every:
            noise_scale = 0.002 + 0.004 * min(1.0, (no_improve - restart_every) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = T_init * (0.5 + 0.5 * rng.random())
            lr = lr_init * (0.5 + 0.5 * rng.random())
            no_improve = 0

        # Gradually increase T to explore (anti-annealing every 500 iters)
        if it % 500 == 0 and no_improve < 50:
            T = min(T * 2.0, T_init * 20.0)

    return best_a, best_conv_max
