"""

- Coordinate descent for final polish

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(lam, a):
    """Compute (H_λ * a)[j] = sum_i a[i] * λ[j+i] for j = 0, ..., n-1."""
    n = len(a)
    # Vectorized using advanced indexing
    idx = np.arange(n)[:, None] + np.arange(n)[None, :]
    return (lam[idx] * a[None, :]).sum(axis=1)


def smoothed_grad_step(a, T, lr, momentum_v=None, mom=0.9):
    """One step of smoothed gradient descent."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = T * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()
    
    # Gradient of log-sum-exp: grad[j] = 2 * sum_i p[i] * a[|i-j|]... 
    # Actually: d/d a[j] of sum_k p[k] * conv[k]
    # = sum_k p[k] * (a[k-j] + (if k-j≠j) a[k-j])... 
    # More carefully: d conv[k] / d a[j] = a[k-j] + (if k≠2j and 0≤k-j<n) a[k-j]... 
    # Actually d conv[k]/da[j] = a[k-j] (if 0≤k-j<n) + a[j-k] (if 0≤j-k<n and j-k≠k-j)
    # Wait, conv[k] = sum_i a[i]*a[k-i]. d/da[j] = a[k-j] (if 0≤k-j<n) + a[j-k]... no
    # d/da[j] of a[i]*a[k-i]: this is non-zero when i=j (gives a[k-j]) or k-i=j (i.e., i=k-j, gives a[j])
    # So d conv[k]/da[j] = a[k-j] (if 0≤k-j<n) + a[j] (if 0≤k-j<n and k-j≠j)
    # = a[k-j] + a[j] if k≠2j and 0≤k-j<n
    # = 2*a[j] if k=2j
    # = a[k-j] if k-j is in [0,n-1] but j not in [0,n-1] (impossible since j is always in [0,n-1])
    # Actually since j is always in [0,n-1], the condition is just 0≤k-j<n.
    # So: d conv[k]/da[j] = a[k-j] (if 0≤k-j<n) + (a[j] if k-j≠j and 0≤k-j<n else 0)
    # = a[k-j] + a[j] if k≠2j and 0≤k-j<n
    # = 2*a[j] if k=2j (and 0≤k-2j<n which is always true since k≤2n-2 and j≤n-1)
    # Hmm wait, if k=j then k-j=0 which is valid, and k-j≠j (unless k=2j=0... edge case)
    # Let me just use: d conv[k]/da[j] = a[k-j] (if 0≤k-j<n) + a[k-j] (if 0≤k-j<n and k-j≠j)
    # = 2*a[k-j] if 0≤k-j<n and k≠2j
    # = a[j]^2... no, = a[j]*a[j] = a[j]^2... 
    
    # I'm overcomplicating this. Let me use the hankel_mv approach:
    # The gradient of sum_k p[k] * conv[k] with respect to a is:
    # grad = 2 * H_p * a (approximately, ignoring the factor of 2 for the k=2j case)
    # More precisely: grad[j] = sum_k p[k] * d conv[k]/da[j]
    # = sum_k p[k] * (a[k-j] + a[j] if k≠2j else 2*a[j]) for valid k-j
    # ≈ 2 * sum_{k: 0≤k-j<n} p[k] * a[k-j]
    # = 2 * sum_i p[j+i] * a[i]
    # = 2 * (H_p * a)[j]
    
    grad = 2.0 * hankel_mv(p, a)
    
    if momentum_v is None:
        v = -lr * grad
    else:
        v = mom * momentum_v - lr * grad
    
    a_new = project_simplex(a + v)
    return a_new, v


def penalty_method(a_init, time_limit, mu_init=0.1, mu_growth=1.05):
    """Augmented Lagrangian method for min max conv(a,a) s.t. sum(a)=1, a>=0."""
    a = a_init.copy()
    a = a / a.sum()
    n = len(a)
    
    lam = np.zeros(2 * n - 1)
    mu = mu_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    step_a = 1.0 / (mu + 1.0)
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        if M < best_obj - 1e-14:
            best_obj = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Update lambda (dual ascent)
        step_lam = 1.0 / (mu + 1.0)
        lam = np.maximum(lam + step_lam * (conv - M), 0.0)
        
        # Compute gradient
        grad = 2.0 * hankel_mv(lam, a) + mu * (a.sum() - 1.0)
        
        # Step size (adaptive)
        step_a = 1.0 / (np.linalg.norm(2.0 * hankel_mv(lam, a)) + mu + 1e-10)
        
        # Backtracking
        for _ in range(20):
            a_candidate = project_simplex(a - step_a * grad)
            a_candidate = a_candidate / a_candidate.sum()
            M_cand = fast_convolve(a_candidate).max()
            if M_cand < M + 0.5 * step_a * step_a * (np.linalg.norm(grad)**2):
                a = a_candidate
                break
            step_a *= 0.5
        else:
            a = project_simplex(a - step_a * grad)
            a = a / a.sum()
        
        # Update mu
        mu *= mu_growth
        if mu > 1e6:
            mu = 1e6
        
        # Periodic lambda decay to avoid staleness
        if no_improve > 50:
            lam *= 0.5
        if no_improve > 200:
            lam = np.zeros(2 * n - 1)
            mu = mu_init
            no_improve = 0
    
    return best_a, best_obj


def momentum_smoothed_descent(a_init, time_limit, T_init=0.001, lr_init=0.05, mom_init=0.9):
    """Smoothed gradient descent with momentum and adaptive parameters."""
    a = a_init.copy()
    a = a / a.sum()
    n = len(a)
    
    T = T_init
    lr = lr_init
    mom = mom_init
    v = np.zeros(n)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        a_new, v = smoothed_grad_step(a, T, lr, v, mom)
        a = a_new
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt parameters
        if no_improve > 10:
            T *= 1.03
        if no_improve > 30:
            T *= 0.9  # Reset temperature
            lr *= 0.9
        if no_improve > 100:
            # Restart with perturbation
            rng = np.random.default_rng(it * 7919)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            a = a / a.sum()
            v = np.zeros(n)
            T = T_init
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj


def coordinate_descent(a_init, time_limit):
    """Coordinate descent: transfer mass from high-contributing to low-contributing positions."""
    a = a_init.copy()
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    stale = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        
        # Find the position that contributes most to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]
        
        # Find a position to receive the mass (low value, far from peak)
        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            break
        j_min = outside[np.argmin(a[outside])]
        
        # Try to transfer mass
        improved = False
        for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
            delta = a[j_max] * frac
            if delta < 1e-15:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min] += delta
            candidate = np.maximum(candidate, 0.0)
            s = candidate.sum()
            if s < 1e-15:
                continue
            candidate = candidate / s
            M_cand = fast_convolve(candidate).max()
            if M_cand < best_obj - 1e-15:
                a = candidate
                best_obj = M_cand
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            stale = 0
        else:
            stale += 1
            if stale % 50 == 0:
                rng = np.random.default_rng(it * 4567 + 3)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.002)
                candidate = candidate / candidate.sum()
                M_cand = fast_convolve(candidate).max()
                if M_cand < best_obj - 1e-15:
                    a = candidate
                    best_obj = M_cand
                    best_a = a.copy()
                    stale = 0
                elif stale > 500:
                    break
    
    return best_a, best_obj


def multi_peak_subgradient(a_init, time_limit, top_k=20, lr_init=0.01, mom_init=0.93):
    """Multi-peak subgradient with momentum."""
    a = a_init.copy()
    a = a / a.sum()
    n = len(a)
    
    v = np.zeros(n)
    lr = lr_init
    mom = mom_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        
        # Select top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            top_indices = np.argpartition(conv, -top_k)[-top_k:]
            peaks = top_indices
            pv = conv[peaks]
            # Weight by height
            weights = pv / pv.sum()
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]
        
        v = mom * v - lr * sg
        a = project_simplex(a + v)
        a = a / a.sum()
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.9998
        mom = min(0.98, mom + 0.00005)
        
        if no_improve > 200:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.002 + 0.003 * min(1.0, (no_improve - 200) / 300)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            a = a / a.sum()
            v = np.zeros(n)
            lr = lr_init * 0.5
            no_improve = 0
    
    return best_a, best_obj


def generate_structured_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Edge-concentrated (two peaks at edges)
    x = np.arange(n, dtype=np.float64)
    edge = np.exp(-0.5 * (x / (0.1 * n))**2) + np.exp(-0.5 * ((n-1-x) / (0.1 * n))**2)
    inits.append(edge / edge.sum())
    
    # Cosine taper (Hann window)
    hann = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())
    
    # Flat-top like
    pi = np.pi
    a1, a2, a3, a4, a5 = 1.0, 1.93, 1.29, 0.388, 0.028
    flat = a1 - a2*np.cos(2*pi*x/(n-1)) + a3*np.cos(4*pi*x/(n-1)) - a4*np.cos(6*pi*x/(n-1)) + a5*np.cos(8*pi*x/(n-1))
    flat = np.maximum(flat, 1e-10)
    inits.append(flat / flat.sum())
    
    # Periodic pattern
    for freq in [2, 3, 4, 5, 8]:
        h = 1.0 + 0.4 * np.cos(2.0 * pi * freq * x / n)
        h = np.maximum(h, 0.01)
        inits.append(h / h.sum())
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Interpolated parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits


def fw_with_active_lp(a_init, time_limit, max_lp_constraints=200):
    """Frank-Wolfe with restricted LP oracle (only top-K constraints)."""
    a = a_init.copy()
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_obj = M
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_expand = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        # Select active constraints (top peaks)
        conv = fast_convolve(a)
        M = conv.max()
        
        # Use top K peaks as active constraints
        K = min(max_lp_constraints, 2 * n - 1)
        active_idx = np.argpartition(conv, -K)[-K:]
        
        # Build restricted LP
        c = -np.ones(n)
        A_ub = np.zeros((len(active_idx), n))
        for row_i, k in enumerate(active_idx):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j_col in range(js, je + 1):
                A_ub[row_i, j_col] = a[k - j_col]
        b_ub = np.full(len(active_idx), M * m_expand)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'time_limit': min(10, remaining * 0.3)})
            if res.success:
                g = np.maximum(res.x, 0.0)
            else:
                g = None
        except:
            g = None
        
        if g is None or g.sum() < 1e-15:
            m_expand = min(m_expand * 1.5, 5.0)
            if m_expand > 4.0:
                # Fall back to smoothed gradient step
                a_new, _ = smoothed_grad_step(a, 0.01, 0.03)
                a = a_new
                m_expand = 1.0
            continue
        
        g = g / g.sum()
        
        if np.max(np.abs(g - a)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 7919)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                a = a / a.sum()
                m_expand = 1.0
                no_improve = 0
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in np.linspace(0.01, 1.0, 15):
            at = (1 - alpha) * a + alpha * g
            at = np.maximum(at, 0)
            st = at.sum()
            if st < 1e-15:
                continue
            at = at / st
            val = fast_convolve(at).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            a = (1 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            M_new = fast_convolve(a).max()
            m_expand = 1.0
            
            if M_new < best_obj - 1e-14:
                best_obj = M_new
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_expand = min(m_expand * 1.2, 5.0)
        
        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            a = a / a.sum()
            m_expand = 1.0
            no_improve = 0
    
    return best_a, best_obj


def try_and_update(a, state, label=""):
    """Evaluate and update state if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a)))


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid search using penalty method, smoothed gradient,
    restricted LP Frank-Wolfe, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    state = [float('inf'), None]
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.8f}".format(len(parent), ps))
    except:
        pass
    
    if parent is None:
        parent = np.ones(384) / 384
    
    # === Phase 1: Multi-n exploration with fast methods (~300s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()
    
    n_parent = len(state[1]) if state[1] is not None else 384
    candidate_ns = sorted(set([n_parent, 512, 640, 768, 896, 1024]))
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 1536]
    
    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 20:
            break
        
        # Generate initializations
        inits = generate_structured_inits(n_try, 
                                           parent if n_try != n_parent else None, 
                                           rng)
        
        best_for_n = None
        best_obj_n = float('inf')
        
        for idx, init in enumerate(inits[:4]):  # Try top 4 initializations
            remaining = deadline - time.monotonic() - 1100
            if remaining < 10:
                break
            
            # Quick penalty method
            t_alloc = min(remaining, 25)
            a_opt, obj = penalty_method(init, time_limit=t_alloc)
            if obj < best_obj_n:
                best_obj_n = obj
                best_for_n = a_opt
            
            # Quick smoothed gradient
            remaining = deadline - time.monotonic() - 1050
            if remaining > 5:
                a_opt2, obj2 = momentum_smoothed_descent(a_opt, time_limit=min(remaining, 10),
                                                         T_init=0.001, lr_init=0.03)
                if obj2 < best_obj_n:
                    best_obj_n = obj2
                    best_for_n = a_opt2
        
        if best_for_n is not None:
            try_and_update(best_for_n, state, "explore_n{}".format(n_try))
    
    print("  Phase 1 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))
    
    # === Phase 2: Deep optimization on best n (~500s) ===
    print("\n=== Phase 2: Deep Optimization ===")
    t_phase = time.monotonic()
    
    if state[1] is not None:
        n_best = len(state[1])
        
        # Penalty method (long run)
        remaining = deadline - time.monotonic() - 600
        if remaining > 30:
            t_alloc = min(remaining, 200)
            print("  Penalty method, t={:.0f}s".format(t_alloc))
            a_opt, obj = penalty_method(state[1], time_limit=t_alloc, mu_init=0.05, mu_growth=1.03)
            try_and_update(a_opt, state, "penalty_deep")
        
        # Smoothed gradient with different parameters
        for T_init, lr_init in [(0.0005, 0.04), (0.002, 0.02), (0.005, 0.01)]:
            remaining = deadline - time.monotonic() - 400
            if remaining < 15:
                break
            t_alloc = min(remaining, 80)
            print("  Smoothed grad T={}, lr={}".format(T_init, lr_init))
            a_opt, obj = momentum_smoothed_descent(state[1], time_limit=t_alloc,
                                                   T_init=T_init, lr_init=lr_init)
            try_and_update(a_opt, state, "smooth_T{}".format(T_init))
        
        # Multi-peak subgradient
        for top_k, lr in [(10, 0.005), (30, 0.003), (50, 0.002)]:
            remaining = deadline - time.monotonic() - 200
            if remaining < 10:

            t_alloc = min(remaining, 60)
            print("  Multi-peak SG k={}".format(top_k))
            a_opt, obj = multi_peak_subgradient(state[1], time_limit=t_alloc,
                                                top_k=top_k, lr_init=lr)
            try_and_update(a_opt, state, "mpsg_k{}".format(top_k))
    
    print("  Phase 2 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    print("\n=== Phase 3: Restricted LP FW ===")
    t_phase = time.monotonic()
    
    if state[1] is not None:
        remaining = deadline - time.monotonic() - 200
        if remaining > 30:
            t_alloc = min(remaining, 200)
            print("  Active-set FW, constraints={}, t={:.0f}s".format(100, t_alloc))
            a_opt, obj = fw_with_active_lp(state[1], time_limit=t_alloc, max_lp_constraints=100)
            try_and_update(a_opt, state, "active_fw")
        
        # With more constraints
        remaining = deadline - time.monotonic() - 100
        if remaining > 20:
            t_alloc = min(remaining, 80)
            print("  Active-set FW (200), t={:.0f}s".format(t_alloc))
            a_opt, obj = fw_with_active_lp(state[1], time_limit=t_alloc, max_lp_constraints=200)
