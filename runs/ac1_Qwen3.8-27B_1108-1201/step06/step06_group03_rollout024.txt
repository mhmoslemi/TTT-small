from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_peak_gradient(c, a, peak_indices):
    """Compute subgradient of max peak w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        # d(c[k])/da[i] = 2*a[k-i] if 0 <= k-i < n
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad[j] += 2.0 * a[k - j]
    return grad / len(peak_indices)

def entropic_descent_step(a, grad, lr):
    """Multiplicative (entropic mirror) update."""
    # a_new[i] ∝ a[i] * exp(-lr * grad[i])
    # Then project to simplex
    log_a = np.where(a > 1e-30, np.log(a), -30.0)
    v = log_a - lr * grad
    # Softmax to get probabilities
    v_max = v.max()
    exp_v = np.exp(v - v_max)
    a_new = exp_v / exp_v.sum()
    return a_new

        w_sum = w.sum()
        if w_sum < 1e-30:

        else:
            w = w / w_sum

            js = max(0, k - n + 1)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(a):
    """Compute the objective value (before calling evaluate_sequence)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def kkt_gradient(a, c, n_active_frac=0.1, temperature=0.02):
    """
    Compute KKT-inspired gradient: for each coordinate i,
    compute the weighted sum of a[k-i] over active peaks.
    At optimum, this should be constant over the support.
    """
    n = len(a)
    M = c.max()
    
    # Find active peaks (those close to max)
    threshold = M * (1.0 - n_active_frac)
    active = np.where(c >= threshold)[0]
    
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    
    # Softmax weighting over active peaks
    cv = c[active]
    T = M * temperature
    w = np.exp((cv - cv.max()) / T)
    w = w / w.sum()
    
    # Compute g[i] = sum_k w[k] * 2*a[k-i]
    grad = np.zeros(n)
    for idx in range(len(active)):
        if w[idx] < 1e-12:
            continue
        k = int(active[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idxs = np.arange(js, je + 1)
        contrib = a[k - idxs]
        grad[idxs] += 2.0 * w[idx] * contrib
    
    return grad, active, M

def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def structured_start(n, method, rng):
    """Generate a structured initial sequence of length n."""
    x = np.arange(n, dtype=np.float64)
    if method == 0:
        return np.ones(n) / n
    elif method == 1:
        # Cosine taper
        w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
        return w / w.sum()
    elif method == 2:
        # Gaussian center
        g = np.exp(-0.5 * ((x - (n - 1) / 2) / (0.2 * n)) ** 2)
        return g / g.sum()
    elif method == 3:
        # Edge concentration
        edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 2
        return edge / edge.sum()
    elif method == 4:
        # Two peaks
        g1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
        g2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
        return (g1 + g2) / (g1 + g2).sum()
    elif method == 5:
        # Power law
        w = (x + 1) ** 2
        return w / w.sum()
    elif method == 6:
        # Sinusoidal modulation
        w = 1.0 + 0.5 * np.sin(2 * np.pi * 3 * x / n)
        w = np.maximum(w, 0.01)
        return w / w.sum()
    elif method == 7:
        # Random Dirichlet
        return rng.dirichlet(np.ones(n) * 0.5)
    elif method == 8:
        # Bimodal with gap
        g1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.08 * n)) ** 2)
        g2 = np.exp(-0.5 * ((x - 0.8 * n) / (0.08 * n)) ** 2)
        return (g1 + g2) / (g1 + g2).sum()
    else:
        return np.ones(n) / n

                stagnant = 0
            continue
        
        g = g / g.sum()
        
        if np.max(np.abs(g - f)) < 1e-10:
            stagnant += 1
            if stagnant > 30:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M + 1e-14
        for alpha in np.linspace(0.01, 1.0, 20):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.001
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1
        
        if stagnant > 10:
            m_relax = min(m_relax * 1.01, 1.03)
        
        if verbose and it % 50 == 0:
            print(f"    FW it={it}, obj={best_obj:.10f}, stag={stagnant}")
    
    return best_f, best_obj


def refine_at_length(a, time_limit, methods=None):
    """Apply multiple refinement methods sequentially to a sequence."""
    if methods is None:
        methods = ['kkt', '2coord', 'sa']
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for method in methods:
        elapsed = 0
        total_alloc = time_limit / len(methods)
        
        if time.monotonic() - 0 > elapsed + total_alloc:
            break
        
        if method == 'kkt':
            t_alloc = total_alloc
            a_new, obj_new = kkt_optimize(best_a, time_limit=t_alloc)
            if obj_new < best_obj:
                best_a = a_new
                best_obj = obj_new
        elif method == '2coord':
            a_new, obj_new = two_coord_break(best_a, time_limit=total_alloc)
            if obj_new < best_obj:
                best_a = a_new
                best_obj = obj_new
        elif method == 'sa':
            a_new, obj_new = simulated_annealing(best_a, time_limit=total_alloc)
            if obj_new < best_obj:
                best_a = a_new
                best_obj = obj_new
        elif method == 'fw_lp':
            a_new, obj_new = fw_lp_optimize(best_a, time_limit=total_alloc)
            if obj_new < best_obj:
                best_a = a_new
                best_obj = obj_new
    
    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search for minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    
    Strategy:
    1. Quick multi-length/scoring exploration (200s)
    2. KKT-guided + 2coord refinement on best candidates (500s)
    3. Simulated annealing for escape (300s)
    4. LP-based FW for deep refinement (400s)
    5. Final micro-polish (remaining)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")
    
    n_parent = len(parent)
    
    # === Phase 1: Quick multi-length exploration (200s) ===
    print("\n=== Phase 1: Multi-Length Exploration ===")
    t_phase = time.monotonic()
    
    # Candidate lengths around parent
    lengths = sorted(set([
        max(128, n_parent - 128),
        max(128, n_parent - 64),
        n_parent,
        n_parent + 64,
        n_parent + 128,
        512, 640, 768, 896, 1024, 1152, 1280,
    ]))
    lengths = [l for l in lengths if 128 <= l <= 2048]
    
    # Keep track of best per length
    best_per_length = {}
    
    for n_try in lengths:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 10:
            break
        
        # Generate starts
        starts = [resample_sequence(parent, n_try)]
        for m in range(4):
            starts.append(structured_start(n_try, m, rng))
        
        for si, start in enumerate(starts):
            remaining = deadline - time.monotonic() - 1000
            if remaining < 5:
                break
            
            # Quick KKT optimization (fast, no LP)
            t_alloc = min(remaining, 8.0)
            a_opt, _ = kkt_optimize(start, time_limit=t_alloc)
            score = try_seq(a_opt, f"p1_n{n_try}_s{si}")
            
            if n_try not in best_per_length or score < best_per_length[n_try][1]:
                best_per_length[n_try] = (a_opt, score)
    
    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 2: KKT + 2Coord deep refinement (500s) ===
    print("\n=== Phase 2: KKT + 2Coord Refinement ===")
    t_phase = time.monotonic()
    
    # Refine top 3 lengths
    sorted_lengths = sorted(best_per_length.keys(), key=lambda l: best_per_length[l][1])[:3]
    
    for n_best in sorted_lengths:
        remaining = deadline - time.monotonic() - 700
        if remaining < 30:
            break
        
        a_start = best_per_length[n_best][0]
        t_alloc = min(remaining, 150)
        print(f"  Refining n={n_best}, t={t_alloc:.0f}s")
        
        # KKT optimization
        a_opt, _ = kkt_optimize(a_start, time_limit=t_alloc * 0.4)
        try_seq(a_opt, f"p2_kkt_n{n_best}")
        
        # 2-coordinate breaking
        a_opt2, _ = two_coord_break(a_opt, time_limit=t_alloc * 0.3)
        try_seq(a_opt2, f"p2_2c_n{n_best}")
        
        # More KKT from the 2coord result
        a_opt3, _ = kkt_optimize(a_opt2, time_limit=t_alloc * 0.3)
        try_seq(a_opt3, f"p2_kkt2_n{n_best}")
    
    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 3: Simulated annealing (300s) ===
    print("\n=== Phase 3: Simulated Annealing ===")
    t_phase = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Multiple SA runs with different seeds
        for sa_run in range(3):
            remaining = deadline - time.monotonic() - 400
            if remaining < 20:
                break
            
            # Perturb slightly before SA
            pert_scale = [0.003, 0.01, 0.03][sa_run]
            start = project_simplex(best_seq + rng.standard_normal(n_best) * pert_scale)
            
            t_alloc = min(remaining, 90)
            print(f"  SA run {sa_run}, scale={pert_scale}, t={t_alloc:.0f}s")
            a_opt, _ = simulated_annealing(start, time_limit=t_alloc)
            try_seq(a_opt, f"p3_sa{sa_run}")
    
    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 4: LP-based FW deep refinement (400s) ===
    print("\n=== Phase 4: FW with LP ===")
    t_phase = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        remaining = deadline - time.monotonic() - 200
        if remaining > 30:
            t_alloc = min(remaining, 300)
            print(f"  FW n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "p4_fw")
        
        # Perturbed FW restarts
        for scale in [0.002, 0.005, 0.01]:
            remaining = deadline - time.monotonic() - 150
            if remaining < 20:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 50)
            a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"p4_fw_p{scale}")
    
    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 5: Final polish (remaining) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Alternate KKT and 2coord with tiny perturbations
        polish_rounds = 6
        for ri in range(polish_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            
            t_alloc = min(remaining, 25)
            
            if ri % 2 == 0:
                noise = rng.standard_normal(n_best) * 0.0005
                start = project_simplex(best_seq + noise)
                a_opt, _ = kkt_optimize(start, time_limit=t_alloc, lr_init=0.002)
                try_seq(a_opt, f"p5_kkt_{ri}")
            else:
                a_opt, _ = two_coord_break(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"p5_2c_{ri}")
        
        # Final SA with very low temperature
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_alloc = min(remaining, 20)
            a_opt, _ = simulated_annealing(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "p5_final_sa")
    
    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
