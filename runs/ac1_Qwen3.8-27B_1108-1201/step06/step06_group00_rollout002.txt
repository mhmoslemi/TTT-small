import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(f, conv, beta):
    """Compute smoothed gradient of max_k conv(a,a)[k] using LSE."""
    n = len(f)
    size = len(conv)
    # LSE weights
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        w = np.ones(size) / size
    else:
        w = exp_bc / total
    
    # Gradient: grad[i] = sum_k w[k] * 2 * f[k-i] (with bounds)
    grad = np.zeros(n)
    for k in range(size):
        if w[k] < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[lo:hi+1] += w[k] * 2 * f[k - lo : k - hi + 1][::-1]
        idx = np.arange(lo, hi + 1)
        grad[idx] += w[k] * 2.0 * f[k - idx]
    return grad


def compute_gradient_topk(f, conv, beta, top_k=50):
    """Compute gradient using only top-k peaks for efficiency."""
    n = len(f)
    size = len(conv)
    k_select = min(top_k, size)
    peak_idx = np.argpartition(conv, -k_select)[-k_select:]
    
    bc = beta * conv[peak_idx]
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        w = np.ones(k_select) / k_select
    else:
        w = exp_bc / total
    
    grad = np.zeros(n)
    for i, k in enumerate(peak_idx):
        k = int(k)
        wi = w[i]
        if wi < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += wi * 2.0 * f[k - idx]
    return grad
