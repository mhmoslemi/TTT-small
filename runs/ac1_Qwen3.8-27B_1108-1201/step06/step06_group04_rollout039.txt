```python
import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    # Clean up tiny negative values
    c = np.maximum(c, 0.0)
    return c

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def conv_gradient(a, p):
    """
    Compute the gradient of sum_k p[k] * (a*a)[k] w.r.t. a.
    p has length 2n-1 (weights on the convolution lags).
    Returns a vector of length n.
    """
    n = len(a)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # grad[js:je+1] += 2 * p[k] * a[k - np.arange(js, je+1)]
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * p[k] * a[k - idx]
    return grad
```

```python
def hankel_matvec_fast(h, x):
    """Compute (H @ x) where H[i,j] = h[i+j]. Both h and x are 1D.
    h has length >= 2*len(x)-1, x has length n.
    Returns a vector of length n.
    """
    n = len(x)
    h_rev = h[:2*n-1][::-1]  # Reverse first 2n-1 elements of h
    size = len(h_rev) + n - 1  # 3n-2
    fft_size = 1 << (size - 1).bit_length()
    H = np.fft.rfft(h_rev, fft_size)
    X = np.fft.rfft(x, fft_size)
    conv = np.fft.irfft(H * X, fft_size)[:size]
    idx = 2*(n-1) - np.arange(n)
    return conv[idx]
```

```python
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
```

```python
    grad_x = a * (grad_a - E)

    return val, grad_x


def optimize_lbfgsb_softmax(a0, time_limit, T_init=0.1, T_final=0.0005,
                             n_temps=15, maxiter_per_temp=300, verbose=False):
    """
    Multi-temperature L-BFGS-B optimization using softmax parameterization.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    # Initialize x from log(a) (softmax(log(a)) = a)
    x = np.log(a + 1e-30)
    x -= x.mean()  # Center for numerical stability

    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())

    t0 = time.monotonic()
    temps = np.geomspace(T_init, T_final, n_temps)

    for ti, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 10:
            break

        remaining = time_limit - (time.monotonic() - t0)
        # Estimate time per evaluation
        if ti == 0:
            maxiter = min(maxiter_per_temp, max(50, int(remaining * 5)))
        else:
            maxiter = min(maxiter_per_temp, max(30, int(remaining * 8)))

        result = scipy_minimize(
            fun=lambda xx: softmax_obj_grad(xx, T)[0],
            x0=x,
            jac=lambda xx: softmax_obj_grad(xx, T)[1],
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 3}
        )

        x = result.x - result.x.mean()  # Re-center

        a_new = np.exp(x - x.max())
        a_new = a_new / a_new.sum()

        score = evaluate_sequence(a_new.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_new.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"      T={T:.6f}: score={best_score:.10f}, iter={result.nit}, "
                  f"elapsed={elapsed:.1f}s")

    return best_a, best_score


def multiplicative_weights_optimize(a0, time_limit, T_init=0.01, T_final=0.0001,
                                    n_iters=5000, verbose=False):
    """
    Multiplicative weights (mirror descent) with annealing.
    Directly operates on the simplex.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    T_decay = (T_final / T_init) ** (1.0 / max(1, n_iters - 1))

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit - 1:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient of the smoothed objective w.r.t. a
        grad = 2.0 * hankel_matvec(p, a)

        # Multiplicative weights update: a_new = a * exp(-lr * grad) / Z
        # Choose lr adaptively
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        lr = 0.5 / gnorm

        log_shift = -lr * grad
        log_shift -= log_shift.max()
        weights = np.exp(log_shift)
        a_new = a * weights
        sn = a_new.sum()
        if sn < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / sn

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1

        # Anneal
        T = max(T_final, T * T_decay)

        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng((it * 7919 + n) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MW it={it}, obj={best_obj:.10f}, T={T:.6f}, "
                  f"elapsed={elapsed:.1f}s")

    return best_a, best_obj


def subgradient_polish(a0, time_limit, verbose=False):
    """
    Direct subgradient method on the exact max objective.
    Cycles through all peak lags for diverse subgradients.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    lr = 1.0 / (2.0 * n) * 2.0  # Initial learning rate
    momentum = 0.9
    v = np.zeros(n)
    y = a.copy()
    t_mom = 1.0

    for it in range(200000):
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        # Nesterov step
        y = a + (t_mom - 1) / (t_mom + 1) * (a - (a - v / lr if lr > 0 else 0))
        # Simpler: just use y = a + momentum * v
        y = a + momentum * v
        y = project_simplex(y)

        conv_y = fast_convolve(y)
        k_peak = int(np.argmax(conv_y))

        # Subgradient at y for the peak lag
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * y[k_peak - idx]

        # Update
        v = project_simplex(y - lr * grad) - y
        a = y + v
        a = project_simplex(a)

        # Decay lr and update momentum
        lr *= 0.99999
        momentum = min(0.999, momentum + 0.00001)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        # Periodic restart
        if it > 0 and it % 5000 == 0:
            # Check if stuck
            pass

    return best_a, best_obj


def multi_peak_subgradient(a0, time_limit, verbose=False):
    """
    Subgradient method that uses weighted average of top-K peak subgradients.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)

    for it in range(100000):
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)

        # Use top-K peaks with softmax weighting
        T_sub = 0.001
        scaled = conv / T_sub
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Weighted subgradient
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * p[k] * a[k - idx]

        # Momentum step
        y = a + momentum * v
        v = project_simplex(y - lr * grad) - y
        a = project_simplex(y + v)

        # Adaptive lr
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            lr = min(0.05, lr * 1.05)
        else:
            lr *= 0.999

        if it % 10000 == 0 and it > 0:
            # Slight perturbation to escape
            rng = np.random.default_rng((it + n) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)

    return best_a, best_obj


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Interpolated parent
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # U-shaped (edge concentration)
    x = np.arange(n, dtype=np.float64)
    u = 1.0 / (np.minimum(x + 1, n - x) ** 0.5)
    inits.append(u / u.sum())

    # Gaussian centered
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine (periodic)
    c = np.maximum(0.01, 1.0 + 0.3 * np.cos(2 * np.pi * x / n))
    inits.append(c / c.sum())

    # Sine (anti-periodic)
    s_ = np.maximum(0.01, 1.0 + 0.3 * np.sin(np.pi * x / n))
    inits.append(s_ / s_.sum())

    # Two lobes
    g2 = (np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2))
    inits.append(g2 / g2.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase optimization:
    1. Load parent, try quick L-BFGS-B at several n values
    2. Deep L-BFGS-B at the best n
    3. Multiplicative weights refinement
    4. Subgradient polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {best_score:.10f} (n={len(a_norm)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        try_seq(parent, "default")

    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, best={best_score:.10f}")

    # === Phase 1: Multi-n L-BFGS-B exploration (~400s) ===
    print("\n=== Phase 1: Multi-n L-BFGS-B Exploration ===")
    t1 = time.monotonic()

    candidate_ns = sorted(set([
        n_parent,
        max(64, n_parent - n_parent // 4),
        n_parent + n_parent // 4,
        max(64, n_parent // 2),
        min(4096, n_parent * 2),
        512, 640, 768, 896, 1024, 1280, 1536, 2048,
    ]))
    # Filter to reasonable range
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 3000]
    candidate_ns = sorted(set(candidate_ns))

    phase1_time = min(400, (deadline - time.monotonic()) * 0.3)
    for n_try in candidate_ns:
        if time.monotonic() + 30 > deadline - phase1_time + 400:
            break
        remaining_p1 = (t1 + phase1_time) - time.monotonic()
        if remaining_p1 < 15:
            break

        inits = generate_diverse_inits(n_try, parent, rng)
        best_for_n = None
        best_score_n = float('inf')

        # Try a few inits with short L-BFGS-B
        for ii, init in enumerate(inits[:4]):
            t_alloc = min(remaining_p1, max(5, 40 / len(inits[:4])))
            if t_alloc < 3:
                break
            a_opt, _ = optimize_lbfgsb_softmax(
                init, time_limit=t_alloc,
                T_init=0.1, T_final=0.001, n_temps=8, maxiter_per_temp=100)
            sc = evaluate_sequence(a_opt.tolist())
            if np.isfinite(sc) and sc < best_score_n:
                best_score_n = sc
                best_for_n = a_opt

        if best_for_n is not None:
            try_seq(best_for_n, f"LBFGS_n{n_try}")

    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")

    # === Phase 2: Deep L-BFGS-B at best n (~700s) ===
    print("\n=== Phase 2: Deep L-BFGS-B ===")
    t2 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        phase2_time = min(700, (deadline - time.monotonic()) * 0.4)

        # Run L-BFGS-B with more iterations and finer annealing
        configs = [
            {'T_init': 0.1, 'T_final': 0.0001, 'n_temps': 20},
            {'T_init': 0.05, 'T_final': 0.00005, 'n_temps': 20},
            {'T_init': 0.2, 'T_final': 0.001, 'n_temps': 12},
        ]

        for ci, cfg in enumerate(configs):
            remaining = (t2 + phase2_time) - time.monotonic()
            if remaining < 20:
                break
            t_alloc = min(remaining, max(30, phase2_time / 3))
            print(f"  Deep L-BFGS-B cfg{ci} (T:{cfg['T_init']}->{cfg['T_final']}), "
                  f"n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = optimize_lbfgsb_softmax(
                best_seq, time_limit=t_alloc,
                maxiter_per_temp=500, **cfg)
            try_seq(a_opt, f"deep_lbfgs_{ci}")

    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")

    # === Phase 3: Multiplicative weights refinement (~400s) ===
    print("\n=== Phase 3: Multiplicative Weights ===")
    t3 = time.monotonic()

    if best_seq is not None:
        phase3_time = min(400, (deadline - time.monotonic()) * 0.25)
        mw_configs = [
            {'T_init': 0.01, 'T_final': 0.0005, 'n_iters': 8000},
            {'T_init': 0.005, 'T_final': 0.0002, 'n_iters': 8000},
            {'T_init': 0.02, 'T_final': 0.001, 'n_iters': 6000},
        ]
        for ci, cfg in enumerate(mw_configs):
            remaining = (t3 + phase3_time) - time.monotonic()
            if remaining < 15:
                break
            t_alloc = min(remaining, max(20, phase3_time / len(mw_configs)))
            print(f"  MW cfg{ci} (T:{cfg['T_init']}->{cfg['T_final']}), t={t_alloc:.0f}s")
            a_opt, _ = multiplicative_weights_optimize(
                best_seq, time_limit=t_alloc, **{k: v for k, v in cfg.items()})
            try_seq(a_opt, f"mw_{ci}")

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")

    # === Phase 4: Subgradient polish (~200s) ===
    print("\n=== Phase 4: Subgradient Polish ===")
    t4 = time.monotonic()

    if best_seq is not None:
        phase4_time = min(200, (deadline - time.monotonic()) * 0.15)

        # Multi-peak subgradient
        t_alloc = min(phase4_time, max(30, phase4_time // 2))
        print(f"  Multi-peak subgrad, t={t_alloc:.0f}s")
        a_opt, _ = multi_peak_subgradient(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "multipeak_sg")

        # Standard subgradient
        remaining = (t4 + phase4_time) - time.monotonic()
        if remaining > 15:
            t_alloc = min(remaining, max(20, phase4_time // 3))
            print(f"  Subgrad polish, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "sg_polish")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # === Phase 5: Final L-BFGS-B re-optimization (~remaining) ===
    print("\n=== Phase 5: Final Optimization ===")
    t5 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            # Final deep L-BFGS-B with very fine annealing
            print(f"  Final L-BFGS-B, t={remaining:.0f}s")
            a_opt, _ = optimize_lbfgsb_softmax(
                best_seq, time_limit=remaining,
                T_init=0.01, T_final=0.00001,
                n_temps=25, maxiter_per_temp=500)
            try_seq(a_opt, "final_lbfgs")

            # One more MW pass
            remaining2 = deadline - time.monotonic() - 5
            if remaining2 > 20:
                a_opt2, _ = multiplicative_weights_optimize(
                    best_seq, time_limit=remaining2,
                    T_init=0.005, T_final=0.0001, n_iters=5000)
                try_seq(a_opt2, "final_mw")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
