    # (x*x)[2] = 2*x[0]*x[2] + 2*x[1]^2 = 1 => x[2] = (1-2*(1/4))/2 = (1/2)/2 = 1/4

    x[0] = 1.0

"""
Optimized step function search using flat-convolution initialization,
Nesterov-accelerated smoothed gradient, targeted peak reduction,
and multi-resolution exploration.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def flat_convolution_init(n):
    """Construct sequence with (a*a)[k] constant for k=0,...,n-1."""
    x = np.zeros(n)
    x[0] = 1.0
    for k in range(1, n):
        s = 0.0
        for i in range(1, k):
            s += x[i] * x[k - i]
        x[k] = (1.0 - s) / 2.0
    # Check if all positive
    if np.any(x < -1e-10):
        # Clip and renormalize
        x = np.maximum(x, 0.0)
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return x / s

    if s < 1e-15:

    else:
        a = a / s
    n = len(a)
    k_max = max(1, int(n * top_k_frac))
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    T = T_init
    lr = lr_init
    no_improve = 0
    it = 0
    
    # Nesterov momentum
    y = a.copy()
    m = np.zeros(n)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv_y = fast_convolve(y)
        
        # Top-k weighting
        idx = np.argsort(conv_y)[::-1][:k_max]
        top_vals = conv_y[idx]
        weights = np.exp(top_vals / T - top_vals.max())
        weights = weights / weights.sum()
        
        # Build p vector (sparse: only top-k lags have weight)
        p = np.zeros(2 * n - 1)
        p[idx] = weights
        
        # Gradient: grad[j] = 2 * sum_m a[m] * p[j+m]
        # Using FFT for efficiency
        # grad = 2 * ifft(fft(a) * conj(fft(p)))... but we need linear correlation
        # Actually: grad[j] = 2 * (a * p reversed)[j] where * is convolution
        # Let's use: grad[j] = 2 * sum_{m=0}^{n-1} a[m] * p[j+m]
        # This is a cross-correlation of a with p, restricted to valid range
        
        # Efficient computation via FFT
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        A_fft = np.fft.rfft(a, fft_size)
        P_fft = np.fft.rfft(p, fft_size)
        # (a * p)[k] = sum_i a[i] * p[k-i]
        # We want sum_m a[m] * p[j+m] = (a * p_rev)[j] where p_rev[i] = p[size-1-i]
        p_rev = p[::-1]
        P_rev_fft = np.fft.rfft(p_rev, fft_size)
        corr = np.fft.irfft(A_fft * P_rev_fft, fft_size)[:n]
        grad = 2.0 * corr

        y = a + (momentum / (1 - momentum)) * m if momentum < 1 else a + m
        y = project_simplex(y)

        best_a_trial = a
        best_val = fast_convolve(a).max()

            trial = (1 - alpha) * a + alpha * y
            trial = np.maximum(trial, 0)
            ts = trial.sum()
            if ts > 1e-15:
                trial = trial / ts
                val = fast_convolve(trial).max()
                if val < best_val:
                    best_val = val
                    best_a_trial = trial

            best_a = a.copy()
            no_improve = 0
        else:

            lr = lr_init

            no_improve = 0

            y = a.copy()
            # Small perturbation
            rng = np.random.default_rng(it * 31 + 7)
            a = project_simplex(a + rng.standard_normal(n) * 0.002)
        
        if no_improve > 500:
            break
    
    return best_a, best_obj

One more idea: **alternating minimization over the active set**.

At the optimum, there's a set S of lags where (a*a)[k] = max. The KKT conditions say:
- For k in S: (a*a)[k] = t (the max value)
- For k not in S: (a*a)[k] <= t
- The gradient of the Lagrangian is zero

The Lagrangian is: L = t + sum_{k in S} lambda_k * (t - (a*a)[k]) - mu * (sum(a) - 1) - sum_i nu_i * a_i

At the optimum: grad_a L = 0, which gives:
-2 * sum_{k in S} lambda_k * grad_a [(a*a)[k]] - mu * 1 - nu = 0

This is a system of equations that determines the optimal a. But solving it directly is hard because the active set S changes.

We can use an iterative approach:
1. Start with an estimate of S (e.g., the top-k lags)
2. Solve the KKT system for the current S
3. Update S based on the new solution
4. Repeat

Step function optimization using flat-convolution initialization,
Nesterov-accelerated smoothed gradient with top-k focus,
targeted peak reduction, and multi-resolution exploration.

    return np.fft.irfft(A * A, fft_size)[:size]

    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]

    cond = u * np.arange(1, n + 1) > (cssv - 1)

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    x[0] = 1.0

        for i in range(1, k):

        x[k] = (1.0 - s) / (2.0 * x[0])

    P_rev_fft = np.fft.rfft(p_rev, fft_size)
    corr = np.fft.irfft(A_fft * P_rev_fft, fft_size)[:n]
    return 2.0 * corr


def nesterov_opt(a0, time_limit, T_init=0.003, lr_init=0.02, 
                  top_k_frac=0.05, momentum=0.85):
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    k_max = max(2, int(n * top_k_frac))
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    T = T_init
    lr = lr_init
    no_improve = 0
    it = 0
    y = a.copy()
    m = np.zeros(n)
    rng = np.random.default_rng(int(time.time() * 1000) % 100000)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(y)
        top_idx = np.argsort(conv)[::-1][:k_max]
        top_vals = conv[top_idx]
        w = np.exp((top_vals - top_vals.max()) / T)
        w = w / w.sum()
        
        p = np.zeros(2 * n - 1)
        p[top_idx] = w
        
        grad = compute_grad_topk(a, p)
        
        # Nesterov momentum update
        m = momentum * m - lr * grad
        y = project_simplex(a + m / (1.0 - momentum + 1e-10))
        
        # Line search
        best_trial = a
        best_val = fast_convolve(a).max()
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.1]:
            trial = (1.0 - alpha) * a + alpha * y
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            val = fast_convolve(trial).max()
            if val < best_val:
                best_val = val
                best_trial = trial
        
        a = best_trial
        
        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt
        if no_improve > 0 and no_improve % 30 == 0:
            T *= 1.03
            lr *= 0.99
        if no_improve > 150:
            T = T_init
            lr = lr_init
            m = np.zeros(n)
            y = a.copy()
            no_improve = 0
            a = project_simplex(a + rng.standard_normal(n) * 0.003)
        if no_improve > 500:
            break
    
    return best_a, best_obj


def peak_reduction_opt(a0, time_limit, n_lags=3):
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        top_idx = np.argsort(conv)[::-1][:n_lags]
        p = np.zeros(2 * n - 1)
        p[top_idx] = 1.0 / n_lags
        
        grad = compute_grad_topk(a, p)
        
        improved = False
        for lr in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02]:
            candidate = project_simplex(a - lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_obj - 1e-15:
                a = candidate
                best_obj = val
                best_a = a.copy()
                no_improve = 0
                improved = True
                break
        
        if not improved and it % 3 == 0:
            # Pair moves
            for _ in range(20):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                for frac in [0.03, 0.07, 0.15, 0.25]:
                    candidate = a.copy()
                    amount = a[j] * frac
                    if amount < 1e-15:
                        continue
                    candidate[j] -= amount
                    candidate[i] += amount
                    candidate = np.maximum(candidate, 0)
                    cs = candidate.sum()
                    if cs < 1e-10:
                        continue
                    candidate = candidate / cs
                    val = fast_convolve(candidate).max()
                    if val < best_obj - 1e-15:
                        a = candidate
                        best_obj = val
                        best_a = a.copy()
                        no_improve = 0
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            no_improve += 1
            if no_improve > 200:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                no_improve = 0
            if no_improve > 1000:
                break
    
    return best_a, best_obj
