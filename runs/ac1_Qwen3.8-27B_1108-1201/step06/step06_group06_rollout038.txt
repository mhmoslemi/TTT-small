        return np.array([a[0] * cp[0]] if n == 1 else np.zeros(0))
    full = np.correlate(cp, a, mode='full')
    return full[n - 1: 2 * n - 1].copy()


def compute_lp_grad(a, c, p):
    """Gradient of sum_k c[k]^p w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    if p == 1.0:
        cp = np.maximum(c, 0.0)
    else:
        cp = np.maximum(c, 0.0) ** (p - 1.0)
    result = hankel_matvec(cp, a)
    return p * result


def lp_continuation(a0, time_limit, p_values=None, verbose=True):
    """
    Lp continuation: optimize sum c[k]^p for increasing p.
    Uses projected gradient with Nesterov momentum and backtracking.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    if p_values is None:
        p_values = [2.0, 3.0, 4.0, 6.0, 8.0, 12.0, 16.0, 24.0, 32.0]

    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = fast_convolve(a).max()

    for p in p_values:
        if time.monotonic() - t0 > time_limit - 2.0:
            break

        # Compute Lipschitz constant estimate for step size
        c = fast_convolve(a)
        grad = compute_lp_grad(a, c, p)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            continue

        # Initial step size based on gradient magnitude
        lr = 0.5 / max(gnorm, 1e-10)
        lr = min(lr, 1.0 / n * 0.5)
        beta = 0.85  # momentum

        momentum = np.zeros(n)
        a_cur = a.copy()
        it = 0
        max_iters = 3000
        stalled = 0

        while it < max_iters and time.monotonic() - t0 < time_limit - 1.0:
            it += 1
            c_cur = fast_convolve(a_cur)
            val_cur = np.sum(c_cur ** p)

            grad = compute_lp_grad(a_cur, c_cur, p)

            # Nesterov momentum
            momentum = beta * momentum + (1.0 - beta) * grad
            step = a_cur - lr * momentum
            a_new = project_simplex(step)

            c_new = fast_convolve(a_new)
            val_new = np.sum(c_new ** p)

            if val_new < val_cur - 1e-18:
                a_cur = a_new
                lr = min(lr * 1.02, 0.5 / max(np.linalg.norm(grad), 1e-10))
                lr = min(lr, 1.0 / n * 0.5)
                stalled = 0
            else:
                lr *= 0.5
                stalled += 1

            conv_max = c_cur.max()
            if conv_max < best_conv_max - 1e-12:
                best_conv_max = conv_max
                best_a = a_cur.copy()

            if stalled > 50:
                break

        a = a_cur.copy()
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    p={p}: best_conv_max={best_conv_max:.10f} (elapsed {elapsed:.0f}s, {it} iters)")

    return best_a, best_conv_max


def coordinate_descent(a0, time_limit, seed=42):
    """
    Coordinate descent: optimize one coordinate at a time with exact 1D search.
    For coordinate j, the contribution to conv is:
    conv[k] includes a[j]*a[k-j] and a[j]*a[j-k] (when valid)
    So changing a[j] affects conv linearly in a[j].
    The objective max(conv) is piecewise linear in a[j], so the optimum
    is at a breakpoint or where the max constraint is tight.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Precompute the "influence" of each coordinate on the convolution
    # conv[k] = sum_i a[i]*a[k-i]
    # dconv[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    # But since conv is symmetric, dconv[k]/da[j] = 2*a[k-j] (if both valid) or a[j]^2 (if k=j+0... )
    # Actually: conv[k] = sum_{i: valid} a[i]*a[k-i]
    # dconv[k]/da[j] = a[k-j] (from i=j term) + a[j-k] (from k-i=j term)
    # = a[k-j] + a[j-k] (when both 0<=k-j<n and 0<=j-k<n, i.e., k=j)
    # For k != j: only one term is non-zero

    # For coordinate j, conv[k] as a function of a[j]:
    # conv[k] = a[j]*(a[k-j] + a[j-k]) + (other terms not involving a[j])
    # Let u[k] = a[k-j] if 0<=k-j<n else 0, plus a[j-k] if 0<=j-k<n else 0
    # But we need to be careful: when we change a[j], the "other terms" also change
    # because conv[j+j] = ... includes a[j]^2.

    # Actually, let's just do it numerically:
    # For coordinate j, set a[j] = t and compute the resulting conv peak.
    # The conv peak as a function of t is the max of linear/quadratic functions.
    # For simplicity, use golden section search on t.

    while time.monotonic() - t0 < time_limit - 0.5:
        progress = False

        # Random coordinate order
        perm = rng.permutation(n)
        for j in perm:
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            a_save_j = a[j]
            if a_save_j < 1e-15:
                # Try setting it to a small positive value
                for t in [1e-4, 5e-4, 1e-3]:
                    a[j] = t
                    # Renormalize others
                    rest_sum = 1.0 - t
                    if rest_sum < 1e-10:
                        a[j] = a_save_j
                        break
                    a[np.arange(n) != j] *= rest_sum / (1.0 - a_save_j) if (1.0 - a_save_j) > 1e-10 else 1.0
                    obj = fast_convolve(a).max()
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_a = a.copy()
                        progress = True
                        break
            else:
                # Golden section search on a[j]
                # a[j] can range from 0 to 1 (approximately, since others must be >= 0)
                lo = 0.0
                hi = min(a_save_j * 3, 0.5)

                # Check endpoints and golden section points
                def obj_at(t):
                    a[j] = t
                    rest_sum = 1.0 - t
                    others_sum = 1.0 - a_save_j
                    if others_sum < 1e-15:
                        return fast_convolve(a).max()
                    if rest_sum < 1e-10:
                        return np.inf
                    mask = np.arange(n) != j
                    a[mask] = a[mask] * (rest_sum / others_sum)
                    return fast_convolve(a).max()

                best_t = a_save_j
                best_v = best_obj

                for t in [0.0, a_save_j * 0.5, a_save_j * 0.8, a_save_j * 1.0,
                          a_save_j * 1.2, a_save_j * 1.5, a_save_j * 2.0]:
                    if t >= hi or t < 0:
                        continue
                    v = obj_at(t)
                    if v < best_v:
                        best_v = v
                        best_t = t

                # Fine search around best_t
                if best_t > 0 and best_t < 1:
                    width = a_save_j * 0.1
                    for frac in np.linspace(-1, 1, 9):
                        t = best_t + frac * width
                        if t < 0 or t > 0.9:
                            continue
                        v = obj_at(t)
                        if v < best_v:
                            best_v = v
                            best_t = t

                a[j] = best_t
                others_sum = 1.0 - a_save_j
                rest_sum = 1.0 - best_t
                if others_sum > 1e-15 and rest_sum > 1e-10:
                    mask = np.arange(n) != j
                    a[mask] = best_a[mask] * (rest_sum / others_sum)
                # Actually this scaling is wrong if best_a != a. Let me just normalize.
                a = np.maximum(a, 0)
                ss = a.sum()
                if ss > 1e-15:
                    a = a / ss
                else:
                    a = best_a.copy()
                    break

                obj = fast_convolve(a).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                    progress = True

        if not progress:
            # Try a few random restarts
            for _ in range(3):
                if time.monotonic() - t0 > time_limit - 0.5:
                    break
                a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
                obj = fast_convolve(a).max()
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

    return best_a, best_obj


def subgradient_maxmin(a0, time_limit, T_init=0.005, verbose=True):
    """
    Direct minimization of max(conv) using smoothed gradient with adaptive temperature.
    Different from previous: uses alternating peak targeting and faster convergence.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    T = T_init
    lr = 0.01
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit - 0.5:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.3:
            break

        c = fast_convolve(a)
        c_max = c.max()

        # Smoothed max: softmax approximation
        scaled = c / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Gradient: grad[j] = 2 * sum_k w[k] * dc[k]/da[j] = 2 * sum_k w[k] * a[|k-j|] (approximately)
        # More precisely: grad[j] = 2 * (w * a_flipped correlation at position j)
        # dc[k]/da[j] = a[k-j] (if 0<=k-j<n)
        # grad[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
        # = 2 * sum_m a[m] * w[j+m] (substituting m = k-j)
        # = 2 * hankel_matvec(w, a)[j]
        grad = 2.0 * hankel_matvec(w, a)

        # Gradient step with projection
        a_new = project_simplex(a - lr * grad)
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        if obj_new < c_max - 1e-15:
            a = a_new
            if c_max < best_obj - 1e-14:
                best_obj = c_max
                best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.01, 0.1)
        else:
            no_improve += 1
            lr *= 0.8

        # Adapt temperature
        if no_improve > 20:
            T *= 1.05
        if no_improve > 80:
            T *= 0.7
            lr = 0.01
        if no_improve > 200:
            rng = np.random.default_rng((it * 7919) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = 0.01
            no_improve = 0

    return best_a, best_obj


def wavelet_init(n, seed=0):
    """Generate a wavelet-motivated initial sequence."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64) / n

    # Multi-scale structure
    a = np.ones(n)
    for scale in [1, 2, 4, 8, 16, 32]:
        freq = scale * rng.uniform(0.8, 1.2)
        phase = rng.uniform(0, 2 * np.pi)
        w = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x + phase)
        a *= (0.5 + 0.5 * w)

    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def try_eval(a, label, best_seq, best_score):
    """Evaluate sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return a_norm.copy(), score
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry: Lp-continuation + subgradient + coordinate descent hybrid search.
    Different from previous SOTA: uses smooth-to-sharp continuation, FFT-based
    Hankel operations, and coordinate descent polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Using uniform init: score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Multi-n exploration with Lp-continuation ===
    print("\n=== Phase 1: Multi-n Lp-continuation ===")
    t_phase = time.monotonic()

    # n values to explore
    n_candidates = []
    for n_try in [256, 384, 512, 640, 768, 896, 1024, 1280, 1536, 2048, 3072]:
        n_candidates.append(n_try)
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    # Also add some in between
    n_candidates.append(int(n_parent * 1.5))
    n_candidates.append(int(n_parent * 0.75))
    n_candidates = sorted(set(n for n in n_candidates if 128 <= n <= 4096))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 200:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            # Interpolate parent to new length
            start = np.interp(np.linspace(0, 1, n_try),
                             np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try

        # Allocate time: shorter for new n, longer for parent n
        if n_try == n_parent:
            t_alloc = min(remaining * 0.15, 120)
        else:
            t_alloc = min(remaining * 0.06, 50)

        print(f"  n={n_try}, t_alloc={t_alloc:.0f}s")
        a_opt, _ = lp_continuation(start, time_limit=t_alloc,
                                    p_values=[2.0, 4.0, 8.0, 16.0], verbose=False)
        best_seq, best_score = try_eval(a_opt, f"lp_n{n_try}", best_seq, best_score)

    print(f"  Phase 1 done: best={best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep Lp-continuation on best n ===
    print("\n=== Phase 2: Deep Lp-continuation ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 150:
        t_alloc = min(remaining * 0.35, 450)
        print(f"  Deep Lp n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = lp_continuation(best_seq, time_limit=t_alloc,
                                    p_values=[2.0, 3.0, 4.0, 6.0, 8.0, 12.0, 16.0, 24.0, 32.0],
                                    verbose=True)
        best_seq, best_score = try_eval(a_opt, "deep_lp", best_seq, best_score)

    # Multiple restarts with perturbation
    for scale in [0.005, 0.01, 0.02, 0.04]:
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break
        n_best = len(best_seq)
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc = min(remaining * 0.12, 100)
        a_opt, _ = lp_continuation(f_pert, time_limit=t_alloc,
                                    p_values=[4.0, 8.0, 16.0, 32.0], verbose=False)
        best_seq, best_score = try_eval(a_opt, f"lp_pert_{scale}", best_seq, best_score)

    print(f"  Phase 2 done: best={best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Subgradient max-minimization ===
    print("\n=== Phase 3: Subgradient max-min ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 100:
        t_alloc = min(remaining * 0.2, 180)
        print(f"  Subgradient T=0.003, t={t_alloc:.0f}s")
        a_opt, _ = subgradient_maxmin(best_seq, time_limit=t_alloc, T_init=0.003, verbose=True)
        best_seq, best_score = try_eval(a_opt, "subgrad_T3", best_seq, best_score)

    remaining = deadline - time.monotonic()
    if remaining > 80:
        t_alloc = min(remaining * 0.15, 120)
        print(f"  Subgradient T=0.001, t={t_alloc:.0f}s")
        a_opt, _ = subgradient_maxmin(best_seq, time_limit=t_alloc, T_init=0.001, verbose=True)
        best_seq, best_score = try_eval(a_opt, "subgrad_T1", best_seq, best_score)

    # Different T with restart
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_best = len(best_seq)
        a_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.005)
        t_alloc = min(remaining * 0.15, 80)
        a_opt, _ = subgradient_maxmin(a_pert, time_limit=t_alloc, T_init=0.01, verbose=False)
        best_seq, best_score = try_eval(a_opt, "subgrad_T10", best_seq, best_score)

    print(f"  Phase 3 done: best={best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Lp refinement with higher p ===
    print("\n=== Phase 4: High-p Lp refinement ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 80:
        t_alloc = min(remaining * 0.2, 150)
        print(f"  Lp p=[8,16,32,64], t={t_alloc:.0f}s")
        a_opt, _ = lp_continuation(best_seq, time_limit=t_alloc,
                                    p_values=[8.0, 16.0, 32.0, 64.0], verbose=True)
        best_seq, best_score = try_eval(a_opt, "lphigh", best_seq, best_score)

    # Try slightly different n
    remaining = deadline - time.monotonic()
    if remaining > 60:
        for n_try in [int(n_best * 1.25), int(n_best * 0.8)]:
            if n_try < 128 or n_try > 4096:
                continue
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            start = np.interp(np.linspace(0, 1, n_try),
                             np.linspace(0, 1, n_best), best_seq)
            start = np.maximum(start, 0)
            start = start / start.sum()
            t_alloc = min(remaining * 0.2, 60)
            a_opt, _ = lp_continuation(start, time_limit=t_alloc,
                                        p_values=[4.0, 8.0, 16.0, 32.0], verbose=False)
            best_seq, best_score = try_eval(a_opt, f"lphigh_n{n_try}", best_seq, best_score)

    print(f"  Phase 4 done: best={best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Coordinate descent polish ===
    print("\n=== Phase 5: Coordinate Descent ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 50:
        t_alloc = min(remaining * 0.3, 150)
        print(f"  Coord descent, t={t_alloc:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc, seed=42)
        best_seq, best_score = try_eval(a_opt, "coord", best_seq, best_score)

    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_alloc = min(remaining * 0.2, 60)
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc, seed=123)
        best_seq, best_score = try_eval(a_opt, "coord2", best_seq, best_score)

    print(f"  Phase 5 done: best={best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Final polish ===
    print("\n=== Phase 6: Final Polish ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    # Very fine Lp
    if remaining > 30:
        t_alloc = min(remaining * 0.2, 60)
        a_opt, _ = lp_continuation(best_seq, time_limit=t_alloc,
                                    p_values=[32.0, 64.0], verbose=False)
        best_seq, best_score = try_eval(a_opt, "finelp", best_seq, best_score)

    # Subgradient with very small T
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t_alloc = min(remaining * 0.2, 40)
        a_opt, _ = subgradient_maxmin(best_seq, time_limit=t_alloc, T_init=0.0005, verbose=False)
        best_seq, best_score = try_eval(a_opt, "finesub", best_seq, best_score)

    # Tiny perturbations + quick Lp
    remaining = deadline - time.monotonic()
    for i in range(5):

            break
        noise = rng.standard_normal(n_best) * 0.001
        f_pert = project_simplex(best_seq + noise)
        t_alloc = min(remaining * 0.15, 15)
        a_opt, _ = lp_continuation(f_pert, time_limit=t_alloc,
                                    p_values=[16.0, 32.0], verbose=False)
        best_seq, best_score = try_eval(a_opt, f"micro{i}", best_seq, best_score)
        remaining = deadline - time.monotonic()

    # Coordinate descent one more time
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_opt, _ = coordinate_descent(best_seq, time_limit=min(remaining * 0.3, 30), seed=777)
        best_seq, best_score = try_eval(a_opt, "coord_final", best_seq, best_score)

    if best_seq is None:

        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score) and final_score < best_score:

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")

    return [float(x) for x in best_seq]
