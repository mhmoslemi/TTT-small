                    a2 = lo + gr * (hi - lo)
                    v2 = eval_alpha(a2)
            best_alpha = (lo + hi) / 2.0
            best_val = eval_alpha(best_alpha)

        if best_alpha > 1e-12 and best_val < M - 1e-15:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_factor = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                # Update active set
                thr = M * 0.85
                active_lags = np.where(conv >= thr)[0].tolist()
                if len(active_lags) < 3:
                    active_lags = np.where(conv >= M * 0.6)[0].tolist()
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_factor = min(m_factor * 1.08, 4.0)

        if no_improve > 35:
            # Structured restart
            rng = np.random.default_rng(it * 104729 + 7)
            conv = fast_convolve(best_f)
            peak_k = int(np.argmax(conv))
            noise = rng.standard_normal(n) * 0.004
            # Bias: redistribute mass away from peak contributors
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            noise[js:je + 1] -= 0.002
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_factor = 1.0
            no_improve = 0
            active_lags = np.where(conv >= M * 0.85)[0].tolist()

    return best_f, best_obj


def nesterov_smooth_subgrad(a0, time_limit, T_init=0.001, lr_init=0.02, decay=0.9995):
    """Nesterov-accelerated subgradient with softmax peak weighting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    y = a.copy()
    v = np.zeros(n)
    t_nest = 1.0
    lr = lr_init
    T = T_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.2:
            break

        # Nesterov lookahead
        alpha_n = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / alpha_n
        y = (1.0 + beta) * a - beta * y
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y / ys
        t_nest += 1.0

        # Compute softmax-weighted subgradient
        conv = fast_convolve(y)
        scaled = (conv - conv.max()) / T
        p = np.exp(scaled)
        p = p / p.sum()

        # Subgradient: for each peak lag k with weight p[k],
        # the gradient contribution to coordinate i is 2*p[k]*y[k-i] (windowed)
        sg = np.zeros(n)
        # Vectorized: for each k, add p[k] * 2 * reversed_windowed_y
        for k in range(len(conv)):
            if p[k] < 1e-14:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            seg = y[js:je + 1]
            # The gradient of (y*y)[k] w.r.t. y[i] for i in [js, je] is 2*y[k-i]
            # So sg[i] += p[k] * 2 * y[k - i]
            # For i in [js, je]: k - i ranges from k-je to k-js
            k_minus_i = k - np.arange(js, je + 1)
            vals = y[k_minus_i]
            sg[js:je + 1] += p[k] * 2.0 * vals

        # Update
        v = beta * v - lr * sg
        a_new = a + v
        a_new = project_simplex(a_new)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        lr *= decay

        # Adaptive temperature and restart
        if no_improve > 50:
            T *= 1.1
        if no_improve > 100:
            T *= 0.7
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            y = a.copy()
            v = np.zeros(n)
            t_nest = 1.0
            lr = lr_init
            T = T_init
            no_improve = 0

    return best_a, best_obj


def distribution_match_polish(a0, time_limit):
    """
    Polish by trying to match the auto-convolution to a target flat distribution.
    Uses iterative water-filling / projection style updates.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.3:
            break

        conv = fast_convolve(a)
        M = conv.max()
        peak_k = int(np.argmax(conv))

        # Target: we want conv to be more uniform
        # The "excess" at the peak should be redistributed
        # Find coordinates that contribute to the peak and reduce them,
        # while increasing coordinates that don't contribute to any peak

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)

        # Compute contribution of each coordinate to the peak
        contrib = np.zeros(n)
        for i in range(js, je + 1):
            partner = peak_k - i
            if 0 <= partner < n:
                contrib[i] = a[partner]

        # Find coordinates with low total peak contribution
        total_peak_contrib = np.zeros(n)
        top_peaks = np.argpartition(conv, -5)[-5:]
        for pk in top_peaks:
            pk = int(pk)
            ps = max(0, pk - n + 1)
            pe = min(n - 1, pk)
            for i in range(ps, pe + 1):
                partner = pk - i
                if 0 <= partner < n:
                    total_peak_contrib[i] += a[partner] * (conv[pk] / M)

        # Move mass from high-contribution to low-contribution coordinates
        high_idx = np.where(total_peak_contrib > np.mean(total_peak_contrib) + 0.001)[0]
        low_idx = np.where(total_peak_contrib < np.mean(total_peak_contrib) - 0.001)[0]

        if len(high_idx) == 0 or len(low_idx) == 0:
            stall += 1
        else:
            # Try a small transfer
            rng = np.random.default_rng(it * 7 + 13)
            src = rng.choice(high_idx)
            dst = rng.choice(low_idx)
            
            best_gain = 0
            best_frac = 0
            max_transfer = a[src] * 0.3

            for frac in [0.05, 0.1, 0.15, 0.2, 0.25, 0.3]:
                delta = a[src] * frac
                if delta > max_transfer:
                    continue
                a_test = a.copy()
                a_test[src] -= delta
                a_test[dst] += delta
                a_test = np.maximum(a_test, 0.0)
                s = a_test.sum()
                if s < 1e-15:
                    continue
                a_test = a_test / s
                c_new = fast_convolve(a_test)
                new_max = c_new.max()
                gain = M - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_frac = frac

            if best_gain > 1e-12:
                delta = a[src] * best_frac
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                new_obj = fast_convolve(a).max()
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_a = a.copy()
                stall = 0
            else:
                stall += 1

        # Periodic random perturbation
        if stall > 30:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31 - 1))
            noise_scale = 0.001 + 0.002 * min(1.0, stall / 100.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            stall = 0

        if stall > 300:
            break

    return best_a, best_obj


def simulated_annealing_polish(a0, time_limit, T_start=0.005, T_end=0.0001):
    """Simulated annealing for final polish."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    total_iters = max(1000, int(time_limit * 100))  # Estimate iterations

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.2:
            break

        # Annealing schedule
        progress = min(1.0, it / total_iters)
        T = T_start * (T_end / T_start) ** progress

        rng = np.random.default_rng(it * 2654435761 % (2**31 - 1))

        # Generate perturbation: pick 2-5 random coordinates and redistribute
        n_perturb = rng.integers(2, 6)
        idx = rng.choice(n, size=n_perturb, replace=False)

        a_new = a.copy()
        mass = a[idx].copy()
        # Randomly redistribute
        new_mass = mass * rng.dirichlet(np.ones(n_perturb) * 2.0)
        total_old = mass.sum()
        if total_old > 1e-15:
            new_mass = new_mass * total_old / new_mass.sum()
        a_new[idx] = new_mass

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

        obj_new = fast_convolve(a_new).max()
        delta = obj_new - fast_convolve(a).max()

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_new
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a.copy()

    return best_a, best_obj


def hann_window(n):
    """Generate a raised cosine (Hann) window."""
    if n == 1:
        return np.array([1.0])
    return 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))


def blackman_window(n):
    """Generate a Blackman window."""
    if n == 1:
        return np.array([1.0])
    m = np.arange(n)
    w = 0.42 - 0.5 * np.cos(2 * np.pi * m / (n - 1)) + 0.08 * np.cos(4 * np.pi * m / (n - 1))
    return w


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search combining FW-LP,
    Nesterov subgradient, distribution matching, and simulated annealing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    t0_total = time.monotonic()

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t0_total
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512

    n_parent = len(parent)

    # === Phase 1: Multi-n Exploration ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    # Determine n values to try
    n_candidates = sorted(set([
        max(256, n_parent - 128),
        n_parent,
        min(1200, n_parent + 128),
        384, 512, 640, 768, 896, 1024
    ]))
    n_candidates = [n for n in n_candidates if 256 <= n <= 1200]
    # Remove duplicates while preserving order
    seen = set()
    n_candidates_unique = []
    for n in n_candidates:
        if n not in seen:
            seen.add(n)
            n_candidates_unique.append(n)
    n_candidates = n_candidates_unique

    for n_try in n_candidates:
        if time.monotonic() > deadline - 1100:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            # Interpolate parent to new size
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss > 1e-10:
                start = start / ss
            else:
                start = np.ones(n_try) / n_try

        t_alloc = min(40, deadline - time.monotonic() - 1100)
        if t_alloc < 10:
            break
        print(f"  FW explore n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(start, time_limit=t_alloc, max_lp_time=3.0)
        try_seq(a_opt, f"explore_n{n_try}")

    # Also try structured initializations
    for n_try in [512, 768]:
        if time.monotonic() > deadline - 1050:
            break
        for wname, wf in [("hann", hann_window), ("blackman", blackman_window)]:
            w = wf(n_try)
            w = w / w.sum()
            t_alloc = min(20, deadline - time.monotonic() - 1050)
            if t_alloc < 5:
                break
            print(f"  FW {wname} n={n_try}, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(w, time_limit=t_alloc, max_lp_time=2.0)
            try_seq(a_opt, f"{wname}_n{n_try}")

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW-LP on best ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        t_alloc = min(400, deadline - time.monotonic() - 650)
        if t_alloc > 20:
            print(f"  Deep FW n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, max_lp_time=5.0)
            try_seq(a_opt, "deep_fw")

        # Perturbed restarts
        for scale in [0.003, 0.007, 0.015]:
            t_alloc = min(60, deadline - time.monotonic() - 550)
            if t_alloc < 10:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc, max_lp_time=4.0)
            try_seq(a_opt, f"fw_pert{scale}")

    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Nesterov Subgradient ===
    print("\n=== Phase 3: Nesterov Subgradient ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        configs = [
            {'T_init': 0.0005, 'lr_init': 0.004, 'decay': 0.9997},
            {'T_init': 0.002, 'lr_init': 0.008, 'decay': 0.9995},
            {'T_init': 0.001, 'lr_init': 0.003, 'decay': 0.9998},
        ]
        for ci, cfg in enumerate(configs):
            t_alloc = min(100, deadline - time.monotonic() - 450)
            if t_alloc < 10:
                break
            print(f"  Nesterov cfg{ci} T={cfg['T_init']} lr={cfg['lr_init']}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_smooth_subgrad(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"nest_{ci}")

    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Distribution Matching Polish ===
    print("\n=== Phase 4: Distribution Matching ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(150, deadline - time.monotonic() - 200)
        if t_alloc > 10:
            print(f"  Dist match polish, t={t_alloc:.0f}s")
            a_opt, _ = distribution_match_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "dist_match")

        # Second pass from slightly different start
        if time.monotonic() < deadline - 150:
            t_alloc = min(80, deadline - time.monotonic() - 150)
            if t_alloc > 10:
                perturbed = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.002)
                a_opt, _ = distribution_match_polish(perturbed, time_limit=t_alloc)
                try_seq(a_opt, "dist_match2")

    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Simulated Annealing Polish ===
    print("\n=== Phase 5: Simulated Annealing ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(100, deadline - time.monotonic() - 60)
        if t_alloc > 10:
            print(f"  SA polish, t={t_alloc:.0f}s")
            a_opt, _ = simulated_annealing_polish(best_seq, time_limit=t_alloc,
                                                   T_start=0.003, T_end=0.00005)
            try_seq(a_opt, "sa")

    print(f"  Phase 5 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Final FW + Polish ===
    print("\n=== Phase 6: Final Refinement ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # One more FW pass
        t_alloc = min(60, deadline - time.monotonic() - 30)
        if t_alloc > 5:
            print(f"  Final FW, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, max_lp_time=3.0)
            try_seq(a_opt, "final_fw")

        # Quick SA from best
        if time.monotonic() < deadline - 20:
            t_alloc = min(25, deadline - time.monotonic() - 15)
            if t_alloc > 5:
                a_opt, _ = simulated_annealing_polish(best_seq, time_limit=t_alloc,
                                                       T_start=0.001, T_end=0.00002)
                try_seq(a_opt, "final_sa")

        # Quick distribution match
        if time.monotonic() < deadline - 10:
            t_alloc = min(15, deadline - time.monotonic() - 5)
            if t_alloc > 3:
                a_opt, _ = distribution_match_polish(best_seq, time_limit=t_alloc)
                try_seq(a_opt, "final_dm")

    print(f"  Phase 6 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, total time={time.monotonic()-t0_total:.0f}s)")
    return [float(x) for x in best_seq]
