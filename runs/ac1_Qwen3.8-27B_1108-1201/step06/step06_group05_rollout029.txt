def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_shaping(a0, time_limit, n_iter=None, relaxation=0.5, method='soft'):
    """
    Spectral shaping via SOR:
    1. Compute A = FFT(a)
    2. B = A^2
    3. b = IFFT(B) (the convolution)
    4. Shape b to reduce peaks
    5. B_shaped = FFT(b_shaped)
    6. A_shaped = sqrt(B_shaped)
    7. a_shaped = IFFT(A_shaped)
    8. Project to non-negative simplex
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    L = 1 << (2 * n - 1 - 1).bit_length()  # FFT size >= 2n-1
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    alpha = 0.1  # shaping strength
    
    if n_iter is None:
        n_iter = 100000
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        if it % 1000 == 0:
            print(f"    Spectral it={it}, obj={best_obj:.10f}, no_improve={no_improve}")
        
        # Compute spectrum
        A = np.fft.fft(a, L)
        B = A * A  # element-wise square
        b = np.fft.ifft(B).real  # should be real
        
        # Shape b to reduce peaks
        b_max = b.max()
        b_mean = b.sum() / (2 * n - 1)  # average over valid range
        
        if method == 'hard':
            M = best_obj * 1.05  # target peak
            b_shaped = np.minimum(b, np.maximum(M, b_mean * 2))
        elif method == 'soft':
            # Soft shaping: compress values above mean
            excess = np.maximum(b - b_mean, 0)
            b_shaped = b - alpha * excess * (excess / b_max)
        else:  # 'exp'
            # Exponential suppression
            b_shaped = b * np.exp(-alpha * (b / b_max - 1)**2)
        
        # Ensure b_shaped sum is preserved (roughly)
        # Actually, let's not enforce this - we'll normalize a at the end
        
        B_shaped = np.fft.fft(b_shaped)
        
        # Square root with phase continuity
        # A_shaped[m] should satisfy A_shaped[m]^2 = B_shaped[m]
        # and A_shaped[L-m] = conj(A_shaped[m])
        # A_shaped[0] should be positive real
        
        angles = np.angle(B_shaped)
        magnitudes = np.abs(B_shaped)
        
        # Previous angles (for phase continuity)
        prev_angles = np.angle(A)
        
        # New angles are half of B's angles, but we need to handle the 2π ambiguity
        new_angles = angles / 2.0
        
        # For phase continuity: choose the branch closest to previous
        # B_shaped[m] ≈ B[m] (if shaping is mild), so angle(B_shaped[m]) ≈ angle(B[m]) = 2*angle(A[m]) (mod 2π)
        # So angle(A_new[m]) ≈ angle(A[m]) (mod π)
        # We want to choose angle(A_new[m]) = angle(B_shaped[m])/2 or angle(B_shaped[m])/2 + π
        # such that it's closest to prev_angles[m]
        
        diff = new_angles - prev_angles
        diff = (diff + np.pi) % (2 * np.pi) - np.pi  # wrap to [-π, π]
        
        # If |diff| > π/2, the branch is wrong, add π
        wrong_branch = np.abs(diff) > np.pi / 2
        new_angles[wrong_branch] += np.pi
        
        A_shaped = magnitudes ** 0.5 * np.exp(1j * new_angles)
        
        # Ensure conjugate symmetry
        A_shaped[0] = np.abs(A_shaped[0])  # positive real
        for m in range(1, L // 2 + 1):
            if m < L - m:
                A_shaped[L - m] = np.conj(A_shaped[m])
            elif m == L - m:  # Nyquist
                A_shaped[m] = np.real(A_shaped[m])
        
        a_shaped = np.fft.ifft(A_shaped).real[:n]
        
        # Project to non-negative simplex
        a_shaped = np.maximum(a_shaped, 0)
        ss = a_shaped.sum()
        if ss < 1e-15:
            break
        a_shaped = a_shaped / ss
        
        # Relaxed update
        a_new = (1 - relaxation) * a + relaxation * a_shaped
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        obj = fast_convolve(a_new).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            # Slightly increase shaping strength
            alpha = min(alpha * 1.01, 0.5)
        else:
            a_new = a  # revert
            no_improve += 1
            if no_improve % 100 == 0:
                alpha = max(alpha * 0.9, 0.01)
        
        a = a_new
    
    return best_a, best_obj

def subgrad_multi_peak(a, top_k=10):
    """Compute weighted subgradient from top-k peaks."""
    n = len(a)
    conv = fast_convolve(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    order = np.argsort(pv)[::-1]
    weights = np.zeros(len(peaks))
    for rank, idx in enumerate(order):
        weights[idx] = 1.0 / (rank + 1) ** 1.5
    ws = weights.sum()
    if ws > 0:
        weights /= ws
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx_arr = np.arange(js, je + 1)
        sg[idx_arr] += weights[i] * a[k - idx_arr]
    return sg, conv

def pgd_momentum(a0, time_limit, top_k=20, lr_init=0.02, mom_init=0.9):

def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral shaping via Newton's method for complex square root (new frequency-domain approach)
2. Improved projected subgradient with multi-peak adaptive weighting
3. Softmax-smoothed gradient descent
4. Parametric initialization (Gaussian, cosine, polynomial forms)
5. Systematic n exploration with interpolation
6. Random restarts with adaptive perturbation
"""
import numpy as np
import time


def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_shaping(a0, time_limit, relaxation=0.4, shape_alpha=0.1, use_newton=True):
    """
    Frequency-domain spectral shaping:
    1. A = FFT(a), B = A^2, b = IFFT(B)
    2. Shape b to reduce peaks
    3. B_new = FFT(b_shaped)
    4. A_new = Newton step toward sqrt(B_new)
    5. a_new = IFFT(A_new), project to simplex
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    L = 1 << (2 * n - 1 - 1).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    alpha = shape_alpha

    while time.monotonic() - t0 < time_limit:
        if it % 2000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral it={it}, obj={best_obj:.10f}, no_imp={no_improve}, alpha={alpha:.4f}")
        it += 1

        # Compute spectrum
        A = np.fft.fft(a, L)
        B = A * A
        b = np.fft.ifft(B).real

        # Shape b to reduce peaks
        bmax = b.max()
        bmin = b.min()
        b_mean = b.sum() / (2 * n - 1)

        # Soft peak reduction
        excess = np.maximum(b - b_mean, 0.0)
        b_shaped = b - alpha * excess * (excess / max(bmax, 1e-15))

        # Also slightly boost below-mean values to maintain energy
        deficit = np.maximum(b_mean - b, 0.0)
        b_shaped = b_shaped + 0.5 * alpha * deficit * (deficit / max(b_mean - bmin, 1e-15))

        # Ensure non-negativity of b_shaped (convolution of non-negative seq should be non-negative)
        b_shaped = np.maximum(b_shaped, 0.0)

        B_new = np.fft.fft(b_shaped)

        if use_newton:
            # Newton step for complex square root: x_new = 0.5*(x + B_new/x)
            # Start from current A
            A_safe = np.where(np.abs(A) > 1e-15, A, 1e-15 + 0j)
            A_new = 0.5 * (A + B_new / A_safe)
        else:
            # Direct square root
            A_new = np.sqrt(B_new)
            # Fix branch: A[0] should be positive real
            if A_new[0].real < 0:
                A_new = -A_new

        A_new[0] = complex(A_new[0].real, 0.0)
        for m in range(1, L // 2 + 1):
            if m < L - m:
                A_new[L - m] = np.conj(A_new[m])
            elif m == L - m:
                A_new[m] = complex(A_new[m].real, 0.0)

        a_new = np.fft.ifft(A_new).real[:n]

        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        sc = a_cand.sum()
        if sc < 1e-15:
            break
        a_cand = a_cand / sc

            best_obj = obj
            best_a = a_cand.copy()

def subgrad_multi_peak(a, top_k=10):
    """Compute weighted subgradient from top-k peaks."""
    n = len(a)
    conv = fast_convolve(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    order = np.argsort(pv)[::-1]
    weights = np.zeros(len(peaks))
    for rank, idx in enumerate(order):
        weights[idx] = 1.0 / (rank + 1) ** 1.5
    ws = weights.sum()
    if ws > 0:
        weights /= ws
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx_arr = np.arange(js, je + 1)
        sg[idx_arr] += weights[i] * a[k - idx_arr]
    return sg, conv


def pgd_momentum(a0, time_limit, top_k=20, lr_init=0.02, mom_init=0.9):

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)

        a = a / s

    mom = mom_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        sg, conv = subgrad_multi_peak(a, top_k)
        obj = conv.max()

        best_candidate = a
        best_val = obj
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.125]:
            v_new = mom * v - trial_lr * sg
            candidate = project_simplex(a + v_new)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                v = v_new
                break

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.95

            v = np.zeros(n)
            lr = lr_init
            mom = mom_init
            top_k = 20
            no_improve = 0

    return best_a, best_obj


def smooth_grad(a0, time_limit, T_init=0.001, lr_init=0.01):
    """Smoothed gradient using softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init
    lr = lr_init

    v = np.zeros(n)
    mom = 0.92
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = np.zeros(n)
        for k in range(len(conv)):
            if p[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += 2.0 * p[k] * a[k - idx_arr]

        v = mom * v - lr * grad
        best_candidate = a
        best_val = conv.max()
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            candidate = project_simplex(a + scale * v)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.92

        if no_improve > 30:
            T *= 1.1
            lr *= 0.95
        if no_improve > 250:
            rng = np.random.default_rng((it * 31 + 7) % (2**31 - 1))
            noise = 0.003 + 0.004 * min(1.0, (no_improve - 250) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_obj


def parametric_sequence(n, kind='gaussian', **kwargs):
    """Generate parametric initial sequences."""
    x = np.linspace(0, 1, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'gaussian':
        sigma = kwargs.get('sigma', 0.2)
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    elif kind == 'cosine':
        beta = kwargs.get('beta', 0.5)
        a = 1.0 + beta * np.cos(2 * np.pi * x)
    elif kind == 'bump':
        # Two bumps
        w = kwargs.get('width', 0.15)
        a = np.exp(-((x - 0.3) / w) ** 2) + np.exp(-((x - 0.7) / w) ** 2)
    elif kind == 'sinoid':
        freq = kwargs.get('freq', 3.0)
        amp = kwargs.get('amp', 0.3)
        a = 1.0 + amp * np.sin(2 * np.pi * freq * x)
    elif kind == 'polynomial':
        deg = kwargs.get('deg', 4)
        coeffs = kwargs.get('coeffs', None)
        if coeffs is None:
            # Random polynomial (non-negative on [0,1])
            a = 1.0
            for d in range(1, deg + 1):
                a += 0.1 * np.cos(d * np.pi * x)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s


def try_and_update(a, best_state, label=""):
    """Evaluate sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_state
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    best_seq, best_score = best_state
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return (best_seq, best_score)


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Uses spectral shaping (Newton sqrt in frequency domain)
    as the primary novel method, combined with subgradient, smoothed gradient,
    parametric initialization, multi-n exploration, and random restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    # === Phase 0: Parametric initialization scan (~60s) ===
    print("\n=== Phase 0: Parametric Initialization ===")
    t_start = time.monotonic()
    n_parent = len(parent)

    param_configs = [
        ('gaussian', {'sigma': 0.15}),
        ('gaussian', {'sigma': 0.2}),
        ('gaussian', {'sigma': 0.3}),
        ('cosine', {'beta': 0.3}),
        ('cosine', {'beta': 0.5}),
        ('bump', {'width': 0.12}),
        ('bump', {'width': 0.2}),
        ('sinoid', {'freq': 2.0, 'amp': 0.3}),
        ('sinoid', {'freq': 4.0, 'amp': 0.2}),
    ]

    for kind, kw in param_configs:
        remaining = deadline - time.monotonic() - 1500
        if remaining < 5:
            break
        a_param = parametric_sequence(n_parent, kind=kind, **kw)
        (best_seq, best_score) = try_and_update(a_param, (best_seq, best_score), f"param_{kind}")

    print(f"  Phase 0 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 1: Spectral shaping on best sequence (~400s) ===
    print("\n=== Phase 1: Spectral Shaping ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    # Multiple spectral shaping configurations
    for relax, alpha in [(0.3, 0.05), (0.5, 0.1), (0.4, 0.15), (0.6, 0.08)]:
        remaining = deadline - time.monotonic() - 900
        if remaining < 20:
            break
        t_alloc = min(remaining, 100)
        print(f"  Spectral relax={relax}, alpha={alpha}, t={t_alloc:.0f}s")
        a_opt, _ = spectral_shaping(best_seq, time_limit=t_alloc, relaxation=relax, shape_alpha=alpha)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"spec_r{relax}_a{alpha}")

    # Follow up with PGD
    remaining = deadline - time.monotonic() - 800
    if remaining > 20:
        t_alloc = min(remaining, 80)
        a_opt, _ = pgd_momentum(best_seq, time_limit=t_alloc, top_k=20, lr_init=0.02)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "pgd_p1")

    # Smoothed gradient
    remaining = deadline - time.monotonic() - 700
    if remaining > 15:
        t_alloc = min(remaining, 60)
        a_opt, _ = smooth_grad(best_seq, time_limit=t_alloc, T_init=0.001, lr_init=0.01)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "smooth_p1")

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 2: Multi-n exploration (~400s) ===
    print("\n=== Phase 2: Multi-n Exploration ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    # Generate candidate n values
    candidate_ns = set()
    for n_try in [256, 320, 384, 448, 512, 576, 640, 704, 768, 832, 896, 960, 1024, 1152, 1280, 1408, 1536]:
        candidate_ns.add(n_try)
    # Add multiples of parent n
    for mult in [0.5, 1.0, 1.5, 2.0]:
        n_try = int(n_best * mult)
        if 128 <= n_try <= 2048:
            candidate_ns.add(n_try)
    candidate_ns = sorted([n for n in candidate_ns if 128 <= n <= 2048])

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 350
        if remaining < 30:
            break

        if n_try == n_best and best_seq is not None and len(best_seq) == n_try:
            start = best_seq.copy()
        else:
            start = interp_sequence(best_seq, n_try)

        # Parametric at this n
        a_param = parametric_sequence(n_try, kind='gaussian', sigma=0.2)
        start = project_simplex(0.5 * start + 0.5 * a_param)

        t_alloc = min(remaining, 50)
        print(f"  n={n_try}, t={t_alloc:.0f}s, cur_best={best_score:.8f}")

        # Spectral
        a_opt, _ = spectral_shaping(start, time_limit=t_alloc * 0.5, relaxation=0.5, shape_alpha=0.1)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"spec_n{n_try}")

        # PGD
        remaining = deadline - time.monotonic() - 300
        if remaining > 15:
            a_opt2, _ = pgd_momentum(a_opt, time_limit=min(remaining, t_alloc * 0.3), top_k=15, lr_init=0.015)
            (best_seq, best_score) = try_and_update(a_opt2, (best_seq, best_score), f"pgd_n{n_try}")

        # Smooth
        remaining = deadline - time.monotonic() - 250
        if remaining > 10:
            a_opt3, _ = smooth_grad(a_opt, time_limit=min(remaining, t_alloc * 0.2), T_init=0.001, lr_init=0.008)
            (best_seq, best_score) = try_and_update(a_opt3, (best_seq, best_score), f"sm_n{n_try}")

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 3: Deep spectral + refinement on best n (~300s) ===
    print("\n=== Phase 3: Deep Refinement ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    # Deep spectral with various params
    for relax, alpha in [(0.2, 0.03), (0.3, 0.05), (0.5, 0.08), (0.4, 0.12)]:
        remaining = deadline - time.monotonic() - 200
        if remaining < 20:
            break
        a_opt, _ = spectral_shaping(best_seq, time_limit=min(remaining, 60), relaxation=relax, shape_alpha=alpha)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"deep_r{relax}a{alpha}")

    # PGD variants
    for top_k, lr in [(10, 0.01), (30, 0.008), (5, 0.005)]:
        remaining = deadline - time.monotonic() - 150
        if remaining < 10:
            break
        a_opt, _ = pgd_momentum(best_seq, time_limit=min(remaining, 40), top_k=top_k, lr_init=lr)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"pgd_k{top_k}")

    # Smoothed variants
    for T in [0.0005, 0.001, 0.003, 0.01]:
        remaining = deadline - time.monotonic() - 100
        if remaining < 10:
            break
        a_opt, _ = smooth_grad(best_seq, time_limit=min(remaining, 30), T_init=T, lr_init=0.008)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"sm_T{T}")

    # Spectral again after refinement
    remaining = deadline - time.monotonic() - 60
    if remaining > 15:
        a_opt, _ = spectral_shaping(best_seq, time_limit=min(remaining, 50), relaxation=0.3, shape_alpha=0.05)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "deep_spec2")

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 4: Random restarts (~150s) ===
    print("\n=== Phase 4: Random Restarts ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    restart_scales = [0.002, 0.004, 0.006, 0.01, 0.015, 0.02, 0.03]
    for scale in restart_scales:
        remaining = deadline - time.monotonic() - 80
        if remaining < 15:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc = min(remaining, 30)

        # Spectral on perturbed
        a_opt, _ = spectral_shaping(f_pert, time_limit=t_alloc * 0.6, relaxation=0.5, shape_alpha=0.1)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"rst_{scale}_spec")

        # PGD on perturbed
        remaining = deadline - time.monotonic() - 60
        if remaining > 10:
            a_opt2, _ = pgd_momentum(a_opt, time_limit=min(remaining, t_alloc * 0.4), top_k=20, lr_init=0.01)
            (best_seq, best_score) = try_and_update(a_opt2, (best_seq, best_score), f"rst_{scale}_pgd")

    # Targeted perturbation: perturb only the peak region
    conv = fast_convolve(best_seq)
    peak_idx = np.argmax(conv)
    # Identify which a elements contribute most to the peak
    js = max(0, peak_idx - n_best + 1)
    je = min(n_best - 1, peak_idx)
    peak_support = np.arange(js, je + 1)
    for scale in [0.003, 0.008]:
        remaining = deadline - time.monotonic() - 40
        if remaining < 10:
            break
        f_pert = best_seq.copy()
        f_pert[peak_support] += rng.standard_normal(len(peak_support)) * scale
        f_pert = np.maximum(f_pert, 0)
        f_pert = f_pert / f_pert.sum()
        a_opt, _ = spectral_shaping(f_pert, time_limit=min(remaining, 20), relaxation=0.5, shape_alpha=0.1)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"peak_rst_{scale}")

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Final polishing (~80s) ===
    print("\n=== Phase 5: Final Polish ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    # Quick spectral
    remaining = deadline - time.monotonic() - 25
    if remaining > 10:
        a_opt, _ = spectral_shaping(best_seq, time_limit=min(remaining, 30), relaxation=0.3, shape_alpha=0.05)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "fp_spec")

    # Quick PGD
    remaining = deadline - time.monotonic() - 15
    if remaining > 5:
        a_opt, _ = pgd_momentum(best_seq, time_limit=min(remaining, 20), top_k=20, lr_init=0.01)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "fp_pgd")

    # Quick smooth
    remaining = deadline - time.monotonic() - 10
    if remaining > 5:
        a_opt, _ = smooth_grad(best_seq, time_limit=min(remaining, 15), T_init=0.001, lr_init=0.005)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "fp_smooth")

    # Symmetry
    sym_final = (best_seq + best_seq[::-1]) / 2.0
    sym_final = sym_final / sym_final.sum()
    (best_seq, best_score) = try_and_update(sym_final, (best_seq, best_score), "fp_sym")

    # Tiny perturbation + spectral
    remaining = deadline - time.monotonic() - 5
    if remaining > 5:
        f_tiny = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
        a_opt, _ = spectral_shaping(f_tiny, time_limit=min(remaining, 10), relaxation=0.5, shape_alpha=0.05)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "fp_tiny")

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
