def nesterov_subgrad(a0, time_limit, lr_init=0.005, momentum_init=0.9, verbose=False):

    n = len(a)

            break

        conv_y = fast_convolve(y)
        k_peak = int(np.argmax(conv_y))

        js = max(0, k_peak - n + 1)

        a = y + v_new
        v = v_new
        a = project_simplex(a)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0

            no_improve += 1

            no_improve = 0

(a_new * a_new)[k] ≈ (1-t)^2 M + 2t(1-t) M/sum(g)
= M [(1-t)^2 + 2t(1-t)/sum(g)]
= M (1-t) [(1-t) + 2t/sum(g)]

2t/sum(g) < t

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- LP-based primal-dual iteration (maximize sum(g) s.t. (a*g)_k <= M, g>=0)
- Smoothed gradient descent (log-sum-exp surrogate) for polishing
- Nesterov-accelerated subgradient for fine-tuning
- Adaptive step sizes, multiple n exploration, and random restarts
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 without calling evaluate_sequence."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * len(a) * conv.max() / (s * s))


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where A[k,j] = a[k-j] if 0<=k-j<n. Shape (2n-1, n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def hankel_mv(w, a):
    """Compute W @ a where W_{i,j} = w[i+j]. w has length 2n-1, a has length n."""

def smoothed_grad(a0, time_limit, T_init=0.001, lr_init=0.03, verbose=False):
    """Smoothed gradient descent with log-sum-exp surrogate for min max conv(a,a)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = 2.0 * hankel_mv(p, a)

        # Backtracking line search
        best_candidate = a
        best_val = conv.max()
        trial_lr = lr
        for _ in range(10):
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                break
            trial_lr *= 0.5

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 10:
            T *= 1.08
            lr *= 0.97
        if no_improve > 50:
            T *= 0.7
        if no_improve > 150:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    Smoothed it={it}, obj={best_obj:.10f}, T={T:.6f}, lr={lr:.6f}")

    return best_a, best_obj


def nesterov_subgrad(a0, time_limit, lr_init=0.005, momentum_init=0.9, verbose=False):
    """Nesterov-accelerated subgradient method."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = momentum_init
    y = a.copy()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        y = a + mom * v
        conv_y = fast_convolve(y)
        k_peak = int(np.argmax(conv_y))

        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * y[k_peak - idx]

        v_new = project_simplex(y - lr * grad) - y
        a = y + v_new
        v = v_new
        a = project_simplex(a)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9999
        mom = min(0.995, mom + 0.00003)

        if no_improve > 80:
            lr *= 0.85
            mom = max(0.8, mom * 0.95)
        if no_improve > 250:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = lr_init * 0.5
            mom = momentum_init
            no_improve = 0

    return best_a, best_obj


def multiplicative_weight(a0, time_limit, lr=0.05, eta=0.1, verbose=False):
    """
    Multiplicative weight update approach.
    alternates between:
    - Primal: a[i] *= exp(-lr * (p @ H @ a)[i]) then project to simplex
    - Dual: p[k] *= exp(eta * conv[k]) then normalize
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    # Initialize dual distribution (uniform over peaks)
    p = np.ones(2 * n - 1) / (2 * n - 1)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Primal update: gradient of E_p[conv] w.r.t. a
        # grad[i] = 2 * sum_k p[k] * a[k-i] (for valid k-i)
        # This is 2 * hankel_mv(p, a)
        grad = 2.0 * hankel_mv(p, a)
        a = a * np.exp(-lr * grad)
        a = project_simplex(a)

        # Dual update
        conv = fast_convolve(a)
        p = p * np.exp(eta * (conv - conv.max()))
        p = p / p.sum()

        obj = conv.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            lr *= 0.9
            eta *= 1.1
        if no_improve > 200:
            lr = 0.05
            eta = 0.1
            rng = np.random.default_rng((it * 31 + 11) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            p = np.ones(2 * n - 1) / (2 * n - 1)
            no_improve = 0

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search:
    Phase 1: LP-based primal-dual (main workhorse)
    Phase 2: n-exploration
    Phase 3: Smoothed gradient + MW polishing
    Phase 4: Nesterov fine-tuning
    Phase 5: Final LP + verification
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        # Use own computation for speed
        score = compute_objective(a_norm)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        try_seq(parent, "uniform_384")

    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")

    # === Phase 1: LP-based primal-dual on parent (main workhorse) ===
    print("\n=== Phase 1: LP Primal-Dual ===")
    t1 = time.monotonic()

    remaining = deadline - time.monotonic() - 900
    t_alloc = min(max(60, remaining), 600)
    print(f"  LP-PD n={n_parent}, t={t_alloc:.0f}s")
    a_opt, _ = lp_primal_dual(parent, time_limit=t_alloc, t_init=0.3, verbose=True)
    try_seq(a_opt, "lp_pd_phase1")

    # Second LP pass with different step size
    remaining = deadline - time.monotonic() - 850
    if remaining > 30 and best_seq is not None:
        t_alloc = min(remaining, 200)
        print(f"  LP-PD n={len(best_seq)}, t={t_alloc:.0f}s, t_init=0.15")
        a_opt, _ = lp_primal_dual(best_seq, time_limit=t_alloc, t_init=0.15)
        try_seq(a_opt, "lp_pd_phase1b")

    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")

    # === Phase 2: n-exploration ===
    print("\n=== Phase 2: n-exploration ===")
    t2 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        candidate_ns = sorted(set([
            n_best,
            max(128, n_best - 32),
            n_best + 32,
            n_best + 64,
            max(128, n_best // 2 + 1),
            min(2048, n_best * 2),
        ]))

        for n_try in candidate_ns:
            remaining = deadline - time.monotonic() - 400
            if remaining < 30:
                break

            if n_try == n_best:
                start = best_seq.copy()
            else:
                start = np.interp(np.linspace(0, 1, n_try),
                                  np.linspace(0, 1, n_best), best_seq)
                start = np.maximum(start, 0.0)
                start = start / start.sum()

            t_alloc = min(remaining, 100)
            print(f"  LP-PD n={n_try}, t={t_alloc:.0f}s")
            a_opt, _ = lp_primal_dual(start, time_limit=t_alloc, t_init=0.3)
            try_seq(a_opt, f"lp_n{n_try}")

    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")

    # === Phase 3: Smoothed gradient + MW polishing ===
    print("\n=== Phase 3: Smoothed Gradient & MW ===")
    t3 = time.monotonic()

    if best_seq is not None:
        # Smoothed gradient with different temperatures
        for T_val, lr_val in [(0.0005, 0.02), (0.001, 0.015), (0.002, 0.01)]:
            remaining = deadline - time.monotonic() - 250
            if remaining < 15:
                break
            t_alloc = min(remaining, 60)
            print(f"  Smoothed T={T_val}, lr={lr_val}, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_grad(best_seq, time_limit=t_alloc,
                                     T_init=T_val, lr_init=lr_val)
            try_seq(a_opt, f"smooth_T{T_val}")

        # Multiplicative weight
        remaining = deadline - time.monotonic() - 200
        if remaining > 15:
            t_alloc = min(remaining, 60)
            print(f"  MW lr=0.03, eta=0.05, t={t_alloc:.0f}s")
            a_opt, _ = multiplicative_weight(best_seq, time_limit=t_alloc, lr=0.03, eta=0.05)
            try_seq(a_opt, "mw")

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")

    # === Phase 4: More LP with varied params ===
    print("\n=== Phase 4: LP Variants ===")
    t4 = time.monotonic()

    if best_seq is not None:
        for t_init_val in [0.1, 0.5, 0.2]:
            remaining = deadline - time.monotonic() - 100
            if remaining < 15:
                break
            t_alloc = min(remaining, 60)
            print(f"  LP t_init={t_init_val}, t={t_alloc:.0f}s")
            a_opt, _ = lp_primal_dual(best_seq, time_limit=t_alloc, t_init=t_init_val)
            try_seq(a_opt, f"lp_t{t_init_val}")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # === Phase 5: Nesterov + final polish ===
    print("\n=== Phase 5: Nesterov & Final Polish ===")
    t5 = time.monotonic()

    if best_seq is not None:
        for lr_i in [0.003, 0.008, 0.001]:
            remaining = deadline - time.monotonic() - 30
            if remaining < 10:
                break
            t_alloc = min(remaining, 40)
            print(f"  Nesterov lr={lr_i}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=t_alloc, lr_init=lr_i)
            try_seq(a_opt, f"nesterov_lr{lr_i}")

        # One final LP pass
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            t_alloc = min(remaining, 20)
            a_opt, _ = lp_primal_dual(best_seq, time_limit=t_alloc, t_init=0.2)
            try_seq(a_opt, "final_lp")

        # Final smoothed gradient with very low temperature
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            t_alloc = min(remaining, 15)
            a_opt, _ = smoothed_grad(best_seq, time_limit=t_alloc,
                                     T_init=0.0002, lr_init=0.005)
            try_seq(a_opt, "final_smooth")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = compute_objective(best_seq)

    # Final verification with evaluate_sequence
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    elif not np.isfinite(fs):
        # Shouldn't happen, but fallback
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
