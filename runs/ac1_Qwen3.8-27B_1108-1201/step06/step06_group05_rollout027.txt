def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_grad_full(a, T):
    """Compute gradient of T*logsumexp(conv(a)/T) w.r.t. a, plus the softmax weights."""
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return grad, p, conv

def active_set_optimize(a0, time_limit, peak_tol=0.01):
    """Focus optimization on the active peak set using weighted subgradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()

        if time.monotonic() - t0 > time_limit - 0.05:

        cur_max = conv.max()

        else:
            no_improve += 1

        if total_active < 1e-15:

        w = active / total_active

        for idx, k in enumerate(active_indices):
            wi = w[idx]

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += 2.0 * wi * a[k - idx_arr]
        
        v = mom * v - lr * grad
        best_candidate = a

            c = fast_convolve(candidate)

            if val < best_val - 1e-16:

                best_candidate = candidate
                break

            lr *= 0.92
            mom = min(0.97, mom + 0.003)

def lw_subgradient(a0, time_limit, top_k=50):
    """Loram-Warm start: use top-k peaks with harmonic weights + Nesterov momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)
    best_a = a.copy()
    conv = fast_convolve(a)
    best_max = conv.max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    lr = 0.03
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        it += 1
        
        conv_y = fast_convolve(y)
        # Top-k with harmonic weights
        if top_k >= len(conv_y):
            peaks = np.arange(len(conv_y))
            weights = np.ones(len(peaks))
        else:
            peaks = np.argpartition(conv_y, -top_k)[-top_k:]
            pv = conv_y[peaks]
            order = np.argsort(pv)[::-1]
            weights = np.zeros(len(peaks))
            for rank, idx in enumerate(order):
                weights[idx] = 1.0 / (rank + 1)**1.3
            ws = weights.sum()
            weights /= ws
        
        sg = np.zeros(n)
        for i, k in enumerate(peaks):
            if weights[i] < 1e-14:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * y[k - idx_arr]
        
        # Nesterov step
        t_k = 2.0 / (it + 1.5 + 1e-10)
        v_new = (it / (it + 1.0 + t_k)) * v
        y_new = y + t_k * (a - y) + t_k * v_new
        a_new = project_simplex(y_new - lr * sg)
        
        # Backtracking
        best_candidate = a
        best_val = fast_convolve(a).max()
        for trial in [a_new, 
                       project_simplex(0.5*a + 0.5*a_new),
                       project_simplex(0.7*a + 0.3*a_new)]:
            c = fast_convolve(trial)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = trial
                break
        
        a = best_candidate
        v = v_new
        y = y_new
        
        if best_val < best_max - 1e-15:
            best_max = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            lr = max(0.001, lr * 0.98)
        
        if no_improve > 300:
            rng = np.random.default_rng((it * 31 + 7) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            y = a.copy()
            v = np.zeros(n)
            lr = 0.03
            no_improve = 0
    
    return best_a, best_max

def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s

def symmetrize(a):
    """Make sequence symmetric."""
    s = (a + a[::-1]) / 2.0
    s = np.maximum(s, 0)
    sm = s.sum()
    if sm < 1e-15:
        return a.copy()
    return s / sm
