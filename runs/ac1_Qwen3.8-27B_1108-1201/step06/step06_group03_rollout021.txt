"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an improved Frank-Wolfe
approach with away-steps, cubic line search, and spectral initialization,
followed by fine-grained coordinate descent polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (f*g)[k] = sum_j f[j]*g[k-j]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = j + n
        A[start:end, j] = f
    return A


def solve_lp_oracle(f, M, n):
    """Solve LP: max sum(g) s.t. A_f @ g <= M, g >= 0."""
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def cubic_line_search(f, g, M, n):
    """
    Cubic line search: find alpha in [0,1] minimizing max conv((1-a)f + a*g, (1-a)f + a*g).
    Uses golden section + cubic interpolation.
    """
    def obj(alpha):
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            return 1e10
        ft = ft / s
        return fast_convolve(ft).max()

    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    a_lo, a_hi = 0.0, 1.0
    a1 = a_hi - gr * (a_hi - a_lo)
    a2 = a_lo + gr * (a_hi - a_lo)
    f1, f2 = obj(a1), obj(a2)

    for _ in range(40):
        if f1 < f2:
            a_hi = a2
            a2 = a1
            f2 = f1
            a1 = a_hi - gr * (a_hi - a_lo)
            f1 = obj(a1)
        else:
            a_lo = a1
            a1 = a2
            f1 = f2
            a2 = a_lo + gr * (a_hi - a_lo)
            f2 = obj(a2)

    best_alpha = (a_lo + a_hi) / 2
    best_val = obj(best_alpha)
    return best_alpha, best_val


def fw_optimize_improved(f0, time_limit, use_away=True, verbose=False):
    """
    Improved Frank-Wolfe with away-steps and cubic line search.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:

    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    stagnant = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:

        # Solve LP oracle
        g = solve_lp_oracle(f, M, n)

        if g is None or g.sum() < 1e-15:
            # Try slightly relaxed

        if g is None or g.sum() < 1e-15:
            stagnant += 1
            if stagnant > 30:
                rng = np.random.default_rng(it * 7919 + 3)

                        if M < best_obj - 1e-15:

                            best_f = f.copy()
                            stagnant = 0
                        else:
                            stagnant += 1
                        continue

        # Apply the standard FW step
        if best_alpha > 1e-10:
            f_new = (1.0 - best_alpha) * f + best_alpha * g
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            f = f_new
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW it={it}, obj={best_obj:.10f}, stagnant={stagnant}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def spectral_init(n, parent=None, rng=None):
    """
    Generate a spectrally-motivated initialization.
    Uses the idea that the optimal sequence has a smooth shape
    related to the eigenvector of the Toeplitz-like convolution structure.
    """
    if rng is None:
        rng = np.random.default_rng(42)

    starts = []

    # Resampled parent
    if parent is not None:
        old_n = len(parent)
        x_old = np.linspace(0, 1, old_n)
        x_new = np.linspace(0, 1, n)
        p = np.interp(x_new, x_old, parent)
        p = np.maximum(p, 0)
        s = p.sum()
        if s > 1e-15:
            starts.append(p / s)

    # Uniform
    starts.append(np.ones(n) / n)

    # Cosine (flat-top window shape)
    x = np.arange(n, dtype=np.float64)
    cos_t = np.cos(np.pi * x / (n - 1))
    w = 0.5 * (1 + cos_t)
    w = np.maximum(w, 0.001)
    starts.append(w / w.sum())

    # Raised cosine with wider main lobe
    w2 = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    w2 = np.maximum(w2, 0.001)
    starts.append(w2 / w2.sum())

    # Exponential decay from both ends towards center
    center = (n - 1) / 2.0
    exp_w = np.exp(-0.3 * np.abs(x - center) / n)
    starts.append(exp_w / exp_w.sum())

    # "U-shape" (concentrated at edges)
    edge_w = 1.0 / (0.1 + np.minimum(x, n - 1 - x) / n)
    starts.append(edge_w / edge_w.sum())

    # Bessel-like (smooth with compact support feel)
    from scipy.special import j0
    r = np.abs(x - center) / (0.5 * n)
    bessel_w = np.maximum(j0(np.pi * r), 0) ** 2
    bessel_w = np.maximum(bessel_w, 0.001)
    starts.append(bessel_w / bessel_w.sum())

    return starts


def coordinate_descent_polish(a0, time_limit):
    """
    Fine-grained coordinate descent: try small moves that reduce the peak.
    Uses a priority queue of "candidate moves" based on peak contributions.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    step_sizes = [0.005, 0.002, 0.001, 0.0005, 0.0002]
    ss_idx = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        # Focus on top peaks
        n_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        improved = False
        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)

            if js > je:
                continue

            # Find the top contributing pairs
            products = np.array([a[j] * a[k - j] for j in range(js, je + 1)])
            top_pairs = np.argsort(products)[::-1][:8]

            for pi in top_pairs:
                j1 = js + int(top_pairs[pi])
                j2 = k - j1

                if a[j1] < 1e-10 and a[j2] < 1e-10:
                    continue

                # Try moving mass from the larger of j1, j2 to a random distant location
                src = j1 if a[j1] >= a[j2] else j2
                step = step_sizes[ss_idx]

                for _attempts in range(6):
                    rng = np.random.default_rng((it * 9973 + k * 31 + src * 7 + _attempts) % (2**31))
                    j_target = int(rng.integers(0, n))
                    if abs(j_target - src) < n // 4:
                        continue  # Too close

                    for frac in [step, step * 0.5, step * 2]:
                        delta = a[src] * frac
                        if delta < 1e-12:
                            continue
                        candidate = a.copy()
                        candidate[src] -= delta
                        candidate[j_target] += delta
                        if candidate[src] < -1e-12:
                            continue
                        candidate = np.maximum(candidate, 0)
                        candidate = candidate / candidate.sum()
                        val = fast_convolve(candidate).max()
                        if val < best_obj - 1e-14:
                            a = candidate
                            best_obj = val
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            ss_idx = min(ss_idx + 1, len(step_sizes) - 1)
            if ss_idx == len(step_sizes) - 1 and step_sizes[ss_idx] < 1e-5:
                # Try a small random perturbation
                rng = np.random.default_rng(it * 48271)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-14:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
                    ss_idx = 0
                elif it % 500 == 0:
                    break
            elif not improved and it % 200 == 0:
                rng = np.random.default_rng(it * 48271)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.002)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-14:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
                    ss_idx = 0

    return best_a, best_obj

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix where (f*g)[k] = sum_j f[j]*g[k-j]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M, n):
    """Solve LP: max sum(g) s.t. A_f @ g <= M, g >= 0."""
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def line_search_alpha(f, g, n):

    for i, alpha in enumerate(alphas):
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)

        if s < 1e-15:

        ft = ft / s
        vals[i] = fast_convolve(ft).max()

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final alternating polish (remaining) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    n_best = len(best_seq)
    round_i = 0
    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        round_i += 1

        if round_i % 2 == 1:
            noise = rng.standard_normal(n_best) * 0.0003
            start = project_simplex(best_seq + noise)
            t_alloc = min(remaining, 30)
            a_opt, _ = fw_optimize(start, time_limit=t_alloc)
            do_try(a_opt, f"polish_fw_{round_i}")
        else:
            t_alloc = min(remaining, 25)
            a_opt, _ = coord_descent(best_seq, time_limit=t_alloc, top_k=3)
            do_try(a_opt, f"polish_cd_{round_i}")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        return [float(x) for x in best_seq]
    else:
        fallback = np.ones(768) / 768
        print(f"  FALLBACK: {evaluate_sequence(fallback.tolist()):.10f}")
        return [float(x) for x in fallback]
