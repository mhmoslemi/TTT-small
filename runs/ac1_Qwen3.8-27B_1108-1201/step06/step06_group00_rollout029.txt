def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

So the sum is over j = max(0, n-1-i) to n-1, and the argument of p is i+j, which ranges from i+max(0,n-1-i) to i+n-1 = 2n-2-i... 

If i <= n-1: n-1-i >= 0, so the sum is from j=n-1-i to j=n-1.
p index: from i+(n-1-i) = n-1 to i+(n-1) = 2n-2-i. Since i <= n-1, 2n-2-i >= n-1. ✓

If i > n-1: not applicable since we only compute for i in [0, n-1].

So for all i in [0, n-1]:
(a * v)[2n-2-i] = sum_{j=n-1-i}^{n-1} a[j] * p[i+j]

And p index ranges from n-1 to 2n-2-i (which is at most 2n-2). ✓

Now, what we want is:

The constraint i+j <= 2n-2 is automatic (i <= n-1, j <= n-1, so i+j <= 2n-2). ✓
The constraint i+j >= 0 is automatic. ✓
The constraint j >= 0 is automatic. ✓

So the full sum is: sum_{j=0}^{n-1} a[j] * p[i+j]

But from the convolution computation, we get: sum_{j=n-1-i}^{n-1} a[j] * p[i+j]

These are different! The convolution only gives us part of the sum (j from n-1-i to n-1), not the full sum (j from 0 to n-1).

The issue is that in the convolution (a*v)[m], we need both a[j] and v[m-j] to be within their valid ranges. For v, the valid range is [0, n-1], so we need 0 <= m-j <= n-1, i.e., m-n+1 <= j <= m.

For m = 2n-2-i: j ranges from 2n-2-i-(n-1) = n-1-i to min(2n-2-i, n-1) = n-1.

So we're missing the terms where j < n-1-i. For those terms, v[2n-2-i-j] would have an index > n-1, which is out of bounds.

But in our desired sum, p[i+j] for j < n-1-i would have index i+j < n-1, which is valid. The issue is that v is the reverse of p, and v[m] = p[2n-2-m]. When m = 2n-2-i-j and j < n-1-i, we get m > n-1, which is out of bounds for v (which has length n).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B on log-sum-exp smoothing with softmax parameterization
2. Adaptive multi-beta annealing from smooth to peak-focused
3. Spectral flattening (Gerchberg-Saxton) initialization
4. Multi-length exploration and interpolation
5. Coordinate descent refinement on top active peaks
6. Peak-pair mass transfer for final sub-1e-4 polish

import numpy as np
from scipy.optimize import minimize
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def lse_grad(a, beta, conv=None):
    """Compute LSE(beta*conv(a,a)) value and gradient.
    
    Gradient formula: grad[i] = 2 * sum_{j=0}^{n-1} a[j] * p[i+j]
    where p[k] = exp(beta*conv_k) / sum_k exp(beta*conv_k)
    """
    n = len(a)
    if conv is None:
        conv = fast_convolve(a)
    bc = beta * (conv - conv.max())
    exp_bc = np.exp(bc)
    sum_exp = exp_bc.sum()
    p = exp_bc / sum_exp
    lse_val = conv.max() + np.log(sum_exp) / beta
    # O(n^2) sliding dot product: grad[i] = 2 * sum_j a[j]*p[i+j]
    a_pad = np.concatenate([a, np.zeros(n - 1)])
    mat = np.vstack([a_pad[i:i + n] for i in range(n)])
    grad = 2.0 * (mat @ p)
    return lse_val, grad


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero-cond if False else np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_objective_and_grad(x, a, beta):
    """Objective and gradient for L-BFGS-B in simplex space.
    
    x is on the simplex (sum=1, x>=0). Returns LSE value and projected gradient.
    Uses softmax reparameterization internally for the optimizer.
    """
    val, grad_a = lse_grad(a, beta)
    dot = float(np.dot(grad_a, a))
    # Project gradient: subtract mean to stay on simplex
    proj_grad = grad_a - dot
    return val, proj_grad


def lbfgs_beta_step(f, beta, maxiter=300):
    """One step of L-BFGS-B at given beta, using log-parameterization."""
    n = len(f)
    f_safe = np.maximum(f, 1e-30)
    # Initial x such that softmax(x) = f
    x0 = np.log(f_safe)
    
    a_cur = f_safe.copy()
    
    def obj_and_grad(x):
        # Numerically stable softmax
        x_shift = x - np.max(x)
        exp_x = np.exp(x_shift)
        a = exp_x / exp_x.sum()
        val, grad_a = lse_grad(a, beta)
        dot = float(np.dot(grad_a, a))
        grad_x = a * (grad_a - dot)
        return val, grad_x
    
    try:
        res = minimize(
            obj_and_grad, x0, method='L-BFGS-B', jac=True,
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 40}
        )
        if res.x is not None:
            x_opt = res.x
            exp_x = np.exp(x_opt - np.max(x_opt))
            f_new = exp_x / exp_x.sum()
            return f_new
    except Exception:
        pass
    return f


def spectral_flatten(n, n_iter=80, seed=42):
    """Generate a non-negative sequence with flattened power spectrum."""
    rng = np.random.default_rng(seed)
    a = rng.uniform(0.5, 1.5, n)
    for _ in range(n_iter):
        A = np.fft.rfft(a)
        mag = np.abs(A)
        phase = np.angle(A)
        # Target: flat magnitude (with DC preserved)
        target = np.median(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
        # Blend toward flat
        new_mag = 0.7 * mag + 0.3 * target
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, n)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new /= s
        a = a_new
    return a


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def build_lp_matrix(f, constraint_indices):
    """Build sparse constraint matrix for LP: (f*g)[k] <= M."""
    n = len(f)
    rows, cols, vals = [], [], []
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([idx] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    if not rows:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(len(constraint_indices), n))


def lp_direction(f, M, top_k=40, time_limit=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top peaks, g >= 0, sum(g)=1."""
    n = len(f)
    conv = fast_convolve(f)
    k_sel = min(top_k + 1, 2 * n - 1)
    peak_idx = np.argpartition(conv, -k_sel)[-k_sel:]
    center = n - 1
    if center not in peak_idx:
        peak_idx = np.append(peak_idx, center)
    A = build_lp_matrix(f, peak_idx)
    if A.shape[0] == 0:
        return None
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    # Add sum(g) = 1 as bounds trick: maximize sum => minimize -sum
    # Actually just maximize sum(g) without the =1 constraint, normalize after
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None


def fw_refine(f, time_limit, top_k=35):
    """Frank-Wolfe with LP oracle for refinement."""
    n = len(f)
    f = np.maximum(f, 0).copy()
    f /= f.sum()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_imp = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 4:
            break
        conv = fast_convolve(f)
        M = conv.max()
        lk = min(top_k + no_imp, n)
        g = lp_direction(f, M, top_k=lk, time_limit=min(8, rem * 0.1))
        if g is None:
            no_imp += 5
            if no_imp > 60:
                break
            continue
        diff = np.max(np.abs(g - f))
        if diff < 1e-12:
            no_imp += 1
            if no_imp > 40:
                break
            continue
        # Line search
        best_a = 0.0
        best_v = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            v = fast_convolve(ft).max()
            if v < best_v:
                best_v = v
                best_a = alpha
        if best_a > 1e-6:
            # Bisection refinement
            lo, hi = max(0, best_a - 0.1), min(1, best_a + 0.1)
            for _ in range(8):
                mid = (lo + hi) / 2
                ft = (1 - mid) * f + mid * g
                ft = np.maximum(ft, 0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft /= st
                v = fast_convolve(ft).max()
                if v < best_v:
                    best_v = v
                    best_a = mid
                    lo = mid
                else:
                    hi = mid
            f = np.maximum((1 - best_a) * f + best_a * g, 0)
            f /= f.sum()
            conv_new = fast_convolve(f)
            M_new = conv_new.max()
            if M_new < best_obj - 1e-15:
                best_obj = M_new
                best_f = f.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
    return best_f, best_obj


def coord_descent(f, n_sweeps=50, n_peaks=10, time_limit=30):
    """Coordinate descent: try moving mass from peak contributors."""
    n = len(f)
    f = np.maximum(f, 0).copy()
    f /= f.sum()
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        conv = fast_convolve(f)
        top_k = min(n_peaks, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        for k in peaks:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo >= hi:
                continue
            # Find the pair contributing most to this peak
            js = np.arange(lo, hi + 1)
            contribs = f[js] * f[k - js]
            # Find which single element to reduce
            for j in js:
                other = k - j
                if other < 0 or other >= n or other == j:
                    continue
                # Try moving a small amount from j to a random destination
                for _ in range(5):
                    dest = rng.integers(0, n)
                    if dest == j or dest == other:
                        continue
                    delta = f[j] * 0.05
                    if delta < 1e-10:
                        continue
                    f_trial = f.copy()
                    f_trial[j] -= delta
                    f_trial[dest] += delta
                    if f_trial[j] < 0:
                        continue
                    f_trial = np.maximum(f_trial, 0)
                    f_trial /= f_trial.sum()
                    conv_t = fast_convolve(f_trial)
                    if conv_t.max() < best_obj - 1e-16:
                        best_obj = conv_t.max()
                        best_f = f_trial.copy()
                        f = f_trial.copy()
                        break
    return best_f, best_obj


def subgradient_lse(f, beta, n_steps=2000, lr=None):
    """Subgradient descent with LSE smoothing on the simplex."""
    n = len(f)
    f = np.maximum(f, 0).copy()
    f /= f.sum()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    if lr is None:
        lr = 2.0 / np.sqrt(n)
    
    for step in range(n_steps):
        conv = fast_convolve(f)
        val, grad = lse_grad(f, beta, conv)
        # Project gradient onto tangent space of simplex
        grad_proj = grad - np.mean(grad)
        gn = np.linalg.norm(grad_proj)
        if gn < 1e-15:
            break
        f_new = project_simplex(f - lr * grad_proj / gn)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            lr = min(lr * 1.003, 15.0 / np.sqrt(n))
        else:
            lr *= 0.95
            if lr < 0.001 / np.sqrt(n):
                f = best_f.copy()
                lr = 2.0 / np.sqrt(n)
                break
    return best_f, best_obj


def compute_score(a):
    """Compute the objective value (without calling evaluate_sequence)."""
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def try_update(a, best_tracker):

        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.10f} (n={len(a)})")
    return score

    t_rem = deadline - time.monotonic()
    if t_rem > 30:
        t_alloc = min(80, t_rem * 0.5)
        print(f"  Final L-BFGS-B (beta=50000) n={len(current_best)}, t={t_alloc:.0f}s")
        f_new = lbfgs_beta_step(current_best, beta=50000, maxiter=200)
        if f_new is not None:
            try_update(f_new, best_tracker)
    
    # More coordinate descent
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t_rem = deadline - time.monotonic()
    if t_rem > 20:
        t_alloc = min(60, t_rem * 0.5)
        print(f"  Final CoordDesc t={t_alloc:.0f}s")
        f_opt, _ = coord_descent(current_best, n_sweeps=200, n_peaks=20, time_limit=t_alloc)
        try_update(f_opt, best_tracker)
    
    # One more FW with many constraints
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t_rem = deadline - time.monotonic()
    if t_rem > 25:
        t_alloc = min(50, t_rem * 0.6)
        print(f"  Final FW (top_k=80) t={t_alloc:.0f}s")
        f_opt, _ = fw_refine(current_best, time_limit=t_alloc, top_k=80)
        try_update(f_opt, best_tracker)
    
    # Random perturbations
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t_rem = deadline - time.monotonic()
    if t_rem > 10:
        print(f"  Random perturbations t={t_rem:.0f}s")
        n_cur = len(current_best)
        for i in range(20):
            if time.monotonic() > deadline - 5:
                break
            for strength in [0.003, 0.01, 0.03]:
                f_pert = project_simplex(current_best + rng.standard_normal(n_cur) * strength)
                try_update(f_pert, best_tracker)
    
    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]
    
    # Fallback
    print("WARNING: Using fallback")
    fallback = np.ones(768) / 768.0
    return [float(x) for x in fallback]
