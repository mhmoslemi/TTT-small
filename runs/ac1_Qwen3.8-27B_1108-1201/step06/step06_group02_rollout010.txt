    diff = rows - cols

    A = np.zeros((2*n-1, n))
    A[mask] = f[diff[mask]]
    return A

def solve_lp(f, M):
    """LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = build_A_ub(f)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 20})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def fw_lp(f0, time_limit, lp_timeout=20):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    A_cached = None
    f_cached = None
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        # Solve LP
        if f_cached is None or not np.allclose(f, f_cached, atol=1e-12):
            A_cached = build_A_ub(f)
            f_cached = f.copy()
        
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M * m_exp)
        bounds = [(0, None)] * n
        try:
            res = linprog(c, A_ub=A_cached, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': lp_timeout})
            g = np.maximum(res.x, 0.0) if res.success else None
        except Exception:
            g = None
        
        if g is None or g.sum() < 1e-15:
            m_exp = min(m_exp * 1.2, 3.0)
            if m_exp > 2.5:
                # Perturb and restart
                rng = np.random.default_rng(it * 7919)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                f_cached = None
            continue
        
        g = g / g.sum()
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
                f_cached = None
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in np.linspace(0.05, 1.0, 15):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            ft = ft / ft.sum()
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            f_cached = None
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 3.0)
    
    return best_f, best_obj

def multiplicative_update(f0, time_limit, T_init=0.001):
    """EM-like multiplicative update that preserves non-negativity."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    T = T_init
    eta = 0.1
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(f)
        
        # Softmax weights on the convolution
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()
        
        # Compute "pressure" at each position
        # pressure_j = sum_k w_k * f_{k-j} (valid range)
        # This is the cross-correlation of w with f
        pressure = np.correlate(w, f, mode='full')[:2*n-1]
        # Actually, we want: for each j, sum over k of w_k * f_{k-j}
        # where k-j is in [0, n-1]
        # This is the convolution of w with reversed f, or correlation
        # Let me think: (f*w)_k = sum_j f_j * w_{k-j}
        # We want for each j: sum_k w_k * f_{k-j} = sum_k w_k * f_{k-j}
        # Let l = k-j, then k = j+l: sum_l w_{j+l} * f_l
        # This is the cross-correlation of w and f at lag j
        pressure = np.correlate(w, f, mode='full')[n-1:n-1+n]
        # Hmm, let me just use the hankel product
        # pressure_j = sum_k w_k * a_{k-j} for valid k,j
        # = sum_k w_k * a_{k-j} where 0 <= k-j < n
        # For fixed j: k ranges from j to j+n-1 (but k <= 2n-2)
        # So k ranges from j to min(j+n-1, 2n-2)
        # a_{k-j} ranges from a_0 to a_{min(n-1, 2n-2-j)}
        
        # Simpler: pressure = correlation of w with a (full), take the right part
        # np.correlate(w, a, 'full')[k] = sum_i w[i+k] * a[i] (for valid i)
        # We want: for each j, sum_k w[k] * a[k-j] (valid k-j)
        # = sum_i a[i] * w[i+j] (letting i = k-j, so k = i+j)
        # = np.correlate(w, a, 'full')[n-1+j] ... hmm, let me just compute directly
        
        # Actually, the simplest way:
        # pressure_j = sum_{valid k} w_k * a_{k-j}
        # = (w * reverse(a)) evaluated at appropriate index
        # Or use FFT:
        # pressure = ifft(fft(w, 2n) * conj(fft(a, 2n)))
        # This gives circular cross-correlation
        
        # Let me just use the direct formula via the hankel matrix
        # pressure_j = sum_k w_k * a_{k-j}
        # For the full convolution, this is the "anti-diagonal" sum
        
        # Actually, let me think about it as:
        # b_k = sum_j a_j * a_{k-j}
        # db_k/da_j = a_{k-j} + (if j != k-j) a_{k-j} = 2*a_{k-j} (for j ≠ k-j)
        # or a_{k-j} (for j = k-j, i.e., k = 2j)
        # The "gradient" of max(b) w.r.t. a_j (using smoothed max) is:
        # grad_j = sum_k w_k * db_k/da_j = sum_k w_k * (a_{k-j} + a_{k-j} * [j≠k-j])
        # ≈ 2 * sum_k w_k * a_{k-j} (approximately, ignoring the diagonal)
        
        # Let me just compute: grad_j = sum_k w_k * a_{|k-j|} ... no that's wrong.
        # b_k = sum_j a_j a_{k-j}, so db_k/da_j = a_{k-j} (treating a as independent in each product)
        # But since b_k = sum_j a_j a_{k-j}, and a_j appears in both factors when j = k-j,
        # the derivative is: db_k/da_j = a_{k-j} + a_j * (d a_{k-j}/d a_j)
        # = a_{k-j} + (1 if k=2j else 0)
        # For simplicity, let's use: grad_j = sum_k w_k * a_{k-j} (the "first-order" term)
        
        # Using FFT for efficiency:
        size = 2 * n
        W = np.fft.rfft(w, size)
        A = np.fft.rfft(f, size)
        # cross-correlation: sum_k w[k] * f[k-j] = ifft(W * conj(A))[j] (with appropriate shift)
        # Actually: ifft(|W+A|^2) gives autocorrelation... let me think.
        # correlation(j) = sum_k w[k] * f[k-j]
        # In FFT: ifft(FFT(w) * conj(FFT(f))) gives sum_k w[k] * conj(f)[k+j]
        # = sum_k w[k] * f[k-j] (for real f)
        # So: pressure = ifft(FFT(w, size) * conj(FFT(f, size)))
        # But this gives circular correlation. We need linear.
        # For linear correlation of w (length 2n-1) and f (length n):
        # result length is 3n-2, and we want the first n elements (or a subset)
        
        # Let me just use a simpler approach: compute grad via the hankel matrix
        # grad_j = sum_k w_k * a_{k-j} for valid (k,j)
        # This is equivalent to: (H_w @ a) where H_w is the hankel matrix of w
        # H_w[j, k] = w[j+k] ... no, that's not right either.
        
        # Let me just do it directly with numpy correlate
        # np.correlate(w, f, mode='full') gives:
        # result[m] = sum_i w[i+m] * f[i] (for valid i)
        # We want: for each j, sum_k w[k] * f[k-j]
        # Let i = k-j, so k = i+j: sum_i w[i+j] * f[i]
        # This is np.correlate(w, f, 'full')[j] ... let me check.
        # np.correlate(w, f, 'full')[m] = sum_i f[i] * w[i+m] (for valid i)
        # Wait, the numpy docs say: correlate(a, v)[k] = sum_n a[n+k] * conj(v[n])
        # So correlate(w, f, 'full')[j] = sum_n w[n+j] * conj(f[n]) = sum_n w[n+j] * f[n]
        # Yes! That's exactly what we want: pressure_j = sum_n w[n+j] * f[n]
        
        pressure_full = np.correlate(w, f, mode='full')
        # pressure_full has length (2n-1) + n - 1 = 3n-2
        # pressure_full[j] = sum_n w[n+j] * f[n] for j = 0, ..., 3n-3
        # We want j = 0, ..., n-1 (the first n elements)
        pressure = pressure_full[:n]
        
        # Multiplicative update: a_j <- a_j / (1 + eta * pressure_j)
        # This reduces a_j where pressure is high (contributing to peaks)
        f_new = f / (1.0 + eta * pressure / max(pressure.max(), 1e-15))
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()
        
        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            f = f_new  # still update to explore
            if no_improve > 50:
                f = best_f.copy()
                eta *= 0.8
                T *= 1.1
                no_improve = 0
        
        if it % 100 == 0:
            eta = max(eta * 0.95, 0.01)
    
    return best_f, best_obj

def smoothed_grad(a0, time_limit, T_init=0.001, lr_init=0.02):
    """Smoothed gradient descent with simplex projection."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    T = T_init
    lr = lr_init
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Gradient: grad_j = 2 * sum_k p_k * a_{k-j}
        pressure_full = np.correlate(p, a, mode='full')
        grad = 2.0 * pressure_full[:n]
        
        # Try multiple step sizes
        best_candidate = a
        best_val = conv.max()
        for trial_lr in [lr, lr*0.5, lr*0.25, lr*0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
        
        a = best_candidate
        obj = fast_convolve(a).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 20:
            T *= 1.05
            lr *= 0.97
        if no_improve > 200:
            rng = np.random.default_rng(it * 31 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj
