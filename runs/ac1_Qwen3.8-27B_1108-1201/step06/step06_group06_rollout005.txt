Looking at this problem, I need to minimize `2*n*max(autocorr)/(sum(a))^2` for non-negative sequences. The key insight is that this is a convex optimization problem (max of convex quadratics over a simplex), so I can use fast projected gradient methods with O(n log n) per step via FFT-based convolution.

My improvements over the previous approach:
1. **Weighted Spectral Descent (WSD)**: Replace slow LP oracle with fast FFT-based gradient steps
2. **Nesterov acceleration** for faster convergence
3. **Adaptive top-K peak selection** to focus on the most important lags
4. **Better time allocation** with more iterations per second
5. **Multiple weight strategies** to avoid getting stuck

"""
Optimized search for minimum peak autocorrelation step function.
Uses Weighted Spectral Descent (WSD) with Nesterov acceleration,
adaptive top-K peak targeting, multi-n exploration, and local search.
Key advantage: O(n log n) per iteration vs O(n^2)+ LP in previous approach.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_topk(a, conv, K):
    """Compute subgradient of max_k conv[k] using top-K peaks.
    grad[j] = (1/K) * sum_{k in S} [a[k-j]*I(0<=k-j<n) + a[j-k]*I(0<=j-k<n)]
    """
    n = len(a)
    top_k = np.argsort(conv)[::-1][:K]
    grad = np.zeros(n)
    for k in top_k:
        k = int(k)
        # a[k-j] for valid j: j in [max(0, k-n+1), min(k, n-1)]
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            grad[j_start:j_end + 1] += a[k - j_end:k - j_start + 1][::-1]
        # a[j-k] for valid j: j in [max(k, 0), min(k+n-1, n-1)]
        j_start2 = max(k, 0)
        j_end2 = min(k + n - 1, n - 1)
        if j_start2 <= j_end2:
            grad[j_start2:j_end2 + 1] += a[0:j_end2 - j_start2 + 1]
    grad /= K
    return grad


def compute_hankel_vec(a, w):
    """Compute W*a where W[i,j] = w[i+j] using FFT.
    (W*a)[i] = sum_j w[i+j] * a[j]
    Using: (W*a)[i] = (conv(w, a_flip))[i + n - 1]
    """
    n = len(a)
    a_flip = a[::-1].copy()
    c = fft_convolve(w, a_flip)
    # c has length (2n-1) + n - 1 = 3n-2
    # We need c[n-1 : 2n-1]
    result = c[n - 1:2 * n - 1]
    return result


def wsd_optimize(a0, time_limit, T_init=0.01, lr_init=0.5, K_init=5,

    K = K_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    # Nesterov state
    x_prev = a.copy()
    t_nest = 1.0

    # Estimate Lipschitz constant periodically
    L_estimate = 2.0 * n  # initial guess: max eigenvalue of W is at most sum(w)*n

    while time.monotonic() - t0 < time Limit:
        it += 1

            break

        conv = fast_convolve(a)

            beta = (t_nest - 1.0) / t_new
            y = a + beta * (a - x_prev)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy

        else:

        max_c = conv_y.max()

            scaled -= scaled.max()
            w = np.exp(scaled)
            w = w / w.sum()
            grad_w = compute_hankel_vec(y, w)

            if no_improve > 300:
                # Random restart

                a = project_simplex(best_a + noise)
                x_prev = a.copy()
                t_nest = 1.0
                T = T_init
                lr = lr_init
                K = K_init
                no_improve = 0

        # Periodic Lipschitz update
        if it % 100 == 0:
            # Estimate via power iteration (1 step)
            v = rng.standard_normal(n)
            v = v / np.linalg.norm(v)
            for _ in range(3):

def fw_lp_optimize(a0, time_limit, max_iters=200, seed=42):
    """Frank-Wolfe with LP oracle (limited iterations for speed)."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    # Build sparse matrix
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    row_idx = np.array(rows)
    col_idx = np.array(cols)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        c = -np.ones(n)
        vals = np.empty(len(row_idx))
        for idx in range(len(row_idx)):
            k = row_idx[idx]
            i = col_idx[idx]
            vals[idx] = f[k - i]

        A_mat = lil_matrix((2 * n - 1, n))
        A_mat[row_idx, col_idx] = vals
        A_mat = A_mat.tocsr()
        b = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(remaining * 0.3, 1.0)})
            if res.success and res.fun is not None:
                g = np.maximum(res.x, 0.0)
                sg = g.sum()
                if sg > 1e-15:
                    g = g / sg
                else:
                    g = None
            else:
                g = None
        except Exception:
            g = None

        if g is None:
            m_exp = min(m_exp * 1.3, 5.0)
            if m_exp >= 5.0:
                g = project_simplex(f + rng.standard_normal(n) * 0.005)
                m_exp = 1.0
            else:
                continue

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

    return best_f, best_obj


def local_search(a0, time_limit, seed=42):
    """Local search with various perturbation strategies."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        improved = False

        # Random perturbations at multiple scales
        for scale in [0.0002, 0.0005, 0.001, 0.003, 0.008]:
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(a + pert)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        # Coordinate moves
        if not improved and it % 2 == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                for frac in [0.05, 0.1, 0.2, 0.4]:
                    candidate = a.copy()
                    amount = candidate[i] * frac
                    candidate[i] -= amount
                    candidate[j] += amount
                    candidate = np.maximum(candidate, 0)
                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break

        # Structured perturbation
        if not improved and it % 5 == 0:
            strategy = int(rng.integers(0, 4))
            if strategy == 0:
                # Sinusoidal
                freq = rng.uniform(0.5, 10)
                phase = rng.uniform(0, 2 * np.pi)
                pattern = np.sin(2 * np.pi * freq * np.arange(n) / n + phase) * 0.003
            elif strategy == 1:
                # Triangle wave
                x = np.linspace(0, 2, n)
                pattern = (2 * np.abs(x - 1) - 1) * 0.003
            elif strategy == 2:
                # Edge boost
                decay = np.exp(-np.minimum(np.arange(n), n - 1 - np.arange(n)) * 0.05)
                pattern = (decay - 1) * 0.002
            else:
                # Random block
                block_size = int(rng.integers(n // 10, n // 3))
                start = int(rng.integers(0, n - block_size))
                pattern = np.zeros(n)
                pattern[start:start + block_size] = rng.uniform(-0.003, 0.003)
            candidate = project_simplex(a + pattern)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True

        if not improved:
            stagnation += 1
        else:
            stagnation = 0

        if stagnation > 500:
            break

    return best_a, best_obj


def parametric_init(n, strategy=0, seed=42):
    """Generate parametric initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    if strategy == 0:
        # Uniform
        a = np.ones(n)
    elif strategy == 1:
        # Edge-concentrated
        dist = np.minimum(x, n - 1 - x)
        a = np.exp(-dist * 2.0 / n)
    elif strategy == 2:
        # Two peaks at 1/4 and 3/4
        a = np.exp(-0.5 * ((x - n * 0.25) / (n * 0.1)) ** 2) + \
            np.exp(-0.5 * ((x - n * 0.75) / (n * 0.1)) ** 2)
    elif strategy == 3:
        # Exponential decay from left
        a = np.exp(-x * 3.0 / n)
    elif strategy == 4:
        # Sine envelope
        a = np.abs(np.sin(np.pi * x / n)) ** 0.5
    elif strategy == 5:
        # Random smooth
        a = np.abs(rng.standard_normal(n)).cumsum()
        a = a[::-1]
    else:
        # Power law from edges
        d = np.minimum(x, n - 1 - x) + 1
        a = 1.0 / d ** 0.3

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def try_seq(a, best_seq, best_score, label=""):
    """Evaluate a sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return a_norm.copy(), score
    return best_seq, best_score
