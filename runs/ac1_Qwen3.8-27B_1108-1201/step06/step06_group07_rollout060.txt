import numpy as np
from scipy.special import softmax
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient(a, b, temperature=0.01):
    """
    Compute the (smooth) subgradient of max(b) w.r.t. a using softmax weights.
    grad[j] = sum_k p_k * 2*a[k-j] for valid (k,j).
    """
    n = len(a)
    # Softmax weights on the auto-convolution values
    p = softmax(b / temperature)
    
    # grad[j] = 2 * sum_k p[k] * a[k-j] for valid k-j
    # This is 2 * (p reversed) convolved with a, but we need to be careful
    # (p_rev * a)[j] = sum_k p[k] * a[k-j] = sum_m p[m] * a[j-m+... ]
    # Actually: grad[j] = 2 * sum_k p[k] * a[k-j] * [0<=k-j<n]
    # Let m = k-j, so k = j+m: grad[j] = 2 * sum_m p[j+m] * a[m] * [0<=m<n, 0<=j+m<2n-1]
    
    # Use FFT for efficiency
    # grad[j] = 2 * sum_{m=0}^{n-1} a[m] * p[j+m] (where p is defined on [0, 2n-2])
    # This is a cross-correlation-like operation
    # grad = 2 * ifft(conj(fft(a)) * fft(p, padded)) ... let me think
    
    # Actually, let's just do it directly with the FFT approach:
    # Define q[m] = p[m] for m in [0, 2n-2], and we want:
    # grad[j] = 2 * sum_m a[m] * p[j+m] for m in [0,n-1], j+m in [0, 2n-2]
    # = 2 * (a * p_pad)[reversed]  ... 
    # Let me use a different approach: cross-correlation
    # cross[j] = sum_m a[m] * p[j+m]
    # This is the cross-correlation of a and p, evaluated at lag j
    # = FFT approach: cross = ifft(fft(p, 2*max_len) * conj(fft(a, 2*max_len)))
    
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    # cross-correlation: ifft(P * conj(A)) gives sum_m p[m] * conj(a[m-j+...])
    # Actually: ifft(fft(p)*conj(fft(a)))[k] = sum_m p[m] * a[m-k]  (cross-correlation)
    # We want sum_m a[m] * p[j+m] = sum_m a[m] * p[m+j]
    # Let n' = m: = sum_m a[m] * p[m+j]
    # cross_corr[j] = sum_m p[m+j] * a[m] = (p * a mirrored) 
    # This is the convolution of p with reversed a, shifted...
    
    # Simpler: just use the definition directly with numpy
    # For each j, grad[j] = 2 * sum_{m: 0<=m<n, 0<=j+m<2n-1} a[m] * p[j+m]
    # We can vectorize: for a given j, the valid m range is [max(0, -(j)), min(n-1, 2n-2-j)]
    # = [0, min(n-1, 2n-2-j)] since j >= 0
    
    # Use FFT: define A_full and P_full
    # cross[j] = sum_m a[m] * p[m+j] = (a_cross_corr_p)[j]
    # In FFT terms: cross = ifft(FFT(a) * conj(FFT(p)))  ... no
    # ifft(FFT(x) * conj(FFT(y)))[n] = sum_k x[k] * conj(y[k-n])  (for real: sum x[k]*y[k-n])
    # So ifft(FFT(a) * conj(FFT(p)))[j] = sum_m a[m] * p[m-j]
    # We want sum_m a[m] * p[m+j] = sum_m a[m] * p[m-(-j)]
    # = ifft(FFT(a) * conj(FFT(p)))[-j] = ifft(FFT(a) * conj(FFT(p)))[fft_size - j]
    
    cross = np.fft.irfft(A * np.conj(P), fft_size)
    # cross[j] = sum_m a[m] * p[m-j] (circular, but with enough padding it's linear)
    # We want grad[j] = 2 * sum_m a[m] * p[m+j] = 2 * cross[-j] = 2 * cross[fft_size - j]
    # But we need to be careful with the linear vs circular convolution
    
    # Actually with fft_size >= 3n-1, the linear cross-correlation is embedded:
    # cross[j] for j in [0, fft_size) gives the circular cross-correlation
    # For linear: cross[j] = sum_m a[m]*p[m-j] for j in range, valid when indices are in range
    # p has length 2n-1, a has length n
    # cross[j] = sum_{m: 0<=m<n, 0<=m-j<2n-1} a[m]*p[m-j]
    # We want: sum_{m: 0<=m<n, 0<=m+j<2n-1} a[m]*p[m+j]
    # = cross[-j] in the linear sense
    # In the circular array: cross[fft_size - j] (for j > 0)
    
    # Let me just verify: for j=0, we want sum_m a[m]*p[m] = cross[0] ✓
    # For j=1, we want sum_m a[m]*p[m+1]. In the cross array, cross[fft_size-1] = sum_m a[m]*p[m-1+fft_size]
    # Due to circularity, p[m-1+fft_size] = p[m-1] (if m-1 >= 0) ... this is getting confusing.
    
    # Let me just do it the straightforward way with a loop-free approach:
    # grad[j] = 2 * sum_{m=0}^{min(n-1, 2n-2-j)} a[m] * p[j+m]
    # Create a matrix: M[j,m] = p[j+m] for valid (j,m), then grad = 2 * (M @ a)
    # But this is O(n^2) memory. For n=2048, that's 4M floats = 32MB. OK.
    # Actually for larger n this could be problematic. Let me use the FFT approach properly.
    
    # The correct FFT approach:
    # We want c[j] = sum_m a[m] * p[m+j] for j = 0, 1, ..., 2n-2
    # (with a[m]=0 for m>=n and p[k]=0 for k<0 or k>=2n-1)
    # Define A_padded = [a, 0, ..., 0] (length fft_size)
    # Define P_padded = [0, 0, ..., 0, p, 0, ..., 0] ... 
    # Actually, c[j] = sum_m A[m] * P[m+j] = cross-correlation at lag j
    # = ifft(FFT(A) * conj(FFT(P))) evaluated appropriately
    # In numpy: np.correlate(p, a, mode='full') gives sum_m a[m] * p[m+k] for various k
    # But that's O(n^2). For n up to 2048, it's 4M ops which is fine.
    
    # Let me use a cleaner FFT formulation:
    # c[j] = sum_m a[m] * p[m+j]
    # Let's flip a: a_flip[k] = a[n-1-k]
    # Then c[j] = sum_m a[n-1-(n-1-m)] * p[m+j] = sum_m a_flip[n-1-m] * p[m+j]
    # = (a_flip_rev * p)[n-1+j] ... I'm overcomplicating this.
    
    # Just use the direct approach - it's O(n * (2n)) = O(n^2) which is fine for n <= 4096
    # Vectorized: for each j, we need a[0:min(n, 2n-1-j)] dot p[j:j+min(n, 2n-1-j)]
    # This is a series of dot products of decreasing length.
    
    # Alternative: use the fact that this is a convolution
    # c = sum_m a[m] * p[m+j] = (a * p_reversed)[n-1+j] ... 
    # Let p_rev[k] = p[2n-2-k]. Then (a * p_rev)[k] = sum_m a[m] * p_rev[k-m] = sum_m a[m] * p[2n-2-k+m]
    # Set k = 2n-2-j: sum_m a[m] * p[2n-2-(2n-2-j)+m] = sum_m a[m] * p[j+m] = c[j] ✓
    # So c[j] = (a * p_rev)[2n-2-j]
    
    p_rev = p[::-1]
    conv = np.convolve(a, p_rev)[:2*n-1]  # length 2n-1 + n - 1... no
    # np.convolve(a, p_rev) has length n + (2n-1) - 1 = 3n-2
    # We want conv[2n-2-j] for j = 0, ..., 2n-2
    # conv length is 3n-2, indices 0 to 3n-3
    # 2n-2-j ranges from 2n-2 (j=0) to 0 (j=2n-2)
    # All within [0, 3n-3] ✓
    
    c = conv[2*n-2::-1]  # c[j] = conv[2n-2-j] for j=0,...,2n-2
    # c has length 2n-1, c[j] = sum_m a[m]*p[m+j]
    
    # Now grad[j] = 2 * c[j] for j in [0, n-1]
    grad = 2.0 * c[:n]
    return grad, p


def nesterov_subgradient(a0, time_limit, init_temp=0.005, verbose=False):
    """
    Nesterov-accelerated subgradient method with adaptive temperature.
    Minimizes max(b) where b = a*a, subject to sum(a)=1, a>=0.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    a_y = a.copy()  # Nesterov lookahead
    a_prev = a.copy()
    beta = 1.0  # momentum parameter (will decay)
    it = 0
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    temp = init_temp
    lr = 0.02
    
    t0 = time.monotonic()
    no_improve = 0
    epoch = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Evaluate at lookahead point
        a_eval = np.maximum(a_y, 0.0)
        s_eval = a_eval.sum()
        if s_eval < 1e-15:
            a_eval = a.copy()
            s_eval = s
        else:
            a_eval = a_eval / s_eval
        
        b = fast_convolve(a_eval)
        obj = b.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_eval.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute gradient
        grad, p = compute_subgradient(a_eval, b, temperature=temp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size (decreasing)
        lr_t = lr / (1.0 + 0.01 * it)
        
        # Nesterov update
        a_new = a - lr_t * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
            s_new = s
        else:
            a_new = a_new / s_new
        
        # Update momentum
        beta = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_y = a_new + (beta - 1.0) / (beta + 1.0) * (a_new - a_prev)
        a_y = np.maximum(a_y, 0.0)
        s_y = a_y.sum()
        if s_y > 1e-15:
            a_y = a_y / s_y
        a_prev = a.copy()
        a = a_new
        
        # Adapt temperature
        if no_improve > 30:
            temp *= 1.1  # Increase temperature (smoother)
            if temp > 0.5:
                temp = 0.005  # Reset
        elif no_improve < 5:
            temp = max(0.001, temp * 0.95)  # Decrease (sharper)
        
        # Periodic restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng((it * 7919) % (2**31 - 1))
            scale = min(0.05, 0.005 * np.sqrt(no_improve / 200.0))
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            a_y = a.copy()
            a_prev = a.copy()
            beta = 1.0
            temp = init_temp
            no_improve = 0
            epoch += 1
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NS it={it}, obj={best_obj:.10f}, temp={temp:.4f}, lr={lr_t:.5f}, t={elapsed:.0f}s")
    
    return best_a, best_obj


def coord_descent(a0, time_limit, sweeps_target=500, verbose=False):
    """
    Cyclic coordinate descent with normalization.
    For each coordinate j, find the best value of a[j] (keeping sum=1)
    by trying to reduce the max auto-convolution.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    sweep = 0
    
    while time.monotonic() - t0 < time_limit and sweep < sweeps_target:
        sweep += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            
            old_val = a[j]
            if old_val < 1e-15 and j % 10 == 0:
                # Try adding a small amount
                candidates = [old_val + 0.001 / n]
            else:
                # Try reducing and slightly increasing
                candidates = [old_val * 0.9, old_val * 0.95, old_val * 1.05, old_val * 1.1,
                              old_val * 0.7, old_val * 1.3]
                # Also try 0
                candidates.append(0.0)
            
            best_candidate = old_val
            b_current = fast_convolve(a)
            max_current = b_current.max()
            
            for cand in candidates:
                cand = max(0.0, cand)
                a[j] = cand
                s_new = a.sum()
                if s_new < 1e-15:
                    a[j] = old_val
                    continue
                a_norm = a / s_new
                max_new = fast_convolve(a_norm).max()
                if max_new < max_current - 1e-15:
                    max_current = max_new
                    best_candidate = cand
                    improved = True
            
            a[j] = best_candidate
            s_final = a.sum()
            if s_final > 1e-15:
                a = a / s_final
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
        else:
            if sweep > 20:
                break  # Converged
        
        if verbose and sweep % 50 == 0:
            print(f"      CD sweep={sweep}, obj={best_obj:.10f}, t={time.monotonic()-t0:.0f}s")
    
    return best_a, best_obj


def peak_redistribute(a0, time_limit, verbose=False):
    """
    Directly target the highest peak: redistribute mass from positions
    that contribute to it toward positions that don't.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        b = fast_convolve(a)
        k_peak = int(np.argmax(b))
        peak_val = b[k_peak]
        
        if peak_val <= best_obj * 1.001:
            if no_improve > 50:
                break
            no_improve += 1
            # Try second peak
            b_copy = b.copy()
            b_copy[k_peak] = 0
            k_peak2 = int(np.argmax(b_copy))
            if b_copy[k_peak2] > 0:
                k_peak = k_peak2
                peak_val = b_copy[k_peak2]
            else:
                break
        
        # Find contributing positions
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        
        # Contribution of each position j to the peak
        contrib = np.zeros(n)
        for j in range(js, je + 1):
            contrib[j] = a[j] * a[k_peak - j]
        
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break
        
        # Try different fractions of redistribution
        best_frac = 0
        best_new_obj = best_obj
        for frac in [0.02, 0.05, 0.1, 0.2, 0.3]:
            a_trial = a.copy()
            # Remove mass proportional to contribution
            for j in range(n):
                if contrib[j] > 0:
                    a_trial[j] -= frac * a_trial[j] * (contrib[j] / total_contrib)
            
            # Redistribute to positions outside the peak's influence
            # Try adding to positions that minimize the new max
            outside = [j for j in range(n) if j < js or j > je]
            if not outside:
                outside = [j for j in range(n) if j != js and j != je]
            
            if outside:
                total_removed = a.sum() - a_trial.sum()
                if total_removed > 1e-15:
                    # Distribute uniformly to outside positions
                    for j in outside:
                        a_trial[j] += total_removed / len(outside)
            
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            new_obj = fast_convolve(a_trial).max()
            if new_obj < best_new_obj:
                best_new_obj = new_obj
                best_frac = frac
                best_trial = a_trial.copy()
        
        if best_frac > 0 and best_new_obj < best_obj - 1e-14:
            a = best_trial
            best_obj = best_new_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng((it * 9973 + 5) % (2**31 - 1))
                a = best_a + rng.standard_normal(n) * 0.003
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
    
    return best_a, best_obj


def generate_structured_inits(n, parent, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    t = x / max(n - 1, 1)
    
    # Parent interpolation
    if parent is not None:
        px = np.interp(t, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Beta-like distributions (varying concentration)
    for alpha in [0.3, 0.5, 1.0, 2.0, 5.0]:
        w = (t + 1e-10) ** (alpha - 1) * (1 - t + 1e-10) ** (alpha - 1)
        w = w / w.sum()
        inits.append(w)
    
    # Edge-heavy
    edge_w = 1.0 / (np.minimum(x, n - 1 - x).astype(float) + 1.0) ** 1.2
    inits.append(edge_w / edge_w.sum())
    
    # Cosine bell (reduces center peak)
    cos_w = np.cos(np.pi * t) ** 2 + 0.01
    inits.append(cos_w / cos_w.sum())
    
    # Two peaks
    g2 = np.exp(-((t - 0.2) ** 2) / 0.02) + np.exp(-((t - 0.8) ** 2) / 0.02) + 0.01
    inits.append(g2 / g2.sum())
    
    # Three peaks
    g3 = np.exp(-((t - 0.15) ** 2) / 0.01) + np.exp(-((t - 0.5) ** 2) / 0.03) + np.exp(-((t - 0.85) ** 2) / 0.01) + 0.01
    inits.append(g3 / g3.sum())
    
    # Sawtooth-like
    saw = (1.0 + np.sin(2 * np.pi * 3 * t)) ** 2 + 0.1
    inits.append(saw / saw.sum())
    
    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))
    
    # Perturbed parent
    if parent is not None:
        px = np.interp(t, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px + rng.standard_normal(n) * 0.02, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits

    if best_seq is not None:
        # Try multiple temperatures
        for temp in [0.003, 0.005, 0.01, 0.02]:
            remaining = deadline - time.monotonic()
            if remaining < 200:
                break
            t_alloc = min(80, remaining - 180)
            if t_alloc < 10:
                break
            print(f"  Deep NS temp={temp}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_subgradient(best_seq, time_limit=t_alloc, init_temp=temp)
            try_seq(a_opt, f"deep_ns_t{temp}")
    
    print(f"  P2 done: best={best_score:.8f}, t={time.monotonic()-t2:.0f}s")
    
    # === Phase 3: Coordinate Descent polishing (300s) ===
    print("\n=== Phase 3: Coordinate Descent ===")
    t3 = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()
        if remaining > 100:
            t_alloc = min(120, remaining - 100)
            print(f"  CD n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = coord_descent(best_seq, time_limit=t_alloc, sweeps_target=200)
            try_seq(a_opt, "cd")
            
            # Second pass
            remaining = deadline - time.monotonic()
            if remaining > 60:
                t_alloc = min(60, remaining - 50)
                a_opt2, _ = coord_descent(a_opt, time_limit=t_alloc, sweeps_target=100)
                try_seq(a_opt2, "cd2")
    
    print(f"  P3 done: best={best_score:.8f}, t={time.monotonic()-t3:.0f}s")
    
    # === Phase 4: Peak Redistribution (200s) ===
    print("\n=== Phase 4: Peak Redistribution ===")
    t4 = time.monotonic()
    
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 80:
            t_alloc = min(100, remaining - 80)
            print(f"  PeakRedist t={t_alloc:.0f}s")
            a_opt, _ = peak_redistribute(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_redist")
            
            # Follow up with Nesterov
            remaining = deadline - time.monotonic()
            if remaining > 40:
                t_alloc = min(40, remaining - 30)
                a_opt2, _ = nesterov_subgradient(a_opt, time_limit=t_alloc, init_temp=0.003)
                try_seq(a_opt2, "after_peak_ns")
    
    print(f"  P4 done: best={best_score:.8f}, t={time.monotonic()-t4:.0f}s")
    
    # === Phase 5: Diversified restarts (200s) ===
    print("\n=== Phase 5: Diversified Restarts ===")
    t5 = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        # Try different structured inits at the best n
        inits = generate_structured_inits(n_best, best_seq, rng)
        for i, init in enumerate(inits):
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            t_alloc = min(20, remaining - 40)
            if t_alloc < 5:
                break
            print(f"  Restart {i}/{len(inits)}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_subgradient(init, time_limit=t_alloc, init_temp=0.01)
            try_seq(a_opt, f"restart_{i}")
    
    print(f"  P5 done: best={best_score:.8f}, t={time.monotonic()-t5:.0f}s")
    
    # === Phase 6: Try other n values with best shape (150s) ===
    print("\n=== Phase 6: Multi-n Refinement ===")
    t6 = time.monotonic()
    
    if best_seq is not None:
        for n_try in [int(best_seq.size * 0.75), int(best_seq.size * 1.25), int(best_seq.size * 1.5)]:
            n_try = max(256, min(3072, n_try))
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            if n_try == len(best_seq):
                continue
            interp = np.interp(np.linspace(0, 1, n_try),
                               np.linspace(0, 1, len(best_seq)), best_seq)
            interp = np.maximum(interp, 0)
            interp = interp / interp.sum()
            t_alloc = min(30, remaining - 30)
            if t_alloc < 5:
                break
            print(f"  Multi-n n={n_try}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_subgradient(interp, time_limit=t_alloc, init_temp=0.005)
            try_seq(a_opt, f"multin_{n_try}")
    
    print(f"  P6 done: best={best_score:.8f}, t={time.monotonic()-t6:.0f}s")
    
    # === Phase 7: Final polish (remaining time) ===
    print("\n=== Phase 7: Final Polish ===")
    t7 = time.monotonic()
    
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t_alloc = min(30, remaining - 10)
            print(f"  Final NS t={t_alloc:.0f}s")
            a_opt, _ = nesterov_subgradient(best_seq, time_limit=t_alloc, init_temp=0.002)
            try_seq(a_opt, "final_ns")
        
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_alloc = min(20, remaining - 5)
            a_opt, _ = coord_descent(best_seq, time_limit=t_alloc, sweeps_target=50)
            try_seq(a_opt, "final_cd")
        
        remaining = deadline - time.monotonic()
        if remaining > 10:
            t_alloc = min(10, remaining - 3)
            a_opt, _ = peak_redistribute(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_peak")
    
    print(f"  P7 done: best={best_score:.8f}, t={time.monotonic()-t7:.0f}s")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
