"""
Hybrid optimizer for minimum peak autocorrelation step function.
Uses Hankel matrix power iteration, log-sum-exp gradient descent,
Nesterov-accelerated Frank-Wolfe with LP oracle, and local search.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return np.array([])
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_matvec(p, v):
    """Compute H_p @ v where H_p[j,m] = p[j+m] (Hankel matrix).
    p has length 2n-1, v has length n.
    Result has length n: result[j] = sum_m p[j+m] * v[m]
    """
    n = len(v)
    result = np.zeros(n)
    for j in range(n):
        end = min(j + n, len(p))
        if j < len(p):
            result[j] = np.dot(v[:end - j], p[j:end])
    return result


def hankel_power_method(p, v_init, n_power_iter=30):
    """Find top eigenvector of Hankel matrix H_p using power method.
    Returns non-negative unit vector (by Perron-Frobenius).
    """
    v = v_init.astype(np.float64).copy()
    vn = np.linalg.norm(v)
    if vn < 1e-15:
        v = np.ones(len(v)) / np.sqrt(len(v))
    else:
        v = v / vn

    for _ in range(n_power_iter):
        v = hankel_matvec(p, v)
        # Ensure non-negativity (should hold by PF but numerical issues)
        v = np.maximum(v, 0.0)
        vn = np.linalg.norm(v)
        if vn < 1e-15:
            v = np.ones(len(v)) / np.sqrt(len(v))
        else:
            v = v / vn
    return v


def hankel_power_optimize(a0, time_limit, T_start=0.01, T_end=0.001,
                          n_power=30, blend_alpha=1.0):
    """Hankel power iteration: alternate p=softmax(conv/T) and eigenvector of H_p."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 0.5:
            break

        # Temperature schedule: decrease over time
        frac = min(elapsed / time_limit, 1.0)
        T = T_start * (T_end / T_start) ** frac

        # Compute convolution
        conv = fast_convolve(a)

        # Softmax weights
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros_like(conv)
            p[conv.argmax()] = 1.0
        else:
            p = p / p_sum

        # Power iteration on Hankel
        v = hankel_power_method(p, a, n_power_iter=n_power)
        v_sum = v.sum()
        if v_sum < 1e-15:
            v = a.copy()
        else:
            v = v / v_sum

        # Blend
        a_new = (1.0 - blend_alpha) * a + blend_alpha * v
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / ss

        a = a_new
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt
        if no_improve > 100:
            blend_alpha = max(blend_alpha * 0.9, 0.1)
        if no_improve > 300:
            blend_alpha = 1.0
            # Random perturbation
            rng = np.random.default_rng((it * 104729 + 7) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0

        if it % 500 == 0:
            print(f"    Hankel it={it}, obj={obj:.8f}, best={best_obj:.8f}, T={T:.5f}, alpha={blend_alpha:.3f}")

    return best_a, best_obj


def compute_lse_gradient(a, p, n):
    """Compute gradient of log-sum-exp objective.
    grad[j] = 2 * sum_m a[m] * p[m+j]
    """
    grad = np.zeros(n)
    plen = len(p)
    for j in range(n):
        end = min(j + n, plen)
        if j < plen:
            grad[j] = 2.0 * np.dot(a[:end - j], p[j:end])
    return grad


def lse_gradient_optimize(a0, time_limit, lambda_start=50.0, lambda_end=2000.0,
                          lr_init=0.005):
    """Log-sum-exp smoothed gradient descent on simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    lr = lr_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        if time_limit - elapsed < 0.3:
            break

        frac = min(elapsed / time_limit, 1.0)
        lam = lambda_start * (lambda_end / lambda_start) ** frac

        conv = fast_convolve(a)
        scaled = lam * conv
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros_like(conv)
            p[conv.argmax()] = 1.0
        else:
            p = p / p_sum

        grad = compute_lse_gradient(a, p, n)

        # Backtracking line search
        cur_obj = conv.max()
        best_candidate = a
        best_val = cur_obj
        grad_norm_sq = np.dot(grad, grad)

        trial_lr = lr
        for _ in range(10):
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            if c.max() < best_val - 1e-15:
                best_val = c.max()
                best_candidate = candidate
                if c.max() < cur_obj - 0.3 * trial_lr * grad_norm_sq:
                    break
            trial_lr *= 0.5

        a = best_candidate

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.05, 0.1)
        else:
            no_improve += 1
            lr *= 0.95

        if no_improve > 200:
            lr = lr_init * 0.5
            no_improve = 0
        if no_improve > 500:
            rng = np.random.default_rng((it * 31 + 7) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            lr = lr_init
            no_improve = 0

        if it % 1000 == 0:
            print(f"    LSE it={it}, obj={obj:.8f}, best={best_obj:.8f}, lam={lam:.1f}, lr={lr:.6f}")

    return best_a, best_obj


def prebuild_lp_pattern(n):
    """Pre-compute sparsity pattern for LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    rows = np.array(rows)
    cols = np.array(cols)
    f_idx = rows - cols  # index into f for each matrix entry
    return rows, cols, f_idx


def solve_lp_oracle(f, M, rows, cols, f_idx):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    vals = f[f_idx]

    A = lil_matrix((2 * n - 1, n))
    A[rows, cols] = vals
    A = A.tocsr()

    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def fw_lp_optimize(a0, time_limit, rows=None, cols=None, f_idx=None):
    """Nesterov-accelerated Frank-Wolfe with LP oracle."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    if rows is None:
        rows, cols, f_idx = prebuild_lp_pattern(n)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_f = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    m_exp = 1.0
    prev_f = a.copy()

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        # Nesterov lookahead
        gamma = 2.0 / (it + 2.0)
        if it > 1:
            prev_gamma = 2.0 / (it + 1.0)
            beta = (1.0 - gamma) / prev_gamma
            x_lookahead = a + max(beta, 0.0) * (a - prev_f)
            x_lookahead = project_simplex(x_lookahead)
        else:
            x_lookahead = a.copy()

        # Solve LP oracle at lookahead point
        g = solve_lp_oracle(x_lookahead, M * m_exp, rows, cols, f_idx)
        if g is None:
            m_exp = min(m_exp * 1.5, 10.0)
            if m_exp >= 10.0:
                rng = np.random.default_rng(it * 31 + 7)
                g = project_simplex(a + rng.standard_normal(n) * 0.005)
                m_exp = 1.0
            else:
                continue

        if np.max(np.abs(g - a)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                rng = np.random.default_rng(it * 7919 + 13)
                a = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                conv = fast_convolve(a)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
                prev_f = a.copy()
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = (1.0 - alpha) * a + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Ternary refinement
        if 0 < best_alpha < 1:
            lo = max(0.0, best_alpha - 0.15)
            hi = min(1.0, best_alpha + 0.15)
            for _ in range(10):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.maximum((1 - m1) * a + m1 * g, 0)
                f1 = f1 / f1.sum()
                v1 = fast_convolve(f1).max()
                f2 = np.maximum((1 - m2) * a + m2 * g, 0)
                f2 = f2 / f2.sum()
                v2 = fast_convolve(f2).max()
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            mid = (lo + hi) / 2.0
            ft = np.maximum((1 - mid) * a + mid * g, 0)
            ft = ft / ft.sum()
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = mid

        if best_alpha > 1e-10:
            prev_f = a.copy()
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.2, 10.0)

        if it % 200 == 0:
            print(f"    FW-LP it={it}, M={M:.8f}, best={best_obj:.8f}, m_exp={m_exp:.2f}")

    return best_f, best_obj


def local_search_optimize(a0, time_limit, seed=42):
    """Local search with random perturbations and coordinate moves."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time_limit - (time.monotonic() - t0) < 0.1:
            break

        improved = False

        for scale in [0.0002, 0.0005, 0.001, 0.003, 0.008]:
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(a + pert)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-16:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        if not improved and it % 3 == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j and a[i] > 1e-10:
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    candidate = a.copy()
                    amount = candidate[i] * frac
                    candidate[i] -= amount
                    candidate[j] += amount
                    candidate = np.maximum(candidate, 0)
                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-16:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break

        if not improved and it > 5000:
            break

    return best_a, best_obj


def try_seq(a, label, best_ref):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_ref
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_ref[1]:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return (a_norm.copy(), score)
    return best_ref


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point: multi-phase hybrid search.
    Phase 1: Quick multi-n exploration with Hankel power method
    Phase 2: Deep Hankel power + LSE gradient on best n
    Phase 3: FW-LP refinement
    Phase 4: Local search polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Using uniform init: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Quick multi-n exploration ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = [384, 512, 640, 768, 896, 1024, 1280]
    if n_parent not in candidate_ns:
        candidate_ns.append(n_parent)
    candidate_ns = sorted(set(n for n in candidate_ns if 128 <= n <= 2048))

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic()
        if remaining < 200:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try

        # Quick Hankel power
        t_alloc = min(remaining * 0.05, 30)
        a_opt, _ = hankel_power_optimize(start, time_limit=t_alloc,
                                         T_start=0.02, T_end=0.005, n_power=20)
        best_seq, best_score = try_seq(a_opt, f"hankel_n{n_try}", (best_seq, best_score))

        # Quick LSE gradient
        remaining2 = deadline - time.monotonic()
        if remaining2 > 150:
            a_opt2, _ = lse_gradient_optimize(a_opt, time_limit=min(remaining2 * 0.03, 15),
                                              lambda_start=50, lambda_end=500, lr_init=0.005)
            best_seq, best_score = try_seq(a_opt2, f"lse_n{n_try}", (best_seq, best_score))

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep optimization on best n ===
    print("\n=== Phase 2: Deep Optimization ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 200:
        # Deep Hankel power with decreasing temperature
        t_alloc = min(remaining * 0.4, 500)
        print(f"  Deep Hankel n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = hankel_power_optimize(best_seq, time_limit=t_alloc,
                                         T_start=0.01, T_end=0.0005,
                                         n_power=40, blend_alpha=1.0)
        best_seq, best_score = try_seq(a_opt, "deep_hankel", (best_seq, best_score))

    # LSE gradient with increasing lambda
    remaining = deadline - time.monotonic()
    if remaining > 150:
        t_alloc = min(remaining * 0.3, 350)
        print(f"  Deep LSE n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = lse_gradient_optimize(best_seq, time_limit=t_alloc,
                                         lambda_start=100, lambda_end=3000, lr_init=0.003)
        best_seq, best_score = try_seq(a_opt, "deep_lse", (best_seq, best_score))

    # Another Hankel round with finer params
    remaining = deadline - time.monotonic()
    if remaining > 100:
        n_best = len(best_seq)
        t_alloc = min(remaining * 0.25, 200)
        print(f"  Hankel round 2, t={t_alloc:.0f}s")
        a_opt, _ = hankel_power_optimize(best_seq, time_limit=t_alloc,
                                         T_start=0.005, T_end=0.0003,
                                         n_power=50, blend_alpha=0.8)
        best_seq, best_score = try_seq(a_opt, "hankel_r2", (best_seq, best_score))

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: FW-LP refinement ===
    print("\n=== Phase 3: FW-LP Refinement ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 100:
        rows, cols, f_idx = prebuild_lp_pattern(n_best)
        t_alloc = min(remaining * 0.35, 400)
        print(f"  FW-LP n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc,
                                  rows=rows, cols=cols, f_idx=f_idx)
        best_seq, best_score = try_seq(a_opt, "fw_lp", (best_seq, best_score))

    # Perturbed FW-LP restart
    remaining = deadline - time.monotonic()
    if remaining > 100:
        n_best = len(best_seq)
        for scale in [0.003, 0.008, 0.02]:
            remaining = deadline - time.monotonic()
            if remaining < 80:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            rows, cols, f_idx = prebuild_lp_pattern(n_best)
            t_alloc2 = min(remaining * 0.2, 120)
            a_opt2, _ = fw_lp_optimize(f_pert, time_limit=t_alloc2,
                                       rows=rows, cols=cols, f_idx=f_idx)
            best_seq, best_score = try_seq(a_opt2, f"fw_pert_{scale}", (best_seq, best_score))

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Final polish ===
    print("\n=== Phase 4: Final Polish ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Local search
    if remaining > 50:
        t_alloc = min(remaining * 0.25, 100)
        print(f"  Local search, t={t_alloc:.0f}s")
        a_opt, _ = local_search_optimize(best_seq, time_limit=t_alloc, seed=12345)
        best_seq, best_score = try_seq(a_opt, "local", (best_seq, best_score))

    # Fine LSE
    remaining = deadline - time.monotonic()
    if remaining > 40:
        t_alloc = min(remaining * 0.2, 60)
        a_opt, _ = lse_gradient_optimize(best_seq, time_limit=t_alloc,
                                         lambda_start=500, lambda_end=5000, lr_init=0.001)
        best_seq, best_score = try_seq(a_opt, "fine_lse", (best_seq, best_score))

    # Final FW-LP
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_best = len(best_seq)
        rows, cols, f_idx = prebuild_lp_pattern(n_best)
        t_alloc = min(remaining * 0.3, 50)
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc,
                                  rows=rows, cols=cols, f_idx=f_idx)
        best_seq, best_score = try_seq(a_opt, "final_fw", (best_seq, best_score))

    # Tiny perturbations with quick LSE
    remaining = deadline - time.monotonic()
    if remaining > 20:
        for _ in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            noise = rng.standard_normal(n_best) * 0.0003
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = lse_gradient_optimize(f_pert, time_limit=min(remaining * 0.15, 8),
                                             lambda_start=1000, lambda_end=3000, lr_init=0.001)
            best_seq, best_score = try_seq(a_opt, "micro_perturb", (best_seq, best_score))

    # Another local search
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_opt, _ = local_search_optimize(best_seq, time_limit=min(remaining * 0.25, 25),
                                         seed=99999)
        best_seq, best_score = try_seq(a_opt, "local2", (best_seq, best_score))

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
