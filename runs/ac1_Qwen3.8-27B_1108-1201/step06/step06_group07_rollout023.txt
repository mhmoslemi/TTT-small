def _cross_corr(w, a):
    """Compute cross-correlation: result[j] = sum_m w[m+j] * a[m] for j=0..n-1"""
    n = len(a)
    # result[j] = conv(a[::-1], w)[j + n - 1]
    size = 2 * n - 1  # length of w
    fft_size = 1 << (size + n - 1 - 1).bit_length()  # need at least size + n - 1
    A = np.fft.rfft(a[::-1], fft_size)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Log-sum-exp (penalty) gradient descent on the simplex
- SCA with LP oracle (improved M adaptation)
- Coordinate descent refinement
- Multi-peak subgradient with softmax weighting
- Multi-n exploration with diverse initializations
- Diversified restarts with structured perturbations

import numpy as np
from scipy.optimize import linprog
from scipy.special import softmax
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_conv(a, b):
    """Linear cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    B = np.fft.rfft(b.astype(np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cross_corr_for_grad(w, a):
    """
    Compute grad[j] = sum_{m=0}^{n-1} w[m+j] * a[m] for j=0..n-1.
    This is the cross-correlation needed for the log-sum-exp gradient.
    """
    n = len(a)
    # result[j] = conv(a[::-1], w)[j + n - 1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a[::-1], fft_size)
    W = np.fft.rfft(w, fft_size)
    result = np.fft.irfft(A * W, fft_size)
    return result[n - 1: 2 * n - 1]


def logsumexp_descent(a0, time_limit, beta_init=20, beta_max=500, verbose=False):
    """
    Minimize max(conv(a,a)) via log-sum-exp penalty gradient descent on the simplex.
    """
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    beta = beta_init
    best_a = a.copy()
    best_maxconv = fast_convolve(a).max()

    t0 = time.monotonic()
    lr = 0.05
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)

        # Softmax weights (shifted for numerical stability)
        shifted = beta * (conv - conv.max())
        e = np.exp(shifted)
        e_sum = e.sum()
        if e_sum < 1e-300:
            p = np.zeros_like(conv)
            p[np.argmax(conv)] = 1.0
        else:
            p = e / e_sum

        # Gradient: dL/da[j] = 2 * sum_k p[k] * a[k-j] * I(valid)
        grad = 2.0 * cross_corr_for_grad(p, a)

        # Gradient step
        a_new = project_simplex(a - lr * grad)

        # Line search
        curr_max = conv.max()
        best_alpha = 0.0
        best_val = curr_max
        for alpha in [0.1, 0.3, 0.5, 0.8, 1.0]:
            cand = (1.0 - alpha) * a + alpha * a_new
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            a = (1.0 - best_alpha) * a + best_alpha * a_new
            a = np.maximum(a, 0)
            a = a / a.sum()

        maxconv = fast_convolve(a).max()
        if maxconv < best_maxconv - 1e-13:
            best_maxconv = maxconv
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt parameters
        if no_improve > 20:
            beta = min(beta * 1.12, beta_max)
            lr *= 0.96
        if lr < 0.0005:
            lr = 0.05
            beta = max(beta_init, beta * 0.4)
            no_improve = 0

        # Restart if stuck
        if no_improve > 150:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            no_improve = 0
            beta = beta_init
            lr = 0.05

        if verbose and it % 500 == 0:
            print(f"      LSE it={it}, maxconv={best_maxconv:.12f}, beta={beta:.1f}, obj={2*n*best_maxconv:.8f}")

    return best_a, 2.0 * n * best_maxconv


def build_lp_matrix(n):
    """Build sparse matrix for LP: A[k,j] = f[k-j] for valid pairs."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_oracle(f, M, n, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    k_arr = row_idx
    i_arr = col_idx
    vals = f[k_arr - i_arr]
    A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 10})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def subgradient_topk(a0, time_limit, top_k=8, tau_scale=0.003, verbose=False):
    """Subgradient method with softmax-weighted top-K peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_maxconv = fast_convolve(a).max()

    t0 = time.monotonic()
    lr = 0.04
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        curr_max = conv.max()

        # Top-K peaks with softmax weighting
        k_eff = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_eff)[-k_eff:]
        pv = conv[peaks]
        tau = tau_scale * curr_max + 1e-10
        pw = softmax(pv / tau)

        # Combined gradient
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            w = pw[idx]
            if w < 1e-12:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je + 1] += w * 2.0 * a[k - je:k - js + 1][::-1]

        # Direction
        direction = project_simplex(a - lr * grad) - a

        # Line search
        best_alpha = 0.0
        best_val = curr_max
        for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
            cand = a + alpha * direction
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue

            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            a = a + best_alpha * direction
            a = np.maximum(a, 0)
            a = a / a.sum()

        obj = fast_convolve(a).max()
        if obj < best_maxconv - 1e-13:
            best_maxconv = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.998

            no_improve = 0

            rng = np.random.default_rng((it * 1299709 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0
            lr = 0.04

        if verbose and it % 200 == 0:
            print(f"      SG it={it}, obj={best_maxconv:.12f}, lr={lr:.6f}")

    return best_a, 2.0 * n * best_maxconv


def coord_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent: adjust one element at a time to reduce max(conv).
    Uses analytical update of conv to avoid full recompute each step.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_maxconv = fast_convolve(a).max()

    t0 = time.monotonic()
    sweep = 0
    max_sweeps = 200

    while time.monotonic() - t0 < time_limit and sweep < max_sweeps:
        sweep += 1
        conv = fast_convolve(a)
        any_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            if a[j] < 1e-14:
                continue

            # For changing a[j] by delta:
            # new_conv[k] = conv[k] + 2*delta*a[j]*a[k-j] + delta^2*a[j]^2*I(k=2j)
            # (for valid k-j, i.e., k in [j, j+n-1])
            # Approximate (small delta): new_conv[k] ≈ conv[k] + 2*delta*a[j]*a[k-j]

            aj = a[j]
            k_start = j
            k_end = min(2 * n - 2, j + n - 1)

            # Sensitivity vector (added to conv[k] per unit delta)
            sens = 2.0 * aj * a[0:n]  # a[k-j] for k=j..j+n-1

            curr_max = conv.max()
            best_delta = 0.0
            best_new_max = curr_max

            for frac in [-0.04, -0.02, -0.01, -0.005, 0.005, 0.01, 0.02, 0.04]:
                delta = frac * aj
                if aj + delta < 0:
                    continue
                # New conv in the affected region
                mid = conv[k_start:k_end + 1] + delta * sens
                mid_max = mid.max()
                # Unaffected regions
                left_max = conv[0:k_start].max() if k_start > 0 else 0.0
                right_max = conv[k_end + 1:].max() if k_end + 1 < 2 * n - 1 else 0.0
                new_max = max(left_max, mid_max, right_max)

                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0:
                a[j] += best_delta
                a[j] = max(a[j], 0.0)
                conv = fast_convolve(a)
                if conv.max() < best_maxconv - 1e-14:
                    best_maxconv = conv.max()
                    best_a = a.copy()
                    any_improved = True

        if not any_improved:
            if sweep >= 20:
                break

        if verbose and sweep % 20 == 0:
            print(f"      CD sweep={sweep}, maxconv={best_maxconv:.12f}")

    return best_a, 2.0 * n * best_maxconv


def generate_structured_inits(n, parent, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # From parent (interpolated)
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge-heavy (power law)
    dist_edge = np.minimum(x, n - 1 - x).astype(float) + 1.0
    for p in [1.0, 1.5, 2.0, 3.0]:
        w = 1.0 / dist_edge ** p
        inits.append(w / w.sum())

    # Cosine window
    w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) + 0.05
    inits.append(w / w.sum())

    # Sine modulation
    for freq in [1, 2, 3]:
        w = (1 + 0.3 * np.sin(2 * np.pi * freq * x / n)) ** 2 + 0.1
        inits.append(w / w.sum())

    # Two-peak (Gaussian)
    g2 = (np.exp(-((x - 0.25 * (n - 1)) ** 2) / (2 * (0.08 * n) ** 2)) +
          np.exp(-((x - 0.75 * (n - 1)) ** 2) / (2 * (0.08 * n) ** 2))) + 0.05
    inits.append(g2 / g2.sum())

    # Three-peak
    g3 = (np.exp(-((x - 0.2*(n-1))**2) / (2*(0.06*n)**2)) +
          np.exp(-((x - 0.5*(n-1))**2) / (2*(0.06*n)**2)) +
          np.exp(-((x - 0.8*(n-1))**2) / (2*(0.06*n)**2))) + 0.05
    inits.append(g3 / g3.sum())

    # Dirichlet samples
    for alpha in [0.2, 0.5, 1.0, 3.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha))

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: hybrid optimization combining log-sum-exp gradient descent,
    SCA with LP oracle, coordinate descent, and multi-peak subgradient.
    Explores multiple n values and uses diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)

    # === Phase 1: Quick log-sum-exp across n values ===
    print("\n=== Phase 1: Multi-n LogSumExp ===")
    t1 = time.monotonic()

    ns_to_try = sorted(set([n_parent, 384, 512, 640, 768, 896, 1024, 1280, 1536]))
    ns_to_try = [n for n in ns_to_try if 256 <= n <= 2048]

    for n_try in ns_to_try:
        remaining_after = deadline - time.monotonic()
        if remaining_after < 300:
            break
        if n_try == n_parent and parent is not None:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            start = start / start.sum()

        t_alloc = min(35, remaining_after - 280)
        if t_alloc < 10:
            break
        print(f"  LSE n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = logsumexp_descent(start, time_limit=t_alloc)
        try_seq(a_opt, f"lse_n{n_try}")

    print(f"  P1 done: best={best_score:.8f}, t={time.monotonic()-t1:.0f}s")

    # === Phase 2: Deep SCA on best solution ===
    print("\n=== Phase 2: Deep SCA ===")
    t2 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 500
        if remaining > 30:
            t_alloc = min(remaining, 350)
            print(f"  SCA n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = sca_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "sca_deep")

            remaining = deadline - time.monotonic() - 300
            if remaining > 20:
                t_alloc2 = min(remaining, 120)
                a_opt2, _ = sca_optimize(a_opt, time_limit=t_alloc2)
                try_seq(a_opt2, "sca_deep2")

    print(f"  P2 done: best={best_score:.8f}, t={time.monotonic()-t2:.0f}s")

    # === Phase 3: Log-sum-exp refinement with high beta ===
    print("\n=== Phase 3: LSE High-Beta Refinement ===")
    t3 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 350
        if remaining > 20:
            t_alloc = min(remaining, 150)
            print(f"  LSE-high n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = logsumexp_descent(best_seq, time_limit=t_alloc,
                                         beta_init=50, beta_max=800, verbose=True)
            try_seq(a_opt, "lse_high")

            # Second pass with even higher beta
            remaining = deadline - time.monotonic() - 200
            if remaining > 15:
                t_alloc2 = min(remaining, 80)
                a_opt2, _ = logsumexp_descent(a_opt, time_limit=t_alloc2,
                                              beta_init=100, beta_max=1000)
                try_seq(a_opt2, "lse_high2")

    print(f"  P3 done: best={best_score:.8f}, t={time.monotonic()-t3:.0f}s")

    # === Phase 4: Coordinate descent refinement ===
    print("\n=== Phase 4: Coordinate Descent ===")
    t4 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 250
        if remaining > 15:
            t_alloc = min(remaining, 100)
            print(f"  CD n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = coord_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord")

            # Subgradient after CD
            remaining = deadline - time.monotonic() - 150
            if remaining > 15:
                t_alloc2 = min(remaining, 60)
                a_opt2, _ = subgradient_topk(a_opt, time_limit=t_alloc2, top_k=8)
                try_seq(a_opt2, "sg_after_cd")

    print(f"  P4 done: best={best_score:.8f}, t={time.monotonic()-t4:.0f}s")

    # === Phase 5: Subgradient with various K ===
    print("\n=== Phase 5: Multi-K Subgradient ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        for k in [4, 8, 15, 25]:
            remaining = deadline - time.monotonic() - 100
            if remaining < 15:
                break
            t_alloc = min(remaining, 50)
            tau = 0.002 if k <= 8 else 0.005
            print(f"  SG k={k}, tau={tau}, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_topk(best_seq, time_limit=t_alloc, top_k=k, tau_scale=tau)
            try_seq(a_opt, f"sg_k{k}")

    print(f"  P5 done: best={best_score:.8f}, t={time.monotonic()-t5:.0f}s")

    # === Phase 6: Diversified restarts ===
    print("\n=== Phase 6: Diversified Restarts ===")
    t6 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Different perturbation scales
        scales = [0.002, 0.005, 0.01, 0.02, 0.04]
        for scale in scales:
            remaining = deadline - time.monotonic() - 50
            if remaining < 15:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(30, remaining - 40)
            if t_alloc < 8:
                break
            print(f"  Restart LSE scale={scale}, t={t_alloc:.0f}s")
            a_opt, _ = logsumexp_descent(f_pert, time_limit=t_alloc,
                                         beta_init=30, beta_max=400)
            try_seq(a_opt, f"restart_lse_{scale}")

        # SCA restarts
        for scale in [0.003, 0.01]:
            remaining = deadline - time.monotonic() - 40
            if remaining < 15:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(25, remaining - 35)
            if t_alloc < 8:
                break
            print(f"  Restart SCA scale={scale}, t={t_alloc:.0f}s")
            a_opt, _ = sca_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"restart_sca_{scale}")

    print(f"  P6 done: best={best_score:.8f}, t={time.monotonic()-t6:.0f}s")

    # === Phase 7: Try different n with structured inits ===
    print("\n=== Phase 7: Structured Inits at Different n ===")
    t7 = time.monotonic()

    for n_try in [512, 768, 1024, 1280]:
        remaining = deadline - time.monotonic() - 30
        if remaining < 20:
            break
        inits = generate_structured_inits(n_try, parent, rng)
        for i, init in enumerate(inits[:3]):  # Try top 3 inits
            remaining = deadline - time.monotonic() - 20
            if remaining < 15:
                break
            t_alloc = min(20, remaining - 15)
            if t_alloc < 8:
                break
            a_opt, _ = logsumexp_descent(init, time_limit=t_alloc,
                                         beta_init=30, beta_max=300)
            try_seq(a_opt, f"struct_n{n_try}_i{i}")

    print(f"  P7 done: best={best_score:.8f}, t={time.monotonic()-t7:.0f}s")

    # === Phase 8: Final polish ===
    print("\n=== Phase 8: Final Polish ===")
    t8 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # SCA final
        remaining = deadline - time.monotonic() - 15
        if remaining > 10:
            t_alloc = min(remaining, 40)
            a_opt, _ = sca_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_sca")

        # LSE final
        remaining = deadline - time.monotonic() - 10
        if remaining > 8:
            t_alloc = min(remaining, 25)
            a_opt, _ = logsumexp_descent(best_seq, time_limit=t_alloc,
                                         beta_init=100, beta_max=1000)
            try_seq(a_opt, "final_lse")

        # CD final
        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            a_opt, _ = coord_descent(best_seq, time_limit=min(remaining, 15))
            try_seq(a_opt, "final_cd")

    print(f"  P8 done: best={best_score:.8f}, t={time.monotonic()-t8:.0f}s")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
