"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive-temperature 
logsumexp gradient descent with FFT-based gradient computation.
Key innovations over previous approach:
- Smooth gradient via logsumexp (replaces subgradient)
- FFT-based O(n log n) gradient evaluation
- Exact max-based line search (not smoothed)
- Multi-scale temperature annealing
- Structured spectral initializations
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(p, T, n):
    """
    Compute gradient of T*log(mean(exp(conv/T))) with respect to p.
    Returns gradient vector of length n.
    Uses FFT for O(n log n) computation.
    """
    conv = fast_convolve(p)
    
    # Softmax weights
    conv_shifted = conv - conv.max()  # for numerical stability
    w = np.exp(conv_shifted / T)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(conv)) / len(conv)
    else:
        w = w / w_sum
    
    # Gradient: grad[m] = 2 * sum_k w[k] * p[k-m] * I(0<=k-m<n)
    # This is 2 * cross_correlation(w, p)[m]
    # cross_correlation(w, p)[m] = sum_j w[m+j] * p[j]
    
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    P = np.fft.rfft(p, fft_size)
    # Cross-correlation: ifft(W * conj(P))
    # But we need (w ⊛ p)[m] = sum_j w[m+j] * p[j]
    # Standard: ifft(fft(w) * conj(fft(p))) gives correlation
    cross = np.fft.irfft(W * np.conj(P), fft_size)[:2*n-1]
    
    # The correlation gives (w ⊛ p)[m] = sum_j w[j+m] * p[j] for m in range
    # We want grad[m] = 2 * sum_{j=0}^{n-1} w[m+j] * p[j] for m in [0, n-1]
    # This is exactly cross[m] for m in [0, n-1]
    grad = 2.0 * cross[:n]
    
    return grad, w


def line_search_max_conv(p, direction, n, max_iter=50):
    """
    Line search: find alpha in [0, 1] that minimizes max(conv(p + alpha*(-direction)))
    subject to p - alpha*direction >= 0 and sum = 1 (after projection).
    """
    best_alpha = 0.0
    best_val = fast_convolve(p).max()
    
    # Backtracking line search
    alpha = 1.0
    c1 = 1e-4  # sufficient decrease
    
    for _ in range(max_iter):
        candidate = p - alpha * direction
        candidate = np.maximum(candidate, 0)
        s = candidate.sum()
        if s < 1e-15:
            alpha *= 0.5
            continue
        candidate = candidate / s
        
        val = fast_convolve(candidate).max()
        if val < best_val * (1 - c1 * alpha * abs(direction.max() + 1e-10)):
            return candidate, val
        alpha *= 0.5
        if alpha < 1e-10:
            break
    
    # Try some fixed alphas
    for test_alpha in [0.01, 0.005, 0.001, 0.0005]:
        candidate = p - test_alpha * direction
        candidate = np.maximum(candidate, 0)
        s = candidate.sum()
        if s < 1e-15:
            continue
        candidate = candidate / s
        val = fast_convolve(candidate).max()
        if val < best_val:
            best_val = val
            best_alpha = test_alpha
            best_candidate = candidate
    
    if best_alpha > 0:
        return best_candidate, best_val
    return p, best_val


def gradient_descent_optimize(p0, time_limit, T_init=0.01, T_min=0.0001, lr_init=0.1):
    """
    Adaptive temperature gradient descent on the simplex.
    Minimizes max(conv(p,p)) where sum(p)=1, p>=0.
    """
    p = np.maximum(p0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p)) / len(p)
    else:
        p = p / s
    n = len(p)
    
    T = T_init
    lr = lr_init
    best_p = p.copy()
    best_obj = fast_convolve(p).max()
    stagnant = 0
    it = 0
    t0 = time.monotonic()
    
    v = np.zeros(n)  # momentum
    mom = 0.8
    
    while time.monotonic() - t0 < time_limit - 0.1:
        it += 1
        
        # Compute smoothed gradient
        grad, w = compute_smoothed_gradient(p, T, n)
        
        # Gradient direction (negative gradient + momentum)
        v = mom * v - lr * grad
        direction = v
        
        # Line search
        candidate, val = line_search_max_conv(p, direction, n)
        
        if val < best_obj - 1e-14:
            best_obj = val
            best_p = candidate.copy()
            stagnant = 0
            p = candidate
        else:
            p = candidate  # still move, even if not improving (for exploration)
            stagnant += 1
        
        # Adaptive temperature
        if stagnant < 50:
            T = max(T * 0.995, T_min)
        elif stagnant < 150:
            T = max(T * 0.97, T_min)
        else:
            # Restart with higher T
            T = T_init * 0.5
            lr = lr_init * 0.8
            mom = 0.9
            v = np.zeros(n)
            # Perturb
            rng = np.random.default_rng(it * 7919)
            p = project_simplex(best_p + rng.standard_normal(n) * 0.003)
            stagnant = 0
        
        # Adapt learning rate
        if stagnant > 30 and stagnant < 100:
            lr *= 0.95
        elif stagnant >= 100:
            lr = max(lr * 1.2, lr_init * 0.5)
        
        # Periodic evaluation
        if it % 100 == 0 and it < 10000:
            actual = fast_convolve(p).max()
            if actual < best_obj:
                best_obj = actual
                best_p = p.copy()
    
    return best_p, best_obj


def sign_gradient_descent(p0, time_limit, top_k=3):
    """
    Target the top-k peaks directly using signed gradient.
    For each peak at index k, the gradient contribution is 2*p[k-m] (shifted).
    We move in the direction that reduces all top peaks simultaneously.
    """
    p = np.maximum(p0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p)) / len(p)
    else:
        p = p / s
    n = len(p)
    
    best_p = p.copy()
    best_obj = fast_convolve(p).max()
    lr = 0.05
    it = 0
    t0 = time.monotonic()
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit - 0.1:
        it += 1
        
        conv = fast_convolve(p)
        
        # Find top-k peaks
        if top_k < len(conv):
            peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peak_idx = np.arange(len(conv))
        
        # Compute gradient as weighted sum of peak contributions
        # For peak at k: contribution to grad[m] is 2*p[k-m] (if 0<=k-m<n)
        grad = np.zeros(n)
        for k in peak_idx:
            k = int(k)
            # conv[k] = sum_j p[j]*p[k-j], d/dp[m] = 2*p[k-m] (if valid)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            for m in range(lo, hi + 1):
                grad[m] += 2.0 * p[k - m]
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Projected gradient step
        candidate = p - lr * grad
        candidate = project_simplex(candidate)
        
        val = fast_convolve(candidate).max()
        
        if val < best_obj - 1e-14:
            best_obj = val
            best_p = candidate.copy()
            p = candidate
            stagnant = 0
            lr = min(lr * 1.02, 0.3)
        else:
            stagnant += 1
            lr *= 0.95
            if stagnant > 100:
                # Restart
                p = project_simplex(best_p + np.random.default_rng(it).standard_normal(n) * 0.002)
                stagnant = 0
                lr = 0.05
            if stagnant > 500:
                break
    
    return best_p, best_obj


def generate_init(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng()
    
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine taper
    cos_t = np.cos(np.pi * x / (n - 1))
    w = 0.5 * (1 + cos_t)
    w = np.maximum(w, 0.01)
    inits.append(w / w.sum())
    
    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge / edge.sum())
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - (n-1)/2) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Two peaks (Barker-inspired)
    g1 = np.exp(-0.5 * ((x - 0.25*n) / (0.06*n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75*n) / (0.06*n)) ** 2)
    inits.append((g1 + g2) / (g1 + g2).sum())
    
    # Wilson sequence inspired (alternating weights)
    w_alt = np.ones(n)
    w_alt[::2] = 1.5
    w_alt[1::2] = 1.0
    inits.append(w_alt / w_alt.sum())
    
    # Geometric taper
    geo = 0.98 ** np.minimum(x, n-1-x)
    inits.append(geo / geo.sum())
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 0.2))
    
    # Parent resampled
    if parent is not None:
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        p_res = np.interp(x_new, x_old, parent)
        p_res = np.maximum(p_res, 0)
        p_res = p_res / p_res.sum()
        inits.append(p_res)
    
    return inits


def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s
