def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_oracle(f, M, warm_start=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            g = np.maximum(res.x, 0.0)
            return g
    except Exception:
        pass
    return None

def nesterov_subgradient(a0, time_limit, lr_init=0.005, max_k=50):
    """
    Nesterov-accelerated subgradient method with top-k peak tracking.
    Uses lookahead to achieve O(1/T^2) convergence rate.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    y = a.copy()  # Nesterov lookahead point
    v = np.zeros(n)  # momentum
    t_nest = 1.0
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # Nesterov lookahead
        alpha = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / alpha
        y = (1.0 + beta) * a - beta * y
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y / ys
        t_nest += 1.0

        # Compute top-k subgradient
        conv = fast_convolve(y)

        if max_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -max_k)[-max_k:]

        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * y[k - idx_arr]

        # Nesterov-style update
        v = beta * v - lr * sg
        a_new = a + v
        a_new = project_simplex(a_new)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            a_new = a  # Keep old point if no improvement

        a = a_new
        lr *= (1.0 - 0.0002)

        # Adaptive restart
        if no_improve > 100:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.001 * np.sqrt(no_improve / 100.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            y = a.copy()
            v = np.zeros(n)
            t_nest = 1.0
            no_improve = 0

    return best_a, best_obj


def peak_coordinate_descent(a0, time_limit):
    """
    Targeted coordinate descent: for each top peak, try to reduce it
    by adjusting the variables that contribute to it.
    Uses exact 2-variable optimization (golden section on the line).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        # Get top 5 peaks
        top_k = min(5, len(conv))
        peak_lags = np.argpartition(conv, -top_k)[-top_k:]
        peak_lags = peak_lags[np.argsort(conv[peak_lags])[::-1]]

        improved_this_round = False

        for k_peak in peak_lags:
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            # Variables that affect this peak
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            affected = np.arange(js, je + 1)

            if len(affected) < 2:
                continue

            # Try pairwise swaps: move mass from j1 to j2
            # Focus on pairs where j1 contributes to the peak but j2 doesn't
            # (or contributes less to other peaks)
            peak_val = conv[int(k_peak)]

            # Find best pair to swap
            best_gain = 0
            best_pair = None
            best_frac = 0

            sampled = affected
            if len(sampled) > 30:
                rng_sample = np.random.default_rng(it * 7 + k_peak)
                sampled = rng_sample.choice(affected, size=30, replace=False)

            for ii in range(len(sampled)):
                for jj in range(ii + 1, len(sampled)):
                    j1 = sampled[ii]
                    j2 = sampled[jj]

                    # How much does a[j1] contribute to this peak?
                    contributor1 = a[k_peak - j1] if 0 <= k_peak - j1 < n else 0
                    contributor2 = a[k_peak - j2] if 0 <= k_peak - j2 < n else 0

                    # We want to move mass from the one with higher contribution
                    # to the one with lower contribution
                    if contributor1 >= contributor2 and a[j1] > a[j2] * 0.1:
                        src, dst = j1, j2
                    elif a[j2] > a[j1] * 0.1:
                        src, dst = j2, j1
                    else:
                        continue

                    if a[src] < 1e-8:
                        continue

                    # Try a small transfer
                    for frac in [0.01, 0.05, 0.1, 0.2, 0.3]:
                        delta = a[src] * frac
                        if delta < 1e-10:
                            continue
                        a_test = a.copy()
                        a_test[src] -= delta
                        a_test[dst] += delta
                        a_test = np.maximum(a_test, 0.0)
                        s = a_test.sum()
                        if s < 1e-15:
                            continue
                        a_test = a_test / s
                        c_new = fast_convolve(a_test)
                        new_max = c_new.max()
                        gain = best_obj - new_max
                        if gain > best_gain:
                            best_gain = gain
                            best_pair = (src, dst)
                            best_frac = frac

            if best_pair is not None and best_gain > 1e-12:
                src, dst = best_pair
                delta = a[src] * best_frac
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                c_new = fast_convolve(a)
                best_obj = c_new.max()
                best_a = a.copy()
                improved_this_round = True
                stall = 0
            else:
                stall += 1

        if not improved_this_round:
            stall += 1

        # If stalled, try a small random perturbation
        if stall > 20:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise_scale = 0.0005 + 0.001 * min(1.0, stall / 50.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            stall = 0

        if stall > 200:
            break

    return best_a, best_obj


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]."""
    return np.correlate(p, a, mode='valid')


def smoothed_grad_refine(a0, time_limit, T_init=0.001, lr_init=0.03):
    """Smoothed gradient descent with softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = 2.0 * hankel_mv(p, a)

        # Backtracking line search
        best_candidate = a
        best_val = conv.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            T *= 1.08
            lr *= 0.97
        if no_improve > 100:
            T *= 0.8
        if no_improve > 300:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def micro_polish(a0, time_limit):
    """Very fine perturbation-based polish for the last bit of improvement."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    scales = [0.0001, 0.0003, 0.001, 0.002]
    si = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        scale = scales[si % len(scales)]
        rng = np.random.default_rng((it * 2654435761 + 1) % (2**32))

        # Try several perturbations per iteration
        tried = 0
        while tried < 10 and time.monotonic() - t0 < time_limit - 0.1:
            tried += 1
            # Structured perturbation: biased toward reducing peaks
            conv = fast_convolve(a)
            peak_k = int(np.argmax(conv))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            noise = rng.standard_normal(n) * scale
            # Bias: reduce values that contribute to the peak
            for j in range(js, je + 1):
                partner = peak_k - j
                if 0 <= partner < n:
                    noise[j] -= scale * 2.0 * a[partner]

            candidate = project_simplex(a + noise)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                break

        if it % 100 == 0:
            si += 1

    return best_a, best_obj
