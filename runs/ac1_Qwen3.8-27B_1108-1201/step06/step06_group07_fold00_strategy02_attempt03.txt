            opt_seq = slew_sequence_optimizer(start_seq, time_limit)
            score_val = border_check(opt_seq, n_candidate)
            if score_val < 1.5048 and score_val < best_score:
                best_score = score_val
                best_n = n_candidate
                best_sequence = opt_seq.copy()
            # Evaluate the composed Slepian candidates if n_candidate >= 8
            if n_candidate >= 8:
                s_vec = np.fft.irfft(np.fft.rfft(np.log(np.maximum(opt_seq, 1e-15))))
                # Generate alternative points while preserving simplex
                for i in range(8):
                    candidate = np.zeros(n_candidate)  # Clean slate
                    candidate[i] = np.exp(s_vec[i])
                    score_candidate = objective_function(candidate)
                    if score_candidate > 1.5 and score_candidate < best_score:
                        best_sequence = candidate
                        best_score = score_candidate
        except Exception as e:
            print(f"Failure during n.optimization: {e}")
    
    if best_n is None:
        try_load_parent = False
        if hasattr(globals(), 'height_sequence_1'):
            try:
                if objective_function(height_sequence_1) < best_score:
                    best_sequence = height_sequence_1.copy()
                    best_score = objective_function(best_sequence)
                    return [float(x) for x in best_sequence]
            except:
                pass
        else:
            # Fallback uniform if no parent and no optimization
            base = np.ones(min(256, 512)) / min(256, 512)
            best_sequence = base.copy()
            best_score = objective_function(best_sequence)
    
    # Formatted output in the requested form
    return [float(x) / best_sequence.sum() for x in best_sequence * 10]  # Multiplicative scaling for normalization
