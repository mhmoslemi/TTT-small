```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Momentum projected gradient with LSE smoothing (multi-n sweep)
2. L-BFGS-B polish on softmax-reparameterized LSE objective
3. CAZAC-inspired and diverse initializations
4. Symmetry exploitation (reversal averaging)
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_weights(conv, beta):
    m = len(conv)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.ones(m) / m
    return exp_bc / total

def compute_grad_smoothed(f, w=None):
    """Gradient of sum_k w_k * (f*f)[k] wrt f."""
    n = len(f)
    if w is None:
        conv = fast_convolve(f)
        w = lse_weights(conv, 100.0)
    f_rev = f[::-1]
    conv_fw = np.convolve(f_rev, w, mode='full')
    return 2.0 * conv_fw[n-1 : 2*n-1]

def momentum_gd(f0, time_limit, beta_start=30.0, beta_end=3000.0):
    """Momentum projected gradient on LSE-smoothed max."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    t0 = time.monotonic()
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    v = np.zeros(n)
    momentum = 0.85
    step = 5.0 / (beta * n)
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        conv = fast_convolve(f)
        w = lse_weights(conv, beta)
        grad = compute_grad_smoothed(f, w)

        if gnorm < 1e-18:

        step_adaptive = 1.0 / (np.linalg.norm(v) + 1e-10) * 0.5

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

            no_improve = 0

def softmax(z):
    z = z - z.max()
    ez = np.exp(z)
    return ez / ez.sum()

def lse_objective_softmax(z, beta):
    """LSE-smoothed max of conv, with a = softmax(z)."""
    a = softmax(z)
    conv = fast_convolve(a)
    bc = beta * conv
    bc_max = bc.max()
    return (1.0/beta) * (bc_max + np.log(np.exp(bc - bc_max).sum()))

def lse_gradient_softmax(z, beta):
    """Gradient of lse_objective_softmax."""
    a = softmax(z)
    n = len(a)
    conv = fast_convolve(a)
    w = lse_weights(conv, beta)
    grad_a = compute_grad_smoothed(a, w)
    # Chain rule through softmax: dz_j = a_i * (delta_ij - a_j)
    grad_z = a * (grad_a - np.dot(a, grad_a))
    return grad_z

def lbfgs_polish(f0, time_limit, beta_start=100.0, beta_end=10000.0, max_iter=200):
    """L-BFGS-B polish using softmax reparameterization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    z0 = np.where(f > 1e-10, np.log(f), -10.0)
    
    t0 = time.monotonic()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    beta = beta_start
    for _ in range(20):  # multiple L-BFGS-B runs with increasing beta
        if time.monotonic() - t0 > time_limit - 10:
            break
        
        result = minimize(
            fun=lambda z: lse_objective_softmax(z, beta),
            x0=z0,
            jac=lambda z: lse_gradient_softmax(z, beta),
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12}
        )
        
        z_opt = result.x
        a_opt = softmax(z_opt)
        conv_opt = fast_convolve(a_opt)
        obj_opt = conv_opt.max()
