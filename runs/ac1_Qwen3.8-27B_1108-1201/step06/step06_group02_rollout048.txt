Mirror descent (multiplicative weight) + FW-LP hybrid for minimizing
2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: uses KL-divergence-based mirror descent which is the
natural algorithm for this problem (derived from primal-dual analysis).

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

"""
Hybrid mirror descent + FW-LP + coordinate descent for minimizing
2*n*max(conv(a,a))/(sum(a))^2.
Uses KL-divergence mirror descent (multiplicative weights) as the primary
method, derived from primal-dual analysis of the convex relaxation.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad(p, a):
    """Compute gradient of smoothed objective: 2 * hankel_mv(p, a)."""
    return 2.0 * np.correlate(p, a, mode='valid')


def mirror_descent(a0, time_limit, T_init=0.001, lr_init=0.05):
    """Mirror descent with KL divergence (multiplicative weight)."""
    a = np.maximum(a0.astype(np.float64), 1e-300)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        grad = compute_grad(p, a)
        grad_max = np.max(np.abs(grad))
        if grad_max < 1e-30:
            break
        
        # Normalized step size
        eff_lr = lr / grad_max
        
        for trial_scale in [1.0, 0.5, 0.25, 0.1]:
            log_update = -trial_scale * eff_lr * grad
            log_update = np.clip(log_update, -100, 100)
            a_new = a * np.exp(log_update)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue

            obj = fast_convolve(a_new).max()

def solve_lp_oracle(f, M, time_limit_lp=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit_lp})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def coordinate_descent(a0, time_limit):
    """Coordinate descent: move mass from peak contributors to non-contributors."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    stale = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        
        # Find the index i that most contributes to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        if len(affected) == 0:
            break
        
        # Contribution of a[i] to peak: a[i] * a[k_peak - i]
        partners = a[k_peak - affected]
        contributions = a[affected] * partners
        
        # Find the "hottest" index (most to reduce)
        j_hot = affected[np.argmax(contributions)]
        
        # Find a "cold" index (one that doesn't contribute to the peak)
        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            # If all indices contribute, find the one with least contribution
            j_cold = affected[np.argmin(contributions)]
        else:
            # Among cold indices, pick one with least value (to not increase other peaks too much)
            j_cold = outside[np.argmin(a[outside])]
        
        if a[j_hot] < 1e-10:
            stale += 1
            if stale > 100:
                break
            continue
        
        # Try different transfer amounts
        improved = False
        for frac in [0.001, 0.0005, 0.002, 0.005, 0.01, 0.02]:
            delta = a[j_hot] * frac
            if delta < 1e-12:
                continue
            candidate = a.copy()
            candidate[j_hot] -= delta
            candidate[j_cold] += delta
            candidate = np.maximum(candidate, 0.0)
            cs = candidate.sum()
            if cs < 1e-15:
                continue
            candidate = candidate / cs
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            stale = 0
        else:
            stale += 1
            if stale % 50 == 0:
                rng = np.random.default_rng(stale * 12345 + 7)
                # Random coordinate move
                i1 = rng.integers(n)
                i2 = rng.integers(n)
                if i1 == i2:
                    continue
                delta = a[i1] * 0.01
                candidate = a.copy()
                candidate[i1] -= delta
                candidate[i2] += delta
                candidate = np.maximum(candidate, 0.0)
                candidate = candidate / candidate.sum()
                if fast_convolve(candidate).max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = fast_convolve(a).max()
                    best_a = a.copy()
                    stale = 0
            elif stale > 500:
                break
    
    return best_a, best_obj
