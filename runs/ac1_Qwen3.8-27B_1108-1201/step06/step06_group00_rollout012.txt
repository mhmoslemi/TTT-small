import time

    n = len(a)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def _eval_direction(f, g, alpha):
    """Evaluate objective at (1-alpha)*f + alpha*g."""
    ft = (1.0 - alpha) * f + alpha * g
    ft = np.maximum(ft, 0.0)
    st = ft.sum()
    if st < 1e-15:
        return np.inf
    ft /= st
    return fast_convolve(ft).max()


def _diversify(f, best_f, best_obj, n, seed_offset):
    """Perturb f in frequency domain to escape stagnation."""
    rng = np.random.default_rng(seed_offset * 7919 + 13)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    scale = 0.02 + 0.01 * min(1.0, 10.0 / (seed_offset + 1))
    mag_pert = np.maximum(mag * (1.0 + rng.standard_normal(len(mag)) * scale), 0)
    F_new = mag_pert * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    f_new = project_simplex(f_new)
    conv = fast_convolve(f_new)
    M = conv.max()
    if M < best_obj:
        best_obj = M
        best_f = f_new.copy()
    return f_new, best_obj, best_f


def coord_descent_polish(f0, time_limit, verbose=False):
    """Peak-aware coordinate descent: reduce active peaks by redistributing mass."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    total_iters = 0
    while time.monotonic() - t0 < time_limit:
        batch_time = time_limit - (time.monotonic() - t0)
        if batch_time < 2:
            break

        # Get top peaks
        num_peaks = min(15, len(conv))
        peak_idx = np.argpartition(conv, -num_peaks)[-num_peaks:]
        conv_sorted = conv[peak_idx]

        # For each peak, try to reduce it
        for k in peak_idx:
            k = int(k)
            if conv[k] < best_obj * 0.95:
                continue  # not an active peak

            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo >= hi:
                continue

            # Find the coordinate i in [lo, hi] that contributes most to C[k]
            i_range = np.arange(lo, hi + 1)
            contributions = f[i_range] * f[k - i_range]
            i_max = i_range[np.argmax(contributions)]

            if f[i_max] < 1e-8:
                continue

            # Try moving a fraction of mass from i_max to a random "safe" position
            frac = rng.choice([0.05, 0.1, 0.15, 0.2, 0.3])
            delta = f[i_max] * frac

            # Choose destination: prefer positions far from active peaks
            candidates = rng.choice(n, size=min(8, n), replace=False)
            for j_dest in candidates:
                if j_dest == i_max:
                    continue
                f_trial = f.copy()
                f_trial[i_max] -= delta
                f_trial[j_dest] += delta
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                conv_trial = fast_convolve(f_trial)
                obj_trial = conv_trial.max()
                total_iters += 1
                if obj_trial < best_obj - 1e-12:
                    best_obj = obj_trial
                    best_f = f_trial.copy()
                    f = f_trial.copy()
                    conv = conv_trial
                    break

        # Also try random small perturbations
        if total_iters % 50 == 0:
            for _ in range(3):
                if time.monotonic() - t0 > time_limit:
                    break
                f_pert = project_simplex(f + rng.standard_normal(n) * 0.003)
                conv_pert = fast_convolve(f_pert)
                if conv_pert.max() < best_obj - 1e-12:
                    best_obj = conv_pert.max()
                    best_f = f_pert.copy()
                    f = f_pert.copy()

    return best_f, best_obj


def subgradient_step(f, beta, top_k=40):
    """One subgradient step with LSE-smoothed max."""
    n = len(f)
    conv = fast_convolve(f)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        w = np.ones(len(conv)) / len(conv)
    else:
        w = exp_bc / total

    # Focus on top peaks
    k_top = min(top_k, len(conv))
    peak_idx = np.argpartition(conv, -k_top)[-k_top:]
    w_peaks = w[peak_idx]
    ws = w_peaks.sum()
    if ws > 1e-15:
        w_peaks /= ws
    else:
        w_peaks = np.ones(k_top) / k_top

    # Compute subgradient
    grad = np.zeros(n)
    for idx, k in enumerate(peak_idx):
        k = int(k)
        wi = w_peaks[idx]
        if wi < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += wi * 2.0 * f[k - i_range]

    return grad, conv.max()


def subgradient_optimize(f0, time_limit, beta_start=20.0, beta_end=5000.0, top_k=50):
    """Subgradient descent with adaptive beta and backtracking."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    beta = beta_start
    lr = 1.5 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        grad, cur_obj = subgradient_step(f, beta, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_n = grad / gnorm

        # Backtracking line search
        step = lr
        improved = False
        for _ in range(8):
            f_new = project_simplex(f - step * grad_n)
            obj_new = fast_convolve(f_new).max()
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                improved = True
                no_improve = 0
                break
            step *= 0.5

        if improved:
            lr = min(lr * 1.002, 5.0 / np.sqrt(n))
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.9
            if no_improve > 50:
                f = best_f.copy()
                lr = max(1.0 / np.sqrt(n), 0.005)
                no_improve = 0

        beta = min(beta * 1.005, beta_end)

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def freq_perturb(f, rng, strength=0.03):
    """Perturb f in the frequency domain."""
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Main optimization:
    Phase 1: Short FW on multiple lengths for exploration
    Phase 2: Deep FW with full LP oracle on best length
    Phase 3: Subgradient refinement with increasing beta
    Phase 4: Coordinate descent polish
    Phase 5: Final random perturbations
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768.0
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    total_budget = float(budget_s) - 30.0

    # === Phase 1: Multi-length FW exploration (25% time) ===
    print("\n=== Phase 1: Multi-length FW exploration ===")
    t1_end = time.monotonic() + total_budget * 0.25

    n_candidates = sorted(set([
        n_parent,
        max(128, n_parent - 64),
        min(2000, n_parent + 64),
        max(128, int(n_parent * 0.75)),
        min(2000, int(n_parent * 1.25)),
    ]), reverse=True)
    seen = set()
    n_candidates = [x for x in n_candidates if not (x in seen or seen.add(x))]

    for nc in n_candidates[:4]:
        t_rem = t1_end - time.monotonic()
        if t_rem < 30:
            break
        t_alloc = min(80, t_rem * 0.3)
        if t_alloc < 15:
            break
        if nc == n_parent:
            f_init = parent.copy()
        else:
            f_init = interpolate_to_length(parent, nc)
        print(f"  FW explore: n={nc}, t={t_alloc:.0f}s")
        f_opt, _ = fw_optimize(f_init, time_limit=t_alloc)
        try_sequence(f_opt, best_tracker)

    # === Phase 2: Deep FW on best (40% time) ===
    print(f"\n=== Phase 2: Deep FW optimization ===")
    t2_end = t1_end + total_budget * 0.40
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)
    t_alloc = min(800, t2_end - time.monotonic())
    if t_alloc > 30:
        print(f"  Deep FW: n={n_cur}, t={t_alloc:.0f}s")
        f_opt, _ = fw_optimize(current_best, time_limit=t_alloc, verbose=True)
        try_sequence(f_opt, best_tracker)

        # Second pass if time remains
        t_rem = t2_end - time.monotonic()
        if t_rem > 60:
            current_best = best_tracker[1] if best_tracker[1] is not None else current_best
            n_cur = len(current_best)
            print(f"  Second FW pass: n={n_cur}, t={t_rem:.0f}s")
            f_opt, _ = fw_optimize(current_best, time_limit=t_rem - 5)
            try_sequence(f_opt, best_tracker)

    # === Phase 3: Subgradient refinement (20% time) ===
    print(f"\n=== Phase 3: Subgradient refinement ===")
    t3_end = t2_end + total_budget * 0.20
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)
    t_alloc = min(300, t3_end - time.monotonic())
    if t_alloc > 30:
        print(f"  Subgradient: n={n_cur}, t={t_alloc:.0f}s")
        f_opt, _ = subgradient_optimize(current_best, time_limit=t_alloc,
                                         beta_start=30.0, beta_end=10000.0, top_k=60)
        try_sequence(f_opt, best_tracker)

    # === Phase 4: Coordinate descent polish (10% time) ===
    print(f"\n=== Phase 4: Coordinate descent polish ===")
    t4_end = min(t3_end + total_budget * 0.10, deadline - 30)
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)
    t_alloc = min(200, t4_end - time.monotonic())
    if t_alloc > 20:
        print(f"  Coord descent: n={n_cur}, t={t_alloc:.0f}s")
        f_opt, _ = coord_descent_polish(current_best, time_limit=t_alloc)
        try_sequence(f_opt, best_tracker)

    # === Phase 5: Final perturbations and verification ===
    print(f"\n=== Phase 5: Final refinements ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)
    t_rem = deadline - time.monotonic()
    if t_rem > 15:
        print(f"  Final random perturbations: t={t_rem:.0f}s")
        for attempt in range(30):
            if time.monotonic() > deadline - 5:
                break
            # Small random perturbation
            f_pert = project_simplex(current_best + rng.standard_normal(n_cur) * 0.002)
            try_sequence(f_pert, best_tracker)
            # Frequency perturbation
            f_freq = freq_perturb(current_best, rng, strength=0.005)
            try_sequence(f_freq, best_tracker)
            # Brief coord descent on perturbed
            if time.monotonic() > deadline - 10:
                break
            f_cd, _ = coord_descent_polish(f_pert, time_limit=min(5, deadline - time.monotonic() - 5))
            try_sequence(f_cd, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using uniform fallback")
    fallback = np.ones(448) / 448.0
    return [float(x) for x in fallback]
