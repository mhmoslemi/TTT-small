using projected gradient descent, Frank-Wolfe with LP oracle, and
active-set Newton methods to push below 1.5027.
"""
import numpy as np
from scipy.optimize import linprog
import time
import math


def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def flat_conv_sequence(n):
    """Generate the flat-convolution sequence a[k] = C(2k,k)/4^k, normalized."""
    a = np.zeros(n)
    # C(2k,k)/4^k can be computed recursively
    # a[0] = 1, a[k] = a[k-1] * (2k-1)/(2k)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2 * k - 1) / (2 * k)
    s = a.sum()
    a = a / s
    return a


def symmetric_flat_conv(n):
    """Symmetric version: average of flat conv and its reversal."""
    a = flat_conv_sequence(n)
    b = a[::-1]
    s = a + b
    return s / s.sum()


def product_flat_conv(n):
    """Product of flat conv and its reversal (bump in the middle)."""
    a = flat_conv_sequence(n)
    b = a[::-1]
    s = a * b
    return s / s.sum()


def compute_gradient(a, top_k=30):
    """Compute weighted subgradient from top-k peaks of auto-convolution."""
    n = len(a)
    conv = fast_convolve(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    order = np.argsort(pv)[::-1]
    weights = np.zeros(len(peaks))
    for rank, idx in enumerate(order):
        weights[idx] = 1.0 / (rank + 1) ** 1.5
    ws = weights.sum()
    if ws > 0:
        weights /= ws
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx_arr = np.arange(js, je + 1)
        sg[idx_arr] += weights[i] * a[k - idx_arr]
    return sg, conv


def compute_softmax_gradient(a, T=0.001):
    """Softmax-weighted gradient for smoother optimization."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()
    # Gradient: 2 * M(p) @ a where M(p)_{j,m} = p[j+m]
    grad = np.zeros(n)
    for k in range(len(p)):
        if p[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx_arr = np.arange(js, je + 1)
        grad[idx_arr] += p[k] * a[k - idx_arr]
    return 2.0 * grad, conv


def pgd_optimize(a0, time_limit, top_k=25, lr_init=0.03, mom_init=0.92,
                 use_softmax=False, T_init=0.001):
    """Projected gradient descent with momentum and adaptive step."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = mom_init
    T = T_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        if use_softmax:
            sg, conv = compute_softmax_gradient(a, T)
        else:
            sg, conv = compute_gradient(a, top_k)
        obj = conv.max()

        # Backtracking
        best_candidate = a
        best_val = obj
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            v_new = mom * v - lr * scale * sg
            candidate = project_simplex(a + v_new)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                v = v_new
                break

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.95

        # Adapt
        if no_improve > 20:
            lr *= 0.95
            mom = min(0.98, mom + 0.003)
        if no_improve > 40:
            if use_softmax:
                T *= 1.1
            else:
                top_k = min(60, top_k + 5)
        if no_improve > 150:
            rng = np.random.default_rng((it * 48271 + 17) % (2**31 - 1))
            noise = 0.002 + 0.003 * min(1.0, (no_improve - 150) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init
            mom = mom_init
            T = T_init
            top_k = 25
            no_improve = 0

    return best_a, best_obj


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, m_expand_rate=1.05, m_max=3.0):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        g = solve_lp_oracle(f, M * m_exp)
        if g is None:
            m_exp = min(m_exp * 1.2, m_max * 2)
            no_improve += 1
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 20:
                m_exp = min(m_exp * 1.02, m_max)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, m_max * 2)

        if no_improve > 40:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.005 * min(1.0, no_improve / 80.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def active_set_refine(a0, time_limit, tol=1e-10, max_iter=500):
    """
    Newton-type refinement using active peak identification.
    At the optimum: for active peaks P and support S:
    - sum_{k in P} mu_k * a[k-j] = lambda for all j in S
    - (a*a)[k] = t for all k in P
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1
        if it % 50 == 0:
            print(f"      AS it={it}, obj={best_obj:.10f}")

        conv = fast_convolve(a)
        M = conv.max()
        support = np.where(a > 1e-12)[0]
        if len(support) == 0:
            break

        # Find active peaks (within 0.1% of max)
        threshold = M * (1.0 - 0.0005)
        active_peaks = np.where(conv >= threshold)[0]

        if len(active_peaks) <= 1:
            break

        # Build system: for j in support, sum_{k in P} mu_k * a[k-j] = lambda
        m = len(support)
        p = len(active_peaks)

        A_sys = np.zeros((m, p + 1))
        for i, j in enumerate(support):
            A_sys[i, -1] = 1.0
            for k_idx, k in enumerate(active_peaks):
                idx = k - j
                if 0 <= idx < n:
                    A_sys[i, k_idx] = a[idx]

        try:
            U, S_vals, Vt = np.linalg.svd(A_sys)
            sol = Vt[-1, :]
            mu_vals = np.maximum(sol[:p], 0)
            mu_sum = mu_vals.sum()
            if mu_sum < 1e-15:
                break
            mu_vals /= mu_sum
        except np.linalg.LinAlgError:
            break

        # Gradient direction from mu weights
        grad = np.zeros(n)
        for k_idx, k in enumerate(active_peaks):
            if mu_vals[k_idx] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] -= mu_vals[k_idx] * a[k - idx_arr]

        # Line search
        best_candidate = a
        best_val = M
        for step in [0.0005, 0.002, 0.005, 0.01, 0.02, 0.05]:
            candidate = project_simplex(a + step * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate

        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
        else:
            # Try smaller active set
            threshold = M * (1.0 - 0.001)
            active_peaks_tight = np.where(conv >= threshold)[0]
            if len(active_peaks_tight) < len(active_peaks):
                continue
            if abs(M - best_val) < 1e-13:
                break

    return best_a, best_obj


def peak_equalization(a0, time_limit, max_iter=200):
    """
    Direct peak equalization: iteratively adjust a to equalize the top peaks.
    Uses a first-order approximation to reduce the max while keeping sum=1.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        peaks = np.where(conv >= M * 0.999)[0]

        if len(peaks) == 0:
            break

        # For each peak k, the "contribution" to a[j] is a[k-j]
        # We want to reduce mass at positions contributing to peaks
        penalty = np.zeros(n)
        for k in peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            penalty[idx_arr] += a[k - idx_arr]

        # Move mass away from high-penalty positions
        # Direction: -penalty (reduce where penalty is high)
        # But keep sum = 1, so project
        direction = -penalty
        # Zero out where a is already 0
        direction[a < 1e-15] = 0

        best_candidate = a
        best_val = M
        for alpha in [0.001, 0.003, 0.01, 0.03, 0.1, 0.3]:
            candidate = project_simplex(a + alpha * direction)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate
        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
        elif abs(M - best_val) < 1e-13:
            break

    return best_a, best_obj


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def try_and_update(a, best_seq, best_score, label=""):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return best_seq, best_score

        # PGD on perturbed
        a_opt, _ = pgd_optimize(f_pert, time_limit=t_alloc * 0.6, top_k=30, lr_init=0.02)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"pert{scale}")

        # Quick active-set on perturbed result
        remaining = deadline - time.monotonic() - 80
        if remaining > 10:
            a_opt2, _ = active_set_refine(a_opt, time_limit=min(remaining, 20))
            best_seq, best_score = try_and_update(a_opt2, best_seq, best_score, f"pert_as{scale}")

    # Symmetry check
    sym = (best_seq + best_seq[::-1]) / 2.0
    sym = sym / sym.sum()
    best_seq, best_score = try_and_update(sym, best_seq, best_score, "sym_check")

    # Anti-symmetric (monotone) check
    mono = np.sort(best_seq)[::-1]  # decreasing
    mono = mono / mono.sum()
    best_seq, best_score = try_and_update(mono, best_seq, best_score, "mono_check")

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 6: Final intensive polishing (~150s) ===
    print("\n=== Phase 6: Final Polish ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    # Fine PGD with very small steps
    remaining = deadline - time.monotonic() - 60
    if remaining > 10:
        t_alloc = min(remaining, 40)
        a_opt, _ = pgd_optimize(best_seq, time_limit=t_alloc, top_k=50, lr_init=0.003, mom_init=0.96)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "fine_pg")

    # Active-set with tight tolerance
    remaining = deadline - time.monotonic() - 40
    if remaining > 10:
        t_alloc = min(remaining, 30)
        a_opt, _ = active_set_refine(best_seq, time_limit=t_alloc, tol=1e-12)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "as_tight")

    # Peak equalization
    remaining = deadline - time.monotonic() - 20
    if remaining > 8:
        t_alloc = min(remaining, 20)
        a_opt, _ = peak_equalization(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "peak_final")

    # One more FW-LP
    remaining = deadline - time.monotonic() - 10
    if remaining > 8:
        t_alloc = min(remaining, 15)
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "fw_final")

    # Final symmetry and monotone checks
    sym = (best_seq + best_seq[::-1]) / 2.0
    sym = sym / sym.sum()
    best_seq, best_score = try_and_update(sym, best_seq, best_score, "final_sym")

    mono = np.sort(best_seq)[::-1]
    mono = mono / mono.sum()
    best_seq, best_score = try_and_update(mono, best_seq, best_score, "final_mono")

    # Flat-conv blend
    fc = flat_conv_sequence(len(best_seq))
    for blend in [0.3, 0.5, 0.7]:
        blended = (1 - blend) * best_seq + blend * fc
        blended = np.maximum(blended, 0)
        blended = blended / blended.sum()
        best_seq, best_score = try_and_update(blended, best_seq, best_score, f"blend{blend}")

    print(f"  Phase 6 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = flat_conv_sequence(768)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
