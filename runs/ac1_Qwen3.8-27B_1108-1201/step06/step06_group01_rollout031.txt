import numpy as np
from scipy.ndimage import correlate1d
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smoothed_grad(a, T):
    """Compute the gradient of the smoothed max of auto-convolution."""
    n = len(a)
    conv = fast_convolve(a)
    # Softmax weights (numerically stable)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()
    # Gradient: grad[j] = 2 * sum_m p[j+m] * a[m]
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return grad, conv

def smoothed_nesterov(a0, time_limit, T_init=0.005, lr_init=0.01, max_k=None):

    Smoothed gradient descent with Nesterov momentum.
    Minimizes max_k conv(a,a)[k] subject to a >= 0, sum(a) = 1.
    """

    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    # Nesterov state
    y = a.copy()  # lookahead
    v = np.zeros(n)  # velocity
    t_counter = 1.0
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Nesterov lookahead
        alpha_t = (1.0 + np.sqrt(1.0 + 4.0 * t_counter * t_counter)) / 2.0
        beta = (t_counter - 1.0) / alpha_t
        y_new = (1.0 + beta) * a - beta * y
        y_new = np.maximum(y_new, 0.0)
        ys = y_new.sum()
        if ys < 1e-15:
            y_new = a.copy()
        else:
            y_new = y_new / ys
        y = y_new
        t_counter += 1.0
        
        # Compute gradient at y (lookahead point)
        conv_y = fast_convolve(y)
        scaled = conv_y / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p /= p.sum()
        grad = 2.0 * np.correlate(p, y, mode='valid')
        
        # Update
        v = beta * v - lr * grad
        a_new = a + v
        a_new = project_simplex(a_new)
        
        # Evaluate
        obj = fast_convolve(a_new).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            a_new = a  # Revert if no improvement
        
        a = a_new
        
        # Adaptive parameters
        if no_improve > 30:
            T *= 0.95  # Cool down (sharper)
            lr *= 0.99
        if no_improve > 100:
            T *= 1.1  # Maybe heat up to escape
        if no_improve > 300:
            # Restart with perturbation

            y = a.copy()
            v = np.zeros(n)
            t_counter = 1.0
            T = T_init
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed gradient descent
with Nesterov acceleration, adaptive temperature cooling, and multi-scale
exploration. Exploits the structure of the auto-convolution to guide
mass redistribution away from peak lags. Uses multiple initializations
and sequence lengths to avoid local minima in this non-convex problem.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_soft_gradient(a, T, top_k=None):
    """Compute gradient of smoothed max of auto-convolution using softmax weights."""
    n = len(a)
    conv = fast_convolve(a)
    if top_k is not None and top_k < len(conv):
        # Use only top-k peaks for efficiency when T is small
        idx = np.argpartition(conv, -top_k)[-top_k:]
        p = np.zeros_like(conv)
        vals = conv[idx]
        vals -= vals.max()
        ep = np.exp(vals / T)
        p[idx] = ep
        p_sum = p.sum()
        if p_sum > 0:
            p /= p_sum
        else:
            p = np.ones(len(conv)) / len(conv)
    else:
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p /= p.sum()
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return grad, conv

def coarse_subgradient(a0, time_limit, max_k=20):

        # Compute subgradient: average of 2*a[peak - j] for each peak

            je = min(n - 1, k)

        obj = fast_convolve(a_new).max()

            no_improve = 0

def coordinate_polish(a0, time_limit):
    """
    Coordinate descent polish: cyclically optimize each coordinate using
    a 1D line search (the objective is piecewise quadratic in each coordinate).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        improved = False
        rng = np.random.default_rng(int((time.monotonic() - t0) * 1000))

        # Random coordinate order for better mixing
        order = rng.permutation(n)
        # Only optimize a subset per pass for speed
        subset_size = min(n, 200)
        order = order[:subset_size]

        for j in order:
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            old_val = a[j]
            if old_val < 1e-12:
                # Try setting to a small positive value
                candidates = [0.001, 0.005, 0.01]
            else:
                # Try scaling up and down
                candidates = [old_val * f for f in [0.1, 0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 3.0, 10.0, 100.0]]

            best_candidate = old_val
            best_local_obj = best_obj

            for val in candidates:
                a[j] = val
                a = np.maximum(a, 0.0)
                ss = a.sum()
                if ss < 1e-15:
                    a[j] = old_val
                    continue
                a = a / ss
                obj = fast_convolve(a).max()
                if obj < best_local_obj - 1e-15:
                    best_local_obj = obj
                    best_candidate = val

            a[j] = best_candidate
            a = np.maximum(a, 0.0)
            ss = a.sum()
            if ss > 1e-15:
                a = a / ss

            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:

                best_a = a.copy()

            stall += 1
        else:
            stall = 0

            rng = np.random.default_rng(stall * 997 + 1)

            t_coarse = min(0.3 * n_budget, 60.0)
            if t_coarse > 3:
                a_opt, _ = coarse_subgradient(a_start, time_limit=t_coarse, max_k=20)
                try_seq(a_opt, f"P2a_n{n_try}_coarse")

            # Sub-phase 2b: Smoothed Nesterov (fine-grained, with cooling)
            t_nest = min(0.5 * n_budget, 120.0)
            if t_nest > 5:
                # Try different T values
                for T_val in [0.01, 0.003]:
                    t_sub = t_nest / 2
                    if time.monotonic() > deadline - 300:
                        break
                    if t_sub < 3:
                        break
                    a_opt, _ = smoothed_nesterov(a_opt, time_limit=t_sub,
                                                 T_init=T_val, lr_init=0.006,
                                                 top_k=50)
                    try_seq(a_opt, f"P2b_n{n_try}_T{T_val}")

            # Sub-phase 2c: More Nesterov with smaller T
            t_nest2 = min(0.2 * n_budget, 60.0)
            if t_nest2 > 5:
                a_opt, _ = smoothed_nesterov(a_opt, time_limit=t_nest2,
                                             T_init=0.001, lr_init=0.004,
                                             top_k=20)
                try_seq(a_opt, f"P2c_n{n_try}_T001")

    print(f"  Phase 2 done: best={best_score:.8f} ({time.monotonic()-t_phase_start:.0f}s)")

    # === Phase 3: Perturbation and restart (~300s) ===
    print("\n=== Phase 3: Perturbation & Restart ===")
    t_phase_start = time.monotonic()
    phase3_budget = min(300.0, deadline - time.monotonic() - 120.0)

    if best_seq is not None:
        n_best = len(best_seq)
        perturb_scales = [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]
        time_per_pert = max(5.0, phase3_budget / (len(perturb_scales) * 2))

        for scale in perturb_scales:
            if time.monotonic() > deadline - 120:
                break
            for direction in range(2):
                t_alloc = min(time_per_pert, deadline - time.monotonic() - 120)
                if t_alloc < 3:
                    break

                rng_p = np.random.default_rng(rng.integers(2**31))
                a_pert = project_simplex(best_seq + rng_p.standard_normal(n_best) * scale)
                try_seq(a_pert, f"P3_init_scale{scale}")

                # Short Nesterov from perturbed point
                a_opt, _ = smoothed_nesterov(a_pert, time_limit=t_alloc,
                                             T_init=0.003, lr_init=0.005,
                                             top_k=30)
                try_seq(a_opt, f"P3_nest_scale{scale}_{direction}")

    print(f"  Phase 3 done: best={best_score:.8f} ({time.monotonic()-t_phase_start:.0f}s)")

    # === Phase 4: Final polish (~100s) ===
    print("\n=== Phase 4: Final Polish ===")
    t_phase_start = time.monotonic()
    phase4_budget = min(100.0, deadline - time.monotonic() - 30.0)

    if best_seq is not None:
        # Coordinate descent polish
        t_coord = min(0.5 * phase4_budget, 50.0)
        if t_coord > 5:
            a_opt, _ = coordinate_polish(best_seq, time_limit=t_coord)
            try_seq(a_opt, "P4_coord")

        # Very fine Nesterov
        t_fine = min(0.3 * phase4_budget, 30.0)
        if t_fine > 3:
            a_opt, _ = smoothed_nesterov(a_opt, time_limit=t_fine,
                                         T_init=0.0005, lr_init=0.002,
                                         top_k=10)
            try_seq(a_opt, "P4_fine_nest")

        # Another coordinate pass
        t_coord2 = min(0.2 * phase4_budget, 20.0)
        if t_coord2 > 3:
            a_opt, _ = coordinate_polish(a_opt, time_limit=t_coord2)
            try_seq(a_opt, "P4_coord2")

    print(f"  Phase 4 done: best={best_score:.8f} ({time.monotonic()-t_phase_start:.0f}s)")

    # === Phase 5: Final safety check ===
    if best_seq is None:
        best_seq = np.ones(384) / 384.0
        best_score = safe_eval(best_seq)

    # Verify with actual evaluator
    try:
        final_check = evaluate_sequence(best_seq.tolist())
        if np.isfinite(final_check) and final_check < best_score:
            best_score = final_check
        elif not np.isfinite(final_check):
            # Fallback
            best_seq = np.ones(384) / 384.0
            best_score = safe_eval(best_seq)
    except Exception as e:
        print(f"Evaluation check failed: {e}")

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={elapsed:.0f}s")

    return [float(max(0.0, x)) for x in best_seq]
