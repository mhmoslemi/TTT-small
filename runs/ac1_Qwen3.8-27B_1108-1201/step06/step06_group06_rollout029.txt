def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    cols = []

            rows.append(k)

"""
Optimized search for minimum peak autocorrelation step function.
Uses FW-LP with precomputed sparse structure, top-k smoothed gradient,
peak gradient descent, mirror descent, and local search with multi-n exploration.
Key improvements: faster LP oracle, top-k peak penalization, more aggressive
multi-n exploration, and adaptive time allocation.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, coo_matrix
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

class LPOracle:
    """Precomputed LP oracle for FW step."""
    def __init__(self, n):
        self.n = n
        self.rows = []
        self.cols = []
        for k in range(2 * n - 1):
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            for i in range(i_start, i_end + 1):
                self.rows.append(k)
                self.cols.append(k - i)
        self.rows = np.array(self.rows)
        self.cols = np.array(self.cols)
        self.c = -np.ones(n)

    def solve(self, f, M, time_limit=None):
        n = self.n
        vals = f[self.rows - self.cols]
        A = coo_matrix((vals, (self.rows, self.cols)), shape=(2 * n - 1, n)).tocsr()
        b = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        opts = {'presolve': True}
        if time_limit is not None:
            opts['time_limit'] = time_limit
        try:
            res = linprog(self.c, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs', options=opts)
            if res.success:
                g = np.maximum(res.x, 0.0)
                s = g.sum()
                if s > 1e-15:
                    return g / s
        except Exception:
            pass
        return None

            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = np.maximum((1 - m1) * f + m1 * g, 0)
            f1 = f1 / f1.sum()
            v1 = fast_convolve(f1).max()
            f2 = np.maximum((1 - m2) * f + m2 * g, 0)
            f2 = f2 / f2.sum()
            v2 = fast_convolve(f2).max()
            if v1 < v2:
                hi = m2
            else:
                lo = m1
        mid = (lo + hi) / 2.0
        ft = np.maximum((1 - mid) * f + mid * g, 0)
        ft = ft / ft.sum()
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = mid
    return best_alpha, best_val

def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with precomputed LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    oracle = LPOracle(n)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        g = oracle.solve(f, M * m_exp, time_limit=min(remaining * 0.1, 30))
        if g is None:
            m_exp = min(m_exp * 1.3, 5.0)
            if m_exp >= 5.0:
                rng = np.random.default_rng(it * 31 + 7)
                g = project_simplex(f + rng.standard_normal(n) * 0.01)
                m_exp = 1.0
            else:
                continue

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 60:
                rng = np.random.default_rng(it * 7919 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        best_alpha, best_val = fw_line_search(f, g, M)

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 200 == 0:
                    print(f"    FW it={it}, obj={best_obj:.8f}, M={M:.8f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

    return best_f, best_obj

def compute_smoothed_grad(a, p):
    """Compute gradient of sum_k p[k]*c[k] w.r.t. a.
    grad[j] = 2 * sum_k p[k] * a[k-j] (for valid k-j)
    """
    n = len(a)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-10:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        idx = j_start + np.arange(j_end - j_start + 1)
        grad[idx] += 2.0 * p[k] * a[k - idx]
    return grad

def topk_smoothed_grad(a, k, T):
    """Compute gradient using top-k peak weighting with softmax falloff."""
    n = len(a)
    conv = fast_convolve(a)
    # Get top k indices
    top_idx = np.argsort(conv)[::-1][:k]
    # Softmax weighting within top-k
    vals = conv[top_idx] / T
    vals -= vals.max()
    w = np.exp(vals)
    w = w / w.sum()
    # Build p
    p = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_idx):
        p[idx] = w[i]
    return compute_smoothed_grad(a, p), p

def peak_grad_descent(a0, time_limit, verbose=False, use_topk=True, k=5, T=0.001):
    """Gradient descent with peak or top-k smoothed objective."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    lr = 0.05
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        if use_topk:
            grad, _ = topk_smoothed_grad(a, min(k, 2 * n - 1), T)
        else:
            conv = fast_convolve(a)
            k_peak = int(np.argmax(conv))
            grad = np.zeros(n)
            j_start = max(0, k_peak - n + 1)
            j_end = min(k_peak, n - 1)
            for j in range(j_start, j_end + 1):
                grad[j] = 2.0 * a[k_peak - j]

        best_candidate = a
        best_val = fast_convolve(a).max()
        for trial_lr in [lr, lr * 0.5, lr * 0.1, lr * 0.05, lr * 0.01]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    PGD it={it}, obj={best_obj:.8f}")
        else:
            no_improve += 1

        if no_improve > 30:
            lr *= 0.92
        if no_improve > 100:
            T *= 1.1
            k = min(k + 1, 2 * n - 1)
        if no_improve > 300:
            rng = np.random.default_rng(it * 2654435761)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            no_improve = 0
            lr = 0.05
            T = 0.001
            k = 5

    return best_a, best_obj

def mirror_descent(a0, time_limit, verbose=False):
    """Mirror descent with multiplicative weight updates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    eta = 0.08
    T = 0.002
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = compute_smoothed_grad(a, p)

        log_a = np.log(a + 1e-30)
        new_log_a = log_a - eta * grad
        new_log_a -= new_log_a.max()
        a_new = np.exp(new_log_a)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj - 1e-15:
            a = a_new
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    MD it={it}, obj={best_obj:.8f}")
        else:
            no_improve += 1

        if no_improve > 20:
            eta *= 0.93
        if no_improve > 80:
            T = max(T * 1.1, 0.0005)
        if no_improve > 250:
            rng = np.random.default_rng(it * 104729 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            no_improve = 0
            eta = 0.08
            T = 0.002

    return best_a, best_obj

def local_search(a0, time_limit, seed=42):
    """Local search with coordinate moves and structured perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        improved = False

        # Coordinate moves (move mass from high to low positions)
        for attempt in range(3):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            hi, lo = (i, j) if a[i] > a[j] else (j, i)
            for frac in [0.02, 0.05, 0.1, 0.15, 0.25, 0.4]:
                candidate = a.copy()
                amount = candidate[hi] * frac
                candidate[hi] -= amount
                candidate[lo] += amount
                candidate = np.maximum(candidate, 0)
                ss = candidate.sum()
                if ss < 1e-10:
                    continue
                candidate = candidate / ss
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Random perturbation
        if not improved:
            for scale in [0.0003, 0.0008, 0.002, 0.005, 0.01, 0.02]:
                pert = rng.standard_normal(n) * scale
                candidate = project_simplex(a + pert)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break

        # Structured: sinusoidal perturbation
        if not improved and it % 5 == 0:
            freq = rng.uniform(0.5, 6.0)
            phase = rng.uniform(0, 2 * np.pi)
            pattern = np.sin(2 * np.pi * freq * np.arange(n) / n + phase) * 0.003
            candidate = project_simplex(a + pattern)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True

        if not improved and it > 3000:
            break

    return best_a, best_obj

def try_seq(a, label, best_ref):
    """Evaluate a sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_ref
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_ref[1]:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return (a_norm.copy(), score)
    return best_ref

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid search for minimum peak autocorrelation.
    Phase 1: Multi-n quick scan
    Phase 2: Deep FW-LP + perturbed restarts
    Phase 3: Top-k gradient + mirror descent
    Phase 4: Asymmetric structure exploration
    Phase 5: Final polish (local search, fine-tuning)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Using uniform init: score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Multi-n Quick Scan ===
    print("\n=== Phase 1: Multi-n Quick Scan ===")
    t_phase = time.monotonic()

    candidate_ns = [256, 384, 512, 640, 768, 896, 1024, 1280, 1536, 2048, 2560, 3072]
    if n_parent not in candidate_ns:
        candidate_ns.append(n_parent)
    candidate_ns = sorted(set([x for x in candidate_ns if 128 <= x <= 3072]))

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic()
        if remaining < 250:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            # Interpolate parent to new length
            x_orig = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            start = np.interp(x_new, x_orig, parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try

        # Quick peak gradient
        t_alloc = min(remaining * 0.025, 12)
        a_opt, _ = peak_grad_descent(start, time_limit=t_alloc, use_topk=True, k=3, T=0.002)
        best_seq, best_score = try_seq(a_opt, f"pgd_n{n_try}", (best_seq, best_score))

        # Quick FW if time allows
        remaining2 = deadline - time.monotonic()
        if remaining2 > 200 and n_try <= 1536:
            t_alloc2 = min(remaining2 * 0.02, 10)
            a_opt2, _ = fw_optimize(a_opt, time_limit=t_alloc2)
            best_seq, best_score = try_seq(a_opt2, f"fw_n{n_try}", (best_seq, best_score))

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW-LP ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 300:
        t_alloc = min(remaining * 0.3, 500)
        print(f"  Deep FW n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
        best_seq, best_score = try_seq(a_opt, "deep_fw", (best_seq, best_score))

    # Perturbed FW restarts at different scales
    for scale in [0.002, 0.005, 0.01, 0.02, 0.04]:
        remaining = deadline - time.monotonic()
        if remaining < 120:
            break
        n_best = len(best_seq)
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc2 = min(remaining * 0.1, 100)
        a_opt2, _ = fw_optimize(f_pert, time_limit=t_alloc2)
        best_seq, best_score = try_seq(a_opt2, f"fw_pert_{scale}", (best_seq, best_score))

    # Fresh FW from parent at parent's n
    remaining = deadline - time.monotonic()
    if remaining > 200 and n_best != n_parent:
        t_alloc3 = min(remaining * 0.15, 150)
        print(f"  FW from parent n={n_parent}, t={t_alloc3:.0f}s")
        a_opt3, _ = fw_optimize(parent, time_limit=t_alloc3, verbose=True)
        best_seq, best_score = try_seq(a_opt3, "fw_parent_n", (best_seq, best_score))

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Top-k Gradient + Mirror Descent ===
    print("\n=== Phase 3: Advanced Gradient Methods ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    # Top-k gradient with different k values
    for k_val in [3, 5, 10, 20]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        t_alloc = min(remaining * 0.1, 80)
        print(f"  Top-k grad (k={k_val}), t={t_alloc:.0f}s")
        a_opt, _ = peak_grad_descent(best_seq, time_limit=t_alloc,
                                     use_topk=True, k=k_val, T=0.001, verbose=True)
        best_seq, best_score = try_seq(a_opt, f"topk{k_val}", (best_seq, best_score))

    # Mirror descent
    remaining = deadline - time.monotonic()
    if remaining > 80:
        t_alloc = min(remaining * 0.15, 100)
        print(f"  Mirror descent, t={t_alloc:.0f}s")
        a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc, verbose=True)
        best_seq, best_score = try_seq(a_opt, "mirror", (best_seq, best_score))

    # Mirror descent from perturbed start
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_best = len(best_seq)
        for scale in [0.003, 0.01, 0.03]:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            a_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            a_opt, _ = mirror_descent(a_pert, time_limit=min(remaining * 0.12, 40))
            best_seq, best_score = try_seq(a_opt, f"mirror_p{scale}", (best_seq, best_score))

    # Peak gradient (single peak focus)
    remaining = deadline - time.monotonic()
    if remaining > 50:
        t_alloc = min(remaining * 0.1, 60)
        print(f"  Single peak grad, t={t_alloc:.0f}s")
        a_opt, _ = peak_grad_descent(best_seq, time_limit=t_alloc,
                                     use_topk=False, T=0.001)
        best_seq, best_score = try_seq(a_opt, "single_peak", (best_seq, best_score))

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Asymmetric Structure Exploration ===
    print("\n=== Phase 4: Structure Exploration ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 100:
        x = np.arange(n_best, dtype=np.float64)
        structures = []

        # Exponential decay (strongly asymmetric)
        for rate in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08]:
            exp_d = np.exp(-rate * x)
            structures.append((f"exp_{rate}", exp_d / exp_d.sum()))

        # Power law
        for alpha in [0.3, 0.5, 0.7, 1.0, 1.5]:
            pw = (x + 1.0) ** (-alpha)
            structures.append((f"pw_{alpha}", pw / pw.sum()))

        # Sine positive part (varied phases)
        for phase in [0.0, 0.3, 0.5, 0.7]:
            sin_s = np.maximum(np.sin(np.pi * x / n_best + phase * np.pi), 0)
            if sin_s.sum() > 1e-10:
                structures.append((f"sin_{phase}", sin_s / sin_s.sum()))

        # Asymmetric Gaussian pair
        g1 = np.exp(-0.5 * (x / (0.15 * n_best)) ** 2)
        g2 = np.exp(-0.5 * ((x - 0.8 * n_best) / (0.1 * n_best)) ** 2)
        structures.append(("asym_gauss", (g1 + g2) / (g1 + g2).sum()))

        # "Cliff" structure: high then sharp drop
        cliff = np.ones(n_best) * 0.1
        cliff[:n_best // 3] = 1.0
        cliff[n_best // 3:] *= np.linspace(1.0, 0.01, n_best - n_best // 3)
        structures.append(("cliff", cliff / cliff.sum()))

        si = 0
        for name, struct in structures:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            for alpha in [0.4, 0.7, 0.9]:
                remaining = deadline - time.monotonic()
                if remaining < 30:
                    break
                blend = project_simplex((1 - alpha) * best_seq + alpha * struct)
                t_alloc = min(remaining * 0.04, 12)
                a_opt, _ = peak_grad_descent(blend, time_limit=t_alloc,
                                             use_topk=True, k=5, T=0.002)
                best_seq, best_score = try_seq(a_opt, f"{name}_a{alpha}", (best_seq, best_score))
            si += 1

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final Polish ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    # Final FW pass
    if remaining > 60:
        t_alloc = min(remaining * 0.2, 80)
        print(f"  Final FW, t={t_alloc:.0f}s")
        a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, "final_fw", (best_seq, best_score))

    # Local search
    remaining = deadline - time.monotonic()
    if remaining > 40:
        t_alloc = min(remaining * 0.25, 70)
        print(f"  Local search, t={t_alloc:.0f}s")
        a_opt, _ = local_search(best_seq, time_limit=t_alloc, seed=12345)
        best_seq, best_score = try_seq(a_opt, "local1", (best_seq, best_score))

    # Fine top-k gradient (very fine)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_alloc = min(remaining * 0.2, 40)
        print(f"  Fine topk grad, t={t_alloc:.0f}s")
        a_opt, _ = peak_grad_descent(best_seq, time_limit=t_alloc,
                                     use_topk=True, k=3, T=0.0003)
        best_seq, best_score = try_seq(a_opt, "fine_topk", (best_seq, best_score))

    # Mirror descent final
    remaining = deadline - time.monotonic()
    if remaining > 25:
        t_alloc = min(remaining * 0.15, 30)
        a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, "final_mirror", (best_seq, best_score))

    # Micro perturbations
    remaining = deadline - time.monotonic()
    if remaining > 15:
        for _ in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 2:
                break
            noise = rng.standard_normal(n_best) * 0.0003
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = peak_grad_descent(f_pert, time_limit=min(remaining * 0.15, 3),
                                         use_topk=True, k=3, T=0.0005)
            best_seq, best_score = try_seq(a_opt, "micro", (best_seq, best_score))

    # Another local search
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_opt, _ = local_search(best_seq, time_limit=min(remaining * 0.3, 15), seed=77777)
        best_seq, best_score = try_seq(a_opt, "local2", (best_seq, best_score))

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(768) / 768
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
