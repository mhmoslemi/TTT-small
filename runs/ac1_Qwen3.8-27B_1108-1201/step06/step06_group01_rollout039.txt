    n = len(v)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed-max gradient descent
with FFT-based gradients, multi-scale n exploration, and coordinate descent polish.

Key insight: The objective is equivalent to finding a probability distribution p
on {0,...,n-1} that minimizes the peak of the distribution of X+Y where X,Y
are i.i.d. with distribution p. We use smoothed L_infinity surrogate gradients.
"""

import numpy as np
from scipy.signal import correlate as sp_correlate
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    # Clean up tiny negative values from FFT roundoff
    c[np.abs(c) < 1e-15] = 0.0
    return c


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_obj(a, T):
    """
    Compute gradient of L_T(a) = T*log(sum exp(c[k]/T)) where c=conv(a,a).
    Also returns the actual max of c.
    """
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    
    # Softmax weights (numerically stable)
    c_scaled = c / T
    c_scaled -= c_scaled.max()
    exp_c = np.exp(c_scaled)
    Z = exp_c.sum()
    w = exp_c / Z
    
    # First term: correlate(w, a, 'valid') gives sum_j w[m+j]*a[j] for m=0..n-1
    first = sp_correlate(w, a, mode='valid')
    
    # Second term: w[2m]*a[m] for m=0..n-1
    second = w[0:2*n-1:2] * a
    
    grad = (first + second) / T
    return grad, c_max, c

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed-max gradient descent
with FFT-based gradients, Nesterov acceleration, multi-scale n exploration,
and coordinate descent polish.

Key insight: The objective (with sum(a)=1) equals 2n*max(conv(a,a)), which is
2n times the peak of the sum-distribution of two i.i.d. draws from the
probability distribution a. We minimize the L_infinity norm of the
auto-convolution using a log-sum-exp surrogate with annealed temperature.
"""

import numpy as np
from scipy.signal import correlate as sp_correlate
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    c[np.abs(c) < 1e-15] = 0.0
    return c


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad(a, T):
    """
    Compute gradient of L_T(a) = T*log(sum_k exp(c[k]/T)) where c=conv(a,a).
    Returns (gradient, c_max, c).
    Gradient formula: (2/T) * correlate(w, a, 'valid') where w = softmax(c/T).
    """
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()

    c_scaled = c / T
    c_scaled -= c_scaled.max()
    exp_c = np.exp(c_scaled)
    Z = exp_c.sum()
    w = exp_c / Z

    grad = (2.0 / T) * sp_correlate(w, a, mode='valid')
    return grad, c_max, c

    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    T = T_init
    t0 = time.monotonic()
    no_improve = 0
    total_it = 0

    y = a.copy()
    v = np.zeros(n)
    t_nest = 1.0

    while time.monotonic() - t0 < time_limit - 0.05:
        total_it += 1

        if use_nesterov:
            alpha_n = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
            beta = (t_nest - 1.0) / alpha_n
            y_new = (1.0 + beta) * a - beta * y
            y_new = np.maximum(y_new, 0.0)
            ys = y_new.sum()
            if ys < 1e-15:
                y = a.copy()
            else:

            t_nest += 1.0
        else:

        found = False

            if use_nesterov:
                a_trial = project_simplex(a + beta * v - alpha * grad)
            else:
                a_trial = project_simplex(a - alpha * grad)

            c_trial = fast_convolve(a_trial)
            new_max = c_trial.max()

            if new_max < c_y_max - 1e-13:
                if use_nesterov:
                    v = beta * v - alpha * grad
                a = a_trial
                c_y_max = new_max
                found = True
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break

        if found:
            if c_y_max < best_obj - 1e-14:
                best_obj = c_y_max
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            if use_nesterov:
                v *= beta  # decay momentum when no step taken

        # Anneal temperature
        if no_improve > 40:
            T *= 0.75
            if T < T_end:
                break
            no_improve = 0
            lr_init *= 0.85

        # Restart if very stuck
        if no_improve > 200:
            rng = np.random.default_rng(total_it * 7919 + 13)
            noise_scale = 0.002 * min(3.0, no_improve / 100.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            y = a.copy()
            v = np.zeros(n)
            t_nest = 1.0
            no_improve = 0
            T = max(T, T_end * 3)

    return best_a, best_obj


def nesterov_subgrad(a0, time_limit, lr_init=0.003, top_k=30):
    """
    Nesterov-accelerated subgradient method using top-k peak weighting.
    More aggressive than smoothed descent for late-stage refinement.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    y = a.copy()
    v = np.zeros(n)
    t_nest = 1.0
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit - 0.05:
        it += 1

        alpha_n = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / alpha_n
        y_new = (1.0 + beta) * a - beta * y
        y_new = np.maximum(y_new, 0.0)
        ys = y_new.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y_new / ys
        t_nest += 1.0

        conv = fast_convolve(y)
        k_eff = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_eff)[-k_eff:]
        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.3
        ws = weights.sum()
        weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += 2.0 * weights[i] * y[k - idx_arr]

        v = beta * v - lr * sg
        a_new = project_simplex(a + v)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            a_new = a

        a = a_new
        lr *= (1.0 - 0.0003)

        if no_improve > 80:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            ns = 0.001 * min(3.0, no_improve / 50.0)
            a = project_simplex(best_a + rng.standard_normal(n) * ns)
            y = a.copy()
            v = np.zeros(n)
            t_nest = 1.0
            no_improve = 0

    return best_a, best_obj


def coordinate_polish(a0, time_limit):
    """
    Pairwise coordinate descent targeting the peak lag.
    Tries to reduce a[j]*a[k-j] products at the peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_convolve(a).max()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    c = fast_convolve(a)

    t0 = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit - 0.05:
        it += 1
        peak_k = int(np.argmax(c))
        js_start = max(0, peak_k - n + 1)
        js_end = min(n - 1, peak_k)

        improved = False

        # Try pairwise adjustments among elements contributing to the peak
        for j in range(js_start, js_end + 1):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            partner = peak_k - j
            if partner < 0 or partner >= n or partner == j:
                continue
            if a[j] < 1e-10 or a[partner] < 1e-10:
                continue

            # Try moving mass between j and partner to reduce their product
            # while preserving sum
            for frac in [0.05, 0.1, 0.2, 0.3, 0.45]:
                delta = min(a[j], a[partner]) * frac
                # Direction 1: move from j to partner
                a_t = a.copy()
                a_t[j] -= delta
                a_t[partner] += delta
                a_t = np.maximum(a_t, 0)
                st = a_t.sum()
                if st > 1e-15:
                    a_t = a_t / st
                    c_t = fast_convolve(a_t)
                    if c_t.max() < best_obj - 1e-13:
                        a = a_t
                        c = c_t
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break
                # Direction 2: move from partner to j
                a_t = a.copy()
                a_t[partner] -= delta
                a_t[j] += delta
                a_t = np.maximum(a_t, 0)
                st = a_t.sum()
                if st > 1e-15:
                    a_t = a_t / st
                    c_t = fast_convolve(a_t)
                    if c_t.max() < best_obj - 1e-13:
                        a = a_t
                        c = c_t
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break

        # Also try: move mass from a peak-contributing element to a
        # non-contributing element
        if not improved and it % 3 == 0:
            for j in range(js_start, js_end + 1):
                if time.monotonic() - t0 > time_limit - 0.5:
                    break
                if a[j] < 1e-8:
                    continue
                # Find a target that doesn't contribute to the peak
                for target in range(0, n):
                    if target == j:
                        continue
                    tp = peak_k - target
                    if 0 <= tp < n and tp != target:
                        continue  # this target also contributes to peak
                    delta = a[j] * 0.1
                    a_t = a.copy()
                    a_t[j] -= delta
                    a_t[target] += delta
                    a_t = np.maximum(a_t, 0)
                    st = a_t.sum()
                    if st > 1e-15:
                        a_t = a_t / st
                        c_t = fast_convolve(a_t)
                        if c_t.max() < best_obj - 1e-13:
                            a = a_t
                            c = c_t
                            best_obj = c.max()
                            best_a = a.copy()
                            improved = True
                            break

        if improved:
            stall = 0
        else:
            stall += 1

        if stall > 80:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            c = fast_convolve(a)
            stall = 0

        if stall > 400:
            break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: multi-phase hybrid optimization combining smoothed gradient
    descent, Nesterov subgradient, and coordinate polish across multiple n.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t0_total
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return score

    t0_total = time.monotonic()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Parent load failed: {e}")

    n_parent = len(parent) if parent is not None else 768

    # === Phase 1: Multi-n exploration with fast smoothed descent (~350s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    n_candidates = sorted(set(
        [n_parent] + list(range(384, 1537, 64))
    ))
    n_candidates = [x for x in n_candidates if 256 <= x <= 1600]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 1100:
            break

        if n_try == n_parent and parent is not None:
            a_init = parent.copy()
        elif parent is not None:

                                np.linspace(0, 1, n_parent), parent)
            a_init = np.maximum(a_init, 0)
            ss = a_init.sum()
            a_init = a_init / ss if ss > 1e-10 else np.ones(n_try) / n_try
        else:
            a_init = np.ones(n_try) / n_try

            break

        a_opt, _ = smoothed_descent(a_init, time_limit=t_alloc,
                                     T_init=0.08, T_end=0.002, lr_init=0.15)
        try_seq(a_opt, f"p1_n{n_try}")
