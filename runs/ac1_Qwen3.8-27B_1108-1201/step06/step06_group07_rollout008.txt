"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- FFT-based spectral shaping for good initializations
- Improved SCA with adaptive M and golden-section line search
- Subgradient descent with softmax peak weights

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_from_convolve(conv):
    """Compute objective from convolution (assuming sum(a)=1)."""
    n_half = (len(conv) + 1) // 2
    return 2.0 * n_half * conv.max()


def spectral_shaping(a, max_iters=500, alpha_start=0.90, alpha_end=0.65):
    """Flatten the magnitude spectrum to reduce convolution peak."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    for it in range(max_iters):
        alpha = alpha_start + (alpha_end - alpha_start) * it / max(1, max_iters - 1)
        A = np.fft.rfft(a, 2 * n - 1)
        mag = np.abs(A).copy()
        phase = np.angle(A)
        mag_mean = mag.mean()
        if mag_mean < 1e-15:
            break
        # Compress dynamic range: new_mag = mag^alpha * mean^(1-alpha)
        mag_new = np.power(mag, alpha) * np.power(mag_mean, 1.0 - alpha)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, 2 * n - 1)[:n]
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def build_lp_indices(n):
    """Build sparse matrix indices for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_oracle(f, M, n, k_arr, i_arr):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    vals = f[k_arr - i_arr]
    A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def subgradient_refine(a0, time_limit, temperature=0.01):
    """Subgradient descent with softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = 0.03

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)
        max_val = conv.max()

        # Softmax-weighted subgradient over all peaks
        scaled = conv / (temperature * max_val + 1e-15)
        exp_scaled = np.exp(scaled - scaled.max())
        weights = exp_scaled / exp_scaled.sum()

        # Full subgradient
        grad = np.zeros(n)
        for k in range(len(conv)):
            w = weights[k]
            if w < 1e-8:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += w * 2.0 * a[k - j]

        # Update
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        # Line search
        best_candidate = a
        best_val = max_val
        for alpha in [0.3, 0.5, 0.7, 1.0]:
            cand = (1 - alpha) * a + alpha * a_new
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_candidate = cand

        a = best_candidate
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            lr *= 0.9
        if lr < 0.001:
            lr = 0.03
            no_improve = 0

        if no_improve > 100:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31 - 1))
            a = np.maximum(best_a + rng.standard_normal(n) * 0.002, 0)
            a = a / a.sum()
            lr = 0.03
            no_improve = 0

    return best_a, best_obj


def coordinate_descent(a0, time_limit, max_rounds=200):
    """Coordinate descent: optimize each element (or small group) at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        # Random order of coordinates
        order = rng.permutation(n)

        for idx_i in order:
            if time.monotonic() - t0 > time_limit - 1.0:
                break
            i = int(idx_i)

            # Try a few values for a[i]
            # Current value
            ai = a[i]

            # Candidate values: currently optimal region
            # The contribution of a[i] to conv[k] for various k:
            # conv[k] includes a[i]*a[k-i] for valid k-i
            # We want to reduce the peak, so try reducing a[i] slightly
            # or increasing it (if it helps distribute mass)

            candidates = []
            if ai > 1e-6:
                candidates.extend([ai * 0.5, ai * 0.7, ai * 0.9, ai * 1.1, ai * 1.3])
            candidates.extend([0.0, ai * 2.0])
            # Also try values based on neighbors
            if i > 0:
                candidates.append(a[i-1] * 0.8)
            if i < n - 1:
                candidates.append(a[i+1] * 0.8)

            best_cand = ai
            best_cand_obj = best_obj

            current_peak = fast_convolve(a).max()

            for cand_val in candidates:
                cand_val = max(0.0, cand_val)
                a_test = a.copy()
                # Renormalize: adjust a[i] and scale rest
                diff = cand_val - ai
                if diff == 0:
                    continue
                # Simply set a[i] = cand_val and normalize
                a_test[i] = cand_val
                s_test = a_test.sum()
                if s_test < 1e-15:
                    continue
                a_test = a_test / s_test
                obj_test = fast_convolve(a_test).max()
                if obj_test < best_cand_obj - 1e-12:
                    best_cand_obj = obj_test
                    best_cand = cand_val

            if best_cand != ai:
                a[i] = best_cand
                a = a / a.sum()
                obj = fast_convolve(a).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()

    return best_a, best_obj


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    t = x / max(1, n - 1)

    # From parent (interpolated)
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Perturbed parent
        px2 = np.maximum(px + rng.standard_normal(n) * 0.02, 0)
        ps2 = px2.sum()
        if ps2 > 1e-10:
            inits.append(px2 / ps2)

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge-heavy
    edge_w = 1.0 / (np.minimum(x, n - 1 - x).astype(float) + 1.0) ** 1.2
    inits.append(edge_w / edge_w.sum())

    # Cosine bump
    cos_bump = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    cos_bump = np.maximum(cos_bump, 0.01)
    inits.append(cos_bump / cos_bump.sum())

    # Two-sided (low in middle)
    two_side = np.exp(-((t - 0.25)**2) / 0.02) + np.exp(-((t - 0.75)**2) / 0.02) + 0.01
    inits.append(two_side / two_side.sum())

    # Sine modulation
    sine = (1.0 + 0.3 * np.sin(4 * np.pi * t)) ** 2 + 0.05
    inits.append(sine / sine.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    return inits

            try_seq(a_shape, f"multi_n{n_alt}_spec")

            t_alloc = min(60, remaining - 30)
            if t_alloc > 20:
                a_opt, _ = sca_optimize(a_shape, time_limit=t_alloc)
                try_seq(a_opt, f"multi_n{n_alt}")

    print(f"  P6 done: best={best_score:.8f}, t={time.monotonic()-t6:.0f}s")

    # === Final: One more coordinate descent on best ===
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            a_final, _ = coordinate_descent(best_seq, time_limit=min(remaining, 30), max_rounds=50)
            try_seq(a_final, "final_coord2")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
