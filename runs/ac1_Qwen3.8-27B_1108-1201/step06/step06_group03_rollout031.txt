def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 for normalized a."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_subgradient(a0, time_limit, T_init=0.001, lr_init=0.01, top_k_frac=0.05, mom=0.9):

                            cs = candidate.sum()
                            if cs < 1e-15:

                            score_new = 2.0 * n * val / (1.0)

                                best_a = a.copy()

                        break

                    break

        # Compute autocorrelation
        conv = fast_convolve(a)

        # In frequency domain: |A(ω)|² = DFT(R) where R is the circular autocorrelation
        # We want to reduce peaks in R, i.e., make |A(ω)|² flatter
        # Take DFT of the centered autocorrelation
        fft_size = 1 << (2 * n).bit_length()
        R_ext = np.zeros(fft_size)
        R_ext[:2 * n - 1] = conv
        spec = np.fft.rfft(R_ext)

        # Damp peaks: reduce large spectral values
        spec_abs = np.abs(spec)
        spec_max = spec_abs.max()
        if spec_max < 1e-15:
            break
        # Soft thresholding
        threshold = 0.95 * spec_max
        spec_new = spec.copy()
        mask = spec_abs > threshold
        spec_new[mask] = spec_new[mask] * 0.8 + np.sign(spec[mask]) * threshold * 0.2

        # Inverse DFT to get modified autocorrelation
        R_new = np.fft.irfft(spec_new, fft_size)[:2 * n - 1]
        R_new = np.maximum(R_new, 0)

        # Now we need to find a non-negative a such that conv(a,a) ≈ R_new
        # Use iterative approach: a_new = project(a + eta * (R_new - conv) correlated back)
        # Subgradient of (1/2)||conv(a,a) - R_new||² w.r.t. a
        diff = conv - R_new
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if abs(diff[k]) < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += diff[k] * (a[k - idx_arr] + a[idx_arr])

        # Gradient step with projection
        lr = 0.01 / (np.max(np.abs(grad)) + 1e-15)
        a_new = project_simplex(a - lr * grad)

        score_new = compute_score(a_new)
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def fw_lp_oracle(f, M, n):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    stagnant = 0
    it = 0
    m_relax = 1.005

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        g = fw_lp_oracle(f, M * m_relax, n)
        if g is None or g.sum() < 1e-15:
            g = fw_lp_oracle(f, M * 1.02, n)

        if g is None or g.sum() < 1e-15:
            stagnant += 1
            if stagnant > 30:
                rng = np.random.default_rng(it * 7919 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                M = fast_convolve(f).max()
                m_relax = 1.005
                stagnant = 0
            continue

        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-10:
            stagnant += 1
            if stagnant > 40:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                M = fast_convolve(f).max()
                stagnant = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = M + 1e-14
        for alpha in np.linspace(0.02, 1.0, 25):
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-8:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        if stagnant > 15:
            m_relax = min(m_relax * 1.03, 1.05)

        if verbose and it % 50 == 0:
            print(f"    FW it={it}, obj={best_obj:.10f}, stag={stagnant}")

    return best_f, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length via interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Multi-phase exploration to minimize
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    # === Try uniform sequences ===
    print("\n=== Uniform sequences ===")
    for n_try in [128, 256, 512, 768, 1024, 2048]:
        uniform = np.ones(n_try) / n_try
        try_seq(uniform, f"uniform_{n_try}")

    # Set initial best from uniform if nothing better
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback uniform: {best_score:.10f}")

    if parent is None:
        parent = best_seq.copy()

    n_parent = len(parent)

    # === Phase 1: Short exploration across lengths (120s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_phase = time.monotonic()

    lengths = sorted(set([n_parent, max(256, n_parent - 128), n_parent + 128, 384, 512, 768, 1024, 1536]))
    lengths = [l for l in lengths if 128 <= l <= 2048]

    for n_try in lengths:
        if time.monotonic() > deadline - 1400:
            break

        # Try a few different starts
        starts = [resample_sequence(parent, n_try), np.ones(n_try) / n_try]
        # Cosine taper
        x = np.arange(n_try, dtype=np.float64)
        cos_w = 0.5 * (1 + np.cos(np.pi * x / (n_try - 1)))
        cos_w = np.maximum(cos_w, 0.01)
        starts.append(cos_w / cos_w.sum())
        # Gaussian
        g = np.exp(-0.5 * ((x - (n_try - 1) / 2) / (0.3 * n_try)) ** 2)
        starts.append(g / g.sum())
        # Random
        starts.append(rng.dirichlet(np.ones(n_try) * 0.5))

        for si, start in enumerate(starts[:3]):
            if time.monotonic() > deadline - 1350:
                break
            t_alloc = min(15.0, deadline - time.monotonic() - 1300)
            if t_alloc < 5:
                break
            a_opt = smoothed_subgradient(start, time_limit=t_alloc, T_init=0.001, lr_init=0.01)
            try_seq(a_opt[0], f"sg_n{n_try}_s{si}")

    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep optimization on best (500s) ===
    print("\n=== Phase 2: Deep Optimization ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Deep smoothed subgradient with multiple configs
        configs = [
            {'T_init': 0.0005, 'lr_init': 0.008, 'top_k_frac': 0.03, 'mom': 0.85},
            {'T_init': 0.001, 'lr_init': 0.012, 'top_k_frac': 0.05, 'mom': 0.9},
            {'T_init': 0.003, 'lr_init': 0.006, 'top_k_frac': 0.1, 'mom': 0.95},
            {'T_init': 0.0002, 'lr_init': 0.004, 'top_k_frac': 0.02, 'mom': 0.8},
        ]

        for ci, cfg in enumerate(configs):
            if time.monotonic() > deadline - 700:
                break
            t_alloc = min(120.0, deadline - time.monotonic() - 650)
            if t_alloc < 10:
                break
            print(f"  Subgrad config {ci}: T={cfg['T_init']}, lr={cfg['lr_init']}, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_subgradient(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"deep_sg_{ci}")

        # FW on best
        if time.monotonic() < deadline - 400:
            t_alloc = min(200.0, deadline - time.monotonic() - 380)
            if t_alloc > 20:
                print(f"  FW on best, t={t_alloc:.0f}s")
                a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc)
                try_seq(a_opt, "deep_fw")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Peak coordinate descent (200s) ===
    print("\n=== Phase 3: Peak Coordinate Descent ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(200.0, deadline - time.monotonic() - 250)
        if t_alloc > 10:
            print(f"  Peak CD, t={t_alloc:.0f}s")
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_cd")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Spectral flattening (150s) ===
    print("\n=== Phase 4: Spectral Flattening ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(150.0, deadline - time.monotonic() - 200)
        if t_alloc > 10:
            print(f"  Spectral, t={t_alloc:.0f}s")
            a_opt, _ = spectral_flatten(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "spectral")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Micro-polish with tiny perturbations (remaining) ===
    print("\n=== Phase 5: Micro-Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        while time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic() - 30
            if remaining < 5:
                break

            # Try tiny perturbation + short optimization
            noise_scale = 0.0005
            perturbed = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)

            # Quick subgradient
            t_alloc = min(20.0, remaining * 0.4)
            a_opt, _ = smoothed_subgradient(perturbed, time_limit=t_alloc,
                                            T_init=0.0003, lr_init=0.003,
                                            top_k_frac=0.03, mom=0.85)
            try_seq(a_opt, "polish_sg")

            # Quick coordinate descent
            t_alloc = min(15.0, remaining * 0.3)
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "polish_cd")

            # Try slightly different lengths
            if remaining > 50:
                for n_try in [n_best - 32, n_best + 32]:
                    if n_try < 64 or n_try > 4096:
                        continue
                    resampled = resample_sequence(best_seq, n_try)
                    t_alloc = min(10.0, remaining * 0.2)
                    a_opt, _ = smoothed_subgradient(resampled, time_limit=t_alloc,
                                                    T_init=0.0005, lr_init=0.005,
                                                    top_k_frac=0.04)
                    try_seq(a_opt, f"polish_n{n_try}")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
