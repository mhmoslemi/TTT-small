"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer with temperature-annealed smooth max (log-sum-exp)
2. Active-set identification and direct KKT Newton solve
3. Frank-Wolfe with LP oracle for global moves

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_obj_and_grad(a, T):
    """Compute smooth objective and gradient for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    # Smooth max
    conv_shifted = (conv - conv.max()) / T
    log_sum_exp = conv.max() + T * np.log(np.sum(np.exp(conv_shifted)))
    obj = 2.0 * n * log_sum_exp
    # Weights for the smooth max
    w = np.exp(conv_shifted)
    w /= w.sum()
    # Gradient: 4n * correlate(w, a, 'valid')
    grad = 4.0 * n * np.correlate(w, a, mode='valid')
    return obj, grad, conv

def adam_smooth(a0, time_limit, T_init=0.01, T_min=0.0001, T_decay=0.999,
                lr_init=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer with annealed temperature for smooth max."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    T = T_init
    lr = lr_init
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max() * 2 * n
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        
        obj, grad, conv = smooth_obj_and_grad(a, T)
        true_obj = 2.0 * n * conv.max()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Step in negative gradient direction, project to simplex
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        candidate = project_simplex(a - step)
        
        # Check improvement
        new_conv = fast_convolve(candidate)
        new_true_obj = 2.0 * n * new_conv.max()
        
        if new_true_obj < true_obj - 1e-15:
            a = candidate
            no_improve = 0
            if new_true_obj < best_obj * 0.9999:
                best_obj = new_true_obj
                best_a = a.copy()
        else:
            # Try smaller step
            for scale in [0.5, 0.25, 0.125, 0.0625]:
                candidate = project_simplex(a - scale * step)
                new_conv = fast_convolve(candidate)
                new_true_obj = 2.0 * n * new_conv.max()
                if new_true_obj < true_obj - 1e-15:
                    a = candidate
                    no_improve = 0
                    if new_true_obj < best_obj * 0.9999:
                        best_obj = new_true_obj
                        best_a = a.copy()
                    break
            else:
                no_improve += 1
        
        # Anneal temperature
        T = max(T_min, T * T_decay)
        
        # Adapt learning rate
        if no_improve > 50:
            lr *= 0.9
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(t * 31 + 7)
            noise = 0.001 + 0.002 * min(1.0, no_improve / 500.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = lr_init
            T = max(T_init * 0.5, T_min * 10)
            no_improve = 0
    
    return best_a, best_obj / (2 * n)  # Return max(conv) for normalized a


def active_set_solve(a0, time_limit, tol=1e-14, max_iter=5000):
    """
    Solve the KKT system for the active peak set.
    At optimum: for active peaks k in P, c[k] = M.
    We use Newton's method on the system:
      F(a, M) = [c[k] - M for k in P; sum(a) - 1] = 0
    with the constraint a >= 0 handled by projection.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Identify active peaks (within tolerance of max)
        threshold = M - max(1e-10, M * 1e-6)
        active = np.where(conv >= threshold)[0]
        
        if len(active) <= 1:
            break
        
        # Build the system: for each active peak k, c[k] = M
        # Unknowns: a[0..n-1] and M (total n+1 unknowns)
        # Equations: len(active) peak equations + 1 sum equation = len(active) + 1
        # We need len(active) + 1 <= n + 1, i.e., len(active) <= n
        
        if len(active) > n:
            # Too many active peaks, use only the top n
            top_idx = np.argsort(conv[active])[::-1][:n]
            active = active[top_idx]
        
        # Newton step: solve J * da = -F
        # F = [c[k] - M for k in active; sum(a) - 1]
        # J = [dc[k]/da[j] for k in active, j in 0..n-1; dc[k]/dM for k in active; 0; 
        #     1 for all j; 0]
        
        m_peaks = len(active)
        # J is (m_peaks + 1) x (n + 1)
        # F is (m_peaks + 1)
        
        F = np.zeros(m_peaks + 1)
        J = np.zeros((m_peaks + 1, n + 1))
        
        for i, k in enumerate(active):
            F[i] = conv[k] - M
            # dc[k]/da[j] = 2*a[k-j] if 0 <= k-j < n
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    J[i, j] = 2.0 * a[idx]
            J[i, n] = -1.0  # dc[k]/dM = -1
        
        # Sum equation
        F[m_peaks] = a.sum() - 1.0
        J[m_peaks, :n] = 1.0
        J[m_peaks, n] = 0.0
        
        # Solve J * dx = -F using least squares (since J may be rectangular)
        try:
            dx, residuals, rank, sv = np.linalg.lstsq(J, -F, rcond=tol)
        except np.linalg.LinAlgError:
            break
        
        da = dx[:n]
        dM = dx[n]
        
        # Damped Newton step
        step_size = 1.0
        for _ in range(20):
            a_new = a + step_size * da
            if (a_new < -1e-10).any():
                step_size *= 0.5
                continue
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            
            conv_new = fast_convolve(a_new)
            M_new = conv_new.max()
            
            # Check if all active peaks are satisfied
            if np.all(conv_new[active] <= M_new + 1e-10) and M_new < M + 1e-10:
                a = a_new
                break
            step_size *= 0.5
        
        if step_size <= 1e-10:
            break
        
        new_M = fast_convolve(a).max()
        if new_M < best_obj - 1e-15:
            best_obj = new_M
            best_a = a.copy()
        
        # Check convergence
        if np.abs(F).max() < 1e-12:
            if it > 10:
                break
    
    return best_a, best_obj


def fw_lp_oracle(f, M, max_time=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'max_time': max_time})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        g = fw_lp_oracle(f, M * m_exp, max_time=min(20, remaining * 0.3))
        if g is None:
            m_exp *= 1.2
            if m_exp > 5.0:
                m_exp = 1.0
            no_improve += 1
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-14:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)
        
        if no_improve > 40:
            rng = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj


def try_and_update(a, best_state, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_state
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    best_seq, best_score = best_state
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return (best_seq, best_score)


def interp_sequence(parent, n_target):
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np
