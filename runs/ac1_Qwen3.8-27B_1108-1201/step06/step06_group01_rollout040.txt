def fw_lp_exact(f, M):
    """
    Solve exact FW subproblem:
    Minimize max_k (b_a[k] + 2*(a*g)[k])
    subject to sum(g) = 1, g >= 0
    
    where b_a = conv(a,a), M = max(b_a)
    
    This is equivalent to:
    Minimize t
    subject to b_a[k] + 2*(a*g)[k] <= t for all k
    sum(g) = 1
    g >= 0
    """
    n = len(f)
    b_a = fast_convolve(f)
    
    # LP variables: g[0],...,g[n-1], t
    # Minimize t
    # Constraints:
    #   For each k: 2*(a*g)[k] - t <= -b_a[k]
    #   sum(g) = 1
    #   g >= 0, t free (but t >= 0 effectively)
    
    # Using scipy linprog:
    # c = [0, 0, ..., 0, 1] (minimize t, which is the last variable)
    # A_ub: for each k (2n-1 constraints): 2*(cross_conv row for k) - 1 <= -b_a[k]
    # A_eq: sum(g) = 1
    # bounds: g[i] >= 0, t free
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # A_ub: (2n-1) x (n+1)
    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    
    for k in range(2 * n - 1):
        # 2*(a*g)[k] - t <= -b_a[k]
        # (a*g)[k] = sum_j a[k-j]*g[j] for valid j
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[k, j] = 2.0 * f[k - j]
        A_ub[k, n] = -1.0  # coefficient of t
        b_ub[k] = -b_a[k]
    
    # A_eq: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if res.success:
        g = np.maximum(res.x[:n], 0.0)
        return g
    return None
