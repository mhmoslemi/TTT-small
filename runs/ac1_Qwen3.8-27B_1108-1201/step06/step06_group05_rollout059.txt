Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using advanced dual-based optimization.

2. Fast Frank-Wolfe with efficient LP oracle

4. Multi-start with diverse initializations
5. Adaptive n selection

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def conv_gradient_at_peak(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    idx_arr = np.arange(js, je + 1)
    grad[idx_arr] = a[k - idx_arr]
    return grad

def dual_subgradient(a, mu, lr):
    """
    Primal-dual step: 
    - mu are dual variables for constraints (a*a)[k] <= M
    - Primal update: a <- project_simplex(a - lr * grad)
    - Dual update: mu <- max(0, mu + lr * ((a*a)[k] - M))
    """
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    
    # Primal gradient: weighted sum of gradients at active peaks
    grad = np.zeros(n)
    active = mu > 1e-10
    for k in np.where(active)[0]:
        if mu[k] > 1e-10:
            grad += mu[k] * conv_gradient_at_peak(a, k)
    grad *= 2.0  # factor of 2 from auto-convolution
    
    # Primal step
    a_new = project_simplex(a - lr * grad)
    
    # Dual step
    conv_new = fast_convolve(a_new)
    M_new = conv_new.max()
    mu_new = np.maximum(mu + lr * (conv_new - M_new), 0.0)
    
    return a_new, mu_new, M

    a = np.maximum(a0.astype(np.float64), 0.0)

import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve_and_obj(a):
    """Return (convolution, max_value, objective)."""
    conv = fast_convolve(a)
    M = conv.max()
    s = a.sum()
    n = len(a)
    obj = 2.0 * n * M / (s * s) if s > 1e-15 else np.inf
    return conv, M, obj


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, n):
    """Build constraint matrix A where A[k,m] = f[k-m] if valid."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            A[k, m_lo:m_hi + 1] = f[k - m_lo:k - m_hi + 1][::-1]
    return A


def lp_oracle(f, M, timeout=20):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0. Returns g normalized."""
    n = len(f)
    c = -np.ones(n)
    A = build_constraint_matrix(f, n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None

                if f[j] < 1e-8:
                    continue
                # Away direction: f - (f_j / (1 - f_j)) * (f - e_j * ... )
                # Simpler: just try removing j
                ft = f.copy()
                ft[j] = 0.0
                # Rescale to simplex
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < away_val:
                    away_val = val
                    away_alpha = 1.0
                    away_idx = j

        # Choose the better step
        if away_val < best_val - 1e-12:
            f = f.copy()
            f[away_idx] = 0.0
            f = f / f.sum()
            new_val = away_val
        elif best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            new_val = best_val
        else:
            no_improve += 1
            if no_improve > 30:
                # Perturb and restart
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
                continue

        conv = fast_convolve(f)
        M = conv.max()
        obj = 2.0 * n * M / (f.sum() ** 2)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.004 * min(1.0, no_improve / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def smoothed_fista(f0, time_limit, T_init=0.0005, n_inner=5000):
    """FISTA on the log-sum-exp smoothed approximation of max(conv)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    T = T_init
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    # FISTA momentum
    y = f.copy()
    t_k = 1.0
    v = np.zeros(n)

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Gradient of T * log(sum(exp(conv_k / T)))
        conv_y = fast_convolve(y)
        scaled = conv_y / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient: 2 * sum_k p_k * conv(a, e_k)
        grad = np.zeros(n)
        # Efficient: grad[j] = 2 * sum_k p_k * a[k-j] (if valid)
        # = 2 * (p * a)[j] in circular sense, but we use linear
        # Actually grad[j] = 2 * sum_k p[k] * a[k-j] for valid k-j
        # This is a cross-correlation
        from scipy.signal import fftconvolve
        # p is defined on [0, 2n-2], a on [0, n-1]
        # grad[j] = 2 * sum_k p[k] * a[k-j] = 2 * (p cross-correlate a)[j]
        # = 2 * (flip(p) * a)[j] in convolution terms
        p_flip = p[::-1]
        conv_pa = np.convolve(p_flip, a_y := y, mode='full')
        # We need the part corresponding to linear convolution
        # (p cross a)[j] = sum_k p[k] a[k-j]
        # Using FFT: this is ifft(fft(p) * conj(fft(a)))
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        P = np.fft.rfft(p, fft_size)
        Y = np.fft.rfft(y, fft_size)
        grad = 2.0 * np.fft.irfft(P * np.conj(Y), fft_size)[:n]

        # Lipschitz constant estimate
        L = 2.0 * p.max() * 2.0 * n  # rough estimate
        L = max(L, 4.0)
        step = 1.0 / L

        # Gradient step
        z = project_simplex(y - step * grad)

        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = z + ((t_k - 1.0) / t_new) * (z - f)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        f = z
        t_k = t_new

        # Evaluate
        conv_f = fast_convolve(f)
        M = conv_f.max()
        obj = 2.0 * n * M / (f.sum() ** 2)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt T
        if no_improve == 100:
            T *= 0.8
            t_k = 1.0
        if no_improve == 300:
            T = max(0.0001, T * 0.7)
            t_k = 1.0
        if no_improve > 500:
            # Restart with perturbation
            rng = np.random.default_rng(it * 13 + 5)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            y = f.copy()
            t_k = 1.0
            T = T_init
            no_improve = 0

    return best_f, best_obj


def kkt_refine(a0, time_limit, tol=1e-10, max_iter=500):
    """
    KKT-based refinement: identify active peaks and support,
    then solve the balancing QP to find optimal dual multipliers,
    and use them for a Newton step.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max() / (a.sum() ** 2)

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        M = conv.max()
        support = np.where(a > 1e-10)[0]

        if len(support) < 2:
            break

        # Find active peaks (within 0.1% of max)
        threshold = M * (1.0 - 0.0005)
        active = np.where(conv >= threshold)[0]

        if len(active) < 2:
            # Single peak: just do gradient step
            js = max(0, active[0] - n + 1)
            je = min(n - 1, active[0])
            grad = np.zeros(n)
            grad[js:je + 1] = 2.0 * a[active[0]:active[0] - js + 1][::-1]
            for step in [0.001, 0.005, 0.01, 0.05]:
                candidate = project_simplex(a - step * grad)
                val = fast_convolve(candidate).max()
                if val < M - 1e-14:
                    a = candidate
                    break
            continue

        # Build the balancing matrix B: B[i, k] = a[active[k] - support[i]]
        m_s = len(support)
        m_a = len(active)
        B = np.zeros((m_s, m_a))
        for i, j in enumerate(support):
            for k_idx, k in enumerate(active):
                idx = k - j
                if 0 <= idx < n:
                    B[i, k_idx] = a[idx]

        # Solve QP: min ||B @ mu - c * 1||^2 s.t. mu >= 0, sum(mu) = 1
        # Equivalently: find mu that makes B @ mu as flat as possible
        # This is: min mu^T B^T (I - 11^T/m) B mu s.t. mu >= 0, 1^T mu = 1
        # Simplification: use the fact that at optimum, B @ mu should be constant
        # So we want: B @ mu = c * 1, which gives (m_s) equations
        # With sum(mu) = 1, we have m_a + 1 unknowns and m_s + 1 equations

        # Use SVD approach: find mu in null space of (B - c*1_row)
        # Alternative: minimize variance of B @ mu
        # var(B @ mu) = mu^T B^T (I/m_s - 1/m_s^2 * 11^T) B mu

        C = B.T @ ((np.eye(m_s) / m_s) - (np.ones((m_s, m_s)) / (m_s * m_s))) @ B
        # Minimize mu^T C mu s.t. mu >= 0, sum(mu) = 1
        # Use projection gradient:
        mu = np.ones(m_a) / m_a
        for _ in range(50):
            grad_mu = 2.0 * C @ mu
            # Remove mean (constraint sum(mu)=1)
            grad_mu -= grad_mu.mean()
            mu = mu - 0.5 * grad_mu
            mu = np.maximum(mu, 0)
            sm = mu.sum()
            if sm < 1e-15:
                mu = np.ones(m_a) / m_a
            else:
                mu = mu / sm

        # Use mu to compute gradient direction
        # The direction that "flattens" the active peaks
        # At the KKT point: sum_k mu_k * a[k-j] = constant for j in support
        # If not balanced, move a to reduce the imbalance
        grad = np.zeros(n)
        for k_idx, k in enumerate(active):
            w = mu[k_idx]
            if w < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += w * a[k - idx_arr]

        # Normalize gradient by support values to get the "excess"
        # The KKT condition says grad[j] should be constant on support
        # If not, adjust a to make it so
        grad_on_support = grad[support]
        mean_grad = grad_on_support.mean()
        # Direction: reduce a where grad is above mean, increase where below
        direction = np.zeros(n)
        direction[support] = grad_on_support - mean_grad

        # Take a step in the direction that balances
        best_candidate = a
        best_val = M
        for step in [0.0005, 0.001, 0.005, 0.01, 0.02, 0.05]:
            candidate = project_simplex(a - step * direction)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate
        obj = 2.0 * n * fast_convolve(a).max() / (a.sum() ** 2)

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()

        # Convergence check
        if abs(M - best_val) < 1e-13:
            if it > 20:
                break

    return best_a, best_obj


def pgd_refine(a0, time_limit, top_k=20, lr_init=0.01):
    """Projected gradient descent with momentum for final polishing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    mom = 0.9
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Multi-peak gradient
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.3
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        grad = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += weights[i] * a[k - idx_arr]

        v = mom * v - lr * grad
        best_candidate = a
        best_val = M
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            candidate = project_simplex(a + scale * v)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate

        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.9

        if no_improve > 30:
            lr *= 0.95
        if no_improve > 100:
            rng = np.random.default_rng((it * 48271 + 17) % (2**31 - 1))
            noise = 0.001 + 0.002 * min(1.0, (no_improve - 100) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def try_and_update(a, best_state, label=""):
    """Evaluate sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_state
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    best_seq, best_score = best_state
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return (best_seq, best_score)


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase convex optimization search.
    Uses FW with LP oracle, smoothed FISTA, KKT refinement,
    and multi-n exploration to minimize 2n*max(conv)/(sum)^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Quick optimization (~80s) ===
    print(f"\n=== Phase 1: Quick Opt (parent n={n_parent}) ===")
    t_start = time.monotonic()

    # Symmetric version
    sym = (best_seq + best_seq[::-1]) / 2.0
    sym = sym / sym.sum()
    (best_seq, best_score) = try_and_update(sym, (best_seq, best_score), "sym")

    # Quick PGD
    remaining = deadline - time.monotonic() - 1500
    if remaining > 20:
        t_alloc = min(remaining, 40)
        a_opt, _ = pgd_refine(best_seq, time_limit=t_alloc, top_k=20, lr_init=0.02)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "pgd1")

    # Quick FW
    remaining = deadline - time.monotonic() - 1400
    if remaining > 20:
        t_alloc = min(remaining, 40)
        a_opt, _ = fw_standard(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "fw1")

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # === Phase 2: Multi-n exploration (~250s) ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    # Try various n values
    n_candidates = set()
    for n_try in [384, 512, 640, n_parent, 896, 1024, 1024, 1280, 1536]:
        if 256 <= n_try <= 2048:
            n_candidates.add(n_try)

    for n_try in sorted(n_candidates):
        remaining = deadline - time.monotonic() - 900
        if remaining < 25:
            break

        if n_try == n_best:
            start = best_seq.copy()
        else:
            start = interp_sequence(best_seq, n_try)

        t_alloc = min(remaining, 60)
        print(f"  n={n_try}, alloc={t_alloc:.0f}s")

        # Quick FW
        a_opt, _ = fw_standard(start, time_limit=t_alloc * 0.5)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"fw_n{n_try}")

        # Quick PGD
        remaining = deadline - time.monotonic() - 850
        if remaining > 15:
            a_opt2, _ = pgd_refine(a_opt, time_limit=min(remaining, t_alloc * 0.3), top_k=15)
            (best_seq, best_score) = try_and_update(a_opt2, (best_seq, best_score), f"pgd_n{n_try}")

        # KKT quick
        remaining = deadline - time.monotonic() - 830
        if remaining > 10:
            a_opt3, _ = kkt_refine(a_opt2, time_limit=min(remaining, 15))
            (best_seq, best_score) = try_and_update(a_opt3, (best_seq, best_score), f"kkt_n{n_try}")

    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # === Phase 3: Deep FW on best n (~500s) ===
    print(f"\n=== Phase 3: Deep FW (n={len(best_seq)}) ===")
    t_start = time.monotonic()

    remaining = deadline - time.monotonic() - 400
    if remaining > 30:
        t_alloc = min(remaining, 400)
        print(f"  Deep FW, alloc={t_alloc:.0f}s")
        a_opt, _ = fw_standard(best_seq, time_limit=t_alloc, lp_timeout=30)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "deep_fw")

    # Away-step FW
    remaining = deadline - time.monotonic() - 250
    if remaining > 30:
        t_alloc = min(remaining, 200)
        print(f"  Away-step FW, alloc={t_alloc:.0f}s")
        a_opt, _ = fgw_away(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "away_fw")

    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # === Phase 4: Smoothed FISTA (~300s) ===
    print(f"\n=== Phase 4: Smoothed FISTA ===")
    t_start = time.monotonic()

    for T_init in [0.0003, 0.001, 0.003]:
        remaining = deadline - time.monotonic() - 200
        if remaining < 20:
            break
        t_alloc = min(remaining, 100)
        print(f"  FISTA T={T_init}, alloc={t_alloc:.0f}s")
        a_opt, _ = smoothed_fista(best_seq, time_limit=t_alloc, T_init=T_init)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"fista_T{T_init}")

    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # === Phase 5: KKT refinement + PGD (~250s) ===
    print(f"\n=== Phase 5: KKT + PGD Refinement ===")
    t_start = time.monotonic()

    # Deep KKT
    remaining = deadline - time.monotonic() - 150
    if remaining > 20:
        t_alloc = min(remaining, 120)
        print(f"  KKT refine, alloc={t_alloc:.0f}s")
        a_opt, _ = kkt_refine(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "kkt_deep")

    # Multiple PGD configs
    for top_k, lr_init in [(10, 0.005), (30, 0.003), (50, 0.002)]:
        remaining = deadline - time.monotonic() - 100
        if remaining < 15:
            break
        t_alloc = min(remaining, 40)
        a_opt, _ = pgd_refine(best_seq, time_limit=t_alloc, top_k=top_k, lr_init=lr_init)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"pgd_k{top_k}")

    print(f"  Phase 5 best: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # === Phase 6: Final polishing (~150s) ===
    print(f"\n=== Phase 6: Final Polish ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    # Tiny perturbations + quick optimization
    for i in range(8):
        remaining = deadline - time.monotonic() - 30
        if remaining < 15:
            break
        noise = rng.standard_normal(n_best) * (0.0005 + 0.001 * i / 8.0)
        f_pert = project_simplex(best_seq + noise)
        t_alloc = min(remaining, 15)
        a_opt, _ = pgd_refine(f_pert, time_limit=t_alloc, top_k=20, lr_init=0.005)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"pert{i}")

        remaining = deadline - time.monotonic() - 25
        if remaining > 10:
            a_opt2, _ = kkt_refine(a_opt, time_limit=min(remaining, 10))
            (best_seq, best_score) = try_and_update(a_opt2, (best_seq, best_score), f"pert_kkt{i}")

    # Final FW
    remaining = deadline - time.monotonic() - 20
    if remaining > 15:
        a_opt, _ = fw_standard(best_seq, time_limit=min(remaining, 25))
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "final_fw")

    # Symmetry check
    sym_final = (best_seq + best_seq[::-1]) / 2.0
    sym_final = sym_final / sym_final.sum()
    (best_seq, best_score) = try_and_update(sym_final, (best_seq, best_score), "final_sym")

    # Final KKT on best
    remaining = deadline - time.monotonic() - 10
    if remaining > 5:
        a_opt, _ = kkt_refine(best_seq, time_limit=min(remaining, 8))
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "final_kkt")

    print(f"  Phase 6 best: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
