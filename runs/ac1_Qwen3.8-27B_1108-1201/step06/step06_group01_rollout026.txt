"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA (Fast Iterative Shrinkage-Thresholding) with softmax-smoothed peaks
2. Peak-targeted coordinate descent with 1D line search
3. Structured perturbation for escape from local minima
4. Multi-n exploration with time budgeting
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_matvec(p, a):
    """Compute (Hankel matrix from p) @ a efficiently.
    p has length 2n-1, a has length n.
    Result[i] = sum_{m=0}^{n-1} p[i+m] * a[m]
    """
    n = len(a)
    # Create index matrix
    i_idx = np.arange(n)[:, None]
    m_idx = np.arange(n)[None, :]
    M = p[i_idx + m_idx]
    return M @ a

def smooth_gradient(a, T):
    """Compute smoothed gradient of max(conv(a,a)) using softmax.
    T is the temperature. Returns gradient vector.
    """
    n = len(a)
    conv = fast_convolve(a)
    # Softmax weights
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()
    # Gradient: 2 * H(p) @ a where H(p) is the Hankel matrix from p
    grad = 2.0 * hankel_matvec(p, a)
    return grad, conv.max(), p

def fista_smoothed(a0, time_limit, T_init=0.001, lr_init=0.1):
    """FISTA (Nesterov-accelerated proximal gradient) on smoothed objective."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    y = a.copy()  # Nesterov lookahead
    t_nest = 1.0
    lr = lr_init
    T = T_init
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Nesterov momentum
        alpha_nest = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / alpha_nest
        y_new = (1.0 + beta) * a - beta * y
        y = np.maximum(y_new, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y / ys
        t_nest += 1.0
        
        # Compute smoothed gradient at y
        grad, conv_max, p = smooth_gradient(y, T)
        
        # Gradient step + projection
        y_step = y - lr * grad
        a_new = project_simplex(y_step)
        
        # Evaluate
        obj = fast_convolve(a_new).max()
        
        # Backtracking if needed
        if obj > conv_max + 1e-8:
            lr *= 0.8
            y_step = y - lr * grad
            a_new = project_simplex(y_step)
            obj = fast_convolve(a_new).max()
        
        a = a_new
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt temperature and lr
        if no_improve > 20:
            T *= 1.05
            lr *= 0.98
        if no_improve > 100:
            T *= 1.2
            # Reset Nesterov
            y = best_a.copy()
            t_nest = 1.0
            no_improve = 50
        
        if no_improve > 200:
            # Structured restart
            break
    
    return best_a, best_obj

def coord_descent_1d(a0, time_limit, top_k=5):
    """Coordinate descent: optimize one coordinate at a time with 1D line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        # Get top peaks
        k_peaks = min(top_k, len(conv))
        peak_idx = np.argpartition(conv, -k_peaks)[-k_peaks:]
        peak_idx = peak_idx[np.argsort(conv[peak_idx])[::-1]]
        
        improved = False
        
        for k in peak_idx:
            if time.monotonic() - t0 > time_limit - 1.0:
                break
            
            k = int(k)
            # Find indices that contribute to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            if je < js:
                continue
            
            # For coordinate descent: try setting a[j] to 0 and renormalizing
            # Or try transferring mass from one j to another
            # Simple approach: for each j in [js, je], try reducing a[j] by a small amount
            # and distributing to the index with lowest contribution
            
            # Find the partner with highest contribution
            contributions = np.array([a[k - j] * a[j] for j in range(js, je + 1)])
            # Actually, the contribution of a[j] to conv[k] is a[j] * a[k-j]
            # (times 2 if j != k-j)
            
            # Try moving mass from the highest contributor to a random non-contributor
            if len(contributions) > 1:
                max_contrib_idx = js + np.argmax(contributions)
                
                # Find a good target: an index that doesn't contribute to this peak
                # and has low value
                all_indices = np.setdiff1d(np.arange(n), np.arange(js, je + 1))
                if len(all_indices) > 0:
                    # Pick the one with lowest current value
                    target = all_indices[np.argmin(a[all_indices])]
                    
                    # Try transferring
                    for frac in [0.1, 0.2, 0.3, 0.5]:
                        delta = a[max_contrib_idx] * frac
                        if delta < 1e-10:
                            break
                        a_test = a.copy()
                        a_test[max_contrib_idx] -= delta
                        a_test[target] += delta
                        a_test = np.maximum(a_test, 0.0)
                        a_test = a_test / a_test.sum()
                        c = fast_convolve(a_test)
                        if c.max() < best_obj - 1e-12:
                            a = a_test
                            best_obj = c.max()
                            best_a = a.copy()
                            improved = True
                            break
        
        if not improved:
            stall += 1
        else:
            stall = 0
        
        if stall > 50:
            # Perturb
            rng = np.random.default_rng(stall * 7919 + 13)
            scale = 0.001 * min(1.0, stall / 20.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            stall = 0
        
        if stall > 200:
            break
    
    return best_a, best_obj

def multi_start_explore(n_candidates, time_per, parent):
    """Try multiple starting points for each length."""
    results = []
    for n in n_candidates:
        # Generate several starting points
        starts = []
        if parent is not None and len(parent) > 0:
            # Interpolate parent to length n
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            p = np.maximum(p, 0)
            p = p / p.sum()
            starts.append(p)
        
        # Add some structured starts
        starts.append(np.ones(n) / n)
        
        # Cosine-shaped
        x = np.linspace(0, np.pi, n)
        c = np.cos(x) ** 2
        c = c / c.sum()
        starts.append(c)
        
        # U-shaped (high at edges)
        u = 1.0 - np.cos(np.linspace(0, np.pi, n))
        u = u / u.sum()
        starts.append(u)
        
        for start in starts:
            if time_per < 5:
                break
            a_opt, obj = fista_smoothed(start, time_limit=time_per)
            results.append((a_opt, obj, n))
    
    return results

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA (Nesterov-accelerated proximal gradient) with softmax-smoothed peak gradient
2. Peak-targeted 1D coordinate descent with bisection line search
3. Dual-guided structured perturbation for escaping local minima
4. Parameterized initialization (cosine, exponential, U-shape)
5. Multi-n exploration with adaptive time budgeting
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_matvec(p, a):
    """Compute Hankel(p) @ a where Hankel(p)[i,m] = p[i+m].
    p has length 2n-1, a has length n.
    Uses np.correlate for efficiency.
    """
    return np.correlate(p, a, mode='valid')


def compute_smoothed_grad(a, T):
    """Compute smoothed gradient of max(conv(a,a)).
    Returns (gradient, max_conv, peak_weights).
    """
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()
    grad = 2.0 * hankel_matvec(p, a)
    return grad, conv.max(), p, conv


def fista_optimize(a0, time_limit, T_init=0.001, lr_init=0.1, use_line_search=True):
    """FISTA with adaptive temperature and optional line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    y = a.copy()
    t_nest = 1.0
    lr = lr_init
    T = T_init

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Nesterov momentum
        alpha_nest = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / alpha_nest
        y = (1.0 + beta) * a - beta * y
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y / ys
        t_nest += 1.0

        # Compute smoothed gradient
        grad, conv_max, p, conv = compute_smoothed_grad(y, T)

        # Gradient step
        step = lr * grad
        y_step = y - step
        a_new = project_simplex(y_step)

        # Evaluate
        obj_new = fast_convolve(a_new).max()

        if use_line_search and obj_new > conv_max:
            # Backtracking line search
            lr *= 0.7
            step = lr * grad
            y_step = y - step
            a_new = project_simplex(y_step)
            obj_new = fast_convolve(a_new).max()

        a = a_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive temperature: cool when improving, heat when stuck
        if no_improve > 10 and no_improve % 10 == 0:
            T *= 1.03
        if no_improve > 50 and no_improve % 25 == 0:
            T *= 0.9  # Reheat to escape
            lr *= 1.05
            y = best_a.copy()
            t_nest = 1.0
        if no_improve > 150:
            break

    return best_a, best_obj


def coord_descent_optimize(a0, time_limit, top_k=5):
    """1D coordinate descent: optimize one coordinate at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        k_peaks = min(top_k, len(conv))
        peak_idx = np.argpartition(conv, -k_peaks)[-k_peaks:]
        peak_idx = peak_idx[np.argsort(conv[peak_idx])[::-1]]

        improved = False

        for k in peak_idx:
            if time.monotonic() - t0 > time_limit - 1.0:
                break

            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)

            if je < js:
                continue

            # Find the index in [js, je] with highest contribution to this peak
            # Contribution of a[j] to conv[k]: a[j] * a[k-j]
            contributions = np.array([a[j] * a[k - j] for j in range(js, je + 1)])
            if contributions.max() < 1e-15:
                continue

            src = js + int(np.argmax(contributions))

            # Find target: index outside [js, je] (or inside with low contribution)
            outside = np.setdiff1d(np.arange(n), np.arange(js, je + 1))
            if len(outside) == 0:
                outside = np.arange(n)

            # Target: lowest value among outside indices
            target = outside[int(np.argmin(a[outside]))]

            if target == src:
                continue

            # 1D line search: a[src] -> a[src] - delta, a[target] -> a[target] + delta
            # Binary search for optimal delta
            lo = 0.0
            hi = a[src] * 0.9
            best_delta = 0.0
            best_val = best_obj

            for _ in range(15):
                mid = (lo + hi) / 2.0
                if mid < 1e-12:
                    break
                a_test = a.copy()
                a_test[src] -= mid
                a_test[target] += mid
                a_test = np.maximum(a_test, 0.0)
                ts = a_test.sum()
                if ts < 1e-15:
                    break
                a_test = a_test / ts
                val = fast_convolve(a_test).max()
                if val < best_val:
                    best_val = val
                    best_delta = mid
                    lo = mid
                else:
                    hi = mid

            if best_delta > 1e-10 and best_val < best_obj - 1e-13:
                a[src] -= best_delta
                a[target] += best_delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_obj = best_val
                best_a = a.copy()
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1

        # Try also the "reverse" direction: move mass TO a peak contributor
        # from a non-contributor (to balance)
        if stall > 10 and stall % 5 == 0:
            # Find the global minimum and try boosting it
            min_idx = int(np.argmin(a))
            # Find a peak contributor at another peak
            if len(peak_idx) > 0:
                k0 = int(peak_idx[0])
                js0 = max(0, k0 - n + 1)
                je0 = min(n - 1, k0)
                if min_idx < js0 or min_idx > je0:
                    # min_idx doesn't contribute to peak k0
                    for frac in [0.05, 0.1, 0.2]:
                        delta = a[min_idx] * frac
                        if delta < 1e-10:
                            break
                        a_test = a.copy()
                        a_test[min_idx] -= delta
                        a_test[src if a_test[src] > a_test[target] else target] += delta
                        a_test = np.maximum(a_test, 0.0)
                        a_test = a_test / a_test.sum()
                        val = fast_convolve(a_test).max()
                        if val < best_obj - 1e-13:
                            best_obj = val
                            best_a = a_test.copy()
                            improved = True
                            a = a_test
                            break

        if stall > 80:
            # Structured perturbation guided by peak structure
            rng = np.random.default_rng(it * 31 + 7)
            conv = fast_convolve(best_a)
            peak_k = int(np.argmax(conv))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            noise = rng.standard_normal(n) * 0.003
            # Bias: reduce contributors to the main peak
            for j in range(js, je + 1):
                partner = peak_k - j
                if 0 <= partner < n:
                    noise[j] -= 0.003 * a[partner]
            a = project_simplex(best_a + noise)
            stall = 0

        if stall > 300:
            break

    return best_a, best_obj

def dual_guided_perturb(a, conv, scale=0.005):
    """Generate a perturbation that targets reducing the main peaks."""
    n = len(a)
    rng = np.random.default_rng(np.random.randint(1, 2**31))

    noise = rng.standard_normal(n) * scale

    # Weight the noise by peak contribution
    p = conv / conv.max()
    p = p ** 3  # Sharpen
    p = p / p.sum()

    # For each peak, reduce the indices that contribute most
    peak_lags = np.argsort(conv)[-10:]
    for k in peak_lags:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            partner = k - j
            if 0 <= partner < n:
                weight = a[j] * a[partner] * p[k]
                noise[j] -= weight * 2.0

    return noise
