Optimized step function search using Nesterov momentum on smoothed
auto-convolution peak, spectral flattening (GS), and local search.
Targets improvement over 1.504824.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Step function optimization using Nesterov momentum on smoothed auto-convolution,
spectral flattening (Gerchberg-Saxton), LP-based Frank-Wolfe, and local search.
Targets improvement over 1.504824 (n=768).
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_fft(a, p, T):
    """
    Compute gradient of smoothed objective T*logsumexp(conv/T).
    grad[i] = (2/T) * sum_m a[m] * p[m+i]
    Uses FFT cross-correlation.
    """
    n = len(a)
    p_flip = p[::-1]
    conv_len = n + len(p) - 1  # n + (2n-1) - 1 = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p_flip, fft_size)
    conv = np.fft.irfft(A * P, fft_size)[:conv_len]
    # grad[i] = (2/T) * conv[2n-2-i]
    idx = 2 * n - 2 - np.arange(n)
    return (2.0 / T) * conv[idx]


def nesterov_smoothed(a0, time_limit, n, T_init=None, lr_init=0.05, beta=0.9, seed=42):
    """
    Nesterov momentum on smoothed auto-convolution peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    n = len(a)
    
    if T_init is None:
        conv0 = fast_convolve(a)
        T_init = conv0.max() * 0.05
    T = T_init
    lr = lr_init
    v = np.zeros(n)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        max_conv = conv.max()
        
        # Softmax weights
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        else:
            p = p / p_sum
        
        # Gradient
        grad = compute_grad_fft(a, p, T)
        
        # Nesterov step
        v = beta * v + grad
        candidate = a - lr * v
        candidate = project_simplex(candidate)
        
        # Check improvement
        c_new = fast_convolve(candidate)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-14:
            a = candidate
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        elif obj_new < max_conv:
            a = candidate
            no_improve += 1
        else:
            no_improve += 1
            # Try smaller step
            for factor in [0.5, 0.25, 0.1]:
                candidate2 = a - lr * factor * v
                candidate2 = project_simplex(candidate2)
                c2 = fast_convolve(candidate2)
                if c2.max() < max_conv:
                    a = candidate2
                    break
            else:
                v = np.zeros(n)
        
        # Adapt temperature
        if no_improve > 30:
            T *= 0.92
            no_improve = 0
        if no_improve > 100:
            T *= 0.8
            lr *= 0.9
        if no_improve > 300:
            # Restart
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            T = T_init * 0.5
            lr = lr_init * 0.5
            no_improve = 0
    
    return best_a, best_obj


def spectral_flatten(a0, time_limit, seed=42):
    """
    Gerchberg-Saxton style spectral flattening.
    Iteratively equalizes the magnitude spectrum of the zero-padded sequence.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    N = 1 << (2 * n - 2).bit_length()  # next power of 2 >= 2n-1
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        
        # Zero-pad and compute FFT
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        
        # Equalize magnitudes
        mag = np.abs(A)
        target_mag = np.sqrt(np.sum(mag ** 2) / N)
        
        # Preserve phase, set magnitude
        phase = A / np.maximum(mag, 1e-30)
        A_new = target_mag * phase
        A_new[0] = np.abs(A_new[0])  # ensure DC is real positive
        
        # Inverse FFT
        a_new = np.real(np.fft.ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a = project_simplex(a + rng.standard_normal(n) * 0.001)
            continue
        a_new = a_new / s
        
        # Check
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
        else:
            # Blend
            for alpha in [0.1, 0.3, 0.5]:
                blend = (1 - alpha) * a + alpha * a_new
                blend = project_simplex(blend)
                obj_blend = fast_convolve(blend).max()
                if obj_blend < best_obj - 1e-14:
                    best_obj = obj_blend
                    best_a = blend.copy()
                    a = blend
                    break
            else:
                a = a_new  # still update to explore
        
    return best_a, best_obj


def lp_fw_step(f, M, n, time_limit_per_lp=2.0):
    """
    One FW step with LP oracle.
    Solve: max sum(g) s.t. (f*g)[k] <= M, g >= 0.
    """
    # Build constraint matrix
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    
    if not rows:
        return None
    
    rows = np.array(rows)
    cols = np.array(cols)
    
    # Values: for row k, col i, value is f[k-i]
    vals = f[rows - cols]  # f[k-i] where k=rows, i=cols
    
    A_mat = lil_matrix((2 * n - 1, n))
    A_mat[rows, cols] = vals
    A_mat = A_mat.tocsr()
    
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        t0 = time.monotonic()
        res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit_per_lp})
        if res.success and res.fun is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def lp_fw_optimize(a0, n_iters=50, time_limit=100):
    """FW with LP oracle, limited iterations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_obj = M
    
    t0 = time.monotonic()
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        g = lp_fw_step(a, M, n)
        if g is None:
            # Fall back to gradient direction
            T = M * 0.1
            conv = fast_convolve(a)
            scaled = conv / T
            scaled -= scaled.max()
            p = np.exp(scaled)
            p = p / p.sum()
            grad = compute_grad_fft(a, p, T)
            g = project_simplex(a - 0.01 * grad)
        
        # Line search
        best_alpha = 0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1 - alpha) * a + alpha * g
            ft = np.maximum(ft, 0)
            s = ft.sum()
            if s < 1e-15:
                continue
            ft = ft / s
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            a = (1 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()
    
    return best_a, best_obj


def local_search(a0, time_limit, seed=42):
    """Local search with gradient-guided moves."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        
        improved = False
        
        # Gradient-guided step
        conv = fast_convolve(a)
        T = conv.max() * 0.05
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        grad = compute_grad_fft(a, p, T)
        
        for lr in [0.001, 0.003, 0.01, 0.03]:
            candidate = project_simplex(a - lr * grad)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-14:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break
        
        # Random perturbation
        if not improved:
            for scale in [0.001, 0.003, 0.01, 0.03]:
                pert = rng.standard_normal(n) * scale
                candidate = project_simplex(a + pert)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-14:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break
        
        # Pair exchange
        if not improved and it % 3 == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                for frac in [0.05, 0.1, 0.2, 0.4]:
                    candidate = a.copy()
                    amount = min(candidate[i], candidate[j]) * frac
                    candidate[i] -= amount
                    candidate[j] += amount
                    candidate = np.maximum(candidate, 0)
                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-14:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved and it > 5000:
            break
    
    return best_a, best_obj


def try_update(a, label, best_seq, best_score):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return a_norm.copy(), score
    return best_seq, best_score
