```python
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def peak_shave(a0, time_limit, top_k=5, step_init=0.01):
    """
    Iterative peak shaving: at each step, identify the peak and
    move in the direction that reduces it most while preserving
    non-negativity and the sum constraint.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Find top-k peaks
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]
        
        # Compute gradient direction: for each peak k, the gradient contribution
        # to a[j] is a[k-j] (for valid j)
        # We want to REDUCE the peak, so we move in the direction -sum a[k-j]
        # But we need to be careful: reducing a[j] might increase other peaks
        
        # Direction: d[j] = -sum_{k in peaks} w_k * a[k-j] + correction for sum constraint
        # where w_k are weights (e.g., proportional to how much peak k exceeds the average)
        
        # Simple approach: equal weights for top peaks
        d = np.zeros(n)
        for k in peak_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            d[js:je+1] -= a[k - (js:je+1)]
        
        # Project d onto the subspace where sum(d) = 0 (to maintain the simplex)
        d = d - d.mean()
        
        # Also, we should not decrease a[j] below 0, so clip the direction
        # For j where a[j] is small, we shouldn't decrease it much
        # Simple approach: scale the step to avoid going negative
        if d.min() < 0:
            max_step = min(1.0, -a[d < 0].min() / (-d[d < 0]).max()) if (d < 0).any() and a[d < 0].min() > 0 else 1.0
        else:
            max_step = 1.0
        
        # Try several step sizes
        best_candidate = a
        best_val = M
        for scale in [step, step*0.5, step*0.25, step*0.1]:
            scale_eff = min(scale, max_step * 0.9)
            candidate = a + scale_eff * d
            candidate = np.maximum(candidate, 0)
            sc = candidate.sum()
            if sc < 1e-15:
                continue
            candidate = candidate / sc
            val = fast_convolve(candidate).max()
            if val < best_val:
                best_val = val
                best_candidate = candidate
        
        a = best_candidate
        
        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
            step = min(step * 1.1, 0.1)
        else:
            no_improve += 1
            step *= 0.95
            if no_improve > 100:

                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                step = step_init
                no_improve = 0
    
    return best_a, best_obj
```

```python
def fw_line_search(f, g, n_evals=20):
    """Find the optimal alpha in [0,1] that minimizes max_k ((1-a)f + a*g) * B_k * ((1-a)f + a*g)"""
    n = len(f)
    
    def obj_at_alpha(alpha):
        x = (1 - alpha) * f + alpha * g
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-15:
            return float('inf')
        x = x / s
        return fast_convolve(x).max()
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    a, b = 0.0, 1.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = obj_at_alpha(c)
    fd = obj_at_alpha(d)
    
    for _ in range(n_evals):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = obj_at_alpha(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = obj_at_alpha(d)
    
    return (a + b) / 2
```

```python
def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_obj(a):
    """Compute the objective max(a*a) given sum(a)=1."""
    return fast_convolve(a).max()


def golden_section_minimize(func, a, b, n_iter=25):
    """Minimize a univariate function on [a,b] using golden section search."""
    gr = (np.sqrt(5) - 1) / 2
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(n_iter):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = func(d)
    return (a + b) / 2


def fw_golden(f, g):
    """Compute the optimal alpha for FW step using golden section search."""
    n = len(f)

    def obj_at_alpha(alpha):
        x = (1.0 - alpha) * f + alpha * g
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-15:
            return 1e10
        x = x / s
        return fast_convolve(x).max()

    alpha = golden_section_minimize(obj_at_alpha, 0.0, 1.0, n_iter=20)
    return alpha, obj_at_alpha(alpha)


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'max_time': 60})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def smoothed_grad_optimize(a0, time_limit, T_init=0.002, lr_init=0.015, mom_init=0.92):

    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()

    T = T_init
    lr = lr_init
    mom = mom_init
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = eval_obj(a)

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Softmax weights for the peaks
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            p = p / p_sum

        # Gradient: sum_k p[k] * a[k-j] for each j
        grad = np.zeros(n)
        for k in range(len(conv)):
            if p[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += p[k] * a[k - idx]

        # Momentum update
        v = mom * v - lr * grad
        v -= v.mean()  # Project to sum=0 subspace

        # Backtracking line search
        best_candidate = a
        best_val = M
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            candidate = np.maximum(a + scale * v, 0.0)
            sc = candidate.sum()
            if sc < 1e-15:
                continue
            candidate = candidate / sc
            val = eval_obj(candidate)
            if val < best_val - 1e-18:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.9

        # Adaptive parameters
        if no_improve > 20:
            T *= 1.1
            lr *= 0.97
        if no_improve > 100:
            T *= 0.7

            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            v = np.zeros(n)

def evolutionary_refine(a0, time_limit, pop_size=6, n_gen_max=500):
    """
    Simple evolutionary strategy: maintain a population, create offspring
    via crossover and mutation, select best.
    """
    n = len(a0)
    rng = np.random.default_rng(42)

    # Initialize population
    population = [a0.copy()]
    for _ in range(pop_size - 1):
        pert = a0 + rng.standard_normal(n) * 0.02
        population.append(project_simplex(pert))

    # Evaluate initial population
    scores = [eval_obj(pop) for pop in population]
    best_idx = int(np.argmin(scores))
    best_a = population[best_idx].copy()
    best_obj = scores[best_idx]

    t0 = time.monotonic()
    gen = 0

    while time.monotonic() - t0 < time_limit and gen < n_gen_max:
        gen += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Sort population by score
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]

        # Create offspring
        offspring = []
        n_offspring = pop_size // 2 + 1

        for i in range(n_offspring):
            if i == 0:
                # Elite: keep best
                offspring.append(population[0].copy())
            elif i == 1:
                # Crossover of top two
                w = rng.random() * 0.4 + 0.3  # weight in [0.3, 0.7]
                child = w * population[0] + (1 - w) * population[1]
                child = project_simplex(child)
                offspring.append(child)
            else:
                # Mutation of a random parent
                parent_idx = int(rng.integers(0, min(3, pop_size)))
                scale = 0.005 + 0.01 * rng.random()
                child = population[parent_idx] + rng.standard_normal(n) * scale
                child = project_simplex(child)
                offspring.append(child)

        # Evaluate offspring
        off_scores = [eval_obj(off) for off in offspring]

        # Select: keep best from combined population + offspring
        combined = population + offspring
        combined_scores = scores + off_scores
        new_order = np.argsort(combined_scores)[:pop_size]

        population = [combined[i] for i in new_order]
        scores = [combined_scores[i] for i in new_order]

        if scores[0] < best_obj - 1e-15:
            best_obj = scores[0]
            best_a = population[0].copy()

    return best_a, best_obj

def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def make_hann(n):
    """Create a Hann window sequence."""
    j = np.arange(n)
    w = 0.5 * (1 + np.cos(2 * np.pi * j / (n - 1))) if n > 1 else np.array([1.0])
    s = w.sum()
    return w / s


def make_hamming(n):
    """Create a Hamming window sequence."""
    j = np.arange(n)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * j / (n - 1)) if n > 1 else np.array([1.0])
    s = w.sum()
    return w / s


def make_triangle(n):
    """Create a triangle sequence."""
    j = np.arange(n)
    mid = (n - 1) / 2.0
    w = 1 - np.abs(j - mid) / (mid + 1e-10)
    w = np.maximum(w, 0.01)
    s = w.sum()
    return w / s


def make_raised_cosine(n, beta=1.0):
    """Create a raised cosine sequence."""
    j = np.arange(n)
    t = j / (n - 1) if n > 1 else np.array([0.5])
    w = np.cos(np.pi * (t - 0.5)) ** 2
    s = w.sum()
    return w / s

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Structured initializations at parent length (~60s) ===
    print(f"\n=== Phase 1: Structured Inits (n={n_parent}) ===")
    t_start = time.monotonic()

    structured_inits = {
        "parent": parent,
        "hann": make_hann(n_parent),
        "hamming": make_hamming(n_parent),
        "triangle": make_triangle(n_parent),
        "raised_cos": make_raised_cosine(n_parent),
        "uniform": np.ones(n_parent) / n_parent,
    }

    # Also try a few random inits
    for i in range(3):
        structured_inits[f"rand{i}"] = project_simplex(rng.standard_normal(n_parent) + 2.0)

    for name, init in structured_inits.items():
        remaining = deadline - time.monotonic() - 1500
        if remaining < 5:
            break
        (best_seq, best_score) = try_update(init, (best_seq, best_score), f"init_{name}")
        # Quick optimization from this init
        t_alloc = min(remaining, 8)
        a_opt, _ = smoothed_grad_optimize(init, time_limit=t_alloc, T_init=0.003, lr_init=0.02)
        (best_seq, best_score) = try_update(a_opt, (best_seq, best_score), f"quick_{name}")

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 2: Multi-n exploration (~200s) ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    candidate_ns = sorted(set([
        n_best, n_best // 2, n_best * 2,
        384, 512, 640, 768, 896, 1024, 1280, 1536
    ]))
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 2048]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1000
        if remaining < 20:
            break

        start = interp_sequence(best_seq, n_try)
        t_alloc = min(remaining, 50)

        # Quick FW
        t_fw = min(t_alloc * 0.5, 25)
        a_opt, _ = fw_optimize(start, time_limit=t_fw)
        (best_seq, best_score) = try_update(a_opt, (best_seq, best_score), f"fw_n{n_try}")

        # Quick smoothed grad
        remaining = deadline - time.monotonic() - 950
        if remaining > 10:
            t_sg = min(remaining, t_alloc * 0.4)
            a_opt2, _ = smoothed_grad_optimize(a_opt, time_limit=t_sg, T_init=0.002, lr_init=0.015)
            (best_seq, best_score) = try_update(a_opt2, (best_seq, best_score), f"sg_n{n_try}")

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 3: Deep FW optimization (~300s) ===
    print(f"\n=== Phase 3: Deep FW (n={len(best_seq)}) ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    remaining = deadline - time.monotonic() - 800
    if remaining > 30:
        t_alloc = min(remaining, 250)
        print(f"  Running FW with t={t_alloc:.0f}s")
        a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_update(a_opt, (best_seq, best_score), "deep_fw")

    # Perturbed FW restarts
    for scale in [0.003, 0.008, 0.015, 0.03]:
        remaining = deadline - time.monotonic() - 600
        if remaining < 15:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc = min(remaining, 50)
        a_opt, _ = fw_optimize(f_pert, time_limit=t_alloc)
        (best_seq, best_score) = try_update(a_opt, (best_seq, best_score), f"fw_pert_{scale}")

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 4: Peak equalization + Smoothed gradient (~300s) ===
    print(f"\n=== Phase 4: Peak Equalization & Smoothed Grad ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    # Peak equalization
    remaining = deadline - time.monotonic() - 500
    if remaining > 20:
        t_alloc = min(remaining, 120)
        print(f"  Peak equalize t={t_alloc:.0f}s")
        a_opt, _ = peak_equalize(best_seq, time_limit=t_alloc, n_peaks=5)
        (best_seq, best_score) = try_update(a_opt, (best_seq, best_score), "peak_eq")

    # Smoothed gradient with different T
    for T_init in [0.0005, 0.001, 0.003, 0.01]:
        remaining = deadline - time.monotonic() - 400
        if remaining < 10:
            break
        t_alloc = min(remaining, 80)
        a_opt, _ = smoothed_grad_optimize(best_seq, time_limit=t_alloc, T_init=T_init, lr_init=0.012)
        (best_seq, best_score) = try_update(a_opt, (best_seq, best_score), f"sg_T{T_init}")

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Evolutionary refinement (~250s) ===
    print(f"\n=== Phase 5: Evolutionary Refinement ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    remaining = deadline - time.monotonic() - 300
    if remaining > 20:
        t_alloc = min(remaining, 200)
        print(f"  Evolution t={t_alloc:.0f}s")
        a_opt, _ = evolutionary_refine(best_seq, time_limit=t_alloc, pop_size=6, n_gen_max=300)
        (best_seq, best_score) = try_update(a_opt, (best_seq, best_score), "evo")

    # Second evolutionary round with different params
    remaining = deadline - time.monotonic() - 200
    if remaining > 15:
        t_alloc = min(remaining, 100)
        a_opt, _ = evolutionary_refine(best_seq, time_limit=t_alloc, pop_size=8, n_gen_max=200)
        (best_seq, best_score) = try_update(a_opt, (best_seq, best_score), "evo2")

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 6: Spectral method + final polish (~250s) ===
    print(f"\n=== Phase 6: Spectral & Final Polish ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    # Spectral optimization
    remaining = deadline - time.monotonic() - 150
    if remaining > 20:
        t_alloc = min(remaining, 100)
        print(f"  Spectral t={t_alloc:.0f}s")
        a_opt, _ = spectral_optimize(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_update(a_opt, (best_seq, best_score), "spectral")

    # Final FW polish
    remaining = deadline - time.monotonic() - 100
    if remaining > 15:
        t_alloc = min(remaining, 60)
        a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_update(a_opt, (best_seq, best_score), "final_fw")

    # Tiny perturbations + quick optimize
    for i in range(4):
        remaining = deadline - time.monotonic() - 50
        if remaining < 8:
            break
        noise_scale = 0.001 * (i + 1)
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
        t_alloc = min(remaining, 15)
        a_opt, _ = smoothed_grad_optimize(f_pert, time_limit=t_alloc, T_init=0.001, lr_init=0.01)
        (best_seq, best_score) = try_update(a_opt, (best_seq, best_score), f"micro_pert_{i}")

    # Final peak equalization
    remaining = deadline - time.monotonic() - 30
    if remaining > 10:
        a_opt, _ = peak_equalize(best_seq, time_limit=min(remaining, 30), n_peaks=8)
        (best_seq, best_score) = try_update(a_opt, (best_seq, best_score), "final_peakeq")

    print(f"  Phase 6 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 7: Try a different n one more time ===
    remaining = deadline - time.monotonic() - 30
    if remaining > 30:
        n_best = len(best_seq)
        for n_try in [n_best + 64, n_best - 64, n_best + 128]:
            if n_try < 256 or n_try > 2048:
                continue
            remaining = deadline - time.monotonic() - 20
            if remaining < 10:
                break
            start = interp_sequence(best_seq, n_try)
            t_alloc = min(remaining, 20)
            a_opt, _ = fw_optimize(start, time_limit=t_alloc)
            (best_seq, best_score) = try_update(a_opt, (best_seq, best_score), f"final_n{n_try}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
