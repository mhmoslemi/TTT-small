The SCA iteration:
1. Given current f and bound M, solve the LP to get g
2. Rescale g to have the same norm as f
3. Blend: f_new = (1-t)*f + t*g
4. Update M based on the new max convolution
5. Repeat

The key parameter is M: it should be slightly above the current max to make the LP feasible but tight enough to be informative.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Adam optimizer with entropy-regularized subgradient (O(n log n) per step)
- SCA with LP oracle for polishing
- Structured simulated annealing for global exploration
- Multi-n exploration with diverse initializations
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_gradient(a, beta=100.0):
    """
    Compute the entropic-regularized subgradient of max_k (a*a)_k.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    conv = fast_convolve(a)
    size = 2 * n - 1
    
    # Entropic weights
    conv_shifted = conv - conv.max()
    w = np.exp(beta * conv_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(size)
        w[np.argmax(conv)] = 1.0
    else:
        w = w / w_sum
    
    # Compute gradient via FFT cross-correlation
    # grad_i = 2 * sum_k w_k * a_{k-i} for valid k,i
    # This equals 2 * (w_rev * a)[2n-2-i] where w_rev is reversed w
    w_rev = w[::-1]
    
    fft_size = 1 << (size + n - 1 - 1).bit_length()
    W = np.fft.rfft(w_rev, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(W * A, fft_size)[:size + n - 1]
    
    # grad_i = 2 * corr[2n-2-i] for i = 0, ..., n-1
    grad = 2.0 * corr[2*n-2-n+1:2*n-2+1][::-1]
    # Let me re-index: we want grad[i] = 2 * corr[2n-2-i]
    # for i in 0..n-1, that's corr[2n-2], corr[2n-3], ..., corr[n-1]
    # which is corr[n-1 : 2n-2+1][::-1]
    grad = 2.0 * corr[n-1:2*n-1][::-1]
    
    return grad, conv.max()

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def adam_optimize(a0, time_limit, beta_schedule='cosine', lr=0.01, 
                  beta_init=50.0, beta_final=500.0, verbose=False):
    """
    Adam optimization on the simplex with entropy-regularized subgradient.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        
        # Temperature schedule
        progress = min(1.0, elapsed / time_limit)
        if beta_schedule == 'cosine':
            beta = beta_final - (beta_final - beta_init) * 0.5 * (1 + np.cos(np.pi * progress))
        elif beta_schedule == 'linear':
            beta = beta_init + (beta_final - beta_init) * progress
        else:
            beta = beta_init
        
        # Compute gradient
        grad, current_max = fast_gradient(a, beta)
        
        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        
        # Learning rate decay
        lr_t = lr * np.sqrt(1 - b2 ** t) / (1 - b1 ** t)
        
        # Update
        a_raw = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_raw)
        
        # Line search (simple: try a few steps)
        best_candidate = a
        best_val = current_max
        for alpha in [1.0, 0.5, 0.25]:
            cand = (1 - alpha) * a + alpha * a_new
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_candidate = cand
        
        a = best_candidate
        obj = fast_convolve(a).max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > 2000:
            rng = np.random.default_rng((t * 12345 + 67890) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.01
        
        if verbose and t % 10000 == 0:
            print(f"    Adam t={t}, obj={best_obj:.10f}, beta={beta:.1f}, no_imp={no_improve}")
    
    return best_a, best_obj

def build_lp_sparse(n):
    """Build sparse constraint matrix for the LP oracle."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

def solve_lp_oracle(f, M, n, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    k_arr = row_idx
    i_arr = col_idx
    vals = f[k_arr - i_arr]
    
    A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, 
                      method='highs', options={'presolve': True, 'time_limit': 5})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def sca_optimize(a0, time_limit, verbose=False):
    """SCA with LP oracle."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    row_idx, col_idx = build_lp_sparse(n)
    
    conv = fast_convolve(a)
    M = conv.max() * 1.001
    best_a = a.copy()
    best_obj = conv.max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        
        g_raw = solve_lp_oracle(a, M, n, row_idx, col_idx)
        
        if g_raw is None:
            M *= 1.02
            no_improve += 1
            if no_improve > 30:
                M = fast_convolve(a).max() * 1.005
                no_improve = 0
            continue
        
        g = g_raw / g_raw.sum()
        
        if np.max(np.abs(g - a)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                rng = np.random.default_rng((it * 7919) % (2**31 - 1))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                M = fast_convolve(a).max() * 1.005
                no_improve = 0
            continue
        
        # Line search
        best_val = fast_convolve(a).max()
        best_ft = a.copy()
        for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * a + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_ft = ft
        
        if best_val < best_obj - 1e-13:
            a = best_ft
            best_obj = best_val
            best_a = a.copy()
            M = best_val * 0.999
            no_improve = 0
        else:
            no_improve += 1
            M *= 0.998
        
        if no_improve > 40:
            rng = np.random.default_rng((it * 31831 + 7) % (2**31 - 1))
            scale = 0.002 + 0.002 * min(1.0, (no_improve - 40) / 50.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            M = fast_convolve(a).max() * 1.005
            no_improve = 0
    
    return best_a, best_obj

def simulated_annealing(a0, time_limit, verbose=False):
    """
    Structured simulated annealing: move mass between positions.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    temp = 0.01
    temp_min = 0.0001
    cooling = 0.9995
    rng = np.random.default_rng(42)
    
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Choose a move: transfer mass from i to j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Amount to transfer (fraction of a[i])
        frac = rng.uniform(0.01, 0.1)
        delta = frac * a[i]
        
        if delta > a[i]:
            delta = a[i] * 0.5
        
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        new_obj = fast_convolve(a_new).max()
        old_obj = fast_convolve(a).max()
        delta_obj = new_obj - old_obj
        
        # Accept/reject
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / temp):
            a = a_new
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Cool down
        temp = max(temp_min, temp * cooling)
        
        #Restart if severely stuck
        if no_improve > 5000:
            a = best_a.copy()
            temp = 0.01
            no_improve = 0
        
        if verbose and it % 10000 == 0:
            print(f"    SA it={it}, obj={best_obj:.10f}, temp={temp:.6f}")
    
    return best_a, best_obj

def pair_descent(a0, time_limit, verbose=False):
    """
    Coordinate pair descent: for each pair (i,j), find optimal mass transfer.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    for sweep in range(10000):
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        improved = False
        # Random subset of pairs for speed
        rng = np.random.default_rng(sweep * 7919 + 13)
        num_pairs = min(200, n * n // 4)
        
        for _ in range(num_pairs):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            # Find best delta to move from i to j
            # Try several values
            best_delta = 0.0
            best_val = fast_convolve(a).max()
            
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                delta = frac * a[i]
                if delta <= 0:
                    continue
                a_test = a.copy()
                a_test[i] -= delta
                a_test[j] += delta
                a_test = np.maximum(a_test, 0)
                if a_test[i] < 0:
                    continue
                a_test = a_test / a_test.sum()
                val = fast_convolve(a_test).max()
                if val < best_val:
                    best_val = val
                    best_delta = delta
            
            # Also try moving from j to i
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                delta = frac * a[j]
                if delta <= 0:
                    continue
                a_test = a.copy()
                a_test[j] -= delta
                a_test[i] += delta
                a_test = np.maximum(a_test, 0)
                if a_test[j] < 0:
                    continue
                a_test = a_test / a_test.sum()
                val = fast_convolve(a_test).max()
                if val < best_val:
                    best_val = val
                    best_delta = -delta  # negative means j to i
            
            if best_delta != 0 and best_val < best_obj - 1e-14:
                if best_delta > 0:
                    a[i] -= best_delta
                    a[j] += best_delta
                else:
                    a[j] -= best_delta
                    a[i] += best_delta
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_obj = fast_convolve(a).max()
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_obj

def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Edge-heavy
    edge_w = 1.0 / (np.minimum(x, n - 1 - x).astype(float) + 1.0) ** 1.0
    inits.append(edge_w / edge_w.sum())
    
    # Periodic modulation
    for freq in [2, 3, 5]:
        mod = 1.0 + 0.3 * np.sin(2 * np.pi * freq * x / n)
        inits.append(mod / mod.sum())
    
    # Two-peak
    g2 = np.exp(-0.5 * ((x - 0.25 * n) / (0.05 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.05 * n)) ** 2)
    inits.append(g2 / g2.sum())
    
    # Anti-symmetric modulation (small in center)
    center = n / 2
    anti = 1.0 - 0.5 * np.exp(-0.5 * ((x - center) / (0.1 * n)) ** 2)
    anti = np.maximum(anti, 0.01)
    inits.append(anti / anti.sum())
    
    return inits
