def subgradient_max(a, top_k=10, temperature=0.0):
    """
    Compute (weighted) subgradient of max_k conv(a,a)[k].
    Returns (gradient, max_value, peak_indices).
    """
    n = len(a)
    conv = fast_convolve(a)
    if top_k >= len(conv):
        top_k = len(conv)
    
    # Get top-k peaks
    peak_indices = np.argpartition(conv, -top_k)[-top_k:]
    peak_values = conv[peak_indices]
    
    # Weight by temperature (0 = hard max)
    if temperature > 0:
        scaled = peak_values / temperature
        scaled -= scaled.max()
        weights = np.exp(scaled)
        weights /= weights.sum()
    else:
        # Hard max: uniform weight on all peaks within 1% of max
        max_val = conv.max()
        mask = peak_values > max_val * 0.99
        if not mask.any():
            mask = np.zeros(len(peak_indices), dtype=bool)
            mask[np.argmax(peak_values)] = True
        weights = np.zeros(len(peak_indices))
        weights[mask] = 1.0 / mask.sum()
    
    max_val = conv.max()
    
    # Compute gradient
    grad = np.zeros(n)
    for i in range(len(peak_indices)):
        if weights[i] < 1e-15:
            continue
        k = int(peak_indices[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        indices = np.arange(js, je + 1)
        partners = k - indices
        grad[indices] += weights[i] * 2.0 * a[partners]

                stall = 0
            else:
                stall += 1
        
        if not improved:
            stall += 1
        
        # Perturbation if stalled
        if stall > 15:
            scale = 0.001 * min(1.0 + stall / 20.0, 3.0)
            rng = np.random.default_rng((it * 48271 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            obj = compute_obj_raw(a)
            if obj > best_obj * 1.001:
                a = best_a.copy()
            stall = 0
        
        if stall > 200:
            break
    
    return best_a, best_obj


def spectral_equalize(a0, time_limit, n_iters_target=500):
    """
    CMA-style spectral equalization: alternate between flattening the power
    spectrum in frequency domain and projecting to the simplex in time domain.
    """
    a = a0.astype(np.float64).copy()
    a = a / a.sum()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_obj = compute_obj_raw(a)
    
    t0 = time.monotonic()
    alpha = 0.3  # blending factor
    beta = 0.5   # spectrum flattening strength
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Compute current spectrum
        A = np.fft.fft(a, fft_size)
        P = np.abs(A) ** 2
        
        # Target: geometric mean of non-zero bins
        nonzero = P > 1e-30
        if nonzero.sum() > 1:
            log_P = np.zeros_like(P)
            log_P[nonzero] = np.log(P[nonzero])
            target_log = np.mean(log_P[nonzero])
            P_target = np.exp(target_log)
        else:
            P_target = np.mean(P)
        
        # Flatten: adjust amplitudes
        P_new = (1.0 - beta) * P + beta * P_target
        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        
        # Optionally reduce DC to maintain zero-mean-like property
        # (not required, but can help)
        
        # Inverse FFT and project
        a_new = np.real(np.fft.ifft(A_new, fft_size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        
        # Blend with current
        a_blend = (1.0 - alpha) * a + alpha * a_new
        a_blend = project_simplex(a_blend)
        
        obj = compute_obj_raw(a_blend)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_blend.copy()
            a = a_blend
        else:
            # Try smaller step
            for trial_alpha in [alpha * 0.5, alpha * 0.1]:
                a_trial = (1.0 - trial_alpha) * a + trial_alpha * a_new
                a_trial = project_simplex(a_trial)
                obj_trial = compute_obj_raw(a_trial)
                if obj_trial < best_obj - 1e-15:
                    best_obj = obj_trial
                    best_a = a_trial.copy()
                    a = a_trial
                    break
            else:
                # No improvement: adjust parameters
                alpha = max(0.05, alpha * 0.95)
                beta = min(0.9, beta * 1.02)
        
        # Periodic restart
        if it % 200 == 0 and compute_obj_raw(a) > best_obj * 1.005:
            a = best_a.copy()
    
    return best_a, best_obj


def basin_hopping(a0, time_limit, n_samples=5):
    """
    Basin hopping: random perturbations with greedy acceptance.
    Uses structured perturbations biased toward peak reduction.
    """
    a = a0.astype(np.float64).copy()
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = compute_obj_raw(a)
    
    t0 = time.monotonic()
    it = 0
    n_no_improve = 0
    base_scale = 0.001
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Adaptive scale
        scale = base_scale * (1.0 + 0.2 * np.sin(it * 0.01))
        if n_no_improve > 100:
            scale *= (1.0 + n_no_improve / 200.0)
        
        rng = np.random.default_rng((it * 2654435761) % (2**32))
        
        # Try multiple perturbations
        found = False
        for _ in range(n_samples):
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            
            # Structured perturbation
            conv = fast_convolve(a)
            peak_k = int(np.argmax(conv))
            
            noise = rng.standard_normal(n) * scale
            
            # Bias: reduce coordinates that contribute to the peak
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            for j in range(js, je + 1):
                partner = peak_k - j
                if 0 <= partner < n:
                    noise[j] -= scale * a[partner] * 1.5
                    noise[partner] -= scale * a[j] * 1.5
            
            candidate = project_simplex(a + noise)
            obj = compute_obj_raw(candidate)
            
            if obj < best_obj - 1e-15:
                a = candidate
                best_obj = obj
                best_a = a.copy()
                n_no_improve = 0
                found = True
                break
        
        if not found:
            n_no_improve += 1
        
        # Occasionally try larger jumps
        if it % 500 == 0:
            rng2 = np.random.default_rng((it * 7 + 13) % (2**31))
            big_noise = rng2.standard_normal(n) * scale * 5.0
            candidate = project_simplex(a + big_noise)
            obj = compute_obj_raw(candidate)
            if obj < best_obj - 1e-15:
                a = candidate
                best_obj = obj
                best_a = a.copy()
                n_no_improve = 0
    
    return best_a, best_obj


def multi_n_explore(a_best, time_limit):
    """Try interpolating to nearby sequence lengths and quick-refining."""
    n_best = len(a_best)
    candidates = sorted(set([n_best - 32, n_best - 16, n_best + 16, n_best + 32]))
    candidates = [n for n in candidates if 128 <= n <= 2000]
    
    best_overall = a_best.copy()
    best_score = evaluate_sequence(a_best.tolist())
    
    t0 = time.monotonic()
    time_per = max(1.0, (time_limit - 1.0) / max(1, len(candidates)))
    
    for n_try in candidates:
        if time.monotonic() - t0 > time_limit - time_per:
            break
        
        # Interpolate
        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, n_try)
        a_interp = np.interp(x_new, x_old, a_best)
        a_interp = np.maximum(a_interp, 0)
        s = a_interp.sum()
        if s < 1e-15:
            continue
        a_interp = a_interp / s
        
        score = evaluate_sequence(a_interp.tolist())
        if score < best_score:
            best_score = score
            best_overall = a_interp.copy()
        
        # Quick Nesterov refinement
        t_alloc = max(2.0, time_per - 2.0)
        a_ref, _ = nesterov_optimize(a_interp, time_limit=t_alloc, top_k=10, lr_init=0.002)
        score_ref = evaluate_sequence(a_ref.tolist())
        if score_ref < best_score:
            best_score = score_ref
            best_overall = a_ref.copy()
            print(f"    Multi-n: n={n_try}, score={score_ref:.8f}")
    
    return best_overall, best_score


def final_polish(a0, time_limit):
    """Ultra-fine pairwise optimization for the last bit of improvement."""
    a = a0.astype(np.float64).copy()
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = compute_obj_raw(a)
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        conv = fast_convolve(a)
        # Get top 3 peaks
        num_peaks = min(3, len(conv))
        peak_lags = np.argpartition(conv, -num_peaks)[-num_peaks:]
        
        improved = False
        for kp in peak_lags:
            if time.monotonic() - t0 > time_limit - 0.3:
                break
            
            kp = int(kp)
            js = max(0, kp - n + 1)
            je = min(n - 1, kp)
            affected = np.arange(js, je + 1)
            
            if len(affected) < 2:
                continue
            
            # Pick a few promising pairs
            contributions = np.array([a[j] * a[kp - j] for j in affected
                                      if 0 <= kp - j < n])
            
            # Try top contributor with random others
            top_src = affected[np.argmax(contributions)]
            
            for attempt in range(5):
                if time.monotonic() - t0 > time_limit - 0.2:
                    break
                
                rng = np.random.default_rng((int(kp) * 100 + attempt + int(time.monotonic() * 10)) % (2**31))
                dst = rng.choice(affected)
                if dst == top_src or a[top_src] < 1e-8:
                    continue
                
                a_test, _ = pair_ternary_search(a.copy(), int(top_src), int(dst), n_iter=25)
                obj = compute_obj_raw(a_test)
                
                if obj < best_obj - 1e-14:
                    a = a_test
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            # Tiny perturbation
            rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.0003)
            obj = compute_obj_raw(a)
            if obj > best_obj:
                a = best_a.copy()
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search combining Nesterov subgradient,
    pairwise coordinate optimization, spectral equalization, basin hopping,
    multi-n exploration, and final micro-polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    t0_total = time.monotonic()
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t0_total
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        parent = np.ones(384) / 384
        try_seq(parent, "uniform")
    
    n_parent = len(parent)
    
    # === Phase 1: Nesterov Subgradient (~400s) ===
    print("\n=== Phase 1: Nesterov Subgradient ===")
    t_phase = time.monotonic()
    
    configs = [
        {'top_k': 15, 'lr_init': 0.003},
        {'top_k': 8, 'lr_init': 0.005},
        {'top_k': 25, 'lr_init': 0.002},
    ]
    
    for ci, cfg in enumerate(configs):
        remaining = deadline - time.monotonic()
        t_alloc = min(130, remaining - 1200)
        if t_alloc < 20:
            break
        print(f"  Nesterov k={cfg['top_k']}, lr={cfg['lr_init']}, t={t_alloc:.0f}s")
        a_opt, _ = nesterov_optimize(best_seq, time_limit=t_alloc, top_k=cfg['top_k'], lr_init=cfg['lr_init'])
        try_seq(a_opt, f"nest_{ci}")
    
    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 2: Peak Pair Optimization (~400s) ===
    print("\n=== Phase 2: Peak Pair Optimization ===")
    t_phase = time.monotonic()
    
    remaining = deadline - time.monotonic()
    t_alloc = min(350, remaining - 800)
    if t_alloc > 30:
        print(f"  Pair opt, t={t_alloc:.0f}s")
        a_opt, _ = peak_pair_optimize(best_seq, time_limit=t_alloc, n_pairs_per_round=300)
        try_seq(a_opt, "pairs")
    
    # Second pass with different settings
    remaining = deadline - time.monotonic()
    t_alloc = min(200, remaining - 600)
    if t_alloc > 20:
        print(f"  Pair opt 2, t={t_alloc:.0f}s")
        a_opt, _ = peak_pair_optimize(best_seq, time_limit=t_alloc, n_pairs_per_round=500)
        try_seq(a_opt, "pairs2")
    
    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 3: Spectral Equalization (~300s) ===
    print("\n=== Phase 3: Spectral Equalization ===")
    t_phase = time.monotonic()

        print(f"  Spectral, t={t_alloc:.0f}s")

    # === Phase 6: Final Polish (~150s) ===
    print("\n=== Phase 6: Final Polish ===")
    t_phase = time.monotonic()
    
    remaining = deadline - time.monotonic()
    t_alloc = min(100, remaining - 50)
    if t_alloc > 10:
        print(f"  Final polish, t={t_alloc:.0f}s")
        a_opt, _ = final_polish(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "polish")
    
    # One more Nesterov pass
    remaining = deadline - time.monotonic()
    t_alloc = min(40, remaining - 20)
    if t_alloc > 5:
        print(f"  Final Nesterov, t={t_alloc:.0f}s")
        a_opt, _ = nesterov_optimize(best_seq, time_limit=t_alloc, top_k=10, lr_init=0.001)
        try_seq(a_opt, "final_nest")
    
    # Final micro-polish
    remaining = deadline - time.monotonic()
    t_alloc = min(20, remaining - 5)
    if t_alloc > 3:

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, total time={time.monotonic()-t0_total:.0f}s)")
    return [float(x) for x in best_seq]
