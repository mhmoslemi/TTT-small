"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Fast FFT-based RG peak shaving with adaptive step size and softmax-weighted peaks
- Multiplicative weights (mirror descent) variant for better conditioning
- Multi-n exploration with interpolation/reinterpolation
- Simulated annealing for global exploration
- Boundary zeroing heuristic to eliminate trivial boundary peaks
- Coordinate descent polish for final refinement

import numpy as np
import time
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, w, max_sig=50):
    """Compute weighted subgradient: g[i] = 2 * sum_k w[k] * a[k-i]."""
    n = len(a)
    g = np.zeros(n)
    sig = np.argsort(w)[::-1][:max_sig]
    for ki in sig:
        k = int(ki)
        wk = w[k]
        if wk < 1e-14:
            break
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[lo:hi + 1] += 2.0 * wk * a[k - idx]
    return g


def peak_shave_rg(a0, time_limit, k_soft=15, use_mw=False):
    """
    RG-style peak shaving with softmax-weighted multiple peaks.
    use_mw: if True, use multiplicative weights update instead of projected gradient.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_a = a.copy()
    best_obj = conv.max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    temperature = 0.01
    consec_worse = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Softmax weights over peaks
        T = max(temperature, 1e-7)
        shifted = (conv - M) / T
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            # All mass on one peak
            peak_idx = np.argmax(conv)
            w = np.zeros(len(conv))
            w[peak_idx] = 1.0
        else:
            w /= w_sum

        # Compute subgradient
        g = compute_subgradient(a, w, max_sig=max(10, 2 * k_soft))

        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-20:
            break

        # Adaptive step size
        base_eta = M / (g_norm_sq + 1e-20)
        # Safe step (don't go negative)
        pos_mask = g > 1e-15
        if pos_mask.any():
            safe_eta = (a[pos_mask] / g[pos_mask]).min()
            safe_eta = max(safe_eta * 0.99, 0)
        else:
            safe_eta = 1.0

        max_eta = min(1.0, safe_eta)
        if max_eta < 1e-10:
            max_eta = 1e-8

        # Try geometrically spaced step sizes
        n_tries = 8
        if base_eta > 1e-10:
            log_lo = math.log(max(base_eta * 0.01, 1e-10))
            log_hi = math.log(min(base_eta * 10, max_eta))
            if log_hi <= log_lo:
                log_hi = log_lo + 0.5
            etas = np.exp(np.linspace(log_lo, log_hi, n_tries))
        else:
            etas = np.linspace(1e-6, max_eta, n_tries)

        etas = np.clip(etas, 0, max_eta)
        etas = sorted(set(etas.tolist()), reverse=True)

        best_eta = 0.0
        best_val = M

        for eta in etas:
            if eta < 1e-12:
                continue
            if use_mw:
                log_a = np.log(a + 1e-30) - eta * g
                log_a -= log_a.max()
                a_new = np.exp(log_a)
                sn = a_new.sum()
                if sn < 1e-15:
                    continue
                a_new = a_new / sn
            else:
                a_new = a - eta * g
                a_new = np.maximum(a_new, 0.0)
                sn = a_new.sum()
                if sn < 1e-15:
                    continue
                a_new = a_new / sn

            val = fast_convolve(a_new).max()
            if val < best_val:
                best_val = val
                best_eta = eta

        if best_eta > 1e-12:
            if use_mw:
                log_a = np.log(a + 1e-30) - best_eta * g
                log_a -= log_a.max()
                a = np.exp(log_a)
                a = a / a.sum()
            else:
                a = a - best_eta * g
                a = np.maximum(a, 0.0)
                a = a / a.sum()
        else:
            consec_worse += 1
            if consec_worse > 20:
                break

        # Adapt temperature: gradually focus on top peak
        if it % 50 == 0:
            temperature *= 0.95
            temperature = max(temperature, 0.0005)

        # Periodic perturbation to escape local minima
        if no_improve > 80:
            rng = np.random.default_rng((it * 2654435761) % (2**31))
            scale = 0.002 + 0.003 * min(1.0, (no_improve - 80) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            no_improve = 0
            consec_worse = 0
            temperature = 0.01

        if no_improve > 1000:
            break

    return best_a, best_obj


def mul_weight_descent(a0, time_limit, lr_init=0.05, decay=0.999):
    """
    Mirror descent (multiplicative weights) on the simplex.
    Naturally maintains positivity.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_a = a.copy()
    best_obj = conv.max()

    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    temperature = 0.01

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_a = a.copy()
            no_improve = 0

        # Softmax over peaks
        T = max(temperature, 1e-7)
        shifted = (conv - M) / T
        w = np.exp(shifted)
        w /= w.sum()

        # Subgradient
        g = compute_subgradient(a, w, max_sig=30)

        # Multiplicative weights update with line search
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-20:
            break

        base_lr = M / (g_norm_sq + 1e-20)
        candidates = [base_lr * f for f in [0.1, 0.3, 0.5, 1.0, 2.0, 5.0]]

        best_lr = 0.0
        best_val = M
        for trial_lr in candidates:
            log_a = np.log(a + 1e-30) - trial_lr * g

            val = fast_convolve(a_new).max()
            if val < best_val:
                best_val = val
                best_lr = trial_lr

            log_a = np.log(a + 1e-30) - best_lr * g
            log_a -= log_a.max()
            a = np.exp(log_a)
            a = a / a.sum()
            no_improve += 1
        else:
            no_improve += 1

        # Anneal temperature

        # Restart

def coordinate_polish(a0, time_limit):

    Targeted coordinate descent: for each top peak, try mass transfers
    that reduce it. Uses golden section on 2D sub-problems.

            break

        conv = fast_convolve(a)
        M = conv.max()

            best_obj = M
            best_a = a.copy()

        peak_lags = np.argpartition(conv, -top_k)[-top_k:]
        peak_lags = peak_lags[np.argsort(conv[peak_lags])[::-1]]
        peak_lags = peak_lags.tolist()

        improved = False

        for k_peak in peak_lags:
            if time.monotonic() - t0 > time_limit - 1.0:
                break
            k_peak = int(k_peak)

            # Variables affecting this peak
            lo = max(0, k_peak - n + 1)
            hi = min(n - 1, k_peak)
            affected = np.arange(lo, hi + 1)

            if len(affected) < 2:
                continue

            # For each source (high contributor), try moving mass to a destination
            # that doesn't contribute much to any peak
            # Score: contribution to this peak = a[k_peak - j]
            contributions = np.array([a[k_peak - j] for j in affected])

            # Sample pairs for efficiency
            rng = np.random.default_rng(it * 31 + k_peak)

            if n_affected <= 40:
                sources = affected
            else:
                # Focus on top contributors
                top_src = np.argsort(contributions)[::-1][:40]
                sources = affected[top_src]

            # Destinations: low contribution to all top peaks
            peak_contrib = np.zeros(n)
            for pk in peak_lags[:2]:
                pk = int(pk)
                lo2 = max(0, pk - n + 1)
                hi2 = min(n - 1, pk)
                for j in range(lo2, hi2 + 1):
                    peak_contrib[j] += a[pk - j]

            # Distant from peaks (low peak_contrib)
            low_contrib = np.argsort(peak_contrib)[:50]

            best_gain = 0.0

            best_dst = -1
            best_frac = 0.0

            for src in sources:
                src = int(src)
                if a[src] < 1e-10:
                    continue
                for dst in low_contrib:
                    dst = int(dst)
                    if dst == src:
                        continue
                    # Try small transfers
                    for frac in [0.05, 0.1, 0.2]:
                        delta = a[src] * frac
                        if delta < 1e-12:
                            continue
                        a_test = a.copy()

                        a_test[dst] += delta
                        a_test = np.maximum(a_test, 0.0)
                        s = a_test.sum()
                        a_test = a_test / s
                        c_new = fast_convolve(a_test)
                        gain = M - c_new.max()
                        if gain > best_gain:

                            best_src = src

            scale = 0.002 * min(2.0, stall / 30.0)

            stall = 0

        if stall > 300:

    return best_a, best_obj


def simulated_annealing(a0, time_limit, n_restarts=50):
    """
    Simulated annealing on the probability simplex.
    Proposal: random mass transfer between two coordinates.
    Acceptance: Metropolis criterion on the peak value.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    current_obj = best_obj

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    T_start = 0.005
    T_end = 0.0001
    total_steps = n_restarts * 1000

    while time.monotonic() - t0 < time_limit:
        # Current temperature (exponential cooling)
        elapsed = time.monotonic() - t0
        T = T_start * (T_end / T_start) ** (elapsed / time_limit)
        T = max(T, T_end)

        # Batch of proposals
        batch = min(200, total_steps)
        for _ in range(batch):
            # Random mass transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            frac = rng.uniform(0.01, 0.3)
            delta = a[i] * frac
            if delta < 1e-10:
                continue

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()

            new_obj = fast_convolve(a_new).max()

            # Metropolis acceptance
            if new_obj < current_obj:
                a = a_new
                current_obj = new_obj
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
            else:
                ratio = np.exp(-(new_obj - current_obj) / T)
                if rng.random() < ratio:
                    a = a_new
                    current_obj = new_obj

        if time.monotonic() - t0 > time_limit - 1.0:
            break

    return best_a, best_obj


def interpolate_sequence(source, target_n):
    """Interpolate a sequence to a different length."""
    n_src = len(source)
    if n_src == target_n:
        return source.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, target_n)
    result = np.interp(x_tgt, x_src, source)
    return np.maximum(result, 0.0)

            start = start / start.sum()
            print(f"  Final RG n={n_try}, t={t_alloc:.0f}s", end=" ")
            a_opt, obj = peak_shave_rg(start, time_limit=t_alloc, k_soft=10)
            sc = try_seq(a_opt, f"final_n{n_try}")
            print(f"-> obj={obj:.8f}, score={sc:.8f}")

    # Very final: one more RG pass
    if best_seq is not None:
        t_alloc = max(5, deadline - time.monotonic() - 5)
        if t_alloc > 3:
            a_opt, obj = peak_shave_rg(best_seq, time_limit=t_alloc, k_soft=5)
            try_seq(a_opt, "very_final")

    print(f"  Phase 6 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, total time={time.monotonic()-t0_total:.0f}s)")
    return [float(x) for x in best_seq]
