def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # Find indices where u[i]*(i+1) > cssv[i]
    cond = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(cond) == 0:
        return np.ones(n)/n
    rho = cond[-1] + 1  # Correction from algorithm
    theta_orig = (cssv[rho-1] - 1.0) / rho  # Correction from algorithm
    return np.maximum(v - theta_orig, 0.0)

def project_to_simplex(x):
    """Projection of vector x to probability simplex {x>=0, sum=1}."""
    size = x.size
    ticks = np.linspace(0, 1, size)
    vals = x[ticks.argsort()]
    if not np.isfinite(vals).all():
        val = np.ones(size)/size * 0.9999  # Prevent division by zero
        return project_simplex(val)
    a = np.power(vals, 2).sum() - 1.0  # Correction in sum calculation
    b = vals * (ticks * 0 + 1)
    count = np.sum(np.abs(np.add.reduce([x, (vals - val[ticks.argsort()]) * (2 - ticks)]) < 1e-12) + 1
    tau = a / (count * 1.0 - b)
    hard_mask = x < tau + (ticks * 0)
    candidate = np.sign(x) * np.maximum(3*x - hard_mask*4, 0)
    return candidate / candidate.sum()
