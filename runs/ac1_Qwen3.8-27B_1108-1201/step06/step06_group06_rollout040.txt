    """Euclidean projection onto probability simplex."""

"""
Optimized minimax convolution step function search.
Uses FFT-accelerated smoothed gradient descent with adaptive temperature,
coordinate descent, simulated annealing, and multi-length exploration.
Key insight: for non-negative f, the convolution peak can be anywhere,
so we use softmax-weighted subgradients to simultaneously reduce top peaks.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad(a, p):
    """Compute gradient of sum_k p[k]*(a*a)[k] w.r.t. a.
    grad[j] = 2 * sum_i a[i] * p[i+j] for i=0..n-1
    Using FFT for efficiency.
    """
    n = len(a)
    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    # grad[j] = 2 * dot(a, p[j:j+n])
    # This equals 2 * (a * p_reversed)[2n-2-j] where * is linear convolution
    # FFT approach: IFFT(FFT(a) * FFT(p_reversed))
    p_rev = p[::-1]
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p_rev, fft_size)
    conv = np.fft.irfft(A * P, fft_size)[:conv_len]
    # grad[j] = 2 * conv[2n-2-j] for j=0..n-1
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return grad

def coordinate_descent(a0, time_limit, seed=42):
    """
    Coordinate descent: adjust pairs of coordinates to reduce the max convolution.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        it += 1

        improved = False
        # Try random pairs
        n_tries = min(200, max(50, 3000000 // max(n, 1)))
        for _ in range(n_tries):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            # Move amount from i to j (or vice versa)
            for direction in [1, -1]:
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    amount = a[i] * frac if direction == 1 else a[j] * frac
                    if amount < 1e-15:
                        continue
                    candidate = a.copy()
                    if direction == 1:
                        candidate[i] -= amount
                        candidate[j] += amount
                    else:
                        candidate[j] -= amount
                        candidate[i] += amount
                    if candidate[i] < -1e-12 or candidate[j] < -1e-12:
                        continue
                    candidate = np.maximum(candidate, 0)
                    ss = candidate.sum()
                    if ss < 1e-10:
                        continue
                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved and it > 200:
            # Try broader random perturbations
            for scale in [0.001, 0.003, 0.008]:
                pert = rng.standard_normal(n) * scale
                candidate = project_simplex(a + pert)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and it > 500:
            break

    return best_a, best_obj


def simulated_annealing(a0, time_limit, seed=42):
    """
    Simulated annealing: random moves with Boltzmann acceptance.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    current_obj = best_obj

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    temp = 0.0005
    min_temp = 1e-7
    cooling = 0.9995

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        # Random move
        move_type = rng.integers(0, 3)
        if move_type == 0:
            # Gaussian perturbation
            scale = temp * 10
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(a + pert)
        elif move_type == 1:
            # Coordinate swap
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                candidate = a.copy()
                amt = min(a[i], a[j]) * rng.uniform(0.05, 0.3)
                candidate[i] -= amt
                candidate[j] += amt
                candidate = np.maximum(candidate, 0)
                ss = candidate.sum()
                if ss > 1e-10:
                    candidate = candidate / ss
                else:
                    candidate = a
        else:
            # Local block modification
            start = int(rng.integers(0, n - 10))
            end = start + int(rng.integers(5, min(20, n - start)))
            candidate = a.copy()
            candidate[start:end] += rng.standard_normal(end - start) * temp * 5
            candidate = project_simplex(candidate)

        c = fast_convolve(candidate)
        new_obj = c.max()
        delta = new_obj - current_obj

        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            a = candidate
            current_obj = new_obj
            if new_obj < best_obj - 1e-15:
                best_obj = new_obj
                best_a = a.copy()

        temp = max(temp * cooling, min_temp)

    return best_a, best_obj


def multi_peak_gradient(a0, time_limit):
    """
    Focus on reducing the top K peaks simultaneously.
    Uses a "peeling" strategy: identify active peaks and minimize their sum.
    """

            break
        it += 1

        conv = fast_convolve(a)
        max_val = conv.max()

        # Find top K peaks
        top_k = np.argsort(conv)[::-1][:K]
        # Create soft focus: weight peaks by how close they are to max
        weights = np.zeros(len(conv))
        for k in top_k:
            weights[k] = np.exp(-(conv.max() - conv[k]) / 0.001)
        w_sum = weights.sum()
        if w_sum > 1e-15:
            p = weights / w_sum
        else:
            p = np.ones(len(conv)) / len(conv)

        grad = compute_grad(a, p)

        # Backtracking
        best_candidate = a
        best_val = max_val
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            if c.max() < best_val - 1e-15:
                best_val = c.max()
                best_candidate = candidate
                break
            if c.max() < best_val:
                best_val = c.max()
                best_candidate = candidate

        a = best_candidate
        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()

        # Adapt K and lr
        if it % 100 == 0:

    return best_a, best_obj

def interpolate_sequence(src, new_n):
    """Interpolate a sequence to a new length."""
    n_src = len(src)
    if n_src == new_n:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_new = np.linspace(0, 1, new_n)
    result = np.interp(x_new, x_src, src)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(new_n) / new_n
    return result


def random_structured_init(n, seed):
    """Generate a random structured initialization."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    choice = int(rng.integers(0, 6))

    if choice == 0:
        # Exponential decay
        rate = rng.uniform(0.002, 0.02)
        f = np.exp(-rate * x)
    elif choice == 1:
        # Two-sided Gaussian (off-center)
        center = rng.uniform(0.1, 0.4) * n
        width = rng.uniform(0.05, 0.2) * n
        f = np.exp(-0.5 * ((x - center) / width) ** 2)
    elif choice == 2:
        # Power law
        alpha = rng.uniform(0.3, 1.5)
        f = 1.0 / (x + 1.0) ** alpha
    elif choice == 3:
        # Sinusoidal modulation
        freq = rng.uniform(1, 5)
        f = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        # Triangular (off-center)
        peak = rng.integers(n // 4, 3 * n // 4)
        f = np.maximum(0, 1 - np.abs(x - peak) / (n * 0.6))
    else:
        # Random smooth
        f = np.exp(-0.005 * x) * (1 + 0.3 * np.sin(2 * np.pi * 3 * x / n))

    f = np.maximum(f, 0)
    s = f.sum()
    if s > 1e-15:
        f = f / s
    else:
        f = np.ones(n) / n
    return f

        best_seq, best_score = try_seq(a_opt, f"perturb_{pert_scale}", (best_seq, best_score), verbose=False)

    # Try adjacent lengths
    for n_adj in [n_best - 64, n_best + 64, n_best - 128, n_best + 128]:
        if n_adj < 128 or n_adj > 2560:
            continue
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        start_adj = interpolate_sequence(best_seq, n_adj)
        t_alloc = min(remaining * 0.06, 40)
        a_opt, _ = smoothed_grad_optimize(start_adj, time_limit=t_alloc,
                                           T_start=0.003, T_end=0.0001, lr_start=0.05)
        best_seq, best_score = try_seq(a_opt, f"adj_n{n_adj}", (best_seq, best_score), verbose=False)

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Coordinate descent and simulated annealing ===
    print("\n=== Phase 4: Local Polish ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t_alloc = min(remaining * 0.2, 120)
        print(f"  Coordinate descent (t={t_alloc:.0f}s)")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc, seed=12345)
        best_seq, best_score = try_seq(a_opt, "coord", (best_seq, best_score), verbose=False)

    # Simulated annealing
    remaining = deadline - time.monotonic()
    if remaining > 50:
        t_alloc = min(remaining * 0.15, 80)
        print(f"  Simulated annealing (t={t_alloc:.0f}s)")
        a_opt, _ = simulated_annealing(best_seq, time_limit=t_alloc, seed=999)
        best_seq, best_score = try_seq(a_opt, "anneal", (best_seq, best_score), verbose=False)

    # Fine smoothed gradient
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_alloc = min(remaining * 0.2, 60)
        print(f"  Fine smoothed (t={t_alloc:.0f}s)")
        a_opt, _ = smoothed_grad_optimize(best_seq, time_limit=t_alloc,
                                           T_start=0.0005, T_end=0.00005, lr_start=0.02)
        best_seq, best_score = try_seq(a_opt, "fine_smooth", (best_seq, best_score), verbose=False)

    # Another coordinate descent pass
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t_alloc = min(remaining * 0.2, 40)
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc, seed=77777)
        best_seq, best_score = try_seq(a_opt, "coord2", (best_seq, best_score), verbose=False)

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Ultra-fine final tuning ===
    print("\n=== Phase 5: Ultra-Fine Tuning ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)

    # Very small perturbations with gradient refinement
    for _ in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        noise = rng.standard_normal(n_best) * 0.0003
        perturbed = project_simplex(best_seq + noise)
        t_alloc = min(remaining * 0.1, 10)
        a_opt, _ = smoothed_grad_optimize(perturbed, time_limit=t_alloc,
                                           T_start=0.0003, T_end=0.00005, lr_start=0.01)
        best_seq, best_score = try_seq(a_opt, "ultra_fine", (best_seq, best_score), verbose=False)

    # Final coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 10:
        t_alloc = min(remaining * 0.3, 20)
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc, seed=424242)
        best_seq, best_score = try_seq(a_opt, "final_coord", (best_seq, best_score), verbose=False)

    # Final game iteration
    remaining = deadline - time.monotonic()
    if remaining > 8:
        t_alloc = min(remaining * 0.4, 15)
        a_opt, _ = game_iteration(best_seq, time_limit=t_alloc, T_start=0.0005)
        best_seq, best_score = try_seq(a_opt, "final_game", (best_seq, best_score), verbose=False)

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
