Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses subgradient descent with mirror mapping, Nesterov acceleration,
multi-peak weighting, adaptive step sizes, structured initializations,
and periodic restarts to escape local minima.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgrad_at_peak(a, k):
    """Subgradient of autocorrelation at lag k with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    js = np.arange(lo, hi + 1)
    grad[js] = 2.0 * a[k - js]
    return grad


def multi_peak_subgrad(a, top_k=5):
    """Weighted subgradient from top-k peaks."""
    n = len(a)
    conv = fast_convolve(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    total = pv.sum()
    if total < 1e-30:
        return conv, np.zeros(n)
    weights = pv / total
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-15:
            continue
        grad += weights[i] * subgrad_at_peak(a, int(peaks[i]))
    return conv, grad

    log_a = np.log(a + 1e-30) - eta * grad
    log_a -= log_a.max()
    a_new = np.exp(log_a)

    if s < 1e-30:

    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    eta = eta_init
    mom = mom_init
    v = np.zeros(n)
    a_prev = a.copy()
    y = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv, grad = multi_peak_subgrad(y, top_k)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        v = mom * v - eta * grad
        y = mirror_step(a, v, 1.0)

        # Cross-step (Nesterov)
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        a_new = (gamma * y - (it + 1.0) * y + it * a_prev + it * y) / (gamma + it)
        # Simplified: just use y as new a
        a_prev = a.copy()
        a = y

        # Adaptive step size
        if no_improve > 30:
            eta *= 1.05
        if no_improve > 100:
            eta *= 0.95
        if no_improve > 400:
            rng = np.random.default_rng((it * 31627 + 1) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            eta = eta_init
            mom = mom_init
            no_improve = 0
            it = 0
            a_prev = a.copy()
            y = a.copy()

        # Slowly increase momentum
        mom = min(0.95, mom + 0.0002)

    return best_a, best_obj


def plain_subgrad(a0, time_limit, eta_init=0.03, top_k=3):
    """Plain subgradient with mirror mapping (no momentum)."""
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    eta = eta_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv, grad = multi_peak_subgrad(a, top_k)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = mirror_step(a, grad, eta)

        if no_improve > 20:
            eta *= 1.03
        if no_improve > 80:
            eta *= 0.97
        if no_improve > 300:
            rng = np.random.default_rng((it * 48271 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            eta = eta_init
            no_improve = 0

    return best_a, best_obj


def smoothed_softmax_descent(a0, time_limit, T_init=0.001, lr_init=0.02):
    """Smoothed objective via softmax peak weighting + gradient step."""
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Weighted subgradient
        grad = np.zeros(n)
        for k_idx in range(len(conv)):
            if p[k_idx] < 1e-15:
                continue
            grad += p[k_idx] * subgrad_at_peak(a, k_idx)

        # Projected gradient step
        candidate = project_simplex(a - lr * grad)
        c_val = fast_convolve(candidate).max()

        if c_val < best_obj - 1e-14:
            best_obj = c_val
            best_a = candidate.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = candidate

        if no_improve > 15:
            T *= 1.03
            lr *= 0.98
        if no_improve > 50:
            T *= 0.7
        if no_improve > 200:
            rng = np.random.default_rng((it * 69621 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    """Cyclic coordinate descent optimizing one coordinate at a time."""
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            # Optimize a[j] given rest fixed
            # c[n-1] = sum a[i]^2 includes a[j]^2
            # Other peaks: c[k] includes a[j]*a[k-j]
            # For simplicity, just try moving mass from j to the min coordinate
            j_min = int(np.argmin(a))
            if j_min == j:
                continue
            for frac in [0.001, 0.005, 0.01, 0.02]:
                delta = a[j] * frac
                if delta < 1e-12:
                    continue
                candidate = a.copy()
                candidate[j] -= delta
                candidate[j_min] += delta
                if candidate[j] < 0:
                    continue
                c_val = fast_convolve(candidate).max()
                if c_val < best_obj - 1e-15:
                    a = candidate
                    best_obj = c_val
                    best_a = a.copy()
                    improved = True
                    break
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                rng = np.random.default_rng(it * 12345 + 1)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                no_improve = 0
                if fast_convolve(a).max() < best_obj:
                    best_obj = fast_convolve(a).max()
                    best_a = a.copy()

    return best_a, best_obj


def generate_structured(n, rng):
    """Generate diverse structured initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    # Cosine taper
    c = 1.0 + 0.5 * np.cos(2.0 * np.pi * x / n)
    inits.append(c / c.sum())
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g / g.sum())
    # Edge-concentrated
    e = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(e / e.sum())
    # Multi-frequency
    for freq in [2, 3, 4, 6, 8]:
        h = 1.0 + 0.3 * np.cos(2.0 * np.pi * freq * x / n)
        h = np.maximum(h, 0.01)
        inits.append(h / h.sum())
    return inits


def try_and_update(a, state, label):
    """Evaluate and update best state."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a)))


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization:
    1. Load parent and evaluate
    2. Try structured initializations at multiple n
    3. Apply Nesterov subgradient, plain subgradient, smoothed softmax
    4. Coordinate descent for final polish
    5. Adaptive n-search with resampling
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    state = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.8f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        state = [float(evaluate_sequence(parent.tolist())), parent.copy()]
        print("Default parent: n=384, score={:.8f}".format(state[0]))

    n_parent = len(parent)
    print("Initial best: {:.8f}".format(state[0]))

    # === Phase 1: Quick structured exploration (~200s) ===
    print("\n=== Phase 1: Structured Exploration ===")
    t_phase = time.monotonic()

    test_ns = sorted(set([n_parent, 256, 384, 512, 640, 768, 1024, 1280]))
    test_ns = [n for n in test_ns if 128 <= n <= 2048]

    for n_try in test_ns:
        remaining = deadline - time.monotonic() - 1300
        if remaining < 10:
            break

        if n_try == n_parent and parent is not None:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss > 1e-10:
                start = start / ss
            else:
                start = np.ones(n_try) / n_try

        t_alloc = min(remaining, 30)
        # Quick Nesterov subgradient
        a_opt, _ = nesterov_subgrad(start, time_limit=t_alloc, eta_init=0.05, top_k=5)
        try_and_update(a_opt, state, "nest_n{}".format(n_try))

        remaining = deadline - time.monotonic() - 1250
        if remaining > 5:
            a_opt2, _ = plain_subgrad(a_opt, time_limit=min(remaining, 15),
                                      eta_init=0.03, top_k=3)
            try_and_update(a_opt2, state, "plain_n{}".format(n_try))

    print("  Phase 1 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 2: Deep Nesterov optimization (~500s) ===
    print("\n=== Phase 2: Deep Nesterov ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])
        remaining = deadline - time.monotonic() - 800
        if remaining > 30:
            t_alloc = min(remaining, 400)
            a_opt, _ = nesterov_subgrad(state[1], time_limit=t_alloc,
                                        eta_init=0.04, top_k=7, mom_init=0.5)
            try_and_update(a_opt, state, "deep_nest")

        remaining = deadline - time.monotonic() - 600
        if remaining > 20:
            a_opt2, _ = nesterov_subgrad(state[1], time_limit=min(remaining, 150),
                                         eta_init=0.02, top_k=3, mom_init=0.3)
            try_and_update(a_opt2, state, "deep_nest_loweta")

        # Perturbed restarts
        for scale in [0.005, 0.01, 0.02]:
            remaining = deadline - time.monotonic() - 400
            if remaining < 15:
                break
            f_pert = project_simplex(state[1] + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 50)
            a_opt3, _ = nesterov_subgrad(f_pert, time_limit=t_alloc,
                                         eta_init=0.03, top_k=5)
            try_and_update(a_opt3, state, "nest_pert_{}".format(scale))

    print("  Phase 2 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 3: Smoothed softmax descent (~300s) ===
    print("\n=== Phase 3: Smoothed Softmax ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        for T_init, lr_init in [(0.0005, 0.03), (0.001, 0.02), (0.005, 0.01), (0.0001, 0.05)]:
            remaining = deadline - time.monotonic() - 300
            if remaining < 15:
                break
            t_alloc = min(remaining, 80)
            a_opt, _ = smoothed_softmax_descent(state[1], time_limit=t_alloc,
                                                T_init=T_init, lr_init=lr_init)
            try_and_update(a_opt, state, "smooth_T{}".format(T_init))

    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 4: Coordinate descent + plain subgradient (~300s) ===
    print("\n=== Phase 4: Coordinate + Plain SG ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        remaining = deadline - time.monotonic() - 200
        if remaining > 10:
            t_alloc = min(remaining, 100)
            a_opt, _ = coordinate_descent(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "coord")

        remaining = deadline - time.monotonic() - 150
        if remaining > 10:
            t_alloc = min(remaining, 80)
            a_opt2, _ = plain_subgrad(state[1], time_limit=t_alloc,
                                      eta_init=0.02, top_k=2)
            try_and_update(a_opt2, state, "plain_top2")

        remaining = deadline - time.monotonic() - 100
        if remaining > 10:
            t_alloc = min(remaining, 60)
            a_opt3, _ = nesterov_subgrad(state[1], time_limit=t_alloc,
                                         eta_init=0.01, top_k=10, mom_init=0.7)
            try_and_update(a_opt3, state, "nest_wide")

    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 5: Final polish (~150s) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Multiple tiny perturbations with quick optimization
        for i in range(8):
            remaining = deadline - time.monotonic() - 20
            if remaining < 5:
                break
            noise_scale = 0.001 + 0.0005 * i
            f_pert = project_simplex(state[1] + rng.standard_normal(len(state[1])) * noise_scale)
            t_alloc = min(remaining, 10)
            a_opt, _ = plain_subgrad(f_pert, time_limit=t_alloc, eta_init=0.01, top_k=3)
            try_and_update(a_opt, state, "polish_{}".format(i))

        # One final Nesterov pass with very small eta
        remaining = deadline - time.monotonic() - 10
        if remaining > 5:
            a_opt, _ = nesterov_subgrad(state[1], time_limit=min(remaining, 15),
                                        eta_init=0.005, top_k=5, mom_init=0.8)
            try_and_update(a_opt, state, "final_nest")

    print("  Phase 5 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(384) / 384
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
