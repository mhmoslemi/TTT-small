def solve_active_lp(f, M, n, active_rows, row_idx, col_idx):
    """Solve LP with only active constraints."""
    # Filter to active rows
    active_mask = np.isin(row_idx, active_rows)
    r_idx = row_idx[active_mask]
    c_idx = col_idx[active_mask]
    vals = f[r_idx - c_idx]
    
    from scipy.sparse import csr_matrix
    # Map active rows to 0..len(active_rows)-1
    row_map = {k: i for i, k in enumerate(active_rows)}
    r_idx_mapped = np.array([row_map[k] for k in r_idx])
    
    A_sparse = csr_matrix((vals, (r_idx_mapped, c_idx)), shape=(len(active_rows), n))
    b_ub = np.full(len(active_rows), M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-15:
            return g
    return None

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import linprog
from scipy.special import softmax
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def solve_small_lp(f, M, active_ks):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_ks, g >= 0."""
    n = len(f)
    m = len(active_ks)
    if m == 0:
        return None
    A = np.zeros((m, n))
    for idx, k in enumerate(active_ks):
        js = max(0, k - (n - 1))
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A[idx, j] = f[k - j]
    c = -np.ones(n)
    b = np.full(m, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def mirror_descent_v2(a0, time_limit, eta_init=0.03):
    """Improved mirror descent with adaptive temperature."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    eta = eta_init
    temperature = 0.01 * fast_convolve(a).max()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        max_val = conv.max()

        # Adaptive temperature
        avg_val = conv.mean()
        temperature = max(0.001 * max_val, (max_val - avg_val) * 0.1)

        # Weighted peaks
        weights = np.exp((conv - max_val) / temperature)
        weights /= weights.sum()

        # Subgradient (only for non-negligible weights)
        grad = np.zeros(n)
        significant = np.where(weights > 0.01 / n)[0]
        for k in significant:
            w = weights[k]
            if w < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je + 1] += w * 2.0 * a[k - je:k - js + 1][::-1]

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        # Blend with line search
        best_candidate = a
        best_val = max_val
        for alpha in [0.3, 0.5, 0.7, 1.0]:
            cand = (1 - alpha) * a + alpha * a_new
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_candidate = cand

        a = best_candidate
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            eta = min(eta * 1.05, 0.1)  # Increase step slightly
        else:
            no_improve += 1
            eta *= 0.95

        if eta < 1e-4:
            eta = eta_init
            no_improve = 0

        if no_improve > 100:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31 - 1))
            a = np.maximum(best_a + rng.standard_normal(n) * 0.003, 0)
            a = a / a.sum()
            eta = eta_init
            no_improve = 0

    return best_a, best_obj


def subgradient_adaptive(a0, time_limit, top_k=10):
    """Subgradient with adaptive step size and binary line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = 0.05
    best_lr = 0.05

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            weights = (pv - pv.min() + 1e-10)
            weights = weights ** 2  # Emphasize the largest

        for idx in range(len(peaks)):
            w = weights[idx]
            if w < 1e-12:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je + 1] += w * 2.0 * a[k - je:k - js + 1][::-1]

        direction = np.maximum(a - lr * grad, 0)
        direction = direction / direction.sum() - a

        # Binary line search
        lo, hi = 0.0, 1.0
        best_alpha = 0.0
        best_val = conv.max()
        for _ in range(10):
            mid = (lo + hi) / 2.0
            cand = a + mid * direction
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                hi = mid
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
                lo = mid
            else:
                hi = mid

        if best_alpha > 1e-10:
            a = a + best_alpha * direction
            a = np.maximum(a, 0)
            a = a / a.sum()
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
                best_lr = lr
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve % 10 == 0:
            lr *= 0.9
        if lr < 0.001:
            lr = best_lr
            no_improve = 0

        if no_improve > 80:
            rng = np.random.default_rng((it * 1299709 + 3) % (2**31 - 1))
            a = np.maximum(best_a + rng.standard_normal(n) * 0.002, 0)
            a = a / a.sum()
            no_improve = 0
            lr = best_lr

    return best_a, best_obj


def peak_targeted(a0, time_limit):
    """
    Direct peak targeting: iteratively reduce the highest peak by
    finding the optimal redistribution.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.2:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        peak_val = conv[k_peak]

        # Find positions contributing to this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)

        # Contribution of each pair (j, k_peak - j)
        contrib = np.zeros(n)
        for j in range(js, je + 1):
            contrib[j] = a[j] * a[k_peak - j]

        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break

        # Try to reduce the peak by moving mass from high-contrib pairs
        # to positions that don't contribute to this peak
        # Strategy: reduce a[j] where contrib is highest, add to a[i] where i is far from peak
        frac = 0.05  # Fraction of mass to move
        move = np.zeros(n)

        # Source: positions with high contribution
        src_mask = contrib > 0.1 * total_contrib / max(1, je - js + 1)
        src_positions = np.where(src_mask)[0]
        if len(src_positions) == 0:
            src_positions = np.array([js, je])

        # Destination: positions far from the peak span
        dest_candidates = np.array([j for j in range(n) if j < js - 2 or j > je + 2])
        if len(dest_candidates) == 0:
            dest_candidates = np.array([0, n - 1])

        total_src = a[src_positions].sum()
        if total_src < 1e-15:
            break

        amount = frac * total_src
        # Reduce sources proportionally to their contribution
        src_contrib = contrib[src_positions]
        src_total = src_contrib.sum()
        if src_total > 1e-15:
            red = amount * src_contrib / src_total
        else:
            red = np.full(len(src_positions), amount / len(src_positions))
        for i, j in enumerate(src_positions):
            move[j] -= red[i]
            if a[j] - red[i] < 0:
                red[i] = a[j]
                move[j] = -a[j]

        actual_amount = -move.sum()  # How much we actually removed
        if actual_amount < 1e-15:
            break

        # Distribute to destinations uniformly (or proportional to current value)
        dest_vals = a[dest_candidates]
        dest_sum = dest_vals.sum()
        if dest_sum > 1e-15:
            add = actual_amount * dest_vals / dest_sum
        else:
            add = np.full(len(dest_candidates), actual_amount / len(dest_candidates))
        for i, j in enumerate(dest_candidates):
            move[j] += add[i]

        a_new = np.maximum(a + move, 0)
        a_new = a_new / a_new.sum()
        new_obj = fast_convolve(a_new).max()

        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            frac *= 0.8  # Try smaller moves

        if frac < 0.001:
            frac = 0.05
            if no_improve > 50:
                rng = np.random.default_rng((it * 9973 + 5) % (2**31 - 1))
                a = np.maximum(best_a + rng.standard_normal(n) * 0.002, 0)
                a = a / a.sum()
                no_improve = 0
                frac = 0.05

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    inits.append(np.ones(n) / n)

    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    inits.append(hann / hann.sum())

    # Edge-heavy
    edge_w = 1.0 / (np.minimum(x, n - 1 - x).astype(float) + 1.0) ** 1.5
    inits.append(edge_w / edge_w.sum())

    # Two-peak
    g2 = np.exp(-0.5 * ((x - 0.2 * (n - 1)) / (0.08 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.8 * (n - 1)) / (0.08 * n)) ** 2)
    inits.append(g2 / g2.sum())

    # Sinusoidal modulation
    t = x / max(n - 1, 1)
    asym = (1.0 + 0.3 * np.sin(2 * np.pi * 3 * t)) ** 2 + 0.1
    inits.append(asym / asym.sum())

    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Perturbed parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px + rng.standard_normal(n) * 0.05, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

    rng = np.random.default_rng(seed)

        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)

            return float('inf')
