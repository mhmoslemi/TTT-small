def fast_convolve(a):

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])

def solve_lp_oracle(f, M, method='highs'):
    """LP oracle: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method,
                      options={'presolve': True, 'max_time': 30})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

        best_alpha = 0.0

def smoothed_lbfgs(a0, time_limit, beta_init=500.0, beta_max=50000.0):
    """
    Log-sum-exp smoothed optimization using L-BFGS-B.
    Works on the simplex via projected gradient trick.
    Uses annealing of the smoothing parameter beta.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()

    # Annealing schedule
    n_anneal = 8
    for phase in range(n_anneal):
        if time.monotonic() - t0 > time_limit - 5:
            break
        beta = beta_init * (10.0 ** (phase * 0.5))
        beta = min(beta, beta_max)
        phase_time = (time_limit - (time.monotonic() - t0)) / (n_anneal - phase)

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            sx = x.sum()
            if sx < 1e-15:
                return 1e10, np.zeros(n)
            xa = x / sx
            conv = fast_convolve(xa)
            # Log-sum-exp: log(sum exp(beta*conv)) / beta
            max_c = conv.max()
            scaled = beta * (conv - max_c)
            log_sum = max_c * beta + np.log(np.sum(np.exp(scaled)))
            obj = log_sum / beta - max_c  # shift for numerical stability
            # Actually let's just use: obj = logsumexp(beta*conv)/beta
            # For gradient: d(obj)/da_j = sum_k p_k * 2*a[k-j] / (sum a)^2 adjusted
            # Since we normalize, the gradient is:
            # grad_raw[m] = sum_k p_k * 2 * a[k-m]
            # Then projected: grad[m] = grad_raw[m] - mean(grad_raw * a) * ... 
            # Actually for scale-invariant objective on simplex:
            # We just need the direction, so compute:
            p = np.exp(scaled)
            p = p / p.sum()
            grad_raw = 2.0 * hankel_mv(p, a)
            # Project gradient onto tangent space of simplex:
            # Remove the component in the direction of a (to maintain normalization)
            # Actually, since we clip to 0 and renormalize, let's just use:
            grad = grad_raw - np.sum(grad_raw * a) * a  # tangent projection
            return float(obj), grad

        # Run L-BFGS-B
        try:
            res = minimize(
                obj_and_grad, a.copy(),
                method='L-BFGS-B',
                bounds=[(0, None)] * n,
                options={'maxiter': 200, 'maxfun': 5000,
                         'ftol': 1e-14, 'gtol': 1e-8,
                         'maxls': 30, 'max_corr': 20}
            )
            if res.x is not None:
                candidate = np.maximum(res.x, 0.0)
                cs = candidate.sum()
                if cs > 1e-15:
                    candidate = candidate / cs
                    obj_val = fast_convolve(candidate).max()
                    if obj_val < best_obj:
                        best_obj = obj_val
                        best_a = candidate.copy()
                        print(f"    L-BFGS-B phase {phase}: obj={obj_val:.10f}, beta={beta:.0f}")
        except Exception as e:
            print(f"    L-BFGS-B phase {phase} error: {e}")

        a = best_a.copy()

    return best_a, best_obj

def active_set_newton(a0, time_limit, tol=1e-12, max_iter=2000):
    """Regularized active-set Newton using KKT conditions."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1
        if time.monotonic() - t0 > time_limit - 1:
            break

        conv = fast_convolve(a)
        M = conv.max()
        support = np.where(a > 1e-8)[0]
        if len(support) == 0:
            break

        # Active peaks: within 0.5% of max
        threshold = M * (1.0 - 0.005)
        active_peaks = np.where(conv >= threshold)[0]
        if len(active_peaks) <= 1:
            # Single peak: use simple gradient
            k = int(active_peaks[0])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad = np.zeros(n)
            grad[idx_arr] = a[k - idx_arr]
            candidate = project_simplex(a - 0.01 * grad)
            val = fast_convolve(candidate).max()
            if val < best_obj:
                best_obj = val
                best_a = candidate.copy()
                a = candidate
            else:
                break

        m = len(support)
        p = len(active_peaks)
        # Build KKT system: for j in support, sum_k mu_k * a[k-j] = nu
        A_sys = np.zeros((m, p + 1))
        for i, j in enumerate(support):
            A_sys[i, -1] = 1.0
            for k_idx, k in enumerate(active_peaks):
                idx = k - j
                if 0 <= idx < n:
                    A_sys[i, k_idx] = a[idx]

        # Add regularization
        A_reg = A_sys + 1e-10 * np.eye(m, p + 1)
        try:
            U, S_vals, Vt = np.linalg.svd(A_reg)
            sol = Vt[-1, :]
            mu_vals = np.maximum(sol[:p], 0)
            mu_sum = mu_vals.sum()
            if mu_sum < 1e-15:
                break
            mu_vals /= mu_sum
        except np.linalg.LinAlgError:
            break

        # Gradient step
        grad = np.zeros(n)
        for k_idx, k in enumerate(active_peaks):
            if mu_vals[k_idx] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] -= mu_vals[k_idx] * a[k - idx_arr]

        best_candidate = a
        best_val = M
        for step in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
            candidate = project_simplex(a + step * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate
        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()

        if abs(M - best_val) < 1e-14:
            break

    return best_a, best_obj


def softmax_smoothed_grad(a0, time_limit, T_init=0.001, lr_init=0.01):
    """Smoothed gradient using softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init
    lr = lr_init
    v = np.zeros(n)
    mom = 0.92
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        grad = 2.0 * hankel_mv(p, a)
        v = mom * v - lr * grad
        best_candidate = a
        best_val = conv.max()
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            candidate = project_simplex(a + scale * v)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break
        a = best_candidate
        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.9
        if no_improve > 20:
            T *= 1.08
            lr *= 0.96
        if no_improve > 50:
            T *= 0.85
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31 - 1))
            noise = 0.002 + 0.003 * min(1.0, (no_improve - 200) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            v = np.zeros(n)
            no_improve = 0
    return best_a, best_obj


def try_and_update(a, best_state, label=""):
    """Evaluate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_state
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    best_seq, best_score = best_state
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return (best_seq, best_score)


def interp_sequence(parent, n_target):
    """Interpolate parent to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def structured_init(n, kind='hann'):
    """Generate structured initial sequences."""
    if kind == 'hann':
        x = np.arange(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) if n > 2 else np.ones(n)
    elif kind == 'gaussian':
        x = np.arange(n)
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif kind == 'raised_cosine':
        x = np.linspace(0, np.pi, n)
        a = 1 - 0.5 * np.cos(x)
    elif kind == 'triangle':
        x = np.arange(n)
        center = (n - 1) / 2.0
        a = 1.0 - np.abs(x - center) / center if n > 2 else np.ones(n)
        a = np.maximum(a, 0)
    elif kind == 'bimodal':
        x = np.arange(n)
        c1 = n * 0.25
        c2 = n * 0.75
        sigma = n * 0.08
        a = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    elif kind == 'flat_top':
        # Flat-top window coefficients
        x = np.arange(n)
        w = np.pi / n
        a = (1 - 0.216 * np.cos(2 * w * (x + 0.5))
             + 0.019 * np.cos(4 * w * (x + 0.5)))
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search combining:
    - Structured initialization
    - Multi-n exploration
    - PGD with momentum
    - Log-sum-exp L-BFGS-B (new)
    - Deficit-filling (new)
    - FW-LP with LP oracle
    - Active-set Newton
    - Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    # === Phase 1: Multi-n exploration with structured init (~400s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_p1 = time.monotonic()

    n_parent = len(parent)
    # Candidate n values to try
    candidate_ns = sorted(set([
        n_parent,
        max(128, n_parent // 2),
        min(4096, n_parent * 2),
        256, 384, 512, 640, 768, 896, 1024, 1280, 1536, 2048
    ]))
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 4096]

    # Try structured inits at parent size first
    for kind in ['hann', 'gaussian', 'bimodal', 'flat_top', 'raised_cosine']:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 10:
            break
        a_init = structured_init(n_parent, kind)
        # Quick PGD
        t_alloc = min(remaining, 30)
        a_opt, _ = pgd_momentum(a_init, time_limit=t_alloc, top_k=20, lr_init=0.03)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"init_{kind}")

    # Multi-n exploration
    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1000
        if remaining < 20:
            break

        if n_try == len(best_seq):
            start = best_seq.copy()
        else:
            start = interp_sequence(best_seq, n_try)

        t_alloc = min(remaining, 60)
        print(f"  n={n_try}, t={t_alloc:.0f}s")

        # PGD
        a_opt, _ = pgd_momentum(start, time_limit=t_alloc * 0.5, top_k=20, lr_init=0.02)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"pgd_n{n_try}")

        # Quick smoothed
        remaining = deadline - time.monotonic() - 950
        if remaining > 15:
            a_opt2, _ = softmax_smoothed_grad(a_opt, time_limit=min(remaining, 20), T_init=0.002)
            (best_seq, best_score) = try_and_update(a_opt2, (best_seq, best_score), f"smooth_n{n_try}")

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_p1:.0f}s, n={len(best_seq)})")

    # === Phase 2: Deep optimization at best n (~600s) ===
    print("\n=== Phase 2: Deep Optimization ===")
    t_p2 = time.monotonic()
    n_best = len(best_seq)

    # PGD with various configs
    remaining = deadline - time.monotonic() - 500
    if remaining > 30:
        t_alloc = min(remaining, 120)
        print(f"  PGD deep, n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = pgd_momentum(best_seq, time_limit=t_alloc, top_k=30, lr_init=0.015, mom_init=0.93)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "pgd_deep")

    # L-BFGS-B smoothed (new key method)
    remaining = deadline - time.monotonic() - 400
    if remaining > 30 and n_best <= 2048:
        t_alloc = min(remaining, 150)
        print(f"  L-BFGS-B smoothed, n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = smoothed_lbfgs(best_seq, time_limit=t_alloc, beta_init=200.0, beta_max=20000.0)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "lbfgs")

    # Deficit filling (new)
    remaining = deadline - time.monotonic() - 300
    if remaining > 20:
        t_alloc = min(remaining, 80)
        print(f"  Deficit filling, n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = deficit_filling(best_seq, time_limit=t_alloc, T_init=0.05)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "deficit")

    # FW-LP
    remaining = deadline - time.monotonic() - 200
    if remaining > 30:
        t_alloc = min(remaining, 100)
        print(f"  FW-LP, n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "fw")

    # Active-set Newton
    remaining = deadline - time.monotonic() - 100
    if remaining > 20:
        t_alloc = min(remaining, 80)
        print(f"  Active-set Newton, t={t_alloc:.0f}s")
        a_opt, _ = active_set_newton(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "as_newton")

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_p2:.0f}s)")

    # === Phase 3: Multi-restart + refinement (~500s) ===
    print("\n=== Phase 3: Multi-Restart Refinement ===")
    t_p3 = time.monotonic()
    n_best = len(best_seq)

    # Try different perturbations
    for i, scale in enumerate([0.002, 0.005, 0.01, 0.02, 0.03]):
        remaining = deadline - time.monotonic() - 200
        if remaining < 20:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc = min(remaining, 60)

        # Quick PGD
        a_opt, _ = pgd_momentum(f_pert, time_limit=t_alloc * 0.4, top_k=20, lr_init=0.02)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"pert_{scale}")

        # Active-set Newton on perturbed
        remaining = deadline - time.monotonic() - 150
        if remaining > 15:
            a_opt2, _ = active_set_newton(a_opt, time_limit=min(remaining, 30))
            (best_seq, best_score) = try_and_update(a_opt2, (best_seq, best_score), f"pert_as_{scale}")

    # Try at slightly different n values around best
    for n_offset in [-128, -64, 64, 128, 256]:
        remaining = deadline - time.monotonic() - 100
        if remaining < 20:
            break
        n_try = n_best + n_offset
        if n_try < 128 or n_try > 4096:
            continue
        start = interp_sequence(best_seq, n_try)
        t_alloc = min(remaining, 50)
        a_opt, _ = pgd_momentum(start, time_limit=t_alloc * 0.5, top_k=20, lr_init=0.02)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"n{n_try}")
        if remaining > 20:
            a_opt2, _ = active_set_newton(a_opt, time_limit=min(remaining - 20, 20))
            (best_seq, best_score) = try_and_update(a_opt2, (best_seq, best_score), f"nas_n{n_try}")

    # Symmetrization check
    sym = (best_seq + best_seq[::-1]) / 2.0
    sym = sym / sym.sum()
    (best_seq, best_score) = try_and_update(sym, (best_seq, best_score), "sym")

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_p3:.0f}s)")

    # === Phase 4: Final intensive polish (~300s) ===
    print("\n=== Phase 4: Final Polish ===")
    t_p4 = time.monotonic()
    n_best = len(best_seq)

    # L-BFGS-B with tighter parameters
    remaining = deadline - time.monotonic() - 100
    if remaining > 30 and n_best <= 2048:
        t_alloc = min(remaining, 100)
        print(f"  Final L-BFGS-B, t={t_alloc:.0f}s")
        a_opt, _ = smoothed_lbfgs(best_seq, time_limit=t_alloc, beta_init=1000.0, beta_max=50000.0)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "final_lbfgs")

    # Multiple active-set Newton runs
    for _ in range(3):
        remaining = deadline - time.monotonic() - 50
        if remaining < 15:
            break
        a_opt, _ = active_set_newton(best_seq, time_limit=min(remaining, 40))
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "final_as")

    # Deficit filling final
    remaining = deadline - time.monotonic() - 30
    if remaining > 15:
        a_opt, _ = deficit_filling(best_seq, time_limit=min(remaining, 30), T_init=0.01)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "final_deficit")

    # Tiny perturbation sweeps
    for _ in range(4):
        remaining = deadline - time.monotonic() - 15
        if remaining < 10:
            break
        noise = rng.standard_normal(n_best) * 0.001
        f_pert = project_simplex(best_seq + noise)
        a_opt, _ = pgd_momentum(f_pert, time_limit=min(remaining, 15), top_k=15, lr_init=0.01)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "micro_pert")

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_p4:.0f}s)")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
