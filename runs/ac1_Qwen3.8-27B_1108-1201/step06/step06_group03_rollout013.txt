    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

        return np.ones(n) / n

Now let me think about the peak surgery more carefully.

**Peak Surgery**: Given a peak at index `k` with value `conv[k] = sum_j a[j]*a[k-j]`:

The most effective way to reduce this peak is to:
1. Find the position `j*` where `a[j*]` is large AND `a[k-j*]` is large (these contribute most to the peak)
2. Move some mass from `j*` (or `k-j*`) to a position `j_target` where:
   - `a[j_target] * a[k-j_target]` is small (won't increase this peak much)
   - The new position doesn't create problems for other peaks

For the target position, we want to minimize the "damage" to all peaks:
`damage(j_target, delta) = max_k [2 * delta * (sum over peaks k of a[k - j_target]) * weight_k]`

This is approximately `2 * delta * (sum of weighted a values at positions that would interact with j_target)`

Let me simplify: the increase in peak `k` from adding `delta` to `a[j_target]` is approximately `2 * delta * a[k - j_target]` (for the peaks that include `j_target` in their support).

So we want to minimize `max_k 2 * delta * a[k - j_target]` over the top peaks, which means we want `j_target` to be in a "quiet" region where the convolution is low.

Actually, a simpler heuristic: add the mass to a position where the current convolution is minimal. This is because if `conv[k]` is small at all `k` that involve `j_target`, then adding mass there won't create new peaks.

def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_peak_grad(a, peak_indices, weights):
    """Compute gradient of weighted sum of peak values."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(len(peak_indices)):
        wi = weights[i]
        if wi < 1e-15:

        k = int(peak_indices[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += wi * 2.0 * a[k - idx]
    return grad

def peak_surgery(a0, time_limit, max_rounds=5000):

    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    step_frac = 0.05
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_rounds:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        n_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        improved_this_round = False

        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)

            if js > je:
                continue

            j_arr = np.arange(js, je + 1)
            contributions = a[j_arr] * a[k - j_arr]
            if contributions.max() < 1e-12:
                continue

            best_local = np.argmax(contributions)
            src = js + best_local

            if a[src] < 1e-10:
                continue

            # Find quiet target position
            scores = np.zeros(n)
            for k2 in peak_indices[:3]:
                s2 = max(0, k2 - n + 1)
                e2 = min(n - 1, k2)
                for j in range(s2, e2 + 1):
                    scores[j] += a[k2 - j]
            scores[src] = 1e10

            # Also penalize positions near src to avoid local oscillation
            src_lo = max(0, src - 3)
            src_hi = min(n - 1, src + 3)
            scores[src_lo:src_hi + 1] += 0.5 * scores[src_lo:src_hi + 1].max() + 1e-6

            target = int(np.argmin(scores))

            for frac in [step_frac, step_frac * 0.5, step_frac * 2.0]:
                delta = a[src] * frac
                if delta < 1e-12:
                    continue
                candidate = a.copy()
                candidate[src] -= delta
                candidate[target] += delta
                candidate = np.maximum(candidate, 0.0)
                cs = candidate.sum()
                if cs < 1e-15:
                    continue
                candidate /= cs
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
                    improved_this_round = True
                    break

def block_refine(a0, time_limit, block_size=32):
    """Block coordinate descent: optimize groups of coordinates together."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    n_blocks = (n + block_size - 1) // block_size

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        improved_any = False

        for b in range(n_blocks):
            start = b * block_size
            end = min(n, start + block_size)
            if end - start < 2:
                continue

            # Try small perturbations within this block
            block = a[start:end].copy()
            block_sum = block.sum()

            if block_sum < 1e-15:
                continue

            # Generate a few candidate modifications
            rng = np.random.default_rng((it * 100000 + b * 7919 + 42) % (2 ** 31))

            for _attempt in range(6):
                # Random direction within the block that preserves non-negativity
                direction = rng.standard_normal(end - start)
                # Remove mean to try to preserve sum approximately
                direction -= direction.mean()

                for scale in [0.02, 0.01, 0.005]:
                    new_block = block + scale * direction * block_sum
                    new_block = np.maximum(new_block, 0.0)
                    nb_sum = new_block.sum()
                    if nb_sum < 1e-15:
                        continue

                    candidate = a.copy()
                    candidate[start:end] = new_block
                    cs = candidate.sum()
                    candidate /= cs
                    val = fast_convolve(candidate).max()
                    if val < best_obj - 1e-15:
                        a = candidate
                        best_obj = val
                        best_a = a.copy()
                        block = a[start:end].copy()
                        block_sum = block.sum()
                        improved_any = True
                        break

        if not improved_any and it > 500:
            break

    return best_a, float(best_obj)


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_diverse_starts(n, parent, rng):
    """Generate diverse initial sequences of length n."""
    starts = []

    if parent is not None:
        starts.append(resample_sequence(parent, n))
        # Shifted parent
        shifted = np.roll(parent, n // 10)
        starts.append(resample_sequence(shifted, n))

    x = np.arange(n, dtype=np.float64)

    # Cosine taper
    w = 0.5 * (1 + np.cos(np.pi * x / max(n - 1, 1)))
    w = np.maximum(w, 0.01)
    starts.append(w / w.sum())

    # Gaussian
    g = np.exp(-0.5 * ((x - n / 2) / (0.15 * n)) ** 2)
    starts.append(g / g.sum())

    # Two Gaussian bumps
    g1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())

    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    starts.append(edge / edge.sum())

    # Random Dirichlet
    starts.append(rng.dirichlet(np.ones(n) * 0.2))
    starts.append(rng.dirichlet(np.ones(n) * 1.0))

    # Comb-like pattern
    comb = np.ones(n)
    comb[::3] = 0.3
    starts.append(comb / comb.sum())

    return starts


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Uses smooth max + Adam optimization with:
    - Multi-length exploration
    - Adaptive hyperparameter scheduling
    - Peak surgery for targeted improvement
    - Block refinement for final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 45.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform")

    n_parent = len(parent)

    # === Phase 1: Quick multi-length exploration (120s) ===
    print("\n=== Phase 1: Multi-length Exploration ===")
    t_phase = time.monotonic()

    lengths_to_try = sorted(set([
        n_parent,
        max(256, n_parent - 128),
        n_parent + 128,
        512, 640, 768, 896, 1024, 1152, 1280
    ]))
    lengths_to_try = [l for l in lengths_to_try if 256 <= l <= 1600]

    for n_try in lengths_to_try:
        remaining = deadline - time.monotonic() - 700
        if remaining < 15:
            break

        starts = generate_diverse_starts(n_try, parent, rng)

        for si, start in enumerate(starts[:3]):
            remaining = deadline - time.monotonic() - 650
            if remaining < 10:
                break
            t_alloc = min(remaining, 20.0)
            a_opt, _ = smooth_max_adam(start, time_limit=t_alloc,
                                       T_init=0.001, K_init=20, lr_init=0.008)
            try_seq(a_opt, f"explore_n{n_try}_s{si}")

    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep optimization on best (450s) ===
    print("\n=== Phase 2: Deep Smooth Max ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple configs for deep optimization
        configs = [
            {'T_init': 0.0003, 'K_init': 10, 'lr_init': 0.004},
            {'T_init': 0.001, 'K_init': 20, 'lr_init': 0.006},
            {'T_init': 0.0005, 'K_init': 15, 'lr_init': 0.003},
        ]

        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 350
            if remaining < 30:
                break
            t_alloc = min(remaining, 140.0)
            print(f"  Config {ci} (T={cfg['T_init']}, K={cfg['K_init']}), t={t_alloc:.0f}s")
            a_opt, _ = smooth_max_adam(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"deep_{ci}")

        # Perturbed restarts
        for scale in [0.002, 0.005, 0.01, 0.02]:
            remaining = deadline - time.monotonic() - 300
            if remaining < 20:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 50.0)
            a_opt, _ = smooth_max_adam(f_pert, time_limit=t_alloc,
                                       T_init=0.0004, K_init=12, lr_init=0.005)
            try_seq(a_opt, f"pert_{scale}")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Peak surgery (250s) ===
    print("\n=== Phase 3: Peak Surgery ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 250
        if remaining > 20:
            t_alloc = min(remaining, 200.0)
            print(f"  Peak surgery, t={t_alloc:.0f}s")
            a_opt, obj = peak_surgery(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_surgery")

            # Second round of peak surgery on the result
            remaining = deadline - time.monotonic() - 200
            if remaining > 15:
                t_alloc = min(remaining, 100.0)
                a_opt2, obj2 = peak_surgery(a_opt, time_limit=t_alloc)
                try_seq(a_opt2, "peak_surgery_2")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: More deep optimization with different strategies (250s) ===
    print("\n=== Phase 4: Refinement ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Try a different length (slightly different)
        for delta in [-64, 64, -32, 32]:
            n_new = n_best + delta
            if n_new < 256 or n_new > 1600:
                continue
            remaining = deadline - time.monotonic() - 150
            if remaining < 15:
                break
            a_resampled = resample_sequence(best_seq, n_new)
            t_alloc = min(remaining, 40.0)
            a_opt, _ = smooth_max_adam(a_resampled, time_limit=t_alloc,
                                       T_init=0.0004, K_init=12, lr_init=0.004)
            try_seq(a_opt, f"relen_{n_new}")

        # Block refinement
        remaining = deadline - time.monotonic() - 100
        if remaining > 15:
            t_alloc = min(remaining, 80.0)
            print(f"  Block refine, t={t_alloc:.0f}s")
            a_opt, _ = block_refine(best_seq, time_limit=t_alloc, block_size=24)
            try_seq(a_opt, "block_refine")

        # Another peak surgery round
        remaining = deadline - time.monotonic() - 50
        if remaining > 15:
            t_alloc = min(remaining, 60.0)
            a_opt, _ = peak_surgery(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_refine")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final polishing (remaining time) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        polish_iter = 0

        while time.monotonic() < deadline - 5:
            polish_iter += 1
            remaining = deadline - time.monotonic()

            if polish_iter % 3 == 1:
                # Smooth max with very small T (near-exact max)
                noise = rng.standard_normal(n_best) * 0.001
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining - 5, 20.0)
                if t_alloc > 5:
                    a_opt, _ = smooth_max_adam(start, time_limit=t_alloc,
                                               T_init=0.0001, K_init=8, lr_init=0.002)
                    try_seq(a_opt, f"polish_sm{polish_iter}")

            elif polish_iter % 3 == 2:
                # Peak surgery
                t_alloc = min(remaining - 5, 25.0)
                if t_alloc > 5:
                    a_opt, _ = peak_surgery(best_seq, time_limit=t_alloc)
                    try_seq(a_opt, f"polish_pk{polish_iter}")

            else:
                # Block refinement
                t_alloc = min(remaining - 5, 15.0)
                if t_alloc > 3:
                    a_opt, _ = block_refine(best_seq, time_limit=t_alloc, block_size=16)
                    try_seq(a_opt, f"polish_bl{polish_iter}")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
