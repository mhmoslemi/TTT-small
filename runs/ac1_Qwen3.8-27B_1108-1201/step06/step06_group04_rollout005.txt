import numpy as np
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(w, a):
    """Compute W @ a where W_{i,j} = w[i+j]."""
    return np.correlate(w, a, mode='valid')


def compute_subgrad_topk(conv, a, K=5, temperature=0.01):
    """Compute weighted subgradient using top-K peaks with exponential weights."""
    n = len(a)
    cmax = conv.max()
    # Soft-max weights on top peaks
    scaled = (conv - cmax) / temperature
    w = np.exp(scaled)
    w = w / w.sum()
    # Only keep top K to save computation
    topk_idx = np.argpartition(conv, -K)[-K:]
    w[:len(w) - K] = 0.0  # zero out non-top-K (approximate)
    # Actually, let's just use top K
    w = np.zeros(len(conv))
    w[topk_idx] = np.exp((conv[topk_idx] - cmax) / temperature)
    w = w / w.sum()
    # Gradient: sum_k w[k] * 2 * a[k-i] for valid i
    # grad[i] = 2 * sum_k w[k] * a[k-i] where 0<=k-i<n
    # This is 2 * (w * a) flipped and truncated
    # grad = 2 * correlate(w, a, 'valid') but need to be careful with indexing
    # conv_wa[j] = sum_i w[i+j] * a[i] for j = 0..n-1
    # But we want grad[i] = 2 * sum_k w[k] * a[k-i]
    # Let's just compute it directly for the non-zero w entries
    grad = np.zeros(n)
    for k in topk_idx:
        wk = w[k]
        if wk < 1e-15:
            continue
        # contributing i: max(0, k-n+1) <= i <= min(k, n-1)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        indices = np.arange(js, je + 1)
        grad[indices] += wk * a[k - indices]
    grad *= 2.0
    return grad


def mirror_descent(a0, time_limit, K=5, T_init=0.01, lr_init=0.05,
                   momentum=0.3, verbose=False):
    """Mirror descent with momentum on the simplex (multiplicative weights)."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)  # momentum in log-space
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        grad = compute_subgrad_topk(conv, a, K=K, temperature=T)

        # Mirror descent update in log-space
        log_a = np.log(a)
        v = momentum * v - lr * grad
        log_a_new = log_a + v
        # Renormalize (exponential normalizing constant)
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0.0)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Backtrack
            lr *= 0.95

        a = a_new

        # Adapt temperature
        if no_improve > 20:
            T *= 1.1
        if no_improve > 60:
            T *= 0.7
        if no_improve > 150:
            # Restart from best with small perturbation
            rng = np.random.default_rng((it * 48271 + 17) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = lr_init
            T = T_init
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    MD it={it}, obj={best_obj:.10f}, T={T:.6f}")

    return best_a, best_obj


def block_coordinate_descent(a0, time_limit, block_size=32, verbose=False):
    """Optimize blocks of coordinates while keeping others fixed."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        cmax = conv.max()

        # Pick a random block
        rng = np.random.default_rng(it * 31 + 7)
        start = rng.integers(0, max(1, n - block_size + 1))
        end = min(start + block_size, n)
        block = np.arange(start, end)

        # Try to improve by redistributing mass within the block
        # Keep sum of block constant
        block_sum = a[block].sum()
        if block_sum < 1e-15:
            continue

        # Try several redistributions
        candidates = []
        # Uniform within block
        candidates.append(np.full(len(block), block_sum / len(block)))
        # Linear
        x = np.linspace(0, 1, len(block))
        candidates.append(x / x.sum() * block_sum)
        candidates.append((1 - x) / (1 - x).sum() * block_sum)
        # Cosine
        c = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        c = np.maximum(c, 0.01)
        candidates.append(c / c.sum() * block_sum)
        # Gaussian centered
        g = np.exp(-0.5 * ((x - 0.5) / 0.2) ** 2)
        candidates.append(g / g.sum() * block_sum)
        # Current
        candidates.append(a[block].copy())

        best_candidate = a[block].copy()
        best_block_obj = cmax

        for cand in candidates:
            a_test = a.copy()
            a_test[block] = cand
            a_test = a_test / a_test.sum()
            obj = fast_convolve(a_test).max()
            if obj < best_block_obj - 1e-14:
                best_block_obj = obj
                best_candidate = cand.copy()

        a[block] = best_candidate
        a = a / a.sum()

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            block_size = max(8, min(n // 4, block_size * 2))
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    BCD it={it}, obj={best_obj:.10f}, bs={block_size}")

    return best_a, best_obj


def solve_qp_block(w, a_init, n, max_iters=100, time_limit=3.0):
    """Solve min a^T W a s.t. sum(a)=1, a>=0 using active set on a block."""
    t0 = time.monotonic()
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    active = set(np.where(a > 1e-12)[0])
    if not active:
        active = set(range(n))

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if len(active) == 0:
            return np.ones(n) / n
        if len(active) == 1:
            a = np.zeros(n)
            a[list(active)[0]] = 1.0
            return a

        act = sorted(active)
        k = len(act)
        if k > 500:
            # Too many active, subsample
            rng = np.random.default_rng(it)
            act = list(rng.choice(act, size=500, replace=False))
            act = sorted(act)
            k = len(act)

        W_AA = np.array([[w[act[i] + act[j]] for j in range(k)] for i in range(k)])
        W_AA_reg = W_AA + 1e-10 * np.eye(k)

        try:
            x = np.linalg.solve(W_AA_reg, np.ones(k))
        except np.linalg.LinAlgError:
            try:
                x = np.linalg.lstsq(W_AA_reg, np.ones(k), rcond=None)[0]
            except Exception:
                x = np.ones(k) / k

        x_sum = x.sum()
        if abs(x_sum) < 1e-14:
            x = np.ones(k) / k
            x_sum = 1.0

        a_act = x / x_sum

        removed = False
        min_val = float('inf')
        min_idx = -1
        for i, idx in enumerate(act):
            if a_act[i] < -1e-10:
                if a_act[i] < min_val:
                    min_val = a_act[i]
                    min_idx = idx
        if min_idx >= 0:
            active.discard(min_idx)
            a = np.zeros(n)
            for i, idx in enumerate(act):
                if idx != min_idx:
                    a[idx] = max(0.0, a_act[i])
            s2 = a.sum()
            if s2 > 1e-15:
                a = a / s2
            continue

        a = np.zeros(n)
        for i, idx in enumerate(act):
            a[idx] = max(0.0, a_act[i])

        Wa = hankel_mv(w, a)
        grad = 2.0 * Wa
        mu = np.mean([grad[idx] for idx in act])

        inactive = set(range(n)) - active
        if inactive:
            min_viol = 0.0
            min_j = -1
            for j in inactive:
                viol = grad[j] - mu
                if viol < min_viol:
                    min_viol = viol
                    min_j = j
            if min_viol < -1e-6:
                active.add(min_j)
                continue

        return a

    return a

def coordinate_descent_polish(a0, time_limit, verbose=False):

        if len(contrib) == 0 or len(contrib) == n:

        else:
            non_contrib = np.setdiff1d(np.arange(n), contrib)
            if len(non_contrib) == 0:
                continue

            # Pick a large contributor and a random non-contributor
            idx_max = contrib[np.argmax(a[contrib])]
            rng = np.random.default_rng(it * 7)
            idx_min = non_contrib[rng.integers(0, len(non_contrib))]

            for frac in [0.003, 0.007, 0.015, 0.03]:
                shift = a[idx_max] * frac
                if shift < 1e-14:

                a_test = a.copy()
                a_test[idx_max] -= shift
                a_test[idx_min] += shift
                if a_test[idx_max] < 0:
                    continue
                a_test = a_test / a_test.sum()
                obj = fast_convolve(a_test).max()
                if obj < best_obj - 1e-14:

            else:

                continue
            no_improve = 0

        if verbose and it % 200 == 0:

    return best_a, best_obj

            break

        conv = fast_convolve(a)
        cmax = conv.max()

        # Active set: lags within 5% of max
        threshold = cmax * 0.95
        active_lags = np.where(conv >= threshold)[0]
        if len(active_lags) == 0:
            active_lags = np.array([int(np.argmax(conv))])

        # Subgradient: weighted average of gradients at active lags
        weights = conv[active_lags]
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for k in active_lags:
            wk = weights[list(active_lags).index(k)]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            indices = np.arange(js, je + 1)
            grad[indices] += wk * a[k - indices]
        grad *= 2.0

        # Newton step: minimize 0.5*a^T(H) a + (grad - mu*1)^T a
        # where H approximates the Hessian of the smoothed max
        # Hessian of conv[a,k] at a is 2*B_k
        # For the max, use weighted average of Hessians
        # H = 2 * sum_k w_k * B_k
        # This is a Hankel-like matrix: H[i,j] = 2 * sum_k w_k * [i+j==k]
        # = 2 * w[i+j] (for valid i,j)
        # So H is actually a Hankel matrix with entries 2*w[i+j]!
        # This means we can compute H @ v using hankel_mv!

        # Solve (H + lambda*I) d = -(grad - mu*1) on the active set of coordinates
        # Simplified: use gradient step with Armijo line search
        # (full Newton would require inverting an n x n Hankel matrix)

        # Use the structure: H @ v = 2 * hankel_mv(w_hankel, v)
        # where w_hankel[k] = 2 * sum_{active} w_k * delta_{i+j,k} ...
        # Actually H[i,j] = 2 * (sum_k weights_k * [i+j==k]) = 2 * weights_hankel[i+j]
        # where weights_hankel is just the weighted convolution weights

        # Let's just do a scaled gradient step with line search
        mu = np.mean(grad)  # approximate Lagrange multiplier
        direction = -grad + mu

        best_step = None
        best_val = cmax
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            a_test = a + step * direction
            a_test = project_simplex(a_test)
            obj = fast_convolve(a_test).max()
            if obj < best_val - 1e-15:
                best_val = obj
                best_step = a_test

        if best_step is not None:
            a = best_step
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            # Try small random perturbation
            rng = np.random.default_rng(it * 13)
            a = project_simplex(a + rng.standard_normal(n) * 0.001)

        if verbose and it % 50 == 0:
            print(f"    SSN it={it}, obj={best_obj:.10f}, |A|={len(active_lags)}")

    return best_a, best_obj


def fourier_perturb(a, seed, strength=0.01):
    """Generate a perturbation using a random Fourier mode."""
    rng = np.random.default_rng(seed)
    n = len(a)
    freq1 = rng.integers(1, max(2, n // 4))
    freq2 = rng.integers(1, max(2, n // 4))
    if freq2 == freq1:
        freq2 = (freq1 + 1) % max(1, n // 4)
    phase1 = rng.uniform(0, 2 * np.pi)
    phase2 = rng.uniform(0, 2 * np.pi)
    x = np.arange(n)
    pert = strength * (np.cos(2 * np.pi * freq1 * x / n + phase1) +
                       np.cos(2 * np.pi * freq2 * x / n + phase2))
    return pert


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)

    x = np.arange(n, dtype=np.float64)
    # Edge concentration (U-shaped)
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge / edge.sum())
    # Cosine window
    c1 = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    c1 = np.maximum(c1, 0.01)
    inits.append(c1 / c1.sum())
    # Raised cosine with different period
    c2 = 0.5 + 0.5 * np.cos(4 * np.pi * x / n)
    c2 = np.maximum(c2, 0.01)
    inits.append(c2 / c2.sum())
    # Linear ramp
    r = np.linspace(0.1, 1.0, n)
    inits.append(r / r.sum())
    # V-shaped
    v = np.abs(2.0 * x / (n - 1) - 1.0) + 0.05
    inits.append(v / v.sum())
    # Bimodal
    g1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.08 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.8 * n) / (0.08 * n)) ** 2)
    inits.append((g1 + g2) / (g1 + g2).sum())
    # Triangular
    tri = 1.0 - np.abs(2.0 * x / (n - 1) - 1.0)
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search with mirror descent,
    block coordinate descent, improved QP, and semismooth Newton.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384

    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")

    # === Phase 1: Multi-n exploration (250s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t1 = time.monotonic()

    candidate_ns = sorted(set([
        n_parent,
        max(128, n_parent - 128),
        n_parent + 128,
        n_parent + 256,
        max(256, n_parent // 2),
    ]))

    # Filter to reasonable range
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 2048]
    if not candidate_ns:
        candidate_ns = [n_parent]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1100
        if remaining < 30:
            break

        if n_try == n_parent and parent is not None:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 1e-10)
            start = start / start.sum()

        # Quick mirror descent
        t_alloc = min(remaining, 25)
        print(f"  MD n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = mirror_descent(start, time_limit=t_alloc, K=5, T_init=0.01, lr_init=0.03)
        try_seq(a_opt, f"md_n{n_try}")

        # Quick QP
        remaining = deadline - time.monotonic() - 1050
        if remaining > 15:
            t_alloc = min(remaining, 20)
            a_opt2, _ = qp_primal_dual_v2(a_opt, time_limit=t_alloc,
                                          T_init=0.01, use_topk=True, topk=8)
            try_seq(a_opt2, f"qp_n{n_try}")

    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")

    # === Phase 2: Deep optimization (700s) ===
    print("\n=== Phase 2: Deep Optimization ===")
    t2 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple methods in sequence
        # 2a: Mirror descent
        remaining = deadline - time.monotonic() - 500
        if remaining > 20:
            t_alloc = min(remaining, 150)
            print(f"  Deep MD K=8, t={t_alloc:.0f}s")
            a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc,
                                      K=8, T_init=0.005, lr_init=0.02, momentum=0.4)
            try_seq(a_opt, "deep_md_k8")

        # 2b: QP with different settings
        configs = [
            {'T_init': 0.005, 'use_topk': False, 'topk': 30, 'blend': 0.4},
            {'T_init': 0.01, 'use_topk': True, 'topk': 8, 'blend': 0.5},
            {'T_init': 0.02, 'use_topk': True, 'topk': 5, 'blend': 0.3},
        ]
        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 350
            if remaining < 20:
                break
            t_alloc = min(remaining, 100)
            print(f"  QP2 config {ci}, t={t_alloc:.0f}s")
            a_opt, _ = qp_primal_dual_v2(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"qp2_{ci}")

        # 2c: Block coordinate descent
        remaining = deadline - time.monotonic() - 200
        if remaining > 15:
            t_alloc = min(remaining, 80)
            print(f"  BCD, t={t_alloc:.0f}s")
            a_opt, _ = block_coordinate_descent(best_seq, time_limit=t_alloc, block_size=16)
            try_seq(a_opt, "bcd")

        # 2d: Semismooth Newton
        remaining = deadline - time.monotonic() - 150
        if remaining > 15:
            t_alloc = min(remaining, 80)
            print(f"  Semismooth Newton, t={t_alloc:.0f}s")
            a_opt, _ = semismooth_newton(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "ssn")

    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")

    # === Phase 3: Alternating refinement (400s) ===
    print("\n=== Phase 3: Alternating Refinement ===")
    t3 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Alternate between methods, always starting from current best
        methods = [
            ("md", lambda a, t: mirror_descent(a, time_limit=t, K=5, T_init=0.005, lr_init=0.01)),
            ("qp", lambda a, t: qp_primal_dual_v2(a, time_limit=t, T_init=0.01, use_topk=True, topk=6, blend=0.3)),
            ("bcd", lambda a, t: block_coordinate_descent(a, time_limit=t, block_size=8)),
            ("ssn", lambda a, t: semismooth_newton(a, time_limit=t)),
        ]

        mi = 0
        while time.monotonic() < deadline - 200:
            remaining = deadline - time.monotonic() - 180
            if remaining < 15:
                break
            t_alloc = min(remaining, 60)
            name, func = methods[mi % len(methods)]
            print(f"  Alternating: {name}, t={t_alloc:.0f}s, current={best_score:.10f}")
            a_opt, _ = func(best_seq, t_alloc)
            try_seq(a_opt, f"alt_{name}_{mi}")
            mi += 1

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")

    # === Phase 4: Coordinate descent polish (150s) ===
    print("\n=== Phase 4: CD Polish ===")
    t4 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 80
        if remaining > 10:
            t_alloc = min(remaining, 120)
            print(f"  CD Polish, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "cd_polish")

            # Another round with smaller steps
            remaining = deadline - time.monotonic() - 50
            if remaining > 10:
                t_alloc = min(remaining, 50)
                a_opt2, _ = coordinate_descent_polish(a_opt, time_limit=t_alloc)
                try_seq(a_opt2, "cd_polish2")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # === Phase 5: Final micro-optimizations (60s) ===
    print("\n=== Phase 5: Final Micro-Optimizations ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Tiny mirror descent
        remaining = deadline - time.monotonic() - 20
        if remaining > 10:
            t_alloc = min(remaining, 40)
            a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc,
                                      K=3, T_init=0.001, lr_init=0.003, momentum=0.5)
            try_seq(a_opt, "final_md")

        # Tiny QP
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            t_alloc = min(remaining, 15)
            a_opt, _ = qp_primal_dual_v2(best_seq, time_limit=t_alloc,
                                         T_init=0.002, use_topk=True, topk=3, blend=0.2)
            try_seq(a_opt, "final_qp")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
