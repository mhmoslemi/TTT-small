import numpy as np
from scipy.optimize import linprog
from scipy import sparse
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def binomial_initial(n):
    """Generate initial sequence from binomial coefficients (flat spectrum target)."""
    # c_k = binom(2k, k) / 4^k
    # Use log to avoid overflow
    k = np.arange(n, dtype=np.float64)
    # log(c_k) = log(binom(2k,k)) - k*log(4)
    # log(binom(2k,k)) = lgamma(2k+1) - 2*lgamma(k+1)
    import math
    log_c = np.zeros(n)
    for i in range(n):
        log_c[i] = math.lgamma(2*i+1) - 2*math.lgamma(i+1) - i*math.log(4)
    c = np.exp(log_c)
    c = c / c.sum()
    return c

def build_sparse_conv_matrix(a):
    """Build sparse matrix A where (A @ g)[k] = (a*g)[k] (convolution)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def fw_lp_step(a, M, max_time=30):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns the optimal g."""
    n = len(a)
    A = build_sparse_conv_matrix(a)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': max_time, 'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

    best_obj = M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    step = 1.0  # Initial step (full step)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        # Solve LP
        g = fw_lp_step(a, M, max_time=min(remaining*0.5, 30))
        if g is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Check if g is the same as a (convergence)
        if np.max(np.abs(g - a)) < 1e-12:
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        # Line search
        ft = a + step * (g - a)
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            no_improve += 1
            continue
        ft = ft / st
        
        conv_new = fast_convolve(ft)
        M_new = conv_new.max()
        
        if M_new < best_obj - 1e-15:
            best_obj = M_new
            best_a = ft.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Reduce step
            step *= 0.8
            if step < 0.01:
                step = 0.5
                # Try a different direction: random perturbation
                rng = np.random.default_rng(it)
                a_pert = project_simplex(a + rng.standard_normal(n) * 0.01)
                conv_pert = fast_convolve(a_pert)
                if conv_pert.max() < M:
                    a = a_pert
                    M = conv_pert.max()
        
        a = ft
        M = conv_new.max() if not np.isfinite(conv_new.max()) else M_new
        
        if verbose and it % 20 == 0:
            print(f"    FW it={it}, M={M:.10f}, best={best_obj:.10f}, step={step:.4f}")
    
    return best_a, best_obj

def pgd_optimize(a0, time_limit, top_k=20, lr_init=0.01):
    """Projected gradient descent with multi-peak subgradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    v = np.zeros(n)
    mom = 0.9
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Multi-peak subgradient
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            weights = pv / pv.sum()
        
        sg = np.zeros(n)
        for i, k in enumerate(peaks):
            if weights[i] < 1e-14:
                continue
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * a[k - idx_arr]
        
        # Gradient step
        v = mom * v - lr * sg
        candidate = project_simplex(a + v)
        c = fast_convolve(candidate)
        val = c.max()
        
        if val < best_obj - 1e-15:
            best_obj = val
            best_a = candidate.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = candidate
        
        # Adapt
        if no_improve > 30:
            lr *= 0.9
            if no_improve > 100:
                v = np.zeros(n)
                a = best_a.copy()
                no_improve = 0
                lr = lr_init
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 31 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj
