1. Initialize a (from parent or random)
2. Normalize: a = a / sum(a)
3. Set τ (temperature for log-sum-exp)
4. For each iteration:

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.special import softmax
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def logsumexp(x, tau):
    """Compute (1/tau) * log(sum(exp(tau * x))) stably."""
    x_max = x.max()
    return x_max + np.log(np.sum(np.exp(tau * (x - x_max)))) / tau

def smooth_max_grad(x, tau):
    """Compute weights w = softmax(tau * x) and the smooth max value."""
    x_shifted = x - x.max()
    w = np.exp(tau * x_shifted)
    w_sum = w.sum()
    w = w / w_sum
    val = x.max() + np.log(w_sum) / tau
    return val, w

def compute_smooth_grad(w, a):
    """Compute gradient of smooth max of conv(a,a) w.r.t. a.
    grad[i] = 2 * sum_k w[k] * a[k-i]
    """
    n = len(a)
    L = 1 << (3 * n - 1).bit_length()
    w_pad = np.zeros(L)
    w_pad[:2*n-1] = w
    a_pad = np.zeros(L)
    a_pad[:n] = a
    W = np.fft.rfft(w_pad)
    A = np.fft.rfft(a_pad)
    corr = np.fft.irfft(W * np.conj(A), L)
    return 2.0 * corr[:n]

def smooth_max_gd(a0, time_limit, tau_init=50.0, tau_max=5000.0, 
                   lr_init=0.1, verbose=False):
    """Projected gradient descent with smooth max approximation."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    tau = tau_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        sm_val, w = smooth_max_grad(conv, tau)
        
        # Exact objective for tracking
        exact_obj = conv.max()
        
        # Gradient
        grad = compute_smooth_grad(w, a)
        
        # Project gradient onto tangent space of simplex
        grad_proj = grad - grad.mean()
        
        # Backtracking line search
        best_alpha = 0.0
        best_f = sm_val
        alpha = lr
        for _ in range(20):
            a_new = a - alpha * grad_proj
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-15:
                alpha *= 0.5
                continue
            a_new = a_new / sn
            conv_new = fast_convolve(a_new)
            sm_new, _ = smooth_max_grad(conv_new, tau)
            if sm_new < best_f:
                best_f = sm_new
                best_alpha = alpha
            alpha *= 0.5
            if alpha < 1e-12:
                break
        
        if best_alpha > 0:
            a = a - best_alpha * grad_proj
            a = np.maximum(a, 0)
            a = a / a.sum()
            
            exact_new = fast_convolve(a).max()
            if exact_new < best_obj - 1e-12:
                best_obj = exact_new
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Anneal temperature
        if it % 50 == 0:
            tau = min(tau * 1.3, tau_max)
        
        if no_improve > 30:
            # Restart with perturbation
            rng = np.random.default_rng(it * 31 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            tau = tau_init
            lr = lr_init
            no_improve = 0
        
        if no_improve > 100:
            lr *= 0.5
            if lr < 1e-6:
                lr = lr_init
        
        if verbose and it % 100 == 0:
            print(f"    SM-GD it={it}, exact={best_obj:.10f}, tau={tau:.1f}, lr={lr:.6f}")
    
    return best_a, best_obj

Let me now think about a completely different approach: **Iterative Water-Filling / Proportional Fairness**.

The idea: 
1. Start with some a
2. Compute conv(a, a)
3. The "bottleneck" is the peak. We want to "flatten" the convolution.
4. For the peak position k*, we want to reduce `(a*a)[k*]`. This means reducing `sum_j a[j] * a[k*-j]`.
5. We can do this by reducing some a[j] that have large products with a[k*-j].
6. The freed mass should be redistributed to positions that don't contribute to the peak.

This is essentially what the peak_shave function does, but let me make it more sophisticated:

**Iterative Peak Equalization (IPE):**
1. Compute conv(a, a)
2. Find the top K peaks
3. For each peak, compute the "sensitivity" of each a[j] to that peak
4. Redistribute mass from high-sensitivity positions to low-sensitivity positions
5. Use a water-filling criterion: equalize the peaks

This is similar to the "water-filling" algorithm in information theory.

def peak_equalization(a0, time_limit, K=5, verbose=False):
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        
        # Find top K peaks
        if K < len(conv):
            peaks = np.argpartition(conv, -K)[-K:]
        else:
            peaks = np.arange(len(conv))
        
        peak_vals = conv[peaks]
        max_peak = peak_vals.max()
        mean_peak = peak_vals.mean()
        
        # For each peak, compute sensitivity of each a[j]
        # sens[j] = sum over peaks of weight * 2 * a[peak - j]
        # This is the gradient contribution from the top peaks
        weights = peak_vals / peak_vals.sum()
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            k = int(peaks[idx])
            w = weights[idx]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            # grad[j] += w * 2 * a[k-j] for j in [js, je]
            for j in range(js, je + 1):
                grad[j] += w * 2.0 * a[k - j]
        
        # Water filling: reduce high-grad positions, increase low-grad positions
        # Target: make grad more uniform
        grad_mean = grad.mean()
        # Move mass from above-average grad to below-average grad
        excess = np.maximum(grad - grad_mean, 0)
        deficit = np.maximum(grad_mean - grad, 0)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            no_improve += 1
            break
        
        # Fraction to move
        frac = 0.05
        delta = np.zeros(n)
        delta[excess > 0] = -frac * excess[excess > 0]
        # Distribute to deficit positions proportional to their deficit
        if deficit.sum() > 1e-15:
            delta[deficit > 0] += frac * (-delta.sum()) * deficit[deficit > 0] / deficit.sum()
        
        a_new = a + delta
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        new_obj = fast_convolve(a_new).max()

            a = a_new

            no_improve = 0
            frac = 0.05

**New idea: Use the dual problem.**

The primal problem is:
```
min M
s.t. (a*a)[k] <= M for all k = 0, ..., 2n-2
     sum_j a_j = 1
     a_j >= 0
```

The Lagrangian is:
```
L(a, M, λ, μ) = M + sum_k λ_k ((a*a)[k] - M) - μ (sum_j a_j - 1) - sum_j ν_j a_j
```

At the optimum, the KKT conditions give:
- `(a*a)[k] <= M` (primal feasibility)
- `λ_k >= 0` (dual feasibility)
- `λ_k ((a*a)[k] - M) = 0` (complementary slackness)
- `∂L/∂a_j = 0` for a_j > 0: `sum_k λ_k * 2 * a[k-j] - μ = 0`

So at the optimum, for all j with a_j > 0:
`sum_k λ_k * a[k-j] = μ/2` (constant)

This means that the "weighted cross-correlation" of λ and a is constant on the support of a.

This gives us an **iterative algorithm**:
1. Initialize λ (e.g., λ_k = 1 for all k where (a*a)[k] is close to M)
2. Given λ, find a that satisfies `sum_k λ_k * a[k-j] = const` for all j in support(a)
3. Update M = max_k (a*a)[k]
4. Update λ based on which constraints are tight
5. Iterate

Step 2 is interesting: we need `cross_corr(λ, a)[j] = const` for j in support. This is a linear constraint on a. If we fix the support, we can solve for a using linear algebra.

But the support is unknown and changes during the iteration. This makes it complex.

However, we can use a **multiplicative weight update** based on the dual:
- If `sum_k λ_k * a[k-j] > μ/2`, decrease a_j
- If `sum_k λ_k * a[k-j] < μ/2`, increase a_j

- For each coordinate i, find the optimal value of a_i (given all others fixed) that minimizes the max of the convolution
- This is a 1D optimization problem (unimodal in a_i)

For a fixed a_{-i} (all coordinates except i), the convolution is:
`(a*a)[k] = (a_{-i}*a_{-i})[k] + a_i * (a_{k-i} + a_{k-i})` (approximately)

More precisely:
`(a*a)[k] = sum_j a_j a_{k-j}` 
= `sum_{j≠i} a_j a_{k-j}` + `a_i a_{k-i}` + `a_{k-i} a_i` (if k-i ≠ i)
= `sum_{j≠i} a_j a_{k-j}` + `2 a_i a_{k-i}` (if k-i ≠ i)
= `sum_{j≠i} a_j a_{k-j}` + `a_i²` (if k-i = i, i.e., k = 2i)

So `(a*a)[k]` is linear in a_i (for k ≠ 2i) and quadratic (for k = 2i).

The max of a set of linear (and one quadratic) functions of a_i is convex in a_i. So we can find the minimum by ternary search or by finding where the derivative changes sign.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Smooth-max projected gradient descent with Adam optimizer and temperature annealing
- SCA (Successive Convex Approximation) with LP oracle for fine polish
- Spectral shaping initializations (targeting flat power spectrum)
- Multi-n exploration with diverse restarts
- Peak equalization via water-filling heuristic
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.special import softmax
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def logsumexp_smooth(x, tau):
    """Compute (1/tau)*log(sum(exp(tau*x))) stably. Returns (value, softmax_weights)."""
    x_max = x.max()
    scaled = tau * (x - x_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = x_max + np.log(exp_scaled.sum()) / tau
    w = exp_scaled / exp_scaled.sum()
    return log_sum, w


def compute_smooth_gradient(w, a):
    """
    Compute grad[i] = 2 * sum_k w[k] * a[k-i] for i in 0..n-1.
    w has length 2n-1, a has length n.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    L = 1 << (3 * n - 1).bit_length()
    w_pad = np.zeros(L)
    w_pad[:2 * n - 1] = w
    a_pad = np.zeros(L)
    a_pad[:n] = a
    W = np.fft.rfft(w_pad)
    A = np.fft.rfft(a_pad)
    # Cross-correlation: ifft(FFT(w) * conj(FFT(a)))[i] = sum_k w[k] * a[(k-i) mod L]
    corr = np.fft.irfft(W * np.conj(A), L)
    return 2.0 * corr[:n]

                  lr_init=0.05, use_adam=True, verbose=False):

    Projected gradient descent with log-sum-exp smooth approximation of max.
    Uses Adam optimizer with backtracking line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    tau = tau_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    restart_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.05:
            break

        conv = fast_convolve(a)
        sm_val, w = logsumexp_smooth(conv, tau)
        exact_obj = conv.max()

        grad = compute_smooth_gradient(w, a)
        grad_proj = grad - grad.mean()

        if use_adam:
            adam_t += 1
            m = beta1 * m + (1 - beta1) * grad_proj
            v = beta2 * v + (1 - beta2) * grad_proj ** 2
            m_hat = m / (1 - beta1 ** adam_t)
            v_hat = v / (1 - beta2 ** adam_t)

        else:

        best_f = sm_val + 1e-10

        alpha = 1.0

            sn = a_new.sum()
            if sn < 1e-15:
