def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve_fft(a, fft_size=None):
    """Linear auto-convolution via FFT with specified size."""
    n = len(a)
    if fft_size is None:
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:2*n-1]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (f*g)[k] = sum_j A[k,j]*g[j]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

    for k in range(1, N):
        phase = rng.uniform(0, 2 * np.pi)
        A[k] = target_amp * np.exp(1j * phase)
    # Conjugate symmetry for real signal
    for k in range(1, N // 2 + 1):
        if k < N - k:
            A[N - k] = np.conj(A[k])
    
    for _ in range(n_iter):
        a = np.real(np.fft.irfft(A, N))[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
            break
        a = a / s
        # Update spectrum
        A = np.fft.rfft(a, N)
        # Flatten magnitudes (keep DC)
        mags = np.abs(A)
        mags[0] = 1.0
        # For non-DC, set to geometric mean or a fraction
        non_dc = mags[1:]
        if non_dc.max() > 0:
            target = non_dc.mean() * 0.8  # Slightly below mean for convergence
            mags[1:] = target
        # Preserve phase
        phases = np.angle(A)
        A = mags * np.exp(1j * phases)
        # Rebuild full spectrum
        A_full = np.zeros(N, dtype=np.complex128)
        A_full[:len(A)] = A  # rfft gives N//2+1 bins
        for k in range(1, len(A)):
            if N - k < len(A):
                A_full[N - k] = np.conj(A[k])
        A = A_full
    
    a = np.real(np.fft.irfft(A, N))[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def spectral_flat_init_v2(n, rng, n_iter=100, flatten_strength=0.7):
    """
    Improved spectral flattening: uses a softer approach.
    Instead of hard flattening, use power compression.
    """
    N = 1 << (2 * n - 1).bit_length()
    
    # Initialize with a windowed sequence
    x = np.arange(n, dtype=np.float64)
    # Raised cosine window
    a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    
    for i in range(n_iter):
        A = np.fft.rfft(a, N)
        mags = np.abs(A)
        phases = np.angle(A)
        
        # Power compression: |A_new| = |A|^alpha where alpha < 1 flattens
        alpha = flatten_strength
        mags[0] = 1.0  # Fix DC
        mags[1:] = mags[1:] ** alpha
        # Normalize non-DC to have the right total power
        # We want sum(|A|^2) to be consistent with sum(a^2)
        # For now, just rescale
        non_dc_power = (mags[1:] ** 2).sum()
        target_power = (N - 1) * (1.0 / N) ** 2  # ≈ 1/N per bin
        # Actually, let's just normalize so the total energy is right
        # Keep it simple: don't renormalize, just let the projection handle it
        
        A_new = mags * np.exp(1j * phases)
        a_new = np.real(np.fft.irfft(A_new, N))[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
            break
        a_new = a_new / s
        
        if i > 0:
            # Track convergence
            if np.max(np.abs(a_new - a)) < 1e-10:
                a = a_new
                break
        a = a_new
    
    return a


def log_sum_exp_obj(a, T=10.0):
    """Compute the log-sum-exp approximation to max(conv(a,a))."""
    conv = fast_convolve(a)
    # Use log-sum-exp for numerical stability
    max_c = conv.max()
    if max_c < 1e-30:
        return max_c, conv
    shifted = T * (conv - max_c)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    lse = max_c + np.log(np.exp(shifted).sum()) / T
    return lse, conv


def log_domain_descent(a0, time_limit, T_init=5.0, T_final=50.0, lr_init=0.01, n_inner=3):
    """
    Gradient descent in log-domain with log-sum-exp smoothed objective.
    Parameterization: a[j] = exp(x[j]) / sum(exp(x))
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    # Initialize x from a
    x = np.log(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    T = T_init
    lr = lr_init
    v = np.zeros(n)  # momentum
    mom = 0.9
    
    t0 = time.monotonic()
    it = 0
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        
        # Current a from x
        x_max = x.max()
        exp_x = np.exp(x - x_max)
        a = exp_x / exp_x.sum()
        
        # Compute convolution and gradient
        conv = fast_convolve(a)
        max_c = conv.max()
        
        if max_c < best_obj - 1e-12:
            best_obj = max_c
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Log-sum-exp weighted gradient
        T_eff = min(T, 200.0)
        shifted = T_eff * (conv - max_c)
        shifted = np.clip(shifted, -500, 0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w = w / w_sum
        
        # Gradient of max(conv) w.r.t. a
        # (conv)[k] = sum_j a[j] * a[k-j]
        # d(conv[k])/da[m] = a[k-m] + a[m-k] (when indices valid)
        # Weighted sum: grad[m] = sum_k w[k] * (a[k-m] + a[m-k])
        grad = np.zeros(n)
        for k in range(len(conv)):
            if w[k] < 1e-12:
                continue
            # a[k-m] is valid when 0 <= k-m <= n-1, i.e., k-n+1 <= m <= k
            # a[m-k] is valid when 0 <= m-k <= n-1, i.e., k <= m <= k+n-1
            lo1 = max(0, k - n + 1)
            hi1 = min(n - 1, k)
            lo2 = max(0, k)
            hi2 = min(n - 1, k + n - 1)
            
            if lo1 <= hi1:
                m_arr = np.arange(lo1, hi1 + 1)
                grad[m_arr] += w[k] * a[k - m_arr]
            if lo2 <= hi2:
                m_arr = np.arange(lo2, hi2 + 1)
                grad[m_arr] += w[k] * a[m_arr - k]
        
        # Gradient in x-space: dL/dx[m] = a[m] * (grad[m] - sum_j a[j]*grad[j])
        # (from the softmax/normalize structure)
        mean_grad = (a * grad).sum()
        grad_x = a * (grad - mean_grad)
        
        # Momentum
        v = mom * v - lr * grad_x
        x = x + v
        
        # Anneal T and lr
        if stagnant > 30:
            T = min(T * 1.1, T_final)
            lr *= 0.98
        if stagnant > 100:
            # Restart with perturbation
            rng = np.random.default_rng(it * 7919)
            x = np.log(best_a) + rng.standard_normal(n) * 0.01
            v = np.zeros(n)
            T = T_init
            lr = lr_init
            stagnant = 0
    
    # Final: compute a from x
    x_max = x.max()
    a = np.exp(x - x_max)
    a = a / a.sum()
    obj = fast_convolve(a).max()
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()
    
    return best_a, best_obj


def fw_scc_optimize(f0, time_limit, verbose=False):
    """
    Frank-Wolfe with Strongly Convex Combination (SCC) step.
    Uses LP oracle with adaptive M and backtracking line search.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    stagnant = 0
    it = 0
    m_relax = 1.001

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.2:
            break

        # Solve LP oracle
        g = solve_lp_oracle(f, M * m_relax)
        if g is None or g.sum() < 1e-15:
            g = solve_lp_oracle(f, M * 1.01)
        
        if g is None or g.sum() < 1e-15:
            stagnant += 1
            if stagnant > 30:
                rng = np.random.default_rng(it * 7919 + 3)
                scale = 0.003 + 0.003 * min(1.0, stagnant / 100.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.001
                stagnant = 0
            continue

        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-10:
            stagnant += 1
            if stagnant > 40:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue

        # SCC step: find optimal alpha in [0, 1]
        # Objective: phi(alpha) = max_k ((1-alpha)*f + alpha*g) * ((1-alpha)*f + alpha*g) at lag k
        # We use the current active set (peaks)
        peaks = np.argsort(conv)[::-1][:5]
        
        best_alpha = 1.0 / max(it, 1)  # Default FW step
        best_val = np.inf
        
        # Try a grid of alphas
        alpha_grid = np.concatenate([
            np.linspace(0.01, 0.5, 10),
            np.linspace(0.5, 1.0, 10),
            [1.0 / max(it, 1)]
        ])
        
        for alpha in alpha_grid:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            # Evaluate on active peaks (faster than full conv)
            # Full conv for accuracy
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Fine refinement around best alpha
        if best_alpha > 0.01 and best_alpha < 0.99:
            for alpha in np.linspace(max(0, best_alpha - 0.05), min(1, best_alpha + 0.05), 10):
                ft = (1.0 - alpha) * f + alpha * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
        
        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1
        
        # Adapt relaxation
        if stagnant > 15:
            m_relax = min(m_relax * 1.03, 1.05)
        elif stagnant == 0:
            m_relax = 1.001

        if verbose and it % 50 == 0:
            print(f"    FW-SCC it={it}, obj={best_obj:.10f}, stagnant={stagnant}")

    return best_f, best_obj


def peak_targeted_local_search(a0, time_limit):
    """
    Local search that targets specific peaks by redistributing mass.
    More aggressive than before: tries to "cancel" peaks by 
    creating destructive interference in the convolution.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        
        conv = fast_convolve(a)
        
        # Find the top peaks
        n_peaks = min(3, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]
        
        improved = False
        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            # Pairs (j, k-j) that contribute to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            # Find the largest contributing pair
            if js > je:
                continue
            
            contributions = np.array([a[j] * a[k - j] for j in range(js, je + 1)])
            top_idx = np.argsort(contributions)[::-1][:3]
            
            for ti in range(len(top_idx)):
                j = js + top_idx[ti]
                k_j = k - j
                
                # Strategy 1: Reduce a[j] or a[k_j] and add to a "cancelling" position
                # A position that creates negative correlation at lag k
                # For lag k, we want to reduce a[j]*a[k_j]
                # Adding mass at position p changes conv[k] by a[p]*a[k-p] + a[k-p]*a[p] (roughly)
                # We want a[p]*a[k-p] to be small (or negative, but can't be negative)
                
                for src in [j, k_j]:
                    if a[src] < 1e-10:
                        continue
                    
                    # Try moving a small fraction to a random location
                    for _ in range(5):
                        p = int(rng.integers(0, n))
                        if p == src or p == k_j or p == j:
                            continue
                        
                        # Choose delta
                        for frac in [0.01, 0.005, 0.02]:
                            delta = a[src] * frac
                            candidate = a.copy()
                            candidate[src] -= delta
                            candidate[p] += delta
                            if candidate[src] < -1e-12:
                                continue
                            candidate = np.maximum(candidate, 0)
                            candidate = candidate / candidate.sum()
                            val = fast_convolve(candidate).max()
                            if val < best_obj - 1e-13:
                                a = candidate
                                best_obj = val
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    
                    if improved:
                        break
            
            if improved:
                break
        
        if not improved:
            # Try a global perturbation
            if it % 50 == 0:
                scale = max(0.001, 0.01 / (1 + it / 100))
                candidate = project_simplex(a + rng.standard_normal(n) * scale)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-13:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
            
            if it > 5000:
                break
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using cubic interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    # Use linear interpolation (cubic can introduce negative values)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def chirp_init(n, rng):
    """Initialize using a real chirp (Zadoff-Chu-like) sequence, made non-negative."""
    # Generate a complex chirp
    if n % 2 == 0:
        roots = np.arange(n)
        # Use a_prime = 1 for simplicity
        x = np.exp(-1j * np.pi * roots * roots / n)
    else:
        roots = np.arange(n)
        x = np.exp(-1j * np.pi * roots * roots / n)
    
    # Take absolute value and add noise
    a = np.abs(x) + rng.uniform(0, 0.1, n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def generate_diverse_starts(n, parent, rng):
    """Generate diverse high-quality initializations."""
    starts = []
    
    # Parent resampled
    if parent is not None:
        starts.append(resample_sequence(parent, n))
    
    # Spectral flattening (multiple random inits)
    for _ in range(3):
        starts.append(spectral_flat_init_v2(n, rng, n_iter=80, flatten_strength=0.6))
    
    # Chirp-based
    starts.append(chirp_init(n, rng))
    
    # Raised cosine with random phase modulation (made non-negative)
    x = np.arange(n, dtype=np.float64)
    base = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    phase = rng.uniform(0, 2 * np.pi, n)
    modulated = base * (0.5 + 0.5 * np.cos(phase))
    starts.append(modulated / modulated.sum())
    
    # Two-lobe with random center
    center1 = rng.uniform(0.2, 0.5)
    center2 = rng.uniform(0.5, 0.8)
    width = rng.uniform(0.05, 0.2)
    g1 = np.exp(-0.5 * ((x / n - center1) / width) ** 2)
    g2 = np.exp(-0.5 * ((x / n - center2) / width) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())
    
    # Uniform
    starts.append(np.ones(n) / n)
    
    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    starts.append(edge / edge.sum())
    
    return starts


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Multi-phase approach:
    1. Spectral initialization + quick FW across multiple lengths (200s)
    2. Log-domain gradient descent on best candidates (200s)
    3. Deep FW-SCC on best (400s)
    4. Peak-targeted local search (200s)
    5. Final polish with alternating methods (remaining time)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")
    
    n_parent = len(parent)
    
    # === Phase 1: Diverse initialization + quick optimization (200s) ===
    print("\n=== Phase 1: Diverse Init + Quick FW ===")
    t_phase = time.monotonic()
    
    # Candidate lengths
    lengths = sorted(set([n_parent, 
                          max(200, n_parent - 100), 
                          n_parent + 100,
                          384, 512, 640, 768, 896, 1024, 1280, 1536]))
    lengths = [l for l in lengths if 192 <= l <= 2048]
    
    # For each length, try a few spectral inits with short FW
    for n_try in lengths:
        remaining = deadline - time.monotonic() - 900
        if remaining < 20:
            break
        
        starts = generate_diverse_starts(n_try, parent, rng)
        
        # Evaluate all starts quickly
        scored_starts = []
        for si, start in enumerate(starts):
            obj = fast_convolve(start).max()
            scored_starts.append((obj, si, start))
        
        scored_starts.sort()
        
        # Take top 3 starts and run short FW on each
        for rank in range(min(3, len(scored_starts))):
            remaining = deadline - time.monotonic() - 850
            if remaining < 10:
                break
            _, si, start = scored_starts[rank]
            t_alloc = min(remaining, 25)
            a_opt, _ = fw_scc_optimize(start, time_limit=t_alloc)
            label = f"fw_n{n_try}_s{si}_r{rank}"
            try_seq(a_opt, label)
    
    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 2: Log-domain gradient descent (200s) ===
    print("\n=== Phase 2: Log-Domain Descent ===")
    t_phase = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        configs = [
            {'T_init': 3.0, 'lr_init': 0.02, 'T_final': 50.0},
            {'T_init': 5.0, 'lr_init': 0.015, 'T_final': 100.0},
            {'T_init': 10.0, 'lr_init': 0.01, 'T_final': 200.0},
        ]
        
        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 500
            if remaining < 20:
                break
            t_alloc = min(remaining, 60)
            # Start from best or a perturbed version
            if ci == 0:
                start = best_seq
            else:
                start = project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)
            
            print(f"  Log-descent config {ci} (T={cfg['T_init']}), t={t_alloc:.0f}s")
            a_opt, _ = log_domain_descent(start, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"logdesc_{ci}")
        
        # Also try log-descent from spectral inits
        for ci in range(2):
            remaining = deadline - time.monotonic() - 400
            if remaining < 15:
                break
            spec_init = spectral_flat_init_v2(n_best, rng, n_iter=100, flatten_strength=0.5)
            t_alloc = min(remaining, 50)
            a_opt, _ = log_domain_descent(spec_init, time_limit=t_alloc,
                                           T_init=5.0, lr_init=0.01, T_final=100.0)
            try_seq(a_opt, f"logdesc_spec_{ci}")
    
    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 3: Deep FW-SCC (400s) ===
    print("\n=== Phase 3: Deep FW-SCC ===")
    t_phase = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Main deep FW
        remaining = deadline - time.monotonic() - 450
        if remaining > 30:
            t_alloc = min(remaining, 350)
            print(f"  Deep FW-SCC n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_scc_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_fw_scc")
        
        # Perturbed restarts
        for scale in [0.002, 0.005, 0.01, 0.02]:
            remaining = deadline - time.monotonic() - 250
