def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_nonneg_sum1(v):
    """Project onto {x : x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-30:
        return np.ones(len(v)) / len(v)
    return v / s

def subgradient_at_peak(a, peak_idx):
    """Subgradient of conv(a,a)[peak_idx] w.r.t. a."""
    n = len(a)
    k = peak_idx
    sg = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        sg[j] += 2.0 * a[k - j]
    return sg

def cosine_init(n, p=1.0):
    """Generate cosine-tapered initialization."""
    j = np.arange(n)
    a = np.sin(np.pi * (j + 0.5) / n) ** p
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def random_spectral_init(n, rng):
    """Generate initialization from random spectral phases."""
    # Generate random phases
    phases = rng.uniform(0, 2 * np.pi, n)
    # Create complex spectrum with random phases
    spec = np.exp(1j * phases)
    # Inverse FFT
    a = np.fft.ifft(spec).real
    # Project to non-negative, sum to 1
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def symmetric_init(n, p=1.0, rng=None):
    """Generate symmetric cosine-tapered initialization."""
    half = n // 2
    if rng is not None:
        # Random smooth profile
        freqs = rng.uniform(0, 1, min(5, half))
        amps = rng.uniform(0.1, 1.0, min(5, half))
        j = np.arange(n)
        a = np.ones(n)
        for f, amp in zip(freqs, amps):
            a += amp * np.cos(2 * np.pi * f * j / n)
    else:
        a = cosine_init(n, p)
    # Symmetrize
    a = (a + a[::-1]) / 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def fw_lp_optimize_v2(f0, time_limit, m_tol=1.001):
    """
    Frank-Wolfe with LP oracle, optimized for speed.
    Uses the block-Toeplitz structure of the constraint matrix.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break
        
        # Solve LP: max sum(g) s.t. (f*g)[k] <= M*m_relax, g >= 0
        # Build constraint matrix (block Toeplitz)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j+n, j] = f
        b_ub = np.full(2 * n - 1, M * m_relax)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'max_time': max(1, remaining * 0.3)})
            if res.success:
                g = np.maximum(res.x, 0.0)
                g_sum = g.sum()
                if g_sum < 1e-15:
                    g = None
                else:
                    g = g / g_sum
            else:
                g = None
        except Exception:
            g = None
        
        if g is None:
            m_relax *= 1.5
            if m_relax > 10:
                m_relax = 1.0
                no_improve += 1
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.2, 10.0)
        
        if no_improve > 40:
            rng = np.random.default_rng(it * 7919)
            f = project_nonneg_sum1(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0
    
    return best_f, best_obj

def evaluate_and_update(a, best_seq, best_score, label=""):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)

    a = a / s

        best_score = score

        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return best_seq, best_score
