def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, peak_indices, peak_weights):
    """Compute weighted subgradient of max_k conv[a,a][k] at given peaks."""
    n = len(a)
    g = np.zeros(n)
    for i in range(len(peak_indices)):
        if peak_weights[i] < 1e-14:
            continue
        k = int(peak_indices[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            indices = np.arange(j_start, j_end + 1)
            g[indices] += 2.0 * peak_weights[i] * a[k - indices]
    return g


def multiplicative_shaving(a0, time_limit, eta_init=0.15, top_k=8, verbose=False):
    """
    Iterative Multiplicative Peak Shaving (IMPS).
    Uses multiplicative weight updates to reduce the peak autocorrelation.

    a = a / a.sum()
    n = len(a)

        M = conv.max()

            best_a = a.copy()
            stagnant = 0

def nesterov_projected_gradient(a0, time_limit, verbose=False):
    """
    Nesterov-accelerated projected gradient descent on the simplex.
    Minimizes max_k conv[a,a][k].
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    # Yekaterina (Nesterov momentum)
    y = a.copy()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    mom = 0.5  # momentum parameter (starts moderate)
    lr = 0.3  # initial learning rate (Lipschitz of each component is ~2, so 1/L = 0.5)

    t0 = time.monotonic()
    it = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break

        # Compute subgradient at y
        conv_y = fast_convolve(y)
        num_peaks = min(5, len(conv_y))
        peak_indices = np.argpartition(conv_y, -num_peaks)[-num_peaks:]
        peak_values = conv_y[peak_indices]
        peak_max = peak_values.max()
        T = 0.02 * peak_max
        w = np.exp((peak_values - peak_max) / max(T, 1e-15))
        w = w / w.sum()
        g = compute_subgradient(y, peak_indices, w)

        # Gradient step
        x_new = project_simplex(y - lr * g)

        # Nesterov momentum update
        beta = (1.0 + mom) / 2.0
        y_new = beta * x_new + (1.0 - beta) * y

        a = y_new
        conv = fast_convolve(a)
        M = conv.max()

        if M < best_obj - 1e-15:
            best_obj = M
            best_a = a.copy()
            stagnant = 0
            # Slightly increase momentum when making progress
            mom = min(0.95, mom * 1.002)
        else:
            stagnant += 1
            if stagnant > 20:
                mom *= 0.99
            if stagnant > 100:
                lr *= 0.95
                mom = max(0.3, mom * 0.98)

        # Backtracking: if not improving, reduce lr
        if stagnant > 50:
            lr *= 0.9
            lr = max(lr, 0.01)

    return best_a, best_obj


def peak_coordinate_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent that directly targets the top peaks.
    For each peak, redistributes mass from contributing positions to non-contributing ones.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    step_scale = 0.005

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break

        conv = fast_convolve(a)
        n_peaks = min(3, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        improved = False
        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)

            if j_start > j_end:
                continue

            # Find positions with high contribution to this peak
            # Contribution of position j to peak k is a[j] * a[k-j]
            indices = np.arange(j_start, j_end + 1)
            contribs = a[indices] * a[k - indices]
            # Sort by contribution (descending)
            order = np.argsort(contribs)[::-1]

            improved_this_peak = False
            for oi in range(min(8, len(order))):
                idx_in_range = order[oi]
                j1 = indices[idx_in_range]
                j2 = k - j1

                if a[j1] < 1e-10:
                    continue

                # Try to reduce this pair by moving mass
                # Move a small amount from j1 (or j2) to a random distant position
                for _attempt in range(3):
                    j_target = np.random.randint(0, n)
                    if j_target == j1 or j_target == j2:
                        continue

                    for src in [j1, j2]:
                        if a[src] < 1e-10:
                            continue
                        for frac in [step_scale, step_scale * 0.5, step_scale * 2.0]:
                            delta = a[src] * frac
                            if delta < 1e-12:
                                continue
                            candidate = a.copy()
                            candidate[src] -= delta
                            candidate[j_target] += delta
                            if candidate[src] < -1e-12:
                                continue
                            candidate = np.maximum(candidate, 0)
                            cs = candidate.sum()
                            if cs < 1e-15:
                                continue
                            candidate = candidate / cs
                            val = fast_convolve(candidate).max()
                            if val < best_obj - 1e-15:
                                a = candidate
                                best_obj = val
                                best_a = a.copy()
                                improved_this_peak = True
                                improved = True
                                break
                        if improved_this_peak:
                            break

                if improved_this_peak:
                    break

        if not improved:
            step_scale *= 0.92
            if step_scale < 1e-7:
                break
            if it % 500 == 0:
                # Global perturbation
                perturb = np.random.randn(n) * step_scale * 3
                candidate = project_simplex(a + perturb)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
                    step_scale *= 1.5
            elif it > 5000:
                break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.ones(new_n) / new_n
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_starts(n, parent, rng):

    starts.append((g1 + g2 + g3) / (g1 + g2 + g3).sum())

    # Sine-based (alternative to cosine)
    sin_t = np.sin(np.pi * x / (n - 1 + 0.1))

    rc = np.maximum(rc, 0.001)

def imps_with_restarts(a0, time_limit, n_restarts=3, verbose=False):
    """
    IMPS with periodic restarts to escape local optima.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    total_restarts = 0
    time_per_phase = time_limit / (n_restarts + 1)

    for ri in range(n_restarts + 1):
        phase_end = t0 + (ri + 1) * time_per_phase
        if phase_end > time.monotonic() + time_limit:
            phase_end = time.monotonic() + time_limit

        if verbose:
            print(f"    IMPS restart {ri}/{n_restarts}, phase_time={time_per_phase:.1f}s")

        result_a, result_obj = multiplicative_shaving(
            a, time_limit=max(1.0, phase_end - time.monotonic()),
            eta_init=0.15 / (1.0 + 0.3 * ri), top_k=8, verbose=verbose
        )

        if result_obj < best_obj - 1e-15:
            best_obj = result_obj
            best_a = result_a.copy()

        # Prepare for next restart with perturbation
        if ri < n_restarts:
            scale = 0.002 * (1.0 + 0.5 * ri)
            perturb = np.random.randn(n) * scale
            a = project_simplex(best_a + perturb)
            total_restarts += 1

    return best_a, best_obj

        if phase % 5 == 0:
            # IMPS
            result_a, result_obj = multiplicative_shaving(
                a, t_alloc, eta_init=0.12, top_k=10
            )
        elif phase % 5 == 1:
            # Nesterov projected gradient
            result_a, result_obj = nesterov_projected_gradient(
                a, t_alloc
            )
        elif phase % 5 == 2:
            # IMPS with different params
            result_a, result_obj = multiplicative_shaving(
                a, t_alloc, eta_init=0.05, top_k=4
            )
        elif phase % 5 == 3:
            # Coordinate descent
            result_a, result_obj = peak_coordinate_descent(
                a, t_alloc
            )
        else:
            # IMPS
            result_a, result_obj = multiplicative_shaving(
                a, t_alloc, eta_init=0.2, top_k=12
            )

        if result_obj < best_obj - 1e-15:
            best_obj = result_obj
            best_a = result_a.copy()
            a = result_a.copy()
        else:
            # Use the result anyway (might be in a different basin)
            a = result_a.copy()

        phase += 1

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search for minimizing
    2*n*max(conv(a,a))/(sum(a))^2.

    Phases:
    1. Multi-length exploration with IMPS (400s)
    2. Deep hybrid optimization on top candidates (600s)
    3. Fine-tuning with restarts (400s)
    4. Final polish (remaining)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 90.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")

    n_parent = len(parent)

    # === Phase 1: Multi-length exploration with IMPS (400s) ===
    print("\n=== Phase 1: Multi-length Exploration ===")
    t_phase = time.monotonic()
    phase1_budget = 400.0

    # Candidate lengths - focus around parent but explore wider
    lengths = sorted(set([
        max(128, n_parent // 2),
        max(192, n_parent * 3 // 4),
        n_parent,
        n_parent + 64,
        n_parent + 128,
        n_parent + 256,
        n_parent + 384,
        max(512, n_parent + 512),
        max(1024, n_parent + 1024),
        max(1536, int(n_parent * 1.75)),
        max(2048, int(n_parent * 2.5)),
        max(2560, int(n_parent * 3.0)),
    ]))
    lengths = [l for l in lengths if 128 <= l <= 4096]
    # Remove duplicates and sort
    lengths = sorted(set(lengths))

    print(f"  Trying lengths: {lengths}")

    # For each length, try a few starting points with short IMPS
    results_by_length = {}

    for n_try in lengths:
        remaining = deadline - time.monotonic() - (phase1_budget - (time.monotonic() - t_phase)) - 600
        if remaining < 20:
            break

        starts = generate_starts(n_try, parent, rng)
        # Try up to 3 starts per length
        for si, start in enumerate(starts[:3]):
            t_alloc = min(remaining, max(8.0, phase1_budget / (len(lengths) * 3)))
            if t_alloc < 5:
                break
            t_start = time.monotonic()
            a_opt, obj = multiplicative_shaving(start, time_limit=t_alloc, eta_init=0.1, top_k=6)
            elapsed = time.monotonic() - t_start
            score = try_seq(a_opt, f"P1_n{n_try}_s{si}({elapsed:.0f}s)")

            if n_try not in results_by_length:
                results_by_length[n_try] = []
            results_by_length[n_try].append((score, a_opt))

    # Sort by best score per length
    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Deep optimization on top candidates (700s) ===
    print("\n=== Phase 2: Deep Hybrid Optimization ===")
    t_phase = time.monotonic()
    phase2_budget = 700.0

    # Get top 4 candidates (by score)
    all_candidates = []
    for n_try, results in results_by_length.items():
        if results:
            best_in_len = min(results, key=lambda x: x[0])
            all_candidates.append((best_in_len[0], best_in_len[1], n_try))

    # Also include the overall best
    if best_seq is not None:
        all_candidates.append((best_score, best_seq.copy(), len(best_seq)))

    all_candidates.sort(key=lambda x: x[0])

    # Deduplicate by length (keep best per length, take top 4 lengths)
    seen_lengths = set()
    top_candidates = []
    for score, seq, n in all_candidates:
        if n not in seen_lengths:
            seen_lengths.add(n)
            top_candidates.append((score, seq, n))
        if len(top_candidates) >= 4:
            break

    print(f"  Top candidates: {[(f'{s:.6f}', n) for s, _, n in top_candidates]}")

    for ci, (cscore, cseq, cn) in enumerate(top_candidates):
        remaining = deadline - time.monotonic() - (phase2_budget - (time.monotonic() - t_phase)) - 300
        if remaining < 30:
            break

        t_alloc = min(remaining, max(60.0, phase2_budget / len(top_candidates)))
        print(f"  Deep hybrid on n={cn}, t={t_alloc:.0f}s")

        # Run hybrid optimizer
        a_opt, obj = hybrid_optimize(cseq, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, f"P2_hybrid_n{cn}")

        # Also try IMPS with restarts
        remaining2 = deadline - time.monotonic() - (phase2_budget - (time.monotonic() - t_phase)) - 300
        if remaining2 > 30:
            t_alloc2 = min(remaining2, 60.0)
            a_opt2, obj2 = imps_with_restarts(a_opt, time_limit=t_alloc2, n_restarts=2, verbose=True)
            try_seq(a_opt2, f"P2_imps_r_n{cn}")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Fine-tuning on the overall best (400s) ===
    print("\n=== Phase 3: Fine-tuning ===")
    t_phase = time.monotonic()
    phase3_budget = 400.0

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple IMPS runs with different parameters
        configs = [
            {'eta_init': 0.08, 'top_k': 5},
            {'eta_init': 0.15, 'top_k': 10},
            {'eta_init': 0.25, 'top_k': 15},
            {'eta_init': 0.05, 'top_k': 3},
            {'eta_init': 0.35, 'top_k': 20},
        ]

        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - (phase3_budget - (time.monotonic() - t_phase)) - 100
            if remaining < 15:
                break
            t_alloc = min(remaining, max(20.0, phase3_budget / len(configs)))
            a_opt, obj = multiplicative_shaving(
                best_seq, time_limit=t_alloc,
                eta_init=cfg['eta_init'], top_k=cfg['top_k']
            )
            try_seq(a_opt, f"P3_imps_{ci}_n{n_best}")

        # Nesterov projected gradient
        remaining = deadline - time.monotonic() - 100
        if remaining > 30:
            t_alloc = min(remaining, 80.0)
            a_opt, obj = nesterov_projected_gradient(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, f"P3_nesterov_n{n_best}")

        # Coordinate descent
        remaining = deadline - time.monotonic() - 80
        if remaining > 20:
            t_alloc = min(remaining, 60.0)
            a_opt, obj = peak_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, f"P3_coord_n{n_best}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Final micro-polish (remaining) ===
    print("\n=== Phase 4: Final Micro-Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Alternating tiny perturbations + short optimization
        polish_configs = [
            (0.0005, 'imps', 0.02, 3),
            (0.001, 'nesterov', 0.1, 0),
            (0.002, 'imps', 0.05, 5),
            (0.0003, 'imps', 0.01, 2),
            (0.0015, 'nesterov', 0.05, 0),
        ]

        for pi, (noise_scale, method, eta, top_k) in enumerate(polish_configs):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            t_alloc = min(remaining, 25.0)

            noise = rng.standard_normal(n_best) * noise_scale
            start = project_simplex(best_seq + noise)

            if method == 'imps':
                a_opt, _ = multiplicative_shaving(start, t_alloc, eta_init=eta, top_k=top_k)
            else:
                a_opt, _ = nesterov_projected_gradient(start, t_alloc)

            try_seq(a_opt, f"P4_polish_{pi}")

        # Final coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 10:
            t_alloc = min(remaining, 20.0)
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "P4_final_coord")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
