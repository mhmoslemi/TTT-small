Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, conv, top_k=20, temp=0.001):
    """Compute smoothed subgradient of max(conv) w.r.t. a."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    # Softmax weights
    pv = conv[peaks].copy()
    pv -= pv.max()
    w = np.exp(pv / temp)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    # Compute subgradient
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        # ∂g[k]/∂a[j] = 2*a[k-j] if 0<=k-j<n else 0
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            kj = k - j
            if 0 <= kj < n:
                sg[j] += 2.0 * w[i] * a[kj]
    
    return sg

def mirror_descent(a0, time_limit, verbose=False):
    """Mirror descent (exponentiated gradient) with smoothed subgradients."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    eta = 0.1
    top_k = max(10, n // 20)
    temp = 0.0005
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        # Compute smoothed subgradient
        sg = compute_subgradient(a, conv, top_k=top_k, temp=temp)
        
        # Exponentiated gradient update
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * sg
        log_a_new -= log_a_new.max()  # for numerical stability
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            a = a_new
            conv = conv_new
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
            eta = min(eta * 1.02, 0.5)  # slight increase

            no_improve += 1
            eta *= 0.7  # decrease step size

    return best_a, best_obj

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def coordinate_descent(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)

            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-14:
                continue
            
            # Try a range of multipliers
            cur_peak = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_peak
            
            # Logarithmic scan of multipliers
            for mult in [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0]:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                if new_val >= 1.0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            # Refine with bisection if we found an improvement
            if best_j_val != old_val:
                lo = min(old_val, best_j_val) * 0.9
                hi = max(old_val, best_j_val) * 1.1
                for _ in range(8):
                    mid = (lo + hi) / 2.0
                    if mid < 0 or mid >= 1.0:
                        break
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        break
                    scale = (1.0 - mid) / denom
                    f_trial = f.copy()
                    f_trial[j] = mid
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()
                    
                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj:
                        best_j_obj = val
                        best_j_val = mid
                        lo = mid
                    else:
                        hi = mid
                
                # Apply best
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                    
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
        
        if not improved and passes > 2:
            break
        if verbose and passes % 5 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.10f}")
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
