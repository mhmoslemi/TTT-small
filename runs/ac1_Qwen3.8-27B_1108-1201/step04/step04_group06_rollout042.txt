```python
import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_subgradient(a, conv, top_k=50, temp=0.01):
    """Compute softmax-weighted subgradient of max(conv)."""
    n = len(a)
    # Use top-k peaks with softmax weights
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    
    # Subgradient: grad[j] = sum_{k in active} w_k * (a[k-j] + a[j-k])
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        # a[k-j] is valid when 0 <= k-j <= n-1, i.e., k-n+1 <= j <= k
        # a[j-k] is valid when 0 <= j-k <= n-1, i.e., k <= j <= k+n-1
        # Combined: max(0, k-n+1) <= j <= min(n-1, k+n-1)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k + n - 1)
        for j in range(j_start, j_end + 1):
            contrib = 0.0
            if 0 <= k - j <= n - 1:
                contrib += a[k - j]
            if 0 <= j - k <= n - 1:
                contrib += a[j - k]
            grad[j] += w[idx] * contrib
    return grad
```

def eg_optimize(a0, time_limit, lr_init=1.0, top_k=50, temp_init=0.01, verbose=False):
    """Exponentiated gradient (mirror descent) on the probability simplex."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    log_a = np.log(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    eta = lr_init
    temp = temp_init
    no_improve = 0
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient
        g = compute_subgradient(a, conv, top_k=top_k, temp=temp)
        
        # Mirror descent step (in log domain)
        log_a_new = log_a - eta * g
        # Renormalize
        log_a_new -= np.logaddexp.reduce(log_a_new)  # log-sum-exp
        a_new = np.exp(log_a_new)
        
        # Check for numerical issues
        if not np.all(np.isfinite(a_new)) or a_new.sum() < 1e-15:
            eta *= 0.5
            if eta < 1e-10:
                break
            continue
        
        # Accept or reject step
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            # Slightly increase eta if improving
            eta = min(eta * 1.002, 5.0)
            # Slowly increase temperature (focus less on peaks)
            temp *= 1.001
        else:
            no_improve += 1
            eta *= 0.7
            if eta < 1e-10:
                break
        
        log_a = np.log(np.maximum(a_new, 1e-30))
        a = a_new
        conv = conv_new
        step += 1
        
        if verbose and step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    EG step {step}, obj={best_obj:.8f}, eta={eta:.4f}, t={elapsed:.1f}s")
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix C such that C @ g gives conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_step(a, conv, C=None):
    """One LP-based direction finding step."""
    n = len(a)
    M = conv.max()
    if C is None:
        C = build_conv_matrix(a)
    
    c = -np.ones(n)
    b = np.full(2 * n - 1, M * 1.0001)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1.0 + 1e-8:
                # Line search
                d = g - a
                best_a = a.copy()
                best_obj = 2.0 * n * M
                for t in [0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03]:
                    a_try = a + t * d
                    a_try = np.maximum(a_try, 0.0)
                    s = a_try.sum()
                    if s < 1e-15:
                        continue
                    a_try = a_try / s
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        break
                if best_obj < 2.0 * n * M - 1e-12:
                    return best_a, True
    except Exception:
        pass
    return a.copy(), False


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based iterative optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    C = build_conv_matrix(a)
    
    while time.monotonic() - t0 < time_limit:
        a_new, improved = lp_step(a, conv, C)
        if improved:
            a = a_new
            conv = fast_convolve(a)
            C = build_conv_matrix(a)  # Update matrix
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        iteration += 1
        if verbose and iteration % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iteration}, obj={best_obj:.8f}, t={elapsed:.1f}s")
        
        if no_improve >= 50:
            # Perturb and restart
            rng = np.random.default_rng(iteration * 31 + 7)
            scale = 0.002 + 0.003 * min(iteration // 50, 10)
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            C = build_conv_matrix(a)
            no_improve = 0
            if iteration > 500:
                break
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Various smooth profiles
    inits['sine'] = (np.sin(np.pi * x / n)**2 + 0.01)
    inits['sine'] /= inits['sine'].sum()
    
    inits['cosine'] = (np.cos(np.pi * x / n)**2 + 0.01)
    inits['cosine'] /= inits['cosine'].sum()
    
    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n))**2)
    inits['gaussian'] = g / g.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 1e-10)
                inits['parent_interp'] = pp / pp.sum()
    
    return inits
