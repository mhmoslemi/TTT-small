```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid LP + gradient approach
with improved step sizing, constraint generation for LP, and frequency-domain polish.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g) (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def compute_soft_gradient(a, conv, top_k=30, temp=0.003):
    """Compute softmax-weighted gradient of the peak autocorrelation."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad


def project_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lp_optimize_fast(a0, time_limit, verbose=False):
    """
    Fast LP-based optimization with proper simplex-preserving updates.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    prev_direction = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        # Build LP
        C = build_conv_matrix(a)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        except Exception:
            no_improve += 1
            if no_improve > 100:
                break
            continue

        if not res.success:
            no_improve += 1
            continue

        g = np.maximum(res.x, 0.0)
        g_sum = g.sum()

        if g_sum <= 1.0 + 1e-8:
            no_improve += 1
        else:
            # Normalize g to sum 1
            g_norm = g / g_sum
            d = g_norm - a

            # Line search: try several step sizes
            improved = False
            for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]:
                a_try = (1 - t) * a + t * g_norm
                # a_try is automatically on the simplex
                conv_try = fast_convolve(a_try)
                M_try = conv_try.max()
                obj_try = 2.0 * n * M_try

                if obj_try < best_obj - 1e-12:
                    a = a_try
                    conv = conv_try
                    M = M_try
                    best_obj = obj_try
                    best_a = a.copy()
                    prev_direction = d
                    improved = True
                    break

            if improved:
                no_improve = 0
            else:
                no_improve += 1

        iteration += 1

        if verbose and iteration % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iteration}, obj={best_obj:.8f}, t={elapsed:.1f}s")

        # Stagnation handling
        if no_improve >= 100:
            rng = np.random.default_rng(iteration)
            # Structured perturbation: reduce peak, add noise
            peak_idx = int(np.argmax(conv))
            # Compute gradient at peak
            grad = np.zeros(n)
            j_start = max(0, peak_idx - n + 1)
            j_end = min(n - 1, peak_idx)
            idxs = np.arange(j_start, j_end + 1)
            grad[idxs] = a[peak_idx - idxs]

            # Perturb: move away from peak, add noise
            scale = 0.002 * (1 + no_improve / 200)
            a = best_a - scale * grad / (np.linalg.norm(grad) + 1e-10)
            a = a + rng.standard_normal(n) * scale * 0.5
            a = project_simplex(a)
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0

            if no_improve > 500 or (time.monotonic() - t0 > time_limit * 0.9):
                break

    return best_a, best_obj


def nesterov_gradient(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=40, temp=0.002):
    """Nesterov-accelerated gradient descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    mom = momentum

    while time.monotonic() - t0 < time_limit:
        y = project_simplex(a + mom * v)
        conv_y = fast_convolve(y)
        grad = compute_soft_gradient(y, conv_y, top_k=top_k, temp=temp)

        v = mom * v - lr * grad
        a_new = project_simplex(a + v)

        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break

        lr *= 0.9997
        temp *= 1.002
        mom = min(0.97, mom + 0.0001)

    return best_a, best_obj


def freq_domain_polish(a0, time_limit):
    """
    Frequency-domain shaping: modify the power spectrum to flatten autocorrelation.
    Uses iterative phase adjustment (Gerchberg-Saxton inspired).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        # Compute DFT
        A = np.fft.rfft(a, 2 * n)
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP-based direction finding with simplex-preserving convex updates
2. Nesterov gradient for warm-up and polishing
3. Frequency-domain Gerchberg-Saxton-style shaping
4. Multi-restart with structured perturbations targeting the peak autocorrelation
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g) (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def compute_soft_gradient(a, conv, top_k=40, temp=0.002):
    """Compute softmax-weighted gradient of the peak autocorrelation."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad


def project_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lp_optimize_core(a0, time_limit, verbose=False):
    """
    LP-based optimization: iteratively find direction g maximizing sum(g)
    subject to conv(a,g) <= M, g>=0, then take a convex step toward g.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 80

    while time.monotonic() - t0 < time_limit:
        # Build and solve LP
        C = build_conv_matrix(a)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception:
            no_improve += 1
            if no_improve > 200:
                break
            continue

        if not res.success:
            no_improve += 1
            if no_improve > 200:
                break
            continue

        g = np.maximum(res.x, 0.0)
        g_sum = g.sum()

        if g_sum <= 1.0 + 1e-9:
            no_improve += 1
        else:
            # Normalize g to be on the simplex
            g_norm = g / g_sum

            # Line search: try step sizes from large to small
            improved = False
            for t in [0.4, 0.25, 0.15, 0.1, 0.06, 0.03, 0.01]:
                a_try = (1.0 - t) * a + t * g_norm
                conv_try = fast_convolve(a_try)
                M_try = conv_try.max()
                obj_try = 2.0 * n * M_try

                if obj_try < best_obj - 1e-13:
                    a = a_try
                    conv = conv_try
                    M = M_try
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                no_improve = 0
            else:
                no_improve += 1

        iteration += 1

        if verbose and iteration % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iteration}, obj={best_obj:.8f}, t={elapsed:.1f}s")

        # Stagnation: structured restart
        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 31 + 7)
            conv_b = fast_convolve(best_a)
            # Target the top peaks

            a = a_pert
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_no_improve = min(200, max_no_improve + 20)

            if max_no_improve > 200 or (time.monotonic() - t0 > time_limit * 0.95):

def nesterov_gradient(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=50, temp=0.002):
    """Nesterov-accelerated gradient descent on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    mom = momentum

    while time.monotonic() - t0 < time_limit:
        y = project_simplex(a + mom * v)
        conv_y = fast_convolve(y)
        grad = compute_soft_gradient(y, conv_y, top_k=top_k, temp=temp)

        v = mom * v - lr * grad
        a_new = project_simplex(a + v)

        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break

        lr *= 0.9996
        temp *= 1.003
        mom = min(0.97, mom + 0.0001)

    return best_a, best_obj


def freq_shaping(a0, time_limit):
    """
    Gerchberg-Saxton-inspired frequency domain shaping to flatten autocorrelation.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    fft_n = 2 * n

    iteration = 0
    while time.monotonic() - t0 < time_limit:
        # Go to frequency domain
        A = np.fft.rfft(a, fft_n)
        P = np.abs(A) ** 2
        phase = np.angle(A)

        # Compress power spectrum toward uniform (flattens autocorrelation)
        P_mean = P[1:].mean() if len(P) > 1 else P[0]
        alpha = 0.05 + 0.05 * (iteration % 20) / 20.0
        P_new = (1.0 - alpha) * P + alpha * np.full_like(P, P_mean)
        P_new[0] = P[0]  # Keep DC
        P_new = np.maximum(P_new, 1e-10)

        # Partially randomize phase (keep real signal)
        phase_noise = rng.uniform(-np.pi, np.pi, len(phase)) * 0.2
        phase_new = phase + phase_noise
        phase_new[0] = 0.0

        A_new = np.sqrt(P_new) * np.exp(1j * phase_new)
        a_new = np.fft.irfft(A_new, fft_n)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            iteration += 1
            continue
        a_new = a_new / s_new

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

        iteration += 1
        if iteration > 50000:
            break

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['sine1'] = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine1'] /= inits['sine1'].sum()
    inits['sine2'] = (np.sin(2 * np.pi * x / n) ** 2 + 0.01)
    inits['sine2'] /= inits['sine2'].sum()
    inits['sine3'] = (np.sin(3 * np.pi * x / n) ** 2 + 0.01)
    inits['sine3'] /= inits['sine3'].sum()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['random'] /= inits['random'].sum()

    # Bimodal
    edge = max(1, n // 6)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # Alternating
    alt = np.ones(n)
    alt[::2] = 2.0
    alt[1::2] = 0.5
    inits['alternating'] = alt / alt.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent'] = pp / pps

    return inits

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = [float('inf')]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    n_parent = len(parent) if parent is not None else 768

    # =========================================================
    # Phase 1: Gradient warm-up (quick, diverse)
    # =========================================================
    print("=== Phase 1: Gradient Warm-up ===")
    t1 = time.monotonic()
    phase1_budget = min(180.0, (deadline - time.monotonic()) * 0.12)

    lengths_to_try = [n_parent]
    if n_parent <= 1024:
        lengths_to_try.append(1024)
    lengths_to_try = [l for l in lengths_to_try if 128 <= l <= 2048]

    for n in lengths_to_try[:2]:
        if time.monotonic() > deadline - 600:
            break

        inits = generate_inits(n, parent, rng)

        # Priority order
        priority = ['parent', 'sine1', 'uniform', 'sine2', 'bimodal', 'random',
                    'gaussian', 'alternating', 'sine3']
        ordered = [nm for nm in priority if nm in inits]
        ordered += [nm for nm in inits if nm not in priority]

        t_per_init = max(8.0, phase1_budget / (len(lengths_to_try) * len(ordered)))

        for name in ordered[:5]:
            if time.monotonic() > deadline - 500:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 400)
            if t_alloc < 5:
                break

            a0 = inits[name]
            a_opt, _ = nesterov_gradient(a0, time_limit=t_alloc,
                                         lr_init=0.015, momentum=0.92,
                                         top_k=60, temp=0.001)
            score = try_candidate(a_opt, best_score)
            if score < best_score[0]:
                best_sequence = a_opt.copy()
                print(f"  Grad n={n} {name}: {score:.6f} *NEW BEST*")
            else:
                print(f"  Grad n={n} {name}: {score:.6f}")

    print(f"Phase 1 done ({time.monotonic()-t1:.0f}s). Best: {best_score[0]:.6f}")

    # =========================================================
    # Phase 2: LP deep optimization
    # =========================================================
    print("=== Phase 2: LP Deep Optimization ===")
    t2 = time.monotonic()

    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 300
        if remaining > 60:
            # Primary LP run
            t_alloc = min(remaining * 0.5, 400.0)
            print(f"  LP primary (n={n_best}), t={t_alloc:.0f}s")
            a_opt, _ = lp_optimize_core(best_sequence, time_limit=t_alloc, verbose=True)
            score = try_candidate(a_opt, best_score)
            if score < best_score[0] + 1e-10:
                if np.isfinite(score) and score < (best_sequence is not None and float(evaluate_sequence(best_sequence.tolist())) or float('inf')):
                    best_sequence = a_opt.copy()
                print(f"  LP primary: {score:.6f}")
            else:
                print(f"  LP primary: {score:.6f}")

            # Verify and update
            s_check = float(evaluate_sequence(best_sequence.tolist()))
            if np.isfinite(s_check):
                best_score[0] = min(best_score[0], s_check)

            # Second LP run with perturbation
            if time.monotonic() < deadline - 200:
                t_alloc2 = min(remaining * 0.3, 250.0)
                noise = rng.standard_normal(n_best) * 0.005
                a_pert = np.maximum(best_sequence + noise, 0)
                a_pert = project_simplex(a_pert)
                print(f"  LP perturbed, t={t_alloc2:.0f}s")
                a_opt2, _ = lp_optimize_core(a_pert, time_limit=t_alloc2)
                score2 = try_candidate(a_opt2, best_score)
                if np.isfinite(score2) and score2 < best_score[0]:
                    best_sequence = a_opt2.copy()
                    print(f"  LP perturbed: {score2:.6f} *NEW BEST*")
                else:
                    print(f"  LP perturbed: {score2:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t2:.0f}s). Best: {best_score[0]:.6f}")

    # =========================================================
    # Phase 3: Frequency-domain shaping
    # =========================================================
    print("=== Phase 3: Frequency-domain Shaping ===")
    t3 = time.monotonic()

    if best_sequence is not None and time.monotonic() < deadline - 100:
        t_alloc = min(80.0, deadline - time.monotonic() - 50)
        a_opt, _ = freq_shaping(best_sequence, time_limit=t_alloc)
        score = try_candidate(a_opt, best_score)
        if np.isfinite(score) and score < best_score[0]:
            best_sequence = a_opt.copy()
            print(f"  FreqShaping: {score:.6f} *NEW BEST*")
        else:
            print(f"  FreqShaping: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t3:.0f}s). Best: {best_score[0]:.6f}")

    # =========================================================
    # Phase 4: LP multi-restart
    # =========================================================
    print("=== Phase 4: LP Multi-Restart ===")
    t4 = time.monotonic()

    if best_sequence is not None and time.monotonic() < deadline - 40:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 25
        n_restarts = max(1, min(4, int(remaining / 80)))
        t_per = max(15.0, remaining / (n_restarts + 1))

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 10)
            if t_alloc < 10:
                break

            if ri == 0:
                a0 = best_sequence.copy()
            else:
                # Structured perturbation targeting peaks
                conv_b = fast_convolve(best_sequence)
                top_k_peaks = np.argpartition(conv_b, -10)[-10:]
                scale = 0.001 * (ri + 1)
                a0 = best_sequence.copy()
                for pk in top_k_peaks:
                    k = int(pk)
                    j_start = max(0, k - n_best + 1)
                    j_end = min(n_best - 1, k)
                    idxs = np.arange(j_start, j_end + 1)
                    a0[idxs] -= scale * best_sequence[k - idxs]
                a0 += rng.standard_normal(n_best) * scale * 0.5
                a0 = project_simplex(a0)

            a_opt, _ = lp_optimize_core(a0, time_limit=t_alloc)
            score = try_candidate(a_opt, best_score)
            if np.isfinite(score) and score < best_score[0]:
                best_sequence = a_opt.copy()
                print(f"  LP restart {ri}: {score:.6f} *NEW BEST*")
            else:
                print(f"  LP restart {ri}: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t4:.0f}s). Best: {best_score[0]:.6f}")

    # =========================================================
    # Phase 5: Final gradient polish
    # =========================================================
    print("=== Phase 5: Final Polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic() - 3
        if remaining > 8:
            # Short gradient
            t_g = min(remaining * 0.5, 40.0)
            a_opt, _ = nesterov_gradient(best_sequence, time_limit=t_g,
                                         lr_init=0.004, momentum=0.95,
                                         top_k=100, temp=0.0005)
            score = try_candidate(a_opt, best_score)
            if np.isfinite(score) and score < best_score[0]:
                best_sequence = a_opt.copy()
                print(f"  Final Grad: {score:.6f} *NEW BEST*")

            # Very short LP
            remaining = deadline - time.monotonic() - 3
            if remaining > 10:
                a_opt, _ = lp_optimize_core(best_sequence, time_limit=remaining)
                score = try_candidate(a_opt, best_score)
                if np.isfinite(score) and score < best_score[0]:
                    best_sequence = a_opt.copy()
                    print(f"  Final LP: {score:.6f} *NEW BEST*")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score[0] = min(best_score[0], final_score)
    else:
        # Should not happen, but just in case
        best_sequence = np.ones(256) / 256
        best_score[0] = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score[0]:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
