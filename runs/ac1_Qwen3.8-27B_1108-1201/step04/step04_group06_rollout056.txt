import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g) (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def objective(a):
    """Compute 2*n*max(conv)/(sum)^2 without calling evaluate_sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def lp_solve(a, M, C=None):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0."""
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

                         options={'presolve': True})
        if result.success:
            g = result.x
            if g is not None and g.sum() > 1e-10:
                return g, True
    except Exception:
        pass
    return None, False

def compute_soft_gradient(a, conv, top_k=30, temp=0.003):
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]

    scaled -= scaled.max()
    w = np.exp(scaled)

    if w_sum < 1e-15:

        j_end = min(n - 1, k)

        grad[idxs] += w[idx] * a[k - idxs]

        if not success or g is None:
            no_improve += 1
        else:
            d = g - a
            norm_d = np.linalg.norm(d)
            if norm_d < 1e-10:
                no_improve += 1
            else:
                best_t = 0.0
                best_obj_try = 2.0 * n * M
                candidates = [0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
                for t in candidates:
                    a_try = a + t * d
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_obj_try - 1e-12:
                        best_obj_try = obj_try
                        best_t = t
                    elif best_t > 0:
                        break

                if best_t > 0:
                    a = a + best_t * d
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
                    conv = fast_convolve(a)
                    cur_obj = 2.0 * n * conv.max()
                    if cur_obj < best_obj - 1e-12:
                        best_obj = cur_obj
                        best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1

        iteration += 1
        if verbose and iteration % 50 == 0:
            print(f"    LP iter {iteration}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")

        if no_improve >= max_no_improve:
            # Perturbation
            strat = perturb_count % 4
            if strat == 0:
                scale = 0.003 + 0.004 * min(perturb_count, 10)
                noise = rng.standard_normal(n) * scale
                a = np.maximum(best_a + noise, 0)
            elif strat == 1:
                x = np.arange(n) / n
                freq = 1 + (perturb_count % 7)
                phase = perturb_count * 1.3
                pert = 0.015 * np.sin(2 * np.pi * freq * x + phase)
                a = np.maximum(best_a + pert, 0)
            elif strat == 2:
                # Swap two random groups
                n_swap = max(3, n // 50)
                idx1 = rng.choice(n, n_swap, replace=False)
                idx2 = rng.choice(n, n_swap, replace=False)
                a_new = best_a.copy()
                a_new[idx1], a_new[idx2] = best_a[idx2].copy(), best_a[idx1].copy()
                a = a_new
            else:
                # Blend with random
                a_rand = rng.dirichlet(np.ones(n) * 0.3)
                a = 0.8 * best_a + 0.2 * a_rand

            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()

            no_improve = 0
            perturb_count += 1
            max_no_improve = min(300, 50 + 25 * perturb_count)
            if perturb_count > 60:
                break

    return best_a, best_obj


def _lp_solve(a, M, C=None):
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success:
            g = result.x
            if g is not None and np.all(np.isfinite(g)) and g.sum() > 1e-10:
                return g, True
    except Exception:
        pass
    return None, False
