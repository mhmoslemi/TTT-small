            if no_improve > 40:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                B = build_conv_matrix(f)
                no_improve = 0
            continue

        alpha, obj_val = line_search(f, target, current_obj)

        if alpha > 1e-8:
            f = (1.0 - alpha) * f + alpha * target
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-15:
                f = f / fs
                B = build_conv_matrix(f)

            if obj_val < best_obj - 1e-13:
                best_obj = obj_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}, alpha={alpha:.4f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            scale = 0.002 + min(no_improve * 0.0002, 0.01)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            B = build_conv_matrix(f)
            no_improve = 0

    return best_f, best_obj


def pairwise_swap_search(f0, time_limit, rng, n_swaps_per_iter=200):
    """Local search via pairwise swaps targeting peak reduction."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    delta_base = 0.003
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(best_f)
        peak_idx = np.argmax(conv)
        current_peak = conv[peak_idx]

        # Find elements contributing to the peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        contrib_indices = np.arange(js, je + 1)
        contrib_weights = best_f[peak_idx - contrib_indices]

        improveround = 0
        for _ in range(n_swaps_per_iter):
            if time.monotonic() - t0 > time_limit:
                break

            # Pick i from contributors to peak (weighted) and j from non-contributors
            if len(contrib_indices) > 0 and np.sum(contrib_weights) > 0:
                w_norm = contrib_weights / contrib_weights.sum()
                w_norm = np.clip(w_norm, 1e-6, None)
                w_norm /= w_norm.sum()
                i = contrib_indices[np.random.choice(len(contrib_indices), p=w_norm)]
            else:
                i = rng.integers(n)

            # j should be away from peak-contributing positions
            non_contrib = np.setdiff1d(np.arange(n), contrib_indices)
            if len(non_contrib) > 0:
                j = non_contrib[rng.integers(len(non_contrib))]
            else:
                j = (i + n // 2) % n

            if i == j:
                continue

            # Try swap: f[i] -= delta, f[j] += delta
            delta = delta_base * best_f.sum() / n
            # Ensure non-negativity
            delta = min(delta, best_f[i] * 0.5)
            if delta < 1e-15:
                continue

            f_trial = best_f.copy()
            f_trial[i] -= delta
            f_trial[j] += delta
            f_trial = np.maximum(f_trial, 0.0)
            ts = f_trial.sum()
            if ts < 1e-15:
                continue
            f_trial = f_trial / ts

            obj = fast_convolve(f_trial).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f_trial.copy()
                improveround += 1
                no_improve = 0
            else:
                no_improve += 1

        if no_improve > n_swaps_per_iter * 3:
            delta_base *= 0.9
            if delta_base < 1e-5:
                delta_base = 0.002
                no_improve = 0

    return best_f, best_obj


def subgradient_optimize(f0, time_limit, lr=0.002, mom=0.94, top_k=50, temp=0.00005):
    """Smoothed subgradient descent with momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    rng = np.random.default_rng(999)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.003

        if no_improve > 400:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            cur_lr = max(lr * 0.7, 0.0003)
            cur_temp = temp
            no_improve = 0

    return best_f, best_obj


def nelder_mead_1d(f0, time_limit, n_basis=8):
    """Optimize a low-dimensional parameterization using Nelder-Mead."""
    n = len(f0)
    f_base = f0.copy()
    s = f_base.sum()
    f_base = f_base / s

    # Build basis: cosines and polynomials
    x = np.linspace(0, 1, n)
    basis = []
    for k in range(1, n_basis + 1):
        basis.append(np.cos(2 * np.pi * k * x))
        # Make zero-mean
        basis[-1] = basis[-1] - basis[-1].mean()

    basis = np.array(basis)  # shape (n_basis, n)

    best_obj = fast_convolve(f_base).max()
    best_params = np.zeros(n_basis)
    best_f = f_base.copy()

    def objective(params):
        a = f_base + params @ basis
        a = np.maximum(a, 0.0)
        sa = a.sum()
        if sa < 1e-15:
            return 1e10
        a = a / sa
        return fast_convolve(a).max()

    t0 = time.monotonic()
    # Simplex
    simplex = [np.zeros(n_basis)]
    scale = 0.01
    for i in range(n_basis):
        pt = np.zeros(n_basis)
        pt[i] = scale
        simplex.append(pt)

    func_vals = [objective(p) for p in simplex]
    n_iter = 0

    while time.monotonic() - t0 < time_limit:
        n_iter += 1
        # Sort
        order = np.argsort(func_vals)
        simplex = [simplex[i] for i in order]
        func_vals = [func_vals[i] for i in order]

        # Check best
        if func_vals[0] < best_obj - 1e-13:
            best_obj = func_vals[0]
            best_params = simplex[0].copy()
            a = f_base + best_params @ basis
            a = np.maximum(a, 0.0)
            best_f = a / a.sum()

        # Centroid of all but worst
        centroid = np.mean(simplex[:-1], axis=0)

        # Reflection
        sigma = 1.0
        x_r = centroid + sigma * (centroid - simplex[-1])
        f_r = objective(x_r)

        if f_r < func_vals[0]:
            # Expansion
            gamma = 2.0
            x_e = centroid + gamma * (centroid - simplex[-1])
            f_e = objective(x_e)
            if f_e < f_r:
                simplex[-1] = x_e
                func_vals[-1] = f_e
            else:
                simplex[-1] = x_r
                func_vals[-1] = f_r
        elif f_r < func_vals[-2]:
            simplex[-1] = x_r
            func_vals[-1] = f_r
        else:
            # Contraction
            rho_c = 0.5
            x_c = centroid + rho_c * (simplex[-1] - centroid)
            f_c = objective(x_c)
            if f_c < func_vals[-1]:
                simplex[-1] = x_c
                func_vals[-1] = f_c
            else:
                # Shrink
                for i in range(1, len(simplex)):
                    simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                    func_vals[i] = objective(simplex[i])

        # Convergence check
        if max(func_vals) - min(func_vals) < 1e-14:
            break

    return best_f, best_obj


def generate_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine modulation
    for freq in [1, 2, 3, 5]:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x)
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())

    # Gaussian bump
    for sigma in [0.1, 0.2, 0.35]:
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        inits.append(a / a.sum())

    # Edge-weighted
    a = 1.0 / (0.1 + np.abs(x - 0.5) ** 2)
    inits.append(a / a.sum())

    # Beta-like
    a = (x + 0.01) ** 1.5 * (1 - x + 0.01) ** 1.5
    inits.append(a / a.sum())

    # Random smooth
    coef = rng.standard_normal(6) * 0.2
    a = 1.0 + sum(c * np.cos(2 * np.pi * (k + 1) * x) for k, c in enumerate(coef))
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())

    # Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP Frank-Wolfe, pairwise swap search, subgradient, Nelder-Mead,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
            try_seq(parent, "parent")
    except Exception:
        pass

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")
    t_start = time.monotonic()

    candidates = []
    if parent is not None:
        n_p = len(parent)
        candidates.append(("parent", parent))
    else:
        n_p = 384
        candidates.append(("uniform", np.ones(n_p) / n_p))

    inits = generate_structured_inits(n_p, rng)
    for i, ini in enumerate(inits):
        candidates.append((f"init_{i}", ini))

    # Try a few nearby lengths
    for n_try in [n_p - 32, n_p + 32, 320, 448]:
        if n_try < 100 or n_try == n_p:
            continue
        if parent is not None:
            px = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_p), parent)
        else:
            px = np.ones(n_try) / n_try
        px = np.maximum(px, 0)
        px = px / px.sum()
        candidates.append((f"len{n_try}", px))

    for label, cand in candidates:
        if time.monotonic() > deadline - 300:
            break
        try_seq(cand, label)

    if best_seq is None:
        best_seq = np.ones(384) / 384
        try_seq(best_seq, "fallback")

    print(f"  Phase 1 best: {best_score:.8f} (n={len(best_seq)})")

    n_best = len(best_seq)

    # === Phase 2: LP Frank-Wolfe (primary) ===
    print("\n=== Phase 2: LP Frank-Wolfe ===")
    remaining = deadline - time.monotonic()
    t_fw_full = min(remaining * 0.35, 600)

    if t_fw_full > 60:
        print(f"  Full LP FW, budget={t_fw_full:.0f}s")
        a_opt, _ = fw_optimize(best_seq, time_limit=t_fw_full * 0.6,
                               use_reduced=False, verbose=True)
        try_seq(a_opt, "fw_full")

        rem = deadline - time.monotonic()
        if rem > 60:
            t_fw_red = min(rem * 0.3, 300)
            a_opt, _ = fw_optimize(best_seq, time_limit=t_fw_red,
                                   use_reduced=True, reduced_k=30)
            try_seq(a_opt, "fw_reduced")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Pairwise swap local search ===
    print("\n=== Phase 3: Pairwise Swap Search ===")
    remaining = deadline - time.monotonic()
    t_swap = min(remaining * 0.25, 400)

    if t_swap > 30:
        print(f"  Swap search, budget={t_swap:.0f}s")
        a_opt, _ = pairwise_swap_search(best_seq, time_limit=t_swap * 0.7, rng=rng)
        try_seq(a_opt, "swap1")

        rem = deadline - time.monotonic()
        if rem > 30:
            a_opt, _ = pairwise_swap_search(best_seq, time_limit=min(rem * 0.4, 150), rng=rng)
            try_seq(a_opt, "swap2")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Subgradient polishing ===
    print("\n=== Phase 4: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    t_sg = min(remaining * 0.2, 300)

    if t_sg > 20:
        print(f"  SG polish, budget={t_sg:.0f}s")
        a_opt, _ = subgradient_optimize(best_seq, time_limit=t_sg * 0.6,
                                        lr=0.002, mom=0.94, top_k=n_best // 5, temp=0.00004)
        try_seq(a_opt, "sg1")

        rem = deadline - time.monotonic()
        if rem > 20:
            a_opt, _ = subgradient_optimize(best_seq, time_limit=min(rem * 0.4, 120),
                                            lr=0.0008, mom=0.96, top_k=n_best // 3, temp=0.00001)
            try_seq(a_opt, "sg_fine")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Nelder-Mead on cosine parameterization ===
    print("\n=== Phase 5: Nelder-Mead Parameterization ===")
    remaining = deadline - time.monotonic()
    t_nm = min(remaining * 0.15, 200)

    if t_nm > 20:
        print(f"  Nelder-Mead, budget={t_nm:.0f}s")
        a_opt, _ = nelder_mead_1d(best_seq, time_limit=t_nm, n_basis=10)
        try_seq(a_opt, "nm")

    print(f"  Phase 5 best: {best_score:.8f}")

    # === Phase 6: Restart + re-optimize ===
    print("\n=== Phase 6: Restart & Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 60:
        for attempt in range(5):
            if time.monotonic() > deadline - 30:
                break
            noise_scale = 0.003 * (attempt + 1)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            t_each = min((deadline - time.monotonic() - 20) * 0.4, 80)
            if t_each < 15:
                break

            # Quick FW + swap
            a_opt, _ = fw_optimize(f_pert, time_limit=t_each * 0.5,
                                   use_reduced=True, reduced_k=25)
            try_seq(a_opt, f"restart_{attempt}")

            rem = deadline - time.monotonic() - 15
            if rem > 15:
                a_opt, _ = pairwise_swap_search(a_opt, time_limit=min(rem * 0.5, 40), rng=rng)
                try_seq(a_opt, f"restart_{attempt}_swap")

    print(f"  Phase 6 best: {best_score:.8f}")

    # === Phase 7: Try nearby lengths ===
    print("\n=== Phase 7: Length Exploration ===")
    remaining = deadline - time.monotonic()

    if remaining > 40 and best_seq is not None:
        for n_try in [n_best - 16, n_best + 16, n_best - 64, n_best + 64]:
            if n_try < 128 or time.monotonic() > deadline - 30:
                continue
            px = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_best), best_seq)
            px = np.maximum(px, 0)
            px = px / px.sum()
            t_each = min((deadline - time.monotonic() - 20) * 0.3, 50)
            if t_each < 10:
                break
            a_opt, _ = fw_optimize(px, time_limit=t_each, use_reduced=True, reduced_k=25)
            try_seq(a_opt, f"len{n_try}")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
