1. Spectral shaping (Gerchberg-Saxton variant) for global structure
2. Nesterov accelerated projected subgradient for conviction-based optimization
3. Multi-length exploration with interpolation
4. Coordinate descent polishing
Key insight: asymmetric sequences reduce center convolution peak.
"""

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_flatten(f0, time_limit, alpha=0.5, n_iter_max=5000, verbose=False):
    """
    Spectral shaping: flatten the power spectrum to reduce autocorrelation peaks.
    Uses Gerchberg-Saxton style iteration.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()  # Actually need 2n-1 for linear conv
    # For linear convolution of length n, we need FFT size >= 2n-1
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size *= 2
    
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and it < n_iter_max:
        it += 1
        
        # Compute FFT
        A = np.fft.fft(f, fft_size)
        P = np.abs(A)**2
        
        # Compute target: flat spectrum
        # P_target = 1 (uniform)
        # Correction factor: (P_mean / P)^alpha
        P_safe = np.maximum(P, 1e-30)
        P_mean = P.sum() / fft_size
        
        # Gentle correction
        correction = (P_mean / P_safe) ** alpha
        A_new = A * correction
        
        # Inverse FFT
        f_new = np.real(np.fft.ifft(A_new, fft_size))[:n]
        
        # Project to non-negative and normalize
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new = f_new / s_new
        
        # Evaluate
        conv = fast_convolve(f_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    Spectral iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            # Try with smaller alpha
            if no_improve % 100 == 0:
                alpha = max(alpha * 0.9, 0.05)
        
        f = f_new
        
        if no_improve > 2000:
            break
    
    return best_f, best_obj

def nesterov_subgrad(f0, time_limit, verbose=False, top_k=5):
    """
    Nesterov accelerated projected subgradient for minimizing max_k (a*a)[k]
    on the simplex.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    t_k = 1.0
    f_prev = f.copy()
    no_improve = 0
    it = 0
    lr_base = 2.0 / (2.0 * np.linalg.norm(f, 2))  # Initial step size
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov momentum
        momentum = (t_k - 1) / (t_k + 1)
        y = f + momentum * (f - f_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = best_f.copy()
        else:
            y = y / sy
        
        # Compute convolution at y
        conv_y = fast_convolve(y)
        M_y = conv_y.max()
        
        # Find top-k peaks for subgradient
        k_actual = min(top_k, len(conv_y))
        peak_indices = np.argpartition(conv_y, -k_actual)[-k_actual:]
        peak_vals = conv_y[peak_indices]
        
        # Weight peaks by softmax
        scaled = peak_vals / 0.001
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Compute subgradient
        # grad of (a*a)[k] w.r.t. a[m] = 2*a[k-m] if 0<=k-m<n, else 0
        sg = np.zeros(n)
        for i in range(len(peak_indices)):
            wi = w[i]
            if wi < 1e-12:
                continue
            k = int(peak_indices[i])
            # (a*a)[k] = sum_j a[j] * a[k-j]
            # d/d a[m] = 2 * a[k-m] (if valid)
            for m in range(max(0, k - n + 1), min(n, k + 1)):
                idx = k - m
                if 0 <= idx < n:
                    sg[m] += 2.0 * wi * y[idx]
        
        # Adaptive step size
        lr = lr_base / (t_k + 1)
        
        # Update
        f_prev = f.copy()
        f_new = y - lr * sg
        f_new = project_simplex(f_new)
        
        # Evaluate
        conv = fast_convolve(f_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    Nesterov iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        # Update Nesterov parameter
        t_k = (1 + np.sqrt(1 + 4 * t_k**2)) / 2
        
        # Restart if stuck
        if no_improve > 500:
            # Reset with perturbation
            rng = np.random.default_rng(it)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            f_prev = f.copy()
            t_k = 1.0
            no_improve = 0
            lr_base = 2.0 / (2.0 * np.linalg.norm(f, 2))
        
        f = f_new
    
    return best_f, best_obj

def multi_peak_subgrad(f0, time_limit, verbose=False):
    """
    Subgradient method that directly uses the convolution peak structure.
    Uses exponential weighting of peaks for a better descent direction.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    temp = 0.01  # Temperature for peak weighting
    lr = 0.01
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Weight peaks by exponential of (peak - M) / temp
        # This gives more weight to peaks near the maximum
        diff = conv - M
        exp_w = np.exp(diff / temp)
        exp_w /= exp_w.sum()
        
        # Compute weighted subgradient
        # sg[m] = sum_k exp_w[k] * 2 * (f[k-m] if 0<=k-m<n)
        # This is 2 * (f * reverse(exp_w))[m] ... let me think
        # Actually: sg[m] = sum_{k} exp_w[k] * 2 * f[k-m] (where f[k-m]=0 if out of range)
        # = 2 * sum_k exp_w[k] * f[k-m]
        # This is a cross-correlation of exp_w and f
        # Using FFT: it's the inverse FFT of FFT(exp_w) * conj(FFT(f))
        # Or more simply: it's the linear cross-correlation
        
        # Actually, let me just compute it directly using the convolution structure
        # sg[m] = 2 * sum_{k: 0<=k-m<n} exp_w[k] * f[k-m]
        # Let j = k-m: sg[m] = 2 * sum_{j: 0<=j<n, 0<=j+m<2n-1} exp_w[j+m] * f[j]
        # = 2 * (cross_corr(exp_w, f))[m]
        
        # Use FFT for efficiency
        size = 2 * n
        fft_size = 1 << (size - 1).bit_length()
        F = np.fft.rfft(f, fft_size)
        W = np.fft.rfft(exp_w, fft_size)
        # Cross-correlation: ifft(conj(F) * W)
        cross = np.fft.irfft(np.conj(F) * W, fft_size)[:n]
        sg = 2.0 * cross
        
        # Adaptive learning rate
        sg_norm = np.linalg.norm(sg)
        if sg_norm > 1e-15:
            step = lr * M / sg_norm
            step = min(step, 0.5)  # Cap the step
        else:
            step = 0.0
        
        f_new = f - step * sg
        f_new = project_simplex(f_new)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            # Slowly decrease temperature
            temp = max(temp * 0.999, 0.001)
        else:
            no_improve += 1
        
        f = f_new
        
        if verbose and it % 500 == 0:
            print(f"    Multi-peak iter {it}: obj={best_obj:.10f}, temp={temp:.6f}")
        
        if no_improve > 300:
            # Restart
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            temp = 0.01
            no_improve = 0
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def coordinate_descent(f0, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate, find the best value by
    binary search, keeping others proportionally scaled.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-14:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            # Try several multiplier values
            multipliers = [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial /= f_trial.sum()
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_flatten(f0, time_limit, alpha=0.3, verbose=False):
    """
    Spectral shaping: flatten the power spectrum to reduce autocorrelation peaks.
    Gerchberg-Saxton style iteration: modify spectrum, project to non-negative.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size *= 2

    it = 0
    no_improve = 0
    cur_alpha = alpha

    while time.monotonic() - t0 < time_limit:
        it += 1

        A = np.fft.fft(f, fft_size)
        P = np.abs(A) ** 2
        P_safe = np.maximum(P, 1e-30)
        P_mean = P.sum() / fft_size

        correction = (P_mean / P_safe) ** cur_alpha
        A_new = A * correction
        f_new = np.real(np.fft.ifft(A_new, fft_size))[:n]
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new = f_new / s_new

        conv = fast_convolve(f_new)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 200 == 0:
                cur_alpha = max(cur_alpha * 0.85, 0.03)

        f = f_new
        if no_improve > 3000:
            break

    return best_f, best_obj


def gradient_descent_ls(f0, time_limit, verbose=False, top_k=1, temp_init=0.003):
    """
    Gradient descent with line search for minimizing max_k (a*a)[k] on simplex.
    Uses weighted peaks for direction, backtracking line search.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    temp = temp_init
    alpha_mem = 0.1

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Find weighted peaks for direction
        k_actual = min(top_k, len(conv))
        peak_indices = np.argpartition(conv, -k_actual)[-k_actual:]
        peak_vals = conv[peak_indices]

        # Exponential weighting over all peaks near max
        diff = conv - M
        exp_w = np.exp(diff / temp)
        wsum = exp_w.sum()
        if wsum < 1e-30:
            exp_w = np.zeros(len(conv))
            exp_w[int(np.argmax(conv))] = 1.0
        else:
            exp_w /= wsum

        # Compute subgradient: sg[m] = 2 * sum_j f[j] * exp_w[j+m]
        sg = np.zeros(n)
        for m in range(n):
            sg[m] = 2.0 * np.dot(f, exp_w[m:m + n])

        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break

        sg = sg / sg_norm

        # Backtracking line search
        alpha = min(alpha_mem * 1.5, 5.0)
        best_val = M
        best_alpha = 0.0

        for _ in range(20):
            f_trial = f - alpha * sg
            f_trial = project_simplex(f_trial)
            val = fast_convolve(f_trial).max()
            if val < best_val - 1e-15:
                best_val = val
                best_alpha = alpha
                alpha_mem = alpha
                break
            alpha *= 0.7

        if best_alpha > 0:
            f_new = f - best_alpha * sg
            f_new = project_simplex(f_new)
            f = f_new
            obj = fast_convolve(f).max()

            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                temp *= 0.995
            else:
                no_improve += 1
        else:
            no_improve += 1
            alpha_mem *= 0.5

        # Adapt temperature
        if no_improve > 100:
            temp = min(temp * 1.3, 0.1)

        # Restart if stuck
        if no_improve > 400:
            rng = np.random.default_rng(it * 31 + 17)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            no_improve = 0
            temp = temp_init
            alpha_mem = 0.1

        if verbose and it % 200 == 0:
            print(f"    GD iter {it}: obj={best_obj:.10f}, temp={temp:.6f}, alpha={alpha_mem:.4f}")

    return best_f, best_obj


def nesterov_subgrad(f0, time_limit, verbose=False):
    """
    Nesterov accelerated projected subgradient for minimizing max_k (a*a)[k].
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolive(f) if False else fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    t_k = 1.0
    f_prev = f.copy()
    no_improve = 0
    it = 0
    temp = 0.005

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        momentum = (t_k - 1) / (t_k + 1)
        y = f + momentum * (f - f_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = best_f.copy()
        else:
            y = y / sy

        conv_y = fast_convolve(y)
        M_y = conv_y.max()

        # Subgradient from weighted peaks
        diff = conv_y - M_y
        exp_w = np.exp(diff / temp)
        wsum = exp_w.sum()
        if wsum < 1e-30:
            exp_w = np.zeros(len(conv_y))
            exp_w[int(np.argmax(conv_y))] = 1.0
        else:
            exp_w /= wsum

        sg = np.zeros(n)
        for m in range(n):
            sg[m] = 2.0 * np.dot(y, exp_w[m:m + n])

        # Step size
        lr = 2.0 / (t_k + 1)
        f_prev = f.copy()
        f_new = y - lr * sg
        f_new = project_simplex(f_new)

        conv = fast_convolve(f_new)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0

        if no_improve > 500:
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            f_prev = f.copy()
            t_k = 1.0
            no_improve = 0
            temp *= 1.5

        f = f_new

        if verbose and it % 500 == 0:
            print(f"    Nesterov iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate, find the best value by
    trying multipliers, keeping others proportionally scaled.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 3.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial /= f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def double_length(f):
    """Create a sequence of double length by mirroring and blending."""
    n = len(f)
    # Create a smooth extension: blend reversed with new
    rev = f[::-1]
    # Cross-fade
    alpha = np.linspace(1.0, 0.0, n)
    beta = 1.0 - alpha
    new_seq = np.concatenate([f, f * 0.5 + rev * 0.5])
    # Smooth the join
    half = n // 4
    join = n - half // 2
    for i in range(half):
        w = i / half
        new_seq[n + i] = (1 - w) * new_seq[n + i] + w * f[n - 1 - i] * 0.5
    new_seq = np.maximum(new_seq, 0)
    ss = new_seq.sum()
    if ss > 1e-15:
