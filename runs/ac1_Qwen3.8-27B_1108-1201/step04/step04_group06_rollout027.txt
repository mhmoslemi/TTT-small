import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build matrix C such that (C @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    m = 2 * n - 1
    C = np.zeros((m, n))
    for j in range(n):
        C[j:j+n, j] = a
    return C

def solve_lp_direction(a, M, C):
    """Solve: max sum(g) s.t. C@g <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A_ub = C  # shape (2n-1, n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 5})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            if g.sum() > 1e-10:
                return g
    except:
        pass
    return None

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s
