        v = momentum * v + (1 - momentum) * grad
        f_new = project_simplex(f - lr * v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr = max(lr * 0.88, lr_min)

        T = max(T * 0.996, T_min)

        if no_improve > 400:
            rng = np.random.default_rng((it * 331 + 17) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            lr = max(lr * 1.3, 0.002)
            T = max(T_init * 0.2, T_min)
            no_improve = 0

    return best_f, best_obj


def subgradient_polish(f0, time_limit, lr=0.001, top_k_frac=0.06, temp=0.00003,
                       momentum=0.7):
    """Subgradient polish with peak weighting and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    top_k = max(3, int(n * top_k_frac))
    v = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        sg = peak_weighted_subgradient(f, conv, top_k, temp)
        sg = sg - np.mean(sg)

        v = momentum * v + (1 - momentum) * sg
        f_new = project_simplex(f - lr * v)

        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1

        if it % 60 == 0:
            lr *= 0.995
            temp *= 1.005

        if no_improve > 200:
            rng = np.random.default_rng((it * 733 + 41) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0006)
            v = np.zeros(n)
            lr = 0.0006
            no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    # U-shape (edge-concentrated)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    # Two humps
    h1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.12 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.8 * n) / (0.12 * n)) ** 2)
    inits.append((h1 + h2) / (h1 + h2).sum())

    # Three humps
    h3a = np.exp(-0.5 * ((x - 0.15 * n) / (0.1 * n)) ** 2)
    h3b = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    h3c = np.exp(-0.5 * ((x - 0.85 * n) / (0.1 * n)) ** 2)
    inits.append((h3a + h3b + h3c) / (h3a + h3b + h3c).sum())

    # Mirror-symmetric random
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.5)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    if n % 2 == 1:
        sym[half_n] = half[-1]
    inits.append(sym / sym.sum())

    # Sparse-ish (Zipf-like)
    zipf_w = 1.0 / (np.arange(n) + 1) ** 0.7
    inits.append(zipf_w / zipf_w.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Perturbed parent
        pp = px + rng.standard_normal(n) * 0.02
        pp = np.maximum(pp, 0)
        pp_s = pp.sum()
        if pp_s > 1e-10:
            inits.append(pp / pp_s)

    return inits


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        score = evaluate_sequence(a_list)
        return score
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search with:
    - Dual-aware LP Frank-Wolfe
    - Peak-priority subgradient with momentum
    - Coarse-to-fine length exploration
    - Adaptive restarts
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = evaluate_safe(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_safe(parent):.8f}")
    except Exception:
        pass

    # Evaluate parent
    if parent is not None:
        try_seq(parent, "parent_raw")

    base_n = len(parent) if parent is not None else 384

    # === Phase 1: Quick FW on parent and nearby lengths (250s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t1_start = time.monotonic()
    phase1_budget = 250.0

    # Optimize parent directly
    rem = phase1_budget - (time.monotonic() - t1_start)
    if rem > 50 and parent is not None:
        print(f"  FW on parent n={base_n}, t={rem:.0f}s")
        a_opt, _ = fw_optimize(parent, time_limit=min(120.0, rem - 10))
        try_seq(a_opt, "p1_parent_fw")

    # Try a few nearby lengths
    lengths_to_try = []
    for delta in [4, -4, 8, -8, 16, -16, 32, -32]:
        cn = base_n + delta
        if 200 <= cn <= 600:
            lengths_to_try.append(cn)

    for n in lengths_to_try[:4]:
        rem = phase1_budget - (time.monotonic() - t1_start)
        if rem < 30:
            break
        inits = generate_inits(n, parent, rng)
        best_init = None
        best_obj = float('inf')
        for fi in inits[:4]:
            co = fast_convolve(fi).max()
            if co < best_obj:
                best_obj = co
                best_init = fi
        if best_init is not None:
            t_alloc = min(40.0, rem - 10)
            print(f"  FW n={n}, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(best_init, time_limit=t_alloc)
            try_seq(a_opt, f"p1_n{n}")

    print(f"  Phase 1 done. Best: {best_score:.8f}")

    # === Phase 2: Deep optimization (600s) ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    t2_start = time.monotonic()
    phase2_budget = 600.0

    if best_seq is not None:
        f_work = best_seq.copy()
        f_work = f_work / f_work.sum()
        n_work = len(f_work)

        # Cycle: FW -> Smoothed GD -> FW -> Subgrad -> FW
        methods = [
            ("fw", 100),
            ("sgd", 80),
            ("fw", 80),
            ("sg", 60),
            ("fw", 80),
            ("sgd", 60),
        ]

        for mname, t_base in methods:
            rem = phase2_budget - (time.monotonic() - t2_start)
            if rem < 30:
                break
            t_alloc = min(t_base, rem - 20)
            if t_alloc < 20:
                break

            if mname == "fw":
                print(f"  FW (t={t_alloc:.0f}s)")
                a_opt, _ = fw_optimize(f_work, time_limit=t_alloc)
            elif mname == "sgd":
                T0 = max(0.001, 0.004 * fast_convolve(f_work).max())
                print(f"  Smoothed GD (t={t_alloc:.0f}s, T={T0:.5f})")
                a_opt, _ = smoothed_grad_descent(f_work, time_limit=t_alloc,
                                                  T_init=T0, T_min=0.00003)
            else:  # sg
                print(f"  Subgrad (t={t_alloc:.0f}s)")
                a_opt, _ = subgradient_polish(f_work, time_limit=t_alloc,
                                               lr=0.0008, top_k_frac=0.05,
                                               temp=0.00002)

            try_seq(a_opt, f"p2_{mname}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()
                n_work = len(f_work)

    print(f"  Phase 2 done. Best: {best_score:.8f}")

    # === Phase 3: Length exploration (350s) ===
    print(f"\n=== Phase 3: Length Exploration ===")
    t3_start = time.monotonic()
    phase3_budget = 350.0

    if best_seq is not None:
        n_best = len(best_seq)
        f_best = best_seq / best_seq.sum()

        # Try a range of lengths
        length_candidates = []
        for delta in range(-40, 41, 4):
            cn = n_best + delta
            if 256 <= cn <= 550:
                length_candidates.append(cn)

        # Sort by distance from current best
        length_candidates.sort(key=lambda x: abs(x - n_best))

        for n in length_candidates[:8]:
            rem = phase3_budget - (time.monotonic() - t3_start)
            if rem < 40:
                break
            if n == n_best:
                continue

            # Initialize by interpolation
            px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_best), f_best)
            px = project_simplex(px)

            # Quick FW
            t_alloc = min(40.0, rem - 20)
            a_opt, _ = fw_optimize(px, time_limit=t_alloc)
            try_seq(a_opt, f"p3_n{n}_fw")

            # If improved, do a bit more
            rem2 = phase3_budget - (time.monotonic() - t3_start)
            if rem2 > 30:
                a_opt2, _ = smoothed_grad_descent(a_opt, time_limit=min(25.0, rem2 - 15),
                                                   T_init=0.002, T_min=0.0001)
                try_seq(a_opt2, f"p3_n{n}_sgd")

    print(f"  Phase 3 done. Best: {best_score:.8f}")

    # === Phase 4: Fine polish (350s) ===
    print(f"\n=== Phase 4: Fine Polish ===")
    t4_start = time.monotonic()
    phase4_budget = 350.0

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()
        n_work = len(f_work)

        # Multiple rounds of increasingly fine optimization
        configs = [
            {"method": "fw", "t": 80},
            {"method": "sg", "lr": 0.0005, "top_k_frac": 0.04, "temp": 0.000015, "t": 50},
            {"method": "fw", "t": 60},
            {"method": "sg", "lr": 0.0003, "top_k_frac": 0.03, "temp": 0.00001, "t": 40},
            {"method": "sgd", "T_init": 0.0005, "T_min": 0.00002, "t": 40},
            {"method": "fw", "t": 40},
            {"method": "sg", "lr": 0.0002, "top_k_frac": 0.02, "temp": 0.000005, "t": 30},
        ]

        for i, cfg in enumerate(configs):
            rem = phase4_budget - (time.monotonic() - t4_start)
            if rem < 25:
                break
            t_alloc = min(cfg["t"], rem - 10)
            if t_alloc < 15:
                break

            m = cfg["method"]
            if m == "fw":
                a_opt, _ = fw_optimize(f_work, time_limit=t_alloc)
            elif m == "sg":
                a_opt, _ = subgradient_polish(f_work, time_limit=t_alloc,
                                               lr=cfg["lr"], top_k_frac=cfg["top_k_frac"],
                                               temp=cfg["temp"])
            else:
                a_opt, _ = smoothed_grad_descent(f_work, time_limit=t_alloc,
                                                  T_init=cfg["T_init"], T_min=cfg["T_min"])
            try_seq(a_opt, f"p4_{m}_{i}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

    print(f"  Phase 4 done. Best: {best_score:.8f}")

    # === Phase 5: Final tricks (remaining) ===
    print(f"\n=== Phase 5: Final Tricks ===")

    if best_seq is not None:
        n_best = len(best_seq)
        f_work = best_seq / best_seq.sum()

        # Mirror-symmetric projection + optimize
        half_n = n_best // 2
        symmetric = np.zeros(n_best)
        symmetric[:half_n] = f_work[:half_n]
        symmetric[half_n:] = f_work[:half_n][::-1]
        if n_best % 2 == 1:
            symmetric[half_n] = f_work[half_n]
        symmetric = project_simplex(symmetric)
        if np.max(np.abs(symmetric - f_work)) > 1e-8:
            a_opt, _ = fw_optimize(symmetric, time_limit=20)
            try_seq(a_opt, "p5_sym")

        # Tiny perturbations
        for trial in range(5):
            rem = total_deadline - time.monotonic()
            if rem < 15:
                break
            noise = rng.standard_normal(n_best) * 0.0003
            f_pert = project_simplex(f_work + noise)
            a_opt, _ = subgradient_polish(f_pert, time_limit=min(12, rem - 5),
                                           lr=0.0003, top_k_frac=0.02, temp=0.000008)
            try_seq(a_opt, f"p5_pert{trial}")

        # One final tiny FW
        rem = total_deadline - time.monotonic()
        if rem > 25:
            a_opt, _ = fw_optimize(f_work, time_limit=min(25, rem - 5))
            try_seq(a_opt, "p5_final_fw")

    # Fallback
    if best_seq is None:
        print("  WARNING: No valid sequence found, using uniform fallback")
        best_seq = np.ones(base_n) / base_n
        best_score = evaluate_safe(best_seq)

    # Final verification
    fs = evaluate_safe(best_seq)
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = evaluate_safe(best_seq)

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
