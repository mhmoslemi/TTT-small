"""
Multi-scale peak-suppression optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive peak-weighted gradient descent with periodic LP-FW direction finding,
multi-temperature mixing, exact-like line search, and multi-length exploration.
Key improvements over previous: multi-temperature gradient, smarter LP constraint
handling, cubic line search, and more aggressive exploration of lengths.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def conv_gradient_peaks(conv, a, peaks, weights):
    """Compute gradient of weighted sum of selected convolution peaks.
    
    d/d(a[i]) conv(a,a)[k] = 2*a[k-i] if 0 <= k-i <= n-1, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    for i in range(len(peaks)):
        wi = weights[i]
        if wi < 1e-14:
            continue
        k = int(peaks[i])
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo > i_hi:
            continue
        seg = a[max(0, k - n + 1): min(n - 1, k) + 1][::-1]
        grad[i_lo:i_hi + 1] += 2.0 * wi * seg
    return grad


def multi_temp_gradient(conv, a, T_list, top_k_list):
    """Compute gradient using a mixture of multiple temperatures."""
    n = len(a)
    total = np.zeros(n)
    for T, top_k in zip(T_list, top_k_list):
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / max(T, 1e-12)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        total += conv_gradient_peaks(conv, a, peaks, w)
    return total


def solve_lp_direction(f, M):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def line_search_cubic(f, g, f0, g0, best_val):
    """Cubic interpolation line search along f -> g on the simplex."""
    # We want to minimize max(conv((1-a)f + a*g, (1-a)f + a*g)) for a in [0,1]
    # conv is quadratic in a, so each entry is quadratic, and the max is piecewise quadratic.
    # Use a sampling + refinement approach.
    
    def obj_at_alpha(alpha):
        ft = (1.0 - alpha) * f0 + alpha * g0
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            return 1e10
        ft = ft / st
        return fast_convolve(ft).max()
    
    # Coarse grid
    alphas = np.linspace(0, 1, 21)
    vals = [obj_at_alpha(a) for a in alphas]
    max_vals = np.array(vals)
    best_idx = np.argmin(max_vals)
    best_a = alphas[best_idx]
    best_v = max_vals[best_idx]
    
    # Refine around best
    lo = alphas[max(0, best_idx - 1)]
    hi = alphas[min(len(alphas) - 1, best_idx + 1)]
    
    # Cubic interpolation using 3 points
    a1 = max(0.0, best_a - (hi - lo) * 0.3)
    a2 = best_a
    a3 = min(1.0, best_a + (hi - lo) * 0.3)
    v1 = obj_at_alpha(a1)
    v2 = best_v
    v3 = obj_at_alpha(a3)
    
    # Steffen's method for cubic interpolation
    if abs(v3 - v1) > 1e-15:
        d1 = (v2 - v1) / (a2 - a1 + 1e-30)
        d2 = (v3 - v2) / (a3 - a2 + 1e-30)
        b = (d1 * d2) / (d1 + d2) if abs(d1 + d2) > 1e-30 else (d1 + d2) / 2.0
        a_new = a2 - b * (a2 - a1)
    else:
        a_new = (a1 + a3) / 2.0
    
    a_new = max(0.0, min(1.0, a_new))
    v_new = obj_at_alpha(a_new)
    
    if v_new < best_v:
        best_a = a_new
        best_v = v_new
    
    # One more refinement

            a_try = best_a * frac
            a_try = max(0.0, min(1.0, a_try))
            v_try = obj_at_alpha(a_try)
            if v_try < best_v:
                best_a = a_try
                best_v = v_try
    
    return best_a, best_v

            rng = np.random.default_rng((it * 7919 + n * 104729) % (2**31))
            scale = 0.003 + 0.003 * min(no_improve // 20, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def multi_temp_grad_descent(f0, time_limit, verbose=False):
    """Multi-temperature gradient descent on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 0.005
    lr_min = 0.00005

    # Multi-temperature schedule
    base_T = 0.003 * M

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it > 20000:
            break

        conv = fast_convolve(f)
        M_now = conv.max()
        
        # Adaptive temperatures
        T1 = max(base_T * (0.5 + 0.5 * M_now / M), 1e-6)
        T2 = T1 * 5.0
        T3 = T1 * 20.0
        top_k1 = max(3, int(n * 0.01))
        top_k2 = max(5, int(n * 0.05))
        top_k3 = max(10, int(n * 0.15))

        grad = multi_temp_gradient(conv, f, [T1, T2, T3], [top_k1, top_k2, top_k3])
        grad = grad / (len(T1) * 3)  # normalize by number of temperature levels
        
        # Project gradient to tangent space
        grad = grad - np.mean(grad)

        f_new = project_simplex(f - lr * grad)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 200 == 0:
                print(f"    MTGD iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                lr = max(lr * 0.9, lr_min)

        # Periodic aggressive step
        if it % 500 == 0 and no_improve > 100:
            # Take a step toward the LP direction
            g = solve_lp_direction(f, M_now * 1.01)
            if g is not None:
                g_sum = g.sum()
                if g_sum > 1e-15:
                    g = g / g_sum
                    f_try = project_simplex(0.7 * f + 0.3 * g)
                    conv_try = fast_convolve(f_try)
                    if conv_try.max() < best_obj:
                        f = f_try
                        best_obj = conv_try.max()
                        best_f = f.copy()
                        no_improve = 0

        # Periodic restart
        if no_improve > 500:
            rng = np.random.default_rng((it * 331 + 17) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            lr = max(lr * 2.0, 0.002)
            no_improve = 0

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.001, top_k_frac=0.05, temp=0.00003):
    """Subgradient polish focusing on top peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    top_k = max(3, int(n * top_k_frac))

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it > 10000:
            break

        conv = fast_convolve(f)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / max(temp, 1e-12)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = conv_gradient_peaks(conv, f, peaks, w)
        sg = sg - np.mean(sg)

        f = project_simplex(f - lr * sg)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-13:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 100 == 0:
            lr *= 0.995
            temp *= 1.01

        if no_improve > 200:
            rng = np.random.default_rng((it * 733 + 41) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            lr = max(lr, 0.0003)
            no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    # U-shape
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Two humps
    h1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.08 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.8 * n) / (0.08 * n)) ** 2)
    inits.append((h1 + h2) / (h1 + h2).sum())

    # Mirror symmetric
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.8)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    if n % 2 == 1:
        sym[half_n] = half[-1]
    inits.append(sym / sym.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search with multi-temperature gradient,
    LP-FW direction finding, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # === Phase 1: Exploration (250s) ===
    print("=== Phase 1: Exploration ===")
    t_p1 = time.monotonic()
    p1_budget = 250.0

    all_candidates = []

    if parent is not None:
        n_p = len(parent)
        t_alloc = min(80.0, p1_budget - 60.0)
        if t_alloc > 20:
            print(f"  FW-LP parent n={n_p}, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(parent, time_limit=t_alloc)
            all_candidates.append(a_opt)
            try_seq(a_opt, "parent_fw")

    # Try other lengths
    base_n = len(parent) if parent is not None else 384
    lengths = [base_n]
    for d in [16, -16, 32, -32, 48, 64, -48, 96, 128, -64]:
        cn = base_n + d
        if 192 <= cn <= 768 and cn not in lengths:
            lengths.append(cn)

    # Sort by distance from base
    other_lengths = sorted([l for l in lengths if l != base_n], key=lambda l: abs(l - base_n))

    for n in other_lengths[:4]:
        rem = p1_budget - (time.monotonic() - t_p1)
        if rem < 40:
            break
        t_alloc = min(40.0, rem - 30.0)
        if t_alloc < 15:
            break

        inits = generate_inits(n, parent, rng)
        # Pick best init quickly
        best_init = inits[0]
        best_init_obj = float('inf')
        for fi in inits[:4]:
            ci = fast_convolve(fi)
            oi = ci.max()
            if oi < best_init_obj:
                best_init_obj = oi
                best_init = fi

        print(f"  FW-LP n={n}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_init, time_limit=t_alloc)
        all_candidates.append(a_opt)
        try_seq(a_opt, f"explore_n{n}")

    print(f"  Phase 1 done: best={best_score:.6f}")

    # === Phase 2: Deep Optimization (1100s) ===
    print("\n=== Phase 2: Deep Optimization ===")
    t_p2 = time.monotonic()
    p2_budget = 1100.0

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()
        n_best = len(f_work)

        n_cycles = 8
        t_per_cycle = p2_budget / n_cycles

        for cycle in range(n_cycles):
            rem = p2_budget - (time.monotonic() - t_p2)
            if rem < 30:
                break

            # Sub-phase A: LP-FW
            t_fw = min(t_per_cycle * 0.45, rem * 0.4)
            if t_fw > 20:
                print(f"  Cycle {cycle}: FW (t={t_fw:.0f}s)")
                a_opt, _ = fw_lp_optimize(f_work, time_limit=t_fw)
                try_seq(a_opt, f"p2_fw_{cycle}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()
                    n_best = len(f_work)

            # Sub-phase B: Multi-temp gradient
            rem = p2_budget - (time.monotonic() - t_p2)
            t_mt = min(t_per_cycle * 0.4, rem * 0.4)
            if t_mt > 15:
                print(f"  Cycle {cycle}: MTGD (t={t_mt:.0f}s)")
                a_opt, _ = multi_temp_grad_descent(f_work, time_limit=t_mt)
                try_seq(a_opt, f"p2_mt_{cycle}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()
                    n_best = len(f_work)

            # Sub-phase C: Quick SG polish
            rem = p2_budget - (time.monotonic() - t_p2)
            t_sg = min(t_per_cycle * 0.15, rem * 0.3)
            if t_sg > 10:
                a_opt, _ = sg_polish(f_work, time_limit=t_sg,
                                      lr=0.0008, top_k_frac=0.04, temp=0.00005)
                try_seq(a_opt, f"p2_sg_{cycle}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()
                    n_best = len(f_work)

    print(f"  Phase 2 done: best={best_score:.6f}")

    # === Phase 3: Fine Polish (300s) ===
    print("\n=== Phase 3: Fine Polish ===")
    t_p3 = time.monotonic()
    p3_budget = 300.0

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()

        # Progressively finer SG polish
        configs = [
            {"lr": 0.0006, "top_k_frac": 0.03, "temp": 0.00003},
            {"lr": 0.0003, "top_k_frac": 0.02, "temp": 0.00002},
            {"lr": 0.0002, "top_k_frac": 0.015, "temp": 0.00001},
            {"lr": 0.0001, "top_k_frac": 0.01, "temp": 0.000005},
        ]

        for i, cfg in enumerate(configs):
            rem = p3_budget - (time.monotonic() - t_p3)
            if rem < 25:
                break
            t_alloc = min(60.0, rem - 15.0)
            print(f"  SG round {i} (t={t_alloc:.0f}s)")
            a_opt, _ = sg_polish(f_work, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"p3_sg_{i}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Fine multi-temp gradient
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 30:
            t_alloc = min(80.0, rem - 10.0)
            print(f"  Fine MTGD (t={t_alloc:.0f}s)")
            a_opt, _ = multi_temp_grad_descent(f_work, time_limit=t_alloc)
            try_seq(a_opt, "p3_fine_mt")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Final LP-FW
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 25:
            t_alloc = min(50.0, rem - 10.0)
            print(f"  Final FW (t={t_alloc:.0f}s)")
            a_opt, _ = fw_lp_optimize(f_work, time_limit=t_alloc)
            try_seq(a_opt, "p3_final_fw")

    print(f"  Phase 3 done: best={best_score:.6f}")

    # === Phase 4: Final Tricks (100s) ===
    print("\n=== Phase 4: Final Tricks ===")

    if best_seq is not None:
        n_best = len(best_seq)
        f_work = best_seq / best_seq.sum()

        # Try mirror-symmetric version
        mirrored = 0.5 * f_work + 0.5 * f_work[::-1]
        mirrored = project_simplex(mirrored)
        if np.max(np.abs(mirrored - f_work)) > 1e-7:
            a_opt, _ = sg_polish(mirrored, time_limit=15,
                                  lr=0.0004, top_k_frac=0.02, temp=0.00001)
            try_seq(a_opt, "p4_mirror")

        # Tiny perturbations
        for attempt in range(4):
            if time.monotonic() > deadline - 15:
                break
            noise = rng.standard_normal(n_best) * 0.0004
            f_pert = project_simplex(f_work + noise)
            a_opt, _ = sg_polish(f_pert, time_limit=8,
                                  lr=0.0002, top_k_frac=0.015, temp=0.000008)
            try_seq(a_opt, f"p4_pert{attempt}")

        # One more short FW
        if time.monotonic() < deadline - 25:
            a_opt, _ = fw_lp_optimize(f_work, time_limit=20)
            try_seq(a_opt, "p4_fw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
