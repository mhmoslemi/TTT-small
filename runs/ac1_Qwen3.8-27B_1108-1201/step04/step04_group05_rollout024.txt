"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-method approach:
1. LP-based Frank-Wolfe with active-set reduction and warm starting
2. Exact subgradient mirror descent with adaptive peak weighting

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def exact_subgradient(a, peaks, weights):
    """Compute exact subgradient of max_k conv(a,a)[k] at point a.
    
    grad[j] = sum_k weights[k] * 2*a[k-j] for valid k-j in [0, n-1]
    """
    n = len(a)
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-15:
            continue
        k = int(peaks[i])
        w = weights[i]
        # For gradient at position j: 2*a[k-j] if 0 <= k-j < n
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # sg[j] += 2 * w * a[k-j] for j in [js, je]
        # a[k-j] for j in [js, je] is a[k-je] ... a[k-js]
        # i.e., a[k-je], a[k-je+1], ..., a[k-js]
        idx = np.arange(k - je, k - js + 1)
        sg[js:je + 1] += 2.0 * w * a[idx]
    return sg


def solve_lp_reduced(f, M, active_frac=0.3):
    """Solve LP with only active constraints for speed.
    
    max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0
    """
    n = len(f)
    conv_f = fast_convolve(f)
    # Select active constraints: those where conv_f >= M * (1 - active_frac)
    threshold = M * (1.0 - active_frac)
    active_mask = conv_f >= threshold
    # Always include the maximum constraint
    max_idx = np.argmax(conv_f)
    active_mask[max_idx] = True
    
    active_k = np.where(active_mask)[0]
    n_active = len(active_k)
    
    if n_active > 2 * n:
        # If too many active, fall back to top constraints + max
        top_count = min(2 * n, n_active)
        top_indices = np.argsort(conv_f)[::-1][:top_count]
        active_k = top_indices
        n_active = top_count
    
    c = -np.ones(n)
    A_ub = np.zeros((n_active, n))
    for lua_i, k_idx in enumerate(active_k):
        # (f*g)[k_idx] = sum_j f[j] * g[k_idx - j]
        # For g[m]: coefficient is f[k_idx - m] if 0 <= k_idx - m < n
        m_idx = np.arange(n)
        f_idx = k_idx - m_idx
        valid = (f_idx >= 0) & (f_idx < n)
        A_ub[lua_i, valid] = f[f_idx[valid]]
    b_ub = np.full(n_active, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    
    # Fallback: full LP
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

                shave = best_f.copy()
                for k in peaks:
                    js = max(0, k - n + 1)
                    je = min(n - 1, k)
                    # Reduce proportionally to contribution
                    contrib = best_f[js:je + 1] * best_f[k - je:k - js + 1][::-1]
                    total_c = contrib.sum()
                    if total_c > 1e-15:
                        reduction = 0.05 * best_f * (contrib / total_c)
                        shave -= reduction
                shave = np.maximum(shave, 0)
                ss = shave.sum()
                if ss > 1e-15:
                    f = shave / ss
                else:
                    f = project_simplex(best_f + rng_local.standard_normal(n) * 0.01)
            else:
                f = project_simplex(rng_local.dirichlet(np.ones(n) * 0.6))

            conv = fast_convolve(f)
            M = conv.max()
            m_mult = 1.0
            no_improve = 0

    return best_f, best_obj


def mirror_descent_optimize(f0, time_limit, lr=0.005, momentum=0.9, top_k=20, temp=1e-5):
    """Mirror descent with exact subgradients and entropic regularizer."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    # Add small epsilon to avoid log(0)
    eps = 1e-12
    f_safe = f + eps

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    rng_local = np.random.default_rng(9999)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        conv_max = conv.max()

        # Select peaks with softmax weighting
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute exact subgradient
        sg = exact_subgradient(f_safe, peaks, w)

        # Momentum update
        v = momentum * v - cur_lr * sg

        # Mirror descent update: f_new[j] = f[j] * exp(v[j])
        # Normalized to sum to 1
        log_f = np.log(f_safe) + v
        # Subtract max for numerical stability
        log_f -= log_f.max()
        f_new = np.exp(log_f)
        f_new = np.maximum(f_new, 0.0)
        sn = f_new.sum()
        if sn < 1e-15:
            f = project_simplex(f - cur_lr * sg)
        else:
            f = f_new / sn

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate
        cur_lr *= 0.9998
        cur_temp *= 1.003

        # Restart
        if no_improve > 400:
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.001)
            no_improve = 0

    return best_f, best_obj


def peak_shaving_optimize(f0, time_limit, n_shaves=5):
    """Targeted peak shaving: iteratively reduce the worst convolution peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    rng_local = np.random.default_rng(5555)

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()

        # Find top peaks
        n_peaks = min(n_shaves, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        improved = False
        for k in peak_indices:
            if conv[k] < M * 0.99:
                continue
            # Find which j values contribute most to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            contrib = f[js:je + 1] * f[k - je:k - js + 1][::-1]
            total_c = contrib.sum()
            if total_c < 1e-15:
                continue

            # Try reducing the largest contributors
            for frac in [0.02, 0.05, 0.1, 0.2]:
                f_new = f.copy()
                reduction = frac * total_c / (je - js + 1)
                # Reduce proportionally to contribution
                reduce_amt = frac * f[js:je + 1] * (contrib / total_c)
                f_new[js:je + 1] -= reduce_amt
                f_new = np.maximum(f_new, 0.0)
                sn = f_new.sum()
                if sn < 1e-15:
                    continue
                f_new = f_new / sn
                val = fast_convolve(f_new).max()
                if val < best_obj - 1e-15:
                    f = f_new
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    no_improve = 0
                    break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                # Perturb and continue
                scale = 0.005 * (1 + no_improve / 50)
                f = project_simplex(best_f + rng_local.standard_normal(n) * scale)
                no_improve = 0

        if M < 1.0001 and best_obj < 1.001:
            break

    return best_f, best_obj


def gradient_flow_optimize(f0, time_limit, lr=0.002, n_peaks=15):
    """Projected gradient descent with exact subgradients and adaptive step."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    v = np.zeros(n)
    momentum = 0.9

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    rng_local = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()

        # Get peaks
        pk = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -pk)[-pk:]
        pv = conv[peaks]
        # Uniform weights over peaks (or softmax)
        w = np.ones(len(peaks)) / len(peaks)

        # Exact subgradient
        sg = exact_subgradient(f, peaks, w)

        # Momentum
        v = momentum * v - cur_lr * sg

        # Projected gradient step
        f_new = project_simplex(f + v)
        val = fast_convolve(f_new).max()

        if val < best_obj - 1e-15:
            f = f_new
            best_obj = val
            best_f = f.copy()
            no_improve = 0
        else:
            # Try half step
            v_half = momentum * v - 0.5 * cur_lr * sg
            f_half = project_simplex(f + v_half)
            val_half = fast_convolve(f_half).max()
            no_improve += 1
            if val_half < best_obj - 1e-15:
                f = f_half
                best_obj = val_half
                best_f = f.copy()
                no_improve = 0
                v = v_half
            else:
                cur_lr *= 0.95
                if no_improve > 300:
                    f = project_simplex(best_f + rng_local.standard_normal(n) * 0.002)
                    v = np.zeros(n)
                    cur_lr = max(lr * 0.5, 0.0003)
                    no_improve = 0

        cur_lr = max(cur_lr * 0.9999, 0.0001)

    return best_f, best_obj


def try_sequence(a, label, state):
    """Evaluate a sequence and update state if it's better."""
    a = np.maximum(a, 0.0)
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return score


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.4))

    # Edge concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 1.5
    inits.append(edge_w / edge_w.sum())

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    cos_t = np.cos(np.pi * x / (2.0 * max(n - 1, 1)))
    cos_t = np.maximum(cos_t, 0.01)
    inits.append(cos_t / cos_t.sum())

    # Two peaks
    peak = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)) +
            np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)))
    inits.append(peak / peak.sum())

    # Sine window
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())

    # Power law
    pl = (x + 1.0) ** (-0.4)
    inits.append(pl / pl.sum())

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Also try parent with perturbation
        px2 = px + rng.standard_normal(n) * 0.01
        px2 = np.maximum(px2, 0)
        ps2 = px2.sum()
        if ps2 > 1e-10:
            inits.append(px2 / ps2)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
            s = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(s):
                print(f"  Parent score: {s:.6f}")
                state['best_seq'] = parent.copy()
                state['best_score'] = s
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        state['best_seq'] = parent.copy()
        state['best_score'] = float(evaluate_sequence(parent.tolist()))

    # === Phase 1: Quick multi-length exploration (25% time) ===
    print("\n=== Phase 1: Multi-length Exploration ===")
    phase1_end = time.monotonic() + (float(budget_s) * 0.25)
    p_len = len(parent)

    lengths = sorted(set([256, 320, 384, 448, 512, 640, 768]))
    # Put parent length first
    if p_len in lengths:
        lengths.remove(p_len)
    lengths.insert(0, p_len)

    for n in lengths:
        if time.monotonic() > phase1_end - 40:
            break
        inits = generate_inits(n, parent, rng)
        n_try = min(2, len(inits))

        t_each = min(t_each, 60)

        for i in range(n_try):
            if time.monotonic() > phase1_end - 20:
                break
            t_alloc = min(t_each, phase1_end - time.monotonic() - 15)
            if t_alloc < 15:
                break
            f_init = inits[i]
            a_opt, obj = fw_lp_optimize(f_init, time_limit=t_alloc, active_frac=0.2)
            try_sequence(a_opt, f"p1_n{n}_i{i}", state)

    print(f"  Phase 1 best: {state['best_score']:.6f}")

    # === Phase 2: Deep LP-FW optimization (35% time) ===
    print(f"\n=== Phase 2: Deep LP-FW (n={len(state['best_seq'])}) ===")
    phase2_end = time.monotonic() + (float(budget_s) * 0.35)
    n_best = len(state['best_seq'])

    # Main deep run
    rem = phase2_end - time.monotonic() - 20
    if rem > 30:
        a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=min(rem, 200),
                                     verbose=True, active_frac=0.15)
        try_sequence(a_opt, "p2_deep", state)

    # Perturbation-based restarts
    perturb_scales = [0.002, 0.005, 0.01, 0.02, 0.05]
    for ps in perturb_scales:
        rem = phase2_end - time.monotonic() - 20
        if rem < 20:
            break
        t_alloc = min(rem // len(perturb_scales), 80)
        if t_alloc < 15:
            break
        n_cur = len(state['best_seq'])
        noise = rng.standard_normal(n_cur) * ps
        f_init = project_simplex(state['best_seq'] + noise)
        a_opt, obj = fw_lp_optimize(f_init, time_limit=t_alloc, active_frac=0.2)
        try_sequence(a_opt, f"p2_pert{ps}", state)

    # Mirror blend
    rem = phase2_end - time.monotonic() - 15
    if rem > 20:
        n_cur = len(state['best_seq'])
        f_mirror = project_simplex(0.7 * state['best_seq'] + 0.3 * state['best_seq'][::-1])
        a_opt, obj = fw_lp_optimize(f_mirror, time_limit=min(rem, 60), active_frac=0.25)
        try_sequence(a_opt, "p2_mirror", state)

    print(f"  Phase 2 best: {state['best_score']:.6f}")

    # === Phase 3: Alternative methods (25% time) ===
    print(f"\n=== Phase 3: Alternative Methods ===")

    rem = phase3_end - time.monotonic() - 30

    # === Phase 4: Final polishing (15% time) ===
    print(f"\n=== Phase 4: Final Polishing ===")
    phase4_end = deadline - 10
    n_final = len(state['best_seq'])

    rem = phase4_end - time.monotonic() - 20
    if rem > 15:
        a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=min(rem, 60),
                                     active_frac=0.1)
        try_sequence(a_opt, "p4_fwl", state)

    # Fine mirror descent
    rem = phase4_end - time.monotonic() - 15
    if rem > 15:
        a_opt, obj = mirror_descent_optimize(state['best_seq'], time_limit=min(rem, 50),
                                              lr=0.001, momentum=0.95,
                                              top_k=max(5, n_final // 15), temp=1e-5)
        try_sequence(a_opt, "p4_md_fine", state)

    # Fine gradient flow
    rem = phase4_end - time.monotonic() - 10
    if rem > 10:
        a_opt, obj = gradient_flow_optimize(state['best_seq'], time_limit=min(rem, 40),
                                             lr=0.0005, n_peaks=max(5, n_final // 20))
        try_sequence(a_opt, "p4_gf_fine", state)

    # Final LP-FW squeeze
    rem = phase4_end - time.monotonic() - 5

    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs
    else:
        # Emergency fallback
        state['best_seq'] = np.ones(256) / 256
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))

    print(f"\nFINAL SCORE: {state['best_score']:.6f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
