import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def subgradient_step(a, conv, top_k=50, temp=0.01):
    """
    Compute soft subgradient of max(conv(a,a)) at a.
    Returns the gradient vector.
    """
    n = len(a)
    conv_size = len(conv)  # 2n-1
    
    # Softmax weights on the convolution values
    # We want to weight the peaks heavily
    scaled = conv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(conv_size)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    # Gradient: for each lag k with weight w[k],
    # d(c[k])/da[i] = 2*a[k-i] if 0 <= k-i < n, else 0
    # So grad[i] = sum_k w[k] * 2*a[k-i] (where valid)
    grad = np.zeros(n)
    
    for k in range(conv_size):
        if w[k] < 1e-12:
            continue
        # a[k-i] is valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
        # and 0 <= i < n
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        idxs = np.arange(i_start, i_end + 1)
        # a[k - idxs] gives the values
        grad[idxs] += w[k] * 2.0 * a[k - idxs]
    
    return grad

def spectral_refine(a0, time_limit, n_iters=500):

    Spectral domain optimization: modify the spectrum to flatten autocorrelation,

    n = len(a)

    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute spectrum
        A = np.fft.fft(a, L)
        power = np.abs(A) ** 2

        # Compute autocorrelation from spectrum
        ac = np.fft.ifft(power).real[:L]

        # Find the peak
        peak_idx = np.argmax(ac)
        peak_val = ac[peak_idx]

        # Try to reduce the peak by modifying the spectrum
        # The peak at lag k* corresponds to spectral coherence at frequency 2*pi*k*/L
        # Strategy: add a "conjugate" cosine to cancel the peak
        freq = 2.0 * np.pi * peak_idx / L
        amplitude = 0.02 * (peak_val / (power.sum() / L + 1e-15))
        amplitude = min(amplitude, 0.1)

        # Modify: add -amplitude * cos(freq * f_index) to the power spectrum
        f_indices = np.arange(L)
        correction = -amplitude * np.cos(freq * f_indices)
        # Also add to maintain symmetry
        power_mod = power + correction
        power_mod = np.maximum(power_mod, 0)

        # Reconstruct: take square root (magnitude) and keep phase
        mag_new = np.sqrt(power_mod)
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)

        # Inverse FFT to get time domain
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:

            best_a = a_new.copy()
            a = a_new
        else:

                phase_noise = rng.standard_normal(L) * 0.05
                A_pert = A * np.exp(1j * phase_noise)
                a_pert = np.fft.ifft(A_pert).real[:n]

def lp_polish(a0, time_limit):
    """LP-based direction finding for final polishing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), 2.0 * len(a0) * fast_convolve(a0).max()
    a = a / s
    n = len(a)

    # Build convolution matrix
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Solve LP: max sum(g) s.t. C@g <= M*1.001, g >= 0
        c_vec = -np.ones(n)
        b_vec = np.full(2 * n - 1, M * 1.001)
        bounds = [(0, None)] * n

        try:
            result = linprog(c_vec, A_ub=C, b_ub=b_vec, bounds=bounds,
                             method='highs', options={'presolve': True,
                             'time_limit': max(1, time_limit - (time.monotonic() - t0) - 5)})
            if result.success:
                g = result.x
                g_sum = g.sum()
                if g_sum > 1.0 + 1e-6:
                    d = g - a
                    # Line search
                    for t in [0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
                        a_try = np.maximum(a + t * d, 0.0)
                        s_try = a_try.sum()
                        if s_try < 1e-15:
                            continue
                        a_try = a_try / s_try
                        conv_try = fast_convolve(a_try)
                        obj_try = 2.0 * n * conv_try.max()
                        if obj_try < best_obj - 1e-10:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            no_improve = 0
                            break
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        # Update M
        conv = fast_convolve(a)
        M = conv.max()
        # Rebuild C
        C = np.zeros((2 * n - 1, n))
        for j in range(n):
            C[j:j + n, j] = a

        iteration += 1
        if no_improve > 20:
            # Perturb
            rng = np.random.default_rng(iteration * 17 + 3)
            noise = rng.standard_normal(n) * 0.003
            a = np.maximum(best_a + noise, 0.0)
            ss = a.sum()
            if ss > 1e-15:
                a = a / ss
            no_improve = 0
            if iteration > 100:
                break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Sine-based
    s1 = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine1'] = s1 / s1.sum()

    s2 = (np.sin(2 * np.pi * x / n) ** 2 + 0.01)
    inits['sine2'] = s2 / s2.sum()

    # Cosine-based (non-negative, peaks at center)
    c1 = (1.0 + np.cos(2 * np.pi * x / n)) / 2.0
    c1 += 0.01
    inits['cos1'] = c1 / c1.sum()

    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # Alternating
    alt = np.ones(n)
    alt[::2] = 1.3
    alt[1::2] = 0.7
    inits['alternating'] = alt / alt.sum()

    # Mother wavelet-like
    mw = np.exp(-0.5 * ((x - n / 2) / (n / 6)) ** 2) * (1 - 2 * (x - n / 2) / (n / 3)) ** 2
    mw = np.maximum(mw, 0) + 0.01
    inits['mexican'] = mw / mw.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid fast subgradient + spectral + LP search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    # =========================================================
    # Phase 1: Fast subgradient scan across lengths and initializations
    # =========================================================
    print("=== Phase 1: Fast Subgradient Scan ===")
    t_start = time.monotonic()

    # Prioritize parent length, then nearby
    parent_n = len(parent) if parent is not None else 768
    lengths = sorted(set([384, 512, parent_n, 768, 1024, 1536]))
    # Reorder: parent first
    lengths.sort(key=lambda x: (x != parent_n, abs(x - parent_n)))
    lengths = lengths[:5]  # max 5 lengths

    scan_results = []
    total_length_budget = 250.0  # seconds for phase 1

    for n in lengths:
        if time.monotonic() > deadline - 600:
            break
        remaining = min(deadline - time.monotonic() - 500, total_length_budget - (time.monotonic() - t_start))
        if remaining < 20:
            continue

        inits = generate_initializations(n, parent, rng)
        # Prioritize parent-based
        priority = ['parent', 'parent_interp', 'sine1', 'cos1', 'random', 'uniform', 'bimodal']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:3]  # top 3 per length

        t_per = max(10.0, remaining / len(init_names))

        for name in init_names:
            if time.monotonic() > deadline - 500:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 400)
            if t_alloc < 5:
                break

            a0 = inits[name]
            a_opt, obj = polyak_step_subgradient(
                a0, time_limit=t_alloc, lower_bound=1.50,
                top_k_init=100, temp_init=0.005, step_cap=0.12, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  SubGrad n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: Deeper subgradient on top candidates
    # =========================================================
    print("=== Phase 2: Deep Subgradient ===")
    t_start2 = time.monotonic()

    # Pick top candidates (diverse lengths)
    top_candidates = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen:
            top_candidates.append((n, a_opt, score))
            seen.add(n)
        if len(top_candidates) >= 3:
            break

    if best_sequence is not None and len(best_sequence) not in seen:
        top_candidates.append((len(best_sequence), best_sequence.copy(), best_score))

    remaining2 = deadline - time.monotonic() - 400.0
    n_cand = max(1, len(top_candidates))
    t_per2 = max(20.0, remaining2 * 0.5 / n_cand)

    for i, (n, a0, sc) in enumerate(top_candidates):
        if time.monotonic() > deadline - 350:
            break
        t_alloc = min(t_per2, deadline - time.monotonic() - 300)
        if t_alloc < 10:
            break

        # Use multiple configurations
        configs = [
            {'lower_bound': 1.50, 'top_k_init': 60, 'temp_init': 0.003, 'step_cap': 0.10},
            {'lower_bound': 1.50, 'top_k_init': 200, 'temp_init': 0.01, 'step_cap': 0.08},
        ]
        for ci, cfg in enumerate(configs):
            if time.monotonic() > deadline - 300:
                break
            t_alloc2 = min(t_alloc / (len(configs) - ci) * 0.85, deadline - time.monotonic() - 250)
            if t_alloc2 < 5:
                break
            a_opt, obj = polyak_step_subgradient(a0, time_limit=t_alloc2, **cfg)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep SG {i}.{ci} (n={n}): {score:.6f} *NEW BEST*")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Spectral refinement
    # =========================================================
    print("=== Phase 3: Spectral Refinement ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        remaining3 = deadline - time.monotonic() - 300.0
        t_spec = min(remaining3 * 0.3, 80.0)
        if t_spec > 10:
            a_opt, obj = spectral_refine(best_sequence, time_limit=t_spec)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Spectral: {score:.6f} *NEW BEST*")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Subgradient polish (very fine)
    # =========================================================
    print("=== Phase 4: Fine Polish ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        remaining4 = deadline - time.monotonic() - 200.0

        # Multiple fine-tuning runs with high top_k (focus on actual peaks)
        fine_configs = [
            {'lower_bound': 1.50, 'top_k_init': 20, 'temp_init': 0.001, 'step_cap': 0.05},
            {'lower_bound': 1.50, 'top_k_init': 50, 'temp_init': 0.002, 'step_cap': 0.04},
            {'lower_bound': 1.50, 'top_k_init': 10, 'temp_init': 0.0005, 'step_cap': 0.03},
        ]
        n_fine = len(fine_configs)
        t_per_fine = max(10.0, remaining4 * 0.4 / n_fine)

        for fi, cfg in enumerate(fine_configs):
            if time.monotonic() > deadline - 150:
                break
            t_alloc = min(t_per_fine, deadline - time.monotonic() - 100)
            if t_alloc < 5:
                break
            a_opt, obj = polyak_step_subgradient(best_sequence, time_limit=t_alloc, **cfg)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Fine {fi}: {score:.6f} *NEW BEST*")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: LP polish (if time allows)
    # =========================================================
    print("=== Phase 5: LP Polish ===")
    t_start5 = time.monotonic()

    if best_sequence is not None:
        remaining5 = deadline - time.monotonic() - 50.0
        t_lp = min(remaining5 * 0.5, 120.0)
        if t_lp > 15:
            a_opt, obj = lp_polish(best_sequence, time_limit=t_lp)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP: {score:.6f} *NEW BEST*")

    print(f"Phase 5 done ({time.monotonic()-t_start5:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 6: Final subgradient micro-steps
    # =========================================================
    print("=== Phase 6: Final Micro ===")

    if best_sequence is not None and time.monotonic() < deadline - 10:
        remaining6 = deadline - time.monotonic() - 5
        if remaining6 > 10:
            # Very aggressive peak focus
            a_opt, obj = polyak_step_subgradient(
                best_sequence, time_limit=remaining6,
                lower_bound=1.50, top_k_init=5, temp_init=0.0001, step_cap=0.02)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Micro: {score:.6f} *NEW BEST*")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
