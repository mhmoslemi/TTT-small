def tournament_selection(
    population: np.ndarray,
    fitnesses: List[float],
    tournament_size: int = 3
) -> int:
    """Selects an individual from the population through tournament selection"""
    candidate_idx = random.randint(0, population.shape[0]-1)
    candidates = random.sample(range(population.shape[0]), tournament_size)
    winners = [candidate_idx] + [i for i in candidates if fitnesses[i] <= fitnesses[candidate_idx]]
    if not winners:
        return random.randint(0, population.shape[0]-1)
    return random.choice(winners)
