    Q = build_Q(k, n)

            break

        a_curr = np.zeros(n)
        a_curr[S_list] = a_S

    """Approximate prolate/slepian sequence using raised cosine with bandwidth param."""
    x = np.arange(n, dtype=np.float64)
    t = (x - (n - 1) / 2) / ((n - 1) / 2)
    a = np.sinc(c * t) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def make_bimodal(n):
    """Bimodal: mass at edges."""
    a = np.zeros(n)
    edge = max(1, n // 6)
    a[:edge] = 1.0
    a[-edge:] = 1.0
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def interpolate_sequence(a_src, n_target):
    """Interpolate a sequence to a new length."""
    if len(a_src) == n_target:
        return a_src.copy()
    s = a_src.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a_src / s
    x_src = np.linspace(0, 1, len(a_src))
    x_tgt = np.linspace(0, 1, n_target)
    a_new = np.interp(x_tgt, x_src, a_norm)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn > 0:
        a_new = a_new / sn
    return a_new


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization:
    Phase 1: Quick exploration with diverse initializations
    Phase 2: Deep LP optimization on best candidate
    Phase 3: Polish with coordinate descent and fine gradient
    Phase 4: Try alternative sequence lengths
    Phase 5: Final LP micro-polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    def update_best(a, label=""):
        nonlocal best_sequence, best_score
        if a is None or len(a) == 0:
            return
        a_valid = [float(max(0.0, min(1000.0, x))) for x in a]
        try:
            score = float(evaluate_sequence(a_valid))
        except Exception:
            score = float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = np.array(a_valid, dtype=np.float64)
            print(f"  {label}: score={score:.8f}, n={len(a)}")

    # === Phase 1: Quick Exploration (120s) ===
    print("=== Phase 1: Quick Exploration ===")
    t_p1_start = time.monotonic()
    p1_budget = min(120.0, (deadline - time.monotonic()) * 0.1)

    target_lengths = [512, 768, 1024]
    candidates = []

    for n in target_lengths:
        if time.monotonic() > deadline - 1200:
            break

        # Generate initializations
        inits = {}
        inits['raised_cos'] = make_raised_cosine(n)
        inits['slepian'] = make_slepian_like(n, c=0.25)
        inits['slepian_wide'] = make_slepian_like(n, c=0.4)
        inits['bimodal'] = make_bimodal(n)
        inits['uniform'] = np.ones(n) / n

        if parent is not None:
            inits['parent'] = interpolate_sequence(parent, n)
            # Parent with small perturbation
            p_pert = inits['parent'] + rng.standard_normal(n) * 0.01
            p_pert = np.maximum(p_pert, 0)
            p_pert /= p_pert.sum()
            inits['parent_pert'] = p_pert

        # Quick gradient on each init
        t_per_init = max(5.0, p1_budget / (len(target_lengths) * 6))

        for name, a0 in inits.items():
            if time.monotonic() > deadline - 1100:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 1100)
            if t_alloc < 3:
                break

            a_opt, obj = nesterov_optimize(a0, time_limit=t_alloc,
                                           lr_init=0.01, momentum=0.91,
                                           top_k=40, temp_init=0.002)
            candidates.append((obj, n, name, a_opt.copy()))
            update_best(a_opt, f"P1 n={n} {name}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_p1_start:.0f}s). Best so far: {best_score:.6f}")

    # === Phase 2: Deep LP Optimization (1200s) ===
    print("=== Phase 2: Deep LP ===")
    t_p2_start = time.monotonic()

    # Pick top candidates (best 2-3 from different lengths)
    top_candidates = []
    seen_n = set()
    for obj, n, name, a in candidates:
        if n not in seen_n and len(top_candidates) < 2:
            top_candidates.append((n, a))
            seen_n.add(n)

    # Always include the overall best if not already there
    if best_sequence is not None and len(best_sequence) not in seen_n:
        top_candidates.append((len(best_sequence), best_sequence.copy()))

    remaining_p2 = deadline - time.monotonic() - 400.0
    if remaining_p2 < 100:
        remaining_p2 = 100

    for i, (n, a0) in enumerate(top_candidates):
        if time.monotonic() > deadline - 350:
            break
        t_alloc = max(30.0, remaining_p2 / (len(top_candidates) - i))
        t_alloc = min(t_alloc, deadline - time.monotonic() - 300)
        if t_alloc < 10:
            break

        print(f"  LP cand {i}: n={n}, t={t_alloc:.0f}s")

        # Main LP run
        a_opt, obj = lp_optimize(a0, time_limit=t_alloc * 0.7,
                                  top_k_init=3, relax_init=1.0001, verbose=True)
        update_best(a_opt, f"P2 LP{i} n={n}")

        # Second LP run with different parameters
        if time.monotonic() < deadline - 200:
            t_rem = min(t_alloc * 0.3, deadline - time.monotonic() - 150)
            if t_rem > 10:
                a_opt2, obj2 = lp_optimize(a_opt, time_limit=t_rem,
                                           top_k_init=5, relax_init=1.00005)
                update_best(a_opt2, f"P2 LP{i}b n={n}")
                a_opt = a_opt2 if obj2 < obj else a_opt

        # Quick gradient polish after LP
        if time.monotonic() < deadline - 150:
            t_gpol = min(10.0, deadline - time.monotonic() - 130)
            if t_gpol > 3:
                a_gp, ogp = nesterov_optimize(a_opt, time_limit=t_gpol,
                                              lr_init=0.003, momentum=0.95,
                                              top_k=50, temp_init=0.001)
                update_best(a_gp, f"P2 gradpol{i}")

    print(f"Phase 2 done ({time.monotonic()-t_p2_start:.0f}s). Best: {best_score:.6f}")

    # === Phase 3: Polish (250s) ===
    print("=== Phase 3: Polish ===")
    t_p3_start = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 200.0
        if remaining > 50:
            # Multiple LP micro-runs with perturbation
            n_polish = min(4, int(remaining / 30))
            for pi in range(n_polish):
                if time.monotonic() > deadline - 150:
                    break
                t_alloc = min(remaining / (n_polish - pi) * 0.8, deadline - time.monotonic() - 130)
                if t_alloc < 8:
                    break

                if pi == 0:
                    a0 = best_sequence.copy()
                else:
                    scale = 0.002 * (pi + 1)
                    noise = rng.standard_normal(len(best_sequence)) * scale
                    a0 = np.maximum(best_sequence + noise, 0)
                    s = a0.sum()
                    if s > 1e-15:
                        a0 = a0 / s
                    else:
                        a0 = best_sequence.copy()

                a_opt, obj = lp_optimize(a0, time_limit=t_alloc,
                                         top_k_init=3, relax_init=1.00005)
                update_best(a_opt, f"P3 polish{pi}")

            # Coordinate descent
            remaining = deadline - time.monotonic() - 100.0
            if remaining > 10:
                t_cd = min(remaining * 0.4, 60.0)
                if t_cd > 5:
                    a_cd, obj_cd = coordinate_descent(best_sequence, time_limit=t_cd)
                    update_best(a_cd, f"P3 coorddesc")

            # Fine gradient
            remaining = deadline - time.monotonic() - 80.0
            if remaining > 8:
                t_fg = min(remaining * 0.6, 30.0)
                if t_fg > 3:
                    a_fg, obj_fg = nesterov_optimize(best_sequence, time_limit=t_fg,
                                                     lr_init=0.002, momentum=0.96,
                                                     top_k=80, temp_init=0.0005)
                    update_best(a_fg, f"P3 finegrad")

    print(f"Phase 3 done ({time.monotonic()-t_p3_start:.0f}s). Best: {best_score:.6f}")

    # === Phase 4: Alternative lengths (150s) ===
    print("=== Phase 4: Length Exploration ===")
    t_p4_start = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 50.0
        alt_lengths = [512, 1024, 1280]
        # Only try lengths different from current best
        current_n = len(best_sequence)
        alt_lengths = [ln for ln in alt_lengths if ln != current_n][:2]

        for ln in alt_lengths:
            if time.monotonic() > deadline - 40:
                break
            t_alloc = min(remaining / len(alt_lengths), deadline - time.monotonic() - 30)
            if t_alloc < 20:
                break

            a_interp = interpolate_sequence(best_sequence, ln)
            # Quick optimize at new length
            a_opt, obj = lp_optimize(a_interp, time_limit=t_alloc * 0.7,
                                     top_k_init=3, relax_init=1.0001)
            update_best(a_opt, f"P4 n={ln}")

    print(f"Phase 4 done ({time.monotonic()-t_p4_start:.0f}s). Best: {best_score:.6f}")

    # === Phase 5: Final micro-polish (30s) ===
    print("=== Phase 5: Final Polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 5:
        t_final = min(20.0, deadline - time.monotonic() - 3)
        if t_final > 5:
            a_final, obj_final = lp_optimize(best_sequence, time_limit=t_final,
                                             top_k_init=3, relax_init=1.00003)
            update_best(a_final, "P5 final")

    # Fallback
    if best_sequence is None:
        best_sequence = make_raised_cosine(256)
        update_best(best_sequence, "fallback")

    # Final verification
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_sequence]
    try:
        final_score = float(evaluate_sequence(final_list))
    except Exception:
        final_score = float('inf')

    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return final_list
