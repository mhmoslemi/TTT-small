Multi-strategy local search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses spectral analysis, window reoptimization, pairwise coordinate descent,
and LP-based Frank-Wolfe as a finishing tool.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(f, M):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except:
        pass
    return None

def fw_step(f, M_time=30):

    conv = fast_convolve(f)
    M = conv.max()
    g = solve_lp(f, M)
    if g is None:
        g = solve_lp(f, M * 1.05)
    if g is None:
        return f, M
    gs = g.sum()
    if gs < 1e-15:
        return f, M
    g = g / gs
    
    # Line search
    best_alpha = 0.0
    best_val = M
    for alpha in np.linspace(0.1, 1.0, 15):
        ft = (1 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0)

        if s < 1e-15:

        ft = ft / s

        end = start + window_size
        idx = np.arange(start, end)
        
        # Fix everything outside the window
        # The convolution at lag k depends on a[j] * a[k-j]
        # We need to find the best values for a[idx] that minimize max conv

        # that preserve the sum
        window_sum = f[idx].sum()
        
        # Try a few different distributions
        candidates = []
        # Uniform
        candidates.append(np.full(window_size, window_sum / window_size))
        # Triangular
        tri = np.abs(np.arange(window_size) - window_size / 2)
        tri = tri / tri.sum() * window_sum
        candidates.append(tri)
        # Random
        candidates.append(rng.dirichlet(np.ones(window_size)) * window_sum)
        # Current (sorted)

            f_new[idx] = cand
            val = fast_convolve(f_new).max()

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses targeted gradient descent with level-set targeting, LP-based Frank-Wolfe,
Fourier-mode perturbation, and pairwise coordinate descent in a cascaded fashion.
Key innovation: adaptive target-level gradient that focuses on the most
offending convolution lags, combined with multi-scale exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M):
    """LP oracle: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_line_search(f, g, M_current):
    """Coarse-to-fine line search for FW step."""
    best_alpha = 0.0
    best_val = M_current

    alphas = np.linspace(0.05, 1.0, 16)
    for alpha in alphas:
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            continue
        ft = ft / s
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha

    # Refine around best
    if 0.05 < best_alpha < 0.95:
        lo = max(0.0, best_alpha - 0.06)
        hi = min(1.0, best_alpha + 0.06)
        for _ in range(7):
            mid = (lo + hi) / 2.0
            ft = (1.0 - mid) * f + mid * g
            ft = np.maximum(ft, 0.0)
            s = ft.sum()
            if s < 1e-15:
                break
            ft = ft / s
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
                lo = mid
            else:
                hi = mid

    return best_alpha, best_val


def fw_step(f):
    """One Frank-Wolfe step using LP oracle."""
    conv = fast_convolve(f)
    M = conv.max()
    g = solve_lp(f, M)
    if g is None:
        g = solve_lp(f, M * 1.03)
    if g is None:
        return f, M, False

    gs = g.sum()
    if gs < 1e-15:
        return f, M, False
    g = g / gs

    # Check fixed point
    if np.max(np.abs(g - f)) < 1e-10:
        return f, M, False

    alpha, val = fw_line_search(f, g, M)
    if alpha > 1e-10:
        f_new = (1.0 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-15:
            f_new = f_new / s
            return f_new, val, True
    return f, M, False


def targeted_gradient(f0, time_limit, n_peaks=8, init_lr=0.008, mom=0.91):
    """
    Gradient method that targets the most offending convolution lags.
    Uses an adaptive target level that tightens over time.
    """
    n = len(f0)
    f = np.maximum(f0, 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    lr = init_lr
    target = best_obj
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(12345)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Find peaks exceeding target (or top peaks if none exceed)
        if (conv > target).sum() > 0:
            peaks = np.where(conv > target)[0]
            if len(peaks) > n_peaks:
                # Take the worst ones
                peak_vals = conv[peaks]
                worst_idx = np.argsort(peak_vals)[-n_peaks:]
                peaks = peaks[worst_idx]
        else:
            # Tighten target and take top peaks
            target = max(best_obj * 0.95, target * 0.97)
            peaks = np.argpartition(conv, -min(n_peaks, len(conv)))[-min(n_peaks, len(conv)):]

        # Weight by excess over target
        pv = conv[peaks]
        excess = np.maximum(pv - target, 1e-10)
        w = excess ** 2  # Quadratic weighting for emphasis on worst peaks
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Nesterov-like momentum step
        v = mom * v - lr * sg
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            conv = conv_new
            no_improve = 0
            # Tighten target gradually
            target = min(target * 0.998, best_obj * 0.995)
        else:
            no_improve += 1
            lr *= 0.97

        # Adaptive restart
        if no_improve > 150:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = init_lr * 0.6
            no_improve = 0
            target = fast_convolve(f).max()

    return best_f, best_obj


def fourier_search(f0, time_limit):
    """
    Search using random Fourier mode perturbations.
    Explores directions in the frequency domain that can reduce convolution peaks.
    """
    n = len(f0)
    best_f = f0.copy()
    best_obj = fast_convolve(f0).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(8765)
    x = np.arange(n)

    # Try different scales of perturbation
    scales = [0.005, 0.01, 0.02, 0.04, 0.06, 0.08]
    for scale in scales:
        if time.monotonic() - t0 > time_limit * 0.8:
            break
        for _ in range(30):
            if time.monotonic() - t0 > time_limit * 0.95:
                break
            # Random 1-3 Fourier modes
            n_modes = rng.integers(1, 4)
            pert = np.zeros(n)
            for _m in range(n_modes):
                k = rng.integers(1, max(2, n // 3))
                amp = rng.uniform(0.3, 1.0) * scale
                phi = rng.uniform(0, 2 * np.pi)
                pert += amp * np.cos(2 * np.pi * k * x / n + phi)

            f_new = project_simplex(best_f + pert)
            val = fast_convolve(f_new).max()
            if val < best_obj - 1e-15:
                best_obj = val
                best_f = f_new.copy()

    return best_f, best_obj


def pairwise_cd(f0, time_limit, n_trials_per_round=100):
    """
    Pairwise coordinate descent: move mass between pairs of coordinates.
    Tries to find the optimal transfer amount for each pair.
    """
    n = len(f0)
    best_f = f0.copy()
    best_obj = fast_convolve(f0).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(3456)

    while time.monotonic() - t0 < time_limit:
        improved = False
        for _ in range(n_trials_per_round):
            if time.monotonic() - t0 > time_limit - 1:
                break
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j or best_f[j] < 1e-12:
                continue

            # Try several transfer fractions
            for frac in [0.02, 0.05, 0.1, 0.2, 0.4]:
                delta = best_f[j] * frac
                if delta < 1e-14:
                    continue
                f_new = best_f.copy()
                f_new[i] += delta
                f_new[j] -= delta
                if f_new[j] < 0:
                    f_new[i] += f_new[j]
                    f_new[j] = 0.0
                s = f_new.sum()
                if s < 1e-15:
                    continue
                f_new = f_new / s
                val = fast_convolve(f_new).max()
                if val < best_obj - 1e-15:
                    best_obj = val
                    best_f = f_new.copy()
                    improved = True
                    break

        if not improved:
            # Try structured pairs (symmetric positions)
            structured_improved = False
            for offset in [n // 2, n // 3, n // 4, n // 5]:
                for i in range(0, n - offset, max(1, n // 30)):
                    j = i + offset
                    if j >= n:
                        break
                    for frac in [0.02, 0.05, 0.1]:
                        delta = best_f[j] * frac
                        if delta < 1e-14:
                            continue
                        f_new = best_f.copy()
                        f_new[i] += delta
                        f_new[j] -= delta
                        if f_new[j] < 0:
                            f_new[i] += f_new[j]
                            f_new[j] = 0.0
                        s = f_new.sum()
                        if s < 1e-15:
                            continue
                        f_new = f_new / s
                        val = fast_convolve(f_new).max()
                        if val < best_obj - 1e-15:
                            best_obj = val
                            best_f = f_new.copy()
                            improved = True
                            structured_improved = True
                            break
                    if structured_improved:
                        break
                if structured_improved:
                    break

            if not improved:
                break

    return best_f, best_obj


def smooth_and_resolve(f0, time_limit):
    """
    Apply Gaussian smoothing then re-optimize. This can escape local minima
    by reducing high-frequency oscillations that contribute to convolution peaks.
    """
    n = len(f0)
    t0 = time.monotonic()
    best_f = f0.copy()
    best_obj = fast_convolve(f0).max()

    # Try different smoothing levels
    for sigma_factor in [0.5, 1.0, 2.0, 3.0, 5.0]:
        if time.monotonic() - t0 > time_limit * 0.5:
            break
        sigma = sigma_factor * n / 50.0
        # Simple Gaussian via repeated box blur
        kernel_size = max(3, int(4 * sigma) | 1)  # odd
        kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (sigma ** 2 + 1e-10))
        kernel = kernel / kernel.sum()

        # Apply smoothing (circular)
        f_smooth = np.convolve(f0, kernel, mode='same')
        f_smooth = np.maximum(f_smooth, 0)
        s = f_smooth.sum()
        if s < 1e-15:
            continue
        f_smooth = f_smooth / s

        val = fast_convolve(f_smooth).max()

        # Blend with original
        for blend in [0.3, 0.5, 0.7, 1.0]:
            f_mix = (1 - blend) * f0 + blend * f_smooth
            f_mix = np.maximum(f_mix, 0)
            s = f_mix.sum()
            if s < 1e-15:
                continue
            f_mix = f_mix / s
            val = fast_convolve(f_mix).max()
            if val < best_obj - 1e-15:
                best_obj = val
                best_f = f_mix.copy()

        # Quick FW steps from the smoothed version
        if time.monotonic() - t0 < time_limit * 0.8:
            f_cur = f_smooth.copy()
            for _ in range(5):
                f_cur, _, ok = fw_step(f_cur)
                if not ok:
                    break
                val = fast_convolve(f_cur).max()
                if val < best_obj - 1e-15:
                    best_obj = val
                    best_f = f_cur.copy()

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive restarts."""
    n = len(f0)
    f = np.maximum(f0, 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(9999 + n)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 3:
            break

        it += 1
        f_new, obj, improved = fw_step(f)

        if improved:
            f = f_new
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    FW iter {it}: obj={obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Restart strategies
        if no_improve > 20:
            strategies = no_improve % 4
            if strategies == 0:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            elif strategies == 1:
                f = project_simplex(0.8 * best_f + 0.2 * best_f[::-1])
            elif strategies == 2:
                # Edge boost
                x = np.arange(n)
                boost = 1.0 + 0.05 * np.exp(-((x - n / 2) / (n / 10)) ** 2)
                f = project_simplex(best_f * boost)
            else:
                # Random window flip
                ws = n // 5
                start = rng.integers(0, n - ws)
                f = best_f.copy()
                f[start:start + ws] = f[start + ws - 1:start - 1:-1]
                f = project_simplex(f)

            no_improve = 0
            f = np.maximum(f, 0)
            fs = f.sum()
            if fs > 1e-15:
                f = f / fs

    return best_f, best_obj


def try_sequence(a, label, state):
    """Evaluate and update best."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    if a.sum() < 1e-10:
        return float('inf')
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining targeted gradient,
    LP-based Frank-Wolfe, Fourier perturbation, smoothing, and pairwise CD.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    state = {'best_seq': None, 'best_score': float('inf')}

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}")
            s = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(s):
                print(f"  Parent score: {s:.6f}")
                state['best_score'] = s
                state['best_seq'] = parent.copy()
    except Exception:
        pass

    if state['best_seq'] is None:
        parent = np.ones(384) / 384
        state['best_seq'] = parent.copy()
        state['best_score'] = float(evaluate_sequence(parent.tolist()))
        print(f"No parent. Uniform n=384: {state['best_score']:.6f}")

    # === Phase 1: Exploration (25% of time) ===
    print("\n=== Phase 1: Exploration ===")
    p1_end = time.monotonic() + float(budget_s) * 0.25

    # Quick targeted gradient on parent
    f_best = state['best_seq'].copy()
    n_best = len(f_best)
    t_alloc = min(60, p1_end - time.monotonic() - 30)
    if t_alloc > 15:
        print(f"  Targeted gradient on parent (n={n_best}, t={t_alloc:.0f}s)")
        f_opt, _ = targeted_gradient(f_best, time_limit=t_alloc, n_peaks=6, init_lr=0.006)
        try_sequence(f_opt, "tg_parent", state)

    # Try nearby lengths
    adjacent_lengths = [n_best - 32, n_best - 16, n_best + 16, n_best + 32, n_best + 64]
    for n_try in adjacent_lengths:
        if n_try < 100 or n_try > 1000:
            continue
        if time.monotonic() > p1_end - 40:
            break
        # Interpolate
        x_old = np.linspace(0, 1, len(f_best))
        x_new = np.linspace(0, 1, n_try)
        f_interp = np.interp(x_new, x_old, f_best)
        f_interp = np.maximum(f_interp, 0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp = f_interp / s

        # Quick optimization: a few FW steps
        if time.monotonic() < p1_end - 20:
            for _ in range(4):
                f_interp, _, _ = fw_step(f_interp)

        try_sequence(f_interp, f"len{n_try}", state)

    # Fourier search from parent
    if time.monotonic() < p1_end - 20:
        t_alloc = min(40, p1_end - time.monotonic() - 10)
        if t_alloc > 10:
            f_opt, _ = fourier_search(f_best, time_limit=t_alloc)
            try_sequence(f_opt, "fourier_p1", state)

    print(f"  Phase 1 best: {state['best_score']:.6f}")

    # === Phase 2: Intensive optimization (50% of time) ===
    print(f"\n=== Phase 2: Intensive Optimization ===")
    p2_end = time.monotonic() + float(budget_s) * 0.50

    f_best = state['best_seq'].copy()
    n_best = len(f_best)

    # 2a: LP-FW (major allocation)
    t_alloc = min(200, (p2_end - time.monotonic()) * 0.40)
    if t_alloc > 30:
        print(f"  2a: LP-FW (t={t_alloc:.0f}s)")
        f_opt, _ = lp_fw_optimize(f_best, time_limit=t_alloc, verbose=True)
        try_sequence(f_opt, "lfpw_main", state)
        f_best = state['best_seq'].copy()

    # 2b: Targeted gradient (second run, different params)
    t_alloc = min(120, (p2_end - time.monotonic()) * 0.25)
    if t_alloc > 20:
        print(f"  2b: Targeted gradient (t={t_alloc:.0f}s)")
        f_opt, _ = targeted_gradient(f_best, time_limit=t_alloc, n_peaks=12, init_lr=0.004, mom=0.93)
        try_sequence(f_opt, "tg2", state)
        f_best = state['best_seq'].copy()

    # 2c: Smooth and resolve
    t_alloc = min(80, (p2_end - time.monotonic()) * 0.20)
    if t_alloc > 15:
        print(f"  2c: Smooth & resolve (t={t_alloc:.0f}s)")
        f_opt, _ = smooth_and_resolve(f_best, time_limit=t_alloc)
        try_sequence(f_opt, "smooth", state)
        f_best = state['best_seq'].copy()

    # 2d: Fourier search
    t_alloc = min(60, (p2_end - time.monotonic()) * 0.15)
    if t_alloc > 10:
        print(f"  2d: Fourier search (t={t_alloc:.0f}s)")
        f_opt, _ = fourier_search(f_best, time_limit=t_alloc)
        try_sequence(f_opt, "fourier2", state)
        f_best = state['best_seq'].copy()

    # 2e: Pairwise CD
    t_alloc = min(80, (p2_end - time.monotonic()) * 0.20)
    if t_alloc > 15:
        print(f"  2e: Pairwise CD (t={t_alloc:.0f}s)")
        f_opt, _ = pairwise_cd(f_best, time_limit=t_alloc)
        try_sequence(f_opt, "cd2", state)
        f_best = state['best_seq'].copy()

    # 2f: Another LP-FW pass from best
    t_alloc = min(100, (p2_end - time.monotonic()) * 0.3)
    if t_alloc > 20:
        print(f"  2f: LP-FW round 2 (t={t_alloc:.0f}s)")
        f_opt, _ = lp_fw_optimize(f_best, time_limit=t_alloc, verbose=False)
        try_sequence(f_opt, "lfpw_r2", state)
        f_best = state['best_seq'].copy()

    print(f"  Phase 2 best: {state['best_score']:.6f}")

    # === Phase 3: Final polishing (25% of time) ===
    print(f"\n=== Phase 3: Final Polishing ===")
    p3_end = deadline - 5

    f_best = state['best_seq'].copy()
    n_best = len(f_best)

    # 3a: Fine targeted gradient
    t_alloc = min(80, (p3_end - time.monotonic()) * 0.30)
    if t_alloc > 15:
        print(f"  3a: Fine TG (t={t_alloc:.0f}s)")
        f_opt, _ = targeted_gradient(f_best, time_limit=t_alloc, n_peaks=4, init_lr=0.002, mom=0.95)
        try_sequence(f_opt, "tg_fine", state)
        f_best = state['best_seq'].copy()

    # 3b: LP-FW final
    t_alloc = min(60, (p3_end - time.monotonic()) * 0.25)
    if t_alloc > 15:
        print(f"  3b: LP-FW final (t={t_alloc:.0f}s)")
        f_opt, _ = lp_fw_optimize(f_best, time_limit=t_alloc, verbose=False)
        try_sequence(f_opt, "lfpw_final", state)
        f_best = state['best_seq'].copy()

    # 3c: Fine pairwise CD
    t_alloc = min(60, (p3_end - time.monotonic()) * 0.25)
    if t_alloc > 10:
        print(f"  3c: Fine CD (t={t_alloc:.0f}s)")
        f_opt, _ = pairwise_cd(f_best, time_limit=t_alloc, n_trials_per_round=200)
        try_sequence(f_opt, "cd_fine", state)
        f_best = state['best_seq'].copy()

    # 3d: Very fine gradient
    t_alloc = min(40, (p3_end - time.monotonic()) * 0.20)
    if t_alloc > 10:
        print(f"  3d: Ultra-fine TG (t={t_alloc:.0f}s)")
        f_opt, _ = targeted_gradient(f_best, time_limit=t_alloc, n_peaks=2, init_lr=0.001, mom=0.97)
        try_sequence(f_opt, "tg_ultra", state)
        f_best = state['best_seq'].copy()

    # 3e: One last FW squeeze
    t_rem = p3_end - time.monotonic() - 3
    if t_rem > 10:
        print(f"  3e: Final FW squeeze (t={t_rem:.0f}s)")
        f_cur = f_best.copy()
        for _ in range(min(20, int(t_rem // 10))):
            if time.monotonic() > p3_end - 3:
                break
            f_cur, _, ok = fw_step(f_cur)
            if not ok:
                break
            s = float(evaluate_sequence(f_cur.tolist()))
            if np.isfinite(s) and s < state['best_score']:
                state['best_score'] = s
                state['best_seq'] = f_cur.copy()
                print(f"    Final FW: {s:.6f}")

    # Final verification
    final_score = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(final_score):
        state['best_score'] = final_score

    print(f"\nFINAL SCORE: {state['best_score']:.6f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
