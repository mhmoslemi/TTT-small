"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an enhanced multi-phase approach:
1. LP-based Frank-Wolfe with adaptive constraint relaxation
2. Subgradient method with backtracking line search and momentum
3. Frequency-domain smoothing for escape from local minima
4. Multi-length exploration and final polish

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build the constraint matrix for the LP: (f*g)[k] <= M for all k."""
    n = len(f)
    # A_ub has shape (2n-1, n)
    # Row k represents constraint (f*g)[k] = sum_j f[j]*g[k-j] <= M
    # For row k, the coefficient of g[j] is f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] appears in (f*g)[k] for k = j, j+1, ..., j+n-1
        # with coefficient f[k-j]
        A[j:j + n, j] = f
    return A

def sg_optimize(f0, time_limit, lr=0.002, momentum=0.93, top_k=20, temperature=0.0005,
                decay_lr=0.9995, decay_temp=1.002, verbose=True):
    """
    Subgradient optimization with adaptive peak weighting and line search.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temperature
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break
        
        conv = fast_convolve(f)
        M = float(conv.max())
        
        # Select top-k peaks with softmax weighting
        k_sel = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_sel)[-k_sel:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            L = max(0, k - n + 1)
            R = min(n - 1, k)
            idx = np.arange(L, R + 1)
            sg[idx] += w[i] * 2.0 * f[k - idx]
        
        # Momentum update
        v = momentum * v - cur_lr * sg
        
        # Project
        f_new = project_simplex(f + v)
        
        # Check improvement
        conv_new = fast_convolve(f_new)
        obj_new = float(conv_new.max())
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        cur_lr *= decay_lr
        cur_temp *= decay_temp
        
        # Restart if stuck
        if no_improve > 100:
            rng = np.random.default_rng(it * 104729 + 7)
            noise_scale = 0.002 * min(2.0, no_improve / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.7, 0.0003)
            cur_temp = temperature * 1.5
            no_improve = 0
    
    return best_f, best_obj


def freq_smooth_optimize(f0, time_limit, verbose=True):

    s = f.sum()

        f = np.ones(len(f)) / len(f)

        f = f / s
    n = len(f)

    best_obj = float(conv.max())
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0

        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break
        
        # Compute DFT
        fft_size = 1 << (2 * n - 2).bit_length()
        A = np.fft.rfft(f, fft_size)
        
        # Low-pass filter: keep low frequencies, attenuate high
        # Use a raised cosine transition
        half = fft_size // 2
        freq = np.arange(half + 1) / fft_size
        # Attenuation: 1 for low freq, decreases for high
        cutoff = 0.1  # Keep first 10% of frequencies fully
        atten = np.ones(half + 1)
        high_start = int(cutoff * half)
        if high_start < half:
            transition = np.linspace(1.0, 0.3, half - high_start)
            atten[high_start:] = transition

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    
    # Gaussian
    x = np.arange(n, dtype=np.float64)
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Cosine (Hann window)
    h = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    h = np.maximum(h, 0.01)
    inits.append(h / h.sum())
    
    # Edge-weighted
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Random smooth
    rough = rng.standard_normal(n)
    # Smooth via FFT
    fft_size = 1 << (n - 1).bit_length()
    R = np.fft.rfft(rough, fft_size)
    half = fft_size // 2
    R[int(0.2 * half):] *= 0.1
    smooth = np.fft.irfft(R, fft_size)[:n]
    smooth = np.abs(smooth) + 0.1
    inits.append(smooth / smooth.sum())
    
    # Interpolated parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 2.0))
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an enhanced multi-phase approach:
1. LP-based Frank-Wolfe with adaptive constraint relaxation and careful line search
2. Subgradient method with exponential peak weighting, backtracking, and momentum
3. Frequency-domain smoothing to escape local minima
4. Multi-length exploration and adaptive time allocation
Key improvements: exponential (not softmax) peak weighting for sharper focus,
theoretical-gap-based adaptive learning rate, and structured restarts.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, n):
    """Build constraint matrix: row k has (f*g)[k] = sum_j f[k-j]*g[j]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle, adaptive M relaxation, and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = float(conv.max())
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    m_relax = 1.0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break

        A_lp = build_lp_matrix(f, n)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * m_relax)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c, A_ub=A_lp, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': max(1.0, rem * 0.05)})
        except Exception:
            m_relax = min(m_relax * 1.2, 5.0)
            continue

        if not res.success:
            m_relax = min(m_relax * 1.15, 5.0)
            if m_relax > 3.0:
                m_relax = 1.0
                rng = np.random.default_rng(it * 7919 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = float(conv.max())
                no_improve = 0
            continue

        g = res.x
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                break
            m_relax = min(m_relax * 1.03, 2.5)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = float(fast_convolve(ft).max())
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.04)
            hi = min(1.0, best_alpha + 0.04)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = float(fast_convolve(ft).max())
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = float(conv.max())
            m_relax = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    FW iter {it}: obj={M:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.1, 4.0)

        if no_improve > 70:
            rng = np.random.default_rng(it * 31 + 17)
            scale = 0.001 + 0.004 * min(1.0, no_improve / 120.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = float(conv.max())
            m_relax = 1.0
            no_improve = 0

    return best_f, best_obj


def sg_optimize(f0, time_limit, lr=0.002, momentum=0.93, top_k=20,
                temperature=0.0005, decay_lr=0.9995, decay_temp=1.002,
                verbose=False):
    """Subgradient with exponential peak weighting and adaptive step."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()
    # Theoretical lower bound for max autocorrelation (sum=1): 1/(2n-1)
    lower_bound = 1.0 / (2 * n - 1)

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temperature
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        conv = fast_convolve(f)
        M = float(conv.max())

        # Adaptive learning rate based on gap to lower bound
        gap = M - lower_bound
        if gap > 1e-8:
            lr_scale = min(1.0, gap * 200.0)
        else:
            lr_scale = 0.1

        k_sel = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_sel)[-k_sel:]
        pv = conv[peaks].copy()

        # Exponential weighting (sharper than softmax)
        pv_shifted = pv - pv.max()
        w = np.exp(pv_shifted / cur_temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            L = max(0, k - n + 1)
            R = min(n - 1, k)
            idx = np.arange(L, R + 1)
            sg[idx] += w[i] * 2.0 * f[k - idx]

        v = momentum * v - cur_lr * lr_scale * sg
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = float(conv_new.max())

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        cur_lr *= decay_lr
        cur_temp *= decay_temp

        if no_improve > 120:
            rng = np.random.default_rng(it * 104729 + 7)
            noise_scale = 0.002 * min(2.0, no_improve / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.7, 0.0003)
            cur_temp = temperature * 1.5
            no_improve = 0

    return best_f, best_obj


def freq_smooth_optimize(f0, time_limit, verbose=False):
    """Frequency-domain smoothing to reduce high-frequency content."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    alpha = 0.1

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        fft_size = 1 << (2 * n - 2).bit_length()
        A = np.fft.rfft(f, fft_size)
        half = fft_size // 2
        high_start = max(1, int(0.15 * half))
        atten = np.ones(half + 1)
        atten[high_start:] = np.linspace(1.0, 0.2, half - high_start + 1)

        A_filtered = A * atten
        f_new = np.fft.irfft(A_filtered, fft_size)[:n]
        f_new = np.maximum(f_new, 0.0)
        fnew_sum = f_new.sum()
        if fnew_sum < 1e-15:
            break
        f_new = f_new / fnew_sum

        f_mix = (1 - alpha) * f + alpha * f_new
        f_mix = np.maximum(f_mix, 0.0)
        f_mix = f_mix / f_mix.sum()

        conv_new = fast_convolve(f_mix)
        obj_new = float(conv_new.max())

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_mix.copy()
            no_improve = 0
        else:
            no_improve += 1
            alpha = max(0.03, alpha * 0.8)

        f = f_mix

        if no_improve > 80:
            break

    return best_f, best_obj


def coordinated_descent(f0, time_limit, verbose=False):
    """
    Coordinate-style descent: find the pair (j, m) that gives best
    single-pair mass redistribution, execute it, repeat.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        conv = fast_convolve(f)
        M = float(conv.max())
        k_peak = int(np.argmax(conv))

        # Find the element contributing most to the peak
        L = max(0, k_peak - n + 1)
        R = min(n - 1, k_peak)
        idx = np.arange(L, R + 1)
        contrib = f[idx] * f[k_peak - idx]

        # Moving mass from j to m changes (a*a)[k] by:
        # -delta * a[k-j] - delta * a[j-k] + delta * a[k-m] + delta * a[m-k]
        # (for the peak at k_peak, and other peaks)
        # Simplified: just try moving mass from the top contributor to a "safe" position
        j_star = idx[np.argmax(contrib)]

        # Find "safe" position: one that minimizes impact on other peaks
        # Approximate: position with smallest sum |a[m] |a[k'-m]| over non-peak k'
        # Too expensive, so use heuristic: position with smallest f value
        # (moving to a zero-contribution area)
        mask = np.ones(n, dtype=bool)
        mask[j_star] = False
        # Also avoid positions that are partners of j_star for the peak
        partner = k_peak - j_star
        if 0 <= partner < n:
            mask[partner] = False

        if mask.any():
            candidates = np.where(mask)[0]
            # Score each candidate by its "danger" (how much it contributes to peaks)
            danger = np.zeros(len(candidates))
            for ci, m in enumerate(candidates):
                # m contributes to peaks at k = m + j for all j
                # Approximate danger as sum of f[j] for j where m+j or m-j is near peak
                danger[ci] = np.sum(f)  # Simplified
            # Just pick the one with smallest f value (least "active")
            m_star = candidates[np.argmin(f[candidates])]
        else:
            m_star = np.argmin(f)

        # Try multiple delta values
        best_delta = 0.0
        best_val = M
        dmax = min(f[j_star], 0.1 / n)
        for delta_frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            delta = delta_frac * dmax
            if delta < 1e-15:
                continue
            f_try = f.copy()
            f_try[j_star] -= delta
            f_try[m_star] += delta
            f_try = np.maximum(f_try, 0.0)
            f_try = f_try / f_try.sum()
            val = float(fast_convolve(f_try).max())
            if val < best_val:
                best_val = val
                best_delta = delta

        if best_delta > 0:
            f[j_star] -= best_delta
            f[m_star] += best_delta
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = float(conv.max())

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            rng = np.random.default_rng(it * 6151 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)

    x = np.arange(n, dtype=np.float64)
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())

    h = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    h = np.maximum(h, 0.01)
    inits.append(h / h.sum())

    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    rough = rng.standard_normal(n)
    fft_sz = 1 << (n - 1).bit_length()
    R = np.fft.rfft(rough, fft_sz)
    half = fft_sz // 2
    R[int(0.2 * half):] *= 0.1
    smooth = np.fft.irfft(R, fft_sz)[:n]
    smooth = np.abs(smooth) + 0.1
    inits.append(smooth / smooth.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    # Asymmetric bump
    x_norm = x / max(n - 1, 1)
    asym = np.exp(-4.0 * (x_norm - 0.3) ** 2) + 0.1 * np.exp(-4.0 * (x_norm - 0.7) ** 2)
    inits.append(asym / asym.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: Quick exploration with diverse inits
    Phase 2: Deep FW-LP refinement
    Phase 3: Subgradient optimization with multiple configs
    Phase 4: Coordinated descent + freq smoothing
    Phase 5: Final fine polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        try:
            score = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    n_default = 384
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
            n_default = len(parent)
    except Exception as e:
        print(f"Could not load parent: {e}")
        n_default = 384

    n_base = n_default

    # === Phase 1: Quick Exploration (~150s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_phase1_end = time.monotonic() + 150

    if parent is not None:
        t1 = min(80, t_phase1_end - time.monotonic() - 60)
        if t1 > 15:
            print(f"  FW on parent (n={n_base}), t={t1:.0f}s")
            a_opt, _ = fw_lp_optimize(parent, time_limit=t1)
            try_seq(a_opt, "fw_parent")

    # Try other inits briefly
    inits = generate_inits(n_base, parent, rng)
    for fi in [1, 2, 5]:
        if time.monotonic() > t_phase1_end - 10:
            break
        if fi < len(inits):
            t_rem = t_phase1_end - time.monotonic() - 5
            if t_rem > 10:
                a_opt, _ = fw_lp_optimize(inits[fi], time_limit=min(t_rem, 30))
                try_seq(a_opt, f"init_{fi}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Deep FW-LP (~500s) ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase2_end = time.monotonic() + min(500, (deadline - time.monotonic()) - 700)

    if best_seq is not None:
        n_b = len(best_seq)
        t2 = t_phase2_end - time.monotonic() - 100
        if t2 > 30:
            print(f"  Deep FW (n={n_b}), t={t2:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t2, verbose=True)
            try_seq(a_opt, "deep_fw")

        # Perturbed restart
        t_rem = t_phase2_end - time.monotonic()
        if t_rem > 30:
            noise = rng.standard_normal(n_b) * 0.005
            f_pert = project_simplex(best_seq + noise)
            a_opt2, _ = fw_lp_optimize(f_pert, time_limit=min(t_rem - 10, 150))
            try_seq(a_opt2, "fw_pert")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Subgradient (~400s) ===
    print("\n=== Phase 3: Subgradient ===")
    t_phase3_end = time.monotonic() + min(400, (deadline - time.monotonic()) - 350)

    if best_seq is not None:
        n_b = len(best_seq)
        sg_configs = [
            {'lr': 0.003, 'mom': 0.92, 'top_k': 15, 'temp': 0.0003, 'dlr': 0.9995, 'dt': 1.002},
            {'lr': 0.002, 'mom': 0.94, 'top_k': 30, 'temp': 0.0006, 'dlr': 0.9996, 'dt': 1.0015},
            {'lr': 0.004, 'mom': 0.90, 'top_k': 10, 'temp': 0.0002, 'dlr': 0.9994, 'dt': 1.003},
            {'lr': 0.001, 'mom': 0.95, 'top_k': 50, 'temp': 0.001, 'dlr': 0.9997, 'dt': 1.001},
        ]

        for si, cfg in enumerate(sg_configs):
            t_rem = t_phase3_end - time.monotonic()
            if t_rem < 15:
                break
            t_alloc = min(t_rem - 5, 120)
            print(f"  SG {si} (lr={cfg['lr']}, k={cfg['top_k']}), t={t_alloc:.0f}s")
            a_opt, _ = sg_optimize(
                best_seq, time_limit=t_alloc,
                lr=cfg['lr'], momentum=cfg['mom'],
                top_k=cfg['top_k'], temperature=cfg['temp'],
                decay_lr=cfg['dlr'], decay_temp=cfg['dt']
            )
            try_seq(a_opt, f"sg_{si}")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Coordinated Descent + Freq Smoothing (~200s) ===
    print("\n=== Phase 4: Coord Descent + Freq Smooth ===")
    t_phase4_end = time.monotonic() + min(200, (deadline - time.monotonic()) - 100)

    if best_seq is not None:
        n_b = len(best_seq)

        # Coordinated descent
        t_rem = t_phase4_end - time.monotonic()
        if t_rem > 30:
            t_alloc = min(t_rem - 20, 100)
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_opt, _ = coordinated_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord")

        # Freq smoothing
        t_rem = t_phase4_end - time.monotonic()
        if t_rem > 20:
            a_opt, _ = freq_smooth_optimize(best_seq, time_limit=min(t_rem - 5, 80))
            try_seq(a_opt, "freq_smooth")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Final Fine Polish (~remaining) ===
    print("\n=== Phase 5: Final Polish ===")

    if best_seq is not None:
        n_b = len(best_seq)

        # Fine subgradient
        t_rem = deadline - time.monotonic() - 30
        if t_rem > 30:
            a_opt, _ = sg_optimize(
                best_seq, time_limit=min(t_rem - 20, 100),
                lr=0.0005, momentum=0.96, top_k=30, temperature=0.0001,
                decay_lr=0.9998, decay_temp=1.001
            )
            try_seq(a_opt, "final_sg")

        # Tiny perturbations
        t_rem = deadline - time.monotonic() - 15
        for pi in range(5):
            t_rem = deadline - time.monotonic() - 5
            if t_rem < 10:
                break
            noise = rng.standard_normal(n_b) * 0.001
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = sg_optimize(
                f_pert, time_limit=min(t_rem - 2, 30),
                lr=0.0003, momentum=0.96, top_k=20, temperature=0.00005
            )
            try_seq(a_opt, f"pert_{pi}")

        # One more FW pass on best
        t_rem = deadline - time.monotonic() - 10
        if t_rem > 20:
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=min(t_rem - 5, 60))
            try_seq(a_opt, "final_fw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(n_base) / n_base
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
