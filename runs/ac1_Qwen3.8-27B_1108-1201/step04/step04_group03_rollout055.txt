            # Project gradient to tangent space of simplex (subtract mean component)
            # Actually, since we're normalizing inside, the effective gradient is:
            # dx is in tangent space if sum(dx)=0. The gradient in that space is g - mean(g)*1
            # But L-BFGS-B works in the full space with bounds, so let's just use g.
            # The normalization inside objective/grad handles the constraint approximately.
            return g
        
        # Run L-BFGS-B
        res = minimize(objective, a, method='L-BFGS-B', jac=grad_func,
                       bounds=[(0, None)] * n,
                       options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-8,
                                'max_fun': 500})
        
        a_new = np.maximum(res.x, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
            conv_new = fast_convolve(a_new)
            obj_new = conv_new.max()
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                print(f"    T={T:.1f}: improved to {obj_new:.10f} (score={2*n*obj_new:.8f})")
            a = a_new
    
    return best_a, best_obj


def coord_obj(a, j, x_new):
    """
    Compute max(conv(a,a)) after setting a[j] = x_new, given current a.
    Efficient: only modify affected b[k] values.
    """
    n = len(a)
    old = a[j]
    delta = x_new - old
    
    # Get current b
    b = fast_convolve(a)
    
    # Modify b for the change in a[j]:
    # For k in [j, j+n-1] intersect [0, 2n-2]:
    #   if k == 2*j: b[k] += delta*(x_new + old)
    #   else: b[k] += 2*delta*a[k-j]
    k_lo = max(0, j)
    k_hi = min(2 * n - 2, j + n - 1)
    
    if k_lo <= k_hi:
        ks = np.arange(k_lo, k_hi + 1)
        # General update: 2*delta*a[k-j]
        idx = ks - j  # = k-j, ranges from k_lo-j to k_hi-j, in [0, n-1]
        b[k_lo:k_hi + 1] += 2.0 * delta * a[idx]
        # Fix the k=2j term: we added 2*delta*a[j] but should have added delta*(x_new+old)
        # delta*(x_new+old) - 2*delta*a[j] = delta*(x_new+old-2*old) = delta*(x_new-old) = delta^2
        if 2 * j >= k_lo and 2 * j <= k_hi:
            b[2 * j] += delta * (x_new - old)  # = delta^2
    
    return b.max()


def coordinate_descent(a0, time_limit, n_sweeps_max=1000):
    """
    Exact coordinate descent: for each coordinate j, find the x that minimizes
    max(conv(a,a)) after setting a[j]=x, with a>=0 and sum(a)=1.
    Since objective is homogeneous, we work with sum(a)=1.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    # Precompute: for each j, the range of k affected and the corresponding a values
    # affect[j] = (k_start, k_end, a_segment) where a_segment = a[k_start-j : k_end-j+1]
    # But a changes, so we just compute on the fly.
    
    no_improve = 0
    sweep = 0
    
    while time.monotonic() - t0 < time_limit and sweep < n_sweeps_max:
        sweep += 1
        sweep_improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            
            # Define 1D objective: max_b after setting a[j] = x
            def f1d(x):
                return coord_obj(a, j, x)
            
            # Bounds: x in [0, 1] (since sum(a)=1 and a>=0)
            # Try a few candidate points first for speed
            # The optimal x is where the max of the modified b is minimized
            # Try: 0, old_val, and the point that equalizes the top peaks
            
            # Quick candidates
            candidates = [0.0, old_val]
            # Try to reduce the peak: if b at the peak is large, reduce a[j]
            # Try a few fractions
            if old_val > 0:
                candidates.extend([old_val * 0.5, old_val * 0.7, old_val * 1.3,
                                    old_val * 0.3, old_val * 1.5])
            candidates = [max(0.0, min(1.0, c)) for c in candidates]
            
            # Evaluate candidates
            best_x = old_val
            best_f = f1d(old_val)
            for c in candidates:
                if abs(c - best_x) < 1e-12:
                    continue
                val = f1d(c)
                if val < best_f - 1e-15:
                    best_f = val
                    best_x = c
            
            # Refine with Brent's method in a neighborhood
            lo = max(0.0, best_x - old_val * 0.5 - 0.01)
            hi = min(1.0, best_x + old_val * 0.5 + 0.01)
            if hi - lo > 1e-10:
                try:
                    res = minimize_scalar(f1d, bounds=(lo, hi), method='brent',
                                          options={'xatol': 1e-10, 'maxiter': 30})
                    if res.success and res.fun < best_f - 1e-15:
                        best_x = res.x
                        best_f = res.fun
                except Exception:
                    pass
            
            if best_x != old_val and best_f < best_obj - 1e-15:
                a[j] = best_x
                # Renormalize
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                sweep_improved = True
        
        # Check overall improvement after full sweep
        cur_obj = fast_convolve(a).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5:
                # Try random perturbations
                rng = np.random.default_rng(sweep * 7919)
                perturb = a * np.exp(rng.standard_normal(n) * 0.01)
                perturb = np.maximum(perturb, 0)
                ps = perturb.sum()
                if ps > 1e-15:
                    perturb /= ps
                    perturb_obj = fast_convolve(perturb).max()
                    if perturb_obj < best_obj:
                        a = perturb
                        best_obj = perturb_obj
                        best_a = a.copy()
                no_improve = 0
                if no_improve > 10 and sweep > 50:
                    break
    
    return best_a, best_obj


def coord_descent_fast(a0, time_limit):
    """
    Faster coordinate descent using precomputed b and incremental updates.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    b = fast_convolve(a)
    best_obj = b.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    sweep = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old = a[j]
            if old < 1e-15:
                # Try setting to a small positive value
                # The change in b[k] for k in [j, j+n-1]: b[k] += 2*new*a[k-j]
                # (ignoring the quadratic term which is negligible for small new)
                # Just skip for now
                continue
            
            # Precompute the influence: for k in [max(0,j), min(2n-2, j+n-1)]
            k_lo = max(0, j)
            k_hi = min(2 * n - 2, j + n - 1)
            ks = np.arange(k_lo, k_hi + 1)
            a_seg = a[ks - j]  # a[k-j] for k in range
            
            # b_modified(x) = b + 2*(x-old)*a_seg for k != 2j
            #                b + (x^2 - old^2) for k = 2j
            # = b + 2*(x-old)*a_seg + (x^2-old^2 - 2*(x-old)*a[j]) * indicator(k==2j)
            # = b + 2*(x-old)*a_seg + (x-old)*(x+old-2*old) * indicator(k==2j)
            # = b + 2*(x-old)*a_seg + (x-old)^2 * indicator(k==2j)
            
            def f1d_fast(x):
                delta = x - old
                b_mod = b.copy()
                b_mod[k_lo:k_hi + 1] += 2.0 * delta * a_seg
                if 2 * j >= k_lo and 2 * j <= k_hi:
                    b_mod[2 * j] += delta * delta
                return b_mod.max()
            
            # Find optimal x
            # Quick scan
            candidates = [0.0, old]
            if old > 1e-10:
                for frac in [0.3, 0.5, 0.7, 1.3, 1.5, 0.1]:
                    candidates.append(old * frac)
            candidates = [max(0.0, min(1.0, c)) for c in candidates]
            
            best_x = old
            best_f = f1d_fast(old)
            for c in candidates:
                if abs(c - old) < 1e-12:
                    continue
                val = f1d_fast(c)
                if val < best_f - 1e-14:
                    best_f = val
                    best_x = c
            
            # Brent refinement
            lo = max(0.0, best_x - max(old * 0.5, 0.02))
            hi = min(1.0, best_x + max(old * 0.5, 0.02))
            if hi - lo > 1e-12:
                try:
                    res = minimize_scalar(f1d_fast, bounds=(lo, hi), method='brent',
                                          options={'xatol': 1e-11, 'maxiter': 25})
                    if res.success and res.fun < best_f - 1e-14:
                        best_x = float(res.x)
                        best_f = float(res.fun)
                except Exception:
                    pass
            
            if abs(best_x - old) > 1e-14:
                a[j] = best_x
                # Update b
                delta = best_x - old
                b[k_lo:k_hi + 1] += 2.0 * delta * a_seg
                if 2 * j >= k_lo and 2 * j <= k_hi:
                    b[2 * j] += delta * delta
                # Renormalize (approximately - just scale)
                s = a.sum()
                if s > 1e-15 and abs(s - 1.0) > 1e-10:
                    scale = 1.0 / s
                    a *= scale
                    b *= scale * scale
                    best_obj_candidate = b.max()
                    # Don't update best here, wait for sweep end
        
        # End of sweep: renormalize properly
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        b = fast_convolve(a)
        cur_obj = b.max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3:
                # Perturbation
                rng = np.random.default_rng(sweep * 104729)
                scale = 0.005 * min(1.0, no_improve / 5.0)
                a = a * np.exp(rng.standard_normal(n) * scale)
                a = np.maximum(a, 0)
                a = a / a.sum()
                b = fast_convolve(a)
                if b.max() < best_obj:
                    best_obj = b.max()
                    best_a = a.copy()
                no_improve = 0
                if sweep > 20 and no_improve > 8:
                    break
    
    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    
    # Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    # Cosine-based
    x = np.arange(n, dtype=np.float64)
    c = np.abs(np.cos(np.pi * (x + 0.5) / n)) + 0.1
    inits.append(c / c.sum())
    
    # Gaussian centered
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Edge-weighted
    edge = 1.0 / (np.minimum(x, n - 1 - x).astype(float) + 1.0)
    inits.append(edge / edge.sum())
    
    # Multi-frequency
    mf = 1.0
    for freq in range(1, 5):
        mf += 0.3 * np.cos(2 * np.pi * freq * x / n)
    mf = np.abs(mf) + 0.1
    inits.append(mf / mf.sum())
    
    # From parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits


def peak_balance_optimize(a0, time_limit):
    """
    Try to equalize the top peaks using projected gradient on the "excess" objective.
    minimize sum_k max(0, b[k] - M)^2 for a target M slightly above the current max.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    b = fast_convolve(a)
    best_obj = b.max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    M_target = best_obj * 0.99  # Try to get below current max
    lr = 0.01
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        excess = np.maximum(b - M_target, 0.0)
        
        if excess.max() < 1e-14:
            M_target *= 0.98
            continue
        
        # Gradient of sum_k excess[k]^2 w.r.t. a[j]:
        # = 2 * sum_k excess[k] * db[k]/da[j]
        # = 2 * sum_k excess[k] * 2*a[k-j] (for valid k-j, k!=2j) + ...
        # Approximate: g[j] = 4 * sum_k excess[k] * a[k-j] (for valid range)
        # Compute via convolution-like operation
        g = np.zeros(n)
        for j in range(n):
            k_lo = max(0, j)
            k_hi = min(2 * n - 2, j + n - 1)
            g[j] = 4.0 * np.dot(excess[k_lo:k_hi + 1], a[k_lo - j:k_hi - j + 1])
        
        # Projected gradient step (onto simplex)
        a = a - lr * g
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        
        b = fast_convolve(a)
        cur_max = b.max()
        if cur_max < best_obj - 1e-14:
            best_obj = cur_max
            best_a = a.copy()
        
        lr *= 0.995
        if it % 100 == 0:
            M_target = max(best_obj * 0.97, best_obj - 0.01 * best_obj)
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: L-BFGS-B smooth optimization on parent and alternatives (~300s)
    Phase 2: Fast coordinate descent polish (~500s)
    Phase 3: Multi-length exploration (~300s)
    Phase 4: Peak balancing (~200s)
    Phase 5: Final coordinate descent (~350s)
    Phase 6: Safety margin (~150s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_parent = len(parent)
            print(f"Parent loaded: n={n_parent}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")
        n_parent = 384
        parent = np.ones(n_parent) / n_parent
    
    if best_seq is None:
        best_seq = parent.copy()
        best_score = evaluate_sequence(best_seq.tolist())
    
    # === Phase 1: L-BFGS-B smooth optimization (~300s) ===
    print("\n=== Phase 1: L-BFGS-B Smooth Optimization ===")
    n_best = len(best_seq)
    
    rem = deadline - time.monotonic() - 1300
    if rem > 60:
        t_alloc = min(rem, 300)
        print(f"  L-BFGS-B on best (n={n_best}), t={t_alloc:.0f}s")
        a_opt, _ = lbfgsb_smooth_optimize(best_seq, time_limit=t_alloc,
                                           T_start=10.0, T_end=200.0, n_T_steps=6)
        try_seq(a_opt, "lbfgsb_parent")
    
    # Try a couple alternative inits
    rem = deadline - time.monotonic() - 1200
    inits = generate_inits(n_best, parent, rng)
    for ii in range(1, min(4, len(inits))):
        rem = deadline - time.monotonic() - 1100
        if rem < 30:
            break
        t_alloc = min(rem, 60)
        print(f"  L-BFGS-B init {ii}, t={t_alloc:.0f}s")
        a_opt, _ = lbfgsb_smooth_optimize(inits[ii], time_limit=t_alloc,
                                           T_start=10.0, T_end=100.0, n_T_steps=4)
        try_seq(a_opt, f"lbfgsb_init{ii}")
    
    print(f"  Phase 1 best: {best_score:.8f}")
    
    # === Phase 2: Fast coordinate descent (~500s) ===
    print("\n=== Phase 2: Coordinate Descent ===")
    rem = deadline - time.monotonic() - 800
    if rem > 60:
        t_alloc = min(rem, 500)
        print(f"  Coord descent on best (n={len(best_seq)}), t={t_alloc:.0f}s")
        a_opt, _ = coord_descent_fast(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "coord1")
    
    # Another pass with perturbation
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_alloc = min(rem, 200)
        # Perturb and run coord descent again
        perturbed = best_seq * np.exp(rng.standard_normal(len(best_seq)) * 0.005)
        perturbed = np.maximum(perturbed, 0)
        perturbed = perturbed / perturbed.sum()
        print(f"  Coord descent on perturbed, t={t_alloc:.0f}s")
        a_opt, _ = coord_descent_fast(perturbed, time_limit=t_alloc)
        try_seq(a_opt, "coord_pert")
    
    print(f"  Phase 2 best: {best_score:.8f}")
    
    # === Phase 3: Multi-length exploration (~300s) ===
    print("\n=== Phase 3: Multi-Length Exploration ===")
    for n_try in [512, 768, 1024]:
        rem = deadline - time.monotonic() - 400
        if rem < 40:
            break
        t_alloc = min(rem, 120)
        if t_alloc < 20:
            break
        print(f"  Trying n={n_try}, t={t_alloc:.0f}s")
        
        # Initialize from parent
        x_new = np.linspace(0, 1, n_try)
        x_old = np.linspace(0, 1, len(parent))
        a_init = np.interp(x_new, x_old, parent)
        a_init = np.maximum(a_init, 0)
        a_init = a_init / a_init.sum()
        
        # Quick L-BFGS-B
        t1 = t_alloc * 0.5
        a_opt, _ = lbfgsb_smooth_optimize(a_init, time_limit=t1,
                                           T_start=10.0, T_end=100.0, n_T_steps=4)
        score_try = try_seq(a_opt, f"lbfgsb_n{n_try}")
        
        # If good, do coord descent
        rem = deadline - time.monotonic() - 300
        if rem > 20 and score_try < best_score + 0.01:
            t2 = min(rem, t_alloc * 0.5)
            a_opt2, _ = coord_descent_fast(a_opt, time_limit=t2)
            try_seq(a_opt2, f"coord_n{n_try}")
    
    print(f"  Phase 3 best: {best_score:.8f} (n={len(best_seq)})")
    
    # === Phase 4: Peak balancing (~200s) ===
    print("\n=== Phase 4: Peak Balancing ===")
    rem = deadline - time.monotonic() - 250
    if rem > 30:
        t_alloc = min(rem, 200)
        print(f"  Peak balance, t={t_alloc:.0f}s")
        a_opt, _ = peak_balance_optimize(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "peak_balance")
    
    print(f"  Phase 4 best: {best_score:.8f}")
    
    # === Phase 5: Final coordinate descent (~300s) ===
    print("\n=== Phase 5: Final Coord Descent ===")
    rem = deadline - time.monotonic() - 100
    if rem > 30:
        t_alloc = min(rem, 300)
        print(f"  Final coord descent, t={t_alloc:.0f}s")
        a_opt, _ = coord_descent_fast(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "final_coord")
    
    # Another L-BFGS-B pass with high T
    rem = deadline - time.monotonic() - 60
    if rem > 30:
        t_alloc = min(rem, 80)
        print(f"  Final L-BFGS-B high T, t={t_alloc:.0f}s")
        a_opt, _ = lbfgsb_smooth_optimize(best_seq, time_limit=t_alloc,
                                           T_start=50.0, T_end=500.0, n_T_steps=4)
        try_seq(a_opt, "final_lbfgsb")
    
    print(f"  Phase 5 best: {best_score:.8f}")
    
    # === Phase 6: Final micro-perturbations ===
    print("\n=== Phase 6: Micro Perturbations ===")
    rem = deadline - time.monotonic() - 30
    for _ in range(20):
        rem = deadline - time.monotonic() - 20
        if rem < 10:
            break
        scale = rng.uniform(0.0005, 0.003)
        perturbed = best_seq * np.exp(rng.standard_normal(len(best_seq)) * scale)
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps > 1e-15:
            perturbed /= ps
            t_min = min(rem, 10)
            a_opt, _ = coord_descent_fast(perturbed, time_limit=t_min)
            try_seq(a_opt, "micro_pert")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
