dc[1]/da[1] = 2*a[0]*I(0<=0<=n-1) = 2*x. ✓ (d/dy = 2x)

c[2] = sum_i a[i]*a[2-i]*I(0<=2-i<=n-1). For i=0: a[0]*a[2]. For i=1: a[1]*a[1]. For i=2: a[2]*a[0]. So c[2] = xz + y^2 + zx = 2xz + y^2.
dc[2]/da[0] = 2*a[2]*I(0<=2<=n-1) = 2*z. Check: d(2xz+y^2)/dx = 2z. ✓
dc[2]/da[1] = 2*a[1]*I(0<=1<=n-1) = 2*y. Check: d(2xz+y^2)/dy = 2y. ✓
dc[2]/da[2] = 2*a[0]*I(0<=0<=n-1) = 2*x. Check: d(2xz+y^2)/dz = 2x. ✓

So the formula is: `dc[k]/da[j] = 2 * a[k-j]` if `0 <= k-j < n`, else 0.

This is equivalent to saying the gradient of c with respect to a is `2 * convolve(a, reversed_a)`... no, it's simpler than that.

The gradient vector `g_a` has components:
`g_a[j] = sum_k (weight_k) * dc[k]/da[j] = sum_k (weight_k) * 2 * a[k-j] * I(0 <= k-j < n)`

where `weight_k` are the weights from the log-sum-exp (or whatever we're using to combine the peaks).

`g_a[j] = 2 * sum_{k: 0<=k-j<n} weight_k * a[k-j] = 2 * sum_{m: 0<=m<n} weight_{j+m} * a[m]`

(where `weight_k = 0` for k outside [0, 2n-2]).

This is a convolution-like operation! `g_a = 2 * convolve(a, weight_reversed)` (with appropriate zero-padding).

So the gradient computation is O(n log n) using FFT! This makes the log-sum-exp approach feasible.

    # grad[j] = 2 * sum_k w[k] * a[k-j] * I(0<=k-j<n)
    # = 2 * convolve(rev_a, w)[n-1 : 3n-2]
    rev_a = a[::-1]
    fft_size = 1 << (n + len(w) - 1 - 1).bit_length()
    RA = np.fft.rfft(rev_a, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(RA * W, fft_size)
    grad = 2.0 * conv[n - 1: n - 1 + n]
    return grad


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_optimize(f0, time_limit, lr_init=0.05, eps_init=10.0, eps_min=0.5,
                 eps_decay=1.003, lr_decay=0.9995, restart_interval=200):
    """Log-sum-exp gradient optimization with multiplicative weight updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_eps = eps_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Compute LSE gradient
        grad = compute_lse_gradient(conv, f, cur_eps)

        # Multiplicative weight update
        f_new = f * np.exp(-cur_lr * grad)
        f_new = np.maximum(f_new, 1e-30)
        f_new = f_new / f_new.sum()

        # Line search: try a few step sizes
        best_f_try = f_new
        best_obj_try = M
        for mult in [1.0, 0.5, 2.0, 0.25]:
            f_try = f * np.exp(-cur_lr * mult * grad)
            f_try = np.maximum(f_try, 1e-30)
            f_try = f_try / f_try.sum()
            obj_try = fast_convolve(f_try).max()
            if obj_try < best_obj_try:
                best_obj_try = obj_try
                best_f_try = f_try

        f = best_f_try
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update hyperparameters
        cur_eps = min(cur_eps * eps_decay, 500.0)
        cur_lr *= lr_decay

        # Restart if stuck
        if no_improve > restart_interval:
            rng = np.random.default_rng(hash((it, n)) % (2**31))
            noise_scale = 0.003 + 0.005 * min(1.0, no_improve / (restart_interval * 3))
            f = best_f * np.exp(rng.standard_normal(n) * noise_scale)
            f = f / f.sum()
            cur_lr = max(lr_init * 0.5, 0.01)
            cur_eps = eps_init * 0.5
            no_improve = 0

    return best_f, best_obj


def peak_shave_optimize(f0, time_limit, top_k=5, lr_init=0.02, decay=0.999):
    """Peak-shaving: focus gradient on the top-k peaks only."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Select top-k peaks
        k_eff = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_eff)[-k_eff:]
        
        # Uniform weights on top peaks
        w = np.zeros(len(conv))
        w[peaks] = 1.0 / k_eff

        # Compute gradient with these weights
        rev_a = f[::-1]
        fft_size = 1 << (n + len(w) - 1 - 1).bit_length()
        RA = np.fft.rfft(rev_a, fft_size)
        W = np.fft.rfft(w, fft_size)
        conv_g = np.fft.irfft(RA * W, fft_size)
        grad = 2.0 * conv_g[n - 1: n - 1 + n]

        # Multiplicative update
        f_new = f * np.exp(-cur_lr * grad)
        f_new = np.maximum(f_new, 1e-30)
        f_new = f_new / f_new.sum()
        f = f_new

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= decay

        if no_improve > 100:
            rng = np.random.default_rng(it * 7 + n)
            f = best_f * np.exp(rng.standard_normal(n) * 0.003)
            f = f / f.sum()
            cur_lr = max(lr_init * 0.5, 0.002)
            no_improve = 0

    return best_f, best_obj


def block_coordinate_descent(f0, time_limit, block_size=16, n_blocks_per_iter=4):
    """Optimize blocks of consecutive coordinates via small-scale Nelder-Mead."""
    from scipy.optimize import minimize
    
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Randomly select blocks to optimize
        rng = np.random.default_rng(it * 13 + n)
        for _ in range(n_blocks_per_iter):
            start = rng.integers(0, max(1, n - block_size))
            end = min(start + block_size, n)
            block_idx = np.arange(start, end)
            
            # Optimize this block keeping rest fixed
            fixed = f.copy()
            
            def block_obj(x):
                temp = fixed.copy()
                temp[block_idx] = x
                temp = np.maximum(temp, 0.0)
                ts = temp.sum()
                if ts < 1e-15:
                    return 1e10
                temp = temp / ts
                return fast_convolve(temp).max()
            
            x0 = f[block_idx].copy()
            res = minimize(block_obj, x0, method='Nelder-Mead',
                          options={'maxiter': 50, 'xatol': 1e-8, 'fatol': 1e-10})
            if res.fun < block_obj(x0):
                f[block_idx] = np.maximum(res.x, 0.0)
                f = f / f.sum()

        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()

    return best_f, best_obj


def generate_asymmetric_inits(n, rng):
    """Generate structured asymmetric initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Linear ramp
    a = (x + 1)
    inits.append(a / a.sum())
    
    # Quadratic ramp
    a = (x + 1) ** 2
    inits.append(a / a.sum())
    
    # Exponential ramp (various alphas)
    for alpha in [1.0, 2.0, 3.0, 5.0]:
        a = np.exp(alpha * x / n)
        inits.append(a / a.sum())
    
    # Sigmoid shapes
    for beta in [2.0, 4.0, 6.0]:
        a = 1.0 / (1.0 + np.exp(-beta * (x / n - 0.5)))
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # Two-level
    for frac in [0.3, 0.4, 0.5, 0.6]:
        m = int(n * frac)
        a = np.ones(n)
        a[:m] = 1.0 + 0.5
        a[m:] = 1.0
        inits.append(a / a.sum())
    
    # Cosine window (asymmetric)
    a = 0.5 * (1 + np.cos(np.pi * x / n))
    a = a ** 2  # Sharpen
    a = np.maximum(a, 1e-10)
    inits.append(a / a.sum())
    
    # Power law: a[i] = (i/n)^p
    for p in [0.5, 1.0, 2.0]:
        a = (x / n) ** p + 0.01
        inits.append(a / a.sum())
    
    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search using log-sum-exp smoothed
    gradient with multiplicative weight updates across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})", flush=True)
        return score

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.8f}", flush=True)
            try_seq(parent, "parent")
    except Exception:
        pass

    # === Phase 1: Quick exploration across multiple n and initializations ===
    print("\n=== Phase 1: Quick Exploration ===", flush=True)
    n_values = [256, 384, 512, 768]
    rem = deadline - time.monotonic() - 1200
    t_per_n = max(10, rem / (len(n_values) * 3))

    for n in n_values:
        rem = deadline - time.monotonic() - 1100
        if rem < 30:
            break
        print(f"  n={n}, t_available={rem:.0f}s", flush=True)
        
        # Generate initializations
        inits = generate_asymmetric_inits(n, rng)
        # Also try parent interpolated to this length
        if parent is not None:
            px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            px = np.maximum(px, 0)
            if px.sum() > 1e-10:
                inits.append(px / px.sum())
        
        # Try top few initializations
        for ii, init in enumerate(inits[:5]):
            rem = deadline - time.monotonic() - 1000
            if rem < 10:
                break
            t_alloc = min(t_per_n, rem)
            a_opt, _ = lse_optimize(init, time_limit=t_alloc, lr_init=0.08,
                                     eps_init=5.0, eps_min=0.3, eps_decay=1.005)
            try_seq(a_opt, f"n{n}_init{ii}")

    print(f"  Phase 1 best: {best_score:.8f}", flush=True)

    # === Phase 2: Deep optimization on best ===
    print("\n=== Phase 2: Deep Optimization ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 700
        if rem > 20:
            t_alloc = min(rem, 500)
            print(f"  LSE deep, t={t_alloc:.0f}s", flush=True)
            a_opt, _ = lse_optimize(best_seq, time_limit=t_alloc, lr_init=0.03,
                                     eps_init=3.0, eps_min=0.2, eps_decay=1.004,
                                     lr_decay=0.9997, restart_interval=300)
            try_seq(a_opt, "lse_deep")

            # Peak shaving
            rem = deadline - time.monotonic() - 500
            if rem > 20:
                t_alloc = min(rem, 300)
                print(f"  Peak shave, t={t_alloc:.0f}s", flush=True)
                a_opt, _ = peak_shave_optimize(best_seq, time_limit=t_alloc,
                                                top_k=8, lr_init=0.015, decay=0.999)
                try_seq(a_opt, "peak_shave")

                # More peak shaving with different k
                rem = deadline - time.monotonic() - 400
                if rem > 20:
                    a_opt, _ = peak_shave_optimize(best_seq, time_limit=min(rem, 200),
                                                    top_k=3, lr_init=0.01, decay=0.999)
                    try_seq(a_opt, "peak_shave_3")

    print(f"  Phase 2 best: {best_score:.8f}", flush=True)

    # === Phase 3: Block coordinate descent ===
    print("\n=== Phase 3: Block Coordinate Descent ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem, 300)
            print(f"  BCD, t={t_alloc:.0f}s, n={len(best_seq)}", flush=True)
            a_opt, _ = block_coordinate_descent(best_seq, time_limit=t_alloc,
                                                 block_size=12, n_blocks_per_iter=6)
            try_seq(a_opt, "bcd")

    print(f"  Phase 3 best: {best_score:.8f}", flush=True)

    # === Phase 4: Fine-tuning with very small steps ===
    print("\n=== Phase 4: Fine-Tuning ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem, 200)
            print(f"  Fine LSE, t={t_alloc:.0f}s", flush=True)
            a_opt, _ = lse_optimize(best_seq, time_limit=t_alloc, lr_init=0.005,
                                     eps_init=1.0, eps_min=0.1, eps_decay=1.002,
                                     lr_decay=0.9999, restart_interval=500)
            try_seq(a_opt, "fine_lse")

            # Peak shave with tiny steps
            rem = deadline - time.monotonic() - 60
            if rem > 15:
                a_opt, _ = peak_shave_optimize(best_seq, time_limit=min(rem, 100),
                                                top_k=4, lr_init=0.003, decay=0.9999)
                try_seq(a_opt, "fine_peak")

    print(f"  Phase 4 best: {best_score:.8f}", flush=True)

    # === Phase 5: Try different n values from best ===
    print("\n=== Phase 5: Length Exploration ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 30:
            # Try a larger n by interpolating
            for new_n in [len(best_seq) * 2, len(best_seq) + 128]:
                rem = deadline - time.monotonic() - 30
                if rem < 20:
                    break
                px = np.interp(np.linspace(0, 1, new_n),
                              np.linspace(0, 1, len(best_seq)), best_seq)
                px = np.maximum(px, 0)
                px = px / px.sum()
                t_alloc = min(rem, 80)
                a_opt, _ = lse_optimize(px, time_limit=t_alloc, lr_init=0.05,
                                         eps_init=3.0, eps_min=0.3, eps_decay=1.004)
                try_seq(a_opt, f"n{new_n}_interp")

    # === Phase 6: Final perturbations ===
    print("\n=== Phase 6: Final Perturbations ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        attempts = 0
        while rem > 5 and attempts < 10:
            attempts += 1
            for noise_scale in [0.001, 0.003, 0.01]:
                rem = deadline - time.monotonic() - 15
                if rem < 5:
                    break
                f_pert = best_seq * np.exp(rng.standard_normal(len(best_seq)) * noise_scale)
                f_pert = f_pert / f_pert.sum()
                t_alloc = min(rem, 30)
                a_opt, _ = lse_optimize(f_pert, time_limit=t_alloc, lr_init=0.01,
                                         eps_init=1.0, eps_min=0.2, eps_decay=1.002,
                                         lr_decay=0.9998, restart_interval=100)
                try_seq(a_opt, f"perturb_{noise_scale}")
                rem = deadline - time.monotonic() - 15
                if rem < 5:
                    break

    # Fallback
    if best_seq is None:
        n_fb = 384
        x = np.arange(n_fb, dtype=np.float64)
        best_seq = np.exp(2.0 * x / n_fb)
        best_seq = best_seq / best_seq.sum()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Fallback score: {best_score:.8f}", flush=True)

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
