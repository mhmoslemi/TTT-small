import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_max_gradient(p, conv, temp):
    """Gradient of T*log(sum exp(conv/T)) w.r.t. p (on the simplex)."""
    scaled = conv / temp
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    w = exp_s / exp_s.sum()
    n = len(p)
    # grad[i] = 2 * sum_m p[m] * w[m+i]
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(p, w[i:i+n])
    return grad

def softmax_x_to_p(x):
    x = x - x.max()
    ex = np.exp(x)
    return ex / ex.sum()

def p_to_x(p):
    p = np.maximum(p, 1e-30)
    return np.log(p)

def lbfgs_smoothed(p_init, temps, max_iter_per_temp=100, verbose=False):
    """Run L-BFGS-B on softmax parameterization with temperature annealing."""
    x = p_to_x(p_init)
    n = len(p_init)
    best_p = p_init.copy()
    best_val = fast_convolve(p_init).max()
    
    for t_idx, temp in enumerate(temps):
        def objective(x_var):
            p = softmax_x_to_p(x_var)
            conv = fast_convolve(p)
            return temp * np.log(np.sum(np.exp((conv - conv.max()) / temp)))
        
        def gradient(x_var):
            p = softmax_x_to_p(x_var)
            conv = fast_convolve(p)
            grad_p = smoothed_max_gradient(p, conv, temp)
            # Chain rule: d/dx_j = p_j * (grad_p[j] - dot(p, grad_p))
            return p * (grad_p - np.dot(p, grad_p))

                       options={'maxiter': max_iter_per_temp, 'gtol': 1e-6, 'ftol': 1e-12})

        conv_new = fast_convolve(p_new)
        val = conv_new.max()
        
        if val < best_val:
            best_val = val
            best_p = p_new.copy()
        
        x = res.x
        if verbose:
            print(f"    T={temp:.4f}: obj={objective(res.x):.8f}, max_conv={val:.10f}")
    
    return best_p, best_val

def mirror_descent(p_init, time_limit, init_lr=0.05, decay=0.98, top_k=32, temp=0.01):
    """Mirror descent (multiplicative weights) on the simplex."""
    p = np.maximum(p_init, 1e-15)
    p = p / p.sum()
    n = len(p)
    
    conv = fast_convolve(p)
    best_val = conv.max()
    best_p = p.copy()
    
    lr = init_lr
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            w = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            scaled = pv / temp
            scaled -= scaled.max()
            ew = np.exp(scaled)
            w = ew / ew.sum()
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            # sg[j] += 2*w[i]*p[k-j] for valid j
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * p[k - idx]
        
        # Mirror descent update: p_new = p * exp(-lr * sg) / Z
        log_p_new = np.log(p) - lr * sg
        log_p_new -= log_p_new.max()
        p_new = np.exp(log_p_new)
        p_new /= p_new.sum()
        
        conv_new = fast_convolve(p_new)
        val = conv_new.max()
        
        if val < best_val - 1e-14:
            best_val = val
            best_p = p_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                # Backtrack: reduce lr
                lr *= 0.5
                if lr < 1e-8:
                    # Restart with noise
                    rng = np.random.default_rng(int(time.monotonic() * 1000))
                    p = project_simplex(best_p + rng.standard_normal(n) * 0.005)
                    conv = fast_convolve(p)
                    lr = init_lr * 0.1
                    no_improve = 0
                continue
        
        p = p_new
        conv = conv_new
        lr *= decay
    
    return best_p, best_val

def solve_lp_oracle(f, M, time_limit=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def fw_lp_step(f, M, verbose=False):
    """One Frank-Wolfe step with LP oracle."""
    n = len(f)
    g = solve_lp_oracle(f, M, time_limit=20)
    if g is None:
        return f, M
    gs = g.sum()
    if gs < 1e-15:
        return f, M
    g = g / gs
    
    # Line search
    best_alpha = 0.0
    best_val = M
    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        ft = (1 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0)
        s = ft.sum()
        if s < 1e-15:
            continue
        ft = ft / s
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    if best_alpha > 0:
        f_new = (1 - best_alpha) * f + best_alpha * g
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        return f_new, best_val
    return f, M

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed L-BFGS optimization with temperature annealing on softmax parameterization
2. Mirror descent (multiplicative weights) for refinement
3. LP-based Frank-Wolfe for global moves
4. Multi-start with diverse initializations and adaptive time allocation

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_conv_obj_and_grad(x, temp):
    """
    Objective and gradient for softmax parameterization.
    x: parameter vector (not yet softmax'd)
    Returns (objective_value, gradient_wrt_x)
    """
    x = x - x.max()
    ex = np.exp(x)
    p = ex / ex.sum()
    n = len(p)

    conv = fast_convolve(p)

    # Smoothed objective: T * log(sum exp(conv/T))
    scaled = conv / temp
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    w = exp_s / exp_s.sum()
    obj = temp * (scaled.max() + np.log(exp_s.sum()))

    # Gradient of sum_k w_k * conv[k] w.r.t. p_i
    # = 2 * sum_m p[m] * w[m+i]
    grad_p = np.zeros(n)
    for i in range(n):
        grad_p[i] = 2.0 * np.dot(p, w[i:i + n])

    # Chain rule for softmax: d/dx_j = p_j * (grad_p[j] - dot(p, grad_p))
    mean_g = np.dot(p, grad_p)
    grad_x = p * (grad_p - mean_g)

    return obj, grad_x


def lbfgs_optimize(p_init, temps, max_iter_per_temp=80, verbose=False):
    """
    Run L-BFGS-B on softmax parameterization with temperature annealing.
    p_init: initial point on the simplex
    temps: list of temperatures (high to low)
    Returns: (best_p, best_max_conv)
    """
    p_init = np.maximum(p_init, 1e-15)
    p_init = p_init / p_init.sum()
    x = np.log(p_init)
    n = len(p_init)

    best_p = p_init.copy()
    best_val = fast_convolve(p_init).max()
    clip_val = best_val

    for t_idx, temp in enumerate(temps):
        res = minimize(
            lambda xx: smoothed_conv_obj_and_grad(xx, temp)[0],
            x,
            jac=lambda xx: smoothed_conv_obj_and_grad(xx, temp)[1],
            method='L-BFGS-B',
            options={'maxiter': max_iter_per_temp, 'gtol': 1e-7, 'ftol': 1e-14}
        )

        x_new = res.x - res.x.max()
        ex = np.exp(x_new)
        p_new = ex / ex.sum()
        p_new = np.maximum(p_new, 0)
        p_new = p_new / p_new.sum()

        conv_new = fast_convolve(p_new)
        val = conv_new.max()

        if val < best_val:
            best_val = val
            best_p = p_new.copy()

        x = res.x

        # Also try the exact max at this point
        if val < clip_val:
            clip_val = val

        if verbose:
            print(f"      T={temp:.5f}: smooth_obj={res.fun:.8f}, max_conv={val:.10f}")

    return best_p, best_val


def mirror_descent(p_init, time_limit, init_lr=0.05, decay=0.995, top_k=32, temp=0.01):
    """Mirror descent (multiplicative weights) on the simplex."""
    p = np.maximum(p_init, 1e-15).astype(np.float64)
    p = p / p.sum()
    n = len(p)

    conv = fast_convolve(p)
    best_val = conv.max()
    best_p = p.copy()

    lr = init_lr
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        # Compute weighted subgradient
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            w = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            scaled = pv / temp
            scaled -= scaled.max()
            ew = np.exp(scaled)
            w = ew / ew.sum()

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * p[k - idx]

        # Mirror descent update
        log_p_new = np.log(p) - lr * sg
        log_p_new -= log_p_new.max()
        p_new = np.exp(log_p_new)
        p_new /= p_new.sum()

        conv_new = fast_convolve(p_new)
        val = conv_new.max()

        if val < best_val - 1e-15:
            best_val = val
            best_p = p_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                lr *= 0.5
                if lr < 1e-10:
                    rng = np.random.default_rng(it * 7919 + 13)
                    p = project_simplex(best_p + rng.standard_normal(n) * 0.003)
                    conv = fast_convolve(p)
                    lr = init_lr * 0.05
                    no_improve = 0
                continue

        p = p_new
        conv = conv_new
        lr *= decay

    return best_p, best_val


def solve_lp_oracle(f, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_lp_iteration(p_init, time_limit):
    """Run Frank-Wolfe with LP oracle for a time limit."""
    p = np.maximum(p_init, 0).astype(np.float64)
    p = p / p.sum()
    n = len(p)

    conv = fast_convolve(p)
    M = conv.max()
    best_p = p.copy()
    best_M = M

    t0 = time.monotonic()
    m_expand = 1.0

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        g = solve_lp_oracle(p, M * m_expand, time_limit=min(15, rem - 3))
        if g is None:
            m_expand *= 1.3
            if m_expand > 10.0:
                m_expand = 1.0
            continue

        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        if np.max(np.abs(g - p)) < 1e-10:
            m_expand = min(m_expand * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * p + alpha * g
            ft = np.maximum(ft, 0)
            s = ft.sum()
            if s < 1e-15:
                continue
            ft = ft / s
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            p = (1.0 - best_alpha) * p + best_alpha * g
            p = np.maximum(p, 0)
            p = p / p.sum()
            conv = fast_convolve(p)
            M = conv.max()
            m_expand = 1.0

            if M < best_M:
                best_M = M
                best_p = p.copy()
        else:
            m_expand = min(m_expand * 1.1, 5.0)

    return best_p, best_M


def generate_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine
    c = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.005
    inits.append(c / c.sum())

    # Gaussian
    center = (n - 1) / 2.0
    g = np.exp(-0.5 * ((x - center) / (n / 4.0)) ** 2)
    inits.append(g / g.sum())

    # Sine (Hann window)
    s = 0.5 * (1 - np.cos(2 * np.pi * x / n)) + 0.005
    inits.append(s / s.sum())

    # Power law (heavy tail)
    pw = (x + 1.0) ** (-0.5)
    inits.append(pw / pw.sum())

    # Inverse power law
    ipw = (n - x + 1.0) ** (-0.3)
    inits.append(ipw / ipw.sum())

    # Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # Dirichlet (spread)
    inits.append(rng.dirichlet(np.ones(n) * 3.0))

    # Parent-based
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                inits.append(px / ps)

    return inits


def symmetrize(p):
    """Make sequence symmetric: p'[i] = (p[i] + p[n-1-i]) / 2."""
    n = len(p)
    p_sym = 0.5 * (p + p[::-1])
    p_sym = np.maximum(p_sym, 0)
    s = p_sym.sum()
    if s > 1e-15:
        p_sym = p_sym / s
    return p_sym

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.ones(384) / 384

    final_score = float(evaluate_sequence(best_seq.tolist()))
    if not np.isfinite(final_score):
        best_seq = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"FINAL SCORE: {final_score:.8f} (n={len(best_seq)})")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    return [float(x) for x in best_seq]
