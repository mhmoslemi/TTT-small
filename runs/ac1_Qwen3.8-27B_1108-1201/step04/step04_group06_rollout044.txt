from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build matrix C such that C @ g = convolve(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j+n, j] = a
    return C

def lp_with_dual(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0. Returns (g, dual_vars, success)."""
