Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed gradient descent with simplex projection
2. Continuation method (decreasing smoothing parameter)
3. CVXPY convex optimization for final polishing
4. Multiple initializations from different parametric families

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def proj_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 1 and np.all(v >= 0):
        # Already on or inside the simplex (but we need sum = 1)
        # Actually if sum < 1, we need to adjust
        pass
    # Standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def compute_soft_gradient(a, w):
    """Compute gradient of log-sum-exp given softmax weights w."""
    # g[i] = 2 * sum_k w[k] * a[k-i] for valid k-i
    # = 2 * correlate(w, a, mode='full')[i]
    n = len(a)
    g = 2.0 * np.correlate(w, a, mode='full')
    return g[:n]

def logsumexp(c, mu):
    """Compute mu * logsumexp(c/mu) stably."""
    c_scaled = c / mu
    c_max = c_scaled.max()
    return mu * (c_max + np.log(np.exp(c_scaled - c_max).sum()))

def soft_max_descent(a0, time_limit, mu_init=None, mu_min=1e-4, verbose=False):

    Minimize max(conv(a,a)) using log-sum-exp smoothing and projected gradient.
    a is normalized to sum to 1.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
