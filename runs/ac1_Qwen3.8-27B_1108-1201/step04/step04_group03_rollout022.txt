Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast coordinate descent
with incremental convolution updates, combined with FW-LP for big moves
and random restarts. Key innovation: exact 1D minimax subproblem solved
via bisection on the convex 1D function, maintaining the convolution

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    Coordinate descent optimization with random ordering and restarts.
    """

    s = f.sum()
    if s < 1e-15:

    else:

    n = len(f)

            best_f = f.copy()

        if best_obj - obj < 1e-14:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Gerchberg-Saxton (flat spectrum design) for initialization
2. Fast coordinate descent with incremental convolution updates
3. Frank-Wolfe with LP oracle for global moves
4. Random restarts and multi-scale search

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def gs_flat_spectrum(f0, n_iters=5000, relaxation=1.0, seed=42):
    """
    Gerchberg-Saxton algorithm for flat magnitude spectrum.
    Projects between flat-spectrum set and non-negative set.
    """
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    N = 1 << (2 * n - 1).bit_length()  # FFT size for linear conv
    target_mag = 1.0 / np.sqrt(N)  # For flat spectrum of a normalized sequence
    
    rng = np.random.default_rng(seed)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    for it in range(n_iters):
        # FFT
        A = np.fft.fft(f, N)
        
        # Replace magnitude with target
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        phase = A / mag
        A_new = target_mag * phase
        # Make DC component real and positive
        A_new[0] = np.abs(A_new[0])
        
        # IFFT
        f_new = np.fft.ifft(A_new).real[:n]
        
        # Project onto non-negative simplex
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            f = f  # Keep old
        else:
            f_new = f_new / s
            # Relaxed update
            f = (1 - relaxation) * f + relaxation * f_new
        
        # Track best
        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj:
            best_obj = obj
            best_f = f.copy()
    
    return best_f, best_obj

def coord_descent_pass(f, conv, order):
    """One pass of coordinate descent. Returns updated f and conv."""
    n = len(f)
    for i in order:
        i = int(i)
        x_old = f[i]
        if x_old < 1e-15:
            continue  # Skip zero coordinates
        
        # Determine affected range
        k_min = max(0, i)
        k_max = min(2 * n - 2, i + n - 1)
        
        ks = np.arange(k_min, k_max + 1)
        ki = ks - i  # k - i values
        
        # Compute coefficients
        # c_k(x) = a_k * x^2 + b_k * x + d_k
        a_vec = np.zeros(len(ks))
        b_vec = np.zeros(len(ks))
        
        mask_quad = (ki == i)  # k = 2i
        a_vec[mask_quad] = 1.0
        # b[k] = 2 * f[k-i] for non-quadratic terms
        mask_lin = ~mask_quad
        b_vec[mask_lin] = 2.0 * f[ki[mask_lin]]
        
        # d = conv[ks] - a*x_old^2 - b*x_old
        d_vec = conv[ks] - a_vec * x_old**2 - b_vec * x_old
        
        # Solve min_x max_k (a_k*x^2 + b_k*x + d_k) for x >= 0
        # Use bisection on the subgradient
        # First, find a good bracket
        
        def eval_at(x):
            return np.max(a_vec * x * x + b_vec * x + d_vec)
        
        # Try current value and neighbors
        candidates_x = np.array([0.0, x_old * 0.5, x_old * 0.75, x_old,
                                  x_old * 1.25, x_old * 1.5, x_old * 2.0])
        best_x = x_old
        best_val = eval_at(x_old)
        for xc in candidates_x:
            if xc < 0:
                continue
            vc = eval_at(xc)
            if vc < best_val:
                best_val = vc
                best_x = xc
        
        # Golden section refinement
        if best_x < 1e-15:
            lo, hi = 0.0, max(x_old * 0.5, 1e-10)
        else:
            span = max(x_old * 0.5, 1e-12)
            lo = max(0.0, best_x - span)
            hi = best_x + span
        
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(30):
            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)
            f1 = eval_at(x1)
            f2 = eval_at(x2)
            if f1 < f2:
                hi = x2
            else:
                lo = x1
            if hi - lo < 1e-13 * max(1.0, best_x):
                break
        
        x_new = max(0.0, (lo + hi) / 2.0)
        val_new = eval_at(x_new)
        if val_new < best_val:
            best_x = x_new
        
        # Update
        f[i] = best_x
        conv[ks] = a_vec * best_x**2 + b_vec * best_x + d_vec
        conv[ks] = np.maximum(conv[ks], 0.0)
    
    return f, conv

def coord_descent_optimize(f0, time_limit, verbose=True):
    """Coordinate descent with re-normalization and periodic recompute."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    pass_count = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        order = rng.permutation(n)
        f, conv = coord_descent_pass(f, conv, order)
        
        # Renormalize and recompute
        s = f.sum()
        if s < 1e-15:
            break
        f = f / s
        conv = fast_convolve(f)
        
        obj = conv.max()
        pass_count += 1
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and pass_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass {pass_count}, obj={best_obj:.10f}, elapsed={elapsed:.1f}s")
        
        if no_improve > 200:
            # Random restart
            pert = rng.standard_normal(n) * (0.005 + 0.01 * min(1.0, no_improve / 500.0))
            f = project_simplex(best_f + pert)
            conv = fast_convolve(f)
            no_improve = 0
    
    return best_f, best_obj

def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[j]*g[k-j]
    # For each k, the constraint is: sum_{j} f[min(n-1,k)]... 
    
    # Actually, (f*g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[j] * g[k-j]
    # Let l = k-j, then j = k-l, and the constraint is:
    # sum_{l} f[k-l] * g[l] <= M
    
    # For a given k, the coefficients of g[l] are f[k-l] (when 0<=k-l<n and 0<=l<n)
    # So the row k of A_ub has: A_ub[k, l] = f[k-l] if 0 <= k-l < n else 0
    
    A_ub = np.zeros((2 * n - 1, n))

        ks = np.arange(l, l + n)
        A_ub[ks, l] = f[np.arange(l + n - 1, l - 1, -1)]
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gs_optimize(f0, time_limit, seed=42, verbose=True):
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    N = 1 << (2 * n - 1).bit_length()
    target_mag = 1.0 / np.sqrt(N)

    rng = np.random.default_rng(seed)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    it = 0
    relax = 0.8

    while time.monotonic() - t0 < time_limit:
        it += 1

        A = np.fft.fft(f, N)
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        phase = A / mag
        A_new = target_mag * phase
        A_new[0] = np.abs(A_new[0])

        f_new = np.fft.ifft(A_new).real[:n]
        f_new = np.maximum(f_new, 0.0)
        s2 = f_new.sum()
        if s2 < 1e-15:
            f_new = f
        else:
            f_new = f_new / s2

        f = (1.0 - relax) * f + relax * f_new

        if it % 50 == 0:
            conv = fast_convolve(f)
            obj = conv.max()
            if obj < best_obj:
                best_obj = obj
                best_f = f.copy()
                if verbose:
                    print(f"    GS it={it}, obj={best_obj:.10f}, t={time.monotonic()-t0:.1f}s")
            # Perturb if stuck
            if it % 500 == 0:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.01)

    conv = fast_convolve(f)
    obj = conv.max()
    if obj < best_obj:
        best_obj = obj
        best_f = f.copy()

    return best_f, best_obj


def lsx_objective_and_grad(z, beta, n):
    """Compute log-sum-exp smoothed objective and gradient w.r.t. z (softmax param)."""
    z_max = z.max()
    ez = np.exp(z - z_max)
    f = ez / ez.sum()

    conv = fast_convolve(f)

    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # Objective: 2*n/β * log(sum exp(β*c_k))
    # = 2*n/β * (β*max(conv)/β + log(sum exp(β*(conv - max))))
    # Simplify: 2*n * (max(conv) + (1/β)*log(sum exp(β*(conv - max(conv)))))
    obj = 2.0 * n * (conv.max() + np.log(w_sum) / beta)

    # Gradient w.r.t. f: g_j = 2n * sum_k w_k * 2*f[k-j] = 4n * (w cross-correl f)_j
    # cross_correl(j) = sum_l f[l] * w[j+l]
    # Use FFT: N >= n + (2n-1) - 1 = 3n-2
    N2 = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, N2)
    F = np.fft.rfft(f, N2)
    corr = np.fft.irfft(W * np.conj(F), N2)[:n]
    g = 4.0 * n * corr

    # Gradient w.r.t. z: grad_z = f * (g - g.dot(f))
    gf = g.dot(f)
    grad_z = f * (g - gf)

    return obj, grad_z


def lsx_optimize(f0, time_limit, beta_init=5.0, beta_max=500.0, verbose=True):
    """Log-sum-exp smoothed optimization with softmax parameterization."""
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    # Initialize z from f
    z = np.full(n, -10.0)
    mask = f > 1e-10
    z[mask] = np.log(f[mask] + 1e-30)

    best_f = f.copy()
    best_true_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    total_time = time_limit
    it = 0
    no_improve = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / 30000.0)
        beta = beta_init * (beta_max / beta_init) ** progress

        # Use scipy minimize with a few iterations of L-BFGS-B
        res = minimize(
            lambda z_: lsx_objective_and_grad(z_, beta, n)[0],
            z,
            jac=lambda z_: lsx_objective_and_grad(z_, beta, n)[1],
            method='L-BFGS-B',
            options={'maxiter': 5, 'ftol': 1e-12, 'gtol': 1e-8}
        )

        z = res.x
        # Get f from z
        z_max = z.max()
        ez = np.exp(z - z_max)
        f = ez / ez.sum()

        # Check true objective
        conv = fast_convolve(f)
        true_obj = conv.max()

        if true_obj < best_true_obj - 1e-14:
            best_true_obj = true_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 500 == 0:
            print(f"    LSX it={it}, beta={beta:.1f}, true_obj={best_true_obj:.10f}, t={time.monotonic()-t0:.1f}s")

        if no_improve > 500:
            # Restart with perturbation
            pert = rng.standard_normal(n) * 0.02
            f_pert = project_simplex(best_f + pert)
            z = np.full(n, -10.0)
            mask = f_pert > 1e-10
