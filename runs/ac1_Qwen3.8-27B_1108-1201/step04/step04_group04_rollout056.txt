def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coord_step(f, c, i):
    """Optimize coordinate i. Returns updated f and c."""
    n = len(f)
    l_start = i  # since i >= 0
    l_end = min(2*n - 2, i + n - 1)
    
    # Max of unaffected c_k
    C = 0.0

        C = max(C, float(c[:l_start].max()))
    if l_end < 2*n - 2:

    has_quad = (2*i >= l_start and 2*i <= l_end)
    f_i_old = f[i]
    
    # Precompute the slice we'll use repeatedly
    sl = slice(l_start, l_end+1)
    f_sl = f[l_start - i : l_end + 1 - i]  # f[k-i] for k in [l_start, l_end]
    c_sl = c[sl]

        if has_quad:
            vals = vals.copy()
            vals[2*i - l_start] += t*t

            m = C
        return m

    lo = -f_i_old

            m2 = m1

            m1 = m2
            f_m1 = f_m2
            m2 = a + gr * (b - a)

    return f, c


def coord_descent(f, max_sweeps, rng, time_limit=None):
    """Run coordinate descent for a given number of sweeps."""
    n = len(f)
    f = f.copy()
    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        order = rng.permutation(n)
        for i in order:
            f, c = coord_step(f, c, i)
        
        obj = c.max()
        if obj < best_obj:
            best_obj = obj
            best_f = f.copy()
    
    # Normalize for return
    s = best_f.sum()
    if s > 1e-15:
        best_f = best_f / s
    
    return best_f, best_obj


def solve_lp_dir(f, M):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c_vec = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def fw_step(f, M, m_exp=1.0):
    """One Frank-Wolfe step with LP direction."""
    n = len(f)
    g = solve_lp_dir(f, M * m_exp)
    if g is None:
        return None, m_exp * 1.1
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None, m_exp
    g = g / g_sum
    
    # Line search
    best_alpha = 0.0
    best_val = fast_convolve(f).max()
    for alpha in [0.1, 0.3, 0.5, 0.7, 0.9]:
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    if best_alpha > 1e-10:
        f_new = (1.0 - best_alpha) * f + best_alpha * g
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        return f_new, 1.0
    
    return None, min(m_exp * 1.05, 3.0)

"""
Coordinate descent optimizer for minimizing 2*n*max(auto_convolve(a))/sum(a)^2.
Uses exact 1D golden-section coordinate optimization, LP-based Frank-Wolfe for
large moves, periodic Gaussian perturbations, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coord_step(f, c, i):
    """Optimize coordinate i of f given current autocorrelation c.
    Uses golden-section search on the convex 1D subproblem.
    Modifies f and c in-place."""
    n = len(f)
    l_start = i
    l_end = min(2 * n - 2, i + n - 1)

    # Max of unaffected c_k
    C = 0.0
    if l_start > 0:
        C = max(C, float(c[:l_start].max()))
    if l_end < 2 * n - 2:
        C = max(C, float(c[l_end + 1:].max()))

    has_quad = (2 * i >= l_start and 2 * i <= l_end)
    f_i_old = f[i]

    # Precompute slices
    f_sl = f[l_start - i: l_end + 1 - i]
    c_sl = c[l_start: l_end + 1]
    quad_idx = 2 * i - l_start if has_quad else -1

    # Golden section search
    lo = -f_i_old
    hi = f_i_old * 2.0 + 0.05
    if hi < 0.01:
        hi = 0.01

    # Ensure hi is beyond the minimum
    def obj_at(t):
        vals = c_sl + 2.0 * t * f_sl
        if has_quad:
            m = float(np.maximum(vals[quad_idx] + t * t, np.delete(vals, quad_idx)).max())
        else:
            m = float(vals.max())
        return m if m > C else C

    obj_lo = obj_at(lo)
    obj_hi = obj_at(hi)
    expand = 0
    while obj_hi < obj_lo and expand < 25:
        hi = hi * 2.0 + 0.02
        obj_hi = obj_at(hi)
        expand += 1

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_gs = lo
    b_gs = hi
    m1 = b_gs - gr * (b_gs - a_gs)
    m2 = a_gs + gr * (b_gs - a_gs)
    f_m1 = obj_at(m1)
    f_m2 = obj_at(m2)

    for _ in range(60):
        if f_m1 < f_m2:
            b_gs = m2
            m2 = m1
            f_m2 = f_m1
            m1 = b_gs - gr * (b_gs - a_gs)
            f_m1 = obj_at(m1)
        else:
            a_gs = m1
            m1 = m2
            f_m1 = f_m2
            m2 = a_gs + gr * (b_gs - a_gs)
            f_m2 = obj_at(m2)

    t_opt = (a_gs + b_gs) / 2.0
    new_val = max(0.0, f_i_old + t_opt)
    delta = new_val - f_i_old

    if abs(delta) < 1e-14:
        return

    # Update c first (using old f values)
    c[l_start: l_end + 1] += 2.0 * delta * f_sl
    if has_quad:
        c[2 * i] += delta * delta
    f[i] = new_val


def coord_descent_run(f_init, rng, time_limit, max_sweeps=100000):
    """Run coordinate descent. Returns (best_f_normalized, best_maxc)."""
    n = len(f_init)
    f = np.maximum(f_init.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    c = fast_convolve(f)
    best_obj = float(c.max())
    best_f = f.copy()

    t0 = time.monotonic()
    sweep_count = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit and sweep_count < max_sweeps:
        sweep_count += 1
        order = rng.permutation(n)
        for i in order:
            coord_step(f, c, int(i))

        obj = float(c.max())
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 20:
                # Perturb to escape
                scale = 0.005 + 0.005 * min(stagnant // 20, 3)
                f = best_f + rng.standard_normal(n) * scale
                f = np.maximum(f, 0.0)
                fs = f.sum()
                if fs < 1e-15:
                    f = best_f.copy()
                else:
                    f = f / fs
                c = fast_convolve(f)
                stagnant = 0

    return best_f, best_obj


def solve_lp_dir(f, M):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c_vec = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def fw_direction(f, M, m_exp=1.0):
    """One LP-FW step. Returns (f_new or None, m_exp)."""
    n = len(f)
    g = solve_lp_dir(f, M * m_exp)
    if g is None:
        return None, min(m_exp * 1.1, 5.0)
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None, m_exp
    g = g / g_sum

    if np.max(np.abs(g - f)) < 1e-10:
        return None, min(m_exp * 1.03, 3.0)

    c_cur = fast_convolve(f)
    best_alpha = 0.0
    best_val = float(c_cur.max())

    for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = float(fast_convolve(ft).max())
        if val < best_val:
            best_val = val
            best_alpha = alpha

    if best_alpha > 1e-10:
        f_new = (1.0 - best_alpha) * f + best_alpha * g
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        return f_new, 1.0

    return None, min(m_exp * 1.05, 4.0)


def generate_inits(n, parent, rng):
    """Generate diverse initializations for coordinate n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.2))

    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 1.5
    inits.append(edge_w / edge_w.sum())

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    c1 = np.abs(np.cos(np.pi * x / n)) ** 2 + 0.02
    inits.append(c1 / c1.sum())

    h1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.08 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.8 * n) / (0.08 * n)) ** 2)
    inits.append((h1 + h2) / (h1 + h2).sum())

    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.5)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    inits.append(sym / sym.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Structured noise on uniform
    noise = rng.standard_normal(n) * 0.1
    init_n = (np.ones(n) + noise)
    init_n = np.maximum(init_n, 0)
    inits.append(init_n / init_n.sum())

    return inits
