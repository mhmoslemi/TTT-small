def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)

    A = np.fft.rfft(a, fft_size)

def softmax(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    return e / e.sum()

    s_max = s.max()
    e = np.exp(s - s_max)
    weights = e / e.sum()
    
    # Only use weights above threshold for efficiency
    active = weights > threshold
    if not active.any():
        active[np.argmax(weights)] = True
    weights_active = weights[active]
    weights_active = weights_active / weights_active.sum()
    
    value = mu * (s_max + np.log(e.sum()))
    
    # Gradient: sum_k p_k * 2 * a_{k-j}
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        wk = weights[k]
        if wk < 1e-15:
            continue
        # (a*a)_k = sum_j a_j * a_{k-j}
        # d/da_j = 2 * a_{k-j} for valid j
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # grad[j] += 2 * wk * a[k-j]
        # For j in [js, je], k-j ranges from k-je to k-js
        # a[k-j] for j in [js, je] is a[k-je : k-js+1] reversed... 
        # Actually: for j in [js, je], the contribution is a[k-j]
        # Let's just do it directly
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * wk * a[k - idx]
    
    return value, grad

def smooth_max_softopt(f0, time_limit, mu_init=0.001, mu_end=0.00001, 
                        lr_init=0.1, lr_end=0.001, verbose=False):
    """Optimize using smooth max with annealing, in softmax parameterization."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    # Initialize z from a: z = log(a) (with small epsilon)
    z = np.log(np.maximum(a, 1e-15))
    
    # Momentum for Nesterov
    z_prev = z.copy()
    best_obj = np.inf
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    
    total_iters = 5000  # approximate
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / total_iters)
        
        # Anneal mu and lr
        mu = mu_init * (mu_end / mu_init) ** progress
        lr = lr_init * (lr_end / lr_init) ** progress
        
        a = np.maximum(softmax(z), 1e-15)
        a = a / a.sum()
        
        # Nesterov lookahead
        z_nest = z + 0.9 * (z - z_prev)
        a_nest = np.maximum(softmax(z_nest), 1e-15)
        a_nest = a_nest / a_nest.sum()
        
        # Compute gradient at Nesterov point
        val, grad_a = smooth_max_grad(a_nest, mu, threshold=0.005)
        
        if not np.isfinite(val):
            mu *= 2
            continue
        
        # Convert gradient from a-space to z-space
        # df/dz = a * (df/da - a·(df/da))
        dot = np.dot(a_nest, grad_a)
        grad_z = a_nest * (grad_a - dot)
        
        # Update
        z_prev = z.copy()
        z = z - lr * grad_z
        
        # Evaluate
        a = np.maximum(softmax(z), 1e-15)
        a = a / a.sum()
        b = fast_convolve(a)
        obj = b.max()
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose and it % 100 == 0:
                print(f"    iter {it}: obj={obj:.8f}, mu={mu:.6f}")
    
    return best_a, best_obj

def nesterov_sg(f0, time_limit, lr_init=0.005, top_frac=0.3, verbose=False):
    """Nesterov-accelerated subgradient on the simplex, focusing on top peaks."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    # Nesterov state
    a_curr = a.copy()
    a_prev = a.copy()
    beta = 0.9  # momentum parameter
    
    t0 = time.monotonic()
    it = 0
    cur_lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov lookahead
        a_nest = a_curr + beta * (a_curr - a_prev)
        a_nest = np.maximum(a_nest, 0)
        sn = a_nest.sum()
        if sn < 1e-15:
            a_nest = a_curr
        else:
            a_nest = a_nest / sn
        
        b = fast_convolve(a_nest)
        M = b.max()
        
        # Find top peaks
        n_peaks = max(10, int(top_frac * len(b)))
        peaks = np.argpartition(b, -n_peaks)[-n_peaks:]
        
        # Compute subgradient
        sg = np.zeros(n)
        for k in peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * a_nest[k - idx]
        
        # Scale subgradient (it's for the max, but we're using top peaks)
        # The true subgradient of max is 2*a_{k*-j} for the argmax k*
        # Using multiple peaks gives a smoothed version
        sg = sg / len(peaks)
        
        # Projected gradient step
        a_curr = a_curr - cur_lr * sg
        a_curr = np.maximum(a_curr, 0)
        sc = a_curr.sum()
        if sc < 1e-15:
            a_curr = best_a.copy()
        else:
            a_curr = a_curr / sc
        
        a_prev = a_curr
        
        # Evaluate
        obj = fast_convolve(a_curr).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_curr.copy()
        
        cur_lr *= 0.9995
        
        if it % 2000 == 0 and verbose:
            print(f"    iter {it}: obj={best_obj:.8f}, lr={cur_lr:.6f}")
    
    return best_a, best_obj

def fw_lp_simple(f0, time_limit, verbose=False):
    """Simplified FW-LP optimization."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    b = fast_convolve(a)
    M = b.max()
    best_obj = M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = a
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': max(1, (time_limit - (time.monotonic() - t0)) * 0.8)})
        
        if not res.success:
            no_improve += 1
            if no_improve > 20:
                # Try with relaxed constraint
                b_ub = np.full(2 * n - 1, M * 1.1)
                res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                              options={'time_limit': max(1, (time_limit - (time.monotonic() - t0)) * 0.8)})
                if not res.success:
                    break
            continue
        
        g = res.x
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            at = (1 - alpha) * a + alpha * g
            at = np.maximum(at, 0)
            st = at.sum()
            if st < 1e-15:
                continue
            at = at / st
            val = fast_convolve(at).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            a = (1 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            b = fast_convolve(a)
            M = b.max()
            if M < best_obj:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 30:
            # Restart with perturbation
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 100000)
            a = best_a * np.exp(rng.standard_normal(n) * 0.01)
            a = np.maximum(a, 0)
            a = a / a.sum()
            b = fast_convolve(a)
            M = b.max()
            no_improve = 0
    
    return best_a, best_obj

    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_fn(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    total = e.sum()
    if total < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / total


def compute_smooth_max_grad(a, mu, frac_threshold=0.003):
    """
    Compute value and gradient of mu*log(sum(exp(b_k/mu))) where b = a*a.
    Gradient is w.r.t. a (primal space).
    """
    b = fast_convolve(a)
    n = len(a)

    s = b / mu
    s_max = s.max()
    e = np.exp(s - s_max)
    total_e = e.sum()
    weights = e / total_e

    value = mu * (s_max + np.log(total_e))

    # Determine active set (weights above threshold)
    threshold = frac_threshold
    active_mask = weights >= threshold
    if not active_mask.any():
        active_mask[np.argmax(weights)] = True

    active_idx = np.where(active_mask)[0]
    active_w = weights[active_idx]
    w_sum = active_w.sum()
    if w_sum < 1e-15:
        active_w = np.zeros(len(active_idx))
        active_w[np.argmax(weights[active_idx])] = 1.0
    else:
        active_w /= w_sum

    # Compute gradient: grad_j = 2 * sum_k w_k * a_{k-j}
    grad = np.zeros(n)
    for i in range(len(active_idx)):
        k = active_idx[i]
        wk = active_w[i]
        if wk < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * wk * a[k - idx]

    return value, grad


def smooth_max_optimize(f0, time_limit, mu_init=0.005, mu_end=0.00005,
                        lr_init=0.5, lr_end=0.005, momentum=0.9):
    """
    Optimize smooth max using softmax parameterization with Nesterov momentum.
    Works in z-space where a = softmax(z), automatically satisfying a>0, sum(a)=1.
    """
    a = np.maximum(f0.astype(np.float64), 1e-15)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    # Initialize z from a
    z = np.log(a)
    z_prev = z.copy()
    v = np.zeros(n)  # momentum

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / 8000.0)

        # Anneal parameters
        mu = mu_init * (mu_end / mu_init) ** progress
        lr = lr_init * (lr_end / lr_init) ** progress

        # Nesterov lookahead
        z_nest = z + momentum * v
        a_nest = softmax_fn(z_nest)
        a_nest = np.maximum(a_nest, 1e-15)
        a_nest = a_nest / a_nest.sum()

        # Compute gradient at Nesterov point
        val, grad_a = compute_smooth_max_grad(a_nest, mu)

        if not np.isfinite(val):
            mu *= 2.0
            continue

        # Convert to z-space: df/dz = a * (df/da - a . (df/da))
        dot = np.dot(a_nest, grad_a)
        grad_z = a_nest * (grad_a - dot)

        # Momentum update
        v = momentum * v - lr * grad_z
        z = z_nest + v  # Actually: z_new = z_nest - lr*grad, with v updated
        # Let me redo: standard Nesterov
        # v_new = momentum * v_old + grad
        # z_new = z_nest - lr * v_new
        # But z_nest = z_old + momentum * v_old
        # So z_new = z_old + momentum*v_old - lr*(momentum*v_old + grad)
        # This is getting confused. Let me use a simpler formulation.

        # Actually let me use a cleaner Nesterov:
        # y = x + beta*(x - x_prev)  [lookahead]
        # x_new = y - lr * grad(y)
        # x_prev = x

        # I'll restart with a cleaner formulation below
        break

    # Clean Nesterov implementation
    a = np.maximum(f0.astype(np.float64), 1e-15)
    a = a / a.sum()
    z = np.log(a)
    z_prev = z.copy()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / 8000.0)
        mu = mu_init * (mu_end / mu_init) ** progress
        lr = lr_init * (lr_end / lr_init) ** progress

        # Nesterov lookahead
        z_nest = z + momentum * (z - z_prev)
        a_nest = softmax_fn(z_nest)
        a_nest = np.maximum(a_nest, 1e-15)
        a_nest = a_nest / a_nest.sum()

        # Compute gradient at Nesterov point
        val, grad_a = compute_smooth_max_grad(a_nest, mu)
        if not np.isfinite(val):
            mu *= 2.0
            z_prev = z.copy()
            continue

        # Convert to z-space
        dot = np.dot(a_nest, grad_a)
        grad_z = a_nest * (grad_a - dot)

        # Update
        z_prev = z.copy()
        z = z_nest - lr * grad_z

        # Evaluate at actual point
        a = softmax_fn(z)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        b = fast_convolve(a)
        obj = b.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart if stuck
        if no_improve > 500:
            rng_local = np.random.default_rng(it * 7919 + 13)
            z = z + rng_local.standard_normal(n) * 0.01
            z_prev = z.copy()
            no_improve = 0

    return best_a, best_obj


def nesterov_subgrad(f0, time_limit, lr_init=0.004, top_frac=0.25, decay=0.9995):
    """
    Nesterov-accelerated subgradient method on the simplex.
    Focuses on the top fraction of peaks with rank-based weighting.
    """
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    a_curr = a.copy()
    a_prev = a.copy()
    beta = 0.9

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    cur_lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        a_nest = a_curr + beta * (a_curr - a_prev)
        a_nest = np.maximum(a_nest, 0.0)
        sn = a_nest.sum()
        if sn < 1e-15:
            a_nest = a_curr.copy()
        else:
            a_nest = a_nest / sn

        b = fast_convolve(a_nest)
        M = b.max()

        # Select top peaks
        n_peaks = max(5, int(top_frac * len(b)))
        n_peaks = min(n_peaks, len(b))
        peaks = np.argpartition(b, -n_peaks)[-n_peaks:]

        # Rank-based weighting (emphasize the highest peaks)
        p_sorted = b[peaks]
        order = np.argsort(p_sorted)[::-1]
        w = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            w[idx] = 1.0 / (rank + 1) ** 1.2
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += w[i] * 2.0 * a_nest[k - idx]

        # Projected gradient step
        a_curr = a_curr - cur_lr * sg
        a_curr = np.maximum(a_curr, 0.0)
        sc = a_curr.sum()
        if sc < 1e-15:
            a_curr = best_a.copy()
        else:
            a_curr = a_curr / sc

        a_prev = a_curr

        obj = fast_convolve(a_curr).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_curr.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= decay

        if no_improve > 300:
            rng_local = np.random.default_rng(int(time.monotonic() * 100000) % 999983)
            a_curr = best_a * np.exp(rng_local.standard_normal(n) * 0.008)
            a_curr = np.maximum(a_curr, 0)
            a_curr = a_curr / a_curr.sum()
            a_prev = a_curr.copy()
            cur_lr = max(lr_init * 0.6, 0.0005)
            no_improve = 0

    return best_a, best_obj


def peak_proportional_descent(f0, time_limit, lr_init=0.002, target_frac=0.95):
    """
    Peak-proportional descent: reduce a_j proportionally to how much it
    contributes to peaks that exceed the target level.
    This is different from standard subgradient - it uses the actual excess.
    """
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    cur_lr = lr_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        M = b.max()
        mean_b = b.mean()

        # Target: reduce peaks that are above target_frac * M
        target = target_frac * M

        # Compute "excess" for each peak
        excess = np.maximum(b - target, 0.0)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            break

        # Distribute blame: for each peak k with excess e_k,
        # a_j gets blame proportional to e_k * a_{k-j}
        blame = np.zeros(n)
        active_peaks = np.where(excess > 0.01 * M)[0]
        for k in active_peaks:
            ek = excess[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            blame[idx] += ek * a[k - idx]

        # Normalize blame to sum to 1
        b_sum = blame.sum()
        if b_sum < 1e-15:
            break
        blame = blame / b_sum

        # "Cold" entries: those with low blame
        # Move mass from high-blame to low-blame entries
        # a_new = a - lr * (blame - mean_blame) where mean_blame = 1/n
        # Project onto simplex
        a_new = a - cur_lr * (blame - 1.0 / n)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = best_a.copy()
        else:
            a_new = a_new / sn

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()
            cur_lr *= 0.999
        else:
            # Try smaller step
            a_try = a - cur_lr * 0.5 * (blame - 1.0 / n)
            a_try = np.maximum(a_try, 0.0)
            st = a_try.sum()
            if st > 1e-15:
                a_try = a_try / st
                obj_try = fast_convolve(a_try).max()
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    cur_lr *= 0.999
                else:
                    cur_lr *= 0.95
            else:
                cur_lr *= 0.9

    return best_a, best_obj


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    b = fast_convolve(a)
    M = b.max()
    best_obj = M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # Solve LP
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = a
        b_ub = np.full(2 * n - 1, M * m_relax)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': max(1.0, remaining * 0.7)})

        if not res.success:
            m_relax = min(m_relax * 1.2, 5.0)
            no_improve += 1
            if m_relax > 5.0 or no_improve > 30:
                # Perturb and restart
                rng_local = np.random.default_rng(int(time.monotonic() * 1000) % 99991)
                a = best_a * np.exp(rng_local.standard_normal(n) * 0.01)
                a = np.maximum(a, 0)
                a = a / a.sum()
                b = fast_convolve(a)
                M = b.max()
                m_relax = 1.0
                no_improve = 0
            continue

        g = res.x
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            at = (1 - alpha) * a + alpha * g
            at = np.maximum(at, 0)
            st = at.sum()
            if st < 1e-15:
                continue
            at = at / st
            val = fast_convolve(at).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            a = (1 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            b = fast_convolve(a)
            M = b.max()
            m_relax = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.1, 4.0)

    return best_a, best_obj


def refine_polish(f0, time_limit):
    """Very fine polishing using alternating small steps."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = 0.0005

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        M = b.max()
        # Top 20 peaks
        k_peaks = max(5, 20)
        peaks = np.argpartition(b, -k_peaks)[-k_peaks:]

        sg = np.zeros(n)
        for k in peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * a[k - idx]
        sg /= len(peaks)

        a_new = a - lr * sg
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn > 1e-15:
            a_new = a_new / sn
            obj = fast_convolve(a_new).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new.copy()
                lr *= 0.999
            else:
                lr *= 0.95

        if lr < 1e-7:
            break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-strategy search combining:
    - Smooth max optimization (softmax parameterization + Nesterov)
    - Nesterov subgradient
    - Peak-proportional descent
    - FW-LP
    - Fine polishing
    - Multiple n values and random restarts
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            update_best(parent, "parent_initial")
    except Exception as e:
        print(f"Could not load parent: {e}")

    # Determine n values to try
    n_values = []
    if parent is not None:
        n_values.append(len(parent))
    for n_try in [384, 512, 768]:
        if n_try not in n_values:
            n_values.append(n_try)

    # === Phase 1: Smooth max optimization on various n ===
    print("\n=== Phase 1: Smooth Max Optimization ===")

    time_per_n = max(40, (deadline - time.monotonic() - 700) / len(n_values))

    for n in n_values:
        rem = deadline - time.monotonic() - 500
        if rem < 30:
            break

        # Create initialization
        if parent is not None and n == len(parent):
            init = parent.copy()
        elif parent is not None:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            init = np.interp(x_new, x_old, parent)
            init = np.maximum(init, 0)
            init = init / init.sum()
        else:
            init = np.ones(n) / n

        t_alloc = min(time_per_n, rem)
        print(f"  Smooth max n={n}, t={t_alloc:.0f}s")
        a_opt, obj = smooth_max_optimize(init, time_limit=t_alloc * 0.6,
                                          mu_init=0.01, mu_end=0.0001,
                                          lr_init=0.3, lr_end=0.005, momentum=0.85)
        update_best(a_opt, f"smooth_n{n}")

        # Nesterov SG follow-up
        rem = deadline - time.monotonic() - 350
        if rem > 20:
            t2 = min(time_per_n * 0.4, rem)
            a_opt2, _ = nesterov_subgrad(a_opt, time_limit=t2,
                                          lr_init=0.003, top_frac=0.25)
            update_best(a_opt2, f"nesterov_n{n}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Peak-proportional descent ===
    print("\n=== Phase 2: Peak-Proportional Descent ===")

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(200, rem)
            print(f"  Peak-proportional, n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = peak_proportional_descent(best_seq, time_limit=t_alloc,
                                                  lr_init=0.003, target_frac=0.9)
            update_best(a_opt, "peak_prop")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: FW-LP ===
    print("\n=== Phase 3: FW-LP ===")

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(250, rem)
            print(f"  FW-LP, n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            update_best(a_opt, "fw_lp")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Random restarts with diverse methods ===
    print("\n=== Phase 4: Random Restarts ===")

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 100
        n_restarts = max(2, int(rem / 40))
        t_per_restart = max(15, rem / n_restarts)

        for ri in range(n_restarts):
            rem = deadline - time.monotonic() - 50
            if rem < t_per_restart * 0.5:
                break

            # Different perturbation strategies
            if ri % 3 == 0:
                # Multiplicative noise
                noise = rng.standard_normal(n_best) * (0.005 + 0.005 * ri)
                a_pert = best_seq * np.exp(noise)
            elif ri % 3 == 1:
                # Add uniform noise and re-project
                a_pert = best_seq + rng.uniform(0, 0.01, n_best)
            else:
                # Mix with random Dirichlet
                a_rand = rng.dirichlet(np.ones(n_best) * 0.3)
                a_pert = 0.8 * best_seq + 0.2 * a_rand

            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            t_alloc = min(t_per_restart, rem)

            # Alternate methods
            if ri % 3 == 0:
                a_opt, _ = smooth_max_optimize(a_pert, time_limit=t_alloc,
                                                mu_init=0.003, mu_end=0.00005,
                                                lr_init=0.1, lr_end=0.003, momentum=0.85)
                label = f"restart_smooth_{ri}"
            elif ri % 3 == 1:
                a_opt, _ = nesterov_subgrad(a_pert, time_limit=t_alloc,
                                             lr_init=0.002, top_frac=0.2)
                label = f"restart_nest_{ri}"
            else:
                a_opt, _ = peak_proportional_descent(a_pert, time_limit=t_alloc,
                                                      lr_init=0.002, target_frac=0.85)
                label = f"restart_peak_{ri}"

            update_best(a_opt, label)

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Fine polish ===
    print("\n=== Phase 5: Fine Polish ===")

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_alloc = min(rem * 0.5, 100)
            print(f"  Fine polish, t={t_alloc:.0f}s")
            a_opt, _ = refine_polish(best_seq, time_limit=t_alloc)
            update_best(a_opt, "fine_polish")

            rem = deadline - time.monotonic() - 25
            if rem > 15:
                t2 = min(rem * 0.6, 60)
                print(f"  Final Nesterov, t={t2:.0f}s")
                a_opt2, _ = nesterov_subgrad(best_seq, time_limit=t2,
                                              lr_init=0.0003, top_frac=0.1,
                                              decay=0.9998)
                update_best(a_opt2, "final_nest")

    # === Phase 6: Try sequence growth ===
    print("\n=== Phase 6: Sequence Growth ===")

    if best_seq is not None:
        n_cur = len(best_seq)
        rem = deadline - time.monotonic() - 20
        for growth_factor in [2.0, 1.5]:
            n_new = int(n_cur * growth_factor)
            if n_new > 2048 or rem < 30:
                continue

            x_old = np.linspace(0, 1, n_cur)
            x_new = np.linspace(0, 1, n_new)
            a_grown = np.interp(x_new, x_old, best_seq)
            a_grown = np.maximum(a_grown, 0)
            a_grown = a_grown / a_grown.sum()

            t_alloc = min(rem, 100)
            print(f"  Growth to n={n_new}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_subgrad(a_grown, time_limit=t_alloc,
                                         lr_init=0.002, top_frac=0.2)
            update_best(a_opt, f"grown_n{n_new}")
            rem = deadline - time.monotonic() - 20

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
