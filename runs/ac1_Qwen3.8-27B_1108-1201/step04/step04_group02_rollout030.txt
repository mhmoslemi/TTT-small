```python
def fista_smooth(a0, mu_init, mu_final, time_limit):
    a = a0.copy()
    n = len(a)
    mu = mu_init
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    L = 1.0  # initial Lipschitz estimate
    
    best_a = a.copy()
    best_val = smooth_obj(a, mu)
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        grad = smooth_grad(y, mu)
        
        # Backtracking line search for L
        while True:
            a_new = project_simplex(y - grad / L)
            # Check descent
            if smooth_obj(a_new, mu) <= smooth_obj(y, mu) - 0.5 * L * np.sum((a_new - y)**2):
                break
            L *= 2
        
        # FISTA update
        a_prev = a.copy()
        a = a_new
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        y = a + ((t - 1) / t_new) * (a - a_prev)
        y = np.maximum(y, 0)
        y = y / y.sum()
        t = t_new
        
        # Check best
        val = smooth_obj(a, mu)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        
        # Anneal mu
        mu *= 0.999
        
    return best_a, best_val
```

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

    rho = np.nonzero(cond)[0][-1]

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_value(b, mu):
    """Compute smooth max approximation."""
    b_max = b.max()
    shifted = (b - b_max) / mu
    shifted -= shifted.max()  # numerical stability
    return mu * np.log(np.sum(np.exp(shifted))) + b_max


def smooth_max_grad(a, mu):
    """Compute gradient of smooth max of auto-convolution."""
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    shifted = (b - b_max) / mu
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # grad[i] = (1/mu) * sum_k w[k] * (a[k-i] + a[i-k])
    term1 = np.correlate(weights, a, 'full')[:n]
    term2 = np.convolve(weights, a, 'full')[:n]
    
    return (term1 + term2) / mu


def solve_lp_oracle(f, M, time_limit=10):
    """LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def sg_polish(a0, time_limit, lr=0.001, top_k=100, temp=0.0001):
    """Fine subgradient polishing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    b = fast_convolve(a)
    best_obj = b.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    it = 0
    
    def compute_sg(a, b, top_k, temp):
        if top_k >= len(b):
            peaks = np.arange(len(b))
            w = np.full(len(peaks), 1.0/len(peaks))
        else:
            peaks = np.argpartition(b, -top_k)[-top_k:]
            pv = b[peaks]
            scaled = pv / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w /= w.sum()
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
        return sg
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        sg = compute_sg(a, b, top_k, temp)
        v = 0.95 * v - lr * sg
        a = project_simplex(a + v)
        b = fast_convolve(a)
        obj = b.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
        
        lr *= 0.9999
        
        if it % 1000 == 0 and time.monotonic() - t0 > time_limit - 20:
            break
    
    return best_a, best_obj
