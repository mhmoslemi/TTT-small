    """Linear auto-convolution via FFT."""

    # In terms of convolution: (v corr s)[i] = (v * s_flipped)[i + n - 1]

    # Gradient: grad_i = s_i * (corr_i - wc)
    grad = s * (corr - wc)

def adam_smoothed_max(z0, tau_init, tau_min, tau_decay, lr, beta1, beta2, epsilon,
                      time_limit, n_checks=100):
    """Optimize z to minimize smoothed max of auto-convolution of softmax(z)."""
    n = len(z0)
    z = z0.copy().astype(np.float64)
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    tau = tau_init
    
    t0 = time.monotonic()
    best_obj = np.inf
    best_z = z.copy()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, (c_max, s) = compute_grad_softmax(z, tau)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Anneal temperature
        tau = max(tau_min, tau * tau_decay)
        
        # Track best
        if t % n_checks == 0:
            s_check = np.exp(z - z.max())
            s_check /= s_check.sum()
            c_check = fast_convolve(s_check)
            obj = 2.0 * n * c_check.max()
            if obj < best_obj:
                best_obj = obj
                best_z = z.copy()
    
    # Final check
    s_final = np.exp(best_z - best_z.max())
    s_final /= s_final.sum()
    c_final = fast_convolve(s_final)
    best_obj = 2.0 * n * c_final.max()
    
    return best_z, s_final, best_obj

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def compute_lp_gradient(z, p, c=None, s=None):
    """
    Compute gradient of (sum_k c_k^p)^(1/p) w.r.t. z, where s = softmax(z), c = s*s.
    Returns (gradient, objective_value).
    """
    n = len(z)
    if c is None or s is None:
        s = softmax(z)
        c = fast_convolve(s)
    
    # f_p = (sum c_k^p)^(1/p)
    # To avoid overflow, compute in log space
    log_cp = p * np.log(np.maximum(c, 1e-300))
    log_sum = np.logaddexp.reduce(log_cp)  # log(sum c_k^p)
    log_fp = log_sum / p
    fp = np.exp(log_fp)
    
    # u_k = (c_k / f_p)^(p-1)
    # This is bounded in [0, 1]
    with np.errstate(divide='ignore', invalid='ignore'):
        ratio = np.maximum(c, 1e-300) / fp
        u = np.power(ratio, p - 1)
    
    # corr[u, s][i] = sum_j u[j+i] * s[j] for i = 0..n-1
    # Using FFT cross-correlation
    fft_size = 1 << (3 * n - 1).bit_length()
    U = np.fft.rfft(u, fft_size)
    S_flip = np.fft.rfft(s[::-1], fft_size)

    corr = corr_full[n-1:n-1+n]

    uc = (u * c).sum()
    
    # gradient_i = s_i * (corr[i] - uc)
    grad = s * (corr - uc)
    
    return grad, fp

def lp_solve_direction(a, M):
    """
    Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0.
    Returns g or None if infeasible.
    """
    n = len(a)
    # Build the convolution matrix: (a*g)_k = sum_j a[j] * g[k-j]
    # For k = 0..2n-2, j ranges over valid values
    # We need: for each k, sum over valid j of a[j] * g[k-j] <= M
    # Let l = k-j, so g[l] appears in constraint k with coefficient a[k-l]
    # For constraint k: sum over l of a[k-l] * g[l] <= M, where 0 <= k-l <= n-1
    # i.e., max(0, k-n+1) <= l <= min(n-1, k)

    # Number of constraints: 2n-1
    # Number of variables: n

        l_start = max(0, k - n + 1)
        l_end = min(n - 1, k)

            A[k, l] = a[k - l]
    
    b = np.full(2 * n - 1, M)

            g = np.maximum(result.x, 0)

                return g

        pass
    return None

"""

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def stable_softmax(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def compute_lp_grad_and_obj(z, p):
    """
    Compute gradient and value of (sum_k c_k^p)^(1/p) w.r.t. z,
    where s = softmax(z) and c = s * s (linear auto-convolution).
    Returns (gradient, objective_value, c_max).
    """
    n = len(z)
    s = stable_softmax(z)
    c = fast_convolve(s)
    c = np.maximum(c, 0.0)  # numerical safety

    # Compute f_p = (sum c_k^p)^(1/p) using log-sum-exp
    log_cp = p * np.log(np.maximum(c, 1e-300))
    max_log = log_cp.max()
    log_sum = max_log + np.log(np.sum(np.exp(log_cp - max_log)))
    log_fp = log_sum / p
    fp = np.exp(log_fp)

    # u_k = (c_k / f_p)^(p-1), bounded in [0, 1]
    with np.errstate(divide='ignore', invalid='ignore'):
        ratio = np.maximum(c, 1e-300) / fp
        u = np.power(ratio, p - 1)
    u = np.where(np.isfinite(u), u, 0.0)

    # Cross-correlation: corr[i] = sum_j u[j+i] * s[j] for i = 0..n-1
    # corr[i] = (u * s_flip)[i + n - 1] where s_flip = s reversed
    fft_size = 1 << (3 * n - 1).bit_length()
    U = np.fft.rfft(u, fft_size)
    S_flip = np.fft.rfft(s[::-1], fft_size)
    corr_full = np.fft.irfft(U * S_flip, fft_size)
    corr = corr_full[n - 1:n - 1 + n]
    corr = np.maximum(corr, 0.0)  # numerical safety

    # Scalar: sum_k u_k * c_k
    uc = (u * c).sum()

    # Gradient: grad_i = s_i * (corr_i - uc)
    grad = s * (corr - uc)

    return grad, fp, c.max()

            g = np.maximum(result.x, 0)
            if g.sum() > 1e-10:
