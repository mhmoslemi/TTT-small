    """Compute gradient of smoothed max of conv(f,f) w.r.t. f."""
    n = len(f)
    # grad[i] = 2 * sum_k weights[k] * f[k-i] for valid k-i
    # = 2 * cross_corr(weights, f)[i]
    # = 2 * conv(weights, f[::-1])[i + n - 1]
    f_rev = f[::-1]
    size = len(weights) + n - 1

    result = np.fft.irfft(W * F, fft_size)

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad_fft(f, weights):
    """Compute gradient via FFT: grad[i] = 2*sum_k weights[k]*f[k-i]"""
    n = len(f)
    size = len(weights) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f[::-1], fft_size)
    result = np.fft.irfft(W * F, fft_size)
    return 2.0 * result[n-1:n-1+n]

def sigmoid_max_gd(f0, time_limit, T0=50, T1=500, lr0=0.01, mom=0.9):
    """Smoothed max gradient descent with FFT-accelerated gradient."""
    f = f0.copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    T = T0
    lr = lr0
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        
        # Compute softmax weights (smoothed max)
        T_conv = T * conv
        T_conv -= T_conv.max()
        exp_T = np.exp(T_conv)
        weights = exp_T / exp_T.sum()
        
        # FFT gradient
        grad = compute_grad_fft(f, weights)
        
        # Momentum update
        v = mom * v - lr * grad
        f_new = project_simplex(f + v)
        
        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            f = f_new  # still update for momentum
            no_improve += 1
        
        # Adaptive T and lr
        T = min(T * 1.002, T1)
        lr = max(lr * 0.9995, 0.001)
        
        if no_improve > 100:
            # Restart with perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = lr0 * 0.5
            T = T0
            no_improve = 0
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated smoothed-max gradient descent with momentum
2. Multiplicative weight (mirror descent) with adaptive temperature
3. Top-k peak focus with exponential weighting
4. Diverse restarts with structured initializations
5. Multi-scale initialization (optimize longer seq, downsample)
6. Coordinate-level refinement for final polish

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_fft(f, weights):
    """Compute gradient of smoothed-max conv via FFT.
    grad[i] = 2 * sum_k weights[k] * f[k-i] for valid k-i.
    This equals 2 * (weights * f_rev)[n-1+n-1] where f_rev = f[::-1].
    """
    n = len(f)
    lw = len(weights)
    size = lw + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f[::-1], fft_size)
    result = np.fft.irfft(W * F, fft_size)
    return 2.0 * result[n - 1:n - 1 + n]


def sigmoid_max_gd(f0, time_limit, T0=50.0, T_max=2000.0, lr0=0.02,
                   mom=0.85, T_growth=1.003, lr_decay=0.9997,
                   restart_every=200, verbose=False):
    """Smoothed max gradient descent with FFT-accelerated gradient and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    T = T0
    lr = lr0
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        # Softmax weights for smoothed max
        Tc = T * conv
        Tc -= Tc.max()
        exp_T = np.exp(Tc)
        wsum = exp_T.sum()
        if wsum < 1e-300:
            weights = np.full_like(conv, 1.0 / len(conv))
        else:
            weights = exp_T / wsum

        # FFT gradient
        grad = compute_grad_fft(f, weights)

        # Momentum + step
        v = mom * v - lr * grad
        f_new = f + v
        f_new = project_simplex(f_new)

        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 5000 == 0:
                print(f"    it={it}, obj={best_obj:.8f}, T={T:.1f}, lr={lr:.6f}")
        else:
            no_improve += 1

        f = f_new

        # Adaptive schedules
        T = min(T * T_growth, T_max)
        lr = max(lr * lr_decay, 0.0005)

        # Adaptive momentum
        if no_improve > 50:
            mom = min(mom + 0.001, 0.98)
        elif no_improve < 10:
            mom = max(mom - 0.001, 0.8)

        # Restart
        if no_improve > restart_every:
            rng = np.random.default_rng(hash((it, int(time.monotonic()))) % (2**31))
            noise_scale = 0.002 + 0.004 * min(1.0, (no_improve - restart_every) / 200.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            lr = lr0 * 0.6
            T = max(T0, T * 0.5)
            mom = 0.85
            no_improve = 0

    return best_f, best_obj


def mult_weight_optimize(f0, time_limit, lr_init=0.04, top_k=20,
                         temp_init=0.0004, decay_lr=0.9996, decay_temp=1.0015,
                         verbose=False):
    """Multiplicative weight (mirror descent) with top-k peak focus."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        # Top-k peaks with softmax weighting
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            pv = conv
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]

        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute gradient for multiplicative update
        # d(conv[k])/d(log f[j]) = f[j] * d(conv[k])/d(f[j]) = f[j] * 2*f[k-j]
        log_grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            log_grad[idx] += 2.0 * w[i] * f[k - idx] * f[k - idx]

        # Multiplicative update
        f = f * np.exp(-cur_lr * log_grad)
        f = np.maximum(f, 1e-30)
        f = f / f.sum()

        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 5000 == 0:
                print(f"    MW it={it}, obj={best_obj:.8f}")
        else:
            no_improve += 1

        cur_lr *= decay_lr
        cur_temp *= decay_temp

        if no_improve > 200:
            rng = np.random.default_rng(hash((it, 999)) % (2**31))
            f = best_f * np.exp(rng.standard_normal(n) * 0.004)
            f = f / f.sum()
            cur_lr = max(lr_init * 0.5, 0.005)
            cur_temp = temp_init * 1.5
            no_improve = 0

    return best_f, best_obj

        Tc = T * conv

        wsum = exp_T.sum()
        if wsum < 1e-300:
            weights = np.full_like(conv, 1.0 / len(conv))

        v = mom * v - lr * grad

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()

            no_improve = 0

            lr = lr_init * 0.5
            T = T0
            no_improve = 0

    return best_f, best_obj

def generate_structured_inits(n, parent, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # Edge-weighted (emphasize boundaries)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.01
    inits.append(c / c.sum())

    # Sine wave
    s = np.abs(np.sin(2 * np.pi * x / (n // 3))) + 0.01
    inits.append(s / s.sum())

    # Two-scale pattern
    period = max(3, n // 8)
    two_scale = np.ones(n)
    two_scale[::2] = 1.5
    inits.append(two_scale / two_scale.sum())

    # Random smooth
    rough = rng.standard_normal(n)
    # Simple boxcar smoothing
    kernel_size = max(3, n // 30)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(rough, kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    # Parent interpolation
    if parent is not None:
        px = np.interp(x, np.arange(len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def multiscale_init(parent, target_n, rng, time_limit):
    """Optimize a longer sequence, then downsample to target length."""
    long_n = target_n * 3
    if long_n > 2000:
        long_n = 2000

    # Start with uniform or parent interpolated
    if parent is not None:
        long_seq = np.interp(np.arange(long_n), np.arange(len(parent)), parent)
        long_seq = long_seq / long_seq.sum()
    else:
        long_seq = np.ones(long_n) / long_n

    # Quick optimization on long sequence
    t_alloc = min(time_limit, 120)
    opt_seq, _ = sigmoid_max_gd(long_seq, t_alloc, T0=30, T_max=500, lr0=0.015, mom=0.8)

    # Downsample by averaging blocks
    block_size = long_n // target_n
    if block_size < 1:
        return long_seq[:target_n] / long_seq[:target_n].sum()

    result = np.zeros(target_n)
    for i in range(target_n):
        start = i * block_size
        end = start + block_size if i < target_n - 1 else long_n
        result[i] = opt_seq[start:end].mean()

    result = np.maximum(result, 0)
    rs = result.sum()
    if rs > 1e-15:
        result = result / rs
    return result
