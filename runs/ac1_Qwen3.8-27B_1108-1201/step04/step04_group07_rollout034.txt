from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build B matrix where (B*g)[k] = (f*g)[k]."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B


def solve_linearized_lp(f, conv_f, top_k=50):
    """Solve reduced linearized LP: min t s.t. t >= conv_f[k] + 2*(B*g)[k] for top k,
    g >= -f, sum(g)=0. Returns direction g or None."""
    n = len(f)
    B = build_conv_matrix(f)
    
    peaks = np.argpartition(conv_f, -top_k)[-top_k:]
    peaks = np.sort(peaks)
    k_used = len(peaks)
    
    c = np.zeros(n + 1)
    c[0] = 1.0
    
    A_ub = np.zeros((k_used, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B[peaks, :]
    b_ub = -conv_f[peaks]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def solve_maxsum_lp(f, M):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None and np.any(res.x > 0):
        return res.x
    return None


def line_search_for_direction(f, target, current_obj, max_alpha=1.0):
    """Find best alpha for f_new = (1-alpha)*f + alpha*target."""
    best_alpha = 0.0
    best_obj = current_obj
    
    alphas = np.linspace(0.05, max_alpha, 15)
    for alpha in alphas:
        f_new = (1.0 - alpha) * f + alpha * target
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            continue
        obj = fast_convolve(f_new / s).max()
        if obj < best_obj:
            best_obj = obj
            best_alpha = alpha
    
    if best_alpha > 0:
        lo = max(0.01, best_alpha * 0.7)
        hi = min(max_alpha, best_alpha * 1.3)
        for _ in range(10):
            mid = (lo + hi) / 2.0
            f_new = (1.0 - mid) * f + mid * target
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s < 1e-15:
                break
            obj = fast_convolve(f_new / s).max()
            if obj < best_obj:
                best_obj = obj
                best_alpha = mid
                lo = mid
            else:
                hi = mid
    
    return best_alpha, best_obj


def coordinate_descent_sweeps(a, max_sweeps=50, time_limit=None, t0=None, verbose=False):
    """
    Coordinate descent for minimizing max(conv(a,a))/(sum(a))^2.
    Maintains convolution incrementally for O(n) per coordinate step.
    Returns (optimized_a, best_obj).
    """
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    if time_limit is None:
        time_limit = 60.0
    if t0 is None:
        t0 = time.monotonic()
    
    sweep = 0
    stagnation = 0
    
    while sweep < max_sweeps and (time.monotonic() - t0) < time_limit:
        improved = False
        for j in range(n):
            if (time.monotonic() - t0) > time_limit:
                break
            
            old = a[j]
            S0 = 1.0 - old
            if S0 < 1e-12:
                continue
            
            lo_k = max(0, j)
            hi_k = min(2 * n - 2, j + n - 1)
            
            unchanged_max = 0.0
            if lo_k > 0:
                unchanged_max = max(unchanged_max, conv[:lo_k].max())
            if hi_k + 1 < 2 * n - 1:
                unchanged_max = max(unchanged_max, conv[hi_k + 1:].max())
            
            ks = np.arange(lo_k, hi_k + 1)
            kjs = ks - j
            is_quad = (ks == 2 * j)
            
            base2 = conv[ks].astype(np.float64) - 2.0 * old * a[kjs]
            slope = 2.0 * a[kjs].astype(np.float64)
            quad = np.zeros(len(ks))
            
            if is_quad.any():
                base2[is_quad] = conv[ks[is_quad]] - old * old
                slope[is_quad] = 0.0
                quad[is_quad] = 1.0
            
            # Candidate t values
            candidates = [old, 0.0]
            
            # Top few peaks for candidate generation
            top_idx = np.argsort(base2 + slope * old + quad * old * old)[-5:]
            for idx in top_idx:
                b = base2[idx]
                s_ = slope[idx]
                q = quad[idx]
                denom = 2.0 * q * S0 - s_
                if abs(denom) > 1e-15:
                    t_cand = (2.0 * b - s_ * S0) / denom
                    if 0 <= t_cand <= 2.0:
                        candidates.append(t_cand)
            
            candidates.extend([old * 0.7, old * 1.3])
            candidates = [max(0.0, min(c, 2.0)) for c in candidates]
            
            best_t = old
            best_val = np.inf
            seen = set()
            for t in candidates:
                key = round(t, 8)
                if key in seen:
                    continue
                seen.add(key)
                vals = base2 + slope * t + quad * t * t
                max_val = max(unchanged_max, vals.max())
                st = S0 + t
                if st < 1e-10:
                    continue
                obj = max_val / (st * st)
                if obj < best_val:
                    best_val = obj
                    best_t = t
            
            # Fine refinement
            span = max(0.005, old * 0.1)
            lo = max(0.0, best_t - span)
            hi = best_t + span
            for t in np.linspace(lo, hi, 30):
                vals = base2 + slope * t + quad * t * t
                max_val = max(unchanged_max, vals.max())
                st = S0 + t
                if st < 1e-10:
                    continue
                obj = max_val / (st * st)
                if obj < best_val:
                    best_val = obj
                    best_t = max(0.0, t)
            
            if abs(best_t - old) > 1e-14:
                delta = best_t - old
                for k in range(lo_k, hi_k + 1):
                    if k == 2 * j:
                        conv[k] += best_t * best_t - old * old
                    else:
                        conv[k] += 2.0 * delta * a[k - j]
                a[j] = best_t
                improved = True
        
        # Renormalize
        s = a.sum()
        if abs(s - 1.0) > 1e-8:
            a = a / s
            conv = fast_convolve(a)
        
        cur_obj = conv.max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        if verbose and sweep % 10 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, stagnation={stagnation}")
        
        if stagnation > 5:
            break
        
        sweep += 1
    
    return best_a, best_obj


def subgradient_optimize(a0, time_limit, lr=0.002, mom=0.94, top_k=60, temp=0.00005, cooldown=1.003):
    """Subgradient descent with softmax-weighted multi-peak targeting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
        
        v = mom * v - cur_lr * sg
        a_new = project_simplex(a + v)
        conv = fast_convolve(a_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            a_new = a  # revert
        
        a = a_new
        cur_lr *= 0.9998
        cur_temp *= cooldown
        
        if no_improve > 200:
            rng = np.random.default_rng((it * 7919 + 31) % 1000000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0
    
    return best_a, best_obj


def fw_lp_optimize(a0, time_limit, top_k=40, verbose=False):
    """Frank-Wolfe with linearized LP oracle."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        current_obj = conv.max()
        
        g = solve_linearized_lp(a, conv, top_k=top_k)
        if g is None:
            no_improve += 3
            if no_improve > 50:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue
        
        target = a + g
        target = np.maximum(target, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts
        
        if np.max(np.abs(target - a)) < 1e-11:
            no_improve += 1
            if no_improve > 30:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                no_improve = 0
            continue
        
        alpha, obj_val = line_search_for_direction(a, target, current_obj)
        if alpha > 1e-9:
            a = (1.0 - alpha) * a + alpha * target
            a = np.maximum(a, 0.0)
            fs = a.sum()
            if fs < 1e-15:
                a = best_a.copy()
            else:
                a = a / fs
            
            if obj_val < best_obj - 1e-14:
                best_obj = obj_val
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    FW-LP iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 40:
            a = project_simplex(best_a + rng.standard_normal(n) * (0.001 + no_improve * 5e-5))
            no_improve = 0
    
    return best_a, best_obj


def symmetrize(a):
    """Create symmetric version: a'[j] = (a[j] + a[n-1-j])/2."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym


def try_periodic_perturbation(a, rng, num_freqs=20):
    """Try adding small sinusoidal perturbations at various frequencies."""
    n = len(a)
    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()
    
    base_x = np.arange(n, dtype=np.float64)
    for freq in range(1, num_freqs + 1):
        for phase in [0.0, np.pi / 2, np.pi]:
            for amp in [0.003, 0.008, 0.015]:
                pert = a + amp * np.sin(2 * np.pi * freq * base_x / n + phase)
                pert = project_simplex(pert)
                obj = fast_convolve(pert).max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = pert.copy()
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search leveraging coordinate descent as
    the primary fine-tuning tool, with subgradient and LP-FW for coarser moves.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(384) / 384
        print("No parent, using uniform n=384")
    
    n = len(parent)
    
    # === Phase 1: Quick initialization ===
    print("=== Phase 1: Initialization ===")
    t_start = time.monotonic()
    
    candidates = [("parent", parent)]
    
    # Symmetric version
    a_sym = symmetrize(parent)
    candidates.append(("sym", a_sym))
    
    # Some structured inits
    x = np.arange(n, dtype=np.float64)
    for sigma_frac in [0.2, 0.3, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
        candidates.append((f"gauss_{sigma_frac}", g / g.sum()))
    
    for label, cand in candidates:
        try_seq(cand, f"init_{label}")
    
    # Periodic perturbation on best
    a_period, _ = try_periodic_perturbation(best_seq, rng, num_freqs=30)
    try_seq(a_period, "periodic")
    
    print(f"  Phase 1 best: {best_score:.8f} (n={len(best_seq)})")
    
    # === Phase 2: Subgradient for medium-scale optimization ===
    print("\n=== Phase 2: Subgradient Optimization ===")
    remaining = deadline - time.monotonic() - 300
    t_sg = min(remaining * 0.25, 300)
    
    if t_sg > 20:
        a_opt, _ = subgradient_optimize(best_seq, time_limit=t_sg * 0.5,
                                         lr=0.003, mom=0.93, top_k=80, temp=0.0001)
        try_seq(a_opt, "sg_coarse")
        
        a_opt, _ = subgradient_optimize(best_seq, time_limit=t_sg * 0.5,
                                         lr=0.001, mom=0.95, top_k=120, temp=0.00003)
        try_seq(a_opt, "sg_fine")    
        # Symmetry attempt
        a_sym = symmetrize(best_seq)
        a_opt, _ = subgradient_optimize(a_sym, time_limit=min(t_sg * 0.3, 60),
                                         lr=0.002, mom=0.94, top_k=80, temp=0.00008)
        try_seq(a_opt, "sg_sym")
    
    print(f"  Phase 2 best: {best_score:.8f}")
    
    # === Phase 3: LP Frank-Wolfe for large jumps ===
    print("\n=== Phase 3: LP Frank-Wolfe ===")
    remaining = deadline - time.monotonic() - 400
    t_fw = min(remaining * 0.2, 250)
    
    if t_fw > 30:
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_fw * 0.6, top_k=40)
        try_seq(a_opt, "fw_lp")
        
        # Perturbed restart
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            a_pert = project_simplex(best_seq + rng.standard_normal(n) * 0.008)
            a_opt, _ = fw_lp_optimize(a_pert, time_limit=min(rem * 0.5, 120), top_k=35)
            try_seq(a_opt, "fw_lp_pert")
    
    print(f"  Phase 3 best: {best_score:.8f}")
    
    # === Phase 4: Coordinate Descent (main fine-tuner) ===
    print("\n=== Phase 4: Coordinate Descent ===")
    remaining = deadline - time.monotonic() - 200
    t_cd = min(remaining * 0.4, 500)
    
    if t_cd > 10:
        print(f"  Running CD, budget={t_cd:.0f}s")
        a_opt, cd_obj = coordinate_descent_sweeps(best_seq, max_sweeps=200,
                                                  time_limit=t_cd * 0.5, verbose=True)
        try_seq(a_opt, "cd1")
        
        # More CD from the result
        rem = deadline - time.monotonic() - 150
        if rem > 10:
            a_opt2, _ = coordinate_descent_sweeps(a_opt, max_sweeps=200,
                                                   time_limit=min(rem * 0.6, 200), verbose=True)
            try_seq(a_opt2, "cd2")
        
        # CD on symmetric version
        rem = deadline - time.monotonic() - 100
        if rem > 10:
            a_sym = symmetrize(best_seq)
            a_opt3, _ = coordinate_descent_sweeps(a_sym, max_sweeps=100,
                                                   time_limit=min(rem * 0.5, 100))
            try_seq(a_opt3, "cd_sym")
    
    print(f"  Phase 4 best: {best_score:.8f}")
    
    # === Phase 5: Alternating subgradient + CD ===
    print("\n=== Phase 5: Alternating SG + CD ===")
    remaining = deadline - time.monotonic() - 100
    
    for phase in range(4):
        rem = deadline - time.monotonic() - 60
        if rem < 15:
            break
        
        # Subgradient step
        t_sg_step = min(rem * 0.3, 60)
        lr_step = 0.001 / (phase + 1)
        a_opt, _ = subgradient_optimize(best_seq, time_limit=t_sg_step,
                                         lr=lr_step, mom=0.95, top_k=100,
                                         temp=0.00002 / (phase + 1))
        try_seq(a_opt, f"alt_sg_{phase}")
        
        # CD step
        rem = deadline - time.monotonic() - 50
        if rem < 10:
            break
        t_cd_step = min(rem * 0.5, 120)
        a_opt2, _ = coordinate_descent_sweeps(best_seq, max_sweeps=100,
                                               time_limit=t_cd_step)
        try_seq(a_opt2, f"alt_cd_{phase}")
        
        # Periodic perturbation + CD
        rem = deadline - time.monotonic() - 40
        if rem < 10:
            break
        a_period, _ = try_periodic_perturbation(best_seq, rng, num_freqs=15)
        try_seq(a_period, f"alt_per_{phase}")
        if a_period.sum() > 0:
            a_opt3, _ = coordinate_descent_sweeps(a_period, max_sweeps=50,
                                                   time_limit=min(rem * 0.4, 60))
            try_seq(a_opt3, f"alt_per_cd_{phase}")
    
    print(f"  Phase 5 best: {best_score:.8f}")
    
    # === Phase 6: Final intensive CD polish ===
    print("\n=== Phase 6: Final CD Polish ===")
    rem = deadline - time.monotonic() - 30
    if rem > 5:
        a_opt, _ = coordinate_descent_sweeps(best_seq, max_sweeps=500, time_limit=rem)
        try_seq(a_opt, "final_cd")
        
        # One more from symmetric
        rem = deadline - time.monotonic() - 15
        if rem > 5:
            a_sym = symmetrize(best_seq)
            a_opt2, _ = coordinate_descent_sweeps(a_sym, max_sweeps=200, time_limit=rem)
            try_seq(a_opt2, "final_cd_sym")
    
    # === Final verification ===
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")
    
    return [float(x) for x in best_seq]
