"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Strategy:
1. LSE (log-sum-exp) gradient descent with Adam optimizer and multi-peak weighting
2. Simulated annealing with pairwise mass-transfer moves
3. Exploration of different sequence lengths with structured initializations
4. Final high-precision polish
"""

import numpy as np
import time
import math

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1) > (cssv - 1))
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_optimize_v2(f0, time_limit, beta_start=10.0, beta_end=5000.0, 
                    lr_init=0.01, top_k=None, use_adam=True, seed=0):
    """LSE gradient descent with Adam and multi-peak weighting."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    if top_k is None:
        top_k = max(4, n // 16)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    adam_beta1 = 0.85
    adam_beta2 = 0.999
    eps = 1e-8
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)

        if top_k >= len(conv):

            if w[idx] < 1e-12:

            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx_range = np.arange(js, je + 1)
            grad[idx_range] += 2.0 * w[idx] * f[(k_int - idx_range) % n]  # WRONG!
        
        # Actually, let me use FFT for the gradient
        # grad[i] = 2 * sum_k w_extended[k] * f[k-i] where w_extended has w at peak positions
        # This is a cross-correlation: (w_extended * f_flipped)[offset]
        
        # Simpler: use the full convolution approach
        # w_full[k] = w[idx] if k is at peaks[idx], else 0
        # grad = 2 * cross_correlation(w_full, f)
        # cross_correlation(x, y)[i] = sum_k x[k] * y[k-i]
        # = convolution of x with reversed y, shifted

            k = int(peaks[idx])
            # Contribution to grad[i] from peak k: 2*wi*f[k-i] for 0<=k-i<n
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for i in range(lo, hi):
                ki = k - i
                if 0 <= ki < n:
                    grad[i] += 2.0 * wi * f[ki]

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1) > (cssv - 1))
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_multi_peak(f, peaks, weights):
    """Compute gradient of smoothed objective using top-k peak weighting.
    grad[i] = 2 * sum_k weights[k_idx] * f[peaks[k_idx] - i] for valid indices.
    """
    n = len(f)
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        wi = weights[idx]
        if wi < 1e-14:
            continue
        k = int(peaks[idx])
        # grad[i] gets contribution 2*wi*f[k-i] for 0 <= k-i < n
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for i in range(lo, hi):
            ki = k - i
            if 0 <= ki < n:
                grad[i] += 2.0 * wi * f[ki]
    return grad


def compute_gradient_fft(f, peaks, weights):

def lse_optimize_v2(f0, time_limit, beta_start=10.0, beta_end=5000.0,
                    lr_init=0.01, top_k=None, seed=0, use_fft_grad=True):
    """LSE gradient descent with Adam optimizer and multi-peak weighting."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    if top_k is None:
        top_k = max(4, n // 16)
    if top_k > 2 * n - 1:
        top_k = 2 * n - 1

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    # Adam state
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    adam_b1 = 0.85
    adam_b2 = 0.999
    eps = 1e-8

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Multi-peak softmax weighting
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            pv = conv.copy()
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]

        scaled = beta * pv
        scaled -= scaled.max()
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()

        # Compute gradient
        if use_fft_grad and n > 100:
            grad = compute_gradient_fft(f, peaks, w)
        else:
            grad = compute_gradient_multi_peak(f, peaks, w)

        # Adam update
        lr = lr_init / (1.0 + 0.02 * no_improve)
        m_vec = adam_b1 * m_vec + (1 - adam_b1) * grad
        v_vec = adam_b2 * v_vec + (1 - adam_b2) * (grad * grad)
        m_hat = m_vec / (1 - adam_b1 ** it)
        v_hat = v_vec / (1 - adam_b2 ** it)
        update = m_hat / (np.sqrt(v_hat) + eps)

        f_new = project_simplex(f - lr * update)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                # Perturb and restart
                scale = 0.001 + 0.002 * min(1.0, (no_improve - 300) / 200.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                m_vec = np.zeros(n)
                v_vec = np.zeros(n)
                no_improve = 0
                beta = min(beta * 1.5, beta_end)

        # Ramp beta
        if beta < beta_end:
            beta *= 1.003

        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-v2 it={it}, obj={best_obj:.10f}, beta={beta:.1f}, lr={lr:.6f}, no_imp={no_improve}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def simulated_annealing_transfer(f0, time_limit, seed=0, t_init=None, t_final=None,
                                  max_steps_per_s=500):
    """Simulated annealing with pairwise mass-transfer moves."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    cur_obj = best_obj

    if t_init is None:
        t_init = 0.01 * best_obj
    if t_final is None:
        t_final = 1e-7 * best_obj

    t0 = time.monotonic()
    total_time = time_limit
    it = 0
    step_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / total_time)
        # Geometric cooling
        temp = t_init * (t_final / t_init) ** progress

        # Pick random pair (i, j) with i != j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Determine transfer direction and amount
        # Transfer from the one with more mass to the one with less (biased)
        # Or just random direction
        if f[i] < 1e-15:
            continue

        # Propose delta: transfer from i to j (positive) or j to i (negative)
        max_pos = f[i] * 0.5  # max we can take from i
        max_neg = f[j] * 0.5  # max we can take from j
        # Random delta in [-max_neg, max_pos]
        delta = rng.uniform(-max_neg, max_pos)
        if abs(delta) < 1e-15:
            continue

        # Apply move
        f_new = f.copy()
        f_new[i] -= delta
        f_new[j] += delta
        if f_new[i] < 0 or f_new[j] < 0:
            continue

        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        # Metropolis acceptance
        delta_obj = obj_new - cur_obj
        if delta_obj < 0 or rng.random() < math.exp(-delta_obj / max(temp, 1e-15)):
            f = f_new
            cur_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_f = f.copy()

        step_count += 1
        if it % 5000 == 0:
            print(f"    SA it={it}, best={best_obj:.10f}, cur={cur_obj:.10f}, T={temp:.2e}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def smart_sa(f0, time_limit, seed=0):
    """Simulated annealing with peak-directed moves."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    cur_obj = best_obj

    t0 = time.monotonic()
    total_time = time_limit
    t_init = 0.005 * best_obj
    t_final = 1e-8 * best_obj
    it = 0

    # Only consider coordinates with significant mass
    active_thresh = 1e-8

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / total_time)
        temp = t_init * (t_final / t_init) ** progress

        # Find current peak
        conv = fast_convolve(f)
        k_peak = int(np.argmax(conv))

        # Find source: coordinate that contributes most to peak
        # Contribution of f[i] to b[k_peak] is 2*f[k_peak-i] (for valid k_peak-i)
        lo = max(0, k_peak - n + 1)
        hi = min(n, k_peak + 1)
        contributions = np.zeros(n)
        for i in range(lo, hi):
            ki = k_peak - i
            if 0 <= ki < n:
                contributions[i] = 2.0 * f[ki]

        # Source: high contribution, significant mass
        sources = np.where((contributions > 0.1 * contributions.max()) & (f > active_thresh))[0]
        if len(sources) == 0:
            sources = np.where(f > active_thresh)[0]
        if len(sources) == 0:
            sources = np.arange(n)

        # Target: low contribution to peak
        target_thresh = 0.1 * contributions.max()
        targets = np.where((contributions < target_thresh) & (f < 0.5))[0]
        if len(targets) == 0:
            targets = np.where(contributions < 0.5 * contributions.max())[0]
        if len(targets) == 0:
            targets = np.arange(n)

        i = int(sources[rng.integers(0, len(sources))])
        j = int(targets[rng.integers(0, len(targets))])
        if i == j:
            # Find another target
            candidates = np.delete(targets, i)
            if len(candidates) == 0:
                continue
            j = int(candidates[rng.integers(0, len(candidates))])

        # Transfer amount: small fraction of f[i]
        frac = rng.uniform(0.02, 0.15)
        delta = f[i] * frac
        if delta < 1e-15:
            continue

        f_new = f.copy()
        f_new[i] -= delta
        f_new[j] += delta

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        delta_obj = obj_new - cur_obj
        if delta_obj < 0 or rng.random() < math.exp(-delta_obj / max(temp, 1e-15)):
            f = f_new
            cur_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_f = f.copy()

        if it % 3000 == 0:
            print(f"    SmartSA it={it}, best={best_obj:.10f}, cur={cur_obj:.10f}, T={temp:.2e}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def structured_init_prime(n, p):
    """Create structured initialization based on quadratic residues mod p."""
    if n <= 0:
        return np.ones(1)
    # Map indices 0..n-1 to residues
    # Use quadratic residue indicator
    qr = np.zeros(n)
    for i in range(n):
        # Check if i is a quadratic residue mod p (or 0)
        if i % p == 0:
            qr[i] = 1.0
        else:
            # Euler's criterion: a^((p-1)/2) ≡ 1 (mod p) if QR
            r = pow(i % p, (p - 1) // 2, p)
            if r == 1:
                qr[i] = 1.0
            else:
                qr[i] = 0.3  # Non-QR gets some mass
    s = qr.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return qr / s
