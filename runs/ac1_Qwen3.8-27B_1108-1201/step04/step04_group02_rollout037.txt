```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated gradient
descent on a smooth (logsumexp) approximation of the max, with temperature
annealing and multi-scale exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_grad(f, T, fft_size=None):
    """
    Compute the smooth maximum T*logsumexp(conv(f,f)/T) and its gradient.
    Returns (smooth_value, gradient).
    """
    n = len(f)
    conv_val = fast_convolve(f)
    scaled = conv_val / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    w /= w_sum
    smooth_val = T * (np.log(w_sum) + scaled.max())
    
    # Gradient: grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * f[j]
    # Compute via FFT: grad[i] = 2 * (f * w_rev)[i + 2n - 2]
    w_rev = w[::-1]
    if fft_size is None:
        fft_size = 1 << (3 * n - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_full = np.fft.irfft(F * W, fft_size)[:3 * n - 2]
    grad = 2.0 * conv_full[2 * n - 2:3 * n - 2]
    
    return smooth_val, grad


def nesterov_optimize(f0, time_limit, T_start=None, T_end=None, verbose=True, seed=0):
    """
    Nesterov-accelerated gradient descent on smooth max of auto-convolution.
    f0 is on the simplex (sum=1, non-negative).
    """
    n = len(f0)
    f = f0.copy().astype(np.float64)
    
    # Compute initial peak for T initialization
    conv_init = fast_convolve(f)
    peak_init = conv_init.max()
    
    if T_start is None:
        T_start = peak_init * 20  # Start with broad "view"
    if T_end is None:
        T_end = peak_init * 0.05  # End with sharp "view"
    
    # Nesterov state
    y = f.copy()
    t_k = 1.0
    T = T_start
    
    # Track best
    best_f = f.copy()
    best_peak = peak_init
    
    # Annealing: exponential from T_start to T_end
    # We'll estimate total iterations and anneal accordingly
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    # Precompute FFT size
    fft_size = 1 << (3 * n - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit - 1:
        it += 1
        
        # Compute gradient at y
        smooth_val, grad_y = smooth_max_grad(y, T, fft_size)
        
        # Nesterov step size
        # Use lr = 1/t_k (standard Nesterov for 1-Lipschitz)
        # Scale by a factor to make steps meaningful
        # The gradient magnitude on the simplex is roughly O(peak/T)
        # We want the step to be roughly O(1/n) to explore the simplex
        lr = 1.0 / (t_k * np.linalg.norm(grad_y))
        lr = min(lr, 0.5)  # Cap step size
        lr = max(lr, 1e-6)  # Floor
        
        f_new = project_simplex(y - lr * grad_y)
        
        # Check convergence
        if np.max(np.abs(f_new - f)) < 1e-14:
            no_improve += 1
        else:
            no_improve = 0
        
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new
        y = f_new + beta * (f_new - f)
        y = project_simplex(y)
        f = f_new
        t_k = t_new
        
        # Anneal T
        # Exponential: T = T_start * (T_end/T_start)^(it/total_est)
        # But we don't know total_est, so use a slow exponential decay
        T = max(T * 0.9995, T_end)
        
        # Evaluate actual peak (every 10 iterations to save time)
        if it % 10 == 0 or it == 1:
            conv_f = fast_convolve(f)
            peak_f = conv_f.max()
            if peak_f < best_peak:
                best_peak = peak_f
                best_f = f.copy()
                if verbose and it % 1000 == 0:
                    rem = time_limit - (time.monotonic() - t0)
                    print(f"    it={it}, peak={peak_f:.8f}, T={T:.6f}, lr={lr:.6f}, rem={rem:.0f}s")
        
        # Escape from local minima
        if no_improve > 500:
            rng = np.random.default_rng(it * 7 + seed)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
            y = f.copy()
            t_k = 1.0
            no_improve = 0
        
        if it % 100 == 0:
            rem = time_limit - (time.monotonic() - t0)
            if rem < 20:
                break
    
    return best_f, best_peak


def parametric_search(n, time_limit, seed=0):
    """Quick 1D/2D parametric search for good starting points."""
    best_f = None
    best_obj = np.inf
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    # Geometric: a[i] = exp(t * (i - center))
    for t in np.geomspace(0.001, 5.0, 50):
        a = np.exp(t * (x - center))
        a = a / a.sum()
        conv = fast_convolve(a)
        obj = conv.max()
        if obj < best_obj:
            best_obj = obj
            best_f = a.copy()
    
    # Also try negative t (by symmetry should be same, but let's verify)
    for t in np.geomspace(0.001, 5.0, 20):
        a = np.exp(-t * (x - center))
        a = a / a.sum()
        conv = fast_convolve(a)
        obj = conv.max()
        if obj < best_obj:
            best_obj = obj
            best_f = a.copy()
    
    # Cosine: a[i] = (cos(pi * (i/n - 0.5))**p + eps)
    for p in [0.5, 1, 2, 3, 4, 6, 8]:
        a = (np.cos(np.pi * (x / n - 0.5)) ** p + 0.01)
        a = np.maximum(a, 0)
        a = a / a.sum()
        conv = fast_convolve(a)
        obj = conv.max()
        if obj < best_obj:
            best_obj = obj
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated gradient
descent on a smooth logsumexp approximation of max(auto-convolution), with
temperature annealing, multi-scale n exploration, and coordinate polishing.
Key idea: larger n lowers the theoretical bound (2n/(2n-1) -> 1), and
Nesterov convergence is much faster than subgradient methods.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_and_grad(f, T, fft_size):
    """
    Compute T*logsumexp(conv(f,f)/T) and its gradient w.r.t. f.
    Returns (smooth_value, gradient).
    """
    n = len(f)
    conv_val = fast_convolve(f)
    scaled = conv_val / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    w /= w_sum
    smooth_val = T * (np.log(w_sum) + scaled.max())

    # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * f[j]
    # = 2 * (f * w_rev)[i + 2n - 2]  where * is linear convolution
    w_rev = w[::-1]
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_full = np.fft.irfft(F * W, fft_size)[:3 * n - 2]
    grad = 2.0 * conv_full[2 * n - 2:3 * n - 2]

    return smooth_val, grad


def nesterov_optimize(f0, time_limit, verbose=False, seed=0):
    """
    Nesterov-accelerated gradient descent on smooth max of auto-convolution.
    Operates on the simplex (sum=1, f>=0).
    """
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv_init = fast_convolve(f)
    peak_init = conv_init.max()

    T_start = peak_init * 30.0
    T_end = peak_init * 0.03
    T = T_start

    fft_size = 1 << (3 * n - 1).bit_length()

    y = f.copy()
    t_k = 1.0
    best_f = f.copy()
    best_peak = peak_init

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    lr_base = 0.02

    while time.monotonic() - t0 < time_limit - 2:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        _, grad_y = smooth_max_and_grad(y, T, fft_size)

        grad_norm = np.linalg.norm(grad_y)
        if grad_norm < 1e-15:
            break

        lr = lr_base / t_k
        f_new = project_simplex(y - lr * grad_y / grad_norm * peak_init * 100)

        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new
        y = f_new + beta * (f_new - f)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys
        else:
            y = f_new.copy()

        f = f_new
        t_k = t_new

        # Anneal T
        T = max(T * 0.9997, T_end)

        # Evaluate actual peak periodically
        if it % 20 == 0:
            p = fast_convolve(f).max()
            if p < best_peak * (1 - 1e-10):
                best_peak = p
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 500 == 0:
                    print(f"      it={it}, obj={2*n*best_peak:.6f}, T/T_start={T/T_start:.4f}, rem={rem:.0f}s")
            else:
                no_improve += 1

        # Escape from stagnation
        if no_improve > 300:
            rng = np.random.default_rng(it * 13 + seed)
            scale = 0.02 * (1 + 0.5 * (no_improve - 300) / 300)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            y = f.copy()
            t_k = 1.0
            no_improve = 0

    # Final check
    p = fast_convolve(f).max()
    if p < best_peak:
        best_peak = p
        best_f = f.copy()

    return best_f, best_peak


def parametric_init(n, max_evals=200):
    """Generate diverse parametric initializations. Returns best found."""
    best_f = None
    best_obj = np.inf
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    evals = 0

    candidates = []

    # Geometric: exp(t * (i - center))
    for t in np.concatenate([
        np.geomspace(0.001, 2.0, 30),
        np.linspace(0, 8.0, 20),
    ]):
        a = np.exp(t * (x - center) / max(1, n * 0.1))
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            candidates.append(a / s)

    # Inverse geometric
    for t in np.geomspace(0.1, 5.0, 15):
        a = 1.0 / (1.0 + t * np.abs(x - center) / n + 1e-10)
        candidates.append(a / a.sum())

    # Cosine power
    for p in [0.3, 0.5, 1, 1.5, 2, 3, 4, 6, 8, 12]:
        a = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 2 * p
        a = np.arccos(1 - np.exp(-a)) + 0.001
        a = np.maximum(a, 0.001)
        candidates.append(a / a.sum())

    # Power law
    for p in np.linspace(-1.5, 1.5, 15):
        a = (x + 1.0) ** p
        a = np.maximum(a, 1e-10)
        candidates.append(a / a.sum())

    # Symmetric bump: exp(-t * (x-center)^2 / n)
    for t in np.geomspace(0.1, 20.0, 15):
        a = np.exp(-t * (x - center) ** 2 / n)
        candidates.append(a / a.sum())

    # Evaluate all
    for a in candidates:
        if evals >= max_evals:
            break
        evals += 1
        conv = fast_convolve(a)
        obj = conv.max()
        if obj < best_obj:
            best_obj = obj
            best_f = a.copy()

    if best_f is None:
        best_f = np.ones(n) / n
        best_obj = fast_convolve(best_f).max()

    return best_f, best_obj


def coordinate_polish(f0, time_limit, n_trials=5000, seed=0):
    """
    Local coordinate search: randomly perturb pairs of coordinates to reduce peak.
    """
    n = len(f0)
    f = f0.copy().astype(np.float64)
    conv = fast_convolve(f)
    best_peak = conv.max()
    best_f = f.copy()
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    trials = 0

    while time.monotonic() - t0 < time_limit - 2 and trials < n_trials:
        trials += 1
        if trials % 2000 == 0:
            rem = time_limit - (time.monotonic() - t0)
            if rem < 3:
                break

        # Pick two random indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # Try moving mass from i to j (or vice versa)
        for delta in [0.001, 0.005, 0.01, 0.02]:
            # Move from i to j
            if f[i] > delta + 1e-12:
                f_try = f.copy()
                f_try[i] -= delta
                f_try[j] += delta
                ct = fast_convolve(f_try)
                pt = ct.max()
                if pt < best_peak:
                    best_peak = pt
                    best_f = f_try.copy()
                    f = f_try
                    conv = ct
                break
            # Move from j to i
            elif f[j] > delta + 1e-12:
                f_try = f.copy()
                f_try[j] -= delta
                f_try[i] += delta
                ct = fast_convolve(f_try)
                pt = ct.max()
                if pt < best_peak:
                    best_peak = pt
                    best_f = f_try.copy()
                    f = f_try
                    conv = ct
                break

    return best_f, best_peak


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses Nesterov-accelerated smooth max optimization
    with temperature annealing, multi-n exploration, and coordinate polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, peak={fast_convolve(parent).max():.6f}")
    except Exception:
        pass

    # === Phase 1: Quick parametric search across multiple n values ===
    print("=== Phase 1: Parametric Search ===")
    t_phase1_end = min(60.0, budget_s * 0.05)

    n_candidates = [512, 768, 1024, 1536, 2048]
    if parent is not None:
        n_parent = len(parent)
        if n_parent not in n_candidates:
            n_candidates.insert(0, n_parent)

    param_results = {}  # n -> (best_f, best_peak)

    for n in n_candidates:
        if time.monotonic() > deadline - t_phase1_end - 100:
            break
        f_init, peak = parametric_init(n, max_evals=150)
        param_results[n] = (f_init, peak)
        obj = 2 * n * peak
        print(f"  n={n}: parametric best obj={obj:.6f}")

        # Try parent resampled
        if parent is not None:
            x_new = np.linspace(0, 1, n)
            x_old = np.linspace(0, 1, len(parent))
            p_resamp = np.interp(x_new, x_old, parent)
            p_resamp = np.maximum(p_resamp, 0)
            ps = p_resamp.sum()
            if ps > 0:
                p_resamp /= ps
                p_peak = fast_convolve(p_resamp).max()
                if p_peak < peak:
                    param_results[n] = (p_resamp, p_peak)
                    peak = p_peak
                    f_init = p_resamp
                    obj = 2 * n * peak
                    print(f"  n={n}: parent resample obj={obj:.6f}")

    # === Phase 2: Nesterov optimization on top candidates ===
    print("\n=== Phase 2: Nesterov Optimization ===")

    # Sort by parametric result
    sorted_ns = sorted(param_results.keys(), key=lambda n: param_results[n][1])
    top_ns = sorted_ns[:3]  # Focus on top 3 n values

    # Allocate time: more time for larger n (better potential)
    time_weights = {}
    for n in top_ns:
        time_weights[n] = 1.0 + 0.3 * n / max(top_ns)
    total_w = sum(time_weights.values())
    avail_time = deadline - time.monotonic() - 60
    for n in list(time_weights.keys()):
        time_weights[n] *= max(30, avail_time / total_w)

    for n in top_ns:
        if time.monotonic() > deadline - 40:
            break
        t_alloc = min(time_weights[n], deadline - time.monotonic() - 35)
        if t_alloc < 20:
            break

        f_init, _ = param_results[n]
        print(f"  Nesterov on n={n}, time={t_alloc:.0f}s")

        # Multiple restarts
        n_restarts = max(2, int(t_alloc // 60))
        t_per_restart = t_alloc / n_restarts

        for r in range(n_restarts):
            rem = deadline - time.monotonic() - 30
            if rem < 15:
                break
            t_r = min(t_per_restart, rem)
            if t_r < 10:
                break

            if r == 0:
                f_r = f_init.copy()
            elif r == 1:
                f_r = project_simplex(f_init + rng.standard_normal(n) * 0.01)
            elif r == 2:
                # Blend with uniform
                f_r = project_simplex(0.7 * f_init + 0.3 * np.ones(n) / n)
            else:
                f_r = project_simplex(f_init + rng.standard_normal(n) * 0.03)

            a_opt, peak = nesterov_optimize(f_r, time_limit=t_r, verbose=(r == 0), seed=seed + r)
            try_seq(a_opt, f"n{n}_r{r}")

    # === Phase 3: Try the specific n that gave best parametric result with focused runs ===
    print(f"\n=== Phase 3: Focused Optimization ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)

        # Several focused Nesterov runs with different configs
        n_focused = max(3, int((deadline - time.monotonic() - 30) // 50))
        for k in range(n_focused):
            rem = deadline - time.monotonic() - 25
            if rem < 15:
                break

            if k == 0:
                f_init = best_seq.copy()
            elif k == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.005)
            elif k == 2:
                f_init = project_simplex(0.9 * best_seq + 0.1 * np.ones(n_best) / n_best)
            elif k == 3:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)
            else:
                f_init = project_simplex(0.8 * best_seq + 0.2 * (best_seq[np.argsort(best_seq)][::-1]))

            t_alloc = min(rem - 5, 40)
            a_opt, _ = nesterov_optimize(f_init, time_limit=t_alloc, seed=seed + 100 + k)
            try_seq(a_opt, f"focus_{k}")

    # === Phase 4: Coordinate polishing ===
    print(f"\n=== Phase 4: Coordinate Polish ===")
    print(f"  Before polish: {best_score:.8f}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = coordinate_polish(best_seq, time_limit=min(rem, 45), seed=seed + 999)
            try_seq(a_opt, "polish")

        # Another round with more trials if time permits
        rem = deadline - time.monotonic() - 8
        if rem > 8:
            a_opt, _ = coordinate_polish(best_seq, time_limit=min(rem, 30),
                                         n_trials=20000, seed=seed + 998)
            try_seq(a_opt, "polish2")

    # === Phase 5: Try a few more n values if time remains ===
    rem = deadline - time.monotonic() - 10
    if rem > 30:
        print(f"\n=== Phase 5: Additional n exploration ===")
        extra_ns = [n for n in [384, 512, 768, 1024, 1536, 2048, 3072]
                    if n not in top_ns or n not in [len(best_seq) if best_seq is not None else 0]]
        for n in extra_ns[:3]:
            rem = deadline - time.monotonic() - 8
            if rem < 20:
                break
            f_init, _ = parametric_init(n, max_evals=80)
            if parent is not None:
                x_new = np.linspace(0, 1, n)
                x_old = np.linspace(0, 1, len(parent))
                p_r = np.maximum(np.interp(x_new, x_old, parent), 0)
                ps = p_r.sum()
                if ps > 0:
                    p_r /= ps
                    if fast_convolve(p_r).max() < fast_convolve(f_init).max():
                        f_init = p_r
            a_opt, _ = nesterov_optimize(f_init, time_limit=min(rem - 5, 30), seed=seed + 500)
            try_seq(a_opt, f"extra_n{n}")

    # Fallback
    if best_seq is None:
        print("  WARNING: No valid sequence found, using fallback")
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
