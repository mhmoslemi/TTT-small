import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, top_k=10, temperature=0.001):
    """Compute weighted subgradient of max(conv) using top-k peaks."""
    n = len(a)
    conv = fast_convolve(a)
    
    # Get top-k peaks
    k_actual = min(top_k, len(conv))
    peak_indices = np.argpartition(conv, -k_actual)[-k_actual:]
    
    # Softmax weights
    peak_vals = conv[peak_indices]
    scaled = peak_vals / temperature
    scaled -= scaled.max()
    weights = np.exp(scaled)
    weights /= weights.sum()
    
    # Compute subgradient
    sg = np.zeros(n)
    for i in range(len(peak_indices)):
        if weights[i] < 1e-10:
            continue
        k = int(peak_indices[i])
        # sg[m] += 2 * w_i * a[k-m] for 0 <= k-m < n
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        m_range = np.arange(js, je + 1)
        a_reflected = a[k - m_range]  # a[k-m] for m in m_range
        sg[m_range] += 2.0 * weights[i] * a_reflected
    
    return sg, conv.max()

def adam_subgradient_descent(a0, time_limit, top_k=20, lr=0.01, beta1=0.9, beta2=0.999, 
                              eps=1e-8, temperature=0.001, verbose=False):
    """Adam-based subgradient descent on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    
    best_conv_max = fast_convolve(a).max()
    best_a = a.copy()
    no_improve = 0
    
    t0 = time.monotonic()
    cur_temp = temperature
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Compute subgradient
        sg, cur_max = compute_subgradient(a, top_k=top_k, temperature=cur_temp)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Update in the negative subgradient direction
        a_raw = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to simplex
        a = project_simplex(a_raw)
        
        # Evaluate
        conv_max = fast_convolve(a).max()
        
        if conv_max < best_conv_max - 1e-12:
            best_conv_max = conv_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Anneal temperature
        cur_temp = max(temperature * 0.999, 0.0001)
        
        # Learning rate schedule
        lr_t = lr / (1 + 0.001 * t)
        
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(t)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            cur_temp = temperature
            lr_t = lr
        
        if verbose and t % 1000 == 0:
            print(f"    SGD iter {t}: conv_max={best_conv_max:.10f}, F={2*n*best_conv_max:.6f}")
    
    return best_a, best_conv_max

def compute_subgradient_fast(a, peak_indices, weights, fft_size):

    n = len(a)
    # Create weight sequence
    w = np.zeros(2*n-1)
    w[peak_indices] = weights

    # This equals conv(flip(w), a) evaluated at appropriate positions
    # Actually: (w ⋆ a)[m] = sum_k w[k] * a[k-m] = conv(flip(w), a)[m + (2n-2)]
    
    # Let's use the direct FFT approach
    # FFT(w) * conj(FFT(a_padded)) gives cross-correlation in frequency domain
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(W * np.conj(A), fft_size)
    
    # cross[m] = sum_k w[k] * conj(a[k-m]) ... hmm, for real signals
    # Actually for real signals: cross[m] = sum_k w[k] * a[(m-k) mod L]
    # This is circular cross-correlation
    
    # We want: sg[m] = 2 * sum_k w[k] * a[k-m] for k-m in [0, n-1]
    # = 2 * sum_k w[k] * a[k-m] (with a[i]=0 for i outside [0,n-1])
    
    # The circular cross-correlation gives us sum_k w[k] * a[(m-k) mod L]
    # For m in [0, n-1] and k in [0, 2n-2], k-m is in [-(n-1), 2n-2]
    # We need k-m in [0, n-1], so the circular version might have wrap-around issues
    
    # Let's just do it directly for simplicity
    sg = np.zeros(n)
    for i in range(len(peak_indices)):
        if weights[i] < 1e-12:
            continue
        k = int(peak_indices[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je+1] += 2.0 * weights[i] * a[k - je:k - js + 1][::-1]
    return sg

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast Adam subgradient descent
with multi-peak weighting, multi-restart exploration, adaptive hyperparameters,
and coordinate descent polishing. Explores multiple sequence lengths and
initializations to escape local minima.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize_simplex(a):
    """Fast clip-and-normalize to simplex (approximate projection)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_subgradient(a, top_k, temperature, fft_size=None):
    """Compute weighted subgradient of max(conv) using top-k peaks with softmax weighting."""
    n = len(a)
    conv = fast_convolve(a)
    cur_max = conv.max()

    k_actual = min(top_k, len(conv))
    peak_indices = np.argpartition(conv, -k_actual)[-k_actual:]
    peak_vals = conv[peak_indices]

    if temperature < 1e-10:
        weights = np.zeros(k_actual)
        weights[np.argmax(peak_vals)] = 1.0
    else:
        scaled = peak_vals / temperature
        scaled -= scaled.max()
        weights = np.exp(scaled)
        ws = weights.sum()
        if ws < 1e-15:
            weights = np.zeros(k_actual)
            weights[np.argmax(peak_vals)] = 1.0
        else:
            weights /= ws

    sg = np.zeros(n)
    for i in range(k_actual):
        if weights[i] < 1e-12:
            continue
        k = int(peak_indices[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * weights[i] * a[k - idx]

    return sg, cur_max, conv


def adam_subgrad_search(a0, time_limit, top_k=20, base_lr=0.05, beta1=0.85,
                        beta2=0.999, eps=1e-8, base_temp=0.005, verbose=False,
                        restart_every=300, restart_scale=0.003):
    """Adam-based subgradient descent on the simplex with restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    conv = fast_convolve(a)
    best_conv_max = conv.max()
    best_a = a.copy()
    no_improve = 0

    t0 = time.monotonic()
    cur_temp = base_temp

    while time.monotonic() - t0 < time_limit:
        t += 1
        lr = base_lr / (1.0 + 0.0005 * t)

        sg, cur_max, conv = compute_subgradient(a, top_k, cur_temp)

        m = beta1 * m + (1.0 - beta1) * sg
        v = beta2 * v + (1.0 - beta2) * (sg * sg)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_raw = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = normalize_simplex(a_raw)

        conv_max = fast_convolve(a).max()

        if conv_max < best_conv_max - 1e-13:
            best_conv_max = conv_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_temp = max(base_temp * (0.9995 ** t), 0.0005)

        if no_improve > restart_every:
            rng = np.random.default_rng(t * 17 + 3)
            noise = rng.standard_normal(n) * restart_scale
            a = project_simplex(best_a + noise)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            cur_temp = base_temp

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam iter {t}, elapsed={elapsed:.1f}s, "
                  f"conv_max={best_conv_max:.10f}, F={2*n*best_conv_max:.6f}")

    return best_a, best_conv_max


def coordinate_descent_polish(a0, time_limit, n_trials=6):
    """Coordinate descent: try adjusting individual coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_conv_max = fast_convolve(a).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-10:
                continue

            multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]
            best_j_val = old_val
            best_j_obj = fast_convolve(a).max()

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = a.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial /= f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a /= a.sum()

                if best_j_obj < best_conv_max - 1e-14:
                    best_conv_max = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved and passes > 2:
            break

    return best_a, best_conv_max


def interpolate_sequence(f, new_n):
    """Linear interpolation of a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
