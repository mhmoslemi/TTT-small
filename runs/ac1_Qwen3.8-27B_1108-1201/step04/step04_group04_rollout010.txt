import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Optimized step function coefficient minimizer using:
1. LP-based Frank-Wolfe with adaptive constraint relaxation
2. Nesterov-accelerated projected gradient on log-sum-exp smoothed objective
3. Active-set subgradient polishing focused on top peaks
4. Multi-length exploration and diverse initializations

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(f, T):
    """Compute gradient of T*log(sum(exp(conv(f)/T))) w.r.t. f."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = w / ws

    fft_size = 1 << (3 * n).bit_length()
    A = np.fft.rfft(f, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(np.conj(A) * P, fft_size)
    return 2.0 * cross[:n], conv


def lp_direction(f, M):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    n_cons = 2 * n - 1
    c = -np.ones(n)
    A_ub = np.zeros((n_cons, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(n_cons, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None

def nesterov_smoothed(f0, time_limit, T_init=0.004, T_min=0.00003, lr_init=0.008, verbose=False):
    """Nesterov-accelerated projected gradient on log-sum-exp smoothed max-conv."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    y = f.copy()
    t_mom = 1.0
    T = T_init
    lr = lr_init

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute gradient at momentum point y
        grad_y, _ = compute_smoothed_grad(y, T)
        grad_y = grad_y - np.mean(grad_y)

        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new

        # Gradient step
        x_new = project_simplex(y - lr * grad_y)

        # Nesterov extrapolation
        f_new = x_new + beta * (x_new - f)
        f_new = project_simplex(f_new)

        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            y = x_new
            t_mom = t_new
            if verbose and it % 200 == 0:

        else:
            no_improve += 1
            if no_improve % 40 == 0:
                lr *= 0.88

        # Anneal
        T = max(T * 0.9975, T_min)
        lr = max(lr * 0.9997, 0.00005)

        # Reset momentum periodically
        if no_improve > 150:
            f = best_f.copy()
            y = best_f.copy()
            t_mom = 1.0
            no_improve = 0

        # Full restart
        if no_improve > 400:
            rng = np.random.default_rng((it * 37 + 11) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0012)

            t_mom = 1.0

            no_improve = 0

def generate_inits(n, parent, rng):
    """Generate diverse initializations for optimization."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.25))

    x = np.arange(n, dtype=np.float64)
    # U-shape (edge concentrated)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Half-sine
    hs = np.sin(np.pi * (x + 0.5) / n) + 0.1
    inits.append(hs / hs.sum())

    # Two humps
    h1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.12 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.12 * n)) ** 2)
    two_hump = h1 + h2 + 0.01
    inits.append(two_hump / two_hump.sum())

    # Mirror symmetric random
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.4)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    if n % 2 == 1:
        sym[half_n] = half[half_n - 1]
    inits.append(sym / sym.sum())

    # Cosine with oscillation
    c1 = 0.5 + 0.3 * np.cos(2 * np.pi * x / n) + 0.05 * np.cos(4 * np.pi * x / n) + 0.1
    inits.append(c1 / c1.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

            lr_start = max(0.004, 0.006 * fast_convolve(f_work).max())
            print(f"  Nesterov sub-phase {sub} (t={t_alloc:.0f}s, T={T_start:.5f}, lr={lr_start:.5f})")
            a_opt, _ = nesterov_smoothed(f_work, time_limit=t_alloc,
                                          T_init=T_start, T_min=0.00002,
                                          lr_init=lr_start, verbose=True)
            sc = try_seq(a_opt, f"p2_nest_{sub}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()
                n_work = len(f_work)

    print(f"  Phase 2 best: {best_score:.6f}")

    # === Phase 3: LP-FW refinement (300s) ===
    print(f"\n=== Phase 3: LP-FW Refinement ===")
    t3_start = time.monotonic()
    p3_budget = 300.0

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()

        # Two rounds of FW with decreasing perturbation
        for round_i in range(2):
            rem = p3_budget - (time.monotonic() - t3_start)
            if rem < 30:
                break
            t_alloc = min(120.0, rem - 20.0)
            print(f"  FW round {round_i} (t={t_alloc:.0f}s)")
            a_opt, _ = fw_lp_optimize(f_work, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, f"p3_fw_{round_i}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

    print(f"  Phase 3 best: {best_score:.6f}")

    # === Phase 4: Active-set polish (300s) ===
    print(f"\n=== Phase 4: Active-Set Polish ===")
    t4_start = time.monotonic()
    p4_budget = 300.0

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()

        configs = [
            {"k_top": 20, "temp_init": 0.0001},
            {"k_top": 10, "temp_init": 0.00005},
            {"k_top": 8, "temp_init": 0.00002},
            {"k_top": 5, "temp_init": 0.00001},
        ]

        for i, cfg in enumerate(configs):
            rem = p4_budget - (time.monotonic() - t4_start)
            if rem < 25:
                break
            t_alloc = min(70.0, rem - 15.0)
            print(f"  AS polish {i} (k={cfg['k_top']}, t={t_alloc:.0f}s)")
            a_opt, _ = active_set_polish(f_work, time_limit=t_alloc, **cfg, verbose=True)
            try_seq(a_opt, f"p4_as_{i}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

    print(f"  Phase 4 best: {best_score:.6f}")

    # === Phase 5: Multi-length exploration (200s) ===
    print(f"\n=== Phase 5: Multi-Length Exploration ===")
    t5_start = time.monotonic()
    p5_budget = 200.0

    if best_seq is not None:
        f_best = best_seq / best_seq.sum()
        n_best = len(f_best)

        # Try a few lengths near the best
        for delta in [8, -8, 32, -32, 48]:
            rem = p5_budget - (time.monotonic() - t5_start)
            if rem < 25:
                break
            new_n = n_best + delta
            if new_n < 128 or new_n > 1024:
                continue

            px = np.interp(np.linspace(0, 1, new_n),
                           np.linspace(0, 1, n_best), f_best)
            px = project_simplex(px)

            # Quick FW + Nesterov
            t_each = min(30.0, rem / 2.0 - 10.0)
            if t_each < 10:
                continue
            a_opt, _ = fw_lp_optimize(px, time_limit=t_each)
            sc = try_seq(a_opt, f"p5_n{new_n}_fw")

            rem = p5_budget - (time.monotonic() - t5_start)
            if rem > 20 and best_seq is not None:
                f_new = best_seq / best_seq.sum()
                t_each2 = min(25.0, rem - 10.0)
                if t_each2 > 10:
                    a_opt2, _ = nesterov_smoothed(f_new, time_limit=t_each2,
                                                    T_init=0.002, T_min=0.00002,
                                                    lr_init=0.005)
                    try_seq(a_opt2, f"p5_n{new_n}_nest")

    print(f"  Phase 5 best: {best_score:.6f}")

    # === Phase 6: Final micro-polish (remaining time) ===
    print(f"\n=== Phase 6: Final Micro-Polish ===")

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()
        n_work = len(f_work)

        # Tiny perturbations + quick polish
        for i in range(5):
            if time.monotonic() > deadline - 20:
                break
            noise_scale = 0.0003 * (i + 1)
            f_pert = project_simplex(f_work + rng.standard_normal(n_work) * noise_scale)

            # Quick active set polish
            a_opt, _ = active_set_polish(f_pert, time_limit=15, k_top=5, temp_init=0.00001)
            try_seq(a_opt, f"p6_pert_{i}")

            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # One final very short FW
        if time.monotonic() < deadline - 30:
            a_opt, _ = fw_lp_optimize(f_work, time_limit=25)
            try_seq(a_opt, "p6_final_fw")

        # Mirror symmetry check
        if time.monotonic() < deadline - 20:
            n_m = len(f_work)
            mirrored = 0.5 * (f_work + f_work[::-1])
            mirrored = project_simplex(mirrored)
            a_opt, _ = active_set_polish(mirrored, time_limit=10, k_top=5, temp_init=0.00001)
            try_seq(a_opt, "p6_mirror")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
