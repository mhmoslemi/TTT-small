    fft_size = 1 << (size - 1).bit_length()

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) LP-based Frank-Wolfe with golden-section line search,
(2) Smoothed gradient descent with Nesterov momentum,
(3) Peak-aware subgradient polishing with adaptive temperature,
(4) Multi-length exploration and solution pool averaging.
Key improvements: momentum-based gradient, better temperature scheduling,
solution pool with periodic re-optimization, and adaptive peak weighting.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_obj(f):
    """Compute 2*n*max_conv for normalized f."""
    n = len(f)
    conv = fast_convolve(f)
    return conv.max()


def build_lp_matrix(f):
    """Build the LP constraint matrix for the Frank-Wolfe direction."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def fw_lp_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with golden-section line search and adaptive constraint."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break

        g = solve_lp_direction(f, M * m_exp)
        if g is None:
            m_exp *= 1.15
            if m_exp > 8.0:
                m_exp = 1.0
                no_improve += 5
            if no_improve > 60:
                _fw_restart(f, best_f, n, it)
                f = f.copy()
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 3
            m_exp = min(m_exp * 1.05, 5.0)
            if no_improve > 80:
                break
            continue

        # Golden section line search
        best_alpha = 0.0
        best_val = M
        lo, hi = 0.0, 1.0
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = 1.0 - gr
        c2 = gr

        x1 = c1 * lo + c2 * hi
        x2 = c2 * lo + c1 * hi

        ft1 = np.maximum((1.0 - x1) * f + x1 * g, 0.0)
        ft1 = ft1 / ft1.sum()
        v1 = fast_convolve(ft1).max()

        ft2 = np.maximum((1.0 - x2) * f + x2 * g, 0.0)
        ft2 = ft2 / ft2.sum()
        v2 = fast_convolve(ft2).max()

        for _ in range(18):
            if remaining < 0.05:
                break
            if v1 < v2:
                hi = x2
                x2 = x1
                v2 = v1
                x1 = c1 * lo + c2 * hi
                ft1 = np.maximum((1.0 - x1) * f + x1 * g, 0.0)
                ft1 = ft1 / ft1.sum()
                v1 = fast_convolve(ft1).max()
            else:
                lo = x1
                x1 = x2
                v1 = v2
                x2 = c2 * lo + c1 * hi
                ft2 = np.maximum((1.0 - x2) * f + x2 * g, 0.0)
                ft2 = ft2 / ft2.sum()
                v2 = fast_convolve(ft2).max()

        best_alpha = (lo + hi) / 2.0
        best_val = min(v1, v2)

        if best_alpha > 1e-10 and best_val < M - 1e-14:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 5.0)

        if no_improve > 50:
            _fw_restart(f, best_f, n, it)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def _fw_restart(f, best_f, n, it):
    """Perturb f away from local optimum."""
    rng = np.random.default_rng((it * 7919 + n * 104729) % (2**31))
    scale = 0.003 + 0.003 * (it % 5)
    noise = rng.standard_normal(n) * scale
    f[:] = project_simplex(best_f + noise)
    return f


def smoothed_gradient(conv, a, T):
    """Compute gradient of T*log(sum(exp(conv/T))) w.r.t. a using FFT cross-correlation."""
    n = len(a)
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = w / ws
    fft_size = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(np.conj(A) * P, fft_size)
    return 2.0 * cross[:n]


def sgd_with_momentum(f0, time_limit, T_init=0.004, T_min=0.00005, momentum=0.85,
                      lr_init=0.01, verbose=False):
    """Gradient descent with Nesterov momentum on simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    T = T_init
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    velocity = np.zeros(n)
    f_prev = f.copy()

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.05:
            break

        # Nesterov lookahead
        f_look = project_simplex(f + momentum * (f - f_prev))
        f_prev = f.copy()

        conv = fast_convolve(f_look)
        grad = smoothed_gradient(conv, f_look, T)
        grad = grad - np.mean(grad)

        velocity = momentum * velocity - lr * grad
        f = project_simplex(f + velocity)

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    SGD+M iter {it}: obj={best_obj:.10f}, T={T:.6f}, lr={lr:.6f}")
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr = max(lr * 0.9, 0.00005)

        # Adaptive temperature
        T = max(T * 0.998, T_min)

        # Adaptive momentum
        if no_improve > 200:
            momentum = max(momentum - 0.01, 0.6)

        # Restart
        if no_improve > 400:
            rng = np.random.default_rng((it * 331 + 17) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            velocity = np.zeros(n)
            f_prev = f.copy()
            lr = max(lr * 2.0, 0.005)
            T = max(T_init * 0.4, T_min)
            no_improve = 0
            momentum = 0.85

    return best_f, best_obj


def peak_aware_subgradient(f0, time_limit, lr_init=0.001, temp_init=0.00003,
                           top_k_frac=0.1, verbose=False):
    """Subgradient method with peak-aware weighting (squared excess over mean)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    T = temp_init
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    top_k = max(3, int(n * top_k_frac))

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.05:
            break

        conv = fast_convolve(f)
        conv_mean = conv.mean()

        # Peak-aware: weight by (peak - mean)^2 for top peaks
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peak_idx]
        excess = pv - conv_mean
        # Use squared excess for sharper focus on the worst peaks
        w = excess ** 2 + 1e-10
        w = w / w.sum()

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peak_idx)):
            if w[i] < 1e-12:
                continue
            k = int(peak_idx[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        f = project_simplex(f - lr * sg)
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive cooling
        if it % 50 == 0:
            lr *= 0.998
            T = T * 1.01

        # Restart
        if no_improve > 300:
            rng = np.random.default_rng((it * 733 + 41) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            lr = lr_init * 0.5
            no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)

    # U-shape (edge concentrated)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.8
    inits.append(edge_w / edge_w.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / (n - 1))) + 0.02
    inits.append(c1 / c1.sum())

    # Two humps
    h1 = np.exp(-0.5 * ((x - 0.2 * (n - 1)) / (0.12 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.8 * (n - 1)) / (0.12 * n)) ** 2)
    inits.append((h1 + h2) / (h1 + h2).sum())

    # Modulated uniform
    for freq in [1, 2, 3]:
        mod = 1.0 + 0.15 * np.cos(2.0 * np.pi * freq * x / n)
        inits.append(mod / mod.sum())

    # Random symmetric
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.4)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    inits.append(sym / sym.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def try_and_update(seq, label, state):
    """Evaluate a sequence and update best if better."""
    if seq is None:
        return
    a = np.maximum(seq, 0.0)
    a = np.clip(a, 0.0, 1000.0)

        state['best_score'] = score
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return score


def interpolate_seq(source, target_n):
    """Interpolate a sequence to a different length."""
    src_n = len(source)
    if src_n == target_n:
        return source.copy()
    x_src = np.linspace(0, 1, src_n)
    x_tgt = np.linspace(0, 1, target_n)
    result = np.interp(x_tgt, x_src, source)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-10:
        result = result / s
    return result

    rem = p5_budget - (time.monotonic() - t_start)
    t_alloc = min(30.0, rem - 5.0)
    if t_alloc > 5:
        a_opt, _ = peak_aware_subgradient(f_work, time_limit=t_alloc,
                                          lr_init=0.0003, temp_init=0.00001,
                                          top_k_frac=0.05)
        try_and_update(a_opt, "p5_sg_fine", state)

    # Final tiny LP
    if state['best_seq'] is not None:
        f_work = state['best_seq'] / state['best_seq'].sum()
    rem = p5_budget - (time.monotonic() - t_start)
    t_alloc = min(25.0, rem - 5.0)
    if t_alloc > 5:
        a_opt, _ = fw_lp_optimize(f_work, time_limit=t_alloc)
        try_and_update(a_opt, "p5_fw_final", state)

    # Average with best and re-optimize
    if state['best_seq'] is not None:
        f_work = state['best_seq'] / state['best_seq'].sum()
        rem = p5_budget - (time.monotonic() - t_start)
        if rem > 10:
            # Mirror average
            mirrored = 0.5 * f_work + 0.5 * f_work[::-1]
            mirrored = project_simplex(mirrored)
            if np.max(np.abs(mirrored - f_work)) > 1e-6:
                a_opt, _ = peak_aware_subgradient(mirrored, time_limit=10,
                                                  lr_init=0.0005, temp_init=0.00002)
                try_and_update(a_opt, "p5_mirror", state)

    # Final verification
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(384) / 384
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))

    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs
    else:
        state['best_seq'] = np.ones(384) / 384
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))

    print(f"\nFINAL SCORE: {state['best_score']:.6f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
