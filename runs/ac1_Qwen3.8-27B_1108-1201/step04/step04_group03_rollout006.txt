def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def multiplicative_weight_v2(f0, time_limit, lr_init=0.01, top_k=20, 
                             temp_init=0.0005, nesterov_mom=0.9):
    """
    Improved multiplicative weight method with:
    - Nesterov momentum on the log-scale
    - Adaptive top-k selection
    - Better temperature scheduling
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    log_f = np.log(np.maximum(f, 1e-300))
    v = np.zeros(n)  # momentum on log-scale
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init

        scaled -= scaled.max()

        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(peaks))
            w[0] = 1.0
        else:
            w /= ws
        
        # Compute gradient signal
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += w[i] * 2.0 * f[k - idx]

        v = nesterov_mom * v - lr * grad
        log_f = log_f + v  # Note: v has negative gradient, so this increases where gradient is low

        f_new = np.exp(log_f)
        f_new = np.maximum(f_new, 1e-30)
        f_new = f_new / f_new.sum()
        log_f = np.log(f_new)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

            best_obj = obj_new
            best_f = f_new.copy()

        lr *= 0.9997
        temp *= 1.001

            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            log_f = np.log(np.maximum(f, 1e-300))
            v = np.zeros(n)
            lr = max(lr_init * 0.5, 0.001)
            no_improve = 0

def projected_gradient_line_search(f0, time_limit, t_init=0.0005):
    """

        # Line search: find best alpha
        best_alpha = 0.0
        best_f_new = f.copy()
        best_obj_new = obj

            if obj_try < best_obj_new:

def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []

    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    edge2 = (1 + np.cos(2 * np.pi * x / n)) ** 2 + 0.01
    inits.append(edge2 / edge2.sum())

        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:

def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def peak_rebalancing(f0, time_limit):
    """
    New approach: iteratively identify the top peaks and redistribute mass
    from positions that contribute most to those peaks to positions that
    contribute less, while maintaining the sum constraint.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = 0.005
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find the top peak
        k_star = np.argmax(conv)
        
        # Gradient at the peak: d(conv[k*])/d(f[j]) = 2*f[k*-j] for valid j
        j_valid = np.arange(max(0, k_star - n + 1), min(n, k_star + 1))
        grad_peak = 2.0 * f[k_star - j_valid]
        
        # We want to decrease f where grad_peak is large
        # and increase f where grad_peak is small
        # Use a "redistribution" approach:
        # delta[j] = -lr * grad_peak[j] + lr * mean(grad_peak)
        # This maintains the sum
        
        delta = np.zeros(n)
        delta[j_valid] = -lr * grad_peak + lr * grad_peak.mean()
        
        f_new = f + delta
        f_new = project_simplex(f_new)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            lr *= 0.5
            if lr < 1e-6:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                lr = 0.003
                no_improve = 0
        elif no_improve > 10:
            lr *= 0.9
    
    return best_f, best_obj

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def smooth_max_value_and_grad(x, t):
    """
    Compute smooth max objective and gradient in softmax parameterization.
    Objective: n * (max(conv) + t * log(sum(exp(conv/t)) - 1))
    which approximates n * max(conv) as t->0.
    Returns (value, grad_x).
    """
    a = softmax(x)
    n = len(a)
    conv = fast_convolve(a)
    
    # Smooth max
    conv_max = conv.max()
    conv_shifted = conv - conv_max
    exp_conv = np.exp(conv_shifted / t)
    log_sum = np.log(exp_conv.sum())
    smooth_val = conv_max + t * log_sum
    obj = n * smooth_val
    
    # Softmax weights over peaks
    p = exp_conv / exp_conv.sum()
    
    # Gradient w.r.t. a: grad_a[j] = 2 * sum_k p[k] * a[k-j]
    # Compute via convolution: grad_a = 2 * correlate(p, a)
    # grad_a[j] = 2 * sum_{m=0}^{n-1} a[m] * p[j+m]
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = np.fft.rfft(a_rev, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_ap = np.fft.irfft(A_rev * P, fft_size)
    grad_a = 2.0 * conv_ap[n - 1:2 * n - 1]
    
    # Gradient w.r.t. x: grad_x[i] = a[i] * (grad_a[i] - sum_j a[j]*grad_a[j])
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    
    return obj, grad_x


def smooth_max_value(x, t):
    """Just the value (for scipy minimize)."""
    a = softmax(x)
    n = len(a)
    conv = fast_convolve(a)
    conv_max = conv.max()
    conv_shifted = conv - conv_max
    exp_conv = np.exp(conv_shifted / t)
    return n * (conv_max + t * np.log(exp_conv.sum()))


def lbgfs_smooth_optimize(f0, time_limit, t_init=0.0008, t_end=0.00005, max_iter=5000):
    """
    L-BFGS optimization of smooth max in softmax space.
    Uses scipy's L-BFGS-B with custom gradient.
    """
    n = len(f0)
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Initialize x from a (inverse softmax)
    x = np.log(np.maximum(a, 1e-30))
    
    # Convex combination parameter to interpolate between temperatures
    best_t = t_init
    best_x = x.copy()
    best_val = float('inf')
    
    t0 = time.monotonic()
    t = t_init
    n_iters = 0
    
    while t0 + time.monotonic() < time_limit + t0 and time.monotonic() - t0 < time_limit:
        if t < t_end:
            break
        
        # Run L-BFGS at current temperature
        def obj_func(xv):
            return smooth_max_value(xv, t)
        
        def grad_func(xv):
            _, g = smooth_max_value_and_grad(xv, t)
            return g
        
        try:
            res = minimize(obj_func, x, method='L-BFGS-B', jac=grad_func,
                          options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-8,
                                   'maxls': 50})
        except Exception:
            break
        
        n_iters += 1
        x_new = res.x
        a_new = softmax(x_new)
        conv = fast_convolve(a_new)
        actual_val = n * conv.max()
        
        if actual_val < best_val:
            best_val = actual_val
            best_x = x_new.copy()
            best_t = t
        
        # Only keep the new x if it improved
        if actual_val < best_val + 0.01 * best_val:
            x = x_new
        
        # Anneal temperature
        t *= 0.92
        n_iters += 1
        
        if time.monotonic() - t0 > time_limit:
            break
    
    a_best = softmax(best_x)
    return a_best, best_val

def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def spectral_equalize(f0, time_limit, alpha_init=0.08):
    """
    Iteratively flatten the power spectrum to reduce auto-correlation peaks.
    Uses the fact that auto-correlation = IFFT(|FFT(a)|^2).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    alpha = alpha_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute power spectrum
        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(f, fft_size)
        P = np.abs(A) ** 2
        
        # Flattening: reduce high-P bins, boost low-P bins
        P_mean = P.mean()
        P_std = P.std() + 1e-10
        modulation = np.exp(-alpha * (P - P_mean) / P_std)
        A_new = A * modulation
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            break
        a_new = a_new / a_new_sum
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = a_new.copy()
            f = a_new
            no_improve = 0
        else:
            f = a_new  # still update, might help later
            no_improve += 1
        
        if no_improve > 30:
            alpha *= 0.8
            if alpha < 0.005:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                alpha = alpha_init * 0.5
                no_improve = 0
    
    return best_f, best_obj


def mult_weight_v2(f0, time_limit, lr_init=0.008, top_k=15,
                   temp_init=0.0003, momentum=0.85):
    """
    Multiplicative weight with Nesterov momentum on log-scale.

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    log_f = np.log(np.maximum(f, 1e-300))
    v = np.zeros(n)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        f = np.exp(log_f)
        f = f / f.sum()
        
        conv = fast_convolve(f)
        
        k_sel = max(5, min(top_k, len(conv)))
        peaks = np.argpartition(conv, -k_sel)[-k_sel:]
        
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(peaks))
            w[0] = 1.0
        else:
            w /= ws
        
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += w[i] * 2.0 * f[k - idx]
        
        v = momentum * v - lr * grad
        log_f = log_f + v
        
        f_new = np.exp(log_f)
        f_new = np.maximum(f_new, 1e-30)
        f_new = f_new / f_new.sum()
        log_f = np.log(f_new)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        lr *= 0.9997
        temp *= 1.001
        
        if no_improve > 120:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            log_f = np.log(np.maximum(f, 1e-300))

    return best_f, best_obj


def peak_rebalancing(f0, time_limit, lr_init=0.004):
    """
    Iteratively reduce the highest peaks by redistributing mass.
    Uses adaptive step size and multi-peak awareness.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find top peaks above threshold
        threshold = M * 0.95
        peak_indices = np.where(conv >= threshold)[0]
        
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(conv)])
        
        # Compute weighted gradient
        pv = conv[peak_indices]
        weights = (pv / pv.sum()) ** 2  # emphasize higher peaks more
        
        grad = np.zeros(n)
        for i in range(len(peak_indices)):
            if weights[i] < 1e-10:
                continue
            k = int(peak_indices[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += weights[i] * 2.0 * f[k - idx]
        
        # Project gradient to tangent space of simplex
        grad_mean = grad.mean()
        grad_tangent = grad - grad_mean
        
        # Try a few step sizes
        best_step = 0
        best_obj_step = M
        for step in [lr, lr * 0.5, lr * 2]:
            f_try = f - step * grad_tangent
            f_try = project_simplex(f_try)
            conv_try = fast_convolve(f_try)
            obj_try = conv_try.max()
            if obj_try < best_obj_step:
                best_obj_step = obj_try
                best_step = step
                f_new = f_try
        
        if best_step > 0:
            f = f_new
            if best_obj_step < best_obj - 1e-15:
                best_obj = best_obj_step
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 30:
            lr *= 0.7
            if lr < 1e-5:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                lr = lr_init * 0.5
                no_improve = 0
    
    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())
    
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())
    
    # Multi-frequency structure
    mf = 1.0
    for freq in [2, 3, 5]:
        mf = mf * (1.0 + 0.3 * np.cos(2.0 * np.pi * freq * x / n))
    inits.append(mf / mf.sum())
    
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits


def try_and_update(a, best_seq, best_score, label):
    """Evaluate sequence and update best if improved. Returns (best_seq, best_score, score)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_seq, best_score, float('inf')
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score, score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: FW-LP on parent (300s)
    Phase 2: L-BFGS smooth max optimization (300s)
    Phase 3: MW with momentum, multiple configs (250s)
    Phase 4: Spectral equalization (200s)
    Phase 5: Peak rebalancing + multi-n exploration (250s)
    Phase 6: Final L-BFGS polish + perturbations (remainder)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            sc = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={sc:.8f}")
            best_seq, best_score, _ = try_and_update(parent, best_seq, best_score, "parent")
            if not np.isfinite(best_score):
                parent = None
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        parent = np.ones(384) / 384
        best_seq, best_score, _ = try_and_update(parent, best_seq, best_score, "uniform384")
    
    # === Phase 1: FW-LP on parent ===
    print("\n=== Phase 1: FW-LP on Parent ===")
    rem = deadline - time.monotonic() - 1300
    if rem > 30:
        t_alloc = min(rem, 300)
        print(f"  FW-LP, n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        best_seq, best_score, _ = try_and_update(a_opt, best_seq, best_score, "fw_parent")
    print(f"  Phase 1 best: {best_score:.8f}")
    
    # === Phase 2: L-BFGS smooth max ===
    print("\n=== Phase 2: L-BFGS Smooth Max ===")
    rem = deadline - time.monotonic() - 900
    if rem > 60:
        t_alloc = min(rem, 300)
        print(f"  L-BFGS smooth, t={t_alloc:.0f}s")
        a_opt, _ = lbgfs_smooth_optimize(best_seq, time_limit=t_alloc,
                                         t_init=0.001, t_end=0.00008)
        best_seq, best_score, _ = try_and_update(a_opt, best_seq, best_score, "lbgfs")
    print(f"  Phase 2 best: {best_score:.8f}")
    
    # === Phase 3: MW with momentum ===
    print("\n=== Phase 3: MW with Momentum ===")
    mw_configs = [
        {'lr_init': 0.006, 'top_k': 12, 'temp_init': 0.0002, 'momentum': 0.82},
        {'lr_init': 0.012, 'top_k': 20, 'temp_init': 0.0004, 'momentum': 0.88},
        {'lr_init': 0.004, 'top_k': 8, 'temp_init': 0.00015, 'momentum': 0.78},
    ]
    rem = deadline - time.monotonic() - 500
    t_per = max(20, rem // max(1, len(mw_configs)))
    for ci, cfg in enumerate(mw_configs):
        rem = deadline - time.monotonic() - (500 - ci * 50)
        if rem < 20:
            break
        t_alloc = min(t_per, rem)
        print(f"  MW {ci} (lr={cfg['lr_init']}, k={cfg['top_k']}), t={t_alloc:.0f}s")
        a_opt, _ = mult_weight_v2(best_seq, time_limit=t_alloc, **cfg)
        best_seq, best_score, _ = try_and_update(a_opt, best_seq, best_score, f"mw_{ci}")
    print(f"  Phase 3 best: {best_score:.8f}")
    
    # === Phase 4: Spectral equalization ===
    print("\n=== Phase 4: Spectral Equalization ===")
    rem = deadline - time.monotonic() - 300
    if rem > 30:
        t_alloc = min(rem, 200)
        print(f"  Spectral eq, t={t_alloc:.0f}s")
        a_opt, _ = spectral_equalize(best_seq, time_limit=t_alloc)
        best_seq, best_score, _ = try_and_update(a_opt, best_seq, best_score, "spectral")
        
        # Second spectral pass with different alpha
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            a_opt2, _ = spectral_equalize(best_seq, time_limit=min(rem, 100),
                                          alpha_init=0.04)
            best_seq, best_score, _ = try_and_update(a_opt2, best_seq, best_score, "spectral2")
    print(f"  Phase 4 best: {best_score:.8f}")
    
    # === Phase 5: Peak rebalancing + multi-n ===
    print("\n=== Phase 5: Peak Rebalancing ===")
    rem = deadline - time.monotonic() - 150
    if rem > 30:
        t_alloc = min(rem, 150)
        print(f"  Peak rebalance, t={t_alloc:.0f}s")
        a_opt, _ = peak_rebalancing(best_seq, time_limit=t_alloc)
        best_seq, best_score, _ = try_and_update(a_opt, best_seq, best_score, "rebalance")
    print(f"  Phase 5 best: {best_score:.8f}")
    
    # Try a different n value
    rem = deadline - time.monotonic() - 100
    if rem > 60:
        n_alt = 256
        inits = generate_inits(n_alt, best_seq, rng)
        print(f"\n  Trying n={n_alt} with parent interpolation...")
        for ii in [0, 5]:  # uniform and parent-interpolated
            if ii < len(inits):
                a_opt, _ = fw_lp_optimize(inits[ii], time_limit=min(40, rem - 20))
                best_seq, best_score, _ = try_and_update(a_opt, best_seq, best_score, f"n{ii}_{n_alt}")
                rem = deadline - time.monotonic() - 60
                if rem < 30:
                    break
    
    # === Phase 6: Final polish ===
    print("\n=== Phase 6: Final Polish ===")
    if best_seq is not None:
        # L-BFGS with very small temperature
        rem = deadline - time.monotonic() - 40
        if rem > 30:
            t_alloc = min(rem, 80)
            print(f"  Final L-BFGS, t={t_alloc:.0f}s")
            a_opt, _ = lbgfs_smooth_optimize(best_seq, time_limit=t_alloc,
                                             t_init=0.0003, t_end=0.00003)
            best_seq, best_score, _ = try_and_update(a_opt, best_seq, best_score, "final_lbgfs")
        
        # Fine FW-LP
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            t_alloc = min(rem, 60)
            print(f"  Final FW-LP, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            best_seq, best_score, _ = try_and_update(a_opt, best_seq, best_score, "final_fw")
        
        # Tiny perturbations
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            for pi in range(4):
                rem = deadline - time.monotonic() - 5
                if rem < 5:
                    break
                noise = rng.standard_normal(len(best_seq)) * 0.0008
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = peak_rebalancing(f_pert, time_limit=min(rem, 15), lr_init=0.002)
                best_seq, best_score, _ = try_and_update(a_opt, best_seq, best_score, f"perturb_{pi}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
