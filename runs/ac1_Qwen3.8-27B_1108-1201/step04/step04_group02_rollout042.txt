def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)
    conv = fast_convolve(a)
    s = np.sum(a)
    if s < 0.01:
        return float('inf'), conv, s
    return 2.0 * n * np.max(conv) / (s * s), conv, s


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_1d_subproblem(a, j, conv, s, n):
    """
    Solve the 1D subproblem for coordinate j.
    Returns the optimal value x >= 0 for a[j].
    """
    aj = a[j]
    s0 = s - aj
    if s0 < 1e-15:
        return max(aj, 1e-10)

    k_lo = j
    k_hi = min(j + n, 2 * n - 1)
    k_diag = 2 * j

    # Compute constants for the 1D problem
    # M(x) = max(A0, A1(x), A2(x))
    # A0 = max of conv outside [k_lo, k_hi)
    # A1(x) = (conv[k_diag] - aj^2) + x^2
    # A2(x) = max over k in [k_lo,k_hi), k!=k_diag of (conv[k] + 2*(x-aj)*a[k-j])

    # A0: max outside
    if k_lo > 0 and k_hi < 2 * n - 1:
        A0 = max(np.max(conv[:k_lo]), np.max(conv[k_hi:]))
    elif k_lo > 0:
        A0 = np.max(conv[:k_lo])
    elif k_hi < 2 * n - 1:
        A0 = np.max(conv[k_hi:])
    else:
        A0 = 0.0

    # A1 base
    if k_diag < 2 * n - 1:
        A1_base = conv[k_diag] - aj * aj
    else:
        A1_base = 0.0

    # A2 arrays
    ks = np.arange(k_lo, k_hi)
    idx = ks - j
    valid = (idx >= 0) & (idx < n) & (ks != k_diag)
    if valid.any():
        ks_v = ks[valid]
        idx_v = idx[valid]
        conv_v = conv[ks_v].copy()
        a_v = a[idx_v].copy()
    else:
        ks_v = np.array([], dtype=np.int64)
        idx_v = np.array([], dtype=np.int64)
        conv_v = np.array([], dtype=np.float64)
        a_v = np.array([], dtype=np.float64)

    # Evaluate objective at a given x
    def f_of_x(x):
        if x < 0 or x > 1000.0:
            return float('inf'), 0.0
        s_new = s0 + x
        if s_new < 0.01:
            return float('inf'), 0.0
        M = A0
        quad_val = A1_base + x * x
        if quad_val > M:
            M = quad_val
        if len(ks_v) > 0:
            lin_max = np.max(conv_v + 2.0 * (x - aj) * a_v)
            if lin_max > M:
                M = lin_max
        return 2.0 * n * M / (s_new * s_new), M

    # Generate candidate x values
    cands = [0.0, aj]

    # A1/A0 crossover
    if A0 > A1_base and A0 - A1_base > 0:
        x_c = np.sqrt(A0 - A1_base)
        if 0 < x_c < 1000.0:
            cands.append(x_c)

    # A2 critical points and crossovers
    if len(ks_v) > 0 and np.max(a_v) > 1e-15:
        # Active k at current point (x=aj): the one with max conv_v
        active = np.argmax(conv_v)
        d = 2.0 * a_v[active]
        c = conv_v[active]
        if d > 1e-15:
            # Critical point in linear regime: x = s0 - 2*c/d
            x_cp = s0 - 2.0 * c / d
            if 0 < x_cp < 1000.0:
                cands.append(x_cp)
            # Crossover A2 = A0: x = aj + (A0 - c) / d
            x_cr = aj + (A0 - c) / d
            if 0 < x_cr < 1000.0:
                cands.append(x_cr)
            # Crossover A2 = A1: conv_v[active] + 2*(x-aj)*a_v[active] = A1_base + x^2
            # x^2 - 2*a_v[active]*x + (A1_base - conv_v[active]) + 2*aj*a_v[active] = 0
            # x^2 - 2*a_v[active]*x + (A1_base - c + 2*aj*a_v[active]) = 0
            disc = 4.0 * a_v[active] ** 2 - 4.0 * (A1_base - c + 2.0 * aj * a_v[active])
            if disc > 0:
                for sign in [1, -1]:
                    x_ca = a_v[active] + sign * np.sqrt(disc) / 2.0
                    if 0 < x_ca < 1000.0:
                        cands.append(x_ca)

    # A1 critical: x = A1_base / s0 (minimizes (A1_base + x^2)/(s0+x)^2)
    if s0 > 1e-15 and A1_base > 0:
        x_a1c = A1_base / s0
        if 0 < x_a1c < 1000.0:
            cands.append(x_a1c)

    # Heuristic values
    for factor in [0.3, 1.5, 3.0, 0.1]:
        cands.append(aj * factor)
    cands.append(s0 * 0.1)
    cands.append(s0 * 0.5)

    # Evaluate all candidates
    best_x = aj
    best_f, _ = f_of_x(aj)

    for x in cands:
        if x < 0 or x > 1000.0:
            continue
        f_val, _ = f_of_x(x)
        if f_val < best_f:
            best_f = f_val
            best_x = x

    # Golden section refinement
    eps = max(aj * 0.3, s0 * 0.1, 1e-6)
    lo = max(0.0, best_x - eps)
    hi = min(1000.0, best_x + eps)

                f2 = f1
                x1 = hi - gr * (hi - lo)
                f1, _ = f_of_x(x1)
            else:
                lo = x1
                x1 = x2
                f1 = f2
                x2 = lo + gr * (hi - lo)
                f2, _ = f_of_x(x2)
        x_g = (lo + hi) / 2.0
        f_g, _ = f_of_x(x_g)

def coordinate_descent(a0, time_limit, seed=42, verbose=True, perturb_every=200):
    """
    Exact coordinate descent with 1D golden section search.
    Cycles through coordinates in random order, solving each exactly.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        n = len(a)
        a = np.ones(n) / n
        s = 1.0
    n = len(a)

    conv = fast_convolve(a)
    objective = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    best_obj = objective

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    sweep = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit - 3:
        sweep += 1
        obj_start = objective
        perm = rng.permutation(n)

        for j in range(n):
            jj = perm[j]
            aj = a[jj]
            s0 = s - aj
            if s0 < 1e-15:
                continue

            x_opt = solve_1d_subproblem(a, jj, conv, s, n)

            if abs(x_opt - aj) > 1e-15:
                delta = x_opt - aj
                k_lo = jj
                k_hi = min(jj + n, 2 * n - 1)
                k_diag = 2 * jj

                # Update conv
                if k_diag < 2 * n - 1:
                    conv[k_diag] += x_opt * x_opt - aj * aj

                ks = np.arange(k_lo, k_hi)
                idx = ks - jj
                valid = (idx >= 0) & (idx < n) & (ks != k_diag)
                if valid.any():
                    conv[ks[valid]] += 2.0 * delta * a[idx[valid]]

                a[jj] = x_opt
                s = s0 + x_opt
                new_obj = 2.0 * n * conv.max() / (s * s)

                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    stagnant = 0
                else:
                    stagnant += 1

        if stagnant > perturb_every:
            # Perturb to escape local minimum
            scale = best_a.max() * 0.05
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 0.01:
                a = best_a.copy()
                s = a.sum()
            else:
                conv = fast_convolve(a)
                objective = 2.0 * n * conv.max() / (s * s)
                if objective > best_obj * 1.5:
                    # Perturbation was too aggressive, revert
                    a = best_a.copy()
                    s = a.sum()
                    conv = fast_convolve(a)
                    objective = best_obj
            stagnant = 0

        if verbose and sweep % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweep}, best_obj={best_obj:.10f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def subgradient_refine(a0, time_limit, seed=123, verbose=True):
    """
    Subgradient descent with softmax peak weighting as a complement to CD.
    Uses Barzilai-Borwein step sizes.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        n = len(a)
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    no_improve = 0
    lr = 0.005
    temp = 0.5
    top_k = min(64, 2 * n - 1)

    while time.monotonic() - t0 < time_limit - 2:
        # Compute subgradient with softmax weighting
        size = len(conv)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= wsum

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        a_new = project_simplex(a - lr * sg)
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            no_improve = 0
        else:
            no_improve += 1

        # Adapt parameters
        lr *= 0.9995
        temp *= 1.005

        if no_improve > 300:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
            conv = fast_convolve(a)
            lr = 0.005
            temp = 0.5
            no_improve = 0

    return best_a * a.sum(), best_obj  # Return unnormalized (scale doesn't matter)

def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    h = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    inits.append(h / h.sum())

    # Raised cosine (cosine bell)
    c = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 2 + 0.01
    inits.append(c / c.sum())

    # Linear ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    inits.append(r / r.sum())
    inits.append(r[::-1] / r.sum())

    # Quadratic
    q = (x - n / 2.0) ** 2 + 1.0
    inits.append(q / q.sum())

    # Gaussian center
    g = np.exp(-((x - n / 2.0) / (n / 6.0)) ** 2)
    inits.append(g / g.sum())

    # Random Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Random Dirichlet (spread)
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    # Interpolated parent
    if parent is not None and len(parent) > 1:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())
            # Slightly perturbed parent
            pp = px / px.sum() + rng.standard_normal(n) * 0.02
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                inits.append(pp / pp.sum())

    return inits

                a_cd, _ = coordinate_descent(a_sg, time_limit=t_cd, seed=3000)
                try_sequence(a_cd, best_score)

    # === Phase 4: Final polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"  Current best: {best_score[0]:.10f}")

    if best_score[1] is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            n_best = len(best_score[1])
            a_opt, _ = coordinate_descent(best_score[1], time_limit=rem, seed=9999,
                                         perturb_every=500)
            try_sequence(a_opt, best_score)

    # === Fallback ===
    if best_score[1] is None or not np.isfinite(best_score[0]):
        print("  FALLBACK: using uniform sequence")
        fallback = np.ones(256) / 256
        try_sequence(fallback, best_score)

    # Final verification
    if best_score[1] is not None:
        final_score = float(evaluate_sequence(best_score[1].tolist()))
        if np.isfinite(final_score):
            best_score[0] = final_score
            print(f"\nFINAL: score={final_score:.10f}, n={len(best_score[1])}")
        else:
            fallback = np.ones(384) / 384
            best_score[0] = float(evaluate_sequence(fallback.tolist()))
            best_score[1] = fallback
            print(f"\nFINAL (fallback): score={best_score[0]:.10f}")

    result = [float(x) for x in best_score[1]]
    print(f"Returned sequence length: {len(result)}")
    return result
