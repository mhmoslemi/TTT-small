"""
Multi-method local search for minimizing
    2*n*max(conv(a,a)) / (sum(a))^2
over non-negative step-function heights.

The implementation combines:
- L-BFGS-B on a smooth log-sum-exp approximation of the maximum convolution
  tap, using a softmax parameterization with one reference coordinate fixed.
- An active-peak LP direction for first-order max-reduction.
- Exact profile coordinate descent for final polishing.
- Short searches at neighbouring lengths.

The best sequence found is tracked globally and returned.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog


_GLOBAL = {'score': float('inf'), 'p': None}
_OPT = {
    'n': 0,
    'ref': 0,
    'idx': None,
    'deadline': 0.0,
    'stop': False,
    'local_score': float('inf'),
    'local_p': None,
}

    'valid': False,
    'y': None,
    'beta': None,
    'alpha': None,
    'val': None,
    'grad': None,
}
_BOUNDS_MIN = -50.0
_BOUNDS_MAX = 50.0


def _reset_state():
    _GLOBAL['score'] = float('inf')
    _GLOBAL['p'] = None
    _OPT['n'] = 0
    _OPT['ref'] = 0
    _OPT['idx'] = None
    _OPT['deadline'] = 0.0
    _OPT['stop'] = False
    _OPT['local_score'] = float('inf')
    _OPT['local_p'] = None
    _CACHE['valid'] = False
    _CACHE['y'] = None
    _CACHE['beta'] = None
    _CACHE['alpha'] = None

def normalize_positive(p):
    p = np.asarray(p, dtype=np.float64).copy()
    if p.ndim != 1:
        p = p.ravel()
    if p.size == 0:
        return np.array([1.0], dtype=np.float64)
    p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(p.size, dtype=np.float64) / float(p.size)
    return p / s


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([float(a[0]) * float(a[0])], dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    out = np.asarray(out, dtype=np.float64)
    out = np.maximum(out, 0.0)
    return out


def compute_lse_grad(weights, f):
    weights = np.asarray(weights, dtype=np.float64).ravel()
    f = np.asarray(f, dtype=np.float64).ravel()
    n = f.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if weights.size != 2 * n - 1:
        return np.zeros(n, dtype=np.float64)
    if n == 1:
        return np.array([2.0 * float(weights[0]) * float(f[0])], dtype=np.float64)

    f_flipped = f[::-1].copy()
    total_len = weights.size + f.size - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def score_of(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size == 0:
        return float('inf')
    p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return float('inf')
    c = fast_convolve(p)
    if c.size == 0:
        return float('inf')
    val = 2.0 * p.size * float(np.max(c)) / (s * s)
    if not np.isfinite(val):
        return float('inf')
    return float(val)


def _update_tracking(p, c=None):
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size == 0:
        return
    p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return
    if c is None:
        c = fast_convolve(p)
    c = np.asarray(c, dtype=np.float64).ravel()
    if c.size == 0:
        return
    m = float(np.max(c))
    if not np.isfinite(m):
        return
    val = 2.0 * p.size * m / (s * s)
    if not np.isfinite(val):
        return
    if val < _GLOBAL['score']:
        _GLOBAL['score'] = val
        _GLOBAL['p'] = p.copy()
    if p.size == _OPT['n'] and val < _OPT['local_score'] - 1e-15:
        _OPT['local_score'] = val
        _OPT['local_p'] = p.copy()


def _cache_invalid():
    _CACHE['valid'] = False
    _CACHE['y'] = None
    _CACHE['beta'] = None
    _CACHE['alpha'] = None
    _CACHE['val'] = None
    _CACHE['grad'] = None


def _p_from_y(y):
    n = _OPT['n']
    if n <= 0:
        return np.array([1.0], dtype=np.float64)
    y = np.asarray(y, dtype=np.float64)
    if y.size != n - 1:
        y = np.zeros(max(0, n - 1), dtype=np.float64)
    z = np.empty(n, dtype=np.float64)
    if _OPT['idx'] is not None:
        z[_OPT['idx']] = y
    z[_OPT['ref']] = 0.0
    if not np.all(np.isfinite(z)):
        z = np.zeros(n, dtype=np.float64)
    z -= float(np.max(z))
    w = np.exp(z)
    sw = float(w.sum())
    if not np.isfinite(sw) or sw <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return w / sw


def _compute_full(y, beta, alpha):
    y = np.asarray(y, dtype=np.float64)
    if y.size == 0:
        p = np.ones(1, dtype=np.float64)
    else:
        if not np.all(np.isfinite(y)):
            y = np.zeros(y.size, dtype=np.float64)
        p = _p_from_y(y)

    c = fast_convolve(p)
    _update_tracking(p, c)

    n = _OPT['n']
    out_grad = np.zeros(max(0, n - 1), dtype=np.float64)

    if beta is None or not np.isfinite(beta) or beta <= 0.0:
        beta = 1e-8
    if alpha is None or not np.isfinite(alpha):
        alpha = 0.0

    bc = beta * c
    mx = float(np.max(bc))
    if not np.isfinite(mx):
        return 1e100, out_grad, p, c

    ex = np.exp(bc - mx)
    ws = float(np.sum(ex))
    if not np.isfinite(ws) or ws <= 1e-300:
        ex = np.ones_like(bc)
        ws = float(ex.size)

    val = float((mx + np.log(ws)) / beta)
    w = ex / ws

    gp = compute_lse_grad(w, p)
    if alpha != 0.0:
        val += float(alpha * np.dot(c, c))
        gp = gp + alpha * (2.0 * compute_lse_grad(c, p))

    sdot = float(np.dot(p, gp))
    gfull = gp - p * sdot

    if _OPT['idx'] is not None and len(_OPT['idx']) > 0:
        out_grad = gfull[_OPT['idx']].copy()
    if not np.all(np.isfinite(out_grad)):
        out_grad = np.zeros(out_grad.size, dtype=np.float64)

    return val, out_grad, p, c


def _lse_value(y, beta, alpha):
    if time.monotonic() > _OPT['deadline']:
        _OPT['stop'] = True
        return 1e100
    y_arr = np.asarray(y, dtype=np.float64)
    if (_CACHE['valid'] and
            _CACHE['beta'] == beta and
            _CACHE['alpha'] == alpha and
            np.array_equal(_CACHE['y'], y_arr)):
        return _CACHE['val']

    val, grad, p, c = _compute_full(y_arr, beta, alpha)
    _CACHE['valid'] = True
    _CACHE['y'] = y_arr
    _CACHE['beta'] = beta
    _CACHE['alpha'] = alpha
    _CACHE['val'] = val
    _CACHE['grad'] = grad
    return val


def _lse_grad(y, beta, alpha):
    n = _OPT['n']
    grad_len = max(0, n - 1)
    if time.monotonic() > _OPT['deadline']:
        _OPT['stop'] = True
        return np.zeros(grad_len, dtype=np.float64)

    y_arr = np.asarray(y, dtype=np.float64)
    if (_CACHE['valid'] and
            _CACHE['beta'] == beta and
            _CACHE['alpha'] == alpha and
            np.array_equal(_CACHE['y'], y_arr)):
        return _CACHE['grad']

    val, grad, p, c = _compute_full(y_arr, beta, alpha)
    _CACHE['valid'] = True
    _CACHE['y'] = y_arr
    _CACHE['beta'] = beta
    _CACHE['alpha'] = alpha
    _CACHE['val'] = val
    _CACHE['grad'] = grad
    return grad


def _minimize_callback(xk):
    if _OPT['stop']:
        return True
    if time.monotonic() > _OPT['deadline']:
        _OPT['stop'] = True
        return True
    return False


def run_lse_phase(p0, beta, alpha, time_budget, overall_deadline):
    now = time.monotonic()
    deadline = min(now + time_budget, overall_deadline)
    p = normalize_positive(p0)
    n = p.size

    if n < 3 or deadline - now < 2.0:
        _update_tracking(p)
        return p

    s0 = score_of(p)
    ref = int(np.argmax(p))
    idx = np.array([i for i in range(n) if i != ref], dtype=np.int64)

    _OPT['n'] = n
    _OPT['ref'] = ref
    _OPT['idx'] = idx
    _OPT['deadline'] = float(deadline)
    _OPT['stop'] = False
    _OPT['local_score'] = float(s0)
    _OPT['local_p'] = p.copy()
    _cache_invalid()
    _update_tracking(p)

    tiny = 1e-300
    y = np.log(np.maximum(p[idx], tiny)) - np.log(max(float(p[ref]), tiny))
    y = np.clip(y, _BOUNDS_MIN, _BOUNDS_MAX)

    bounds = [(_BOUNDS_MIN, _BOUNDS_MAX)] * (n - 1)
    options = {
        'disp': False,
        'maxiter': 50000,
        'ftol': 1e-12,
        'gtol': 1e-9,
        'maxcor': 25,
    }

    res = None
    try:
        res = minimize(
            _lse_value,
            y,
            jac=_lse_grad,
            args=(beta, alpha),
            method='L-BFGS-B',
            bounds=bounds,
            options=options,
            callback=_minimize_callback,
        )
    except Exception as exc:
        print("    LSE minimize error:", repr(exc))

    if res is not None and hasattr(res, 'x') and res.x is not None:
        xr = np.asarray(res.x, dtype=np.float64)
        if xr.shape[0] == n - 1 and np.all(np.isfinite(xr)):
            px = _p_from_y(xr)
            _update_tracking(px)

    if _OPT['local_p'] is not None:
        return normalize_positive(_OPT['local_p'])
    return p


def lp_direction_phase(p0, time_budget, overall_deadline):
    now = time.monotonic()
    deadline = min(now + time_budget, overall_deadline)
    p = normalize_positive(p0)
    n = p.size

    if n < 2 or deadline - now < 2.0:
        _update_tracking(p)
        return p

    c = fast_convolve(p)
    local_score = 2.0 * n * float(np.max(c))
    local_p = p.copy()
    _update_tracking(p, c)

    no_improve = 0
    it = 0

    while time.monotonic() < deadline - 2.0:
        it += 1
        c = fast_convolve(p)
        M = float(np.max(c))
        cur_score = 2.0 * n * M
        if cur_score < local_score - 1e-15:
            local_score = cur_score
            local_p = p.copy()
        _update_tracking(p, c)

        K = min(16 + 16 * no_improve, 512, c.size)
        if K < 1:
            K = 1
        active = np.argsort(c)[-K:].astype(np.int64, copy=False)

        A_ub = np.zeros((K, n + 1), dtype=np.float64)
        for row, k in enumerate(active):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                idx_i = np.arange(lo, hi + 1, dtype=np.int64)
                A_ub[row, idx_i] = 2.0 * p[k - idx_i]
            A_ub[row, n] = -1.0

        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0

        lower = -p - 1e-12
        upper = np.maximum(4.0 * p, 0.002)
        bounds = [(float(lower[i]), float(upper[i])) for i in range(n)]
        bounds.append((None, None))

        c_obj = np.zeros(n + 1, dtype=np.float64)
        c_obj[n] = 1.0

        res = None
        try:
            res = linprog(
                c_obj,
                A_ub=A_ub,
                b_ub=np.zeros(K, dtype=np.float64),
                A_eq=A_eq,
                b_eq=np.zeros(1, dtype=np.float64),
                bounds=bounds,
                method='highs',
                options={'presolve': True, 'time_limit': 3.0},
            )
        except Exception:
            res = None

        v = None
        tval = 0.0
        if res is not None:
            x = getattr(res, 'x', None)
            if x is not None:
                x = np.asarray(x, dtype=np.float64)
                if x.size == n + 1 and np.all(np.isfinite(x)):
                    v = x[:n].copy()
                    tval = float(x[n])

        if v is None or tval > -1e-12:
            no_improve += 1
            if no_improve > 4:
                break
            continue

        v = v - float(np.mean(v))
        if not np.all(np.isfinite(v)) or np.linalg.norm(v) < 1e-14:
            no_improve += 1
            if no_improve > 4:
                break
            continue

        best_M = M
        best_p = None
        best_a = None

        for a in (0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0):
            pn = p + a * v
            pn = np.maximum(pn, 0.0)
            ss = float(pn.sum())
            if ss <= 1e-15:
                continue
            pn = pn / ss
            cn = fast_convolve(pn)
            Mn = float(np.max(cn))
            if Mn < best_M - 1e-15:
                best_M = Mn
                best_p = pn.copy()
                best_a = a

        if best_a is not None:
            lo = max(0.0, best_a * 0.5 - 0.01)
            hi = min(1.0, best_a * 1.5 + 0.01)
            if hi > lo:
                for a2 in np.linspace(lo, hi, 18):
                    pn = p + a2 * v
                    pn = np.maximum(pn, 0.0)
                    ss = float(pn.sum())
                    if ss <= 1e-15:
                        continue
                    pn = pn / ss
                    cn = fast_convolve(pn)
                    Mn = float(np.max(cn))
                    if Mn < best_M - 1e-15:
                        best_M = Mn
                        best_p = pn.copy()
                        best_a = a2

        if best_p is not None:
            best_score = 2.0 * n * best_M
            if best_score < local_score - 1e-14:
                p = best_p
                local_p = p.copy()
                local_score = best_score
                _update_tracking(p)
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 4:
            break

        if it % 25 == 0:
            print(f"    LP it={it}, current={cur_score:.10f}, best={local_score:.10f}")

    _update_tracking(local_p)
    return normalize_positive(local_p)


def coord_polish_phase(p0, time_budget, overall_deadline):
    now = time.monotonic()
    deadline = min(now + time_budget, overall_deadline)
    p = normalize_positive(p0)
    n = p.size

    if n < 2 or deadline - now < 2.0:
        _update_tracking(p)
        return p

    c = fast_convolve(p)
    local_score = 2.0 * n * float(np.max(c))
    local_p = p.copy()
    _update_tracking(p, c)

    grid_full = np.linspace(0.0, 1.0, 32)
    pass_count = 0

    while time.monotonic() < deadline - 2.0:
        improved_any = False
        c = fast_convolve(p)
        cur_score = 2.0 * n * float(np.max(c))
        if cur_score < local_score - 1e-15:
            local_score = cur_score
            local_p = p.copy()
        _update_tracking(p, c)

        for j in range(n):
            if time.monotonic() > deadline - 2.0:
                break

            s = float(p[j])
            denom = 1.0 - s
            if denom < 1e-10:
                continue

            A = c.astype(np.float64, copy=True)
            A[j:j + n] -= 2.0 * s * p
            A[2 * j] += s * s
            A /= (denom * denom)
            np.maximum(A, 0.0, out=A)

            B = np.zeros(2 * n - 1, dtype=np.float64)
            B[j:j + n] = 2.0 * p / denom
            B[2 * j] = 0.0

            u = grid_full.reshape(-1, 1)
            one = 1.0 - u
            x2 = u * u
            vals = A[None, :] * one * one + B[None, :] * u * one
            vals = np.maximum(vals, x2)
            maxvals = vals.max(axis=1)

            ib = int(np.argmin(maxvals))
            xb = float(grid_full[ib])
            wb = float(maxvals[ib])

            lo = max(0.0, xb - 0.03)
            hi = min(1.0, xb + 0.03)
            if hi > lo:
                grid = np.linspace(lo, hi, 24)
                u = grid.reshape(-1, 1)
                one = 1.0 - u
                x2 = u * u
                vals = A[None, :] * one * one + B[None, :] * u * one
                vals = np.maximum(vals, x2)
                maxvals = vals.max(axis=1)
                ib2 = int(np.argmin(maxvals))
                wb2 = float(maxvals[ib2])
                if wb2 < wb - 1e-15:
                    xb = float(grid[ib2])
                    wb = wb2

            if abs(xb - s) < 1e-12:
                continue

            pn = p * ((1.0 - xb) / denom)
            pn[j] = xb
            ss = float(pn.sum())
            if ss <= 1e-15:
                continue
            pn = pn / ss

            cn = fast_convolve(pn)
            Mn = float(np.max(cn))
            score_n = 2.0 * n * Mn
            if score_n < local_score - 1e-14:
                p = pn
                c = cn
                local_score = score_n
                local_p = p.copy()
                _update_tracking(p, c)
                improved_any = True

        pass_count += 1
        if not improved_any:
            break
        if pass_count >= 5:
            break

    _update_tracking(local_p)
    return normalize_positive(local_p)


def _make_length(source, m):
    source = np.asarray(source, dtype=np.float64).ravel()
    n = source.size
    if n == 0:
        return np.ones(int(m), dtype=np.float64) / float(m)
    m = int(m)
    if m <= 0:
        m = n
    if m == n:
        return source.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, source)
    return normalize_positive(out)

        if _GLOBAL['p'] is not None and len(_GLOBAL['p']) == len(current):
            current = normalize_positive(_GLOBAL['p'])
        print(f"After final LSE: score={_GLOBAL['score']:.10f}, length={len(current)}")

    # Exact coordinate polish.
    if time.monotonic() < overall_deadline - 70.0:
        tb = 100.0 * scale
        current = coord_polish_phase(current, tb, overall_deadline)
        if _GLOBAL['p'] is not None and len(_GLOBAL['p']) == len(current):
            current = normalize_positive(_GLOBAL['p'])
        print(f"After coordinate polish: score={_GLOBAL['score']:.10f}, length={len(current)}")

    # Try a few nearby lengths.
    alt_lengths = [384, 480, 512, 576, 448]
    current_n = len(current)
    for m in alt_lengths:
        if time.monotonic() > overall_deadline - 70.0:
            break
        if m == current_n:
            continue

        source = _GLOBAL['p'] if _GLOBAL['p'] is not None else current
        source = normalize_positive(source)
        p_alt = _make_length(source, m)

        for beta, alpha, base_budget in [(100.0, 0.05, 20.0), (1500.0, 0.0, 25.0)]:
            tb = base_budget * scale
            if tb < 3.0:
                continue
            p_alt = run_lse_phase(p_alt, beta, alpha, tb, overall_deadline)
            if _GLOBAL['p'] is not None and len(_GLOBAL['p']) == len(p_alt):
                p_alt = normalize_positive(_GLOBAL['p'])

        tb = 12.0 * scale
        if tb >= 3.0:
            p_alt = lp_direction_phase(p_alt, tb, overall_deadline)
            if _GLOBAL['p'] is not None and len(_GLOBAL['p']) == len(p_alt):
                p_alt = normalize_positive(_GLOBAL['p'])

        print(f"After length {m}: global score={_GLOBAL['score']:.10f}")

    # Final return.
    best = _GLOBAL['p'] if _GLOBAL['p'] is not None else current
    best = np.asarray(best, dtype=np.float64).ravel()
    if best.size == 0 or not np.isfinite(best).all():
        best = np.ones(448, dtype=np.float64) / 448.0
    best = np.nan_to_num(best, nan=0.0, posinf=0.0, neginf=0.0)
    best = np.clip(best, 0.0, 1000.0)
    s = float(best.sum())
    if not np.isfinite(s) or s <= 1e-15:
        best = np.ones(448, dtype=np.float64) / 448.0
    else:
        best = best / s

    final_list = [float(x) for x in best]

    try:
        final_score = float(evaluate_sequence(final_list))
    except Exception:
        final_score = score_of(best)

    if not np.isfinite(final_score):
        fallback = np.ones(448, dtype=np.float64) / 448.0
        final_list = [float(x) for x in fallback]
        final_score = score_of(fallback)

    print(f"FINAL: score={final_score:.10f}, length={len(final_list)}")
    return final_list
