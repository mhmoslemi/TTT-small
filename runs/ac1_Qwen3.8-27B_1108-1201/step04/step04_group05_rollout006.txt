"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved LP-based Frank-Wolfe
with Barzilai-Borwein subgradient polishing, spectral restarts, adaptive 
constraint tightening, and multi-scale optimization.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_for_direction(f, M, max_iter=5000):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Uses the structure that the constraint matrix is a product of f and a Toeplitz matrix.
    """
    n = len(f)
    c = -np.ones(n)
    
    # Build the constraint matrix: A_ub has shape (2n-1, n)
    # A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # i.e., (f * g)[k] = sum_j f[k-j] * g[j] (with f out of range = 0)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # (f*g)[k] gets contribution f[k-j] * g[j] for k = j, j+1, ..., j+n-1
        A_ub[j:j + n, j] = f
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, res
    except Exception:
        pass
    return None, None


def golden_section_search(f_curr, g_dir, n_eval_max=30):
    """
    1D golden section search for alpha in [0,1] minimizing
    max_conv((1-alpha)*f_curr + alpha*g_dir) after projection and normalization.
    """
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    
    lo = 0.0
    hi = 1.0
    
    def obj(alpha):
        f_new = (1 - alpha) * f_curr + alpha * g_dir
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            return 1e10
        f_new = f_new / s
        return fast_convolve(f_new).max()
    
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)
    fc = obj(c)
    fd = obj(d)
    
    for _ in range(n_eval_max):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - phi * (hi - lo)
            fc = obj(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + phi * (hi - lo)
            fd = obj(d)
    
    best_alpha = (lo + hi) / 2
    return best_alpha, obj(best_alpha)


def fw_lp_optimize(f0, time_limit, verbose=False, M_init=None):
    """
    Improved Frank-Wolfe with LP oracle.
    Key improvements: adaptive M, golden section line search, 
    dual-guided restarts.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    if M_init is not None:
        M = min(M, M_init)
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng_local = np.random.default_rng(12345 + n)
    
    # Track dual information for restarts
    last_dual = None

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        # Solve LP
        g, lp_res = solve_lp_for_direction(f, M)
        if g is None:
            # LP failed, tighten M slightly and retry
            M *= 0.99
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Fixed point check
        diff = np.max(np.abs(g - f))
        if diff < 1e-9:
            no_improve += 1
            if no_improve > 30:
                break
            # Try to tighten M to escape
            M *= 0.995
            continue

        # Golden section line search
        best_alpha, best_val = golden_section_search(f, g, n_eval_max=25)

        if best_alpha > 1e-10 and best_val < best_obj:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:
                f = f / fs
            conv = fast_convolve(f)
            M = conv.max()

            best_obj = M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 20 == 0:
                print(f"    FW-LP iter {it}: obj={best_obj:.10f}, M={M:.8f}")
        else:
            no_improve += 1

        # Update M adaptively
        # If we're making progress, keep M at current level
        # If stuck, try tightening
        if no_improve > 10 and no_improve % 5 == 0:
            M *= 0.99

        # Restart strategy
        if no_improve > 35:
            strategies = ["gaussian", "dual_guide", "mirror", "peak_push"]
            strat = strategies[no_improve % len(strategies)]
            
            if strat == "gaussian":
                scale = 0.003 + 0.003 * (no_improve // 50)
                noise = rng_local.standard_normal(n) * scale
                f = project_simplex(best_f + noise)
                
            elif strat == "dual_guide":
                # Use the convolution peaks to guide the perturbation
                conv_bf = fast_convolve(best_f)
                peaks = np.argsort(conv_bf)[-n // 4:]
                # Push mass away from positions that contribute to peaks
                push = np.zeros(n)
                for k in peaks:
                    for j in range(n):
                        idx = k - j
                        if 0 <= idx < n:
                            push[j] += best_f[idx] * conv_bf[k]
                f = project_simplex(best_f - 0.05 * push / (np.max(np.abs(push)) + 1e-10))
                
            elif strat == "mirror":
                blend = 0.2 + 0.1 * (no_improve // 50)
                f = project_simplex((1 - blend) * best_f + blend * best_f[::-1])
                
            else:  # peak_push
                conv_bf = fast_convolve(best_f)
                max_k = np.argmax(conv_bf)
                # Reduce contribution at the peak
                f_new = best_f.copy()
                for j in range(n):
                    idx = max_k - j
                    if 0 <= idx < n:
                        f_new[j] -= 0.01 * best_f[idx]
                        f_new[idx] -= 0.01 * best_f[j]
                f = project_simplex(f_new)

            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def bb_subgradient_polish(f0, time_limit, top_k=None):
    """
    Barzilai-Borwein subgradient method for polishing.
    Uses the subgradient of max_k (a*a)[k] with BB step sizes.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    if top_k is None:
        top_k = max(5, n // 10)

    t0 = time.monotonic()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    # BB parameters
    alpha = 1.0 / n
    s_prev = None
    y_prev = None
    prev_grad_val = None
    
    no_improve = 0
    rng_local = np.random.default_rng(55555)
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        curr_obj = conv.max()
        
        # Find top peaks for subgradient
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        # Weighted subgradient (uniform over top peaks for stability)
        # or use the single max peak
        max_k = int(np.argmax(conv))
        
        # Subgradient at k: g_j = 2 * a[k-j] (for valid k-j)
        # More precisely: d/d a_j [(a*a)[k] |_{a}] = a[k-j] + (1 if 2j=k else 0) * a[j]
        # Actually: (a*a)[k] = sum_j a_j a_{k-j}
        # d/d a_j = a_{k-j} + a_j * [k-j = j] = a_{k-j} + a_j * [2j=k]
        # But since we're taking max, the subgradient is:
        # For the argmax k*: g_j = a_{k*-j} + a_j * [2j = k*]
        # But this needs care. Let's use the simpler form.
        
        # Simple subgradient: for the peak at k*, the gradient of (a*a)[k*] w.r.t. a_j is:
        # a_{k*-j} (if k*-j != j) or 2*a_j (if k*-j == j, i.e., 2j=k*)
        # Actually no: (a*a)[k] = sum_{j} a_j * a_{k-j}
        # d/d a_j of this = a_{k-j} + (if we also have the term a_{k-j} * a_j when k-j is the index)
        # Wait, let's be very careful.
        # (a*a)[k] = sum_{j=0}^{n-1} a_j * a_{k-j} where a_i = 0 for i outside [0,n-1]
        # This is a function of the vector a. The partial derivative w.r.t. a_j is:
        # Term where the first factor is a_j: a_{k-j}
        # Term where the second factor is a_j: a_{k-j} (from the term a_{k-j} * a_j in the sum)
        # Wait, the sum is over j', and the term is a_{j'} * a_{k-j'}.
        # d/d a_j of sum_{j'} a_{j'} * a_{k-j'}:
        # = sum_{j'} [d(a_{j'})/d(a_j)] * a_{k-j'} + a_{j'} * d(a_{k-j'})/d(a_j)
        # = a_{k-j} + sum_{j': k-j'=j} a_{j'}
        # = a_{k-j} + a_{k-j} (since when k-j'=j, j'=k-j, so a_{j'} = a_{k-j})
        # Wait, that's only if k-j is in range.
        # Actually: sum_{j': k-j'=j} a_{j'} = a_{k-j} (there's exactly one such j', namely j'=k-j)
        # But we also need k-j to be in [0, n-1].
        # And the first term a_{k-j} also requires k-j in [0,n-1].
        # So if k-j is in [0,n-1]: derivative = a_{k-j} + a_{k-j} = 2*a_{k-j}
        # If k-j is not in [0,n-1]: derivative = 0
        # BUT: if j = k-j (i.e., 2j=k), then the term a_j * a_j = a_j^2 appears once,
        # and its derivative is 2*a_j. This is consistent with the formula 2*a_{k-j} = 2*a_j.
        
        # So the subgradient of (a*a)[k] w.r.t. a is:
        # sg_j = 2 * a_{k-j} if 0 <= k-j < n, else 0
        
        sg = np.zeros(n)
        for k in [max_k]:  # Use single peak for simplicity
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                sg[j] += 2.0 * f[k - j]
        
        # Normalize subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        sg = sg / sg_norm
        
        # BB step size
        if s_prev is not None and y_prev is not None:
            sy = np.dot(s_prev, y_prev)
            yy = np.dot(y_prev, y_prev)
            if yy > 1e-15:
                alpha = sy / yy
                alpha = np.clip(alpha, 1e-6, 1.0)
        
        # Line search (simple backtracking)
        step = alpha
        best_alpha_try = step
        best_val = curr_obj
        for _ in range(5):
            f_new = project_simplex(f - step * sg)
            val = fast_convolve(f_new).max()
            if val < best_val:
                best_val = val
                best_alpha_try = step
            step *= 0.5
        
        if best_val < best_obj - 1e-14:
            f = project_simplex(f - best_alpha_try * sg)
            best_obj = fast_convolve(f).max()
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                f = project_simplex(best_f + rng_local.standard_normal(n) * 0.005)
                no_improve = 0
                best_obj = fast_convolve(f).max()
                if best_obj > fast_convolve(best_f).max():
                    f = best_f.copy()
                    best_obj = fast_convolve(f).max()
        
        # Update BB
        if s_prev is not None:
            s_prev = f - s_prev  # This isn't quite right for BB, let me fix
            # Actually for BB: s_k = x_k - x_{k-1}, y_k = grad_k - grad_{k-1}
            # But we're using subgradients, so this is approximate
        
        if curr_obj < best_obj + 1e-10 and no_improve > 500:
            break

    return best_f, best_obj


def spectral_restart(f, rng):
    """
    Create a restart point using spectral analysis.
    Identify frequency components that cause peaks and attenuate them.
    """
    n = len(f)
    # Compute DFT
    F = np.fft.rfft(f)
    # The "flat" DFT magnitude would be constant
    # Attenuate the peaks in the DFT
    mag = np.abs(F)
    phase = np.angle(F)
    
    # Soft-clip the magnitude
    m = mag.max()
    if m < 1e-15:
        return f.copy()
    mag_clipped = np.minimum(mag, m * 0.7)
    # But keep some of the original to maintain structure
    mag_new = 0.5 * mag + 0.5 * mag_clipped
    
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    fs = f_new.sum()
    if fs < 1e-10:
        return f.copy()
    return f_new / fs


def two_phase_optimize(f0, time_limit):
    """
    Two-phase: first use FW-LP for fast convergence, then BB subgradient for polishing.
    """
    n = len(f0)
    
    # Phase A: FW-LP (60% of time)
    t_a = time_limit * 0.6
    f_opt, obj_a = fw_lp_optimize(f0, time_limit=t_a)
    
    # Phase B: BB subgradient (40% of time)
    t_b = time_limit - (time.monotonic() - (time.monotonic() - t_a))  # remaining
    t_b = time_limit * 0.4
    f_opt2, obj_b = bb_subgradient_polish(f_opt, time_limit=t_b, top_k=max(5, n // 8))
    
    if obj_b < obj_a:
        return f_opt2, obj_b
    return f_opt, obj_a


def generate_inits_v2(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge-concentrated (triangular)
    tri = 1.0 - np.abs(2.0 * x / (n - 1) - 1.0)
    inits.append(tri / tri.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Two-peak symmetric
    peak = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)) +
            np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)))
    inits.append(peak / peak.sum())

    # Cosine
    cos_t = np.cos(np.pi * x / (2.0 * (n - 1))) ** 2
    inits.append(cos_t / cos_t.sum())

    # Sine
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    # Chirp-like (for flat autocorrelation)
    chirp = np.abs(np.exp(1j * np.pi * x ** 2 / (2 * n)))
    # This is just 1, so use a modulated version
    chirp2 = np.abs(np.cos(np.pi * x * (x + 1) / (2 * n)))
    chirp2 = np.maximum(chirp2, 0.01)
    inits.append(chirp2 / chirp2.sum())

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Also a smoothed version
            from scipy.ndimage import gaussian_filter1d
            px_smooth = gaussian_filter1d(px, sigma=np.sqrt(n) * 0.1)
            px_smooth = np.maximum(px_smooth, 0)
            ps2 = px_smooth.sum()
            if ps2 > 1e-10:
                inits.append(px_smooth / ps2)

    return inits


def try_sequence(a, label, state):
    """Evaluate a sequence and update state if it's better."""
    a = np.maximum(a, 0.0)
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return score


def constrained_minimize_peak(f0, time_limit):
    """
    Alternative approach: directly minimize the peak using iteratively
    reweighted least squares (IRLS) on the active constraints.
    
    Idea: At each iteration, identify the "active" constraints (where 
    conv(a,a) is near max), and solve a QP that minimizes the weighted
    sum of the active convolution values.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    t0 = time.monotonic()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    no_improve = 0
    rng_local = np.random.default_rng(88888)
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        curr_max = conv.max()
        
        # Identify active constraints (top 20% of convolution values)
        threshold = np.percentile(conv, 80)
        active = conv >= threshold
        num_active = active.sum()
        
        if num_active == 0:
            active = np.zeros(len(conv), dtype=bool)
            active[np.argmax(conv)] = True
            num_active = 1
        
        # Build the QP: minimize sum_k w_k * (conv(a,a)[k])^2 
        # subject to a >= 0, sum(a) = 1
        # where w_k = 1 for active k, small for others
        # The gradient of sum_k w_k * (conv(a,a)[k])^2 w.r.t. a_j is:
        # 2 * sum_k w_k * conv(a,a)[k] * (d conv[a,a][k] / d a_j)
        # = 2 * sum_k w_k * conv[k] * 2 * a[k-j] (for valid k-j)
        
        # This is a nonlinear optimization. Use a simple gradient step.
        grad = np.zeros(n)
        for k in range(len(conv)):
            if not active[k]:
                continue
            w_k = conv[k]  # weight by current value
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += 4.0 * w_k * f[k - j]
        
        # Gradient step with adaptive step size
        # Use Armijo line search
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        alpha = 0.1 / (grad_norm * n)
        improved = False
        for _ in range(10):
            f_new = project_simplex(f - alpha * grad)
            val = fast_convolve(f_new).max()
            if val < curr_max - 1e-12:
                f = f_new
                improved = True
                break
            alpha *= 0.5
        
        if improved:
            new_obj = fast_convolve(f).max()
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 100:
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.003)
            no_improve = 0
    
    return best_f, best_obj
