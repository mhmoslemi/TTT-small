"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed mirror descent,
restricted LP-FW, and coordinate descent with multi-length search.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(a, p):
    """Compute gradient of tau*logsumexp((a*a)/tau) w.r.t. a.
    p is the softmax weights over the convolution lags.
    Returns gradient of length n."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    # grad_j = sum_k p[k] * a[k-j] = cross-correlation
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    # cross-correlation: ifft(fft(p) * conj(fft(a)))
    corr = np.fft.irfft(P * np.conj(A), fft_size)
    return corr[:n]

    n = len(a)

    eta = eta_init

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_init(n, seed=42):
    """Generate structured initial sequences."""
    rng = np.random.default_rng(seed)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Sine-squared
    j = np.arange(n)
    s = np.sin(np.pi * (j + 0.5) / n) ** 2
    inits.append(s / s.sum())
    
    # Cosine window
    c = np.cos(np.pi * j / (n - 1)) ** 2
    inits.append(c / c.sum())
    
    # U-shaped (edge-concentrated)
    u = 1.0 / (1.0 + 10.0 * (j + 0.5) * (n - j - 0.5) / (n / 2.0) ** 2)
    inits.append(u / u.sum())
    
    # Gaussian
    center = n / 2.0
    g = np.exp(-(j - center) ** 2 / (2 * (n / 4.0) ** 2))
    inits.append(g / g.sum())
    
    # Random
    r = rng.random(n)
    inits.append(r / r.sum())
    
    return inits

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed mirror descent with

structured initializations, and multi-length exploration.
"""

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def enforce_symmetry(a):
    """Enforce palindromic symmetry: a = (a + a[::-1]) / 2, then normalize."""
    a = 0.5 * (a + a[::-1])
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:

    return a

    p: softmax weights over convolution lags (length 2n-1).
    Returns gradient of length n: grad[j] = sum_k p[k] * a[k-j].

    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(P * np.conj(A), fft_size)

    """LP-FW using only top-k peaks as constraints for faster LP solves."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:

    if use_symmetry:
        f = enforce_symmetry(f)
    n = len(f)

    M = conv.max()
    best_obj = M

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            if use_symmetry:
                ft = enforce_symmetry(ft)
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Refine line search
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.15)
            hi = min(1.0, best_alpha + 0.15)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                if use_symmetry:
                    ft = enforce_symmetry(ft)
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            if use_symmetry:
                f_new = enforce_symmetry(f_new)
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    RLP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            a_pert = best_f + rng.standard_normal(n) * 0.004
            if use_symmetry:
                f = enforce_symmetry(a_pert)
            else:
                f = project_simplex(a_pert)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def smart_coordinate_descent(f0, time_limit, use_symmetry=True, verbose=False):
    """Coordinate descent with multi-candidate search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    if use_symmetry:
        f = enforce_symmetry(f)
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        if use_symmetry:
            indices = list(range(n // 2))
        else:
            indices = list(range(n))

        for j in indices:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_j_obj = fast_convolve(f).max()

            # Multiplier candidates
            multipliers = [0.2, 0.4, 0.6, 0.8, 1.25, 1.5, 2.0, 3.0]

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.95:
                    continue

                if use_symmetry:
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    f_trial[n - 1 - j] = new_val
                    other_sum = 1.0 - 2.0 * new_val
                    old_other_sum = 1.0 - 2.0 * old_val
                    if old_other_sum < 1e-10 or other_sum < 1e-10:
                        continue
                    scale = other_sum / old_other_sum
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    mask[n - 1 - j] = False
                    f_trial[mask] *= scale
                    f_trial = np.maximum(f_trial, 0)
                    st = f_trial.sum()
                    if st < 1e-15:
                        continue
                    f_trial /= st
                else:
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    old_other_sum = 1.0 - old_val
                    if old_other_sum < 1e-10:
                        continue
                    scale = (1.0 - new_val) / old_other_sum
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = np.maximum(f_trial, 0)
                    st = f_trial.sum()
                    if st < 1e-15:
                        continue
                    f_trial /= st

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                if use_symmetry:
                    f[j] = best_j_val
                    f[n - 1 - j] = best_j_val
                    other_sum = 1.0 - 2.0 * best_j_val
                    old_other_sum = 1.0 - 2.0 * old_val
                    if old_other_sum > 1e-10:
                        scale = other_sum / old_other_sum
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        mask[n - 1 - j] = False
                        f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                else:
                    f[j] = best_j_val
                    old_other_sum = 1.0 - old_val
                    if old_other_sum > 1e-10:
                        scale = (1.0 - best_j_val) / old_other_sum
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 2:
            break

        if verbose and passes % 5 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.10f}")

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_init(n, seed=42):
    """Generate diverse structured initial sequences."""
    rng = np.random.default_rng(seed)
    j = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Sine-squared (concentrated center)
    s = np.sin(np.pi * (j + 0.5) / n) ** 2
    inits.append(s / s.sum())

    # Cosine window
    c = np.cos(np.pi * j / max(n - 1, 1)) ** 2
    inits.append(c / c.sum())

    # U-shaped (edge-concentrated)
    center = (n - 1) / 2.0
    dist = np.abs(j - center) / center
    u = 1.0 / (1.0 + 4.0 * dist ** 2)
    inits.append(u / u.sum())

    # Broad Gaussian
    g = np.exp(-0.5 * (2 * dist) ** 2)
    inits.append(g / g.sum())

    # Double peak
    peak1 = np.exp(-0.5 * ((j - 0.25 * n) / (0.1 * n)) ** 2)
    peak2 = np.exp(-0.5 * ((j - 0.75 * n) / (0.1 * n)) ** 2)
    dp = peak1 + peak2
    inits.append(dp / dp.sum())

    # Random (smooth)
    r = rng.random(n)
    # Smooth it
    kernel = np.ones(5) / 5.0
    r_smooth = np.convolve(r, kernel, mode='same')
    inits.append(r_smooth / r_smooth.sum())

    # Palindromic random
    half = (rng.random(n // 2 + 1) + 1e-6)
    r_pal = np.concatenate([half, half[:-1][::-1]])
    inits.append(r_pal / r_pal.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: structured init, smoothed mirror descent,
    restricted LP-FW, coordinate descent, symmetry, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if len(a) == 0:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)

    # === Phase 1: Structured Initialization ===
    print("=== Phase 1: Structured Initialization ===")
    t1_start = time.monotonic()
    t1_budget = min(90, max(30, budget_s * 0.04))

    best_init = parent.copy()
    best_init_obj = fast_convolve(parent).max()
    try_seq(best_init, "parent")

    for i, init in enumerate(structured_init(n_parent, seed=seed + i * 7)):
        if time.monotonic() - t1_start > t1_budget:
            break
        obj = fast_convolve(init).max()
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = init.copy()
            print(f"  Structured init {i}: obj={obj:.10f}")

    # Quick SMD on best structured init
    rem_t1 = t1_budget - (time.monotonic() - t1_start)
    if rem_t1 > 15:
        a_opt, obj = smoothed_mirror_descent(best_init, time_limit=min(rem_t1, 30),
                                              tau_init=0.01, tau_min=0.002,
                                              eta_init=0.1, use_symmetry=True)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_opt.copy()

    try_seq(best_init, "best_init")

    # === Phase 2: Smoothed Mirror Descent (main) ===
    print(f"\n=== Phase 2: Smoothed Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 500
    if rem > 30:
        t_smd = min(rem * 0.6, 350)
        print(f"  SMD main t={t_smd:.0f}s")
        a_opt, _ = smoothed_mirror_descent(best_init, time_limit=t_smd,
                                           tau_init=0.003, tau_min=0.0003,
                                           eta_init=0.06, use_symmetry=True, verbose=True)
        try_seq(a_opt, "smd_main")

        # Second SMD with different params
        rem = deadline - time.monotonic() - 350
        if rem > 20:
            a_opt, _ = smoothed_mirror_descent(best_init, time_limit=min(rem, 150),
                                               tau_init=0.01, tau_min=0.001,
                                               eta_init=0.12, use_symmetry=False, verbose=False)
            try_seq(a_opt, "smd_asym")

    # === Phase 3: Restricted LP-FW ===
    print(f"\n=== Phase 3: Restricted LP-FW (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else best_init
    rem = deadline - time.monotonic() - 350
    if rem > 20:
        t_lpfw = min(rem * 0.55, 250)
        print(f"  RLP-FW t={t_lpfw:.0f}s")
        a_opt, _ = restricted_lp_fw(base, time_limit=t_lpfw, top_k=35,
                                     use_symmetry=True, verbose=True)
        try_seq(a_opt, "rlpfw_sym")

        # Asymmetric LP-FW
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_lpfw2 = min(rem * 0.5, 120)
            print(f"  RLP-FW asym t={t_lpfw2:.0f}s")
            a_opt, _ = restricted_lp_fw(base, time_limit=t_lpfw2, top_k=30,
                                         use_symmetry=False, verbose=False)
            try_seq(a_opt, "rlpfw_asym")

    # === Phase 4: Coordinate Descent ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 15:
            t_cd = min(rem * 0.7, 200)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = smart_coordinate_descent(best_seq, time_limit=t_cd,
                                                 use_symmetry=True)
            try_seq(a_opt, "cd_sym")

            # Asymmetric CD
            rem = deadline - time.monotonic() - 80
            if rem > 10:
                a_opt, _ = smart_coordinate_descent(best_seq, time_limit=min(rem, 80),
                                                     use_symmetry=False)
                try_seq(a_opt, "cd_asym")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        candidate_lengths = []
        for factor in [1.5, 2.0, 0.75]:
            cn = int(n_best * factor)
            cn = max(128, min(4096, cn))
            if cn != n_best and cn not in candidate_lengths:
                candidate_lengths.append(cn)

        for target_n in candidate_lengths:
            rem = deadline - time.monotonic() - 80
            if rem < 30:
                break

            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.4, 130)
            print(f"  Explore n={target_n}, t={t_alloc:.0f}s")

            # SMD on interpolated
            a_opt, _ = smoothed_mirror_descent(f_interp, time_limit=t_alloc * 0.5,
                                               tau_init=0.005, tau_min=0.0005,
                                               eta_init=0.08, use_symmetry=True)
            try_seq(a_opt, f"smd_n{target_n}")

            # RLP-FW on result
            rem = deadline - time.monotonic() - 60
            if rem > 15:
                a_opt, _ = restricted_lp_fw(a_opt, time_limit=min(rem, 60),
                                             top_k=30, use_symmetry=True)
                try_seq(a_opt, f"rlpfw_n{target_n}")

    # === Phase 6: Final Polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Fine SMD
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            a_opt, _ = smoothed_mirror_descent(best_seq, time_limit=min(rem, 60),
                                               tau_init=0.001, tau_min=0.0002,
                                               eta_init=0.03, use_symmetry=True)
            try_seq(a_opt, "smd_fine")

        # Brief LP-FW
        rem = deadline - time.monotonic() - 25
        if rem > 10:
            a_opt, _ = restricted_lp_fw(best_seq, time_limit=min(rem, 40),
                                         top_k=25, use_symmetry=True)
            try_seq(a_opt, "final_lpfw")

        # Tiny perturbations with SMD
        for perturb_idx in range(4):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * 0.003
            f_pert = best_seq + noise
            f_pert = enforce_symmetry(f_pert)
            a_opt, _ = smoothed_mirror_descent(f_pert, time_limit=min(rem, 20),
                                               tau_init=0.001, tau_min=0.0003,
                                               eta_init=0.02, use_symmetry=True)
            try_seq(a_opt, f"perturb_{perturb_idx}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
