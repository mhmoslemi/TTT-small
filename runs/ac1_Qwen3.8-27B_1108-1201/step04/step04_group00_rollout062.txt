def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    v_acc = np.zeros(n)

        exp_w = np.exp(scaled)

        if true_max < best_obj - 1e-15:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed Adam optimization (dense gradients via FFT)
2. LP-based Frank-Wolfe with adaptive constraint relaxation
3. Coordinate descent polishing with adaptive multipliers
4. Multi-length and multi-initialization exploration

Key innovation over previous state of the art: uses a differentiable
log-sum-exp surrogate of the peak functional with Adam optimizer,
providing rich gradient information at each step, combined with
annealing of the smoothing parameter to sharpen the objective.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, a)
    f_interp = np.maximum(f_interp, 0.0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def smoothed_adam(a0, time_limit, t_init=3.0, t_final=3000.0, lr=0.008,
                  beta1=0.88, beta2=0.999, eps=1e-8, restart_interval=3000,
                  noise_scale=0.004):
    """Adam optimization on log-sum-exp smoothed peak objective.
    
    Minimizes (1/t)*log(sum(exp(t*conv[k]))) with annealing t -> t_final.
    Gradient: 2 * correlate(softmax(t*conv), a, 'valid')
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    m = np.zeros(n)
    v_acc = np.zeros(n)
    step_ct = 0
    t0 = time.monotonic()
    best_obj = float('inf')
    best_a = a.copy()
    patience = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        scaled = t_init * (t_final / t_init) ** ((time.monotonic() - t0) / time_limit) * conv
        t_cur = t_init * (t_final / t_init) ** ((time.monotonic() - t0) / time_limit)
        scaled = t_cur * conv
        scaled -= scaled.max()
        exp_w = np.exp(scaled)
        w = exp_w / exp_w.sum()

        true_max = conv.max()
        if true_max < best_obj - 1e-14:
            best_obj = true_max
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        # Gradient: 2 * sum_j w[i+j] * a[j] for each i
        c = np.correlate(w, a, mode='valid')
        grad = 2.0 * c

        # Adam update
        step_ct += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v_acc = beta2 * v_acc + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** step_ct)
        v_hat = v_acc / (1.0 - beta2 ** step_ct)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new)

        # Periodic restart to escape local minima
        if patience > restart_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v_acc = np.zeros(n)
            step_ct = 0
            patience = 0
            noise_scale *= 0.8

    return best_a, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        js = max(0, 0 - (n - 1 - j))  # k - j >= 0 => k >= j
        je = min(2 * n - 2, (n - 1 - j) + (2 * n - 2))  # k - j <= n-1 => k <= j + n - 1
        # A[k, j] = f[j] when 0 <= k-j < n, i.e., j <= k <= j+n-1
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A_ub[k_start:k_end, j] = f[j]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30.0})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def sg_polish(f0, time_limit, lr=0.003, mom=0.94, top_k=50, temp=0.00005):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.003

        if no_improve > 400:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj


def make_structural_init(n, rng):
    """Create a non-uniform structural initialization."""
    a = np.zeros(n)
    # Random "comb" pattern with varying amplitude
    period = rng.integers(3, max(4, n // 30))
    for i in range(n):
        pos_frac = i / max(1, n - 1)
        envelope = 0.5 + 0.5 * np.sin(np.pi * pos_frac)
        a[i] = envelope * (1.0 + 0.5 * np.sin(2 * np.pi * i / period))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a

            rem = deadline - time.monotonic() - 30
            if rem > 15:
                a_opt, _ = sg_polish(best_seq, time_limit=min(rem, 100),
                                     lr=0.002, mom=0.94,
                                     top_k=max(20, len(best_seq) // 6),
                                     temp=0.00003)
                try_seq(a_opt, "sg_after_cd")

    # === Phase 6: Final LP-FW + fine polish ===
    print(f"\n=== Phase 6: Final refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_final_lpfw = min(rem * 0.4, 100.0)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_final_lpfw)
            try_seq(a_opt, "final_lpfw")

            rem = deadline - time.monotonic() - 10
            if rem > 10:
                a_opt, _ = smoothed_adam(best_seq, time_limit=min(rem, 60),
                                         t_init=50.0, t_final=5000.0,
                                         lr=0.003, beta1=0.92, beta2=0.999)
                try_seq(a_opt, "fine_adam")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            n_best = len(best_seq)
            for p in range(2):
                if rem < 4:
                    break
                noise = rng.standard_normal(n_best) * 0.001
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = smoothed_adam(f_pert, time_limit=min(rem, 25),
                                         t_init=20.0, t_final=2000.0, lr=0.002)
                try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence([float(max(0.0, x)) for x in best_seq]))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(max(0.0, x)) for x in best_seq]
