Let me also think about **what makes the LP direction good or bad**:

The LP direction g (after normalizing) is the point in the simplex that maximizes sum while keeping C(a)*g <= M. If g is close to a, the LP is "saturated" and we're near a local optimum. If g is far from a, we have room to improve.

A key insight: **the LP direction depends on which constraints are active**. If only the peak constraint is active, the direction is determined by the peak. If multiple constraints are active, the direction is a compromise.

**Improved approach: Use the dual variables to guide the optimization.**

The dual of the LP is:
min M * sum(y)
s.t. C(a)^T * y >= 1
y >= 0

The dual variables y_k tell us which "input" constraints (the ones with C(a)[k,j] = a[j] shifted) are "expensive." Large y_k means that the k-th convolution constraint is "tight" in the dual sense.

We could use the dual variables to construct a better direction:
- Weight the peaks by their dual values
- Use a weighted subgradient of the max function

This is essentially the "weighted subgradient" method.

Minimizes 2*n*max(conv(a,a))/sum(a)^2 using improved LP-based iterative optimization
with exact grid line search, adaptive constraint tightening, and structured perturbations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = convolve(a, g)."""
    n = len(a)
    # C has shape (2n-1, n)
    # C[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # So C[k, j] = a[k-j] for valid indices
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def objective_value(a, n):
    """Compute the objective 2*n*max(conv)/sum^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def lp_direction(a, M, C):
    """
    Solve LP: max sum(g) s.t. C@g <= M, g >= 0.
    Returns (g, success).
    """
    n = len(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True,
                                  'dual_feasibility_tolerance': 1e-10,
                                  'primal_feasibility_tolerance': 1e-10})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0)
            if g.sum() > 1e-10:
                return g, True
    except Exception:
        pass
    return None, False


def grid_line_search(a, g, n, n_grid=30):
    """
    Grid line search: find best t in [0,1] for a_new = (1-t)*a_norm + t*g_norm.
    Returns (a_best, obj_best).
    """
    a_norm = a / a.sum()
    g_norm = g / g.sum()
    
    best_obj = float('inf')
    best_t = 0.0
    best_a = a_norm.copy()
    
    # Evaluate at t=0
    conv0 = fast_convolve(a_norm)
    best_obj = conv0.max()
    
    for t in np.linspace(0.01, 1.0, n_grid):
        a_try = (1 - t) * a_norm + t * g_norm
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        conv_try = fast_convolve(a_try)
        obj_try = conv_try.max()
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
            best_a = a_try.copy()
    
    return best_a, best_obj

def nesterov_gradient(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=30, temp_init=0.003):
    """Nesterov-accelerated gradient descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    y = a.copy()
    conv_y = fast_convolve(y)
    best_obj = 2.0 * n * conv_y.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        
        conv_y = fast_convolve(y)
        # Compute soft gradient
        peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        peak_vals = conv_y[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
        
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idxs = np.arange(j_start, j_end + 1)
            grad[idxs] += w[idx] * y[k - idxs]
        
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.999
        temp *= 1.002
        mom = min(0.97, mom + 0.0003)
        
        if no_improve >= 300:
            break
    
    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['sine1'] = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine1'] /= inits['sine1'].sum()
    inits['sine2'] = (np.sin(2 * np.pi * x / n) ** 2 + 0.01)
    inits['sine2'] /= inits['sine2'].sum()
    inits['sine3'] = (np.sin(3 * np.pi * x / n) ** 2 + 0.01)
    inits['sine3'] /= inits['sine3'].sum()
    
    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Alternating
    alt = np.ones(n)
    alt[::2] = 2.0
    alt[1::2] = 0.5
    inits['alternating'] = alt / alt.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    inits['gaussian'] = g / g.sum()
    
    # Random low-freq
    freqs = rng.integers(1, 5, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    sig = np.zeros(n)
    for f, p in zip(freqs, phases):
        sig += np.sin(2 * np.pi * f * x / n + p)
    sig = np.abs(sig) + 0.1
    inits['lowfreq'] = sig / sig.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
                # Flipped parent
                inits['parent_flip'] = p[::-1] / ps
                # Smoothed parent
                k = np.ones(3) / 3
                ps_smooth = np.convolve(p / ps, k, mode='same')
                ps_smooth = np.maximum(ps_smooth, 0)
                inits['parent_smooth'] = ps_smooth / ps_smooth.sum()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                inits['parent_interp'] = pp / pp.sum()
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using improved LP-based iterative optimization
with grid line search, adaptive constraint tightening, structured perturbations,
multi-length exploration, and coordinate descent polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = full linear convolution of a and g."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_direction(a, M, C):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0. Returns (g, success)."""
    n = len(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True,
                                  'dual_feasibility_tolerance': 1e-10,
                                  'primal_feasibility_tolerance': 1e-10})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0)
            if g.sum() > 1e-10:
                return g, True
    except Exception:
        pass
    return None, False


def grid_line_search(a, g, n, n_grid=40):

    a_norm = a / a.sum()

    g_norm = g / g_s

def nesterov_gradient(a0, time_limit, lr_init=0.01, momentum=0.91, top_k=40, temp_init=0.002):
    """Nesterov-accelerated gradient descent with softmax peak weighting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    y = a.copy()
    conv_y = fast_convolve(y)
    best_obj = 2.0 * n * conv_y.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y

        conv_y = fast_convolve(y)
        peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        peak_vals = conv_y[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idxs = np.arange(j_start, j_end + 1)
            grad[idxs] += w[idx] * y[k - idxs]

        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.999
        temp *= 1.003
        mom = min(0.97, mom + 0.0002)

        if no_improve >= 300:
            break

    return best_a, best_obj

                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-11:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    no_improve = 0
                    improved = True
                    break
            if not improved:
                no_improve += 1
        if no_improve > 800:
            break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid LP + Nesterov gradient search with:
    - Multi-length exploration (512, 768, 896)
    - LP with adaptive epsilon and coarse-fine grid line search
    - Structured perturbations (shift, flip, smooth, noise, blend)
    - Coordinate descent polishing
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    # ============ Phase 1: Quick gradient exploration ============
    print("=== Phase 1: Gradient Exploration ===")
    t_start = time.monotonic()
    phase1_limit = 250.0

    lengths = [512, 768, 896]
    scan_results = []

    for n in lengths:
        if time.monotonic() > deadline - 1000:
            break
        inits = generate_inits(n, parent, rng)

        priority = []
        for name in ['parent', 'parent_flip', 'parent_smooth', 'parent_interp',
                     'sine1', 'sine2', 'alternating', 'bimodal', 'triangular',
                     'uniform', 'random', 'lowfreq', 'sine3', 'gaussian']:
            if name in inits:
                priority.append(name)

        t_per = max(12.0, (phase1_limit - 10) / (len(lengths) * 3))

        for name in priority[:4]:
            if time.monotonic() > deadline - 900:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 800)
            if t_alloc < 8:
                break

            a0 = inits[name]
            a_opt, obj = nesterov_gradient(a0, time_limit=t_alloc,
                                            lr_init=0.012, momentum=0.91,
                                            top_k=40, temp_init=0.002)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic() - t_start:.0f}s). Best: {best_score:.6f}")
    for sc, nn, nm, _ in scan_results[:5]:
        print(f"  {sc:.6f}  n={nn}  {nm}")

    # ============ Phase 2: Deep LP optimization on top candidates ============
    print("=== Phase 2: LP Deep Optimization ===")
    t_start2 = time.monotonic()

    top_configs = []
    seen_lengths = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen_lengths:
            top_configs.append((n, a_opt, score))
            seen_lengths.add(n)
        if len(top_configs) >= 3:
            break

    if best_sequence is not None:
        bn = len(best_sequence)
        if bn not in seen_lengths:
            top_configs.append((bn, best_sequence.copy(), best_score))
            seen_lengths.add(bn)

    remaining = deadline - time.monotonic() - 200.0
    n_cand = max(1, len(top_configs))
    t_per_lp = max(60.0, remaining * 0.60 / n_cand)

    for i, (n, a0, sc) in enumerate(top_configs):
        if time.monotonic() > deadline - 180:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 130)
        if t_alloc < 30:
            break

        print(f"  LP cand {i}: n={n}, init={sc:.6f}, t={t_alloc:.0f}s")
        a_opt, obj = lp_optimize_improved(a0, time_limit=t_alloc,
                                           eps_init=1e-3, eps_min=1e-7, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP {i} (n={n}): {score:.6f} *** NEW BEST ***")
            else:
                print(f"  LP {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic() - t_start2:.0f}s). Best: {best_score:.6f}")

    # ============ Phase 3: Polish with multiple LP variants ============
    print("=== Phase 3: Polish ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 80.0
        if remaining > 50:
            polish_strategies = [
                ('direct', 1e-4),
                ('noise', 5e-4),
                ('smooth', 1e-4),
                ('flip', 1e-4),
            ]
            for pi, (pname, peps) in enumerate(polish_strategies):
                if time.monotonic() > deadline - 50:
                    break
                t_alloc = min(remaining / (len(polish_strategies) - pi) * 0.8,
                              deadline - time.monotonic() - 40)
                if t_alloc < 15:
                    break

                bs = best_sequence.copy()
                if pname == 'direct':
                    a0 = bs
                elif pname == 'noise':
                    noise = rng.standard_normal(len(bs)) * 0.001
                    a0 = np.maximum(bs + noise, 0)
                    a0 = a0 / a0.sum()
                elif pname == 'smooth':
                    k = np.ones(3) / 3.0
                    a0 = np.convolve(bs, k, mode='same')
                    a0 = np.maximum(a0, 0)
                    a0 = a0 / a0.sum()
                else:
                    a0 = bs[::-1].copy()

                a_opt, obj = lp_optimize_improved(a0, time_limit=t_alloc,
                                                   eps_init=peps, eps_min=1e-7)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Polish {pname}: {score:.6f} *** NEW BEST ***")
                else:
                    print(f"  Polish {pname}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic() - t_start3:.0f}s). Best: {best_score:.6f}")

    # ============ Phase 4: Coordinate descent + gradient fine-tuning ============
    print("=== Phase 4: Fine-tuning ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 25.0
        if remaining > 15:
            t_cd = min(remaining * 0.4, 50.0)
            if t_cd > 5:
                a_opt, obj = coordinate_polish(best_sequence, time_limit=t_cd)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  CoordDesc: {score:.6f} *** NEW BEST ***")
                else:
                    print(f"  CoordDesc: {score:.6f}")

        remaining = deadline - time.monotonic() - 15.0
        if remaining > 10:
            grad_configs = [
                {'lr_init': 0.003, 'momentum': 0.95, 'top_k': 80, 'temp_init': 0.001},
                {'lr_init': 0.008, 'momentum': 0.90, 'top_k': 50, 'temp_init': 0.003},
            ]
            for gi, cfg in enumerate(grad_configs):
                if time.monotonic() > deadline - 12:
                    break
                t_a = min(remaining / (len(grad_configs) - gi),
                          deadline - time.monotonic() - 10)
                if t_a < 3:
                    break
                a_opt, obj = nesterov_gradient(best_sequence, time_limit=t_a, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Grad {gi}: {score:.6f} *** NEW BEST ***")

    print(f"Phase 4 done ({time.monotonic() - t_start4:.0f}s). Best: {best_score:.6f}")

    # ============ Phase 5: Final LP micro-polish ============
    print("=== Phase 5: Final ===")
    if best_sequence is not None and time.monotonic() < deadline - 8:
        t_final = min(25.0, deadline - time.monotonic() - 4)
        if t_final > 5:
            a_opt, obj = lp_optimize_improved(best_sequence, time_limit=t_final,
                                               eps_init=5e-5, eps_min=1e-7)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final: {score:.6f} *** NEW BEST ***")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
