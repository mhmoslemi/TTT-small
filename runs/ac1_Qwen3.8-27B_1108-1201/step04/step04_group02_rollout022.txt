```python
"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using annealed smoothed gradient
descent with softmax parameterization and L-BFGS-B optimizer.
Uses multiple sequence lengths and restarts for robustness.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_objective_and_grad(a, tau):
    """
    Compute the smoothed objective 2*n*smooth_max(conv(a,a))/S^2
    and its gradient with respect to a, on the simplex (S=1).
    Returns (obj, grad_a).
    """
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf, np.zeros(n)
    
    conv = fast_convolve(a)
    conv_max = conv.max()
    
    # Softmax weights
    shifted = (conv - conv_max) / tau
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    smooth_val = conv_max + tau * np.log(exp_shifted.sum())
    
    # Objective (assuming S=1 for gradient, but compute general)
    obj = 2.0 * n * smooth_val / (S * S)
    
    # Gradient of smooth_max w.r.t. a:
    # d(smooth_max)/da[m] = sum_k w[k] * d(conv[k])/da[m]
    # d(conv[k])/da[m] = a[k-m]*I(0<=k-m<n) + a[m-k]*I(0<=m-k<n)
    # grad_conv[m] = sum_k w[k] * a[|k-m|] (with a[x]=0 for x>=n)
    
    # Using correlation trick:
    a_sym = np.concatenate([a[::-1], a[1:]])  # length 2n-1
    # grad_conv[m] = correlate(a_sym, w, 'full')[m + n - 1]
    # But for efficiency, use FFT:
    fft_size2 = 1 << (2 * (2 * n - 1) - 1 - 1).bit_length()
    AS = np.fft.rfft(a_sym, fft_size2)
    W = np.fft.rfft(w, fft_size2)
    corr = np.fft.irfft(AS * W, fft_size2)
    # correlate(a_sym, w) in numpy = corr reversed? Let me use np.correlate for correctness
    # Actually, let me verify: np.correlate(a, v, 'full')[k] = sum_n a[n+k] * conj(v[n])
    # I want sum_k w[k] * a_sym[k-m+n-1]
    # Let's just use np.correlate for correctness (n=384 is fast enough)
    
    # Actually for n=384, np.correlate is O(n^2) ~ 300K ops, which is fine.
    corr = np.correlate(a_sym, w, mode='full')
    grad_conv = corr[n-1:2*n-1]  # length n
    
    # Full gradient of obj w.r.t. a:
    # d_obj/da[m] = 2*n/S^2 * grad_conv[m] - 4*n*smooth_val/S^3
    grad_a = 2.0 * n / (S * S) * grad_conv - 4.0 * n * smooth_val / (S ** 3)
    
    return obj, grad_a


def softmax_params_to_simplex(x):
    """Convert logits to simplex (probability vector)."""
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    return exp_x / exp_x.sum()


def simplex_to_softmax_params(a):
    """Convert simplex point to logits (with small epsilon to avoid log(0))."""
    return np.log(np.maximum(a, 1e-300))


def objective_from_logits(x, tau):
    """Objective function taking logits as input."""
    a = softmax_params_to_simplex(x)
    obj, grad_a = smooth_objective_and_grad(a, tau)
    return obj, a, grad_a


def grad_from_logits(x, tau):
    """Gradient of objective w.r.t. logits."""
    a = softmax_params_to_simplex(x)
    obj, grad_a = smooth_objective_and_grad(a, tau)
    # Chain rule: d_obj/dx[m] = a[m] * (grad_a[m] - sum_j grad_a[j]*a[j])
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)
    return obj, grad_x

    """

    Uses L-BFGS-B on the logit parameterization.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-10)
    a = a / a.sum()
    
    best_obj = evaluate_sequence(a.tolist())
```
