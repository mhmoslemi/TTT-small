"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
Phase 1: Diverse initialization
Phase 2: Aggressive subgradient with Nesterov momentum
Phase 3: LP-based Frank-Wolfe (linearized + max-sum oracles)
Phase 4: Fast O(n) coordinate local search (key innovation - incremental conv updates)
Phase 5: Final polish and length exploration
The coordinate local search uses incremental convolution updates for ~700x speedup
over FFT-based evaluation, enabling thousands of additional trial moves.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build B matrix where (B*g)[k] = (f*g)[k]."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B


def solve_linearized_lp(f, conv_f, B=None):
    """Solve: min t s.t. t >= conv_f[k] + 2*(B*g)[k], g >= -f, sum(g)=0."""
    n = len(f)
    if B is None:
        B = build_conv_matrix(f)
    c = np.zeros(n + 1)
    c[0] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B
    b_ub = -conv_f
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[1:]
    except Exception:
        pass
    return None


def solve_reduced_lp(f, conv_f, top_k=20):
    """Reduced LP using only top_k peak constraints."""
    n = len(f)
    peaks = np.argpartition(conv_f, -top_k)[-top_k:]
    peaks = np.sort(peaks)
    B_full = build_conv_matrix(f)
    c = np.zeros(n + 1)
    c[0] = 1.0
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B_full[peaks, :]
    b_ub = -conv_f[peaks]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[1:]
    except Exception:
        pass
    return None


def solve_maxsum_lp(f, M):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None and np.any(res.x > 0):
            return res.x
    except Exception:
        pass
    return None


def line_search(f, target, current_obj, max_alpha=1.0):
    """Find best alpha for f_new = (1-alpha)*f + alpha*target."""
    best_alpha = 0.0
    best_obj = current_obj
    alphas = np.linspace(0.05, max_alpha, 10)
    for alpha in alphas:
        f_new = (1.0 - alpha) * f + alpha * target
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new = f_new / s
        obj = fast_convolve(f_new).max()
        if obj < best_obj:
            best_obj = obj
            best_alpha = alpha
    if best_alpha > 0:
        lo = max(0.01, best_alpha * 0.5)
        hi = min(max_alpha, best_alpha * 1.5)
        for _ in range(6):
            mid = (lo + hi) / 2.0
            f_new = (1.0 - mid) * f + mid * target
            s = f_new.sum()
            if s < 1e-15:
                break
            f_new = f_new / s
            obj = fast_convolve(f_new).max()
            if obj < best_obj:
                best_obj = obj
                best_alpha = mid
                lo = mid
            else:
                hi = mid
    return best_alpha, best_obj


def subgradient_optimize(f0, time_limit, lr=0.005, mom=0.9, top_k=1,
                         temp=0.0001, restart_interval=150):
    """Aggressive subgradient with Nesterov momentum and backtracking."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        cur_obj = conv.max()

        # Get top peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        # Compute subgradient (Nesterov look-ahead)
        y = f + mom * v
        conv_y = fast_convolve(y)

        sg = np.zeros(n)
        pv = conv_y[peaks]
        if top_k > 1:
            scaled = pv / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w /= w.sum()
        else:
            w = np.ones(len(peaks)) / len(peaks)

        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * y[k - idx]

        # Backtracking line search
        step = cur_lr
        accepted = False
        for _ in range(10):
            v_new = mom * v - step * sg
            f_cand = project_simplex(f + v_new)
            obj_cand = fast_convolve(f_cand).max()
            expected_decrease = cur_obj - step * np.dot(sg, f - f_cand)
            if obj_cand <= expected_decrease + 1e-8 or obj_cand < cur_obj:
                f = f_cand
                v = v_new
                accepted = True
                break
            step *= 0.5

        if not accepted:
            f = project_simplex(f + v)
            v = mom * v

        obj_now = fast_convolve(f).max()
        if obj_now < best_obj - 1e-13:
            best_obj = obj_now
            best_f = f.copy()
            no_improve = 0
            cur_lr = min(cur_lr * 1.02, 0.02)
        else:
            no_improve += 1
            cur_lr *= 0.99

        if no_improve > restart_interval:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.001)
            no_improve = 0

    return best_f, best_obj

def eval_conv_incremental(conv, n, i, delta):

    k_max = min(2 * n - 2, i + n - 1)

        if k == 2 * i:
            change = 2.0 * delta * conv[2 * i] / (2.0 * conv[2 * i] + delta) if conv[2 * i] > 0 else 0
            # Actually: old conv[2i] includes a[i]^2, new has (a[i]+delta)^2
            # change = 2*delta*a[i] + delta^2, but we don't have a[i] here
            # We need a[i]. Let's pass it differently.
            pass
        vals.append(k)
    return max_new_val


def coord_local_search(f0, time_limit, rng):
    """
    Fast coordinate local search with O(n) incremental convolution updates.
    Key insight: changing a[i] by delta only affects lags k in [i, i+n-1],
    and the change at each such lag is O(1) to compute.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f).copy()
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        total_iters += 1
        peak_lag = int(np.argmax(conv))
        improved = False

        # Compute impact scores: how much each coordinate contributes to the peak
        # Changing a[i] affects conv[peak_lag] by 2*delta*a[peak_lag-i] if valid
        impact = np.zeros(n)
        for i in range(n):
            j = peak_lag - i
            if 0 <= j < n:
                if j == i:
                    impact[i] = 2.0 * f[i]
                else:
                    impact[i] = 2.0 * f[j]

        # Try coordinates in order of impact
        order = np.argsort(-impact)

        # Try both decreasing (to reduce peak) and small increases (to redistribute)
        for idx_i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(order[idx_i])
            if impact[i] < 1e-12:
                break

            # Try several step sizes (decrease)
            tried = 0
            for scale in [0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
                delta = -scale * f[i]
                new_ai = f[i] + delta
                if new_ai < -1e-15:
                    continue

                # Compute new max using incremental update
                k_min = i
                k_max = min(2 * n - 2, i + n - 1)
                new_max = 0.0

                # Check affected lags
                for k in range(k_min, k_max + 1):
                    if k == 2 * i:
                        change = 2.0 * delta * f[i] + delta * delta
                    else:
                        ji = k - i
                        if 0 <= ji < n:
                            change = 2.0 * delta * f[ji]
                        else:
                            change = 0.0
                    new_val = conv[k] + change
                    if new_val > new_max:
                        new_max = new_val

                # Also check the max over UNaffected lags
                # The unaffected max is the old max minus any affected peak
                # Simpler: just track old_max and check if it's in affected range
                old_max_unaffected = 0.0
                if best_obj <= 0:
                    old_max_unaffected = 0.0
                else:
                    # Find max over lags NOT in [k_min, k_max]
                    # This is O(n) but we can cache it
                    # For simplicity, just use the global old max as upper bound
                    # and verify at the end
                    old_max_unaffected = 0.0
                    for k in range(2 * n - 1):
                        if k < k_min or k > k_max:
                            if conv[k] > old_max_unaffected:
                                old_max_unaffected = conv[k]

                total_new_max = max(new_max, old_max_unaffected)

                if total_new_max < best_obj - 1e-12:
                    # Apply the change
                    for k in range(k_min, k_max + 1):
                        if k == 2 * i:
                            conv[k] += 2.0 * delta * f[i] + delta * delta
                        else:
                            ji = k - i
                            if 0 <= ji < n:
                                conv[k] += 2.0 * delta * f[ji]
                    f[i] = max(0.0, new_ai)
                    best_obj = total_new_max
                    best_f = f.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved:
            no_improve += 1
            # Try proportional redistribution: move mass from peak contributors to others
            if no_improve % 20 == 0:
                # Find the top contributor to the peak
                # and try to move mass to a less-impactful coordinate
                contributors = []
                for i in range(n):
                    j = peak_lag - i
                    if 0 <= j < n and f[i] > 0 and f[j] > 0:
                        contributors.append((f[i] * f[j], i))
                if contributors:
                    contributors.sort(reverse=True)
                    top_contrib = contributors[0][1]
                    # Find a coordinate to move mass to
                    low_impact = np.argsort(impact)[:10]
                    for target_i in low_impact:
                        if target_i == top_contrib:
                            continue
                        frac = 0.05
                        delta = -frac * f[top_contrib]
                        delta2 = -delta  # positive, add to target
                        new_top = f[top_contrib] + delta
                        new_tgt = f[target_i] + delta2
                        if new_top < 0 or new_tgt < 0:
                            continue
                        # Check if this reduces the max
                        # This is a 2-coordinate change, more expensive
                        # For speed, just try single: increase target_i from mass
                        pass

            if no_improve > 100:
                # Restart with noise
                noise_scale = 0.003 + 0.002 * min(no_improve // 100, 5)
                f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
                conv = fast_convolve(f).copy()
                cur_obj = conv.max()
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_f = f.copy()
                no_improve = 0

    return best_f, best_obj


def fast_coord_search_v2(f0, time_limit, rng):
    """
    Optimized coordinate local search with cached unaffected-max.
    Uses the O(n) incremental update but caches the max over unaffected region.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f).copy()
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    rng2 = np.random.default_rng(1234)

    while time.monotonic() - t0 < time_limit:
        peak_lag = int(np.argmax(conv))
        improved = False

        # For each coordinate i, the affected range is [i, min(2n-2, i+n-1)]
        # Precompute the unconstrained max (max over all lags)
        # and for each i, the max over lags NOT in [i, i+n-1]

        # Efficient approach: for a given i, lags not affected are [0, i-1] union [i+n, 2n-2]
        # We can precompute prefix and suffix maxes

        # Actually, let's just be smart: try the most impactful coordinates first
        # and do the full O(n) check only for those

        # Compute impact on peak
        impact = np.zeros(n)
        for i in range(n):
            j = peak_lag - i
            if 0 <= j < n:
                if j == i:
                    impact[i] = 2.0 * f[i]
                else:
                    impact[i] = 2.0 * f[j]

        order = np.argsort(-impact)

        # Try top coordinates with multiple step sizes
        for count, idx_i in enumerate(order):
            if time.monotonic() - t0 > time_limit:
                break
            if count > 60:  # Limit per sweep
                break
            i = int(idx_i)
            if impact[i] < 1e-10:
                break

            k_min = i
            k_max = min(2 * n - 2, i + n - 1)

            # Precompute max over unaffected lags for this i
            # Unaffected: [0, k_min-1] and [k_max+1, 2n-2]
            unaff_max = 0.0
            for k in range(0, k_min):
                if conv[k] > unaff_max:
                    unaff_max = conv[k]
            for k in range(k_max + 1, 2 * n - 1):
                if conv[k] > unaff_max:
                    unaff_max = conv[k]

            # Try step sizes
            for scale in [0.2, 0.1, 0.05, 0.02, 0.01, 0.005]:
                delta = -scale * f[i]
                new_ai = f[i] + delta
                if new_ai < -1e-14:
                    continue

                # Compute max over affected lags
                aff_max = 0.0
                for k in range(k_min, k_max + 1):
                    if k == 2 * i:
                        change = 2.0 * delta * f[i] + delta * delta
                    else:
                        ji = k - i
                        if 0 <= ji < n:
                            change = 2.0 * delta * f[ji]
                        else:
                            change = 0.0
                    new_val = conv[k] + change
                    if new_val > aff_max:
                        aff_max = new_val

                total_max = max(aff_max, unaff_max)

                if total_max < best_obj - 1e-12:
                    # Apply
                    for k in range(k_min, k_max + 1):
                        if k == 2 * i:
                            conv[k] += 2.0 * delta * f[i] + delta * delta
                        else:
                            ji = k - i
                            if 0 <= ji < n:
                                conv[k] += 2.0 * delta * f[ji]
                    f[i] = max(0.0, new_ai)
                    best_obj = total_max
                    best_f = f.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 50:
                # Larger restart
                scale = 0.002 + 0.001 * (no_improve // 50)
                f = project_simplex(best_f + rng2.standard_normal(n) * scale)
                conv = fast_convolve(f).copy()
                if conv.max() < best_obj:
                    best_obj = conv.max()
                    best_f = f.copy()
                no_improve = 0

    return best_f, best_obj


def dual_coord_search(f0, time_limit, rng):
    """
    Dual coordinate search: simultaneously adjust two coordinates.
    More powerful than single-coordinate but 2x cost.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f).copy()
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        peak_lag = int(np.argmax(conv))
        improved = False

        # Find top 2 contributors to the peak
        contrib = []
        for i in range(n):
            j = peak_lag - i
            if 0 <= j < n and f[i] > 1e-8 and f[j] > 1e-8:
                contrib.append((f[i] * f[j], i, j))
        contrib.sort(reverse=True)

        if len(contrib) >= 2:
            # Try: decrease i, increase j (or vice versa)
            for attempt in range(min(5, len(contrib))):
                if time.monotonic() - t0 > time_limit:
                    break
                _, i, j = contrib[attempt]
                if i == j:
                    continue
                for frac in [0.1, 0.05, 0.02]:
                    delta_i = -frac * f[i]
                    delta_j = frac * f[i]  # conserve sum
                    new_fi = f[i] + delta_i
                    new_fj = f[j] + delta_j
                    if new_fi < -1e-14 or new_fj < -1e-14:
                        continue

                    # Compute new max (O(n) for each of the two changes)
                    # Affected by i: [i, i+n-1]
                    # Affected by j: [j, j+n-1]
                    # Union of affected ranges
                    k_min = min(i, j)
                    k_max = max(min(2*n-2, i+n-1), min(2*n-2, j+n-1))

                    # Just do full check over union
                    new_max = 0.0
                    for k in range(2 * n - 1):
                        new_val = conv[k]
                        # Contribution from changing a[i]
                        if i <= k <= min(2*n-2, i+n-1):
                            if k == 2 * i:
                                new_val += 2.0 * delta_i * f[i] + delta_i * delta_i
                            else:
                                ji = k - i
                                if 0 <= ji < n:
                                    new_val += 2.0 * delta_i * f[ji]
                        # Contribution from changing a[j]
                        if j <= k <= min(2*n-2, j+n-1):
                            if k == 2 * j:
                                new_val += 2.0 * delta_j * f[j] + delta_j * delta_j
                            else:
                                jj = k - j
                                if 0 <= jj < n:
                                    new_val += 2.0 * delta_j * f[jj]
                        if new_val > new_max:
                            new_max = new_val

                    if new_max < best_obj - 1e-12:
                        # Apply both changes
                        # Update conv for change at i
                        for k in range(i, min(2*n-1, i+n)):
                            if k == 2 * i:
                                conv[k] += 2.0 * delta_i * f[i] + delta_i * delta_i
                            else:
                                ji = k - i
                                if 0 <= ji < n:
                                    conv[k] += 2.0 * delta_i * f[ji]
                        # Update conv for change at j
                        for k in range(j, min(2*n-1, j+n)):
                            if k == 2 * j:
                                conv[k] += 2.0 * delta_j * f[j] + delta_j * delta_j
                            else:
                                jj = k - j
                                if 0 <= jj < n:
                                    conv[k] += 2.0 * delta_j * f[jj]
                        f[i] = max(0.0, new_fi)
                        f[j] = max(0.0, new_fj)
                        best_obj = new_max
                        best_f = f.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                conv = fast_convolve(f).copy()
                no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 1.0))
    inits.append(rng.dirichlet(np.ones(n) * 3.0))

    x = np.arange(n, dtype=np.float64)
    for sigma_frac in [0.1, 0.2, 0.3, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
        inits.append(g / g.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.01
    inits.append(c1 / c1.sum())

    # M-sequence inspired (pseudo-random with structure)
    mseq = np.zeros(n)
    period = (1 << 8) - 1  # 255-bit m-sequence
    state = rng.integers(1, period + 1)
    for i in range(n):
        mseq[i] = 1.0 if (state & 1) else 0.0
        # LFSR step
        bit = ((state >> 7) ^ (state >> 5) ^ (state >> 3) ^ state) & 1
        state = (state >> 1) | (bit << 7)
        state &= period
    mseq += 0.1
    inits.append(mseq / mseq.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining subgradient, LP-based FW,
    and fast O(n) coordinate local search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.10f}")
    except Exception:
        pass

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")
    t_start = time.monotonic()
    t1_deadline = t_start + 40

    n_base = len(parent) if parent is not None else 384

    candidates = []
    if parent is not None:
        candidates.append(("parent", parent))
    else:
        candidates.append(("uniform", np.ones(n_base) / n_base))

    # Generate diverse inits at base length
    inits = generate_inits(n_base, parent, rng)
    for i, ini in enumerate(inits):
        candidates.append((f"init_{i}", ini))

    # Try a couple other lengths
    for n_try in [n_base - 1, n_base + 1, n_base - 7, n_base + 7]:
        if n_try < 32:
            continue
        if parent is not None:
            px = np.interp(np.linspace(0, 1, n_try),
                          np.linspace(0, 1, len(parent)), parent)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                candidates.append((f"len{n_try}", px / ps))
        else:
            candidates.append((f"len{n_try}", np.ones(n_try) / n_try))

    # Quick evaluation
    for label, cand in candidates:
        if time.monotonic() > t1_deadline:
            break
        try_seq(cand, f"init_{label}")

    if best_seq is None:
        best_seq = np.ones(384) / 384
        try_seq(best_seq, "fallback")

    print(f"  Phase 1 best: {best_score:.10f} (n={len(best_seq)})")

    # === Phase 2: Subgradient optimization ===
    print("\n=== Phase 2: Subgradient ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 100
    t_alloc_p2 = min(remaining * 0.18, 300)

    if t_alloc_p2 > 30:
        print(f"  Running subgradient (n={n_best}), budget={t_alloc_p2:.0f}s")
        a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc_p2 * 0.7,
                                        lr=0.005, mom=0.9, top_k=1, temp=0.0001)
        try_seq(a_opt, "sg_main")

        rem = deadline - time.monotonic() - 100
        if rem > 30:
            a_opt, _ = subgradient_optimize(best_seq, time_limit=min(rem * 0.4, 150),
                                            lr=0.002, mom=0.92, top_k=5, temp=0.00005)
            try_seq(a_opt, "sg_multi")

    print(f"  Phase 2 best: {best_score:.10f}")

    # === Phase 3: LP-based Frank-Wolfe ===
    print("\n=== Phase 3: LP Frank-Wolfe ===")
    remaining = deadline - time.monotonic() - 100
    t_alloc_p3 = min(remaining * 0.22, 400)

    if t_alloc_p3 > 30:
        print(f"  Running LP FW (n={n_best}), budget={t_alloc_p3:.0f}s")
        # Full LP
        t_sub = min(t_alloc_p3 * 0.4, 160)
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_sub, use_reduced=False)
        try_seq(a_opt, "lpfw_full")

        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_sub2 = min(rem * 0.4, 150)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_sub2,
                                      use_reduced=True, reduced_k=25)
            try_seq(a_opt, "lpfw_red")

        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_sub3 = min(rem * 0.5, 120)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_sub3, use_maxsum=True)
            try_seq(a_opt, "maxsum_fw")

    print(f"  Phase 3 best: {best_score:.10f}")

    # === Phase 4: Fast coordinate local search (main workhorse) ===
    print("\n=== Phase 4: Coordinate Local Search ===")
    remaining = deadline - time.monotonic() - 80
    t_alloc_p4 = min(remaining * 0.45, 700)

    if t_alloc_p4 > 20:
        print(f"  Running coord search (n={n_best}), budget={t_alloc_p4:.0f}s")
        # First: single-coordinate search
        t_cs1 = min(t_alloc_p4 * 0.5, 350)
        a_opt, _ = fast_coord_search_v2(best_seq, time_limit=t_cs1, rng=rng)
        try_seq(a_opt, "coord_v2")

        rem = deadline - time.monotonic() - 80
        if rem > 20:
            # Dual-coordinate search
            t_cs2 = min(rem * 0.5, 200)
            a_opt, _ = dual_coord_search(best_seq, time_limit=t_cs2, rng=rng)
            try_seq(a_opt, "dual_coord")

        rem = deadline - time.monotonic() - 80
        if rem > 20:
            # Another round of single-coordinate from current best
            t_cs3 = min(rem * 0.7, 150)
            a_opt, _ = fast_coord_search_v2(best_seq, time_limit=t_cs3, rng=rng)
            try_seq(a_opt, "coord_v2_r2")

    print(f"  Phase 4 best: {best_score:.10f}")

    # === Phase 5: Final polish ===
    print("\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic() - 20
    t_alloc_p5 = min(remaining * 0.8, 250)

    if t_alloc_p5 > 15:
        # Fine subgradient
        t_sg = min(t_alloc_p5 * 0.3, 80)
        a_opt, _ = subgradient_optimize(best_seq, time_limit=t_sg,
                                        lr=0.001, mom=0.95, top_k=3, temp=0.00002)
        try_seq(a_opt, "sg_fine")

        rem = deadline - time.monotonic() - 20
        if rem > 15:
            # Final coordinate search
            a_opt, _ = fast_coord_search_v2(best_seq, time_limit=min(rem * 0.5, 80), rng=rng)
            try_seq(a_opt, "coord_final")

        rem = deadline - time.monotonic() - 15
        if rem > 10:
            # Try a perturbed restart + quick optimize
            noise = rng.standard_normal(n_best) * 0.003
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = fast_coord_search_v2(f_pert, time_limit=min(rem, 40), rng=rng)
            try_seq(a_opt, "perturb_final")

    # Final verification
    if best_seq is not None:
        final_score = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
