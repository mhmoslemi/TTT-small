"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed FISTA (accelerated gradient descent)
2. LP-based Frank-Wolfe for global direction finding
3. Multi-length exploration with diverse initializations
4. Stochastic local search polishing

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(a, temp):
    """Compute G = sum_k w_k * grad(conv_k) where w is softmax of conv/temp."""
    n = len(a)
    conv = fast_convolve(a)
    # Softmax weights
    shifted = conv - conv.max()
    exp_v = np.exp(shifted / temp)
    Z = exp_v.sum()
    if Z < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w = exp_v / Z
    
    # Cross-correlation: cross_corr[i] = sum_j w[i+j] * a[j]
    cross_corr = np.correlate(w, a, mode='valid')
    
    # G_i = 2 * cross_corr[i] - w[2i] * a[i]
    G = 2.0 * cross_corr
    # Apply correction for k=2i
    idx_2i = 2 * np.arange(n)
    mask = idx_2i < len(conv)
    G[mask] -= w[idx_2i[mask]] * a[mask]
    
    return G, conv


def fista_optimize(a0, time_limit, temp=0.01, lr0=0.01, verbose=False):
    """FISTA (accelerated gradient descent) on the smoothed objective."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    # Initial gradient
    G, conv = smooth_gradient(a, temp)
    obj = conv.max()
    best_obj = obj
    best_a = a.copy()
    
    # FISTA variables
    b = a.copy()  # the extrapolated point
    t = 1.0
    lr = lr0
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break
        
        # Compute gradient at b
        G_b, conv_b = smooth_gradient(b, temp)
        
        # Gradient step
        a_new = b - lr * G_b
        
        # Project onto simplex
        a_new = project_simplex(a_new)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        
        # Backtracking line search (simplified: just use fixed lr for now)
        # Actually for FISTA with non-smooth projection, we need to be careful
        # Let's use a simpler momentum approach instead
        
        # Simple momentum (Nesterov-like)
        b_new = a_new + (t / t_new - 1.0) * (a_new - a)
        
        a = a_new
        b = b_new
        t = t_new
        
        # Evaluate
        conv = fast_convolve(a)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        # Decrease lr slowly
        lr *= 0.9999
        
        # Restart if stuck
        if no_improve > 200:
            # Restart FISTA
            b = a.copy()
            t = 1.0
            lr = lr0 * 0.5
            no_improve = 0
        
        if verbose and it % 1000 == 0:
            print(f"    FISTA it={it}, obj={obj:.10f}, best={best_obj:.10f}, lr={lr:.6f}")
    
    return best_a, best_obj


def fw_lp_optimize(a0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)

        a = a / s

    M = conv.max()

    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:

        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        
        # Solve LP
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):

        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n
        
        g = None
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': 10})
            if res.success:
                g = res.x
        except Exception:
            pass
        
        if g is None:
            m_exp = min(m_exp * 1.3, 5.0)
            if m_exp > 4.0:
                # Skip LP, do a random move
                rng = np.random.default_rng(it)
                a = project_simplex(a + rng.standard_normal(n) * 0.01)
                conv = fast_convolve(a)
                M = conv.max()
                m_exp = 1.0
                continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - a)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.02, 2.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            at = (1.0 - alpha) * a + alpha * g
            at = project_simplex(at)
            val = fast_convolve(at).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = project_simplex(a)
            conv = fast_convolve(a)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()

            else:
                no_improve += 1

            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)

            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            conv = fast_convolve(a)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_a, best_obj


def stochastic_polish(a0, time_limit, step_size=0.001, verbose=False):
    """Stochastic local search: random perturbations, keep improvements."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    cur_step = step_size
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        rng = np.random.default_rng(it * 7919 + 13)
        # Random perturbation
        noise = rng.standard_normal(n) * cur_step
        a_try = project_simplex(a + noise)
        
        val = fast_convolve(a_try).max()
        if val < best_obj - 1e-15:
            best_obj = val
            best_a = a_try.copy()
            a = a_try
            cur_step = max(cur_step * 1.01, 0.0001)  # Slightly increase
        else:
            # Decay step size
            cur_step *= 0.999
            # Occasionally reset to best
            if it % 100 == 0:
                a = best_a.copy()
        
        # Also try local coordinate changes
        if it % 10 == 0:
            i = it % n
            j = (i + n // 2) % n
            # Try swapping a small amount between i and j
            for delta in [0.001, 0.005, 0.01]:
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                if a_try[i] >= 0 and a_try[j] >= 0:
                    a_try = a_try / a_try.sum()
                    val = fast_convolve(a_try).max()
                    if val < best_obj - 1e-15:
                        best_obj = val
                        best_a = a_try.copy()
                        a = a_try
                        break
    
    return best_a, best_obj


def generate_inits(n, parent=None, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)

    # Cosine
    c = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 2 + 0.01
    inits.append(c / c.sum())
    
    # Gaussian bump
    g = np.exp(-0.5 * ((x - n/2) / (n/4))**2) + 0.01
    inits.append(g / g.sum())
    
    # U-shaped
    u = ((x - n/2)**2) / (n**2/4) + 0.1
    inits.append(u / u.sum())
    
    # Ramp
    r = np.linspace(0.1, 1.0, n)
    inits.append(r / r.sum())
    
    # Reverse ramp
    rr = np.linspace(1.0, 0.1, n)
    inits.append(rr / rr.sum())
    
    # Sine
    sn = np.sin(np.pi * x / n) + 0.1
    inits.append(sn / sn.sum())
    
    # Bimodal
    b1 = np.exp(-0.5 * ((x - n*0.25) / (n/8))**2) + np.exp(-0.5 * ((x - n*0.75) / (n/8))**2) + 0.01
    inits.append(b1 / b1.sum())
    
    # Random
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)
    
    # Another random
    r2 = rng.dirichlet(np.ones(n) * 0.2)
    inits.append(r2)
    
    # Parent-based
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits
