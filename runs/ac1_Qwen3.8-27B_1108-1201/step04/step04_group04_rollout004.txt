Actually, let me reconsider the coordinate descent. The key insight is:

For a fixed i, if I change a[i] by delta (and renormalize), the change in the objective can be computed efficiently. But the renormalization step makes it tricky.

Alternative: don't renormalize. Instead, work with the unnormalized objective:
F(a) = 2n * max_k conv(a,a)[k] / (sum(a))^2

For coordinate descent on this:
- Fix all a[j] for j ≠ i
- Vary a[i] from 0 to some max
- Evaluate F(a) (or an approximation)
- Pick the best a[i]

Since F is scale-invariant, we can fix sum(a) = 1 without loss of generality. So when we change a[i], we need to adjust other coordinates to maintain sum = 1.

Simplest approach: change a[i], then scale all other coordinates to maintain the sum:
a[j] *= (1 - a[i]) / (1 - a[i]_old) for j ≠ i

This is a "proportional renormalization" and it's a valid step in the simplex.

Let me implement this:

```python
def coord_desc_step(f, i, n_evals=20):
    """Try to improve f by changing coordinate i with proportional renormalization."""
    n = len(f)
    ai = f[i]
    # Try different values of a[i]
    # a[i] ranges from 0 to 1 (in the simplex)
    # When a[i] = x, the others are scaled by (1-x)/(1-ai)
    # Evaluate objective for a few x values
    
    best_obj = float('inf')
    best_x = ai
    
    # Sample x values
    xs = np.linspace(0, min(1, 2*ai + 0.01), n_evals)
    xs = np.append(xs, ai)  # include current
    
    for x in xs:
        if x < 0 or x >= 1:
            continue
        f_trial = f.copy()
        f_trial[i] = x
        if x < 1:
            # Scale others proportionally
            others_sum = 1 - ai
            if others_sum > 1e-10:
                f_trial[np.arange(n) != i] *= (1 - x) / others_sum
        f_trial = np.maximum(f_trial, 0)
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial /= s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj:
            best_obj = obj
            best_x = x
    
    # Refine around best_x with a few more evaluations
    if best_x != ai:
        f[i] = best_x
        if best_x < 1:
            f[np.arange(n) != i] *= (1 - best_x) / (1 - ai)
    
    return f
```

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_nonnormalized(a):
    """Compute the objective for a (assumed non-negative, sum > 0)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient(f):
    """Compute subgradient of max_k conv(f,f)[k] w.r.t. f."""
    n = len(f)
    conv = fast_convolve(f)
    # Find all k where conv[k] is within epsilon of max
    max_val = conv.max()
    # Use top-k for a smoother gradient
    k = max(1, int(np.sum(conv >= max_val - 0.01 * max_val)))
    # Weighted subgradient
    idx = np.where(conv >= max_val - 0.01 * max_val)[0]
    if len(idx) == 0:
        idx = np.array([np.argmax(conv)])
    # For each active k, subgrad[i] += 2 * f[k-i] (if valid)
    sg = np.zeros(n)
    for ki in idx:
        ki = int(ki)
        js = max(0, ki - n + 1)
        je = min(n - 1, ki)
        for j in range(js, je + 1):
            sg[j] += 2.0 * f[ki - j]
    sg /= len(idx)  # average
    return sg, conv

def smooth_subgradient(f, T=0.005):
    """Compute smoothed subgradient using log-sum-exp."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    # subgrad[i] = 2 * sum_k w[k] * f[k-i] (if valid)
    # This is a cross-correlation
    fft_size = 1 << (3 * n).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(np.conj(F) * W, fft_size)
    return 2.0 * cross[:n], conv

            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Bisection refinement
        if best_alpha > 0.01:
            lo = max(0, best_alpha - 0.15)
            hi = min(1.0, best_alpha + 0.15)
            for _ in range(8):
                mid = (lo + hi) / 2
                ft = (1 - mid) * f + mid * g
                ft = np.maximum(ft, 0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft /= st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f = (1 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0)
            f /= f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"      FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 4.0)
            
        if no_improve > 30:
            rng = np.random.default_rng((it * 7919 + n * 104729) % (2**31))
            scale = 0.001 + 0.001 * min(no_improve // 30, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj

def solve_lp_dir(f, M):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None
