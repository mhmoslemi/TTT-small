    """Adam optimizer on the simplex."""
    n = len(a0)
    v = a0.copy()  # parameter
    m = np.zeros(n)  # first moment
    u = np.zeros(n)  # second moment
    t = 0
    best_a = a0.copy()
    best_val = np.inf
    
    for i in range(num_iters):
        t += 1
        a = np.maximum(v, 1e-15)
        a = a / a.sum()
        
        obj, grad_a = lp_objective_and_grad(a, p)
        
        if obj < best_val:
            best_val = obj
            best_a = a.copy()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_a
        u = beta2 * u + (1 - beta2) * grad_a**2
        m_hat = m / (1 - beta1**t)
        u_hat = u / (1 - beta2**t)

    a = np.maximum(v, 1e-15)
    a = a / a.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Lp-norm smoothing with Adam on
log-parameterization, coordinate descent, peak equalization via Newton steps,
and multi-scale temperature annealing. Key ideas: Lp approximation of L-infinity
for smooth optimization, softmax parameterization for automatic simplex constraint,
FFT-based gradient computation, and targeted peak equalization for final polish.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_lp(a, p):
    """Compute gradient of sum(b^p) w.r.t. a, where b = conv(a,a). Uses FFT."""
    n = len(a)
    b = fast_convolve(a)
    w = b ** (p - 1)  # length 2n-1
    # grad_a[j] = 2p * sum_{m=0}^{n-1} a[m] * w[j+m]
    # Using FFT: c[m] = (a_ext * w_rev)[2n-2-m]
    fft_size = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    w_rev = w[::-1]
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(A * W, fft_size)
    c = conv[2 * n - 2 : n - 2 : -1]
    return 2.0 * p * c


def softmax_ppm(v):
    """Stable softmax."""
    v_shift = v - v.max()
    e = np.exp(v_shift)
    return e / e.sum()


def adam_logparam(v0, p, num_iters, lr=0.005, beta1=0.9, beta2=0.999, eps=1e-8,
                  time_limit=None, t_start=None):
    """Adam optimizer on log-parameterization (softmax) for Lp objective."""
    n = len(v0)
    v = v0.copy().astype(np.float64)
    m = np.zeros(n)
    u = np.zeros(n)
    t = 0
    best_a = softmax_ppm(v)
    best_b = fast_convolve(best_a)
    best_val = np.sum(best_b ** p)
    best_maxb = best_b.max()

    if t_start is None:
        t_start = time.monotonic()

    for i in range(num_iters):
        if time_limit is not None and time.monotonic() - t_start > time_limit:
            break
        t += 1
        a = softmax_ppm(v)
        b = fast_convolve(a)
        obj = np.sum(b ** p)
        cur_maxb = b.max()
        if obj < best_val:
            best_val = obj
            best_a = a.copy()
            best_maxb = cur_maxb
        grad_a = compute_grad_lp(a, p)
        grad_v = a * (grad_a - np.dot(a, grad_a))
        m = beta1 * m + (1.0 - beta1) * grad_v
        u = beta2 * u + (1.0 - beta2) * grad_v ** 2
        m_hat = m / (1.0 - beta1 ** t)
        u_hat = u / (1.0 - beta2 ** t)
        v = v - lr * m_hat / (np.sqrt(u_hat) + eps)

    a_final = softmax_ppm(v)
    b_final = fast_convolve(a_final)
    if np.sum(b_final ** p) < best_val:
        best_a = a_final
        best_val = np.sum(b_final ** p)
        best_maxb = b_final.max()
    return best_a, best_val, v


def coordinate_descent(a, num_sweeps=3, time_limit=None, t_start=None):
    """Coordinate descent: optimize one coordinate at a time on the simplex."""
    n = len(a)
    a = a.copy()
    if t_start is None:
        t_start = time.monotonic()

    for sweep in range(num_sweeps):
        if time_limit is not None and time.monotonic() - t_start > time_limit:
            break
        b = fast_convolve(a)
        for j in range(n):
            if a[j] < 1e-15:
                continue
            # Remove a[j], optimize it
            a_j = a[j]
            rest_sum = 1.0 - a_j
            if rest_sum <= 0:
                continue
            # b[k] = a[j]*a[k-j] + (contribution from rest)
            # We want to find a_j that minimizes max_k b[k]
            # b[k] is linear in a_j: b[k] = c_k + d_k * a_j
            # where c_k = b[k] - a_j*a[k-j] (for valid k-j) and d_k = a[k-j]
            # Actually b[k] = sum_m a[m]*a[k-m]. The terms involving a[j] are:
            #   m=j: a[j]*a[k-j]
            #   k-m=j, i.e., m=k-j: a[k-j]*a[j]
            # So b[k] = rest_k + a[j]*a[k-j] + a[k-j]*a[j] (if both valid)
            # = rest_k + 2*a[j]*a[k-j] (if 0<=k-j<n)
            # = rest_k + a[j]*a[k-j] (if only one term valid, which happens at boundaries)
            # For simplicity, use: b[k] ≈ b[k]_current - a_j*a[k-j] + new_a_j*a[k-j]
            # (approximately linear in a_j)
            # Try a few candidate values
            candidates = [a_j, a_j * 0.5, a_j * 1.5, a_j * 0.8, a_j * 1.2,
                          a_j + 0.002, a_j - 0.002, 1.0 / n]
            candidates = [max(0.0, min(1.0, c)) for c in candidates]
            cur_max = b.max()
            best_cand = a_j
            for cand in candidates:
                a[j] = cand
                # Renormalize
                s = a.sum()
                if s < 1e-15:
                    a[j] = a_j
                    break
                a = a / s
                b_test = fast_convolve(a)
                if b_test.max() < cur_max:
                    cur_max = b_test.max()
                    best_cand = cand
            a[j] = best_cand
            s = a.sum()
            if s > 1e-15:
                a = a / s
            b = fast_convolve(a)
    return a


def peak_equalization_newton(a, num_steps=50, time_limit=None, t_start=None):
    """Newton-like step to equalize the top peaks."""
    n = len(a)
    a = a.copy()
    if t_start is None:
        t_start = time.monotonic()
    b = fast_convolve(a)
    best_maxb = b.max()
    best_a = a.copy()

    for step in range(num_steps):
        if time_limit is not None and time.monotonic() - t_start > time_limit:
            break
        b = fast_convolve(a)
        cur_max = b.max()
        if cur_max < best_maxb:
            best_maxb = cur_max
            best_a = a.copy()

        # Find top peaks
        s = max(3, min(10, n // 20))
        peaks = np.argpartition(b, -s)[-s:]
        peak_vals = b[peaks]
        mean_peak = peak_vals.mean()

        # Build Jacobian: J[i, j] = db[peaks[i]]/da[j] = 2*a[peaks[i]-j] (valid)
        J = np.zeros((s, n))
        for i, k in enumerate(peaks):
            for j in range(n):
                diff = k - j
                if 0 <= diff < n:
                    J[i, j] = 2.0 * a[diff]

        # Target: reduce peaks to mean_peak
        # Solve J @ da = (mean_peak - peak_vals) + lambda * ones, with sum(da)=0
        # Use pseudoinverse with constraint
        target = mean_peak - peak_vals
        # Add normalization constraint
        J_aug = np.vstack([J, np.ones(s)])
        target_aug = np.append(target, 0.0)
        # Minimum norm solution
        try:
            da = np.linalg.lstsq(J_aug, target_aug, rcond=None)[0]
        except np.linalg.LinAlgError:
            break
        # Scale step
        step_size = 0.3 / (np.linalg.norm(da) + 1e-10)
        da = da * step_size
        a_new = a + da
        a_new = project_simplex(a_new)
        b_new = fast_convolve(a_new)
        if b_new.max() < cur_max:
            a = a_new
        else:
            # Try smaller step
            a_new = a - 0.1 * da
            a_new = project_simplex(a_new)
            b_new = fast_convolve(a_new)
            if b_new.max() < cur_max:
                a = a_new
            else:
                break

    return best_a


def fista_logparam(v0, p, num_iters, lr=None, time_limit=None, t_start=None):
    """FISTA on the log-parameterization for Lp objective."""
    n = len(v0)
    v = v0.copy().astype(np.float64)

    best_maxb = fast_convolve(best_a).max()

    if lr is None:
        # Estimate Lipschitz constant

        w0 = b0 ** (p - 1)
        g0 = compute_grad_lp(best_a, p)
        lr = 0.01 / (np.linalg.norm(g0) + 1e-10)
        lr = min(lr, 0.1)

    if t_start is None:

    for i in range(num_iters):
        if time_limit is not None and time.monotonic() - t_start > time_limit:
            break

        b = fast_convolve(a)

            best_val = obj
            best_a = a.copy()
            best_maxb = cur_maxb

        grad_a = compute_grad_lp(a, p)
        grad_v = a * (grad_a - np.dot(a, grad_a))

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista ** 2)) / 2.0
        y_v = v + (t_fista - 1.0) / t_new * (v - v_prev)
        a_y = softmax_ppm(y_v)
        b_y = fast_convolve(a_y)
        grad_a_y = compute_grad_lp(a_y, p)
        grad_v_y = a_y * (grad_a_y - np.dot(a_y, grad_a_y))

        v_prev = v.copy()
        v = y_v - lr * grad_v_y
        t_fista = t_new

    return best_a, best_maxb, v


def peak_weighted_gradient(a, num_iters=500, lr=0.001, time_limit=None, t_start=None):
    """Gradient step weighted by peak values (sharper focus on top peaks)."""
    n = len(a)
    a = a.copy()
    if t_start is None:
        t_start = time.monotonic()
    b = fast_convolve(a)
    best_maxb = b.max()
    best_a = a.copy()
    cur_lr = lr
    momentum = np.zeros(n)
    beta = 0.9

    for i in range(num_iters):
        if time_limit is not None and time.monotonic() - t_start > time_limit:
            break
        b = fast_convolve(a)
        cur_max = b.max()
        if cur_max < best_maxb:
            best_maxb = cur_max
            best_a = a.copy()

        # Weight peaks by exp(tau * (b[k] - max))
        tau = 50.0
        shifted = b - b.max()
        w = np.exp(tau * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient = 2 * sum_k w[k] * a[k-j] for each j (same as compute_grad but with weights)
        grad = np.zeros(n)
        fft_size = 1 << (3 * n).bit_length()
        A = np.fft.rfft(a, fft_size)
        w_rev = w[::-1]
        W = np.fft.rfft(w_rev, fft_size)
        conv = np.fft.irfft(A * W, fft_size)
        c = conv[2 * n - 2 : n - 2 : -1]
        grad = 2.0 * c

        # Remove mean to preserve normalization
        grad = grad - grad.mean()

        momentum = beta * momentum - cur_lr * grad
        a_new = project_simplex(a + momentum)
        b_new = fast_convolve(a_new)
        if b_new.max() < cur_max:
            a = a_new
        else:
            # Reject step, reduce lr
            cur_lr *= 0.95
            momentum = momentum * 0.5

    return best_a
