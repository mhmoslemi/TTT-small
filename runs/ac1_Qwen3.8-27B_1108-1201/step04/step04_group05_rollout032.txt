"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed gradient descent
with softmax parameterization, FISTA acceleration, adaptive temperature,
and multi-length exploration. Uses the key insight that for non-negative
sequences, the convolution structure allows efficient FFT-based gradients.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_correlate(a, w):
    """Compute c[m] = sum_j a[j] * w[j+m] for m = 0, ..., len(a)-1.
    Uses FFT: c = ifft(conj(FFT(a)) * FFT(w))[:len(a)]
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    c = np.fft.irfft(np.conj(A) * W, fft_size)
    return c[:n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_grad(a, T):
    """Compute the gradient of T*log(sum exp(b[k]/T)) w.r.t. a.
    
    Returns grad where grad[i] = 2 * sum_j a[j] * w[i+j]
    with w = softmax(b/T).
    """
    b = fast_convolve(a)
    # Softmax weights
    b_scaled = b / T
    b_scaled -= b_scaled.max()  # for numerical stability
    w = np.exp(b_scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # Fallback: uniform weights
        w = np.ones(len(b)) / len(b)
    else:
        w = w / w_sum
    # grad[i] = 2 * sum_j a[j] * w[i+j] = 2 * correlate(a, w)[i]
    grad = 2.0 * fast_correlate(a, w)
    return grad, b, w


def optimize_fgk(a0, time_limit, T_init=0.01, T_min=0.0001, lr_init=0.01,
                 momentum=0.9, seed=42, verbose=False):
    """FISTA-style accelerated gradient descent on the simplex.
    
    Minimizes max_k b[k] where b = convolve(a, a), over a >= 0, sum(a) = 1.
    Uses smoothed approximation with adaptive temperature.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    
    y = a.copy()  # FISTA momentum variable
    t_k = 1.0
    T = T_init
    lr = lr_init
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    no_improve = 0
    
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break
        
        # Compute gradient at y
        grad_y, b_y, w_y = smoothed_grad(y, T)
        
        # Gradient step
        y_new = project_simplex(y - lr * grad_y)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = y_new + ((t_k - 1.0) / t_new) * (y_new - a)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = y_new.copy()
            y_sum = 1.0
        y = y / y_sum
        a = y_new
        t_k = t_new
        
        # Evaluate
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt parameters
        if it % 100 == 0:
            T = max(T * 0.995, T_min)
            lr = max(lr * 0.999, 0.001)
        
        if verbose and it % 200 == 0:
            score = 2.0 * n * best_obj
            print(f"    FGK iter {it}: obj={best_obj:.10f}, score={score:.6f}, T={T:.6f}, lr={lr:.6f}")
        
        # Restart if stuck
        if no_improve > 200:
            # Try perturbation
            if rng.random() < 0.5:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            else:
                # Try a different T
                T = T_init * 0.5
                lr = lr_init * 0.5
            y = a.copy()
            t_k = 1.0
            no_improve = 0
    
    return best_a, best_obj


def optimize_softmax_param(a0, time_limit, T_init=0.01, T_min=0.00005, 
                            lr_init=0.05, momentum=0.9, seed=42, verbose=False):
    """Gradient descent on softmax parameterization z where a = softmax(z).
    
    This automatically maintains a >= 0 and sum(a) = 1.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    z = np.log(a)  # log-space parameterization
    
    v = np.zeros(n)  # momentum
    T = T_init
    lr = lr_init
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    no_improve = 0
    
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        # Compute gradient w.r.t. z
        grad_a, b, w = smoothed_grad(a, T)
        # Convert to grad_z: grad_z[i] = a[i] * (grad_a[i] - dot(grad_a, a))
        dot_ga_a = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot_ga_a)
        
        # Momentum update
        v = momentum * v - lr * grad_z
        z = z + v
        # Re-center z for numerical stability
        z = z - z.mean()
        
        # Evaluate
        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()
        obj = fast_convolve(a_new).max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            a = a_new  # still update a
        
        # Adapt
        if it % 200 == 0:
            T = max(T * 0.99, T_min)
            lr = max(lr * 0.999, 0.005)
        
        if verbose and it % 500 == 0:
            score = 2.0 * n * best_obj
            print(f"    Softmax iter {it}: obj={best_obj:.10f}, score={score:.6f}")
        
        if no_improve > 500:
            # Restart
            a_cur = np.exp(best_a if False else z - z.max())
            a_cur = a_cur / a_cur.sum()
            a_cur = project_simplex(a_cur + rng.standard_normal(n) * 0.003)
            z = np.log(np.maximum(a_cur, 1e-15))
            z = z - z.mean()
            v = np.zeros(n)
            T = max(T_init * 0.3, T_min * 10)
            lr = lr_init * 0.3
            no_improve = 0
    
    return best_a, best_obj


def optimize_direct(a0, time_limit, T_init=0.005, T_min=0.00001,
                    lr_init=0.005, seed=42, verbose=False):
    """Direct gradient descent on the simplex with adaptive step size.
    
    Uses backtracking line search for robustness.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    
    v = np.zeros(n)  # momentum
    T = T_init
    lr = lr_init
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    no_improve = 0
    
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)
    momentum = 0.9
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        grad, b, w = smoothed_grad(a, T)
        
        # Momentum
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        
        # Check improvement
        obj_new = fast_convolve(a_new).max()
        
        if obj_new < fast_convolve(a).max():
            a = a_new
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a.copy()
                no_improve = 0
        else:
            # Rejected step - reduce momentum
            v *= 0.5
            no_improve += 1
            if no_improve > 50:
                lr *= 0.9
                v = -lr * grad
        
        # Adapt
        if it % 300 == 0:
            T = max(T * 0.99, T_min)
        
        if verbose and it % 500 == 0:
            score = 2.0 * n * best_obj
            print(f"    Direct iter {it}: obj={best_obj:.10f}, score={score:.6f}")
        
        if no_improve > 500:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            T = max(T_init * 0.3, T_min * 5)
            lr = lr_init * 0.5
            no_improve = 0
    
    return best_a, best_obj


def pairwise_swap_polish(a0, time_limit, seed=42):
    """Local search via pairwise mass transfers on the simplex."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    no_improve = 0
    step_sizes = [0.01, 0.02, 0.05, 0.1, 0.2]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        # Try random pairs
        for _ in range(200):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            for frac in step_sizes:
                delta = a[j] * frac
                if delta < 1e-15 or a[i] < 1e-15:
                    continue
                # Move mass from j to i
                a_new = a.copy()
                a_new[i] += delta
                a_new[j] -= delta
                a_new = np.maximum(a_new, 0.0)
                s = a_new.sum()
                if s < 1e-15:
                    continue
                a_new = a_new / s
                obj = fast_convolve(a_new).max()
                if obj < best_obj - 1e-12:
                    a = a_new
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0
    
    return best_a, best_obj


def binomial_init(n, rng):
    """Initialize using the binomial coefficient pattern."""
    a = np.zeros(n)
    for j in range(n):
        a[j] = 1.0
        for k in range(2):
            pass
        # Use recurrence: a[j] = a[j-1] * (2j-1)/(2j)
        if j > 0:
            a[j] = a[j-1] * (2*j - 1) / (2*j)
    # Symmetrize
    a = (a + a[::-1]) / 2.0
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Binomial (symmetrized)
    inits.append(binomial_init(n, rng))
    
    # Binomial (asymmetric, left-heavy)
    a_bin = np.ones(n)
    for j in range(1, n):
        a_bin[j] = a_bin[j-1] * (2*j - 1) / (2*j)
    inits.append(a_bin / a_bin.sum())
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5*(n-1)) / (0.15*n))**2)
    inits.append(g / g.sum())
    
    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n-1-x) + 1.0)**0.5
    inits.append(edge_w / edge_w.sum())
    
    # Two-peak
    peak = (np.exp(-((x - n/4.0)**2) / (2.0*(n/10.0)**2)) +
            np.exp(-((x - 3.0*n/4.0)**2) / (2.0*(n/10.0)**2)))
    inits.append(peak / peak.sum())
    
    # Sine window
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    # Cosine
    cos_t = np.cos(np.pi * x / (2.0*(n-1)))
    cos_t = np.maximum(cos_t, 0.01)
    inits.append(cos_t / cos_t.sum())
    
    # Parent interpolation
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 1e-10)
        inits.append(px / px.sum())
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed gradient descent
with FISTA acceleration, adaptive temperature annealing, softmax parameterization,
and pairwise swap polishing. Key insight: for non-negative sequences, the
auto-convolution gradient has a clean FFT structure, enabling fast iterations.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_correlate(a, w):
    """Compute c[m] = sum_j a[j] * w[j+m] for m = 0, ..., len(a)-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rftp = None
    W = np.fft.rfft(w, fft_size)
    c = np.fft.irfft(np.conj(A) * W, fft_size)
    return c[:n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(a, T):
    """Compute gradient of T*log(sum exp(b[k]/T)) w.r.t. a.
    Returns (grad, b, w) where grad[i] = 2 * sum_j a[j] * w[i+j].
    """
    b = fast_convolve(a)
    b_scaled = b / T
    b_scaled -= b_scaled.max()
    w = np.exp(b_scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(b)) / len(b)
    else:
        w = w / w_sum
    grad = 2.0 * fast_correlate(a, w)
    return grad, b, w


def optimize_fista(a0, time_limit, T_init=0.01, T_min=0.00001,

    n = len(a0)

    y = a.copy()
    t_k = 1.0

    best_a = a.copy()

    momentum = 0.9

        y = y_next

def pairwise_swap_polish(a0, time_limit, seed=42):

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        n_trials = 300
        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit - 0.2:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:

            fracs = [0.02, 0.05, 0.1, 0.2, 0.4]

def binomial_init(n, rng):
    """Initialize using symmetrized binomial coefficient pattern."""
    a_left = np.ones(n)
    for j in range(1, n):
        a_left[j] = a_left[j - 1] * (2 * j - 1) / (2 * j)
    a = (a_left + a_left[::-1]) / 2.0
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)
    inits.append(binomial_init(n, rng))

    # Asymmetric binomial
    a_bin = np.ones(n)
    for j in range(1, n):
        a_bin[j] = a_bin[j - 1] * (2 * j - 1) / (2 * j)
    inits.append(a_bin / a_bin.sum())

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.12 * n)) ** 2)
    inits.append(g / g.sum())

    # Edge weight
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.8
    inits.append(edge_w / edge_w.sum())

    # Two-peak
    peak = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 12.0) ** 2)) +
            np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 12.0) ** 2)))
    inits.append(peak / peak.sum())

    # Sine
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())

    # Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.2))

    # Cosine taper
    cos_t = np.cos(np.pi * x / (2.0 * (n - 1)))
    cos_t = np.maximum(cos_t, 0.005)
    inits.append(cos_t / cos_t.sum())

    if parent is not None and len(parent) > 0:
        if len(parent) != n:
            px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        else:
            px = parent.copy()
        px = np.maximum(px, 1e-10)
        inits.append(px / px.sum())

    return inits

            f_init = project_simplex(state['best_seq'] + noise)
            a_opt, obj = optimize_fista(f_init, time_limit=ta,
                                         T_init=0.005, T_min=0.00001, lr_init=0.005,
                                         seed=seed + 4000 + int(ps * 10000))
            try_sequence(a_opt, f"pert_{ps}", state)

        # Mirror blend
        rem = phase2_end - time.monotonic() - 10
        if rem > 12:
            f_mir = project_simplex(0.55 * state['best_seq'] + 0.45 * state['best_seq'][::-1])
            a_opt, obj = optimize_fista(f_mir, time_limit=min(rem, 80),
                                         T_init=0.005, T_min=0.00001, lr_init=0.005,
                                         seed=seed + 5000)
            try_sequence(a_opt, "mirror", state)

    # === Phase 3: Explore other lengths (14%) ===
    print(f"\n=== Phase 3: Other Lengths ===")
    phase3_end = time.monotonic() + float(budget_s) * 0.14

    tried_n = len(state['best_seq']) if state['best_seq'] is not None else 0
    for n in [512, 768, 1024, 640]:
        if time.monotonic() > phase3_end - 15:
            break
        if n == tried_n:
            continue
        inits = generate_inits(n, parent, rng)
        for i in range(min(2, len(inits))):
            if time.monotonic() > phase3_end - 8:
                break
            ta = min(70, phase3_end - time.monotonic() - 5)
            if ta < 10:
                break
            a_opt, obj = optimize_fista(inits[i], time_limit=ta,
                                         T_init=0.01, T_min=0.0001, lr_init=0.01,
                                         seed=seed + 6000 + n * 7)
            try_sequence(a_opt, f"otherlen_n{n}_i{i}", state)

    # === Phase 4: Final polishing (18%) ===
    print(f"\n=== Phase 4: Final Polishing ===")
    phase4_end = deadline

    if state['best_seq'] is not None:
        print(f"  Current best: {state['best_score']:.6f}")

        # Low-T FISTA
        rem = phase4_end - time.monotonic() - 30
        if rem > 10:
            ta = min(rem * 0.3, 120)
            a_opt, obj = optimize_fista(state['best_seq'], time_limit=ta,
                                         T_init=0.001, T_min=0.000001, lr_init=0.002,
                                         seed=seed + 7000, verbose=True)
            try_sequence(a_opt, "FISTA_polish", state)

        # Block refine
        rem = phase4_end - time.monotonic() - 15
        if rem > 10:
            ta = min(rem * 0.3, 80)
            a_opt, obj = block_refine(state['best_seq'], time_limit=ta, seed=seed + 8000)
            try_sequence(a_opt, "block", state)

        # Pairwise swap
        rem = phase4_end - time.monotonic() - 8
        if rem > 8:
            ta = min(rem * 0.4, 80)
            a_opt, obj = pairwise_swap_polish(state['best_seq'], time_limit=ta,
                                               seed=seed + 8500)
            try_sequence(a_opt, "swap", state)

        # Final squeeze
        rem = phase4_end - time.monotonic() - 2
        if rem > 5:
            a_opt, obj = optimize_fista(state['best_seq'], time_limit=min(rem, 50),
                                         T_init=0.0005, T_min=0.0000005, lr_init=0.001,
                                         seed=seed + 9000)
            try_sequence(a_opt, "final", state)

    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(256) / 256
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))
        print(f"  Fallback uniform: {state['best_score']:.6f}")

    # Verify
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs
    print(f"\nFINAL SCORE: {state['best_score']:.6f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
