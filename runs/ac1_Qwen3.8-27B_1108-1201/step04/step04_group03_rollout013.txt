import numpy as np

def fast_autoconv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_cross_corr(p, a):
    """Compute c_m = sum_k p_k * a_{k-m} for m=0..n-1 using FFT cross-correlation."""
    n = len(a)
    L = 1 << (len(p) + n - 2).bit_length()
    P = np.fft.fft(p, L)
    A = np.fft.fft(a, L)
    C = np.fft.ifft(P * np.conj(A))[:n].real
    return C


def lse_and_grad(w, tau):
    """Compute log-sum-exp objective and gradient w.r.t. w (softmax parameterization)."""
    n = len(w)
    w_shift = w - w.max()
    exp_w = np.exp(w_shift)
    a = exp_w / exp_w.sum()

    conv = fast_autoconv(a)

    tau_conv = tau * (conv - conv.max())
    exp_tc = np.exp(tau_conv)
    exp_sum = exp_tc.sum()
    p = exp_tc / exp_sum
    obj = conv.max() + np.log(exp_sum) / tau

    s = np.dot(p, conv)
    c = compute_cross_corr(p, a)
    grad = 2.0 * a * ((1.0 - a) * c - s)

    return obj, grad

def fista_simplex(a0, time_limit, tau_smooth=50.0):
    """FISTA (accelerated proximal gradient) on the simplex with smooth peak approximation."""
    a = normalization(a0)
    n = len(a)

    t0 = time.monotonic()
    a_prev = a.copy()

    no_improve = 0

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = a + (t_mom - 1.0) / t_new * (a - a_prev)

        if ys < 1e-15:

        y = y / ys

        conv = fast_autoconv(y)
        tau_c = tau_smooth * (conv - conv.max())
        exp_tc = np.exp(tau_c)
        p = exp_tc / exp_tc.sum()
        c = compute_cross_corr(p, y)
        grad = 2.0 * c

        # Adaptive step size (backtracking-lite)
        a_new = project_simplex(y - alpha * grad)

        # Check objective
        obj = fast_autoconv(a_new).max()
        if obj < best_obj - 1e-15:

            no_improve = 0

            alpha *= 0.95

                a_prev = a.copy()
                t_mom = 1.0
                alpha = 1.0 / (4.0 * n)
                no_improve = 0

        a = a_new
        t_mom = t_new

def generate_initializations(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    # Uniform
    inits.append(np.ones(n) / n)

    # Parent (resized)
    if parent is not None and len(parent) > 0:

        def resize_func(t):
            idx = t * (len(parent) - 1)
            lo = int(np.floor(idx))
            hi = min(lo + 1, len(parent) - 1)
            frac = idx - lo
            return parent[lo] * (1 - frac) + parent[hi] * frac

        x = np.linspace(0, 1, n)
        p_resized = np.array([resize_func(t) for t in x])
        inits.append(normalization(p_resized))

    # Hann window
    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    inits.append(normalization(hann))

    # Raised cosine (different shape)
    rc = np.cos(np.pi * (x - 0.5)) ** 2
    inits.append(normalization(rc))

    # Gaussian centered
    mu = n / 2
    sigma = n / 4
    g = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
    inits.append(normalization(g))

    # Edge-weighted
    d = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float) + 1
    inits.append(normalization(1.0 / d))

    # Random Dirichlet (several)
    for alpha_val in [0.5, 1.0, 2.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha_val))

    # Random smooth (low-pass filtered random)
    for _ in range(3):
        r = rng.standard_normal(n)
        # Smooth via repeated averaging
        for _ in range(3):
            r = (r + np.roll(r, 1)) / 2
        inits.append(normalization(r))

    return inits

            print(f"  L-BFGS on parent (n={n_main}), t={t_alloc:.0f}s")
            a_opt, _ = run_continuation_lse(parent, tau_start=3.0, tau_end=8000.0,
                                            tau_mult=1.4, max_iter_stage=200, time_limit=t_alloc)
            try_seq(a_opt, "lbfgs_parent")

    # Try a few other inits quickly
    inits = generate_initializations(n_main, parent, rng)
    for i, init in enumerate(inits[1:6], 1):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.08, 50.0)
        if t_alloc < 10:
            break
        a_opt, _ = run_continuation_lse(init, tau_start=5.0, tau_end=3000.0,
                                        tau_mult=1.5, max_iter_stage=150, time_limit=t_alloc)
        try_seq(a_opt, f"init_{i}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Deep L-BFGS with restarts ===
    print("\n=== Phase 2: Deep L-BFGS with Restarts ===")

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()

        # Multiple L-BFGS runs with different tau schedules
        configs = [
            {'tau_start': 2.0, 'tau_end': 15000.0, 'tau_mult': 1.3, 'max_iter': 400},
            {'tau_start': 5.0, 'tau_end': 10000.0, 'tau_mult': 1.4, 'max_iter': 300},
            {'tau_start': 1.0, 'tau_end': 20000.0, 'tau_mult': 1.25, 'max_iter': 500},
            {'tau_start': 10.0, 'tau_end': 5000.0, 'tau_mult': 1.5, 'max_iter': 250},
        ]

        t_per = max(20.0, rem * 0.30 / len(configs))

        for ci, cfg in enumerate(configs):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(t_per, rem * 0.25)
            if t_alloc < 15:
                break
            print(f"  L-BFGS config {ci} (tau {cfg['tau_start']}->{cfg['tau_end']}), t={t_alloc:.0f}s")
            a_opt, _ = run_continuation_lse(
                best_seq, tau_start=cfg['tau_start'], tau_end=cfg['tau_end'],
                tau_mult=cfg['tau_mult'], max_iter_stage=cfg['max_iter'], time_limit=t_alloc)
            try_seq(a_opt, f"lbfgs_deep_{ci}")

        # Perturbed restarts
        for ri in range(4):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            scale = [0.01, 0.005, 0.02, 0.003][ri]
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem * 0.12, 60.0)
            if t_alloc < 15:
                break
            print(f"  Restart {ri} (scale={scale}), t={t_alloc:.0f}s")
            a_opt, _ = run_continuation_lse(
                a_pert, tau_start=5.0, tau_end=10000.0,
                tau_mult=1.4, max_iter_stage=250, time_limit=t_alloc)
            try_seq(a_opt, f"restart_{ri}")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Try other sizes ===
    print("\n=== Phase 3: Alternative Sizes ===")

    for n_alt in [512, 256]:
        if len(best_seq) == n_alt:
            continue
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.10, 80.0)
        if t_alloc < 20:
            break
        print(f"  Trying n={n_alt}, t={t_alloc:.0f}s")
        # Resize best to new size
        x = np.linspace(0, 1, n_alt)
        a_resized = np.interp(x, np.linspace(0, 1, len(best_seq)), best_seq)
        a_resized = normalization(a_resized)
        a_opt, _ = run_continuation_lse(
            a_resized, tau_start=3.0, tau_end=8000.0,
            tau_mult=1.4, max_iter_stage=250, time_limit=t_alloc)
        try_seq(a_opt, f"size_{n_alt}")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: FISTA Polish ===
    print("\n=== Phase 4: FISTA Polish ===")

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 30:
            t_alloc = min(rem * 0.15, 120.0)
            print(f"  FISTA polish, t={t_alloc:.0f}s")
            a_opt, _ = fista_simplex(best_seq, time_limit=t_alloc, tau_smooth=80.0)
            try_seq(a_opt, "fista_polish")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Fine perturbation search ===
    print("\n=== Phase 5: Fine Perturbation Search ===")

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()

        # Many small perturbations + quick L-BFGS
        scales = [0.002, 0.001, 0.003, 0.0015, 0.0005, 0.004, 0.0008, 0.0025]
        for pi, scale in enumerate(scales):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_alloc = min(rem * 0.10, 30.0)
            if t_alloc < 10:
                break
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(best_seq + noise)
            a_opt, _ = run_continuation_lse(
                a_pert, tau_start=10.0, tau_end=10000.0,
                tau_mult=1.5, max_iter_stage=100, time_limit=t_alloc)
            try_seq(a_opt, f"fine_{pi}")

        # Also try FISTA with different tau_smooth
        for ts in [30.0, 200.0]:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t_alloc = min(rem * 0.08, 40.0)
            if t_alloc < 10:
                break
            a_opt, _ = fista_simplex(best_seq, time_limit=t_alloc, tau_smooth=ts)
            try_seq(a_opt, f"fista_ts{ts}")

    print(f"  Phase 5 best: {best_score:.8f}")

    # === Phase 6: Final L-BFGS with very high tau ===
    print("\n=== Phase 6: Final High-Tau L-BFGS ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t_alloc = min(rem * 0.5, 120.0)
            print(f"  Final L-BFGS (high tau), t={t_alloc:.0f}s")
            a_opt, _ = run_continuation_lse(
                best_seq, tau_start=20.0, tau_end=50000.0,
                tau_mult=1.3, max_iter_stage=500, time_limit=t_alloc)
            try_seq(a_opt, "final_lbfgs")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
