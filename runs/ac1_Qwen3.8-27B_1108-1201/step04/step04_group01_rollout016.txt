def solve_lp(f, M, A_cache):
    n = len(f)

                      method='highs', options={'presolve': True, 'time_limit': 20})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def adam_lse_optimize(f0, time_limit, beta_start=10.0, beta_end=5000.0, lr_init=0.01):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_init
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_lse_grad(w, f)
        
        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            f = f_new  # still take the step (Adam handles this)
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.95
            if no_improve > 100:
                # Restart from best
                f = best_f.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                lr = lr_init * max(0.1, 1.0 - no_improve / 500.0)
                t = max(t, 100)
                no_improve = 0

        # Ramp beta

            beta *= 1.005

        if t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam-LSE t={t}, obj={best_obj:.12f}, beta={beta:.1f}, lr={lr:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def penalty_refine(f0, time_limit):

    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    # Start threshold slightly above current max
    T = best_obj * 0.95
    penalty = 10.0
    lr = 0.01
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)

        v = b2 * v + (1 - b2) * grad ** 2
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. AFW (Away-Frank-Wolfe) with LP oracle for global structure optimization
2. Adam + Log-Sum-Exp exponential gradient descent for smooth local polishing
3. Direct max-penalty refinement with adaptive threshold
4. Second Adam-LSE pass and coordinate descent for final convergence
Multi-phase with dynamic time allocation and restart mechanisms.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - j] for j in js))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def compute_lse_grad(weights, f):
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def solve_lp_direction(f, M):
    n = len(f)
    A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 25})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

        v = b2 * v + (1 - b2) * (grad * grad)
        m_hat = m / (1.0 - b1 ** t)
