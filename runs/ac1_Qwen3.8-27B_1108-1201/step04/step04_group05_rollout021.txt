Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA (accelerated projected subgradient) on the probability simplex
2. Smoothed BFGS optimization in softmax parameterization with log-sum-exp
3. Multi-length exploration and adaptive T scheduling

import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    s = exp_x.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return exp_x / s

def compute_smoothed_obj_and_grad(x, T):
    """
    Compute objective and gradient of 2*n*logsumexp_T(conv(softmax(x)))
    where logsumexp_T(v) = max(v) + log(sum(exp(T*(v - max(v)))))/T
    """
    a = softmax(x)
    n = len(a)
    conv = fast_convolve(a)
    
    # Log-sum-exp with temperature T
    conv_max = conv.max()
    scaled = T * (conv - conv_max)
    # Numerical stability: subtract max (which is 0 since we subtracted conv_max)
    max_scaled = scaled.max()  # should be 0
    exp_scaled = np.exp(scaled - max_scaled)
    lse = conv_max + (max_scaled + np.log(exp_scaled.sum())) / T
    
    # Weights for gradient
    weights = exp_scaled / exp_scaled.sum()
    
    # Objective
    obj = 2.0 * n * lse
    
    # Gradient of lse w.r.t. a:
    # grad_a[m] = 2 * sum_k weights[k] * a[k-m]
    # = 2 * convolve(weights, a[::-1], 'full')[n-1 : 2n-1]
    grad_a = 2.0 * np.convolve(weights, a[::-1], 'full')[n-1:2*n-1]
    
    # Chain rule through softmax:
    # d(obj)/dx_j = sum_i d(obj)/da_i * da_i/dx_j
    # da_i/dx_j = a_i * (delta_ij - a_j)
    # d(obj)/da_i = 2*n * grad_a_i  (since obj = 2*n*lse and d(lse)/da = grad_a)
    # Wait, I already included the 2 in grad_a. Let me re-check.
    # 
    # lse = conv_max + log(sum(exp(T*(conv-conv_max))))/T
    # d(lse)/d(conv_k) = weights[k]
    # d(conv_k)/da_m = 2*a[k-m] (as derived earlier)
    # d(lse)/da_m = sum_k weights[k] * 2*a[k-m] = grad_a[m] (as computed)
    # obj = 2*n*lse
    # d(obj)/da_m = 2*n*grad_a[m]
    # d(obj)/dx_j = sum_m 2*n*grad_a[m] * a_m * (delta_mj - a_j)
    # = 2*n * (a_j * grad_a[j] - a_j * sum_m grad_a[m]*a_m)
    
    grad_x = 2.0 * n * (a * grad_a - a * np.dot(grad_a, a))
    
    return obj, grad_x

def fista_optimize(f0, time_limit, verbose=False):
    """
    FISTA (accelerated projected subgradient) on the probability simplex.
    Minimizes max_k conv_k(a) subject to a >= 0, sum(a) = 1.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    t0 = time.monotonic()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    x = f.copy()
    y = f.copy()
    t = 1.0
    step = 1.0 / (2.0 * n)  # Initial step size
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        M = conv_y.max()
        
        # Find peaks (top few)
        top_k = min(50, len(conv_y))
        peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        # Only use peaks within 1e-6 of max (or all top_k)
        peak_mask = conv_y[peaks] >= M - 1e-8 * max(M, 1e-10)
        active_peaks = peaks[peak_mask]
        if len(active_peaks) == 0:
            active_peaks = [np.argmax(conv_y)]
        
        # Subgradient: sg[m] = 2 * sum over active peaks k of y[k-m]
        sg = np.zeros(n)
        for k in active_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je+1] += 2.0 * y[je:js-1:-1]
        
        # Gradient step
        x_new = project_simplex(y - step * sg)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = np.maximum(y_new, 0.0)
        y_new = y_new / max(y_new.sum(), 1e-30)
        
        x = x_new
        y = y_new
        t = t_new
        
        # Evaluate
        conv_x = fast_convolve(x)
        obj = conv_x.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = x.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    FISTA iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        # Adaptive step size
        if no_improve > 100:
            step *= 0.8
            no_improve = 0
            # Reset momentum
            y = x.copy()
            t = 1.0
        
        if no_improve > 500:
            break
    
    return best_f, best_obj

def smoothed_bfgs(x0, T_start, T_end, time_limit, max_iter_per_T=100, verbose=False):
    """
    Smoothed BFGS optimization in softmax parameterization.
    Gradually increases T from T_start to T_end.
    """
    n = len(x0)
    x = x0.copy()
    t0 = time.monotonic()
    
    # Generate a schedule of T values
    n_levels = int(np.log(T_end / T_start) / np.log(3)) + 1
    T_values = T_start * (3.0 ** np.arange(n_levels))
    T_values = T_values[T_values <= T_end]
    T_values = np.append(T_values, T_end)
    
    for T in T_values:
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        deadline_T = time.monotonic() + min(60, (time_limit - (time.monotonic() - t0)) / max(1, n_levels - 1))
        
        best_T_obj = float('inf')
        best_T_x = x.copy()
        
        Try running BFGS at this T:
        for attempt in range(max_iter_per_T):
            if time.monotonic() > deadline_T:
                break
            
            obj, grad = compute_smoothed_obj_and_grad(x, T)
            
            res = minimize(
                lambda xx: compute_smoothed_obj_and_grad(xx, T)[0],
                x,
                jac=lambda xx: compute_smoothed_obj_and_grad(xx, T)[1],
                method='BFGS',
                options={'maxiter': 50, 'gtol': 1e-6, 'gtol_2': 1e-6}
            )
            
            if res.fun < best_T_obj:
