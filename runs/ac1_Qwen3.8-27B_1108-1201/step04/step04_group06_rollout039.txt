def fast_convolve(a):
    """Linear auto-convolution via FFT."""

        j_end = min(n - 1, l)

    if result.success:

        t_opt = result.x[n]

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Successive Linear Programming (SLP).
The key insight is that the non-convex problem can be solved via a sequence of LPs
by linearizing the quadratic autocorrelation constraints at the current iterate.
The linearization provides a lower bound, ensuring that the SLP makes progress.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_slp_matrix(a_k):
    """Build the LP constraint matrix for the SLP linearization.
    Returns A (shape (2n-1, n+1)) such that A @ [a; t] <= c_k.
    """
    n = len(a_k)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n + 1))
    for l in range(num_rows):
        j_start = max(0, l - n + 1)
        j_end = min(n - 1, l)
        js = np.arange(j_start, j_end + 1)
        A[l, js] = 2.0 * a_k[l - js]
        A[l, n] = -1.0
    return A


def solve_slp_lp(a_k, c_k, A_cache=None):
    """Solve the SLP LP: min t s.t. A@a <= c_k, sum(a)=1, a>=0."""
    n = len(a_k)
    if A_cache is None:
        A_cache = build_slp_matrix(a_k)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    try:
        result = linprog(c_obj, A_ub=A_cache, b_ub=c_k, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success:
            return result.x[:n], result.x[n], True
    except Exception as e:
        pass
    return None, None, False

    n = len(a_k)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n + 1))
    A[:, n] = -1.0
    
    # conv(a_k, a)[l] = sum_j a_k[l-j] * a[j]
    # This is a Toeplitz matrix applied to a
    # Row l of the Toeplitz matrix has a_k[l], a_k[l-1], ..., a_k[l-n+1] (padded with 0s)
    
    # Using scipy's toeplitz or manually:
    # For l < n: row l is [0, ..., 0, a_k[l], a_k[l-1], ..., a_k[0], 0, ..., 0]
    #   (l+1 entries from a_k[l] down to a_k[0], starting at position 0)
    # For l >= n: row l is [a_k[l-n+1], a_k[l-n+2], ..., a_k[n-1], 0, ..., 0]
    #   (n entries, starting at position 0)
    
    # Actually, let me just use the direct formula:
    # A[l, j] = 2 * a_k[l-j] if 0 <= l-j < n, else 0
    
    for j in range(n):
        # a_k[l-j] is non-zero when 0 <= l-j < n, i.e., j <= l < j+n
        # So for column j, the non-zero rows are l = j, j+1, ..., j+n-1
        rows = np.arange(j, min(j + n, num_rows))
        A[rows, j] = 2.0 * a_k[rows - j]
    
    return A

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Successive Linear Programming (SLP).
Key insight: linearize the quadratic autocorrelation constraints at the current iterate
to get a valid LP (since the linearization is a lower bound on the true quadratic).
This enables fast, reliable optimization via HiGHS LP solver.
Additional strategies: multi-init diversification, asymmetric initialization (ramps),
extension to larger n, and restart-based local optimum escaping.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_slp_matrix(a_k):
    """Build SLP constraint matrix A such that A @ [a; t] <= c_k enforces
    2*(conv(a_k, a))[l] - t <= c_k[l] for all l."""
    n = len(a_k)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n + 1))
    A[:, n] = -1.0
    for j in range(n):
        rows_start = j
        rows_end = min(j + n, num_rows)
        rows = np.arange(rows_start, rows_end)
        A[rows, j] = 2.0 * a_k[rows - j]
    return A


def solve_slp_lp(a_k, c_k, A=None):
    """Solve the SLP LP: min t s.t. A@[a;t] <= c_k, sum(a)=1, a>=0."""
    n = len(a_k)
    if A is None:
        A = build_slp_matrix(a_k)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    try:
        result = linprog(c_obj, A_ub=A, b_ub=c_k, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success:
            return result.x[:n], result.x[n], True
    except Exception:
        pass
    return None, None, False


def slp_optimize(a0, time_limit, max_iter=10000, verbose=False):
    """SLP optimization with line search and adaptive restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = float(np.max(c))
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    restart_count = 0
    rng_local = np.random.default_rng(42 + n)

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        A = build_slp_matrix(a)
        a_lp, t_lp, success = solve_slp_lp(a, c, A)

        if not success:
            no_improve += 3
            if no_improve > 30:
                break
            continue

        a_lp = np.maximum(a_lp, 0.0)
        s_lp = a_lp.sum()
        if s_lp < 1e-15:
            no_improve += 3
            if no_improve > 30:
                break
            continue
        a_lp = a_lp / s_lp

        # Line search with damping
        improved = False
        for damp in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_try = (1.0 - damp) * a + damp * a_lp
            c_try = fast_convolve(a_try)
            obj_try = float(np.max(c_try))
            if obj_try < best_obj - 1e-15:
                a = a_try
                c = c_try
                best_obj = obj_try
                best_a = a_try.copy()
                no_improve = 0
                improved = True
                break

        if not improved:
            no_improve += 1

        if no_improve > 80:
            # Restart with perturbation
            restart_count += 1
            if restart_count > 30:
                break
            scale = 0.002 + 0.003 * min(restart_count, 10)
            noise = rng_local.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a < 1e-15:
                a = best_a.copy()
            else:
                a = a / s_a
            c = fast_convolve(a)
            no_improve = 0

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SLP iter {it}: obj={best_obj:.10f}, t={elapsed:.1f}s")

    return best_a, best_obj


def gradient_polish(a0, time_limit, lr_init=0.005, momentum=0.92, top_k=50, temp=0.002):
    """Gradient-based polishing using softmax-weighted peak gradient."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = float(np.max(c))
    best_a = a.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Compute gradient of softmax of top peaks
        peaks = np.argpartition(c, -top_k)[-top_k:]
        peak_vals = c[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-10:
                continue
            k = int(peaks[idx])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idxs = np.arange(j_start, j_end + 1)
            grad[idxs] += w[idx] * a[k - idxs]

        # Nesterov update
        y = a + momentum * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y

        c_y = fast_convolve(y)
        peaks_y = np.argpartition(c_y, -top_k)[-top_k:]
        pv_y = c_y[peaks_y]
        sc_y = pv_y / temp
        sc_y -= sc_y.max()
        w_y = np.exp(sc_y)
        ws_y = w_y.sum()
        if ws_y < 1e-15:
            w_y = np.zeros(len(peaks_y))
            w_y[np.argmax(pv_y)] = 1.0
        else:
            w_y /= ws_y

        grad_y = np.zeros(n)
        for idx in range(len(peaks_y)):
            if w_y[idx] < 1e-10:
                continue
            k = int(peaks_y[idx])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idxs = np.arange(j_start, j_end + 1)
            grad_y[idxs] += w_y[idx] * y[k - idxs]

        v = momentum * v - lr * grad_y
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        c = fast_convolve(a)
        cur_obj = float(np.max(c))
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.999
        if no_improve > 200:
            break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations emphasizing asymmetric structures."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Asymmetric ramps (key: break palindromic symmetry)
    ramp_d = np.arange(n, 0, -1, dtype=np.float64)
    inits['ramp_down'] = ramp_d / ramp_d.sum()

    ramp_u = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp_up'] = ramp_u / ramp_u.sum()

    # Exponential (asymmetric)
    ed = np.exp(-x / (n * 0.25))
    inits['exp_decay'] = ed / ed.sum()
    eu = np.exp(x / (n * 0.25))
    inits['exp_rise'] = eu / eu.sum()

    # Quadratic ramp
    q = (x / n) ** 2 + 0.01
    inits['quad_ramp'] = q / q.sum()

    # Cubic
    cu = (x / n) ** 3 + 0.01
    inits['cubic'] = cu / cu.sum()

    # Asymmetric V (minimum off-center)
    vmin = 0.3
    vv = np.abs(x / n - vmin) + 0.01
    inits['v_asym'] = vv / vv.sum()

    # Power law variations
    p1 = (x + 1.0) ** 0.5
    inits['sqrt_ramp'] = p1 / p1.sum()

    p2 = (x + 1.0) ** (-0.5)
    inits['inv_sqrt'] = p2 / p2.sum()

    # Sine-based asymmetric
    s1 = np.sin(np.pi * x / (n * 0.7)) ** 2 + 0.01
    inits['sine_asym'] = s1 / s1.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits
