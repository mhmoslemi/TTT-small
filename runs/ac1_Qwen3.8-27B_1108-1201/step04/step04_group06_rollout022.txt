    """Find g with sum(g)=1, g>=0 that minimizes max_k (a*g)[k]."""
    n = len(a)
    C = build_conv_matrix(a)  # (2n-1) x n matrix
    
    # Minimize t s.t. C@g - t*1 <= 0, sum(g) = 1, g >= 0
    # Variables: g[0..n-1], t (total n+1 variables)
    # Objective: minimize t
    c = np.zeros(n+1)
    c[n] = 1  # minimize t
    
    # Inequalities: C@g - t <= 0, i.e., [C, -1] @ [g, t] <= 0
    A_ub = np.hstack([C, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1
    b_eq = [1]
    
    bounds = [(0, None)]*n + [(None, None)]  # g >= 0, t free
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if result.success:
        g = result.x[:n]
        t_opt = result.x[n]

    return None, None

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using asymmetric initializations
(linear/exponential ramps), LP-based direction finding (minimize peak of
a*g over unit simplex), subgradient polishing, and multi-length search.
Key insight: asymmetric sequences have lower peak autocorrelation than

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def compute_obj(a):
    """Compute full objective 2*n*max_b/sum_a^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:

    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:

    return a / s

                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
                # Also try reversed parent
                pr = pp[::-1]
                prs = pr.sum()
                if prs > 1e-10:
                    inits['parent_rev'] = pr / prs

    return inits


def lp_direction(a, C=None):
    """
    Solve LP: min t s.t. C_a @ g <= t, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None).
    """
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)

    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t

    # Inequality: C@g - t <= 0  =>  [C, -1] @ [g, t] <= 0
    A_ub = np.hstack([C, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if result.success:
            g = result.x[:n]
            t_opt = result.x[n]
            if g.sum() > 1e-10:
                return g / g.sum(), t_opt
    except Exception:
        pass
    return None, None


def lp_max_sum_direction(a, M, C=None):
    """
    Solve LP: max sum(g) s.t. C_a @ g <= M, g >= 0.
    Returns (g, g_sum, success).
    """
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)

    c = -np.ones(n)
    A_ub = C
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return None, 0.0, False


def subgradient_step(a, conv, n, top_k=10, rng=None):
    """
    Compute subgradient of max_k conv[k] and take a step.
    Returns new a.
    """
    # Find peaks
    if top_k < len(conv):
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    else:
        peaks = np.arange(len(conv))

    peak_vals = conv[peaks]
    max_val = peak_vals.max()

    # Softmax weights over peaks
    temp = 0.01 * max_val + 1e-10
    scaled = (peak_vals - max_val) / temp
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    # Compute gradient: grad[j] = sum_k w_k * 2 * a[k-j] (where valid)
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-10:
            continue
        k = int(peaks[idx])
        # grad[j] += 2 * w[idx] * a[k-j] for j where 0 <= k-j < n
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        a_idx = np.arange(k - j_end, k - j_start + 1)
        grad[idxs] += 2.0 * w[idx] * a[a_idx]

    return grad


def gradient_optimize(a0, time_limit, lr_init=0.005, momentum=0.9, top_k=20):
    """Nesterov-accelerated gradient descent on the objective."""
    a = normalize(a0.astype(np.float64))
    n = len(a)

    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = compute_obj(a)
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + mom * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy

        conv_y = fast_convolve(y)
        grad = subgradient_step(y, conv_y, n, top_k=top_k)

        # Gradient step
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn > 1e-15:
            a = a_new / sn
        else:
            a = best_a.copy()

        conv = fast_convolve(a)
        cur_obj = compute_obj(a)

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive updates
        lr *= 0.999
        mom = min(0.97, mom + 0.0001)

        if no_improve >= 300:
            break

    return best_a, best_obj


def lp_optimize(a0, time_limit, verbose=False):
    """
    LP-based iterative optimization.
    Alternates between two LP formulations and uses line search.
    """
    a = normalize(a0.astype(np.float64))
    n = len(a)

    C = build_conv_matrix(a)
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = compute_obj(a)
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    perturb_count = 0
    use_min_peak_lp = (n <= 3000)  # Use min-peak LP for reasonable n

    while time.monotonic() - t0 < time_limit:
        improved = False

        if use_min_peak_lp and iteration % 2 == 0:
            # LP 1: min t s.t. C_a @ g <= t, sum(g)=1, g>=0
            g, t_opt = lp_direction(a, C)
            if g is not None and t_opt < M * 0.999:
                # Direction is g - a (both have sum 1)
                d = g - a
                norm_d = np.linalg.norm(d)
                if norm_d > 1e-12:
                    # Try step sizes
                    for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
                        a_try = a + t * d
                        a_try = np.maximum(a_try, 0.0)
                        st = a_try.sum()
                        if st < 1e-15:
                            continue
                        a_try = a_try / st
                        conv_try = fast_convolve(a_try)
                        obj_try = 2.0 * n * conv_try.max()
                        if obj_try < best_obj * a.sum()**2 - 1e-10:
                            # Compare properly
                            pass
                        if obj_try < best_obj - 1e-12:
                            a = a_try
                            C = build_conv_matrix(a)
                            conv = conv_try
                            M = conv.max()
                            best_obj = obj_try
                            best_a = a.copy()
                            improved = True
                            break
        else:
            # LP 2: max sum(g) s.t. C_a @ g <= M, g >= 0
            g, g_sum, success = lp_max_sum_direction(a, M * 1.0001, C)
            if success and g_sum > 1.0 + 1e-8:
                d = g / g_sum - a  # normalize g to sum=1, then direction
                norm_d = np.linalg.norm(d)
                if norm_d > 1e-12:
                    for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
                        a_try = a + t * d
                        a_try = np.maximum(a_try, 0.0)
                        st = a_try.sum()
                        if st < 1e-15:
                            continue
                        a_try = a_try / st
                        conv_try = fast_convolve(a_try)
                        obj_try = 2.0 * n * conv_try.max()
                        if obj_try < best_obj - 1e-12:
                            a = a_try
                            C = build_conv_matrix(a)
                            conv = conv_try
                            M = conv.max()
                            best_obj = obj_try
                            best_a = a.copy()
                            improved = True
                            break

        if improved:
            no_improve = 0
        else:
            no_improve += 1

        iteration += 1

        if verbose and iteration % 50 == 0:
            print(f"      LP iter {iteration}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")

        # Stagnation handling
        if no_improve >= 60:
            rng = np.random.default_rng(iteration * 31 + perturb_count * 7)
            scale = 0.002 + 0.004 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = normalize(a)
            C = build_conv_matrix(a)
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1

            # Also try a gradient burst on perturbation
            if perturb_count % 3 == 0:
                a_g, obj_g = gradient_optimize(a, time_limit=1.0, lr_init=0.003, top_k=30)
                if obj_g < best_obj:
                    best_obj = obj_g
                    best_a = a_g.copy()
                    a = a_g.copy()
                    C = build_conv_matrix(a)
                    conv = fast_convolve(a)
                    M = conv.max()

            if perturb_count > 50:
                break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, n_trials=100000):
    """
    Coordinate descent: try moving mass between pairs to reduce peak.
    Uses the peak lag to guide which positions to modify.
    """
    a = normalize(a0.astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    best_obj = compute_obj(a)
    best_a = a.copy()
    peak_idx = int(np.argmax(conv))

    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0

    # Find positions that contribute most to the peak
    j_start = max(0, peak_idx - n + 1)
    j_end = min(n - 1, peak_idx)
    peak_positions = np.arange(j_start, j_end + 1)
    # Weight by contribution
    contrib = a[peak_positions] * a[peak_idx - peak_positions]
    top_contrib = peak_positions[np.argsort(contrib)[-n // 4:]] if n >= 16 else peak_positions

    while time.monotonic() - t0 < time_limit:
        batch = min(n_trials // 10, 5000)
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit - 0.1:
                break

            # Choose i from peak-contributing positions, j from elsewhere
            if len(top_contrib) > 1:
                i = top_contrib[rng.integers(0, len(top_contrib))]
            else:
                i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            # Try small transfer
            delta = min(a[i], a[j]) * 0.05 * (1 if rng.random() < 0.5 else -1)
            if a[i] + delta < 0 or a[j] - delta < 0:
                continue

            a_try = a.copy()
            a_try[i] += delta
            a_try[j] -= delta
            a_try = np.maximum(a_try, 0.0)
            st = a_try.sum()
            if st < 1e-15:
                continue
            a_try = a_try / st

            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()

            if obj_try < best_obj - 1e-11:
                a = a_try
                best_obj = obj_try
                best_a = a.copy()
                conv = conv_try
                peak_idx = int(np.argmax(conv))
                no_improve = 0
            else:
                no_improve += 1

        if no_improve > 5000:
            break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search with asymmetric initializations,
    LP-based optimization, and gradient polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    # =========================================================
    # Phase 1: Quick scan across lengths with asymmetric inits
    # =========================================================
    print("=== Phase 1: Quick Scan ===")
    t_start = time.monotonic()
    phase1_limit = 250.0

    lengths = [128, 256, 384, 512, 768, 1024, 1536, 2048]
    # Filter based on parent length if available
    if parent is not None:
        plen = len(parent)
        # Prioritize lengths near parent
        lengths = [n for n in lengths if abs(n - plen) <= plen * 0.5] or lengths

    scan_results = []
    inits_priority = ['ramp', 'exp_ramp', 'ramp2', 'parent_interp', 'parent', 'uniform', 'random', 'sine1']

    for n in lengths:
        if time.monotonic() > deadline - 900:
            break
        inits = generate_inits(n, parent, rng)
        t_per = max(5.0, (phase1_limit - (time.monotonic() - t_start)) / (len(lengths) * 3))

        for name in inits_priority[:4]:
            if time.monotonic() > deadline - 850:
                break
            if name not in inits:
                continue
            t_alloc = min(t_per, deadline - time.monotonic() - 800)
            if t_alloc < 3:
                break

            a0 = inits[name]
            a_opt, obj = gradient_optimize(a0, time_limit=t_alloc, lr_init=0.008, top_k=30)
            score = evaluate_sequence(a_opt.tolist())
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Scan n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: Deep LP optimization on top candidates
    # =========================================================
    print("=== Phase 2: LP Deep Optimization ===")
    t_start2 = time.monotonic()

    # Pick top unique (length, best_init) candidates
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, a_opt, score))
            seen.add(key)
        if len(top_configs) >= 5:
            break

    # Always include the best
    if best_sequence is not None:
        blen = len(best_sequence)
        if not any(c[0] == blen for c in top_configs):
            top_configs.append((blen, best_sequence.copy(), best_score))

    remaining = deadline - time.monotonic() - 400.0
    n_cand = max(1, len(top_configs))
    t_per_lp = max(30.0, remaining * 0.7 / n_cand)

    for i, (n, a0, init_score) in enumerate(top_configs):
        if time.monotonic() > deadline - 350:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 300)
        if t_alloc < 10:
            break

        print(f"  LP cand {i}: n={n}, init_score={init_score:.6f}, t={t_alloc:.0f}s")

        # First do gradient to get a good starting point for LP
        t_grad = min(20.0, t_alloc * 0.15)
        if t_grad > 3:
            a_g, obj_g = gradient_optimize(a0, time_limit=t_grad, lr_init=0.005, top_k=40)
            score_g = evaluate_sequence(a_g.tolist())
            if np.isfinite(score_g) and score_g < init_score:
                a0 = a_g
                init_score = score_g
            if np.isfinite(score_g) and score_g < best_score:
                best_score = score_g
                best_sequence = a_g.copy()

        # Then LP
        t_lp = t_alloc - t_grad - 10
        if t_lp > 10:
            a_opt, obj = lp_optimize(a0, time_limit=t_lp, verbose=(i == 0))
            score = evaluate_sequence(a_opt.tolist())
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  LP {i} (n={n}): {score:.6f} *NEW BEST*")
                else:
                    print(f"  LP {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Intensive LP polish on the single best
    # =========================================================
    print("=== Phase 3: Intensive Polish ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 150.0

        # Multiple LP runs with different perturbations
        n_polish = max(1, int(remaining / 60))
        n_polish = min(n_polish, 5)
        t_per_polish = remaining / n_polish * 0.85

        for pi in range(n_polish):
            if time.monotonic() > deadline - 120:
                break
            t_alloc = min(t_per_polish, deadline - time.monotonic() - 100)
            if t_alloc < 15:
                break

            if pi == 0:
                a0 = best_sequence.copy()
            else:
                scale = 0.001 * (pi + 1)
                noise = rng.standard_normal(len(best_sequence)) * scale
                a0 = np.maximum(best_sequence + noise, 0)
                a0 = normalize(a0)

            a_opt, obj = lp_optimize(a0, time_limit=t_alloc)
            score = evaluate_sequence(a_opt.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish {pi}: {score:.6f} *NEW BEST*")
            else:
                print(f"  Polish {pi}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Gradient + coordinate descent fine-tuning
    # =========================================================
    print("=== Phase 4: Fine-tuning ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0
        n_best = len(best_sequence)

        # Multiple gradient runs with different params
        grad_configs = [
            {'lr_init': 0.003, 'momentum': 0.93, 'top_k': 50},
            {'lr_init': 0.002, 'momentum': 0.95, 'top_k': 100},
            {'lr_init': 0.008, 'momentum': 0.88, 'top_k': 30},
            {'lr_init': 0.001, 'momentum': 0.97, 'top_k': 200},
        ]

        t_per_grad = remaining * 0.4 / len(grad_configs)
        for gi, cfg in enumerate(grad_configs):
            if time.monotonic() > deadline - 30:
                break
            t_alloc = min(t_per_grad, deadline - time.monotonic() - 20)
            if t_alloc < 3:
                break

            a_opt, obj = gradient_optimize(best_sequence, time_limit=t_alloc, **cfg)
            score = evaluate_sequence(a_opt.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Grad {gi}: {score:.6f} *NEW BEST*")

        # Coordinate descent
        remaining = deadline - time.monotonic() - 20.0
        t_cd = min(remaining * 0.5, 60.0)
        if t_cd > 5:
            a_opt, obj = coordinate_descent(best_sequence, time_limit=t_cd)
            score = evaluate_sequence(a_opt.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CoordDesc: {score:.6f} *NEW BEST*")

        # Final short LP
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 10:
            a_opt, obj = lp_optimize(best_sequence, time_limit=remaining)
            score = evaluate_sequence(a_opt.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: {score:.6f} *NEW BEST*")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Try extending/improving with different structure
    # =========================================================
    if best_sequence is not None and time.monotonic() < deadline - 30:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 10.0

        # Try a mixed init: blend best with ramp
        for blend in [0.3, 0.5, 0.7]:
            if time.monotonic() > deadline - 15:
                break
            x = np.arange(n_best, dtype=np.float64)
            ramp = normalize((x + 1.0) / n_best)
            a_blend = normalize(blend * best_sequence + (1 - blend) * ramp)

            t_alloc = min(remaining * 0.3, 40.0)
            if t_alloc < 5:
                break
            a_opt, obj = lp_optimize(a_blend, time_limit=t_alloc)
            score = evaluate_sequence(a_opt.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Blend {blend}: {score:.6f} *NEW BEST*")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = evaluate_sequence(best_sequence.tolist())

    # Final verification
    final_score = evaluate_sequence(best_sequence.tolist())
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
