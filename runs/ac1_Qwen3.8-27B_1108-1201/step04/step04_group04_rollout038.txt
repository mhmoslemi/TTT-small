def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""

    # Alternative: just use the fact that grad = 2 * IFFT(conj(FFT(a_padded)) * FFT(p_padded))[:n]
    # Let me verify: if we pad a to length L and p to length L, then
    # circular cross-correlation: C[i] = sum_j a[j] * conj(p[j-i]) 
    # Hmm, let me just do it with a direct approach for the gradient using the structure.
    #
    # Actually the cleanest way: conv[k] = sum_j a[j] * a[k-j]
    # d conv[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[i-(k-i)] (if 0<=2i-k<n and 2i-k != k-i)
    # Wait, I'm overcomplicating this. 
    #
    # Simplest correct approach: 
    # grad[i] = sum_k p_k * d(conv[k])/da[i]
    # d(conv[k])/da[i] = a[k-i] + a[2i-k] (each term present if index in [0,n-1])
    # 
    # Let's compute: S1[i] = sum_k p[k] * a[k-i] for 0<=k-i<n, 0<=k<2n-1
    #   = sum_{j=0}^{n-1} a[j] * p[j+i] for 0<=j+i<2n-1
    # S2[i] = sum_k p[k] * a[2i-k] for 0<=2i-k<n, 0<=k<2n-1
    #   = sum_{j=0}^{n-1} a[j] * p[2i-j] for 0<=2i-j<2n-1
    #
    # S1 is cross-correlation: S1[i] = sum_j a[j]*p[i+j]
    # S2[i] = sum_j a[j]*p[2i-j]
    #
    # For S1: pad a to length 3n, pad p to length 3n, compute cross-correlation
    # For S2: this is like a "reversed" cross-correlation
    #
    # Let me just use: pad both to length L >= 3n-1
    # A = FFT(a_padded), P = FFT(p_padded)
    # S1[i] = IFFT(conj(A) * P)[i] ... no that's circular correlation
    #
    # You know what, let me just use a simpler but correct method:
    # grad[i] = 2 * sum_j a[j] * p_{i+j} where the sum is over j such that 0<=j<n and 0<=i+j<2n-1
    #         + (correction for the double-counting at k=i when 2i-k = k-i, i.e., k=2i... 
    #           actually the two terms a[k-i] and a[2i-k] are the same when k-i = 2i-k, i.e., k=2i)
    #
    # Let me just verify with a small example and use the FFT approach:
    # The standard result: if c = a * a (linear conv), then dc/da is a "convolution" operation.
    # Specifically, the gradient of (1/2)||c - t||^2 w.r.t. a is the "correlation" of (c-t) with a.
    # By analogy, the gradient of sum_k p_k * c_k w.r.t. a[i] = 
    #   sum_k p_k * (a[k-i] + a[2i-k] valid)
    # 
    # Let me just compute it via FFT:
    # Define q = p (length 2n-1), pad to length L
    # Define a_pad = a (length n), pad to length L  
    # Linear cross-correlation of a and q: r[m] = sum_j a[j] * q[m+j] for valid j,m
    # We want: grad[i] = r[i] + r_rev[i] where r_rev[i] = sum_j a[j] * q[2i-j] for valid
    # 
    # Actually, I realize the cleanest approach for this specific structure:
    # 
    # conv = a * a (linear, length 2n-1)
    # d conv[k] / da[i] = a[k-i] (if 0 <= k-i < n) + a[i-k+i] ... 
    # 
    # Let me just be very explicit. conv[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
    # d conv[k]/da[i]:
    #   from a[j] term: a[k-i] if j=i and 0<=k-i<n
    #   from a[k-j] term: a[i-(k-i)] = a[2i-k] if k-j=i, i.e., j=k-i, and 0<=2i-k<n, and j!=i (to avoid double count)
    #   If j=i AND k-j=i (i.e., k=2i): both terms contribute a[i]*a[i], derivative is 2*a[i]... 
    #   Wait no. If k=2i and j=i, then the term is a[i]*a[i], and d/da[i] of a[i]^2 = 2*a[i].
    #   But in our formula, we get a[k-i] = a[i] from the j term, and a[2i-k] = a[0]... no.
    #   
    #   Let me be more careful. conv[k] = sum_j a[j]*a[k-j] (implicit valid range).
    #   d/d(a[i]) of a[j]*a[k-j]:
    #     if j = i: a[k-i]
    #     if k-j = i, i.e., j = k-i: a[i]
    #     if both (j=i and k-j=i, i.e., k=2i): 2*a[i]
    #   
    #   So d conv[k]/da[i] = a[k-i]*I(0<=k-i<n) + a[i]*I(0<=k-i<n and k-i!=i) 
    #                        + a[i]*I(0<=k-i<n and k-i==i)  [the extra from the second term when j=k-i=i]
    #   = a[k-i]*I(0<=k-i<n) + a[i]*I(0<=k-i<n)  [since when k-i=i, both terms give a[i])
    #   Wait, when k-i=i (k=2i): first term gives a[i], second term (j=k-i=i) gives a[i]. Total: 2a[i].
    #   When k-i != i: first term gives a[k-i] (if valid), second term gives a[i] (if k-i is valid, i.e., 0<=k-i<n).
    #   
    #   So: d conv[k]/da[i] = a[k-i]*I(0<=k-i<n) + a[i]*I(0<=k-i<n)
    #                     = (a[k-i] + a[i]) * I(0<=k-i<n)
    #
    #   Therefore: grad[i] = sum_k p[k] * (a[k-i] + a[i]) * I(0<=k-i<n)
    #                = sum_{j=0}^{n-1} p[j+i] * (a[j] + a[i])  [substituting j=k-i, valid when 0<=j<2n-1-i and j>=0]
    #   
    #   Wait, k ranges from 0 to 2n-2. j = k-i ranges from -i to 2n-2-i. Valid j: max(0,-i) to min(n-1, 2n-2-i).
    #   Since i >= 0: j ranges from 0 to min(n-1, 2n-2-i).
    #   For i <= n-1: 2n-2-i >= n-1 (since 2n-2-i >= n-1 iff n-1 >= i, which is true).
    #   So j ranges from 0 to n-1.
    #
    #   grad[i] = sum_{j=0}^{n-1} p[i+j] * a[j] + a[i] * sum_{j=0}^{n-1} p[i+j]
    #
    #   The first term is a cross-correlation. The second term is a[i] times a partial sum of p.
    #
    # Let me verify: for i=0, grad[0] = sum_j p[j]*a[j] + a[0]*sum_j p[j]
    # conv[0] = a[0]^2, so d conv[0]/da[0] = 2*a[0]. 
    # From formula: d conv[0]/da[0] = (a[0-0] + a[0])*I(0<=0<n) = 2*a[0]. Correct!
    # From grad formula: the k=0 term contributes p[0]*(a[0]+a[0]) = 2*p[0]*a[0]. 
    # Other k terms: for k>=1, j=k-0=k, need 0<=k<n. So k=1,...,n-1 contribute p[k]*a[k] + p[k]*a[0].
    # Total: 2*p[0]*a[0] + sum_{k=1}^{n-1} p[k]*a[k] + a[0]*sum_{k=1}^{n-1}p[k]
    #       = sum_{j=0}^{n-1} p[j]*a[j] + p[0]*a[0] + a[0]*sum_{k=1}^{n-1}p[k]
    #       = sum_j p[j]*a[j] + a[0]*sum_j p[j]. Correct!

    n = len(a)
    L = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, L)
    P = np.fft.rfft(p, L)
    # cross[i] = sum_j a[j]*p[i+j] = IFFT(conj(A)*P) ... 
    # Actually: circular cross-correlation of a and p:
    # corr[i] = sum_j a[j] * conj(p[j-i]) for circular
    # But we want sum_j a[j]*p[i+j] which is different.
    # Let's use: sum_j a[j]*p[i+j] = (a circ p_rev)[n-1-i] ... this is getting confusing.
    # 
    # Let me just use the direct formula with a smaller FFT:
    # Define v of length 3n-1: v[i+j] += a[j]*p[i+j] ... no.
    # 
    # Cleanest: define q of length 3n-1 where q[k] = norm of... 
    # 
    # Actually, let me just do: 
    # first_term[i] = sum_{j=0}^{n-1} a[j] * p[i+j] for i=0,...,n-1
    # This is a cross-correlation. We can compute it as:
    # Pad a to length 2n-1 (append zeros), pad p to length 3n-2 (we only need p[0]..p[2n-2])
    # Then first_term[i] for i=0..n-1 is the cross-correlation of a_padded and p at lag i in a specific direction.
    #
    # Let me use: np.correlate doesn't work well for this. Let me use FFT:
    # first_term = cross-correlation where (a cross p)[i] = sum_j a[j]*p[i+j]
    # This equals: if we define a_rev = a[::-1], then (a cross p)[i] = (a_rev * p)[2n-2-i] (linear conv)
    # Hmm, let me just verify with small n.
    #
    # For n=3, a=[a0,a1,a2], p=[p0,p1,p2,p3,p4] (length 5=2n-1)
    # first_term[0] = a0*p0 + a1*p1 + a2*p2
    # first_term[1] = a0*p1 + a1*p2 + a2*p3
    # first_term[2] = a0*p2 + a1*p3 + a2*p4
    #
    # This is: first_term[i] = sum_j a[j]*p[i+j]
    # If I define flipped_a = [a2, a1, a0] (reversed), and compute linear conv of flipped_a with p:
    # conv_flipped[k] = sum_j flipped_a[j]*p[k-j]
    # conv_flipped[2] = a2*p0 + a1*p1 + a0*p2  -- not quite what we want
    #
    # Alternative: compute linear conv of a with p:
    # (a*p)[k] = sum_j a[j]*p[k-j] for valid j
    # (a*p)[0] = a0*p0
    # (a*p)[1] = a0*p1 + a1*p0
    # (a*p)[2] = a0*p2 + a1*p1 + a2*p0
    # (a*p)[3] = a0*p3 + a1*p2 + a2*p1
    # (a*p)[4] = a0*p4 + a1*p3 + a2*p2
    # (a*p)[5] = a1*p4 + a2*p3
    # (a*p)[6] = a2*p4
    #
    # first_term[0] = a0*p0 + a1*p1 + a2*p2 = (a*p)[4]  (when n=3, 2n-2=4)
    # first_term[1] = a0*p1 + a1*p2 + a2*p3 = (a*p)[3]  (2n-2-1=3)
    # first_term[2] = a0*p2 + a1*p3 + a2*p4 = (a*p)[2]  (2n-2-2=2)
    # So first_term[i] = (a*p)[2n-2-i]!
    
    # Compute linear convolution of a (length n) and p (length 2n-1) = length 3n-2
    conv_len = n + (2 * n - 1) - 1  # = 3n-2
    fft_len = 1 << (conv_len - 1).bit_length()
    A_full = np.fft.rfft(a, fft_len)
    P_full = np.fft.rfft(p, fft_len)
    ap = np.fft.irfft(A_full * P_full, fft_len)[:conv_len]
    # first_term[i] = ap[2n-2-i] for i=0..n-1
    first_term = ap[2 * n - 2 : n - 1 : -1]  # ap[2n-2], ap[2n-3], ..., ap[n-1]
    
    # second term: a[i] * sum_{j=0}^{n-1} p[i+j] for i=0..n-1
    # partial sums of p
    p_csum = np.cumsum(p)
    # sum_{j=0}^{n-1} p[i+j] = p_csum[i+n-1] - p_csum[i-1] (with p_csum[-1]=0)
    second_sum = np.zeros(n)
    for i in range(n):
        end = i + n - 1
        if end < len(p):
            s = p_csum[end]
            if i > 0:
                s -= p_csum[i - 1]
            second_sum[i] = s
        else:
            # p has length 2n-1, so i+n-1 < 2n-1 iff i < n, which is always true
            # Actually 2n-1 elements means indices 0..2n-2, so max index is 2n-2
            # i+n-1 <= 2n-2 iff i <= n-1, which is true.
            pass  # shouldn't reach here
    
    return first_term + a * second_sum


def obj_and_grad(a, T):
    """Compute smoothed objective and gradient for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    obj = T * (np.log(ws) + scaled.max())  # = T * log(sum(exp(conv/T)))
    p = w / ws
    grad = smoothed_grad_conv(conv, a, T)
    return obj, grad, conv, p


def smoothed_gd(f0, time_limit, T_init=None, T_min=None, verbose=False):
    """
    Smoothed gradient descent on the simplex for minimizing max(conv).
    Uses adaptive temperature and learning rate.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    conv_range = conv.max() - conv.min()
    if T_init is None:
        T_init = max(conv_range * 0.1, 1e-5 / max(n, 1))
    if T_min is None:
        T_min = max(1e-7 / max(n, 1), 1e-10)
    
    best_obj = conv.max()
    best_f = f.copy()
    T = T_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 0.005
    lr_min = 1e-6

    while time.monotonic() - t0 < time_limit:
        it += 1

        obj, grad, conv, p = obj_and_grad(f, T)
        
        # Project gradient to tangent space of simplex
        grad_tan = grad - np.mean(grad)
        
        # Adaptive learning rate: scale by T/conv_range to be dimensionless
        conv_range = max(conv.max() - conv.min(), 1e-12)
        effective_lr = lr * T / conv_range
        
        f_new = project_simplex(f - effective_lr * grad_tan)
        
        conv_new = fast_convolve(f_new)
        hard_obj = conv_new.max()

        if hard_obj < best_obj - 1e-14:
            best_obj = hard_obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                lr = max(lr * 0.9, lr_min)

        # Anneal temperature
        T = max(T * 0.995, T_min)

        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng((it * 104729 + 7) % (2**31))
            noise_scale = 0.001 * min(1.0, no_improve / 200.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            lr = max(lr * 2.0, 0.003)
            T = max(T_init * 0.1, T_min)
            no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SGD it={it}, obj={best_obj:.10f}, T={T:.2e}, lr={lr:.2e}, t={elapsed:.0f}s")

    return best_f, best_obj


def active_set_gd(f0, time_limit, T_init=None, T_min=None, verbose=False):
    """
    Gradient descent that focuses on the active set (peaks near max).
    Uses a weighted combination of the full gradient and the active-set gradient.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    conv_range = conv.max() - conv.min()
    if T_init is None:
        T_init = max(conv_range * 0.05, 1e-5 / max(n, 1))
    if T_min is None:
        T_min = max(1e-8 / max(n, 1), 1e-10)

    best_obj = conv.max()
    best_f = f.copy()
    T = T_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 0.003
    lr_min = 1e-7
    top_k_frac = 0.1

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        current_max = conv.max()
        
        # Identify active set: top-k peaks
        top_k = max(3, int(n * top_k_frac))
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peak_indices]
        
        # Build a "peaked" distribution emphasizing the top peaks
        margin = 0.1 * conv_range if conv_range > 0 else 0.01
        active_mask = conv >= (current_max - margin)
        if active_mask.sum() == 0:
            active_mask[peak_indices[0]] = True
        
        # Compute gradient using only active peaks (weighted)
        T_local = max(T, 1e-8)
        scaled_active = conv[active_mask] / T_local
        scaled_active -= scaled_active.max()
        w_active = np.exp(scaled_active)
        if w_active.sum() < 1e-300:
            w_active = np.zeros(len(w_active))
            w_active[np.argmax(scaled_active)] = 1.0
        else:
            w_active /= w_active.sum()
        
        # p_full for the smoothed objective
        scaled_full = conv / T_local
        scaled_full -= scaled_full.max()
        w_full = np.exp(scaled_full)
        p_full = w_full / w_full.sum()
        
        # Compute gradient using p_full (standard smoothed gradient)
        grad = smoothed_grad_conv(conv, f, T_local)
        
        # Project to tangent space
        grad_tan = grad - np.mean(grad)
        
        conv_range = max(current_max - conv.min(), 1e-12)
        effective_lr = lr * T_local / conv_range
        
        f_new = project_simplex(f - effective_lr * grad_tan)
        conv_new = fast_convolve(f_new)
        hard_obj = conv_new.max()

        if hard_obj < best_obj - 1e-14:
            best_obj = hard_obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 15 == 0:
                lr = max(lr * 0.92, lr_min)

        T = max(T * 0.993, T_min)
        
        # Adapt top_k: increase when stuck (focus more)
        if no_improve > 100:
            top_k_frac = min(top_k_frac * 0.8, 0.02)
        
        if no_improve > 300:
            rng = np.random.default_rng((it * 331 + 13) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0005)
            lr = max(lr * 1.5, 0.002)
            T = max(T_init * 0.05, T_min)
            top_k_frac = 0.1
            no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      AS-GD it={it}, obj={best_obj:.10f}, T={T:.2e}, t={elapsed:.0f}s")

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate i, the objective is a piecewise quadratic in a[i].
    We find the minimizer by checking the peak structure.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    # For coordinate descent, when we change a[i] by delta (and subtract delta/n from all
    # others to maintain sum=1), the change in conv[k] is:
    # conv_new[k] = conv[k] + delta*(f[k-i] + f[2i-k]) + delta^2 (if k=2i) - (delta/n)*corrections
    # This is complex. Let's use a simpler approach: 
    # perturb a[i] by +eps and -eps, evaluate, take the better direction.
    
    eps = 0.01 / n
    order = np.arange(n)
    np.random.default_rng(42).shuffle(order)
    coord_it = 0
    
    while time.monotonic() - t0 < time_limit:
        for i in order:
            if time.monotonic() - t0 >= time_limit:
                break
            coord_it += 1
            
            # Try increasing a[i]
            f_trial = f.copy()
            f_trial[i] += eps
            # Renormalize
            f_trial = f_trial / f_trial.sum()
            conv_trial = fast_convolve(f_trial)
            obj_trial = conv_trial.max()
            
            if obj_trial < best_obj - 1e-15:
                f = f_trial
                best_obj = obj_trial
                best_f = f.copy()
                continue
            
            # Try decreasing a[i]
            if f[i] > 2 * eps:
                f_trial = f.copy()
                f_trial[i] -= eps
                f_trial = f_trial / f_trial.sum()
                conv_trial = fast_convolve(f_trial)
                obj_trial = conv_trial.max()
                
                if obj_trial < best_obj - 1e-15:
                    f = f_trial
                    best_obj = obj_trial
                    best_f = f.copy()
                    continue
            
            # Also try a bigger step if small step helped
            # (skipping for now to keep it simple)
        
        # Decay eps
        eps *= 0.99
        if eps < 1e-10 / n:
            break

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    # Edge-concentrated (U-shape)
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Two humps
    h1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.08 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.08 * n)) ** 2)
    inits.append((h1 + h2) / (h1 + h2).sum())
    
    # Cosine window
    c = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c / c.sum())
    
    # Mirror-symmetric
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.4)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    if n % 2 == 1:
        sym[half_n] = half[-1] + half[0]
    inits.append(sym / sym.sum())
    
    # From parent (resampled)
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    a = np.maximum(a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 0.01:
        return float('inf')
    score = float(evaluate_sequence(a.tolist()))
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main optimization: multi-phase search combining smoothed gradient descent,
    active-set focused optimization, coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return float('inf')
        a = np.maximum(a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = evaluate_safe(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_safe(parent):.8f}")
    except Exception:
        pass

    # === Phase 1: Multi-length exploration (250s) ===
    print("\n=== Phase 1: Multi-length Exploration ===")
    t_p1 = time.monotonic()
    p1_budget = 250.0

    all_best = []  # list of (normalized_f, n)

    if parent is not None:
        n_parent = len(parent)
        # Quick optimize parent length
        t_alloc = min(80.0, p1_budget - 30.0)
        print(f"  Optimizing parent n={n_parent} (t={t_alloc:.0f}s)")
        f_opt, _ = smoothed_gd(parent, time_limit=t_alloc, T_init=None, T_min=None)
        all_best.append(f_opt)
        try_seq(f_opt, "p1_parent")

        # Try nearby lengths
        for delta in [-4, -8, 4, 8, -16, 16, -32, 32]:
            rem = p1_budget - (time.monotonic() - t_p1)
            if rem < 20:
                break
            new_n = n_parent + delta
            if new_n < 64 or new_n > 2048:
                continue
            # Interpolate parent to new length
            px = np.interp(np.linspace(0, 1, new_n),
                          np.linspace(0, 1, n_parent), parent)
            px = np.maximum(px, 0)
            px = px / px.sum()
            t_alloc2 = min(40.0, rem - 10.0)
            f_opt2, _ = smoothed_gd(px, time_limit=t_alloc2)
            all_best.append(f_opt2)
            try_seq(f_opt2, f"p1_n{new_n}")
            rem = p1_budget - (time.monotonic() - t_p1)
    else:
        # No parent: try a few standard lengths
        for n_try in [256, 384, 512]:
            rem = p1_budget - (time.monotonic() - t_p1)
            if rem < 30:
                break
            inits = generate_inits(n_try, None, rng)
            best_init = min(inits, key=lambda x: fast_convolve(x).max())
            t_alloc = min(60.0, rem - 10.0)
            f_opt, _ = smoothed_gd(best_init, time_limit=t_alloc)
            all_best.append(f_opt)
            try_seq(f_opt, f"p1_n{n_try}")

    print(f"  Phase 1 done: best={best_score:.8f}")

    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = evaluate_safe(best_seq)

    # === Phase 2: Deep optimization (1050s) ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    t_p2 = time.monotonic()
    p2_budget = 1050.0

    f_work = best_seq / best_seq.sum()
    n_work = len(f_work)

    # Cycle between different optimizers
    cycles = 0
    while time.monotonic() - t_p2 < p2_budget and time.monotonic() < deadline - 60:
        cycles += 1
        rem = p2_budget - (time.monotonic() - t_p2)
        if rem < 30:
            break

        # Sub-phase A: Smoothed GD (longer runs)
        t_a = min(rem * 0.4, 200.0)
        if t_a > 20:
            T_init = max(1e-5, 0.005 * fast_convolve(f_work).max())
            print(f"  Cycle {cycles}: Smoothed GD (t={t_a:.0f}s, T={T_init:.2e})")
            f_opt, _ = smoothed_gd(f_work, time_limit=t_a, T_init=T_init,
                                    T_min=max(1e-9, T_init * 0.001))
            sc = try_seq(f_opt, f"p2_sgd_{cycles}")
            if f_opt is not None and sc < best_score:
                f_work = f_opt / f_opt.sum()
            rem = p2_budget - (time.monotonic() - t_p2)

        # Sub-phase B: Active-set GD
        rem = p2_budget - (time.monotonic() - t_p2)
        if rem > 30:
            t_b = min(rem * 0.35, 150.0)
            if t_b > 20:
                T_init_b = max(1e-6, 0.002 * fast_convolve(f_work).max())
                print(f"  Cycle {cycles}: Active-Set GD (t={t_b:.0f}s)")
                f_opt_b, _ = active_set_gd(f_work, time_limit=t_b, T_init=T_init_b,
                                            T_min=max(1e-10, T_init_b * 0.001))
                sc = try_seq(f_opt_b, f"p2_asgd_{cycles}")
                if f_opt_b is not None and sc < best_score:
                    f_work = f_opt_b / f_opt_b.sum()
                rem = p2_budget - (time.monotonic() - t_p2)

        # Sub-phase C: Perturbation exploration
        rem = p2_budget - (time.monotonic() - t_p2)
        if rem > 20:
            n_perturb = min(5, int(rem / 15))
            for pi in range(n_perturb):
                rem = p2_budget - (time.monotonic() - t_p2)
                if rem < 15:
                    break
                noise_scale = 0.002 + 0.003 * pi
                f_pert = project_simplex(f_work + rng.standard_normal(n_work) * noise_scale)
                f_opt_p, _ = smoothed_gd(f_pert, time_limit=min(12.0, rem - 5.0),
                                          T_init=max(1e-5, 0.003 * fast_convolve(f_pert).max()),
                                          T_min=1e-9)
                sc = try_seq(f_opt_p, f"p2_pert{pi}_c{cycles}")
                if f_opt_p is not None and sc < best_score:
                    f_work = f_opt_p / f_opt_p.sum()

    print(f"  Phase 2 done: best={best_score:.8f}")

    # === Phase 3: Ultra-fine polishing (400s) ===
    print(f"\n=== Phase 3: Ultra-Fine Polishing ===")
    t_p3 = time.monotonic()
    p3_budget = 400.0

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()
        n_work = len(f_work)
        conv_work = fast_convolve(f_work)
        current_max = conv_work.max()

        # Very fine smoothed gradient with tiny T
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 30:
            T_very_fine = max(1e-7, current_max * 0.001)
            t_fine = min(150.0, rem - 20.0)
            print(f"  Ultra-fine SGD (T={T_very_fine:.2e}, t={t_fine:.0f}s)")
            f_opt, _ = smoothed_gd(f_work, time_limit=t_fine,
                                    T_init=T_very_fine, T_min=T_very_fine * 0.01)
            try_seq(f_opt, "p3_ultra_sgd")
            if f_opt is not None:
                f_work = f_opt / f_opt.sum()

        # Active set with very focused top-k
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 30:
            T_as = max(1e-8, current_max * 0.0005)
            t_as = min(100.0, rem - 20.0)
            print(f"  Ultra-fine AS-GD (T={T_as:.2e}, t={t_as:.0f}s)")
            f_opt, _ = active_set_gd(f_work, time_limit=t_as,
                                      T_init=T_as, T_min=T_as * 0.01)
            try_seq(f_opt, "p3_ultra_as")
            if f_opt is not None:
                f_work = f_opt / f_opt.sum()

        # Coordinate descent (very expensive but precise)
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 60:
            t_cd = min(120.0, rem - 20.0)
            print(f"  Coordinate Descent (t={t_cd:.0f}s)")
            f_opt, _ = coordinate_descent(f_work, time_limit=t_cd)
            try_seq(f_opt, "p3_cd")
            if f_opt is not None:
                f_work = f_opt / f_opt.sum()

        # One more pass of fine SGD after CD
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 20:
            T_final = max(1e-8, 0.0005 * fast_convolve(f_work).max())
            t_final = min(60.0, rem - 10.0)
            f_opt, _ = smoothed_gd(f_work, time_limit=t_final,
                                    T_init=T_final, T_min=T_final * 0.01)
            try_seq(f_opt, "p3_final_sgd")

    print(f"  Phase 3 done: best={best_score:.8f}")

    # === Phase 4: Final tricks ===
    print(f"\n=== Phase 4: Final Tricks ===")

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()
        n_work = len(f_work)
        rem_total = deadline - time.monotonic()

        # Try mirror symmetry
        if rem_total > 30:
            mirrored = 0.5 * f_work + 0.5 * f_work[::-1]
            mirrored = project_simplex(mirrored)
            if np.max(np.abs(mirrored - f_work)) > 1e-8:
                T_m = max(1e-7, 0.001 * fast_convolve(mirrored).max())
                t_m = min(30.0, rem_total / 3.0)
                f_opt_m, _ = smoothed_gd(mirrored, time_limit=t_m,
                                          T_init=T_m, T_min=T_m * 0.01)
                try_seq(f_opt_m, "p4_mirror")
                rem_total = deadline - time.monotonic()

        # Try adjacent lengths (n±1, n±2, n±4)
        if rem_total > 60:
            for delta in [1, -1, 2, -2, 4, -4]:
                rem_total = deadline - time.monotonic()
                if rem_total < 20:
                    break
                new_n = n_work + delta
                if new_n < 32 or new_n > 4096:
                    continue
                f_interp = np.interp(np.linspace(0, 1, new_n),
                                     np.linspace(0, 1, n_work), f_work)
                f_interp = np.maximum(f_interp, 0)
                f_interp = f_interp / f_interp.sum()
                T_i = max(1e-7, 0.002 * fast_convolve(f_interp).max())
                t_i = min(20.0, rem_total / 2.0)
                f_opt_i, _ = smoothed_gd(f_interp, time_limit=t_i,
                                          T_init=T_i, T_min=T_i * 0.01)
                try_seq(f_opt_i, f"p4_len{new_n}")
                rem_total = deadline - time.monotonic()

        # Tiny perturbation + quick optimize
        if rem_total > 30:
            for _ in range(3):
                rem_total = deadline - time.monotonic()
                if rem_total < 10:
                    break
                f_p = project_simplex(f_work + rng.standard_normal(n_work) * 0.0003)
                T_p = max(1e-8, 0.001 * fast_convolve(f_p).max())
                f_opt_p, _ = smoothed_gd(f_p, time_limit=min(10.0, rem_total - 5.0),
                                          T_init=T_p, T_min=T_p * 0.01)
                try_seq(f_opt_p, "p4_tiny_perturb")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = evaluate_safe(best_seq)

    # Final verification
    final_score = evaluate_safe(best_seq)
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_seq = np.ones(384) / 384
        best_score = evaluate_safe(best_seq)

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
