```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-start with diverse initializations and multiple sequence lengths
2. Log-sum-exp smoothed gradient descent with adaptive temperature
3. Coordinate descent for 1D convex subproblems
4. LP-based Frank-Wolfe for global escape
5. Adaptive time allocation
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_objective_and_grad(a, T):
    """
    Compute log-sum-exp smoothed objective and gradient.
    f_T(a) = T * log(sum_k exp(conv[k]/T))
    grad = (1/T) * sum_k p_k * grad_conv[k] where p_k = exp(conv[k]/T) / sum exp
    """
    n = len(a)
    conv = fast_convolve(a)
    # Numerical stability
    conv_shifted = conv / T - (conv.max() / T)
    p = np.exp(conv_shifted)
    p_sum = p.sum()
    p = p / p_sum
    
    obj = T * (np.log(p_sum) + conv.max() / T)
    
    # Gradient: sum_k p_k * grad_conv[k]
    # grad_conv[k][j] = a[k-j] + a[j-k] (with 0 for out of bounds)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        pk = p[k]
        # For each j: grad[j] += pk * (a[k-j] if valid + a[j-k] if valid)
        # a[k-j] valid when 0 <= k-j < n => j in [max(0,k-n+1), min(n-1,k)]
        # a[j-k] valid when 0 <= j-k < n => j in [k, min(n-1, k+n-1)]
        js1 = max(0, k - n + 1)
        je1 = min(n - 1, k)
        if js1 <= je1:
            idx = np.arange(js1, je1 + 1)
            grad[idx] += pk * a[k - idx]
        
        js2 = max(0, k)
        je2 = min(n - 1, k + n - 1)
        if js2 <= je2:
            idx = np.arange(js2, je2 + 1)
            grad[idx] += pk * a[idx - k]
    
    return obj, grad, conv


def smooth_grad_descent(a_init, time_limit, T_init=0.01, T_final=0.0001, 
                        lr_init=0.005, verbose=False):
    """
    Gradient descent on the smoothed objective with adaptive parameters.
    Works on the simplex (sum=1, a>=0).
    """
    n = len(a_init)
    a = project_simplex(a_init.astype(np.float64))
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)  # momentum
    mom = 0.9
    T = T_init
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        obj, grad, conv = smooth_objective_and_grad(a, T)
        
        # Momentum step
        v = mom * v - lr * grad
        a_new = project_simplex(a + v)
        
        # Evaluate true objective
        conv_new = fast_convolve(a_new)
        true_obj = conv_new.max()
        
        if true_obj < best_obj - 1e-14:
            best_obj = true_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        conv = conv_new
        
        # Adaptive parameters
        T = max(T * 0.999, T_final)
        lr = lr * 0.9995
        
        if no_improve > 100:
            # Reset with perturbation
            rng = np.random.default_rng(it)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            T = T_init * 0.5
            lr = lr_init * 0.5
            no_improve = 0
            conv = fast_convolve(a)
        
        if verbose and it % 100 == 0:
            print(f"    SGD iter {it}: obj={best_obj:.8f}, T={T:.6f}, lr={lr:.6f}")
    
    return best_a, best_obj

    For each coordinate j, the subproblem is 1D convex minimization.
    """

    a = project_simplex(a_init.astype(np.float64))
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()

        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        
        for j in range(n):
            x_old = a[j]
            
            # Compute the 1D function f(x) = max_k conv_k(x)
            # where conv_k(x) depends linearly (or quadratically) on x
            
            # For each k, the change in conv[k] when a[j] goes from x_old to x:
            # delta_conv[k] = (x - x_old) * c_k + (quadratic term if k=2j)
            # where c_k = a[k-j] + a[j-k] (with 0 for out of bounds)

                if 0 <= j - k < n:
                    val += a[j - k]

            # conv[k] = sum_i a[i] * a[k-i]
            # When we change a[j] from x_old to x:
            # The terms involving a[j] are: a[j]*a[k-j] (from i=j) and a[k-j]*a[j] (from i=k-j)
            # If j != k-j: both terms = x * a[k-j] (old was x_old * a[k-j])
            #   delta = 2 * (x - x_old) * a[k-j] (if 0 <= k-j < n)
            # If j == k-j (i.e., k = 2j): term = x^2 (old was x_old^2)

            # where delta_k = 2*(x-x_old)*a[k-j] if k!=2j and 0<=k-j<n
            #                       (x^2 - x_old^2) if k==2j
            #                       0 otherwise
            
            # For the 1D problem: minimize max_k (conv[k] + delta_k(x)) subject to x >= 0
            # (and we don't need an upper bound since we'll renormalize later)
            
            # Key: the function is convex in x (max of convex functions)
            # Use bisection on subgradient or ternary search
            
            # Ternary search on [0, x_max] where x_max = max(2*x_old, x_old + 0.01)
            x_lo = 0.0
            x_hi = max(2.0 * x_old + 0.001, 0.01)
            
            # First, coarse grid to narrow down
            grid = np.linspace(x_lo, x_hi, 20)
            vals = np.zeros(len(grid))
            for i, x in enumerate(grid):
                new_conv = conv.copy()
                # Linear part
                mask = np.arange(2*n-1) != 2*j
                new_conv[mask] += c[mask] * (x - x_old)
                # Quadratic part (k = 2j)
                if 2*j < 2*n-1:
                    new_conv[2*j] = conv[2*j] + (x*x - x_old*x_old)
                vals[i] = new_conv.max()
            
            # Narrow to best 3 grid points
            best_idx = np.argmin(vals)
            search_lo = grid[max(0, best_idx - 1)]
            search_hi = grid[min(len(grid) - 1, best_idx + 1)]
            
            # Ternary search refinement
            for _ in range(20):
                m1 = search_lo + (search_hi - search_lo) / 3.0
                m2 = search_hi - (search_hi - search_lo) / 3.0
                
                new_conv1 = conv.copy()
                mask = np.arange(2*n-1) != 2*j
                new_conv1[mask] += c[mask] * (m1 - x_old)
                if 2*j < 2*n-1:
                    new_conv1[2*j] = conv[2*j] + (m1*m1 - x_old*x_old)
                f1 = new_conv1.max()
                
                new_conv2 = conv.copy()
                new_conv2[mask] += c[mask] * (m2 - x_old)
                if 2*j < 2*n-1:
                    new_conv2[2*j] = conv[2*j] + (m2*m2 - x_old*x_old)
                f2 = new_conv2.max()
                
                if f1 < f2:
                    search_hi = m2
                else:
                    search_lo = m1
            
            x_opt = (search_lo + search_hi) / 2.0
            
            # Apply
            a[j] = x_opt
            conv = fast_convolve(a)
            new_obj = conv.max()
            
            if new_obj < best_obj - 1e-13:
                best_obj = new_obj
                best_a = a.copy()
                improved = True
        
        # Renormalize
        a = a / a.sum()
        conv = fast_convolve(a)
        if conv.max() < best_obj:
            best_obj = conv.max()
            best_a = a.copy()
        
        if not improved and sweep > 0:
            break
    
    return best_a, best_obj


def solve_lp_oracle(f, M, time_limit=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix efficiently
    # (f*g)[k] = sum_j f[j]*g[k-j] <= M
    # This is a convolution constraint
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] appears in (f*g)[k] for k = j, j+1, ..., j+n-1
        # with coefficient f[0], f[1], ..., f[n-1]
        A_ub[j:j+n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = project_simplex(f0.astype(np.float64))
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        
        g = solve_lp_oracle(f, M * m_exp, time_limit=min(30, rem - 5))
        if g is None:
            m_exp *= 1.3
            if m_exp > 10.0:
                break
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = best_obj
        
        alphas = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        for alpha in alphas:
            ft = project_simplex((1.0 - alpha) * f + alpha * g)
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Fine search
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(8):
                if hi - lo < 1e-5:
                    break
                mid = (lo + hi) / 2.0
                ft = project_simplex((1.0 - mid) * f + mid * g)
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-8:
            f = project_simplex((1.0 - best_alpha) * f + best_alpha * g)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
    
    return best_f, best_obj
