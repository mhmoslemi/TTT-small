"""
Multi-method optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines mirror descent (multiplicative/softmax updates), LP-based Frank-Wolfe,
smoothed gradient descent with Euclidean projection, and subgradient polishing.
Incorporates symmetry exploitation, multi-scale initialization, and adaptive restarts.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_corr_fft(a, p):
    """Compute g[i] = sum_{j=0}^{n-1} a[j]*p[j+i] for i=0..n-1 via FFT.
    a has length n, p has length >= n (typically 2n-1).
    """
    n = len(a)
    m = len(p)
    total = n + m - 1
    fft_size = 1 << (total - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    corr = np.fft.irfft(np.conj(A) * P, fft_size)
    return corr[:n]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_dir(f, M):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def mirror_descent(f0, time_limit, T_init=0.004, T_min=0.00003, eta_init=0.4,
                   verbose=False):
    """Mirror descent (multiplicative updates) with log-sum-exp smoothing."""
    f = np.maximum(f0.astype(np.float64), 1e-30)
    s = f.sum()
    f = f / s
    n = len(f)

    T = T_init
    eta = eta_init
    t0 = time.monotonic()
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws

        g = 2.0 * cross_corr_fft(f, w)

        # Mirror descent step: a_new[i] = exp(log(f[i]) - eta*g[i]) / Z
        log_f = np.log(np.maximum(f, 1e-30))
        exponents = log_f - eta * g
        exponents -= exponents.max()
        f_new = np.exp(exponents)
        f_new /= f_new.sum()
        f_new = np.maximum(f_new, 0.0)
        f_new /= f_new.sum()

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            # Slightly increase eta on success
            eta = min(eta * 1.01, 2.0)
        else:
            no_improve += 1
            f = f_new  # still move to explore
            if no_improve % 5 == 0:
                eta = max(eta * 0.93, 0.05)

        T = max(T * 0.9985, T_min)

        if no_improve > 300:
            rng = np.random.default_rng((it * 7919 + n * 13) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            f = np.maximum(f, 1e-30)
            f /= f.sum()
            eta = eta_init * 0.5
            T = max(T_init * 0.05, T_min)
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    MD iter {it}: obj={best_obj:.10f}, T={T:.6f}, eta={eta:.4f}")

    return best_f, best_obj

    while time.monotonic() - t0 < time_limit:

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []

    # U-shape (high at edges)

    inits.append(edge_w / edge_w.sum())

    # Inverse U (high in middle)
    inv_edge = (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    inits.append(inv_edge / inv_edge.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    # Mirror-symmetric random
    half_n = n // 2

    sym = np.zeros(n)

    sym[half_n:] = half[::-1]

        px = np.maximum(px, 0)
        ps = px.sum()

                print(f"  Cycle {cycle}: SGD (t={t_b:.0f}s)")
                a_opt, _ = smoothed_grad_descent(f_work, time_limit=t_b, T_init=T_s)
                try_seq(a_opt, f"p2_sgd_{cycle}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()

            # Sub-phase C: LP-FW
            rem = p2_budget - (time.monotonic() - t_p2)
            t_c = min(t_per_cycle * 0.25, rem * 0.3)
            if t_c > 15:
                print(f"  Cycle {cycle}: FW-LP (t={t_c:.0f}s)")
                a_opt, _ = fw_lp_optimize(f_work, time_limit=t_c)
                try_seq(a_opt, f"p2_fw_{cycle}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()

            # Sub-phase D: Symmetry polish
            rem = p2_budget - (time.monotonic() - t_p2)
            if rem > 10:
                a_sym, _ = symmetry_polish(f_work, 5.0)
                try_seq(a_sym, f"p2_sym_{cycle}")

    print(f"  Phase 2 done. Best: {best_score:.8f}")

    # === Phase 3: Fine polish (450s) ===
    print("\n=== Phase 3: Fine Polish ===")
    t_p3 = time.monotonic()
    p3_budget = 450.0

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()

        # Multiple subgradient rounds with decreasing parameters
        configs = [
            {"lr": 0.0008, "top_k_frac": 0.04, "temp": 0.00002},
            {"lr": 0.0004, "top_k_frac": 0.03, "temp": 0.00001},
            {"lr": 0.0002, "top_k_frac": 0.02, "temp": 0.000005},
            {"lr": 0.0001, "top_k_frac": 0.02, "temp": 0.000002},
        ]

        for i, cfg in enumerate(configs):
            rem = p3_budget - (time.monotonic() - t_p3)
            if rem < 25:
                break
            t_alloc = min(70.0, rem - 15.0)
            print(f"  SG round {i} (t={t_alloc:.0f}s)")
            a_opt, _ = sg_polish(f_work, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"p3_sg_{i}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Very fine mirror descent
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 40:
            T_init = max(0.0002, 0.0008 * fast_convolve(f_work).max())
            t_alloc = min(90.0, rem - 15.0)
            print(f"  Fine MD (t={t_alloc:.0f}s, T={T_init:.6f})")
            a_opt, _ = mirror_descent(f_work, time_limit=t_alloc,
                                      T_init=T_init, T_min=0.00001, eta_init=0.2)
            try_seq(a_opt, "p3_fine_md")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Fine smoothed gradient
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 30:
            T_init = max(0.0002, 0.0005 * fast_convolve(f_work).max())
            t_alloc = min(60.0, rem - 10.0)
            print(f"  Fine SGD (t={t_alloc:.0f}s)")
            a_opt, _ = smoothed_grad_descent(f_work, time_limit=t_alloc,
                                             T_init=T_init, T_min=0.00002)
            try_seq(a_opt, "p3_fine_sgd")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Final LP-FW
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 30:
            t_alloc = min(50.0, rem - 10.0)
            print(f"  Final FW (t={t_alloc:.0f}s)")
            a_opt, _ = fw_lp_optimize(f_work, time_limit=t_alloc)
            try_seq(a_opt, "p3_final_fw")

    print(f"  Phase 3 done. Best: {best_score:.8f}")

    # === Phase 4: Final tricks (remaining time) ===
    print("\n=== Phase 4: Final Tricks ===")

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()
        n_best = len(best_seq)

        # Symmetry polish
        a_sym, _ = symmetry_polish(f_work, 10.0)
        try_seq(a_sym, "p4_sym")
        if best_seq is not None:
            f_work = best_seq / best_seq.sum()

        # Tiny perturbations + quick MD
        for trial in range(5):
            if time.monotonic() > deadline - 20:
                break
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(f_work + noise)
            a_opt, _ = mirror_descent(f_pert, time_limit=8.0, T_init=0.001, eta_init=0.1)
            try_seq(a_opt, f"p4_pert_{trial}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # One more subgradient with very small step
        if time.monotonic() < deadline - 15:
            a_opt, _ = sg_polish(f_work, time_limit=10.0, lr=0.00005,
                                  top_k_frac=0.02, temp=0.000001)
            try_seq(a_opt, "p4_final_sg")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
