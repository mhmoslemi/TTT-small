def fast_convolve(a):

    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Implements improved LP-based Frank-Wolfe with:
- Pre-built sparse LP structure
- Asymmetric initialization biased toward low-peak convolution

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _eval_alpha(f, g, alpha, n):

    if st < 1e-15:

    while time.monotonic() - t0 < time_limit:
        improved = False
        n_trials = min(200, max(50, 200000 // max(1, no_improve // 10)))

            i = rng_local.integers(n)
            j = rng_local.integers(n)
            if i == j:
                continue
            
            # Try both directions: move from i to j, and from j to i
            for src, dst in [(i, j), (j, i)]:
                if f[src] < 1e-15:
                    continue
                
                # Find optimal delta
                # conv_new[k] = conv[k] + 2*delta*(f[k-dst] - f[k-src]) + delta^2 * q[k]

                    kd = k - dst
                    ks = k - src
                    if 0 <= kd < n:
                        val += f[kd]
                    if 0 <= ks < n:
                        val -= f[ks]
                    L[k] = 2.0 * val

                q = np.zeros(2 * n - 1)
                q[2 * dst] += 1.0
                q[src + dst] -= 2.0

                # Max delta is f[src] (can't go negative)

                    if (new_conv < 0).any():

                        f[dst] += delta
                        conv = conv + delta * L + delta * delta * q
                        best_obj = new_max

                    break

            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.001)

            no_improve = 0

            m_relax = 1.0
            no_improve = 0
            use_line_search = False

    return best_f, best_obj


def _asymmetric_init(n, rng, mode=0):
    """Generate asymmetric initializations that are biased toward low peak convolution."""
    x = np.arange(n, dtype=np.float64)
    if mode == 0:
        # Asymmetric ramp with noise
        a = (x + 1.0) ** 0.7 * (1.0 + 0.3 * rng.standard_normal(n))
    elif mode == 1:
        # Off-center Gaussian (skewed left)
        mu = n * 0.3
        sigma = n * 0.2
        a = np.exp(-0.5 * ((x - mu) / sigma) ** 2) + 0.1 * rng.standard_normal(n)
    elif mode == 2:
        # Two asymmetric bumps
        a = (np.exp(-((x - n * 0.25) ** 2) / (2.0 * (n * 0.12) ** 2)) +
             0.4 * np.exp(-((x - n * 0.7) ** 2) / (2.0 * (n * 0.15) ** 2)))
        a = a + 0.05 * rng.standard_normal(n)
    elif mode == 3:
        # Power-law decay
        a = (n - x) ** 1.2 * (1.0 + 0.2 * rng.standard_normal(n))
    else:
        # Sinusoidal window (asymmetric)
        a = np.sin(np.pi * (x + 1) / (n + 1)) ** 0.5 * (1.0 + 0.1 * rng.standard_normal(n))

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def pair_swap_polish(f0, time_limit, rng_seed=0):
    """
    Pair-swap coordinate descent: move small amounts of mass between pairs.
    Uses exact quadratic update for the convolution to avoid recomputation.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    rng_local = np.random.default_rng(rng_seed)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve_rounds = 0

    while time.monotonic() - t0 < time_limit:
        improved_this_round = False
        n_trials = 300

        for _ in range(n_trials):
            i = int(rng_local.integers(n))
            j = int(rng_local.integers(n))
            if i == j:
                continue

            for src, dst in [(i, j), (j, i)]:
                if f[src] < 1e-12:
                    continue

                # Compute L[k] = 2*(f[k-dst] - f[k-src])
                L = np.zeros(2 * n - 1)
                L[dst:dst + n] += 2.0 * f
                L[src:src + n] -= 2.0 * f

                # Quadratic term (sparse: only 3 non-zero entries)
                q_idx = [2 * dst, src + dst, 2 * src]
                q_val = [1.0, -2.0, 1.0]

                delta_max = f[src]

                # Try a few delta values (log-spaced)
                for frac in [0.1, 0.05, 0.02, 0.5]:
                    delta = frac * delta_max
                    if delta < 1e-12:
                        continue

                    # Compute new max efficiently
                    new_conv = conv + delta * L
                    for qi, qv in zip(q_idx, q_val):
                        if 0 <= qi < len(new_conv):
                            new_conv[qi] += delta * delta * qv

                    new_max = new_conv.max()
                    if new_max < best_obj - 1e-13:
                        f[src] -= delta
                        f[dst] += delta
                        conv = new_conv
                        best_obj = new_max
                        best_f = f.copy()
                        improved_this_round = True
                        break
                if improved_this_round:
                    break
            if improved_this_round:
                break

        if not improved_this_round:
            no_improve_rounds += 1
            if no_improve_rounds > 8:
                break
            # Small perturbation to escape
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.001)
            conv = fast_convolve(f)
            no_improve_rounds = 0

    return best_f, best_obj


def sg_polish(f0, time_limit, rng_seed=0):
    """Subgradient descent with adaptive step size and multi-peak weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    rng_local = np.random.default_rng(rng_seed)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    lr = 0.004
    mom = 0.9
    temp = 0.0001
    top_k = max(5, n // 8)

    while time.monotonic() - t0 < time_limit:
        # Find peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        # Softmax weighting
        pv = conv[peaks]
        scaled = pv / max(temp, 1e-10)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Momentum update
        v = mom * v - lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        lr *= 0.9995
        temp *= 1.003

        if no_improve > 200:
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def try_sequence(a, label, state):
    """Evaluate a sequence and update state if better."""
    a = np.maximum(a, 0.0)
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return score


def generate_diverse_inits(n, parent, rng, count=8):
    """Generate diverse initializations biased toward asymmetric shapes."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Asymmetric ramp
    a = (x + 1.0) ** 0.8
    inits.append(a / a.sum())

    # Power law decay
    a = (n - x + 1) ** 1.1
    inits.append(a / a.sum())

    # Off-center Gaussian (left)
    mu, sig = n * 0.3, n * 0.18
    a = np.exp(-0.5 * ((x - mu) / sig) ** 2)
    inits.append(a / a.sum())

    # Off-center Gaussian (right)
    mu, sig = n * 0.7, n * 0.18
    a = np.exp(-0.5 * ((x - mu) / sig) ** 2)
    inits.append(a / a.sum())

    # Two asymmetric bumps
    a = (np.exp(-((x - n * 0.2) ** 2) / (2 * (n * 0.1) ** 2)) +
         0.5 * np.exp(-((x - n * 0.75) ** 2) / (2 * (n * 0.14) ** 2)))
    inits.append(a / a.sum())

    # Sine window (naturally asymmetric in its effect on convolution)
    a = np.sin(np.pi * (x + 0.5) / n) ** 0.6
    inits.append(a / a.sum())

    # Random Dirichlet (low concentration = spread out)
    inits.append(rng.dirichlet(np.ones(n) * 0.4))

    # Parent interpolation with asymmetry
    if parent is not None and len(parent) > 0:
        px = np.interp(x, np.linspace(0, len(parent) - 1, n), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Reversed parent
        px_r = px[::-1]
        ps_r = px_r.sum()
        if ps_r > 1e-10:
            inits.append(px_r / ps_r)

    # Add noise to a few
    for i in range(min(2, len(inits))):
        noisy = np.maximum(inits[i] * (1 + 0.1 * rng.standard_normal(n)), 0)
        ns = noisy.sum()
        if ns > 1e-10:
            inits.append(noisy / ns)

    return inits[:count]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining FW-LP, pair-swap
    coordinate descent, and subgradient polishing with asymmetric
    initialization to find sequences with low peak autocorrelation.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(10.0, float(budget_s) - 20.0)
    deadline = time.monotonic() + total_budget

    state = {'best_seq': None, 'best_score': float('inf')}

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}")
            s = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(s):
                print(f"  Parent score: {s:.6f}")
                state['best_score'] = s
                state['best_seq'] = parent.copy()
    except Exception:
        pass

    # === Phase 1: Multi-init exploration (25% budget) ===
    print("=== Phase 1: Multi-init Exploration ===")
    phase1_time = total_budget * 0.25
    phase1_end = time.monotonic() + phase1_time

    # First, quick-optimize the parent
    if parent is not None:
        n_p = len(parent)
        t_quick = min(80, phase1_time * 0.4)
        if t_quick > 20:
            print(f"  Quick FW on parent (n={n_p}), t={t_quick:.0f}s")
            a_opt, _ = fw_lp_optimize(parent, time_limit=t_quick, rng_seed=seed)
            try_sequence(a_opt, "parent_fw", state)

    # Explore diverse initializations at the best length
    n_best_len = len(state['best_seq']) if state['best_seq'] is not None else 384

    # Also try a couple other lengths
    candidate_lengths = [n_best_len]
    for alt_n in [360, 420]:
        if alt_n not in candidate_lengths:
            candidate_lengths.append(alt_n)

    for n in candidate_lengths:
        if time.monotonic() > phase1_end - 30:
            break

        inits = generate_diverse_inits(n, parent, rng, count=6)
        n_to_try = min(4, len(inits))
        t_each = max(15, (phase1_end - time.monotonic() - 20) / n_to_try)
        t_each = min(t_each, 60)

        for i in range(n_to_try):
            if time.monotonic() > phase1_end - 15:
                break
            t_alloc = min(t_each, phase1_end - time.monotonic() - 10)
            if t_alloc < 10:
                break
            f_init = inits[i]
            a_opt, _ = fw_lp_optimize(f_init, time_limit=t_alloc, rng_seed=seed + i * 100)
            try_sequence(a_opt, f"n{n}_init{i}", state)

    print(f"  Phase 1 best: {state['best_score']:.6f}")

    # === Phase 2: Deep optimization (45% budget) ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    phase2_time = total_budget * 0.45
    phase2_end = time.monotonic() + phase2_time

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])

        # Main deep FW run
        t_main = min(phase2_time * 0.4, 200)
        if t_main > 30:
            print(f"  Deep FW (n={n_best}), t={t_main:.0f}s")
            a_opt, _ = fw_lp_optimize(state['best_seq'], time_limit=t_main,
                                       verbose=True, rng_seed=seed + 1000)
            try_sequence(a_opt, "deep_fw", state)

        # Pair-swap polish
        rem = phase2_end - time.monotonic() - 20
        if rem > 20:
            t_ps = min(rem * 0.4, 80)
            print(f"  Pair-swap polish, t={t_ps:.0f}s")
            a_opt, _ = pair_swap_polish(state['best_seq'], time_limit=t_ps, rng_seed=seed + 2000)
            try_sequence(a_opt, "pair_swap_1", state)

        # Perturbation + re-optimize
        perturbations = [0.003, 0.008, 0.015, 0.03]
        for ps in perturbations:
            rem = phase2_end - time.monotonic() - 15
            if rem < 20:
                break
            t_alloc = min(rem * 0.3, 60)
            if t_alloc < 10:
                break
            noise = rng.standard_normal(n_best) * ps
            f_init = project_simplex(state['best_seq'] + noise)
            a_opt, _ = fw_lp_optimize(f_init, time_limit=t_alloc, rng_seed=seed + int(ps * 10000))
            try_sequence(a_opt, f"pert_{ps}", state)

        # Mirror blend
        rem = phase2_end - time.monotonic() - 15
        if rem > 20:
            f_mirror = project_simplex(0.8 * state['best_seq'] + 0.2 * state['best_seq'][::-1])
            a_opt, _ = fw_lp_optimize(f_mirror, time_limit=min(rem, 50), rng_seed=seed + 5000)
            try_sequence(a_opt, "mirror", state)

    print(f"  Phase 2 best: {state['best_score']:.6f}")

    # === Phase 3: Fine polishing (25% budget) ===
    print(f"\n=== Phase 3: Fine Polishing ===")
    phase3_time = total_budget * 0.25
    phase3_end = time.monotonic() + phase3_time

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])

        # Extended pair-swap
        rem = phase3_end - time.monotonic() - 20
        if rem > 15:
            t_ps = min(rem * 0.5, 100)
            print(f"  Extended pair-swap, t={t_ps:.0f}s")
            a_opt, _ = pair_swap_polish(state['best_seq'], time_limit=t_ps, rng_seed=seed + 3000)
            try_sequence(a_opt, "pair_swap_2", state)

        # Subgradient polish
        rem = phase3_end - time.monotonic() - 15
        if rem > 15:
            t_sg = min(rem * 0.5, 60)
            print(f"  SG polish, t={t_sg:.0f}s")
            a_opt, _ = sg_polish(state['best_seq'], time_limit=t_sg, rng_seed=seed + 4000)
            try_sequence(a_opt, "sg_polish", state)

        # Final short FW
        rem = phase3_end - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = fw_lp_optimize(state['best_seq'], time_limit=min(rem, 40), rng_seed=seed + 6000)
            try_sequence(a_opt, "fw_final", state)

    print(f"  Phase 3 best: {state['best_score']:.6f}")

    # === Phase 4: Final squeeze (5% budget) ===
    print(f"\n=== Phase 4: Final Squeeze ===")
    if state['best_seq'] is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            # One more pair-swap
            a_opt, _ = pair_swap_polish(state['best_seq'], time_limit=min(rem * 0.5, 30),
                                         rng_seed=seed + 9000)
            try_sequence(a_opt, "final_ps", state)
            rem = deadline - time.monotonic() - 3
            if rem > 5:
                a_opt, _ = fw_lp_optimize(state['best_seq'], time_limit=min(rem, 20),
                                           rng_seed=seed + 9500)
                try_sequence(a_opt, "final_fw", state)

    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(384) / 384
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))
        print(f"  Fallback uniform: {state['best_score']:.6f}")

    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs

    print(f"\nFINAL SCORE: {state['best_score']:.6f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
