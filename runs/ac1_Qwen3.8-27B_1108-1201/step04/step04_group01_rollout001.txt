def lse_gradient(f, conv, beta):
    """Compute the LSE gradient: 2 * cross_correlation(softmax(beta*conv), f)."""
    n = len(f)
    # Numerically stable softmax

    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()

    conv_result = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv_result[n - 1 : n - 1 + n]

    result = np.fft.irfft(C * F, fft_size)
    return 4.0 * result[n - 1 : n - 1 + n]

        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v.copy())[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient(f, conv, beta):
    """Compute LSE gradient: 2 * cross_correlation(softmax(beta*conv), f)."""
    n = len(f)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    # grad[i] = 2 * sum_{m=0}^{n-1} w[i+m] * f[m]
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f[::-1], fft_size)
    result = np.fft.irfft(W * F, fft_size)
    return 2.0 * result[n - 1: n - 1 + n]


def variance_gradient(f, conv):
    """Gradient of sum(conv_k^2) w.r.t. f: 4 * cross_correlation(conv, f)."""
    n = len(f)
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    C = np.fft.rfft(conv, fft_size)
    F = np.fft.rfft(f[::-1], fft_size)
    result = np.fft.irfft(C * F, fft_size)
    return 4.0 * result[n - 1: n - 1 + n]


def build_lp_matrix(f):
    """Build sparse constraint matrix for LP: A[k,j] = f[k-j] if valid."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            ks = (k - js).astype(int)
            rows.extend(ks.tolist())
            cols.extend(js.tolist())
            vals.extend(f[ks].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns g or None."""
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 25})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, lp_timeout=25):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    M = best_obj

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M * m_exp)
        if g is None:
            m_exp = min(m_exp * 1.15, 8.0)
            if m_exp >= 8.0:
                m_exp = 1.0
                g = np.ones(n) / n
            else:
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 4.0)
            continue

        # Coarse line search
        best_alpha = 0.0
        best_val = best_obj
        alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Bisection refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = np.maximum((1.0 - mid) * f + mid * g, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 6.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.004 * min(1.0, (no_improve - 50) / 60.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj

def _lse_heavyball(f0, time_limit, beta_start, beta_end, var_weight,
                   best_f=None, best_obj=None):
    """LSE with heavy-ball momentum - the workhorse optimizer."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    if best_f is None:
        conv = fast_convolve(f)
        best_obj = conv.max()
        best_f = f.copy()
    n = len(f)

    v = np.zeros(n)
    momentum = 0.90
    beta = beta_start
    lr = 2.5 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        grad = lse_gradient(f, conv, beta)
        if var_weight > 0:
            grad = grad + var_weight * variance_gradient(f, conv)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad = grad / grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.001, 5.0 / np.sqrt(n))
        else:
            no_improve += 1
            if no_improve % 15 == 0:
                lr *= 0.93
                momentum = max(0.82, momentum - 0.003)
            if no_improve > 120:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(2.0 / np.sqrt(n), 0.002)
                momentum = 0.90
                beta = min(beta * 1.8, beta_end)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.004

        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-HB it={it}, obj={best_obj:.12f}, beta={beta:.1f}, "
                  f"lr={lr:.6f}, t={elapsed:.0f}s")

    return best_f, best_obj


def coord_descent_polish(f0, time_limit, n_iterations_max=50000):
    """Coordinate-wise descent: for each coordinate, find optimal value via 1D search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit and it < n_iterations_max:
        if no_improve > 5000:
            break
        it += 1
        # Pick a random coordinate
        i = rng.integers(0, n)
        # Try changing a_i while maintaining sum = 1
        # Redistribute: new_a_i = f[i] + delta, and scale others by (1-f[i]-delta)/(1-f[i])
        # Simpler: just try a few delta values
        base_conv_max = conv.max()
        best_delta = 0.0
        best_val = base_conv_max

        # Try small perturbations
        magnitudes = [0.001, 0.003, 0.01, 0.03]
        for mag in magnitudes:
            for sign in [1.0, -1.0]:
                delta = sign * mag * max(f[i], 0.01)
                if f[i] + delta < 0:
                    continue
                f_trial = f.copy()
                f_trial[i] += delta
                # Renormalize others to keep sum = 1
                others_sum = 1.0 - f_trial[i]
                if others_sum <= 0:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                if f[mask].sum() > 1e-15:
                    f_trial[mask] = f_trial[mask] * (others_sum / f[mask].sum())
                else:
                    f_trial[mask] = others_sum / (n - 1)
                f_trial = np.maximum(f_trial, 0.0)
                f_trial = f_trial / f_trial.sum()
                conv_t = fast_convolve(f_trial)
                val_t = conv_t.max()
                if val_t < best_val:
                    best_val = val_t
                    best_delta = delta
                    best_trial = f_trial.copy()

        if best_delta != 0.0:
            f = best_trial
            conv = fast_convolve(f)
            if conv.max() < best_obj - 1e-14:
                best_obj = conv.max()
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_f, best_obj


def diverse_initialization(n, seed=0):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian bump
    x = np.arange(n)
    sigma = n / 4.0
    g = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    inits.append(g / g.sum())

    # Cosine bump
    c = np.maximum(np.cos(np.pi * (x - n / 2) / n), 0) ** 2
    if c.sum() > 0:
        inits.append(c / c.sum())

    # Two bumps
    b1 = np.exp(-0.5 * ((x - n * 0.3) / (n * 0.15)) ** 2)
    b2 = np.exp(-0.5 * ((x - n * 0.7) / (n * 0.15)) ** 2)
    two = b1 + b2
    if two.sum() > 0:
        inits.append(two / two.sum())

    # Random
    r = rng.random(n) + 0.1
    inits.append(r / r.sum())

    # Power law
    pl = (x + 1) ** (-0.5)
    inits.append(pl / pl.sum())

    # Symmetric V-shape
    v = 1.0 - np.abs(2.0 * x / n - 1.0)
    inits.append(v / v.sum())

    return inits


def try_sequence(a, best_tracker, verbose=True):
    """Evaluate and track best."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        if verbose:
            print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score
