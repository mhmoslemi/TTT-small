        final_sequence_float = [float(x) for x in final_sequence]
        
        # Calculate the score of the final sequence
        score = evaluate_sequence(final_sequence_float)
    else:
        score = np.inf
        
    # Check if we have time to refine further
    remaining_time = budget_seconds - (time() - start_time)
    if remaining_time > 0:
        # If there's time, refine the sequence with additional evaluations
        for i in range(len(final_sequence_float)):
            final_sequence_float[i] = max(0, final_sequence_float[i])
            if random.random() > 0.5:
                # Mutate by adding a small random value
                delta = (1 if random.random() > 0.5 else -1) * random.uniform(0, 0.001)
                final_sequence_float[i] += delta
            else:
                # No mutation needed
                pass
            # Keep non-negative and cap
            final_sequence_float[i] = max(0, final_sequence_float[i])
            if final_sequence_float[i] > 1000:
                final_sequence_float[i] = 1000
            # Re-normalize if necessary? But the evaluator handles it.
            score_val = evaluate_sequence(final_sequence_float)
    else:
        score_val = score
        
    return score if score < score_val else score_val

def propose_candidate_function(height_sequence_1):

    Proposes a candidate sequence of non-negative coefficients for the given problem.
    
    The function takes the parent sequence (height_sequence_1) as input and generates a new
    sequence of length 448 using a multi-start optimization process.
    
    Args:
        height_sequence_1: A one-dimensional NumPy array representing the initial (parent) sequence.
    
    Returns:
        list[float]: The best sequence found during the optimization.
    """

    parent = height_sequence_1
    
    # Available time is 1800 seconds (includes time to find optimal shift)
    start_time = time()

    seq_len = parent.shape[0]
    init_candidates = combine(parent, seq_len)
    
    # Choose to use the best candidate among several evaluations
    best_candidate = None
    for candidate in init_candidates:
        try:
            candidate_score = evaluate_sequence(candidate.tolist())
            if np.isnan(candidate_score) or np.isinf(candidate_score):
                continue
                
            # Compare to our best
            if best_candidate is None or candidate_score < evaluate_sequence(best_candidate.tolist()):
                best_candidate = candidate
        except Exception:  # Skip any that fail
            continue
            
    # Now that we have one candidate, refine further if time allows
    if time.time() - start_time < 1800:
        n_phase = min(10, int(1800/(time.time() - start_time)))
        for i in range(n_phase):
            # Keep only three candidates to avoid bloat
            extend_candidates = extract_candidates(best_candidate, 3)
            for cand in extend_candidates:
                cand_score = evaluate_sequence(cand.tolist())
                cand_score = evaluate_sequence(cand_score)
                if cand_score < best_candidate_score:
                    best_candidate = cand
                    best_candidate_score = cand_score
                    
    return [float(x) for x in best_candidate.tolist() if x > 0]
