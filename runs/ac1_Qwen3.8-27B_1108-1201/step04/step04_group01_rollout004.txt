import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute 2*n*max(conv)/(sum)^2 without calling evaluate_sequence."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def coordinate_descent_pass(a, num_sweeps=1, seed=0):

    For each coordinate i, find the optimal value of a[i] (1D optimization).

            #                     - t_old^2 (if k=2i)
            # Simpler: just set a[i]=0, compute conv, then optimize
            a[i] = 0.0

            S = a.sum()  # sum without a[i]
            
            if S < 1e-15:
                a[i] = 1.0
                continue
            
            # Define f(t) = 2n * max_k(g_k(t)) / (S+t)^2
            # where g_k(t) = B[k] + 2t*a[k-i] (if 0<=k-i<n, k≠2i)
            #               g_{2i}(t) = B[2i] + t^2
            #               g_k(t) = B[k] (otherwise)
            
            # Compute coefficients
            # c[k] = coefficient of t in g_k(t) (linear part)
            # For k where 0<=k-i<n and k≠2i: c[k] = 2*a[k-i]
            # For k=2i: quadratic, g[t] = B[2i] + t^2
            
            # For the 1D optimization, I'll use a simple approach:
            # Evaluate f(t) at a set of candidate t values and pick the best
            
            # The optimal t should be near the current value or where the peak is minimized
            # Upper bound: t_max such that t^2 > (S+t)^2 * best_obj / (2n)
            # i.e., t^2 > best_obj/(2n) * (S+t)^2
            # This gives t > S * sqrt(best_obj/(2n)) / (1 - sqrt(best_obj/(2n)))
            # For practical purposes, t_max = max(10*t_old, S) should suffice
            
            t_upper = max(10.0 * max(t_old, 1e-6), S * 2.0, 1.0)
            
            # Grid search + refinement
            t_candidates = np.geomspace(1e-8, t_upper, 40)
            t_candidates = np.append(t_candidates, 0.0)
            
            best_t = t_old
            best_f = best_obj
            
            for t in t_candidates:
                # g_k(t) = B[k] + 2t*a[k-i] for valid k (0<=k-i<n, k≠2i)
                # g_{2i}(t) = B[2i] + t^2
                a[i] = t
                f_val = compute_obj(a)
                if f_val < best_f:
                    best_f = f_val
                    best_t = t
            
            # Refine around best_t
            if best_t > 0:
                t_lo = max(0.0, best_t * 0.8)
                t_hi = best_t * 1.2
                for t in np.linspace(t_lo, t_hi, 20):
                    a[i] = t
                    f_val = compute_obj(a)
                    if f_val < best_f:
                        best_f = f_val
                        best_t = t
            
            a[i] = best_t
            if best_f < best_obj - 1e-15:
                best_obj = best_f
    
    return a, best_obj

2. LSE smoothed gradient descent with adaptive beta for global optimization

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_obj_from_conv(a, conv):
    """Compute objective given a and its precomputed convolution."""
    n = len(a)
    s = a.sum()
    if s < 0.01:

    return 2.0 * n * conv.max() / (s * s)

            for t in t_grid:
                f = eval_at_t(t)
                if f < best_f:
                    best_f = f
                    best_t = t
            
            # Refine with finer grid around best_t
            if best_t > 1e-10:
                t_lo = max(0.0, best_t * 0.7)
                t_hi = best_t * 1.3
                t_fine = np.linspace(t_lo, t_hi, 25)
                for t in t_fine:
                    f = eval_at_t(t)
                    if f < best_f:
                        best_f = f
                        best_t = t
            elif best_t < 1e-8:
                # Try very small values
                t_fine = np.linspace(0.0, max(t_old, S_rest / n) * 0.5, 15)
                for t in t_fine:
                    f = eval_at_t(t)
                    if f < best_f:
                        best_f = f
                        best_t = t
            
            # Update
            if best_f < best_obj - 1e-15:
                # Update conv incrementally
                delta = best_t - t_old
                if abs(delta) > 1e-15:
                    conv[k_s:k_e + 1] = conv_base + 2.0 * best_t * avals
                    if 2 * i >= k_s and 2 * i <= k_e:
                        idx_2i = 2 * i - k_s
                        conv[k_s + idx_2i] = conv_base[idx_2i] + best_t * best_t
                    a[i] = best_t
                    S = S_rest + best_t
                    best_obj = best_f
                    best_a = a.copy()
                    best_conv = conv.copy()
                    improved = True
            
            # If no improvement, restore
            if not (best_f < best_obj - 1e-15):
                conv[k_s:k_e + 1] = conv_saved
                a[i] = t_old
                S = S_rest + t_old
        
        if verbose and sweep % 5 == 0:
            print(f"    CD sweep {sweep}/{num_sweeps}, obj={best_obj:.10f}, improved={improved}")
    
    return best_a, best_conv, best_obj


def lse_optimize(a0, time_limit, beta_start=30.0, beta_end=20000.0,
                 lr_start=None, momentum=0.92):
    """LSE smoothed gradient descent with adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    if lr_start is None:
        lr_start = 2.0 / np.sqrt(n)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    beta = beta_start
    lr = lr_start
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        
        # Softmax weights (numerically stable)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        
        # Gradient: grad[i] = 2 * sum_j w[i+j] * a[j]
        # = 2 * cross_correlation(a, w)[i]
        # Using FFT: conv(a[::-1], w)[n-1+i] = sum_j a[n-1-j]*w[n-1+i+j-(n-1)] 
        # Let me use: grad[i] = 2 * (a_rev * w)[n-1+i] where * is convolution
        # a_rev = a[::-1]
        # (a_rev * w)[m] = sum_j a_rev[j] * w[m-j] = sum_j a[n-1-j] * w[m-j]
        # Let k = n-1-j: = sum_k a[k] * w[m-n+1+k]
        # We want sum_k a[k] * w[i+k], so m-n+1 = i, m = i+n-1
        # grad[i] = 2 * (a_rev * w)[i+n-1]
        
        a_rev = a[::-1]
        total_len = 2 * n - 1 + n - 1
        fft_size = 1 << (total_len - 1).bit_length()
        A_rev = np.fft.rfft(a_rev, fft_size)
        W = np.fft.rfft(w, fft_size)
        conv_cross = np.fft.irfft(A_rev * W, fft_size)
        grad = 2.0 * conv_cross[n - 1 : 2 * n - 1]
        
        # Momentum + step
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        
        obj_new = 2.0 * n * fast_convolve(a_new).max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            lr = min(lr * 1.002, 6.0 / np.sqrt(n))
        else:
            no_improve += 1
            if no_improve % 15 == 0:
                lr *= 0.94
            if no_improve > 80:
                a = best_a.copy()
                v = np.zeros(n)
                lr = max(lr_start * 0.6, 0.001)
                beta = min(beta * 1.5, beta_end)
                no_improve = 0
        
        if beta < beta_end:
            beta *= 1.003
        
        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it}, obj={best_obj:.10f}, beta={beta:.0f}, lr={lr:.5f}, t={elapsed:.0f}s")
    
    return best_a, best_obj
