"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Frank-Wolfe,
smooth gradient descent, and adaptive local search.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build the constraint matrix for the LP.
    A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    Returns A_ub (shape 2n-1 x n) and b_ub (shape 2n-1).
    """
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, time_limit=60):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = build_lp_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def smooth_gradient(f, conv, beta=100):
    """Compute smooth gradient of log-sum-exp approximation of max(conv)."""
    n = len(f)
    b = conv
    max_b = b.max()
    scaled = beta * (b - max_b)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient of (1/beta)*log(sum exp(beta*b[k])) w.r.t. a[j]
    # = sum_k p[k] * 2*a[k-j] (for valid j)
    grad = np.zeros(n)
    for k in range(len(b)):
        if p[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * p[k] * f[k - idx]
    return grad

            for _ in range(20):

                    break
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                ft = ft / ft.sum()
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    FW-LP it={it}, obj={best_obj:.10f}, alpha={best_alpha:.4f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 50:
            # Random restart
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
    
    return best_f, best_obj


def smooth_descent(f0, time_limit, beta=50, lr=0.01, mom=0.9, 
                   lp_every=100, verbose=True):
    """Smooth gradient descent with momentum and periodic LP escapes."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_beta = beta
    cur_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break
        
        grad = smooth_gradient(f, conv, beta=cur_beta)
        v = mom * v - cur_lr * grad
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Anneal beta and decay lr
        cur_beta *= 1.01
        cur_lr *= 0.9995
        
        # LP escape
        if lp_every > 0 and no_improve > lp_every and rem > 30:
            g = solve_lp(f, best_obj * 1.01, time_limit=min(20, rem - 10))
            if g is not None:
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs
                    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        ft = project_simplex((1 - alpha) * f + alpha * g)
                        c_try = fast_convolve(ft).max()
                        if c_try < best_obj:
                            best_obj = c_try
                            best_f = ft.copy()
                            f = ft
                            conv = fast_convolve(f)
                            break
            no_improve = 0
            cur_beta = beta
            cur_lr = lr
            v = np.zeros(n)
        
        if no_improve > 500:
            rng = np.random.default_rng(it * 13 + 1)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            v = np.zeros(n)
            cur_beta = beta
            cur_lr = lr
            no_improve = 0
    
    return best_f, best_obj


def local_search(f0, time_limit, n_perturb=100, verbose=True):
    """Local search: try many small perturbations and keep improvements."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break
        
        # Try random perturbations
        improved = False
        for _ in range(n_perturb):
            rng = np.random.default_rng(hash((it, f.tobytes()[:100])) % 2**32)
            # Different perturbation strategies
            strategy = rng.integers(0, 4)
            
            if strategy == 0:
                # Gaussian perturbation
                f_try = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            elif strategy == 1:
                # Coordinate perturbation
                j = rng.integers(0, n)
                f_try = best_f.copy()
                delta = rng.uniform(-0.01, 0.01)
                f_try[j] += delta
                f_try = project_simplex(f_try)
            elif strategy == 2:
                # Swap perturbation
                j1, j2 = rng.integers(0, n, size=2)
                f_try = best_f.copy()
                delta = rng.uniform(0.001, 0.01)
                f_try[j1] += delta
                f_try[j2] -= delta
                f_try = np.maximum(f_try, 0)
                f_try = f_try / f_try.sum()
            else:
                # Peak-targeted: reduce the tallest peak
                conv_try = fast_convolve(best_f)
                k_peak = np.argmax(conv_try)
                f_try = best_f.copy()
                # Reduce elements that contribute to the peak
                js = max(0, k_peak - n + 1)
                je = min(n - 1, k_peak)
                for j in range(js, je + 1):
                    f_try[j] *= 0.999
                f_try = project_simplex(f_try)
            
            obj_try = fast_convolve(f_try).max()
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_f = f_try.copy()
                improved = True
        
        if not improved and it > 100:

            pass
    
    return best_f, best_obj
