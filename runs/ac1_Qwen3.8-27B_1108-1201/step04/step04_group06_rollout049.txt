    scaled = peak_vals / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    s = w.sum()
    if s < 1e-15:
        w = np.zeros(k_use)
        w[0] = 1.0
    else:
        w /= s

    grad = np.zeros(n)
    for idx in range(k_use):
        if w[idx] < 1e-12:
            continue
        k = int(sorted_idx[idx])
        # db[k]/da[j] = 2*a[k-j] if 0 <= k-j < n
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            idxs = np.arange(j_min, j_max + 1)
            grad[idxs] += w[idx] * 2.0 * a[k - idxs]

    return grad


def nesterov_optimize(a0, time_limit, lr_init=0.02, momentum=0.9,
                       top_k=40, temp_init=0.002, temp_decay=1.004):
    """Nesterov-accelerated gradient descent with adaptive parameters."""
    a = normalize(a0.copy())
    n = len(a)
    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = compute_objective(a)
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + mom * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy

        conv_y = fast_convolve(y)
        grad = gradient_step(y, conv_y, lr, top_k=top_k, temperature=temp)

        # Update
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn > 1e-15:
            a = a_new / sn
        else:
            a = best_a.copy()
            v = np.zeros(n)
            continue

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        iter_count += 1

        # Adaptive updates
        lr *= 0.9997
        temp *= temp_decay
        mom = min(0.97, mom + 0.0001)

        if no_improve >= 300:
            break

    return best_a, best_obj


def mirror_descent_step(a, conv, beta, damp=0.5):
    """
    One mirror descent step: a_i <- (1-damp)*a_i + damp * exp(-beta * cross_i)
    where cross_i = sum_k (soft_lam)_k * a[k-i]
    """
    n = len(a)
    # Compute soft lambda weights
    top_k = max(10, n // 8)
    sorted_idx = np.argsort(conv)[::-1][:top_k]
    peak_vals = conv[sorted_idx]
    temp = max(1e-8, peak_vals[0] * 0.01)
    w = peak_vals / temp
    w -= w.max()
    w = np.exp(w)
    w /= w.sum()

    # Compute cross[i] = sum_k w_k_normalized * a[k-i]
    # where w is placed at positions sorted_idx
    lam = np.zeros(2 * n - 1)
    for k, wi in zip(sorted_idx, w):
        lam[int(k)] = wi

    # cross[i] = sum_j a[j] * lam[j+i]
    # = (flip(a) * lam)[n-1+i] using convolution
    a_flip = a[::-1]
    conv_fl = fast_convolve_xy(a_flip, lam)
    cross = conv_fl[n - 1: n - 1 + n]

    # Exponential update
    scaled = -beta * cross
    scaled -= scaled.max()
    new_part = np.exp(scaled)
    new_part = normalize(new_part)

    a_new = (1 - damp) * a + damp * new_part
    return normalize(a_new)


def mirror_descent_optimize(a0, time_limit, beta_init=2.0, beta_growth=1.005, damp=0.5):
    """Mirror descent with exponential family updates."""
    a = normalize(a0.copy())
    n = len(a)
    best_a = a.copy()
    best_obj = compute_objective(a)
    beta = beta_init

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        a = mirror_descent_step(a, conv, beta, damp=damp)

        cur_obj = compute_objective(a)
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        beta *= beta_growth
        if no_improve > 500:
            break

    return best_a, best_obj


def lp_gradient_hybrid(a0, time_limit, use_cutting=True):
    """
    Hybrid LP + gradient optimization.
    Alternates between LP steps (for direction) and gradient steps (for fine-tuning).
    """
    a = normalize(a0.copy())
    n = len(a)
    conv = fast_convolve(a)
    best_obj = compute_objective(a)
    best_a = a.copy()

    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0
    lp_fail = 0
    perturb_count = 0
    rng_local = np.random.default_rng(int(time.monotonic() * 1000) % 100000)

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Try LP step
        if use_cutting and n <= 2048:
            g, success = lp_step_cutting(a, time_per_lp=min(3.0, remaining * 0.1))
        else:
            g, success = lp_step_full(a)

        if success and g.sum() > 1e-10:
            g = normalize(g)
            d = g - a
            norm_d = np.linalg.norm(d)

            if norm_d > 1e-10:
                # Backtracking line search
                best_t = 0.0
                best_a_try = a.copy()
                best_obj_try = best_obj
                t = 1.0
                for _ in range(10):
                    a_try = a + t * d
                    a_try = normalize(a_try)
                    obj_try = compute_objective(a_try)
                    if obj_try < best_obj_try - 1e-12:
                        best_obj_try = obj_try
                        best_a_try = a_try.copy()
                        best_t = t
                        break
                    t *= 0.5
                    if t < 0.001:
                        break

                if best_t > 0:
                    a = best_a_try
                    conv = fast_convolve(a)
                    if compute_objective(a) < best_obj - 1e-12:
                        best_obj = compute_objective(a)
                        best_a = a.copy()
                    no_improve = 0
                    lp_fail = 0
                else:
                    lp_fail += 1
            else:
                lp_fail += 1
        else:
            lp_fail += 1

        # If LP is struggling, do gradient steps
        if lp_fail >= 3 or not success:
            a_curr = gradient_short(a, time_per_step=min(2.0, remaining * 0.05),
                                    top_k=50, temp=0.002, lr=0.01)
            obj_curr = compute_objective(a_curr)
            if obj_curr < best_obj - 1e-12:
                best_obj = obj_curr
                best_a = a_curr.copy()
                a = a_curr.copy()
            lp_fail = 0
            no_improve = 0

        no_improve += 1
        iter_count += 1

        # Perturbation on stagnation
        if no_improve >= 50:
            scale = 0.002 + 0.003 * min(perturb_count, 10)
            noise = rng_local.standard_normal(n) * scale
            a = best_a + noise
            a = normalize(a)
            conv = fast_convolve(a)
            no_improve = 0
            perturb_count += 1
            if perturb_count > 60:
                break

    return best_a, best_obj


def gradient_short(a, time_per_step, top_k=50, temp=0.002, lr=0.01):
    """Short burst of gradient steps."""
    a = normalize(a.copy())
    n = len(a)
    t0 = time.monotonic()
    conv = fast_convolve(a)

    while time.monotonic() - t0 < time_per_step:
        grad = gradient_step(a, conv, lr, top_k=top_k, temperature=temp)
        a_new = a - lr * grad
        a = normalize(a_new)
        conv = fast_convolve(a)
        lr *= 0.99

    return a


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Exponential decay (various rates)
    for r_name, r in [('exp_fast', 0.97), ('exp_med', 0.99), ('exp_slow', 0.998)]:
        a = r ** x
        inits[r_name] = normalize(a)

    # Power law decay
    for alpha in [0.3, 0.5, 0.8]:
        a = (x + 1.0) ** (-alpha)
        inits[f'pow_{alpha}'] = normalize(a)

    # Shifted triangular (asymmetric)
    peak = int(0.3 * n)
    tri = np.zeros(n)
    for i in range(n):
        tri[i] = max(0, 1 - abs(i - peak) / (0.7 * n))
    tri = np.maximum(tri, 0.01)
    inits['tri_shift'] = normalize(tri)

    # Sine (centered)
    s1 = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine'] = normalize(s1)

    # Random (Dirichlet)
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Bimodal (mass at edges)
    edge = max(1, n // 6)
    bi = np.zeros(n)
    bi[:edge] = np.linspace(1.0, 0.1, edge)
    bi[-edge:] = np.linspace(0.1, 1.0, edge)
    bi += 0.01
    inits['bimodal'] = normalize(bi)

    # Parent-based
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                inits['parent_interp'] = normalize(pp)

    return inits


def extend_sequence(a_short, n_target):
    """Extend a sequence to target length via interpolation."""
    n_short = len(a_short)
    a_norm = normalize(a_short)
    a_ext = np.interp(np.linspace(0, 1, n_target),
                      np.linspace(0, 1, n_short), a_norm)
    return normalize(a_ext)


def coordinate_polish(a0, time_limit):
    """Local coordinate descent polish."""
    a = normalize(a0.copy())
    n = len(a)
    best_a = a.copy()
    best_obj = compute_objective(a)
    rng = np.random.default_rng(99999)
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 500:
            break
        targets = []
        for _ in range(min(100, max(10, 200000 // n))):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            targets.append((i, j))

        for (i, j) in targets:
            for delta in [0.002, 0.005, 0.01, 0.02, -0.002, -0.005, -0.01, -0.02]:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                a_try = normalize(a_try)
                obj_try = compute_objective(a_try)
                if obj_try < best_obj - 1e-10:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    no_improve = 0
                    break
            else:
                no_improve += 1

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid LP + gradient + mirror descent search
    with cutting-plane LP acceleration and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(parent)}, score~{compute_objective(parent):.6f}")
    except Exception:
        pass

    # =========================================================
    # Phase 1: Diverse exploration across lengths (250s)
    # =========================================================
    print("=== Phase 1: Diverse Exploration ===")
    t_start = time.monotonic()
    phase1_budget = 250.0

    lengths = [384, 512, 768, 1024]
    scan_results = []

    for n in lengths:
        if time.monotonic() > deadline - 600:
            break
        inits = generate_inits(n, parent, rng)

        # Prioritize promising inits
        priority = ['parent', 'parent_interp', 'exp_med', 'exp_slow',
                    'pow_0.5', 'tri_shift', 'sine', 'random']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:5]

        t_per_init = max(8.0, (phase1_budget - 10) / (len(lengths) * 5))

        for name in init_names:
            if time.monotonic() > deadline - 500:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 300)
            if t_alloc < 5:
                break

            a0 = inits[name]
            a_opt, obj = nesterov_optimize(a0, time_limit=t_alloc,
                                           lr_init=0.015, momentum=0.91,
                                           top_k=50, temp_init=0.002,
                                           temp_decay=1.003)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Explore n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: Deep LP+gradient hybrid on top candidates (1100s)
    # =========================================================
    print("=== Phase 2: Deep Hybrid Optimization ===")
    t_start2 = time.monotonic()

    # Collect top unique candidates
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n,)
        if key not in seen:
            top_configs.append((n, a_opt))
            seen.add(key)
        if len(top_configs) >= 4:
            break

    if best_sequence is not None and len(best_sequence) not in seen:
        top_configs.append((len(best_sequence), best_sequence.copy()))

    remaining = deadline - time.monotonic() - 200.0
    n_cands = max(1, len(top_configs))
    t_per_cand = max(30.0, remaining * 0.7 / n_cands)

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 300:
            break
        t_alloc = min(t_per_cand, deadline - time.monotonic() - 200)
        if t_alloc < 15:
            break

        print(f"  Hybrid cand {i}: n={n}, obj={compute_objective(a0):.6f}, t={t_alloc:.0f}s")

        # Run hybrid LP+gradient
        a_opt, obj = lp_gradient_hybrid(a0, time_limit=t_alloc * 0.7, use_cutting=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Hybrid {i} (n={n}): {score:.6f} *NEW BEST*")
        else:
            print(f"  Hybrid {i} (n={n}): {score:.6f}")

        # Also try mirror descent on the result
        if time.monotonic() < deadline - 250:
            t_md = min(30.0, deadline - time.monotonic() - 200)
            if t_md > 10:
                a_md, _ = mirror_descent_optimize(a_opt, time_limit=t_md,
                                                   beta_init=3.0, beta_growth=1.003)
                score_md = float(evaluate_sequence(a_md.tolist()))
                if np.isfinite(score_md) and score_md < best_score:
                    best_score = score_md
                    best_sequence = a_md.copy()
                    print(f"  Mirror {i} (n={n}): {score_md:.6f} *NEW BEST*")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Focused LP polish on best (400s)
    # =========================================================
    print("=== Phase 3: Focused LP Polish ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 100.0
        if remaining > 30:
            for pi in range(4):
                if time.monotonic() > deadline - 80:
                    break
                t_alloc = min(remaining / (4 - pi) * 0.7, deadline - time.monotonic() - 60)
                if t_alloc < 10:
                    break

                if pi == 0:
                    a0 = best_sequence.copy()
                elif pi == 1:
                    noise = rng.standard_normal(len(best_sequence)) * 0.002
                    a0 = normalize(best_sequence + noise)
                elif pi == 2:
                    noise = rng.standard_normal(len(best_sequence)) * 0.005
                    a0 = normalize(best_sequence + noise)
                else:
                    # Try a different init mixed with best
                    n_best = len(best_sequence)
                    alt = rng.dirichlet(np.ones(n_best) * 0.3)
                    a0 = normalize(0.7 * best_sequence + 0.3 * alt)

                a_opt, obj = lp_gradient_hybrid(a0, time_limit=t_alloc, use_cutting=True)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Polish {pi}: {score:.6f} *NEW BEST*")
                else:
                    print(f"  Polish {pi}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Final fine-tuning (remaining - 20s)
    # =========================================================
    print("=== Phase 4: Final Fine-tuning ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0

        # Mirror descent polish
        t_md = min(remaining * 0.3, 60.0)
        if t_md > 5:
            a_opt, _ = mirror_descent_optimize(best_sequence, time_limit=t_md,
                                                beta_init=5.0, beta_growth=1.004, damp=0.3)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final Mirror: {score:.6f} *NEW BEST*")

        # Short Nesterov with very low temp
        remaining = deadline - time.monotonic() - 10.0
        t_ng = min(remaining * 0.3, 40.0)
        if t_ng > 5:
            a_opt, _ = nesterov_optimize(best_sequence, time_limit=t_ng,
                                         lr_init=0.005, momentum=0.95,
                                         top_k=80, temp_init=0.0005,
                                         temp_decay=1.002)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final Nesterov: {score:.6f} *NEW BEST*")

        # Coordinate descent
        remaining = deadline - time.monotonic() - 10.0
        t_cd = min(remaining * 0.8, 30.0)
        if t_cd > 5:
            a_opt, _ = coordinate_polish(best_sequence, time_limit=t_cd)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CoordDesc: {score:.6f} *NEW BEST*")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512

    # Final verification
    final_score = float(evaluate_sequence([float(x) for x in best_sequence]))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
