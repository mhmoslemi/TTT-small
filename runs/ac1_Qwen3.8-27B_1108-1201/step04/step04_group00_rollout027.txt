Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using subgradient optimization with
soft-max smoothing, momentum, adaptive step size, and multi-length search.
Uses larger sequence lengths (up to 4096) for better optimization surface.

import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    r = np.nonzero(cond)[0][-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_subgradient(f, conv, temperature, top_k=100):
    """Compute softmax-weighted subgradient of the convolution."""
    n = len(f)
    # Use top-k peaks for efficiency
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    # Softmax weights over the selected peaks
    vals = conv[peaks] / temperature
    vals -= vals.max()
    w = np.exp(vals)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        int_max = np.argmax(vals)
        w[int_max] = 1.0

        je = min(n - 1, k)
        m_arr = np.arange(js, je + 1)
        sg[m_arr] += 2.0 * w[i] * f[k - m_arr]

def gaussian_init(n, sigma_frac=0.3):
    """Initialize with a Gaussian-like shape."""
    j = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def uniform_init(n):
    return np.ones(n) / n

def optimize_subgradient(f0, time_limit, lr=0.01, momentum=0.9, top_k=100, temp=0.001):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp

        conv = fast_convolve(f)
        M = conv.max()

        if sg_norm > 1e-15:
            sg = sg / sg_norm
        
        # Momentum update
        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        
        # Evaluate
        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()
        
        if M_new < M - 1e-14:
            f = f_new
            if M_new < best_obj - 1e-14:

                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

            cur_lr *= 0.9
            v *= 0.5

            v = np.zeros(n)
            cur_lr = lr * 0.5
            no_improve = 0

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-14:
                continue
            
            best_j_val = old_val
            best_j_obj = fast_convolve(f).max()
            
            multipliers = [0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 2:
            break
    
    return best_f, best_obj

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    r = np.nonzero(cond)[0][-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_sg(f, conv, top_k, soft_temp):
    """Compute soft-max weighted subgradient."""
    n = len(f)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]

    vals -= vals.max()
    w = np.exp(vals)
    ws = w.sum()

        w = np.zeros(len(peaks))
        w[imax] = 1.0

        je = min(n - 1, k)
        m_arr = np.arange(js, je + 1)
        sg[m_arr] += 2.0 * w[i] * f[k - m_arr]

def softmax_parameterize(z):
    z = z - z.max()
    a = np.exp(z)
    return a / a.sum()

def objective_z(z, T):
    f = softmax_parameterize(z)
    conv = fast_convolve(f)
    vals = conv / T
    vals -= vals.max()
    log_sum = np.log(np.exp(vals).sum()) + vals.max()
    obj = T * log_sum
    
    # Gradient w.r.t. z: chain rule
    # d obj / d z = d obj / d f * d f / d z
    # d f / d z is the Jacobian of softmax: diag(f) - outer(f, f)
    # d obj / d f is the gradient we computed above
    
    n = len(f)
    f_grad = np.zeros(n)  # d obj / d f
    weights = np.exp(vals) / np.exp(vals).sum()
    for i in range(len(conv)):
        if weights[i] < 1e-12:
            continue
        k = i
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        m_arr = np.arange(js, je + 1)
        f_grad[m_arr] += 2.0 * weights[i] * f[k - m_arr]
    
    # d f / d z: the Jacobian of softmax is J[i,j] = f[i](delta_ij - f[j])
    # So (J^T * f_grad)[i] = f_grad[i] * f[i] - f[i] * sum_j f_grad[j] * f[j]
    # = f[i] * (f_grad[i] - dot(f_grad, f))
    dot = np.dot(f_grad, f)
    z_grad = f * (f_grad - dot)
    
    return obj, z_grad

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(np.asarray(v, dtype=np.float64))[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    r = np.nonzero(cond)[0][-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(np.asarray(v, dtype=np.float64) - theta, 0.0)
