4. If not, identify the most violated constraint and remove it (or add a new one)
5. Repeat

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
- Fast subgradient descent with momentum and adaptive step sizing
- Coordinate descent with analytic optimal search
- Multi-length exploration with diverse initializations
- LP-based Frank-Wolfe for final polishing

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(conv, a, top_k=5, temperature=0.001):
    """Compute weighted subgradient of max_k conv[k] w.r.t. a."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    # Softmax weights over peaks
    pv = conv[peaks]
    scaled = (pv - pv.max()) / max(temperature, 1e-10)
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
    else:
        w /= ws
    
    # Gradient: for each peak k, grad[t] += 2 * w_i * a[k-t] if valid
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        # Valid t: 0 <= t < n and 0 <= k-t < n
        # i.e., max(0, k-n+1) <= t <= min(n-1, k)
        t_lo = max(0, k - n + 1)
        t_hi = min(n - 1, k)
        if t_lo > t_hi:
            continue
        # grad[t] = 2 * w_i * a[k-t]
        # a[k-t] for t in [t_lo, t_hi] is a[k-t_hi : k-t_lo+1][::-1]
        idx = np.arange(t_lo, t_hi + 1)
        sg[t_lo:t_hi+1] += 2.0 * w[i] * a[k - idx]
    
    return sg

def subgradient_optimize(f0, time_limit, top_k=10, temperature=0.001, 
                         lr=0.01, momentum=0.9, verbose=False):
    """Subgradient descent with momentum for minimizing max_conv."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        
        sg = compute_subgradient(conv, f, top_k=top_k, temperature=cur_temp)
        
        # Momentum update
        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            f = f_new  # still move (for exploration)
        
        # Adaptive learning rate
        cur_lr *= 0.9995
        cur_temp *= 1.005
        
        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng(it * 7919)
            noise_scale = min(0.01, 0.001 * (no_improve // 100))
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.001)
            cur_temp = temperature
            no_improve = 0
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter {it}: obj={best_obj:.10f}, lr={cur_lr:.6f}, t={elapsed:.0f}s")
    
    return best_f, best_obj

def coord_descent_analytic(f0, time_limit, verbose=False):
    """
    Coordinate descent where for each coordinate j, we find the optimal value
    by solving a 1D problem (since fixing all other coords makes the objective
    a quadratic in a[j]).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            
            # If we set f[j] = x, the rest need to sum to 1-x
            # The convolution at position k involves terms a[j]*a[k-j] and a[k-j]*a[j]
            # = 2*a[j]*a[k-j] (for the cross terms when j != k-j)
            # or a[j]^2 (when k-j == j, i.e., k == 2j)
            
            # For a fixed j and value x:
            # conv[k] depends on x through terms: a[j]*a[k-j] + a[k-j]*a[j] = 2*a[k-j]*x
            # (when k-j != j) or a[j]^2 = x^2 (when k-j == j, i.e., k == 2j)
            
            # But we also need to rescale the other elements to maintain sum=1.
            # If f[j] = x, other elements scale by (1-x)/(1-old_val).
            
            # This makes the problem more complex. Let's use a simpler approach:
            # Try a set of candidate values and pick the best.
            
            if old_val < 1e-14:
                # Try setting to a small positive value
                candidates = [0.001, 0.01, 0.02]
            else:
                candidates = [old_val * m for m in [0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0, 3.0]]
                candidates.append(0.0)
            
            best_x = old_val
            best_x_obj = fast_convolve(f).max()
            
            for x in candidates:
                x = max(0.0, x)
                if x >= 1.0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    if x < 1e-10:
                        f_trial = f.copy()
                        f_trial[j] = 0
                        f_trial = f_trial / f_trial.sum()
                    else:
                        continue
                else:
                    scale = (1.0 - x) / denom
                    f_trial = f.copy()
                    f_trial[j] = x
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = np.maximum(f_trial, 0)
                    ss = f_trial.sum()
                    if ss < 1e-15:
                        continue
                    f_trial = f_trial / ss
                
                val = fast_convolve(f_trial).max()
                if val < best_x_obj - 1e-15:
                    best_x_obj = val
                    best_x = x
            
            if best_x != old_val:
                # Apply the update
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_x) / denom
                    f[j] = best_x
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    ss = f.sum()
                    if ss > 1e-15:
                        f = f / ss
                
                if best_x_obj < best_obj - 1e-14:
                    best_obj = best_x_obj
                    best_f = f.copy()
                    improved = True

            break
        if verbose and passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass {passes}: obj={best_obj:.10f}, t={elapsed:.0f}s")
    
    return best_f, best_obj

    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid strategy combining:
- Fast projected subgradient descent with momentum and adaptive step sizing
- LP-based Frank-Wolfe for structured direction finding
- Analytic coordinate descent with exhaustive multiplier search
- Multi-length exploration with diverse initializations (cosine, geometric, sparse)
- Iterative interpolation between scales for progressive refinement
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(conv, a, top_k=10, temperature=0.001):
    """Compute weighted subgradient of max_k conv[k] w.r.t. a."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = (pv - pv.max()) / max(temperature, 1e-10)
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[0] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        t_lo = max(0, k - n + 1)
        t_hi = min(n - 1, k)
        if t_lo > t_hi:
            continue
        idx = np.arange(t_lo, t_hi + 1)
        sg[t_lo:t_hi + 1] += 2.0 * w[i] * a[k - idx]
    return sg


def subgradient_optimize(f0, time_limit, top_k=10, temperature=0.001,
                         lr=0.02, momentum=0.85, verbose=False):
    """Projective subgradient descent with momentum and adaptive scheduling."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    cur_topk = top_k
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        sg = compute_subgradient(conv, f, top_k=cur_topk, temperature=cur_temp)
        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        f = f_new
        cur_lr *= 0.9997
        cur_temp *= 1.004
        cur_topk = max(5, min(50, cur_topk + (1 if it % 50 == 0 else 0)))
        if no_improve > 150:
            rng = np.random.default_rng(it * 7919 + 3)
            noise_scale = min(0.015, 0.002 * (1 + no_improve // 100))
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.4, 0.003)
            cur_temp = temperature
            cur_topk = top_k
            no_improve = 0
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter {it}: obj={best_obj:.10f}, lr={cur_lr:.5f}, t={elapsed:.0f}s")
    return best_f, best_obj


def lpfw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive constraint relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    while time.monotonic() - t0 < time_limit:
        it += 1
        c_vec = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            m_exp *= 1.15
            if m_exp > 20.0:
                m_exp = 1.0
                sg = compute_subgradient(conv, f, top_k=5, temperature=0.001)
                g = project_simplex(f - 0.01 * sg)
            else:
                continue
        else:
            g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.1, 10.0)
            continue
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        if best_alpha > 1e-10:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 30 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.12, 20.0)
        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    return best_f, best_obj


def coord_descent_optimize(f0, time_limit, verbose=False):
    """Coordinate descent with multipliers and rescaling."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    multipliers_coarse = [0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 3.0]
    multipliers_fine = [0.7, 0.85, 1.0, 1.15, 1.3]
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        muls = multipliers_coarse if passes <= 3 else multipliers_fine
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-14:
                candidates = [0.001, 0.005, 0.01]
            else:
                candidates = [old_val * m for m in muls]
                candidates.append(0.0)
            best_x = old_val
            best_x_obj = best_obj
            denom = 1.0 - old_val
            for x in candidates:
                x = max(0.0, x)
                if x >= 0.999:
                    continue
                if denom < 1e-10:
                    if x < 1e-12:
                        f_trial = f.copy()
                        f_trial[j] = 0.0
                        ss = f_trial.sum()
                        if ss < 1e-15:
                            continue
                        f_trial = f_trial / ss
                    else:
                        continue
                else:
                    scale = (1.0 - x) / denom
                    f_trial = f.copy()
                    f_trial[j] = x
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = np.maximum(f_trial, 0.0)
                    ss = f_trial.sum()
                    if ss < 1e-15:
                        continue
                    f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_x_obj - 1e-15:
                    best_x_obj = val
                    best_x = x
            if best_x != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_x) / denom
                    f[j] = best_x
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0.0)
                    ss = f.sum()
                    if ss > 1e-15:
                        f = f / ss
                if best_x_obj < best_obj - 1e-14:
                    best_obj = best_x_obj
                    best_f = f.copy()
                    improved = True
        if not improved and passes > 3:
            break
        if verbose and passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass {passes}: obj={best_obj:.10f}, t={elapsed:.0f}s")
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def generate_initializer(n, method, rng):
    """Generate a diverse initial sequence of length n."""
    if method == 'uniform':
        return np.ones(n) / n
    elif method == 'cosine':
        x = np.arange(n) / n
        a = 0.5 + 0.5 * np.cos(np.pi * x)
        a = np.maximum(a, 1e-12)
        a /= a.sum()
        return a
    elif method == 'raised_cosine':
        x = np.arange(n) / n
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * x)
        a = np.maximum(a, 1e-12)
        a /= a.sum()
        return a
    elif method == 'gaussian':
        x = np.arange(n) / n
        sig = rng.uniform(0.2, 0.4)
        a = np.exp(-0.5 * ((x - 0.5) / sig) ** 2)
        a = np.maximum(a, 1e-12)
        a /= a.sum()
        return a
    elif method == 'geometric':
        r = rng.uniform(0.92, 0.999)
        a = r ** np.arange(n)
        a /= a.sum()
        return a
    elif method == 'symmetric_geometric':
        half = n // 2
        r = rng.uniform(0.9, 0.99)
        a = np.zeros(n)
        for i in range(half):
            a[i] = r ** i
            a[n - 1 - i] = r ** i
        a = np.maximum(a, 1e-12)
        a /= a.sum()
        return a
    elif method == 'sparse':
        a = np.zeros(n)
        n_active = max(2, int(n * rng.uniform(0.3, 0.7)))
        positions = rng.choice(n, size=n_active, replace=False)
        a[positions] = 1.0
        a /= a.sum()
        return a
    elif method == 'chirp':
        x = np.arange(n) / n
        bandwidth = rng.uniform(2.0, 8.0)
        a = np.abs(np.sin(np.pi * bandwidth * x)) ** 2
        a = np.maximum(a, 1e-12)
        a /= a.sum()
        return a
    else:
        return np.ones(n) / n

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384

    # === Phase 0: Diverse initializations (30s) ===
    print("=== Phase 0: Diverse initializations ===")
    init_budget = min(30.0, (deadline - time.monotonic()) * 0.05)
    init_candidates = []
    n_parent = len(parent)

    methods = ['uniform', 'cosine', 'raised_cosine', 'gaussian',
               'geometric', 'symmetric_geometric', 'sparse', 'chirp']
    methods_per = max(1, int(init_budget / (len(methods) * 0.5)))

    for method in methods:
        for trial in range(min(methods_per, 3)):
            if time.monotonic() > deadline - init_budget:
                break
            n_init = n_parent
            a_init = generate_initializer(n_init, method, rng)
            score = float(evaluate_sequence(a_init.tolist()))
            if np.isfinite(score):
                init_candidates.append((score, a_init, method))
            if method == parent and trial == 0:
                init_candidates.append((float(evaluate_sequence(parent.tolist())), parent.copy(), 'parent'))

    init_candidates.sort(key=lambda x: x[0])
    print(f"  Top init: {init_candidates[0][0]:.6f} ({init_candidates[0][2]})")
    for sc, a, m in init_candidates[:3]:
        print(f"    {m}: {sc:.6f}")

    # === Phase 1: Subgradient on best candidates (60% of time) ===
    print(f"\n=== Phase 1: Subgradient optimization ===")
    sg_budget = (deadline - time.monotonic()) * 0.55
    top_n_candidates = min(5, len(init_candidates))

    for idx in range(top_n_candidates):
        if time.monotonic() > deadline - sg_budget * 0.3:
            break
        sc, a_init, method = init_candidates[idx]
        alloc = sg_budget / max(1, top_n_candidates - idx)
        if alloc < 10:
            break
        print(f"  SG on {method} (n={len(a_init)}, init_score={sc:.6f}, t={alloc:.0f}s)")
        a_opt, _ = subgradient_optimize(a_init, time_limit=alloc, top_k=10,
                                        temperature=0.001, lr=0.015, momentum=0.85)
        try_seq(a_opt, f"sg_{method}")

    # === Phase 2: LP-FW refinement (20% of remaining) ===
    print(f"\n=== Phase 2: LP-FW refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        lpfw_budget = (deadline - time.monotonic()) * 0.25
        n_best = len(best_seq)
        if lpfw_budget > 30:
            t_alloc = min(lpfw_budget, 200)
            print(f"  LP-FW n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = lpfw_optimize(best_seq, time_limit=t_alloc, verbose=True)

            if target_n == n_best:

            a_opt, _ = subgradient_optimize(f_interp, time_limit=t_alloc * 0.6,
                                            top_k=10, temperature=0.001, lr=0.01, momentum=0.85)
            a_opt2, _ = coord_descent_optimize(a_opt, time_limit=t_alloc * 0.4)
            try_seq(a_opt2, f"interp_n{target_n}")

    # === Phase 5: Final polish with diverse restarts ===
    print(f"\n=== Phase 5: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        n_best = len(best_seq)
        if rem > 30:
            # Multiple short SG runs with perturbations
            n_restarts = min(4, int(rem / 15))
            for r in range(n_restarts):
                rem = deadline - time.monotonic()
                if rem < 10:
                    break
                if r == 0:
                    f_start = best_seq.copy()
                elif r == 1:
                    f_start = project_simplex(best_seq + rng.standard_normal(n_best) * 0.005)
                elif r == 2:
                    f_start = 0.7 * best_seq + 0.3 * best_seq[::-1]
                    f_start = project_simplex(f_start)
                else:
                    f_start = 0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)
                    f_start = project_simplex(f_start)
                t_alloc = min(rem * 0.5, 40)
                a_opt, _ = subgradient_optimize(f_start, time_limit=t_alloc,
                                                top_k=15, temperature=0.0005,
                                                lr=0.008, momentum=0.9)
                try_seq(a_opt, f"polish_{r}")

            # One final CD
            rem = deadline - time.monotonic()
            if rem > 15:
                a_opt, _ = coord_descent_optimize(best_seq, time_limit=min(rem, 50))
                try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
