"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using golden-section coordinate descent
with analytical 1D subproblem solving, multi-length search, and subgradient polishing.

Key innovation: For each coordinate update, the 1D subproblem (find optimal value
of one coordinate given others fixed, with rescaling) is a convex optimization
problem (minimize max of quadratics). We solve it exactly via golden section search,
each evaluation being O(n) using precomputed convolution structure.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def subgradient_descent(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=30):
    """
    Subgradient method with per-coordinate adaptive step sizes (AdaGrad-like).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    C = fast_convolve(a)
    best_obj = C.max()
    best_a = a.copy()
    
    v = np.zeros(n)  # momentum
    G = np.zeros(n)  # AdaGrad accumulator
    lr = lr_init
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        C = fast_convolve(a)
        # Find top-k peaks
        k_actual = min(top_k, len(C))
        peak_indices = np.argpartition(C, -k_actual)[-k_actual:]
        
        # Compute subgradient
        sg = np.zeros(n)
        for k_idx in peak_indices:
            k = int(k_idx)
            # c[k] = sum_i a[i]*a[k-i]
            # d(c[k])/d(a[j]) = a[k-j] + a[j] (if both valid and k-j != j)
            #                  = 2*a[j] (if k == 2j)
            #                  = a[k-j] (if only one term valid)
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                j = i
                kj = k - i
                if kj < 0 or kj >= n:
                    continue
                if j == kj:
                    sg[j] += 2.0 * a[j]
                else:
                    sg[j] += a[kj]
                    sg[kj] += a[j]
        
        # Normalize subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm > 1e-15:
            sg = sg / sg_norm
        
        # Update with momentum and AdaGrad
        v = momentum * v - lr * sg
        G += sg ** 2
        a = a + v / (np.sqrt(G) + 1e-8)
        a = project_simplex(a)
        
        C_new = fast_convolve(a)
        obj = C_new.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Decay learning rate
        lr *= 0.999
        
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(it * 137 + 42)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            G = np.zeros(n)
            lr = lr_init * 0.5
            no_improve = 0
    
    return best_a, best_obj

def find_optimal_t(p, q, r, lo=0.0, hi=1.0-1e-10):
    """Find t in [lo, hi] minimizing f(t) = max_k (p[k]*t^2 + q[k]*t + r[k])."""
    best_t = lo
    best_val = np.inf
    
    # Check vertices of convex quadratics
    mask_convex = p > 1e-15
    for idx in np.where(mask_convex)[0]:
        t_v = -q[idx] / (2.0 * p[idx])
        if lo <= t_v <= hi:
            vals = p * t_v**2 + q * t_v + r
            v = vals.max()
            if v < best_val:
                best_val = v
                best_t = t_v
    
    # Check endpoints
    for t_c in [lo, hi]:
        vals = p * t_c**2 + q * t_c + r
        v = vals.max()
        if v < best_val:
            best_val = v
            best_t = t_c
    
    # Also do a golden section search to be safe (the function might not be unimodal
    # if some p[k] < 0, but golden section still tends to find good solutions)
    # ... actually, let's just add a fine grid search as well
    # Sample at 100 points
    t_grid = np.linspace(lo, hi, 200)
    for t_g in t_grid:
        vals = p * t_g**2 + q * t_g + r
        v = vals.max()
        if v < best_val:
            best_val = v
            best_t = t_g
    
    # Refine around best_t with golden section in a small window
    if best_t > lo + 1e-8 and best_t < hi - 1e-8:
        window = min(0.01, (hi - lo) * 0.01)
        t_lo = max(lo, best_t - window)
        t_hi = min(hi, best_t + window)
        # Golden section in [t_lo, t_hi]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = t_hi - gr * (t_hi - t_lo)
        c2 = t_lo + gr * (t_hi - t_lo)
        f_c1 = (p * c1**2 + q * c1 + r).max()
        f_c2 = (p * c2**2 + q * c2 + r).max()
        for _ in range(40):
            if f_c1 < f_c2:
                t_hi = c2
                c2 = c1
                f_c2 = f_c1
                c1 = t_hi - gr * (t_hi - t_lo)
                f_c1 = (p * c1**2 + q * c1 + r).max()
            else:
                t_lo = c1
                c1 = c2
                f_c1 = f_c2
                c2 = t_lo + gr * (t_hi - t_lo)
                f_c2 = (p * c2**2 + q * c2 + r).max()
        t_ref = (t_lo + t_hi) / 2.0
        v_ref = (p * t_ref**2 + q * t_ref + r).max()
        if v_ref < best_val:
            best_val = v_ref
            best_t = t_ref
    
    return best_t, best_val
