4. Random restarts and perturbation for escaping local minima

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(f, M, time_limit=30):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def compute_subgradient(f, conv, top_k=64, temp=0.001):
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg

def coord_descent(a, n_iterations=1000, n_grid=20, rng=None):

    n = len(a)
    a = a.copy()
    a = a / a.sum()
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    for it in range(n_iterations):
        if time.monotonic() - t0 > 300:  # time limit per call
            break
        improved = False
        for i in range(n):
            # Try scaling a[i]
            best_scale = 1.0
            best_val = conv.max()
            # Grid search
            scales = np.concatenate([np.linspace(0.1, 1.0, n_grid//2), np.linspace(1.0, 2.0, n_grid//2)])
            for scale in scales:
                a_try = a.copy()
                a_try[i] *= scale
                if a_try[i] < 0:
                    continue
                a_try /= a_try.sum()
                conv_try = fast_convolve(a_try)
                val = conv_try.max()
                if val < best_val:
                    best_val = val
                    best_scale = scale

                a /= a.sum()
                conv = fast_convolve(a)
                improved = True
                if conv.max() < best_obj:
                    best_obj = conv.max()
                    best_a = a.copy()
        
        if not improved:

    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Efficient cyclic coordinate descent with vectorized 1D optimization
2. LP-based Frank-Wolfe for global direction finding
3. Subgradient descent with momentum for moderate-scale improvements
4. Multi-length exploration with diverse structured initializations
5. Adaptive time allocation and final polishing

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def compute_subgradient(f, conv, top_k=64, temp=0.001):
    """Compute subgradient of max(conv) with softmax weighting."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg


def cyclic_coord_descent(a, time_limit=60.0, n_grid=40, randomize=False):
    """
    Cyclic coordinate descent for minimizing max(conv(a,a)) subject to a>=0.
    Uses vectorized 1D optimization per coordinate.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        # Determine coordinate order
        if randomize:
            order = np.random.permutation(n)
        else:
            order = np.arange(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit - 0.05:
                break
            
            v = a[i]
            
            # Build grid of trial values
            if v > 1e-10:
                t_vals = np.concatenate([
                    np.linspace(0.0, v, n_grid // 3 + 1),
                    np.linspace(v, v * 2.5, 2 * n_grid // 3)
                ])
            else:
                t_vals = np.linspace(0.0, 2.0 / n, n_grid)
            
            t_vals = np.maximum(t_vals, 0.0)
            
            # Precompute affected entries
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            other = np.arange(k_start, k_end + 1) - i
            base = conv[k_start:k_end+1].copy() - v * a[other]
            coefs = a[other].copy()
            
            # Handle diagonal (k = 2i)
            has_diag = (2 * i >= k_start) and (2 * i <= k_end) and (2 * i < 2*n-1)
            if has_diag:
                diag_pos = 2 * i - k_start
            
            # Unaffected max
            mask = np.zeros(2*n-1, dtype=bool)
            mask[k_start:k_end+1] = True
            if mask.sum() < 2*n-1:
                max_unaff = conv[~mask].max()
            else:
                max_unaff = 0.0
            
            # Vectorized evaluation
            t_col = t_vals[:, None]
            linear_vals = base[None, :] + t_col * coefs[None, :]
            
            if has_diag:
                linear_vals[:, diag_pos] = base[diag_pos] + t_vals**2
            
            max_affected = linear_vals.max(axis=1)
            max_total = np.maximum(max_affected, max_unaff)
            
            # Renormalization factor
            new_sum = 1.0 - v + t_vals
            new_sum = np.maximum(new_sum, 1e-15)
            normalized_max = max_total / (new_sum ** 2)
            
            best_idx = np.argmin(normalized_max)
            best_t = t_vals[best_idx]
            best_val = normalized_max[best_idx]
            
            # Compare with current
            cur_val = conv.max()  # since sum = 1
            
            if best_val < cur_val * (1.0 - 1e-12):
                a[i] = best_t
                ns = a.sum()
                if ns < 1e-15:
                    break
                a /= ns
                conv = fast_convolve(a)
                improved = True
                if conv.max() < best_obj - 1e-15:
                    best_obj = conv.max()
                    best_a = a.copy()
        
        if not improved:
            stall_count += 1
            if stall_count > 5:
                break
        else:
            stall_count = 0
    
    return best_a, best_obj


def pairwise_coord_descent(a, time_limit=60.0, n_grid=30, max_pairs_per_pass=50):
    """Pairwise coordinate descent: optimize pairs (i,j) while keeping a[i]+a[j] fixed."""
    n = len(a)
    a = a.astype(np.float64).copy()

    s = a.sum()

    rng = np.random.default_rng(123)
    
    while time.monotonic() - t0 < time_limit:

        # Select random pairs

            if time.monotonic() - t0 > time_limit - 0.1:
                break
            
            i, j = rng.choice(n, size=2, replace=False)
            v_i, v_j = a[i], a[j]
            s_ij = v_i + v_j
            if s_ij < 1e-12:
                continue
            
            # Trial values for a[i] (with a[j] = s_ij - a[i])
            t_vals = np.linspace(0.0, s_ij, n_grid + 1)

            best_t = v_i
            best_val = conv.max()
            
            # Affected entries: those involving index i or j
            # k involves i if k-i in [0,n-1], i.e., k in [i, i+n-1]
            # k involves j if k-j in [0,n-1], i.e., k in [j, j+n-1]

            k_ends = [min(i+n-1, 2*n-2), min(j+n-1, 2*n-2)]
            
            # Union of affected ranges
            all_affected = set()
            for ks, ke in zip(k_starts, k_ends):
                all_affected.update(range(ks, ke+1))
            all_affected = np.array(sorted(all_affected))
            
            if len(all_affected) == 0:
                continue
            
            # For each k in all_affected, compute the new value as a function of t
            # conv[k] = sum_{p+q=k} a[p]*a[q]
            # Terms involving a[i]: a[i]*a[k-i] (if valid)
            # Terms involving a[j]: a[j]*a[k-j] (if valid)
            # If k = i+j: also a[i]*a[j]
            
            # Let's just do it numerically for simplicity
            for t in t_vals:
                aj = s_ij - t
                if aj < 0:
                    continue
                a_try = a.copy()
                a_try[i] = t
                a_try[j] = aj
                ns = a_try.sum()  # should still be 1
                if ns < 1e-15:
                    continue
                c_try = fast_convolve(a_try)
                val = c_try.max()
                if val < best_val - 1e-15:
                    best_val = val
                    best_t = t

                a[i] = best_t
                a[j] = s_ij - best_t
                a = np.maximum(a, 0)
                a /= a.sum()
                conv = fast_convolve(a)
                improved = True
                if conv.max() < best_obj - 1e-15:
                    best_obj = conv.max()
                    best_a = a.copy()
        
        if not improved:
            break
    
    return best_a, best_obj
