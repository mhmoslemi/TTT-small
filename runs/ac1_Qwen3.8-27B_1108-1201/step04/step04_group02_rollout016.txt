Minimizes 2*n*max(conv(a,a))/sum(a)^2 using smoothed log-sum-exp optimization
with Adam optimizer, adaptive temperature annealing, and population-based search.
"""

from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_convolve2(x, y):
    """Linear convolution of x and y via FFT."""
    size = len(x) + len(y) - 1
    fft_size = 1 << (size - 1).bit_length()
    X = np.fft.rfft(x, fft_size)
    Y = np.fft.rfft(y, fft_size)
    return np.fft.irfft(X * Y, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_soft_gradient(f, p):
    """
    Compute gradient of t*log(sum(exp(conv/t))) w.r.t. f.
    = 2 * sum_k p[k] * f[k-i] for valid indices
    = 2 * (p * reverse(f))[n-1 : 2n-1]
    """
    n = len(f)
    b = f[::-1]
    conv_pb = fast_convolve2(p, b)
    return 2.0 * conv_pb[n-1 : 2*n-1]

def compute_max_subgradient(f, conv, top_k=16, temp=0.001):
    """Subgradient of max(conv) with softmax weighting."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum
    
    # Build p vector
    p = np.zeros(len(conv))
    p[peaks] = w
    
    return compute_soft_gradient(f, p)

    if s < 1e-15:
        f = np.ones(len(f)) / len(f)

    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t Adam = 0
    t_smooth = t_start
    t0 = time.monotonic()
    no_improve = 0
    total_steps = 0

        total_steps += 1

        scaled = conv / t_smooth
        scaled -= scaled.max()
        p = np.exp(scaled)
        p /= p.sum()

        grad = compute_soft_gradient(f, p)

            m_prev = m.copy()
            m = beta1 * m + (1 - beta1) * grad
            f_look = f - (beta1 * lr) * m / (np.sqrt(v) + eps)
            # Recompute gradient at look-ahead point? Too expensive.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve2(x, y):
    """Linear convolution of x and y via FFT."""
    size = len(x) + len(y) - 1
    fft_size = 1 << (size - 1).bit_length()
    X = np.fft.rfft(x, fft_size)
    Y = np.fft.rfft(y, fft_size)
    return np.fft.irfft(X * Y, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_soft_gradient(f, p):
    """
    Compute 2 * sum_k p[k] * f[k-i] for valid indices (cross-correlation).
    Uses FFT: result = 2 * (p * reverse(f))[n-1 : 2n-1]
    """
    n = len(f)
    b = f[::-1]
    conv_pb = fast_convolve2(p, b)
    return 2.0 * conv_pb[n - 1: 2 * n - 1]


def compute_subgradient(f, conv, top_k=16, temp=0.0005):
    """Subgradient of max(conv) with softmax weighting over top peaks."""
    n = len(f)
    L = len(conv)
    if top_k >= L:
        peaks = np.arange(L)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum
    p = np.zeros(L)
    p[peaks] = w
    return compute_soft_gradient(f, p)

def lp_escape(f, M, time_limit=15):
    """Try to find a direction g with (f*g)[k] <= M for all k."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            g = res.x
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None

        # Micro perturbations
        rem = deadline - time.monotonic() - 5
        if rem > 4:
            for _ in range(3):
                rem = deadline - time.monotonic() - 4
                if rem < 3:
                    break
                f_pert = project_simplex(best_seq / best_seq.sum() + rng.standard_normal(n_best) * 0.002)
                a_opt, _ = subgradient_polish(f_pert, time_limit=min(rem, 8),
                                              lr_start=0.0005, top_k=n_best // 3, temp=0.00003)
                try_seq(a_opt, "micro")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
