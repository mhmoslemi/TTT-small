Let me think about the smoothed FISTA more carefully:

We want to minimize `f(a) = max_k (a*a)[k]` over the simplex.

Smooth approximation: `f_τ(a) = τ * log(sum_k exp((a*a)[k] / τ))`

Gradient: `∇f_τ(a) = (2/sum_k exp((a*a)[k]/τ)) * sum_k exp((a*a)[k]/τ) * a_{k-i}` for component `i`.

Wait, let me recompute. `f_τ(a) = τ * log(sum_k exp(q_k / τ))` where `q_k = (a*a)[k]`.

`∂f_τ/∂a_i = τ * (1/sum_k exp(q_k/τ)) * sum_k exp(q_k/τ) * (1/τ) * ∂q_k/∂a_i`
`= (1/sum_k exp(q_k/τ)) * sum_k exp(q_k/τ) * ∂q_k/∂a_i`
`= sum_k p_k * 2*a_{k-i}`

where `p_k = exp(q_k/τ) / sum_j exp(q_j/τ)` is the softmax over the convolution values.

So the gradient is a weighted average of `2*a_{k-i}` over `k`, weighted by the softmax.

The Lipschitz constant of `f_τ` is at most `2/n` (since `||a||_2 <= 1` for `a` on the simplex with `sum=1`... actually `||a||_2 <= 1` and the gradient involves `a` itself, so the Lipschitz constant is bounded).

Actually, `|∂f_τ/∂a_i| = |sum_k p_k * 2*a_{k-i}| <= 2*max_k a_{k-i} <= 2/n` (for uniform) or up to 2 in the worst case.

The Lipschitz constant `L` of `f_τ` satisfies: `||∇f_τ(a) - ∇f_τ(b)|| <= L * ||a - b||`. Since `∇f_τ(a)_i = 2 * sum_k p_k(a) * a_{k-i}`, and both `p_k` and `a` depend on `a`, the Lipschitz constant is bounded by something like `4/(n*τ)` or similar. In practice, we can just use a backtracking line search.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.where(u * np.arange(1, n+1) > (cssv - 1))[0])
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def solve_lp_primal(f, M):
    """LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n-1, k)
        for j in range(js, je+1):
            A_ub[k, j] = f[k-j]
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True})
        if res.success:
            return res.x
    except:
        pass
    return None

def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0
    rng_local = np.random.default_rng(999 + n)
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        g = solve_lp_primal(f, M * m_exp)
        if g is None:
            m_exp *= 1.1
            if m_exp > 5.0:
                m_exp = 1.0
                # Try a simpler direction
                conv = fast_convolve(f)
                k_star = np.argmax(conv)
                g = np.zeros(n)
                g[k_star % n] = 1.0  # crude direction
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Fixed point check
        diff = np.max(np.abs(g - f))
        if diff < 1e-12:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 3.0)
            if no_improve > 30:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Bisection refinement
        if 0.05 < best_alpha < 0.95:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:
                f = f / fs
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)
        
        # Restart
        if no_improve > 40:
            strategies = ["gaussian", "mirror", "edge_boost"]
            strat = strategies[no_improve % len(strategies)]
            if strat == "gaussian":
                scale = 0.003 + 0.003 * (no_improve // 50)
                f = project_simplex(best_f + rng_local.standard_normal(n) * scale)
            elif strat == "mirror":
                blend = 0.3 + 0.1 * (no_improve // 50)
                f = project_simplex((1 - blend) * best_f + blend * best_f[::-1])
            else:
                x = np.arange(n, dtype=np.float64)
                boost = np.exp(-0.5 * (x / (n * 0.1)) ** 2)
                f = project_simplex(best_f * (1 + 0.1 * boost))
            
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj

**Key realization**: The LP-FW is already finding the exact optimal direction at each step. The convergence rate is O(1/k) which is the best possible for first-order methods on non-smooth convex problems. To do better, we need either:
1. A method that exploits the specific structure (like the quadratic nature of the constraints)
2. A second-order method
3. A method that converges faster than O(1/k)

For a convex problem that's the max of quadratic functions, we could potentially use:
- **Second-order Frank-Wolfe**: Use curvature information to get a better step size
- **Accelerated Frank-Wolfe (AGD)**: O(1/k^2) rate
- **Pairwise FW**: Often faster in practice

Actually, **Accelerated Gradient Descent (AGD) / Nesterov's accelerated method** is exactly FISTA, and it gives O(1/k^2) for smooth convex problems. For non-smooth, it's still O(1/k) in the worst case, but with smoothing (as I described), we can get O(1/tau + k^2 * tau^2) type rates, and by choosing tau ~ 1/k, we get O(1/k) overall. So smoothing doesn't fundamentally help the rate.

However, in practice, FISTA with smoothing often converges faster than FW because:
1. Each iteration is cheaper (no LP solve)

3. The effective step size can be larger

**Alternative: Semi-definite programming relaxation**

Where T_k is the matrix for the k-th autocorrelation. This is a QP with many quadratic constraints.

This relaxation might be tight or not, but solving it might give a good lower bound and potentially the optimal solution.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(f, M):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g (normalized to sum 1) or None."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix vectorized
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # Column j: A[k, j] = f[k-j] for k in [j, j+n-1]
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None

def gradient_refine(f0, time_limit, lr_init=0.005, decay=0.9995,
                    mom_init=0.9, top_k=None, temp_init=0.00005,
                    seed=123):
    """Adaptive gradient descent with multi-peak softmax weighting and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    if top_k is None:
        top_k = max(10, n // 4)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_mom = mom_init
    cur_temp = temp_init
    rng_local = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        # Find top-k peaks in convolution
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= w_sum

        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Momentum update
        v = cur_mom * v - cur_lr * sg
        f = project_simplex(f + v)

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive schedule
        cur_lr *= decay
        cur_temp *= 1.003
        cur_mom = min(cur_mom * 1.0002, 0.98)

        if no_improve > 300:
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.5, 0.001)
            cur_temp = max(temp_init * 0.5, 0.00002)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, seed=42):
    """Coordinate descent with pair swaps for fine polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng_local = np.random.default_rng(seed)
    no_full_pass = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        no_full_pass += 1
        improved = False
        order = rng_local.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            # Try moving mass from a few candidate sources to j
            candidates = [int((j + 1) % n), int((j - 1) % n),
                          int(j ^ (n // 3)), int(rng_local.integers(n))]
            for source in candidates:
                source = source % n
                if source == j or f[source] < 1e-10:
                    continue
                for frac in [0.1, 0.25, 0.5]:
                    delta = f[source] * frac
                    f_new = f.copy()
                    f_new[source] -= delta
                    f_new[j] += delta
                    f_new = np.maximum(f_new, 0.0)
                    sn = f_new.sum()
                    if sn < 1e-15:
                        continue
                    f_new = f_new / sn
                    val = fast_convolve(f_new).max()
                    if val < best_obj - 1e-16:
                        f = f_new
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            if no_full_pass > 3:
                break
        else:
            no_full_pass = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian center (wide)
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())

    # Gaussian center (narrow)
    g2 = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())

    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Sine window
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())

    # Two-peak
    peak = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 8.0) ** 2)) +
            np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 8.0) ** 2)))
    inits.append(peak / peak.sum())

    # Random Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Random Dirichlet (uniform-ish)
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    # Power law
    pl = (x + 1.0) ** (-0.3)
    inits.append(pl / pl.sum())

    # Cosine
    cos_t = np.cos(np.pi * x / (2.0 * (n - 1)))
    cos_t = np.maximum(cos_t, 0.01)
    inits.append(cos_t / cos_t.sum())

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        px = np.interp(x, np.linspace(0, n - 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Symmetrized parent
        pxs = 0.5 * (px + px[::-1])
        pxs = np.maximum(pxs, 0)
        ps2 = pxs.sum()
        if ps2 > 1e-10:
            inits.append(pxs / ps2)

    return inits


def try_sequence(a, label, state):
    """Evaluate a sequence and update state if better."""
    a = np.maximum(a, 0.0)
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: Multi-length exploration with quick FW-LP
    Phase 2: Deep optimization on best candidates
    Phase 3: Gradient refinement + coordinate descent polishing
    Phase 4: Final FW-LP squeeze
    """
    rng = np.random.default_rng(seed)
    total_budget = max(100.0, float(budget_s))
    deadline = time.monotonic() + total_budget - 15.0

    state = {'best_seq': None, 'best_score': float('inf')}

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
            s_parent = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(s_parent):
                print(f"  Parent score: {s_parent:.8f}")
                if s_parent < state['best_score']:
                    state['best_score'] = s_parent
                    state['best_seq'] = parent.copy()
    except Exception:
        pass

    # === Phase 1: Multi-length exploration ===
    print("\n=== Phase 1: Multi-length Exploration ===")
    phase1_end = time.monotonic() + total_budget * 0.22

    # Quick optimization of parent first
    if parent is not None:
        n_p = len(parent)
        t1 = min(80.0, phase1_end - time.monotonic() - 40.0)
        if t1 > 20:
            print(f"  Quick FW-LP on parent (n={n_p}), budget={t1:.0f}s")
            a_opt, obj = fw_lp_optimize(parent, time_limit=t1, seed=seed)
            try_sequence(a_opt, "parent_quick", state)

    # Multi-length exploration
    lengths = [288, 320, 352, 368, 384, 400, 416, 448]
    if parent is not None:
        # Put parent length first
        if len(parent) in lengths:
            lengths.remove(len(parent))
        lengths.insert(0, len(parent))
    else:
        lengths = [320, 384, 448]

    for idx, n in enumerate(lengths):
        if time.monotonic() > phase1_end - 30:
            break
        elapsed = time.monotonic() - (deadline - total_budget * 0.22)
        remaining_phases = len(lengths) - idx
        t_per = (phase1_end - time.monotonic() - 20.0) / max(1, remaining_phases)
        t_per = min(t_per, 60.0)
        t_per = max(t_per, 15.0)

        inits = generate_inits(n, parent, rng)
        n_try = min(3, len(inits))

        for i in range(n_try):
            if time.monotonic() > phase1_end - 15:
                break
            t_alloc = min(t_per / max(1, n_try - i), phase1_end - time.monotonic() - 10)
            if t_alloc < 8:
                break
            f_init = inits[i]
            a_opt, obj = fw_lp_optimize(f_init, time_limit=t_alloc, seed=seed + idx * 100 + i)
            try_sequence(a_opt, f"n{n}_i{i}", state)

    print(f"  Phase 1 best: {state['best_score']:.8f}")

    # === Phase 2: Deep optimization ===
    print("\n=== Phase 2: Deep Optimization ===")
    phase2_end = time.monotonic() + (deadline - time.monotonic()) * 0.45

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        print(f"  Best so far: n={n_best}, score={state['best_score']:.8f}")

        # Main deep FW-LP
        t_main = min(180.0, (phase2_end - time.monotonic()) * 0.4)
        if t_main > 20:
            a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=t_main,
                                        verbose=True, seed=seed + 500)
            try_sequence(a_opt, "deep_main", state)

        # Perturbation-based restarts
        perturb_scales = [0.003, 0.008, 0.015, 0.04, 0.08]
        for ps in perturb_scales:
            rem = phase2_end - time.monotonic() - 15
            if rem < 15:
                break
            t_alloc = min(rem * 0.25, 80.0)
            if t_alloc < 10:
                break
            noise = rng.standard_normal(n_best) * ps
            f_init = project_simplex(state['best_seq'] + noise)
            a_opt, obj = fw_lp_optimize(f_init, time_limit=t_alloc, seed=seed + int(ps * 1000))
            try_sequence(a_opt, f"pert_{ps:.3f}", state)

        # Mirror blend
        rem = phase2_end - time.monotonic() - 15
        if rem > 20:
            for blend in [0.2, 0.4, 0.6]:
                if rem < 15:
                    break
                f_mirror = project_simplex((1 - blend) * state['best_seq'] +
                                           blend * state['best_seq'][::-1])
                a_opt, obj = fw_lp_optimize(f_mirror, time_limit=min(rem, 50),
                                            seed=seed + 700 + int(blend * 10))
                try_sequence(a_opt, f"mirror_{blend}", state)
                rem = phase2_end - time.monotonic() - 10

    # Try a different length if we have time
    rem = phase2_end - time.monotonic() - 20
    if rem > 30 and state['best_seq'] is not None:
        n_curr = len(state['best_seq'])
        alt_lengths = [n_curr + 16, n_curr - 16, n_curr + 32, n_curr - 32]
        alt_lengths = [l for l in alt_lengths if 200 <= l <= 600]
        for n_alt in alt_lengths[:2]:
            if time.monotonic() > phase2_end - 20:
                break
            f_init = np.interp(np.arange(n_alt),
                               np.linspace(0, n_alt - 1, len(state['best_seq'])),
                               state['best_seq'])
            f_init = np.maximum(f_init, 0)
            f_init /= f_init.sum()
            a_opt, obj = fw_lp_optimize(f_init, time_limit=min(rem, 50),
                                        seed=seed + 900)
            try_sequence(a_opt, f"alt_n{n_alt}", state)

    # === Phase 3: Gradient refinement ===
    print("\n=== Phase 3: Gradient Refinement ===")
    phase3_end = time.monotonic() + (deadline - time.monotonic()) * 0.35

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])

        # Long gradient refinement
        rem = phase3_end - time.monotonic() - 10
        if rem > 15:
            a_opt, obj = gradient_refine(state['best_seq'], time_limit=min(rem, 120),
                                         lr_init=0.003, decay=0.9997,
                                         mom_init=0.92, top_k=max(8, n_best // 5),
                                         temp_init=0.00004, seed=seed + 2000)
            try_sequence(a_opt, "grad_long", state)

        # Coordinate descent
        rem = phase3_end - time.monotonic() - 10
        if rem > 10:
            a_opt, obj = coordinate_descent(state['best_seq'], time_limit=min(rem, 80),
                                            seed=seed + 3000)
            try_sequence(a_opt, "cd", state)

        # Fine gradient
        rem = phase3_end - time.monotonic() - 5
        if rem > 10:
            a_opt, obj = gradient_refine(state['best_seq'], time_limit=min(rem, 60),
                                         lr_init=0.001, decay=0.9998,
                                         mom_init=0.95, top_k=max(5, n_best // 8),
                                         temp_init=0.00002, seed=seed + 4000)
            try_sequence(a_opt, "grad_fine", state)

            a_opt, obj = coordinate_descent(state['best_seq'], time_limit=min(rem, 30),
                                            seed=seed + 5000)
            try_sequence(a_opt, "cd_fine", state)

    # === Phase 4: Final squeeze ===
    print("\n=== Phase 4: Final Squeeze ===")
    if state['best_seq'] is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 15:
            # Final FW-LP
            a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=min(rem, 60),
                                        seed=seed + 6000)
            try_sequence(a_opt, "final_fwl", state)

            rem2 = deadline - time.monotonic() - 3
            if rem2 > 8:
                a_opt, obj = coordinate_descent(state['best_seq'], time_limit=min(rem2, 20),
                                                seed=seed + 7000)
                try_sequence(a_opt, "final_cd", state)

    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(384) / 384
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))
        print(f"  Fallback uniform: {state['best_score']:.6f}")

    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs

    print(f"\nFINAL SCORE: {state['best_score']:.8f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
