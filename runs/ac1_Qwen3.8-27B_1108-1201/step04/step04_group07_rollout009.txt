    """Compute H @ v for the LSE objective."""
    n = len(a)
    b = np.convolve(a, a)
    scaled = beta * b
    scaled -= scaled.max()
    w = np.exp(scaled)
    s = w / w.sum()  # softmax weights
    
    # Bv = 2 * conv(v, a)
    Bv = 2.0 * np.convolve(v, a)  # length 2n-1

def project_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([max(v[0], 0.0)])

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-sum-exp smoothed
objective with L-BFGS, combined with LP-based Frank-Wolfe, subgradient
polishing, multi-peak targeting, and local search. Key innovation: adaptive
temperature LSE optimization with Hessian-aware line search enables fast
convergence to near-flat autocorrelation without LP overhead.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def convolve(x, y):
    """Linear convolution of two sequences via FFT."""
    size = len(x) + len(y) - 1
    fft_size = 1 << (size - 1).bit_length()
    X = np.fft.rfft(x, fft_size)
    Y = np.fft.rfft(y, fft_size)
    return np.fft.irfft(X * Y, fft_size)[:size]


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n <= 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_grad(a, beta, lambda_scale=500.0, target_sum=1.0):
    """Computes LSE objective and gradient with scale penalty."""
    n = len(a)
    b = fast_convolve(a)
    scaled = beta * b
    scaled_max = scaled.max()
    w = np.exp(scaled - scaled_max)
    w_sum = w.sum()
    s = w / w_sum
    lse_val = (scaled_max + np.log(w_sum)) / beta

    s_a = a.sum()
    penalty = 0.5 * lambda_scale * (s_a - target_sum) ** 2
    pen_grad_coeff = lambda_scale * (s_a - target_sum)

    # LSE gradient: 2 * cross-correlation(s, a)
    grad = np.zeros(n)
    for i in range(n):
        k_lo = i
        k_hi = min(i + n, 2 * n - 1)
        if k_lo < k_hi:
            grad[i] = 2.0 * np.dot(s[k_lo:k_hi], a[:k_hi - k_lo])

    obj = lse_val + penalty
    grad += pen_grad_coeff
    return obj, grad

    )

def build_conv_matrix_rows(f, peaks):
    """Build rows of the convolution matrix for specific peak indices."""
    n = len(f)
    rows = []
    for k in peaks:
        row = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            row[j] = f[k - j]
        rows.append(row)
    return np.array(rows)


def solve_linearized_lp_reduced(f, conv_f, top_k=20):
    """Solve reduced linearized LP for Frank-Wolfe direction."""
    n = len(f)
    if top_k >= len(conv_f):
        peaks = np.arange(len(conv_f))
    else:
        peaks = np.argpartition(conv_f, -top_k)[-top_k:]

    B_rows = build_conv_matrix_rows(f, peaks)
    n_constraints = len(peaks)

    c = np.zeros(n + 1)
    c[0] = 1.0

    A_ub = np.zeros((n_constraints, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B_rows
    b_ub = -conv_f[peaks]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x[1:]
    return None


def solve_maxsum_lp(f, M):
    """Solve max-sum LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None and res.x.sum() > 0:
        return res.x
    return None

def subgradient_polish(f0, time_limit, lr=0.002, mom=0.92, top_k=50, temp=0.0001):
    """Subgradient descent with softmax multi-peak weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        tk = min(top_k, len(conv))
        peaks = np.argpartition(conv, -tk)[-tk:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-13:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        cur_temp *= 1.005

        if no_improve > 200:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def local_search(f0, time_limit, rng, base_scale=0.004):
    """Random perturbation local search with adaptive scale."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    scale = base_scale

    while time.monotonic() - t0 < time_limit:
        trials = 200
        for _ in range(trials):
            if time.monotonic() - t0 > time_limit:
                break
            noise = rng.standard_normal(n) * scale
            fp = project_simplex(best_f + noise)
            obj = fast_convolve(fp).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = fp.copy()
        scale *= 0.96
        if scale < 1e-7:
            scale = 0.002
            fj = project_simplex(best_f + rng.standard_normal(n) * 0.008)
            oj = fast_convolve(fj).max()
            if oj < best_obj:
                best_obj = oj
                best_f = fj.copy()

    return best_f, best_obj


def multi_peak_direct(f0, time_limit, rng):
    """Directly target peak lags by reducing their contribution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    step = 0.01

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        peaks = np.where(conv > best_obj * 0.999)[0]
        if len(peaks) == 0:
            peaks = np.array([np.argmax(conv)])

        # For each peak, try to reduce it by adjusting the contributing elements
        improved = False
        for k in peaks[:10]:
            # Elements contributing to conv[k] are a[j] and a[k-j] for valid j
            js = range(max(0, k - n + 1), min(n, k + 1))
            for j in js:
                partner = k - j
                if partner < 0 or partner >= n:
                    continue
                # Try reducing the larger of a[j], a[partner]
                if a_val := f[j] > f[partner]:
                    idx = j
                else:
                    idx = partner
                if f[idx] < 1e-10:
                    continue
                # Reduce and compensate
                f_try = f.copy()
                reduction = f[idx] * step
                f_try[idx] -= reduction
                f_try = np.maximum(f_try, 0.0)
                fs = f_try.sum()
                if fs < 1e-15:
                    continue
                f_try = f_try / fs
                obj = fast_convolve(f_try).max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_f = f_try.copy()
                    f = f_try.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            step *= 0.9
            if step < 1e-5:
                # Reset with perturbation
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                step = 0.01
                conv = fast_convolve(f)
                if conv.max() < best_obj:
                    best_obj = conv.max()
                    best_f = f.copy()

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    center = 0.5 * (n - 1)

    # Various Gaussian widths
    for sigma_frac in [0.2, 0.3, 0.4, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        g = np.maximum(g, 0.01)
        inits.append(g / g.sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(np.pi * (x - center) / (0.6 * n)))
    rc = np.maximum(rc, 0.01)
    inits.append(rc / rc.sum())

    # Two peaks
    two = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + \
          np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    two = np.maximum(two, 0.01)
    inits.append(two / two.sum())

    # Three peaks
    three = np.exp(-0.5 * ((x - 0.2 * n) / (0.08 * n)) ** 2) + \
            np.exp(-0.5 * ((x - 0.5 * n) / (0.08 * n)) ** 2) + \
            np.exp(-0.5 * ((x - 0.8 * n) / (0.08 * n)) ** 2)
    three = np.maximum(three, 0.01)
    inits.append(three / three.sum())

    # U-shape (high at edges)
    edge_w = 1.0 / (0.3 * np.abs(x - center) / n + 0.15)
    inits.append(edge_w / edge_w.sum())

    # Beta distribution
    for alpha_b in [1.5, 2.5, 4.0]:
        b = (x + 1) ** alpha_b * (n - x) ** alpha_b
        b = np.maximum(b, 0.01)
        inits.append(b / b.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Perturbed parent
        px_pert = project_simplex(px / ps + rng.standard_normal(n) * 0.02)
        inits.append(px_pert)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining LSE-LBFGS, LP Frank-Wolfe,
    subgradient polish, and direct peak targeting.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10 or len(a) < 2:
            return float('inf')
        a = a.astype(np.float64)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.9f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.9f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        print("No parent, using uniform init")

    n_p = len(parent)

    # === Phase 1: Initialization and quick evaluation ===
    print("\n=== Phase 1: Initialization ===")
    t_start = time.monotonic()

    inits = generate_inits(n_p, parent, rng)
    for label, ini in [("parent", parent)] + [(f"init_{i}", ini) for i, ini in enumerate(inits)]:
        if time.monotonic() > deadline - 400:
            break
        try_seq(ini, label)

    # Try other lengths
    for n_try in [320, 448, 512]:
        if time.monotonic() > deadline - 400:
            break
        if n_try == n_p:
            continue
        px = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_p), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            px = px / ps
            try_seq(px, f"len{n_try}")

    if best_seq is None:
        best_seq = parent.copy()
        try_seq(best_seq, "fallback")

    n_best = len(best_seq)
    print(f"  Phase 1 best: {best_score:.9f} (n={n_best})")

    # === Phase 2: LSE-LBFGS with adaptive temperature ===
    print("\n=== Phase 2: LSE-LBFGS Optimization ===")
    remaining = deadline - time.monotonic()
    t_p2 = min(remaining * 0.35, 500.0)

    current = best_seq.copy()
    betas = [30, 60, 100, 200, 400, 800]
    per_beta = t_p2 / len(betas)

    for bi, beta in enumerate(betas):
        if time.monotonic() > deadline - 300:
            break
        t_each = min(per_beta, deadline - time.monotonic() - 250)
        if t_each < 5:
            break
        print(f"  LSE beta={beta}, budget={t_each:.0f}s")

        a_opt, _ = lse_optimize(current, beta=beta, max_iter=int(t_each * 300))
        try_seq(a_opt, f"lse_b{beta}")

        # Also try from perturbed start
        if time.monotonic() < deadline - 270:
            a_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)
            a_opt2, _ = lse_optimize(a_pert, beta=beta, max_iter=int(t_each * 150))
            try_seq(a_opt2, f"lse_b{beta}_pert")

        # Update current to best found so far
        current = best_seq.copy()

    print(f"  Phase 2 best: {best_score:.9f}")

    # === Phase 3: LP Frank-Wolfe for larger jumps ===
    print("\n=== Phase 3: LP Frank-Wolfe ===")
    remaining = deadline - time.monotonic()
    t_p3 = min(remaining * 0.25, 350.0)

    if t_p3 > 20:
        t_each = t_p3 / 4
        # Linearized LP
        t1 = min(t_each, deadline - time.monotonic() - 200)
        if t1 > 10:
            print(f"  Linearized FW, budget={t1:.0f}s")
            a_opt, _ = frank_wolfe_lp(best_seq, time_limit=t1, use_linearized=True, top_k=25)
            try_seq(a_opt, "linfw")

        # Max-sum LP
        t2 = min(t_each, deadline - time.monotonic() - 180)
        if t2 > 10:
            print(f"  Max-sum FW, budget={t2:.0f}s")
            a_opt, _ = frank_wolfe_lp(best_seq, time_limit=t2, use_linearized=False)
            try_seq(a_opt, "maxfw")

        # From perturbed start
        t3 = min(t_each, deadline - time.monotonic() - 150)
        if t3 > 10:
            a_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.015)
            print(f"  LinFW perturbed, budget={t3:.0f}s")
            a_opt, _ = frank_wolfe_lp(a_pert, time_limit=t3, use_linearized=True, top_k=30)
            try_seq(a_opt, "linfw_pert")

        t4 = min(t_each, deadline - time.monotonic() - 120)
        if t4 > 10:
            a_opt, _ = frank_wolfe_lp(best_seq, time_limit=t4, use_linearized=True, top_k=40)
            try_seq(a_opt, "linfw_k40")

    print(f"  Phase 3 best: {best_score:.9f}")

    # === Phase 4: Subgradient polishing ===
    print("\n=== Phase 4: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    t_p4 = min(remaining * 0.2, 250.0)

    if t_p4 > 15:
        t_each = t_p4 / 3
        # Coarse
        t1 = min(t_each, deadline - time.monotonic() - 100)
        if t1 > 8:
            a_opt, _ = subgradient_polish(best_seq, time_limit=t1,
                                          lr=0.003, mom=0.92, top_k=n_best // 5, temp=0.0002)
            try_seq(a_opt, "sg_coarse")

        # Fine
        t2 = min(t_each, deadline - time.monotonic() - 80)
        if t2 > 8:
            a_opt, _ = subgradient_polish(best_seq, time_limit=t2,
                                          lr=0.001, mom=0.95, top_k=n_best // 3, temp=0.00005)
            try_seq(a_opt, "sg_fine")

        # Very fine
        t3 = min(t_each, deadline - time.monotonic() - 60)
        if t3 > 5:
            a_opt, _ = subgradient_polish(best_seq, time_limit=t3,
                                          lr=0.0005, mom=0.97, top_k=n_best // 2, temp=0.00002)
            try_seq(a_opt, "sg_vfine")

    print(f"  Phase 4 best: {best_score:.9f}")

    # === Phase 5: Direct peak targeting ===
    print("\n=== Phase 5: Direct Peak Targeting ===")
    remaining = deadline - time.monotonic()
    t_p5 = min(remaining * 0.15, 200.0)

    if t_p5 > 10:
        a_opt, _ = multi_peak_direct(best_seq, time_limit=t_p5, rng=rng)
        try_seq(a_opt, "peak_direct")

    print(f"  Phase 5 best: {best_score:.9f}")

    # === Phase 6: Local search and final LSE polish ===
    print("\n=== Phase 6: Local Search & Final Polish ===")
    remaining = deadline - time.monotonic()
    t_p6 = min(remaining * 0.2, 250.0)

    if t_p6 > 10:
        # Local search
        t_ls = min(t_p6 * 0.4, 80)
        if t_ls > 5:
            a_opt, _ = local_search(best_seq, time_limit=t_ls, rng=rng, base_scale=0.003)
            try_seq(a_opt, "local")

        # Final LSE with high beta
        t_lse = min(t_p6 * 0.4, 80)
        if t_lse > 5:
            a_opt, _ = lse_optimize(best_seq, beta=2000, max_iter=int(t_lse * 200),
                                    lambda_scale=500.0, ftol=1e-15, gtol=1e-10)
            try_seq(a_opt, "lse_final_high")

        # Alternating perturb + LSE
        rem = deadline - time.monotonic() - 40
        for attempt in range(5):
            if rem < 8:
                break
            t_each = min(rem / (5 - attempt), 30)
            noise_scale = 0.003 * (attempt + 1)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            a_opt, _ = lse_optimize(f_pert, beta=500, max_iter=int(t_each * 100))
            try_seq(a_opt, f"alt_lse_{attempt}")
            rem = deadline - time.monotonic() - 30

        # Final local search
        rem = deadline - time.monotonic() - 25
        if rem > 5:
            a_opt, _ = local_search(best_seq, time_limit=min(rem, 40), rng=rng, base_scale=0.001)
            try_seq(a_opt, "local_final")

    # === Phase 7: Try nearby lengths ===
    rem = deadline - time.monotonic() - 20
    if rem > 20 and best_seq is not None:
        print("\n=== Phase 7: Length Variation ===")
        for n_try in [n_best - 32, n_best + 32, n_best - 16, n_best + 16]:
            if n_try < 64 or rem < 10:
                continue
            px = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_best), best_seq)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                px = px / ps
                # Quick LSE optimization
                t_each = min(rem * 0.4, 25)
                a_opt, _ = lse_optimize(px, beta=200, max_iter=int(t_each * 100))
                try_seq(a_opt, f"len{n_try}")
                rem = deadline - time.monotonic() - 15

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f} (n=384)")

    return [float(x) for x in best_seq]
