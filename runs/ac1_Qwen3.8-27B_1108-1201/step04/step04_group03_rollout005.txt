"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LSE-smoothed optimization.
"""

import numpy as np
import time

def next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (int(n) - 1).bit_length()

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = next_pow2(size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad(a, beta):
    """Compute LSE gradient w.r.t. a, and the convolution."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    p = exp_s / exp_s.sum()
    
    # grad[j] = 2 * sum_m a[m] * p[j+m]
    # = conv(a[::-1], p)[n-1 : 2n-1]
    fft_size = next_pow2(3 * n)
    a_rev = a[::-1]
    grad_full = np.fft.irfft(
        np.fft.rfft(a_rev, fft_size) * np.fft.rfft(p, fft_size), fft_size
    )[:3 * n - 2]
    grad = 2.0 * grad_full[n - 1:2 * n - 1]
    return grad, conv, p

def lse_mirror_descent(a0, time_limit, beta_init=10.0, beta_rate=1.001, max_beta=200.0):
    """Mirror descent (multiplicative update) with LSE objective."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    beta = beta_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 200 == 1:
            beta = min(beta * beta_rate, max_beta)
        
        grad, conv, p = compute_grad(a, beta)
        
        # Center the gradient for the multiplicative update
        grad_c = grad - np.mean(grad)
        
        # Adaptive step size
        eta = 1.0 / (np.max(np.abs(grad_c)) + 1e-10)
        
        # Multiplicative update
        log_factor = -eta * grad_c
        # Clip for numerical stability
        log_factor = np.clip(log_factor, -20, 20)
        a_new = a * np.exp(log_factor)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            # Try smaller step
            if no_improve > 50:
                eta *= 0.5
                log_factor = -eta * grad_c
                log_factor = np.clip(log_factor, -20, 20)
                a_new = a * np.exp(log_factor)
                a_new = np.maximum(a_new, 1e-30)
                a_new = a_new / a_new.sum()
                conv_new = fast_convolve(a_new)
                if conv_new.max() < best_obj:
                    best_obj = conv_new.max()
                    best_a = a_new.copy()
                    a = a_new
                    no_improve = 0
                else:
                    # Restart
                    if no_improve > 100:

                        a = best_a * np.exp(rng.standard_normal(n) * 0.01)
                        a = np.maximum(a, 1e-30)
                        a = a / a.sum()
                        no_improve = 0
                        beta = max(beta_init, beta * 0.5)
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LSE-smoothed optimization with FFT-based gradients.
Key innovations:
- Log-sum-exp smooth approximation for well-defined gradients over all peaks simultaneously
- FFT-based O(n log n) gradient computation via cross-correlation
- Hybrid mirror descent (multiplicative) and projected gradient with adaptive step
- Adaptive beta schedule for progressive sharpening
- Multi-length exploration and symmetry-breaking perturbations
- Coordinate-wise refinement in the final phase
"""

import numpy as np
import time


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (int(n) - 1).bit_length()


def _fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_and_conv(a, beta):
    """Compute LSE gradient w.r.t. a, the convolution, and softmax weights."""
    n = len(a)
    conv = _fast_convolve(a)
    scaled = beta * conv
    max_val = scaled.max()
    scaled -= max_val
    exp_s = np.exp(scaled)
    p = exp_s / exp_s.sum()

    # grad[j] = 2 * sum_{m=0}^{n-1} a[m] * p[j+m]
    # = 2 * conv(a[::-1], p_full)[n-1 : 2n-1]
    fft_size = _next_pow2(3 * n + 1)
    a_rev = a[::-1]
    grad_full = np.fft.irfft(
        np.fft.rfft(a_rev, fft_size) * np.fft.rfft(p, fft_size), fft_size
    )[:3 * n - 2]
    grad = 2.0 * grad_full[n - 1:2 * n - 1]
    return grad, conv, p


def _lse_mirror_descent(a0, time_limit, beta_init=8.0, beta_rate=1.002, max_beta=300.0,

    it = 0

def _lse_projected_grad(a0, time_limit, beta_init=10.0, beta_rate=1.003, max_beta=400.0,
                        verbose=False):
    """Projected gradient descent with LSE objective and BB step size."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = _fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    beta = float(beta_init)
    no_improve = 0
    it = 0
    prev_a = a.copy()
    prev_grad = None

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 30 == 1:
            beta = min(beta * beta_rate, max_beta)

        grad, conv, _ = _compute_grad_and_conv(a, beta)
        grad_proj = grad - np.mean(grad)

        # BB step size
        if prev_grad is not None:
            s_vec = a - prev_a
            y_vec = grad_proj - prev_grad
            sTs = np.dot(s_vec, s_vec)
            sTy = np.dot(s_vec, y_vec)
            yTy = np.dot(y_vec, y_vec) + 1e-15

            if sTs > 1e-18 and sTy > 1e-15:
                alpha_bb1 = sTs / sTy
                alpha_bb2 = sTy / yTy
                step = 0.5 * (alpha_bb1 + alpha_bb2)
                step = float(np.clip(step, 0.001, 5.0))
            else:
                step = 1.0 / (np.linalg.norm(grad_proj) + 1e-10)
        else:
            step = 1.0 / (np.linalg.norm(grad_proj) + 1e-10)

        # Line search with a few candidates
        best_alpha = step
        best_val = conv.max()
        for trial in [step * 0.5, step, step * 1.5, step * 2.0]:
            if trial < 1e-6:
                continue
            a_trial = _project_simplex(a - trial * grad_proj)
            val = _fast_convolve(a_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = trial

        a = _project_simplex(a - best_alpha * grad_proj)

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                rng = np.random.default_rng(int(time.monotonic() * 100000) % 1000000)
                a = _project_simplex(best_a + rng.standard_normal(n) * 0.008)
                no_improve = 0
                beta = max(beta_init, beta * 0.6)
                prev_grad = None

        prev_a = a.copy()
        prev_grad = grad_proj.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            conv_now = _fast_convolve(a)
            print(f"      PG iter {it}: obj={conv_now.max():.8f}, best={best_obj:.8f}, beta={beta:.1f}, t={elapsed:.0f}s")

    return best_a, best_obj


def _multi_scale_lse(a0, time_limit, verbose=False):
    """Alternates between mirror descent and projected gradient, with periodic
    switching and different beta regimes."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = _fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    use_md = True
    switch_interval = 200  # seconds per phase

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        phase_time = min(switch_interval, rem)

        if use_md:
            a_opt, obj_opt = _lse_mirror_descent(a, time_limit=phase_time)
        else:
            a_opt, obj_opt = _lse_projected_grad(a, time_limit=phase_time)

        if obj_opt < best_obj - 1e-15:
            best_obj = obj_opt
            best_a = a_opt.copy()
            a = a_opt.copy()
        else:
            # Keep best, add small perturbation for next phase
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
            a = _project_simplex(best_a + rng.standard_normal(n) * 0.003)

        use_md = not use_md

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"      Multi-scale: best={best_obj:.8f}, t={elapsed:.0f}s, next={'MD' if use_md else 'PG'}")

    return best_a, best_obj


def _coord_refine(a0, time_limit):
    """Coordinate-wise refinement: optimize a few coordinates at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s
    n = len(a)

    conv = _fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        # Pick a random coordinate to optimize
        j = rng.integers(0, n)
        # Try different values for a[j]
        current = a[j]
        best_val = best_obj
        best_v = current

        for scale in [0.5, 0.7, 1.3, 1.5, 0.3, 0.1, 2.0]:
            v_try = current * scale
            if v_try < 0 or v_try > 1:
                continue
            a_try = a.copy()
            a_try[j] = v_try
            a_try_sum = a_try.sum()
            if a_try_sum < 1e-15:
                continue
            a_try = a_try / a_try_sum
            val = _fast_convolve(a_try).max()
            if val < best_val:
                best_val = val
                best_v = v_try

        if best_val < best_obj - 1e-15:
            a[j] = best_v
            a = a / a.sum()
            best_obj = best_val
            best_a = a.copy()

    return best_a, best_obj

        t_alloc = min(rem, t_p4 * 0.3)
        if t_alloc > 10:
            print(f"  Coord refine, t={t_alloc:.0f}s")
            a_opt, _ = _coord_refine(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord")

        # Tiny perturbations with quick polish
        rem = deadline - time.monotonic()
        if rem > 20:
            for pi in range(4):
                rem = deadline - time.monotonic()
                if rem < 10:
                    break
                t_alloc = min(rem, 15.0)
                noise = rng.standard_normal(n_best) * 0.002
                a_pert = _project_simplex(best_seq + noise)
                a_opt, _ = _lse_mirror_descent(a_pert, time_limit=t_alloc,
                                               beta_init=50.0, beta_rate=1.002, max_beta=200.0)
                try_seq(a_opt, f"tiny_pert_{pi}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = min(best_score, final_score)
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
