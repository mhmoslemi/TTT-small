Optimized multi-phase search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses sparse-matrix LP-based Frank-Wolfe with momentum, multi-length exploration,
and coordinate-wise polishing.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, lil_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)

    if not cond.any():

    """Build sparse matrix A where A[k,j] = f[j] if max(0,k-n+1) <= j <= min(n-1,k)."""
    n = len(f)
    num_rows = 2 * n - 1
    row_idx = []
    col_idx = []
    data = []
    for j in range(n):
        k_lo = j
        k_hi = min(j + n - 1, 2 * n - 2)
        cnt = k_hi - k_lo + 1
        row_idx.extend(range(k_lo, k_hi + 1))
        col_idx.extend([j] * cnt)
        data.extend([f[j]] * cnt)
    return csr_matrix((data, (row_idx, col_idx)), shape=(num_rows, n))

    n = len(f)

    c = -np.ones(n)

def coord_descent_polish(f0, time_limit, verbose=False):
    """Coordinate-wise optimization: for each coordinate, find best value via bisection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        # Randomize coordinate order each pass
        order = np.random.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 >= time_limit:
                break
            
            # Try to optimize coordinate i by bisection on [0, 2*f[i]+0.1/n]
            old_val = f[i]
            lo = 0.0
            hi = max(0.0, 2.0 * old_val + 0.5 / n)
            
            # Evaluate objective at current
            def obj_at(val):
                f[i] = val
                ft = np.maximum(f, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    return float('inf')
                ft = ft / st
                return fast_convolve(ft).max()
            
            # Bisection: we want to find val that minimizes obj
            # Try a few candidate values
            candidates = [old_val, old_val * 0.5, old_val * 1.5, 0.0, 2.0 * old_val]
            best_cand_val = old_val
            best_cand_obj = obj_at(old_val)
            
            for cv in candidates:
                cv = max(0.0, cv)
                fv = obj_at(cv)
                if fv < best_cand_obj:
                    best_cand_obj = fv
                    best_cand_val = cv
            
            # Refine with bisection around best
            if best_cand_val != old_val:
                mid_lo = min(old_val, best_cand_val)
                mid_hi = max(old_val, best_cand_val)
                for _ in range(6):
                    mid = (mid_lo + mid_hi) / 2.0
                    fm = obj_at(mid)
                    if fm < best_cand_obj:
                        best_cand_obj = fm
                        best_cand_val = mid
                        mid_hi = mid
                    else:
                        mid_lo = mid
            
            f[i] = best_cand_val
            
            if best_cand_obj < best_obj - 1e-14:
                best_obj = best_cand_obj
                best_f = f.copy()
                improved = True
        
        f = np.maximum(f, 0.0)
        f = f / f.sum()
        
        if not improved:
            # Try small random perturbations
            rng = np.random.default_rng(time.monotonic() * 1000 % 100000)
            idx = rng.integers(0, n, size=min(10, n))
            for i in idx:
                f[i] = max(0, f[i] * (1 + rng.standard_normal() * 0.1))
            f = project_simplex(f)
            co = fast_convolve(f).max()
            if co < best_obj - 1e-14:
                best_obj = co
                best_f = f.copy()
            else:
                f = best_f.copy()
                break
    
    return best_f, best_obj


def subgradient_polish(f0, time_limit, lr=0.001, temp=0.00003, top_k_frac=0.05):
    """Subgradient method on top-k peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    top_k = max(5, int(n * top_k_frac))
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        f = project_simplex(f - lr * sg)
        co = fast_convolve(f).max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
        
        if it % 100 == 0:
            lr *= 0.995
            temp *= 1.01
    
    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())
    
    # Two humps
    h1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits.append((h1 + h2) / (h1 + h2).sum())
    
    # Mirror-symmetric
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.5)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    inits.append(sym / sym.sum())
    
    # Interpolated parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits

Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses sparse-matrix LP-based Frank-Wolfe with momentum, multi-length exploration,
coordinate descent polishing, and subgradient refinement.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_sparse_conv_matrix(f):
    """Build sparse matrix A where A[k,j] = f[j] if max(0,k-n+1) <= j <= min(n-1,k)."""
    n = len(f)
    num_rows = 2 * n - 1
    row_parts = []
    col_parts = []
    data_parts = []
    for j in range(n):
        k_lo = j
        k_hi = min(j + n - 1, 2 * n - 2)
        cnt = k_hi - k_lo + 1
        row_parts.append(np.arange(k_lo, k_hi + 1))
        col_parts.append(np.full(cnt, j, dtype=np.int32))
        data_parts.append(np.full(cnt, f[j]))
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    data = np.concatenate(data_parts)
    return csr_matrix((data, (rows, cols)), shape=(num_rows, n))


def solve_lp_direction(f, M):
    """Solve LP: max 1^T g s.t. conv_matrix(f) @ g <= M, g >= 0. Returns normalized g or None."""
    n = len(f)
    A = build_sparse_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        if gs > 1e-15:
            return g / gs
    return None


def line_search_fw(f, g, M_max):
    """Coarse grid + bisection line search for FW step."""
    best_alpha = 0.0
    best_val = M_max
    grid = [0.01, 0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5, 0.65, 0.8, 1.0]
    for alpha in grid:
        ft = f + alpha * (g - f)
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            continue
        ft = ft / s
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    if best_alpha > 0.02:
        lo = max(0.0, best_alpha - 0.06)
        hi = min(1.0, best_alpha + 0.06)
        for _ in range(8):
            mid = (lo + hi) / 2.0
            ft = f + mid * (g - f)
            ft = np.maximum(ft, 0.0)
            s = ft.sum()
            if s < 1e-15:
                break
            ft = ft / s
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
                hi = mid
            else:
                lo = mid
    return best_alpha, best_val


def fw_optimize(f0, time_limit, use_momentum=True, verbose=False):
    """LP-based Frank-Wolfe with optional momentum and adaptive constraint relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = float(conv.max())
    best_obj = M
    best_f = f.copy()
    momentum_dir = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        g = solve_lp_direction(f, M * m_relax)
        if g is None:
            m_relax = min(m_relax * 1.15, 5.0)
            if m_relax > 4.5:
                break
            continue

        if use_momentum:
            momentum_dir = 0.6 * momentum_dir + 0.4 * (g - f)
            g_raw = f + momentum_dir
            g_raw = np.maximum(g_raw, 0.0)
            gm = g_raw.sum()
            if gm < 1e-15:
                g_use = g
            else:
                g_use = g_raw / gm
        else:
            g_use = g

        alpha, val = line_search_fw(f, g_use, M)

        if alpha > 1e-10:
            f_new = f + alpha * (g_use - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            M_new = float(fast_convolve(f_new).max())
            if M_new < best_obj - 1e-14:
                best_obj = M_new
                best_f = f_new.copy()
                no_improve = 0
                m_relax = 1.0
            else:
                no_improve += 1
            f = f_new
            M = M_new
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.08, 5.0)

        if no_improve > 25:
            rng2 = np.random.default_rng((it * 7919 + n * 104729) % (2**31))
            scale = 0.002 + 0.003 * min(no_improve // 25, 6)
            f = project_simplex(best_f + rng2.standard_normal(n) * scale)
            M = float(fast_convolve(f).max())
            m_relax = 1.0
            no_improve = 0
            momentum_dir = np.zeros(n)

        if verbose and it % 150 == 0:
            print(f"    FW iter {it}: obj={best_obj:.10f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def coord_descent_polish(f0, time_limit, verbose=False):
    """Coordinate-wise bisection polish on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    best_obj = float(fast_convolve(f).max())
    best_f = f.copy()
    t0 = time.monotonic()
    rng_cd = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng_cd.permutation(n)
        for i in order:
            if time.monotonic() - t0 >= time_limit:
                break
            old_val = f[i]
            if old_val < 1e-15:
                continue
            f[i] = 0.0
            ft = f / max(f.sum(), 1e-15)
            obj_zero = float(fast_convolve(ft).max())
            f[i] = old_val * 2.0
            st = f.sum()
            ft2 = f / st
            obj_double = float(fast_convolve(ft2).max())
            f[i] = old_val

            if obj_zero < obj_double:
                lo_v, hi_v = 0.0, old_val
            else:
                lo_v, hi_v = old_val, old_val * 2.0

            best_v = old_val
            best_o = float(fast_convolve(f / f.sum()).max())

            for _ in range(7):
                mid_v = (lo_v + hi_v) / 2.0
                f[i] = mid_v
                st = f.sum()
                if st < 1e-15:
                    f[i] = best_v
                    break
                ft = f / st
                ov = float(fast_convolve(ft).max())
                if ov < best_o:
                    best_o = ov
                    best_v = mid_v
                    hi_v = mid_v
                else:
                    lo_v = mid_v
            f[i] = best_v
            if best_o < best_obj - 1e-13:
                best_obj = best_o
                best_f = f.copy()
                improved = True

        f = np.maximum(f, 0.0)
        fs = f.sum()
        if fs > 1e-15:
            f = f / fs
        if not improved:
            f = best_f.copy()
            break

    return best_f, best_obj


def subgradient_polish(f0, time_limit, lr=0.001, temp=0.00003, top_k_frac=0.05):
    """Subgradient method focusing on top-k peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    best_obj = float(fast_convolve(f).max())
    best_f = f.copy()
    t0 = time.monotonic()
    top_k = max(5, int(n * top_k_frac))
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[int(np.argmax(pv))] = 1.0
        else:
            w /= ws
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            ii = np.arange(js, je + 1)
            sg[ii] += 2.0 * w[idx] * f[k - ii]
        f = project_simplex(f - lr * sg)
        co = float(fast_convolve(f).max())
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
        if it % 80 == 0:
            lr *= 0.994
            temp *= 1.012

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())
    h1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits.append((h1 + h2) / (h1 + h2).sum())
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.5)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    inits.append(sym / sym.sum())
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    return inits


def try_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and return updated (best_seq, best_score)."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(a, 0.0).astype(np.float64)
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry: multi-phase search combining sparse LP-FW, coordinate descent,
    and subgradient polishing with multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    n_base = len(parent) if parent is not None else 368

    # === Phase 1: Multi-length exploration (280s) ===
    print("=== Phase 1: Multi-length Exploration ===")
    t_p1 = time.monotonic()
    p1_budget = 280.0

    if parent is not None:
        t_alloc = min(90.0, p1_budget - 60.0)
        if t_alloc > 20:
            print(f"  FW on parent n={n_base}, t={t_alloc:.0f}s")
            a_opt, obj = fw_optimize(parent, time_limit=t_alloc)
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "parent_fw")

    lengths = set()
    for delta in [-48, -32, -24, -16, -8, 8, 16, 24, 32, 48, 64, -64]:
        L = n_base + delta
        if 128 <= L <= 1024:
            lengths.add(L)
    lengths.discard(n_base)
    length_list = sorted(lengths, key=lambda l: abs(l - n_base))
    if len(length_list) > 4:
        length_list[:4] = sorted(length_list[:4], key=lambda l: abs(l - n_base))
        rng.shuffle(length_list[4:])

    for L in length_list[:10]:
        rem = p1_budget - (time.monotonic() - t_p1)
        if rem < 25:
            break
        t_alloc = min(28.0, rem - 15.0)
        if t_alloc < 12:
            break
        inits = generate_inits(L, parent, rng)
        best_init = inits[0]
        best_init_obj = float('inf')
        for fi in inits:
            co = float(fast_convolve(fi).max())
            if co < best_init_obj:
                best_init_obj = co
                best_init = fi
        print(f"  FW n={L}, t={t_alloc:.0f}s, init_obj={best_init_obj:.6f}")
        a_opt, obj = fw_optimize(best_init, time_limit=t_alloc)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"n{L}")

    print(f"  Phase 1 done: best={best_score:.6f}, elapsed={time.monotonic()-t_p1:.0f}s")

    # === Phase 2: Deep optimization (1150s) ===
    print("\n=== Phase 2: Deep Optimization ===")
    t_p2 = time.monotonic()
    p2_budget = 1150.0

    if best_seq is not None:
        f_work = best_seq.astype(np.float64) / best_seq.sum()

        num_rounds = 10
        t_per = p2_budget / num_rounds

        for r in range(num_rounds):
            rem = p2_budget - (time.monotonic() - t_p2)
            if rem < 30:
                break

            # Sub-phase A: FW
            t_fw = min(t_per * 0.55, rem * 0.55)
            if t_fw > 20:
                print(f"  Round {r}/10: FW (t={t_fw:.0f}s)")
                a_opt, obj = fw_optimize(f_work, time_limit=t_fw, use_momentum=True)
                best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"p2_fw_{r}")
                if best_seq is not None:
                    f_work = best_seq.astype(np.float64) / best_seq.sum()

            # Sub-phase B: Subgradient polish
            rem = p2_budget - (time.monotonic() - t_p2)
            t_sg = min(t_per * 0.25, rem * 0.3)
            if t_sg > 15:
                print(f"  Round {r}/10: SG (t={t_sg:.0f}s)")
                t_sg_actual = min(t_sg, time.monotonic() < deadline - 60 and 999 or 0)
                a_opt, _ = subgradient_polish(f_work, time_limit=t_sg, lr=0.0006, temp=0.00002, top_k_frac=0.04)
                best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"p2_sg_{r}")
                if best_seq is not None:
                    f_work = best_seq.astype(np.float64) / best_seq.sum()

            # Sub-phase C: Coordinate descent
            rem = p2_budget - (time.monotonic() - t_p2)
            t_cd = min(t_per * 0.18, rem * 0.3)
            if t_cd > 12 and time.monotonic() < deadline - 60:
                print(f"  Round {r}/10: CD (t={t_cd:.0f}s)")
                a_opt, _ = coord_descent_polish(f_work, time_limit=t_cd)
                best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"p2_cd_{r}")
                if best_seq is not None:
                    f_work = best_seq.astype(np.float64) / best_seq.sum()

    print(f"  Phase 2 done: best={best_score:.6f}, elapsed={time.monotonic()-t_p2:.0f}s")

    # === Phase 3: Fine polish (280s) ===
    print(f"\n=== Phase 3: Fine Polish ===")
    t_p3 = time.monotonic()
    p3_budget = 280.0

    if best_seq is not None and time.monotonic() < deadline - 40:
        f_work = best_seq.astype(np.float64) / best_seq.sum()

        sg_configs = [
            (0.0004, 0.000015, 0.03),
            (0.00025, 0.000008, 0.025),
            (0.00015, 0.000004, 0.02),
            (0.0001, 0.000002, 0.015),
        ]

        for i, (lr_v, temp_v, tkf) in enumerate(sg_configs):
            rem = p3_budget - (time.monotonic() - t_p3)
            if rem < 20 or time.monotonic() > deadline - 30:
                break
            t_alloc = min(55.0, rem - 15.0)
            if t_alloc < 12:
                break
            print(f"  SG polish {i} (t={t_alloc:.0f}s, lr={lr_v}, T={temp_v})")
            a_opt, _ = subgradient_polish(f_work, time_limit=t_alloc, lr=lr_v, temp=temp_v, top_k_frac=tkf)
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"p3_sg_{i}")
            if best_seq is not None:
                f_work = best_seq.astype(np.float64) / best_seq.sum()

        # Final coordinate descent
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 30 and time.monotonic() < deadline - 30:
            t_alloc = min(70.0, rem - 15.0)
            print(f"  Final CD (t={t_alloc:.0f}s)")
            a_opt, _ = coord_descent_polish(f_work, time_limit=t_alloc)
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "p3_cd")
            if best_seq is not None:
                f_work = best_seq.astype(np.float64) / best_seq.sum()

        # Final FW
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 30 and time.monotonic() < deadline - 30:
            t_alloc = min(60.0, rem - 15.0)
            print(f"  Final FW (t={t_alloc:.0f}s)")
            a_opt, _ = fw_optimize(f_work, time_limit=t_alloc, use_momentum=True)
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "p3_fw")

    print(f"  Phase 3 done: best={best_score:.6f}")

    # === Phase 4: Final tricks ===
    print(f"\n=== Phase 4: Final Tricks ===")

    if best_seq is not None and time.monotonic() < deadline - 20:
        f_work = best_seq.astype(np.float64) / best_seq.sum()
        n_b = len(f_work)

        # Mirror blend
        mirrored = 0.5 * f_work + 0.5 * f_work[::-1]
        mirrored = project_simplex(mirrored)
        if np.max(np.abs(mirrored - f_work)) > 1e-8:
            a_opt, _ = subgradient_polish(mirrored, time_limit=15, lr=0.0003, temp=0.00001, top_k_frac=0.02)
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "p4_mirror")

        # Small perturbations
        for p in range(4):
            if time.monotonic() > deadline - 15:
                break
            noise = rng.standard_normal(n_b) * 0.0008
            fp = project_simplex(f_work + noise)
            a_opt, _ = subgradient_polish(fp, time_limit=8, lr=0.0002, temp=0.000008, top_k_frac=0.02)
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"p4_pert_{p}")
            if best_seq is not None:
                f_work = best_seq.astype(np.float64) / best_seq.sum()

        # Tiny final FW
        if time.monotonic() < deadline - 25:
            a_opt, _ = fw_optimize(f_work, time_limit=25, use_momentum=True)
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "p4_final_fw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(368, dtype=np.float64) / 368
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_seq = np.ones(368, dtype=np.float64) / 368
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
