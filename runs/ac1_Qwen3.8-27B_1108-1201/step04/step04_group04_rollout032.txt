from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient_at_peaks(f, top_k=10, temperature=0.0):
    """Compute weighted subgradient from top peaks of convolution."""
    n = len(f)
    conv = fast_convolve(f)
    
    # Get top k peaks
    k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k)[-k:]
    pv = conv[peaks]
    
    if temperature > 0:
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
    else:
        # Hard max
        w = np.zeros(k)
        w[np.argmax(pv)] = 1.0
    
    # Compute subgradient
    sg = np.zeros(n)
    for i in range(k):
        if w[i] < 1e-12:
            continue
        pk = int(peaks[i])
        js = max(0, pk - n + 1)
        je = min(n - 1, pk)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[pk - idx]
    
    return conv, sg

def peaking_descent(f0, time_limit, top_k=15, temp_init=None, temp_min=1e-6, lr_init=0.005):
    """Peak-chasing descent with adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    if temp_init is None:
        # Set initial temperature so that ~10% of peaks are "active"
        sorted_conv = np.sort(conv)[::-1]
        temp_init = max((sorted_conv[0] - sorted_conv[min(n, len(sorted_conv)-1)]) / np.log(max(n, 2)), 1e-8)
    
    T = temp_init
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv, sg = compute_subgradient_at_peaks(f, top_k, T)
        
        # Project gradient to tangent space
        sg = sg - np.sum(sg) / n  # Actually, for simplex constraint sum(f)=1, tangent is sum=0
        sg = sg - np.mean(sg)
        
        f_new = project_simplex(f - lr * sg)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                lr *= 0.9
                T *= 0.95
        
        # Anneal temperature
        T = max(T * 0.998, temp_min)
        
        # Periodic restart
        if no_improve > 200:
            rng = np.random.default_rng((it * 631 + n) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            lr = lr_init * 0.5
            T = temp_init * 0.3
            no_improve = 0
        
        if it % 500 == 0 and verbose:
            print(f"    PD iter {it}: obj={best_obj:.10f}, T={T:.8f}, lr={lr:.6f}")
    
    return best_f, best_obj

The Newton direction satisfies: `H * d = -grad`, i.e., `2*Q_{k*} * d = -2*Q_{k*} * f`, so `d = -f`. That's degenerate!

This is because `conv_{k*}` is a quadratic form, and the Newton step for a pure quadratic is to jump to the minimum in one step (which is at `f = 0`). But we're constrained to the simplex, so this doesn't directly help.

However, we can use a "constrained Newton" approach: minimize `conv_{k*}(f)` subject to `sum(f) = 1, f >= 0`. This is a QP!

The KKT conditions are:
`2*Q_{k*} f + lambda * 1 + mu = 0` (stationarity, where lambda is for sum constraint, mu for non-negativity)
`sum(f) = 1`
`f >= 0, mu >= 0, f[m]*mu[m] = 0`

This is a QP that we can solve! And the solution gives the exact minimizer of the active peak. Then we check if other peaks are violated, and if so, add them.

This is essentially a **quatic programming approach** that could be much more efficient than LP-FW.

"""
Multi-strategy optimizer for minimizing 2*n*max(auto_convolve(a,a))/(sum(a))^2.
Uses: (1) Improved LP-Frank-Wolfe with multi-peak constraints,

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_multi_peak(f, M_list, k_list):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M_k for k in k_list, g >= 0.
    This is a multi-peak constraint version.
    """
    n = len(f)
    c = -np.ones(n)
    
    # Build constraint matrix: for each k in k_list, (f*g)[k] <= M_k
    # (f*g)[k] = sum_j f[j] * g[k-j] ... wait, it's cross-correlation
    # Actually, the constraint is that the cross-convolution (f*g)_lin[k] <= M_k
    # where (f*g)_lin[k] = sum_j f[j] * g[k-j] for valid j.
    
    # For a fixed k: (f*g)[k] = sum_{j: max(0,k-n+1) <= j <= min(n-1,k)} f[j] * g[k-j]
    # = sum_{j: valid} f[j] * g[k-j]
    # Let i = k-j, so j = k-i:
    # = sum_{i: max(0,n-1-k) <= i <= min(n-1, k)} f[k-i] * g[i]
    
    # So the coefficient of g[i] in constraint k is f[k-i] (if valid).
    
    A_ub = np.zeros((len(k_list), n))
    for row, k in enumerate(k_list):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[row, i] = f[j]
    
    b_ub = np.array(M_list)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def solve_lp_single_peak(f, M):
    """Original single-peak LP direction."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def line_search(f, g, M, n_iter=20):
    """
    Golden section line search for alpha in (0,1):
    minimize max_conv((1-alpha)*f + alpha*g)
    Returns best alpha and the corresponding point.
    """
    def obj(alpha):
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            return float('inf')
        ft = ft / s
        return fast_convolve(ft).max()
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a = 0.0
    b = 1.0
    c_ = b - gr * (b - a)
    d_ = a + gr * (b - a)
    fc = obj(c_)
    fd = obj(d_)
    
    for _ in range(n_iter):
        if fc < fd:
            b = d_
            d_ = c_
            fd = fc
            c_ = b - gr * (b - a)
            fc = obj(c_)
        else:
            a = c_
            c_ = d_
            fc = fd
            d_ = a + gr * (b - a)
            fd = obj(d_)
    
    best_alpha = (a + b) / 2.0
    best_obj = obj(best_alpha)
    return best_alpha, best_obj


def fw_optimize_improved(f0, time_limit, use_multi_peak=False, verbose=False):

    Improved LP-Frank-Wolfe with:
    - Multi-peak constraints

    s = f.sum()

    n = len(f)

    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()

    it = 0

            g = solve_lp_multi_peak(f, M_list, list(peaks))
        else:
            g = solve_lp_single_peak(f, M * m_relax)
        
        if g is None:
            m_relax = min(m_relax * 1.15, 5.0)
            if m_relax > 3.0:
                # Try with relaxed constraint
                g = solve_lp_single_peak(f, M * 1.5)
            if g is None:

                g = np.ones(n) / n
            else:
                m_relax = 1.0

        if g_sum < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:

                rng = np.random.default_rng((it * 7919 + hash(f.tobytes()) % 100000) % (2**31))
                noise = rng.standard_normal(n) * 0.003
                f = project_simplex(best_f + noise)
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.0
                no_improve = 0
            continue
        
        # Golden section line search
        alpha, obj_val = line_search(f, g, M, n_iter=15)

            f_new = (1.0 - alpha) * f + alpha * g
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()

def peak_equalization(f0, time_limit, verbose=False):
    """
    Directly try to equalize the top peaks of the convolution.
    At each step, identify the top peaks and nudge mass to reduce the highest.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = 0.002
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find all peaks within 5% of max
        threshold = M * 0.95
        active_peaks = np.where(conv >= threshold)[0]
        
        if len(active_peaks) == 0:
            active_peaks = np.array([np.argmax(conv)])
        
        # Compute weighted subgradient
        w = (conv[active_peaks] - threshold) / (M - threshold + 1e-15)
        w = np.maximum(w, 0)
        w /= w.sum()
        
        sg = np.zeros(n)
        for i, pk in enumerate(active_peaks):
            if w[i] < 1e-10:
                continue
            js = max(0, int(pk) - n + 1)
            je = min(n - 1, int(pk))
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[int(pk) - idx]
        
        # Project to tangent space
        sg = sg - np.mean(sg)
        
        f_new = project_simplex(f - lr * sg)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.01, 0.01)
        else:
            no_improve += 1
            lr *= 0.97
        
        if no_improve > 100:
            rng = np.random.default_rng((it * 449 + 7) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            lr = 0.002
            no_improve = 0
        
        if verbose and it % 200 == 0:
            rem = time_limit - (time.monotonic() - t0)
            print(f"    PEAK-EQ iter {it}: obj={best_obj:.10f}, rem={rem:.0f}s")
    
    return best_f, best_obj


def freq_smooth(f0, time_limit, alpha=0.1, verbose=False):
    """
    Frequency-domain smoothing: dampen high frequencies to reduce time-domain peaks.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    # FFT size
    fft_size = 1 << (2 * n - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute DFT of f (zero-padded)
        F = np.fft.rfft(f, fft_size)
        F_abs = np.abs(F)
        
        # Compute a "smoothing" mask that dampens high frequencies
        freqs = np.fft.rfftfreq(fft_size)
        # Smooth: keep low freqs, dampen high
        max_freq = 0.3  # fraction of Nyquist to keep
        mask = np.ones(len(F_abs))
        for i in range(1, len(F_abs)):
            if freqs[i] > max_freq:
                mask[i] = 1.0 - alpha * (freqs[i] / max_freq - 1.0)
                mask[i] = max(mask[i], 0.1)
        
        # Apply smoothing in frequency domain
        F_smooth = F * mask
        f_smooth = np.fft.irfft(F_smooth, fft_size)[:n]
        f_smooth = np.maximum(f_smooth, 0.0)
        
        s2 = f_smooth.sum()
        if s2 < 1e-15:
            f_smooth = f.copy()
            break
        f_smooth = f_smooth / s2
        
        # Blend
        f_new = (1 - alpha) * f + alpha * f_smooth
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break  # Frequency smoothing isn't helping
        
        if verbose and it % 100 == 0:
            print(f"    FREQ iter {it}: obj={best_obj:.10f}")
    
    return best_f, best_obj


def diverse_inits(n, parent, rng):
    """Generate diverse initializations using various structured patterns."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Edge-concentrated (U-shape)
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x).astype(float) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Strong edge (exponential decay from edges)
    decay = 0.3
    e1 = np.exp(-decay * x)
    e2 = np.exp(-decay * (n - 1 - x))
    strong_edge = e1 + e2 + 0.01
    inits.append(strong_edge / strong_edge.sum())
    
    # Asymmetric: more mass on left
    asym = np.exp(-0.5 * (x / (0.3 * n))**2) + 0.5 * np.exp(-0.5 * ((n-1-x) / (0.3 * n))**2) + 0.01
    inits.append(asym / asym.sum())
    
    # Asymmetric: more mass on right
    asym2 = 0.5 * np.exp(-0.5 * (x / (0.3 * n))**2) + np.exp(-0.5 * ((n-1-x) / (0.3 * n))**2) + 0.01
    inits.append(asym2 / asym2.sum())
    
    # Cosine with phase shift (breaks symmetry)
    for phase in [0.0, 0.3, 0.7, 1.1]:
        c = np.abs(np.cos(np.pi * (x / n + phase))) + 0.05
        inits.append(c / c.sum())
    
    # Two humps at asymmetric positions
    for pos1, pos2 in [(0.2, 0.8), (0.15, 0.75), (0.25, 0.9), (0.1, 0.7)]:
        h1 = np.exp(-((x - pos1 * (n-1)) / (0.1 * n))**2)
        h2 = np.exp(-((x - pos2 * (n-1)) / (0.1 * n))**2)
        two = h1 + h2 + 0.001
        inits.append(two / two.sum())
    
    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 0.2))
    
    # Sine-based (can be asymmetric with phase)
    for freq, phase in [(2, 0.5), (3, 0.3), (4, 0.7), (1, 0.4)]:
        s_ = np.abs(np.sin(freq * np.pi * x / n + phase)) + 0.1
        inits.append(s_ / s_.sum())
    
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Perturbed parent
            inits.append(project_simplex(px / ps + rng.standard_normal(n) * 0.01))
    
    return inits

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_single_peak(f, M):
    """Solve LP: max 1^T g s.t. (f*g)_lin[k] <= M for all k, g >= 0."""
    n = len(f)

    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None
