import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_lse_grad(weights, f):
    """Compute gradient of LSE-smoothed max convolution."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]

def lse_optimize(f0, time_limit, beta_start=20.0, beta_end=5000.0, 
                 lr_init=None, momentum_init=0.90, plateau_beta=100, plateau_dur=50):
    """LSE gradient descent with adaptive parameters and plateaus."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    if lr_init is None:
        lr_init = 2.0 / n

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_init
    momentum = momentum_init
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    plateau_count = 0
    best_beta_at_plateau = beta

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Softmax weights
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        # Gradient via FFT
        grad = compute_lse_grad(w, f)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad = grad / grad_norm

        # Momentum update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 5.0 / n)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.93
                momentum = max(0.85, momentum - 0.003)
            if no_improve > 100:

                scale = 0.001 + 0.002 * min(1.0, no_improve / 200.0)

                v = np.zeros(n)
                lr = max(2.0 / n, 0.001)

def sa_optimize(f0, time_limit, T_init=0.01, T_end=0.0001, n_iter_per_temp=50):
    """Simulated annealing on the simplex for fine-tuning."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    cur_obj = best_obj

    T = T_init
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    total_iters = 0

    while time.monotonic() - t0 < time_limit and T > T_end:
        for _ in range(n_iter_per_temp):
            if time.monotonic() - t0 > time_limit:
                break

            total_iters += 1
            # Propose perturbation
            scale = T * 0.5
            f_new = project_simplex(f + rng.standard_normal(n) * scale)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()

            delta = obj_new - cur_obj
            if delta < 0 or np.random.random() < np.exp(-delta / T):
                f = f_new
                cur_obj = obj_new
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_f = f.copy()
            else:
                f = f  # reject, keep current

        # Cool down
        T *= 0.95

        if total_iters % 5000 == 0:
            print(f"    SA T={T:.6f}, best={best_obj:.10f}, iters={total_iters}")

    return best_f, best_obj


def topk_subgrad_optimize(f0, time_limit, top_k=32, lr_init=0.005, momentum=0.95, temp=0.0001):
    """Subgradient method with top-K peak selection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    v = np.zeros(n)
    no_improve = 0
    it = 0
    cur_lr = lr_init
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Top-K peaks with softmax weighting
        k = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k)[-k:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(k)
            w[np.argmax(pv)] = 1.0
        else:
            w = w / w_sum

        # Subgradient
        sg = np.zeros(n)
        for i in range(k):
            if w[i] < 1e-14:
                continue
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[kk - idx]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            cur_lr = min(cur_lr * 1.002, 0.01)
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                cur_lr *= 0.95
            if no_improve > 200:
                rng = np.random.default_rng(it * 13 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                v = np.zeros(n)
                cur_lr = max(lr_init * 0.5, 0.001)
                no_improve = 0
                cur_temp *= 1.5

        cur_temp *= 1.003

        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG it={it}, obj={best_obj:.12f}, lr={cur_lr:.6f}, temp={cur_temp:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj

def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_grad_from_weights(w, f)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            no_improve += 200
            break
        grad = grad / grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.002, 6.0 / n)
        else:
            no_improve += 1
            if no_improve >= 20:
                lr *= 0.94
                momentum = max(0.86, momentum - 0.002)
            if no_improve > restart_after:
                rng = np.random.default_rng(it * 37 + 13)
                scale = 0.0005 + 0.003 * min(1.0, (no_improve - restart_after) / 100.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                v = np.zeros(n)
                lr = max(2.0 / n, 0.001)
                beta = min(beta * 1.8, beta_end)
                momentum = 0.89
                no_improve = 0
                plateau_count = 0

        if beta < beta_end:
            if plateau_count < plateau_len:
                plateau_count += 1
            else:
                beta = min(beta * beta_growth, beta_end)
                plateau_count = 0

        if it % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it}, obj={best_obj:.12f}, beta={beta:.0f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_f, best_obj


def multi_peak_subgrad(f0, time_limit, lr_init=0.004, momentum=0.94,
                       top_k=64, temp_init=0.00008, temp_growth=1.004):
    """Subgradient with softmax weighting over top-K peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    v = np.zeros(n)
    no_improve = 0
    it = 0
    cur_lr = lr_init
    cur_temp = temp_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        k = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k)[-k:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(k)
            w[np.argmax(pv)] = 1.0
        else:
            w = w / w_sum

        sg = np.zeros(n)
        for i in range(k):
            if w[i] < 1e-14:
                continue
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[kk - idx]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            cur_lr = min(cur_lr * 1.001, 0.008)
        else:
            no_improve += 1
            if no_improve % 40 == 0:
                cur_lr *= 0.96
            if no_improve > 250:
                rng = np.random.default_rng(it * 29 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.0008)
                v = np.zeros(n)
                cur_lr = max(lr_init * 0.4, 0.0008)
                no_improve = 0

        cur_temp = min(cur_temp * temp_growth, 0.5)

        if it % 1500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MP-SG it={it}, obj={best_obj:.12f}, lr={cur_lr:.6f}, t={elapsed:.0f}s")

    return best_f, best_obj


def sa_finetune(f0, time_limit, T_init=0.003, T_end=0.00003, cooling=0.97,
                n_per_T=25, scale_factor=0.4):
    """Simulated annealing with adaptive step size for fine-tuning."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    cur_obj = best_obj

    T = T_init
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    total = 0
    accept_count = 0

    while time.monotonic() - t0 < time_limit and T > T_end:
        for _ in range(n_per_T):
            if time.monotonic() - t0 > time_limit:
                break
            total += 1

            scale = T * scale_factor
            f_new = project_simplex(f + rng.standard_normal(n) * scale)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()

            delta = obj_new - cur_obj
            if delta < 0 or (T > 1e-12 and np.random.random() < np.exp(-delta / max(T, 1e-12))):
                f = f_new
                cur_obj = obj_new
                accept_count += 1
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_f = f.copy()

        T *= cooling
        if total % 20000 == 0:
            print(f"    SA T={T:.7f}, best={best_obj:.12f}, iters={total}, accept={accept_count/total:.2f}")

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, n_sweeps=3):
    """Coordinate-wise perturbation: try small adjustments to each pair of adjacent elements."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    rng = np.random.default_rng(2024)
    improving = True
    while improving and time.monotonic() - t0 < time_limit:
        improving = False
        for i in range(n - 1):
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            for eps in [0.001, 0.0005, 0.0002]:
                f_try = f.copy()
                f_try[i] += eps
                f_try[i + 1] -= eps
                if f_try[i] < 0 or f_try[i + 1] < 0:
                    continue
                conv_try = fast_convolve(f_try)
                obj_try = conv_try.max()
                if obj_try < best_obj - 1e-15:
                    f = f_try
                    best_f = f.copy()
                    best_obj = obj_try
                    improving = True
                    break
    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.10f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization pipeline:
    1. FW-LP for global structure
    2. LSE gradient descent (main workhorse) with adaptive beta
    3. Multi-peak subgradient refinement
    4. Simulated annealing for fine-tuning
    5. Final high-beta LSE polish
    6. Length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.10f}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = np.ones(448) / 448
        try_sequence(parent, best_tracker)

    n_parent = len(parent)

    # Phase 1: FW-LP (moderate time)
    print("\n=== Phase 1: FW-LP ===")
    t_alloc = min(180, deadline - time.monotonic() - 650)
    if t_alloc > 25:
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(parent, time_limit=t_alloc)
        try_sequence(a_opt, best_tracker)

    # Phase 2: LSE (main phase, generous time)
    print("\n=== Phase 2: LSE (main) ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(450, deadline - time.monotonic() - 350)
    if t_alloc > 30:
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, _ = lse_optimize(current, time_limit=t_alloc,
                                beta_start=12.0, beta_end=12000.0,
                                lr_init=2.5 / len(current), momentum_init=0.92,
                                beta_growth=1.003, plateau_len=100, restart_after=150)
        try_sequence(a_opt, best_tracker)

    # Phase 3: Multi-peak subgradient
    print("\n=== Phase 3: Multi-Peak Subgradient ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(200, deadline - time.monotonic() - 200)
    if t_alloc > 20:
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, _ = multi_peak_subgrad(current, time_limit=t_alloc,
                                       lr_init=0.003, momentum=0.95,
                                       top_k=max(32, len(current) // 8),
                                       temp_init=0.00005, temp_growth=1.003)
        try_sequence(a_opt, best_tracker)

    # Phase 4: Simulated annealing
    print("\n=== Phase 4: Simulated Annealing ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(200, deadline - time.monotonic() - 100)
    if t_alloc > 20:
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, _ = sa_finetune(current, time_limit=t_alloc,
                                T_init=0.002, T_end=0.00002,
                                cooling=0.975, n_per_T=20, scale_factor=0.35)
        try_sequence(a_opt, best_tracker)

    # Phase 5: Final high-beta LSE polish
    print("\n=== Phase 5: Final LSE Polish ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(150, deadline - time.monotonic() - 30)
    if t_alloc > 15:
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, _ = lse_optimize(current, time_limit=t_alloc,
                                beta_start=500.0, beta_end=30000.0,
                                lr_init=0.3 / len(current), momentum_init=0.93,
                                beta_growth=1.002, plateau_len=150, restart_after=200)
        try_sequence(a_opt, best_tracker)

    # Phase 6: Length exploration (only if time remains)
    print("\n=== Phase 6: Length Exploration ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent

    alt_lengths = [420, 475, 500, 384]
    explored = 0
    for n_alt in alt_lengths:
        if n_alt == n_parent:
            continue
        t_remaining = deadline - time.monotonic() - 15
        if t_remaining < 45:
            break
        if explored >= 2:
            break
        t_alloc = min(50, t_remaining / max(1, len(alt_lengths) - explored))
        if t_alloc < 20:
            break
        explored += 1

        x_old = np.linspace(0, 1, len(current))
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current)
        f_interp = np.maximum(f_interp, 0)
        if f_interp.sum() > 1e-10:
            f_interp = f_interp / f_interp.sum()
        print(f"  Trying n={n_alt}")
        a_opt, _ = lse_optimize(f_interp, time_limit=t_alloc,
                                beta_start=15.0, beta_end=3000.0,
                                lr_init=2.5 / n_alt, momentum_init=0.91)
        try_sequence(a_opt, best_tracker)

    # Final output
    final_seq = best_tracker[1]
    if final_seq is not None:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]
    else:
        seq = np.ones(448) / 448
        return [float(x) for x in seq]
