This creates sidebands at frequencies ±1 in the auto-spectrum, which in the time domain correspond to a modulation of the auto-convolution.

The effect on the central peak: the auto-convolution at position n-1 gets contributions from all frequencies. The modulation might reduce the peak at n-1 while increasing peaks at other positions.

def precise_coord_descent(f0, time_limit, verbose=False):
    """Precise coordinate descent using analytical conv_k(x) formula."""
    f = normalize(f0)
    n = len(f)
    best_f = f.copy()
    best_peak = fast_convolve(f).max()
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        conv = fast_convolve(f)
        
        for i in range(n):
            old = f[i]
            if old < 1e-10 or 1-old < 1e-10:
                continue
            om = 1.0 - old  # one_minus_old
            
            # Compute cross[k] = a[k-i] if 0 <= k-i < n, else 0
            cross = np.zeros(2*n-1)
            cross[i:i+n] = f
            
            # R_k = conv[k] - 2*old*cross[k] + old^2 * delta(k,2i)
            R = conv - 2.0*old*cross
            R[2*i] += old*old
            
            # C_k = 2*cross[k] for k != 2i, 0 for k=2i
            C = 2.0*cross
            C[2*i] = 0.0
            
            delta = np.zeros(2*n-1)
            delta[2*i] = 1.0
            
            # conv_k(x) = R*(1-x)^2/om^2 + C*x*(1-x)/om + delta*x^2
            # Grid search for best x
            # Coarse grid
            x_grid = np.linspace(0, 1, 30)
            # Actually, let's restrict to a reasonable range
            # x should be close to old; let's try [max(0, old-2*old), min(1, old+2*old)]
            # But also include 0 and 1
            x_lo = max(0.0, old * 0.1)
            x_hi = min(1.0, old * 3.0)
            x_grid = np.linspace(x_lo, x_hi, 30)
            
            # Evaluate max conv at each grid point (vectorized)
            best_x = old
            best_val = conv.max()
            
            for x in x_grid:
                t = (1.0-x)/om
                # conv_at_x = R * t^2 + C * x * t + delta * x^2
                conv_x = R * t*t + C * x * t + delta * x*x
                val = conv_x.max()
                if val < best_val:
                    best_val = val
                    best_x = x
            
            # Fine grid around best_x
            if best_x != old:
                eps = (x_hi - x_lo) / 30.0
                fine_grid = np.linspace(max(0, best_x - eps), min(1, best_x + eps), 30)
                for x in fine_grid:
                    t = (1.0-x)/om
                    conv_x = R * t*t + C * x * t + delta * x*x
                    val = conv_x.max()
                    if val < best_val:
                        best_val = val
                        best_x = x
                
                # Update f
                f[i] = best_x
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                f[mask] *= (1.0 - best_x) / om
                f /= f.sum()  # numerical safety
                
                if best_val < best_peak - 1e-15:
                    best_peak = best_val
                    best_f = f.copy()
                    improved = True
                conv = fast_convolve(f)
        
        if not improved and pass_count > 2:
            break
    
    return best_f, best_peak
