        else:
            no_improve += 1

        f = f_new
        cur_lr *= 0.9997
        cur_temp *= 1.0008

        if no_improve > 250:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 99999999)
            scale = 0.001 + 0.003 * min(1.0, no_improve / 500.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.6, 0.0003)
            cur_temp = temp_init * 2.5
            no_improve = 0

    return best_f, best_obj


def coord_descent_polish(f0, time_limit, verbose):
    """
    Coordinate descent: pick the top peak, find best 2-coord transfer to reduce it.
    Uses bisection on the transfer amount.
    """
    f = np.maximum(f0.copy(), 0.0)
    f = f / f.sum()
    n = len(f)

    conv = fast_conv(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_conv(f)
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]

        k = peak_idx
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            break

        idx_arr = np.arange(js, je + 1)
        # Impact of each coordinate j on peak k: proportional to 2*f[k-j]
        # We want to decrease the coordinate with highest f[j]*f[k-j]
        contributions = f[idx_arr] * f[k - idx_arr]
        src = idx_arr[np.argmax(contributions)]

        # Find target: a coordinate NOT in the peak's support (zero impact on this peak)
        in_peak = set(idx_arr.tolist())
        out_peak = [i for i in range(n) if i != src and i not in in_peak]

        if not out_peak:
            out_peak = [i for i in range(n) if i != src]

        # Pick target that minimens impact: prefer coordinates not in any top peak
        # For simplicity, pick randomly from out_peak with bias toward small values
        n_cand = min(10, len(out_peak))
        candidates = rng.choice(len(out_peak), size=n_cand, replace=False)
        best_target = out_peak[candidates[0]]
        best_trial_obj = peak_val

        for ci in range(n_cand):
            target = out_peak[candidates[ci]]
            # Estimate: moving mass from src to target changes peak by
            # -2*f[src]*f[k-src] + 2*f[target]*f[k-target] (if target in S_k)
            # Since target not in S_k, the second term is 0 for this peak
            # But it might affect other peaks. Just try it.
            delta = 0.3 * f[src]
            f_trial = f.copy()
            f_trial[src] -= delta
            f_trial[target] += delta
            f_trial = np.maximum(f_trial, 0)
            f_trial = f_trial / f_trial.sum()
            conv_trial = fast_conv(f_trial)
            obj_trial = conv_trial.max()
            if obj_trial < best_trial_obj:
                best_trial_obj = obj_trial
                best_target = target

        # Bisection for best delta
        delta_lo = 0.0
        delta_hi = 0.5 * f[src]
        best_delta = 0.0
        best_peak = peak_val

        for _ in range(12):
            mid = (delta_lo + delta_hi) / 2.0
            f_trial = f.copy()
            f_trial[src] -= mid
            f_trial[best_target] += mid
            f_trial = np.maximum(f_trial, 0)
            f_trial = f_trial / f_trial.sum()
            conv_trial = fast_conv(f_trial)
            p = conv_trial.max()
            if p < best_peak:
                best_peak = p
                best_delta = mid
                delta_lo = mid
            else:
                delta_hi = mid

        if best_delta > 1e-10:
            f[src] -= best_delta
            f[best_target] += best_delta
            f = np.maximum(f, 0)
            f = f / f.sum()
            conv = fast_conv(f)
            obj = conv.max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
        else:
            # No improvement from this peak, try a random perturbation
            if it % 50 == 0:
                i1 = rng.integers(0, n)
                i2 = rng.integers(0, n)
                if i1 != i2:
                    delta = 0.01 * f[i1]
                    f[i1] -= delta
                    f[i2] += delta
                    f = np.maximum(f, 0)
                    f = f / f.sum()

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    # Edge-weighted
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian centered
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())

    # Raised cosine
    c = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.05
    inits.append(c / c.sum())

    # Power law from edges
    pw = np.minimum(x, n - 1 - x) ** 0.5 + 0.1
    inits.append(pw / pw.sum())

    # Interpolated parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase, multi-n search:
    Phase 1: Quick exploration of n values (250s)
    Phase 2: Deep mirror descent (500s)
    Phase 3: Subgradient with momentum (500s)
    Phase 4: Coordinate descent polish (300s)
    Phase 5: Final fine polish (200s)
    """
    rng = np.random.default_rng(42)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(compute_objective(a_norm))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print("  NEW BEST [{}]: {:.10f} (n={})".format(label, score, len(a)))
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            pscore = compute_objective(parent)
            print("Parent loaded: n={}, score={:.10f}".format(len(parent), pscore))
            try_seq(parent, "parent")
    except Exception:
        pass

    # === Phase 1: Quick multi-n exploration ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    n_candidates = [256, 320, 384, 448, 512, 640]
    t1_per = 25.0  # seconds per n

    for nc in n_candidates:
        rem = deadline - time.monotonic() - 1400
        if rem < 10:
            break
        t_alloc = min(t1_per, rem)
        inits = generate_inits(nc, parent, rng)
        phase_best = None
        phase_best_obj = float('inf')
        for ii, init in enumerate(inits):
            rem2 = deadline - time.monotonic() - 1400
            if rem2 < 5:
                break
            t_init = max(2.0, t_alloc / len(inits))
            a_opt, obj_opt = subgradient_nesterov(
                init, time_limit=t_init, lr_init=0.01, momentum=0.92,
                top_k=nc // 5, temp_init=0.002, verbose=False
            )
            if obj_opt < phase_best_obj:
                phase_best_obj = obj_opt
                phase_best = a_opt
        if phase_best is not None:
            score = 2.0 * nc * phase_best_obj
            if score < best_score:
                best_score = score
                best_seq = phase_best.copy()
                print("  Phase1 n={}: {:.10f}".format(nc, score))

    print("  Phase 1 best: {:.10f}".format(best_score))

    # === Phase 2: Deep mirror descent ===
    print("\n=== Phase 2: Deep Mirror Descent ===")
    if best_seq is not None:
        n_best = len(best_seq)
        mw_configs = [
            (0.04, 0.0005, 0.9995, 1.0003, n_best // 8, 150),
            (0.06, 0.0003, 0.999, 1.0005, n_best // 6, 200),
            (0.03, 0.001, 0.9997, 1.0002, n_best // 10, 300),
            (0.05, 0.0008, 0.9993, 1.0004, n_best // 5, 180),
        ]
        for ci, cfg in enumerate(mw_configs):
            rem = deadline - time.monotonic() - 900
            if rem < 10:
                break
            t_alloc = min(cfg[5], rem)
            lr_i, temp_i, d_lr, d_temp, tk, _ = cfg
            a_opt, _ = mirror_descent(
                best_seq, time_limit=t_alloc,
                lr_init=lr_i, temp_init=temp_i,
                decay_lr=d_lr, decay_temp=d_temp,
                top_k=tk, restart_every=150, verbose=False
            )
            try_seq(a_opt, "mw_{}".format(ci))

    print("  Phase 2 best: {:.10f}".format(best_score))

    # === Phase 3: Subgradient with momentum ===
    print("\n=== Phase 3: Subgradient Nesterov ===")
    if best_seq is not None:
        n_best = len(best_seq)
        sg_configs = [
            (0.008, 0.92, n_best // 6, 0.003),
            (0.005, 0.94, n_best // 5, 0.002),
            (0.012, 0.90, n_best // 8, 0.004),
            (0.003, 0.95, n_best // 4, 0.001),
        ]
        for si, cfg in enumerate(sg_configs):
            rem = deadline - time.monotonic() - 500
            if rem < 10:
                break
            t_alloc = min(rem, 120)
            lr_i, mom_i, tk_i, temp_i = cfg
            a_opt, _ = subgradient_nesterov(
                best_seq, time_limit=t_alloc,
                lr_init=lr_i, momentum=mom_i,
                top_k=tk_i, temp_init=temp_i, verbose=False
            )
            try_seq(a_opt, "sg_{}".format(si))

        # Try from a perturbed start
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            noise = rng.standard_normal(n_best) * 0.008
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = subgradient_nesterov(
                f_pert, time_limit=min(rem, 100),
                lr_init=0.006, momentum=0.93,
                top_k=n_best // 5, temp_init=0.003, verbose=False
            )
            try_seq(a_opt, "sg_perturbed")

    print("  Phase 3 best: {:.10f}".format(best_score))

    # === Phase 4: Coordinate descent polish ===
    print("\n=== Phase 4: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_alloc = min(rem, 250)
            print("  CD polish, t={:.0f}s".format(t_alloc))
            a_opt, _ = coord_descent_polish(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "cd")

    print("  Phase 4 best: {:.10f}".format(best_score))

    # === Phase 5: Final fine polish ===
    print("\n=== Phase 5: Final Polish ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine mirror descent
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_alloc = min(rem, 80)
            a_opt, _ = mirror_descent(
                best_seq, time_limit=t_alloc,
                lr_init=0.005, temp_init=0.0001,
                decay_lr=0.9998, decay_temp=1.0001,
                top_k=max(5, n_best // 15), restart_every=300, verbose=False
            )
            try_seq(a_opt, "fine_mw")

        # Very fine subgradient
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_alloc = min(rem, 60)
            a_opt, _ = subgradient_nesterov(
                best_seq, time_limit=t_alloc,
                lr_init=0.001, momentum=0.96,
                top_k=max(5, n_best // 10), temp_init=0.0005, verbose=False
            )
            try_seq(a_opt, "fine_sg")

        # Tiny perturbations with quick refinement
        rem = deadline - time.monotonic() - 30
        for pi in range(4):
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem - 5, 20)
            a_opt, _ = subgradient_nesterov(
                f_pert, time_limit=t_alloc,
                lr_init=0.002, momentum=0.95,
                top_k=max(5, n_best // 8), temp_init=0.001, verbose=False
            )
            try_seq(a_opt, "perturb_{}".format(pi))
            rem = deadline - time.monotonic() - 10

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(compute_objective(best_seq))

    # Final verification with evaluate_sequence
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print("\nFINAL SCORE: {:.10f} (n={})".format(best_score, len(best_seq)))
    return [float(x) for x in best_seq]
