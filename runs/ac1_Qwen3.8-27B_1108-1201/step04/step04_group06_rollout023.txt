"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Coarse Nesterov gradient optimization with soft-peak gradient
2. LP-based fine optimization with cutting-plane constraint selection
3. Multi-length exploration (1024-2048) with hierarchical extension
4. Adaptive restarts, perturbation, and multi-phase time allocation
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

# Cache for precomputed sparsity patterns
_SPARSE_CACHE = {}

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def convolve_direction(a, g):
    """Compute conv(a, g) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(A * G, fft_size)[:size]

def get_sparse_pattern(n):
    """Get or compute the sparsity pattern for convolution matrix of size n."""
    if n in _SPARSE_CACHE:
        return _SPARSE_CACHE[n]
    total = n * n
    rows = np.zeros(total, dtype=np.int32)
    cols = np.zeros(total, dtype=np.int32)
    idx = 0
    for k in range(2 * n - 1):
        j0 = max(0, k - n + 1)
        j1 = min(n - 1, k)
        count = j1 - j0 + 1
        rows[idx:idx + count] = k
        cols[idx:idx + count] = np.arange(j0, j1 + 1)
        idx += count
    pattern = (rows, cols)
    _SPARSE_CACHE[n] = pattern
    return pattern

def build_C(a):
    """Build sparse matrix C where C @ g = conv(a, g). C[k,j] = a[k-j]."""
    n = len(a)
    rows, cols = get_sparse_pattern(n)
    total = n * n
    vals = np.zeros(total, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j0 = max(0, k - n + 1)
        j1 = min(n - 1, k)
        count = j1 - j0 + 1
        v_start = k - j1
        v_end = k - j0
        vals[idx:idx + count] = a[v_start:v_end + 1]
        idx += count
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def gradient_soft(a, conv, top_k=50, temp=0.01):
    """Compute (sub)gradient of soft-peak of convolution."""
    n = len(a)
    k_eff = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_eff)[-k_eff:]
    pv = conv[peaks]
    s = pv / max(temp, 1e-10)
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-15:
        wf = np.zeros(len(conv))
        wf[peaks[np.argmax(pv)]] = 1.0
    else:
        wf = np.zeros(len(conv))
        wf[peaks] = w / ws
    grad = np.zeros(n)
    for m in range(n):
        ks = m
        ke = min(2 * n - 1, m + n)
        if ks < ke:
            grad[m] = np.dot(wf[ks:ke], a[:ke - ks])
    return grad

def grad_optimize(a0, t_limit, lr=0.01, mom=0.91, top_k=40, temp=0.003):
    """Nesterov-accelerated gradient descent with soft-peak gradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(y)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_imp = 0
    while time.monotonic() - t0 < t_limit:
        y = np.maximum(a + mom * v, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y /= sy
        cy = fast_convolve(y)
        g = gradient_soft(y, cy, top_k=top_k, temp=temp)
        v = mom * v - lr * g
        a_new = np.maximum(a + v, 0.0)
        sn = a_new.sum()
        if sn > 1e-15:
            a = a_new / sn
        else:
            a = best_a.copy()
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        lr *= 0.9994
        temp *= 1.003
        mom = min(0.97, mom + 0.00015)
        if no_imp >= 400:
            break
    return best_a, best_obj

def lp_optimize(a0, t_limit, verbose=False):
    """
    LP-based iterative optimization.
    At each step: solve max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Then take a step toward g with careful sizing.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    perturb_count = 0
    max_no_improve = 60
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < t_limit:
        # Solve LP
        C = build_C(a)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds_lp = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds_lp,
                          method='highs', options={'presolve': True,
                                                   'dual_feasibility_tolerance': 1e-9})
            success = res.success
        except Exception:
            success = False

        if success and res.x is not None:
            g = res.x
            g = np.maximum(g, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                d = g - a
                norm_d = np.linalg.norm(d)
                if norm_d > 1e-12:
                    # Step sizing: try several step sizes
                    best_t = 0.0
                    best_a_try = a.copy()
                    best_obj_try = best_obj
                    candidates = [0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
                    for t in candidates:
                        a_try = a + t * d
                        a_try = np.maximum(a_try, 0.0)
                        s_try = a_try.sum()
                        if s_try < 1e-15:
                            continue
                        a_try = a_try / s_try
                        conv_try = fast_convolve(a_try)
                        obj_try = 2.0 * n * conv_try.max()
                        if obj_try < best_obj_try - 1e-13:
                            best_obj_try = obj_try
                            best_a_try = a_try.copy()
                            best_t = t
                            break
                    if best_t > 0:
                        a = best_a_try
                        conv = fast_convolve(a)
                        M = conv.max()
                        cur_obj = 2.0 * n * M
                        if cur_obj < best_obj - 1e-12:
                            best_obj = cur_obj
                            best_a = a.copy()
                        no_improve = 0
                        iteration += 1
                        if verbose and iteration % 50 == 0:
                            print(f"    LP iter {iteration}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")
                        continue

        # No improvement - handle stagnation
        no_improve += 1
        if no_improve >= max_no_improve:
            scale = 0.002 + 0.004 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(300, 60 + 30 * perturb_count)
            if perturb_count > 60:
                break

        # Periodic gradient restart
        if no_improve >= 30 and no_improve % 30 == 0:
            a_g, obj_g = grad_optimize(a, time_limit=min(3.0, t_limit - (time.monotonic() - t0)),
                                       lr=0.005, top_k=60, temp=0.002)
            if obj_g < best_obj:
                best_obj = obj_g
                best_a = a_g.copy()
                a = a_g.copy()
                conv = fast_convolve(a)
                M = conv.max()
            no_improve = 0

    return best_a, best_obj

    rng = np.random.default_rng(999)
    no_imp = 0
    while time.monotonic() - t0 < t_limit:

                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:

                if a[i] + delta < 0 or a[j] - delta < 0:

                a_try[i] += delta
                a_try[j] -= delta

                obj_try = 2.0 * n * conv_try.max()

                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    no_imp = 0
                    break
            else:
                no_imp += 1
        if no_imp > 500:
            break
    return best_a, best_obj

def interpolate_to_length(a_src, n_target):
    """Interpolate a sequence to a new length."""
    n_src = len(a_src)
    s = a_src.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a_src / s
    if n_target == n_src:
        return a_norm.copy()
    a_ext = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, n_src), a_norm)
    a_ext = np.maximum(a_ext, 0.0)
    ses = a_ext.sum()
    if ses > 1e-15:
        a_ext /= ses
    return a_ext

    s2 = np.sin(2 * np.pi * x / n) ** 2 + 0.05

    # Edge-concentrated (U-shape, non-palindromic)
    edge_w = np.exp(-0.5 * ((x - 0.1 * n) / (0.15 * n)) ** 2) + np.exp(-0.5 * ((x - 0.9 * n) / (0.15 * n)) ** 2)
    inits['bimodal'] = edge_w / edge_w.sum()

    # Asymmetric: more mass on one side

    inits['asym'] = asym / asym.sum()

    # Power law (decreasing)
    pw = (n - x + 1.0) ** 0.5
    inits['power'] = pw / pw.sum()

    # Linear ramp
    ramp = x + 1.0
    inits['ramp'] = ramp / ramp.sum()

        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():

    return inits
