from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(f, M):
    n = len(f)
    c = -np.ones(n)
    # Build A_ub as a sparse-like structure using the Toeplitz property
    # (f*g)[k] = sum_j f[j] * g[k-j] <= M
    # For fixed f, this is linear in g: sum_j f[j] * g[k-j] <= M
    # The coefficient of g[j] in constraint k is f[k-j] (if 0 <= k-j < n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] appears in constraints k where 0 <= k-j < n, i.e., j <= k <= j+n-1
        # The coefficient is f[k-j]
        A_ub[j:j+n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

                        je = min(n - 1, k)
                        idx = np.arange(js, je + 1)
                        sg[idx] += 2.0 * best_f[k - idx]
                    sg /= (len(peaks) + 1e-10)
                    f = project_simplex(best_f - 0.003 * sg)
                else:
                    # Edge boost
                    x = np.arange(n, dtype=np.float64)
                    boost = np.exp(-0.5 * (x / (n * 0.08)) ** 2)
                    f = project_simplex(best_f * (1 + 0.05 * boost))

                conv = fast_convolve(f)
                M = conv.max()
                m_factor = 1.0
                no_improve = 0
                step_scale = 1.0
                continue
            continue

        # Line search: coarse then fine
        best_alpha = 0.0
        best_val = M

        # Adaptive alpha range based on convergence
        if no_improve == 0:
            alphas = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        else:
            alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0]

        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if 0.03 < best_alpha < 0.97:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-9:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
                conv = fast_convolve(f)
                M = conv.max()
            else:
                f = f / fs
                conv = fast_convolve(f)
                M = conv.max()
                m_factor = 1.0

            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.12f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_factor = min(m_factor * 1.1, 5.0)

    return best_f, best_obj


def subgradient_polish(f0, time_limit, seed=0):
    """Subgradient descent with multi-peak weighting and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    lr = 0.003
    mom = 0.92
    top_k = max(5, n // 6)
    temp = 0.00004
    rng_local = np.random.default_rng(88888 + seed)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1:
            break

        # Find top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Momentum update
        v = mom * v - lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-16:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9996
        temp *= 1.003

        if no_improve > 300:
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = max(0.003 * 0.3, 0.0005)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, seed=0):
    """
    Subgradient-guided coordinate descent with exact 1D line search.
    For each coordinate, finds the best mass transfer partner and optimal delta.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng_local = np.random.default_rng(77777 + seed)

    # Active set: conv positions near the max
    conv = fast_convolve(f)
    active_tol = best_obj * 0.9
    active_set = np.where(conv >= active_tol)[0]

    stall = 0
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 2:
            break

        # Update active set periodically
        if stall % 50 == 0:
            conv = fast_convolve(f)
            active_set = np.where(conv >= best_obj * 0.9)[0]

        improved_this_round = False
        order = rng_local.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit - 2:
                break

            # Build candidate sources
            candidates = list(set([
                (j + 1) % n,
                (j - 1) % n,
                (j + n // 3) % n,
                (j - n // 3) % n,
                (j + n // 2) % n,
            ]))
            # Add sources from active set
            for k in active_set[:15]:
                js_max = max(0, k - n + 1)
                je_max = min(n - 1, k)
                if js_max <= j <= je_max:
                    src = k - j
                    if 0 <= src < n and src != j:
                        candidates.append(src)

            best_delta_this_j = 0.0
            best_source = -1
            best_val = best_obj

            for source in candidates:
                source = int(source) % n
                if source == j:
                    continue
                if f[source] < 1e-12:
                    continue

                max_delta = f[source] * 0.95
                # Binary search on delta (convex function in delta)
                lo = 0.0
                hi = max_delta
                # First check if moving helps at all
                mid_test = max_delta * 0.3
                f_new = f.copy()
                f_new[j] += mid_test
                f_new[source] -= mid_test
                f_new = np.maximum(f_new, 0.0)
                val_test = fast_convolve(f_new).max()
                if val_test >= best_obj and mid_test < max_delta:
                    # Try smaller
                    mid_test = max_delta * 0.1
                    f_new = f.copy()
                    f_new[j] += mid_test
                    f_new[source] -= mid_test
                    f_new = np.maximum(f_new, 0.0)
                    val_test = fast_convolve(f_new).max()
                    if val_test >= best_obj:
                        continue

                # Full binary search
                for _ in range(14):
                    mid = (lo + hi) / 2.0
                    f_new = f.copy()
                    f_new[j] += mid
                    f_new[source] -= mid
                    f_new = np.maximum(f_new, 0.0)
                    sn = f_new.sum()
                    if sn < 1e-15:
                        break
                    fn = f_new / sn
                    val = fast_convolve(fn).max()
                    if val < best_val:
                        best_val = val
                        best_delta_this_j = mid
                        best_source = source
                        lo = mid
                    else:
                        hi = mid

            if best_source >= 0 and best_delta_this_j > 1e-12:
                f_new = f.copy()
                f_new[j] += best_delta_this_j
                f_new[best_source] -= best_delta_this_j
                f_new = np.maximum(f_new, 0.0)
                f = f_new / f_new.sum()
                best_obj = best_val
                best_f = f.copy()
                improved_this_round = True
                stall = 0

        if not improved_this_round:
            # Try random pairs as escape
            any_improve = False
            for _ in range(30):
                if time.monotonic() - t0 > time_limit - 2:
                    break
                i = int(rng_local.integers(n))
                j2 = int(rng_local.integers(n))
                if i == j2:
                    continue
                max_d = min(f[i], f[j2]) * 0.5
                if max_d < 1e-12:
                    continue
                lo, hi = 0.0, max_d
                for _ in range(10):
                    mid = (lo + hi) / 2.0
                    for direction in [1, -1]:
                        f_new = f.copy()
                        if direction == 1:
                            f_new[i] -= mid
                            f_new[j2] += mid
                        else:
                            f_new[i] += mid
                            f_new[j2] -= mid
                        f_new = np.maximum(f_new, 0.0)
                        sn = f_new.sum()
                        if sn < 1e-15:
                            continue
                        fn = f_new / sn
                        val = fast_convolve(fn).max()
                        if val < best_obj:
                            best_obj = val
                            f = fn
                            best_f = f.copy()
                            if direction == 1:
                                lo = mid
                            else:
                                lo = mid
                            any_improve = True
                            break
                        else:
                            hi = mid
                    if any_improve:
                        break
                if any_improve:
                    break

            if not any_improve:
                stall += 1
                if stall > 5:
                    break
                # Small perturbation to escape
                f = project_simplex(f + rng_local.standard_normal(n) * 0.0005)

    return best_f, best_obj


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.7))

    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    sin_w = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sin_w / sin_w.sum())

    cos_t = np.cos(np.pi * x / (2.0 * max(n - 1, 1)))
    cos_t = np.maximum(cos_t, 0.05)
    inits.append(cos_t / cos_t.sum())

    # Two-peak
    peak = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 6.0) ** 2)) +
            np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 6.0) ** 2)))
    inits.append(peak / peak.sum())

    # Power law
    pl = (x + 1.0) ** (-0.3)
    inits.append(pl / pl.sum())

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Uses LP-based Frank-Wolfe as primary optimizer,
    subgradient and coordinate descent for polishing, with multi-length
    exploration and smart time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    state = {'best_seq': None, 'best_score': float('inf')}

    def try_seq(a, label):
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < state['best_score']:
            state['best_score'] = score
            state['best_seq'] = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            s = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={s:.8f}")
            if np.isfinite(s):
                state['best_score'] = s
                state['best_seq'] = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        print("No parent available, using uniform n=384")
        s = float(evaluate_sequence(parent.tolist()))
        state['best_score'] = s
        state['best_seq'] = parent.copy()

    n_parent = len(parent)

    # === Phase 1: Deep FW on parent (35% budget) ===
    print(f"\n=== Phase 1: Deep FW (n={n_parent}) ===")
    t1_budget = float(budget_s) * 0.30
    t1 = min(t1_budget, deadline - time.monotonic() - 400)
    if t1 > 30:
        print(f"  Allocating {t1:.0f}s")
        a_opt, obj = fw_optimize(parent, time_limit=t1, verbose=True, seed=1)
        try_seq(a_opt, "fw_parent_deep")

    # === Phase 2: Subgradient polish (10% budget) ===
    print(f"\n=== Phase 2: Subgradient Polish ===")
    if state['best_seq'] is not None:
        t2 = min(float(budget_s) * 0.10, deadline - time.monotonic() - 200)
        if t2 > 20:
            print(f"  Allocating {t2:.0f}s")
            a_opt, obj = subgradient_polish(state['best_seq'], time_limit=t2, seed=2)
            try_seq(a_opt, "sg_polish")

    # === Phase 3: Coordinate descent (15% budget) ===
    print(f"\n=== Phase 3: Coordinate Descent ===")
    if state['best_seq'] is not None:
        t3 = min(float(budget_s) * 0.15, deadline - time.monotonic() - 100)
        if t3 > 15:
            print(f"  Allocating {t3:.0f}s")
            a_opt, obj = coordinate_descent(state['best_seq'], time_limit=t3, seed=3)
            try_seq(a_opt, "cd_main")

    # === Phase 4: Perturbation + FW restarts (20% budget) ===
    print(f"\n=== Phase 4: Perturbation + FW ===")
    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        pert_configs = [
            (0.001, "gauss"),
            (0.003, "gauss"),
            (0.01, "gauss"),
            (0.03, "gauss"),
            (0.0, "mirror"),
            (0.0, "sg_dir"),
        ]
        for ps, ptype in pert_configs:
            rem = deadline - time.monotonic() - 20
            if rem < 25:
                break
            n_remaining = len(pert_configs) - pert_configs.index((ps, ptype))
            t_alloc = min(rem / max(n_remaining, 1), 100)
            if t_alloc < 15:
                t_alloc = min(rem, 30)
                if t_alloc < 10:
                    break

            if ptype == "gauss":
                noise = rng.standard_normal(n_best) * ps
                f_init = project_simplex(state['best_seq'] + noise)
            elif ptype == "mirror":
                f_init = project_simplex(0.8 * state['best_seq'] + 0.2 * state['best_seq'][::-1])
            else:  # sg_dir
                conv_cur = fast_convolve(state['best_seq'])
                peaks = np.where(conv_cur >= conv_cur.max() * 0.98)[0]
                sg = np.zeros(n_best)
                for k in peaks:
                    js = max(0, k - n_best + 1)
                    je = min(n_best - 1, k)
                    idx = np.arange(js, je + 1)
                    sg[idx] += 2.0 * state['best_seq'][k - idx]
                sg /= (len(peaks) + 1e-10)
                f_init = project_simplex(state['best_seq'] - 0.01 * sg)

            a_opt, obj = fw_optimize(f_init, time_limit=t_alloc, seed=int(ps * 10000))
            try_seq(a_opt, f"pert_{ptype}_{ps}")

    # === Phase 5: Nearby lengths (10% budget) ===
    print(f"\n=== Phase 5: Nearby Lengths ===")
    if state['best_seq'] is not None:
        nearby = []
        for dn in [-16, 16, -32, 32, -48, 48]:
            nl = n_parent + dn
            if nl >= 128 and nl <= 1024:
                nearby.append(nl)

        for n_try in nearby[:4]:
            rem = deadline - time.monotonic() - 15
            if rem < 25:
                break
            t_alloc = min(rem / 2, 80)
            if t_alloc < 10:
                break

            # Interpolate to new length
            x_new = np.linspace(0, 1, n_try)
            x_old = np.linspace(0, 1, len(state['best_seq']))
            f_init = np.interp(x_new, x_old, state['best_seq'])
            f_init = np.maximum(f_init, 0)
            f_init = f_init / f_init.sum()

            a_opt, obj = fw_optimize(f_init, time_limit=t_alloc, seed=n_try)
            try_seq(a_opt, f"n{n_try}")

    # === Phase 6: Final polish with remaining time ===
    print(f"\n=== Phase 6: Final Polish ===")
    if state['best_seq'] is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 15:
            # Alternate between CD and SG
            half = rem / 2
            a_opt, obj = coordinate_descent(state['best_seq'], time_limit=half, seed=42)
            try_seq(a_opt, "cd_final")

            rem2 = deadline - time.monotonic() - 5
            if rem2 > 10:
                a_opt, obj = subgradient_polish(state['best_seq'], time_limit=rem2, seed=43)
                try_seq(a_opt, "sg_final")

        # One more CD pass
        rem = deadline - time.monotonic() - 3
        if rem > 8:
            a_opt, obj = coordinate_descent(state['best_seq'], time_limit=rem, seed=44)
            try_seq(a_opt, "cd_last")

    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(384) / 384
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))
        print(f"  Fallback uniform: {state['best_score']:.8f}")

    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {state['best_score']:.8f} (n={len(state['best_seq'])})")
    print(f"{'='*50}")

    return [float(x) for x in state['best_seq']]
