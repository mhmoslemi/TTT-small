def nesterov_subgrad_optimize(f0, time_limit, lr_init=0.01, momentum=0.9,
                               top_k=40, temp_init=0.001, temp_end=0.00005,
                               verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    y = f.copy()  # Nesterov lookahead point
    v = np.zeros(n)  # momentum velocity
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr_init
    cur_temp = temp_init
    cur_mom = momentum

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        y = f + cur_mom * v

        # Compute subgradient at y
        conv_y = fast_convolve(y)
        sg = compute_subgradient(y, conv_y, top_k, cur_temp)

        # Momentum update
        v = cur_mom * v - cur_lr * sg

        # Project onto simplex
        f_new = project_simplex(y + v)

        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

def build_conv_matrix_rows(f, peaks):
    """Build rows of the convolution matrix for given peak indices."""
    n = len(f)
    k = len(peaks)
    B = np.zeros((k, n))
    for i, pk in enumerate(peaks):
        js = max(0, pk - n + 1)
        je = min(n - 1, pk)
        B[i, js:je+1] = f[pk - js:pk - js + (je-js+1)][::-1]
    return B


def solve_linearized_lp_reduced(f, conv, top_k=30):
    """Solve reduced linearized LP: min t s.t. t >= conv[k] + 2*(B*g)[k], g >= -f, sum(g)=0."""
    n = len(f)
    size = len(conv)
    if top_k >= size:
        peaks = np.arange(size)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    peaks = np.sort(peaks)
    k = len(peaks)

    # Variables: [t, g_0, ..., g_{n-1}]
    c = np.zeros(n + 1)
    c[0] = 1.0

    # Build constraint matrix: -t + 2*(B*g) <= -conv[peaks]
    B_rows = np.zeros((k, n))
    for i, pk in enumerate(peaks):
        js = max(0, pk - n + 1)
        je = min(n - 1, pk)
        # (f*g)[pk] contribution from g[j] is f[pk-j] for valid j
        # B_row[i, j] = f[pk-j] if 0<=pk-j<n
        for j in range(js, je+1):
            B_rows[i, j] = f[pk - j]

    A_ub = np.zeros((k, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B_rows
    b_ub = -conv[peaks]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None

            # Quick scan
            if old_val > 1e-10:
                for scan in np.linspace(0, old_val * 3, 8):
                    candidates.append(scan)
            else:
                for scan in np.linspace(0, 1.0 / n * 3, 5):
                    candidates.append(scan)

            best_cand = old_val
            best_cand_obj = best_obj

            for v in candidates:
                if v < 0:
                    continue
                f[i] = v
                fs = f.sum()
                if fs < 1e-15:
                    continue
                f_norm = f / fs
                obj = objective_value(f_norm)
                if obj < best_cand_obj - 1e-15:
                    best_cand_obj = obj
                    best_cand = v

            if best_cand != old_val:
                f[i] = best_cand
                fs = f.sum()
                if fs > 1e-15:
                    f = f / fs
                    passed = True

            f[i] = old_val  # restore

        if best_cand_obj < best_obj - 1e-14 if 'best_cand_obj' in dir() else False:
            pass

        # Recompute best
        cur_obj = objective_value(f)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            total_passes += 1
            if verbose and total_passes % 5 == 0:
                print(f"    CoordDescent pass {total_passes}: obj={best_obj:.10f}")

    return best_f, best_obj


def coordinate_descent_fine(f0, time_limit, n_samples=20):
    """More refined coordinate descent using golden section search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = objective_value(f)
    best_f = f.copy()

    t0 = time.monotonic()

    # Golden ratio
    gr = (np.sqrt(5) - 1) / 2

    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[i]
            # Search range for f[i]
            lo = 0.0
            hi = max(old_val * 3.0, 1.0 / n * 4.0, 0.05)

            # First, do a coarse scan to find a good bracket
            grid = np.linspace(lo, hi, n_samples)
            best_grid_obj = float('inf')
            best_grid_val = old_val

            for v in grid:
                f[i] = v
                fs = f.sum()
                if fs < 1e-15:
                    continue
                fn = f / fs
                obj = objective_value(fn)
                if obj < best_grid_obj:
                    best_grid_obj = obj
                    best_grid_val = v

            f[i] = old_val

            if best_grid_obj < best_obj - 1e-14:
                # Refine with golden section around the best grid point
                idx = int(np.argmin([objective_value(np.concatenate([f[:i], [v], f[i+1:]])) if f.sum() > 1e-15 else inf for v in grid]))
                # Just use the grid result for simplicity
                f[i] = best_grid_val
                fs = f.sum()
                f = f / fs
                improved = True

                cur_obj = objective_value(f)
                if cur_obj < best_obj - 1e-14:
                    best_obj = cur_obj
                    best_f = f.copy()

        if not improved:
            break

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)

    # U-shape (theoretically motivated)
    x = np.arange(n, dtype=np.float64)
    center = 0.5 * (n - 1)
    u = 1.0 / (0.5 * np.abs(x - center) / n + 0.1)
    inits.append(u / u.sum())

    # Stronger U-shape
    u2 = 1.0 / (0.3 * np.abs(x - center) / n + 0.05)
    inits.append(u2 / u2.sum())

    # Gaussian
    for sigma_frac in [0.2, 0.35, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        inits.append(g / g.sum())

    # Cosine (periodic)
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 1.0))

    # Edge-heavy with random modulation
    edge = 1.0 / (0.5 * np.abs(x - center) / n + 0.05)
    mod = 1.0 + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 5) * x / n)
    em = edge * mod
    inits.append(em / em.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def fast_obj_eval(f):
    """Quick objective evaluation."""
    n = len(f)

    return conv.max()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining Nesterov gradient,
    LP Frank-Wolfe, and coordinate descent with multi-restart.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.sum() < 1e-15:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
    except Exception:
        pass

    # Phase 0: Initialization
    print("=== Phase 0: Initialization ===")
    t_start = time.monotonic()

    candidates = []
    if parent is not None:
        n_p = len(parent)
        candidates.append(("parent", parent))
    else:
        n_p = 384
        candidates.append(("uniform", np.ones(n_p) / n_p))

    # Generate diverse inits at parent length
    other_inits = generate_inits(n_p, parent, rng)
    for i, ini in enumerate(other_inits):
        candidates.append((f"init_{i}", ini))

    # Try a few other lengths
    for n_try in [320, 448, 512]:
        if n_try == n_p:
            continue
        ini = generate_inits(n_try, parent, rng)[1]  # U-shape
        candidates.append((f"len{n_try}_u", ini))

    for label, cand in candidates:
        if time.monotonic() > deadline - 300:
            break
        try_seq(cand, f"init_{label}")

    if best_seq is None:
        n_p = 384
        best_seq = np.ones(n_p) / n_p
        try_seq(best_seq, "fallback")

    print(f"  Phase 0 best: {best_score:.8f} (n={len(best_seq)})")

    n_best = len(best_seq)

    # Phase 1: Nesterov accelerated subgradient
    print("\n=== Phase 1: Nesterov Subgradient ===")
    remaining = deadline - time.monotonic() - 200
    t_p1 = min(remaining * 0.3, 400)

    if t_p1 > 30:
        # Run with different hyperparameters
        for cfg in [
            dict(lr_init=0.01, momentum=0.9, top_k=30, temp_init=0.001, temp_end=0.00002),
            dict(lr_init=0.005, momentum=0.95, top_k=50, temp_init=0.0005, temp_end=0.00001),
        ]:
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            t_each = min(rem * 0.6, t_p1)
            a_opt, _ = nesterov_subgrad_optimize(best_seq, time_limit=t_each, verbose=True, **cfg)
            try_seq(a_opt, f"nesterov_{cfg['lr_init']}")

    print(f"  Phase 1 best: {best_score:.8f} (n={len(best_seq)})")

    # Phase 2: LP Frank-Wolfe
    print("\n=== Phase 2: LP Frank-Wolfe ===")
    remaining = deadline - time.monotonic() - 150
    t_p2 = min(remaining * 0.35, 500)

    if t_p2 > 30:
        for rk in [20, 35]:
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            t_each = min(rem * 0.5, t_p2)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_each, reduced_k=rk, verbose=True)
            try_seq(a_opt, f"lpfw_k{rk}")

    print(f"  Phase 2 best: {best_score:.8f} (n={len(best_seq)})")

    # Phase 3: Coordinate descent
    print("\n=== Phase 3: Coordinate Descent ===")
    remaining = deadline - time.monotonic() - 100
    t_p3 = min(remaining * 0.3, 300)

    if t_p3 > 20:
        a_opt, _ = coordinate_descent_fine(best_seq, time_limit=min(t_p3 * 0.7, 200), n_samples=16)
        try_seq(a_opt, "coord_descent")

        # Second pass
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            a_opt, _ = coordinate_descent_fine(best_seq, time_limit=min(rem * 0.5, 100), n_samples=25)
            try_seq(a_opt, "coord_descent2")

    print(f"  Phase 3 best: {best_score:.8f} (n={len(best_seq)})")

    # Phase 4: Multi-restart
    print("\n=== Phase 4: Multi-Restart ===")
    remaining = deadline - time.monotonic() - 50

    restart_configs = [
        # (perturb_scale, phase1_lr, phase1_mom, label)
        (0.005, 0.008, 0.92, "restart_small"),
        (0.015, 0.012, 0.88, "restart_med"),
        (0.03, 0.015, 0.85, "restart_large"),
    ]

    for scale, lr, mom, label in restart_configs:
        rem = deadline - time.monotonic() - 30
        if rem < 30:
            break
        t_total = min(rem * 0.4, 200)

        # Perturb
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        try_seq(f_pert, f"{label}_pert")

        # Nesterov
        t_sg = t_total * 0.5
        if t_sg > 10:
            a_opt, _ = nesterov_subgrad_optimize(f_pert, time_limit=t_sg,
                                                  lr_init=lr, momentum=mom,
                                                  top_k=40, temp_init=0.0008, temp_end=0.00003)
            try_seq(a_opt, f"{label}_nesterov")

        # LP FW
        rem = deadline - time.monotonic() - 25
        if rem > 15:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem * 0.4, 60), reduced_k=20)
            try_seq(a_opt, f"{label}_lpfw")

        # Coord descent
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, _ = coordinate_descent_fine(best_seq, time_limit=min(rem * 0.5, 40), n_samples=20)
            try_seq(a_opt, f"{label}_coord")

    print(f"  Phase 4 best: {best_score:.8f} (n={len(best_seq)})")

    # Phase 5: Final polish - alternate between methods
    print("\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic() - 25

    for round_num in range(5):
        rem = deadline - time.monotonic() - 20
        if rem < 10:
            break

        # Quick Nesterov
        t_each = min(rem * 0.25, 40)
        if t_each > 5:
            a_opt, _ = nesterov_subgrad_optimize(best_seq, time_limit=t_each,
                                                  lr_init=0.002, momentum=0.97,
                                                  top_k=n_best // 3, temp_init=0.0002, temp_end=0.00002)
            try_seq(a_opt, f"polish_nest_{round_num}")

        rem = deadline - time.monotonic() - 15
        if rem < 10:
            break

        # Quick LP
        t_each = min(rem * 0.3, 30)
        if t_each > 5:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_each, reduced_k=15)
            try_seq(a_opt, f"polish_lp_{round_num}")

        rem = deadline - time.monotonic() - 10
        if rem < 5:
            break

        # Quick coord
        t_each = min(rem * 0.4, 30)
        if t_each > 5:
            a_opt, _ = coordinate_descent_fine(best_seq, time_limit=t_each, n_samples=15)
            try_seq(a_opt, f"polish_coord_{round_num}")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
