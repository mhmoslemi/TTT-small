def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(conv, a, T):
    """Compute gradient of (1/T)*log(sum_k exp(T*conv[k]))."""
    n = len(a)
    shifted = T * conv - T * conv.max()
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws

    # This is a cross-correlation of w and a
    grad = np.zeros(n)

        if k_end > k_start:

        k_max = np.argmax(conv)
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        grad = np.zeros(n)
        grad[js:je + 1] = 2.0 * f[k_max - je:k_max - js + 1][::-1]
        
        # We want to move towards a vertex that decreases the objective
        # Try: move mass from high-grad coordinate to low-grad coordinate
        # Or: just pick the vertex with minimum gradient
        v_idx = np.argmin(grad)
        g = np.zeros(n)
        g[v_idx] = 1.0

        # Also try: move mass away from the coordinate that contributes most to the peak
        away_idx = js + np.argmax(f[js:je + 1])
        
        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-15:
                f = f / fs
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 80:
            break

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit):
    """Full coordinate descent with line search for fine polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng_local = np.random.default_rng(9876)

    max_rounds = 200
    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit - 2:
            break
        improved = False
        order = rng_local.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit - 2:
                break
            # Find best partner to exchange mass with
            # Try a few candidates: neighbors, and random
            candidates = [(j + 1) % n, (j - 1) % n, (j + n // 2) % n,
                          rng_local.integers(n)]
            best_new_f = None
            best_new_val = best_obj
            
            for source in candidates:
                source = int(source) % n
                if source == j:
                    continue
                # Try different fractions
                for frac in [0.03, 0.08, 0.15, 0.3, 0.5]:
                    delta = f[source] * frac
                    if delta < 1e-15:
                        continue
                    f_new = f.copy()
                    f_new[j] += delta
                    f_new[source] -= delta
                    if f_new[source] < 0:
                        f_new[j] += f_new[source]
                        f_new[source] = 0.0
                    f_new = np.maximum(f_new, 0.0)
                    sn = f_new.sum()
                    if sn < 1e-15:
                        continue
                    f_new = f_new / sn
                    val = fast_convolve(f_new).max()
                    if val < best_new_val - 1e-15:
                        best_new_val = val
                        best_new_f = f_new.copy()
            
            if best_new_f is not None:
                f = best_new_f
                best_obj = best_new_val
                best_f = f.copy()
                conv = fast_convolve(f)
                improved = True

        if not improved:
            break

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.003, mom=0.93, top_k=30, temp=0.00005):
    """Subgradient descent with multi-peak softmax weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    rng_local = np.random.default_rng(7777)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1:
            break
        
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.003

        if no_improve > 300:
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0003)
            no_improve = 0

    return best_f, best_obj


def try_sequence(a, label, state):
    """Evaluate a sequence and update state if it's better."""
    a = np.maximum(a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return score


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.4))

    # Edge concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    inits.append(edge_w / edge_w.sum())

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Sine
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    rc = np.maximum(rc, 0.01)
    inits.append(rc / rc.sum())

    # Two-peak
    peak = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)) +
            np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)))
    inits.append(peak / peak.sum())

    # "Wavy" - oscillatory envelope
    wave = (1.0 + 0.3 * np.cos(4 * np.pi * x / n)) * np.exp(-0.5 * (x / (0.3 * n)) ** 2)
    wave = np.maximum(wave, 0.01)
    inits.append(wave / wave.sum())

    # Power law
    pl = (x + 1.0) ** (-0.3)
    inits.append(pl / pl.sum())

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: Multi-length exploration with continuation + FW
    Phase 2: Deep optimization with LP-based FW
    Phase 3: Polishing with CD and SG
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    state = {'best_seq': None, 'best_score': float('inf')}

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}")
            s = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(s):
                print(f"  Parent score: {s:.6f}")
                state['best_seq'] = parent.copy()
                state['best_score'] = s
    except Exception:
        pass

    total_time = float(budget_s)

    # === Phase 1: Multi-length exploration with continuation ===
    print("=== Phase 1: Continuation + Multi-length Exploration ===")
    phase1_end = time.monotonic() + (total_time * 0.28)

    # Quick continuation on parent
    if parent is not None:
        n_p = len(parent)
        t1 = min(60, max(20, phase1_end - time.monotonic() - 40))
        if t1 > 10:
            print(f"  Continuation on parent (n={n_p}), t={t1:.0f}s")
            a_opt, obj = continuation_optimize(parent, time_limit=t1, T_max=500, n_T_steps=8)
            try_sequence(a_opt, "parent_cont", state)

    # Multi-length exploration
    lengths = [256, 320, 384, 512]
    if parent is not None:
        if len(parent) not in lengths:
            lengths.insert(0, len(parent))
        else:
            lengths.insert(0, len(parent))
        lengths.sort(key=lambda x: abs(x - len(parent)))

    # Remove duplicates, keep order
    seen = set()
    unique_lengths = []
    for L in lengths:
        if L not in seen:
            seen.add(L)
            unique_lengths.append(L)
    lengths = unique_lengths

    for n in lengths:
        if time.monotonic() > phase1_end - 40:
            break

        inits = generate_inits(n, parent, rng)
        n_try = min(3, len(inits))
        t_per = max(8, (phase1_end - time.monotonic() - 30) / (n_try * max(1, len(lengths) - 1)))
        t_per = min(t_per, 40)

        for i in range(n_try):
            if time.monotonic() > phase1_end - 20:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 15)
            if t_alloc < 6:
                break
            f_init = inits[i]
            
            # Use continuation for smooth phase, then quick FW
            a_cont, _ = continuation_optimize(f_init, time_limit=t_alloc * 0.5, T_max=200, n_T_steps=6)
            t_rem = t_alloc * 0.4
            if t_rem > 3:
                a_opt, obj = fw_lp_optimize(a_cont, time_limit=t_rem)
            else:
                a_opt = a_cont
            try_sequence(a_opt, f"cont_n{n}_i{i}", state)

    print(f"  Phase 1 best: {state['best_score']:.6f}")

    # === Phase 2: Deep optimization ===
    print(f"\n=== Phase 2: Deep FW-LP Optimization ===")
    phase2_end = time.monotonic() + (total_time * 0.42)

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        print(f"  Optimizing n={n_best}, budget={(phase2_end - time.monotonic()):.0f}s")

        # Main deep run
        t_main = min(200, (phase2_end - time.monotonic()) * 0.4)
        if t_main > 20:
            a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=t_main, verbose=True)
            try_sequence(a_opt, "deep_main", state)

        # Vertex FW (faster, can do more iterations)
        rem = phase2_end - time.monotonic() - 20
        if rem > 15:
            a_opt, obj = vertex_fw_optimize(state['best_seq'], time_limit=min(rem, 100))
            try_sequence(a_opt, "vertex_fw", state)

        # Perturbation runs
        perturb_scales = [0.003, 0.008, 0.02, 0.05]
        for ps in perturb_scales:
            rem = phase2_end - time.monotonic() - 20
            if rem < 15:
                break
            t_alloc = min(rem / (len(perturb_scales) - perturb_scales.index(ps) + 1), 80)
            if t_alloc < 10:
                break
            noise = rng.standard_normal(n_best) * ps
            f_init = project_simplex(state['best_seq'] + noise)
            a_opt, obj = fw_lp_optimize(f_init, time_limit=t_alloc)
            try_sequence(a_opt, f"pert_{ps}", state)

        # Mirror blend
        rem = phase2_end - time.monotonic() - 15
        if rem > 15:
            f_mirror = project_simplex(0.7 * state['best_seq'] + 0.3 * state['best_seq'][::-1])
            a_opt, obj = fw_lp_optimize(f_mirror, time_limit=min(rem, 60))
            try_sequence(a_opt, "mirror", state)

        # Try a different length near the best
        rem = phase2_end - time.monotonic() - 10
        if rem > 20 and state['best_seq'] is not None:
            n_alt = n_best + 32 if n_best < 500 else n_best - 32
            if n_alt >= 128:
                n_alt = max(128, n_alt)
                a_resized = np.interp(np.linspace(0, 1, n_alt),
                                      np.linspace(0, 1, n_best), state['best_seq'])
                a_resized = np.maximum(a_resized, 0)
                a_resized = a_resized / a_resized.sum()
                a_opt, obj = fw_lp_optimize(a_resized, time_limit=min(rem, 50))
                try_sequence(a_opt, f"alt_n{n_alt}", state)

    # === Phase 3: Polishing ===
    print(f"\n=== Phase 3: Polishing ===")
    phase3_end = deadline - 10

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        print(f"  Current best: {state['best_score']:.6f}")

        # SG polish - long
        rem = phase3_end - time.monotonic() - 30
        if rem > 15:
            a_opt, obj = sg_polish(state['best_seq'], time_limit=min(rem, 120),
                                   lr=0.0015, mom=0.94,
                                   top_k=max(10, n_best // 4), temp=0.00002)
            try_sequence(a_opt, "sg_long", state)

        # CD polish
        rem = phase3_end - time.monotonic() - 15
        if rem > 10:
            a_opt, obj = coordinate_descent_polish(state['best_seq'], time_limit=min(rem, 60))
            try_sequence(a_opt, "cd", state)

        # Fine SG
        rem = phase3_end - time.monotonic() - 10
        if rem > 10:
            a_opt, obj = sg_polish(state['best_seq'], time_limit=min(rem, 50),
                                   lr=0.0005, mom=0.96,
                                   top_k=max(5, n_best // 10), temp=0.00001)
            try_sequence(a_opt, "sg_fine", state)

        # Final FW squeeze
        rem = phase3_end - time.monotonic()
        if rem > 10:
            a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=min(rem, 40))
            try_sequence(a_opt, "final_fw", state)

        # Another CD pass
        rem = phase3_end - time.monotonic()
        if rem > 8:
            a_opt, obj = coordinate_descent_polish(state['best_seq'], time_limit=min(rem, 25))
            try_sequence(a_opt, "cd_final", state)

    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(256) / 256
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))
        print(f"  Fallback uniform: {state['best_score']:.6f}")

    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs

    print(f"\nFINAL SCORE: {state['best_score']:.6f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
