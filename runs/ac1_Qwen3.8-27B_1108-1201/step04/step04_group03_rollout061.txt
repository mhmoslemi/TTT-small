import time

    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(f, M):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using accelerated coordinate descent
with efficient peak-level convolution updates, diverse structural initializations,
FW-LP warm starts, multiplicative weight optimization, and adaptive method switching.
Key innovation: exact 1D subproblem solutions within coordinate descent using
the structure that each peak is either constant, linear, or quadratic in the
free coordinate.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    For each coordinate j, solve the exact 1D subproblem using the structure:
    each peak is constant, linear (positive slope), or quadratic in a[j].
    """
    f = np.maximum(f0.astype(np.float64), 0.0)

    else:

        pass_count += 1
        order = np.arange(n)
        if pass_count % 2 == 0:
            rng = np.random.default_rng(pass_count * 104729 + 7)
            rng.shuffle(order)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            S0 = 1.0 - old_val

            # Compute background convolution (remove j's contribution)
            conv = fast_convolve(f)
            k_min = j
            k_max = min(j + n - 1, 2 * n - 2)

            # Subtract j's contribution from conv
            if k_max >= k_min:
                ks = np.arange(k_min, k_max + 1)
                others = ks - j
                valid = (others >= 0) & (others < n)
                if np.any(valid):
                    ks_v = ks[valid]
                    o_v = others[valid]
                    if 2 * j in ks_v:
                        # Handle quadratic term separately

                        lin_mask = ~quad_mask
                        if np.any(lin_mask):

                        if np.any(quad_mask):
                            conv[2 * j] -= old_val * old_val
                    else:
                        conv[ks_v] -= 2.0 * old_val * f[o_v]

            # C_out: max of bg outside [k_min, k_max]
            C_out = 0.0
            if k_min > 0:

def mult_weight_optimize(f0, time_limit, lr_init=0.04, top_k=15, temp_init=0.0004):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        k_eff = min(top_k, 2 * n - 1)

        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        log_update = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_contrib = 2.0 * w[i] * f[k - idx]
            log_update[idx] -= cur_lr * grad_contrib

        f = f * np.exp(log_update)
        f = np.maximum(f, 1e-25)
        f = f / f.sum()

        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.002

        if no_improve > 120:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 999983)
            f = best_f * np.exp(rng.standard_normal(n) * 0.005)
            f = f / f.sum()
            cur_lr = max(lr_init * 0.6, 0.008)
            cur_temp = temp_init * 2.0
            no_improve = 0

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.002, mom=0.93, top_k=20):
    """Subgradient polish with momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    v = np.zeros(n)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        k_eff = min(top_k, 2 * n - 1)
        if k_eff >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -k_eff)[-k_eff:]

        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        tw = weights.sum()
        if tw > 0:
            weights /= tw

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += weights[i] * 2.0 * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)

        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        if no_improve > 150:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0005)
            no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    x = np.arange(n, dtype=np.float64)
    # Hann window
    h = 0.5 * (1 + np.cos(2 * np.pi * x / max(n - 1, 1)))
    h = np.maximum(h, 1e-6)
    inits.append(h / h.sum())

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine (single period)
    c = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c / c.sum())

    # Two-hump
    h2 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits.append(h2 / h2.sum())

    # Edge-weighted
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search with coordinate descent as primary
    optimizer, FW-LP for warm starts, and MW/SG for escape from local minima.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})", flush=True)
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}", flush=True)
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        n_p = 384
    else:
        n_p = len(parent)

    # === Phase 1: Initialization exploration (~150s) ===
    print("\n=== Phase 1: Initialization ===", flush=True)
    inits = generate_inits(n_p, parent, rng)

    # Quick FW-LP on top candidates
    t1_total = min(120, max(30, remaining_budget(deadline) - 1500))
    t_per_init = t1_total / min(3, len(inits))

    for ii in range(min(3, len(inits))):
        rem = remaining_budget(deadline)
        if rem < 150:
            break
        t_alloc = min(t_per_init, rem - 1400)
        if t_alloc < 10:
            break
        print(f"  FW-LP init[{ii}] (n={n_p}), t={t_alloc:.0f}s", flush=True)
        a_opt, _ = fw_lp_optimize(inits[ii], time_limit=t_alloc)
        try_seq(a_opt, f"fw_init_{ii}")

    print(f"  Phase 1 best: {best_score:.8f}", flush=True)

    # === Phase 2: Coordinate Descent (main workhorse, ~600s) ===
    print("\n=== Phase 2: Coordinate Descent ===", flush=True)
    if best_seq is not None:
        rem = remaining_budget(deadline)
        t2 = min(550, max(100, rem - 900))
        print(f"  CD pass, t={t2:.0f}s", flush=True)
        a_opt, _ = coord_descent(best_seq, time_limit=t2, n_cand=20, restart_patience=50)
        try_seq(a_opt, "cd_main")

        # Second CD pass with different params
        rem = remaining_budget(deadline)
        if rem > 300:
            t2b = min(200, rem - 700)
            if t2b > 20:
                print(f"  CD pass 2, t={t2b:.0f}s", flush=True)
                a_opt2, _ = coord_descent(best_seq, time_limit=t2b, n_cand=15, restart_patience=30)
                try_seq(a_opt2, "cd_2")

    print(f"  Phase 2 best: {best_score:.8f}", flush=True)

    # === Phase 3: MW Optimization (~250s) ===
    print("\n=== Phase 3: Multiplicative Weight ===", flush=True)
    if best_seq is not None:
        mw_configs = [
            {'lr_init': 0.03, 'top_k': 10, 'temp_init': 0.0003},
            {'lr_init': 0.05, 'top_k': 20, 'temp_init': 0.0006},
            {'lr_init': 0.02, 'top_k': 25, 'temp_init': 0.0002},
        ]
        rem = remaining_budget(deadline)
        t_per_mw = max(30, min(100, (rem - 400) / len(mw_configs)))

        for ci, cfg in enumerate(mw_configs):
            rem = remaining_budget(deadline)
            if rem < 200:
                break
            t_alloc = min(t_per_mw, rem - 300)
            if t_alloc < 15:
                break
            print(f"  MW cfg{ci} (lr={cfg['lr_init']}, k={cfg['top_k']}), t={t_alloc:.0f}s", flush=True)
            a_opt, _ = mult_weight_optimize(
                best_seq, time_limit=t_alloc,
                lr_init=cfg['lr_init'], top_k=cfg['top_k'],
                temp_init=cfg['temp_init']
            )
            try_seq(a_opt, f"mw_{ci}")

    print(f"  Phase 3 best: {best_score:.8f}", flush=True)

    # === Phase 4: Subgradient + CD alternating (~300s) ===
    print("\n=== Phase 4: SG + CD Alternating ===", flush=True)
    if best_seq is not None:
        for round_idx in range(4):
            rem = remaining_budget(deadline)
            if rem < 150:
                break
            # SG step
            t_sg = min(80, rem - 120)
            if t_sg > 10:
                print(f"  SG round {round_idx}, t={t_sg:.0f}s", flush=True)
                a_opt, _ = sg_polish(best_seq, time_limit=t_sg, lr=0.0015, mom=0.93, top_k=25)
                try_seq(a_opt, f"sg_r{round_idx}")

            # CD step
            rem = remaining_budget(deadline)
            if rem < 100:
                break
            t_cd = min(80, rem - 80)
            if t_cd > 10:
                print(f"  CD round {round_idx}, t={t_cd:.0f}s", flush=True)
                a_opt, _ = coord_descent(best_seq, time_limit=t_cd, n_cand=15, restart_patience=30)
                try_seq(a_opt, f"cd_r{round_idx}")

    print(f"  Phase 4 best: {best_score:.8f}", flush=True)

    # === Phase 5: Try different length (~200s) ===
    print("\n=== Phase 5: Different Lengths ===", flush=True)
    for n_try in [512, 448]:
        rem = remaining_budget(deadline)
        if rem < 100:
            break
        print(f"  Trying n={n_try}", flush=True)
        inits_n = generate_inits(n_try, best_seq if best_seq is not None else parent, rng)
        # Quick FW-LP
        t_fw = min(50, rem - 60)
        if t_fw > 10:
            a_opt, _ = fw_lp_optimize(inits_n[0], time_limit=t_fw)
            try_seq(a_opt, f"fw_n{n_try}")
        # Brief CD
        rem = remaining_budget(deadline)
        if rem > 80:
            t_cd = min(60, rem - 40)
            if t_cd > 10:
                a_opt, _ = coord_descent(a_opt if 'a_opt' in dir() else inits_n[0],
                                         time_limit=t_cd, n_cand=15, restart_patience=25)
                try_seq(a_opt, f"cd_n{n_try}")

    print(f"  Phase 5 best: {best_score:.8f}", flush=True)

    # === Phase 6: Final polish (~remaining time) ===
    print("\n=== Phase 6: Final Polish ===", flush=True)
    if best_seq is not None:
        rem = remaining_budget(deadline)
        if rem > 30:
            # Final CD with very fine candidates
            t_final = min(rem - 20, 100)
            if t_final > 10:
                print(f"  Final CD, t={t_final:.0f}s", flush=True)
                a_opt, _ = coord_descent(best_seq, time_limit=t_final, n_cand=25, restart_patience=20)
                try_seq(a_opt, "final_cd")

        # Tiny perturbations
        rem = remaining_budget(deadline)
        if rem > 40:
            for _ in range(3):
                rem = remaining_budget(deadline)
                if rem < 15:
                    break
                noise = rng.standard_normal(len(best_seq)) * 0.0005
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = coord_descent(f_pert, time_limit=min(rem - 10, 20), n_cand=10, restart_patience=10)
                try_seq(a_opt, "perturb")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]


def remaining_budget(deadline):
    """Return seconds remaining until deadline."""
    return max(0.0, deadline - time.monotonic())
