Combined with active-set QP solver and subgradient polishing.
"""

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_qp_active_set(q, n, max_iter=500):
    """Solve min p^T Q p s.t. sum(p)=1, p>=0, Q[j,l]=q[j+l] using active set."""
    Q = np.array([q[j:j+n] for j in range(n)])
    Q = 0.5 * (Q + Q.T)  # Ensure symmetry (should be symmetric already)
    
    S = list(range(n))
    
    for it in range(max_iter):
        k = len(S)
        if k == 0:
            return np.ones(n) / n
        
        Q_SS = Q[np.ix_(S, S)]
        ones = np.ones(k)
        
        try:
            # Add small regularization for numerical stability
            reg = 1e-12 * np.trace(Q_SS) / k
            v = np.linalg.solve(Q_SS + reg * np.eye(k), ones)
        except np.linalg.LinAlgError:
            v = np.linalg.lstsq(Q_SS, ones, rcond=None)[0]
        
        mu = 2.0 / v.sum()
        if mu <= 0 or not np.isfinite(mu):
            # Fallback
            return np.ones(n) / n
        p_S = 0.5 * mu * v
        
        if np.all(p_S >= -1e-10):
            p_S = np.maximum(p_S, 0.0)
            ps = p_S.sum()
            if ps < 1e-15:
                return np.ones(n) / n
            p_S /= ps
            p = np.zeros(n)
            p[S] = p_S
            return p
        
        # Remove the most negative index
        worst = np.argmin(p_S)
        S.pop(worst)
    
    # Fallback
    return np.ones(n) / n

def primal_dual_optimize(n, alpha_schedule, time_limit, verbose=False):
    """
    Primal-dual multiplicative weights algorithm.
    alpha_schedule: function of iteration -> alpha for q update
    """
    p = np.ones(n) / n
    c = fast_convolve(p)
    q = c / c.sum()
    
    best_obj = c.max()
    best_p = p.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Inner: solve QP
        p_new = solve_qp_active_set(q, n)
        if p_new is None:
            p_new = p
        
        # Evaluate
        c_new = fast_convolve(p_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_p = p_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        p = p_new
        c = c_new
        
        # Outer: update q
        alpha = alpha_schedule(it)
        # q[k] proportional to c[k]^alpha
        c_pos = np.maximum(c, 1e-30)
        log_q = alpha * np.log(c_pos)
        log_q -= log_q.max()
        q_new = np.exp(log_q)
        q_s = q_new.sum()
        if q_s < 1e-30:
            q_new = np.ones(2*n-1) / (2*n-1)
        else:
            q_new /= q_s
        
        # Blend with uniform for stability
        eps = 0.05
        q = (1-eps) * q_new + eps * np.ones(2*n-1) / (2*n-1)
        q /= q.sum()
        
        if verbose and it % 100 == 0:
            print(f"  PD iter {it}: obj={best_obj:.10f}, spread={c.max()-c.min():.6f}, it_time={time.monotonic()-t0:.1f}s")
        
        if no_improve > 500:
            break
    
    return best_p, best_obj

def sg_polish(p0, time_limit, lr=0.002, top_k=20):
    """Subgradient polishing."""
    p = p0.astype(np.float64).copy()
    s = p.sum()
    p /= s
    n = len(p)
    
    c = fast_convolve(p)
    best_obj = c.max()
    best_p = p.copy()
    v = np.zeros(n)
    mom = 0.9
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(p)
        peaks = np.argpartition(c, -top_k)[-top_k:]
        # Weighted subgradient
        pv = c[peaks]
        w = pv - pv.min() + 1e-10
        w /= w.sum()
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += w[i] * 2.0 * p[k - idx]
        
        v = mom * v - lr * sg
        p = project_simplex(p + v)
        c = fast_convolve(p)
        obj = c.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_p = p.copy()
        
        lr *= 0.9995
    
    return best_p, best_obj

```python
def hankel_matvec(q, v, n):
    """Compute (H*v)[j] where H[j,l] = q[j+l] for j,l in 0..n-1."""
    # H*v is a cross-correlation
    # (H*v)[j] = sum_{l=0}^{n-1} q[j+l] * v[l]
    # This equals the cross-correlation of q and v at specific lags
    
    # Using FFT: 
    # Let Q = FFT(q, 2n), V = FFT(v, 2n)
    # The circular cross-correlation is IFFT(conj(Q) * V)
    # But we want linear cross-correlation
    
    # Alternative: pad and use convolution
    # (H*v)[j] = sum_l q[j+l]*v[l] = sum_l q[m]*v[m-j] where m=j+l
    # = (q * flip(v))[j + n - 1] ... let me think differently
    
    # Actually: (H*v)[j] = sum_{l=0}^{n-1} q[j+l] * v[l]
    # Let u = v (length n), and we want the "reversed cross-correlation"
    # 
    # Define w[l] = v[n-1-l] (reversed v)
    # Then (H*v)[j] = sum_{l=0}^{n-1} q[j+l] * v[l] = sum_{l=0}^{n-1} q[j+l] * w[n-1-l]
    # Let m = n-1-l: = sum_{m=0}^{n-1} q[j+n-1-m] * w[m]
    # = (q * w)[j+n-1] (convolution at position j+n-1)
    
    fft_size = 1 << (2*n - 1).bit_length()
    Q_hat = np.fft.rfft(q, fft_size)
    w = v[::-1]
    W_hat = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(Q_hat * W_hat, fft_size)
    return conv[n-1 : n-1+n]
```

"""
Primal-dual multiplicative weights algorithm for minimizing max auto-convolution,
combined with FFT-accelerated projected gradient QP solving and subgradient polishing.
Exploits the Hankel structure of the QP matrix for O(n log n) mat-vec products.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_matvec(q, v, n):
    """Compute (H*v)[j] where H[j,l] = q[j+l] using FFT-based cross-correlation."""
    fft_size = 1 << (2 * n - 1).bit_length()
    Q_hat = np.fft.rfft(q, fft_size)
    w = v[::-1]
    W_hat = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(Q_hat * W_hat, fft_size)
    return conv[n - 1:2 * n - 1]


def solve_qp_projected(q, n, n_inner=150, warm_start=None):
    """Solve min p^T H(q) p s.t. p in simplex using projected gradient with FFT matvec."""
    if warm_start is not None:
        p = warm_start.astype(np.float64).copy()
        ps = p.sum()
        if ps < 1e-15:
            p = np.ones(n) / n
        else:
            p = p / ps
    else:
        p = np.ones(n) / n

    step = 0.4

    for _ in range(n_inner):
        grad = 2.0 * hankel_matvec(q, p, n)
        p_new = project_simplex(p - step * grad)
        if np.max(np.abs(p_new - p)) < 1e-14:
            break
        p = p_new

    return p


def solve_qp_active_set(q, n, max_iter=500):
    """Solve min p^T H p s.t. sum(p)=1, p>=0 using active set method."""
    Q = np.array([q[j:j + n] for j in range(n)])
    Q = 0.5 * (Q + Q.T)

    S = list(range(n))

    for _ in range(max_iter):
        k = len(S)
        if k == 0:
            return np.ones(n) / n

        Q_SS = Q[np.ix_(S, S)]
        ones = np.ones(k)

        try:
            reg = 1e-10 * max(1.0, np.trace(Q_SS) / k)
            v = np.linalg.solve(Q_SS + reg * np.eye(k), ones)
        except np.linalg.LinAlgError:
            v = np.linalg.lstsq(Q_SS, ones, rcond=None)[0]

        v_sum = v.sum()
        if v_sum <= 1e-15 or not np.isfinite(v_sum):
            return np.ones(n) / n

        p_S = 1.0 / v_sum * v

        if np.all(p_S >= -1e-12):
            p_S = np.maximum(p_S, 0.0)
            ps = p_S.sum()
            if ps < 1e-15:
                return np.ones(n) / n
            p_S /= ps
            p = np.zeros(n)
            p[S] = p_S
            return p

        worst = np.argmin(p_S)
        S.pop(worst)

    return np.ones(n) / n


def primal_dual_optimize(n, time_limit, alpha_start=1.0, alpha_rate=0.003,
                         alpha_max=8.0, n_inner_pg=120, eps_blend=0.03,
                         verbose=False, warm_p=None):
    """
    Primal-dual multiplicative weights for min_p max_k (p*p)[k].
    """
    if warm_p is not None:
        p = warm_p.astype(np.float64).copy()
        p = np.maximum(p, 0)
        ps = p.sum()
        if ps < 1e-15:
            p = np.ones(n) / n
        else:
            p /= ps
    else:
        p = np.ones(n) / n

    c = fast_convolve(p)
    q = np.maximum(c, 1e-30)
    q /= q.sum()

    best_obj = float(c.max())
    best_p = p.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Inner: solve QP
        p_new = solve_qp_projected(q, n, n_inner=n_inner_pg, warm_start=p)

        # Evaluate
        c_new = fast_convolve(p_new)
        obj_new = float(c_new.max())

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_p = p_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        p = p_new
        c = c_new

        # Outer: update q with multiplicative weights
        alpha = min(alpha_start + alpha_rate * it, alpha_max)
        c_pos = np.maximum(c, 1e-30)
        log_q = alpha * np.log(c_pos)
        log_q -= log_q.max()
        q_new = np.exp(log_q)
        q_s = q_new.sum()
        if q_s < 1e-30:
            q_new = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            q_new /= q_s

        # Blend with uniform for numerical stability
        q = (1.0 - eps_blend) * q_new + eps_blend * (1.0 / (2 * n - 1))
        q /= q.sum()

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PD iter {it}: best={best_obj:.10f}, spread={c.max()-c.min():.8f}, "
                  f"alpha={alpha:.2f}, t={elapsed:.1f}s")

        if no_improve > 400:
            # Try a larger perturbation
            rng = np.random.default_rng(it)
            p = project_simplex(best_p + rng.standard_normal(n) * 0.005)
            no_improve = 0

    return best_p, best_obj


def primal_dual_with_qswitch(n, time_limit, verbose=False):
    """
    Enhanced primal-dual that periodically switches between PG and active set,
    and uses adaptive alpha with cycling.
    """
    p = np.ones(n) / n
    c = fast_convolve(p)
    q = np.maximum(c, 1e-30)
    q /= q.sum()

    best_obj = float(c.max())
    best_p = p.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    alpha = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        phase = (it // 300) % 3

        if phase == 0:
            # PG mode (fast)
            p_new = solve_qp_projected(q, n, n_inner=100, warm_start=p)
        elif phase == 1:
            # Active set mode (accurate) - only if n is small enough
            if n <= 400:
                p_new = solve_qp_active_set(q, n)
            else:
                p_new = solve_qp_projected(q, n, n_inner=200, warm_start=p)
        else:
            # PG with higher accuracy
            p_new = solve_qp_projected(q, n, n_inner=250, warm_start=p)

        c_new = fast_convolve(p_new)
        obj_new = float(c_new.max())

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_p = p_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        p = p_new
        c = c_new

        # Update q with cycling alpha
        alpha = min(1.0 + 0.004 * it, 10.0)
        c_pos = np.maximum(c, 1e-30)
        log_q = alpha * np.log(c_pos)
        log_q -= log_q.max()
        q_new = np.exp(log_q)
        q_s = q_new.sum()
        if q_s < 1e-30:
            q_new = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            q_new /= q_s
        eps = 0.02
        q = (1.0 - eps) * q_new + eps / (2 * n - 1)
        q /= q.sum()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PD2 iter {it}: best={best_obj:.10f}, alpha={alpha:.2f}, t={elapsed:.1f}s")

        if no_improve > 600:
            rng = np.random.default_rng(42 + it)
            p = project_simplex(best_p + rng.standard_normal(n) * 0.003)
            c = fast_convolve(p)
            no_improve = 0

    return best_p, best_obj


def subgradient_polish(p0, time_limit, lr=0.0015, top_k=30, mom=0.92, temp=1e-5):
    """Smoothed subgradient descent for final polishing."""
    p = p0.astype(np.float64).copy()
    p = np.maximum(p, 0)
    ps = p.sum()
    if ps < 1e-15:
        p = np.ones(len(p)) / len(p)
    else:
        p /= ps
    n = len(p)

    c = fast_convolve(p)
    best_obj = float(c.max())
    best_p = p.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    it = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(p)

        if top_k >= len(c):
            peaks = np.arange(len(c))
        else:
            peaks = np.argpartition(c, -top_k)[-top_k:]

        pv = c[peaks]
        scaled = (pv - pv.min()) / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * p[k - idx]

        v = mom * v - cur_lr * sg
        p = project_simplex(p + v)

        c = fast_convolve(p)
        obj = float(c.max())
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_p = p.copy()

        cur_lr *= 0.9996
        cur_temp *= 1.003

    return best_p, best_obj


def local_refine(p0, time_limit, seed=0):
    """Local refinement with random perturbations + subgradient steps."""
    rng = np.random.default_rng(seed)
    p = p0.astype(np.float64).copy()
    p = np.maximum(p, 0)
    ps = p.sum()
    p /= ps
    n = len(p)

    c = fast_convolve(p)
    best_obj = float(c.max())
    best_p = p.copy()

    t0 = time.monotonic()
    scale = 0.003
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Random perturbation
        noise = rng.standard_normal(n) * scale
        p_try = project_simplex(best_p + noise)
        c_try = fast_convolve(p_try)
        if c_try.max() < best_obj - 1e-14:
            best_obj = float(c_try.max())
            best_p = p_try.copy()

        # Quick SG step from current best
        if it % 10 == 0:
            p_sg = subgradient_polish(best_p, time_limit=0.5, lr=0.001, top_k=20)
            if p_sg[1] < best_obj - 1e-14:
                best_obj = p_sg[1]
                best_p = p_sg[0].copy()

        scale = max(scale * 0.998, 0.0005)

    return best_p, best_obj
