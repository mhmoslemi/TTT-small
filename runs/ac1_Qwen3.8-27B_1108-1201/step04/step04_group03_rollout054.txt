    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]

    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

    c = fast_convolve(a)

def lse_grad_a(a, T):

        mag = np.abs(A)
        phase = np.angle(A)

            break
        a_new = a_new / s
        if np.max(np.abs(a_new - a)) < 1e-12:
            break
        a = a_new

def project_simplex(v):

    rho = np.nonzero(cond)[0][-1]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-sum-exp smooth optimization
with L-BFGS, geometric temperature annealing, spectral flattening for diversity,
and fine-grained final polish. Key innovation: uses a smooth surrogate (LSE) of
the non-smooth max function with exact gradients computed via FFT-based
cross-correlation, enabling efficient second-order optimization.
"""

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def softmax(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def lse_value_and_p(a, T):
    """Compute LSE_T(conv(a,a)) and the softmax weights p."""
    c = fast_convolve(a)
    c_max = c.max()
    shifted = (c - c_max) / T
    p = np.exp(shifted)
    p_sum = p.sum()
    p /= p_sum
    lse_val = c_max + T * np.log(p_sum)
    return lse_val, p, c


def lse_grad_a(a, T):
    """
    Gradient of LSE_T(conv(a,a)) w.r.t. a.
    Uses the identity: dLSE/da[j] = 2T * sum_m a[m]*p[m+j]
    computed via FFT cross-correlation.
    """
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    p = np.exp((c - c_max) / T)
    p /= p.sum()

    size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    P = np.fft.rfft(p, size)
    corr = np.fft.irfft(P * np.conj(A), size)[:n]
    return 2.0 * T * corr


def lse_obj_grad_x(x, n, T):
    """
    Objective and gradient for L-BFGS using softmax parameterization.
    Objective: 2n * LSE_T(conv(softmax(x), softmax(x)))

    a = softmax(x)
    lse_val, p, c = lse_value_and_p(a, T)
    val = 2.0 * n * lse_val

    size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    P = np.fft.rfft(p, size)
    corr = np.fft.irfft(P * np.conj(A), size)[:n]
    grad_a = 2.0 * T * corr

    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)
    return val, grad_x


def lbfgs_lse(a_init, T, maxiter=1000, ftol=1e-14, gtol=1e-10):
    """Run L-BFGS to minimize LSE_T objective using softmax parameterization."""
    n = len(a_init)
    a_safe = np.maximum(a_init, 1e-30)
    x0 = np.log(a_safe)

    n = len(a_init)

    def fun(x):
        a = softmax(x)
        lse_val, p, c = lse_value_and_p(a, T)
        return 2.0 * n * lse_val

    def fun_and_grad(x):
        a = softmax(x)
        lse_val, p, c = lse_value_and_p(a, T)
        val = 2.0 * n * lse_val

        size = 1 << (3 * n - 1).bit_length()
        A = np.fft.rfft(a, size)
        P = np.fft.rfft(p, size)
        corr = np.fft.irfft(P * np.conj(A), size)[:n]
        grad_a = 2.0 * T * corr

        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)
        return val, grad_x

    res = minimize(
        fun_and_grad, x0,
        method='L-BFGS-B',

def spectral_flatten(a, beta=0.6, iterations=100):
    """
    Iterative spectral flattening: compresses the dynamic range of |FFT(a)|
    to make the convolution flatter. Used for generating diverse initializations.
    """
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    size = 1 << (2 * n - 1).bit_length()
    for _ in range(iterations):
        A = np.fft.rfft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)
        mag_new = mag ** beta
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        delta = np.max(np.abs(a_new - a))
        a = a_new
        if delta < 1e-14:
            break
    return a


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
