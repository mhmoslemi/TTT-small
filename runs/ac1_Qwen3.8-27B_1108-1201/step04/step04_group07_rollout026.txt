def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_max(conv, t):
    """Smooth approximation of max using log-sum-exp."""
    m = conv.max()
    return m + np.log(np.exp(t * (conv - m)).sum()) / t

"""

2. Exact coordinate descent with bounded 1D minimization

The key innovation is the FFT-based smooth max gradient enabling fast Nesterov steps,

import numpy as np
from scipy.optimize import minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_max_value(conv, t):
    """Compute smooth max value: (1/t)*log(sum(exp(t*x_i)))"""
    m = conv.max()
    shifted = t * (conv - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.exp(shifted).sum()) / t


def smooth_max_grad_fft(a, conv, t):
    """
    Compute gradient of smooth_max(conv(a,a), t) w.r.t. a using FFT.
    grad[i] = (2/t) * sum_{j=0}^{n-1} a[j] * w[i+j]
    where w = softmax(t * conv)
    """
    n = len(a)
    m = conv.max()
    shifted = t * (conv - m)
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[conv.argmax()] = 1.0
        wsum = 1.0
    w /= wsum

    # Compute cross-correlation via FFT
    # grad[i] = (2/t) * conv_result[2n-2-i] where conv_result = a * w_rev (linear convolution)
    w_rev = w[::-1]
    fft_size = 1 << (3 * n - 3).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)[:3 * n - 2]

    # Extract grad[i] = (2/t) * conv_result[2n-2-i] for i=0..n-1
    # Indices: 2n-2, 2n-3, ..., n-1
    grad = (2.0 / t) * conv_result[n - 1:2 * n - 1][::-1]
    return grad


def smooth_max_grad_direct(a, conv, t):
    """Direct O(n^2) computation of smooth max gradient."""
    n = len(a)
    m = conv.max()
    shifted = t * (conv - m)
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[conv.argmax()] = 1.0
        wsum = 1.0
    w /= wsum

    grad = np.zeros(n)
    for i in range(n):
        j_max = min(n, 2 * n - 1 - i)
        if j_max > 0:
            grad[i] = (2.0 / t) * np.dot(a[:j_max], w[i:i + j_max])
    return grad


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coordinate_descent_pass(a, conv, S, top_k=None):
    """
    One pass of coordinate descent. Updates each coordinate using bounded 1D minimization.
    Returns updated a, conv, S, and objective.
    """
    n = len(a)
    if top_k is not None and top_k < n:
        # Only update coordinates associated with top peaks
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        # For each peak k, the affected coordinates are i where 0 <= k-i < n
        # i.e., i in [max(0,k-n+1), min(n-1,k)]
        coords_to_update = set()
        for k in peaks:
            k_lo = max(0, k - n + 1)
            k_hi = min(n - 1, k)
            coords_to_update.update(range(k_lo, k_hi + 1))
        coords_to_update = sorted(coords_to_update)
    else:
        coords_to_update = list(range(n))

    a_new = a.copy()
    obj_best = 2 * n * conv.max() / (S * S)

    for i in coords_to_update:
        a_i = a_new[i]
        if a_i < 1e-15 and S - a_i < 1e-15:
            continue

        # Determine affected convolution range
        k_lo = max(0, i)
        k_hi = min(2 * n - 2, i + n - 1)
        c_vals = 2.0 * a_new[k_lo - i:k_hi - i + 1]  # 2*a[k-i] for k in [k_lo, k_hi]
        conv_affected = conv[k_lo:k_hi + 1].copy()
        M_unaffected = np.max(conv[:k_lo]) if k_lo > 0 else 0.0
        if k_hi < 2 * n - 2:
            M_unaffected = max(M_unaffected, np.max(conv[k_hi + 1:]))

        def obj_1d(x):
            if x < 0:
                x = 0.0
            d = x - a_i
            S_new = S + d
            if S_new < 1e-15:
                return 1e10
            new_conv_aff = conv_affected + d * c_vals
            M_new = max(M_unaffected, new_conv_aff.max()) if len(new_conv_aff) > 0 else M_unaffected
            return 2.0 * n * M_new / (S_new * S_new)

        # Bounded search
        upper = max(3.0 * a_i + 0.5 / n, 1.0 / n)
        res = minimize_scalar(obj_1d, bounds=(0.0, upper), method='bounded',
                              options={'xatol': 1e-10})
        if res.success:
            new_val = res.x
            if new_val < 0:
                new_val = 0.0
            if abs(new_val - a_i) > 1e-15:
                d = new_val - a_i
                a_new[i] = new_val
                S += d
                conv[k_lo:k_hi + 1] += d * c_vals

    obj_new = 2 * n * conv.max() / (S * S)
    return a_new, conv, S, obj_new


def nesterov_smooth_max(a0, t_start, t_end, time_limit, lr_init=0.5,
                        momentum_decay=0.02, restart_interval=200, verbose=False):
    """
    Nesterov accelerated gradient on smooth max objective with annealed temperature.
    Works on simplex (sum=1, non-negative).
    """
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    y = a.copy()  # Nesterov lookahead
    momentum = 0.0

    conv = fast_convolve(y)
    best_obj = 2 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    k = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        k += 1
        # Anneal temperature
        frac = min(1.0, k / max(1, int(time_limit * 30)))
        t_cur = t_start + (t_end - t_start) * frac
        t_cur = max(t_cur, 1.0)

        # Compute gradient at y
        conv_y = fast_convolve(y)
        grad = smooth_max_grad_fft(y, conv_y, t_cur)

        # Project gradient to tangent space of simplex (subtract mean)
        grad_proj = grad - grad.mean()

        # Learning rate with decay
        lr = lr_init / (1.0 + k * momentum_decay)
        lr = min(lr, 2.0 / (t_cur * np.max(conv_y) + 1e-10))
        lr = max(lr, 0.01)

        # Nesterov step
        momentum = (1.0 + (k + 1) * 0.5) / (2.0 + (k + 1) * 0.5)
        y_new = a + momentum * (a - y)
        y = y_new

        # Gradient step
        a_candidate = y - lr * grad_proj

        # Project to simplex
        a = project_simplex(a_candidate)

        # Evaluate
        conv = fast_convolve(a)
        obj = 2 * n * conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= restart_interval:
            # Restart with perturbation
            rng = np.random.default_rng(k * 7 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            y = a.copy()
            no_improve = 0

        if verbose and k % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Nesterov k={k}, t={t_cur:.1f}, obj={best_obj:.10f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj / (2 * n)  # Return max(conv) for sum=1 normalization


def nesterov_with_cd(a0, t_start, t_end, time_limit, lr_init=0.5,

    n = len(a0)

    t0 = time.monotonic()

        t_cur = t_start + (t_end - t_start) * frac
        t_cur = max(t_cur, 1.0)

        # Gradient step
        conv_y = fast_convolve(y)
        grad = smooth_max_grad_fft(y, conv_y, t_cur)
        grad_proj = grad - grad.mean()

        lr = lr_init / (1.0 + k * 0.02)
        lr = min(lr, 2.0 / (t_cur * np.max(conv_y) + 1e-10))
        lr = max(lr, 0.005)

        momentum = (1.0 + (k + 1) * 0.5) / (2.0 + (k + 1) * 0.5)
        y_new = a + momentum * (a - y)
        y = y_new

        a_candidate = y - lr * grad_proj
        a = project_simplex(a_candidate)

            y = a.copy()

        # Evaluate
        conv = fast_convolve(a)
        obj = 2 * n * conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

            print(f"      N+CD k={k}, t={t_cur:.1f}, obj={best_obj:.10f}, elapsed={elapsed:.1f}s")

def peak_targeting_optimize(a0, time_limit, verbose=False):

        # Compute aggregated gradient targeting peaks
        weights = np.zeros(2 * n - 1)
        peak_vals = conv[peak_indices]

            continue

        # Animated learning rate
        lr = 0.1 / (1.0 + 0.01 * k)

def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 1.0))
    inits.append(rng.dirichlet(np.ones(n) * 3.0))

    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)

    # Gaussian bumps (various widths)
    for sigma_frac in [0.1, 0.15, 0.2, 0.3, 0.4]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - mid) / sigma) ** 2)
        inits.append(g / g.sum())

    # Cosine
    for freq in [1, 2, 3]:
        c = np.abs(np.cos(freq * np.pi * x / n)) + 0.1
        inits.append(c / c.sum())

    # Edge-heavy
    dist = np.abs(x - mid) / (0.5 * n)
    for alpha in [0.5, 1.0, 2.0]:
        e = (1.0 - dist) ** alpha + 0.01
        inits.append(e / e.sum())

    # Beta-like
    for ab, bb in [(2, 2), (1.5, 1.5), (3, 3), (1, 2), (2, 1)]:
        b = (x + 1) ** ab * (n - x) ** bb + 1e-6
        inits.append(b / b.sum())

    # Interpolated parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Perturbed parent
        for scale in [0.01, 0.03, 0.05]:
            pp = project_simplex(px / ps + rng.standard_normal(n) * scale)
            inits.append(pp)

    return inits

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
    except Exception:
        pass

    # Phase 1: Initialization
    print("=== Phase 1: Initialization ===")
    t_start = time.monotonic()

    n_base = 384
    if parent is not None:
        n_base = len(parent)

    candidates = []
    inits = generate_initializations(n_base, parent, rng)
    for i, ini in enumerate(inits):
        candidates.append((f"init_{i}", ini, n_base))

    # Different lengths
    for n_try in [256, 320, 448, 512]:
        if n_try == n_base:
            continue
        ini = np.ones(n_try) / n_try
        candidates.append((f"uniform_{n_try}", ini, n_try))
        if parent is not None:
            px = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, len(parent)), parent)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                candidates.append((f"parent_interp_{n_try}", px / ps, n_try))

    for label, cand, n_c in candidates:
        if time.monotonic() > deadline - 1500:
            break
        try_seq(cand, label)

    if best_seq is None:
        best_seq = np.ones(n_base) / n_base
        try_seq(best_seq, "fallback")

    print(f"  Phase 1 best: {best_score:.8f} (n={len(best_seq)})", flush=True)

    # Phase 2: Nesterov smooth max gradient (primary optimization)
    print("\n=== Phase 2: Nesterov Smooth Max Gradient ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 1000
    t_p2 = max(30, min(remaining, 500))

    # Multiple Nesterov runs with different hyperparameters
    nesterov_configs = [
        {'t_start': 5, 't_end': 300, 'lr_init': 1.0, 'label': 'fast_anneal'},
        {'t_start': 10, 't_end': 200, 'lr_init': 0.5, 'label': 'med_anneal'},
        {'t_start': 20, 't_end': 500, 'lr_init': 0.3, 'label': 'slow_anneal'},
    ]

    per_config_time = t_p2 / len(nesterov_configs)
    for cfg in nesterov_configs:
        if time.monotonic() > deadline - 800:
            break
        t_alloc = min(per_config_time * 0.8, deadline - time.monotonic() - 700)
        if t_alloc < 20:
            break
        print(f"  Nesterov [{cfg['label']}]: t=[{cfg['t_start']},{cfg['t_end']}], lr={cfg['lr_init']}, budget={t_alloc:.0f}s")
        a_opt, _ = nesterov_smooth_max(best_seq, t_start=cfg['t_start'], t_end=cfg['t_end'],
                                       time_limit=t_alloc, lr_init=cfg['lr_init'],
                                       verbose=False)
        try_seq(a_opt, cfg['label'])

    print(f"  Phase 2 best: {best_score:.8f}", flush=True)

    # Phase 3: Nesterov + CD hybrid
    print("\n=== Phase 3: Nesterov + Coordinate Descent Hybrid ===")
    remaining = deadline - time.monotonic() - 600
    t_p3 = max(20, min(remaining, 400))

    if t_p3 > 20:
        configs3 = [
            {'t_start': 50, 't_end': 500, 'lr_init': 0.3, 'cd_every': 30, 'cd_top_k': 50, 'label': 'hy1'},
            {'t_start': 20, 't_end': 200, 'lr_init': 0.5, 'cd_every': 50, 'cd_top_k': 30, 'label': 'hy2'},
        ]
        per3 = t_p3 / len(configs3)
        for cfg in configs3:
            if time.monotonic() > deadline - 500:
                break
            t_alloc = min(per3 * 0.8, deadline - time.monotonic() - 400)
            if t_alloc < 15:
                break
            print(f"  Hybrid [{cfg['label']}]: budget={t_alloc:.0f}s")
            a_opt, _ = nesterov_with_cd(best_seq, t_start=cfg['t_start'], t_end=cfg['t_end'],
                                        time_limit=t_alloc, lr_init=cfg['lr_init'],
                                        cd_every=cfg['cd_every'], cd_top_k=cfg['cd_top_k'])
            try_seq(a_opt, cfg['label'])

    print(f"  Phase 3 best: {best_score:.8f}", flush=True)

    # Phase 4: Peak targeting
    print("\n=== Phase 4: Peak Targeting ===")
    remaining = deadline - time.monotonic() - 400
    t_p4 = max(15, min(remaining, 250))

    if t_p4 > 15:
        a_opt, _ = peak_targeting_optimize(best_seq, time_limit=t_p4 * 0.6)
        try_seq(a_opt, "peak_tgt1")

        rem = deadline - time.monotonic() - 300
        if rem > 15:
            # Perturbed restart
            f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.01)
            a_opt, _ = peak_targeting_optimize(f_pert, time_limit=min(rem * 0.6, 100))
            try_seq(a_opt, "peak_tgt2")

    print(f"  Phase 4 best: {best_score:.8f}", flush=True)

    # Phase 5: Pure coordinate descent polishing
    print("\n=== Phase 5: Coordinate Descent Polishing ===")
    remaining = deadline - time.monotonic() - 200
    t_p5 = max(10, min(remaining, 300))

    if t_p5 > 10:
        a_opt, _ = pure_cd_optimize(best_seq, time_limit=t_p5 * 0.6,
                                    top_k_schedule=True)
        try_seq(a_opt, "cd1")

        rem = deadline - time.monotonic() - 150
        if rem > 10:
            # Full CD (no top_k restriction)
            a_opt, _ = pure_cd_optimize(best_seq, time_limit=min(rem * 0.5, 100),
                                        top_k_schedule=None)
            try_seq(a_opt, "cd_full")

    print(f"  Phase 5 best: {best_score:.8f}", flush=True)

    # Phase 6: Alternative lengths + final polish
    print("\n=== Phase 6: Length Exploration & Final Polish ===")
    remaining = deadline - time.monotonic() - 60
    n_best = len(best_seq)

    for n_try in [n_best - 32, n_best + 32, n_best - 64, n_best + 64]:
        if n_try < 64 or time.monotonic() > deadline - 40:
            continue
        px = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_best), best_seq)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps < 1e-10:
            continue
        px = px / ps
        try_seq(px, f"interp_n{n_try}")

        # Quick CD on this length
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, _ = pure_cd_optimize(px, time_limit=min(rem * 0.5, 60))
            try_seq(a_opt, f"cd_n{n_try}")

    # Final: perturbation + Nesterov
    rem = deadline - time.monotonic() - 20
    if rem > 30:
        for scale in [0.005, 0.01, 0.02, 0.03]:
            if rem < 15:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * scale)
            t_each = min(rem * 0.3, 40)
            a_opt, _ = nesterov_smooth_max(f_pert, t_start=50, t_end=300,
                                           time_limit=t_each, lr_init=0.3)
            try_seq(a_opt, f"pert_{scale}")
            rem = deadline - time.monotonic() - 15

    # Final CD polish on best
    rem = deadline - time.monotonic() - 10
    if rem > 10 and best_seq is not None:
        a_opt, _ = pure_cd_optimize(best_seq, time_limit=min(rem, 30))
        try_seq(a_opt, "final_cd")

    # Final verification
    if best_seq is not None:
        final_score = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})", flush=True)
    else:
        best_seq = np.ones(n_base) / n_base
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}", flush=True)

    return [float(x) for x in best_seq]
