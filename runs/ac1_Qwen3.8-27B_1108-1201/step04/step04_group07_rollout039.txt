"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using coordinate descent with normalization,
smooth spectral gradient (minimizing sum|F|^4), LP-based Frank-Wolfe, and subgradient
methods. The key new strategy is fast coordinate descent exploiting the 1D subproblem
structure, combined with frequency-domain optimization for global progress.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build B matrix where (B*g)[k] = (f*g)[k]."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B


def solve_linearized_lp(f, conv_f, B=None):
    """Solve: min t s.t. t >= conv_f[k] + 2*(B*g)[k], g >= -f, sum(g)=0."""
    n = len(f)
    if B is None:
        B = build_conv_matrix(f)
    c = np.zeros(n + 1)
    c[0] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B
    b_ub = -conv_f
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def solve_maxsum_lp(f, M):
    """Solve: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None and np.any(res.x > 0):
        return res.x
    return None


def coordinate_descent(a0, time_limit, n_grid=60, verbose=False):
    """
    Coordinate descent with normalization.
    Minimizes max_k(C_k + 2xD_k + Ex^2) / (s+x)^2 for each coordinate.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)
    order = np.arange(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        np.random.shuffle(order) if it % 10 == 0 else None
        if it % 10 == 0:
            order = rng.permutation(n)

        improved = False
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            ai = a[i]
            s = 1.0 - ai

            if s < 1e-15:
                continue

            # Compute C, D, E vectors
            conv = fast_convolve(a)
            C = conv.copy()
            D = np.zeros(2 * n - 1)
            E = np.zeros(2 * n - 1)

            k_vals = np.arange(2 * n - 1)
            ki = k_vals - i

            valid_cross = (ki >= 0) & (ki < n) & (ki != i)
            valid_self = (ki == i)

            if valid_cross.any():
                C[valid_cross] -= 2.0 * ai * a[ki[valid_cross]]
                D[valid_cross] = a[ki[valid_cross]]
            if valid_self.any():
                C[valid_self] -= ai * ai
                E[valid_self] = 1.0

            # Grid search for optimal x
            x_lo = 0.0
            x_hi = max(ai * 4.0, 0.02)
            x_grid = np.linspace(x_lo, x_hi, n_grid)

            terms = C[None, :] + 2.0 * x_grid[:, None] * D[None, :] + \
                    (x_grid[:, None] ** 2) * E[None, :]
            maxes = terms.max(axis=1)
            f_vals = maxes / (s + x_grid) ** 2

            best_idx = f_vals.argmin()
            best_x = x_grid[best_idx]
            best_f = f_vals[best_idx]

            # Fine refinement
            if best_f < best_obj - 1e-14:
                step = x_hi / n_grid
                fine_lo = max(0.0, best_x - step)
                fine_hi = best_x + step
                x_fine = np.linspace(fine_lo, fine_hi, 20)
                terms_f = C[None, :] + 2.0 * x_fine[:, None] * D[None, :] + \
                          (x_fine[:, None] ** 2) * E[None, :]
                maxes_f = terms_f.max(axis=1)
                f_fine = maxes_f / (s + x_fine) ** 2
                fidx = f_fine.argmin()
                if f_fine[fidx] < best_f:
                    best_x = x_fine[fidx]
                    best_f = f_fine[fidx]

            # Try also near zero and at current value
            extra_x = np.array([0.0, ai])
            terms_e = C[None, :] + 2.0 * extra_x[:, None] * D[None, :] + \
                      (extra_x[:, None] ** 2) * E[None, :]
            maxes_e = terms_e.max(axis=1)
            f_e = maxes_e / (s + extra_x) ** 2
            eidx = f_e.argmin()
            if f_e[eidx] < best_f:
                best_x = extra_x[eidx]
                best_f = f_e[eidx]

            if best_f < best_obj - 1e-12:
                a[i] = best_x
                new_s = a.sum()
                if new_s > 1e-15:
                    a = a / new_s
                conv = fast_convolve(a)
                new_obj = conv.max()
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            # Perturb and restart
            scale = 0.003 + 0.002 * (no_improve // 100)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            conv = fast_convolve(a)
            if conv.max() < best_obj:
                best_obj = conv.max()
                best_a = a.copy()
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    CD iter {it}: obj={best_obj:.10f}, no_improve={no_improve}")

    return best_a, best_obj


def spectral_gradient_descent(a0, time_limit, lr=0.002, mom=0.9, 
                               n_restarts=5, verbose=False):
    """
    Gradient descent on the smooth proxy: sum|DFT(a)|^4 / (sum(a))^4.
    This encourages flat auto-convolution via Parseval's theorem.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    # Use FFT size >= 2n-1 for linear convolution analysis
    fft_size = 1 << (2 * n - 1).bit_length()

    def spectral_obj(a):
        F = np.fft.rfft(a, fft_size)
        return np.sum(np.abs(F) ** 4)

    def spectral_grad(a):
        F = np.fft.rfft(a, fft_size)
        # grad of sum|F|^4 w.r.t. a: 8 * ifft(|F|^2 * conj(F))
        grad_full = np.fft.irfft(np.abs(F) ** 2 * F.conj(), fft_size).real[:n]
        return 8.0 * grad_full

    best_a = a.copy()
    best_obj_val = fast_convolve(a).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break

        if restart > 0:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.01 * restart)
            v = np.zeros(n)

        v = np.zeros(n)
        cur_lr = lr
        no_improve = 0
        it = 0

        while time.monotonic() - t0 < time_limit:
            it += 1
            g = spectral_grad(a)
            # Normalize gradient by current objective scale
            Obj = spectral_obj(a)
            if Obj > 1e-30:
                g_scaled = g / Obj
            else:
                g_scaled = g

            v = mom * v - cur_lr * g_scaled
            a_new = project_simplex(a + v)

            # Evaluate
            conv = fast_convolve(a_new)
            obj_val = conv.max()

            if obj_val < best_obj_val - 1e-14:
                best_obj_val = obj_val
                best_a = a_new.copy()
                no_improve = 0
                a = a_new
            else:
                no_improve += 1

            cur_lr *= 0.9995

            if no_improve > 200:
                break

            if verbose and it % 100 == 0:
                print(f"    SG iter {restart}_{it}: obj={best_obj_val:.10f}")

    return best_a, best_obj_val


def subgradient_optimize(a0, time_limit, lr=0.004, mom=0.93, top_k=30, 
                         temp=0.0001, verbose=False):
    """Smoothed subgradient descent on max convolution."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    v = np.zeros(n)
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for ii in range(len(peaks)):
            if w[ii] < 1e-13:
                continue
            k = int(peaks[ii])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[ii] * a[k - idx]

        v = mom * v - cur_lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.005

        if no_improve > 300:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            cur_lr = max(lr * 0.7, 0.001)
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    SG iter {it}: obj={best_obj:.10f}")

    return best_a, best_obj

    while time.monotonic() - t0 < time_limit:

        conv = fast_convolve(f)
        current_obj = conv.max()

        g = solve_linearized_lp(f, conv, B=B)
        if g is None:
            no_improve += 5
            if no_improve > 100:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                B = build_conv_matrix(f)
                no_improve = 0
            continue

        target = f + g
        target = np.maximum(target, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts

        if np.max(np.abs(target - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:

                B = build_conv_matrix(f)
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = current_obj
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_new = (1.0 - alpha) * f + alpha * target

                else:
                    hi = mid

        if best_alpha > 1e-8:

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)
    for sigma_frac in [0.1, 0.2, 0.3, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - mid) / sigma) ** 2)
        inits.append(g / g.sum())

    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.02
    inits.append(c / c.sum())

    # Edge-heavy
    edge = 1.0 / (0.3 * np.abs(x - mid) / n + 0.05)
    inits.append(edge / edge.sum())

    # Two peaks
    two = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + \
          np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2) + 0.01
    inits.append(two / two.sum())

    # Comb pattern
    period = max(2, n // 12)
    comb = np.zeros(n)
    comb[::period] = 1.0
    inits.append(comb / comb.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n),
                       np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

    else:
        candidate = np.ones(384) / 384

    try_seq(candidate, "parent")

    # Try a few other inits
    n_target = len(candidate)
    for ini in generate_inits(n_target, parent, rng)[:4]:
        sc = try_seq(ini, "init")
        if sc < best_score * 1.05:
            candidate = ini.copy()

    # Try different lengths
    for n_try in [320, 448, 512]:
        if n_try == n_target:
            continue
        px = np.interp(np.linspace(0, 1, n_try),
                       np.linspace(0, 1, len(candidate)), candidate)
        px = np.maximum(px, 0)
        px = px / px.sum()
        try_seq(px, f"len{n_try}")

    print(f"  Phase 1 done. Best: {best_score:.8f} (n={len(best_seq)}), "
          f"time={time.monotonic()-t_start:.1f}s")

    n_best = len(best_seq)

    # === Phase 2: Coordinate Descent (main workhorse) ===
    print("\n=== Phase 2: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    t_cd = min(remaining * 0.35, 550)

    if t_cd > 30:
        print(f"  Running CD, budget={t_cd:.0f}s, n={n_best}")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd * 0.5,
                                      n_grid=50, verbose=True)
        try_seq(a_opt, "cd1")

        # Second CD run from best
        rem = deadline - time.monotonic()
        if rem > 60 and t_cd * 0.3 < rem:
            t_cd2 = min(t_cd * 0.3, rem - 50)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd2,
                                          n_grid=40)
            try_seq(a_opt, "cd2")

    print(f"  Phase 2 done. Best: {best_score:.8f}")

    # === Phase 3: Spectral + Subgradient ===
    print("\n=== Phase 3: Spectral Gradient + Subgradient ===")
    remaining = deadline - time.monotonic()
    t_sg = min(remaining * 0.2, 300)

    if t_sg > 30:
        # Spectral gradient
        t_spec = min(t_sg * 0.4, 120)
        a_opt, _ = spectral_gradient_descent(best_seq, time_limit=t_spec,
                                             lr=0.003, mom=0.92,
                                             n_restarts=3)
        try_seq(a_opt, "spectral")

        # Subgradient
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_sub = min(rem * 0.5, 150)
            top_k = max(10, n_best // 8)
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_sub,
                                            lr=0.005, mom=0.93,
                                            top_k=top_k, temp=0.00005)
            try_seq(a_opt, "subgrad")

    print(f"  Phase 3 done. Best: {best_score:.8f}")

    # === Phase 4: LP Frank-Wolfe ===
    print("\n=== Phase 4: Linearized LP Frank-Wolfe ===")
    remaining = deadline - time.monotonic()
    t_fw = min(remaining * 0.25, 400)

    if t_fw > 30:
        print(f"  Running LinFW, budget={t_fw:.0f}s, n={n_best}")
        a_opt, _ = linearized_fw_optimize(best_seq, time_limit=min(t_fw * 0.6, 250))
        try_seq(a_opt, "linfw")

        # Perturbed restart
        rem = deadline - time.monotonic() - 20
        if rem > 30:
            noise = rng.standard_normal(n_best) * 0.005
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = linearized_fw_optimize(f_pert,
                                              time_limit=min(rem * 0.4, 120))
            try_seq(a_opt, "linfw_pert")

    print(f"  Phase 4 done. Best: {best_score:.8f}")

    # === Phase 5: Local search + length exploration ===
    print("\n=== Phase 5: Local Search & Length Exploration ===")
    remaining = deadline - time.monotonic()
    t_local = min(remaining * 0.3, 250)

    if t_local > 20:
        # Local search
        t_ls = min(t_local * 0.4, 100)
        a_opt, _ = local_search(best_seq, time_limit=t_ls, rng=rng)
        try_seq(a_opt, "local")

        # More CD from different starting points
        rem = deadline - time.monotonic() - 20
        for attempt in range(3):
            if rem < 15:
                break
            noise_scale = 0.003 * (attempt + 1)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            t_each = min(rem * 0.3, 60)
            a_opt, _ = coordinate_descent(f_pert, time_limit=t_each, n_grid=40)
            try_seq(a_opt, f"cd_pert{attempt}")
            rem = deadline - time.monotonic() - 15

    # Try nearby lengths
    rem = deadline - time.monotonic() - 15
    if rem > 20:
        for n_try in [n_best - 32, n_best + 32, n_best - 16, n_best + 16]:
            if n_try < 32 or rem < 15:
                continue
            px = np.interp(np.linspace(0, 1, n_try),
                           np.linspace(0, 1, n_best), best_seq)
            px = np.maximum(px, 0)
            px = px / px.sum()
            a_opt, _ = coordinate_descent(px, time_limit=min(rem * 0.3, 40),
                                          n_grid=30)
            try_seq(a_opt, f"len{n_try}")
            rem = deadline - time.monotonic() - 10

    # Final polish with fine CD
    rem = deadline - time.monotonic() - 10
    if rem > 20 and best_seq is not None:
        a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem * 0.7, 80),
                                      n_grid=80)
        try_seq(a_opt, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_sequence(best_seq.tolist())
        if np.isfinite(fs):
            best_score = float(fs)
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
