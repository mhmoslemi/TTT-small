**Algorithm: Active Set Gradient Descent with Backtracking**

1. Initialize a from parent or a good starting point.
2. Repeat:

   b. Compute the subgradient g = (1/|S|) * sum_{k in S} grad_of_bk(a, k).
   c. This is the gradient of the "smooth approximation" max_k b_k ≈ (1/|S|) * sum_{k in S} b_k.
   d. Take a step in the direction -g (with projection onto nonnegativity).
   e. Use backtracking line search to ensure the objective decreases.
   f. Normalize (the objective is scale-invariant, so we can normalize at each step).

The issue with this approach is that the active set changes as we move, so the gradient direction changes. This can cause oscillation.

- Maintain a set of "sample" lags S (initially the active peaks)
- Optimize the objective (1/|S|) * sum_{k in S} b_k (which is smooth)
- After convergence, check if any lag outside S has exceeded the average
- If so, add it to S and repeat

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_C(a):
    n = len(a)
    m = 2 * n - 1
    C = np.zeros((m, n))
    for j in range(n):
        C[j:j+n, j] = a
    return C

def lp_direction(a, M):
    n = len(a)
    C = build_C(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None and res.x.sum() > 1e-10:
            return res.x
    except Exception:
        pass
    return None

def subgradient(a, conv, topk=10):
    """Compute subgradient of max(conv) using top-k averaging."""
    n = len(a)
    # Find top-k peaks
    peaks = np.argpartition(conv, -topk)[-topk:]
    # Weight by how close they are to the max
    peak_vals = conv[peaks]
    max_val = peak_vals.max()
    # Hard active set: only peaks within 1% of max
    active = peaks[peak_vals >= max_val * 0.99]
    if len(active) == 0:
        active = peaks[:1]
    
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        grad[js] += 2.0 * a[k - js]
    grad /= len(active)
    return grad

    v = np.zeros(n)

    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
