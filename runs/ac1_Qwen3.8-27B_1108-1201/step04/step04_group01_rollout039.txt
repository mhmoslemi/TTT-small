"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. KL Mirror Descent with Log-Sum-Exp smoothing (fast coarse optimization)
2. FW-LP for structural refinement
3. High-beta LSE gradient polish (Euclidean)
4. Alternating refinement for final convergence
5. Multi-start with diverse initializations
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1] if cond.any() else n - 1
    if not cond.any():
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """Compute sum_k weights[k] * f[k-i] for each i=0..n-1."""
    n = len(f)
    w_len = len(weights)  # should be 2n-1
    conv_len = w_len + n - 1
    fft_sz = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_sz)
    F_flipped = np.fft.rfft(f[::-1], fft_sz)
    conv = np.fft.irfft(W * F_flipped, fft_sz)[:conv_len]
    return conv[n-1 : n-1+n]


def build_lp_matrix(f):
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(f, M):
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 20})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def kl_mirror_descent(f0, time_limit, beta_start=5.0, beta_end=200.0,
                      lr_start=1.0, lr_end=0.1, top_k=None):
    """KL mirror descent with LSE smoothing.
    Update: f_new[i] ∝ f[i] * exp(-lr * grad[i])
    This naturally stays on the simplex after normalization.
    """
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    beta = beta_start
    lr = lr_start
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        
        # Compute softmax weights
        if top_k is not None and top_k < len(conv):
            # Focus on top-k peaks
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            w = np.zeros(len(conv))
            vals = conv[peaks]
            scaled = beta * vals
            scaled -= scaled.max()
            exp_vals = np.exp(scaled)
            w[peaks] = exp_vals / exp_vals.sum()
        else:
            scaled = beta * conv
            scaled -= scaled.max()
            exp_vals = np.exp(scaled)
            w = exp_vals / exp_vals.sum()
        
        # Gradient
        grad = compute_lse_grad(w, f)
        
        # KL mirror descent update: f_new ∝ f * exp(-lr * grad)
        log_update = -lr * grad
        # Clip for numerical stability
        log_update = np.clip(log_update, -100, 100)
        f_new = f * np.exp(log_update)
        
        s_new = f_new.sum()
        if s_new < 1e-300:
            f = best_f.copy()
            continue
        f_new = f_new / s_new
        
        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 20:
                lr *= 0.95
            if no_improve > 200:
                f = best_f.copy()
                lr = lr_start
                no_improve = 0
        
        # Ramp beta
        if beta < beta_end:
            beta *= 1.005
        # Decay lr
        if lr > lr_end:
            lr = max(lr_end, lr * 0.999)
        
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    KLM it={it}, obj={best_obj:.10f}, beta={beta:.1f}, lr={lr:.4f}, t={elapsed:.0f}s")
    
    return best_f, best_obj


def fw_lp_optimize(f0, time_limit, max_M_expand=5.0):
    """Frank-Wolfe with LP oracle."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        g = solve_lp(f, M * m_exp)
        if g is None:
            m_exp = min(m_exp * 1.15, max_M_expand)
            if m_exp >= max_M_expand:
                m_exp = 1.0
            continue
        
        g_s = g.sum()
        if g_s < 1e-15:
            break
        g = g / g_s
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = np.linspace(0.02, 1.0, 12)
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            # Refine with bisection
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = np.maximum((1.0 - mid) * f + mid * g, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 4.0)
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.002 * min(1.0, (no_improve - 50) / 30.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj


def lse_euclidean_optimize(f0, time_limit, beta_start=50.0, beta_end=10000.0,

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = lr_start
    momentum = 0.90
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        
        if top_k is not None and top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            w = np.zeros(len(conv))
            vals = conv[peaks]
            scaled = beta * vals
            scaled -= scaled.max()
            exp_vals = np.exp(scaled)
            w[peaks] = exp_vals / exp_vals.sum()
        else:
            scaled = beta * conv
            scaled -= scaled.max()
            exp_vals = np.exp(scaled)
            w = exp_vals / exp_vals.sum()
        
        grad = compute_lse_grad(w, f)
        
        # Normalize gradient
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            no_improve += 100
            break
        grad = grad / gn
        
        # Momentum update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:

            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.001, 3.0 / n)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.93
                momentum = max(0.85, momentum - 0.003)
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(lr_start * 0.5, 0.001)
                beta = min(beta * 1.3, beta_end)
                no_improve = 0
        
        if beta < beta_end:
            beta *= 1.004
        
        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it}, obj={best_obj:.10f}, beta={beta:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """Compute grad[j] = sum_k weights[k] * f[k-j] for j=0..n-1.
    weights has length 2n-1, f has length n.
    Uses FFT via convolution with reversed f.
    """
    n = len(f)
    w_len = len(weights)
    conv_len = w_len + n - 1
    fft_sz = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_sz)
    F_rev = np.fft.rfft(f[::-1], fft_sz)
    conv = np.fft.irfft(W * F_rev, fft_sz)[:conv_len]
    return conv[n - 1: n - 1 + n]


def build_lp_matrix(f):
    """Build sparse constraint matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, lp_timeout=15):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns g or None."""
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': lp_timeout})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = np.ones(448) / 448
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    current_best = best_tracker[1].copy() if best_tracker[1] is not None else parent.copy()
    n_cur = len(current_best)

    # === Phase 1: KL Mirror Descent (coarse) ===
    print("\n=== Phase 1: KL Mirror Descent ===")
    t_remaining = deadline - time.monotonic()
    t_alloc = min(200, t_remaining * 0.12)
    if t_alloc > 20:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = kl_mirror_descent(current_best, time_limit=t_alloc,
                                        beta_start=3.0, beta_end=80.0,
                                        lr_start=1.0, lr_end=0.1,
                                        top_k=max(n_cur // 4, 16))
        try_sequence(a_opt, best_tracker)
        if best_tracker[1] is not None:
            current_best = best_tracker[1].copy()
            n_cur = len(current_best)

    # === Phase 2: FW-LP (structural) ===
    print(f"\n=== Phase 2: FW-LP ===")
    t_remaining = deadline - time.monotonic()
    t_alloc = min(400, t_remaining * 0.25)
    if t_alloc > 30:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = fw_lp_optimize(current_best, time_limit=t_alloc)
        try_sequence(a_opt, best_tracker)
        if best_tracker[1] is not None:
            current_best = best_tracker[1].copy()
            n_cur = len(current_best)

    # === Phase 3: LSE Polish ===
    print(f"\n=== Phase 3: LSE Polish ===")
    t_remaining = deadline - time.monotonic()
    t_alloc = min(350, t_remaining * 0.22)
    if t_alloc > 30:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = lse_euclidean_optimize(current_best, time_limit=t_alloc,
                                             beta_start=15.0, beta_end=5000.0,
                                             lr_start=1.0 / n_cur,
                                             top_k=max(n_cur // 3, 12))
        try_sequence(a_opt, best_tracker)
        if best_tracker[1] is not None:
            current_best = best_tracker[1].copy()
            n_cur = len(current_best)

    # === Phase 4: Peak Shaving + Alternating ===
    print(f"\n=== Phase 4: Peak Shaving + Alternating ===")
    t_remaining = deadline - time.monotonic()

    # Peak shaving first
    t_ps = min(80, t_remaining * 0.1)
    if t_ps > 20:
        a_opt, _ = peak_shave(current_best, time_limit=t_ps,
                              top_k=max(6, n_cur // 50), lr=0.003)
        try_sequence(a_opt, best_tracker)
        if best_tracker[1] is not None:
            current_best = best_tracker[1].copy()
            n_cur = len(current_best)

    # Alternating LP + LSE
    t_remaining = deadline - time.monotonic()
    while t_remaining > 100:
        t_lp = min(100, t_remaining * 0.35)
        if t_lp > 20:
            a_opt, _ = fw_lp_optimize(current_best, time_limit=t_lp)
            try_sequence(a_opt, best_tracker)
            if best_tracker[1] is not None:
                current_best = best_tracker[1].copy()
                n_cur = len(current_best)

        t_remaining = deadline - time.monotonic()
        if t_remaining < 100:
            break

        t_lse = min(100, t_remaining * 0.35)
        if t_lse > 20:
            a_opt, _ = lse_euclidean_optimize(current_best, time_limit=t_lse,
                                               beta_start=200.0, beta_end=30000.0,
                                               lr_start=0.0008,
                                               top_k=max(n_cur // 4, 10),
                                               momentum_init=0.88)
            try_sequence(a_opt, best_tracker)
            if best_tracker[1] is not None:
                current_best = best_tracker[1].copy()
                n_cur = len(current_best)

        t_remaining = deadline - time.monotonic()

    # === Phase 5: Alternative lengths ===
    print(f"\n=== Phase 5: Alternative Lengths ===")
    t_remaining = deadline - time.monotonic()

    alt_lengths = [320, 384, 448, 512, 576, 640]
    if n_parent in alt_lengths:
        alt_lengths.remove(n_parent)
    # Prioritize lengths close to what we already have
    alt_lengths.sort(key=lambda x: abs(x - len(current_best)))

    for n_alt in alt_lengths[:4]:
        t_remaining = deadline - time.monotonic()
        if t_remaining < 70:
            break
        t_alloc = min(110, t_remaining - 30)
        if t_alloc < 40:
            break

        # Interpolate to new length
        x_old = np.linspace(0, 1, len(current_best))
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current_best)
        f_interp = np.maximum(f_interp, 0)
        if f_interp.sum() > 1e-10:
            f_interp = f_interp / f_interp.sum()

        t1 = t_alloc * 0.45
        t2 = t_alloc * 0.5
        a_opt, _ = fw_lp_optimize(f_interp, time_limit=t1)
        try_sequence(a_opt, best_tracker)

        if best_tracker[1] is not None and len(best_tracker[1]) == n_alt:
            a_opt, _ = lse_euclidean_optimize(best_tracker[1], time_limit=t2,
                                               beta_start=20.0, beta_end=5000.0,
                                               lr_start=0.5 / n_alt,
                                               top_k=max(n_alt // 3, 10))
            try_sequence(a_opt, best_tracker)

    # === Final: one more polish on the overall best ===
    t_remaining = deadline - time.monotonic()
    if t_remaining > 30 and best_tracker[1] is not None:
        print(f"\n=== Final Polish ({t_remaining:.0f}s) ===")
        final_f = best_tracker[1].copy()
        n_final = len(final_f)
        a_opt, _ = lse_euclidean_optimize(final_f, time_limit=min(t_remaining - 10, 60),
                                           beta_start=500.0, beta_end=50000.0,
                                           lr_start=0.0003,
                                           top_k=max(n_final // 5, 8),
                                           momentum_init=0.85)
        try_sequence(a_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
    else:
        final_seq = np.ones(448) / 448
        best_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
