import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(f):
    """Solve: min M s.t. (f*g)[k] <= M for all k, sum(g) = 1, g >= 0.
    Returns the optimal g."""
    n = len(f)
    # Variables: g[0], ..., g[n-1], M
    # Total: n+1 variables
    # Objective: minimize M -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    # Constraints:
    # (f*g)[k] - M <= 0 for k = 0, ..., 2n-2
    # A_ub has shape (2n-1) x (n+1)
    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        for j in range(n):
            if 0 <= k - j < n:
                A_ub[k, j] = f[j]
        A_ub[k, n] = -1.0  # -M coefficient
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g[i] >= 0, M is free (or M >= 0)
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2*n-1), A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x[:n]
    return None

import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)

        return np.array([max(float(v[0]), 0.0)])

    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def sg_polish(f0, time_limit, lr=0.001, top_k=50, temp=0.0001):
    """Subgradient polishing with very small steps."""
    f = project_simplex(f0.astype(np.float64))
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_M = M
    best_f = f.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp
    mom = 0.95
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break
        
        # Compute subgradient
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= wsum
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        # Update with momentum
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        M = conv.max()
        
        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9998
        cur_temp *= 1.001
        
        if no_improve > 300:
            rng = np.random.default_rng(it + 1000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            v = np.zeros(n)
            cur_lr = lr * 0.5
            no_improve = 0
    
    return best_f, best_M

    r = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(r / r.sum())
    
    # Cosine
    c = np.cos(np.pi * (x / max(n-1, 1) - 0.5)) ** 4 + 0.01
    inits.append(c / c.sum())
    
    # Gaussian
    g = np.exp(-0.5 * ((x - n/2) / (n/4)) ** 2)
    inits.append(g / g.sum())
    
    # Bimodal
    b = np.exp(-0.5 * ((x - n/4) / (n/8)) ** 2) + np.exp(-0.5 * ((x - 3*n/4) / (n/8)) ** 2)
    inits.append(b / b.sum())
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Power law
    p = (x + 1) ** 0.3
    inits.append(p / p.sum())
    
    return inits
