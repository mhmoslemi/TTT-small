def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_lse_grad(weights, f):
    """Compute gradient of log-sum-exp approximation."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective_norm(a):

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_adam_optimize(f0, time_limit, beta_start=20.0, beta_end=5000.0, lr_init=None):
    """Log-sum-exp smoothed optimization with Adam."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    if lr_init is None:
        lr = 2.0 / np.sqrt(n)
    else:
        lr = lr_init
    m = np.zeros(n)
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    b1, b2, eps = 0.8, 0.999, 1e-8

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_lse_grad(w, f)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            no_improve += 100
            break

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        m_hat = m / (1.0 - b1 ** (it + 1))
        v_hat = v / (1.0 - b2 ** (it + 1))

        f_new = project_simplex(f - lr * m_hat / (np.sqrt(v_hat) + eps))
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve == 30:
                lr *= 0.8
            elif no_improve == 60:
                lr *= 0.7
            elif no_improve > 300:
                f = best_f.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                lr = max(2.0 / np.sqrt(n), 0.001)
                beta = min(beta * 1.5, beta_end)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.004

        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam it={it}, obj={best_obj:.10f}, beta={beta:.0f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_f, best_obj


def lse_momentum_optimize(f0, time_limit, beta_start=20.0, beta_end=5000.0):
    """LSE with heavy ball momentum (alternative to Adam)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 3.0 / n
    momentum = 0.92
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_lse_grad(w, f)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            no_improve += 100
            break
        grad = grad / grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.001, 6.0 / n)
        else:
            no_improve += 1
            if no_improve == 20:
                lr *= 0.9
            elif no_improve > 200:
                f = best_f.copy()
                v = np.zeros(n)
                lr = 3.0 / n
                beta = min(beta * 1.3, beta_end)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.005

        if it % 2000 == 0:
            print(f"    LSE-Mom it={it}, obj={best_obj:.10f}, beta={beta:.0f}, lr={lr:.6f}")

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent: for each coord, try candidate values via 1D search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[i]
            s_others = 1.0 - old_val
            if s_others < 1e-12:
                continue

            # Generate candidates around current value
            candidates = [old_val]
            for frac in [0.1, 0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 0.0]:
                t_cand = old_val * frac if frac != 0.0 else 0.0
                if 0.0 <= t_cand < 0.99:
                    candidates.append(t_cand)
            # Random perturbations
            for _ in range(3):
                t_cand = max(0.0, old_val * (1.0 + rng.standard_normal() * 0.4))
                if t_cand < 0.99:
                    candidates.append(t_cand)

            best_t = old_val
            best_t_obj = best_obj

            for t in candidates:
                if t < 0.0 or t >= 0.99:
                    continue
                scale = (1.0 - t) / s_others
                new_f = f * scale
                new_f[i] = t
                conv_t = fast_convolve(new_f)
                obj_t = conv_t.max()
                if obj_t < best_t_obj:
                    best_t_obj = obj_t
                    best_t = t

            # If we found improvement, do a local refinement
            if best_t != old_val and best_t_obj < best_obj * (1.0 - 1e-8):
                # Fine search around best_t
                for _ in range(5):
                    t_fine = best_t * (1.0 + rng.standard_normal() * 0.05)
                    t_fine = max(0.0, min(t_fine, 0.99))
                    if t_fine == old_val:
                        continue
                    scale = (1.0 - t_fine) / s_others
                    new_f = f * scale
                    new_f[i] = t_fine
                    conv_t = fast_convolve(new_f)
                    obj_t = conv_t.max()
                    if obj_t < best_t_obj:
                        best_t_obj = obj_t
                        best_t = t_fine

                # Apply best
                scale = (1.0 - best_t) / s_others
                f = f * scale
                f[i] = best_t
                best_obj = best_t_obj
                best_f = f.copy()
                improved = True

        if not improved and pass_count > 1:
            break

    return best_f, best_obj


def subgradient_polish(f0, time_limit, top_k=32, temp_init=0.0001):
    """Subgradient with top-k peak weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.002
    cur_temp = temp_init
    v = np.zeros(n)
    momentum = 0.93
    no_improve = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        k = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k)[-k:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(k)
            w[np.argmax(pv)] = 1.0
        else:
            w = w / w_sum

        sg = np.zeros(n)
        for idx in range(k):
            if w[idx] < 1e-13:
                continue
            ki = int(peaks[idx])
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += 2.0 * w[idx] * f[ki - idx_arr]

        v = momentum * v - lr * sg
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.92
            if no_improve > 500:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                v = np.zeros(n)
                lr = 0.002
                no_improve = 0

        cur_temp *= 1.003

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate and track best."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def make_symmetric(f):
    """Average with reverse to exploit symmetry."""
    return 0.5 * (f + f[::-1])


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry: multi-phase optimization combining FW-LP, LSE-Adam,
    coordinate descent, subgradient polish, and multi-restart exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        n_default = 448
        parent = np.ones(n_default) / n_default
        try_sequence(parent, best_tracker)

    n = len(parent)

    # === Phase 1: FW-LP on parent ===
    print("\n=== Phase 1: FW-LP ===")
    t1 = min(180, deadline - time.monotonic() - 1200)
    if t1 > 25:
        print(f"  Time: {t1:.0f}s")
        a_opt, _ = fw_lp_optimize(parent, time_limit=t1)
        try_sequence(a_opt, best_tracker)

    # === Phase 2: LSE-Adam (main optimization) ===
    print("\n=== Phase 2: LSE-Adam ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t2 = min(400, deadline - time.monotonic() - 800)
    if t2 > 30:
        print(f"  Time: {t2:.0f}s")
        a_opt, _ = lse_adam_optimize(current, time_limit=t2, beta_start=10.0, beta_end=5000.0)
        try_sequence(a_opt, best_tracker)

    # === Phase 3: Symmetrization + LSE-Momentum ===
    print("\n=== Phase 3: Symmetry + LSE-Momentum ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    sym_f = make_symmetric(current)
    if sym_f.sum() > 1e-10:
        sym_f = sym_f / sym_f.sum()
    try_sequence(sym_f, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else parent
    t3 = min(250, deadline - time.monotonic() - 500)
    if t3 > 30:
        print(f"  Time: {t3:.0f}s")
        a_opt, _ = lse_momentum_optimize(current, time_limit=t3, beta_start=30.0, beta_end=8000.0)
        try_sequence(a_opt, best_tracker)

    # === Phase 4: Coordinate Descent ===
    print("\n=== Phase 4: Coordinate Descent ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t4 = min(300, deadline - time.monotonic() - 300)
    if t4 > 20:
        print(f"  Time: {t4:.0f}s")
        a_opt, _ = coordinate_descent(current, time_limit=t4)
        try_sequence(a_opt, best_tracker)

    # === Phase 5: Subgradient polish ===
    print("\n=== Phase 5: Subgradient Polish ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t5 = min(150, deadline - time.monotonic() - 150)
    if t5 > 15:
        print(f"  Time: {t5:.0f}s")
        a_opt, _ = subgradient_polish(current, time_limit=t5,
                                      top_k=max(16, n // 14), temp_init=0.00005)
        try_sequence(a_opt, best_tracker)

    # === Phase 6: High-beta LSE final ===
    print("\n=== Phase 6: High-beta LSE ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t6 = min(200, deadline - time.monotonic() - 60)
    if t6 > 20:
        print(f"  Time: {t6:.0f}s")
        a_opt, _ = lse_adam_optimize(current, time_limit=t6,
                                     beta_start=200.0, beta_end=20000.0,
                                     lr_init=0.5 / np.sqrt(n))
        try_sequence(a_opt, best_tracker)

    # === Phase 7: Multi-restart exploration ===
    print("\n=== Phase 7: Multi-Restart ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    best_score = best_tracker[0]

    restart_configs = [
        (0.002, 30, 4000),
        (0.003, 50, 6000),
        (0.005, 80, 8000),
        (0.001, 20, 3000),
        (0.004, 60, 5000),
        (0.006, 100, 10000),
        (0.003, 40, 5000),
        (0.002, 25, 3500),
    ]

    for ri, (noise, beta_s, beta_e) in enumerate(restart_configs):
        t_remaining = deadline - time.monotonic() - 20
        if t_remaining < 50:
            break
        t_alloc = min(100, t_remaining - 20)
        if t_alloc < 20:
            break

        # Create perturbation
        perturbed = project_simplex(current + rng.standard_normal(n) * noise)
        # Also try symmetry-broken version
        if ri % 2 == 1:
            perturbed = project_simplex(0.7 * current + 0.3 * rng.dirichlet(np.ones(n)))

        a_opt, _ = lse_adam_optimize(perturbed, time_limit=t_alloc,
                                     beta_start=float(beta_s), beta_end=float(beta_e),
                                     lr_init=1.5 / np.sqrt(n))
        score = try_sequence(a_opt, best_tracker)
        if score < best_score:
            best_score = score
            current = best_tracker[1].copy()

    # === Phase 8: Final CD on best ===
    print("\n=== Phase 8: Final CD ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_rem = deadline - time.monotonic() - 10
    if t_rem > 15:
        a_opt, _ = coordinate_descent(current, time_limit=min(t_rem, 60))
        try_sequence(a_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL SCORE: {final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("\nFALLBACK to parent")
    return [float(x) for x in parent]
