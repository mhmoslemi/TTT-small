"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Mirror descent with soft-peak subgradients (fast iterations)
2. LP-based direction finding with constraint reduction (deeper optimization)
3. Structured restarts (phase shifts, period doubling)
4. Adaptive step sizing and multi-phase time allocation

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse_info(a, k_list):
    """
    For a given set of lags k_list, build the constraint matrix rows.
    Returns C_sub where C_sub @ g gives (a*g)(k) for k in k_list.
    """
    n = len(a)
    C_sub = np.zeros((len(k_list), n))
    for ii, k in enumerate(k_list):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            idxs = np.arange(j_start, j_end + 1)
            C_sub[ii, idxs] = a[k - idxs]
    return C_sub


def soft_gradient(a, conv, top_k=20, temp=0.01):
    """Compute softmax-weighted subgradient of max autocorrelation."""
    n = len(a)
    size = len(conv)
    # Use top-k peaks with softmax weighting
    if top_k >= size:
        peaks = np.arange(size)
        peak_vals = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
    
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-15:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad


def mirror_descent_optimize(a0, time_limit, temp_init=0.01, top_k=20, eta_init=0.1):
    """
    Mirror descent with multiplicative updates and soft-peak subgradients.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = eta_init
    temp = temp_init
    no_improve = 0
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        grad = soft_gradient(a, conv, top_k=top_k, temp=temp)
        
        # Multiplicative update (mirror descent with entropy)
        # a_new_j = a_j * exp(-eta * grad_j) / Z
        log_updates = -eta * grad
        log_updates -= log_updates.max()  # numerical stability
        updates = np.exp(log_updates)
        a_new = a * updates
        s_new = a_new.sum()
        if s_new < 1e-30:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new
        
        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Slightly decrease eta on improvement
            eta *= 0.999
        else:
            no_improve += 1
            # Keep a at best if no improvement
            a = best_a.copy()
            # Decrease eta more aggressively
            eta *= 0.95
            temp *= 1.01
        
        if no_improve > 200:
            # Perturb and restart
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            noise = rng.standard_normal(n) * 0.01
            a = np.maximum(best_a + noise, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            conv = fast_convolve(a)
            no_improve = 0
            stall_count += 1
            eta = eta_init * 0.5
            if stall_count > 20:
                break

    return best_a, best_obj


def lp_direction(a, M, k_list=None, n_constraints=None):
    """
    Solve reduced LP: max sum(g) s.t. (a*g)(k) <= M for top constraints, g >= 0.
    """
    n = len(a)
    conv_ag = fast_convolve(a)  # This is a*a, used to identify important lags
    
    if k_list is None:
        # Use top constraints (where a*a is closest to M)
        if n_constraints is None:
            n_constraints = min(200, 2 * n - 1)
        # Priority: lags where (a*a) is high
        sorted_lags = np.argsort(-conv_ag)
        k_list = sorted_lags[:n_constraints]
    
    k_list = sorted(set(int(k) for k in k_list if 0 <= k < 2 * n - 1))
    C = build_conv_matrix_sparse_info(a, k_list)
    
    c_obj = -np.ones(n)
    b = np.full(len(k_list), M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 10})
        if result.success and result.fun < -1e-10:
            g = np.maximum(result.x, 0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return None, 0.0, False

def nesterov_optimize(a0, time_limit, lr_init=0.01, momentum=0.92, top_k=30, temp_init=0.005):
    """Nesterov-accelerated subgradient descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y

        conv_y = fast_convolve(y)
        grad = soft_gradient(y, conv_y, top_k=top_k, temp=temp)

        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9997
        temp *= 1.002
        mom = min(0.97, mom + 0.0001)

        if no_improve >= 300:
            break

    return best_a, best_obj


def fine_tune(a0, time_limit):
    """
    Very fine polishing: small random perturbations + gradient steps.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    while time.monotonic() - t0 < time_limit:
        # Try small random perturbations
        for scale in [0.001, 0.002, 0.0005]:
            if time.monotonic() - t0 > time_limit - 1:
                break
            for _ in range(100):
                if time.monotonic() - t0 > time_limit - 0.5:
                    break
                noise = rng.standard_normal(n) * scale
                a_try = np.maximum(best_a + noise, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
    
    return best_a, best_obj


def generate_structured_initializations(n, parent, rng):
    """Generate diverse initializations including structured ones."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.1)
    inits['random_medium'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Various sine waves
    for freq in [1, 2, 3, 5]:
        key = f'sine_{freq}'
        arr = (np.sin(np.pi * freq * x / n) ** 2 + 0.001)
        inits[key] = arr / arr.sum()
    
    # Cosine waves
    for freq in [1, 2, 3]:
        key = f'cosine_{freq}'
        arr = (np.cos(np.pi * freq * x / n) ** 2 + 0.001)
        inits[key] = arr / arr.sum()
    
    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Trimodal
    for center_frac in [0.25, 0.5]:
        key = f'trimodal_{int(center_frac*100)}'
        centers = [0, int(center_frac * n), n-1]
        arr = np.zeros(n)
        for c in centers:
            width = max(1, n // 8)
            start = max(0, c - width)
            end = min(n, c + width + 1)
            arr[start:end] = np.exp(-0.5 * ((np.arange(start, end) - c) / (width * 0.5)) ** 2)
        inits[key] = arr / arr.sum()
    
    # Alternating patterns
    alt1 = np.ones(n)
    alt1[::2] = 2.0
    alt1[1::2] = 0.5
    inits['alt_2'] = alt1 / alt1.sum()
    
    alt3 = np.ones(n)
    alt3[::3] = 2.5
    alt3[1::3] = 0.5
    alt3[2::3] = 0.5
    inits['alt_3'] = alt3 / alt3.sum()
    
    # Ramped
    ramp = x.astype(float) + 1
    inits['ramp'] = ramp / ramp.sum()
    
    # Parent-based
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
            
            # Phase-shifted parent
            for shift in [n // 4, n // 3, n // 2]:
                key = f'parent_shift_{shift}'
                ps_arr = np.roll(p / ps, shift)
                inits[key] = ps_arr
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral and structured initializations for diverse starting points
2. Nesterov-accelerated soft-subgradient descent as the fast workhorse
3. LP-based direction finding with adaptive constraint reduction for deep optimization
4. Multi-phase time allocation with hybrid LP/gradient polishing
5. Adaptive step sizing and structured restarts to escape local minima

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def soft_gradient(a, conv, top_k=30, temp=0.01):
    """Compute softmax-weighted subgradient of max autocorrelation."""
    n = len(a)
    size = len(conv)
    if top_k >= size:
        peaks = np.arange(size)
        peak_vals = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]

    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-15:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end < j_start:
            continue
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad


def nesterov_optimize(a0, time_limit, lr_init=0.01, momentum=0.92, top_k=30, temp_init=0.005):

    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0
    stall_rounds = 0

    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        else:
            y = a.copy()

        conv_y = fast_convolve(y)
        grad = soft_gradient(y, conv_y, top_k=top_k, temp=temp)

        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:

        temp *= 1.002

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = eta_init
    temp = temp_init
    no_improve = 0
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        grad = soft_gradient(a, conv, top_k=top_k, temp=temp)

        log_updates = -eta * grad
        log_updates -= log_updates.max()
        updates = np.exp(log_updates)
        a_new = a * updates
        s_new = a_new.sum()
        if s_new < 1e-30:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new

        conv_new = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv_new.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            no_improve = 0
            eta *= 0.998
        else:
            no_improve += 1
            eta *= 0.97
            temp *= 1.005

        if no_improve > 150:
            stall_count += 1
            if stall_count > 20:
                break
            rng = np.random.default_rng(stall_count * 7919)
            noise = rng.standard_normal(n) * 0.008
            a = np.maximum(best_a + noise, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            conv = fast_convolve(a)
            no_improve = 0
            eta = eta_init * 0.6

    return best_a, best_obj


def build_constraint_rows(a, k_list):
    """Build constraint matrix rows for given lags: row k gives coefficients for (a*g)(k)."""
    n = len(a)
    C_sub = np.zeros((len(k_list), n))
    for ii, k in enumerate(k_list):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            idxs = np.arange(j_start, j_end + 1)
            C_sub[ii, idxs] = a[k - idxs]
    return C_sub


def lp_direction(a, M, n_constraints=200, time_limit_lp=5.0):
    """
    Solve reduced LP: max sum(g) s.t. (a*g)(k) <= M for top constraints, g >= 0.
    Returns (g, g_sum, success).
    """
    n = len(a)
    conv_aa = fast_convolve(a)

    # Select most important constraints: top peaks and their neighbors
    if n_constraints >= 2 * n - 1:
        k_list = list(range(2 * n - 1))
    else:
        # Use top peaks plus neighbors
        top_peaks = np.argsort(-conv_aa)[:min(n_constraints // 2, 2 * n - 1)]
        k_set = set()
        for k in top_peaks:
            for offset in range(-3, 4):
                kk = int(k) + offset
                if 0 <= kk < 2 * n - 1:
                    k_set.add(kk)
        # Fill remaining with next highest peaks
        if len(k_set) < n_constraints:
            all_sorted = np.argsort(-conv_aa)
            for k in all_sorted:
                if len(k_set) >= n_constraints:
                    break
                k_set.add(int(k))
        k_list = sorted(list(k_set))[:n_constraints]

    C_sub = build_constraint_rows(a, k_list)
    c_obj = -np.ones(n)
    b_ub = np.full(len(k_list), M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=C_sub, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': time_limit_lp})
        if result.success and result.fun < -1e-8:
            g = np.maximum(result.x, 0)
            g_sum = g.sum()
            if g_sum > 1.0 + 1e-6:
                return g, g_sum, True
    except Exception:
        pass
    return None, 0.0, False


def lp_optimize(a0, time_limit, n_constraints=200, verbose=False):
    """LP-based iterative optimization with adaptive constraint count."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 40

    while time.monotonic() - t0 < time_limit:
        # Adaptive constraint count: start moderate, increase as we converge
        frac = min(1.0, iteration / 100.0)
        nc = int(n_constraints * (0.5 + 0.5 * frac))

        g, g_sum, success = lp_direction(a, M * 1.00001, n_constraints=nc)

        if success:
            d = g - a
            norm_d = np.linalg.norm(d)

            if norm_d > 1e-10:
                # Estimate optimal step from quadratic model at peak
                peak_idx = int(np.argmax(conv))
                # (a*(a+td))(peak) = (a*a)(peak) + t*(a*d)(peak)
                # We want this to stay <= M
                a_d_peak = 0.0
                j_start = max(0, peak_idx - n + 1)
                j_end = min(n - 1, peak_idx)
                if j_end >= j_start:
                    idxs = np.arange(j_start, j_end + 1)
                    a_d_peak = np.dot(a[idxs], a[peak_idx - idxs] - a[idxs] * 0)  # placeholder
                # Simpler: just use geometric
                best_a_try = a.copy()
                best_obj_try = best_obj
                best_t = 0.0

                candidates = [0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
                for t in candidates:
                    a_try = a + t * d
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    conv_try = fast_convolve(a_try)
                    M_try = conv_try.max()
                    obj_try = 2.0 * n * M_try
                    if obj_try < best_obj_try - 1e-12:
                        best_obj_try = obj_try
                        best_a_try = a_try.copy()
                        best_t = t
                        break

                if best_t > 0:
                    a = best_a_try
                    conv = fast_convolve(a)
                    M = conv.max()
                    if best_obj_try < best_obj - 1e-12:
                        best_obj = best_obj_try
                        best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        iteration += 1

        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iteration}, obj={best_obj:.8f}, t={elapsed:.1f}s")

        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 31 + 7 + no_improve)
            scale = 0.003 + 0.002 * min(no_improve // max_no_improve, 5)
            noise = rng.standard_normal(n) * scale
            a = np.maximum(best_a + noise, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_no_improve = min(200, 40 + 30 * (no_improve // 40 + 1))

            if iteration > 800:
                break

    return best_a, best_obj


def spectral_init(n, rng):
    """Generate sequence with flat-ish spectrum via random phase IFFT."""
    # Random phase, flat magnitude
    phase = rng.uniform(0, 2 * np.pi, n)
    spectrum = np.exp(1j * phase)
    # Make it real by conjugate symmetry
    if n % 2 == 0:
        spectrum[n // 2] = 1.0  # Nyquist must be real
    signal = np.fft.ifft(spectrum).real
    # Project to non-negative
    signal = np.maximum(signal, 0)
    s = signal.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return signal / s


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Spectral inits
    for si in range(3):
        inits[f'spectral_{si}'] = spectral_init(n, rng)

    # Sine waves
    for freq in [1, 2, 3]:
        arr = (np.sin(np.pi * freq * x / n) ** 2 + 0.001)
        inits[f'sine_{freq}'] = arr / arr.sum()

    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Alternating
    alt = np.ones(n)
    alt[::2] = 2.0
    alt[1::2] = 0.5
    inits['alt'] = alt / alt.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Parent-based
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                p_norm = p / ps
                inits['parent'] = p_norm.copy()
                # Phase shifts
                for shift in [n // 4, n // 3, n // 2, n // 6]:
                    inits[f'parent_shift_{shift}'] = np.roll(p_norm, shift)
                # Slight noise
                for ns in range(3):
                    noise = rng.standard_normal(n) * 0.01
                    pn = np.maximum(p_norm + noise, 0)
                    pns = pn.sum()
                    if pns > 1e-10:
                        inits[f'parent_noise_{ns}'] = pn / pns
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits


def fine_tune_micro(a0, time_limit):
    """Very fine micro-perturbation search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    # Try coordinate-type moves
    while time.monotonic() - t0 < time_limit:
        for scale in [0.0005, 0.001, 0.002, 0.003]:
            if time.monotonic() > deadline_est(t0, time_limit):
                break
            for _ in range(200):
                if time.monotonic() - t0 > time_limit - 0.5:
                    break
                # Random pair swap
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                a_try = best_a.copy()
                a_try[i] += scale
                a_try[j] -= scale
                if a_try[i] < 0 or a_try[j] < 0:
                    continue
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()

    return best_a, best_obj


def deadline_est(t0, time_limit):
    return t0 + time_limit


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization combining:
    - Spectral/structured initializations
    - Fast Nesterov and mirror descent for broad search
    - LP-based direction finding for deep local optimization
    - Hybrid polishing and micro fine-tuning
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    n = 768

    if parent is not None and len(parent) != n:
        # Interpolate parent to target length
        p_interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent / parent.sum())
        p_interp = np.maximum(p_interp, 0)
        p_interp = p_interp / p_interp.sum()
        parent = p_interp

    # =========================================================
    # Phase 1: Fast scan - Nesterov + Mirror Descent on multiple inits
    # =========================================================
    print("=== Phase 1: Fast Scan ===")
    t_start = time.monotonic()
    phase1_budget = 250.0

    inits = generate_initializations(n, parent, rng)

    # Priority ordering
    priority = ['parent', 'parent_noise_0', 'parent_noise_1', 'parent_shift_192',
                'parent_shift_256', 'spectral_0', 'spectral_1', 'sine_1', 'uniform', 'sine_2']
    init_names = [nm for nm in priority if nm in inits]
    init_names += [nm for nm in inits if nm not in priority]
    init_names = init_names[:8]

    t_per = (phase1_budget - 20) / max(1, len(init_names))
    scan_results = []

    for name in init_names:
        if time.monotonic() > deadline - 600:
            break
        t_alloc = min(t_per, deadline - time.monotonic() - 500)
        if t_alloc < 3:
            break

        a0 = inits[name]
        # Alternate between Nesterov and mirror descent
        if len(scan_results) % 2 == 0:
            a_opt, obj = nesterov_optimize(a0, time_limit=t_alloc,
                                           lr_init=0.012, momentum=0.92, top_k=40, temp_init=0.003)
        else:
            a_opt, obj = mirror_descent_optimize(a0, time_limit=t_alloc,
                                                 top_k=30, temp_init=0.006, eta_init=0.15)
        a_opt = np.maximum(a_opt, 0)
        a_opt = a_opt / a_opt.sum()
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            scan_results.append((score, name, a_opt.copy()))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: LP Deep Optimization on top candidates
    # =========================================================
    print("=== Phase 2: LP Deep Optimization ===")
    t_start2 = time.monotonic()

    lp_candidates = []
    seen_names = set()
    for score, name, a_opt in scan_results[:6]:
        if name not in seen_names:
            lp_candidates.append(a_opt.copy())
            seen_names.add(name)
        if len(lp_candidates) >= 3:
            break

    if best_sequence is not None:
        lp_candidates.append(best_sequence.copy())

    remaining = deadline - time.monotonic() - 400.0
    n_cands = max(1, len(lp_candidates))
    t_per_lp = max(50.0, remaining * 0.55 / n_cands)

    for i, a0 in enumerate(lp_candidates):
        if time.monotonic() > deadline - 350:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 300)
        if t_alloc < 15:
            break

        print(f"  LP cand {i}, t={t_alloc:.0f}s")
        a_opt, obj = lp_optimize(a0, time_limit=t_alloc, n_constraints=300, verbose=True)
        a_opt = np.maximum(a_opt, 0)
        s = a_opt.sum()
        if s > 1e-15:
            a_opt = a_opt / s
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP {i}: {score:.6f} *** NEW BEST ***")
            else:
                print(f"  LP {i}: {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Hybrid polishing - alternate LP and Nesterov
    # =========================================================
    print("=== Phase 3: Hybrid Polishing ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 150.0
        n_rounds = 5
        t_per_round = remaining / n_rounds

        for r in range(n_rounds):
            if time.monotonic() > deadline - 120:
                break
            t_alloc = min(t_per_round, deadline - time.monotonic() - 100)
            if t_alloc < 10:
                break

            if r % 2 == 0:
                # LP with increasing constraints
                nc = 250 + 50 * r
                a_opt, obj = lp_optimize(best_sequence, time_limit=t_alloc, n_constraints=nc)
            else:
                # Nesterov with tightening parameters
                lr = 0.004 * (1 - 0.15 * r)
                topk = 50 + 20 * r
                temp = 0.002 * (1 - 0.1 * r)
                a_opt, obj = nesterov_optimize(best_sequence, time_limit=t_alloc,
                                               lr_init=max(0.001, lr), momentum=0.94,
                                               top_k=topk, temp_init=max(0.0005, temp))

            a_opt = np.maximum(a_opt, 0)
            s = a_opt.sum()
            if s > 1e-15:
                a_opt = a_opt / s
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Hybrid {r}: {score:.6f} *** NEW BEST ***")
                else:
                    print(f"  Hybrid {r}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Final LP with many constraints
    # =========================================================
    print("=== Phase 4: Final LP Refinement ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 40.0
        if remaining > 30:
            t_alloc = min(remaining * 0.8, deadline - time.monotonic() - 20)
            if t_alloc > 20:
                # Use nearly full constraint set
                a_opt, obj = lp_optimize(best_sequence, time_limit=t_alloc,
                                         n_constraints=1000, verbose=True)
                a_opt = np.maximum(a_opt, 0)
                s = a_opt.sum()
                if s > 1e-15:
                    a_opt = a_opt / s
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score):
                    if score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Final LP: {score:.6f} *** NEW BEST ***")
                    else:
                        print(f"  Final LP: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Micro fine-tuning
    # =========================================================
    print("=== Phase 5: Micro Fine-tuning ===")
    if best_sequence is not None and time.monotonic() < deadline - 5:
        t_final = min(40.0, deadline - time.monotonic() - 2)
        if t_final > 5:
            a_opt, obj = fine_tune_micro(best_sequence, time_limit=t_final)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Micro: {score:.6f} *** NEW BEST ***")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(n) / n
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
