Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast Nesterov-accelerated 
subgradient descent with FFT-based convolution, adaptive peak weighting,
and coordinate refinement. Multi-start with diverse initializations.

import numpy as np
import time

def fft_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(f, conv, top_k, temp):
    """Compute weighted subgradient of max_k (f*f)[k]."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        weights = np.ones(len(peaks)) / len(peaks)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        if temp < 1e-15:
            # Hard max
            weights = np.zeros(len(peaks))
            weights[np.argmax(pv)] = 1.0
        else:
            scaled = (pv - pv.max()) / temp
            weights = np.exp(scaled)
            weights /= weights.sum()
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-12:
            continue
        k = int(peaks[i])
        # d/d f[m] (f*f)[k] = 2*f[k-m] for valid m
        # m ranges from max(0, k-n+1) to min(n-1, k)
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        # f[k-m] for m in [m_start, m_end]
        # k - m ranges from k - m_end to k - m_start
        val_start = k - m_end
        val_end = k - m_start
        sg[m_start:m_end+1] += 2.0 * weights[i] * f[val_start:val_end+1][::-1]
    
    return sg

                      nesterov_mom=0.9, lr_init=0.01, verbose=False):

    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    # Nesterov: maintain y (lookahead point) and momentum
    y = f.copy()
    v = np.zeros(n)  # velocity/momentum
    
    t0 = time.monotonic()
    best_obj = np.inf
    best_f = f.copy()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(777)

        it += 1
        
        # Adaptive parameters based on progress
        progress = min(1.0, it / 5000)
        top_k = int(top_k_init + (top_k_fine - top_k_init) * progress)
        top_k = max(1, top_k)
        temp = temp_init * (temp_fine / temp_init) ** progress
        lr = lr_init * (1.0 / (1.0 + 0.001 * it))
        
        # Compute convolution at y
        conv_y = fft_conv(y)
        
        # Subgradient at y
        sg = compute_subgradient(y, conv_y, top_k, temp)

        # Standard: v = mom * v - lr * sg; x = x + v; y = x + mom * v
        v = nesterov_mom * v - lr * sg

        x = project_simplex(x)
        y = x + nesterov_mom * (x - f)
        y = project_simplex(y)
        f = x

        conv_f = fft_conv(f)
        obj = conv_f.max()

            best_f = f.copy()

        else:

            y = f.copy()
            v = np.zeros(n)
            no_improve = 0
            # Bump temperature
            temp_init = max(temp * 0.5, 1e-5)

        order = rng.permutation(n)
        
        for m in order:
            if time.monotonic() - t0 > time_limit:
                break

                continue

            # The subgradient at peak: sg[m] = 2*f[peak_k - m] (if valid)

            # and increase coordinates with sg < 0 (but sg is always >= 0 for non-negative f)

            for _ in range(sweep_size):
                j = rng.integers(0, n)
                if j == m or f[j] < 1e-12:
                    continue
                
                # Try moving δ from m to j
                # Several trial values of δ
                for sign in [1, -1]:

                        if f[m] - sign * delta < 0 or f[j] + sign * delta < 0:
                            continue
                        # Wait, if sign=1, we move from m to j: f[m]-delta, f[j]+delta
                        # If sign=-1, we move from j to m: f[m]+delta, f[j]-delta
                        delta_actual = sign * frac * min(f[m], f[j])
                        
                        f_trial = f.copy()
                        f_trial[m] -= delta_actual
                        f_trial[j] += delta_actual
                        if f_trial[m] < 0 or f_trial[j] < 0:
                            continue
                        
                        conv_trial = fft_conv(f_trial)
                        obj_trial = conv_trial.max()
                        
                        if obj_trial < best_obj - 1e-14:
                            best_obj = obj_trial
                            best_f = f_trial.copy()
                            f = f_trial.copy()
                            conv = conv_trial.copy()
                            improved = True
        
        if not improved:
            # Try a perturbation
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            conv = fft_conv(f)
            if conv.max() < best_obj:
                best_obj = conv.max()
                best_f = f.copy()
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using ultra-fast Nesterov-accelerated
subgradient descent with FFT convolution, adaptive multi-peak temperature scheduling,
greedy peak-reduction refinement, and multi-start diverse initialization.
Key advantage over LP-based Frank-Wolfe: millions of O(n log n) iterations vs hundreds of LP solves.
"""

import numpy as np
import time


def fft_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(f, conv, top_k, temp):
    """Compute softmax-weighted subgradient of max_k (f*f)[k]."""
    n = len(f)
    num_lags = len(conv)
    if top_k >= num_lags:
        peaks = np.arange(num_lags)
        w = np.ones(len(peaks)) / len(peaks)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        if temp < 1e-14:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            s2 = (pv - pv.max()) / temp
            w = np.exp(s2)
            w /= w.sum()

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        ms = max(0, k - n + 1)
        me = min(n - 1, k)
        # f[k-m] for m in [ms, me] = f[k-me : k-ms+1] reversed
        sg[ms:me + 1] += 2.0 * w[i] * f[k - me:k - ms + 1][::-1]
    return sg


def nesterov_optimize(f0, time_limit, top_k_start=20, top_k_end=2,
                      temp_start=0.002, temp_end=1e-5,
                      lr_init=0.02, mom=0.88, verbose=False, seed=123):
    """Nesterov-accelerated subgradient descent with adaptive schedule."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    y = f.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    best_obj = np.inf
    best_f = f.copy()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(seed)
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / 8000)
        top_k = max(1, int(top_k_start + (top_k_end - top_k_start) * progress))
        temp = temp_start * (temp_end / temp_start) ** progress
        cur_lr = lr_init / (1.0 + 0.0003 * it)

        conv_y = fft_conv(y)
        sg = compute_subgrad(y, conv_y, top_k, temp)

        # Nesterov update
        v = mom * v - cur_lr * sg
        x = project_simplex(f + v)
        y = project_simplex(x + mom * (x - f))
        f = x

        conv_f = fft_conv(f)
        obj = conv_f.max()
        total_iters += 1

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 5000 == 0:
                print(f"    NS it={it} obj={best_obj:.10f} tk={top_k} t={temp:.8f}")
        else:
            no_improve += 1

        # Adaptive restart
        if no_improve > 3000 + 1000 * progress:
            scale = 0.002 + 0.003 * (1 - progress)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            y = f.copy()
            v = np.zeros(n)
            no_improve = 0

    if verbose:
        print(f"    NS done: {total_iters} iters, best={best_obj:.10f}")
    return best_f, best_obj

def greedy_peak_reduce(f0, time_limit, verbose=False, seed=789):
    """Greedy: identify peak lag, reduce most dangerous coordinate pair."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fft_conv(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    rounds = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        conv = fft_conv(f)
        peak_k = int(np.argmax(conv))

        # Compute products at peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        indices = np.arange(js, je + 1)
        prods = f[indices] * f[peak_k - indices]

        # Sort by danger
        order = np.argsort(prods)[::-1]
        top_danger = indices[order[:15]]

        # Find safe targets (low product at peak)
        all_prods = np.zeros(n)
        all_prods[indices] = prods
        safe_sorted = np.argsort(all_prods)
        safe_pool = safe_sorted[:10]
        safe_pool = [int(t) for t in safe_pool if f[t] > 1e-8 and t not in top_danger[:5]]

        if not safe_pool:
            safe_pool = [int(rng.integers(0, n))]

        improved = False
        for j in top_danger:
            if time.monotonic() - t0 > time_limit:
                break
            partner = peak_k - int(j)
            if partner < 0 or partner >= n:
                continue
            src = int(j) if f[int(j)] >= f[partner] else int(partner)

            target = safe_pool[rng.integers(0, len(safe_pool))]

            for frac in [0.015, 0.01, 0.025, 0.005, 0.03]:
                delta = frac * f[src]
                if delta < 1e-12:
                    continue
                f_trial = f.copy()
                f_trial[src] -= delta
                f_trial[target] += delta
                if f_trial[src] < 0:
                    continue
                conv_trial = fft_conv(f_trial)
                obj_trial = conv_trial.max()
                if obj_trial < best_obj - 1e-14:
                    best_obj = obj_trial
                    best_f = f_trial.copy()
                    f = f_trial.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Perturb and retry
            scale = 0.001 + 0.003 * (rounds % 5) / 5.0
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            c = fft_conv(f)
            if c.max() < best_obj:
                best_obj = c.max()
                best_f = f.copy()

    if verbose:
        print(f"    Greedy done: {rounds} rounds, best={best_obj:.10f}")
    return best_f, best_obj

def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = []
    inits.append(np.ones(n) / n)

    # Dirichlet with various concentrations
    for conc in [0.3, 0.7, 1.5, 3.0]:
        inits.append(rng.dirichlet(np.ones(n) * conc))

    x = np.arange(n, dtype=np.float64)
    # Gaussian bumps
    for sigma_frac in [0.1, 0.2, 0.35, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
        inits.append(g / g.sum())

    # Cosine
    c = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    inits.append(c / c.sum())

    # U-shape (edges heavy)
    u = 1.0 / (0.3 * np.abs(x - 0.5 * (n - 1)) / n + 0.15)
    inits.append(u / u.sum())

    # Linear ramp
    r = np.linspace(0.3, 1.0, n)
    inits.append(r / r.sum())

    # Two bumps
    for frac in [0.3, 0.4]:
        b = np.exp(-0.5 * ((x - frac * n) / (0.08 * n)) ** 2) + \
            np.exp(-0.5 * ((x - (1 - frac) * n) / (0.08 * n)) ** 2)
        inits.append(b / b.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

        opt, _ = greedy_peak_reduce(best_seq, time_limit=t4, verbose=True, seed=seed + 300)
        try_seq(opt, "greedy")

    # === Phase 5: Final fine Nesterov polish ===
    print("\n=== Phase 5: Final Fine Polish ===")
    remaining = deadline - time.monotonic() - 30
    t5 = min(remaining * 0.8, 120)

    if t5 > 5:
        opt, _ = nesterov_optimize(best_seq, time_limit=t5,
                                   top_k_start=3, top_k_end=1,
                                   temp_start=0.0002, temp_end=5e-7,
                                   lr_init=0.004, mom=0.95, verbose=True, seed=seed + 400)
        try_seq(opt, "final_polish")

        # One more very fine pass
        rem2 = deadline - time.monotonic() - 20
        if rem2 > 10:
            opt2, _ = nesterov_optimize(best_seq, time_limit=rem2 * 0.7,
                                        top_k_start=2, top_k_end=1,
                                        temp_start=0.0001, temp_end=2e-7,
                                        lr_init=0.002, mom=0.97,
                                        verbose=True, seed=seed + 500)
            try_seq(opt2, "ultra_fine")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
