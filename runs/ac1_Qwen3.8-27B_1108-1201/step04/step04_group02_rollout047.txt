def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    """Project onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)}) (was {old:.8f})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass
    
    # ============================================================
    # Phase 1: Parent optimization (if available)
    # ============================================================
    print("=== Phase 1: Parent Optimization ===")
    if parent is not None:
        n_p = len(parent)
        t_remaining = deadline - time.monotonic() - 300
        t1 = min(120, t_remaining)
        if t1 > 10:
            print(f"  Coordinate descent on parent (n={n_p}), t={t1:.0f}s")
            t_start = time.monotonic()
            
            f = parent.copy()
            conv = fast_convolve(f)
            obj = conv.max()
            
            # Run coordinate descent with time limit
            n_sweeps = 0
            while time.monotonic() - t_start < t1 - 2:
                f, conv, obj = coord_descent_pass(f, conv, ternary_iters=50)
                n_sweeps += 1
                # Periodic random sweep
                if n_sweeps % 10 == 0:
                    f, conv, obj = coord_descent_pass(f, conv, ternary_iters=50,
                                                      random_order=True, rng=rng)
                    n_sweeps += 1
            
            print(f"    {n_sweeps} sweeps, obj={obj:.8f}")
            try_seq(f, "parent_cd")
            
            # SG polish
            rem = deadline - time.monotonic() - 200
            if rem > 10:
                f_pol, _ = sg_polish(f, time_limit=min(rem, 60), lr=0.002,
                                      top_k=n_p // 6, temp=0.0005, coord_every=30)
                try_seq(f_pol, "parent_sg")
    
    # ============================================================
    # Phase 2: Multi-length exploration
    # ============================================================
    print("\n=== Phase 2: Multi-length Exploration ===")
    
    # Priority lengths
    lengths = [384, 512, 256, 768]
    if parent is not None:
        if len(parent) in lengths:
            lengths.remove(len(parent))
            lengths.insert(0, len(parent))
    
    # Budget allocation: prioritize 384 and 512
    time_per_length = {}
    total_exploration = min(500, deadline - time.monotonic() - 400)
    weights = {384: 0.30, 512: 0.25, 256: 0.15, 768: 0.20}
    # Adjust for parent
    if parent is not None and len(parent) in weights:
        weights[len(parent)] *= 0.5  # Already explored in Phase 1
    
    wsum = sum(weights.get(n, 0.1) for n in lengths)
    for n in lengths:
        time_per_length[n] = total_exploration * weights.get(n, 0.1) / wsum
    
    for n in lengths:
        if time.monotonic() > deadline - 350:
            break
        
        t_alloc = time_per_length.get(n, 50)
        t_start = time.monotonic()
        print(f"\n  Exploring n={n}, budget={t_alloc:.0f}s")
        
        inits = generate_inits(n, parent, rng)
        print(f"    {len(inits)} initializations")
        
        # Time per initialization
        t_per_init = max(5, t_alloc / len(inits) * 0.8)
        
        best_f_for_n = None
        best_obj_for_n = float('inf')
        
        for init_idx, f_init in enumerate(inits):
            if time.monotonic() - t_start > t_alloc - 5:
                break
            
            t_this = min(t_per_init, deadline - time.monotonic() - 300)
            if t_this < 3:
                break
            
            f = f_init.copy()
            conv = fast_convolve(f)
            obj = conv.max()
            
            # Coordinate descent
            n_sweeps = 0
            t_cd_start = time.monotonic()
            while time.monotonic() - t_cd_start < t_this - 1:
                f, conv, obj = coord_descent_pass(f, conv, ternary_iters=45)
                n_sweeps += 1
                if n_sweeps % 15 == 0:
                    f, conv, obj = coord_descent_pass(f, conv, ternary_iters=45,
                                                      random_order=True, rng=rng)
                    n_sweeps += 1
            
            if obj < best_obj_for_n:
                best_obj_for_n = obj
                best_f_for_n = f.copy()
            
            # Quick eval
            if init_idx % 3 == 0 or obj < best_score / (2 * n):
                score = try_seq(f, f"n{n}_i{init_idx}")
        
        # Deep coordinate descent on best for this n
        rem = min(t_alloc * 0.4, deadline - time.monotonic() - 250)
        if rem > 5 and best_f_for_n is not None:
            f = best_f_for_n.copy()
            conv = fast_convolve(f)
            t_deep_start = time.monotonic()
            n_sweeps = 0
            while time.monotonic() - t_deep_start < rem - 1:
                f, conv, obj = coord_descent_pass(f, conv, ternary_iters=55)
                n_sweeps += 1
                if n_sweeps % 8 == 0:
                    f, conv, obj = coord_descent_pass(f, conv, ternary_iters=55,
                                                      random_order=True, rng=rng)
                    n_sweeps += 1
            print(f"    Deep: {n_sweeps} sweeps, obj={conv.max():.8f}")
            try_seq(f, f"n{n}_deep")
    
    print(f"\n  Phase 2 best: {best_score:.8f}")
    
    # ============================================================
    # Phase 3: Deep optimization on global best
    # ============================================================
    print(f"\n=== Phase 3: Deep Optimization ===")
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Multiple deep coordinate descent runs with different perturbations
        for attempt in range(5):
            rem = deadline - time.monotonic() - 100
            if rem < 15:
                break
            
            t_alloc = min(rem // (5 - attempt), 150)
            
            if attempt == 0:
                f = best_seq.copy()
            elif attempt == 1:
                f = project_simplex(best_seq + rng.standard_normal(n_best) * 0.005)
            elif attempt == 2:
                f = project_simplex(0.9 * best_seq + 0.1 * (best_seq[::-1]))
            elif attempt == 3:
                f = project_simplex(best_seq + rng.standard_normal(n_best) * 0.02)
            else:
                f = project_simplex(np.maximum(best_seq - rng.random(n_best) * 0.01, 0))
            
            conv = fast_convolve(f)
            t_cd_start = time.monotonic()
            n_sweeps = 0
            while time.monotonic() - t_cd_start < t_alloc - 2:
                f, conv, obj = coord_descent_pass(f, conv, ternary_iters=60)
                n_sweeps += 1
                if n_sweeps % 5 == 0:
                    f, conv, obj = coord_descent_pass(f, conv, ternary_iters=60,
                                                      random_order=True, rng=rng)
                    n_sweeps += 1
            
            print(f"  Deep attempt {attempt}: {n_sweeps} sweeps, obj={conv.max():.8f}")
            try_seq(f, f"deep_{attempt}")
        
        # SG + CD alternating polish
        rem = deadline - time.monotonic() - 60
        if rem > 10:
            f, _ = sg_polish(best_seq, time_limit=min(rem, 80), lr=0.001,
                             mom=0.95, top_k=max(32, n_best // 5), temp=0.0003,
                             coord_every=25, ternary_iters=50)
            try_seq(f, "sg_cd_polish")
    
    # ============================================================
    # Phase 4: Final micro-optimization
    # ============================================================
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"  Current best: {best_score:.8f}")
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Very fine coordinate descent (more ternary iterations)
        rem = deadline - time.monotonic() - 20
        if rem > 5:
            f = best_seq.copy()
            conv = fast_convolve(f)
            t_start = time.monotonic()
            n_sweeps = 0
            while time.monotonic() - t_start < rem - 2:
                f, conv, obj = coord_descent_pass(f, conv, ternary_iters=80)
                n_sweeps += 1
                # Occasionally try random order
                if n_sweeps % 3 == 0:
                    f, conv, obj = coord_descent_pass(f, conv, ternary_iters=80,
                                                      random_order=True, rng=rng)
                    n_sweeps += 1
            print(f"  Final CD: {n_sweeps} sweeps, obj={conv.max():.8f}")
            try_seq(f, "final_cd")
        
        # Micro perturbations + polish
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            for _ in range(4):
                rem = deadline - time.monotonic() - 5
                if rem < 3:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
                conv_pert = fast_convolve(f_pert)
                t_micro = min(rem, 8)
                t_start = time.monotonic()
                while time.monotonic() - t_start < t_micro - 0.5:
                    f_pert, conv_pert, _ = coord_descent_pass(f_pert, conv_pert, ternary_iters=60)
                try_seq(f_pert, "micro")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
