def mirror_descent(a0, time_limit, lr_init=0.1, temp=0.01, top_k=50):
    """Mirror descent with entropy regularization on the simplex."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv = fast_convolve(a)
    best_peak = best_conv.max()
    best_obj = 2.0 * n * best_peak  # sum is 1
    
    lr = lr_init
    no_improve = 0
    
    log_a = np.log(a)
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        peak = conv.max()

        if top_k < len(conv):
            peaks_idx = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks_idx]
            scaled = peak_vals / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w /= w.sum()
        else:
            peaks_idx = np.arange(len(conv))
            w = np.ones(len(conv)) / len(conv)
        
        # Subgradient: weighted average of 2*a[k-m] for each peak k
        grad = np.zeros(n)
        for i, k in enumerate(peaks_idx):
            if w[i] < 1e-15:
                continue
            k = int(k)
            # grad[m] += w[i] * 2 * a[k-m] for valid m
            # k-m valid when 0 <= k-m < n, i.e., max(0, k-n+1) <= m <= k
            m_start = max(0, k - n + 1)
            m_end = min(n - 1, k)
            if m_start <= m_end:
                grad[m_start:m_end+1] += w[i] * 2.0 * a[k - np.arange(m_start, m_end+1)]
        
        # Mirror descent update: a_new[m] = a[m] * exp(-lr * grad[m]) / Z
        log_a_new = log_a - lr * grad
        log_a_new -= log_a_new.max()  # numerical stability
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        log_a = np.log(np.maximum(a_new, 1e-300))
        a = a_new

            best_peak = peak
            best_obj = 2.0 * n * peak
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Decay learning rate
        lr *= 0.999
        temp *= 1.01
        
        if no_improve > 500:
            break
    
    return best_a, best_obj

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution a * b via FFT."""
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C

def lp_solve_direction(a, C, M):

    best_t = 0.0
    best_pk = current_peak
    best_a = a.copy()

    # Fine geometric + adaptive grid
    candidates = [0.4, 0.3, 0.25, 0.2, 0.15, 0.12, 0.1, 0.08, 0.06, 0.05, 0.04, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]

    for t in candidates:
        a_try = a + t * d
        if np.any(a_try < -1e-12):
            continue
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        pk = fast_convolve(a_try).max()
        if pk < best_pk - 1e-13:
            best_pk = pk
            best_t = t
            best_a = a_try.copy()

    # Refine around best_t
    if best_t > 0:
        for dt in [0.5 * best_t, 1.5 * best_t, 0.7 * best_t, 1.3 * best_t]:
            a_try = a + dt * d
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            pk = fast_convolve(a_try).max()
            if pk < best_pk - 1e-13:
                best_pk = pk
                best_t = dt
                best_a = a_try.copy()

    return best_t, best_pk, best_a


def lp_optimize(a0, time_limit, verbose=False):
    """Iterative LP-based optimization with smart restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    C = build_conv_matrix(a)

    t0 = time.monotonic()
    no_improve = 0
    perturb_count = 0
    max_no_improve = 40
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        # Solve LP
        g, success = lp_solve_direction(a, C, M)

        if success:
            d = g
            t, new_pk, a_new = line_search(a, d, M, n)

            if t > 0:
                a = a_new
                M = new_pk
                C = build_conv_matrix(a)
                cur_obj = 2.0 * n * M
                if cur_obj < best_obj - 1e-13:
                    best_obj = cur_obj
                    best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        it_count += 1
        if verbose and it_count % 200 == 0:
            print(f"    LP iter {it_count}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s, perturbs={perturb_count}")

        # Stagnation handling
        if no_improve >= max_no_improve:
            rng = np.random.default_rng(it_count * 31 + perturb_count * 7 + 13)
            # Adaptive perturbation scale
            scale = 0.001 + 0.002 * np.sqrt(min(perturb_count, 20))
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
            conv = fast_convolve(a)
            M = conv.max()
            C = build_conv_matrix(a)
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(300, 40 + 25 * perturb_count)

            if perturb_count > 50:
                break

    return best_a, best_obj


def gradient_polish(a0, time_limit, lr_init=0.008, top_k=50, temp_init=0.002):
    """Subgradient polishing with Nesterov momentum."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    t0 = time.monotonic()
    best_a = a.copy()
    conv = fast_convolve(a)
    best_peak = conv.max()
    best_obj = 2.0 * n * best_peak

    v = np.zeros(n)
    y = a.copy()
    mom = 0.9
    lr = lr_init
    temp = temp_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        y = np.maximum(a + mom * v, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy

        conv_y = fast_convolve(y)
        if top_k < len(conv_y):
            peaks_idx = np.argpartition(conv_y, -top_k)[-top_k:]
            peak_vals = conv_y[peaks_idx]
            scaled = peak_vals / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            ws = w.sum()
            if ws > 1e-15:
                w /= ws
            else:
                w = np.zeros(len(peaks_idx))
                w[np.argmax(peak_vals)] = 1.0
        else:
            peaks_idx = np.arange(len(conv_y))
            w = np.ones(len(conv_y)) / len(conv_y)

        grad = np.zeros(n)
        for i in range(len(peaks_idx)):
            if w[i] < 1e-14:
                continue
            k = int(peaks_idx[i])
            m_start = max(0, k - n + 1)
            m_end = min(n - 1, k)
            if m_start <= m_end:
                idxs = np.arange(m_start, m_end + 1)
                grad[idxs] += w[i] * 2.0 * y[k - idxs]

        v = mom * v - lr * grad
        a_new = np.maximum(a + v, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new

        conv = fast_convolve(a)
        pk = conv.max()
        if pk < best_peak - 1e-13:
            best_peak = pk
            best_obj = 2.0 * n * pk
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        temp *= 1.005
        mom = min(0.96, mom + 0.0003)

        if no_improve > 200:
            break

    return best_a, best_obj


def generate_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(0)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5 + 1e-3)
    inits['sine'] = (np.sin(np.pi * x / n) ** 2 + 0.05)
    inits['sine'] /= inits['sine'].sum()
    inits['sine2'] = (np.sin(2 * np.pi * x / n) ** 2 + 0.05)
    inits['sine2'] /= inits['sine2'].sum()
    inits['cos'] = (np.cos(2 * np.pi * x / n) ** 2 + 0.05)
    inits['cos'] /= inits['cos'].sum()

    # Bimodal
    edge = max(1, n // 6)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi[edge:-edge] = 0.3
    inits['bimodal'] = bi / bi.sum()

    # U-shaped
    u = 1.0 / (0.1 + np.abs(x - 0.5 * (n - 1)) / (0.5 * (n - 1)))
    inits['ushape'] = u / u.sum()

    # Power law
    pw = (x + 1) ** 0.5
    inits['power'] = pw / pw.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pp /= pp.sum()
                inits['parent_interp'] = pp

    return inits


def extend_sequence(a_short, n_target):
    """Interpolate a shorter sequence to target length."""
    n = len(a_short)
    s = a_short.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a_short / s
    a_ext = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, n), a_norm)
    a_ext = np.maximum(a_ext, 0)
    ses = a_ext.sum()
    if ses > 1e-15:
        a_ext = a_ext / ses
    return a_ext


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: multi-phase LP + mirror descent optimization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}")
    except Exception:
        pass

    def try_sequence(a):
        nonlocal best_sequence, best_score
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  NEW BEST: {score:.8f} (n={len(a)})")
        return score

    # ================================================================
    # Phase 1: Fast exploration with mirror descent (250s)
    # ================================================================
    print("=== Phase 1: Mirror Descent Exploration ===")
    t_phase = time.monotonic()
    p1_budget = 250.0

    n_candidates = [640, 768, 896, 1024]
    scan_results = []

    for n in n_candidates:
        if time.monotonic() > deadline - 900:
            break
        inits = generate_inits(n, parent, rng)
        priority = ['parent', 'parent_interp', 'uniform', 'sine', 'random', 'bimodal', 'ushape']
        names = [nm for nm in priority if nm in inits][:3]
        if not names:
            names = list(inits.keys())[:3]

        t_per = max(10.0, (p1_budget - 30) / (len(n_candidates) * 3))

        for name in names:
            if time.monotonic() > deadline - 800:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 700)
            if t_alloc < 5:
                break
            a0 = inits[name]
            a_opt, obj = mirror_descent_optimize(a0, time_limit=t_alloc, lr_init=0.04, top_k=40)
            score = try_sequence(a_opt)
            scan_results.append((score, n, a_opt.copy()))

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_phase:.0f}s). Best: {best_score:.6f}")

    # ================================================================
    # Phase 2: LP deep optimization (800s)
    # ================================================================
    print("=== Phase 2: LP Deep Optimization ===")
    t_phase = time.monotonic()

    # Take top candidates from phase 1 + parent
    top_starts = []
    seen_n = set()
    for score, n, a in scan_results:
        if n not in seen_n and score < best_score + 0.1:
            top_starts.append((n, a.copy()))
            seen_n.add(n)
        if len(top_starts) >= 3:
            break

    # Always include best
    if best_sequence is not None:
        bn = len(best_sequence)
        if bn not in seen_n:
            top_starts.append((bn, best_sequence.copy()))
            seen_n.add(bn)

    if not top_starts:
        top_starts = [(768, np.ones(768) / 768)]

    remaining = deadline - time.monotonic() - 350.0
    n_starts = len(top_starts)
    t_per_lp = max(50.0, remaining * 0.65 / max(1, n_starts))

    for i, (n, a0) in enumerate(top_starts):
        if time.monotonic() > deadline - 300:
            break
        t_alloc = min(t_per_lb if (t_per_lb := t_per_lp) else 50, deadline - time.monotonic() - 250)
        if t_alloc < 20:
            break
        print(f"  LP start {i}: n={n}, time={t_alloc:.0f}s")
        a_opt, obj = lp_optimize(a0, time_limit=t_alloc, verbose=True)
        try_sequence(a_opt)

    print(f"Phase 2 done ({time.monotonic()-t_phase:.0f}s). Best: {best_score:.6f}")

    # ================================================================
    # Phase 3: Multi-restart LP polish (400s)
    # ================================================================
    print("=== Phase 3: Multi-Restart Polish ===")
    t_phase = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 150.0
        n_restarts = 5
        t_per_restart = max(10.0, remaining / n_restarts * 0.85)

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 130:
                break
            t_alloc = min(t_per_restart, deadline - time.monotonic() - 120)
            if t_alloc < 10:
                break

            if ri == 0:
                a0 = best_sequence.copy()
            else:
                # Small perturbation based on the structure
                scale = 0.0005 * (ri + 1)
                noise = rng.standard_normal(len(best_sequence)) * scale
                a0 = np.maximum(best_sequence + noise, 0)
                a0 = a0 / a0.sum()

            a_opt, obj = lp_optimize(a0, time_limit=t_alloc)
            try_sequence(a_opt)

            # Also try gradient polish
            if time.monotonic() < deadline - 100:
                t_grad = min(5.0, deadline - time.monotonic() - 100)
                if t_grad > 2:
                    a_g, obj_g = gradient_polish(a_opt, time_limit=t_grad, lr_init=0.004, top_k=80, temp_init=0.001)
                    try_sequence(a_g)

    # Also try extending to slightly different n
    if best_sequence is not None and time.monotonic() < deadline - 100:
        n_best = len(best_sequence)
        for n_try in [n_best + 128, n_best - 128, n_best + 256]:
            if n_try < 128 or n_try > 4096:
                continue
            if time.monotonic() > deadline - 80:
                break
            a_ext = extend_sequence(best_sequence, n_try)
            a_opt, obj = lp_optimize(a_ext, time_limit=min(30.0, deadline - time.monotonic() - 70))
            try_sequence(a_opt)

    print(f"Phase 3 done ({time.monotonic()-t_phase:.0f}s). Best: {best_score:.6f}")

    # ================================================================
    # Phase 4: Final gradient + LP micro-polish (100s)
    # ================================================================
    print("=== Phase 4: Final Polish ===")
    t_phase = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0

        # Gradient polish
        if remaining > 20:
            t_g = min(remaining * 0.4, 30.0)
            a_g, obj_g = gradient_polish(best_sequence, time_limit=t_g, lr_init=0.003, top_k=100, temp_init=0.0008)
            try_sequence(a_g)

        # Short LP
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 10:
            a_lp, obj_lp = lp_optimize(best_sequence, time_limit=min(remaining * 0.7, 40.0))
            try_sequence(a_lp)

    print(f"Phase 4 done ({time.monotonic()-t_phase:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(768) / 768

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
