"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-shape initialization (Hann, Gaussian, uniform, etc.)
2. Exact coordinate descent with vectorized 1D subproblem solving
3. Simulated annealing with structured moves (smooth, swap, shift)
4. Log-sum-exp gradient polishing
5. Final coordinate descent refinement
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def coord_descent_sweep(f, n_sweeps=1, rng=None, randomize=False):
    """Exact coordinate descent: for each coordinate, solve the 1D subproblem."""
    n = len(f)
    f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max() / f.sum()**2
    best_f = f.copy()
    
    for sweep in range(n_sweeps):
        if rng is not None and randomize:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        
        for i in order:
            x_old = f[i]
            S_other = f.sum() - x_old
            
            if S_other + x_old < 1e-15:
                continue
                
            # Compute c_zero (auto-conv with f[i] set to 0)
            f_zero = f.copy()
            f_zero[i] = 0.0
            c_zero = fast_convolve(f_zero)
            
            # g[k] = coefficient of x in c_k(x) for linear terms
            # c_k(x) = c_zero[k] + g[k]*x + x^2*[k==2i]
            g = np.zeros(2 * n - 1)
            k_start = i
            k_end = min(i + n, 2 * n - 1)
            if k_start < k_end:
                idx = np.arange(k_start, k_end)
                g[idx] = 2.0 * f[idx - i]  # f[k-i] for k in [i, i+n-1]
            # For k = 2i, the quadratic term x^2 replaces the linear term 2*x*f[i]
            # But f[i] = x_old, and we're solving for new x
            # The actual contribution at k=2i is x^2 (not 2*x*f[i])
            # So we need: c[2i] = c_zero[2i] + x^2
            # In our formula: c[2i] = c_zero[2i] + g[2i]*x + (extra x^2)
            # We set g[2i] = 0 and add x^2 separately
            
            two_i = 2 * i
            has_quad = (two_i < 2 * n - 1)
            if has_quad:
                g[two_i] = 0.0  # will add x^2 separately
            
            # 1D minimization of: 2n * max_k (c_zero[k] + g[k]*x + x^2*[k==2i]) / (S_other + x)^2
            # x >= 0
            
            # Grid search
            x_hi = max(2.0 * x_old + 0.01, S_other / n * 4)
            x_grid = np.linspace(0.0, x_hi, 100)
            
            # Vectorized: c_matrix[row, k] = c_zero[k] + g[k]*x_grid[row]
            c_matrix = c_zero[np.newaxis, :] + np.outer(x_grid, g)
            if has_quad:
                c_matrix[:, two_i] = c_zero[two_i] + x_grid**2
            
            max_c = c_matrix.max(axis=1)
            # Ensure non-negative (should be, but numerical issues)
            max_c = np.maximum(max_c, 0)
            denom = (S_other + x_grid)**2
            obj = 2.0 * n * max_c / denom
            best_idx = obj.argmin()
            best_x = x_grid[best_idx]
            
            # Refine with golden section
            gr = (np.sqrt(5) - 1) / 2  # golden ratio
            lo = max(0.0, x_grid[max(0, best_idx-1)])
            hi = x_grid[min(len(x_grid)-1, best_idx+1)]
            
            for _ in range(20):
                if hi - lo < 1e-12 * max(1, hi):
                    break
                m1 = hi - gr * (hi - lo)
                m2 = lo + gr * (hi - lo)
                
                c1 = c_zero + g * m1
                if has_quad:
                    c1[two_i] = c_zero[two_i] + m1**2
                obj1 = 2.0 * n * max(c1.max(), 0) / (S_other + m1)**2
                
                c2 = c_zero + g * m2
                if has_quad:
                    c2[two_i] = c_zero[two_i] + m2**2
                obj2 = 2.0 * n * max(c2.max(), 0) / (S_other + m2)**2
                
                if obj1 < obj2:
                    hi = m2
                else:
                    lo = m1
            
            best_x = (lo + hi) / 2.0
            
            if abs(best_x - x_old) > 1e-12:
                f[i] = max(0.0, best_x)
        
        # End of sweep
        obj = 2.0 * n * fast_convolve(f).max() / f.sum()**2
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
    
    return best_f, best_obj

- c_{2i} changes by: (x_new^2 - x_old^2) = delta*(x_new + x_old)

So I can update c_full in O(n) after each coordinate change. That's much cheaper than an FFT.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-shape diverse initialization (Hann, Gaussian, uniform, parent)
2. Fast exact coordinate descent with incremental convolution updates
3. Diversification via structured perturbations (blur, shift, symmetry)

5. Log-sum-exp gradient polishing
6. Final coordinate descent refinement

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """Compute LSE gradient via FFT."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def coord_descent(f0, time_limit, n_sweeps_max=5000, random_order=True, seed=0):
    """
    Exact coordinate descent with incremental convolution updates.
    For each coordinate i, solves the 1D subproblem exactly.
    """
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
        s = 1.0
    else:
        f = f / s
        s = 1.0

    c_full = fast_convolve(f)
    best_obj = 2.0 * n * c_full.max() / s
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5) - 1) / 2
    no_improve = 0

    sweep = 0
    while time.monotonic() - t0 < time_limit and sweep < n_sweeps_max:
        sweep += 1
        if random_order:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        s_total = f.sum()

        for i in order:
            if time.monotonic() - t0 > time_limit:

            x_old = f[i]
            S_other = s_total - x_old

            if S_other < 1e-14 and x_old < 1e-14:
                continue

            two_i = 2 * i

            max_c = np.maximum(max_c, 0.0)

            obj_grid = 2.0 * n * max_c / denom
            best_idx = obj_grid.argmin()
            best_x = x_grid[best_idx]

            # Golden section refinement
            lo = x_grid[max(0, best_idx - 1)]
            hi = x_grid[min(n_grid - 1, best_idx + 1)]

            for _ in range(14):
                if hi - lo < 1e-10:
                    break
                m1 = hi - gr * (hi - lo)
                m2 = lo + gr * (hi - lo)

                c1 = r + g_full * m1
                if two_i < 2 * n - 1:
                    c1[two_i] = r[two_i] + m1 * m1
                obj1 = 2.0 * n * max(c1.max(), 0.0) / (S_other + m1) ** 2

                    hi = m2
                else:
                    lo = m1

            new_x = max(0.0, (lo + hi) / 2.0)

                f[i] = new_x

                    c_full[two_i] -= delta * delta

def sa_optimize(f0, time_limit, seed=0):
    """

    - Local box blur + sharpen
    - Pair exchange
    - Block cyclic shift
    - Random Gaussian perturbation
    - Symmetrization
    """
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    c_full = fast_convolve(f)
    best_obj = 2.0 * n * c_full.max() / f.sum() ** 2
    best_f = f.copy()
    cur_obj = best_obj

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    temp = 0.005
    temp_min = 0.0001
    cooling = 0.9997
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        f_try = f.copy()
        move_type = rng.integers(0, 5)

        if move_type == 0:
            # Local box blur
            radius = rng.integers(2, 6)
            kernel = np.ones(2 * radius + 1) / (2 * radius + 1)
            f_try = np.convolve(f, kernel, mode='same')
            f_try = np.maximum(f_try, 0.0)
        elif move_type == 1:
            # Pair exchange
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = (f[j] - f[i]) * rng.uniform(0.3, 1.0)
                f_try[i] = max(0.0, f[i] + delta)
                f_try[j] = max(0.0, f[j] - delta)
        elif move_type == 2:
            # Block shift
            block_len = rng.integers(max(2, n // 20), max(3, n // 5))
            start = rng.integers(0, max(1, n - block_len))
            shift = rng.integers(-block_len // 2, block_len // 2 + 1)
            if shift != 0:
                f_try[start:start + block_len] = np.roll(
                    f[start:start + block_len], shift)
        elif move_type == 3:
            # Gaussian perturbation
            scale = temp * 0.5
            f_try = f + rng.standard_normal(n) * scale
            f_try = np.maximum(f_try, 0.0)
        else:
            # Symmetrize (partial)
            alpha = rng.uniform(0.1, 0.5)
            f_rev = f[::-1].copy()
            f_try = (1 - alpha) * f + alpha * f_rev

        s_try = f_try.sum()
        if s_try < 1e-14:
            continue
        f_try = f_try / s_try

        c_try = fast_convolve(f_try)
        obj_try = 2.0 * n * c_try.max() / f_try.sum() ** 2

        if obj_try < cur_obj:
            f = f_try
            cur_obj = obj_try
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_f = f_try.copy()
                no_improve = 0
            no_improve = 0
        else:
            delta_obj = obj_try - cur_obj
            if delta_obj < temp * 0.1:
                accept = np.exp(-delta_obj / (temp + 1e-15))
            else:
                accept = 0.0
            if rng.random() < accept:
                f = f_try
                cur_obj = obj_try
            no_improve += 1

        temp = max(temp_min, temp * cooling)
        if no_improve > 200:
            temp = max(temp * 2.0, 0.002)
            no_improve = 0

        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA it={it}, best={best_obj:.10f}, cur={cur_obj:.10f}, "
                  f"temp={temp:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def lse_optimize(f0, time_limit, beta_start=20.0, beta_end=5000.0, seed=0):
    """Log-sum-exp smoothed gradient descent with adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max() / f.sum() ** 2
    best_f = f.copy()

    beta = beta_start
    lr = 3.0 / n
    momentum = 0.90
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Softmax weights
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_lse_grad(w, f)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad = grad / grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = 2.0 * n * conv_new.max() / f_new.sum() ** 2

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 6.0 / n)
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.93
                momentum = max(0.85, momentum - 0.003)
            if no_improve > 100:
                f = best_f + rng.standard_normal(n) * 0.001
                f = project_simplex(f)
                v = np.zeros(n)
                lr = max(3.0 / n, 0.002)
                beta = min(beta * 1.5, beta_end)
                momentum = 0.88
                no_improve = 0

        if beta < beta_end:
            beta *= 1.004

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it}, best={best_obj:.10f}, beta={beta:.1f}, "
                  f"lr={lr:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def build_diverse_initializations(n, parent=None, rng=None):
    """Generate diverse initial sequences of length n."""
    if rng is None:
        rng = np.random.default_rng(123)
    candidates = []

    # Uniform
    candidates.append(np.ones(n) / n)

    # Hann window
    j = np.arange(n)
    hann = 0.5 * (1 + np.cos(2 * np.pi * j / (n - 1)))
    hann = np.maximum(hann, 1e-6)
    candidates.append(hann / hann.sum())

    # Raised cosine (different bandwidth)
    rc = 0.5 * (1 + np.cos(2 * np.pi * j / (n * 1.2 - 1)))
    rc = np.maximum(rc, 1e-6)
    candidates.append(rc / rc.sum())

    # Gaussian
    center = n / 2 - 0.5
    for sigma_frac in [0.3, 0.4, 0.5, 0.6]:
        g = np.exp(-0.5 * ((j - center) / (n * sigma_frac)) ** 2)
        candidates.append(g / g.sum())

    # Sine window
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        sine = np.sin(np.pi * (j + 0.5) / n) ** p
        sine = np.maximum(sine, 1e-8)
        candidates.append(sine / sine.sum())

    # Super-Gaussian
    for p in [2, 4, 8, 16]:
        x = (j - center) / (n * 0.4)
        sg = np.exp(-np.abs(x) ** p)
        candidates.append(sg / sg.sum())

    # Triangular
    tri = np.minimum(j + 1, n - j)
    tri = np.maximum(tri, 1.0)
    candidates.append(tri / tri.sum())

    # Bimodal
    for sep in [0.2, 0.3, 0.4]:
        c1 = n * (0.5 - sep / 2)
        c2 = n * (0.5 + sep / 2)
        bim = np.exp(-0.5 * ((j - c1) / (n * 0.15)) ** 2) + \
              np.exp(-0.5 * ((j - c2) / (n * 0.15)) ** 2)
        candidates.append(bim / bim.sum())

    # Parent (resized)
    if parent is not None:
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        p_interp = np.interp(x_new, x_old, parent)
        p_interp = np.maximum(p_interp, 0)
        if p_interp.sum() > 1e-10:
            candidates.append(p_interp / p_interp.sum())

    # Random smooth
    for _ in range(5):
        r = rng.standard_normal(n)
        kernel = np.exp(-0.5 * (np.arange(-10, 11) / 3) ** 2)
        r = np.convolve(r, kernel / kernel.sum(), mode='same')
        r = np.maximum(r, 0)
        if r.sum() > 1e-10:
            candidates.append(r / r.sum())

    # Remove duplicates (approximately)
    unique = [candidates[0]]
    for c in candidates[1:]:
        if min(np.linalg.norm(c - u) for u in unique) > 0.01:
            unique.append(c)

    return unique


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    if a.sum() < 0.01:
        return float('inf')
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def fw_lp_step(f, M, timeout=15):
    """One FW step with LP oracle."""
    n = len(f)
    # Build constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': timeout})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-14:
                return g / s
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, seed=0):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    m_exp = 1.0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        g = fw_lp_step(f, M * m_exp, timeout=10)
        if g is None:
            m_exp *= 1.15
            if m_exp > 8.0:
                m_exp = 1.0
            continue

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 4.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = 2.0 * n * M
        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = 2.0 * n * fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            # Refine
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = np.maximum((1.0 - mid) * f + mid * g, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = 2.0 * n * fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            m_exp = 1.0

            if 2.0 * n * M < best_obj - 1e-14:
                best_obj = 2.0 * n * M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 6.0)

        if no_improve > 50:
            f = best_f + rng.standard_normal(n) * 0.003
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase optimization:
    Phase 1: Diverse initialization + coordinate descent on each
    Phase 2: FW-LP on best candidate
    Phase 3: Simulated annealing
    Phase 4: LSE polishing
    Phase 5: Final coordinate descent
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_score = float('inf')
    best_seq = None
    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = np.ones(448) / 448
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    print(f"Target: 1.5027, Current best: {best_tracker[0]:.8f}")

    # === Phase 1: Diverse initialization + coordinate descent ===
    print("\n=== Phase 1: Diverse Initialization + Coord Descent ===")
    n_1 = time.monotonic()

    # Choose target length
    n_target = n_parent  # Start with parent's length

    # Generate candidates
    candidates = build_diverse_initializations(n_target, parent, rng)
    print(f"  Generated {len(candidates)} diverse candidates")

    # Evaluate all quickly
    t_per_cand = min(20.0, (deadline - time.monotonic() - 600) / max(1, len(candidates)))
    results = []
    for idx, cand in enumerate(candidates):
        if time.monotonic() > deadline - 600:
            break
        try_sequence(cand, best_tracker)
        # Quick coordinate descent
        t_alloc = min(t_per_cand, deadline - time.monotonic() - 550)
        if t_alloc > 3:
            f_opt, obj = coord_descent(cand, time_limit=t_alloc, seed=seed + idx)
            try_sequence(f_opt, best_tracker)
            results.append((obj, f_opt))

    # Sort by objective
    results.sort(key=lambda x: x[0])
    if results:
        print(f"  Best after Phase 1: {results[0][0]:.10f}")

    # Multi-length exploration
    for n_alt in [384, 512, 320, 640]:
        if time.monotonic() > deadline - 400:
            break
        if n_alt == n_target:
            continue
        best_cur = best_tracker[1] if best_tracker[1] is not None else parent
        x_old = np.linspace(0, 1, len(best_cur))
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, best_cur)
        f_interp = np.maximum(f_interp, 0)
        if f_interp.sum() > 1e-10:
            f_interp = f_interp / f_interp.sum()
        t_alloc = min(40.0, deadline - time.monotonic() - 350)
        if t_alloc > 10:
            f_opt, obj = coord_descent(f_interp, time_limit=t_alloc, seed=seed + 100)
            try_sequence(f_opt, best_tracker)

    print(f"  Phase 1 done. Best: {best_tracker[0]:.8f} (elapsed={time.monotonic()-n_1:.0f}s)")

    # === Phase 2: FW-LP on best ===
    print(f"\n=== Phase 2: FW-LP Refinement ===")
    n_2 = time.monotonic()
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(200, deadline - time.monotonic() - 250)
    if t_alloc > 20:
        print(f"  Allocating {t_alloc:.0f}s")
        f_opt, obj = fw_lp_optimize(current_best, time_limit=t_alloc, seed=seed + 200)
        try_sequence(f_opt, best_tracker)
    print(f"  Phase 2 done. Best: {best_tracker[0]:.8f} (elapsed={time.monotonic()-n_2:.0f}s)")

    # === Phase 3: Simulated Annealing ===
    print(f"\n=== Phase 3: Simulated Annealing ===")
    n_3 = time.monotonic()
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(200, deadline - time.monotonic() - 200)
    if t_alloc > 20:
        print(f"  Allocating {t_alloc:.0f}s")
        f_opt, obj = sa_optimize(current_best, time_limit=t_alloc, seed=seed + 300)
        try_sequence(f_opt, best_tracker)
        # Continue with coord descent on SA result
        t2 = min(100, deadline - time.monotonic() - 150)
        if t2 > 10:
            f_opt2, obj2 = coord_descent(f_opt, time_limit=t2, seed=seed + 350)
            try_sequence(f_opt2, best_tracker)
    print(f"  Phase 3 done. Best: {best_tracker[0]:.8f} (elapsed={time.monotonic()-n_3:.0f}s)")

    # === Phase 4: LSE Polishing ===
    print(f"\n=== Phase 4: LSE Polishing ===")
    n_4 = time.monotonic()
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(200, deadline - time.monotonic() - 100)
    if t_alloc > 20:
        print(f"  Allocating {t_alloc:.0f}s")
        f_opt, obj = lse_optimize(current_best, time_limit=t_alloc,
                                  beta_start=15.0, beta_end=4000.0, seed=seed + 400)
        try_sequence(f_opt, best_tracker)
    print(f"  Phase 4 done. Best: {best_tracker[0]:.8f} (elapsed={time.monotonic()-n_4:.0f}s)")

    # === Phase 5: Final coordinate descent ===
    print(f"\n=== Phase 5: Final Coord Descent ===")
    n_5 = time.monotonic()
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = max(30, deadline - time.monotonic() - 15)
    if t_alloc > 10:
        print(f"  Allocating {t_alloc:.0f}s")
        f_opt, obj = coord_descent(current_best, time_limit=t_alloc,
                                   n_sweeps_max=20000, seed=seed + 500)
        try_sequence(f_opt, best_tracker)
    print(f"  Phase 5 done. Best: {best_tracker[0]:.8f} (elapsed={time.monotonic()-n_5:.0f}s)")

    # === Bonus: One more round if time allows ===
    if time.monotonic() < deadline - 30:
        print(f"\n=== Bonus Round ===")
        current_best = best_tracker[1] if best_tracker[1] is not None else parent
        t_alloc = deadline - time.monotonic() - 10
        if t_alloc > 20:
            # Try LSE with high beta
            f_opt, _ = lse_optimize(current_best, time_limit=min(80, t_alloc * 0.5),
                                    beta_start=200.0, beta_end=10000.0, seed=seed + 600)
            try_sequence(f_opt, best_tracker)
            # Then coord descent
            current_best = best_tracker[1] if best_tracker[1] is not None else current_best
            t2 = deadline - time.monotonic() - 5
            if t2 > 10:
                f_opt2, _ = coord_descent(current_best, time_limit=t2,
                                          n_sweeps_max=50000, seed=seed + 650)
                try_sequence(f_opt2, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
    else:
        final_seq = np.ones(448) / 448
        best_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
