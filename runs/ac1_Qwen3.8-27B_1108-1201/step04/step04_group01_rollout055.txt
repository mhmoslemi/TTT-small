"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Softmax parameterization (avoids simplex projection)
2. Power-sum smoothed objective with adaptive exponent
3. Nesterov-accelerated gradient descent
4. Structured random initializations for diverse exploration
5. FFT-based gradient computation
6. Multi-length search with reinitialization

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_grad_power_sum(a, p):
    """Compute gradient of (sum conv^p)^(1/p) w.r.t. a.
    Returns (grad, P) where P is the power-sum value.
    """
    n = len(a)
    conv = fast_convolve(a)
    
    # For numerical stability, work in log space
    # P = (sum conv^p)^(1/p)
    # grad_a[i] = 2 * P^(1-p) * sum_k conv[k]^(p-1) * a[k-i]
    
    # Handle zeros in conv
    conv_pos = np.maximum(conv, 1e-300)
    
    if p <= 1.0:
        # p=1: P = sum(conv), grad = 2 * sum_k a[k-i] (constant)
        # Actually for p=1, P = sum_k conv[k] = (sum a)^2, grad is constant
        # Not useful, so we require p > 1
        p = 1.01
    
    log_conv = np.log(conv_pos)
    p_log_conv = p * log_conv
    log_sum = p_log_conv.max() + np.log(np.sum(np.exp(p_log_conv - p_log_conv.max())))
    log_P = log_sum / p
    P = np.exp(log_P)
    
    # weights[k] = conv[k]^(p-1) / P^(p-1) = (conv[k]/P)^(p-1)
    log_weights = (p - 1) * (log_conv - log_P)
    # Clip for numerical stability
    log_weights = np.clip(log_weights, -30, 0)
    weights = np.exp(log_weights)
    
    # grad_a[i] = 2/P * sum_k weights[k] * a[k-i]
    # Cross-correlation of weights and a
    # cross[i] = sum_k weights[k] * a[k-i]
    # Using FFT: pad to size >= len(weights) + n - 1
    total_len = len(weights) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    A = np.fft.rfft(a, fft_size)
    # cross-correlation: ifft(conj(W) * A)
    cross = np.fft.irfft(np.conj(W) * A, fft_size)
    # cross[i] for i = 0, ..., n-1
    grad = (2.0 / P) * cross[:n]
    
    return grad, P


def softmax(z):
    z_max = z.max()
    e = np.exp(z - z_max)
    return e / e.sum()


def softmax_grad(z, grad_a):
    """Compute dz = d(objective)/dz given grad_a = d(objective)/da and a = softmax(z)."""
    a = softmax(z)
    return a * (grad_a - np.dot(a, grad_a))


def power_sum_objective(a, p):
    """Compute (sum conv(a,a)^p)^(1/p)."""
    conv = fast_convolve(a)
    conv = np.maximum(conv, 0)
    return (np.sum(conv ** p)) ** (1.0 / p)


def nesterov_power_sum(z0, n, p_start, p_end, time_limit, lr_init=None, momentum=0.9):
    """Nesterov accelerated gradient descent on power-sum objective in softmax space."""
    rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
    
    z = z0.copy()
    a = softmax(z)
    
    if lr_init is None:
        lr_init = 50.0 / n
    
    lr = lr_init
    best_obj = np.inf
    best_a = a.copy()
    best_z = z.copy()
    
    # Nesterov state
    v = np.zeros(n)
    
    # Compute initial objective
    conv = fast_convolve(a)
    p_cur = p_start
    obj = power_sum_objective(a, p_cur)
    best_obj = obj
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Adaptive p
        progress = (time.monotonic() - t0) / time_limit
        p_cur = p_start * (p_end / p_start) ** progress
        
        # Nesterov extrapolation
        a_nest = softmax(z + momentum * v)
        
        # Compute gradient at extrapolated point
        if p_cur < 1.5:
            # Use max subgradient
            conv = fast_convolve(a_nest)
            max_val = conv.max()
            # Subgradient at max lag
            max_lags = np.where(conv >= max_val * 0.9999)[0]
            grad_a = np.zeros(n)

                idx = np.arange(js, je + 1)

        # Convert to z-space gradient

        # Nesterov update

        z_new = z + v
        a_new = softmax(z_new)
        
        # Evaluate
        obj_new = power_sum_objective(a_new, p_cur)
        
        if obj_new < best_obj * (1 - 1e-10):
            best_obj = obj_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            z = z_new
            no_improve = 0
        else:
            z = z + v  # Still update z (Nesterov style)

                lr *= 0.95
            if no_improve > 200:
                # Restart

                no_improve = 0

    return best_a, best_obj


def structured_init(n, rng, style='random'):
    """Generate a structured random initialization."""
    if style == 'random':
        a = rng.random(n)
    elif style == 'bimodal':
        # U-shaped: high at edges, low in middle
        x = np.linspace(-1, 1, n)
        a = 1.0 + 2.0 * x**2 + 0.5 * rng.random(n)
    elif style == 'triangle':
        # Inverted triangle (V-shaped)
        x = np.linspace(-1, 1, n)
        a = 1.0 + 2.0 * np.abs(x) + 0.5 * rng.random(n)
    elif style == 'cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x) + 0.3 * rng.random(n)
    elif style == 'uniform':
        a = np.ones(n)
    elif style == 'exponential_decay':
        a = np.exp(-np.linspace(0, 3, n)) * rng.random(n) + 0.1
    else:
        a = rng.random(n)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

    a_pos = np.maximum(a, 1e-15)
