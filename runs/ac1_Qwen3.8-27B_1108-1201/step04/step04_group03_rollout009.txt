"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Softmax-parameterized L-BFGS with continuation on temperature
2. Subgradient refinement with backtracking line search
3. Multi-start from diverse initializations
4. Direct "peak suppression" heuristic steps

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_C(w, a):
    """Compute C[m] = sum_{j=0}^{n-1} a[j] * w[m+j] for m=0,...,n-1.
    w has length 2n-1."""
    n = len(a)
    C = np.zeros(n)
    for m in range(n):
        j_end = min(n, 2 * n - 1 - m)
        if j_end > 0:
            C[m] = np.dot(a[:j_end], w[m:m + j_end])
    return C


def softmax_obj_grad(theta, T, n):
    """Compute soft-max objective and gradient in theta space."""
    a = np.exp(theta - theta.max())
    a /= a.sum()
    
    b = fast_convolve(a)
    
    # Softmax weights on peaks
    scaled = b / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()
    
    # Objective: 2*n*soft_max(b, T) = 2*n*(max(b) + T*log(sum(exp(scaled))))
    obj = 2.0 * n * (b.max() + T * np.log(e.sum()))
    
    # Gradient
    S_val = np.dot(w, b)
    C = compute_C(w, a)
    grad_theta = 4.0 * n * a * (C - S_val)
    
    return obj, grad_theta


def lbfgs_optimize(a0, time_limit, T_init=None, T_min=1e-8, T_decay=0.5, 
                   max_lbfgs_iter=200, verbose=True):
    """
    L-BFGS optimization in softmax parameter space with temperature continuation.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    b = fast_convolve(a)
    max_b = b.max()
    best_obj = 2.0 * n * max_b
    best_a = a.copy()
    
    if T_init is None:
        T_init = max_b / 10.0
    
    t0 = time.monotonic()
    T = T_init
    
    while time.monotonic() - t0 < time_limit and T > T_min:
        # Initial theta (from current a)
        theta0 = np.log(a + 1e-30)
        
        # Run L-BFGS
        try:
            res = minimize(
                softmax_obj_grad, theta0,
                args=(T, n),
                method='L-BFGS-B',
                jac=True,
                options={
                    'maxiter': max_lbfgs_iter,
                    'gtol': 1e-7,
                    'ftol': 1e-12,
                    'maxfun': 500
                }
            )
        except Exception as e:
            if verbose:
                print(f"  L-BFGS error: {e}")
            T *= T_decay
            continue
        
        if res.fun < softmax_obj_grad(theta0, T, n)[0]:
            theta_opt = res.x
            a_new = np.exp(theta_opt - theta_opt.max())
            a_new /= a_new.sum()
            
            b_new = fast_convolve(a_new)
            actual_obj = 2.0 * n * b_new.max()
            
            if actual_obj < best_obj:
                best_obj = actual_obj
                best_a = a_new.copy()
                a = a_new.copy()
        
        if verbose:
            el = time.monotonic() - t0
            print(f"  T={T:.2e}, obj={best_obj:.8f}, elapsed={el:.1f}s")
        
        T *= T_decay
    
    return best_a, best_obj


def subgradient_refine(a0, time_limit, lr_init=0.01, momentum=0.9, 
                       num_peaks=50, verbose=True):
    """
    Subgradient method with momentum and backtracking line search,
    using softmax-weighted peaks for smoothness.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    b = fast_convolve(a)
    best_obj = 2.0 * n * b.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    v = np.zeros(n)
    lr = lr_init
    temperature = b.max() / 50.0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        
        # Softmax weights on peaks
        scaled = b / temperature
        scaled -= scaled.max()
        e = np.exp(scaled)
        w = e / e.sum()
        
        # Gradient of soft max w.r.t. a
        # grad_a[m] = sum_k w[k] * 2*a[k-m] = 2 * C[m] where C[m] = sum_k w[k]*a[k-m]
        C = compute_C(w, a)
        grad_a = 2.0 * C
        
        # Objective value (smooth)
        obj_smooth = 2.0 * n * (b.max() + temperature * np.log(e.sum()))
        
        # Momentum update
        v = momentum * v - lr * grad_a
        a_new = project_simplex(a + v)
        
        # Evaluate
        b_new = fast_convolve(a_new)
        obj_new = 2.0 * n * b_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
            lr = min(lr * 1.005, 0.05)
        else:
            no_improve += 1
            lr *= 0.97
            temperature *= 0.99  # sharpen the peak weighting
            a = best_a.copy()
        
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(hash((n, int(time.monotonic()*1000))) % (2**31))
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(best_a + noise)
            v = np.zeros(n)
            lr = lr_init * 0.5
            temperature = best_obj / (2 * n) / 30.0
            no_improve = 0
    
    return best_a, best_obj


def peak_suppression(a0, time_limit, verbose=True):
    """
    Direct peak suppression: identify the peak, find contributing pairs,
    and transfer mass from high-contributing positions to safe ones.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    b = fast_convolve(a)
    best_obj = 2.0 * n * b.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step = 0.01
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        peak_idx = np.argmax(b)
        
        # Find contributing pairs for the peak
        # b[peak_idx] = sum_j a[j] * a[peak_idx - j]
        j_start = max(0, peak_idx - n + 1)
        j_end = min(n - 1, peak_idx)
        
        if j_end < j_start:
            break
        
        js = np.arange(j_start, j_end + 1)
        contributions = a[js] * a[peak_idx - js]
        total_contrib = contributions.sum()
        
        if total_contrib < 1e-15:
            break
        
        # Find "safe" positions (low contribution to any peak)
        # A position m is "safe" if a[m] is small OR it doesn't pair with large values
        # Compute total "danger" for each position
        danger = np.zeros(n)
        for k in range(2 * n - 1):
            if b[k] > 0.8 * max_b:  # only consider significant peaks
                j_start_k = max(0, k - n + 1)
                j_end_k = min(n - 1, k)
                js_k = np.arange(j_start_k, j_end_k + 1)
                danger[js_k] += b[k] * a[js_k] * a[k - js_k]
        
        # Remove mass from highest-danger positions, add to lowest-danger
        top_danger = np.argsort(danger)[-10:]  # top 10 most dangerous
        bottom_danger = np.argsort(danger)[:10]  # bottom 10 safest
        
        mass_to_move = step * total_contrib
        
        from_mass = np.sum(a[top_danger]) * 0.1  # move 10% of mass from dangerous
        from_mass = min(from_mass, mass_to_move)
        
        if from_mass < 1e-15:
            step *= 0.9
            continue
        
        # Scale down dangerous positions
        danger_scale = a[top_danger]
        if danger_scale.sum() > 1e-15:
            a[top_danger] -= from_mass * danger_scale / danger_scale.sum()
        
        # Scale up safe positions
        safe_weights = a[bottom_danger] + 0.01
        a[bottom_danger] += from_mass * safe_weights / safe_weights.sum()
        
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        b = fast_convolve(a)
        obj = 2.0 * n * b.max()
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            step = min(step * 1.01, 0.05)
        else:
            a = best_a.copy()
            step *= 0.9
            if step < 1e-5:
                break
    
    return best_a, best_obj


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    # Edge-weighted
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Gaussian (centered)
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Cosine window
    c = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.01
    inits.append(c / c.sum())
    
    # Triangular
    tri = np.minimum(x + 1, n - x)
    inits.append(tri / tri.sum())
    
    # Decay (exponential)
    decay = np.exp(-x / (n / 5.0))
    inits.append(decay / decay.sum())
    
    # Bimodal
    bmodal = np.exp(-0.5 * ((x - 0.3 * (n-1)) / (0.1*n))**2) + np.exp(-0.5 * ((x - 0.7*(n-1)) / (0.1*n))**2)
    inits.append(bmodal / bmodal.sum())
    
    # Parent (interpolated to length n)
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    # Random sparse
    for _ in range(3):
        sparse = np.zeros(n)
        support = rng.choice(n, size=max(10, n//4), replace=False)
        sparse[support] = rng.dirichlet(np.ones(len(support)) * 0.5)
        sparse /= sparse.sum()
        inits.append(sparse)
    
    return inits


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle (kept from previous for compatibility)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve LP: max sum(g) s.t. f*g[k] <= M for all k, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': max(1.0, (time_limit - (time.monotonic()-t0)) / max(1, 1000-it))})
        except Exception:
            break
        
        if not res.success or res.x is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 17)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
    
    return best_f, best_obj
