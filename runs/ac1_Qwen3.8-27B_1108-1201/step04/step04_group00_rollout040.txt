```python
import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    # Build A_ub efficiently
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j+n, j] = f
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def lp_step(f, M, n):
    """One LP-based Frank-Wolfe step."""
    g = solve_lp(f, M, n)
    if g is None:
        return None
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None
    g = g / g_sum
    return g

def line_search(f, g, max_iter=20):
    """Find best alpha in [0,1] for f + alpha*(g-f)."""
    best_alpha = 0.0
    best_val = fast_convolve(f).max()
    
    # Coarse search
    alphas = np.linspace(0, 1, 11)
    vals = []
    for alpha in alphas:
        ft = f + alpha * (g - f)
        ft = np.maximum(ft, 0)
        s = ft.sum()
        if s < 1e-15:
            vals.append(np.inf)
            continue
        ft = ft / s
        vals.append(fast_convolve(ft).max())
    
    best_idx = np.argmin(vals)
    best_alpha = alphas[best_idx]
    best_val = vals[best_idx]
    
    # Fine search (golden section)
    lo = alphas[max(0, best_idx-1)]
    hi = alphas[min(len(alphas)-1, best_idx+1)]
    golden = 0.618
    for _ in range(15):
        m1 = hi - golden * (hi - lo)
        m2 = lo + golden * (hi - lo)
        v1 = eval_alpha(f, g, m1)
        v2 = eval_alpha(f, g, m2)
        if v1 < v2:
            hi = m2
        else:
            lo = m1
        mid = (lo + hi) / 2
        vm = eval_alpha(f, g, mid)
        if vm < best_val:
            best_val = vm
            best_alpha = mid
    
    return best_alpha, best_val

def eval_alpha(f, g, alpha):
    ft = f + alpha * (g - f)
    ft = np.maximum(ft, 0)

    ft = ft / s
    return fast_convolve(ft).max()
```

```python
def eigenvalue_iteration(f0, time_limit, temperature=1.0, power=2.0, verbose=False):
    """Fixed-point iteration based on eigenvalue characterization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute convolution
        conv = fast_convolve(f)
        # Compute lambda weights
        # Use softmax with temperature
        scaled = conv / temperature
        scaled -= scaled.max()
        lam = np.exp(scaled)
        lam /= lam.sum()
        
        # Build Q
        Q = np.zeros((n, n))
        for k in range(2*n - 1):
            if lam[k] < 1e-12:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                l = k - j
                if j == l:
                    Q[j, j] += lam[k]
                else:
                    Q[j, l] += lam[k] * 0.5
        
        # Solve Q a = 1 (with regularization)
        try:
            Q_reg = Q + 1e-8 * np.eye(n)
            a_new = np.linalg.solve(Q_reg, np.ones(n))
        except np.linalg.LinAlgError:
            a_new = np.linalg.lstsq(Q_reg, np.ones(n), rcond=None)[0]
        
        # Project
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = f.copy()
        else:
            a_new = a_new / s_new
        
        # Mix
        t_mix = 0.5  # mixing parameter
        f_new = (1 - t_mix) * f + t_mix * a_new
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        
        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
        else:
            f = f_new  # still update, might help escape
            # Cool down
            temperature *= 1.05
        
        if verbose and it % 10 == 0:
            print(f"  Eigen iter {it}: obj={best_obj:.10f}, temp={temperature:.4f}")
    
    return best_f, best_obj
```

```python
def eigen_cg_step(f, lam, n, eps=1e-6, max_cg_iter=100):
    """Solve (Q + eps*I) a = 1 using conjugate gradient, where Qa is computed via FFT."""
    def Q_times_a(a):
        a_rev = a[::-1]
        fft_size = 1 << (3*n - 1).bit_length()  # enough for cross-correlation
        L = np.fft.rfft(lam, fft_size)
        A = np.fft.rfft(a_rev, fft_size)
        c_full = np.fft.irfft(L * A, fft_size)
        c = c_full[n-1:2*n-1]
        return 0.5 * c + 0.5 * lam[2*np.arange(n)] * a + eps * a
    
    # CG to solve (Q + eps*I) x = ones
    b = np.ones(n)
    x = np.zeros(n)
    r = b - Q_times_a(x)
    p = r.copy()
    rs_old = r @ r
    
    for i in range(max_cg_iter):
        Ap = Q_times_a(p)
        pAp = p @ Ap
        if abs(pAp) < 1e-15:
            break
        alpha = rs_old / pAp
        x += alpha * p
        r -= alpha * Ap
        rs_new = r @ r
        if rs_new < 1e-12 * rs_old:
            break
        p = r + (rs_new / rs_old) * p
        rs_old = rs_new
    
    return x
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
- LP-based Frank-Wolfe for global optimization
- FFT-accelerated proper subgradient descent
- Eigenvalue fixed-point iteration (KKT refinement via CG solve)
- Multi-length exploration with adaptive time allocation

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_corr(lam, a):
    """Compute c[j] = sum_{m=0}^{n-1} lam[j+m] * a[m] for j=0..n-1 via FFT."""
    n = len(a)
    lam_len = len(lam)
    fft_size = 1 << (lam_len + n - 1 - 1).bit_length()
    L = np.fft.rfft(lam, fft_size)
    A = np.fft.rfft(a[::-1], fft_size)
    c_full = np.fft.irfft(L * A, fft_size)
    return c_full[n - 1:2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_Q_matrix(lam, n):
    """Build Q = sum_k lam[k] * Q_k as a dense matrix."""
    Q = np.zeros((n, n))
    for k in range(2 * n - 1):
        wk = lam[k]
        if wk < 1e-13:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            l = k - j
            if j == l:
                Q[j, j] += wk
            else:
                Q[j, l] += wk * 0.5
    return Q


def Q_times_a(lam, a, n):
    """Compute Q*a where Q = sum_k lam[k]*Q_k, using FFT."""
    c = cross_corr(lam, a)
    idx = 2 * np.arange(n)
    idx = np.minimum(idx, 2 * n - 2)
    return 0.5 * c + 0.5 * lam[idx] * a


def cg_solve_Q(lam, n, rhs, max_iter=200, tol=1e-10, eps_reg=1e-8):
    """Solve (Q + eps*I) x = rhs using conjugate gradient."""
    def matvec(x):
        return Q_times_a(lam, x, n) + eps_reg * x

    x = np.zeros(n)
    r = rhs - matvec(x)
    p = r.copy()
    rs = r @ r
    b_norm = np.linalg.norm(rhs)
    for _ in range(max_iter):
        Ap = matvec(p)
        pAp = p @ Ap
        if abs(pAp) < 1e-16:
            break
        alpha = rs / pAp
        x += alpha * p
        r -= alpha * Ap
        rs_new = r @ r
        if np.sqrt(rs_new) < tol * b_norm:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1

        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.1
            if m_exp > 5.0:
                m_exp = 1.0
            no_improve += 1
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.04, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine golden section
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.15)
            hi = min(1.0, best_alpha + 0.15)
            gr = 0.618
            for _ in range(12):
                m1 = hi - gr * (hi - lo)
                m2 = lo + gr * (hi - lo)
                v1 = _eval_alpha(f, g, m1)
                v2 = _eval_alpha(f, g, m2)
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
                mid = (lo + hi) / 2.0
                vm = _eval_alpha(f, g, mid)
                if vm < best_val:
                    best_val = vm
                    best_alpha = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 30 == 0:
                    print(f"    LP-FW it{it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 13 + 5)
            scale = 0.003 + 0.003 * min(no_improve // 80, 4)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def _eval_alpha(f, g, alpha):
    ft = f + alpha * (g - f)
    ft = np.maximum(ft, 0.0)
    st = ft.sum()
    if st < 1e-15:
        return np.inf
    ft = ft / st
    return fast_convolve(ft).max()


def gradient_descent_opt(f0, time_limit, lr=0.01, verbose=False):
    """Proper subgradient descent on the simplex using FFT-accelerated gradients."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0
    temp = 0.5
    cur_lr = lr
    it = 0
    v = np.zeros(n)
    mom = 0.9

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1

        conv = fast_convolve(f)
        # Softmax weights on peaks
        scaled = conv / temp
        scaled -= scaled.max()
        lam = np.exp(scaled)
        lam_sum = lam.sum()
        if lam_sum < 1e-15:
            lam = np.zeros(2 * n - 1)
            lam[np.argmax(conv)] = 1.0
        else:
            lam /= lam_sum

        # Compute gradient: grad[j] = c[j] + lam[2j]*f[j]
        c = cross_corr(lam, f)
        idx = 2 * np.arange(n)
        idx = np.minimum(idx, 2 * n - 2)
        grad = c + lam[idx] * f

        # Momentum + projected gradient step
        v = mom * v - cur_lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    GD it{it}: obj={best_obj:.10f}, temp={temp:.4f}, lr={cur_lr:.6f}")
        else:
            f = f_new
            no_improve += 1

        # Adapt parameters
        cur_lr *= 0.9995
        temp *= 1.004

        if no_improve > 300:
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.001)
            no_improve = 0

    return best_f, best_obj


def eigenvalue_fp_iter(f0, time_limit, verbose=False):
    """Fixed-point iteration: solve Q a = 1 where Q depends on current peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    temp = 0.3
    it = 0
    no_improve = 0
    mix = 0.6

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1

        conv = fast_convolve(f)
        scaled = conv / temp
        scaled -= scaled.max()
        lam = np.exp(scaled)
        lam_sum = lam.sum()
        if lam_sum < 1e-15:
            lam = np.zeros(2 * n - 1)
            lam[np.argmax(conv)] = 1.0
        else:
            lam /= lam_sum

        # Solve (Q + eps*I) a = 1 using CG
        a_new = cg_solve_Q(lam, n, np.ones(n), max_iter=300, tol=1e-8, eps_reg=1e-6)

        # Project
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = f.copy()
        else:
            a_new = a_new / s_new

        # Mix
        f_new = (1.0 - mix) * f + mix * a_new
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if verbose and it % 10 == 0:
                print(f"    EigenFPI it{it}: obj={best_obj:.10f}, temp={temp:.4f}")
        else:
            f = f_new
            no_improve += 1

        temp *= 1.02
        mix = max(0.3, mix * 0.99)

        if no_improve > 100:
            rng = np.random.default_rng(it * 11 + 2)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            no_improve = 0
            temp = max(0.1, temp * 0.5)

def coordinate_descent_opt(f0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.5, 0.7, 0.85, 1.2, 1.4, 1.7]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 4:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    if parent is None:
        parent = np.ones(512) / 512
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n=512, score={p_score:.6f}")

    # === Phase 1: Quick LP-FW on parent ===
    print("\n=== Phase 1: LP-FW on parent ===")
    rem = deadline - time.monotonic()
    t1 = min(150, rem * 0.15)
    if t1 > 30:
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "lpfw_p1")

    # === Phase 2: Multi-length exploration ===
    print("\n=== Phase 2: Multi-length exploration ===")
    base = best_seq if best_seq is not None else parent
    target_ns = [384, 512, 640, 768, 896, 1024]

    for tn in target_ns:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        f_interp = interpolate_sequence(base, tn)
        t_alloc = min(rem * 0.12, 120)
        if t_alloc < 25:
            continue
        print(f"  LP-FW n={tn}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=False)
        try_seq(a_opt, f"lpfw_n{tn}")

    # === Phase 3: Eigenvalue fixed-point on best ===
    print(f"\n=== Phase 3: Eigenvalue FP (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t3 = min(rem * 0.2, 300)
        if t3 > 30:
            print(f"  EigenFPI t={t3:.0f}s, n={len(best_seq)}")
            a_opt, _ = eigenvalue_fp_iter(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "eigen_fp")

    # === Phase 4: Gradient descent polish ===
    print(f"\n=== Phase 4: Gradient Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t4 = min(rem * 0.2, 250)
        if t4 > 30:
            print(f"  GD t={t4:.0f}s, n={len(best_seq)}")
            a_opt, _ = gradient_descent_opt(best_seq, time_limit=t4, lr=0.008, verbose=True)
            try_seq(a_opt, "gd_polish")

    # === Phase 5: Deep LP-FW ===
    print(f"\n=== Phase 5: Deep LP-FW (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t5 = min(rem * 0.25, 300)
        if t5 > 30:
            print(f"  Deep LP-FW t={t5:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t5, verbose=True)
            try_seq(a_opt, "deep_lpfw")

            # Try from perturbed
            rem = deadline - time.monotonic()
            if rem > 60:
                noise = rng.standard_normal(len(best_seq)) * 0.005
                f_pert = project_simplex(best_seq + noise)
                t5b = min(rem * 0.5, 150)
                a_opt, _ = lp_fw_optimize(f_pert, time_limit=t5b, verbose=False)
                try_seq(a_opt, "deep_pert")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t6 = min(rem * 0.2, 200)
        if t6 > 20:
            print(f"  CD t={t6:.0f}s")
            a_opt, _ = coordinate_descent_opt(best_seq, time_limit=t6)
            try_seq(a_opt, "cd")

    # === Phase 7: Final polish rounds ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick eigenvalue refinement
        rem = deadline - time.monotonic()
        if rem > 30:
            t7 = min(rem * 0.3, 100)
            a_opt, _ = eigenvalue_fp_iter(best_seq, time_limit=t7, verbose=False)
            try_seq(a_opt, "eigen_final")

        # Gradient descent with very small lr
        rem = deadline - time.monotonic()
        if rem > 25:
            t7b = min(rem * 0.3, 80)
            a_opt, _ = gradient_descent_opt(best_seq, time_limit=t7b, lr=0.003, verbose=False)
            try_seq(a_opt, "gd_fine")

        # One more LP-FW
        rem = deadline - time.monotonic()
        if rem > 25:
            t7c = min(rem * 0.4, 60)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t7c, verbose=False)
            try_seq(a_opt, "lpfw_final")

        # Tiny perturbations
        for pidx in range(3):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem * 0.3, 30)
            a_opt, _ = gradient_descent_opt(f_pert, time_limit=t_pert, lr=0.002, verbose=False)
            try_seq(a_opt, f"pert_{pidx}")

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
