def compute_soft_subgrad(a, conv, top_k, temp):
    """Weighted subgradient using top-k peaks with softmax weights."""
    n = len(a)

    scaled -= scaled.max()
    w = np.exp(scaled)

        t_quad = 0.2
    
    best_obj = 2.0 * n * M
    best_a = a.copy()
    best_t = 0.0
    
    candidates = sorted(set([t_quad, 0.5, t_quad * 1.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]),
                        reverse=True)
    
    for t in candidates:
        if t <= 0:
            continue
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        a_try = a_try / s_try
        conv_try = fast_autoconvolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            best_t = t
            break
    
    if best_t > 0:
        return best_a, True, best_obj
    return a.copy(), False, 2.0 * n * M


def nesterov_optimize(a0, time_limit, lr_init=0.01, momentum=0.92, top_k=30, temp_init=0.002):
    """Nesterov-accelerated subgradient method."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_autoconvolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
 
        conv_y = fast_autoconvolve(y)
        grad = compute_soft_subgrad(y, conv_y, top_k=top_k, temp=temp)
        
        # Gradient step
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)
            break
        
        conv = fast_autoconvolve(a)
        cur_obj = 2.0 * n * conv.max()
        iter_count += 1
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive parameters
        lr *= 0.9997
        temp *= 1.002
        mom = min(0.97, mom + 0.0001)
        
        if no_improve >= 300:
            break
    
    return best_a, best_obj


def coordinate_refine(a0, time_limit, n_deltas=6):
    """Fast coordinate descent: try small transfers between pairs."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_autoconvolve(a).max()
    a = a / s
    n = len(a)
    
    conv = fast_autoconvolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0
    deltas = [0.002, 0.005, 0.01, 0.02, 0.05, -0.002, -0.005, -0.01, -0.02, -0.05]
    
    while time.monotonic() - t0 < time_limit:
        batch = max(1, int(200000 / n))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            found = False
            for delta in deltas:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                conv_try = fast_autoconvolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    a = a_try
                    conv = conv_try
                    best_obj = obj_try
                    best_a = a.copy()
                    no_improve = 0
                    found = True
                    break
            if not found:
                no_improve += 1
        
        if no_improve > 500:
            break
    
    return best_a, best_obj


def spectral_rebalance(a0, n_steps=50):
    """Try to reduce variance of power spectrum as auxiliary step."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    fft_size = 1 << (2 * n - 2).bit_length()
    for _ in range(n_steps):
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        P_mean = P.mean()
        if P_max < 2 * P_mean + 1e-10:
            break
        # Gradient of max(P) w.r.t. a (through real part of A)
        k_max = int(np.argmax(P))
        # d|A_k|^2/da_j = 2 * a_j * cos(2*pi*j*k/fft_size) ... simplified
        phase = np.cos(2 * np.pi * np.arange(n) * k_max / fft_size)
        a -= 0.001 * a * phase
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = a0 / a0.sum()
            break
        a = a / s
    return a


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    s1 = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine1'] = s1 / s1.sum()
    
    s2 = (np.sin(2 * np.pi * x / n) ** 2 + 0.01)
    inits['sine2'] = s2 / s2.sum()
    
    s3 = (np.sin(np.pi * x * 1.5 / n + 0.3) ** 2 + 0.01)
    inits['sine3'] = s3 / s3.sum()
    
    # Chirp-like
    chirp = (1.0 + np.cos(2 * np.pi * x ** 2 / (n * 1.5))) / 2.0
    chirp = np.maximum(chirp, 0.01)
    inits['chirp'] = chirp / chirp.sum()
    
    # Bimodal
    edge = max(1, n // 6)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Alternating
    alt = np.ones(n)
    alt[::2] = 1.3
    alt[1::2] = 0.7
    inits['alt'] = alt / alt.sum()
    
    # Triangular
    tri = np.ones(n)
    mid = n // 2
    tri[:mid] = np.linspace(0.3, 1.0, mid)
    tri[mid:] = np.linspace(1.0, 0.3, n - mid)
    inits['tri'] = tri / tri.sum()
    
    # From parent
    if parent is not None:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
    
    return inits


def lp_main_loop(a0, time_limit, n_lags=25, verbose=False):
    """Main LP optimization loop with active set and line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_autoconvolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    perturb_count = 0
    iter_count = 0
    use_minimax = False
    
    while time.monotonic() - t0 < time_limit:
        # Alternate between standard LP and minimax LP
        use_minimax = (iter_count % 5 == 4)
        
        a_new, improved, obj_new = lp_step(a, conv, M, n_lags=n_lags, use_minimax=use_minimax)
        
        if improved:
            a = a_new
            conv = fast_autoconvolve(a)
            M = conv.max()
            cur_obj = 2.0 * n * M
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        iter_count += 1
        
        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter {iter_count}, obj={best_obj:.8f}, t={elapsed:.1f}s")
        
        # Stagnation handling
        if no_improve >= 20:
            # Try gradient step for a bit
            if time.monotonic() - t0 < time_limit - 5:
                a_g, obj_g = nesterov_optimize(best_a, time_limit=min(3.0, (time_limit - (time.monotonic() - t0)) * 0.2),
                                               lr_init=0.008, top_k=40, temp_init=0.001)
                if obj_g < best_obj:
                    best_obj = obj_g
                    best_a = a_g.copy()
                a = best_a.copy()
                conv = fast_autoconvolve(a)
                M = conv.max()
            
            # Perturb if still stuck
            if no_improve >= 40:
                rng = np.random.default_rng(1000 + perturb_count * 7919)
                scale = 0.002 + 0.003 * min(perturb_count, 12)
                noise = rng.standard_normal(n) * scale
                a = best_a + noise
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a < 1e-15:
                    a = best_a.copy()
                else:
                    a = a / s_a
                conv = fast_autoconvolve(a)
                M = conv.max()
                no_improve = 0
                perturb_count += 1
                
                if perturb_count > 60:
                    break
    
    return best_a, best_obj
