"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed subgradient descent
with Nesterov acceleration, log-sum-exp smoothing, interleaved pairwise
coordinate descent, and periodic LP-based Frank-Wolfe jumps. Key innovations:
(1) Smoothed gradients via temperature annealing, (2) analytic pairwise swap
line search with O(n) cost, (3) adaptive restart with diversity-preserving
perturbations, (4) multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient(f, b, T, top_k=None):
    """
    Compute smoothed gradient of max(conv(f,f)) using log-sum-exp.
    grad[m] = 2 * sum_j f[j] * w[j+m] (cross-correlation)
    """
    n = len(f)
    b_use = b.copy()
    if top_k is not None and top_k < len(b) and top_k >= 3:
        peaks = np.argpartition(b_use, -top_k)[-top_k:]
        mask = np.zeros(len(b_use), dtype=bool)
        mask[peaks] = True
        b_use[~mask] = -1e10

    # Softmax weights
    b_max = b_use.max()
    exp_vals = np.exp((b_use - b_max) / T)
    w = exp_vals / exp_vals.sum()

    # Cross-correlation: grad[m] = 2 * sum_{j=0}^{n-1} f[j] * w[j+m]
    fft_size = 1 << (2 * len(w) - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w, fft_size)
    xcorr = np.fft.irfft(F * np.conj(W), fft_size)
    grad = 2.0 * xcorr[:n]
    return grad


def lse_max_value(b, T, top_k=None):
    """Compute log-sum-exp approximation of max."""
    b_use = b
    if top_k is not None and top_k < len(b) and top_k >= 3:
        peaks = np.argpartition(b_use, -top_k)[-top_k:]
        b_use = b_use[peaks]
    b_max = b_use.max()
    return b_max + T * np.log(np.sum(np.exp((b_use - b_max) / T)))


def pairwise_swap_best(f, b, j, m, num_trials=30):
    """
    For a pairwise swap (j gets +delta, m gets -delta),
    find the best delta using the analytic quadratic model.
    Returns (best_delta, best_max).
    """
    n = len(f)
    # Linear coefficients: L[k] = 2*f[k-j] - 2*f[k-m] (with valid index checks)
    # For lag k: valid j' in [max(0,k-n+1), min(n-1,k)]
    L = np.zeros(2 * n - 1)
    # g_j[k] = 2*f[k-j] if 0 <= k-j < n
    valid_j = np.arange(0, 2 * n - 1)
    kj = valid_j - j
    mask_j = (kj >= 0) & (kj < n)
    L[mask_j] += 2.0 * f[kj[mask_j]]
    # g_m[k] = 2*f[k-m] if 0 <= k-m < n
    km = valid_j - m
    mask_m = (km >= 0) & (km < n)
    L[mask_m] -= 2.0 * f[km[mask_m]]

    # Quadratic corrections (nonzero for at most 4 lags)
    # Q[k] = -2*I{j+m=k, j!=m} + I{2j=k} + I{2m=k}
    # We'll just add these to b when evaluating

    # Feasible range for delta
    d_min = -f[j] + 1e-15
    d_max = f[m] - 1e-15
    if d_max <= d_min:
        return 0.0, b.max()

    # Evaluate objective at several delta values
    best_delta = 0.0
    best_obj = b.max()

    deltas = np.linspace(d_min, d_max, num_trials)
    # Also add some points near 0
    d0 = min(f[j], f[m]) * 0.01
    deltas = np.unique(np.concatenate([deltas, np.array([-d0*2, -d0, d0, 2*d0])]))
    deltas = deltas[(deltas >= d_min) & (deltas <= d_max)]

    for delta in deltas:
        # b_new[k] = b[k] + delta*L[k] + delta^2*Q[k]
        # Q is sparse, so approximate with just the linear term for speed
        # (the Q terms only affect 4 positions)
        b_new_max = np.max(b + delta * L)
        # Add quadratic correction for the 4 special positions
        corr = delta * delta
        q_positions = []
        if j + m < 2 * n - 1 and j != m:
            q_positions.append((j + m, -2.0 * corr))
        if 2 * j < 2 * n - 1:
            q_positions.append((2 * j, corr))
        if 2 * m < 2 * n - 1:
            q_positions.append((2 * m, corr))

        max_val = b_new_max
        for pos, qv in q_positions:
            val_at_pos = b[pos] + delta * L[pos] + qv
            if val_at_pos > max_val:
                max_val = val_at_pos

        if max_val < best_obj - 1e-15:
            best_obj = max_val
            best_delta = delta

    return best_delta, best_obj


def smoothed_nesterov_optimize(f0, time_limit, T0=500.0, T_min=0.5,
                                lr0=0.02, mom0=0.85, cooldown=0.9998,
                                top_k_frac=0.2, verbose=False):
    """
    Nesterov-accelerated smoothed subgradient descent on the simplex.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    T = T0
    lr = lr0
    mom = mom0
    y = f.copy()  # Nesterov lookahead
    v = np.zeros(n)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        y = f + mom * v

        # Compute smoothed gradient at y
        b_y = fast_convolve(y)
        top_k = max(10, int(n * top_k_frac))
        grad = smoothed_gradient(y, b_y, T, top_k=top_k)

        # Normalized gradient step
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            v = -lr * grad / (grad_norm + 0.1)
        else:
            v = np.zeros(n)

        f_new = project_simplex(y + v)

        # Update parameters
        T = max(T * cooldown, T_min)
        lr = max(lr * cooldown, 1e-5)
        mom = min(mom * 1.0001, 0.97)

        b_new = fast_convolve(f_new)
        obj = b_new.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1

        if no_improve > 150:
            # Adaptive restart
            pert_scale = 0.002 + 0.001 * min(no_improve / 100.0, 3.0)
            f = project_simplex(best_f + rng.standard_normal(n) * pert_scale)
            y = f.copy()
            v = np.zeros(n)
            no_improve = 0
            T = max(T * 3, T0 * 0.2)
            lr = max(lr * 2, lr0 * 0.3)
            if verbose:
                print(f"      Restart at iter {it}, pert={pert_scale:.4f}")

    return best_f, best_obj


def pairwise_cd_optimize(f0, time_limit, rng, num_pairs_per_iter=50):
    """
    Pairwise coordinate descent: randomly try swaps and keep improvements.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    b = fast_convolve(f)
    best_obj = b.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False

        for _ in range(num_pairs_per_iter):
            j, m = rng.integers(0, n, size=2)
            if j == m:
                continue

            delta, obj_val = pairwise_swap_best(f, b, j, m, num_trials=25)

            if delta != 0.0 and obj_val < best_obj - 1e-14:
                f[j] += delta
                f[m] -= delta
                f = np.maximum(f, 0.0)
                fs = f.sum()
                if fs < 1e-15:
                    f = best_f.copy()
                    b = fast_convolve(f)
                    continue
                f = f / fs
                b = fast_convolve(f)
                obj = b.max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_f = f.copy()
                    improved = True

        if not improved and it > 20:
            break

    return best_f, best_obj


def solve_active_lp(f, b, active_lags):
    """
    Solve LP: min t s.t. t >= b[k] + 2*(conv_dir)[k] for k in active_lags,
    conv_dir = direction such that sum(dir)=0, f+dir >= 0.
    Returns direction or None.
    """
    n = len(f)
    k_list = list(active_lags)
    m = len(k_list)
    if m == 0:
        return None

    # Variables: [t, d_0, ..., d_{n-1}]
    c = np.zeros(n + 1)
    c[0] = 1.0

    # Build constraint matrix for active lags
    A_ub = np.zeros((m, n + 1))
    A_ub[:, 0] = -1.0
    b_ub = -np.array([b[k] for k in k_list])

    for i, k in enumerate(k_list):
        # (f*d)[k] = sum_j f[j]*d[k-j] ... no wait
        # We linearize: (f+d)*(f+d) ≈ f*f + 2*f*d at lag k
        # (f*d)[k] = sum_j f[j] * d[k-j] + d[j] * f[k-j] ... 
        # Actually the linearization of (f+d)*(f+d)[k] at d=0 is:
        # b[k] + sum_j (f[j]*d[k-j] + d[j]*f[k-j]) = b[k] + 2*sum_j f[j]*d[k-j]
        # (using the fact that the cross term appears twice due to symmetry)
        # So constraint: t >= b[k] + 2*sum_j f[j]*d[k-j]
        # i.e., -t + 2*sum_j f[j]*d[k-j] <= -b[k]
        # Coefficient of d[j]: 2*f[k-j] if 0 <= k-j < n

        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                A_ub[i, j + 1] += 2.0 * f[kj]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[1:]
    except Exception:
        pass
    return None


def lp_fw_step(f, b, time_budget=2.0):
    """
    One LP-based Frank-Wolfe step using top active constraints.
    """
    n = len(f)
    t0 = time.monotonic()

    # Find top active lags
    top_k = min(20, 2 * n - 1)
    active = np.argpartition(b, -top_k)[-top_k:]
    active = np.sort(active)

    d = solve_active_lp(f, b, active)
    if d is None:
        return f, True

    # Project target to simplex
    target = f + d
    target = np.maximum(target, 0.0)
    ts = target.sum()
    if ts < 1e-15:
        return f, True
    target = target / ts

    # Check if at fixed point
    if np.max(np.abs(target - f)) < 1e-10:
        return f, True

    # Line search
    best_alpha = 0.0
    best_val = b.max()
    for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        f_try = (1 - alpha) * f + alpha * target
        f_try = np.maximum(f_try, 0)
        if f_try.sum() < 1e-15:
            continue
        f_try = f_try / f_try.sum()
        val = fast_convolve(f_try).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha

    if best_alpha > 1e-8:
        f = (1 - best_alpha) * f + best_alpha * target
        f = np.maximum(f, 0)
        f = f / f.sum()

    elapsed = time.monotonic() - t0
    converged = elapsed > time_budget
    return f, converged


def generate_diverse_inits(n, parent, rng, num_inits=5):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3 + 0.5))

    # Cosine hump
    for freq in [1, 2, 3]:
        c = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x / n)
        inits.append(c / c.sum())

    # Gaussian centered
    sigma = 0.3 * n
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
    inits.append(g / g.sum())

    # Edge-heavy
    e = 1.0 / (0.1 + np.abs(x - 0.5 * (n - 1)) / n)
    inits.append(e / e.sum())

    # Interpolated parent
    if parent is not None and len(parent) > 1:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Add noise to parent
    if parent is not None and len(parent) > 1:
        for scale in [0.005, 0.01, 0.02]:
            px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            px = px + rng.standard_normal(n) * scale
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                inits.append(px / ps)

    return inits[:num_inits + 4]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Multi-phase optimization combining:
    1. Diverse initialization
    2. Nesterov-accelerated smoothed subgradient (primary)
    3. Interleaved pairwise coordinate descent
    4. LP-based Frank-Wolfe for large jumps
    5. Final local search polish
    Uses time management to allocate budget across phases.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-15:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={parent_score:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        try_seq(parent, "uniform_start")

    # === Phase 1: Try multiple lengths and initializations ===
    print("\n=== Phase 1: Initialization ===")
    t1_start = time.monotonic()
    n_parent = len(parent)

    # Primary: optimize at parent length
    n_primary = n_parent

    # Quick evaluation of a few inits at primary length
    inits = generate_diverse_inits(n_primary, parent, rng, num_inits=4)
    for i, ini in enumerate(inits):
        if time.monotonic() > deadline - 600:
            break
        try_seq(ini, f"init{i}")

    # Try nearby lengths
    alt_lengths = [n_primary - 32, n_primary + 32, n_primary + 128]
    for n_try in alt_lengths:
        if n_try < 128 or time.monotonic() > deadline - 500:
            continue
        ini = generate_diverse_inits(n_try, parent, rng, num_inits=1)[0]
        try_seq(ini, f"len{n_try}")

    print(f"  Phase 1 best: {best_score:.10f} (n={len(best_seq)})")
    t1_time = time.monotonic() - t1_start
    print(f"  Phase 1 took {t1_time:.1f}s")

    # === Phase 2: Main Nesterov smoothed subgradient optimization ===
    print("\n=== Phase 2: Nesterov Smoothed Subgradient ===")
    remaining = deadline - time.monotonic() - 100
    t2_budget = min(remaining * 0.50, 800)
    n_work = len(best_seq)

    if t2_budget > 60:
        print(f"  n={n_work}, budget={t2_budget:.0f}s")

        # Multiple restarts with different temperatures
        num_restarts = 3
        t_per_restart = t2_budget / num_restarts

        for r in range(num_restarts):
            rem = deadline - time.monotonic() - 80
            if rem < 30:
                break

            if r == 0:
                f_start = best_seq.copy()
            else:
                # Perturbed restart
                scale = 0.005 * (r + 1)
                f_start = project_simplex(best_seq + rng.standard_normal(n_work) * scale)

            T0 = 500.0 / (r + 1)
            lr0 = 0.015 / (1 + 0.3 * r)

            t_budget = min(t_per_restart, rem - 30)
            if t_budget < 20:
                break

            a_opt, _ = smoothed_nesterov_optimize(
                f_start, time_limit=t_budget, T0=T0, lr0=lr0,
                mom0=0.85 + 0.02 * r, top_k_frac=0.15 + 0.05 * r,
                verbose=True
            )
            try_seq(a_opt, f"nesterov_r{r}")

            # Intersperse with pairwise CD
            rem = deadline - time.monotonic() - 60
            if rem > 30:
                t_cd = min(rem * 0.15, 60)
                a_opt, _ = pairwise_cd_optimize(best_seq, time_limit=t_cd, rng=rng)
                try_seq(a_opt, f"pairwise_r{r}")

        print(f"  Phase 2 best: {best_score:.10f}")

    # === Phase 3: LP-based Frank-Wolfe for larger jumps ===
    print("\n=== Phase 3: LP Frank-Wolfe ===")
    remaining = deadline - time.monotonic() - 80
    t3_budget = min(remaining * 0.30, 400)

    if t3_budget > 30:
        print(f"  budget={t3_budget:.0f}s")
        n_work = len(best_seq)
        b = fast_convolve(best_seq)

        it = 0
        f_curr = best_seq.copy()
        while time.monotonic() < deadline - 60:
            it += 1
            f_new, conv = lp_fw_step(f_curr, fast_convolve(f_curr), time_budget=5.0)
            if not conv and fast_convolve(f_new).max() < fast_convolve(f_curr).max() - 1e-14:
                f_curr = f_new
                try_seq(f_new, f"lpfw{it}")
            elif conv:
                break
            if it > 50:
                break

        print(f"  Phase 3 best: {best_score:.10f}")

    # === Phase 4: Intensive pairwise coordinate descent ===
    print("\n=== Phase 4: Pairwise CD Polish ===")
    remaining = deadline - time.monotonic() - 50
    t4_budget = min(remaining * 0.35, 300)

    if t4_budget > 20:
        print(f"  budget={t4_budget:.0f}s")
        n_work = len(best_seq)

        # Multiple rounds with increasing pair sampling
        for round_i in range(3):
            rem = deadline - time.monotonic() - 30
            if rem < 15:
                break
            t_each = min(rem / (3 - round_i), t4_budget / 3)
            num_pairs = 100 + 200 * round_i

            a_opt, _ = pairwise_cd_optimize(best_seq, time_limit=t_each,
                                            rng=np.random.default_rng(seed + round_i),
                                            num_pairs_per_iter=num_pairs)
            try_seq(a_opt, f"cd_polish{round_i}")

        print(f"  Phase 4 best: {best_score:.10f}")

    # === Phase 5: Fine Nesterov with very low temperature ===
    print("\n=== Phase 5: Fine Polish ===")
    remaining = deadline - time.monotonic() - 30
    t5_budget = min(remaining * 0.5, 150)

    if t5_budget > 15:
        n_work = len(best_seq)
        a_opt, _ = smoothed_nesterov_optimize(
            best_seq, time_limit=t5_budget, T0=50.0, T_min=0.1,
            lr0=0.005, mom0=0.95, cooldown=0.99995, top_k_frac=0.1
        )
        try_seq(a_opt, "fine_polish")

    # === Phase 6: Try perturbation + quick optimization ===
    print("\n=== Phase 6: Exploration ===")
    remaining = deadline - time.monotonic() - 20

    if remaining > 20:
        n_work = len(best_seq)
        for attempt in range(5):
            rem = deadline - time.monotonic() - 15
            if rem < 10:
                break
            scale = 0.001 * (attempt + 1)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_work) * scale)
            t_each = min(rem / 2, 20)
            a_opt, _ = smoothed_nesterov_optimize(
                f_pert, time_limit=t_each, T0=100.0, T_min=1.0,
                lr0=0.01, mom0=0.85, cooldown=0.9998, top_k_frac=0.2
            )
            try_seq(a_opt, f"explore{attempt}")

    # === Final ===
    if best_seq is not None:
        final_score = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
