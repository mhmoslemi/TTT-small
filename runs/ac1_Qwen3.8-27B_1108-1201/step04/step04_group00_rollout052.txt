**Another key idea: Use the "active set" approach more intelligently.**

In the LP-FW method, the LP constraint (f*g)[k] <= M is only active for the lags k where the convolution is near M. If we only include these "active" constraints in the LP, the LP becomes much cheaper (fewer constraints).

Specifically, at each step:
1. Compute conv = f*f
2. Find the lags where conv[k] > M * 0.9 (or some threshold)
3. Solve the LP with only these constraints (plus g >= 0)
4. This gives a direction that's "locally optimal" for the active constraints

This is a standard trick in constrained optimization (active set method).

**Even better: Use a "warm-started" active set LP.**

Keep track of which constraints are active across iterations. If the active set changes slowly, we can warm-start the LP solver.

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_optimize(f0, time_limit, lr=0.01, mom=0.9, temp_init=0.001, temp_growth=1.002, top_k=100):
    """Subgradient optimization of max convolution with simplex constraint."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_temp = temp_init
    cur_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        # Compute softmax weights over top-k lags
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        # Update with momentum
        v = mom * v - cur_lr * sg
        f_new = project_simplex(f + v)
        
        # Evaluate
        conv = fast_convolve(f_new)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        cur_temp *= temp_growth
        cur_lr *= 0.9999
        
        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 2**31)
            scale = min(0.01, 0.001 * (no_improve / 100))
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            no_improve = 0
            cur_temp = temp_init * (1 + 0.5 * no_improve)  # this is a bit hacky
            cur_temp = temp_init
            cur_lr = max(lr * 0.5, 0.001)
    
    return best_f, best_obj


def active_set_lpfw(f0, time_limit, active_frac=0.3, verbose=False):
    """LP-FW with active set constraint selection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Select active constraints (top fraction of lags by convolution value)
        n_active = max(10, int(active_frac * len(conv)))
        active_lags = np.argpartition(conv, -n_active)[-n_active:]
        
        # Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_lags, g >= 0
        # (f*g)[k] = sum_j f_j * g_{k-j}
        n_constraints = len(active_lags)
        A_ub = np.zeros((n_constraints, n))
        for ci, k in enumerate(active_lags):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[ci, j] = f[k - j]
        b_ub = np.full(n_constraints, M)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', 
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
        
        if not res.success:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    AS-LPFW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 50:
            break
    
    return best_f, best_obj


def coordinate_descent_v2(f0, time_limit, n_multipliers=8):
    """Improved coordinate descent with golden section search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    multipliers = np.array([0.3, 0.5, 0.7, 0.8, 1.2, 1.5, 2.0, 3.0])
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj  # use current best_obj as reference
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 2:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
