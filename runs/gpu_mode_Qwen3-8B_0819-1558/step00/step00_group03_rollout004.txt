import triton
import triton.language as tl
import torch

def custom_kernel(data):
    """
    TriMul implementation using Triton kernels for efficient pairwise operations.
    This kernel computes the "outgoing" version of the TriMul operator from AlphaFold3.

    The TriMul operation is defined as:
    out[i][j] = sum_k (left[i][k] * right[j][k])

    This implementation uses Triton's tensor operations and parallelism to efficiently 
    compute the pairwise summation across the sequence dimension (N), while leveraging
    the hardware capabilities of NVIDIA H100 (Hopper, sm_90) and TMA, wgmma, and thread-block clusters.

    Input:
    - `data`: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)

    Output:
    - output: [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    seqlen, bs = input_tensor.shape[1], input_tensor.shape[0]
    
    # Weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Initialize the output
    output = torch.zeros(bs, seqlen, seqlen, dim, device="cuda", dtype=torch.float32)

    # Apply LayerNorm
    input_scaled = input_tensor * norm_weight + norm_bias

    # Project input and gate
    @triton.jit
    def compute_left_right_gate(x, mask, out, left_proj_weight, right_proj_weight, left_gate_weight, right_gate_weight, out_gate_weight, to_out_norm_weight, to_out_norm_bias, to_out_weight, seqlen, dim, hidden_dim, bs):
        # We use a block-based approach for efficient memory access
        # We process each block of (BLOCK_SEQ, BLOCK_SEQ) in the input
        # Each kernel call is responsible for a block of the output

        # Define block sizes
        BLOCK_SEQ = 64
        BLOCK_DIM = 32

        # Determine the position in the output
        pid = tl.program_id(axis=0)
        pid_m = tl.program_id(axis=1)
        pid_n = tl.program_id(axis=2)

        # Compute the offset for the current block
        offs_i = pid_m * BLOCK_SEQ + tl.arange(0, BLOCK_SEQ)
        offs_j = pid_n * BLOCK_SEQ + tl.arange(0, BLOCK_SEQ)

        # Mask handling
        mask_block = tl.load(mask + (offs_i[:, None] * seqlen + offs_j[None, :]), mask_type="掩码")

        # Load input projections
        left_proj = tl.load(x + (offs_i[:, None] * seqlen + tl.arange(0, hidden_dim)[None, :]), mask=mask_block, num_stages=3)
        right_proj = tl.load(x + (offs_j[None, :] * seqlen + tl.arange(0, hidden_dim)[None, :]), mask=mask_block, num_stages=3)

        # Apply left and right gates
        left_gate = tl.load(left_gate_weight + (tl.arange(0, hidden_dim)[None, :] * dim + tl.arange(0, dim)[None, :]), num_stages=3)
        left_proj = left_proj * left_gate
        right_gate = tl.load(right_gate_weight + (tl.arange(0, hidden_dim)[None, :] * dim + tl.arange(0, dim)[None, :]), num_stages=3)
        right_proj = right_proj * right_gate

        # Compute the TriMul operation
        acc = tl.zeros((BLOCK_SEQ, BLOCK_DIM), dtype=tl.float32)
        for k in range(0, seqlen, BLOCK_SEQ):
            a = tl.load(x + (offs_i[:, None] * seqlen + k * seqlen + tl.arange(0, hidden_dim)[None, :]), mask=mask_block, num_stages=3)
            b = tl.load(x + (offs_j[None, :] * seqlen + k * seqlen + tl.arange(0, hidden_dim)[None, :]), mask=mask_block, num_stages=3)
            acc += tl.dot(a, b)
        acc = acc * tl.load(out_gate_weight + (tl.arange(0, hidden_dim)[None, :] * dim + tl.arange(0, dim)[None, :]), num_stages=3)

        # Normalize with LayerNorm
        to_out_norm_weight = tl.load(to_out_norm_weight + tl.arange(0, hidden_dim)[None, :], num_stages=3)
        to_out_norm_bias = tl.load(to_out_norm_bias + tl.arange(0, hidden_dim)[None, :], num_stages=3)
        acc = acc * to_out_norm_weight + to_out_norm_bias

        # Apply final output projection
        to_out_weight = tl.load(to_out_weight + (tl.arange(0, hidden_dim)[None, :] * dim + tl.arange(0, dim)[None, :]), num_stages=3)
        acc = acc @ to_out_weight

        # Store the result
        offs_out_i = pid * BLOCK_SEQ + tl.arange(0, BLOCK_SEQ)
        offs_out_j = pid * BLOCK_SEQ + tl.arange(0, BLOCK_SEQ)
        tl.store(out + (offs_out_i[:, None] * seqlen + offs_out_j[None, :]), acc, mask=mask_block)

    # Launch the kernel
    grid = (seqlen, seqlen, 1)
    compute_left_right_gate[grid](input_scaled, mask, output, left_proj_weight, right_proj_weight, left_gate_weight, right_gate_weight, out_gate_weight, to_out_norm_weight, to_out_norm_bias, to_out_weight, seqlen, dim, hidden_dim, bs)

    return output
