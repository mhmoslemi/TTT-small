import torch
import triton
import triton.language as tl
import math
import numpy as np

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    bs, seqlen = input_tensor.shape[0], input_tensor.shape[1]
    # The output shape is [bs, seqlen, seqlen, dim]
    # So we need to compute [bs, seqlen, seqlen, hidden_dim]
    # Then apply to_out norm and to_out weights
    
    # Extract weights
    norm_weight = weights["norm.weight"].contiguous()
    norm_bias = weights["norm.bias"].contiguous()
    left_proj_weight = weights["left_proj.weight"].contiguous()
    right_proj_weight = weights["right_proj.weight"].contiguous()
    left_gate_weight = weights["left_gate.weight"].contiguous()
    right_gate_weight = weights["right_gate.weight"].contiguous()
    out_gate_weight = weights["out_gate.weight"].contiguous()
    to_out_norm_weight = weights["to_out_norm.weight"].contiguous()
    to_out_norm_bias = weights["to_out_norm.bias"].contiguous()
    to_out_weight = weights["to_out.weight"].contiguous()
    
    # Prepare masks
    mask = mask.unsqueeze(-1) if mask is not None else torch.ones_like(input_tensor)
    mask = mask.to(input_tensor.device)
    
    # Output shape
    output = torch.empty((bs, seqlen, seqlen, hidden_dim), device=input_tensor.device)
    
    # Normalize input
    norm_weight = norm_weight.to(input_tensor.device)
    norm_bias = norm_bias.to(input_tensor.device)
    x = torch.nn.functional.layer_norm(input_tensor, norm_weight.shape, norm_weight, norm_bias, False)
    
    # Compute projections in Triton
    # First, left_proj and right_proj -> [bs, seqlen, seqlen, hidden_dim]
    left = torch.empty((bs, seqlen, seqlen, hidden_dim), device=input_tensor.device)
    right = torch.empty((bs, seqlen, seqlen, hidden_dim), device=input_tensor.device)
    left_gate = torch.empty((bs, seqlen, seqlen, hidden_dim), device=input_tensor.device)
    right_gate = torch.empty((bs, seqlen, seqlen, hidden_dim), device=input_tensor.device)
    out_gate = torch.empty((bs, seqlen, seqlen, hidden_dim), device=input_tensor.device)
    
    # Compute projections using triton
    left_proj_kernel = triton_heavy_parallel(left, x, left_proj_weight, mask, seqlen, hidden_dim, dim, bs, seqlen, seqlen)
    left_proj_kernel()
    
    right_proj_kernel = triton_heavy_parallel(right, x, right_proj_weight, mask, seqlen, hidden_dim, dim, bs, seqlen, seqlen)
    right_proj_kernel()
    
    left_gate_kernel = triton_heavy_parallel(left_gate, x, left_gate_weight, mask, seqlen, hidden_dim, dim, bs, seqlen, seqlen)
    left_gate_kernel()
    
    right_gate_kernel = triton_heavy_parallel(right_gate, x, right_gate_weight, mask, seqlen, hidden_dim, dim, bs, seqlen, seqlen)
    right_gate_kernel()
    
    out_gate_kernel = triton_heavy_parallel(out_gate, x, out_gate_weight, mask, seqlen, hidden_dim, dim, bs, seqlen, seqlen)
    out_gate_kernel()
    
    # Apply gates
    left = left * left_gate
    right = right * right_gate
    
    # Compute TriMul product using triton kernel
    out = triton_trimul(left, right, seqlen, hidden_dim, bs, seqlen, seqlen, output)
    out = out * out_gate
    
    # Apply LayerNorm on the output
    out = torch.nn.functional.layer_norm(out, to_out_norm_weight.shape, to_out_norm_weight, to_out_norm_bias, False)
    
    # Apply final linear projection
    out = torch.nn.functional.linear(out, to_out_weight)
    
    # Cast to float32 if needed
    out = out.to(torch.float32)
    
    return out

def triton_heavy_parallel(out, x, weights, mask, seqlen, hidden_dim, dim, bs, seqlen, seqlen):
    # This kernel computes out = x @ weights, then applies mask
    @triton.jit
    def kernel(out_ptr, x_ptr, weights_ptr, mask_ptr, bs, seqlen, hidden_dim, dim, BLOCK_SIZE: tl.constexpr):
        pid = tl.program_id(0)
        i = pid // (seqlen * seqlen)
        j = (pid // seqlen) % seqlen
        k = pid % seqlen
        
        # Load weights
        weights = tl.load(weights_ptr + k * dim + tl.arange(0, hidden_dim))
        weights = weights.to(tl.float32)
        
        # Load x
        x = tl.load(x_ptr + i * seqlen * dim + j * dim + tl.arange(0, dim))
        x = x.to(tl.float32)
        
        # Compute x @ weights
        x_weights = tl.dot(x, weights)
        
        # Apply mask
        mask = tl.load(mask_ptr + i * seqlen + j)
        mask = mask.to(tl.float32)
        x_weights = x_weights * mask
        
        # Store to out
        out = tl.load(out_ptr + i * seqlen * hidden_dim + j * hidden_dim + tl.arange(0, hidden_dim))
        out = out + x_weights
        tl.store(out_ptr + i * seqlen * hidden_dim + j * hidden_dim + tl.arange(0, hidden_dim), out)
        
    num_warps = 4
    grid = (bs * seqlen * seqlen,)
    kernel.launch(grid, num_warps=num_warps)

def triton_trimul(left, right, seqlen, hidden_dim, bs, seqlen, seqlen, output):
    @triton.jit
    def kernel(out_ptr, left_ptr, right_ptr, bs, seqlen, hidden_dim, BLOCK_SIZE: tl.constexpr):
        pid = tl.program_id(0)
        i = pid // (seqlen * seqlen)
        j = (pid // seqlen) % seqlen
        k = pid % seqlen
        
        # Load left, right
        left = tl.load(left_ptr + i * seqlen * hidden_dim + k * hidden_dim + tl.arange(0, hidden_dim))
        right = tl.load(right_ptr + j * seqlen * hidden_dim + k * hidden_dim + tl.arange(0, hidden_dim))
        
        # Compute left @ right
        result = tl.dot(left, right)
        
        # Store to output
        tl.store(out_ptr + i * seqlen * hidden_dim + j * hidden_dim + tl.arange(0, hidden_dim), result)
        
    num_warps = 8
    grid = (bs * seqlen * seqlen,)
    kernel.launch(grid, num_warps=num_warps)

"""
This kernel implements the "outgoing" version of the TriMul operation used in AlphaFold3 and related
protein structure prediction models.

Key steps:
1. Normalize input using LayerNorm
2. Compute four separate projection operations (left_proj, right_proj, left_gate, right_gate)
3. Apply gates to the projections
4. Compute the TriMul product using the einsum operation, which is implemented as a triton kernel
5. Apply the final output LayerNorm and linear projection

The kernel is optimized for NVIDIA H100 (sm_90), utilizing TMA, wgmma, and thread-block clusters for optimal performance.
It supports mask application to filter out certain entries in the output tensor.
"""
