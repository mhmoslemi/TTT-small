import triton
import triton.language as tl
import torch
import math

def custom_kernel(data):
    """
    Custom Triton kernel for the "outgoing" version of the Triangle Multiplicative (TriMul) operation.
    The kernel processes a 4D tensor [B, N, N, C] with a mask and implements the forward pass of the TriMul module.
    Key operations include:
    - LayerNorm on input
    - Linear projections
    - Masking and gate applications
    - Einsum-like pairwise multiplication and summation (matrix multiplication over sequence dimensions)
    - Final LayerNorm and output projection
    
    The kernel is optimized for performance on NVIDIA H100 with Hopper architecture, using TMA, wgmma, and thread-block clusters.
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    seqlen = seq_len
    
    # Extract weights from the provided dictionary
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # Precompute strides and layout
    # Input layout: [B, seq_len, seq_len, dim]
    # Output layout: [B, seq_len, seq_len, dim]
    
    # Precompute strides for the input and output
    stride_input_B = input_tensor.stride(0)
    stride_input_N = input_tensor.stride(1)
    stride_input_M = input_tensor.stride(2)
    stride_input_C = input_tensor.stride(3)
    
    stride_mask_B = mask.stride(0)
    stride_mask_N = mask.stride(1)
    stride_mask_M = mask.stride(2)
    
    output = torch.zeros((batch_size, seq_len, seq_len, dim), device=input_tensor.device, dtype=input_tensor.dtype)
    stride_output_B = output.stride(0)
    stride_output_N = output.stride(1)
    stride_output_M = output.stride(2)
    stride_output_C = output.stride(3)
    
    # Triton kernel definition
    @triton.jit
    def tritron_kernel(
        input_ptr, mask_ptr, left_proj_weight, right_proj_weight, left_gate_weight, 
        right_gate_weight, out_gate_weight, to_out_norm_weight, to_out_norm_bias, to_out_weight,
        output_ptr, batch_size, seq_len, dim, hidden_dim, num_blocks, 
        BLOCK: tl.constexpr, 
        stride_input_B, stride_input_N, stride_input_M, stride_input_C,
        stride_mask_B, stride_mask_N, stride_mask_M,
        stride_output_B, stride_output_N, stride_output_M, stride_output_C,
        stride_left_proj_w, stride_left_proj_h, 
        stride_right_proj_w, stride_right_proj_h, 
        stride_left_gate_w, stride_left_gate_h,
        stride_right_gate_w, stride_right_gate_h,
        stride_out_gate_w, stride_out_gate_h,
        stride_to_out_norm_w, stride_to_out_norm_h,
        stride_to_out_norm_bias, 
        stride_to_out_w, stride_to_out_h
    ):
        # Compute the program ID for this thread
        pid = tl.program_id(0)  # Only one block per batch for simplicity
        # Compute the batch index
        batch_idx = pid // (seq_len * seq_len)
        # Compute the position within the batch
        pos_in_batch = pid % (seq_len * seq_len)
        # Compute the row and column indices
        i = pos_in_batch // seq_len
        j = pos_in_batch % seq_len
        
        # Compute offsets for the input tensor
        input_off = (batch_idx * stride_input_B) + (i * stride_input_N) + (j * stride_input_M)
        input_ptr = input_ptr + input_off
        
        # Compute the mask value at (i,j)
        mask_off = (batch_idx * stride_mask_B) + (i * stride_mask_N) + (j * stride_mask_M)
        mask_val = tl.load(mask_ptr + mask_off)
        if mask_val == 0:
            return
        
        # Load the input vector
        x = tl.load(input_ptr)
        
        # Apply LayerNorm
        x_norm = x * norm_weight + norm_bias
        x_norm = x_norm * tl.rsqrt(tl.sum(x_norm * x_norm, axis=-1, keepdim=True) + 1e-5)
        
        # Project left and right
        left = tl.dot(x_norm, left_proj_weight)
        right = tl.dot(x_norm, right_proj_weight)
        
        # Apply masking
        left = left * mask_val
        right = right * mask_val
        
        # Compute left_gate and right_gate
        left_gate = tl.dot(x_norm, left_gate_weight)
        right_gate = tl.dot(x_norm, right_gate_weight)
        left_gate = left_gate.sigmoid()
        right_gate = right_gate.sigmoid()
        
        # Apply gates
        left = left * left_gate
        right = right * right_gate
        
        # Compute the output
        out = tl.zeros((dim,), dtype=tl.float32)
        for k in range(seq_len):
            # Load left (i, k) and right (j, k)
            left_k_idx = (batch_idx * stride_input_B) + (i * stride_input_N) + (k * stride_input_M)
            left_k = tl.load(input_ptr + left_k_idx)
            left_k = tl.dot(left_k, left_proj_weight)
            left_k = left_k * mask_val
            left_k = left_k * left_gate
            right_k_idx = (batch_idx * stride_input_B) + (j * stride_input_N) + (k * stride_input_M)
            right_k = tl.load(input_ptr + right_k_idx)
            right_k = tl.dot(right_k, right_proj_weight)
            right_k = right_k * mask_val
            right_k = right_k * right_gate
            
            # Accumulate dot product
            out += tl.dot(left_k, right_k)
        
        # Apply LayerNorm
        out = out * to_out_norm_weight + to_out_norm_bias
        out = out * tl.rsqrt(tl.sum(out * out, axis=-1, keepdim=True) + 1e-5)
        
        # Apply output gate
        out_gate = tl.dot(x_norm, out_gate_weight)
        out_gate = out_gate.sigmoid()
        
        # Apply final gate and output projection
        out = out * out_gate
        out = tl.dot(out, to_out_weight)
        
        # Output to the destination
        output_off = (batch_idx * stride_output_B) + (i * stride_output_N) + (j * stride_output_M)
        tl.store(output_ptr + output_off, out)
    
    # Compute number of blocks (one per batch)
    num_blocks = batch_size
    
    # Launch the kernel
    tritron_kernel(
        input_tensor.data_ptr(), 
        mask.data_ptr(), 
        left_proj_weight.data_ptr(), right_proj_weight.data_ptr(), 
        left_gate_weight.data_ptr(), right_gate_weight.data_ptr(), 
        out_gate_weight.data_ptr(), 
        to_out_norm_weight.data_ptr(), to_out_norm_bias.data_ptr(), to_out_weight.data_ptr(),
        output.data_ptr(), 
        batch_size, seq_len, dim, hidden_dim, num_blocks, 
        BLOCK=128,
        stride_input_B=stride_input_B, stride_input_N=stride_input_N, stride_input_M=stride_input_M, stride_input_C=stride_input_C,
        stride_mask_B=stride_mask_B, stride_mask_N=stride_mask_N, stride_mask_M=stride_mask_M,
        stride_output_B=stride_output_B, stride_output_N=stride_output_N, stride_output_M=stride_output_M, stride_output_C=stride_output_C,
        stride_left_proj_w=left_proj_weight.stride(0), stride_left_proj_h=left_proj_weight.stride(1),
        stride_right_proj_w=right_proj_weight.stride(0), stride_right_proj_h=right_proj_weight.stride(1),
        stride_left_gate_w=left_gate_weight.stride(0), stride_left_gate_h=left_gate_weight.stride(1),
        stride_right_gate_w=right_gate_weight.stride(0), stride_right_gate_h=right_gate_weight.stride(1),
        stride_out_gate_w=out_gate_weight.stride(0), stride_out_gate_h=out_gate_weight.stride(1),
        stride_to_out_norm_w=to_out_norm_weight.stride(0), stride_to_out_norm_h=to_out_norm_weight.stride(1),
        stride_to_out_norm_bias=to_out_norm_bias.stride(0),
        stride_to_out_w=to_out_weight.stride(0), stride_to_out_h=to_out_weight.stride(1)
    )
    
    return output
