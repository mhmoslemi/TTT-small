def apply_linear(x, weight):
    """
    Apply a linear transformation (x @ W)
    """
    return torch.matmul(x, weight)

def apply_layer_norm(x, weight, bias, dim):
    """
    Apply LayerNorm to the last dimension
    """
    mean = torch.mean(x, dim=dim, keepdim=True)
    var = torch.var(x, dim=dim, keepdim=True, unbiased=False)
    std = torch.sqrt(var + 1e-6)
    x = (x - mean) / std * weight + bias
    return x
