def TriMulKernel(config, input_tensor, mask, weights, output):
    """
    Main Triton kernel for TriMul operation.
    This kernel uses shared memory and fusions to efficiently compute the TriMul forward pass.
    """
    SEQLen = config["seqlen"]
    BS = config["bs"]
    DIM = config["dim"]
    HIDDEN = config["hidden_dim"]
    TILE = 64

    # Input and output strides
    input_strides = (input_tensor.stride(0), input_tensor.stride(1), input_tensor.stride(2), input_tensor.stride(3))
    output_strides = (output.stride(0), output.stride(1), output.stride(2), output.stride(3))

    # Compute offset for current thread
    pid = tl.program_id(axis=0)  # batch id
    pid_m = tl.program_id(axis=1)
    pid_n = tl.program_id(axis=2)
    pid_k = tl.program_id(axis=3)

    # Block dimensions
    offset_m = pid_m * TILE + tl.arange(0, TILE)
    offset_n = pid_n * TILE + tl.arange(0, TILE)
    offset_k = pid_k * TILE + tl.arange(0, TILE)

    # Ensure that the block fits within the dimension
    offset_m = tl.where(offset_m >= SEQLen, offset_m, offset_m)
    offset_n = tl.where(offset_n >= SEQLen, offset_n, offset_n)
    offset_k = tl.where(offset_k >= SEQLen, offset_k, offset_k)

    # Shared memory for intermediate results
    shared_left = tl.zeros((TILE, HIDDEN), dtype=tl.float32)
    shared_right = tl.zeros((TILE, HIDDEN), dtype=tl.float32)
    shared_output = tl.zeros((TILE, DIM), dtype=tl.float32)

    # Load input data
    for k in range(TILE):
        shared_left[k, :] = input_tensor[pid, offset_m, offset_k, :]

    # Load mask
    mask_load = mask[pid, offset_m, offset_k].to(tl.float32)

    # Apply LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    input_norm = (input_tensor - norm_bias) * norm_weight
    input_norm = input_norm * mask_load

    # Left projection
    left_proj_weight = weights["left_proj.weight"]
    left_proj = shared_left @ left_proj_weight.T
    left_proj = left_proj * mask_load

    # Right projection
    right_proj_weight = weights["right_proj.weight"]
    right_proj = shared_left @ right_proj_weight.T
    right_proj = right_proj * mask_load

    # Left gate
    left_gate_weight = weights["left_gate.weight"]
    left_gate = shared_left @ left_gate_weight.T
    left_gate = left_gate.sigmoid()

    # Right gate
    right_gate_weight = weights["right_gate.weight"]
    right_gate = shared_left @ right_gate_weight.T
    right_gate = right_gate.sigmoid()

    # Apply gates
    left_proj = left_proj * left_gate
    right_proj = right_proj * right_gate

    # Compute output
    for i in range(TILE):
        for j in range(TILE):
            output[pid, i, j, :] = tl.dot(left_proj[i, :], right_proj[j, :])

    to_out_weight = weights["to_out.weight"]
    output = output @ to_out_weight.T

    for i in range(TILE):
        for j in range(TILE):
            output[pid, i, j, :] = output[i, j, :]
