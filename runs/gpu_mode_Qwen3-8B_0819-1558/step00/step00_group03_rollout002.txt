import triton
import triton.language as tl
import torch
import math
import sys

def custom_kernel(data):
    """
    Implements the "outgoing" version of the Triangle Multiplicative Update (TriMul) operator.
    This forward pass involves multiple projections, gating, and a pairwise multiplication sum.
    The implementation is optimized using Triton for performance on NVIDIA H100 (Hopper, sm_90).
    The kernel ensures correctness across different input configurations.
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Output tensor initialization
    output = torch.zeros(batch_size, seq_len, seq_len, dim, device=input_tensor.device, dtype=input_tensor.dtype)

    # Function to perform LayerNorm
    @triton.jit
    def layer_norm(x_ptr, x_scale_ptr, x_bias_ptr, x_out_ptr, N, C, BLOCK: tl.constexpr):
        pid = tl.program_id(axis=0)
        off_n = pid * BLOCK + tl.arange(0, BLOCK)
        mask = off_n < N
        x = tl.load(x_ptr + off_n, mask=mask)
        x = x - tl.load(x_scale_ptr)
        x = x * tl.load(x_scale_ptr + off_n, mask=mask)
        x = x + tl.load(x_bias_ptr + off_n, mask=mask)
        tl.store(x_out_ptr + off_n, x, mask=mask)

    # Function to perform projections and gates
    @triton.jit
    def apply_linear(x_ptr, w_ptr, x_out_ptr, N, C, BLOCK: tl.constexpr):
        pid = tl.program_id(axis=0)
        off_n = pid * BLOCK + tl.arange(0, BLOCK)
        mask = off_n < N
        x = tl.load(x_ptr + off_n, mask=mask)
        w = tl.load(w_ptr + off_n, mask=mask)
        x = x * w
        tl.store(x_out_ptr + off_n, x, mask=mask)

    # Function to apply sigmoid (approximate)
    @triton.jit
    def sigmoid(x_ptr, x_out_ptr, N, BLOCK: tl.constexpr):
        pid = tl.program_id(axis=0)
        off_n = pid * BLOCK + tl.arange(0, BLOCK)
        mask = off_n < N
        x = tl.load(x_ptr + off_n, mask=mask)
        x = 1 / (1 + tl.exp(-x))
        tl.store(x_out_ptr + off_n, x, mask=mask)

    # Function to compute the TriMul operation (i, j, k) and sum over k
    @triton.jit
    def tri_mul(left_ptr, right_ptr, mask_ptr, out_ptr, N, C, BLOCK: tl.constexpr):
        pid = tl.program_id(axis=0)
        i = pid * BLOCK + tl.arange(0, BLOCK)
        mask = tl.load(mask_ptr + i)
        mask = mask == 1
        i_mask = i < N
        i_mask &= mask
        i_vals = tl.where(i_mask, i, 0)
        i_offsets = tl.arange(0, BLOCK)

        for j in range(N):
            j_offsets = tl.arange(0, BLOCK)
            for k in range(N):
                left = tl.load(left_ptr + i_vals + k * N + j_offsets) * tl.load(mask_ptr + i_vals + k * N)
                right = tl.load(right_ptr + j_vals + k * N + i_offsets) * tl.load(mask_ptr + j_vals + k * N)
                dot = tl.sum(left * right)
                tl.store(out_ptr + i_vals + j_offsets, dot)

    # Apply LayerNorm to the input
    layer_norm(input_tensor, norm_weight, norm_bias, input_tensor, seq_len, dim, BLOCK=128)

    # Project input to hidden dimension for left and right
    left = torch.zeros(batch_size, seq_len, seq_len, hidden_dim, device=input_tensor.device)
    right = torch.zeros(batch_size, seq_len, seq_len, hidden_dim, device=input_tensor.device)
    apply_linear(input_tensor, left_proj_weight, left, seq_len, hidden_dim, BLOCK=128)
    apply_linear(input_tensor, right_proj_weight, right, seq_len, hidden_dim, BLOCK=128)

    # Apply mask to left and right
    mask = mask.unsqueeze(-1)
    left = left * mask
    right = right * mask

    # Apply gates
    left_gate = torch.zeros(batch_size, seq_len, seq_len, hidden_dim, device=input_tensor.device)
    right_gate = torch.zeros(batch_size, seq_len, seq_len, hidden_dim, device=input_tensor.device)
    apply_linear(input_tensor, left_gate_weight, left_gate, seq_len, hidden_dim, BLOCK=128)
    apply_linear(input_tensor, right_gate_weight, right_gate, seq_len, hidden_dim, BLOCK=128)
    apply_linear(input_tensor, out_gate_weight, left_gate, seq_len, hidden_dim, BLOCK=128)

    # Apply sigmoid to gates
    sigmoid(left_gate, left_gate, seq_len, BLOCK=128)
    sigmoid(right_gate, right_gate, seq_len, BLOCK=128)
    sigmoid(left_gate, left_gate, seq_len, BLOCK=128)

    # Element-wise multiplication of left and right with their gates
    left = left * left_gate
    right = right * right_gate

    # Compute TriMul operation
    out = torch.zeros(batch_size, seq_len, seq_len, hidden_dim, device=input_tensor.device)
    tri_mul(left, right, mask, out, seq_len, hidden_dim, BLOCK=128)

    # Apply LayerNorm to the output of TriMul
    layer_norm(out, to_out_norm_weight, to_out_norm_bias, out, seq_len, hidden_dim, BLOCK=128)

    # Apply final gate and linear projection
    out = out * left_gate
    output = torch.matmul(out, to_out_weight)

    return output
