import torch
import triton
import triton.language as tl
import math
import numpy as np

def custom_kernel(data):
    """
    Implementation of the outgoing version of the Triangle Multiplicative Update (TriMul) operator.
    This kernel is used in protein structure prediction models like AlphaFold3 and Protenix.
    The kernel computes the forward pass of the TriMul operator efficiently using Triton.
    The output is of the same shape as the input tensor [bs, seq_len, seq_len, dim], 
    and is computed using fused operations for speed and memory efficiency.

    Args:
        data: Tuple containing the input tensors, mask, and model weights.

    Returns:
        output: Processed tensor of shape [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Compute the output tensor
    out = torch.zeros((bs, seq_len, seq_len, dim), device=input_tensor.device, dtype=input_tensor.dtype)
    out = _triton_trimul(
        input_tensor,
        mask,
        norm_weight,
        norm_bias,
        left_proj_weight,
        right_proj_weight,
        left_gate_weight,
        right_gate_weight,
        out_gate_weight,
        to_out_norm_weight,
        to_out_norm_bias,
        to_out_weight,
        seq_len,
        dim,
        hidden_dim,
    )
    return out

@triton.jit
def _triton_trimul(
    input_ptr, mask_ptr, norm_weight_ptr, norm_bias_ptr,
    left_proj_weight_ptr, right_proj_weight_ptr, left_gate_weight_ptr,
    right_gate_weight_ptr, out_gate_weight_ptr, to_out_norm_weight_ptr, to_out_norm_bias_ptr,
    to_out_weight_ptr, seq_len, dim, hidden_dim,
    BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
    N: tl.constexpr, D: tl.constexpr, H: tl.constexpr
):
    # Each thread handles one output element
    # (i, j) is the output index
    # (k) is the loop over sequence indices
    # (d) is the loop over features
    # The kernel is designed to process each (i, j, d) tuple with a single thread

    # i and j are the outer indices for output (i, j)
    # These are determined via thread index
    pid = tl.program_id(axis=0)
    i = pid % seq_len
    j = pid // seq_len

    # d is the feature dimension
    d_offsets = tl.arange(0, D)
    d = d_offsets % dim  # ensure d is within [0, dim)

    # Compute mask
    mask_val = tl.load(mask_ptr + i * seq_len + j)
    if mask_val == 0:
        return

    # Initialize output (i, j, d)
    out = tl.zeros((D,), dtype=tl.float32)

    # Load all feature weights
    norm_weight = tl.load(norm_weight_ptr + d)
    norm_bias = tl.load(norm_bias_ptr + d)
    left_proj_weight = tl.load(left_proj_weight_ptr + d)
    right_proj_weight = tl.load(right_proj_weight_ptr + d)
    left_gate_weight = tl.load(left_gate_weight_ptr + d)
    right_gate_weight = tl.load(right_gate_weight_ptr + d)
    out_gate_weight = tl.load(out_gate_weight_ptr + d)
    to_out_norm_weight = tl.load(to_out_norm_weight_ptr + d)
    to_out_norm_bias = tl.load(to_out_norm_bias_ptr + d)
    to_out_weight = tl.load(to_out_weight_ptr + d)

    # Normalize input
    input_val = tl.load(input_ptr + i * seq_len * dim + j * dim + d)
    input_norm = (input_val - norm_bias) * norm_weight

    # Compute left and right projections
    left_proj = input_norm * left_proj_weight
    right_proj = input_norm * right_proj_weight

    # Compute gates
    left_gate = (input_norm * left_gate_weight).sigmoid()
    right_gate = (input_norm * right_gate_weight).sigmoid()
    out_gate = (input_norm * out_gate_weight).sigmoid()

    # Compute left and right with gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # For all k in [0, seq_len)
    for k in range(seq_len):
        # Mask check
        mask_val_k = tl.load(mask_ptr + i * seq_len + k)
        if mask_val_k == 0:
            continue

        # Load input elements for current k
        input_k = tl.load(input_ptr + i * seq_len * dim + k * dim + d)
        input_k_norm = (input_k - norm_bias) * norm_weight
        input_k_left = input_k_norm * left_proj_weight
        input_k_right = input_k_norm * right_proj_weight

        # Compute left and right for this k with gates
        input_k_left_gate = (input_k_norm * left_gate_weight).sigmoid()
        input_k_right_gate = (input_k_norm * right_gate_weight).sigmoid()
        input_k_left = input_k_left * input_k_left_gate
        input_k_right = input_k_right * input_k_right_gate

        # Compute the product and accumulate
        val = input_k_left * input_k_right
        out += val

    # Apply LayerNorm
    out = (out - to_out_norm_bias) * to_out_norm_weight
    # Apply output gate
    out = out * out_gate
    # Final projection
    out = out * to_out_weight

    # Write output to memory
    out_ptr = input_ptr + i * seq_len * dim + j * dim + d
    tl.store(out_ptr, out)

def _triton_trimul_kernel_launcher(
    input_tensor, mask, norm_weight, norm_bias,
    left_proj_weight, right_proj_weight, left_gate_weight,
    right_gate_weight, out_gate_weight, to_out_norm_weight,
    to_out_norm_bias, to_out_weight, seq_len, dim, hidden_dim
):
    # Determine block size for the kernel
    # We use a block size equal to the sequence length
    # This may need to be adjusted based on the hardware and memory constraints
    N = seq_len
    D = dim
    H = hidden_dim

    # The number of threads per block is equal to the number of elements in the output
    num_threads = N * N * D
    grid = (num_threads,)

    # Launch the kernel
    _triton_trimul[grid](
        input_tensor, mask, norm_weight, norm_bias,
        left_proj_weight, right_proj_weight, left_gate_weight,
        right_gate_weight, out_gate_weight, to_out_norm_weight,
        to_out_norm_bias, to_out_weight, seq_len, dim, hidden_dim,
        N, D, H
    )

def _triton_trimul(
    input_tensor, mask, norm_weight, norm_bias,
    left_proj_weight, right_proj_weight, left_gate_weight,
    right_gate_weight, out_gate_weight, to_out_norm_weight,
    to_out_norm_bias, to_out_weight, seq_len, dim, hidden_dim
):
    # Launch the kernel
    _triton_trimul_kernel_launcher(
        input_tensor, mask, norm_weight, norm_bias,
        left_proj_weight, right_proj_weight, left_gate_weight,
        right_gate_weight, out_gate_weight, to_out_norm_weight,
        to_out_norm_bias, to_out_weight, seq_len, dim, hidden_dim
    )
    return input_tensor
