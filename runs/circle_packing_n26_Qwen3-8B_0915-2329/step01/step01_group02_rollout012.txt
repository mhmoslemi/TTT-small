import numpy as np
import cvxpy as cp

def run_packing():
    # Define the variables
    n = 26
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # Objective: Maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Each circle remains within the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints += [x - r >= -1e-12]
        constraints += [x + r <= 1 + 1e-12]
        constraints += [y - r >= -1e-12]
        constraints += [y + r <= 1 + 1e-12]

    # No two circles overlap
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = cp.sqrt(dx ** 2 + dy ** 2)
            constraints += [dist >= radii[i] + radii[j] - 1e-12]

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)

    # Use SCS solver (for non-convex problems, might need Nesterov's method or other)
    problem.solve(solver=cp.SCS, verbose=False)

    # Check for success
    if not problem.status in ["optimal", "optimal_inaccurate"]:
        raise ValueError(f"Optimization failed: {problem.status}")

    # Extract the centers and radii
    centers = centers.value

    sum_radii = np.sum(radii)
