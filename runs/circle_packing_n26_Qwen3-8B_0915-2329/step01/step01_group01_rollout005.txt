import numpy as np
import math
from scipy.optimize import minimize

def compute_distance(a, b):
    return np.sqrt((a[0] - b[0])**2 + (a[1] - b[1])**2)

def hexagonal_grid(n):
    """Generate a hexagonal grid of n circles in a square."""
    if n < 1:
        return []
    # Generate a triangular (hexagonal) grid of size sqrt(n)
    top = math.ceil(math.sqrt(n))
    bottom = math.floor(math.sqrt(n))
    grid = []
    y = 0
    for row in range(top):
        if row == 0:
            count = 5  # First row has 5 circles
        elif row == 1:
            count = 4  # Second row has 4 circles
        elif row == top - 1:
            count = 2  # Last row has 2 circles
        else:
            count = 3
        x = (row % 2) * 0.5
        for col in range(count):
            grid.append((x + col * 0.9, y))
        y += 0.6
    return grid

def run_packing():

    centers = hexagonal_grid(26)

    radii = np.full(len(centers), 0.1)

    def objective(x):

        x_pos = x[:len(centers) * 2].reshape((len(centers), 2))
        x_rad = x[len(centers) * 2:]

        new_radii = radii + x_rad

        # Clip to [0, 1] square
        new_centers = np.clip(new_centers, 0, 1)
        new_radii = np.clip(new_radii, 0, 1)

        # Compute valid radii considering boundaries
        valid_radii = []
        for i in range(len(new_centers)):
            x, y = new_centers[i]
            r = new_radii[i]
            dist_to_left = x - r
            dist_to_right = 1 - x - r
            dist_to_bottom = y - r
            dist_to_top = 1 - y - r
            if dist_to_left < 0 or dist_to_right < 0 or dist_to_bottom < 0 or dist_to_top < 0:
                valid_radii.append(0)
            else:
                valid_radii.append(r)

        # Check for circle overlaps
        for i in range(len(new_centers)):
            for j in range(i + 1, len(new_centers)):
                dist = compute_distance(new_centers[i], new_centers[j])
                if dist < valid_radii[i] + valid_radii[j] - 1e-8:
                    valid_radii[i] = 0
                    valid_radii[j] = 0

        return -np.sum(valid_radii)  # Want to maximize sum of radii

    # Bounds: Positions are in [0, 1], radii are in [0, 0.5]
    bounds = []
    for _ in range(len(centers) * 2):
        bounds.append((0, 1))
    for _ in range(len(centers)):
        bounds.append((0, 0.5))

    result = minimize(

        x0=np.zeros(len(bounds)),
        method='SLSQP',
        bounds=bounds,
        constraints=[],
