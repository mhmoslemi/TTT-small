from scipy.optimize import minimize
from scipy.spatial import distance
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    # We'll start with a hexagonal grid for 25 circles and one extra
    # We'll generate the initial positions
    centers = []
    radii = []

    # Generate a hexagonal grid of 5 rows
    for i in range(5):

            centers.append([x, y])
            # The radii will be adjusted during optimization

    # Add the 26th circle near a corner (e.g., bottom-left)
    centers.append([0.05, 0.05])
    radii = [0.05] * 26

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Objective function: maximize the sum of radii
    def objective(x):
        # x is a flat array of [x1, y1, r1, x2, y2, r2, ...]
        # Extract centers and radii
        n = len(centers)
        flat_centers = x[:n * 2].reshape(n, 2)
        flat_radii = x[n * 2:].reshape(n)
        return -np.sum(flat_radii)  # Minimize negative of sum to maximize

    # Constraints: non-overlap and boundary
    def constraint_non_overlap(x):
        n = len(centers)
        flat_centers = x[:n * 2].reshape(n, 2)
        flat_radii = x[n * 2:].reshape(n)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((flat_centers[i] - flat_centers[j]) ** 2))
                if dist < flat_radii[i] + flat_radii[j] - 1e-12:

    def constraint_boundary(x):

        flat_centers = x[:n * 2].reshape(n, 2)

            x, y = flat_centers[i]

            if (x - r < 0 - 1e-12 or x + r > 1 + 1e-12 or

    # Optimization bounds
    bounds = []

        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 1.0)])  # x, y, r for each circle

    # Optimization
    result = minimize(
        fun=objective,
        x0=np.concatenate([centers.flatten(), radii]),
        method='L-BFGS-B',

    return final_centers, final_radii, final_sum
