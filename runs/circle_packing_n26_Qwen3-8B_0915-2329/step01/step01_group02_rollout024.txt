import scipy.optimize as opt
from scipy.spatial.distance import pdist, squareform

def create_hexagonal_grid(n, spacing):
    centers = []
    for i in range(n):
        row = i // spacing
        col = i % spacing
        x = col + 0.5 * (row % 2)
        y = row + 0.5
        centers.append([x, y])
    return np.array(centers)
