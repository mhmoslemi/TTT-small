import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial configuration: Hexagonal grid packing with fixed cell size
    num_circles = 26
    rows = 5
    cols = 5
    # Start with all radii = 0.05 for initial check
    radii_initial = np.full(num_circles, 0.05, dtype=np.float64)
    centers = np.zeros((num_circles, 2), dtype=np.float64)

    # Build initial grid
    radius = 0.05
    col_spacing = 2 * radius
    row_spacing = np.sqrt(3) * radius
    for i in range(rows):
        for j in range(cols):
            col_idx = j
            row_idx = i
            x = col_idx * col_spacing + radius
            y = row_idx * row_spacing + radius
            centers[i * cols + j] = [x, y]

    # Optimization parameters
    bounds = []
    for _ in range(num_circles):
        bounds.append((0.0, 1.0))  # x
        bounds.append((0.0, 1.0))  # y
    bounds = np.array(bounds).reshape(-1, 2)

    def objective(radii):
        return -np.sum(radii)

    def constraints_func(x):
        # x contains all variables: [x_0, y_0, r_0, x_1, y_1, r_1, ..., x_n, y_n, r_n]
        # reshape into centers and radii
        n = num_circles
        centers_vars = x[:2 * n].reshape(n, 2)
        radii_vars = x[2 * n:].reshape(n)
        # Check boundary constraints
        for i in range(n):
            x, y = centers_vars[i]
            r = radii_vars[i]
            if not (0 - 1e-12 <= x - r <= 1 + 1e-12 and 0 - 1e-12 <= y - r <= 1 + 1e-12):
                return False
        # Check non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_vars[i, 0] - centers_vars[j, 0]
                dy = centers_vars[i, 1] - centers_vars[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_vars[i] + radii_vars[j] - 1e-12:
                    return False
        return True

    # Prepare initial guess
    initial_guess = np.concatenate([centers.flatten(), radii_initial])

    # Define constraints
    cons = {'type': 'constraint', 'fun': constraints_func}

    # Solve
    result = opt.minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-8}
    )

    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

    centers = result.x[:2 * num_circles].reshape(num_circles, 2)
    radii = result.x[2 * num_circles:].reshape(num_circles)
    total_sum = np.sum(radii)

    return centers, radii, total_sum
