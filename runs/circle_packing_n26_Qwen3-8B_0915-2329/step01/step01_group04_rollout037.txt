import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Initial grid of centers
    grid_size = 26
    spacing = 0.25  # spacing parameter for grid initialization

    # Define grid of centers
    centers = np.zeros((grid_size, 2))
    for i in range(grid_size):
        row = i // (grid_size // 2)
        col = i % (grid_size // 2)
        centers[i, 0] = col * spacing
        centers[i, 1] = row * spacing

    # Define bounds for the centers
    bounds = [(0, 1) for _ in range(2 * grid_size)]

    # Define function to compute total sum of radii
    def total_sum(radii):
        return np.sum(radii)

    # Define constraints for non-overlapping and boundary conditions
    def constraints_func(x):
        # x contains all variables: first half is centers, then radii
        centers = x[:2 * grid_size].reshape(grid_size, 2)
        radii = x[2 * grid_size:]

        # Check if centers and radii are within bounds
        if any(c < -1e-12 or c > 1 + 1e-12 for c in np.concatenate(centers).flat):
            return np.inf
        if any(r < 0 for r in radii):
            return np.inf

        # Check for circle-center overlap with boundary
        for i in range(grid_size):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or

                return np.inf

        # Check for circle-circle overlap
        for i in range(grid_size):

                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.inf

        return -total_sum(radii)

    # We minimize the negative of total_sum
    x0 = np.concatenate([centers.flatten(), np.zeros(grid_size)])

        fun=constraints_func,
        x0=x0,
