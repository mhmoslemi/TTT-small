import numpy as np
import math
from scipy.optimize import minimize

def run_packing():
    # Hexagonal packing with equidistant rows
    # First, generate centers of a hexagonal lattice
    # Number of circles: 26
    # Divide into 5 layers (rows): 5, 5, 5, 5, 6

    # First layer: 5 circles
    # Second layer: 5 circles
    # Third layer: 5 circles
    # Fourth layer: 5 circles
    # Fifth layer: 6 circles (bottom layer)

    centers = []
    radii = []

    # Parameters
    num_layers = 5
    max_radius = 0.1
    spacing = 2 * max_radius
    y_offset = spacing / 2

    # Generate layers
    for layer in range(num_layers):
        num_circles_in_layer = 5 if layer < 4 else 6
        for i in range(num_circles_in_layer):
            if layer % 2 == 0:
                x = i * spacing
            else:
                x = (i + 0.5) * spacing
            y = layer * spacing + y_offset

            centers.append([x, y])
            radii.append(max_radius)

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Optimization step: adjust radii and positions for better packing
    # Define the objective function: maximize sum of radii
    # Define the constraints: non-overlapping, within square

    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum to maximize sum

    def constraints(radii):
        # Constraint: all circles within square
        # Constraint: all circles non-overlapping

        # Check if radii are non-negative
        if np.any(radii < 0):
            return False

        n = len(radii)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if not (0 - r <= x <= 1 + r and 0 - r <= y <= 1 + r):
                return False

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = math.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False

        return True

    # Combine the objective and constraints
    def constraint_func(radii):
        return constraints(radii)

    # Optimization
    bounds = [(0, 1) for _ in range(len(radii))]
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints={"type": "constraint", "fun": constraint_func},
        method="SLSQP"
    )

    # Check result
    if not result.success:
        raise ValueError(f"Optimization failed: {result.message}")

    # Final packing
    centers = np.array(centers)
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
