import numpy as np
import scipy.optimize as optimize
import matplotlib.pyplot as plt

def run_packing():
    # --- Initial layout: Hexagonal packing in the unit square
    # Start with a hexagonal grid and scale it into the unit square
    n = 26
    max_radius_guess = 0.1  # Initial guess for radius of each circle
    centers = np.zeros((n, 2))
    radii = np.full(n, max_radius_guess)

    # Basic hexagonal grid layout inside the unit square
    # We'll use a grid of rows and columns to approximate hexagonal packing
    rows = 5
    cols = 5
    for i in range(rows):
        for j in range(cols):
            x = j / (cols - 1)
            y = i / (rows - 1)
            centers[i * cols + j] = [x, y]

    # Optimization to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)

    def constraint_boundary(radii):
        for i in range(n):
            if radii[i] < 0:
                return False
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12:
                return False
            if y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return False
        return True

    def constraint_overlap(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    bounds = [(0, 1) for _ in range(n)]
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 1000}
    )

    if not result.success:
        raise ValueError("Optimization failed")

    centers = np.array(centers)
    radii = result.x
