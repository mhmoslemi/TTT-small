        y = (i % 2) / 2  # Staggered grid
        centers.append((x, y))

    bounds = [(0, 1) for _ in range(num_vars)]

    # Define objective function
    def objective(x_flat):

                dy = centers_flat[i][1] - centers_flat[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < 1e-12:
                    dist = 1e-12
                max_r = min(max_r, (dist - 1e-12) / 2)
            radii.append(max_r)
