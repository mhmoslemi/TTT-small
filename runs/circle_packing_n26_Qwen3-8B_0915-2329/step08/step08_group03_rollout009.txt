import scipy.optimize as optimize

def run_packing():

    positions = np.random.rand(num_circles, 2) * 2 - 1  # Initial guess slightly outside the unit square
    # Constraint: all centers must be within [0,1]^2
    bounds = [(0, 1) for _ in range(num_circles * 2)]

    def objective(x):

        radii = np.zeros(num_circles)
        for i in range(num_circles):
            # Compute max radius for boundary
            max_r = 1 - np.maximum(centers[i][0], 1 - centers[i][0])

                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                max_r = np.minimum(max_r, (dist - 1e-8) / 2)
            radii[i] = max_r
        return -np.sum(radii)  # Minimize negative sum to maximize sum of radii

    # Optimization using Sequential Least Squares Programming
    result = optimize.minimize(
        objective,

        bounds=bounds,
