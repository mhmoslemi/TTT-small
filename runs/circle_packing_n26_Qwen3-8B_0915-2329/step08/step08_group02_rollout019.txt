import scipy.optimize as optimize

def run_packing():

    # Initial placement: Create a sparse grid of trial points inside the square
    # This creates a more even distribution than a layered hexagonal approach
    centers = []
    for i in range(n):
        # Spread out points both horizontally and vertically to avoid symmetry
        # and ensure more even spacing
        x = i / (n - 1)
        y = (n - 1 - i) / (n - 1)
        centers.append((x, y))
    centers = np.array(centers)

    # Expand the grid to make sure we can tune positions easily
    # Start with some random perturbation to avoid symmetry
    best_centers = centers + np.random.uniform(-0.05, 0.05, size=(n, 2))

    # Define the optimization objective with constraints
    def objective(x_flat):
        centers_flat = x_flat.reshape((n, 2))
        radii = []
        for i in range(n):
            # Max possible radius from boundaries
            max_r = min(1 - max(centers_flat[i][0], 1 - centers_flat[i][0]),
                        1 - max(centers_flat[i][1], 1 - centers_flat[i][1]))
            # Max possible radius based on distances to others
            for j in range(n):
                if i == j:
                    continue
                dx = centers_flat[i][0] - centers_flat[j][0]
                dy = centers_flat[i][1] - centers_flat[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < 1e-10:
                    dist = 1e-10
                max_r = min(max_r, (dist - 1e-10) / 2)
            radii.append(max_r)
        return -sum(radii)

    # Bounds for positions
    bounds = [(0, 1) for _ in range(2 * n)]

    # Optimization parameters
    options = {
        'maxiter': 500,
        'ftol': 1e-10,
        'eps': 1e-8,
        'disp': False
    }

    # Optimization
    result = optimize.minimize(
        objective,
        best_centers.flatten(),
        bounds=bounds,
        method='SLSQP',
        options=options
    )

    best_centers = result.x.reshape((n, 2))
    best_radii = []
    for i in range(n):

            max_r = min(max_r, (dist - 1e-10) / 2)
        best_radii.append(max_r)

    return (best_centers, np.array(best_radii), sum(best_radii))
