import scipy.optimize as optimize

        centers.append((x, y))

        r = 0.5
        x = 0.5 + r * np.cos(angle)
        y = 0.5 + r * np.sin(angle)
        centers.append((x, y))

    # Trim to 26 circles (1+6+12+7)
    centers = centers[:26]

    num_vars = len(centers) * 2
    best_centers = np.array(centers) + np.random.uniform(-0.01, 0.01, size=(len(centers), 2))

    def objective(x_flat):
        centers_flat = x_flat.reshape((len(centers), 2))
        radii = []

            max_r = 1 - max(x_i, 1 - x_i)

            radii.append(max_r)

    # Bounds: x and y between 0 and 1
