import scipy.optimize as optimize

def run_packing():
    num_circles = 26

    # Initial random positions for all circles within [0, 1]^2
    np.random.seed(0)
    centers = np.random.rand(num_circles, 2)
    radii = np.zeros(num_circles)

    def objective(radii):
        # Objective: maximize the sum of the radii
        return -np.sum(radii)  # Minimize negative sum for optimization

    def constraints(x):
        # Nonlinear constraints: for each pair of circles, ensure their distance is at least the sum of radii
        # and ensure that no circle goes outside the unit square
        for i in range(num_circles):
            x_i, y_i = centers[i]
            r_i = x[i]
            # Check if circle is within the square
            if not (0 - 1e-8 <= x_i - r_i <= 1 + 1e-8 and 0 - 1e-8 <= y_i - r_i <= 1 + 1e-8):
                return 1.0

                return 1.0

            for j in range(num_circles):
                if i == j:
                    continue
                x_j, y_j = centers[j]
                r_j = x[j]
                dx = x_i - x_j
                dy = y_i - y_j
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r_i + r_j - 1e-8:
                    return 1.0
        return 0.0

    # Optimize using Sequential Least Squares Programming (SLSQP)
    result = optimize.minimize(
