import numpy as np
import scipy.optimize as optimize

def run_packing():
    num_circles = 26

    # Define variables for positions and radii: (x_0, y_0, r_0, x_1, y_1, r_1, ..., x_25, y_25, r_25)
    num_vars = num_circles * 3
    x0 = np.random.rand(num_vars)

    def objective(x):
        centers = x[:num_circles*2].reshape((num_circles, 2))
        radii = x[num_circles*2:].copy()
        return -np.sum(radii)  # Return negative sum for minimization

    def constraint_distance(x):
        centers = x[:num_circles*2].reshape((num_circles, 2))
        radii = x[num_circles*2:].copy()
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return dist - (radii[i] + radii[j]) + 1e-8
        return 0.0

    def constraint_boundary(x):
        centers = x[:num_circles*2].reshape((num_circles, 2))
        radii = x[num_circles*2:].copy()
        for i in range(num_circles):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-8 or x_i + r_i > 1.0 + 1e-8 or y_i - r_i < -1e-8 or y_i + r_i > 1.0 + 1e-8:
                return x_i - r_i + 1e-8
        return 0.0

    bounds = [(0, 1) for _ in range(num_vars)]

    # Use SLSQP for constrained optimization
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_distance},
            {'type': 'ineq', 'fun': constraint_boundary},
        ],
        method='SLSQP',
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )

    # Extract best centers and radii
    best_centers = result.x[:num_circles * 2].reshape((num_circles, 2))
    best_radii = result.x[num_circles * 2:]

    return (best_centers, best_radii, np.sum(best_radii))
