import scipy.optimize as optimize

def run_packing():

    positions = []

    rows = 5
    cols = 5
    spacing = 0.25
    for i in range(rows):

            y = i * spacing + 0.5 * spacing
            positions.append([x, y])

    # Add corner positions
    positions.append([0.1, 0.1])
    positions.append([0.9, 0.1])
    positions.append([0.1, 0.9])

        centers = x.reshape((num_circles, 2))
        max_radii = []
        for i in range(num_circles):
            x_i, y_i = centers[i]
            # Boundary distance
            dist_to_left = x_i
            dist_to_right = 1 - x_i
            dist_to_bottom = y_i
            dist_to_top = 1 - y_i
            max_r_boundary = np.min([dist_to_left, dist_to_right, dist_to_bottom, dist_to_top])
            max_r = max_r_boundary

                    continue
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                max_r = np.min([max_r, (dist - 1e-8) / 2])
