import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize

def run_packing():

    ]

    best_radii = None

                    y = r * (1.0 / (rows - 1)) if rows > 1 else 0.5

                    x = (c + 0.5) * (1.0 / (cols - 1)) if cols > 1 else 0.5
                    y = r * (1.0 / (rows - 1)) if rows > 1 else 0.5
                centers.append([x, y])
        
        # Add center circle
        centers.append([0.5, 0.5])
        
        # Optimization parameters
        num_vars = len(centers) * 2
        bounds = [(0, 1) for _ in range(num_vars)]
        initial_centers = np.array(centers) + np.random.rand(len(centers), 2) * 0.05

        # Objective function to maximize sum of radii
        def objective(x):
            centers_flat = x.reshape((len(centers), 2))
            max_radii = []
            for i in range(len(centers_flat)):
                max_r = 1 - max(centers_flat[i][0], 1 - centers_flat[i][0])
                max_r = min(max_r, 1 - max(centers_flat[i][1], 1 - centers_flat[i][1]))

                        continue
                    dx = centers_flat[i][0] - centers_flat[j][0]
                    dy = centers_flat[i][1] - centers_flat[j][1]
                    dist = np.sqrt(dx ** 2 + dy ** 2)
                    max_r = min(max_r, (dist - 1e-12) / 2)
                max_radii.append(max_r)
            return -sum(max_radii)

        result = minimize(
            objective,
            initial_centers.flatten(),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 200, 'ftol': 1e-7, 'eps': 1e-8}
