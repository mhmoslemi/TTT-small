from scipy.optimize import minimize
from scipy.stats import qmc  # For Latin hypercube sampling

def run_packing():

    # Generate initial positions using Latin hypercube sampling plus corner and edge placements
    # Create 50 initial points with Latin hypercube sampling
    lhs_samples = qmc.LatinHypercube(d=2, samples=50)
    lhs = lhs_samples.random()
    positions = []
    for point in lhs:
        positions.append(list(point))
    # Add corner and edge placements
    corners = [(0, 0), (0, 1), (1, 0), (1, 1)]
    for corner in corners:
        positions.append(list(corner))

        positions.append([x, 0])
        positions.append([x, 1])

    positions = list(set(map(tuple, positions)))
    positions = [list(p) for p in positions]
    if len(positions) > num_circles:
        positions = positions[:num_circles]
    elif len(positions) < num_circles:
        for _ in range(num_circles - len(positions)):
            positions.append([0.5, 0.5])

    num_vars = len(positions) * 2
    x0 = np.random.rand(num_vars)

    def objective(x):
        centers = x.reshape((len(positions), 2))
        max_radii = []
        for i in range(len(positions)):
            x_i, y_i = centers[i]
            # Boundary distance
            max_r_boundary = 1 - np.maximum(x_i, 1 - x_i)
            max_r_boundary = np.minimum(max_r_boundary, 1 - np.maximum(y_i, 1 - y_i))
            max_r = max_r_boundary
            # Pairwise distance to other circles

                if i == j:
                    continue
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                max_r = np.minimum(max_r, (dist - 1e-8) / 2)
            max_radii.append(max_r)
        return -np.sum(max_radii)  # Minimize negative objective (so that we maximize the sum of radii)
