import opti

            row = i // 4
            col = i % 4

            y = row / 4.0
            # Disperse circles to avoid symmetry
            x += (np.sin(i) / 20.0)
            y += (np.cos(i) / 20.0)
            # Adjust for corners and edge

            y = np.clip(y, 0.0, 1.0)
            positions.append([x, y])
        return np.array(positions)

    #Initial estimate for radius
    def initial_radii(positions):
        radii = []
        for i in range(len(positions)):

                    continue
                dx = positions[i, 0] - positions[j, 0]
                dy = positions[i, 1] - positions[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                r = min(r, (dist - 1e-8)/2)
            r = min(r, (1 - max(positions[i, 0], 1 - positions[i, 0])) / 2)
            r = min(r, (1 - max(positions[i, 1], 1 - positions[i, 1])) / 2)
            radii.append(r)
        return np.array(radii)

    # Optimization
    opt = opti.Opti()
    centers = opt.variable(26, 2)
    radii = opt.variable(26)
