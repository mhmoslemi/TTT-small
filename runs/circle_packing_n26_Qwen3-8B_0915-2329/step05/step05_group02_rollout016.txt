import scipy.optimize as optimize

def run_packing():

        (0.05, 0.5), (0.5, 0.05), (0.95, 0.5), (0.5, 0.95),

    ]

            y = 0.5 + r * 0.2 * np.sin(theta)
            inner_centers.append((x, y))
    inner_centers = inner_centers[:n - len(centers)]
    centers.extend(inner_centers)

    # Trim to 26 circles
    centers = centers[:n]
    centers = np.array(centers)

    num_vars = len(centers) * 2

    best_centers = centers + np.random.uniform(-0.01, 0.01, size=(len(centers), 2))

        radii = []
        for i in range(len(centers_flat)):

            max_r = 1 - max(x_i, 1 - x_i)

                dy = centers_flat[i][1] - centers_flat[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                max_r = min(max_r, (dist - 1e-12) / 2)
            radii.append(max_r)

    # Bounds: x and y between 0 and 1

    # Use SLSQP for constrained optimization
    result = optimize.minimize(
