import scipy.optimize as optimize

def run_packing():

    # Define the number of rows for the hexagonal lattice
    # We assume a hex grid structured as follows:
    #     |  1
    #  2  3  4
    #  5  6  7  8

    # - Two rows (top, bottom)
    # - Four columns (left to right)
    # - A middle row with 8 circles

    # Define the hexagonal grid as a structured lattice
    hex_lattice = []
    for i in range(4):
        for j in range(4):
            if i == 0 or i == 3:
                # Top and bottom rows
                hex_lattice.append((0.25 + 0.25 * j, 0.1 + 0.25 * i))
            else:
                # Middle rows
                if (i + j) % 2 == 0:
                    hex_lattice.append((0.25 + 0.25 * j, 0.25 + 0.25 * i))
                else:
                    hex_lattice.append((0.25 + 0.25 * (j + 1), 0.25 + 0.25 * i))
        # Add another row if needed
    hex_lattice = hex_lattice[:26]

    # Convert to a NumPy array of positions
    num_vars = len(hex_lattice) * 2  # x and y for each circle
    x0 = np.random.rand(num_vars)  # Initial positions

        centers = x.reshape((len(hex_lattice), 2))
        radii = []

        for i in range(len(centers)):

            x_i, y_i = centers[i]

            for j in range(len(centers)):

                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                max_r = np.minimum(max_r, (dist - 1e-8) / 2)

            radii.append(max_r)

        return -np.sum(radii)  # Negative for minimization

    # Bounds for the x and y positions
    bounds = [(0, 1) for _ in range(num_vars)]

    # Run differential evolution
    result = optimize.differential_evolution(

    sum_radii = np.sum(best_radii)

    return (best_centers, np.array(best_radii), sum_radii)
