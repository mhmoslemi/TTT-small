import random
import math
from scipy.optimize import minimize

def run_packing():

    centers = []
    radii = []

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
