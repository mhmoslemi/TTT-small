import numpy as np
import cvxpy as cp

def run_packing():
    # Define number of circles
    n = 26
    centers = np.random.rand(n, 2)  # Initial guess (random positions)
    radii = np.random.rand(n)       # Initial guess (random radii)

    # Problem variables
    x = cp.Variable((n, 2))
    r = cp.Variable(n, non_negative=True)

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = []

    # Ensure circles are inside the unit square
    for i in range(n):
        x_i, y_i = x[i]
        r_i = r[i]
        constraints.append(x_i - r_i <= 1 + 1e-12)
        constraints.append(x_i + r_i <= 1 + 1e-12)
        constraints.append(y_i - r_i <= 1 + 1e-12)
        constraints.append(y_i + r_i <= 1 + 1e-12)

    # Ensure circles don't overlap
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2)
            constraints.append(dist >= r[i] + r[j] - 1e-12)

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.ECOS)

    radii = r.value

    # Compute total radius sum
    total_radii = np.sum(radii)
