import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial hexagonal grid in unit square with 26 circles
    # Generate positions with some random jitter to avoid symmetry
    def generate_hex_grid(n):
        # Generate hexagonal grid positions in [0, 1] x [0, 1]
        # First, determine the number of rows
        rows = int(np.sqrt(n))
        cols = int(np.ceil(n / rows))
        gap = 1 / (rows + 1)
        positions = []
        for i in range(rows):
            for j in range(cols):
                x = j * (1 + gap) + gap
                y = i * (1 + gap) + gap
                if x > 1 or y > 1:
                    continue
                positions.append((x, y))
        # If not enough points, add more manually
        while len(positions) < n:
            x = np.random.uniform(0, 1)
            y = np.random.uniform(0, 1)
            positions.append((x, y))
        return np.array(positions)

    # Generate initial positions
    initial_centers = generate_hex_grid(26)

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize total

    # Constraint functions
    def boundary_constraint(radii):
        return np.array([r for r in radii])  # Enforce non-negative radii

    def overlap_constraint(radii):
        # Check pairwise overlap, returns distances - (radii[i] + radii[j])
        dists = np.zeros(26)
        for i in range(26):
            for j in range(i + 1, 26):
                dx = initial_centers[i, 0] - initial_centers[j, 0]
                dy = initial_centers[i, 1] - initial_centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                dists[i] += dist - (radii[i] + radii[j])
                dists[j] += dist - (radii[i] + radii[j])
        return dists

    # Bounds
    bounds = [(0, 1) for _ in range(26)]

    # Optimization
    result = optimize.minimize(
        fun=objective,
        x0=np.full(26, 0.05),
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': boundary_constraint},
            {'type': 'ineq', 'fun': overlap_constraint}
        ],
        method='SLSQP'
    )

    centers = initial_centers
    radii = result.x
    total_radius = np.sum(radii)

    return (centers, radii, total_radius)
