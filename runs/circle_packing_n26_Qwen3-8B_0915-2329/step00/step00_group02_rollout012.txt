import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():
    # Initial positions following a hexagonal grid pattern
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Create a hexagonal grid arrangement
    for i in range(n):
        row = i // 5
        col = i % 5
        centers[i, 0] = 0.1 + col * 0.2
        centers[i, 1] = 0.1 + row * 0.22

    # Define the optimization problem
    def objective(x):
        # x is a flat array of centers and radii
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:].reshape(n)

        # Compute total sum of radii
        total_sum = np.sum(radii_flat)
        # Penalize for going out of bounds
        boundary_penalty = np.sum(np.maximum(centers_flat[:, 0] - radii_flat, 0) + np.maximum(1 - centers_flat[:, 0] - radii_flat, 0) +
                                 np.maximum(centers_flat[:, 1] - radii_flat, 0) + np.maximum(1 - centers_flat[:, 1] - radii_flat, 0))
        # Penalize for overlapping
        overlap_penalty = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    overlap_penalty += (radii_flat[i] + radii_flat[j] - dist) ** 2

        return -total_sum + 100 * boundary_penalty + 1000 * overlap_penalty

    # Define bounds for positions and radii
    bounds = []
    for i in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])  # x and y of each center
        bounds.extend([(0.0, 1.0)])  # radius of each circle

    # Initial guess: x and y positions follow hex-grid; radii are small
    x_initial = np.zeros(n * 3)
    for i in range(n):
        x_initial[i * 3 : (i + 1) * 3] = centers[i]

    # Use L-BFGS-B to maximize the total sum of radii
    result = optimize.minimize(
        objective,
        x_initial,
        method='L-BFGS-B',
        bounds=bounds,
        options={'gtol': 1e-8, 'maxiter': 100}
    )

    # Extract result
    centers = result.x[:n*2].reshape(n, 2)

    valid, message = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Invalid packing: {message}")

    # Return centers, radii, and sum of radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
