import numpy as np
import cvxpy as cp

def run_packing():
    # Initial centers grid in a semi-hexagonal pattern
    # Rows: 6, Columns: 5, with some reductions
    rows = 6
    cols = 5
    centers = []
    for row in range(rows):
        for col in range(cols):
            if row % 2 == 0:
                x = col / (cols - 1)
                y = row / (rows - 1)
            else:
                x = (col - 0.5) / (cols - 1)
                y = row / (rows - 1)
            centers.append([x, y])
    centers = np.array(centers)
    # Remove 4 circles to keep 26
    centers = centers[:26]

    # Optimization variables
    radii = cp.Variable(26)
    constraints = []

    # Boundary constraint: Each circle must stay within [0,1]^2
    for i in range(26):
        x, y = centers[i]
        constraints.append(x - radii[i] >= 0)
        constraints.append(x + radii[i] <= 1)
        constraints.append(y - radii[i] >= 0)
        constraints.append(y + radii[i] <= 1)

    # Non-overlapping constraint: For all i < j, distance between centers >= sum of radii
    for i in range(26):
        for j in range(i + 1, 26):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            constraints.append(dist >= radii[i] + radii[j])

    # Maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve()

    # Extract results
    centers_result = centers
    radii_result = radii.value
    sum_radii = np.sum(radii_result)

    return (centers_result, radii_result, sum_radii)
