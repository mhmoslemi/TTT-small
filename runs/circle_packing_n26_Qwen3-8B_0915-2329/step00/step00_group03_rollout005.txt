import numpy as np
import cvxpy as cp

def run_packing():
    # Define variables
    n = 26
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n, nonneg=True)

    # Objective: Maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Constraint: Each circle must fit within the square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r >= -1e-8)
        constraints.append(x + r <= 1 + 1e-8)
        constraints.append(y - r >= -1e-8)
        constraints.append(y + r <= 1 + 1e-8)

    # Constraint: No circle overlaps with another
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-8)

    # Solve the optimization
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, eps=1e-5)

    if not prob.success:
        raise ValueError("Optimization failed.")

    # Convert to numpy arrays
    centers = np.round(centers.value, 8)
    radii = np.round(radii.value, 8)
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
