from scipy.optimize import minimize
from typing import Tuple

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial grid-based placement
    x = np.linspace(0, 1, 6)
    y = np.linspace(0, 1, 6)
    grid = np.array([(xi, yi) for xi in x for yi in y])
    # Remove the (0,0) corner point
    centers = np.delete(grid, 0, axis=0)
    # Add one more circle near the bottom-right corner
    centers = np.vstack([centers, [0.95, 0.95]])
    
    # Define the optimization problem
    def objective(radii):
        return -np.sum(radii)  # Minimize -sum(radii)

    def constraint1(radii):
        # Ensure each circle stays inside the square
        return np.array([r for r in radii])

    def constraint2(radii):
        # Ensure circles don't overlap
        n = len(radii)

        idx = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                dists[idx] = dist - (radii[i] + radii[j])
                idx += 1
        return dists

    # Initial guess: all circles have the same radius
    initial_radius = 1.0 / np.sqrt(26)

    bounds = [(0, 1.0) for _ in range(26)]

    constraints = (
