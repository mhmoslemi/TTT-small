import scipy.optimize as optimize

def run_packing():

    # Use hexagonal packing for a 26-circle configuration
    # We'll round up to the nearest hexagonal layer
    
    # Parameters for hexagonal packing
    num_circles = 26
    max_r = 0.2  # Initial guess for radii (to ensure no overlaps)
    
    # Generate hexagonal grid with spacing
    grid = []
    for i in range(5):
        for j in range(5):
            x = j + 0.5 * i
            y = i + 0.5 * j
            if x >= 1 or y >= 1:
                continue
            grid.append((x, y))
    
    # Add additional circles if needed to reach 26
    while len(grid) < num_circles:
        x = np.random.uniform(0, 1)
        y = np.random.uniform(0, 1)
        grid.append((x, y))
    
    # Convert to numpy arrays
    centers = np.array(grid)
    
    # Step 2: Optimize positions and radii to maximize total radius sum
    # Define variables: radii for each circle
    radii = np.full(len(centers), max_r)
    
    # Define the optimization problem
    def objective(x):
        # x contains radii, we finalize radii as the solution
        return -np.sum(x)  # Negative because we are minimizing
    
    def constraint(x):
        # Ensure all circles are within the unit square
        for i in range(len(x)):
            x_i = centers[i][0]
            y_i = centers[i][1]
            r_i = x[i]
            if not (0 <= x_i - r_i and x_i + r_i <= 1 and 0 <= y_i - r_i and y_i + r_i <= 1):
                return False
        return True
    
    def constraint_overlap(x):
        # Ensure no overlaps between circles
        for i in range(len(x)):
            for j in range(i + 1, len(x)):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < x[i] + x[j] - 1e-8:
                    return False
        return True
    
    # Combine all constraints
    cons = [
        {'type': 'ineq', 'fun': constraint},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]
    
    result = optimize.minimize(
        objective,
        radii,
        bounds=[(0, 1) for _ in range(len(radii))],
        constraints=cons
    )
    
    # Reconstruct final configuration
    final_centers = centers
    final_radii = result.x
    sum_radii = np.sum(final_radii)
    
    # Check if the result is valid
    if not result.success:

    return (final_centers, final_radii, sum_radii)
