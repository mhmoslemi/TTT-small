import scipy.optimize as optimize

def run_packing():

    cols = int(np.ceil(n / rows))

        col = i % cols
        x = col * spacing + spacing / 2
        y = row * spacing * np.sqrt(3) / 2
        centers[i] = [x, y]

    # Initial guess for radii: all ones (will be adjusted)
    radii = np.ones(n)

    # Objective function: maximize sum of radii
    def objective(x):
        # x is a flattened array of centers and radii
        # Extract centers and radii
        centers_flat = x[:n * 2].reshape((n, 2))
        radii_flat = x[n * 2:]
        return -np.sum(radii_flat)  # Minimize negative sum to maximize sum

    # Constraints:
    # 1. To stay within the square [0,1]^2
    def square_constraint(x):
        centers_flat = x[:n * 2].reshape((n, 2))
        radii_flat = x[n * 2:]
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -1.0
        return 0.0

    # 2. Non-overlapping circles
    def non_overlap_constraint(x):
        centers_flat = x[:n * 2].reshape((n, 2))

                r_i = radii_flat[i]
                r_j = radii_flat[j]
                dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)

    # Bounds:
