from scipy.optimize import minimize

def run_packing():

    num_circles = 26
    num_rows = 5  # Based on a hexagonal packing of 26 circles
    num_cols = 5  # Mature packing in a square

    # Coordinates for the centers in a hexagonal grid
    centers = []
    for i in range(num_rows):
        for j in range(num_cols):
            x_pos = j + 0.5 * i
            y_pos = i / 2
            if i % 2 == 0:
                x_pos = j + 0.5 * i
            else:
                x_pos = j + 0.5 * (i - 1)
            centers.append((x_pos, y_pos))

    centers = np.array(centers)

    radii = max_radius * np.ones(num_circles)

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize the sum

    def constraints(radii):
        # Ensure each radius is non-negative
        return radii >= 0

    # Define the constraints as bounds
    bounds = [(0, 1) for _ in range(num_circles)]

    # Run the optimization
    result = minimize(objective, radii, bounds=bounds, method='SLSQP')

    opt_radii = result.x

    opt_radii = np.maximum(opt_radii, 1e-10)

    def validate_packing(centers, radii):
