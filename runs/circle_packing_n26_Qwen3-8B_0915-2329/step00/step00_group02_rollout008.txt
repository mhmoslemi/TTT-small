import numpy as np
import cvxpy as cp

def run_packing():
    # Define variables
    n = 26
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)
    tolerance = 1e-5

    # Objective: Maximize total sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        constraints.append(x >= tolerance)
        constraints.append(x <= 1 - tolerance)
        constraints.append(y >= tolerance)
        constraints.append(y <= 1 - tolerance)

    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints.append(dist >= radii[i] + radii[j] - tolerance)

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver='MOSEK', verbose=False)
    except:
        # If any error occurs, return all-zero radii
        return np.zeros((26, 2)), np.zeros(26), 0.0

    if not prob.status == 'optimal':
        # If not optimal, return all-zero radii
        return np.zeros((26, 2)), np.zeros(26), 0.0

    # Get the computed centers and radii
    centers_np = np.array(centers.value)
    radii_np = np.array(radii.value)

    # Compute total sum of radii

    return centers_np, radii_np, total_radii
