import numpy as np
import scipy.optimize as opt
from typing import Tuple

def validate_packing(centers: np.ndarray, radii: np.ndarray) -> Tuple[bool, str]:
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def initial_hexagonal_arrangement(n: int) -> Tuple[np.ndarray, np.ndarray]:
    rows = int(np.ceil(n / 6))
    cols = int(np.ceil(n / rows))
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.01)
    index = 0
    for row in range(rows):
        for col in range(cols):
            if index >= n:
                break
            x = col + 0.5
            y = row + 0.5
            centers[index] = [x, y]
            radii[index] = 0.05
            index += 1
    return centers, radii

def objective_function(params: np.ndarray, centers: np.ndarray, radii: np.ndarray) -> float:
    n = len(centers)
    new_centers = np.reshape(params[:2*n], (n, 2))
    new_radii = params[2*n:]
    sum_radii = np.sum(new_radii)
    for i in range(n):
        for j in range(i + 1, n):
            dx = new_centers[i, 0] - new_centers[j, 0]
            dy = new_centers[i, 1] - new_centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < new_radii[i] + new_radii[j] - 1e-8:
                return -float('inf')
    for i in range(n):
        x, y = new_centers[i]
        r = new_radii[i]
        if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
