import numpy as np
import random
import math

def run_packing():
    # Configuration parameters
    POP_SIZE = 30
    MAX_ITERATIONS = 100
    PARTICLE_DIM = 26 * 2 + 26  # 2 centers per circle, 1 radius per circle
    INERTIA_WEIGHT = 0.8
    C1 = 2.0  # cognitive weight
    C2 = 2.0  # social weight
    
    def fitness(pop):
        fitness_values = []
        for i in range(len(pop)):
            centers = pop[i, :26*2].reshape(26, 2)
            radii = pop[i, 26*2:].reshape(26)
            total_radius = np.sum(radii)
            
            # Check if circle is within unit square
            for j in range(26):
                x, y = centers[j]
                r = radii[j]
                if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                    y - r < -1e-12 or y + r > 1 + 1e-12):
                    total_radius = 0
                    break
            
            # Check if circles overlap
            for j in range(26):
                for k in range(j + 1, 26):
                    dx = centers[j, 0] - centers[k, 0]
                    dy = centers[j, 1] - centers[k, 1]
                    dist = math.sqrt(dx * dx + dy * dy)
                    if dist < radii[j] + radii[k] - 1e-12:
                        total_radius = 0
                        break
                if total_radius == 0:
                    break
            fitness_values.append(total_radius)
        return np.array(fitness_values)
    
    def bounds():
        # For center coordinates, ensure they are within [0,1]^2
        # For radii, ensure they are non-negative
        return np.array([0.0, 1.0] * 26 + [0.0] * 26)

    # Initialize population
    population = np.random.uniform(0, 1, (POP_SIZE, PARTICLE_DIM))
    personal_best = population.copy()
    global_best = population[0, :]

    for iter in range(MAX_ITERATIONS):
        # Evaluate fitness
        fitness_vals = fitness(population)

        # Update personal best
        for i in range(POP_SIZE):
            if fitness_vals[i] > fitness_vals[np.argmax(fitness_vals)]:
                personal_best[i, :] = population[i, :]

        # Update global best
        best_idx = np.argmax(fitness_vals)
        global_best = population[best_idx, :]

        # Update velocity and position
        for i in range(POP_SIZE):
            r1 = random.random()
            r2 = random.random()
            velocity = INERTIA_WEIGHT * population[i, :] + C1 * r1 * (personal_best[i, :] - population[i, :]) + C2 * r2 * (global_best - population[i, :])
            population[i, :] = population[i, :] + velocity

            # Apply bounds
            for j in range(26):
                population[i, j*2] = np.clip(population[i, j*2], 0.0, 1.0)
                population[i, j*2 + 1] = np.clip(population[i, j*2 + 1], 0.0, 1.0)
            for j in range(26):
                population[i, 26*2 + j] = np.clip(population[i, 26*2 + j], 0.0, 1.0)

    # Extract best solution
    best_idx = np.argmax(fitness(population))
    centers = population[best_idx, :26*2].reshape(26, 2)
    radii = population[best_idx, 26*2:].reshape(26)
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
