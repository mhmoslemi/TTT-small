import numpy as np
import scipy.optimize as optimize

def run_packing():

    # Define initial positions: a hexagonal-like arrangement that evenly distributes
    # the circles within the square, with some placed near the boundary and corners.

    # Generate 26 circle centers through a combination of:
    # - Central block (a dense cluster of circles)
    # - Side cluster (closer to the edges of the square)
    # - One or two corner clusters (to increase packing efficiency)
    centers = []

    # Create a 2D grid of points to initialize centers
    grid_points = []
    for i in range(n):
        x = i / (n - 1) * 0.8 + 0.1
        y = (i // 5) / (n // 5 - 1) if n // 5 > 0 else 0.5
        grid_points.append((x, y))

    # Add corner clusters
    corners = [(0.1, 0.1), (0.1, 0.9), (0.9, 0.1), (0.9, 0.9)]
    centers += corners

    # Add side clusters
    sides = []
    for y in [0.1, 0.9]:
        for x in np.linspace(0.1, 0.9, 4):
            sides.append((x, y))
    for x in [0.1, 0.9]:
        for y in np.linspace(0.1, 0.9, 4):
            sides.append((x, y))
    sides = sides[:n - len(corners)]  # Trim to 26 circles
    centers += sides

    # Convert to NumPy array
    centers = np.array(centers)
    num_vars = len(centers) * 2

    # Bounds: x and y between 0 and 1
    bounds = [(0, 1) for _ in range(num_vars)]

    # Initial guess: slightly perturb the centers to avoid overlap
    best_centers = centers + np.random.uniform(-0.05, 0.05, size=(len(centers), 2))

    # Define the loss function to maximize the sum of radii (so we minimize the negative)
    def objective(x_flat):
        centers_flat = x_flat.reshape((len(centers), 2))
        radii = []
        for i in range(len(centers_flat)):
            x_i, y_i = centers_flat[i]
            # Maximum possible radius from the square constraint
            max_r = 1 - max(x_i, 1 - x_i)
            max_r = min(max_r, 1 - max(y_i, 1 - y_i))
            # For each other circle, compute the maximum radius ensuring non-overlap
            for j in range(len(centers_flat)):
                if i == j:
                    continue
                dx = centers_flat[i][0] - centers_flat[j][0]
                dy = centers_flat[i][1] - centers_flat[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                max_r = min(max_r, (dist - 1e-12) / 2)
            radii.append(max_r)
        return -sum(radii)

    # Perform constrained optimization
    result = optimize.minimize(
        objective,
        best_centers.flatten(),
        bounds=bounds,
        method='SLSQP',
        options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-8, 'disp': False}

    # Compute best radii
    best_radii = []
    for i in range(len(best_centers)):

            max_r = min(max_r, (dist - 1e-12) / 2)
        best_radii.append(max_r)

    return (best_centers, np.array(best_radii), sum(best_radii))
