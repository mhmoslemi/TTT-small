import scipy.optimize as optimize

    grid.append([0.9, 0.1])
    grid.append([0.9, 0.9])

    grid = grid[:26]

    num_vars = len(grid) * 2
    x0 = np.random.rand(num_vars)  # Random starting positions

    def objective(x):
        centers = x.reshape((len(grid), 2))
        max_radii = []
        for i in range(len(centers)):
            x_i, y_i = centers[i]

                    continue

                dist = np.sqrt(dx ** 2 + dy ** 2)
                max_r = np.minimum(max_r, (dist - 1e-8) / 2)
