import numpy as np
import random
import scipy.optimize as optimize

def run_packing():
    num_circles = 26

    # Define a pool of possible positions to start with
    def generate_positions(num_circles):
        # Generate a set of positions with some randomness to aid in optimizing
        return [np.random.rand(2) for _ in range(num_circles)]

    positions = generate_positions(num_circles)

    # Function to compute maximum possible radius for each circle based on distance to others
    def compute_radii(centers):
        radii = []
        for i in range(len(centers)):
            max_r = 0.0
            # Check boundary distance
            max_r = min(1 - centers[i][0], centers[i][0], 1 - centers[i][1], centers[i][1])
            # Check distance to other circles
            for j in range(len(centers)):
                if i == j:
                    continue
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                max_r = min(max_r, (dist - 1e-8) / 2)
            radii.append(max_r)
        return radii

    # Objective function to maximize the sum of radii
    def objective(x):
        centers = x.reshape((len(positions), 2))
        radii = compute_radii(centers)
        return -np.sum(radii)  # Negative for minimization

    # Bounds for x and y of each circle's center
    bounds = [(0, 1) for _ in range(len(positions) * 2)]

    # Use SLSQP for constrained optimization
    result = optimize.minimize(
        objective,
        x0=np.random.rand(len(positions) * 2),
        bounds=bounds,
        method='SLSQP',
        options={'maxiter': 1000, 'ftol': 1e-12, 'eps': 1e-12}
    )

    # Extract best centers and radii
    best_centers = result.x.reshape((len(positions), 2))
    best_radii = compute_radii(best_centers)

    return (best_centers, np.array(best_radii), np.sum(best_radii))
