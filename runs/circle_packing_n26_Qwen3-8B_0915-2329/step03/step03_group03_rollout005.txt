import scipy.optimize as optimize

def run_packing():

    centers = []
    radii = []

        # A hexagonal arrangement will have multiple rows
        row = i // 7

        x = col / 6.0 + 0.1
        y = row / 6.0 + 0.1
        centers.append([x, y])

    # Object function for optimization: minimize negative of sum of radii
    def objective(x):

        centers_flat = x[:n * 2]

            max_r = 1 - max(x_i, 1 - x_i)

                if i == j:
                    continue
                dx = centers_2d[i][0] - centers_2d[j][0]
                dy = centers_2d[i][1] - centers_2d[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                max_r = min(max_r, (dist - 1e-12) / 2)
            max_radii.append(max_r)
        return -sum(max_radii)  # Negative for minimization
