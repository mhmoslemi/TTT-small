import scipy.optimize as optimize

def run_packing():
    # Create a hexagonal grid of 25 circles with refined spacing
    centers = []
    radii = []
    
    # Define grid parameters
    rows = 5
    cols = 5
    spacing = 0.3  # Adjust spacing for denser packing
    
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + (i % 2) * (spacing / 2)
            y = i * (spacing * np.sqrt(3) / 2)
            centers.append([x, y])
            radii.append(0.1)
    
    # Add one more circle in the bottom right corner
    centers.append([0.9, 0.9])
    radii.append(0.1)
    
    num_vars = len(centers) * 2
    x = np.random.rand(num_vars) * 2 - 1  # Random initialization slightly outside [0,1]
    
    def objective(x):
        centers_flat = x[:num_vars]
        centers_2d = centers_flat.reshape((len(centers), 2))
        max_radii = []
        
        for i in range(len(centers_2d)):
            x_i, y_i = centers_2d[i]
            max_r = 1 - max(x_i, 1 - x_i)
            max_r = min(max_r, 1 - max(y_i, 1 - y_i))
            
            for j in range(len(centers_2d)):
                if i == j:
                    continue
                dx = centers_2d[i][0] - centers_2d[j][0]
                dy = centers_2d[i][1] - centers_2d[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                max_r = min(max_r, (dist - 1e-12) / 2)
            
            max_radii.append(max_r)
        
        return -sum(max_radii)  # Minimize negative sum to maximize the total
    
    # Set bounds for each coordinate to be within [0,1]
    bounds = [(0, 1) for _ in range(num_vars)]
    
    # Use a more robust optimization method with tighter tolerances
    result = optimize.minimize(
        objective,
        x,
        bounds=bounds,
        method='SLSQP',
        options={'ftol': 1e-10, 'maxiter': 200, 'disp': True}
    )
    
    best_centers = result.x.reshape((len(centers), 2))
