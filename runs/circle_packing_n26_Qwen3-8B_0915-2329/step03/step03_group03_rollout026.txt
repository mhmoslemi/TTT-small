def run_packing():

    # Initial layout with one circle in the center and 25 circles around it
    centers = []
    radii = []

    # Central circle
    centers.append([0.5, 0.5])
    radii.append(0.02)

    for i in range(25):
        angle = 2 * np.pi / 25 * i
        r = 0.02
        if i % 5 == 0:
            r = 0.03  # slightly larger for corner circles
        x = 0.5 + r * np.cos(angle)
        y = 0.5 + r * np.sin(angle)

    def fitness(centers, radii):

        for i in range(n):
            x, y = centers[i]
            if not (0 <= x <= 1 and 0 <= y <= 1):
                return 0
            if radii[i] < 0:
                return 0

                return 0

        return sum(radii)

    def generate_individual():
        indiv = []

    def evaluate(population):
        fitness_scores = []

    def tournament_selection(population, fitness_scores, tournament_size=5):
        selected = []

            candidates = random.sample(fitness_scores, tournament_size)
            best = max(candidates, key=lambda x: x[0])
