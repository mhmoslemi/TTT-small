import numpy as np
import scipy.optimize as optimize
import random

def run_packing():
    num_circles = 26
    initial_centers = []
    for i in range(num_circles):
        initial_centers.append([random.uniform(0, 1), random.uniform(0, 1)])
    
    num_vars = len(initial_centers) * 2  # x and y for each circle
    x0 = np.array([random.uniform(0, 1) for _ in range(num_vars)])
    
    def objective(x):
        centers = x.reshape((len(initial_centers), 2))
        max_radii = []
        for i in range(len(centers)):
            x_i, y_i = centers[i]
            max_r = 1 - np.maximum(x_i, 1 - x_i)
            max_r = np.minimum(max_r, 1 - np.maximum(y_i, 1 - y_i))
            for j in range(len(centers)):
                if i == j:
                    continue
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                max_r = np.minimum(max_r, (dist - 1e-8) / 2)
            max_radii.append(max_r)
        return -np.sum(max_radii)
    
    def constraint(x):
        centers = x.reshape((len(initial_centers), 2))
        for i in range(len(centers)):
            x_i, y_i = centers[i]
            if x_i < -1e-8 or x_i > 1 + 1e-8 or y_i < -1e-8 or y_i > 1 + 1e-8:
                return 1.0
        return 0.0
    
    bounds = [(0, 1) for _ in range(num_vars)]
    
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraint}],
        method='L-BFGS-B',
        options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    best_centers = result.x.reshape((len(initial_centers), 2))
    best_radii = []
    for i in range(len(best_centers)):
        x_i, y_i = best_centers[i]
        max_r = 1 - np.maximum(x_i, 1 - x_i)

            max_r = np.minimum(max_r, (dist - 1e-8) / 2)

    return (best_centers, np.array(best_radii), np.sum(best_radii))
