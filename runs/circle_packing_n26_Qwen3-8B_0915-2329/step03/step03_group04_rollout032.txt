import scipy.optimize as optimize
import random

def run_packing():

    # First, create a hexagonal packing with 5 rows and 5 columns
    num_rows = 5
    num_cols = 5
    spacing = 1.0 / (num_rows - 1)  # Spacing between rows in the x-direction
    
    centers = []
    for row in range(num_rows):
        for col in range(num_cols):
            if row % 2 == 0:
                # Even rows are aligned
                x = col * spacing
                y = row * spacing
            else:
                # Odd rows are shifted by half the spacing
                x = (col + 0.5) * spacing
                y = row * spacing
            centers.append([x, y])
    
    # Add an additional circle at the center
    centers.append([0.5, 0.5])
    
    # Optimization parameters
    max_iter = 500
    learning_rate = 0.05
    tolerance = 1e-6
    num_vars = len(centers) * 2  # Each circle has x and y coordinates

    # Initial positions for centers
    initial_centers = np.array(centers)
    x = np.concatenate([initial_centers[:, 0], initial_centers[:, 1]])

    def objective(x):
        centers_flat = x.reshape((len(centers), 2))
        max_radii = []
        for i in range(len(centers_flat)):
            x_i, y_i = centers_flat[i]
            
            # Compute max radius without overlapping with boundaries
            max_r = 1 - max(x_i, 1 - x_i)

            # Compute max radius without overlapping with other circles
            for j in range(len(centers_flat)):

                    continue
                dx = centers_flat[i][0] - centers_flat[j][0]
                dy = centers_flat[i][1] - centers_flat[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                max_r = min(max_r, (dist - 1e-12) / 2)
            max_radii.append(max_r)
        return -sum(max_radii)  # Negative for minimization

    # Perform optimization
    result = optimize.minimize(
        objective,
        x,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(num_vars)],
        options={'maxiter': max_iter, 'ftol': tolerance}
    )

    # Extract the updated centers and radii

    best_radii = []
