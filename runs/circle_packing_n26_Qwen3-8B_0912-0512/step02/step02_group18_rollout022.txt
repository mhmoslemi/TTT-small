import numpy as np
import cvxpy as cp

def run_packing():

    # Initial positions (centered and spaced)
    centers = np.random.rand(n_circles, 2)  # Random initial positions within [0,1]x[0,1]

    # Variables for optimization
    radii = cp.Variable(n_circles, non_negative=True)

    # Objective: Maximize the total sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []
    for i in range(n_circles):
        x_i, y_i = centers[i]
        # Boundary constraint: circle must be within the unit square
        constraints.append(x_i - radii[i] <= 1.0)

        constraints.append(y_i + radii[i] <= 1.0)

    # Non-overlapping constraint for all pairs of circles
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints.append(dist >= radii[i] + radii[j])

    # Create the problem
    problem = cp.Problem(objective, constraints)

    problem.solve(solver=cp.SCS, eps=1e-5)
