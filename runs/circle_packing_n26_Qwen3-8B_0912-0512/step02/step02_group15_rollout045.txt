    initial_centers[:, 1] = np.clip(initial_centers[:, 1], 0, 1)

    # and boundary distances
    def compute_radii(centers):

                centers[i, 0],
                1 - centers[i, 0],
                centers[i, 1],
                1 - centers[i, 1]
            ])

        return radii
