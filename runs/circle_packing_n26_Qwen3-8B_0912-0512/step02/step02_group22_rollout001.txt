import matplotlib.pyplot as plt
from scipy.optimize import minimize
from itertools import combinations

def run_packing():
    num_circles = 26

    # Generate an initial square grid configuration
    def generate_square_grid(num_circles):
        # Start with an even grid, then shift the last half to fill corners
        points = []
        max_row = int(np.sqrt(num_circles))
        max_col = max_row
        if max_row * max_row < num_circles:
            max_row += 1
            max_col = max_row
        for row in range(max_row):
            for col in range(max_col):
                if row * max_col + col < num_circles:
                    x = col / (max_col - 1)
                    y = row / (max_row - 1)
                    points.append([x, y])
        # Shuffle to prevent bias
        np.random.shuffle(points)
        return np.array(points)
    
    initial_centers = generate_square_grid(num_circles)

    # Define the objective function to maximize total radii
    def objective(positions):
        centers = positions.reshape((num_circles, 2))
        radii = np.zeros(num_circles)
        for i in range(num_circles):
            # Boundary distance
            dist_to_boundary = np.min([centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]])
            # Nearest neighbor distance
            min_neigh_dist = np.inf
            for j in range(num_circles):
                if i != j:
                    dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if dist < min_neigh_dist:
                        min_neigh_dist = dist
            radii[i] = np.min([dist_to_boundary, min_neigh_dist / 2])
        return -np.sum(radii)  # Minimize negative sum

                if dist < 1e-8:
                    return dist - 1e-8
        return 0

    # Optimization bounds
    bounds = [(0, 1) for _ in range(2 * num_circles)]

    # Initial guess
    initial_guess = initial_centers.reshape(-1)

    result = minimize(
