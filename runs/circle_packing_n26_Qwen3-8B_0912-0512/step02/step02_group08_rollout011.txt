import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def hexagonal_packing(n):
    # Hexagonal grid with alternating row offsets
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // cols
        col = i % cols
        row_offset = (row % 2) * (1 / (cols - 1))
        centers[i] = [col / (cols - 1) + row_offset, row / (rows - 1)]
    return centers

def nested_packing(n):
    # Nested concentric layers
    layers = int(np.sqrt(n))
    centers = []
    for i in range(layers):
        layer = i + 1
        radius = layer / 10
        angle = 2 * np.pi / n * i
        for j in range(n):
            ang = angle + 2 * np.pi * j / n
            x = radius * np.cos(ang)
            y = radius * np.sin(ang)
            centers.append([x, y])
    return np.array(centers)

def random_packing(n):
    # Random initial centers
    centers = np.random.rand(n, 2)
    return centers

def pairwise_distances(positions):
    return squareform(pdist(positions))

def obj_func(positions):
    num_circles = len(positions) // 2
    centers = positions.reshape((num_circles, 2))
    # Compute radii based on minimal spacing and boundary
    radii = []
    for i in range(num_circles):
        x, y = centers[i]
        dist_to_boundary = np.min([x, 1 - x, y, 1 - y])
        min_dist = np.inf
        for j in range(num_circles):
            if i != j:
                dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if dist < min_dist:
                    min_dist = dist
        r = np.min([dist_to_boundary, min_dist / 2])
        radii.append(r)
    # Return negative to maximize
    return -np.sum(radii)

def constraint_func(positions):

    centers = positions.reshape((num_circles, 2))

    radii = []

            if dist_matrix[i, j] < radii[i] + radii[j] - 1e-12:
                return dist_matrix[i, j] - (radii[i] + radii[j] - 1e-12)
    return 0
