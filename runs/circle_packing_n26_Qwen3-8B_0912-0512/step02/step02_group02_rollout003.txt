import scipy.optimize as opt

    centers[3] = [0.95, 0.95]  # top-right
    # Fill remaining centers in the center
    for i in range(4, 26):
        angle = (i - 4) * (2 * np.pi / 22)
        r = 0.4  # initial radius to avoid overlap
        x = 0.5 + r * np.cos(angle)
        y = 0.5 + r * np.sin(angle)
        centers[i] = [x, y]

    # Initial guess for radii (all circles to have same radius)
    initial_radii = np.full(26, 1.0 / 26)  # Start with small radii to avoid overlap

    # Define the objective function: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # We minimize negative sum to maximize sum

    # Define constraints:
    # 1. Bounds: radii must be non-negative
    bounds = [(0.0, 1.0) for _ in range(26)]

    # 2. Constraints for non-overlapping circles
    # List of (i, j, min_distance) for circle pairs
    constraints = []
    for i in range(26):
        for j in range(i + 1, 26):
            # Ensure the distance between centers is at least sum of radii
            def distance_constraint(radii):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                # Distance between centers must be >= radii[i] + radii[j]
                return dist - (radii[i] + radii[j])
            constraints.append({'type': 'ineq', 'fun': distance_constraint})

    # Define optimization problem
    result = opt.minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    if not result.success:

    radii = result.x

    # Final validation
    def validate_packing(centers, radii):

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
