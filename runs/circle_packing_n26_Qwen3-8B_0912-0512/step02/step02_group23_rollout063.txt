import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from itertools import combinations

def run_packing():
    num_circles = 26

    # Define initial point configurations
    def generate_hexagonal_grid(num_circles):
        rows = int(np.ceil(np.sqrt(num_circles)))
        cols = int(np.ceil(num_circles / rows))
        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j / (cols - 1)
                y = i / (rows - 1)
                grid.append([x, y])
        # Shuffle to avoid structured pattern
        np.random.shuffle(grid)
        return np.array(grid)

    def generate_rectangular_grid(num_circles):
        rows = int(np.ceil(np.sqrt(num_circles)))
        cols = int(np.ceil(num_circles / rows))
        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j / (cols - 1)
                y = i / (rows - 1)
                grid.append([x, y])
        np.random.shuffle(grid)
        return np.array(grid)

    def generate_fixed_centers(num_circles):
        grid = np.random.rand(num_circles, 2)
        return grid

    # Try 6 different initial point configurations
    initial_configs = []
    for _ in range(6):
        initial_configs.append(generate_hexagonal_grid(num_circles))
        initial_configs.append(generate_rectangular_grid(num_circles))
        initial_configs.append(generate_fixed_centers(num_circles))
    initial_configs = np.array(initial_configs)

    def objective(positions):
        centers = positions.reshape((num_circles, 2))
        radii = np.zeros(num_circles)
        for i in range(num_circles):
            # Boundary distance
            dist_to_boundary = np.min([centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]])
            # Nearest neighbor distance
            min_neigh_dist = np.inf
            for j in range(num_circles):
                if i != j:
                    dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if dist < min_neigh_dist:
                        min_neigh_dist = dist
            radii[i] = np.min([dist_to_boundary, min_neigh_dist / 2])
        return -np.sum(radii)

    def constraint(positions):
        centers = positions.reshape((num_circles, 2))
        for i in range(num_circles):

                if dist < radii[i] + radii[j] - 1e-8:
                    return dist - radii[i] - radii[j] + 1e-8
        return 0

    best_sum = -np.inf
    best_centers = None
    best_radii = None

    for i in range(len(initial_configs)):
        initial_guess = initial_configs[i].reshape(-1)
        bounds = [(0, 1) for _ in range(2 * num_circles)]

        result = minimize(
            objective,
            initial_guess,
            bounds=bounds,
            constraints=[{'type': 'ineq', 'fun': constraint}],
            method='SLSQP',
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-10}
        )

        if result.success:
            centers = result.x.reshape((num_circles, 2))
            radii = np.zeros(num_circles)
            for j in range(num_circles):
                # Boundary distance
                dist_to_boundary = np.min([centers[j, 0], 1 - centers[j, 0], centers[j, 1], 1 - centers[j, 1]])

            sum_radii = np.sum(radii)

                best_centers = centers
