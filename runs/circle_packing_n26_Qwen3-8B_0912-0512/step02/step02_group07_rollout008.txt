    def max_possible_radius(x, y, centers, radii):

        max_r = min(max_r, 1.0 - max(x, y))
        max_r = min(max_r, max(x, y))

        # Check distance to all other circles
        for i, (cx, cy) in enumerate(centers):
            dx = x - cx
            dy = y - cy
            dist = np.sqrt(dx**2 + dy**2)

        return max_r

    def generate_hexagonal_layout(num_circles):

            return np.array([]), np.array([])

        col = num_circles - row * row
        centers = []
        for i in range(row):
            for j in range(col):

                y = i / (row + 1.0)
                centers.append([x, y])

                    y += 0.25 / (row + 1.0)

                    x = j + 0.5 / (row + 1.0)
                    y = i / (row + 1.0) + 0.25 / (row + 1.0)
                    centers.append([x, y])
        return np.array(centers), np.array([0.1] * len(centers))
