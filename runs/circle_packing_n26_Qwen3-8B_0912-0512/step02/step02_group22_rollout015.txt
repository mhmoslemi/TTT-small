import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial import distance

def run_packing():

    # Better hexagonal packing with zig-zag pattern
    # Using a less dense layout to allow for optimization
    # We will try to get 4 rows with staggered circles
    
    def generate_hexagonal_grid(num_circles):
        # 4 rows, 7,6,5,6 circles in each row (staggered)
        positions = []
        for row in range(4):
            if row % 2 == 0:
                # Even row, left-aligned
                row_positions = [(col / 6.0, row / 4.0) for col in range(7)]
            else:
                # Odd row, right-aligned
                row_positions = [(col / 6.0 + 0.5, row / 4.0) for col in range(6)]
            positions.extend(row_positions)
        np.random.shuffle(positions)
        return np.array(positions)
    
    # Generate initial configuration
    initial_centers = generate_hexagonal_grid(num_circles)
    
    # Objective function to maximize the total radii
    def objective(positions):
        centers = positions.reshape((num_circles, 2))
        radii = np.zeros(num_circles)
        for i in range(num_circles):
            # Compute boundary distance
            dist_to_boundary = np.min([centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]])
            # Compute inter-circle distance
            min_neighbor_distance = np.inf
            for j in range(num_circles):
                if i != j:
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < min_neighbor_distance:
                        min_neighbor_distance = d
            # Total radius as constrained by boundary and neighbor
            radii[i] = np.min([dist_to_boundary, min_neighbor_distance / 2])
        return -np.sum(radii)
    
    # Constraint: Prevent circle overlaps
    def constraint(positions):
        centers = positions.reshape((num_circles, 2))
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if dist < 1e-6:
                    return dist - 1e-6
        return 0
    
    # Bounds for center positions
    bounds = [(0, 1) for _ in range(2 * num_circles)]
    
    # Initial guess
    initial_guess = initial_centers.reshape(-1)
    
    # Optimization
    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraint, 'jac': None}],
        method='SLSQP',
        options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8, 'disp': False}
    )
    
    # Extract results
    optimized_centers = result.x.reshape((num_circles, 2))

        optimized_radii[i] = np.min([dist_to_boundary, min_neighbor_distance / 2])
    
    sum_radii = np.sum(optimized_radii)
    
    return (optimized_centers, optimized_radii, sum_radii)
