from scipy.optimize import minimize

def run_packing():

        return centers

    initial_centers = generate_random_centers(num_circles)

    def objective(positions):
        centers = positions.reshape((num_circles, 2))
        radii = np.zeros(num_circles)
        for i in range(num_circles):
            max_r = 0.0
            # Compute distance to all four boundaries
            side_dist = np.array([
                centers[i, 0],
                1.0 - centers[i, 0],
                centers[i, 1],
                1.0 - centers[i, 1]
            ])
            max_r = np.min(side_dist)
            # Compute distance to nearest neighbor
            min_dist = np.inf
            for j in range(i + 1, num_circles):
                dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if dist < min_dist:
                    min_dist = dist
            max_r = np.min([max_r, min_dist / 2.0])
            radii[i] = max_r
        return -np.sum(radii)

                    return dist - 1e-8
        return 0.0
