import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial known configuration with 25 circles arranged in a grid
    centers_base = np.array([
        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
        [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],
        [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
        [0.1, 0.7], [0.3, 0.7], [0.5, 0.7], [0.7, 0.7], [0.9, 0.7],
        [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9]
    ])
    radii_base = np.array([0.1] * 25)

    # Helper function to calculate maximum allowed radius for a circle at (x, y)
    def max_radius(x, y, centers, radii):
        r = 1.0
        for i in range(len(centers)):
            dx = x - centers[i, 0]
            dy = y - centers[i, 1]
            dist = np.sqrt(dx ** 2 + dy ** 2)
            r = min(r, dist - radii[i])
        r = min(r, 1 - max(x, y))
        r = min(r, max(x, y))
        return r

    # Define an objective function for optimization: maximize the sum of radii
    def objective(x):
        # x is a vector of parameters to optimize
        # x[0] is the x position of the new circle
        # x[1] is the y position of the new circle
        # x[2] is the radius of the new circle
        x_pos = x[0]
        y_pos = x[1]
        r_new = max_radius(x_pos, y_pos, centers_base, radii_base)
        return -r_new  # Minimize negative radius for maximization

    # Bounds for optimization: x and y in [0.1, 0.9], radius in [0.01, 0.5]
    bounds = [(0.1, 0.9), (0.1, 0.9), (0.01, 0.5)]

    # Run differential evolution to find the best configuration
    result = opt.differential_evolution(objective, bounds, strategy='best1bin', maxiter=150, popsize=20, tol=1e-10)

    # Extract the best position and radius
    best_x, best_y, best_r = result.x
    best_centers = np.vstack([centers_base, [best_x, best_y]])
    best_radii = np.concatenate([radii_base, [best_r]])

    # Validate the configuration
    def validate_packing(centers, radii):
        n = centers.shape[0]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < 0 or x + r > 1 or
                y - r < 0 or y + r > 1):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"

    is_valid, message = validate_packing(best_centers, best_radii)
    if not is_valid:
        raise ValueError(message)

    total_radius = np.sum(best_radii)
    return best_centers, best_radii, total_radius
