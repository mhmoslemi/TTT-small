import scipy.optimize as opt
from scipy.optimize import Bounds

    def max_possible_radius(x, y, centers, radii):
        max_r = 1.0
        for i in range(len(centers)):
            dx = x - centers[i, 0]
            dy = y - centers[i, 1]
            dist = np.sqrt(dx ** 2 + dy ** 2)
            max_r = min(max_r, dist - radii[i])
        max_r = min(max_r, 1 - max(x, y))
        max_r = min(max_r, max(x, y))
        return max_r

    # Function to compute the objective for optimization: maximize the added circle radius
    def objective(params, centers, radii):

        r = max_possible_radius(x, y, centers, radii)
        return -r
