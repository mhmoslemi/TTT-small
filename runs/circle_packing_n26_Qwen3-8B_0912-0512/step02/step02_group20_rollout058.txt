    def generate_hexagonal_grid(n):

        cols = int(np.ceil(n / rows))
        for r in range(rows):
            for c in range(cols):
                x = c + 0.5 * (r % 2)
                y = r
                grid.append([x, y])
        np.random.shuffle(grid)
        return np.array(grid)

        points = np.array(points)

        # Map the points to the square
        scale_x = 1.0 / total_width if total_width > 0 else 1.0
        scale_y = 1.0 / total_height if total_height > 0 else 1.0
        translated_points = np.array([
            [x * scale_x, y * scale_y] for x, y in points
        ])
        return translated_points
