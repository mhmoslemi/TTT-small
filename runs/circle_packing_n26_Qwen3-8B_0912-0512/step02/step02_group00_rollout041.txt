import cvxpy as cp

def run_packing():
    # Define variable for centers and radii
    n = 26
    centers = cp.Variable((n, 2), name='centers')
    radii = cp.Variable(n, name='radii')
    
    # Define the objective: maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))
    
    # Non-overlap constraint: distance between centers >= sum of radii
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-8)
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        constraints.append(x >= radii[i])
        constraints.append(x <= 1 - radii[i])
        constraints.append(y >= radii[i])
        constraints.append(y <= 1 - radii[i])
    
    # All radii must be non-negative
    constraints.append(radii >= 0)
    
    # Build and solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.OSQP, eps=1e-6)  # Use OSQP for convex optimization

    if not prob.status == 'optimal':
        raise ValueError(f"Optimization failed with status: {prob.status}")

    centers = centers.value
    radii = radii.value
