import scipy.optimize as opt

def run_packing():
    def distance_constraint(i, j):
        def cons(x, centers, radii):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            return dist - radii[i] - radii[j]
        return cons

    def boundary_constraint(i):
        def cons(x, centers, radii):
            x_pos = centers[i, 0]
            y_pos = centers[i, 1]
            r = radii[i]
            return [x_pos - r, x_pos + r, y_pos - r, y_pos + r]

    num_circles = 26

        centers[i, 0] = x
        centers[i, 1] = y

    # Define bounds for optimization
    bounds = [(0.0, 1.0) for _ in range(2 * num_circles)]

    result = opt.minimize(
