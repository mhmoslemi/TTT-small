import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial grid-based packing
    num_circles = 26
    grid_rows = 5
    grid_cols = 6
    spacing = 0.2  # to be adjusted based on optimal packing

    # Initial square grid layout with some circles removed
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    circle_idx = 0
    for row in range(grid_rows):
        for col in range(grid_cols):
            if row % 2 == 0:
                x = col * spacing
                y = row * spacing
            else:
                x = col * spacing + spacing / 2
                y = row * spacing + spacing / 2
            if circle_idx < num_circles:
                centers[circle_idx] = [x, y]
                radii[circle_idx] = spacing / 2
                circle_idx += 1

    def objective(radii):
        return -np.sum(radii)

    def constraint_overlap(i, j):
        return lambda r: np.sqrt(np.sum((centers[i] - centers[j]) ** 2)) - (r[i] + r[j]) >= -1e-8

    def constraint_boundary(i):
        return lambda r: np.min(np.abs(centers[i] - r[i])) >= 1e-8

    cons_overlap = []
    cons_boundary = []

    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            cons_overlap.append(constraint_overlap(i, j))
        cons_boundary.append(constraint_boundary(i))

    # Optimize radii
    bounds = [(0, 1) for _ in range(num_circles)]
    cons = []
    for c in cons_overlap:
        cons.append(c)
    for c in cons_boundary:
        cons.append(c)

    res = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=cons,
        method='SLSQP'
    )

    return centers, res.x, np.sum(res.x)
