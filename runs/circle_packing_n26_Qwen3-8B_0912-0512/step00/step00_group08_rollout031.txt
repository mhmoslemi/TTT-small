import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize

def run_packing():
    n = 26

    # Initial grid-based placement with some adjustment
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col / 4.0 + 0.05
        y = row / 4.0 + 0.05
        if i == 25:  # last circle, place in the center
            x = 0.5
            y = 0.5
        centers[i] = [x, y]

    # Initial guess for radii: based on spacing
    radii = np.full(n, 0.05)

    # Define loss function
    def loss(radii):
        # Penalize circles going out of bounds
        boundary_loss = 0
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                boundary_loss += 1000 * (x - r - 0) ** 2 + 1000 * (x + r - 1) ** 2 + 1000 * (y - r - 0) ** 2 + 1000 * (y + r - 1) ** 2

        # Penalize overlaps between circles
        overlap_loss = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii[i] + radii[j] - 1e-8:
                    overlap_loss += 1000 * (dist - (radii[i] + radii[j] - 1e-8)) ** 2

        return boundary_loss + overlap_loss

    # Optimize radii
    result = minimize(
        loss,
        radii,
        method="L-BFGS-B",
        bounds=[(0, 0.5) for _ in range(n)]
    )

    # Extract final radii and centers
    final_radii = result.x
    final_sum_radii = np.sum(final_radii)

    # If the loss is too high, fallback to a minimal packing
    if loss(final_radii) > 1e6:
        # If optimization fails, use a safe fallback
        fallback_centers = np.random.rand(n, 2)
        fallback_radii = np.full(n, 0.01)
        final_sum_radii = np.sum(fallback_radii)
        return (fallback_centers, fallback_radii, final_sum_radii)

    return (centers, final_radii, final_sum_radii)
