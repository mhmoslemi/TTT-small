import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, NonlinearConstraint
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

# Strategy:
# 1. Start with a hexagonal packing of 26 circles in a square.
# 2. Optimize the positions and radii using a combination of gradient descent and
#    constraint satisfaction to maximize the sum of the radii.
# 3. Fix the positions of the circles near the boundaries and corners to improve packing efficiency.
# 4. Use a non-linear optimization approach with constraints to prevent overlaps and ensure boundary constraints.
# 5. Add a post-processing step to refine the packing and fill gaps between circles.

def run_packing():
    # Initial guess for circle centers (hexagonal packing)
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Create an initial hexagonal packing pattern
    for i in range(n):
        row = i // 5
        col = i % 5
        if row % 2 == 0:
            x = 0.1 + col * 0.2
        else:
            x = 0.15 + col * 0.2
        y = 0.1 + row * 0.25
        centers[i] = [x, y]
    
    # Initial radii guess
    radii = np.full(n, 0.05)
    
    # Optimization problem
    def objective(x):
        return -np.sum(x[1:])  # Maximize sum of radii
    
    # Constraints
    def constraint1(x):
        # Ensure all circles are within the unit square
        for i in range(n):
            x_i, y_i = x[2*i], x[2*i + 1]
            r_i = x[n + i]
            if not (0 <= x_i - r_i and x_i + r_i <= 1):
                return -1
        return 0
    
    def constraint2(x):
        # Ensure no overlaps between circles
        for i in range(n):
            for j in range(i + 1, n):
                x_i, y_i = x[2*i], x[2*i + 1]
                x_j, y_j = x[2*j], x[2*j + 1]
                r_i = x[n + i]
                r_j = x[n + j]
                dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                if dist < r_i + r_j - 1e-8:
                    return -1
        return 0
    
    # Bounds for x and y coordinates (0 to 1)
    bounds_x = [(0, 1)] * n
    bounds_y = [(0, 1)] * n
    bounds_radii = [(0, 1)] * n
    bounds = bounds_x + bounds_y + bounds_radii
    
    # Initial guess for x and y (centers) and radii
    x0 = centers.flatten()
    x0 = x0.tolist()
    x0 += radii.tolist()
    
    # Optimization constraints
    cons = [
        {'type': 'ineq', 'fun': constraint1},
        {'type': 'ineq', 'fun': constraint2}
    ]
    
    # Perform optimization
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Extract final centers and radii
    centers_final = result.x[0:n*2].reshape(n, 2)
    radii_final = result.x[n*2:]
    
    # Post-processing: refine corners and edges
    for i in range(n):
        x, y = centers_final[i]
        r = radii_final[i]
        # Adjust radii to be as large as possible without crossing boundaries
        if x - r < 0:
            radii_final[i] = x
        elif x + r > 1:
            radii_final[i] = 1 - x
        if y - r < 0:
            radii_final[i] = y
        elif y + r > 1:
            radii_final[i] = 1 - y
    
    # Final sum of radii
    sum_radii = np.sum(radii_final)
    
    return (centers_final, radii_final, sum_radii)
