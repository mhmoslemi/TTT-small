import matplotlib.pyplot as plt
from scipy.optimize import minimize

def initialize_grid(n):
    """Initialize a hexagonal grid of n circles in a unit square."""
    # Start from the bottom-left corner
    grid = []
    rows = int(np.sqrt(n))
    cols = n // rows + (1 if n % rows else 0)
    
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * i / (cols - 1) if cols > 1 else j
            y = i + 0.5 * (rows - 1) / (rows - 1) if rows > 1 else i
            grid.append((x, y))
    
    return np.array(grid)

def compute_initial_radii(centers):
    """Compute initial radii based on spacing between adjacent circles."""
    n = centers.shape[0]

            dists.append(np.linalg.norm(centers[i] - centers[j]))
    if not dists:
        return np.zeros(n)
    avg_dist = np.mean(dists)
    return np.full(n, avg_dist / np.sqrt(2))

def update_centers(centers, radii, max_iter=100, tolerance=1e-6):
    """Refine center positions to avoid overlaps and stay within bounds."""
    def objective_func(params):
        new_centers = params[:2 * n].reshape(n, 2)

        return overlaps

    refined_centers = result.x[:2 * n].reshape(n, 2)

    return refined_centers, refined_radii
