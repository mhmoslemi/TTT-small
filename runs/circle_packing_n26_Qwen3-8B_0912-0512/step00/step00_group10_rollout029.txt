import numpy as np
import scipy.optimize as opt
from scipy.spatial import distance

def run_packing():
    # Initial number of circles
    n = 26
    
    # Initial positions based on a hexagonal grid (5 rows, 6 columns)
    # This is a starting guess that fits tightly in the square
    # We will refine this through optimization
    positions = []
    for row in range(5):
        for col in range(6):
            x = col + 0.5 * (row % 2)
            y = row + 0.5
            if 0 <= x <= 1 and 0 <= y <= 1:
                positions.append([x, y])
    
    # Convert to NumPy array
    centers = np.array(positions)
    
    # Initial guess for radii (based on spacing)
    radii = np.full(n, 0.1)
    
    # Objective function: maximize the sum of radii
    def objective(x):
        # x contains all variables: first 2*n are the centers, next n are the radii
        centers_flat = x[:2 * n].reshape(n, 2)
        radii_flat = x[2 * n:].reshape(n)
        
        # Update radii to be nonnegative
        radii_flat = np.clip(radii_flat, 0, np.inf)
        
        # Check if any circle goes outside the unit square
        for i in range(n):
            x_center, y_center = centers_flat[i]
            r = radii_flat[i]
            if not (0 - r <= x_center <= 1 + r and 0 - r <= y_center <= 1 + r):
                return -np.inf  # Penalize invalid configurations
    
        # Check if any circles overlap
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    return -np.inf  # Penalize overlapping circles
        
        # Return negative sum of radii to maximize in minimization
        return -np.sum(radii_flat)
    
    # Bounds: centers must be within [0, 1]^2, radii must be >= 0
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1), (0, 1)])  # x and y for each center
        bounds.extend([(0, np.inf)])    # radius for each circle
    
    # Optimization
    result = opt.minimize(
        fun=objective,
        x0=np.concatenate([centers.flatten(), radii]),
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 500}
    )

    # Calculate sum of radii
    sum_radii = np.sum(optimized_radii)
    
    return optimized_centers, optimized_radii, sum_radii
