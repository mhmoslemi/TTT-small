import numpy as np
import scipy.optimize as optimize

def compute_distance(center1, center2):
    return np.sqrt(np.sum((center1 - center2) ** 2))

def objective_function(x, n_circles):
    x_flat = x.reshape(n_circles, 4)
    centers = x_flat[:, :2]
    radii = x_flat[:, 2:]
    return -np.sum(radii)

def constraint_center_in_box(x, n_circles):
    x_flat = x.reshape(n_circles, 4)
    centers = x_flat[:, :2]
    radii = x_flat[:, 2:]
    return np.concatenate([
        centers[:, 0] - radii[:, 0],
        centers[:, 1] - radii[:, 1],
        1 - centers[:, 0] - radii[:, 0],
        1 - centers[:, 1] - radii[:, 1]
    ])

def constraint_circle_distances(x, n_circles):
    x_flat = x.reshape(n_circles, 4)
    centers = x_flat[:, :2]
    radii = x_flat[:, 2:]
    constraints = []
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dist = compute_distance(centers[i], centers[j])
            constraints.append(dist - radii[i] - radii[j])
    return np.array(constraints)

def run_packing():
    n_circles = 26
    initial_centers = np.random.rand(n_circles, 2)
    initial_radii = np.random.rand(n_circles) * 0.1

    x0 = np.concatenate([initial_centers, initial_radii], axis=1)

    bounds = []
    for _ in range(n_circles):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])
    
    # Optimization: Maximize sum of radii
    result = optimize.minimize(
        objective_function,
        x0,
        args=(n_circles,),
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_center_in_box, 'args': (n_circles,)},
            {'type': 'ineq', 'fun': constraint_circle_distances, 'args': (n_circles,)}
        ]
    )

    x_opt = result.x
    centers = x_opt.reshape(n_circles, 4)[:, :2]
    radii = x_opt.reshape(n_circles, 4)[:, 2:]
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
