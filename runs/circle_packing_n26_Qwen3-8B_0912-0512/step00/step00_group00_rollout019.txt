from scipy.optimize import minimize

def run_packing():

    rows = 6
    cols = 5
    spacing = 1.0 / (cols - 1)
    half_spacing = spacing / 2

            x = j * spacing
            y = i * spacing
            if i % 2 == 1:
                x += half_spacing
            centers.append((x, y))

    centers = np.array(centers)

                    return -np.inf
        return np.sum(radii)
