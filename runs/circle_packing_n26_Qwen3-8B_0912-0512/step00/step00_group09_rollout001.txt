import cvxpy as cp

        centers[i] = [x, y]

    radii[0] = 0.1

    # Now, perform convex optimization to maximize the total sum of radii
    # Define variables
    r = cp.Variable(n)
    x = cp.Variable((n, 2))

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(r))
