# Define the maximum number of circles
N_CIRCLES = 26

def run_packing():
    """Packs 26 circles in a unit square to maximize the sum of radii."""
    # Initial guess: place circles in a grid-like pattern
    def initial_centers():
        centers = []
        for i in range(N_CIRCLES):
            row = i // 5
            col = i % 5
            x = col / 4.0
            y = row / 4.0
            centers.append([x, y])
        return np.array(centers)

    initial_centers = initial_centers()
    initial_radii = np.full(N_CIRCLES, 0.05)

    # Objective function: maximize sum of radii
    def objective(x):
        # x contains all variables: [c1_x, c1_y, r1, c2_x, c2_y, r2, ...]
        centers = x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
        radii = x[2 * N_CIRCLES:]
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraints: circles must not overlap and stay within the unit square
    def constraints(x):
        # Extract variables
        centers = x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
        radii = x[2 * N_CIRCLES:]

        # Constraints for each circle: must stay within unit square
        constraints = []
        for i in range(N_CIRCLES):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12:
                constraints.append((x_i - r_i + 1e-12))
            if x_i + r_i > 1 + 1e-12:
                constraints.append((x_i + r*r_i - 1 - 1e-12))
            if y_i - r_i < -1e-12:
                constraints.append((y_i - r_i + 1e-12))
            if y_i + r_i > 1 + 1e-12:
                constraints.append((y_i + r_i - 1 - 1e-12))

        # Constraints for non-overlapping circles
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                distance = np.sqrt(dx*dx + dy*dy)

                    constraints.append((distance - radii[i] - radii[j] + 1e-12))

    # Initial guess for optimization
    initial_guess = np.concatenate([initial_centers.reshape(-1), initial_radii])

        bounds.extend([(0, 1), (0, 1)])

    result = minimize(

        bounds=bounds,

    valid, msg = validate_packing(centers, radii)

def validate_packing(centers, radii):
    """Validates a circle packing."""
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
