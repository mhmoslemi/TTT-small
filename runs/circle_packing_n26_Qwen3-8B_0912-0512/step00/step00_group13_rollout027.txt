def generate_hexagonal_grid(num_circles, side_length):
    """Generate a hexagonal grid of circles with given side length."""
    grid = []

        row = i // 6
        col = i % 6
        x = col * side_length
        y = row * (np.sqrt(3) / 2) * side_length
        grid.append((x, y))
    return np.array(grid)
