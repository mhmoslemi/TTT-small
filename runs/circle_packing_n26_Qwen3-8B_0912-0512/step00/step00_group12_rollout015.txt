import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial placement: grid grid with some flexibility
    num_circles = 26
    rows = 5
    cols = 6

    # Grid positions with some spread
    x_positions = np.linspace(0.1, 0.9, cols)
    y_positions = np.linspace(0.1, 0.9, rows)
    centers = np.array([[x, y] for y in y_positions for x in x_positions])
    radii = np.full(num_circles, 0.1)

    # Objective function to maximize total radius sum
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    # Constraints
    constraints = []

    # Constraint: All circles within the unit square
    def in_square(x, r):
        return (x[0] - r >= 0) and (x[0] + r <= 1) and (x[1] - r >= 0) and (x[1] + r <= 1)
    
    # Create constraints for each circle
    for i in range(num_circles):
        def in_square_constraint(x, i=i):
            x_i = x[i * 2]
            y_i = x[i * 2 + 1]
            r_i = radii[i]
            return in_square([x_i, y_i], r_i)
        constraints.append({'type': 'ineq', 'fun': in_square_constraint})

    # Constraint: Distances between centers >= sum of radii
    def dist_constraint(x, i, j):
        x_i = x[i * 2]
        y_i = x[i * 2 + 1]
        x_j = x[j * 2]
        y_j = x[j * 2 + 1]
        r_i = radii[i]
        r_j = radii[j]
        dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
        return dist - (r_i + r_j)

    # Create constraints for all pairs
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            def dist_constraint_func(x, i=i, j=j):
                return dist_constraint(x, i, j)
            constraints.append({'type': 'ineq', 'fun': dist_constraint_func})

    # Optimization bounds: x and y in [0, 1], radii >= 0
    bounds = []
    for i in range(num_circles):
        bounds.append((0, 1))
        bounds.append((0, 1))
        bounds.append((0, 1))  # radius

    # Initial guess
    x0 = []
    for i in range(num_circles):
        x0.extend([centers[i][0], centers[i][1], radii[i]])

    result = opt.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    # Extract centers and radii
    centers_opt = np.array(result.x[::3]).reshape(-1, 2)
    radii_opt = result.x[2::3]

    # Validate the result
    valid, message = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Validation failed: {message}")

    return centers_opt, radii_opt, np.sum(radii_opt)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
