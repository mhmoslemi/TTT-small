import numpy as np
import cvxpy as cp

def run_packing():
    n_circles = 26

    # Define variables
    centers = cp.Variable((n_circles, 2))
    radii = cp.Variable(n_circles)

    # Objective: Maximize the sum of the radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints: Each circle lies within the square
    constraints = []
    for i in range(n_circles):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r >= 0)
        constraints.append(x + r <= 1)
        constraints.append(y - r >= 0)
        constraints.append(y + r <= 1)

    # Constraints: Each pair of circles does not overlap
    for i in range(n_circles):
        for j in range(i + 1, n_circles):

            dy = centers[i, 1] - centers[j, 1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints.append(dist >= radii[i] + radii[j])

    # Constraints: Radii must be non-negative
    for i in range(n_circles):
        constraints.append(radii[i] >= 0)

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, eps=1e-5)

    # Extract results
    centers_np = centers.value
    radii_np = radii.value
    sum_radii = np.sum(radii_np)
