import numpy as np
import scipy.optimize as opt
import matplotlib.pyplot as plt

def run_packing():
    # Initialize centers in a hexagonal grid
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Start with a hexagonal grid
    rows = int(np.sqrt(n))
    cols = int(np.ceil(n / rows))
    radius = 0.1
    for i in range(rows):
        for j in range(cols):
            idx = i * cols + j
            if idx >= n:
                break
            x = j + 0.5 * i
            y = i
            centers[idx] = [x, y]
            radii[idx] = radius

    # Resize radii to fit within the square [0,1]^2
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum (i.e., maximize sum)

    def constraint0(radii):
        return np.sum(radii) - 2.636

    def constraint1(radii):
        return np.sum(radii) - 2.636 - 1e-6

    # Set up bounds for radii
    bounds = [(0, 1) for _ in range(n)]

    # Add constraints
    cons = [
        {'type': 'ineq', 'fun': constraint0},
        {'type': 'ineq', 'fun': constraint1}
    ]

    # Optimization
    result = opt.minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-10, 'maxiter': 1000}
    )

    # Update final centers and radii
    final_centers = centers.copy()
    final_radii = result.x

    # Validate the packing
    def validate_packing(centers, radii):
        n = centers.shape[0]
        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"

    is_valid, msg = validate_packing(final_centers, final_radii)
    if not is_valid:
        raise ValueError(f"Packing validation failed: {msg}")

    sum_radii = np.sum(final_radii)
    return (final_centers, final_radii, sum_radii)
