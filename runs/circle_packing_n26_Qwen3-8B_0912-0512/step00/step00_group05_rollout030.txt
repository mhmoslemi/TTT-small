import numpy as np
import cvxpy as cp

def run_packing():
    n = 26
    # Base grid coordinates (hexagonal packing)
    # Start with a hexagonal grid of circles
    # We'll use a grid of rows, with offset rows to simulate a hexagonal packing
    # We'll place the first row with 9 circles, then 8, then 7, etc.
    
    # To fit into the unit square, we will use a grid of positions
    # But for now, let's place the circles in a base hexagonal grid
    # We'll use 5 rows, with 5, 5, 5, 5, 6 circles per row (total 25 circles)
    # Then one more circle at the end to make it 26
    
    # Base spacing (for hexagonal grid)
    spacing = 0.2
    rows = 5
    cols = 5  # first row
    starts = np.linspace(0, 1, rows)
    cols = np.linspace(0, 1, cols)
    
    # Create grid for the first 25 circles
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * (np.sqrt(3) / 2) * spacing
            centers.append([x, y])
    
    # Add one more circle to make 26
    centers.append([0.5, 1.0])
    
    # Initial radii (same for all)
    initial_radii = np.full(n, spacing / 2)
    
    # Now optimize the radii and positions to maximize the sum, satisfying:
    # 1. All circles are within the square
    # 2. All circles are non-overlapping
    # 3. All circles have non-negative radii
    
    # Define variables
    center_vars = cp.Variable((n, 2))
    radius_vars = cp.Variable(n, non_negative=True)
    
    # Objective: Maximize sum of radii
    obj = cp.Maximize(cp.sum(radius_vars))
    
    # Constraints
    # 1. All circles must be within [0, 1] x [0, 1]
    constraints = []
    for i in range(n):
        constraints.append(center_vars[i, 0] >= 0)
        constraints.append(center_vars[i, 0] <= 1)
        constraints.append(center_vars[i, 1] >= 0)
        constraints.append(center_vars[i, 1] <= 1)
    
    # 2. All circles must have non-negative radii
    # Already handled by radius_vars >= 0
    
    # 3. Distance between centers is at least the sum of radii
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((center_vars[i, 0] - center_vars[j, 0])**2 + (center_vars[i, 1] - center_vars[j, 1])**2)
            constraints.append(dist >= radius_vars[i] + radius_vars[j])
    
    # Initialize with the base grid and radii
    center_vars.value = np.array(centers)
    radius_vars.value = initial_radii
    
    # Solve the optimization problem
    problem = cp.Problem(obj, constraints)
    problem.solve()
    
    # Extract the results
    centers = center_vars.value
    radii = radius_vars.value
    sum_radii = np.sum(radii)
    
    return (centers, radii, sum_radii)
