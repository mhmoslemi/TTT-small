import numpy as np
import random
from scipy.optimize import minimize
import matplotlib.pyplot as plt

def run_packing():
    # Initial hexagonal grid of circles with varying radii
    # This is just a rough initial placement to avoid overlapping at first
    def initial_placement():
        centers = []
        radii = []
        for i in range(26):
            x = random.uniform(0, 1)
            y = random.uniform(0, 1)
            r = random.uniform(0.01, 0.05)
            centers.append([x, y])
            radii.append(r)
        return np.array(centers), np.array(radii)

    centers, radii = initial_placement()

    # Define the loss function for optimization
    def loss(params):
        # Unpack parameters
        new_centers = params[:26 * 2].reshape(26, 2)
        new_radii = params[26 * 2:]

        # Check if any radius is negative
        if np.any(new_radii < 0):
            return float('inf')

        # Check boundaries
        boundary_penalty = 0
        for i in range(26):
            x, y = new_centers[i]
            r = new_radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                boundary_penalty += 1000000

        # Check for overlaps
        overlap_penalty = 0
        for i in range(26):
            for j in range(i + 1, 26):
                dx = new_centers[i, 0] - new_centers[j, 0]
                dy = new_centers[i, 1] - new_centers[j, 1]

                if dist < new_radii[i] + new_radii[j] - 1e-12:

        # Return total penalty
        return boundary_penalty + overlap_penalty

    # Optimization using simulated annealing
    initial_guess = np.concatenate([centers.flatten(), radii])
    result = minimize(loss, initial_guess, method='L-BFGS-B')

    # Extract the results

    total_sum = np.sum(optimal_radii)
    return optimal_centers, optimal_radii, total_sum

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
