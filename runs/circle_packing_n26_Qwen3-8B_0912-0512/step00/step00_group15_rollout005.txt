import numpy as np
import scipy.optimize as optimize

def run_packing():

    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # We'll use a simplified layout to seed the optimization
    # First, place rows with hexagonal spacing
    row_height = 0.55  # Adjust to optimize
    col_width = 0.55
    col_start = 0.25
    row_start = 0.25

    col_index = 0
    for i in range(n):
        if i % 3 == 0:
            col_index += 1
        row = i // 3
        col = i % 3
        x = col_start + col * col_width
        y = row_start + row * row_height
        centers[i] = [x, y]
        radii[i] = 0.15  # Placeholder radii

    # Define a helper function to compute the total sum of radii
    def total_sum(radii):
        return np.sum(radii)

    # Optimization: Use gradient descent to maximize radii while ensuring constraints
    # Function that defines the objective and constraints for nlp
    def objective(x):
        return -total_sum(x)  # Minimize negative sum to maximize radii

    # Define constraints:
    # 1. All circles must stay within [0, 1]^2
    # 2. All circles must be non-overlapping
    # 3. Radius >= 0

    # First, fix the initial centers (we will optimize radii only)
    # This is a simplified version that only adjusts radii to fit non-overlapping
    # A better method would be to also refine positions, but this is a starting point

    def constraint_circle(pos, radii):
        for i in range(n):
            x = pos[i][0]
            y = pos[i][1]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return -1
        return 0

    def constraint_overlap(pos, radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = pos[i][0] - pos[j][0]
                dy = pos[i][1] - pos[j][1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    # Optimize radii
    result = optimize.minimize(
        fun=objective,
        x0=radii,
        bounds=[(0, 1)] * n,
        constraints=[
            {"type": "ineq", "fun": lambda x: constraint_circle(centers, x)},
            {"type": "ineq", "fun": lambda x: constraint_overlap(centers, x)}

    )

    centers = np.array(centers)
    radii = result.x
    sum_radii = total_sum(radii)
