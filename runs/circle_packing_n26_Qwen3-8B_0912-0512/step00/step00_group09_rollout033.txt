import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Define initial placement of circles near the boundaries
    # We start by placing all circles near the boundaries, evenly distributed
    n = 26
    centers = np.random.rand(n, 2)
    # Shift circles to be near the boundaries
    centers[:, 0] = np.clip(centers[:, 0], 0.1, 0.9)
    centers[:, 1] = np.clip(centers[:, 1], 0.1, 0.9)
    radii = np.random.rand(n) * 0.15  # Initial guess for radii

    def objective_function(radii, centers):
        # Compute repulsion forces
        repulsion = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    repulsion += (radii[i] + radii[j] - dist) ** 2
        return repulsion

    # Combine centers and radii into a single parameter vector
    def objective(params):
        radii = params[:n]
        centers = params[n:]
        centers = centers.reshape(n, 2)
        return objective_function(radii, centers)

    # Initial guess: radii and centers
    initial_guess = np.hstack((radii, centers.flatten()))
    result = optimize.minimize(
        objective,
        initial_guess,
        method='Nelder-Mead',
        options={'xatol': 1e-6, 'fatol': 1e-6, 'disp': False}
    )
    params = result.x
    radii = params[:n]
    centers = params[n:].reshape(n, 2)

    # Final validation
    def validate(centers, radii):
        n = centers.shape[0]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    if not validate(centers, radii):
        raise ValueError("Invalid packing after optimization")

    total_radius = np.sum(radii)
    return centers, radii, total_radius
