    centers[25, 0] = 0.5
    centers[25, 1] = 0.5
    
    # Optimization objective: maximize the sum of radii
    # Subject to: circles do not overlap and fit within the unit square
    
    # Define the optimization variables: radii
    radii = np.full(26, radius)
    
    # Optimization function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization problem
    
    # Constraints:
    # 1. Distance between circle centers must be >= sum of radii
    # 2. Radii must be non-negative
    # 3. Circle must be inside the unit square
    
    def constraint_dist(radii):
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0
    
    def constraint_boundary(radii):
        for i in range(26):
            x = centers[i, 0]
            y = centers[i, 1]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:

    bounds = [(0, np.inf) for _ in range(26)]

    initial_guess = np.full(26, radius)
    
    # Run optimization
    result = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        constraints=[

        ]
