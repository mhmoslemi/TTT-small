import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Generate initial positions on a hexagonal grid
    num_circles = 26
    positions_hex = []
    for i in range(num_circles):
        row = i // 7
        col = i % 7
        x = col + 0.5 * (row % 2)
        y = row
        positions_hex.append([x, y])
    centers = np.array(positions_hex)
    
    # Center the grid to fit within [0, 1]×[0, 1]
    centers -= np.min(centers, axis=0)
    centers /= np.max(centers, axis=0)

    # Initial guess for radii: uniform radius
    radii = np.full(num_circles, 0.05)

    # Define objective function: maximize sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize negative sum to maximize

    # Define constraints: non-overlapping circles and within square
    def constraints(x):
        r = x
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < r[i] + r[j] - 1e-12:
                    return False
        for i in range(num_circles):
            if centers[i, 0] - r[i] < -1e-12 or centers[i, 0] + r[i] > 1 + 1e-12:
                return False
            if centers[i, 1] - r[i] < -1e-12 or centers[i, 1] + r[i] > 1 + 1e-12:
                return False
        return True

    # Bounds for radii: non-negative
    bounds = Bounds(0, 1)

    # Optimization
    result = opt.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints={"type": "success", "fun": constraints},
        method="SLSQP"
    )

    # Extract final centers and radii
    final_centers = centers
    final_radii = result.x
    sum_radii = np.sum(final_radii)

    return final_centers, final_radii, sum_radii
