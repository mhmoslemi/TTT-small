import numpy as np
import cvxpy as cp
import matplotlib.pyplot as plt

def run_packing():
    n = 26  # Number of circles

    # Initial guess for centers: grid-based
    num_per_row = 5
    num_per_col = 6
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i % num_per_row
        col = i // num_per_row
        centers[i, 0] = col / (num_per_row - 1)
        centers[i, 1] = row / (num_per_col - 1)

    # Define variables
    r = cp.Variable(n)
    x = cp.Variable((n, 2))

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = []
    for i in range(n):
        constraints.append(x[i, 0] >= 0)
        constraints.append(x[i, 0] <= 1)
        constraints.append(x[i, 1] >= 0)
        constraints.append(x[i, 1] <= 1)
        constraints.append(r[i] >= 0)

    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((x[i, 0] - x[j, 0]) ** 2 + (x[i, 1] - x[j, 1]) ** 2)
            constraints.append(dist >= r[i] + r[j])

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve()

    if not problem.status in ["OPTIMAL", "OPTIMAL_INFEASIBLE"]:
        raise RuntimeError(f"Optimization failed: {problem.status}")

    # Extract solution
    centers_opt = x.value
    radii_opt = r.value

    # Return centers, radii, and sum of radii
    return centers_opt, radii_opt, np.sum(radii_opt)
