import numpy as np
import cvxpy as cp
import matplotlib.pyplot as plt

def run_packing():
    # Initial step: generate a hexagonal packing of 26 circles in a square.
    # Place points in a hexagonal grid, then project into the unit square.
    # This is a heuristic to avoid manual coding of initial positions.

    def generate_grid(n):
        max_rows = int(np.sqrt(n))
        max_cols = int(np.ceil(n / max_rows))
        grid = []
        for i in range(max_cols):
            for j in range(max_rows):
                x = i / (max_cols - 1)
                y = j / (max_rows - 1)
                grid.append((x, y))
        # Add remaining points
        for i in range(max_cols - len(grid)):
            x = i / (max_cols - 1)
            y = (max_rows - 1) / (max_rows - 1)
            grid.append((x, y))
        # Add remaining points in last row
        for i in range(max_cols - len(grid)):
            x = (max_cols - 1) / (max_cols - 1)
            y = i / (max_rows - 1)
            grid.append((x, y))
        return np.array(grid)

    # Generate an initial set of circle centers
    centers = generate_grid(26)

    # Radius variables (we will optimize them)
    radii = cp.Variable(26)

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Boundary constraint
    for i in range(26):
        x, y = centers[i]
        constraints.append(x - radii[i] >= 0)
        constraints.append(x + radii[i] <= 1)
        constraints.append(y - radii[i] >= 0)
        constraints.append(y + radii[i] <= 1)

    # Non-overlapping constraint for all pairs of circles
    for i in range(26):
        for j in range(i + 1, 26):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints.append(dist >= radii[i] + radii[j])

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.SCS, max_iter=1000)

    # Extract the results
    centers_opt = centers
    radii_opt = radii.value
    sum_radii = cp.sum(radii).value

    # Return the result
    return (centers_opt, radii_opt, sum_radii)
