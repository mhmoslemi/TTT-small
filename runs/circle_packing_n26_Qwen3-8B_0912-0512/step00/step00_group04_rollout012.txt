import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    # Initial grid-based configuration
    num_circles = 26
    grid_rows = 5
    grid_cols = 5
    centers = []
    radii = []

    # Create a grid of positions
    for row in range(grid_rows):
        for col in range(grid_cols):
            x = col / (grid_cols - 1)
            y = row / (grid_rows - 1)
            centers.append([x, y])
            radii.append(0.0)

    # Convert to numpy arrays
    centers = np.array(centers)
    initial_radii = np.array(radii)

    # Define objective function to maximize total area (which is equivalent to maximizing sum of radii)
    def objective(radii):
        return -np.sum(radii)  # Negative because we use minimization

    # Define constraints
    def constraint_center_in_square(radii):
        return np.array([np.sum((centers[i] - radii[i]) ** 2) for i in range(num_circles)])  # Not used directly

    def constraint_circle_inside(radii):
        return np.array([centers[i, 0] - radii[i] >= -1e-12 and centers[i, 0] + radii[i] <= 1 + 1e-12 for i in range(num_circles)])

    def constraint_no_overlap(radii):
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                constraints.append(dist >= radii[i] + radii[j] - 1e-12)
        return np.array(constraints)

    # Combine constraints into a common function
    def constraint_all(radii):
        return np.concatenate([
            constraint_circle_inside(radii),
            constraint_no_overlap(radii)
        ])

    # Set bounds for radii
    bounds = [(0, 1) for _ in range(num_circles)]

    # Optimization
    result = minimize(
        fun=objective,
        x0=initial_radii,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraint_all},
        method='SLSQP'
    )

    # Extract results
    optimized_radii = result.x
    optimized_centers = centers

    # Validate the packing
    valid, msg = validate_packing(optimized_centers, optimized_radii)
    if not valid:
        raise ValueError(f"Invalid packing: {msg}")

    return optimized_centers, optimized_radii, np.sum(optimized_radii)
