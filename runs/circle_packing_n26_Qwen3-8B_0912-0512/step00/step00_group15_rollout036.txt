import numpy as np
from scipy.optimize import minimize
from itertools import combinations

def run_packing():
    # Define the number of circles
    N = 26

    # Initial guess: Place circles in a hexagonal lattice inside the square
    # Generate 10 rows of circles, using a hexagonal grid
    n_rows = 5
    n_cols = 5
    circles_per_row = [5, 6, 6, 5, 5]
    total = sum(circles_per_row)
    assert total == N

    # Generate hexagonal grid
    centers = []
    for i in range(n_rows):
        row_center = 0.5 + (0.5 - i * 0.25)  # Adjust spacing to make it fit in [0,1]
        for j in range(circles_per_row[i]):
            x = row_center + j * (0.3 / (n_cols - 1))
            y = 0.5
            centers.append([x, y])
    centers = np.array(centers)

    # Initial radius guess: 0.08, which is a reasonable starting point
    radii = np.full(N, 0.08)

    # Define objective function: maximize total radius sum
    def objective(x):
        new_radii = x[:N]
        return -np.sum(new_radii)  # Minimize the negative of the sum

    # Define constraints: all circles must be within square, and no overlaps
    def constraint1(x):
        new_radii = x[:N]
        for i in range(N):
            if centers[i, 0] - new_radii[i] < -1e-12 or centers[i, 0] + new_radii[i] > 1 + 1e-12:
                return -1
            if centers[i, 1] - new_radii[i] < -1e-12 or centers[i, 1] + new_radii[i] > 1 + 1e-12:
                return -1
        return 0

    def constraint2(x):
        new_radii = x[:N]
        for i, j in combinations(range(N), 2):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < new_radii[i] + new_radii[j] - 1e-12:
                return -1
        return 0

    # Bounds: radii ≥ 0, and no upper bound
    bounds = [(0, None) for _ in range(N)]

    # Run optimization
    result = minimize(
        objective,
        x0=np.concatenate((centers.ravel(), radii)),
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2}
        ],
        method='SLSQP'
    )

    # Extract centers and radii
    final_centers = result.x[:N*2].reshape(N, 2)
    final_radii = result.x[N*2:]

    # Validate packing
    valid, message = validate_packing(final_centers, final_radii)
    if not valid:
        raise ValueError(message)

    sum_radii = np.sum(final_radii)
    return final_centers, final_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
