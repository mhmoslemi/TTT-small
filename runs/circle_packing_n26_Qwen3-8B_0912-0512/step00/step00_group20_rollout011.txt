import matplotlib.pyplot as plt
from scipy.optimize import minimize

def run_packing():

                y = i / 5
                grid.append((x, y))
        return np.array(grid)

    centers = initial_grid()
    radii = np.full(26, 0.05)  # Initial small radii

    # Objective function: maximize sum of radii
    def objective(x):
        # x is a 1D array of [x1, y1, r1, x2, y2, r2, ...]
        # We use a penalty function to enforce constraints
        penalty = 0.0
        n = len(centers)
        # Update centers and radii from x
        flat_centers = centers.flatten()
        flat_radii = x[2 * n : 3 * n]
        x_flat = x[:2 * n]
        y_flat = x[2 * n : 3 * n]

        # Enforce boundary constraints and non-overlap
        for i in range(n):
            x_i = x_flat[i]
            y_i = y_flat[i]
            r_i = flat_radii[i]
            # Boundary checks
            if x_i - r_i < -1e-8:
                penalty += (x_i - r_i + 1e-8) ** 2
            if x_i + r_i > 1 + 1e-8:
                penalty += (x_i + r_i - 1 - 1e-8) ** 2
            if y_i - r_i < -1e-8:
                penalty += (y_i - r_i + 1e-8) ** 2
            if y_i + r_i > 1 + 1e8:
                penalty += (y_i + r_i - 1 - 1e-8) ** 2

            # Non-overlap checks
            for j in range(i + 1, n):
                dx = x_flat[i] - x_flat[j]
                dy = y_flat[i] - y_flat[j]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < flat_radii[i] + flat_radii[j] - 1e-8:
                    penalty += (dist - flat_radii[i] - flat_radii[j] + 1e-8) ** 2

        # Objective: maximize sum of radii, but we minimize a negative of it
        return -np.sum(flat_radii) + 100 * penalty

    # Flatten the centers to allow x and y to be optimized
    flat_centers = centers.flatten()
    x0 = np.concatenate([flat_centers, np.full(26, 0.05)])
    result = minimize(objective, x0, method='L-BFGS-B', tol=1e-6, options={'maxiter': 500, 'disp': False})

    # Reconstruct centers and radii
    flat_centers = result.x[:2 * 26]
    flat_radii = result.x[2 * 26 : 3 * 26]

    centers = flat_centers.reshape(26, 2)
    radii = flat_radii

    return centers, radii, np.sum(radii)
