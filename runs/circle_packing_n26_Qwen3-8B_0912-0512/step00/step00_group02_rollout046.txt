import cvxpy as cp

def run_packing():

    def generate_hex_grid(num_circles):
        rows = int(np.sqrt(num_circles))
        cols = (num_circles + rows - 1) // rows
        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j + 0.5 * i
                y = i
                if x < 1 and y < 1:
                    grid.append((x, y))
        return np.array(grid)

    centers = generate_hex_grid(26)
    n = len(centers)

    # Constraint: Each circle lies within [0, 1]^2
    # Constraint: No overlap with other circles
    # Objective: Maximize sum of radii

    # Define variables
    radii = cp.Variable(n)
    x = cp.Variable(n)
    y = cp.Variable(n)

    # Objective: Maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints:
    # Each circle is within [0, 1]^2
    constraints = [
        x >= 0,
        x <= 1,
        y >= 0,
        y <= 1
    ]

    # No overlap between circles
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-8)

    # Add constraints for boundaries
    for i in range(n):
        constraints.append(x[i] - radii[i] >= -1e-8)
        constraints.append(x[i] + radii[i] <= 1 + 1e-8)
        constraints.append(y[i] - radii[i] >= -1e-8)

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)

    # Extract results
    best_centers = np.column_stack((x.value, y.value))

    # Return as a packing
