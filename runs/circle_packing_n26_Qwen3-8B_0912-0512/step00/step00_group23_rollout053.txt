import scipy.optimize as optimize

def run_packing():

    num_circles = 26
    num_rows = 5
    num_cols = 5
    radius = 0.1
    centers = []
    radii = []
    sum_radii = 0

    # Define a hexagonal grid
    for i in range(num_rows):
        for j in range(num_cols):
            x = (j + 0.5 * i) * (1.0 / (num_cols + 0.5))
            y = (i + 0.5) * (1.0 / (num_rows + 1))
            centers.append([x, y])
            radii.append(radius)
            sum_radii += radius

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Define the objective function to maximize the total area
    def objective(x):
        return -np.sum(x)  # We minimize the negative of the total area

    def constraint_overlap(x):

        radii_flat = x[2 * n:]

        for i in range(n):

                dx = centers_flat[i, 0] - centers_flat[j, 0]

                    return -1  # Violation, try to decrease radius
        return 0  # No overlap, success

    x0 = np.concatenate([centers.flatten(), radii])

    # Perform optimization
    res = optimize.minimize(

        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraint_overlap}]

    x_result = res.x
