from scipy.optimize import minimize

def run_packing():

    # Optimize the radii to maximize sum while satisfying constraints
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize actual sum

        constraints=[{'type': 'ineq', 'fun': constraints}]
