import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial grid-based packing
    N = 26
    rows = int(np.sqrt(N))
    cols = (N + rows - 1) // rows

    centers = np.zeros((N, 2))
    radii = np.zeros(N)

    # Initial grid-based placement
    for i in range(N):
        row = i // cols
        col = i % cols
        x = col / (cols - 1) if cols > 1 else 0.5
        y = row / (rows - 1) if rows > 1 else 0.5
        centers[i] = [x, y]

    # Set initial uniform radius
    initial_radius = 0.1
    radii[:] = initial_radius

    # Define the objective function: minimize total "energy" to allow refinement
    def objective(x):
        # x is a flattened array of [x0, y0, r0, x1, y1, r1, ...]
        x_flat = x.reshape(-1, 2)
        r_flat = x_flat[:, -1]
        x_flat = x_flat[:, :-1]

        # Reconstruct centers and radii
        centers = x_flat
        radii = r_flat

        # Objective: Minimize the sum of radii (reverse of what we want, but we negate it)
        sum_radii = np.sum(radii)
        return -sum_radii  # Negative for minimization

    # Define constraints:
    # 1. Circle radius >= 0
    # 2. Center within [0, 1]^2
    # 3. Circle does not go outside the square
    # 4. Circles do not overlap (min distance between centers >= sum of radii)

    # Flatten centers and radii
    initial_guess = centers.flatten().tolist() + radii.tolist()
    x0 = np.array(initial_guess)

    constraints = []

    # Constraint: Radius >= 0
    for i in range(N):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[N*2 + i]})

    # Constraint: Center within [0, 1]^2
    for i in range(N):
        for j in range(2):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: x[i*2 + j] - 1e-12})
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: 1 - x[i*2 + j] - 1e-12})

    # Constraint: No overlap between circles
    for i in range(N):
        for j in range(i + 1, N):
            def constraint_overlap(x, i=i, j=j):
                dx = x[2*i] - x[2*j]
                dy = x[2*i + 1] - x[2*j + 1]
                dist = np.sqrt(dx**2 + dy**2)
                ri = x[N*2 + i]
                rj = x[N*2 + j]
                return dist - ri - rj + 1e-12
            constraints.append({'type': 'ineq', 'fun': constraint_overlap})

    # Use L-BFGS-B for optimization
    result = optimize.minimize(
        objective,
        x0,
        method='L-BFGS-B',
        constraints=constraints,
        options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 1000}
    )

    # Extract final centers and radii
    final_centers = result.x[:N*2].reshape(N, 2)
    final_radii = result.x[N*2:]

    # Validate packing
    valid, msg = validate_packing(final_centers, final_radii)
    if not valid:
        raise RuntimeError(f"Validation failed: {msg}")

    return final_centers, final_radii, np.sum(final_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
