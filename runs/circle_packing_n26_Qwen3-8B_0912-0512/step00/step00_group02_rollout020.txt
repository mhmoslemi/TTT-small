import cvxpy as cp

def run_packing():
    # Define initial centers using a hexagonal lattice
    def hexagonal_grid(n):
        rows = int(np.ceil(np.sqrt(n)))
        cols = int(np.ceil(n / rows))
        centers = []
        for i in range(rows):
            for j in range(cols):
                x = j + 0.5 * i
                y = i
                if 0 <= x <= 1 and 0 <= y <= 1:
                    centers.append([x, y])
        return np.array(centers)
    
    # Initial centers from hexagonal grid
    centers_initial = hexagonal_grid(26)
    n = centers_initial.shape[0]
    
    # Define variables
    radii = cp.Variable(n)
    centers = cp.Variable((n, 2))
    
    # Constraints:
    # 1. All centers must be within the unit square
    constraints = [
        cp.sum_squares(centers[i, 0] - centers_initial[i, 0]) <= 1e-6 for i in range(n),
        cp.sum_squares(centers[i, 1] - centers_initial[i, 1]) <= 1e-6 for i in range(n),
    ]
    
    # 2. Each circle must be inside the unit square
    for i in range(n):
        constraints.append(centers[i, 0] - radii[i] >= -1e-6)
        constraints.append(centers[i, 0] + radii[i] <= 1 + 1e-6)
        constraints.append(centers[i, 1] - radii[i] >= -1e-6)
        constraints.append(centers[i, 1] + radii[i] <= 1 + 1e-6)
    
    # 3. No overlap between circles
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-6)
    
    # Objective: Maximize total radius sum
    objective = cp.Maximize(cp.sum(radii))
    
    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver='SCS', eps=1e-6)
    
    # Extract results
    if problem.status != 'optimal':
        raise ValueError(f"Optimization failed with status: {problem.status}")
    
    centers_final = centers.value
    radii_final = radii.value
    sum_radii = np.sum(radii_final)
    
    # Validate that all circles are within the unit square and radii are non-negative

        if r < 0:
            raise ValueError(f"Circle {i} has negative radius: {r}")
    
    return centers_final, radii_final, sum_radii
