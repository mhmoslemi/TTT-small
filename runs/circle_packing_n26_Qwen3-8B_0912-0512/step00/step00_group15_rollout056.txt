import cvxpy as cp

    radii = []
    for i in range(len(centers)):
        x, y = centers[i]
        # Minimum distance to boundaries
        dist_left = x
        dist_right = 1 - x
        dist_bottom = y
        dist_top = 1 - y
        min_boundary_dist = min(dist_left, dist_right, dist_bottom, dist_top)
        
        # Minimum distance to neighbors (for initial rough estimate)
        min_neighbor_dist = float('inf')
        for j in range(len(centers)):
            if i == j:
                continue
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < min_neighbor_dist:
                min_neighbor_dist = dist
        
        # Use the smaller of the two to determine initial radius
        initial_radius = min(min_boundary_dist, min_neighbor_dist / 2)
        radii.append(initial_radius)
    
    radii = np.array(radii)

    # Define variables
    n = len(radii)
    r = cp.Variable(n)
    
    # Constraints
    constraints = []
    for i in range(n):
        x, y = centers[i]
        # Constraint: x - r[i] >= 0
        # Constraint: x + r[i] <= 1

    problem = cp.Problem(objective, constraints)

    if not problem.success:

    # Extract the solution
    centers = np.array(centers)
    radii = r.value
    sum_radii = np.sum(radii)
    
    # Return the result
    return centers, radii, sum_radii
