def run_packing():

    cols = 5

    centers = []

            x = j * spacing
            y = i * spacing
            centers.append([x, y])

    center = np.array([0.5, 0.5])
    centers = np.vstack([centers, center])

        return dist < radii[i] + radii[j] - 1e-12

    def constraint_func(radii):
