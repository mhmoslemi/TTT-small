import scipy.optimize as opt
from dataclasses import dataclass
from typing import Tuple, List

@dataclass
class PackingConfig:
    centers: np.ndarray
    radii: np.ndarray
    total_sum: float

def objective(config: PackingConfig) -> float:
    return -config.total_sum  # We minimize -sum for maximization

def constraints_circle_inside(config: PackingConfig) -> np.ndarray:
    n = config.centers.shape[0]
    constraints = []
    for i in range(n):
        x, y = config.centers[i]
        r = config.radii[i]
        # Check that the circle is fully inside the square
        # x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
        constraints.append([x - r, x + r, y - r, y + r])
    return np.array(constraints).flatten()

def constraints_no_overlap(config: PackingConfig) -> np.ndarray:
    n = config.centers.shape[0]
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):

            x2, y2 = config.centers[j]

            r2 = config.radii[j]

            dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
            # Non-overlap constraint: dist >= r1 + r2
            constraints.append(dist - (r1 + r2))
    return np.array(constraints)
