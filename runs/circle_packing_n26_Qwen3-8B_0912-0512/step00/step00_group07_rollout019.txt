import numpy as np
import cvxpy as cp

def run_packing():
    # Number of circles
    n = 26

    # Define spatial variables
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # Objective: maximize sum of radii
    obj = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Boundary condition: all circles must be within [0,1]x[0,1]
    for i in range(n):
        constraints.append(centers[i, 0] >= 0)
        constraints.append(centers[i, 0] <= 1)
        constraints.append(centers[i, 1] >= 0)
        constraints.append(centers[i, 1] <= 1)

    # Non-overlap constraint: distance between centers must be >= sum of radii
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-8)

    # Solve the optimization problem
    prob = cp.Problem(obj, constraints)
    prob.solve(solver='SCS', eps=1e-5)

    # Extract the solution
    centers_np = centers.value
    radii_np = radii.value

    # Compute total sum of radii
    total_sum = np.sum(radii_np)

    return (centers_np, radii_np, total_sum)
