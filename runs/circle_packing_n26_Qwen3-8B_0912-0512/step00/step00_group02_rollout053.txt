def calculate_total_penalty(centers: np.ndarray, radii: np.ndarray) -> float:
    """Penalty function to discourage boundary and overlap violations."""
    n = centers.shape[0]
    total_penalty = 0.0

    # Boundary violation penalty
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12:
            total_penalty += 1e6
        if y - r < -1e-12 or y + r > 1 + 1e-12:
            total_penalty += 1e6

    # Pairwise overlap penalty
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[i] + radii[j] - 1e-12:
                total_penalty += 1e6

    return total_penalty

    n = centers.shape[0]

        return False, "NaN values present"
