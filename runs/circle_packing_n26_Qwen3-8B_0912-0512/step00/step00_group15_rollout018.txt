import scipy.optimize as optimize

    # Step 3: Define an optimization problem with constraints
    # Objective: maximize the sum of radii
    def objective(x):
        # x is a vector of (centers and radii), flatten it
        x_flat = x.reshape((-1, 1))
        centers = x_flat[:centers.size].reshape(centers.shape)
        radii = x_flat[centers.size:].reshape(radii.shape)
        return -np.sum(radii)  # We minimize the negative to maximize

            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
