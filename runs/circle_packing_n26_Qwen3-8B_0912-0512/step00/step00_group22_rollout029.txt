import numpy as np
import scipy.optimize as optimize

def run_packing():

    # Columns: 6, 5, 6, 5, 6
    # Start with approximate center positions

    rows = [5, 6, 5, 6, 5]
    col_pattern = [6, 5, 6, 5, 6]

    # Create initial centers
    centers = []

        for j in range(col_pattern[i]):
            x = j / (col_pattern[i] + 1)
            y = i / (len(rows) + 1)
            centers.append((x, y))

    # Convert to NumPy array
    centers = np.array(centers)

    # Initial guess for radii (all equal)
    initial_radii = np.full(len(centers), 0.1)

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # We minimize this to maximize sum

    # Constraints:
    # 1. Non-overlap between circles
    # 2. Circles do not go out of bounds

    # Define the constraints for non-overlap
    def constraint_non_overlap(radii):
        n = len(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)

        return 0

    # Define bounds for radii

    # Set up optimization problem
    cons = [{'type': 'ineq', 'fun': constraint_non_overlap}]
    result = optimize.minimize(
