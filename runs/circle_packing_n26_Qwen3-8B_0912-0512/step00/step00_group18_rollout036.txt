import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, LinearConstraint, NonlinearConstraint

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def make_initial_hexagonal_grid(n_circles):
    rows = int(np.ceil(n_circles / 4))
    cols = int(np.ceil(n_circles / rows))
    x = np.linspace(0.5 / (cols - 1), 1 - 0.5 / (cols - 1), cols)
    y = np.linspace(0.5 / (rows - 1), 1 - 0.5 / (rows - 1), rows)
    centers = np.zeros((n_circles, 2))
    idx = 0
    for i in range(rows):
        for j in range(cols):
            if i % 2 == 0:
                centers[idx] = [x[j], y[i]]
            else:
                centers[idx] = [x[j] - 0.25, y[i]]
            idx += 1
    return centers

def optimize_packing(initial_centers):
    n = initial_centers.shape[0]
    x_bounds = [(0.0, 1.0) for _ in range(n)]
    y_bounds = [(0.0, 1.0) for _ in range(n)]
    bounds = Bounds(x_bounds, y_bounds)

    def objective(radii):
        return -np.sum(radii)

    def constraints_center(centers, radii):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if not (x - r >= -1e-12 and x + r <= 1 + 1e-12 and
                    y - r >= -1e-12 and y + r <= 1 + 1e-12):
                return False
        return True

    def constraints_overlap(centers, radii):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    def pack(x):
        centers = x[:2 * n].reshape((n, 2))
        radii = x[2 * n:]
        return constraints_center(centers, radii) and constraints_overlap(centers, radii)

    constraints = NonlinearConstraint(pack, np.zeros(n), np.ones(n))

    result = opt.minimize(objective, np.concatenate((initial_centers.flatten(), np.full(n, 0.1))),
                          bounds=bounds, constraints=constraints, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
    centers = result.x[:2 * n].reshape((n, 2))
    radii = result.x[2 * n:]
    return centers, radii

def run_packing():
    n = 26
    initial_centers = make_initial_hexagonal_grid(n)
    centers, radii = optimize_packing(initial_centers)
    success, message = validate_packing(centers, radii)
    assert success, message
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
