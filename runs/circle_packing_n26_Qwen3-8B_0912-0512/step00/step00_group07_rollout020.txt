import cvxpy as cp

    # We'll use a convex optimization model with variable radii and centers

    # Define the optimization problem
    x = cp.Variable((n, 2))
    r = cp.Variable(n, non_negative=True)

    # Constraints
    constraints = []
    # All circles must be within the unit square
    for i in range(n):
        constraints.append(x[i, 0] - r[i] >= 0)
        constraints.append(x[i, 0] + r[i] <= 1)

    # Objective: maximize the sum of areas (i.e., sum(r^2))
    objective = cp.Maximize(cp.sum(r ** 2))

    problem = cp.Problem(objective, constraints)
