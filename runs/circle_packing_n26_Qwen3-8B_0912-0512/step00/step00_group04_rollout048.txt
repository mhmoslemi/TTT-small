import matplotlib.pyplot as plt

    num_circles = 26

    radii = []

    rows = 5

            y = 0.5 * row * np.sin(angle)
            grid.append([x, y])
            radii.append(0.05)

    # Convert to NumPy arrays
    centers = np.array(grid)
    radii = np.array(radii)

    # Objective: maximize sum of radii (converted to minimization)
    def objective(params):
        # params = [x1, y1, r1, x2, y2, r2, ...]
        params = params.reshape((num_circles, 3))
        centers = params[:, :2]
        radii = params[:, 2]
        return -np.sum(radii)

    # Constraints
    def constraint1(params):
        params = params.reshape((num_circles, 3))
        centers = params[:, :2]
        radii = params[:, 2]
        # Ensure all circles are inside the unit square
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return 1e6 * (0.5 - (x - r))
        return 0

    def constraint2(params):
        params = params.reshape((num_circles, 3))
        centers = params[:, :2]

        for i in range(num_circles):

                dist_sq = dx*dx + dy*dy
                r_sum = radii[i] + radii[j]
                if dist_sq < (r_sum)**2 - 1e-8:
                    return 1e6 * (0.5 - (dist_sq))
        return 0

    # Initial guess

    result = minimize(
