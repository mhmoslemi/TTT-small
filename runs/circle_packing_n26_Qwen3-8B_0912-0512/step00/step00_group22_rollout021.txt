from scipy.optimize import minimize

    # We place the primary circles in a staggered grid

    # List to store circle positions
    centers = []

    # Radius guess for the main circles
    # Will be adjusted in the optimization step
    radii = []

    # Strip width = sqrt(3) / 2 * r (horizontal spacing)

    # Create initial position for the top-left circle
    # Based on the number of rows and columns in the hexagonal pattern
    num_rows = 6
    num_cols = 4

    # Start at (0.1, 0.1) and build the grid
    for i in range(num_rows):

            if i % 2 == 1:

            centers.append([x, y])
            radii.append(0.05)  # Initial guess for radius

    # Add an extra circle in the center for optimizing gaps
    centers.append([0.5, 0.5])
    radii.append(0.0)

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Objective function: maximize the sum of radii
    def objective(radii_params):
        return -np.sum(radii_params)  # Negative for minimization

    # Constraints: distance between centers ≥ sum of radii (non-overlap)
    def non_overlap_constraint(radii_params):
        n = len(centers)
        dists = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                dists[i, j] = dist
                dists[j, i] = dist
        constraints = dists - (radii_params[i] + radii_params[j] for i, j in zip(range(n), range(n)))
        return constraints

    # Boundary constraint: each circle must be fully inside the unit square
    def boundary_constraint(radii_params):
        dists = np.zeros(n)
        for i in range(n):

        return dists

    # Define bounds for each radius (must be non-negative)
    bounds = [(0.0, 1.0) for _ in range(len(radii))]

    # Optimization
    result = minimize(

        x0=radii,
        bounds=bounds,
        constraints=[
            NonlinearConstraint(non_overlap_constraint, -np.inf, 0),

        method='SLSQP'

    # Extract the final centers and radii
    final_centers = centers
