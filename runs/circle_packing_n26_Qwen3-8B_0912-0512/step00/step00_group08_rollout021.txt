import numpy as np
import scipy.optimize as optimize
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial layout with a 5x5 grid centered in [0, 1]x[0, 1]
    # First 25 circles arranged in a grid
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # 5x5 grid with spacing 0.25
    for i in range(5):
        for j in range(5):
            centers[5*i + j] = [i * 0.25 + 0.125, j * 0.25 + 0.125]
    
    # Add a 26th circle near the top-right corner
    centers[25] = [0.875, 0.875]

    # To optimize, we will minimize the negative of sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraints:
    # 1. Each circle is within the unit square
    # 2. Each circle's radius is non-negative
    # 3. Distance between any two circles ≥ sum of their radii
    # We will minimize the contraints

    # We will use scipy.optimize with bounds and constraints
    # Define bounds: radii must be non-negative
    bounds = [(0, 1) for _ in range(num_circles)]

    def constraint_1(radii):
        # Circle is within unit square
        # Check if the distance to the edge is ≥ radius
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
                return False
        return True

    def constraint_2(radii):
        # Circles do not overlap
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]

                    return False
        return True

    # Create constraints in the form of a list of dictionaries
    cons = [
        {'type': 'ineq', 'fun': constraint_1},

    ]

    # Run optimization
    result = optimize.minimize(
        fun=objective,
        x0=radii,
        method='SLSQP',
        bounds=bounds,
        constraints=cons
    )
