import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, NonlinearConstraint

def run_packing():
    # Initial positions: hexagonal grid
    n = 26
    pos = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.25
        y = row + 0.25
        pos.append([x, y])
    centers = np.array(pos)

    # Initial guess for radii: based on minimum spacing
    radii = np.full(n, 0.05)

    def objective(radii):
        return -np.sum(radii)

    def constraint_func(radii):
        # Non-overlapping constraint
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return 1.0
        return 0.0

    non_overlapping = NonlinearConstraint(
        constraint_func, 
        0.0, 0.0, 
        jac="2-point"
    )

    # Boundary constraint: circles inside [0,1]^2
    def boundary_func(radii):
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8:
                return 1.0
            if y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e8:
                return 1.0
        return 0.0

    boundary = NonlinearConstraint(
        boundary_func,
        0.0, 0.0,
        jac="2-point"
    )

    bounds = Bounds(0, 1.0)

    result = opt.minimize(
        objective, 
        radii, 
        bounds=bounds, 
        constraints=[non_overlapping, boundary], 
        method="SLSQP"
    )

    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

    return centers, result.x, np.sum(result.x)
