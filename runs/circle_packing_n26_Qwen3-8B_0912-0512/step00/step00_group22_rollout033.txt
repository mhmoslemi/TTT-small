import scipy.optimize as optimize

def validate_packing(centers, radii):

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

        r = radii[i]

        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    initial_x = np.linspace(0, 1, 5)
    initial_y = np.linspace(0, 1, 5)
    center_grid = np.array([[x, y] for y in initial_y for x in initial_x])
    center_grid = np.vstack([center_grid, [0.5, 1.0]])
    radii_initial = np.full(26, 0.04)  # Initial guess

    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii

    def constraint_boundary(radii):

            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:

        return 0

    def constraint_overlap(radii):

                dx = center_grid[i, 0] - center_grid[j, 0]
                dy = center_grid[i, 1] - center_grid[j, 1]

    cons = [
