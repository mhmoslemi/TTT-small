        radii_flat = flat_radii

        #Objective is -sum(radii)

        return -total_radius

    def constraint_boundary(x):
        flat_centers = x[::3]
        flat_radii = x[1::3]

        for i in range(len(centers_flat)):

            r_i = radii_flat[i]
            if x_i - r_i < -1e-8 or x_i + r_i > 1 + 1e-8:

            if y_i - r_i < -1e-8 or y_i + r_i > 1 + 1e-8:
