import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Initial configuration: 20 interior circles in hexagonal packing, 6 on the edges/corners
    # Initial centers for hexagonal packing in a unit square (approximate)
    def hexagonal_packing(num_circles):
        row = int(np.floor(np.sqrt(num_circles)))
        col = num_circles - row * (row + 1) // 2
        centers = []
        for i in range(row + 1):
            for j in range(col):
                x = j + 0.5 * i
                y = i + 0.5 * j
                centers.append([x, y])
        # Make sure it fits into the unit square
        if centers:
            max_x = max(c[0] for c in centers)
            max_y = max(c[1] for c in centers)
            scaled_centers = [[x / max_x, y / max_y] for x, y in centers]
            return np.array(scaled_centers)
        return np.array([])

    # Generate initial centers
    num_interior = 20
    interior_centers = hexagonal_packing(num_interior)

    # Add 6 boundary circles
    boundary_centers = np.array([
        [0.1, 0.5],  # Left
        [0.5, 0.1],  # Bottom
        [0.9, 0.5],  # Right
        [0.5, 0.9],  # Top
        [0.1, 0.1],  # Bottom-left corner
        [0.9, 0.9],  # Top-right corner
    ])

    centers = np.vstack((interior_centers, boundary_centers))

    # Initial guess for radii (equal radii for interior, small for boundaries)
    num_interior = len(interior_centers)
    num_boundary = len(boundary_centers)
    initial_radii = np.concatenate(
        (np.full(num_interior, 0.1), np.full(num_boundary, 0.05))
    )

    def objective(radii):
        return -np.sum(radii)  # Maximize sum by minimizing negative sum

    def constraints(radii):
        # Ensure non-overlapping (distance between centers >= sum of radii)
        dist_matrix = squareform(pdist(centers))
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    def constraint_func(radii):
        return constraints(radii)

    # Use SLSQP for constrained optimization
    bounds = [(0, 1) for _ in range(len(initial_radii))]  # Radii must be non-negative

    result = minimize(
        objective,
        initial_radii,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraint_func}],
        options={'ftol': 1e-10, 'maxiter': 1000}
    )

    # Extract optimized results
    radii = result.x
    centers = centers.copy()

    # Validate the packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Validation failed: {msg}")

    # Calculate sum of radii
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
