from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from numpy.random import choice

def run_packing():

    n_circles = 26
    centers = np.zeros((n_circles, 2))
    for i in range(n_circles):
        row = i // 5
        col = i % 5
        centers[i, 0] = col + 0.5
        centers[i, 1] = row + 0.5

    # Initial guess for radii
    radii = np.ones(n_circles) * 0.1

    # Optimization objective: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize the negative sum

    def non_overlapping(radii):

    def compute_constraints(radii):

    # Optimization
    result = minimize(
        objective,

        constraints=compute_constraints,
        bounds=[(0, 1)] * n_circles,
        method="SLSQP"
