import numpy as np
import cvxpy as cp

def run_packing():
    # Define the number of circles
    n = 26

    # Define initial center positions using 5x6 grid with an extra circle
    centers = []

    # Generate initial positions for circles in a 5x6 grid (5 rows x 6 columns)
    # with a small spacing to avoid overfitting
    for i in range(5):
        for j in range(6):
            x = j / 6 + 1e-3
            y = i / 5 + 1e-3
            centers.append([x, y])

    # Add one more circle to make 26, placed near the bottom right corner
    centers.append([0.95, 0.95])

    centers = np.array(centers)

    # Define optimization variables
    radii = cp.Variable(n)

    # Objective: Maximize the total sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Each circle must be inside the unit square
    for i in range(n):
        x, y = centers[i]
        constraints.append(x - radii[i] >= -1e-12)
        constraints.append(x + radii[i] <= 1 + 1e-12)
        constraints.append(y - radii[i] >= -1e-12)
        constraints.append(y + radii[i] <= 1 + 1e-12)

    # No two circles may overlap
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    # Define and solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, eps=1e-6)

    # Extract results
    centers_result = centers
    radii_result = radii.value

    # Calculate total sum of radii
    sum_radii = np.sum(radii_result)

    return centers_result, radii_result, sum_radii
