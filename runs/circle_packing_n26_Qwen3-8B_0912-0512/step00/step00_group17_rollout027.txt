import numpy as np
import scipy.optimize as opt

def generate_hex_grid(n, padding=0.01):
    """Generate a hexagonal grid of n circles within [0,1]^2."""
    max_rows = int(np.sqrt(n))
    max_cols = int(np.ceil(n / max_rows))
    grid = []
    for i in range(max_rows):
        for j in range(max_cols):
            x = j + i * padding
            y = i * (1 + padding)
            if x > 1 - padding or y > 1 - padding:
                continue
            grid.append((x, y))
    return np.array(grid)

def validate_packing(centers, radii):
    n = centers.shape[0]
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"

def run_packing():
    # Start with a hexagonal grid configuration
    centers = generate_hex_grid(26)
    if len(centers) < 26:
        raise ValueError("Not enough circles in grid")

    radii = np.full(len(centers), 0.1)

    bounds = [(0, 1 - 0.1) for _ in range(len(centers))]

    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    # Define constraints: non-overlapping and within unit square
    def constraint(radii):
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or

                    return -np.inf  # Invalid configuration
        return np.sum(radii)

    # Set up constraints with bounds
    cons = [{'type': 'ineq', 'fun': constraint}]

    # Run optimization

        bounds=bounds,
        constraints=cons,
