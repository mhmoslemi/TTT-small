import numpy as np
import cvxpy as cp

def run_packing():
    n = 26
    # Define variables
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n, nonneg=True)

    # Objective: Maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Each circle must be inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r >= -1e-8)
        constraints.append(x + r <= 1 + 1e-8)
        constraints.append(y - r >= -1e-8)
        constraints.append(y + r <= 1 + 1e-8)

    # Circles must not overlap
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-8)

    # Solve the problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver='SCS', eps=1e-6, max_iters=5000, verbose=False)

    # Extract results
    centers_np = centers.value
    radii_np = radii.value
    sum_radii = np.sum(radii_np)

    # Return as per the required format
    return (centers_np, radii_np, sum_radii)
