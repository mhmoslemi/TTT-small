import numpy as np
import scipy.optimize as optimize

def run_packing():

    num_circles = 26
    initial_positions = []
    for i in range(5):
        for j in range(5):
            x = j / 4
            y = i / 4
            initial_positions.append((x, y))

    initial_positions.append((0.75, 0.75))
    
    centers = np.array(initial_positions)
    radii = np.full(num_circles, 0.1)

    def objective(radii):
        return -np.sum(radii)  # We want to maximize sum of radii
    
    def constraints(radii):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        for i in range(num_circles):

            dy = centers[i, 1]

            if dx - r < -1e-8 or dx + r > 1 + 1e-8:
                return False
            if dy - r < -1e-8 or dy + r > 1 + 1e8:
                return False
        return True

    def constraint_func(radii):
        return constraints(radii)

    result = optimize.minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=[(0, 1.0) for _ in range(num_circles)],
