def generate_hexagonal_grid(n: int, square_size: float = 1.0, grid_spacing: float = 0.2) -> np.ndarray:
    """
    Generate a hexagonal grid of points in a square.

    Parameters:
        n: number of points (circles)
        square_size: size of the square (default 1.0)
        grid_spacing: spacing between grid points (default 0.2)

    Returns:
        np.ndarray: (n, 2) array of x, y coordinates
    """
    x = np.linspace(0, square_size, int(np.sqrt(n)) + 1)
    y = np.linspace(0, square_size, int(np.sqrt(n)) + 1)
    grid = np.array([[xi, yi] for xi in x for yi in y])
    return grid[:n]
