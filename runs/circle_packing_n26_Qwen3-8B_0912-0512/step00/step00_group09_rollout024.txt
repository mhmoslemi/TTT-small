import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, cdist
from sklearn.metrics.pairwise import EuclideanDistances

def run_packing():
    """Packs 26 circles in a unit square to maximize sum of radii."""
    # Parameters for the optimization
    n = 26
    max_iterations = 100
    tolerance = 1e-6
    initial_centers = np.random.rand(n, 2)

    def objective(radii):
        """Objective function to maximize the sum of radii."""
        return -np.sum(radii)  # Negative for minimization

    def constraints(radii):
        """Constraints for the optimization problem."""
        constraints = []
        for i in range(n):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i]})
        for i in range(n):
            for j in range(i + 1, n):
                constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: np.sqrt(np.sum((initial_centers[i] - initial_centers[j]) ** 2)) - r[i] - r[j]})
        return constraints

    # Initial guess for radii
    initial_radii = np.full(n, 0.01)

    # Optimization
    result = minimize(objective, initial_radii, method='SLSQP', constraints=constraints, options={'maxiter': max_iterations, 'ftol': tolerance})

    if not result.success:
        raise ValueError("Optimization failed")

    # Extract results
    centers = initial_centers
    radii = result.x
    sum_radii = np.sum(radii)

    # Validate packing
    valid, message = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Validation failed: {message}")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    """Validates that the packing is within a unit square and circles don't overlap."""
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radi radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
