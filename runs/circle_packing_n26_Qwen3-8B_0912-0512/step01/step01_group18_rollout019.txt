import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    # Preliminary hexagonal packing arrangement with some spacing on the edges
    # These are initial guess positions for the centers
    # This arrangement is more refined than before and pushes circles to corners and edges
    
    # Initial positions for circles
    centers = np.array([
        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
        [0.2, 0.2], [0.8, 0.2],

        [0.2, 0.4], [0.8, 0.4],

        [0.2, 0.6], [0.8, 0.6],
        [0.1, 0.7], [0.3, 0.7], [0.5, 0.7], [0.7, 0.7], [0.9, 0.7],
    ])

    # Initial radii: To allow for optimization, all start with a small value
    radii = np.full(26, 0.05)

    # Objective function: maximize sum of radii
    def objective(x):
        # x is a flat vector of variables: first 52 elements are positions (x, y), next 26 are radii
        positions = x[:52].reshape(26, 2)
        radii = x[52:]
        
        # Apply boundary constraints
        for i in range(26):
            x_i, y_i = positions[i]
            r_i = radii[i]
            if (x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12
                or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12):
                return -np.inf  # Penalize for boundary violation

        # Apply circle-circle constraints
        for i in range(26):
            for j in range(i + 1, 26):
                dx = positions[i, 0] - positions[j, 0]
                dy = positions[i, 1] - positions[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -np.inf  # Penalize for overlap

        # Return negative of sum of radii to minimize (as we are maximizing)
        return -np.sum(radii)

    # Bounds for positions and radii

    x0 = np.concatenate((x0_positions, x0_radii))

    # Perform optimization
    result = minimize(

        x0,
        bounds=bounds,
        method='L-BFGS-B',

        options={'maxiter': 1000}

    # Extract optimized values
    positions = result.x[:52].reshape(26, 2)
    radii = result.x[52:]

    centers = positions
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
