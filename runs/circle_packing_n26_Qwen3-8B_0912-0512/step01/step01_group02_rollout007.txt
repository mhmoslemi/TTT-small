import scipy.optimize as opt

def run_packing():

    num_circles = 26

        x = i % 5 / 5.0

        centers.append([x, y])
    centers = np.array(centers)
    radii = np.array([0.1] * num_circles)

    # Define the objective function to maximize the sum of radii
    def objective(radii):
        sum_radii = np.sum(radii)
        return -sum_radii  # Minimize the negative sum to maximize sum_radii

    # 1. Circles must be inside the unit square
    # 2. Circles must not overlap
    # 3. Radii must be non-negative

        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if (x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1):
                return -1
        return 0

        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

        for r in radii:
            if r < 0:
                return -1
        return 0

    bounds = [(0, 1) for _ in range(num_circles)]

    result = opt.minimize(

        bounds=bounds,
