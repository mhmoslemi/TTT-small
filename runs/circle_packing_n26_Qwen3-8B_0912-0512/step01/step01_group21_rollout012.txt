import numpy as np
import math
import random
from scipy.optimize import minimize

def compute_distance(p1, p2):
    return math.hypot(p1[0] - p2[0], p1[1] - p2[1])

def compute_boundary_distance(p):
    x, y = p
    return min(x, 1 - x, y, 1 - y)

def is_valid(centers, radii):
    n = len(centers)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            return False
    for i in range(n):
        for j in range(i+1, n):
            dist = compute_distance(centers[i], centers[j])
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def run_packing():
