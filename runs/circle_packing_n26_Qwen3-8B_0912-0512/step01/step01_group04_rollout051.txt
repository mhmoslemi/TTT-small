    def generate_hexagonal_grid(n):

        if n == 0:
            return np.array([]), np.array([])

            return np.array([[0.5, 0.5]]), np.array([0.5])

        rows = int(np.sqrt(n))

                y = i + (j % 2) * 0.5

                x = x / (max_col - 1) if max_col > 0 else 0.5
                y = y / (rows - 1) if rows > 0 else 0.5
                hex_centers.append([x, y])
        # Trim the list to exactly n circles
        return np.array(hex_centers[:n]), np.array([0.1] * n)

    initial_centers, _ = generate_hexagonal_grid(num_circles)
