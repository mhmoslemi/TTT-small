import numpy as np
from scipy.optimize import Bounds, minimize

def run_packing():
    def generate_initial_guesses(n_circles=26, num_guesses=5):
        guesses = []
        for _ in range(num_guesses):
            # Create random positions within the square
            centers = np.random.rand(n_circles, 2)
            # Generate random radii between 0.01 and 0.1
            radii = np.random.uniform(0.01, 0.1, n_circles)
            guesses.append((centers, radii))
        return guesses

    def objective(x, n_circles=26):
        # x is a flat vector of variables: first 2*n_circles elements are positions (x, y), next n_circles are radii
        positions = x[:2*n_circles].reshape(n_circles, 2)
        radii = x[2*n_circles:]
        
        # Apply boundary constraints
        for i in range(n_circles):
            x_i, y_i = positions[i]
            r_i = radii[i]
            if (x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12
                or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12):
                return -np.inf  # Penalize for boundary violation

        # Apply circle-circle constraints
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = positions[i, 0] - positions[j, 0]
                dy = positions[i, 1] - positions[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -np.inf  # Penalize for overlap

        # Return negative of sum of radii to minimize (as we are maximizing)
        return -np.sum(radii)

    # Generate multiple initial guesses
    initial_guesses = generate_initial_guesses()

    # For each guess, run optimization
    best_sum = 0
    best_centers = np.zeros((26, 2))
    best_radii = np.zeros(26)

    for centers, radii in initial_guesses:
        # Flatten the initial guess
        x0 = np.concatenate((centers.flatten(), radii))
        
        # Define bounds for positions and radii
        bounds_positions = [(0.0, 1.0)] * (2 * 26)
        bounds_radii = [(0.0, 1.0)] * 26
        bounds = bounds_positions + bounds_radii

        # Perform optimization
        result = minimize(objective, x0, bounds=bounds, method='L-BFGS-B', tol=1e-8)

        # Extract optimized values
        if result.success and result.fun < -best_sum:

            radii = result.x[52:]
            best_sum = -result.fun
            best_centers = positions
            best_radii = radii

    return (best_centers, best_radii, best_sum)
