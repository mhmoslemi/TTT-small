import scipy.optimize as optimize

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    n_rows = 5  # rows of circles
    n_cols = 5  # columns per row
    # pad to make it fit 26 circles
    n_cols = 5
    n_rows = 6

    # 2D grid of circle centers in a hexagonal close packing
    centers = np.zeros((n_rows * n_cols, 2))  # 2D positions

    for row in range(n_rows):
        for col in range(n_cols):
            if row % 2 == 0:
                # even rows: regular horizontal spacing
                x = col / (n_cols - 1)
                y = row / (n_rows - 1)
            else:
                # odd rows: shifted horizontally by half a column
                x = (col + 0.5) / (n_cols - 1)
                y = row / (n_rows - 1)
            centers[row * n_cols + col] = [x, y]

    # Trim to 26 circles (remove extra)
    centers = centers[:26]

    # Initial guess for radii
    radii_initial = np.full(26, 0.05)

    # Define the objective function for optimization
    def objective(x: np.ndarray) -> float:

        n = len(radii_initial)

        radii = x[2 * n:]

        # Check if any constraint is violated
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if (x_i - r_i < -1e-8 or x_i + r_i > 1 + 1e-8

                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:

        # Objective: maximize the sum of radii
        return -np.sum(radii)

    # Run optimization
    result = optimize.minimize(

        x0=x0,
        method='L-BFGS-B',

    centers = result.x[: 2 * 26].reshape(26, 2)
