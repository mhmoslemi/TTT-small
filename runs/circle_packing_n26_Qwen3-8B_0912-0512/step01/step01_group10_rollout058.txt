import numpy as np
import scipy.optimize as optimize
from deap import base, creator, algorithms, tools

def run_packing():
    # Number of circles
    num_circles = 26
    population_size = 100
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.9
    
    # Define the GA parameter space
    creator.create("FitnessMax", base.Fitness, weights=(1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMax)
    
    # Genetic algorithm tools
   toolbox = base.Toolbox()
    toolbox.register("random_x", np.random.uniform, 0, 1)
    toolbox.register("random_r", np.random.uniform, 0, 0.1)
    
    # Define the individual
    toolbox.register("individual", tools.initRepeat, creator.Individual, 
                    lambda: [toolbox.random_x(), toolbox.random_x(), toolbox.random_r()], n=num_circles)
    
    # Define the population
    toolbox.register("population", tools.initIterate, list, 
                    lambda: [toolbox.individual() for _ in range(num_circles)])
    
    # Evaluate an individual
    def eval_func(individual):
        centers = np.array([individual[i*3:i*3+2] for i in range(num_circles)])
        radii = np.array([individual[i*3+2] for i in range(num_circles)])
        
        # Check if circles are within the square
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if not (0 - r <= x <= 1 + r and 0 - r <= y <= 1 + r):
                return 0.0,
        
        # Check non-overlapping condition
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return 0.0,
        
        # Return sum of radii as fitness
        return np.sum(radii),
    
    toolbox.register("evaluate", eval_func)
    
    # Genetic operators
    toolbox.register("select", tools.selTournament, tournsize=3)
    toolbox.register("mate", tools.cxTwoPoint)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.05, mutatename="gaussian", indpb=0.1)
    
    # Initialize population
    pop = toolbox.population()
    
    # Run genetic algorithm
    stats = tools.Statistics(lambda ind: ind.fitness.values)
    stats.register("avg", np.mean)
    stats.register("std", np.std)
    stats.register("min", np.min)
    stats.register("max", np.max)
    
    pop, logbook = algorithms.eaSimple(pop, toolbox, cxpb=crossover_rate, mutpb=mutation_rate, 
                                      ngen=generations, stats=stats, verbose=False)
    
    # Find best individual
    best_ind = max(pop, key=lambda x: x.fitness.values[0])
    
    # Extract best solution
    centers = np.array([best_ind[i*3:i*3+2] for i in range(num_circles)])
    radii = np.array([best_ind[i*3+2] for i in range(num_circles)])
    
    # Ensure all circles are non-negative and inside square
    for i in range(num_circles):
        x, y = centers[i]
        r = radii[i]
        if not (0 - r <= x <= 1 + r and 0 - r <= y <= 1 + r):
            raise ValueError("Some circles are outside the unit square")
        if r < 0:
            raise ValueError("Negative radius detected")
    
    return centers, radii, np.sum(radii)
