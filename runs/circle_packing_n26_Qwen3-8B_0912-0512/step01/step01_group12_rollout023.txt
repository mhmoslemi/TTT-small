import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define the positions of 26 circles in a staggered (hexagonal) packing
    # Each row will have increasing and then decreasing circles (to sum to 26)
    # The first 6 rows increase in count, the 7th row has 5 circles
    
    layer_counts = [1, 2, 3, 4, 5, 6, 5]
    row_height = 0.85  # vertical distance between rows (smaller than the width to allow for staggered packing)
    col_width = 0.85   # horizontal distance between adjacent circles in a row
    
    centers = []
    for i in range(len(layer_counts)):
        row = []
        for j in range(layer_counts[i]):
            x = j * col_width
            y = i * row_height
            row.append([x, y])
        # Apply offset for staggered rows (even rows shifted right)
        if i % 2 == 1:
            for idx in range(len(row)):
                row[idx][0] += (col_width / 2)
        centers.extend(row)
    
    # Convert centers to NumPy array
    centers = np.array(centers)
    
    # Now perform optimization to ensure non-overlapping and boundary constraints
    # Objective: Maximize sum of radii (thus, minimize negative sum)
    def objective(x: np.ndarray) -> float:
        # x is of shape (26*2,) -> reshape to (26, 2)
        centers = x.reshape((26, 2))
        # Initial estimate for radii
        radii = np.full(26, 0.05)

        # Check boundaries
        for i in range(26):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if (x_i - r_i < 0 or x_i + r_i > 1) or (y_i - r_i < 0 or y_i + r_i > 1):
                return float('inf')

        # Check circle-circle overlaps
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                if dist < radii[i] + radii[j] - 1e-6:
                    return float('inf')

        # Try to increase radii based on distance to nearest neighbors and boundary
        # Computed radii is minimum of distance to boundary and distance to neighbors / 2
        radii = np.zeros(26)
        for i in range(26):
            x_i, y_i = centers[i]
            # distance to left right top bottom
            r = min(
                x_i,
                1 - x_i,
                y_i,
                1 - y_i
            )
            radii[i] = r

        # Define radius as minimum of: (distance to nearest neighbor) / 2 and boundary distance
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                radii[i] = min(radii[i], dist / 2)
                radii[j] = min(radii[j], dist / 2)

        return -np.sum(radii)

    # Set bounds for centers: [0, 1]
    bounds = [(0, 1) for _ in range(26 * 2)]

    # Initial guess: use the initial guess we built above
    x0 = centers.reshape(-1)

    # Run optimization
    result = optimize.minimize(
        objective,
        x0,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 100, 'ftol': 1e-9}
    )

    # Extract the refined centers and radii
    centers = result.x.reshape((26, 2))
    radii = np.zeros(26)
    for i in range(26):
        x_i, y_i = centers[i]
        r = min(
            x_i,
            1 - x_i,
            y_i,
            1 - y_i
        )
        radii[i] = r

    # Final check for overlaps
    for i in range(26):
        for j in range(i + 1, 26):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            if dist < radii[i] + radii[j] - 1e-6:
                # If overlap, set smaller radius
                radii[i] = dist / 2
                radii[j] = dist / 2

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
