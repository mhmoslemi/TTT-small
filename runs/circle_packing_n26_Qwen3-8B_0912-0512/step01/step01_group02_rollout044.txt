import numpy as np
import random
import math
from scipy.optimize import Bounds

def run_packing():
    # --- Helper Functions ---
    def is_valid(centers, radii):
        n = len(centers)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if not (0 <= x - r and x + r <= 1 and 0 <= y - r and y + r <= 1):
                return False, f"Circle {i} is out of bounds"
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = math.hypot(dx, dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False, f"Circles {i} and {j} overlap"
        return True, "Valid packing"

    def fitness(centers, radii):
        return np.sum(radii)

    # Initialize the population
    def initialize_population(size, num_circles):
        pop = []
        for _ in range(size):
            points = []
            radii = []
            for _ in range(num_circles):
                x = random.uniform(0, 1)
                y = random.uniform(0, 1)
                r = random.uniform(0.01, 0.2)
                points.append([x, y])
                radii.append(r)
            pop.append((points, radii))
        return pop

    # Genetic Algorithm Parameters
    POP_SIZE = 20
    MUTATION_RATE = 0.1
    CROSSOVER_RATE = 0.7
    MAX_GEN = 500

    # Genetic Algorithm Functions
    def fitness_func(individ):
        centers, radii = individ
        if not is_valid(centers, radii)[0]:
            return -1000000  # Penalize invalid individuals
        return fitness(centers, radii)

    def crossover(parent1, parent2):
        child_centers, child_radii = [], []
        for i in range(len(parent1[0])):
            r1 = parent1[1][i]
            r2 = parent2[1][i]
            if random.random() < CROSSOVER_RATE:
                r = (r1 + r2) / 2.0
                child_radii.append(r)
                x = (parent1[0][i][0] + parent2[0][i][0]) / 2.0
                y = (parent1[0][i][1] + parent2[0][i][1]) / 2.0
                child_centers.append([x, y])
            else:
                child_radii.append(parent1[1][i])
                child_centers.append(parent1[0][i])
        return (child_centers, child_radii)

    def mutate(individ):
        centers, radii = individ
        for i in range(len(centers)):

                dy = random.uniform(-0.1, 0.1)

                centers[i][1] += dy

                centers[i][1] = np.clip(centers[i][1], 0, 1)

                # Randomly perturb radius
                dr = random.uniform(-0.05, 0.05)
                new_radius = radii[i] + dr
                if new_radius < 0:
                    new_radius = 0
                elif new_radius > 1:
                    new_radius = 1
                radii[i] = new_radius
        return (centers, radii)

    # Run the genetic algorithm
    population = initialize_population(POP_SIZE, 26)

    for gen in range(MAX_GEN):
        # Evaluate fitness
        population = [(
            p[0],
            p[1]
        ) for p in population]

        # Evaluate fitness for all
        fitness_values = [fitness_func(individ) for individ in population]
        population = sorted(zip(population, fitness_values), key=lambda x: x[1], reverse=True)
        best = population[0]
        print(f"Generation {gen}: Best fitness = {best[1]}")

        # Create next generation
        next_gen = []
        for i in range(POP_SIZE):

                parent2 = population[i // 2 + 1][0]
                child = crossover(parent1, parent2)
                next_gen.append(child)

        population = next_gen

    centers, radii = best_solution
    total_radius = np.sum(radii)
    return (np.array(centers), np.array(radii), total_radius)
