import numpy as np
import cvxpy as cp
import time


def run(budget_s=1100):
    # Load initial h from pre-defined global
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points

    def simulate_overlap(h):
        # Flip to get 1 - h efficiently on domain [0,2]
        one_minus_h = 1 - h
        S = np.correlate(h, one_minus_h, mode="full") * dx
        return S

    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        S = simulate_overlap(h)
        k = np.argmax(S)
        C5 = S[k]

        # Don't proceed if already at target (or below)
        if C5 <= 0.380876:
            break

        # Identify i's contributing to S_k
        forward_indices = np.arange(n_points)
        backward_indices = (np.arange(n_points) - k) % n_points

        # Compute gradient-like direction
        grad = np.zeros(n_points)
        grad += (1 - h[backward_indices])  # from h[i]*1
        grad -= h[forward_indices]          # from -h[i]*h[i+k]
        grad *= dx

        # LP formulation to minimize max overlap at this k
        delta = cp.Variable(n_points)
        obj = cp.Minimize(cp.max(S))
        constraints = [
            h + delta >= 0,
            h + delta <= 1,
            cp.sum(delta) == 0
        ]
        prob = cp.Problem(obj, constraints)

        try:
            prob.solve(solver=cp.ECOS, max_iters=50, verbose=False)
            impr = prob.value
        except:
            prob.solve(solver=cp.OSQP, max_iter=5000, verbose=False)
            impr = prob.value

        if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            break

        # Step once
        h = h + 0.3 * delta.value

        # Enforce constraints with correction
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale_factor = (n_points / 2) / sum_h
        h = h * scale_factor

        # Update best
        C5_current = evaluate_erdos_solution(h)
        if C5_current < best_C5:
            best_C5 = C5_current
            best_h = h.copy()
        elif C5_current > best_C5 + 1e-6:
            # If getting worse, reset to best
            h = best_h.copy()

    # Final post-processing
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h / best_h.sum() * (n_points / 2)  # Normalize again
    best_C5 = evaluate_erdos_solution(best_h)

    return (best_h.tolist(), best_C5, n_points)
