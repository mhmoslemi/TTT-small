import numpy as np
import time
import scipy.signal
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 160
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        if h.shape[0] != n_points:
            h = scipy.interpolate.interp1d(np.linspace(0, 2, h.shape[0]), h, kind='linear')(np.linspace(0, 2, n_points))
    else:
        plateau_width = 80
        h = np.zeros(n_points)
        h[n_points // 2 - plateau_width // 2 : n_points // 2 + plateau_width // 2] = 1.0
        h = h / h.sum() * n_points / 2
    h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = calculate_c5(h, dx)
    start_phase1 = time.time()
    while time.time() - start_phase1 < 0.8 * (budget_s - 0):
        h_flat = h.flatten()

        # Compute gradients using finite differences
        gradients = np.zeros_like(h)
        for k in range(-n_points + 1, n_points):
            shift = (h_flat + np.roll(h_flat, -k)) % n_points
            values = h_flat * (1.0 - np.roll(h_flat, -k))
            corr = scipy.signal.fftconvolve(h_flat, 1 - h_flat, mode='full')
            corr = corr[n_points - 1: 2 * n_points - 1]
            max_c5 = max(corr)
            grad = np.roll(1 - 2 * h_flat, -k)
            gradients += max(0.0, max_c5 - 0.9 * best_c5) * grad

        grad_norm = np.linalg.norm(gradients)
        alpha = 0.1 / max(grad_norm, 1.0)

        # Line search and projection
        h -= alpha * gradients
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2) / np.sum(h)

        current_c5 = calculate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        start_phase1 = time.time()

        # Check budget
        if time.time() - start_time > budget_s - 15:
            break

        # Add some noise for escape from local optima
        if np.random.rand() < 0.05:
            h *= (1.0 + 0.02 * np.random.randn(n_points))
            h = np.clip(h, 0.0, 1.0)
            h = h / np.sum(h) * n_points / 2

    # Phase 2: QCQP refinement
    try:
        h_var = cp.Variable(n_points)
        obj = 0
        for k in range(-n_points + 1, 0):
            obj += cp.norm(h_var - np.roll(h, -k), 2)
        h_refined = cp.Problem(cp.Minimize(obj),
                               [0 <= h_var, h_var <= 1,
                                cp.sum(h_var) == n_points / 2])
        h_refined.solve(solver=cp.OSQP, max_iter=10000)
        h = h_var.value
        if h is not None:
            current_c5 = calculate_c5(h, dx)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5
    except:
        pass

    # Final clean up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (n_points / 2.0) / np.sum(best_h)

    best_c5 = calculate_c5(best_h, dx)
    if 'initial_h_values' in globals():
        best_c5 = min(best_c5, calculate_c5(np.array(initial_h_values, dtype=np.float64), dx))

    # Final check and validate
    if best_c5 < 0.380876:
        final_c5 = best_c5
    else:
        final_c5 = calculate_c5(np.array(initial_h_values, dtype=np.float64), dx)

    return (best_h, final_c5, n_points)

def calculate_c5(h, dx):
    num_h = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr[num_h - 1: 2 * num_h - 1]) * dx
    return c5
