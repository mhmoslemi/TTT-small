import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        # If no initial construction is provided, use a symmetric binary start
        n_points = 100  # Default resolution
        h = np.zeros(n_points, dtype=np.float64)
        on_positions = (np.arange(n_points) % 2 == 0)
        h[on_positions] = 1.0
        h = h * (n_points / (2 * h.sum()))  # Normalize to integrate to 1
    n_points = len(h)

    def compute_overlaps(h):
        dx = 2.0 / n_points
        overlaps = np.zeros(n_points, dtype=np.float64)
        for k in range(n_points):
            indices = np.arange(n_points - k, dtype=int)
            overlaps[k] = np.sum(h[indices] * (1 - h[indices + k])) * dx
        return overlaps

    best_h = h.copy()
    best_c5 = compute_overlaps(h).max()

    while time.time() - start_time < budget_s:
        # Compute overlaps
        overlaps = compute_overlaps(h)
        max_idx = np.argmax(overlaps)
        
        # Compute gradient
        gradient = np.zeros_like(h, dtype=np.float64)
        for i in range(n_points - max_idx):
            gradient[i] += (1 - h[i + max_idx])
            gradient[i + max_idx] -= h[i]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient /= grad_norm
            
        # Learning rate
        learning_rate = 0.05

        # Update step
        h -= learning_rate * gradient

        # Projection: clip to [0,1]
        h = np.clip(h, 0, 1)
        
        # Project back onto the constraint sum(h) = n_points / 2
        total = h.sum()
        if total != 0 and total != n_points:
            alpha = (n_points / 2) / total
            h = np.clip(h * alpha, 0, 1)
        
        # Evaluate
        overlaps = compute_overlaps(h)
        c5 = overlaps.max()
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final constraint check and return
    total = best_h.sum()
    best_h = best_h * (n_points/2) / total
    best_h = np.clip(best_h, 0, 1)
    return best_h, best_c5, n_points
