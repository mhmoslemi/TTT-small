import numpy as np
import time
from scipy.interpolate import interp1d
from evaluate_erdos_solution import evaluate_erdos_solution

def compute_overlap(h, dx):
    shifted = np.zeros((2 * len(h) - 1,))
    for k in range(1 - len(h), len(h)):
        idxs = np.arange(len(h))
        valid = np.logical_and(idxs >= 0, idxs < len(h))
        shifted_k = k - (len(h) - 1)
        idxs_h = idxs + shifted_k
        valid = np.logical_and(idxs_h >= 0, idxs_h < len(h))
        overlap = np.sum(h[idxs] * (1 - h[idxs_h[valid]]))
        shifted[k] = overlap

    return np.max(shifted * dx)

def repair_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    scale = (n_points / 2) / total
    h = h * scale
    return h

def gaussian_perturbation(h, noise_scale):
    perturbed = h + np.random.normal(0, noise_scale, h.shape)
    return repair_feasibility(perturbed, h.size)

def interpolate_h(h, new_size):
    x_old = np.linspace(0, 2, len(h))
    f = interp1d(x_old, h, kind='cubic', fill_value="extrapolate")
    x_new = np.linspace(0, 2, new_size)
    h_new = f(x_new)
    return repair_feasibility(h_new, new_size)

def find_best_h(start_h, init_n, budget_s, final_n):
    h = start_h[:].copy()
    best_h = h.copy()
    best_c5 = compute_overlap(best_h, 2.0 / init_n)
    start_time = time.time()
    
    # Global optimization (simulated annealing)
    T = 1.0
    alpha = 0.99
    max_steps = int(budget_s / 1.5)
    saved_state = (best_h, best_c5)
    
    for step in range(max_steps):
        if time.time() - start_time > budget_s:
            break
            
        h_candidate = gaussian_perturbation(h, T * 0.1)
        c5_candidate = compute_overlap(h_candidate, 2.0 / init_n)
        
        if c5_candidate < best_c5:
            h[:] = h_candidate[:]
            best_h[:] = h_candidate[:]
            best_c5 = c5_candidate
        elif np.random.rand() < np.exp( (best_c5 - c5_candidate) / T):
            h[:] = h_candidate[:]
        
        T *= alpha
    
    # Local refinement (gradient-like updates)
    dx = 2.0 / init_n
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
            
        C = np.correlate(h, 1-h, mode='full') * dx
        k_max = np.argmax(C)
        shift = k_max - (init_n - 1)
        h_shifted = np.roll(h, shift)
        gradient = (1 - h_shifted)
        h -= 0.01 * gradient[:init_n]
        h = repair_feasibility(h, init_n)
        
        c5_candidate = compute_overlap(h, dx)
        if c5_candidate < best_c5:
            best_h[:] = h[:]
            best_c5 = c5_candidate
    
    # Interpolate to higher resolution
    if final_n > init_n and time.time() - start_time < budget_s - 50:
        h_high = interpolate_h(best_h, final_n)
        best_h = h_high
        init_n = final_n
        
    return best_h, best_c5, init_n

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    if 'initial_h_values' in globals() and initial_h_values.size > 0:
        start_h = initial_h_values
        n_points = len(start_h)
    else:
        n_points = 14
        start_h = np.random.uniform(0, 1, n_points)
        start_h = repair_feasibility(start_h, n_points)
    
    best_h, best_c5, n_points = find_best_h(start_h, n_points, budget_s, 56)
    
    elapsed = time.time()
    if elapsed > budget_s - 5:
        print(f"Time budget exceeded at {elapsed} seconds, returning best found solution")
    
    return best_h, best_c5, n_points
