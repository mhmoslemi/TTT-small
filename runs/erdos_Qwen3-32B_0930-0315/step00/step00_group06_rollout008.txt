import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use the initial_h_values provided, interpolate to a higher resolution
    if 'initial_h_values' in globals():
        h_0 = np.array(initial_h_values)
        n_0 = len(h_0)
    else:
        # Fallback initial guess: flat distribution
        n_0 = 95
        h_0 = np.ones(n_0) / n_0 * (n_0 / 2)
    
    # Interpolate to a finer grid for better resolution
    n_points = 256
    h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, n_0), h_0)
    h = h / np.sum(h) * (n_points / 2)  # Normalize to satisfy the integral constraint

    # Define some constants
    dx = 2.0 / n_points
    max_k = 2 * n_points  # Max lag for correlation
    learning_rate = 0.1
    n_iter = 0

    best_c5 = float('inf')
    best_h = h.copy()

    while time.time() - start_time < budget_s:
        # Compute correlation between h and 1 - h
        corr = np.correlate(h, 1 - h, mode="full")
        corr *= dx  # Adjust for the integral
        k_max = np.argmin(corr)  # We minimize the maximum overlap
        c5 = corr[k_max]
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Compute gradient
        # For discrete step function h, we calculate the gradient based on the correlation
        shift = k_max - n_points + 1
        grad = np.zeros_like(h)
        for i in range(n_points):
            if 0 <= i + shift < n_points:
                grad[i] = -(1 - h[i + shift])  # df/dh[i] = -(1 - h[i + shift])

        # Compute negative gradient and apply learning rate
        h_update = h - learning_rate * grad

        # Project back onto constraint set: [0, 1]^n
        h_update = np.clip(h_update, 0, 1)

        # Ensure the integral constraint is met: sum(h) == n_points / 2
        scale = (n_points / 2) / np.sum(h_update)
        h_update *= scale

        # Update h
        h = h_update
        n_iter += 1

        # Decay learning rate over time
        learning_rate *= 0.999  # Smooth decay

    # Ensure the solution is valid
    h = best_h
    h = np.clip(h, 0, 1)  # Final clipping
    scale = (n_points / 2) / np.sum(h)
    h *= scale  # Enforce sum constraint

    # Final evaluation
    corr = np.correlate(h, 1 - h, mode="full")
    c5_bound = np.max(corr * dx)

    return h.tolist(), c5_bound, n_points
