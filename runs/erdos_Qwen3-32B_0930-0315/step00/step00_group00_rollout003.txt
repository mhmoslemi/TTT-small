import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize, Bounds


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    h0 = np.array(initial_h_values)
    n_points = len(h0)
    dx = 2.0 / n_points

    def objective(h):
        h = h.reshape(-1)
        complement = 1.0 - h
        corr = np.correlate(h, complement, mode="full") * dx
        return np.max(corr)

    def constraint_sum(h):
        return h.sum() - n_points / 2

    def constraint_bounds(h):
        return np.array([0 <= h, h <= 1]).flatten()

    def optimize():
        best_h = h0.copy()
        best_c5 = objective(h0)
        learning_rate = 1e-2
        patience = 200
        best_iter = 0

        def wrapper_obj(h_vec):
            return objective(h_vec)

        for iter_count in range(10000):
            elapsed = time.time() - start_time
            if elapsed > budget_s:
                break

            h_flat = best_h.copy()
            # Perform gradient descent with SLSQP
            result = minimize(wrapper_obj, h_flat,
                              bounds=Bounds(0, 1),
                              constraints=({'type': 'eq', 'fun': constraint_sum},),
                              method='SLSQP', options={'maxiter': 5, 'ftol': 1e-10})
            if result.status == 0 and result.success:
                h_new = result.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_iter = iter_count
                    learning_rate = max(1e-3, learning_rate * 0.95)
                elif iter_count - best_iter > patience:
                    break
            else:
                # If minimizer fails, apply random perturbations
                perturbation = np.random.randn(n_points) * 0.05
                h_proposal = np.clip(best_h + perturbation, 0, 1)
                h_proposal = h_proposal / h_proposal.sum() * (n_points / 2)
                c5_proposal = objective(h_proposal)
                if c5_proposal < best_c5:
                    best_c5 = c5_proposal
                    best_h = h_proposal.copy()
        return best_h, best_c5

    h_opt, c5_bound = optimize()
    return h_opt, c5_bound, n_points
