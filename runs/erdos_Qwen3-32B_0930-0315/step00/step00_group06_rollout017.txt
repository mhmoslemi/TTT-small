import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 256  # higher resolution for better precision
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # corresponds to ∫₀² h(x) dx = 1

    # Initialize h
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        # Resample initial_h to n_points
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
        h = np.clip(initial_h, 0, 1)
    else:
        # Create a basic binary pattern with sum(h) = target_sum
        h = np.zeros(n_points)
        h[:int(target_sum)] = 1.0
        np.random.shuffle(h)

    # Start from initial
    h = h.copy()
    h_sum = h.sum()
    h = h / h_sum * target_sum

    # Ensure h stays within [0, 1]
    def project(h):
        h = np.clip(h, 0, 1)
        return h * target_sum / h.sum()

    h = project(h)

    # Start optimization loop
    for iter_num in range(1000):
        if time.time() - start_time > budget_s - 5:
            break

        # Compute correlation for all lags
        h_flipped = h[::-1]
        acf = correlate(h, h, mode='full') * dx
        # Only lags that are valid (1 to n_points-1)
        valid_lags = np.arange(1, n_points)
        acf_k = acf[n_points + valid_lags - 1]  # shift indices
        c5 = np.max(1.0 - acf_k * dx)

        # Find the worst (maximum) overlap
        k_worst = valid_lags[np.argmax(1.0 - acf_k * dx)]

        # Compute gradient at this point
        # The gradient of the overlap with respect to h is the vector:
        # [-h[i + k] for all i where i + k < n_points; +h[i] ... etc.]
        # Implemented via convolution-like indexing
        grad = np.zeros_like(h)
        for k in [k_worst, n_points - k_worst]:
            top_indices = np.arange(n_points - k)
            bottom_indices = top_indices + k
            grad[top_indices] += h[bottom_indices]
            grad[bottom_indices] -= h[top_indices]

        # Normalize gradient
        grad /= np.linalg.norm(grad)

        # Step toward minimizing the worst case
        step_size = 0.1 / n_points  # manual learning rate
        h -= step_size * grad
        h = project(h)

        if iter_num % 10 == 0:
            h_sum = h.sum()
            mean_h = h.mean()
            print(f"Iter: {iter_num}, C₅: {c5:.6f}, H_sum: {h_sum:.2f}, H_mean: {mean_h:.2f}")

    # Compute final C₅ bound
    h_flipped = h[::-1]
    acf = correlate(h, h, mode='full') * dx
    valid_lags = np.arange(1, n_points)
    acf_k = acf[n_points + valid_lags - 1]
    c5_final = np.max(1.0 - acf_k * dx)

    return h.tolist(), float(c5_final), n_points
