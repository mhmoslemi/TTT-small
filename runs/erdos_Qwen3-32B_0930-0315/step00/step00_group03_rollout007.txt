import numpy as np
import cvxpy as cp
import time
from math import floor


def run(seed=42, budget_s=1100, **kwargs):
    try:
        np.random.seed(seed)
        initial_h = initial_h_values
        n_points = len(initial_h)
    except NameError:
        n_points = 70
        initial_h = np.full(n_points, 0.5 / n_points * n_points)
        initial_h *= n_points / 2 / initial_h.sum()

    h = initial_h.copy()
    best_h = h.copy()
    dx = 2.0 / n_points

    best_c5 = np.inf
    start_time = time.time()
    improve_iter = 0

    while time.time() - start_time < budget_s - 1:
        overlap_results = np.correlate(h, 1 - h, mode="full") * dx
        k_bad = np.argmax(overlap_results)
        bad_c5 = overlap_results[k_bad]

        # Compute gradient
        G = np.zeros_like(h)
        for i in range(n_points):
            if i + k_bad < n_points:
                G[i] += (1 - h[(i + k_bad) % n_points])
            if i - k_bad >= 0:
                G[i] -= h[(i - k_bad) % n_points]
        G = G - h * G.sum() / (n_points * h.sum())

        learning_rate = 1.0 / (1 + (time.time() - start_time) / 500)
        h = h - learning_rate * G

        # Proximal projection to feasible region
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        objective = cp.Minimize(cp.sum_squares(h_var - h))
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.GUROBI, verbose=False)
        h = np.clip(h_var.value, 0, 1)

        if bad_c5 < best_c5:
            best_c5 = bad_c5
            best_h = h.copy()
            improve_iter = 0
        else:
            improve_iter += 1
            if improve_iter > 200:
                break

    overlap_results = np.correlate(best_h, 1 - best_h, mode="full") * dx
    final_c5 = np.max(overlap_results)
    return (best_h, final_c5, n_points)
