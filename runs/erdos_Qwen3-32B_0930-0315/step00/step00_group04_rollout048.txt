import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    initial = np.array(initial_h_values)
    h = initial.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    start_time = time.time()
    max_iter = 1000000
    learning_rate = 0.01
    decay_steps = 100
    decay_rate = 0.95

    def project(h):
        h = np.clip(h, 0, 1)
        total = h.sum()
        if total > 0:
            h *= (n_points / 2) / total
        else:
            h[:] = 1 / n_points
        h = np.clip(h, 0, 1)
        return h

    h = project(h)

    best_h = h.copy()
    best_c5 = float('inf')

    iter_idx = 0
    while time.time() - start_time < budget_s and iter_idx < max_iter:
        if iter_idx % 100 == 0:
            learning_rate *= decay_rate

        u = correlate(h, 1 - h, mode='full') * dx
        k_max = np.argmax(u)
        k_max = np.clip(k_max, 0, n_points - 1)

        shifted = np.roll(1 - h, k_max)
        grad = shifted[n_points - 1:]  # correlate output shape matters
        grad *= (2.0 / n_points)
        h -= learning_rate * grad

        h = project(h)

        u = correlate(h, 1 - h, mode='full') * dx
        c5 = np.max(u)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        iter_idx += 1

    return best_h.tolist(), best_c5, n_points
