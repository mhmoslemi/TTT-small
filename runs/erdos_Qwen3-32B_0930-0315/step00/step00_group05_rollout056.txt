import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if available
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Ensure initial_h satisfies constraint sum(h) == n_points / 2
    def project_h(h):
        h = np.clip(h, 0, 1)
        scaling = (n_points / 2) / np.sum(h)
        h = h * scaling
        h = np.clip(h, 0, 1)
        return h

    h = np.array(initial_h_values, dtype=np.float64).copy()
    h = project_h(h)

    start_time = time.time()
    step = 0
    total_time = 0
    best_h = h.copy()
    best_c5 = np.inf
    pat = 0
    alpha = 0.1  # Initial step size

    while total_time < budget_s:
        ok_k = []
        for k in range(1, n_points):
            if not (np.all(k < np.arange(n_points))):
                continue
            overlaps = correlate(h, 1 - h, mode='valid') * dx
            if overlaps.shape[0] == 0:
                continue
            max_ov = overlaps.max()
            ok_k.append((k, max_ov))
        if not ok_k:
            break
        lag_k, max_c5 = max(ok_k, key=lambda x: x[1])
        
        grad = np.zeros(n_points)
        for m in range(n_points - lag_k):
            grad[m] += (1 - h[m + lag_k]) * dx
        for m in range(lag_k, n_points):
            grad[m] -= h[m - lag_k] * dx
        
        h -= alpha * grad
        h = project_h(h)
        overlaps = correlate(h, 1 - h, mode='valid') * dx
        max_c5_current = overlaps.max() if overlaps.size > 0 else 0.0

        # Track best value
        if max_c5_current < best_c5:
            best_c5 = max_c5_current
            best_h = h.copy()
            pat = 0  # Reset patience
        else:
            pat += 1

        # Adaptive learning rate
        if pat >= 30:
            alpha = max(alpha * 0.99, 1e-6)

        # Add small noise every 50 steps to help escape local minima
        if step > 0 and step % 50 == 0:
            noise = np.random.normal(0, 0.1, n_points)
            h += noise
            h = project_h(h)
        
        # Track time budget
        step += 1
        total_time = time.time() - start_time
        if total_time + 0.1 > budget_s:
            break

    h_values = best_h
    c5_bound = best_c5
    return (h_values, c5_bound, n_points)
