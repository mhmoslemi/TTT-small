import numpy as np
import time

def evaluate_C5(h, n_points):
    dx = 2.0 / n_points
    corr = np.zeros(n_points + n_points)
    for k in range(-n_points + 1, n_points):
        if k <= 0:
            idxs = (np.arange(n_points) + n_points + k)
        else:
            idxs = (np.arange(n_points) - k)
        idxs = np.mod(idxs, n_points)
        integral = np.sum(h * (1 - h[idxs]))
        corr[n_points + k] = integral
    return np.max(corr) * dx


def project_to_feasible(h, n_points):
    # Clip to [0, 1]
    h_proj = np.clip(h, 0, 1)
    current_sum = np.sum(h_proj)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        scale = target_sum / current_sum
        h_proj *= scale
        h_proj = np.clip(h_proj, 0, 1)  # Ensure clipping again post scaling
    return h_proj
