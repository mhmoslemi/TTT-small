```python
import numpy as np
import time
from numpy import correlate
from cvxpy import Variable, Problem, Max, Min, sum as cvxsum, OneNorm, Solver, solve
from evaluate_erdos_solution import evaluate_erdos_solution
import scipy.optimize

def run(budget_s=1100):
    initial_h_values = getattr(run, 'initial_h_values', None)
    actual_h = initial_h_values.astype(float).copy() if initial_h_values is not None else None
    n_points = 45  # Starting with parent construction

    np.random.seed(42)
    dx = 2.0 / n_points

    def compute_c5(h):
        h = np.atleast_1d(h)
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    # Normalize h for unit area, handling numerical stability
    def normalize_h(h):
        s = np.sum(h)
        return h * (n_points / 2) / s if s != 0 else h

    def optimize_with_cnnls(h_initial, dx, budget_s_sub):
        h_vec = Variable(n_points)
        constraints = [0 <= h_vec, h_vec <= 1, cvxsum(h_vec) == n_points / 2]
        corr_values = []
        for k in range(2 * n_points - 1):
            start = max(0, k - n_points + 1)
            stop = min(n_points, k + 1)
            shift = k
            h1 = h_vec[shift:n_points]
            h2 = h_vec[0:(n_points - shift)]
            corr_values.append(cvxsum(h1 * (1 - h2)))
        C5 = Max(*corr_values)
        prob = Problem(Minimize(C5), constraints)
        start_time = time.time()
        while time.time() - start_time < budget_s_sub:
            prob.solve(solver='ECOS', max_iter=2000, kktsolver='robust')
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_opt = np.array(h_vec.value).flatten()
                break
            else:
                if abs(prob.solve(solver='ECOS', max_iter=2000)) < 1e-2:
                    break
        if 'h_opt' not in locals():
            h_opt = np.array(h_vec.value).flatten()
        h_opt = np.clip(h_opt, 0, 1)
        h_final = normalize_h(h_opt)
        return h_final, compute_c5(h_final)

    if actual_h is None:
        actual_h = normalize_h(np.random.rand(n_points))
    start_time = time.time()
    best_h = actual_h.copy()
    best_c5 = compute_c5(best_h)
    last_update_time = time.time()
    update_interval = 30

    print_interval = 60
    last_print_time = time.time()

    while time.time() - start_time < budget_s:
        elapsed = time.time() - start_time
        remaining = max(0, budget_s - elapsed)
        h_opt, c5_opt = optimize_with_cnnls(actual_h, dx, remaining / 2)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            last_update_time = time.time()
        actual_h = best_h
        elapsed = time.time() - start_time
        remaining = max(0, budget_s - elapsed)
        if time.time() - last_print_time >= print_interval:
            print(f"Time elapsed: {elapsed:.2f}s, Best C5: {best_c5:.8f}, Remaining: {remaining:.2f}s")
            last_print_time = time.time()
            if remaining < 10:
                break
    final_h = np.clip(best_h, 0, 1)
    final_h = normalize_h(final_h)
    best_c5 = compute_c5(final_h)
    final_result = evaluate_erdos_solution(final_h)
    return final_h.tolist(), final_result['c5'], n_points
```
