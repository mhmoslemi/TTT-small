import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, verbose=False, **kwargs):
    np.random.seed(seed)

    # Initial setup
    n_points = 70  # Number of samples
    dx = 2.0 / n_points

    # Use initial construction if provided
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.ones(n_points) * 0.5

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    start_time = time.time()
    iter_no_improvement = 0
    max_no_improvement = 100

    while time.time() - start_time < budget_s:
        # Step 1: Compute all overlaps
        overlap = np.correlate(h, 1 - h, mode='full') * dx
        # Compute only the valid shifts: k in range(1, n_points)
        k_values = np.arange(1, n_points)
        valid_overlap = overlap[n_points: n_points + len(k_values)]

        # Find the worst k that maximizes overlap
        max_idx = np.argmax(valid_overlap)
        k = max_idx + 1  # shift index
        worst_overlap = valid_overlap[max_idx]

        # Step 2: Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k < n_points:
                grad[i] += 1 - h[i + k]
            if i >= k:
                grad[i] -= h[i - k]
        
        # Step 3: Learning rate schedule
        iter_count = int((time.time() - start_time) / max(1e-6, 0.1))
        lr = 1 / (1 + iter_count / 1000)

        # Step 4: Projected gradient descent
        h_candidate = h - lr * grad
        h = cp.Variable(n_points)
        constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2]
        objective = cp.Minimize(cp.sum_squares(h - h_candidate))
        problem = cp.Problem(objective, constraints)
        problem.solve(solver='ECOS', verbose=False)
        h = h.value

        # Ensure numerical precision
        h = np.clip(h, 0.0, 1.0)

        # Step 5: Compute and track the best
        c5 = compute_c5(h, dx)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            iter_no_improvement = 0
        else:
            iter_no_improvement += 1

        # Early stopping
        if iter_no_improvement >= max_no_improvement:
            break

    # Validate the best solution
    best_h = np.array(best_h)
    assert np.all(best_h >= -1e-6) and np.all(best_h <= 1.0 + 1e-6)
    total_sum = np.sum(best_h)
    assert abs(total_sum - n_points / 2) < 1e-6

    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    h = np.array(h)
    overlap = np.correlate(h, 1 - h, mode='full') * dx
    if len(overlap) < 2 * len(h) - 1:
        pad = np.pad(overlap, ((len(h) - 1,), (0,)), 'constant')  # Safety padding
    else:
        pad = overlap
    valid = pad[n_points: 2 * n_points - 1]
    return np.max(valid)
