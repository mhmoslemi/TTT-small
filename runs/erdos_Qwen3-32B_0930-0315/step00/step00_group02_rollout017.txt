import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Parameters
    n_points = 160
    dx = 2.0 / n_points
    name = f'n{n_points}'

    def objective(h):
        # Compute centered 1 - h[\cdot]
        shifted_h = 1 - h
        corr = np.correlate(h, shifted_h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Initialize h from parent if available
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        if h.shape[0] != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, h.shape[0]), h)
    else:
        # Fallback initialization: uniform plateau
        h = np.zeros(n_points)
        h[n_points // 4 : 3 * n_points // 4] = 0.5
        h *= 2 / h.sum()  # Scale to have area exactly 1

    # Gradient descent with backtracking line search
    max_iter = 1000
    step_size = 0.01
    proj_cnt = 0
    best_hist = [h.copy()]
    best_c5 = objective(h)
    starting_time = time.time()

    for i in range(max_iter):
        # Compute gradient of objective
        shifted_h = 1 - h
        corr = np.correlate(h, shifted_h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - n_points + 1  # Shift index corresponds to k shift in the overlap

        if 0 <= k < n_points:
            # Compute backward and forward gradients
            backward = h[0 : n_points - k]
            forward = h[k : ]
            grad = -(shifted_h[0 : n_points - k] + shifted_h[k : ])
            grad_h = np.zeros(n_points)
            grad_h[0 : n_points - k] += grad[:n_points - k]
            grad_h[k : ] += grad[:n_points - k]
        else:
            # For k outside valid window (not used in max)
            grad_h = np.zeros(n_points)

        if np.linalg.norm(grad_h) < 1e-4:
            print("Converged on gradient norm")
            break

        # Line search with backtracking
        step = step_size
        while True:
            h_hat = h - step * (grad_h / np.max([1.0, np.linalg.norm(grad_h)]))
            # Project to feasible region
            h_proj = np.clip(h_hat, 0, 1)
            h_proj = h_proj * (n_points / 2) / h_proj.sum()
            c5_hat = objective(h_proj)
            if c5_hat < objective(h):
                break
            step *= 0.5
            if step < 1e-6:
                break
        h = h_proj
        c5_hat = objective(h)

        best_hist.append(h.copy())
        best_c5 = min(best_c5, c5_hat)
        proj_cnt += 1

        if time.time() - starting_time > budget_s - 10:
            print("Time budget is running out, stopping early")
            break

        if i % 100 == 0 or c5_hat < 0.381:
            print(f"Iter {i}, c5 = {objective(h):.8f}, best_heuristic_c5 = {best_c5:.8f}")

    # Final refinement with CVXPY
    h_ref = cp.Variable(n_points)
    constraints = [h_ref >= 0, h_ref <= 1, cp.sum(h_ref) == n_points / 2]

    # Previous best solution to warm start
    prev_h = h

    # Build nonlinear convex approximation around prev_h
    def overlap_gradient(h):
        shifted_h = 1 - h
        corr = np.correlate(h, shifted_h, mode="full")
        max_k = np.argmax(corr)
        k_int = max_k - n_points + 1
        if k_int < 0 or k_int >= n_points:
            return np.zeros(n_points)
        backward = h[0 : n_points - k_int]
        forward = h[k_int : ]
        grad = -(shifted_h[0 : n_points - k_int] + shifted_h[k_int : ])
        grad_h = np.zeros(n_points)
        grad_h[0 : n_points - k_int] += grad[:n_points - k_int]
        grad_h[k_int : ] += grad[:n_points - k_int]
        return grad_h

    grad_ref = overlap_gradient(prev_h)

    # Quadratic model to refine current best
    relaxed_objective = grad_ref @ (h_ref - prev_h)
    problem = cp.Problem(cp.Minimize(relaxed_objective), constraints)
    problem.solve(solver=cp.ECOS, warm_start=True)

    if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_best = h_ref.value.clip(0, 1)
        h_best = h_best * (n_points / 2) / h_best.sum()
        c5_best = objective(h_best)
        if c5_best < best_c5:
            best_hist.append(h_best.copy())
            best_c5 = c5_best
    else:
        h_best = best_hist[-1]

    return h_best, best_c5, n_points
