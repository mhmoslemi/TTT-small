import numpy as np
import time
import cvxpy as cp
from scipy.interpolate import interp1d

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def compute_c5(h, n_points):
        dx = 2.0 / n_points
        h = np.clip(h, 0, 1)
        # Constrain the integral
        h /= (np.sum(h) * dx)
        h *= 0.5 * n_points
        h = np.clip(h, 0, 1)
        one_h = 1 - h
        corr = np.correlate(h, one_h, mode="full") * dx
        return np.max(corr)

    def evaluate_population(population, n_points):
        return [compute_c5(ind, n_points) for ind in population]

    def constrain(h):
        h = np.clip(h, 0, 1)
        total_mass = np.sum(h) * (2.0 / len(h))
        scale = 0.5 * len(h) / np.sum(h)
        h = h * scale
        h = np.clip(h, 0, 1)
        return h

    def initialize_population(pop_size, n_points):
        pop = []
        if 'initial_h_values' in globals():
            base_h = constrain(globals()['initial_h_values'].copy())
            base_h = np.convolve(base_h, np.ones(n_points // 2) / (n_points // 2), mode='same')
            base_h *= len(h) * 0.5 / np.sum(base_h)
            base_h = constrain(base_h)
            for _ in range(pop_size):
                h = base_h * (1 + 0.05 * np.random.randn(n_points))
                pop.append(constrain(h))
        else:
            for _ in range(pop_size):
                h = 0.5 * np.random.rand(n_points) + 0.25
                pop.append(constrain(h))
        return np.array(pop)

    def crossover(parents):
        n_points = len(parents[0])
        idx = np.random.choice(len(parents), 2, replace=False)
        p1, p2 = parents[idx]
        split = np.random.randint(1, n_points - 1)
        return np.concatenate((p1[:split], p2[split:])), np.concatenate((p2[:split], p1[split:]))

    def mutate(h, rate=0.2):
        noise = rate * np.random.randn(len(h))
        return constrain(h + noise)

    # Start at n=14 for fast exploration
    n_points = 14
    pop_size = 20
    population = initialize_population(pop_size, n_points)
    objective = evaluate_population(population, n_points)
    best_idx = np.argmin(objective)
    best_h = population[best_idx]
    best_c5 = objective[best_idx]
    history = [best_c5]
    eta = 0.01
    start_time_res = time.time()

    while time.time() - start_time < budget_s * 0.6:
        # Genetic Algorithm
        next_gen = []
        for _ in range(pop_size):
            if np.random.rand() < 0.8:
                # Crossover
                p1 = population[np.random.choice(pop_size)]
                p2 = population[np.random.choice(pop_size)]
                c1, c2 = crossover([p1, p2])
                c1, c2 = mutate(c1, rate=0.15), mutate(c2, rate=0.15)
                next_gen.extend([c1, c2])
            else:
                # Mutation
                parent = population[np.random.choice(pop_size)]
                offspring = mutate(parent, rate=0.15)
                next_gen.append(offspring)
        next_gen = np.array(next_gen)
        population = np.vstack([population, next_gen])[:pop_size]
        objective = evaluate_population(population, n_points)
        best_idx = np.argmin(objective)
        current_h = population[best_idx]
        current_c5 = objective[best_idx]
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        history.append(best_c5)
        # Early stop if converged
        if len(history) > 5 and all(abs(history[-1] - x) < 1e-5 for x in history[-6:-1]):
            break

    # Gradient descent refinement
    grad_steps = 0
    while grad_steps < 50 and time.time() - start_time < budget_s * 0.9:
        h = best_h.copy()
        h_new = h - eta * np.correlate((h * 0.5).round(), -best_h, mode='same')
        h_new = constrain(h_new)
        c5_new = compute_c5(h_new, n_points)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            eta *= 1.2
        else:
            eta *= 0.8
        grad_steps += 1

    # Resolution doubling
    def double_resolution(h):
        n = len(h)
        x = np.linspace(0, 2, n)
        f = interp1d(x, h, kind="cubic")
        x_fine = np.linspace(0, 2, 2*n)
        h_fine = f(x_fine)
        h_fine *= 0.5 * len(h_fine) / np.sum(h_fine)
        h_fine = np.clip(h_fine, 0, 1)
        return h_fine

    n_points_fine = 28
    best_h_fine = double_resolution(best_h) if n_points == 14 else best_h

    final_c5 = compute_c5(best_h_fine, n_points_fine)
    elapsed = time.time() - start_time

    # Return best found solution
    return best_h_fine, final_c5, n_points_fine
