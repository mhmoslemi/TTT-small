import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
import math

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    # Get the current best construction
    h0 = initial_h_values
    n = len(h0)

    # Constraints
    def compute_overlaps(h):
        dx = 2.0 / n
        h_pad_half = np.r_[h, np.zeros(n - 1)]
        one_minus_h = 1 - h
        one_minus_h_pad_half = np.r_[one_minus_h, np.zeros(n - 1)]
        result = correlate(h_pad_half, one_minus_h_pad_half, mode="valid")
        return result * dx

    def compute_objective(h):
        overlaps = compute_overlaps(h)
        return np.max(overlaps)

    # Enforce constraints: sum(h) == n / 2, but only for active elements in the sum
    def project(h):
        h = np.clip(h, 0, 1)
        total = h.sum()
        if abs(total - n / 2) > 1e-6:
            h = h * (n / 2 / total)
        return h

    # Initial step
    h = np.array(project(h0))

    best_h = h.copy()
    best_c5 = compute_objective(h)
    patience = 0

    learning_rate = 0.01
    max_patience = 20
    history_length = 5
    history = []

    while time.time() - t0 < budget_s:
        overlaps = compute_overlaps(h)
        max_index = np.argmax(overlaps)
        k = max_index  # shift index

        # Compute gradient for the worst shift
        grad = np.zeros_like(h)
        for i in range(n - k):
            grad[i] += (1 - h[i + k])
            grad[i + k] -= h[i]

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-8:
            grad = grad / grad_norm

        # Update step
        h = h - learning_rate * grad
        h = project(h)

        # Compute new objective
        new_c5 = compute_objective(h)

        # Learning rate decay
        learning_rate *= 0.999 + 0.001 * (new_c5 < best_c5)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            # Perturbation: drop a few values and refill to escape local optima
            perturb_indices = np.argsort(np.random.rand(n))[:3]
            h[perturb_indices] = np.random.random(3)
            h = project(h)
            patience = 0

        # Append to history and manage
        history.append(new_c5)
        if len(history) > history_length:
            history.pop(0)
        # Test for futility
        if len(history) == history_length and np.std(history) < 1e-6:
            break

    # Final round checks and clean up
    best_h = project(best_h)
    final_c5 = compute_objective(best_h)
    return list(best_h), final_c5, n
