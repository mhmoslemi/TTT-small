import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def run(seed=42, budget_s=1100):
    # Parameters
    n_points = 256
    dx = 2.0 / n_points

    # Initialization
    np.random.seed(seed)
    h = np.zeros(n_points, dtype=np.float64)
    if 'initial_h_values' in globals():
        # Interpolate original to higher resolution
        h_initial_fine = np.interp(
            np.linspace(0, 2, n_points), 
            np.linspace(0, 2, len(initial_h_values)), 
            initial_h_values
        )
        h = h_initial_fine.copy()
    else:
        # Binary approximation with correct area
        ones_needed = n_points // 2
        h[:ones_needed] = 1.0
        h[ones_needed:] = 0.0

    # Optimization parameters
    max_iter = 1000
    step_size = 0.01
    start_time = time.time()

    # Iterative convex optimization
    for iteration in range(max_iter):
        t = time.time()
        if t - start_time > budget_s:
            break

        # Compute auto-correlation of h
        ac = correlate(h, h, mode='full')
        ac_binary = ac[n_points-1:]  # Only need k ≥ 1 overlaps

        # Compute max overlap corresponding to min auto-correlation
        overlap = 1.0 - ac_binary * dx
        idx_max_overlap = np.argmax(overlap)
        k_max = idx_max_overlap
        max_overlap_value = overlap[idx_max_overlap]

        # Gradient update at shift k
        grad_h = np.zeros_like(h)
        grad_h[:n_points - k_max] = -h[k_max:n_points]
        grad_h[k_max:n_points] = -h[:n_points - k_max]

        # Update h
        h += step_size * grad_h

        # Project to domain [0, 1]
        h = np.clip(h, 0.0, 1.0)

        # Normalize to ensure ∫h(x) dx = 1
        area = np.sum(h) * dx
        if abs(area - 1.0) > 1e-8:
            scale = 1.0 / area
            h *= scale

    # Compute final performance metric
    ac_final = correlate(h, h, mode='full')
    ac_final_binary = ac_final[n_points - 1:]  # k ≥ 1
    overlap_final = 1.0 - ac_final_binary * dx
    c5_bound = np.max(overlap_final)

    return h, c5_bound, n_points
