import numpy as np
import cvxpy as cp
import time
import evaluate_erdos_solution

def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    C5 = np.max(np.abs(corr * dx))
    return C5

def project_onto_simplex(v, z=1.0):
    v = np.sort(v)[::-1]
    cssv = np.cumsum(v) - z
    ind = np.arange(1, len(v) + 1) * v >= cssv
    idx = np.max(np.nonzero(ind))
    theta = float(cssv[idx] / (idx + 1.0))
    w = np.maximum(v - theta, 0.0)
    return w

    start_time = time.time()

    best_h = np.copy(h)
    best_C5 = evaluate_C5(h)

    time_per_iter = 5.0  # Approximate time per convex solve
    max_iter = int((budget_s - 30) // time_per_iter)  # Leave 30s for final eval

        # Convert k to inner indexing

        # Define CVXPY problem to minimize max overlap at lag k
        h_opt = cp.Variable(n_points)
        constraints = [
            h_opt >= 0,
            h_opt <= 1,
            cp.sum(h_opt) == n_points / 2.0
        ]

        shifted_h = np.roll(h_opt, inner_k)
        obj = cp.Minimize(cp.sum(h_opt * (1 - shifted_h)))

        prob = cp.Problem(obj, constraints)

        try:
            prob.solve(solver=cp.MOSEK)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)

        new_C5 = evaluate_C5(h_new)

        if new_C5 < best_C5:
            best_h = np.copy(h_new)
            best_C5 = new_C5

        h = h_new  # For next iteration
        total_time = time.time() - start_time
        max_iter -= 1

    # Final evaluation and return
    return best_h.tolist(), min(best_C5, evaluate_C5(best_h)), n_points
