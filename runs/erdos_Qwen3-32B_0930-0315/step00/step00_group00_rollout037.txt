```python
import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

# Helper: compute the objective C₅
def compute_c5(h, dx):
    h = np.array(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

# Project to feasible set
def project_to_feasible(h, n_points):
    dx = 2.0 / n_points
    total = sum(h) * dx
    h = np.clip(h, 0, 1)
    h *= n_points / 2 / sum(h)
    return h

# Tabu Search
def run(seed=42, budget_s=1100, n_points=45):
    np.random.seed(seed)
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = project_to_feasible(h, n_points)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    start_time = time.time()
    tabu_memory = set()
    best_h_key = tuple(h.round(8).tolist())

    num_iters = 0

    while time.time() - start_time < budget_s:
        neighbors = []
        for _ in range(20):
            # Perturbation: swap values between random indices to preserve constraint
            idxs = np.random.choice(n_points, size=5, replace=False)
            copy_h = h.copy()
            for idx in idxs:
                copy_h[idx] += np.random.uniform(-0.1, 0.1)
            copy_h = project_to_feasible(copy_h, n_points)
            neighbors.append(copy_h)

        next_h = None
        for neighbor in neighbors:
            neighbor_key = tuple(neighbor.round(8).tolist())
            if len(tabu_memory) > 10:
                tabu_memory.pop()
            if neighbor_key not in tabu_memory:
                neighbor_c5 = compute_c5(neighbor, dx)
                if neighbor_c5 < best_c5:
                    next_h = neighbor
                    best_c5 = neighbor_c5
        if next_h is None:
            # Fallback: a little gradient-like step
            grad = np.random.randn(n_points)
            step = cp.Variable(n_points)
            constraint = [sum(step) == 0, step >= -h, h + step <= 1]
            obj_step = cp.Minimize(cp.norm(grad - step, 2))
            prob_step = cp.Problem(obj_step, constraint)
            try:
                prob_step.solve(solver='ECOS', verbose=False)
            except:
                pass
            next_h = project_to_feasible((h + step.value).flatten(), n_points)

        h = next_h
        tabu_memory.add(tuple(h.round(8).tolist()))

        if time.time() - start_time >= budget_s:
            break

        num_iters += 1
        # Update best solution
        current_best_c5 = compute_c5(h, dx)
        if current_best_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_best_c5

    return best_h.tolist(), best_c5, n_points
```
