import numpy as np
import time
import random

# Helper functions
def normalize_h(h):
    idx = np.where((h >= 0) & (h <= 1))[0]
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum == 0:
        h.fill(0.5)
        return h
    scaling = h.size / (2 * h_sum)
    h *= scaling
    h = np.clip(h, 0, 1)
    return h

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def generate_random_h(n_points):
    h = np.random.uniform(0, 1, n_points)
    return normalize_h(h)

def tournament_selection(population, fitness, tournament_size):
    idx = np.random.choice(len(population), tournament_size)
    best_idx = np.argmin(fitness[idx])
    return population[idx[best_idx]]

def crossover(parent1, parent2):
    n_points = len(parent1)
    if random.random() < 0.7:  # 70% crossover chance
        crossover_point = np.random.randint(1, n_points - 1)
        child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
    else:
        child = (parent1 + parent2) / 2
    return normalize_h(child)

def mutate(parent, temperature, n_points):
    child = parent.copy()
    num_mutations = np.random.poisson(1)
    for _ in range(num_mutations):
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 1 / temperature)
        child[idx] = np.clip(child[idx] + delta, 0 - 1e-6, 1 + 1e-6)
    return normalize_h(child)
