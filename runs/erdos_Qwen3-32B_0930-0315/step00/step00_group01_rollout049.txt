import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Constants
    dx = 2.0  # Start with n_points=1
    while True:
        n_points_candidate = (time.time() - start_time) / 50  # Adjust heuristic based on progress
        if n_points_candidate > 1000 or time.time() - start_time > budget_s * 0.9:
            break
        n_points = int(round(n_points_candidate / 100)) * 100 + 50  # Ensure it's a multiple of 50
    n_points = max(min(n_points, 1000), 2)  # Ensure n_points is within [2, 1000]

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = initial_h_values / initial_h_values.sum() * n_points / 2
        n_points = len(h_best)
    else:
        # Construct symmetric binary initial guess
        h_best = np.zeros(n_points)
        n_ones = n_points // 2
        h_best[:n_ones] = 1
        np.random.seed(seed)
        np.random.shuffle(h_best)

    best_c5 = float('inf')

    # Optimization loop
    learning_rate = 0.1
    momentum = 0.9
    max_iters = 1000
    velocity = 0

    for iter in range(max_iters):
        if time.time() - start_time > budget_s:
            break

        # Compute objective and gradients
        h = h_best.copy()
        corr = scipy.signal.correlate(h, 1 - h, mode="full")
        c5 = (2.0 / n_points) * max(corr)

        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            h_best = h.copy()

        # Gradient descent
        grad = np.zeros_like(h)
        for k in range(-n_points + 1, n_points):
            idx = np.flatnonzero((h.size + k - 1) >= 0)
            grad_h = (1 - h) if k <= 0 else np.roll(1 - h, -k)
            grad -= (1 / n_points) * (corr[k] * (1 - 2 * h))
        
        # Apply momentum and step
        velocity = momentum * velocity - learning_rate * grad
        h_best += velocity

        # Project back to feasible region
        h_best = np.clip(h_best, 0, 1)
        h_best = h_best / h_best.sum() * n_points / 2  # Ensure the sum constraint

        # Adaptive learning rate
        learning_rate *= 0.999

    elapsed = time.time() - start_time
    print(f"Optimized for {elapsed:.2f}s, found C5 bound: {best_c5:.6f}")

    return (h_best, best_c5, n_points)
