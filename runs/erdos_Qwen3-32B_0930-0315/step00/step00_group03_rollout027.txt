import numpy as np
import time
import numpy.random as npr
import scipy.signal as sig
import copy

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    t_start = time.time()
    n_points = 70
    dx = 2.0 / n_points
    total_mass = n_points / 2.0

    # 0. Initialization steps
    pop_size = 50
    tournament_size = 5
    mutation_rate = 0.15
    start_from_parent_id = 0

    # Build initial population
    pop = []
    if 'initial_h_values' in globals():
        # Strategy 1: use parent construction
        pop.append(np.array(initial_h_values))
        start_from_parent_id = 1

        # Add perturbed versions of the parent
        for i in range(30 - start_from_parent_id):
            h = copy.copy(initial_h_values)
            perturbation = (npr.rand() - 0.5) * 0.1
            h = np.clip(h + perturbation, 0, 1)
            h = total_mass / h.sum() * h
            h = np.clip(h, 0, 1)
            pop.append(h)

    # Fill rest with random samples
    for _ in range(pop_size - len(pop)):
        h = npr.uniform(0, 1, size=n_points)
        h = total_mass / h.sum() * h
        h = np.clip(h, 0, 1)
        pop.append(h)

    pop = np.array(pop)

    # Main GA loop
    best_h = np.median(pop, axis=0)
    best_h = total_mass / best_h.sum() * best_h
    best_h = np.clip(best_h, 0, 1)
    best_c5 = np.inf

    generation = 0
    while time.time() - t_start < budget_s:
        new_pop = []

        # Strength-based selection
        fitness = -np.array([np.max(np.correlate(h, 1 - h, mode="full") * dx) for h in pop])
        fitness -= np.min(fitness)
        fitness += 1e-6

        # Selection
        indices = []
        for _ in range(pop_size // 2):
            tournament = npr.choice(pop_size, size=tournament_size)
            winner = tournament[np.argmax(fitness[tournament])]
            indices.append(winner)

        selected = pop[indices]

        # Crossover
        for i in range(0, len(selected), 2):
            if i+1 < len(selected):
                parent1, parent2 = selected[i], selected[i+1]
                cpoint = npr.randint(1, n_points)
                child1 = np.concatenate([parent1[:cpoint], parent2[cpoint:]])
                child2 = np.concatenate([parent2[:cpoint], parent1[cpoint:]])
                for child in [child1, child2]:
                    med_prop = total_mass / child.sum()
                    med_prop = np.clip(med_prop, 0.9, 1.1)
                    child = np.clip(child * med_prop, 0, 1)
                    child[child < 1e-3] = 0
                    child[child > 1 - 1e-3] = 1
                    new_pop.append(child)

        # Mutation
        for i in range(len(new_pop)):
            if npr.rand() < mutation_rate:
                gene_ids = npr.choice(n_points, size=int(0.05 * n_points), replace=False)
                for idx in gene_ids:
                    step = npr.randn() * 0.1
                    new_pop[i][idx] = np.clip(new_pop[i][idx] + step, 0.0, 1.0)

                # Maintain sum constraint
                s = total_mass / new_pop[i].sum()
                new_pop[i] = np.clip(new_pop[i] * s, 0, 1)

        # Combine and evaluate the new population
        pop = np.vstack([pop, np.array(new_pop)])
        pop = np.unique(pop, axis=0)

        # Get the best solutions for the next generation
        scores = np.array([np.max(np.correlate(h, 1 - h, mode="full") * dx) for h in pop])
        pop = pop[np.argsort(scores)][:pop_size]

        # Update best solution
        best_id = np.argmin(scores)
        current_c5 = scores[best_id]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = pop[best_id].copy()

        # Progress tracking
        generation += 1
        if generation % 20 == 0:
            print(f'Gen {generation}: best c5 = {best_c5:.6f}')

        # Early stopping check
        if best_c5 <= 0.380876:
            break

    # Final checks
    best_h = np.clip(best_h, 0, 1)
    if abs(np.sum(best_h) - total_mass) > 1e-6:
        best_h = total_mass / np.sum(best_h) * best_h

    return best_h.tolist(), best_c5, n_points
