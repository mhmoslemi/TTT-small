```python
import numpy as np
import time
import cvxpy as cp
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Copy initial solution if available
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_val = np.array(initial_h_values, dtype=np.float64)
    else:
        # Random initialization with constraint satisfied
        n_points = 45
        h_val = np.random.randint(0, 2, n_points)
        h_val = h_val / h_val.sum() * n_points / 2
        h_val = np.clip(h_val, 0, 1)

    # Normalize to satisfy mass constraint exactly
    def normalize_h(h):
        h = h.copy()
        mass = h.sum()
        correction = (n_points / 2) - mass
        idx = np.argsort(h)
        for i in idx:
            if correction > np.abs(h[i] - 0.5):
                h[i] += correction * np.sign(h[i] - 0.5)
                correction = 0
            else:
                h[i] = 0.5 if h[i] > 0.5 else 0.5 if h[i] < 0.5 else h[i] + np.sign(correction)
                correction -= np.abs(h[i] - 0.5)
                break
        return h

    h_best = h_val.copy()
    best_c5 = evaluate_erdos_solution(h_best)
    tabu_memory = []

    start_time = time.time()

    while time.time() - start_time < budget_s:

        candidates = []
        # Neighborhood 1: Perturb random points with constraint preservation
        for _ in range(20):
            h_new = h_val + np.random.uniform(-0.5, 0.5, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new)
            if all(np.allclose(h_new, old_h) for old_h in tabu_memory):
                continue
            candidates.append(h_new)

        # Neighborhood 2: Stepwise shift
        for i in range(n_points):
            h_new = h_val.copy()
            h_new = np.roll(h_new, 1)
            h_new = normalize_h(h_new)
            if all(np.allclose(h_new, old_h) for old_h in tabu_memory):
                continue
            candidates.append(h_new)
            h_new = np.roll(h_new, -1)
            h_new = normalize_h(h_new)
            if all(np.allclose(h_new, old_h) for old_h in tabu_memory):
                continue
            candidates.append(h_new)

        # Evaluate all candidates
        for candidate in candidates:
            c5_candidate = evaluate_erdos_solution(candidate)
            if c5_candidate < best_c5:
                h_best = candidate.copy()
                best_c5 = c5_candidate

        # Maintain tabu memory
        tabu_memory.append(h_val.copy())
        if len(tabu_memory) > 10:
            tabu_memory.pop(0)

        # Update current solution (may stay the same if no improvement)
        h_val = h_best.copy()

        # Occasionally reset to prevent stagnation
        if time.time() - start_time > 0.9 * budget_s:
            h_val = h_best.copy()
            # Final local optimization
            variable = cp.Variable(n_points)
            objective = cp.Maximize(variable)
            constraints = [
                cp.correlation(variable @ np.ones(n_points), variable) <= evaluate_erdos_solution(h_best),
                cp.sum(variable) == n_points / 2,
                0 <= variable,
                variable <= 1
            ]
            problem = cp.Problem(objective, constraints)
            problem.solve(solver=cp.CVXOPT)
            if problem.status == 'optimal':
                h_val = np.clip(variable.value, 0, 1)

    return (h_best, best_c5, n_points)
```
