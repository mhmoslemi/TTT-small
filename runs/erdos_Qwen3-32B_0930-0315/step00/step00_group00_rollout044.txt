    def grad_ascent_template(h, lr, weight_decay=0.1, verbose=False):
        h_float = h.astype(float)
        for _ in range(10):
            correl = correlate(h_float, 1 - h_float, mode='full') * dx
            idx = np.argmax(correl)
            center = int((idx - np.floor(n_points/2)) - dx**-1 * np.random.uniform(-2, 2))

            # Update only a small window near max overlap
            window = np.clip(h, 0.0, 1.0).copy()
            mid = int(n_points / 2)
            delta = np.zeros_like(h_float)
            delta[center:center+mid] = -np.linspace(-0.5, 0.5, mid)

            approx_gradient = np.gradient(correl, dx)[idx]
            approx_boundary = h_float * (1 - h_float)

            if np.any(approx_boundary > 0.01) and approx_gradient < 0:
                # When function responds linearly, can use bounding volume heuristics
                h_float = np.clip(h_float + lr * delta * approx_boundary, 0, 1)
                h_float /= (np.sum(h_float)) * dx
                h_float *= n_points / 2
            elif approx_gradient < -0.01:
                # Strong nonlinearity, need cautious Newton-style updates
                h_float[center:center+mid] = np.clip(h_float[center:center+mid] + np.max(delta) * 0.05, 0, 1)
                h_float /= (np.sum(h_float)) * dx
                h_float *= n_points / 2

        return h_float

    while time.time() - start_time < budget_s:
        # Clone state for hill-climbing
        h_float1 = grad_ascent_template(h.astype(float).copy(), 0.05, 0.1)
        h_float2 = grad_ascent_template(h.astype(float).copy(), 0.1, 0.05)

        c5_1 = compute_c5(h_float1, dx)
        c5_2 = compute_c5(h_float2, dx)
        c5_3 = compute_c5(h_float3, dx)

        candidates = [(h_float1, c5_1), (h_float2, c5_2), (h_float3, c5_3)]

        # Maintain leader-board solution style
        h_updates = np.stack([c[0] for c in candidates])
        c5_values = np.array([c[1] for c in candidates])

    h *= (n_points / 2.0) / np.sum(h)
    c5 = compute_c5(h, dx, penalty_active=False)
    return h, c5, n_points
