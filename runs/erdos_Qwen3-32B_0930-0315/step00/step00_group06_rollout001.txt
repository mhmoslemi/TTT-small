import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    # Use initial_h_values from parent construction
    h = initial_h_values
    n_points = len(h)
    dx = 2.0 / n_points
    assert np.all(0 <= h) and np.all(h <= 1), "Initial h not in [0, 1]"
    assert np.isclose(np.sum(h * dx), 1), "Initial h integral not unit"

    # Define objective as function of h
    def evaluate(h):
        h1m = 1 - h
        corrs = correlate(h, h1m, mode='full')
        corrs = corrs[n_points-1:]  # Valid shift ranges only (k=1 to n_points-1)
        return np.max(corrs * dx)

    start_time = time.time()
    max_iter = 10000
    iter_count = 0

    # Gradient-based optimization
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1

        # Compute gradient of overlap w.r.t. h[x]
        h1m = 1 - h
        corrs = correlate(h, h1m, mode='full')
        corrs = corrs[n_points-1:]

        idx = np.argmax(corrs)
        k = idx
        assert 1 <= k <= n_points - 1

        # Project gradient: only modify h and h[k:]
        grad = np.zeros(n_points)
        grad[:n_points - k] -= h1m[k:]
        grad[k:] -= h[:n_points - k]
        grad = grad * dx

        # Take one gradient step and project onto feasible set
        alpha = 0.1
        h = h + alpha * grad
        h = np.clip(h, 0, 1)
        # Normalize the integral to be exactly 1
        h *= n_points / (2.0 * np.sum(h)) if np.sum(h) != 0 else 1.0

        # Evaluate current C_5 bound
        current_bound = evaluate(h)

        # Stop if we reach a tight bound (try to reach 0.38085)
        if current_bound <= 0.38087:
            break

    return (h.tolist(), current_bound, n_points)
