import numpy as np
import cvxpy as cp
import time

dx = 2.0 / 160

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 160
    h = np.zeros(n_points)
    plateau_width = 80
    h[n_points // 2 - plateau_width // 2 : n_points // 2 + plateau_width // 2] = 0.5  # Initial uniform assignment

    best_c5 = np.inf
    best_h = h.copy()

    while time.time() - start_time < budget_s * 0.8:
        # Phase 1: Gradient descent
        h_curr = h.copy()
        for _ in range(200):
            if time.time() - start_time >= budget_s * 0.8:
                break
            # Compute the overlap integral
            corr = np.correlate(h_curr, 1 - h_curr, mode="full")
            corr = corr[n_points - 1 : 2 * n_points - 1]  # Focusing on meaningful lags
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_curr.copy()

            # Gradient direction
            grad = np.zeros(n_points)
            for i in range(n_points):
                c_term = 0.0
                for k in range(n_points):
                    if i + k < n_points:
                        c_term += (1 - h_curr[i + k])
                    if k - i >= 0 and k - i < n_points:
                        c_term += -h_curr[k - i]
                grad[i] = c_term
            grad /= (-dx)

            # Gradient descent update
            alpha = 0.1
            proposed_h = h_curr + alpha * grad
            proposed_h = np.clip(proposed_h, 0.0, 1.0)
            proposed_sum = np.sum(proposed_h)
            if proposed_sum > 0:
                proposed_h *= (n_points / 2.0) / proposed_sum

            if np.linalg.norm(proposed_h - h_curr) < 1e-6:
                break
            h_curr = proposed_h

        # Phase 2: CVXPY refinement
        h_refine = cp.Variable(n_points, nonneg=True)
        constraints = [
            cp.sum(h_refine) == n_points / 2.0,
            h_refine <= 1.0,
            h_refine >= 0.0
        ]
        idx = np.argmax(corr)  # Update based on best_c5
        k = idx
        overlap = 0.0
        for i in range(n_points):
            j = (i + k) % n_points
            overlap += h_refine[i] * (1 - h_refine[j])
        objective = cp.Minimize(overlap * dx)
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.OSQP, max_iter=5000)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h = h_refine.value
            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (n_points / 2.0) / h_sum

        # Perturb to escape local minima
        if np.random.rand() < 0.1:
            noise = np.random.uniform(-0.1, 0.1, n_points)
            h += noise
            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            h = h * (n_points / 2.0) / h_sum

    # Final Evaluation
    corr = np.correlate(best_h, 1 - best_h, mode="full")
    corr = corr[n_points - 1 : 2 * n_points - 1]
    best_c5 = np.max(corr) * dx
    return (best_h, best_c5, n_points)
