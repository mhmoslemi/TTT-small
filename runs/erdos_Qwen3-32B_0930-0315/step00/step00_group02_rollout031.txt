import numpy as np
import time
import evaluate_erdos_solution

def compute_overlap_integral(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = corr.max() * dx
    return c5

def project_simplex(v, c, n_points):
    u = np.sort(v)[::-1]
    cumsum = np.cumsum(u)
    l = np.arange(1, n_points + 1)
    rho = np.sum(l * u - c > 0) + 1
    eps = (np.sum(u[:rho]) - c) / rho
    u = np.clip(u - eps, 0, 1)
    u_clip, u_clip_idx = np.clip(u, 0, 1), np.where(u >= 1)[0]
    rescale = c / np.sum(u_clip)
    h = u_clip * rescale

        h = np.clip(h, 0, 1)
        h = h * c / np.sum(h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    initial_values = evaluate_erdos_solution.initial_h_values
    n_points = 160
    dx = 2.0 / n_points

    # Normalize to ensure the sum is exactly n_points / 2
    h = np.array(initial_values[n_points//25:], dtype=float)
    h *= n_points / 2 / np.sum(h)

    best_h = h.copy()
    best_c5 = compute_overlap_integral(h, n_points)
    best_time = time.time()
    last_improvement_time = time.time()

    start_time = time.time()
    it = 0

    while time.time() - start_time < budget_s:
        # Compute full cross-correlation
        corr = np.correlate(h, 1 - h, mode="full")
        kmax_idx = np.argmax(corr)

        # Get the direction of the gradient at kmax
        kmax = kmax_idx - (n_points - 1)
        grad = np.roll(1 - h, -kmax) - np.roll(h, kmax)

        # Barzilai-Borwein step size
        if it > 0:
            s = h - prev_h
            y = h - prev_h - (grad - prev_grad)
            alpha = np.dot(s, s) / np.dot(s, y)
        else:
            alpha = 0.1

        prev_h = h.copy()
        prev_grad = grad.copy()

        # Step in the negative gradient direction
        h = h - alpha * grad
        h = project_simplex(h, n_points / 2, n_points)
        h = np.clip(h, 0, 1)

        # Evaluate progress
        c5_current = compute_overlap_integral(h, n_points)
        elapsed = time.time() - start_time

        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            last_improvement_time = time.time()

        # Perturbation every few iterations
        if it % 500 == 0 and it > 0:
            h = best_h.copy()
            mask = np.random.rand(n_points) < 0.1
            h[mask] += np.random.uniform(-0.1, 0.1, size=mask.sum())
            h = project_simplex(h, n_points / 2, n_points)
            h = np.clip(h, 0, 1)

        it += 1

        # Break if no improvement for a significant time
        if time.time() - last_improvement_time > 60:
            break

    return best_h, best_c5, n_points
