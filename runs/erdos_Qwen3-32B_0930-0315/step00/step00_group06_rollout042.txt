```python
import numpy as np
import time
from scipy.signal import correlate

def run(budget_s=1100):
    n_points = 256
    dx = 2.0 / n_points

    # Start from initial_h_values or a normalized binary pattern
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
    h = h / h.sum() * (n_points / 2)

    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    lr = 0.1

    while time.time() - start_time < budget_s:
        # Normalize h to have exact sum constraint
        h_sum = h.sum()
        if abs(h_sum - n_points / 2) > 1e-6:
            h = h / h_sum * (n_points / 2)

        # Clamp to [0, 1]
        h = np.clip(h, 0, 1)

        # Compute all overlaps
        corr = correlate(h, 1 - h, mode='full') * dx
        c5 = corr.max()
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Find lag with maximum overlap
        k_max = np.argmax(corr)

        # Compute gradient for that lag
        # For positive overlap, we want to decrease h(x) at some positions to reduce overlap
        idx = np.arange(n_points)
        left_idx = np.clip(idx, 0, n_points - 1)
        right_idx = np.clip(idx + k_max, 0, n_points - 1)

        grad = np.zeros_like(h)
        grad[left_idx] -= h[right_idx]
        grad[right_idx] -= h[left_idx]

        # Update h with projection
        h = h - lr * grad / np.max(np.abs(grad))  # Normalize gradient step

        # Perturbation every 100 steps to escape local minima
        if (time.time() - start_time) % 10 == 0:
            perturbation = np.random.randn(n_points) * 0.1 * h.std()
            h += perturbation

        # Reduce learning rate over time
        lr *= 0.999

    return best_h.tolist(), best_c5, n_points
```
