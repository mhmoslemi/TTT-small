import time
import numpy as np
import scipy.signal
import cvxpy as cp
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values as parent_h_values

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    st_time = time.time()

    # State
    h = parent_h_values if parent_h_values is not None else _construct_h(n_points=42)
    n_points = h.size
    dx = 2.0 / n_points
    h_clamped = np.clip(h, 0, 1)
    h = h_clamped / h_clamped.sum() * (n_points / 2)

    # Optimization parameters
    max_outer_loop = 200
    n_refine = 5
    μ_start = 1e3
    μ_decay = 0.5

    def evaluate(h):
        sum_h = h.sum()
        h_clamped = np.clip(h, 0, 1)
        h_scaled = h_clamped / sum_h * (n_points / 2)
        C = scipy.signal.correlate(h_scaled, 1 - h_scaled, mode="full")
        return (np.max(C) * dx), h_scaled

    C5, h = evaluate(h)
    best_C5 = C5
    best_h = h

    for outer in range(max_outer_loop):
        if time.time() - st_time > budget_s:
            break

        def objective(h_cp):
            c = scipy.signal.correlate(h_cp, 1 - h_cp, mode="full") * dx
            C = cp.max(c)
            sum_h = cp.sum(h_cp)
            penalty = cp.log(h_cp).sum() + cp.log((1 - h_cp)).sum() + cp.log(sum_h - 1e-8)
            obj = C + μ_start * penalty
            return obj

        h_cp = cp.Variable(n_points)
        constraints = [
            0 <= h_cp,
            h_cp <= 1,
            cp.sum(h_cp) >= 1 - 1e-8,
        ]
        problem = cp.Problem(cp.Minimize(objective(h_cp)), constraints)
        problem.solve(solver=cp.CVXOPT, verbose=False, kktsolver="robust")

        if problem.status == "optimal":
            h_prop = h_cp.value
            c_prop, h_prop = evaluate(h_prop)
            if c_prop < best_C5:
                best_C5 = c_prop
                best_h = h_prop

        if outer < 50:
            μ_start *= μ_decay

    for i in range(n_refine):
        if time.time() - st_time > budget_s:
            break

        perturb = np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(best_h + perturb, 0, 1)
        h_perturbed = h_perturbed / h_perturbed.sum() * (n_points / 2)
        c_perturbed, h_perturbed = evaluate(h_perturbed)
        if c_perturbed < best_C5:
            best_C5 = c_perturbed
            best_h = h_perturbed

    return best_h, best_C5, n_points

def _construct_h(n_points):
    h = np.zeros(n_points)
    center = int(n_points / 2)
    midpoint = int(n_points / 4)
    h[max(0, center - midpoint): min(n_points, center + midpoint)] = 0.0
    h[center - midpoint // 2:center + midpoint // 2] = 1.0
    h = h / h.sum() * (n_points / 2)
    return h
