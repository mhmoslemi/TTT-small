import numpy as np
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    overlaps = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlaps)

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 80
        h = np.full(n_points, 0.5, dtype=np.float64)

    dx = 2.0 / n_points

    # Enforce initial constraint sum(h) == n_points / 2
    h *= (n_points / 2.0) / np.sum(h)

    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)

    while time.time() - start_time < budget_s:
        # Compute current overlaps
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        worst_k = int(np.argmax(overlaps))
        worst_k_offset = abs(worst_k - h.shape[0])

        # Build shifted h
        if worst_k < h.shape[0]:
            h_warped = np.roll(h, -worst_k)
        else:
            h_warped = np.roll(h, h.shape[0] - worst_k)

        # Formulate convex problem
        H = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(H * (1 - h_warped) * dx))
        constraints = [
            H >= 0,
            H <= 1,
            cp.sum(H) == n_points / 2.0
        ]

        prob = cp.Problem(objective, constraints)

        try:
            prob.solve(solver=cp.MOSEK, verbose=False, max_iter=100, eps=1e-6)
            h_candidate = prob.solution.primal_vars[0]
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False, max_iters=100, abstol=1e-6, reltol=1e-6)
                h_candidate = prob.solution.primal_vars[0]
            except Exception:
                # Skip bad solve
                continue

        # Project h_candidate into valid space
        h_candidate[h_candidate < 0] = 0
        h_candidate[h_candidate > 1] = 1
        h_candidate *= (n_points / 2.0) / np.sum(h_candidate)

        # Evaluate
        c5 = evaluate_c5(h_candidate, dx)

        # Track best solution
        if c5 < best_c5:
            best_h[:] = h_candidate
            best_c5 = c5
            print(f"Improved C5 bound: {best_c5:.7f} after {time.time()-start_time:.1f}s")

        # Early stopping if bound achieved
        if best_c5 < 0.380876:
            break

        # Stop if no improvement in 30 seconds
        if any(timer > 30 for timer in [time.time() - start_time]):
            break

        # Update h for next iteration
        h[:] = h_candidate

    # Final verification
    dx = 2.0 / len(best_h)
    best_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), best_c5, len(best_h))
