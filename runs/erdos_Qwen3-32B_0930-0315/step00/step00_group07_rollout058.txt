```python
import numpy as np
import cvxpy as cp
import time
import sys

def evaluate_erdos_solution(h_values):
    """Preimported evaluation wrapper"""
    n = len(h_values)
    dx = 2.0 / n
    corr = np.correlate(h_values, 1 - h_values, mode="full") * dx
    c5_bound = np.max(corr)
    return c5_bound

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    # Use provided initial_h_values if available
    global initial_h_values
    n = len(initial_h_values)
    h = np.copy(initial_h_values)
    h = h.astype(np.float64)

    # Normalize the exact sum(h) to n/2 for feasibility
    target_sum = n / 2
    h *= (target_sum / np.sum(h))

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    dx = 2.0 / n

    # LP parameters
    eps = 1e-6
    max_iter = 2000
    iter_print = 20  # For early convergence measure

    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 0.5:
            break

        lag_values = np.correlate(h, 1 - h, mode="full") * dx
        worst_lag = int(np.argmax(lag_values))
        C_k = lag_values[worst_lag]

        # Compute gradient at worst lag k
        grad = np.zeros(n)
        k = worst_lag
        for i in range(n):
            j1 = (i + k) % n
            j2 = (i - k) % n
            grad[i] = 1.0 - h[j1] - h[j2]

        # Normalize gradient
        grad = grad / np.linalg.norm(grad)

        # Create LP variables
        delta = cp.Variable(n)
        constraints = [
            h + delta >= 0.0 - eps,
            h + delta <= 1.0 + eps,
            cp.sum(delta) == 0.0,
        ]
        objective = cp.Maximize(cp.sum(cp.multiply(grad, delta)))

        # Define and solve optimization
        problem = cp.Problem(objective, constraints)

        try:
            problem.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            problem.solve(solver=cp.GUROBI, verbose=False)

        # Retrieve LP step
        if problem.status not in ["optimal", "optimal_inaccurate"]:
            continue  # Skip if LP fails

        step = delta.value

        # Line search to improve stability
        alpha = 0.2
        while True:
            h_attempt = h + alpha * step
            if any(h_attempt < -eps * 2) or any(h_attempt > 1.0 + eps * 2):
                alpha *= 0.5
                if alpha < 1e-5:
                    break
                continue
            h_attempt = np.clip(h_attempt, 0.0, 1.0)
            # Enforce mass constraint
            h_attempt *= (target_sum / np.sum(h_attempt))
            c5 = evaluate_erdos_solution(h_attempt)
            if c5 < best_c5:
                h[:] = h_attempt
                if c5 < best_c5:
                    best_c5 = c5
                    best_h[:] = h.copy()
                break
            alpha *= 0.5
            if alpha < 1e-5:
                break

        # Print progress
        if iter % iter_print == 0 and iter > 0:
            print(f"Iter {iter} / {max_iter} | Best C₅: {best_c5:.6f} | time: {time.time() - start_time:.1f}", flush=True)

    # Final check
    h = best_h
    h = np.clip(h, 0.0, 1.0)
    h *= (target_sum / np.sum(h))
    c5_bound = evaluate_erdos_solution(h)

    return (h, c5_bound, n)
```
