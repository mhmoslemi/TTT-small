import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points

    # Ensure constraint ∑h = n_points / 2
    h_sum = np.sum(h)
    correction = (n_points / 2 - h_sum) / n_points
    h = h + correction

    def objective(h_flat):
        h = h_flat.reshape(-1)
        cor = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(cor)

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    def constraint_bounds(h_flat):
        return np.array([0 <= h_flat, h_flat <= 1]).all()

    bounds = [(0, 1)] * n_points
    constraints = [{'type': 'eq', 'fun': lambda h_flat: constraint_sum(h_flat)}]
    
    result = minimize(
        objective,
        h,
        bounds=bounds,
        constraints=constraints,
        method='L-BFGS-B',
        options={'ftol': 1e-8, 'maxiter': 10000}
    )

    elapsed = time.time() - start_time
    remaining = max(0, budget_s - elapsed)

    # Evaluate C₅
    h_values = result.x
    h_values = np.clip(h_values, 0, 1)
    h_values = h_values / np.sum(h_values) * (n_points / 2)
    
    cor = np.correlate(h_values, 1 - h_values, mode='full') * dx
    c5_bound = np.max(cor)

    return h_values, c5_bound, n_points
