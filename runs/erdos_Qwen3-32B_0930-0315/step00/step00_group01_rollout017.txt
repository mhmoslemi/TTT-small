import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    h1 = 1 - h
    corr = correlate(h, h1, mode='full')
    c5 = (dx * corr).max()
    return c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 47  # reasonable default
        h = np.zeros(n_points)
        n_ones = int(n_points / 2)
        h[:n_ones] = 1.0
        np.random.shuffle(h)

    target_mass = n_points / 2.0

    # Gradient descent hyperparameters
    step_size = 0.01
    max_iter = 30000
    proj_factor = 1.05  # rescaling factor for projection
    iter_without_improvement_limit = 1000
    tol = 1e-6

    best_c5 = np.inf
    best_h = h.copy()
    best_time = time.time()

    all_h = []

    for it in range(max_iter):
        if time.time() - start > budget_s - 1:
            break

        h_prev = h.copy()
        c5_prev = evaluate_erdos_solution(h, n_points)

        # Compute finite difference of the gradient
        grad = np.zeros_like(h)
        for i in range(n_points):
            eps = 1e-5
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus_clipped = np.clip(h_plus, 0, 1)
            h_plus_clipped *= proj_factor * target_mass / h_plus_clipped.sum()

            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus_clipped = np.clip(h_minus, 0, 1)
            h_minus_clipped *= proj_factor * target_mass / h_minus_clipped.sum()

            delta_plus = evaluate_erdos_solution(h_plus_clipped, n_points) - c5_prev
            delta_minus = evaluate_erdos_solution(h_minus_clipped, n_points) - c5_prev
            grad[i] = (delta_plus - delta_minus) / (2 * eps)

        # Projected gradient step
        h -= step_size * grad
        h = np.clip(h, 0, 1)
        h = h * (target_mass / h.sum())

        # Evaluate current C5
        c5 = evaluate_erdos_solution(h, n_points)

        # Track and project if no improvement
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_time = time.time()
        elif it - best_time > iter_without_improvement_limit:
            # Revert to best solution and reduce step size
            h = best_h.copy()
            step_size *= 0.95

    h_values = best_h.tolist()
    c5_bound = best_c5
    return h_values, c5_bound, n_points
