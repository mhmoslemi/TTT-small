import numpy as np
import scipy.optimize as opt
import time


def evaluate(h):
    n_points = len(h)
    dx = 2.0 / n_points
    corrs = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corrs)


def project_constraints(h, total_mass):
    h = np.clip(h, 0, 1)
    h = h * (total_mass / np.sum(h))
    return h


def objective(h, total_mass):
    h = h.reshape(-1)
    h = project_constraints(h, total_mass)
    return evaluate(h)

def run(seed=42, budget_s=1100):
    start_time = time.time()
