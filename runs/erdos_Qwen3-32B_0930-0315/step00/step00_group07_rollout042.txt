import numpy as np
import cvxpy as cp
import time
import math

def compute_overlaps(h, dx):
    n = len(h)
    k_range = np.arange(1, n)
    overlaps = np.zeros_like(k_range, dtype=np.float64)

    for i, k in enumerate(k_range):
        c = h * (1 - np.roll(h, k))
        overlaps[i] = np.sum(c * dx)

    return overlaps

def correct_overlap_lp(h, dx, k, epsilon=1e-6):
    n = len(h)
    h_var = cp.Variable(n)
    ones = np.ones(n)
    constraints = [
        0 <= h_var,
        h_var <= 1,
        cp.sum(h_var) == cp.sum(h),
    ]

    # Construct gradient
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = dx * (1 - h[(i + k) % n] - h[(i - k) % n])
    obj = cp.Minimize(cp.sum(cp.multiply(grad, h_var)))

    prob = cp.Problem(obj, constraints)
    result = prob.solve(solver=cp.ECOS, max_iters=500, abstol=1e-10, reltol=1e-10, feastol=1e-10)
    return h_var.value

def run(seed=42, budget_s=1100, n_points=80):
    np.random.seed(seed)
    dx = 2.0 / n_points

    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=int(n_points / 2), replace=False)
        h[indices] = 1.0

    h = np.clip(h, 0.0, 1.0)
    h = h / h.sum() * (n_points / 2)
    h = np.clip(h, 0.0, 1.0)

    h_best = h.copy()
    c5_best = np.inf
    start = time.time()
    max_iter = 500
    iter_count = 0

    while time.time() - start < budget_s and iter_count < max_iter:
        overlaps = compute_overlaps(h, dx)
        worst_k = np.argmax(overlaps)
        worst_overlap = overlaps[worst_k]

        # Update only if improving
        if worst_overlap < c5_best - 1e-6:
            c5_best = worst_overlap
            h_best = h.copy()

        # Coordinate descent step: minimize overlap for worst_k
        delta_h = correct_overlap_lp(h, dx, worst_k + 1)

        # Line search and validation
        h_new = h + delta_h - h
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = (n_points / 2 / h_new.sum()) * h_new
        h_new = np.clip(h_new, 0.0, 1.0)

        if time.time() - start < budget_s:
            overlaps_new = compute_overlaps(h_new, dx)
            new_max_overlap = np.max(overlaps_new)
            if new_max_overlap < worst_overlap:
                h = h_new
        else:
            break

        iter_count += 1

    h_best = h_best / h_best.sum() * (n_points / 2)
    h_best = np.clip(h_best, 0.0, 1.0)
    overlaps = compute_overlaps(h_best, dx)
    final_c5 = np.max(overlaps)

    return h_best.tolist(), final_c5, n_points
