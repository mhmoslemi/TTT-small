import numpy as np
import time
import cvxpy as cp
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    initial_n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = initial_n_points
    dx = 2.0 / n_points
    h *= n_points / (2.0 * np.sum(h))  # Normalize to total integral 1

    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    iter_count = 0
    no_improve_count = 0
    iter_step = 50
    perturbations_needed = 0

    # Gradient descent with Barzilai-Borwein step size and projection
    while time.time() - start_time < budget_s:
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode="full")
        k_max = int(np.abs(np.argmin(corr) - n_points)) % n_points

        h1 = h
        h2 = h
        grad = np.zeros_like(h)
        for i in range(n_points):
            j1 = (i + k_max) % n_points
            j2 = (i - k_max) % n_points
            grad[i] = -(1 - h[j1]) + h[j2]  # Gradient of negative C5 at k_max

        # Barzilai-Borwein step size
        if iter_count > 0:
            diff_h = h - prev_h
            diff_grad = grad - prev_grad
            alpha = np.dot(diff_h, diff_h) / np.dot(diff_h, diff_grad)
        else:
            alpha = 0.01

        prev_h = h.copy()
        prev_grad = grad.copy()

        # Step and project
        h -= alpha * grad
        h = np.clip(h, 0, 1)  # Clip to [0, 1]
        h *= n_points / 2.0 / np.sum(h)  # Project to sum = n / 2

        # Evaluate current C5
        c5 = evaluate_erdos_solution(h)
        iter_count += 1

        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Add perturbation after no improvement
        if no_improve_count > 500:
            perturb_mask = np.random.rand(n_points) < 0.1  # Perturb 10%
            h[perturb_mask] = np.random.random(np.sum(perturb_mask))
            h = np.clip(h, 0, 1)
            h *= n_points / 2.0 / np.sum(h)  # Re-project
            no_improve_count = 0

    return best_h, best_c5, n_points
