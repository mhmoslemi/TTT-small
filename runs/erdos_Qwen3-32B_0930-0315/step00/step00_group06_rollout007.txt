import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    h1 = np.concatenate([h, h])
    h2 = 1 - np.concatenate([h, h])
    overlap = correlate(h1, h2, mode="full") * dx
    valid_overlap = overlap[n_points:-n_points]
    return np.max(valid_overlap)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    max_time = budget_s

    # Use pre-imported initial_h_values
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    # Normalize sum constraint
    h /= (2.0 / n_points * h.sum()) / 1.0
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    best_time = 0.0

    step = 0
    no_improvement = 0
    max_no_improvement = 100
    learning_rate = 0.01
    decay_rate = 0.995
    perturb_rate = 0.01
    n_perturb = 5
    update_interval = 10

    while time.time() - start_time < max_time:
        step += 1

        # Compute overlaps
        h1 = np.concatenate([h, h])
        h2 = 1 - np.concatenate([h, h])
        overlap = correlate(h1, h2, mode="full") * dx
        valid_overlap = overlap[n_points:-n_points]
        k_max = np.argmax(valid_overlap)
        max_c5 = valid_overlap[k_max]

        # Gradient with respect to h
        h_tiled = np.tile(h, 2)
        grad = np.zeros(n_points)

        for i in range(n_points):
            grad[i] = (h[i] * (1 - h[(i + k_max) % (2 * n_points)])) * dx

        grad[k_max:] += (h[k_max:] * (1 - h[:n_points - k_max])) * dx
        grad[:k_max] += (h[:k_max] * (1 - h[n_points - k_max:])) * dx

        # Update h
        h = cp.Variable(n_points)
        obj = cp.Minimize((h - (best_h - learning_rate * grad))**2).sum()
        constraints = [0 <= h, h <= 1, h.sum() == n_points / 2]
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS)

        if prob.status == cp.OPTIMAL:
            h = h.value
        else:
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            h = cp.Variable(n_points)
            obj = cp.Minimize((h - h)**2).sum()
            constraints = [0 <= h, h <= 1, h.sum() == n_points / 2]
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.ECOS)
            h = h.value

        current_c5 = evaluate_erdos_solution(h, n_points)

        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1

        # Perturb with small noise if no improvement
        if no_improvement >= max_no_improvement:
            h = best_h + perturb_rate * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = cp.Variable(n_points)
            obj = cp.Minimize((h - h)**2).sum()
            constraints = [0 <= h, h <= 1, h.sum() == n_points / 2]
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.ECOS)
            h = h.value
            no_improvement = 0

        # Schedule perturbation at intervals
        if step % n_perturb == 0:
            h = best_h + perturb_rate * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = cp.Variable(n_points)
            obj = cp.Minimize((h - h)**2).sum()
            constraints = [0 <= h, h <= 1, h.sum() == n_points / 2]
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.ECOS)
            h = h.value

        # Learning rate decay
        learning_rate *= decay_rate

    return (best_h, best_c5, n_points)
