import numpy as np
import cvxpy as cp
import time
from functools import partial

def repair_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h = h / h.sum() * (n_points / 2)
    return h

def compute_overlap(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def simulated_annealing(h, n_points, learning_rate=0.01, init_temp=1.0, alpha=0.97, steps=100):
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_overlap(h, dx)
    
    T = init_temp
    for _ in range(steps):
        h_candidate = h + np.random.normal(0, T, n_points)
        h_candidate = repair_feasibility(h_candidate, n_points)
        c5_candidate = compute_overlap(h_candidate, dx)

        if c5_candidate < best_c5 or np.random.rand() < np.exp((best_c5 - c5_candidate) / T):
            h = h_candidate
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate

        T *= alpha
        if T < 1e-6:
            break
    return best_h, best_c5

def gradient_descent(h, n_points, learning_rate=0.01, steps=100):
    dx = 2.0 / n_points
    for _ in range(steps):
        corr = np.correlate(h, 1-h, mode='full')
        k_max = np.argmax(corr)
        shift = n_points - 1 - k_max # Negative shift is rolling to the right
        gradient = (1 - np.roll(h, shift))[:n_points]
        h -= learning_rate * gradient
        h = repair_feasibility(h, n_points)
    return h

def refine_resolution(h_coarse):
    from scipy.interpolate import CubicSpline
    n_points = h_coarse.size * 2
    x_coarse = np.linspace(0, 2, h_coarse.size)
    x_fine = np.linspace(0, 2, n_points)
    cs = CubicSpline(x_coarse, h_coarse)
    h_fine = cs(x_fine)
    h_fine = repair_feasibility(h_fine, n_points)
    return h_fine

def optimize(h_initial, n_points, budget_s):
    start_time = time.time()
    best_h = h_initial
    best_c5 = compute_overlap(h_initial, 2.0 / n_points)
    
    while time.time() - start_time < budget_s:
        # Simulated Annealing
        h, c5 = simulated_annealing(best_h, n_points, steps=100)
        best_h, best_c5 = (h, c5) if c5 < best_c5 else (best_h, best_c5)
        
        # Gradient Descent
        h, c5 = gradient_descent(best_h, n_points, steps=100)
        best_h, best_c5 = (h, c5) if c5 < best_c5 else (best_h, best_c5)

        # Refinement and Budget Check
        if n_points * 2 <= 64 and time.time() - start_time + 400 < budget_s:
            h = refine_resolution(best_h)
            best_h, best_c5 = h, compute_overlap(h, 2.0 / len(h))
            n_points = len(h)

    return best_h, best_c5, n_points

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    
    # Use initial_h_values if provided
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = np.copy(initial_h_values)
    else:
        # Start with n=14 sample points on [0, 2]
        n_points = 14
        h = np.random.uniform(0, 1, n_points)
        h = repair_feasibility(h, n_points)

    h_opt, c5_bound, n_points = optimize(h, n_points, budget_s)
    return h_opt, c5_bound, n_points
