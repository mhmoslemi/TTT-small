import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 47  # starting resolution

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points // 2] = 1.0

    h = h / h.sum() * n_points / 2  # normalize to satisfy sum constraint
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    dx = 2.0 / n_points
    learning_rate = 0.02
    decay_rate = 0.9999

    while time.time() - start_time < budget_s:
        overlaps = np.correlate(h, 1 - h, mode="full")
        overlaps = overlaps * dx
        max_overlap_idx = np.argmax(overlaps)
        k = max_overlap_idx
        overlap = overlaps[k]
        if overlap < best_c5:
            best_c5 = overlap
            best_h = h.copy()

        grad = np.zeros_like(h)
        for i in range(n_points - k):
            grad[i] -= h[i + k]
            grad[i + k] += (1 - h[i])

        grad = grad / n_points  # normalize by dx

        h -= learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        h = h / h.sum() * n_points / 2  # maintain integral constraint

        learning_rate *= decay_rate

    return best_h.tolist(), best_c5, n_points
