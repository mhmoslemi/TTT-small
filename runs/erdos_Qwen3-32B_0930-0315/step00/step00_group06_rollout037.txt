import numpy as np
import time
from numpy import correlate
import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.copy(initial_h_values).astype(float)

    max_iter = 1000000
    lr_mu = 0.5  # Initial learning rate
    lr_sigma = 0.05

    last_best = np.inf
    no_progress = 0
    start_time = time.time()
    budget_limit = start_time + budget_s

    def compute_c5(cost_h):
        cost = correlate(cost_h, 1 - cost_h, mode='full')
        cost = cost * dx
        return np.max(cost)

    def project_sum(h):
        current_sum = h.sum()
        target_sum = n_points / 2
        if current_sum == 0:
            return h
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        return h

    iteration = 0
    while time.time() < budget_limit and iteration < max_iter:
        iteration += 1
        old_h = h.copy()
        corr = correlate(h, 1 - h, mode='full')
        corr = corr * dx
        k_max = np.argmax(corr)

        left = k_max
        right = len(h) - 1 - k_max
        grad = np.zeros_like(h)
        grad[:left] += (1 - h[left:]) * dx
        grad[left:] -= h[:right] * dx

        lr = lr_mu + lr_sigma * np.random.randn()
        h += lr * grad
        h = project_sum(h)

        c5 = compute_c5(h)
        if c5 < last_best:
            last_best = c5
            no_progress = 0
        else:
            no_progress += 1

        if no_progress >= 100:
            # Perturb to escape local optima
            h = h + np.random.randn(*h.shape) * 0.05
            h = project_sum(h)
            no_progress = 0

    return h, last_best, n_points
