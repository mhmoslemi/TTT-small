import numpy as np
from scipy.ndimage import correlate
import time

def evaluate_erdos(h, dx):
    ones = np.ones_like(h)
    correl = correlate(h, 1 - h, mode='full')
    correl = correl * dx
    return np.max(correl)

def project(h, dx, constraint):
    avg = np.sum(h) * dx
    h_clipped = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h_clipped)
    scaling = constraint / current_sum
    h_scaled = h_clipped * scaling
    return h_scaled

def compute_gradient(h, dx, k_max):
    n = len(h)
    grad = np.zeros_like(h)
    for i in range(n):
        if i + k_max < n:
            grad[i] += (1 - h[i + k_max]) * dx
        if i - k_max >= 0:
            grad[i] -= h[i - k_max] * dx
    return grad

def run(budget_s=1100):
    n_points = 70
    dx = 2.0 / n_points
    constraint = n_points / 2.0

    h = initial_h_values.copy()
    h = project(h, dx, constraint)

    start_time = time.time()
    alpha = 0.1
    C5 = evaluate_erdos(h, dx)
    best_h = h.copy()
    best_C5 = C5

    while time.time() - start_time < budget_s:
        correl = correlate(h, 1 - h, mode='full') * dx
        lag_indices = np.arange(-n_points + 1, n_points)
        valid_lags = lag_indices[(correl.shape[0] + 1) // 2:]
        valid_correl = correl[(correl.shape[0] + 1) // 2:]
        k_max_idx = np.argmax(valid_correl)
        k_max = valid_lags[k_max_idx]

        grad = compute_gradient(h, dx, k_max)
        h -= alpha * grad
        h = project(h, dx, constraint)

        C5 = evaluate_erdos(h, dx)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            alpha = max(0.1, alpha * 1.1)
        else:
            alpha = max(0.01, alpha * 0.95)

        if np.random.rand() < 0.1 or (time.time() - start_time) + 60 > budget_s:
            perturbation = np.random.normal(0, 0.01, size=n_points)
            h = project(h + perturbation, dx, constraint)

        if (time.time() - start_time) % 10 < 1:
            h = project(h, dx, constraint)

    return best_h, best_C5, n_points
