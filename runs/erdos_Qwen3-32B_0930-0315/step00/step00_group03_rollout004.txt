import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy().reshape(-1)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    best_h = h.copy()
    best_c5 = np.inf
    
    while time.time() - start_time < budget_s:
        # Compute overlaps for all shifts
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        k_list = np.arange(n_points)
        full_overlaps = []
        for k in k_list:
            ij_off = k % n_points
            idxs1 = np.arange(n_points)
            idxs2 = (np.arange(n_points) + ij_off) % n_points
            overlap = np.sum(h[idxs1] * (1 - h[idxs2])) * dx
            full_overlaps.append(overlap)
        full_overlaps = np.array(full_overlaps)
        current_c5 = np.max(full_overlaps)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            # print(f"Found: {best_c5:.6f}, t={time.time()-start_time:.1f}s")
        
        # Identify most problematic shift
        worst_k = np.argmax(full_overlaps)
        ij_off = worst_k % n_points
        idxs1 = np.arange(n_points)
        idxs2 = (np.arange(n_points) + ij_off) % n_points
        active = h[idxs1] * (1 - h[idxs2])
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            for j in range(n_points):
                if (j + ij_off) % n_points == i:
                    grad[i] += (1 - h[j])
                if (i + ij_off) % n_points == j:
                    grad[i] -= h[j]
        # Normalize gradient
        grad = grad - (2 * h - 1)
        learning_rate = 0.05 / (1 + (time.time() - start_time) / 1000)
        h_proposal = h - learning_rate * grad
        
        # Projection onto feasible region
        h = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum_squares(h - h_proposal))
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1
        ]
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.GUROBI if hasattr(cp, "GUROBI") else cp.OSQP)
        h = np.clip(np.array(h.value).flatten(), 0.0, 1.0)

        # Early stopping if no progress
        if time.time() - start_time + 10 > budget_s:
            break
    
    return (best_h.tolist(), best_c5, n_points)
