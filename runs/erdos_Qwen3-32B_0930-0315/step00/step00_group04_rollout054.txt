import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(budget_s=1100, seed=42, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    
    h_init = initial_h_values
    n_points = h_init.size
    dx = 2 / n_points
    sum_h = n_points / 2

    # Standardize variables
    h_var = cp.Variable(n_points, nonneg=True)

    # Objective weights
    mu = 1e4

    # Reformulate the objective with barrier
    def max_correlation(h):
        correlation = correlate(h, 1 - h, mode='full') * dx
        return cp.Maximize(-cp.min(correlation))

    def barrier_objective(h):
        obj = max_correlation(h)
        barrier_term = mu * cp.sum(cp.log(h) + cp.log(1 - h) + cp.log(sum_h - cp.sum(h) * dx))
        return obj + barrier_term

    # Optimization step within barrier method
    def optimize_step(h_current, mu_val):
        constraints = [
            cp.sum(h_current) * dx == 1,
            h_current >= 1e-6,
            h_current <= 1 - 1e-6,
        ]
        prob = cp.Problem(obj=barrier_objective(h_current), constraints=constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h_current
        h_opt = h_current.value
        h_opt = np.clip(h_opt, 0, 1)  # Force feasibility
        h_opt /= np.sum(h_opt) * dx
        h_opt *= sum_h / (np.sum(h_opt) * dx)
        return h_opt

    h_current = h_init.copy()
    best_h = h_current.copy()
    best_c5 = float('inf')

    it = 0
    mu_decrease_rate = 0.1
    mu_decay_period = 50

    while True:
        elapsed = time.time() - t_start
        if elapsed >= budget_s:
            break

        if it % 50 == 0:
            print(f"Iter {it}, time used {elapsed:.2f}s")
        h_current = optimize_step(h_current, mu)

        if isinstance(h_current, np.ndarray):
            c5 = evaluate_erdos_solution(h_current)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_current
        mu *= (1 - mu_decrease_rate)
        mu = max(mu, 1e-8)
        it += 1

    best_h = np.clip(best_h, 0, 1)
    best_h /= np.sum(best_h) * dx
    best_h *= sum_h / (np.sum(best_h) * dx)
    return (best_h, best_c5, n_points)
