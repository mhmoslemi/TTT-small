import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

# Evaluations
def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    h_correlated = correlate(h, 1 - h, mode="full")
    max_overlap = (h_correlated * dx).max()
    return max_overlap

# Projection for constraints
def project_h(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    total = h_clipped.sum()
    if abs(total) < 1e-8 or abs(h_clipped.shape[0]) < 1:
        return h_clipped
    scale_factor = (n_points / 2.0) / total
    h_scaled = h_clipped * scale_factor
    return np.clip(h_scaled, 0, 1)

# Run the optimization loop
def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Parameters
    n_points = 70

    # Use provided initial guess
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h, n_points)

    start_time = time.time()
    it = 0
    patience = 0
    patience_limit = 10  # Fallback if no improvement for patience_limit iterations

    while time.time() - start_time < budget_s - 0.1 and patience < patience_limit:
        it += 1
        h = project_h(h, n_points)

        h_var = cp.Variable(n_points)
        dx = 2.0 / n_points
        constr = [0 <= h_var, h_var <= 1, h_var.sum() == n_points / 2]

        # Compute the cross-correlation for (1-h) at all lags
        h_vals = h
        ones_minus_h = 1 - h_vals

        corr = -correlate(h_vals, ones_minus_h, mode="full")
        argmax_corr = np.argmax(corr)
        lag = argmax_corr

        # We only optimize with respect to one of the overlaps
        # That overlaps at lag k +=0

        # Build a linear objective to minimize a single convex component of C5
        buff = np.zeros(n_points + n_points - 1)
        buff[lag] = 1
        a = correlate(buff, ones_minus_h, mode="full")
        a_valid = a[0:n_points] * 2 / n_points  # scale by 2 / n_points for dx
        obj = cp.Maximize(a_valid @ h_var)

        prob = cp.Problem(obj, constr)
        prob.solve(solver=cp.GUROBI, verbose=False)
        h = np.clip(h_var.value, 0, 1)

        c5 = evaluate_c5(h, n_points)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

    return best_h, best_c5, n_points
