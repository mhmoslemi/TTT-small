import numpy as np
import time
import cvxpy as cp
from evaluate_erdos_solution import evaluate_erdos_solution


def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    t0 = time.time()

    # Load initial solution
    h0 = initial_h_values
    h = h0.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Validate initialization
    if not np.all((h >= 0) & (h <= 1)) or abs(np.sum(h) - target_sum) > 1e-6:
        raise ValueError("Initial h is invalid")

    best_h = h.copy()
    best_c5 = None
    last_improvement_time = 0
    learning_rate = 0.05
    lr_decay_factor = 0.995
    iterations = 0

    def compute_objective(h):
        # Compute max overlap
        c5 = evaluate_erdos_solution(h)[0]
        return c5

    while time.time() - t0 < budget_s:
        iterations += 1

        h_float = h.astype(np.float64)

        # Compute the overlap values
        corr = np.correlate(h_float, 1 - h_float, mode="full")
        c5_values = corr * dx
        curr_c5 = np.max(c5_values)

        if best_c5 is None or curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h[:] = h_float[:]
            last_improvement_time = time.time()

        # Early stopping if no improvement
        if time.time() - last_improvement_time > 300:
            break

        # Find max overlap lag
        max_idx = np.argmax(c5_values)
        k = max_idx - (n_points - 1)

        # Construct gradient
        grad = np.zeros(n_points)

        for i in range(n_points):
            if 0 <= (i + k) < n_points:
                grad[i] += (1 - h[i + k]) - 2 * h[i] * (1 - h[i + k])
            if -n_points + 1 < (i - k) < n_points:
                grad[i] += h[i - k] - 2 * h[i] * h[i - k]

        grad *= dx

        # Update h with learning rate
        h_float -= learning_rate * grad

        # Clamp to [0, 1]
        h_float = np.clip(h_float, 0, 1)

        # Project to validity constraint sum(h) == n_points / 2
        h_float = project_to_simplex(h_float, target_sum)

        h = h_float

        # Anneal learning rate
        if iterations % 1000 == 0:
            learning_rate *= lr_decay_factor

    best_c5 = compute_objective(best_h)
    return best_h, best_c5, n_points


def project_to_simplex(v, z=1.0):
    # Project vector v to the simplex: {x | x >= 0, sum x_i = z}
    n = len(v)
    v[v < 0] = 0
    u = np.sort(v)[::-1]
    cumsums = np.cumsum(u)
    rho = np.argmax(u.cumsum() - 1 > (z - u)) + 1
    theta = (cumsums[rho - 1] - z) / rho if rho > 0 else 0.0
    return np.maximum(v - theta, 0)
