        h1 = h

        k_max = np.argmax(corr)

        # Gradient descent step
        h -= base_lr * grad

        # Projection onto domain [0, 1]
        h = np.clip(h, 0, 1)

        # Normalize to satisfy the sum(h) == n_points / 2 constraint
        total = np.sum(h)
        h *= (n_points / 2) / total

        if iter_count % perturb_interval == 0:
