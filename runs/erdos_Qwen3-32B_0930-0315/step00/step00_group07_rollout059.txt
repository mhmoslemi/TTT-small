import numpy as np
import cvxpy as cp
import time
from evaluate_erdos_solution import evaluate_erdos_solution


def run(seed=42, budget_s=1100, n_points=80):
    np.random.seed(seed)
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, int(n_points / 2), replace=False)
        h[indices] = 1.0
    h = h.copy()

    best_h = h.copy()
    best_c5 = float("inf")

    start_time = time.time()
    max_iterations = 300

    for iteration in range(max_iterations):
        # Compute current C₅
        h_eval = h if h.shape == (n_points,) else h[0]
        overlaps = np.correlate(h_eval, 1.0 - h_eval, mode="full")
        overlaps = overlaps * dx

        worst_lag_ind = np.array([i for i in range(len(overlaps))
                                  if i >= h.shape[0] - 1 and i <= len(overlaps) - h.shape[0]])

        if len(worst_lag_ind) > 0:
            relevant_overlaps = overlaps[worst_lag_ind]
            idx_flat = worst_lag_ind[np.argmax(relevant_overlaps)]
            k = idx_flat - (len(h) - 1)
        else:
            k = 3

        h_array = h.copy()
        correlated = np.correlate(h_array, 1.0 - h_array, mode="full")
        current_c5 = np.max(correlated * dx)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            print(f"Iteration {iteration}, C5: {best_c5}")

        # Gradient w.r.t. h for target lag k
        idxs = np.arange(h.shape[0])
        locs_k = (idxs + k) % h.shape[0]
        gradient = -dx * (1.0 - 2 * h[locs_k])

        # Variables
        delta_h = cp.Variable(n_points)

        # Objective
        obj = cp.Minimize(cp.sum(cp.multiply(gradient, delta_h)))

        # Constraints
        constraints = [
            delta_h >= -h,
            delta_h <= 1 - h,
            cp.sum(delta_h) == 0
        ]

        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver=cp.ECOS)

        h += delta_h.value
        h = np.clip(h, 0, 1)

        if time.time() - start_time > budget_s:
            print("Time budget exceeded. Returning best solution found.")
            break

    h = best_h
    c5_bound = evaluate_erdos_solution(h)  # Ensure correct evaluation
    return h, c5_bound, n_points
