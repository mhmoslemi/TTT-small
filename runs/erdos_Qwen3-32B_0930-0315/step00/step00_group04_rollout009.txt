import numpy as np
import time
from scipy.ndimage import zoom

def compute_overlap(h, dx):
    C = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(C)

def repair_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h = h / h.sum() * (n_points / 2)
    return h

def compute_gradient(h, dx):
    C = np.correlate(h, 1 - h, mode='full') * dx
    k_max = np.argmax(C)
    shift = k_max - (h.size - 1)
    h_shifted = np.roll(h, shift)
    gradient = 1 - h_shifted
    return gradient[: h.size]

def simulated_annealing(h, T, alpha, steps, dx, n_points, break_early=True):
    best_h = h.copy()
    best_C = compute_overlap(best_h, dx)
    for step in range(steps):
        h_candidate = h + np.random.normal(0, T, h.shape)
        h_candidate = repair_feasibility(h_candidate, n_points)
        C_candidate = compute_overlap(h_candidate, dx)
        if C_candidate < best_C or np.random.rand() < np.exp((best_C - C_candidate) / T):
            h = h_candidate
            if C_candidate < best_C:
                best_h = h_candidate
                best_C = C_candidate
        if T > 1e-6:
            T *= alpha
        if break_early and best_C < 1e-12 and step > 100:
            break
    return best_h, best_C

def refine_resolution(h, factor=2):
    h_fine = zoom(h, factor, order=3)  # cubic interpolation
    n_fine = h_fine.size
    return repair_feasibility(h_fine, n_fine)

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    t_start = time.time()

    # Use initial_h_values if provided
    if 'initial_h_values' in globals():
        h = initial_h_values
        n_points = h.size
    else:
        n_points = 14
        h = np.random.rand(n_points)
        h = repair_feasibility(h, n_points)

    best_h = h.copy()
    best_C = compute_overlap(h, 2.0 / n_points)

    # Coarse optimization
    dx_coarse = 2.0 / n_points
    T = 1.0
    steps_sa = int(budget_s * 0.7 // 0.1)
    optimized_h, C = simulated_annealing(h, T, 0.999, steps_sa, dx_coarse, n_points)
    best_h = optimized_h
    best_C = C

    # Refine resolution and repeat
    resolutions = [28, 56]
    for n_fine in resolutions:
        h_fine = refine_resolution(best_h, n_fine / best_h.size)
        dx_fine = 2.0 / n_fine

        # Residual budget check
        if time.time() - t_start >= budget_s * 0.85:
            break

        steps_per_refinement = int((budget_s - (time.time() - t_start)) // 0.01)
        optimized_h_fine, C_fine = simulated_annealing(h_fine, 0.5, 0.9995, steps_per_refinement, dx_fine, n_fine, break_early=False)
        best_h = optimized_h_fine
        best_C = C_fine

    # Final check and return
    elapsed = time.time() - t_start
    if best_C > compute_overlap(best_h, 2.0 / best_h.size):
        final_C = compute_overlap(best_h, 2.0 / best_h.size)
    else:
        final_C = best_C

    return best_h, final_C, best_h.size
