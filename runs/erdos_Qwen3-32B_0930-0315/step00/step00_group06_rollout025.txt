import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    # Configuration
    n_points = 256
    dx = 2.0 / n_points
    total_area = dx * n_points / 2

    # Initial step function (normalized)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).astype(np.float64).copy()
        h *= total_area / (2 * dx)
    else:
        # No parent: generate uniform random distribution
        h = np.random.uniform(0, 1, n_points)
        h *= total_area / (h.sum() + 1e-8)

    # Optimization parameters
    learning_rate = 0.1
    max_iterations = 100_000
    perturbation_freq = 100
    perturbation_strength = 0.01

    # Clip and re-normalize initial h
    np.clip(h, 0, 1, out=h)
    h *= total_area / (h.sum() + 1e-8)

    start_time = time.time()
    best_h = h.copy()
    best_c5 = float('inf')

    for iter in range(max_iterations):
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break

        # Compute C₅ via cross-correlation
        hh = correlate(h, 1 - h, mode='full')
        hh *= dx  # scale to area

        # Only consider relevant range of shifts: from -n_points+1 to +n_points-1
        idx = np.arange(h.size)
        hh = hh[n_points - 1 : 2 * n_points]
        C5 = hh.max()

        # Track improvement
        if C5 < best_c5:
            best_c5 = C5
            best_h = h.copy()

        # Determine which shift k corresponds to the maximum C₅
        k_max = np.argmax(hh)

        # Compute gradient at k_max
        grad = np.zeros_like(h)
        grad[:-k_max] += 1 - h[k_max:]
        grad[k_max:] -= 1 - h[:-k_max]
        grad *= dx

        # Gradient update
        h -= learning_rate * grad

        # Project back to constraints
        np.clip(h, 0, 1, out=h)
        h *= total_area / (h.sum() + 1e-8)

        # Adaptive learning rate decay
        if iter % 1000 == 0:
            learning_rate *= 0.999

        # Perturb to escape local minima
        if iter % perturbation_freq == 0 and iter > 0:
            noise = 2 * perturbation_strength * (np.random.rand(n_points) - 0.5)
            h += noise
            h *= total_area / (h.sum() + 1e-8)
            np.clip(h, 0, 1, out=h)

        # Debug print
        if iter % 1000 == 0:
            print(f"It {iter}, t {elapsed_time:.1f}s, C₅={best_c5:.10f}")

    return best_h, best_c5, n_points
