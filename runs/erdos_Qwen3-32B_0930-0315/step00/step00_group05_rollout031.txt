import numpy as np
import time
from scipy.optimize import minimize
from math import sqrt

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr * dx)
    return c5

def objective(h, n_points, lambda_penalty):
    dx = 2.0 / n_points
    constraint_violation = (np.sum(h) - n_points / 2) / n_points
    c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
    # Penalize constraint violation
    return c5 + lambda_penalty * constraint_violation ** 2

def optimize_from_initial(initial_h):
    # Initialize optimization
    n_points = len(initial_h)
    h = np.copy(initial_h)
    
    # Ensure initial solution is feasible
    h = np.clip(h, 0.0, 1.0)
    h *= n_points / 2 / np.sum(h)  # scale to meet the sum constraint
    
    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    t0 = time.time()
    
    # Optimization with L-BFGS-B
    sum_h = n_points / 2
    lambda_penalty = 1e4
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def opt_func(h_vector):
        nonlocal best_c5, best_h, lambda_penalty
        t_elapsed = time.time() - t0
        full_time = 1100
        
        if t_elapsed > 0.95 * full_time:
            return 0
        
        c5 = evaluate_erdos_solution(h_vector, n_points)
        if c5 < best_c5:
            best_h[:] = np.copy(h_vector)
            best_c5 = c5
        
        # Progressively reduce penalty as solution converges
        if c5 < 0.381 and t_elapsed > 300:
            lambda_penalty = max(1, lambda_penalty // 10)
        
        # Add controlled coarse perturbations every 100 steps
        if t_elapsed // 10 % 100 == 0 and t_elapsed > 100:
            noise = 0.01 * np.random.randn(n_points)
            h_vector[:] = np.clip(h_vector + noise, 0, 1)
            h_vector *= sum_h / np.sum(h_vector)
        
        constraint_violation = (np.sum(h_vector) - sum_h) / n_points
        obj = c5 + lambda_penalty * constraint_violation**2
        return obj
    
    params = {
        'h0': h.tolist() if isinstance(h, np.ndarray) else h,
        'method': 'L-BFGS-B',
        'bounds': bounds,
        'options': {
            'maxiter': 5000,
            'ftol': 1e-6
        },
        'args': (n_points, lambda_penalty)
    }
    
    result = minimize(lambda x: opt_func(x), **params)
    
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    
    # If time remains and gain is significant, refine with higher resolution
    max_time = time.time() - t0 < 500
    if max_time and final_c5 < 0.381:
        h_highres = np.interp(np.linspace(0, 1, n_points*2), np.linspace(0, 1, n_points), best_h)
        h_highres = np.clip(h_highres, 0, 1)
        h_highres *= sum_h*2 / np.sum(h_highres)
        refine_result = optimize_from_initial(h_highres)
        if refine_result[1] < final_c5:
            return refine_result
    
    return best_h.tolist(), final_c5, n_points

def run(seed=42, budget_s=1100):
