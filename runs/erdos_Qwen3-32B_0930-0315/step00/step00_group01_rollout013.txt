import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=1100, **kwargs):
    START_TIME = time.time()
    np.random.seed(seed)

    # Load the initial solution if it exists
    n_points = 47
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h /= h.sum() * 2  # scale to ensure sum(h) = n_points / 2

    def objective(h_):
        h = h_.reshape(-1)
        h = np.clip(h, 0, 1)
        sum_h = h.sum()
        normalization = n_points / 2
        if abs(sum_h - normalization) > 1e-6:
            h = normalization * h / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        c5 = np.max(np.abs(corr)) * dx
        return c5

    def constraint_sum_h(h_):
        return h_.sum() - (n_points / 2)

    def constraint_lower_bound(h_):
        return h_.min()

    def constraint_upper_bound(h_):
        return 1 - h_.max()

    def optimize_step(h, step_size=0.1):
        # First, compute the max overlap and its index
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        indices = np.where(corr == max_corr)[0]
        k_max = indices[0] if len(indices) > 0 else 0  # Choose one of the worst lags

        # Compute the gradient to reduce that overlap
        n = len(h)
        grad = np.zeros(n)
        if k_max < n:
            grad[:n - k_max] = (1 - h[k_max:]) - h[:n - k_max]

        new_h = h - step_size * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = normalization * new_h / new_h.sum()

        return new_h

    # Total iterations
    best_c5 = np.inf
    best_h = h
    t0 = time.time()
    max_evals = 1000  # conservative limit to avoid time budget overflow
    iters = 0
    while time.time() - t0 < budget_s and iters < max_evals:
        iters += 1
        h = optimize_step(h, step_size=0.01)
        c5 = objective(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Project back to the feasible set once more
    best_h = np.clip(best_h, 0, 1)
    best_h = (n_points / 2) * best_h / best_h.sum()

    # Final validation
    if abs(best_h.sum() - n_points / 2) > 1e-6:
        best_h = (n_points / 2) * best_h / best_h.sum()
    best_h = np.clip(best_h, 0, 1)

    return best_h.round(6).tolist(), best_c5, n_points
