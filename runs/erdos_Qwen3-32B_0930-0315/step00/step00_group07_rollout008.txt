import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, n_points=80):
    t0 = time.time()

    dx = 2.0 / n_points
    h = initial_h_values if initial_h_values is not None else 0.5 * np.ones(n_points)

    # Initial normalization to meet sum constraint
    h_sum = np.sum(h)
    h = h / h_sum * (n_points / 2)

    best_h = h.copy()
    best_c5 = float('inf')

    while time.time() - t0 < budget_s:
        # Compute overlap for all lags
        h_1_minus_h = 1.0 - h
        corr = np.correlate(h, h_1_minus_h, mode="full") * dx
        max_k = np.argmax(np.abs(corr))
        if max_k < n_points:
            worst_k = max_k
        else:
            worst_k = max_k - n_points

        # Compute gradient for worst_k
        def mod_index(i, k):
            return (i + k) % n_points

        gradient = np.zeros(n_points)
        for i in range(n_points):
            term1 = 1.0 - h[mod_index(i, worst_k)]
            term2 = 1.0 - h[mod_index(i - worst_k, n_points)]
            gradient[i] = dx * (term1 + term2)

        # Normalize gradient for regularization and numerical stability
        gradient = gradient / (np.abs(gradient).max() + 1e-6)

        # Form linear program to perturb h to decrease max overlap
        delta = cp.Variable(n_points)
        objective = cp.Minimize(corr[worst_k] - cp.sum(delta * gradient))
        constraints = [
            h + delta >= 0,
            h + delta <= 1,
            cp.sum(delta) == 0
        ]
        prob = cp.Problem(objective, constraints)

        try:
            prob.solve(solver=cp.ECOS, max_iters=150, verbose=False)
            delta_value = delta.value
        except:
            continue

        if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            continue

        # Step size via line search
        step = 0.1
        for _ in range(10):
            new_h = h + step * delta_value
            if np.all((0 <= new_h) & (new_h <= 1)):
                break
            step /= 2.0

        h = new_h

        # Enforce sum constraint exactly
        h_sum = np.sum(h)
        correction = (n_points / 2) / h_sum if h_sum > 1e-6 else 0.0
        h *= correction  # scale to exact constraint

        # Check for best solution
        h_1_minus_h = 1.0 - h
        corr = np.correlate(h, h_1_minus_h, mode="full") * dx
        c5 = np.max(np.abs(corr))

        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Validate best solution
    assert np.allclose(np.sum(best_h), n_points / 2), "Bad normalization"
    assert np.all((0 <= best_h) & (best_h <= 1)), "Out of bounds"

    return (best_h, best_c5, n_points)
