import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Use the initial construction if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Otherwise, start with a balanced binary function
        n_points = 47  # Starting with a reasonably dense discretization
        h = np.zeros(n_points)
        h[:n_points // 2] = 1.0  # Half of the samples set to 1

    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        """Objective function to minimize: max correlation term"""
        h = np.minimum(np.maximum(h, 0), 1)  # Clamp to [0, 1]
        if not np.isclose(np.sum(h), target_sum):
            h = h / np.sum(h) * target_sum  # Enforce sum constraint
        h = h.astype(np.float64)
        corr = correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx  # Scale with dx for C₅

    best_c5 = np.inf
    best_h = np.copy(h)
    t_start = time.time()

    # Gradient descent with line search
    max_iter = 3000
    max_line_search_steps = 20
    grad_threshold = 1e-4
    patience = 100
    wait_counter = 0

    for iter in range(max_iter):
        h_current = np.copy(h)

        # Compute the gradient using finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_perturbed = np.copy(h)
            h_perturbed[i] += eps
            j_p = objective(h_perturbed)
            h_perturbed[i] -= 2 * eps
            j_m = objective(h_perturbed)
            grad[i] = (j_p - j_m) / (2 * eps)

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm

        # Line search with backtracking
        step_size = 0.5
        for _ in range(max_line_search_steps):
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)  # Clamping values to [0,1]

            total = np.sum(h_new)
            if not np.isclose(total, target_sum):
                h_new = h_new / total * target_sum  # Rescaling if needed

            c5_new = objective(h_new)
            if c5_new < objective(h):
                h = h_new
                break
            step_size *= 0.9

        # Check for improvement
        c5_current = objective(h)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = np.copy(h)
            best_counter = 0  # Reset patience counter
        else:
            best_counter += 1

        # Early stopping if progress stalls
        if best_counter >= patience:
            break

        # Time check
        if time.time() - t_start >= budget_s:
            break

    # Final projection
    h = np.copy(best_h)
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * target_sum

    # Final objective computation
    c5_bound = objective(h)

    return (h.tolist(), c5_bound, n_points)
