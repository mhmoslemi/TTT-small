import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

# Global variable provided by the harness
# initial_h_values is a numpy array of shape (n_points,)
# with the starting construction


def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    
    # Read initial_h_values provided by the harness
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    def compute_c5(h_vals):
        th = np.concatenate((h_vals, h_vals))
        not_h = 1.0 - th
        overlap = correlate(h_vals, not_h, mode='full')
        c5 = np.max(overlap * dx)
        return c5

    def constrain_h_sum(h_vals):
        return np.sum(h_vals) / n_points * 2.0

    def project_onto_constraints(h_vals):
        # Normalize to meet integral constraint
        h_vals = h_vals / constrain_h_sum(h_vals) * (n_points / 2)
        # Clamp to [0,1]
        return np.clip(h_vals, 0, 1)

    def gradient_descent(h_vals, learning_rate=0.05, directions=10):
        h_combined = np.concatenate((h_vals, h_vals))
        not_h_combined = 1 - h_combined
        overlap = correlate(h_vals, not_h_combined, mode='full') * dx
        max_lag = np.argmax(overlap)
        lag = max_lag - n_points + 1

        # Apply a small perturbation in upright and left directions
        directions = []
        for dl in range(-5, 6):
            if abs(dl) > 10:
                continue
            mask = np.zeros_like(h_vals)
            for i in range(n_points):
                for r in range(5):
                    idx1 = (i + dl + r) % n_points
                    idx2 = (i - dl - r) % n_points
                    mask[idx1] += 1
                    mask[idx2] += 1
            mask /= mask.sum()
            directions.append(mask)

        best_h = h_vals.copy()
        best_c5 = compute_c5(h_vals)
        for dir_idx in range(10):
            if time.time() - start_time > budget_s - 30:
                break
            direction = directions[dir_idx]
            step = np.sign(compute_c5(best_h) - compute_c5(best_h - learning_rate * direction))
            trial_h = best_h - learning_rate * step * direction
            trial_h = project_onto_constraints(trial_h)
            trial_c5 = compute_c5(trial_h)
            if trial_c5 < best_c5:
                best_h = trial_h
                best_c5 = trial_c5

        return best_h, best_c5

    best_c5 = compute_c5(h)
    best_h = h.copy()

    t_end = time.time()
    while t_end - start_time < budget_s - 30:
        h, c5 = gradient_descent(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        t_end = time.time()

    final_c5 = compute_c5(project_onto_constraints(best_h))
    return best_h, final_c5, n_points
