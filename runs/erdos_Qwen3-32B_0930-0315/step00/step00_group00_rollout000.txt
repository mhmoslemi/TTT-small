import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

dx = 2.0 / len(initial_h_values)
n_points = len(initial_h_values)
target_val = n_points / 2

def evaluate_C5(h):
    h = h / np.sum(h) * target_val
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_simplex(v, z=target_val):
    """Project onto the simplex constraint sum(v) == z and 0 <= v[i] <= 1"""
    v = np.clip(v, 0, 1)
    s = np.sum(v)
    if abs(s - z) < 1e-8:
        return v
    d = v - (s - z) / len(v)
    return np.clip(d, 0, 1)

def gradient_C5(h):
    n = len(h)
    max_corr = -np.inf
    max_k = -1
    corr_all = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr_all)
    best_offsets = np.where(np.isclose(corr_all, max_val))[0]

    grad = np.zeros(n)
    for k in best_offsets:
        overlap = []
        for i in range(n):
            j = i + k
            if j < 0 or j >= n:
                continue
            overlap.append((i, j))
        for i, j in overlap:
            grad[i] += (1 - h[j])
            grad[j] -= h[i] * 1.0
    grad *= dx
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    h = initial_h_values.copy()
    h = h / np.sum(h) * target_val

    best_h = h.copy()
    best_c5 = evaluate_C5(h)
    start_time = time.time()

    iter = 0
    learning_rate = 2.0
    lr_decay = 0.995

    while time.time() - start_time < budget_s:
        # Project h to maintain constraints
        h = project_simplex(h)

        # Compute gradient
        grad = gradient_C5(h)

        # Gradient descent step
        h -= learning_rate * grad

        # Project again to maintain constraints
        h = project_simplex(h)

        # Evaluate
        c5 = evaluate_C5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Adapt learning rate
        learning_rate *= lr_decay
        learning_rate = max(0.001, learning_rate)

        iter += 1
        if iter % 100 == 0:
            print(f"Iter {iter}, C5={best_c5:.6f}, time={time.time() - start_time:.1f}s")

        # Early stopping if C5 does not improve after 2000 iterations
        if iter > 2000 and best_c5 < best_c5:
            print("C5 not improved after 2000 iterations. Stopping.")
            break

    return best_h, best_c5, n_points
