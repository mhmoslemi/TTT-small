import numpy as np
import time
from scipy.signal import correlate

def evaluate_objective(h, dx):
    N = len(h)
    corr = correlate(h, 1 - np.roll(h, 1), mode="full")
    return np.max(corr[1:N]) * dx

def project_feasible(x, n_points):
    x = np.clip(x, 0, 1)
    x *= (n_points / 2) / np.sum(x)  # Enforce sum constraint
    return x

    decay_steps = 100

        max_corr_index = np.argmax(corr[1:n_points])
        shift = max_corr_index + 1

        # Compute gradient from max overlap shift
        shifted = np.roll(h, shift)
        grad = (1 - shifted)

        h -= learning_rate * grad
