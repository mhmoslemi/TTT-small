import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def objective_and_constraints(h, n_points, dx):
    h_values = h
    correlations = (
        correlate(h_values, 1 - h_values, mode='full') * dx
    )
    return np.max(correlations), correlations

def optimize_with_sdp(n_points, initial_h, dx, budget_s, seed):
    t_var = cp.Variable(pos=True)
    x_var = cp.Variable(n_points)

    ones = np.ones(n_points)

    constraints = [
        x_var >= 0,
        x_var <= 1,
        ones @ x_var == n_points / 2,
    ]

    obj = cp.Minimize(t_var)

    prob = cp.Problem(obj, constraints)

    x_val = 0.5 * np.ones(n_points) if initial_h is None else initial_h
    result = None

    start_time = time.time()

    while time.time() - start_time < budget_s:
        print(f"Started optimization epoch at {time.time() - start_time:.2f}s")

        # Add constraint to bound max cross-correlation
        x_h = x_var.copy()

        constraints += [
            (
                cp.sum([x_h[i] * (1 - x_h[i + k]) for i in range(n_points - k)])
                <= t_var * n_points / 2
            )
            for k in range(1, n_points)
        ]
        # Equality to prevent drifting
        constraints += [x_var == x_val]

        prob = cp.Problem(obj, constraints)
        prob.solve(warm_start=True, solver='ECOS')
        x_val = x_var.value

        probs = correlate(x_val, 1 - x_val, mode='full') * dx
        new_C5 = np.max(probs)

        if result is None or new_C5 < result[1]:
            result = (x_val, new_C5, n_points)

        # Clamp to [0, 1] and restore sum constraint as necessary
        x_val = np.clip(x_val, 0, 1)
        x_val /= np.sum(x_val)
        x_val *= n_points / 2

    return result
