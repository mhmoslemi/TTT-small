import numpy as np
import time
from scipy.signal import correlate


def run(seed=42, budget_s=1100):
    # Set random seed if needed
    np.random.seed(seed)

    # Load initial solution
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Normalize to satisfy constraint: ∫ h(x) dx = 1
    total_area = np.sum(initial_h) * dx
    initial_h *= 1 / total_area

    # Ensure sum(h) == n_points / 2 (required for constraint)
    initial_h *= n_points / 2 / np.sum(initial_h)
    assert np.all(initial_h <= 1 + 1e-8), "Initial h values exceed upper bound 1"
    h = initial_h.copy()

    # Setup
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    t0 = time.time()
    iter_count = 0
    learning_rate = 0.01

    def project(h, n_points):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h *= scale
        return h

    while time.time() - t0 < budget_s:
        # Evaluate the current objective
        c5 = evaluate_erdos_solution(h)

        # Check improvement
        if c5 < best_c5 - 1e-6:
            best_h = h.copy()
            best_c5 = c5
            learning_rate *= 1.05  # small increase for finding better solution
        elif c5 > best_c5 + 1e-6:
            learning_rate *= 0.95  # small decrease

        # Find the shift k with maximum overlap
        corr = correlate(h, 1 - h, mode="full")
        corr = corr * dx
        k_max = np.argmax(corr)

        # Compute gradients w.r.t h to minimize corr[k_max]
        grad = np.zeros_like(h)
        grad += (1 - h[(np.arange(n_points) + k_max) % n_points])  # Forward
        grad -= h[(np.arange(n_points) - k_max) % n_points]  # Backward
        grad += (1 - h[(np.arange(n_points) - k_max) % n_points]) * 0  # Symmetry adjustment

        # Update h with gradient descent
        h -= learning_rate * (-grad)
        h = project(h, n_points)

        iter_count += 1

        # Report progress
        if iter_count % 100 == 0:
            print(f"Iteration {iter_count}: C₅ = {best_c5} | Time: {time.time()-t0:.2f}s")

        # Time constraint check
        if time.time() - t0 >= budget_s:
            break

    # Ensure final constraints are met precisely
    best_h = project(best_h, n_points)
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
