import numpy as np
import cvxpy as cp
import time

from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values


def run(seed=42, budget_s=1100, n_points=80):
    np.random.seed(seed)
    dx = 2.0 / n_points

    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h.fill(0.5)

    start_time = time.time()
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)  # Initial value from provided function

    while time.time() - start_time < budget_s:
        # Compute the overlap correlations
        h_minus = 1 - h
        corr = np.correlate(h, h_minus, mode='full')
        corr = corr * dx

        # Identify worst-case lag k with max value
        k = np.argmax(corr)

        if k < n_points:
            # Build shifted h array at lag k for constraint
            h_k = np.roll(h, shift=-k)

            # Define CVX variables and objective
            h_new = cp.Variable(n_points)
            constraints = [
                h_new >= 0,
                h_new <= 1,
                cp.sum(h_new) == n_points / 2,
            ]
            # Minimize overlap at the worst-case lag
            objective = cp.Minimize(cp.sum(cp.multiply(h_new, (1 - h_k))))  # Objective for current lag
            problem = cp.Problem(objective, constraints)

            try:
                if 'MOSEK' in cp.installed_solvers():
                    problem.solve(solver=cp.MOSEK, mosek_params={mosek.dparam.intpnt_tol_feas: 1e-8})
                else:
                    problem.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8)
            except Exception:
                # On failure, fall back to current solution
                h_new = h.reshape(-1)

            if problem.status == cp.OPTIMAL:
                h = h_new.value
                # Re-evaluate C5 after the update
                current_C5 = evaluate_erdos_solution(h)
                if current_C5 < best_C5:
                    best_C5 = current_C5
                    best_h = h.copy()
        else:
            break  # Invalid k

        # Stop early if we reach the target
        if best_C5 <= 0.380876:
            break

    return (best_h, best_C5, n_points)
