import numpy as np
import time
import random
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=1100):
    random.seed(seed)
    np.random.seed(seed)
    
    # Use the provided initial construction if available
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_base = np.array(initial_h_values)
    else:
        # Construct a uniform step function if no initial_h_values is available
        n_points = 70
        h_base = np.ones(n_points) / n_points * (n_points / 2)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h_best = h_base.copy()
    c5_best = evaluate_erdos_solution(h_best)
    
    population_size = 50
    mutation_rate = 0.15
    tournament_size = 5
    generations = 0
    start_time = time.time()
    
    # Normalize initial population
    def repair_ofs(h):
        h = np.clip(h, 0, 1)
        s = h.sum()
        if s == 0:
            return np.zeros_like(h)
        scale = target_sum / s
        h = np.clip(h * scale, 0, 1)
        return h
    
    population = [repair_ofs(h_base + (np.random.rand(n_points) * 0.2 - 0.1)) for _ in range(population_size)]
    
    def evaluate_fitness(h):
        # Fitness is negative of the C₅ value (we maximize fitness to minimize C₅)
        correlations = correlate(h, 1.0 - h, mode='full')
        C5 = correlations.max() * dx
        return -C5
    
    while time.time() - start_time < budget_s:
        fitnesses = [evaluate_fitness(h) for h in population]
        parents = []

        # Tournament selection
        for _ in range(population_size // 2):
            candidates = random.choices(list(enumerate(population)), k=tournament_size)
            selected = min(candidates, key=lambda x: -evaluate_fitness(x[1]))
            parents.append(selected[1])

        next_pop = parents.copy()

        # Crossover and mutation
        for i in range(0, len(parents) // 2):
            if i + 1 >= len(parents):
                break
            p1, p2 = parents[i], parents[i + 1]
            if np.random.rand() < 0.8:
                # One-point crossover
                cross_point = np.random.randint(1, n_points - 1)
                o1, o2 = p1.copy(), p2.copy()
                o1[:cross_point], o2[:cross_point] = p2[:cross_point], p1[:cross_point]
                o1 = repair_ofs(o1)
                o2 = repair_ofs(o2)
                next_pop.extend([o1, o2])

        # Add mutations
        for i in range(len(next_pop)):
            if np.random.rand() < mutation_rate:
                mutate = next_pop[i].copy()
                # Pick random indices to mutate
                mask = np.random.rand(n_points) < 0.05
                mutation = np.random.randn(n_points) * 0.1
                mutate[mask] += mutation[mask]
                mutate = repair_ofs(mutate)
                next_pop[i] = mutate

        # Evaluate all new and keep top population_size
        population = sorted(next_pop, key=evaluate_fitness, reverse=True)[:population_size]
        generations += 1

        # Periodically check best
        best_in_gen = evaluate_fitness(population[0])
        current_best = min(c5_best, -best_in_gen)
        if current_best < c5_best:
            c5_best = current_best
            h_best = population[0]
        
        # If C5 <= 0.38085 already reached, we can return early
        if c5_best <= 0.38085:
            break

    return h_best, c5_best, n_points
