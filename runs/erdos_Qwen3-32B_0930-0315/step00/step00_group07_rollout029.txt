import numpy as np
import cvxpy as cp
import time
from evaluate_erdos_solution import evaluate_erdos_solution
import sys

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    n_points = 80  # Start with n=80, as given in initial_h_values
    dx = 2.0 / n_points

    # Use the initial_h_values provided by the parent construction
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        assert len(h) == n_points and np.all((0.0 <= h) * (h <= 1.0)), "Initial h is not valid"
        total_h = np.sum(h)
        assert np.isclose(total_h, n_points / 2.0, atol=1e-6), "Initial h does not satisfy integral constraint"
    else:
        # Fallback initialization
        h = np.full(n_points, 0.5)
        total_h = n_points * 0.5
        assert np.isclose(total_h, n_points / 2.0, atol=1e-6), "Fallback h does not satisfy integral constraint"

    max_lag = 4  # Since we loop over k in [-4, ..., 4], only lags up to 4 affects max C5
    tolerance = 1e-6
    max_iter = 100
    iter_without_improve = 0

    while time.time() - start_time < budget_s - 1:
        # Evaluate current C5
        h_boxed = np.concatenate([h, h])  # for np.correlate
        corr_with_offset = np.correlate(h_boxed, 1 - h_boxed, mode='full') * dx
        corr_values = corr_with_offset[n_points : n_points + n_points]
        C_k = corr_values
        C5 = np.max(C_k)

        if C5 < best_c5:
            best_c5 = C5
            best_h = h.copy()
            iter_without_improve = 0
        else:
            iter_without_improve += 1

        if iter_without_improve >= 20:
            break

        # Find worst lag k
        worst_k = np.argmax(C_k)

        # Variables for convex formulation
        dh = cp.Variable(n_points)
        constraints = [
            h + dh >= 0,
            h + dh <= 1,
            cp.sum(dh) == 0,  # preserve total sum
        ]

        # Objective: reduce C_k for worst_k
        i = np.arange(n_points)
        wrap_idx = (i + worst_k) % n_points
        obj_expr = cp.sum((h + dh) * (1 - (h[wrap_idx] + dh[wrap_idx]))) * dx
        problem = cp.Problem(cp.Minimize(obj_expr), constraints)

        try:
            problem.solve(solver=cp.ECOS, max_iters=1000)
            if dh.value is not None:
                # Perform update step
                dh_flat = dh.value
                step_size = 0.1
                h_update = h + step_size * dh_flat
                h_update = np.clip(h_update, 0, 1)
                current_sum = np.sum(h_update)
                if not np.isclose(current_sum, n_points / 2.0):
                    scale_factor = (n_points / 2.0) / current_sum
                    h_update *= scale_factor
                    h_update = np.clip(h_update, 0, 1)
                h = h_update
        except Exception:
            # Fail-safe: revert to old solution
            pass

    # Final check and return
    h_values = None
    c5_bound = None
    if best_h is not None:
        h_values = best_h.copy().tolist()
        # Final evaluation
        c5_bound, _, _ = evaluate_erdos_solution(n_points, h_values)
    return h_values, c5_bound, n_points
