import numpy as np
import time
from numpy.fft import fft, ifft

def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    k_vals = np.arange(-n_points + 1, n_points)
    cross_correlation = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(cross_correlation * dx)
    return max_overlap

def project_simplex(v, sum_val):
    """
    Project vector v onto the simplex such that sum(v) = sum_val and 0 ≤ v[i] ≤ 1
    """
    n = len(v)
    v_clipped = np.clip(v, 0, 1).astype(float)
    
    # Normalize to the sum constraint
    total = np.sum(v_clipped)
    if total == 0:
        return v * 0  # Avoid division by zero
        
    scale = sum_val / total
    return np.clip(v * scale, 0, 1)

def compute_gradient(h):
    """
    Compute the gradient of C₅ with respect to h using finite differences
    """
    epsilon = 1e-5
    n = len(h)
    grad = np.zeros(n)
    
    for i in range(n):
        h_plus = np.array(h, dtype=float)
        h_plus[i] += epsilon
        
        h_minus = np.array(h, dtype=float)
        h_minus[i] -= epsilon
        
        # Ensure feasible points
        h_plus_clipped = project_simplex(h_plus, n / 2)
        h_minus_clipped = project_simplex(h_minus, n / 2)
        
        f_plus = evaluate_erdos_solution(h_plus_clipped)
        f_minus = evaluate_erdos_solution(h_minus_clipped)
        
        grad[i] = (f_plus - f_minus) / (2 * epsilon)
    
    return grad
