import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 160  # balance precision and efficiency
        # plateau initialization
        midpoint = n_points // 2
        h = np.ones(n_points, dtype=np.float64)
        h[midpoint:] = 0
        # max sum(h) over [0, 1] with area 1 is n_points / 2
        h = np.clip(h * (n_points / 2) / np.sum(h), 0, 1)

    dx = 2.0 / n_points
    start_time = time.time()
    best_h = h.copy()
    best_c5 = float('inf')
    t0 = start_time
    no_progress_count = 0
    alpha = 0.01

    while time.time() - t0 < budget_s:
        correlations = scipy.signal.correlate(h, 1 - h, mode='full', method='direct') * dx
        # wrap indices for circular overlap
        c5 = np.max(correlations)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_progress_count = 0
        else:
            no_progress_count += 1

        # stagnation or overshoot - reduce learning rate
        if no_progress_count > 100 or c5 > best_c5 + 1e-4:
            alpha *= 0.5
            no_progress_count = 0

        max_idx = np.argmax(correlations)
        k_max = max_idx - n_points + 1
        k_max = k_max % n_points

        # gradient w.r.t. max overlap
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad[i] = (1 - h[j]) - h[(i - k_max) % n_points]

        h -= alpha * grad

        # projection: bounds
        h = np.clip(h, 0, 1)

        # projection: normalization (area = 1)
        total = np.sum(h)
        if total != 0:
            h *= (n_points / 2) / total
        else:
            h[:] = 0.5  # fallback

    # compute final value
    correlations = scipy.signal.correlate(best_h, 1 - best_h, mode='full', method='direct') * dx
    c5_bound = np.max(correlations)

    return (best_h.tolist(), c5_bound, n_points)
