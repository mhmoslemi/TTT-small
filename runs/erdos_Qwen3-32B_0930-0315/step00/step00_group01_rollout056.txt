import numpy as np

import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    # Start with initial_h_values
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    h_sum = n_points / 2

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    start_time = time.time()

    # Hyperparameters
    learning_rate = 0.02

    while time.time() - start_time < budget_s:
        # Compute overlaps for all k
        overlaps = np.zeros(n_points)
        for k in range(1, n_points):
            valid_indices = np.arange(n_points - k)
            overlaps[k] = np.sum(h[valid_indices] * (1 - h[valid_indices + k])) * dx

        # Find worst k
        k_max = np.argmax(overlaps)

        # Compute gradient for worst k
        grad = np.zeros_like(h)
        for i in range(n_points - k_max):
            grad[i] += (1 - h[i + k_max])
            grad[i + k_max] -= h[i]

        # Update h
        h = h - learning_rate * grad

        # Project back into [0, 1] and maintain sum
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * h_sum

        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    return (best_h.tolist(), best_c5, n_points)
