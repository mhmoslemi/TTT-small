import numpy as np
import cvxpy as cp
import time

def cross_correlation(h):
    return np.correlate(h, 1 - h, mode="full") * (2.0 / len(h))

def compute_gradient(h, k, n_points):
    dx = 2.0 / n_points
    G = np.zeros_like(h)
    for i in range(n_points):
        if i + k < n_points:
            G[i] += (1 - h[i + k]) * dx
        if i - k >= 0:
            G[i] -= h[i - k] * dx
    return G

def project_feasible(h, n_points):
    h = cp.Variable(n_points, nonneg=True)
    objective = cp.Minimize(cp.sum_squares(h - h_init))
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2
    ]
    prob = cp.Problem(objective, constraints)
    result = prob.solve(solver=cp.ECOS)
    return h.value

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s

    if 'initial_h_values' in globals():
        h_init = initial_h_values
        n_points = len(h_init)
        assert np.allclose(np.sum(h_init), n_points / 2, atol=1e-6)
    else:
        n_points = 70
        h_init = 0.5 * np.ones(n_points)

    h = np.copy(h_init)
    best_h = np.copy(h)
    best_c5 = cross_correlation(h).max()
    iter_no_improve = 0
    max_iter = 10000
    iteration = 0
    eta = 0.01

    while time.time() - start_time < max_time and iteration < max_iter:
        iteration += 1
        corr_values = cross_correlation(h)
        k_worst = np.argmax(corr_values)
        grad = compute_gradient(h, k_worst, n_points)
        grad /= (1e-8 + np.linalg.norm(grad))  # Normalize gradient

        h_candidate = h - eta * grad
        h_candidate = np.clip(h_candidate, 0, 1)

        # Ensuring sum constraint
        diff_sum = np.sum(h_candidate) - n_points / 2
        alpha = n_points / 2 / np.sum(h_candidate)
        h_candidate = np.clip(h_candidate * alpha, 0, 1)

        # Constrain rounding errors
        h_candidate = h_candidate * (n_points / 2 / h_candidate.sum())

        h = np.copy(h_candidate)
        corr_values = cross_correlation(h)
        c5 = corr_values.max()

        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            iter_no_improve = 0
            eta = 0.1  # Slight increase for greedy step
        else:
            iter_no_improve += 1
            if iter_no_improve > 100:
                eta *= 0.5  # Reduce step
                iter_no_improve = 0

    return best_h, best_c5, n_points
