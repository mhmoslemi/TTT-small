import numpy as np
import time
import cvxpy as cp
from functools import partial

def evaluate_erdos_c5(h_values, dx):
    h = h_values.copy()
    # Compute 1 - h
    one_minus_h = 1.0 - h
    # Compute correlate(h, 1-h, mode='full') in discrete convolution
    # This can be done using FFT convolution for speed
    h_pad = np.pad(h, (0, len(one_minus_h) - 1), mode='constant')
    omh_pad = np.pad(one_minus_h, (0, len(h) - 1), mode='constant')
    corr = np.correlate(h_pad, omh_pad, mode='valid')
    return np.max(corr) * dx

def project_h_constraint(h, n_points):
    # Constraints: sum(h) == n_points / 2 and 0 ≤ h[i] ≤ 1
    # First, clip values to [0, 1]
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if not np.isclose(total, n_points / 2):
        # Project onto l_2 ball if necessary or scale
        scaling_factor = (n_points / 2) / total
        h = h * scaling_factor
        # If any are out of bounds, clip again
        h = np.clip(h, 0, 1)
    return h

def grad_C(h, dx):
    # Approximate gradient of C_5
    n = len(h)
    h = h * 1.0
    one_minus_h = 1.0 - h

    # Compute convolution difference (1 - 2 * h[j]) for each shift
    conv_diff = np.zeros(2 * n - 1)
    for k in range(1 - n, n):
        idxs = (np.arange(n) + k) % n
        conv_diff[k + n - 1] = np.sum(h * one_minus_h[idxs])
    conv_diff *= dx
    max_idx = np.argmax(conv_diff)
    # Retrieve the overlap integral over the most problematic shift
    k = max_idx - (n - 1)
    idxs = (np.arange(n) + k) % n
    overlap = h * (1 - h[idxs])
    grad = (1 - 2 * h[idxs]) * dx

    return grad
