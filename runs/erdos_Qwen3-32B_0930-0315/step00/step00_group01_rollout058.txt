import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 47
        h = np.zeros(n_points, dtype=np.float64)
        h[:n_points // 2] = 1.0  # Simple symmetric initialization

    def project_h(h):
        # Project h to ensure 0 <= h <= 1 and constraint sum(h) = n_points / 2
        h = np.clip(h, 0.0, 1.0)
        total = h.sum()
        if abs(total - n_points / 2) > 1e-6:
            # Normalize to constraint
            h += (n_points / 2 - total) / n_points
            h = np.clip(h, 0.0, 1.0)
        return h

    def compute_overlap(h):
        # Compute the max overlap across all shifts
        dx = 2.0 / n_points
        overlaps = []
        for k in range(-n_points + 1, n_points):
            rev = k < 0
            k = abs(k)
            if rev:
                a = h[k:]
                b = h[: -k]
            else:
                a = h[:-k]
                b = h[k:]
            if len(a) == 0 or len(b) == 0:
                continue
            overlap = (a * (1 - b)).sum() * dx
            overlaps.append(overlap)
        return np.array(overlaps)

    best_h = h.copy()
    best_overlaps = compute_overlap(h)
    best_c5 = best_overlaps.max()

    learning_rate = 0.01
    patience = 0
    max_patience = 100

    while time.time() - start_time < budget_s:
        overlaps = compute_overlap(h)
        worst_k = np.argmax(overlaps)
        c5 = overlaps[worst_k]

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_overlaps = overlaps.copy()
            patience = 0
            learning_rate = min(0.02, learning_rate * 1.2)
        else:
            patience += 1
            if patience >= max_patience:
                learning_rate *= 0.9

        if learning_rate < 1e-6:
            break

        dx = 2.0 / n_points
        grad = np.zeros_like(h)

        # Build gradient for worst_k
        k = worst_k - (n_points - 1)
        if k < 0:
            a = h[-k:]
            b = h[:k]
        else:
            a = h[:-k]
            b = h[k:]

        # For each index i in a and its counterpart i + k in b
        if k < 0:
            for i in range(len(a)):
                grad[i] += (1 - b[i]) * dx
                idx = i + k
                if(idx >= 0):
                    grad[idx] -= a[i] * dx
        else:
            for i in range(len(a)):
                grad[i] += (1 - b[i]) * dx
                idx = i + k
                if(idx < len(h)):
                    grad[idx] -= a[i] * dx

        h = project_h(h - learning_rate * grad)

    final_overlaps = compute_overlap(best_h)
    final_c5 = final_overlaps.max()
    return best_h.tolist(), final_c5, n_points
