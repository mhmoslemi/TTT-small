import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values[:]
    n_points = len(initial_h)

    dx = 2.0 / n_points
    def get_overlap_integral(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:n_points] * dx)

    current_h = np.array(initial_h)
    best_h = current_h.copy()
    best_c5 = get_overlap_integral(current_h)

    t0 = time.time()

    while time.time() - t0 < budget_s * 0.9:
        # Gradient descent step
        h = cp.Variable(n_points)
        objective = cp.Minimize(cp.norm(h - current_h, 1))
        constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS)

        if prob.status != 'optimal':
            break

        current_h = np.array(h.value).flatten()
        current_c5 = get_overlap_integral(current_h)

        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5

        current_h = best_h  # Reset to best before next round

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h / best_h.sum() * (n_points / 2)

    return (best_h.tolist(), best_c5, n_points)
