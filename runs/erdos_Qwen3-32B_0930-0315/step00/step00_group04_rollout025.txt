import numpy as np
import time
from scipy.interpolate import interp1d


def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    t0 = time.time()
    
    n_points = 14
    dx = 2.0 / n_points

    def repair_feasibility(h, n_points):
        h = np.clip(h, 0, 1)
        h = h / h.sum() * (n_points / 2)
        return h

    def compute_overlap(h, dx):
        h1 = h - h.min()
        h1 /= h1.max()
        h1 = h1 / h1.sum() * (n_points / 2)
        C = np.correlate(h1, 1 - h1, mode='full') * dx
        return np.max(C), np.argmax(C)

    def simple_weighted_mean(a, b, beta):
        beta = max(min(beta, 1.0), 0.0)
        return a * (1.0 - beta) + b * beta

    # Use the pre-imported initial_h_values if available
    if 'initial_h_values' in globals():
        h_values = initial_h_values
        h_values = np.array(h_values)
    else:
        # Initialize uniform
        h_values = np.ones(n_points) * 0.5

    h_values = repair_feasibility(h_values, n_points)
    best_C = compute_overlap(h_values, dx)[0]
    best_h = h_values.copy()

    # Simulated annealing
    T = 1.0
    alpha = 0.99
    sa_steps = 2000
    max_shift = 0.25
    threshold_step = max(1.0, n_points // 8)

    for i in range(sa_steps):
        if time.time() - t0 > budget_s * 0.8:
            break

        h_candidate = h_values + np.random.randn(n_points) * T * max_shift
        h_candidate = repair_feasibility(h_candidate, n_points)
        candidate_C, _ = compute_overlap(h_candidate, dx)
        
        if candidate_C < best_C or np.random.rand() < np.exp((best_C - candidate_C) / T):
            h_values = h_candidate
            best_C = candidate_C
            best_h = h_candidate.copy()
            T *= 0.995

        T *= alpha

    # Resolution doubling
    if time.time() - t0 < budget_s * 0.98:
        interp_func = interp1d(np.linspace(0, 2, n_points), h_values, kind='cubic')
        n_points = n_points * 2
        dx = 2.0 / n_points
        h_values = interp_func(np.linspace(0, 2, n_points))

        h_values = repair_feasibility(h_values, n_points)
        best_C, _ = compute_overlap(h_values, dx)
        best_h = h_values.copy()

    # Final local search
    step_size = 0.01
    steps = 100
    for _ in range(steps):
        if time.time() - t0 > budget_s:
            break
        h_values -= step_size * (1.0 - np.roll(h_values, 1))
        h_values = repair_feasibility(h_values, n_points)
        best_C, _ = compute_overlap(h_values, dx)

    elapsed = time.time() - t0
    return best_h, best_C, n_points
