import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    
    # Configuration
    n_points_initial = len(initial_h_values) if 'initial_h_values' in globals() else 95
    n_points = max(128, n_points_initial * 2)  # Use higher resolution for better optimization
    dx = 2.0 / n_points
    eps = 1e-6

    # Initialize h using interpolation from parent if available
    if 'initial_h_values' in globals():
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    else:
        # Alternative initial binary pattern with uniform distribution
        h = np.zeros(n_points)
        h[:n_points // 2] = 1.0  # Start with balanced binary function

    h = h / h.sum() * n_points / 2  # Normalize via sum to satisfy average constraint
    h = np.clip(h, 0.0, 1.0)  # Ensure in [0,1] range

    # Optimization loop settings
    max_iter = 2000
    alpha = 0.2  # Fixed step size
    t0 = time.time()
    best_h = h.copy()
    best_c5 = evaluate_C5(best_h, dx)

    # Optimization loop
    for it in range(max_iter):
        if time.time() - t0 > budget_s:
            break

        # Compute overlap integrals for all shifts
        overlap = correlate(h, 1 - h, mode="full") * dx

        # Find the worst shift (max overlap)
        k_max = np.argmax(overlap)

        # Directly compute its correction gradient via finite difference
        grad = np.zeros_like(h)
        for i in range(n_points):
            if k_max - i >= 0 and k_max - i < n_points:
                grad[i] += h[k_max - i]  # contribution of -h[i] from h[x]*h[x+k]
            if i + k_max < n_points:
                grad[i] -= (1 - h[i + k_max])  # contribution of -(1 - h[x+k]) from h[x]*(1 - h[x+k])
        grad *= dx  # Account for dx in integral

        # Projected gradient ascent
        h += alpha * grad
        h = np.clip(h, 0., 1.)  # Back to feasible range
        h = h / h.sum() * n_points / 2  # Normalize to average constraint

        c5 = evaluate_C5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h[:] = h[:]  # Update best solution

    # Final cleaning and pruning to a smaller n (target 95 or similar)
    best_h = np.clip(best_h, 0., 1.)  # Final clip
    n_sample_points = 95  # Target resolution near initial
    h_values = np.interp(np.linspace(0, 2, n_sample_points), np.linspace(0, 2, n_points), best_h)

    return h_values, best_c5, n_sample_points

def evaluate_C5(h, dx):
    """Evaluates the maximum overlap integral as defined by Erdős constant C5."""
    overlap = correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)
