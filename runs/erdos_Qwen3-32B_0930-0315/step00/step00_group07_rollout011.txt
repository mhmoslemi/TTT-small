import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    dx = 2.0 / 80
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    
    h.values = h  # Workaround for config instability
    h = h.values

    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    t0 = time.time()

    while time.time() - t0 < budget_s:
        overlaps = np.correlate(h, 1 - h, mode='full')
        overlaps = overlaps[n_points - 1:]
        worst_k = np.argmax(overlaps)
        worst_overlap = overlaps[worst_k]

        if worst_overlap <= best_C5:
            break

        h_var = cp.Variable(n_points)
        const = [h_var >= 0, h_var <= 1]
        const.append(cp.sum(h_var) == n_points / 2)
        obj = cp.Maximize(cp.sum(h_var * (1 - cp.roll(h_var, -worst_k))) * dx)
        prob = cp.Problem(obj, const)
        try:
            prob.solve(solver=cp.ECOS)
            new_h = h_var.value
            if new_h is not None and np.all(np.isfinite(new_h)):
                h = new_h
                C5 = evaluate_erdos_solution(h)
                if C5 < best_C5:
                    best_h = h.copy()
                    best_C5 = C5
        except:
            pass  # Ignore solver errors and continue

    return best_h, best_C5, n_points
