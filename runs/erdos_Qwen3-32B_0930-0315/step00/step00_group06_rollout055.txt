import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    dx = 2.0 / len(initial_h_values)
    h = np.array(initial_h_values)
    h = h / np.sum(h) * len(h) / 2  # Normalize to satisfy the integral constraint

    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    step_size = 0.05
    decay_rate = 0.995
    improvement_steps = 0
    max_steps_no_improvement = 100
    max_budget_steps = 100_000

    for i in range(max_budget_steps):
        # Compute overlaps at all lags
        overlap = np.correlate(h, 1 - h, mode="full") * dx

        # Identify the maximum overlap and its position
        c5 = np.max(overlap)
        k = np.argmax(overlap)

        # Early termination if elapsed time exceeds budget
        if time.time() - start_time > budget_s:
            break

        # Determine the shift and compute sensitive indices
        if k < len(h):
            # Right overlap
            grad = (1 - h[k:]) * 1
        elif k >= len(h):
            # Left overlap
            grad = (1 - h[: len(h) - (k - len(h) + 1)]) * 1

        # Update h with gradient descent
        h = h - step_size * grad
        h = np.clip(h, 0, 1)

        # Enforce the integral constraint
        current_sum = np.sum(h)
        h = h * (len(h) / 2) / current_sum

        # Evaluate
        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            improvement_steps = 0
        else:
            improvement_steps += 1

        # Optional step decay
        step_size *= decay_rate

        # Termination check
        if improvement_steps >= max_steps_no_improvement:
            break

    return (best_h, best_c5, len(h))
