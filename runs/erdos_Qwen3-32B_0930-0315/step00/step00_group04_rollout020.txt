import numpy as np
import cvxpy as cp
import time
import scipy.signal

from evaluate_erdos_solution import evaluate_erdos_solution

from collections.abc import Sequence

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Initial parameters
    n_points = 42
    dx = 2.0 / n_points

    # Use initial_h_values as starting point if available
    try:
        h_init = np.array(initial_h_values)
    except NameError:
        h_init = np.zeros(n_points)
        h_init[::3] = 1.0  # Regular starting pattern
        h_init = h_init.clip(0, 1)
        h_init = h_init / h_init.sum() * n_points / 2

    h_var = cp.Variable(n_points)
    
    def objective_function(h):
        corr = scipy.signal.correlate(h, 1 - h, mode="full") * dx
        max_corr = cp.max(corr)
        return max_corr

    def compute_c5(h_values):
        h = np.array(h_values)
        h = h.clip(0, 1)
        h = h / h.sum() * n_points / 2
        corr = scipy.signal.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    # Initial validation of initial_h_values
    h_values = h_init.copy()
    h_values = h_values.clip(0, 1)
    h_values = h_values / h_values.sum() * n_points / 2
    
    best_c5 = compute_c5(h_values)
    best_h = h_values.copy()
    
    # Barrier parameter
    mu = 1.0e4
    t_start = time.time()
    n_outer = 0
    
    while time.time() - t_start < budget_s and n_outer < 1000:
        n_outer += 1

        if n_outer % 50 == 0:
            mu *= 0.9
        if mu < 1e-8:
            break

        # Construct model
        obj = objective_function(h_var)
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        problem = cp.Problem(cp.Minimize(obj + mu * -cp.sum(cp.log(h_var) + cp.log(1 - h_var))), constraints)
        
        try:
            result = problem.solve(solver=cp.CVXOPT, verbose=False)
            h_result = h_var.value

            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                # Process result
                h_result = h_result.clip(0, 1)
                h_result = h_result / h_result.sum() * n_points / 2
                current_c5 = compute_c5(h_result)

                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_result.copy()

                # Early exit if converged
                if abs(current_c5 - best_c5) < 1e-7:
                    break
        except Exception as e:
            # In case of numerical error, just skip this iteration
            pass

        # Finish early if progress is minimal
        if n_outer > 10 and abs(compute_c5(best_h) - best_c5) < 1e-6:
            break

    return (best_h, best_c5, n_points)
