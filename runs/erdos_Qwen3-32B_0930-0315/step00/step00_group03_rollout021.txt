import numpy as np
import time
import cvxpy as cp
import evaluate_erdos_solution

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)
    
    # Use parent-specified n=70 for quick evaluation
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Uniform initialization
        h = np.ones(n_points) * 0.5
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Normalize to meet the area constraint
    def project(h):
        h = np.clip(h, 0, 1)
        mean_h = np.sum(h) / n_points
        area_diff = mean_h - 0.5
        if abs(area_diff) > 1e-6:
            h -= area_diff
            h = np.clip(h, 0, 1)
            # Enforce area constraint tightly
            excess = np.sum(h) - 0.5 * n_points
            if np.abs(excess) > 1e-6:
                h_proj = cp.Variable(n_points)
                constraints = [0 <= h_proj, h_proj <= 1, sum(h_proj) == 0.5 * n_points]
                objective = cp.Minimize(cp.sum_squares(h_proj - h))
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.ECOS, verbose=False)
                h = h_proj.value
        return h
    
    iters_no_improv = 0
    iteration = 0
    learning_rate = 0.01
    
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Compute the overlap for all shifts using cross-correlation
        corr = evaluate_erdos_solution.evaluate_overlap(h, n_points)
        max_idx = np.argmax(corr)
        current_c5 = corr[max_idx]
        shift = max_idx - (n_points - 1)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            iters_no_improv = 0
        else:
            iters_no_improv += 1
            
        if iters_no_improv > 100:
            break  # early stopping
            
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            j_plus = (i + int(shift)) % n_points
            grad[i] += (1 - h[j_plus])
            j_minus = (i - int(shift)) % n_points
            grad[i] -= h[j_minus]
        grad = project(grad)
        
        # Update with decaying learning rate
        learning_rate = 0.01 / (1 + iteration / 1000)
        h = h - learning_rate * grad
        h = project(h)
        
    h = best_h
    corr = evaluate_erdos_solution.evaluate_overlap(h, n_points)
    c5_bound = np.max(corr)
    
    # Final check for constraint feasibility
    assert np.all(h >= 0) and np.all(h <= 1), "Feasibility violation"
    assert abs(np.sum(h) - n_points / 2) < 1e-6, "Integral constraint violation"
    
    return h.tolist(), c5_bound, n_points
