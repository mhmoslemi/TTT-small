import numpy as np
import time

from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    total_area = np.sum(h) * dx
    if not np.isclose(total_area, 1.0, atol=1e-6):
        raise ValueError(f"Integral of h is {total_area:.6f} ≠ 1.0")
    overlaps = correlate(h, 1 - h, mode='full')
    overlaps *= dx
    c5 = np.max(overlaps)
    return c5

    np.random.seed(seed)
    start_time = time.time()

    # Custom high-resolution target
    n_points = 256
    dx = 2.0 / n_points
    h = np.zeros(n_points)

            x = np.linspace(0, 2, n_points)
            f = interp1d(x_parent, h_parent, kind='linear', fill_value='extrapolate', bounds_error=False)
            h = f(x)
            h = np.clip(h, 0.0, 1.0)
            h *= (1.0 / (np.sum(h) * dx)) * 0.5 * n_points  # Renormalize to ensure ∫h(x)dx = 1
    except:
        # Fallback initialization
        h = np.zeros(n_points)
        h[:128] = 1.0  # Approx 1.0 * 128 * dx ≈ 1.0

    h = h.astype(np.float64)

    # Normalize exactly to ∫h(x)dx = 1.0
    h_norm = h * (n_points / (2 * np.sum(h)))  # exact scalar formulation
    h_norm = np.clip(h_norm, 0.0, 1.0)  # enforce [0,1] constraints
    h = h_norm.copy()

    # Optimization loop parameters
    max_iter = 200
    step_size = 0.1
    eval_interval = 5
    best_c5 = np.inf
    best_h = None
    for t in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        estimate_h = np.array(h, copy=True)
        overlaps = correlate(estimate_h, 1 - estimate_h, mode='full')
        overlaps *= dx
        idx = np.argmax(overlaps)
        shift = idx - n_points + 1  # k in the Erdős problem
        if 0 <= shift < n_points * 2:  # valid k range
            # Derivative of overlap at current h[i] for active indices vs inactive
            delta = np.zeros_like(h)
            for i in range(n_points - shift):
                delta[i] -= (1 - estimate_h[i + shift])
            for i in range(n_points - shift):
                delta[i] -= (1 - estimate_h[i + shift]) * ((1 - estimate_h[i + shift]) - estimate_h[i + shift])
            delta *= dx

            # Gradient projection step
            h_proposed = h + step_size * delta / np.max(np.abs(delta) + 1.e-5)

            # Projection step: clip h to [0,1], then rescale to maintain ∫ = 1
            h_proposed = np.clip(h_proposed, 0.0, 1.0)
            h_proposed = h_proposed * (n_points / (2 * np.sum(h_proposed)))  # exact normalization
            h_proposed = np.clip(h_proposed, 0.0, 1.0)  # re-clip after rescaling

            h = h_proposed

        # Prune stale evaluation (every eval_interval)
        if t % eval_interval == 0:
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                print(f"Iter {t}, Elapsed: {elapsed:.1f}s, C5: {c5:.6f}")

    # Final cleanup
    final_h = best_h if best_h is not None else h
    final_c5 = evaluate_erdos_solution(final_h)
    return (final_h.tolist(), final_c5, n_points)
