import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    # Projection onto feasible set: Box constraint and integral constraint
    def project_feasible(h):
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if abs(current_sum - n_points / 2) > 1e-8:
            delta = (n_points / 2 - current_sum)
            h += delta / n_points
            h = np.clip(h, 0, 1)
        return h

    # Compute the overlap maximum and its index
    def compute_C5(h_vec):
        ones_minus_h = 1 - h_vec
        corrs = correlate(h_vec, ones_minus_h, mode='full') * dx
        max_corr = np.max(corrs)
        max_idx = np.argmax(corrs)
        return max_corr, max_idx

    start_time = time.time()
    alpha = 0.1
    t = 0
    h_best = h.copy()
    c5_best, k_best = compute_C5(h)

    while time.time() - start_time < budget_s:
        t += 1
        # Compute gradient at the current maximum overlap shift
        h_vec = h.copy()
        max_corr, k = compute_C5(h_vec)

        if max_corr < c5_best:
            c5_best = max_corr
            h_best = h.copy()

        # Compute gradient of C_5 at the maximizing shift k
        grad = np.zeros(n_points)
        for i in range(n_points - abs(k)):
            if k >= 0:
                idx_h = i
                idx_1h = i + k
            else:
                idx_h = i - k
                idx_1h = i
            grad[idx_h] += (1.0 - h[idx_1h]) * dx
            grad[idx_1h] -= h[idx_h] * dx

        h = project_feasible(h - alpha * grad)

        # Perturb every few steps to avoid local optima
        if t % 50 == 0:
            h += np.random.normal(0, 0.01, n_points)
            h = project_feasible(h)
            alpha *= 0.5

    final_corr, _ = compute_C5(h_best)
    return (h_best, final_corr, n_points)
