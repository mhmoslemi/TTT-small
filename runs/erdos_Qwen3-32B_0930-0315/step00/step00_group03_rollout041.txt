import numpy as np
import time
import random
from scipy.optimize import linprog
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)

    # Discretization parameters
    n_points = 70 if "initial_h_values" in globals() else 70

    if "initial_h_values" in globals():
        h_start = initial_h_values
    else:
        # Build a simple initial step function
        h_start = np.zeros(n_points, dtype=float)
        h_start[::2] = 1.0
        h_start *= (n_points / 2.0) / np.sum(h_start)

    # Genetic Algorithm settings
    pop_size = 50
    mutation_rate = 0.15
    tournament_size = 5
    generations = 0
    max_generations = 2000
    population = [h_start.copy()]

    # Initialize the population
    def create_individual():
        h = np.random.rand(n_points)
        return h / h.sum() * (n_points / 2.0)

    for _ in range(pop_size - 1):
        population.append(create_individual())

    def evaluate(h):
        dx = 2.0 / n_points
        h_flt = h.astype(float)
        correlation = np.correlate(h_flt, 1.0 - h_flt, mode="full")
        C5 = np.max(correlation) * dx
        return C5

    def fitness(h):
        return evaluate(h)

    def clip_and_normalize(h):
        h_clipped = np.clip(h.astype(float), 0, 1)
        total = np.sum(h_clipped)
        if abs(total - n_points / 2.0) > 1e-9:
            scale = (n_points / 2.0) / total if total != 0 else 1.0
            h_clipped = np.clip(h_clipped * scale, 0, 1)
        return h_clipped

    def crossover(parent1, parent2):
        idx = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:idx], parent2[idx:]])
        child2 = np.concatenate([parent2[:idx], parent1[idx:]])
        return [clip_and_normalize(child1), clip_and_normalize(child2)]

    def mutate(h):
        mutated_h = h.copy()
        num_mutation = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, size=num_mutation, replace=False)
        mutated_h[indices] += np.random.normal(0, 0.1, num_mutation)
        mutated_h = np.clip(mutated_h, 0, 1)
        return clip_and_normalize(mutated_h)

    def tournament_selection(pop):
        contestants = [random.choice(pop) for _ in range(tournament_size)]
        best = min(contestants, key=fitness)
        return best.copy()

    best_h = h_start.copy()
    best_c5 = evaluate(best_h)

    while time.time() - start_time < budget_s and generations < max_generations:
        new_population = []
        # Selection and crossover
        for _ in range(pop_size // 2):
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            child1, child2 = crossover(parent1, parent2)
            new_population.extend([child1, child2])

        # Mutation
        for i in range(len(new_population)):
            if random.random() < mutation_rate:
                new_population[i] = mutate(new_population[i])

        # Fitness evaluations
        population = population + new_population
        population = [clip_and_normalize(h) for h in population]
        population.sort(key=fitness)
        population = population[:pop_size]

        # Update best solution
        current_best = min(population, key=fitness)
        current_best_c5 = fitness(current_best)
        if current_best_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_best_c5

        generations += 1
        if generations % 100 == 0:
            print(f"Generation {generations}, Best C5 = {best_c5:.6f}")

    # Final validation
    best_h = clip_and_normalize(best_h)
    assert abs(np.sum(best_h) - n_points / 2.0) < 1e-7, "Constraint violation: mass"
    assert np.all((best_h >= 0.0) & (best_h <= 1.0)), "Constraint violation: domain"

    return best_h, best_c5, n_points
