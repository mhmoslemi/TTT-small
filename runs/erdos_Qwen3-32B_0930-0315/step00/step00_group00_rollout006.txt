import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100, **kwargs):
    # Initialization
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the given parent construction if available
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Default construction: binary-like distribution with exact constraint
        n_points = 45
        h = np.random.rand(n_points)
        h = np.where(h > 0.5, 1.0, 0.0)
        h = h * (n_points / 2) / h.sum()  # Enforce constraint
    
    # Normalize h to ensure sum constraint
    h *= (n_points / 2) / h.sum()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)[1]

    # Tabu parameters
    tabu_size = 15
    tabu_memory = []
    n_neighbors = 20
    improvement_threshold = 1e-5
    no_impv_bound = 1000  # Threshold for restarting with perturbation

    # Helper: Compute the objective function value C₅
    def compute_c5(h):
        dx = 2.0 / n_points
        c5 = np.max(scipy.signal.correlate(h, 1 - h, mode='full')) * dx
        return c5

    # Helper: Perturb h with constraints maintained
    def perturb(h):
        h = h.copy()
        idxs = np.random.choice(n_points, np.random.randint(1, 6), replace=False)
        for i in idxs:
            h[i] += np.random.uniform(-0.2, 0.2)
            h[i] = np.clip(h[i], 0.0, 1.0)

        # Project to constraint surface sum(h) = n_points / 2
        h *= (n_points / 2) / h.sum()
        return h

    # Helper: Perturb h by shifting values left/right
    def shift_perturb(h):
        h = h.copy()
        shift = np.random.choice([-1, 1]) * np.random.rand()
        h = np.roll(h, np.random.randint(-3, 4))  # Wrap around rolling
        h += shift
        h = np.clip(h, 0.0, 1.0)
        h *= (n_points / 2) / h.sum()
        return h

    # Helper: Check if the state is in tabu memory
    def is_tabu(h):
        return any(np.allclose(h, mem, atol=1e-4) for mem in tabu_memory)

    # Main search loop
    iterations = 0
    no_impv_count = 0

    while time.time() - start_time < budget_s:
        # Generate neighbors
        neighbors = []
        for _ in range(n_neighbors):
            if np.random.rand() > 0.5:
                neighbor = perturb(h)
            else:
                neighbor = shift_perturb(h)
            neighbors.append(neighbor)

        # Evaluate and select best non-tabu or tentative improvement
        best_neighbor = None
        best_c5_current = best_c5

        for neighbor in neighbors:
            if is_tabu(neighbor):
                continue
            c5_current = compute_c5(neighbor)
            if c5_current < best_c5_current:
                best_neighbor = neighbor
                best_c5_current = c5_current

        if best_neighbor is not None:
            h = best_neighbor
            if best_c5_current < best_c5:
                best_h = h.copy()
                best_c5 = best_c5_current
                no_impv_count = 0
            else:
                no_impv_count += 1
        else:
            # Accept a tabu move if we get stuck
            h = neighbors[np.random.randint(len(neighbors))]
            no_impv_count += 1
            if no_impv_count > no_impv_bound:
                h = perturb(h)

        # Update tabu memory
        tabu_memory.append(h.copy())
        if len(tabu_memory) > tabu_size:
            tabu_memory.pop(0)

        iterations += 1

    # Return best result
    return (best_h.tolist(), best_c5, n_points)
