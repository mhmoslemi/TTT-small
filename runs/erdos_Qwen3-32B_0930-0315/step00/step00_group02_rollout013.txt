import numpy as np
import math
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Ensure initial_h_values exists
    if 'initial_h_values' not in globals():
        raise ValueError("initial_h_values is not available")
    
    # Discretization and initialization

    n_half = n // 2
    dx = 2.0 / n
    h = np.clip(np.array(initial_h_values, dtype=np.float64), 0.0, 1.0)
    h *= (n_half / np.sum(h))  # normalize to exact constraint

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h)
    
    def project_l1(h):
        # Enforce the integral constraint that sum(h) == n//2
        s = np.sum(h)
        if abs(s - n_half) > 1e-8:
            h *= (n_half / s)
        # Enforce box constraint [0, 1]
        h = np.clip(h, 0.0, 1.0)
        return h

    def compute_c5(h):
        # Use the correlation function to find the max overlap
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    best_c5 = compute_c5(h)
    
    # Optimization loop
    iter_count = 0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stall_counter = 0
    prev_best_c5 = best_c5

    while time.time() - start_time < budget_s:
        # Compute correlation and find max overlap shift
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient for the worst shift
        g = np.zeros(n)
        for i in range(n):
            i_k_max = (i + k_max) % n
            if k_max <= n:
                g[i] = (1 - h[i_k_max]) - h[i]  # contribution to gradient
        
        # Barzilai-Borwein step size
        delta_h = h - np.roll(h, shift=k_max)
        delta_g = -2 * (corr[k_max] / n) * np.roll(h, shift=k_max)
        if np.sum(delta_g**2) > 0:
            alpha = np.dot(delta_h, delta_g) / np.sum(delta_g**2)
        else:
            alpha = 1.0

        # Gradient step
        h -= alpha * g
        h = project_l1(h)

        # After 500 iterations, perturb if no improvement
        iter_count += 1
        if iter_count % 500 == 0 or iter_count == 1:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = np.copy(h)
                stall_counter = 0
                prev_best_c5 = c5
            else:
                stall_counter += 1
                if stall_counter >= 3:
                    # Perturb to escape local minimum
                    perturb_indices = np.random.choice(n, size=min(10, n), replace=False)
                    noise = np.random.randn(len(perturb_indices)) * 0.1
                    h[perturb_indices] += noise
                    h = project_l1(h)
                    stall_counter = 0
        
        # Respect budget
        if time.time() - start_time >= budget_s:
            break

    return best_h, best_c5, best_h.size
