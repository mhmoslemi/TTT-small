import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    def constraint_sum(h):
        return h.sum() - n_points / 2

    def cost(h):
        h = h.reshape(-1)
        one_minus_h = 1 - h
        overlap = correlate(h, one_minus_h, mode="full") * dx
        return overlap.max()

    def callback(x):
        nonlocal start_time, h, c5_best, time_best
        h = x
        current_c5 = cost(h)
        if current_c5 < c5_best:
            c5_best = current_c5
            time_best = time.time()
            print(f"Improved C5: {c5_best:.8f} at time {time.time() - start_time:.1f}s")
        return

    # Ensure constraint is maintained
    clip_and_project = lambda x: np.clip((x * n_points / 2) / x.sum(), 0, 1)

    # Initial projection
    h = clip_and_project(h)
    c5_best = cost(h)
    time_best = start_time
    best_h = h.copy()

    def objective_with_penalty(h_flat):
        h = h_flat.reshape((-1,))
        obj = cost(h)
        # Penalty to maintain the average constraint
        penalty = (h.sum() - n_points / 2) ** 2 * 1e4
        return obj + penalty

    def objective(h_flat, return_value=False):
        h = h_flat.reshape((-1,))
        constraint_val = constraint_sum(h)
        # Return scalar constraint violation only for optimizers
        if not return_value:
            return constraint_val
        else:
            obj = cost(h)
            return (constraint_val,), obj

    bounds = [(0, 1)] * n_points

    # Optimization loop
    max_iter_per_pass = 500
    perturb_every = 50
    perturb_strength = 0.01

    for i in range(1000):
        if time.time() - start_time > budget_s:
            break
        if (i % 20 == 0 or i == 0):
            # Occasionally perturb
            perturbed_h = h + np.random.uniform(-perturb_strength, perturb_strength, n_points)
            perturbed_h = clip_and_project(perturbed_h)
            h = perturbed_h

        res = minimize(
            fun=objective_with_penalty,
            x0=h,
            bounds=bounds,
            method='L-BFGS-B',
            callback=callback
        )
        h = res.x
        h = h.reshape((-1,))
        h = clip_and_project(h)

    h = clip_and_project(h)
    c5_final = cost(h)

    print(f"Final time elapsed: {time.time() - start_time:.1f}s")
    print(f"Best C5 found: {c5_best}")
    print(f"Final C5: {c5_final}")
    print(f"Using n_points = {n_points}")
    return h, c5_best, n_points
