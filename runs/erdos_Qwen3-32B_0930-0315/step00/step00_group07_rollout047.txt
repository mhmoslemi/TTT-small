import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    h_correlate = np.correlate(h, (1 - h), mode="full")
    n = len(h)
    return np.max(h_correlate[:2 * n]) * dx

def optimize_c5(n_points, h0, max_time=1100, tol=1e-6):
    dx = 2.0 / n_points
    h = np.array(h0, dtype=np.float64)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    
    t_start = time.time()
    
    while time.time() - t_start < max_time:
        # Compute all lags
        C = np.correlate(h, (1 - h), mode="full")[:2 * n_points]
        worst_k = np.argmax(C)
        worst_C = C[worst_k]
        
        # Shift indices to [0, n_points]
        k = worst_k % n_points
        
        # Define variables for the subproblem
        delta = cp.Variable(n_points)
        constraints = [
            h + delta >= 0,
            h + delta <= 1,
            cp.sum(delta) == 0
        ]
        
        # Construct the overlap objective for lag k
        overlap = 0
        for i in range(n_points):
            delta_i = delta[i]
            delta_j = delta[(i + k) % n_points] if k <= n_points else delta[(i - n_points) % n_points]
            overlap += h[i]*(1 - h[(i + k) % n_points]) - h[i]*delta_j - delta_i*(1 - h[(i + k) % n_points])
        
        obj = cp.Minimize(cp.Constant(overlap * dx))
        problem = cp.Problem(obj, constraints)
        
        try:
            problem.solve(solver=cp.ECOS, max_iters=100)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h = np.clip(h + delta.value, 0.0, 1.0)
                h = h / np.sum(h) * (n_points / 2)
                
                current_c5 = evaluate_erdos_solution(h, n_points)
                if current_c5 < best_c5 - 1e-7:
                    best_h[:] = h
                    best_c5 = current_c5
        except Exception:
            pass  # recovery strategy: skip on failure, but current h remains valid
        
        if abs(best_c5) < 0.380876:
            break
            
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    if hasattr(initial_h_values, '__iter__'):
        h0 = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h0)
    else:
        n_points = 80
        h0 = np.ones(n_points) / 2

    h, c5 = optimize_c5(n_points, h0, max_time=budget_s)
    
    # Validate constraints
    h = np.clip(h, 0.0, 1.0)
    h = h / np.sum(h) * (n_points / 2)
    valid_c5 = evaluate_erdos_solution(h, n_points)
    
    return h.tolist(), valid_c5, n_points
