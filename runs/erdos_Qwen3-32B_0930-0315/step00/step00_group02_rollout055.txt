import numpy as np
import cvxpy as cp
from time import time
import evaluate_erdos_solution

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    dx = 2.0 / len(initial_h_values)
    h = np.copy(initial_h_values)
    h = h / np.sum(h) * (len(h) / 2)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = time()

    n = len(h)

    def compute_c5(h):
        if h.ndim == 1:
            hh = h[:, np.newaxis]
            hhk = hh * np.kron(np.ones((1, n)), hh.T)
            h1 = 1.0 - hhk
        out = np.zeros(2 * n - 1)
        for k in range(n):
            idx1 = np.arange(n)
            idx2 = (idx1 + k) % n
            out[k] = np.sum(h[idx1] * h1[idx1, idx2])
        for k in range(n, 2 * n - 1):
            k_flip = 2 * n - 1 - k
            out[k] = out[k_flip]
        out *= dx
        return np.max(out)

    def project(h):
        h = np.clip(h, 0, 1)
        while not np.allclose(np.sum(h), n / 2):
            s = np.sum(h)
            if s > n / 2:
                threshold = np.sort(h)[::-1][int(np.ceil(n / 2))]
                h[h > threshold] = threshold
            else:
                h += (n / 2 - s) / n
        return h

    while time() - start_time < budget_s:
        c5_vals = np.zeros(n)
        for k in range(n):
            idx1 = np.arange(n)
            idx2 = (idx1 + k) % n
            c5_vals[k] = np.sum(h[idx1] * (1 - h[idx2]))
        c5_vals *= dx
        max_idx = np.argmax(c5_vals)

        grad = np.zeros(n)
        for i in range(n):
            grad[i] += (1 - h[(i + max_idx) % n])
            grad[i] -= h[(i - max_idx) % n]
        grad *= dx

        h = h - 0.01 * (grad)
        h = project(h)

        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    return best_h, best_c5, n
