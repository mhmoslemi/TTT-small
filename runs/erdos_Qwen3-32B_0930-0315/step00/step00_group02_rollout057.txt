import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Load initial solution if available
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = np.array(initial_h_values, dtype=float)
    else:
        # If no parent, initialize with a plateau profile(cls)
        n_points = 160  # Fixed for balance between accuracy and speed
        h = np.ones(n_points) * 0.5

    # Normalize initial h to satisfy the sum constraint
    h *= n_points / 2 / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float("inf")

    max_iter = 600
    patience = 100
    alpha = 1e-1
    alpha_decay = 0.5
    improvement = 1.0

    # Perform iterative gradient-based optimization
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Compute the overlaps at all lags
        overlaps = correlate(h, 1 - h, mode="full") * dx

        k_max = np.argmax(overlaps)

        # Compute gradient ∇C at current lag
        gradient = np.zeros_like(h)
        for j in range(n_points):
            idx_plus = (j + k_max) % n_points
            idx_minus = (j - k_max) % n_points
            gradient[j] = (1 - h[idx_plus]) - h[idx_minus]

        # Gradient update
        h_new = h - alpha * gradient

        # Projection to [0,1]
        h_new = np.clip(h_new, 0, 1)

        # Projection to maintain sum constraint
        total = np.sum(h_new)
        if total != 0:
            h_new *= (n_points / 2) / total

        # Compute new C5
        c5_new = np.max(correlate(h_new, 1 - h_new, mode="full") * dx)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            improvement = 1.0
        else:
            improvement -= 1e-2
            if improvement <= 0:
                alpha *= alpha_decay
                improvement = 1.0

        h = h_new.copy()

        # Occasionally add small noise to escape local optima
        if i % 50 == 0 and np.random.rand() < 0.2 and n_points > 20:
            noise = np.random.normal(0, 0.01, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            h *= (n_points / 2) / total

    return best_h, best_c5, n_points
