import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    n_points = 70
    dx = 2.0 / n_points

    # Use the global initial_h_values when it exists
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        h = np.random.rand(n_points)
        h = h / h.sum() * (n_points / 2)  # Normalize to density

    t0 = time.time()
    best_h = h.copy()
    best_c5 = evaluate_overlap_integral(h)

    for it in range(1000):
        # Safety check on time
        if time.time() - t0 > budget_s:
            break

        # Create flattened problem with overlap term at top lags
        h_var = cp.Variable(n_points)
        overlap = []

        for k in range(n_points):
            # Compute current overlap at lag k in numpy manually
            if k < n_points:
                lagged = np.zeros_like(h)
                lagged[k:] = 1 - h[:-k]
                overlap_val = h[:n_points - k] @ lagged[:n_points - k] * dx
                overlap.append(overlap_val)
            else:
                lagged = np.zeros_like(h)
                lagged[:n_points - (k - n_points)] = 1 - h[k - n_points:]
                overlap_val = h[k - n_points:] @ lagged[k - n_points:] * dx
                overlap.append(overlap_val)

        max_overlap = np.max(overlap)

        # Formulate problem with bounds and integral constraint
        obj = cp.Minimize(0)  # Dummy
        constraints = [
            0 <= h_var, h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]

        try:
            prob = cp.Problem(obj, constraints)
            result = prob.solve(solver=cp.CVXOPT, verbose=False)
        except:
            h_var.value = h

        # Update h from solution
        h = np.clip(h_var.value, 0, 1)
        h = h / h.sum() * n_points / 2  # Project to integral constraint

        # Evaluate and update best solution
        current_c5 = evaluate_overlap_integral(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h

        # Occasionally perturb for better escape from local optima
        if it % 10 == 0:
            h += 0.001 * np.random.randn(n_points)  # Small random noise
            h = np.clip(h, 0, 1)
            h = h / h.sum() * n_points / 2  # Project to constraint

    return best_h, best_c5, n_points

def evaluate_overlap_integral(h):
    n_points = len(h)
    dx = 2.0 / n_points
    k_values = range(-n_points + 1, n_points)
    overlaps = correlate(h, 1 - h, mode='full') * dx
    return np.max(overlaps)
