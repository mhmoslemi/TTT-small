import numpy as np
import cvxpy as cp
import time
import collections

def evaluate_C5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr *= dx
    return np.max(corr)

def compute_redundancy(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    R = np.zeros(n)
    win_size = int(0.2 * n)
    for i in range(n):
        start_k = max(0, i - win_size)
        end_k = min(len(corr), i + win_size + 1)
        R[i] = np.sum(np.abs(corr[start_k:end_k]))
    return R

def refine_step(x, step=0.1):
    x_dir = np.round(x / step) * step
    return np.clip(x_dir, 0, 1)

def feasibility_projection(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    h_clipped *= (n_points / 2.0) / np.sum(h_clipped)
    return h_clipped
