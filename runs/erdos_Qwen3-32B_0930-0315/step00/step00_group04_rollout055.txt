import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    n_points = 42
    dx = 2.0 / n_points
    h0 = np.array(initial_h_values)
    h0_maker = h0.copy()

    # Ensure feasibility
    h0_maker = np.clip(h0_maker, 0, 1)
    area = np.sum(h0_maker) * dx
    if area > 0:
        h0_maker *= 1 / area * (n_points / 2)

    best_h = h0_maker.copy()
    best_c5 = np.inf

    h = cp.Variable(n_points)
    objective = cp.Minimize(cp.max(
        cp.conv(cp.convolution(h, 1 - h, mode="full") * dx)
    ))
    constraints = [
        0 <= h,
        h <= 1,
        cp.sum(h) * dx == 1
    ]

    problem = cp.Problem(objective, constraints)

    total_budget = budget_s
    iter_max = int(200)
    for iter in range(iter_max):
        t0 = time.time()
        h_init = best_h + 0.01 * np.random.rand(*best_h.shape) - 0.005
        h_init = np.clip(h_init, 0, 1)
        area = np.sum(h_init) * dx
        if area > 0:
            h_init *= 1 / area * (n_points / 2)

        problem.solve(warm_start=True, solver=cp.CVXOPT, verbose=False, maxiters=50, kktsolver="lg")
        if problem.status in ["optimal", "optimal_inaccurate"]:
            h_current = np.array(h.value).flatten()
            h_current = np.clip(h_current, 0, 1)
            area = np.sum(h_current) * dx
            if area > 0:
                h_current *= 1 / area * (n_points / 2)
            c5_current = np.max(np.correlate(h_current, 1 - h_current, mode='full') * dx)
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current

        if time.time() - t_start > total_budget - 10:
            break

    return best_h, best_c5, n_points
