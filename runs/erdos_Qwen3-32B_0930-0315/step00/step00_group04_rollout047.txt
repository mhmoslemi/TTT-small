import numpy as np
import time
from scipy.interpolate import interp1d
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')

    def evaluate_c5(h, n):
        dx = 2.0 / n
        result = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(result)

    def rescale_h(h):
        # Clamp to [0, 1]
        h_clamped = np.clip(h, 0, 1)
        # Rescale to have sum = n / 2
        current_sum = h_clamped.sum()
        if current_sum == 0:
            return h_clamped
        scale_factor = (len(h_clamped) / 2) / current_sum
        return h_clamped * scale_factor

    def repair_h(h):
        return rescale_h(np.clip(h, 0, 1))

    def genetic_algorithm(n, pop_size=20, generations=200, mutation_rate=0.2):
        nonlocal best_h, best_c5
        population = []
        for _ in range(pop_size):
            h = np.random.rand(n) * 2  # Initial values in [0, 2]
            h = repair_h(h)
            population.append(h)
        population = np.array(population)

        for gen in range(generations):
            if time.time() - start_time > budget_s * 0.9:
                break

            # Evaluate objective for all individuals
            c5_scores = np.array([evaluate_c5(h, n) for h in population])
            min_idx = c5_scores.argmin()
            current_best = population[min_idx]
            current_best_c5 = c5_scores[min_idx]
            if current_best_c5 < best_c5:
                best_h = current_best.copy()
                best_c5 = current_best_c5

            # Tournament selection
            selected_indices = []
            for _ in range(pop_size // 2):
                idx = np.random.choice(pop_size, 5)
                selected = idx[c5_scores[idx].argmin()]
                selected_indices.append(selected)
            selected_indices = np.array(selected_indices)

            # Crossover
            offspring = []
            for i in range(0, len(selected_indices), 2):
                if i + 1 >= len(selected_indices):
                    continue

                parent1 = population[selected_indices[i]]
                parent2 = population[selected_indices[i + 1]]
                crossover_point = np.random.randint(1, n - 1)
                child1 = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
                child2 = np.concatenate((parent2[:crossover_point], parent1[crossover_point:]))
                offspring.append(repair_h(child1))
                offspring.append(repair_h(child2))

            offspring = np.array(offspring)
            population[:len(offspring)] = offspring

            # Mutation
            for i in range(pop_size):
                if np.random.rand() < mutation_rate * (1 - gen / generations):
                    mutation_mask = np.random.rand(n) < 0.1
                    population[i] = repair_h(population[i] + np.random.rand(n) * 0.2 * mutation_mask)

    # Start with coarse resolution
    n_points = 14
    if 'initial_h_values' in globals():
        init_h = initial_h_values[:14] if len(initial_h_values) >= 14 else np.array(initial_h_values)
        init_h = np.clip(init_h, 0, 1)
        init_h = init_h / init_h.sum() * 14 / 2
        init_h.resize(14)
        best_h = repair_h(init_h)

    genetic_algorithm(n_points)
    if best_h is not None:
        best_c5 = evaluate_c5(best_h, n_points)

    # Adaptively refine
    while time.time() - start_time < budget_s and n_points * 2 <= 42:
        n_points *= 2
        new_h = interp1d(np.linspace(0, 2, n_points // 2), best_h, kind='cubic')(np.linspace(0, 2, n_points))
        new_h = repair_h(new_h)
        genetic_algorithm(n_points, generations=100, mutation_rate=0.1)
        if best_h is not None and evaluate_c5(best_h, n_points) < best_c5:
            best_c5 = evaluate_c5(best_h, n_points)

    if best_h is None:
        n_points = 14
        best_h = np.array([1.0] * n_points / 2.0)
        best_h = repair_h(best_h)
        best_c5 = evaluate_c5(best_h, n_points)

    return (best_h, best_c5, n_points)
