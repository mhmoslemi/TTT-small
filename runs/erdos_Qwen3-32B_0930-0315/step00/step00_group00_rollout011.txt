import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

n_points = 45
dx = 2.0 / n_points

def evaluate_C5(h):
    corr = correlate(h, 1 - h, mode="full")
    corr *= dx
    return np.max(corr)

def enforce_constraints(h):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    scale = n_points / 2.0 / h_sum
    h *= scale
    return h

def sample_random():
    h = np.random.rand(n_points)
    return enforce_constraints(h)

def tournament_select(pop, fitnesses, size=5):
    indices = np.random.choice(len(pop), size=size, replace=False)
    best_idx = np.argmin(fitnesses[indices])
    return pop[indices[best_idx]]

def crossover(parent1, parent2):
    point = np.random.randint(1, n_points - 1)
    child = np.zeros_like(parent1)
    child[:point] = parent1[:point]
    child[point:] = parent2[point:]
    return enforce_constraints(child)

def mutate(h, mutation_rate, temperature):
    if np.random.rand() < mutation_rate:
        idxs = np.random.choice(n_points, int(mutation_rate * n_points), replace=False)
        delta = np.random.normal(0, 0.1, len(idxs))
        h[idxs] += delta
        h[idxs] = np.clip(h[idxs], 0, 1)
    if temperature > 1e-6:
        idx = np.random.choice(n_points)
        h_new = h.copy()
        h_new[idx] = np.random.rand()
        h_new = enforce_constraints(h_new)
        new_C5 = evaluate_C5(h_new)
        current_C5 = evaluate_C5(h)
        if new_C5 < current_C5 or np.random.rand() < np.exp((current_C5 - new_C5) / temperature):
            h[:] = h_new[:]
    return h

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    pop_size = 75
    mutation_rate = 0.05
    crossover_prob = 0.7
    tournament_size = 5
    init = global_initial_h_values
    pop = [enforce_constraints(init)]
    for _ in range(pop_size - 1):
        pop.append(enforce_constraints(sample_random()))
    best_h = None
    best_C5 = np.inf
    T = 100.0
    cooldown_rate = 0.95
    gen = 0
    cooldown_count = 50
    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitnesses = np.array([evaluate_C5(h) for h in pop])
        for i in range(pop_size):
            if fitnesses[i] < best_C5:
                best_C5 = fitnesses[i]
                best_h = pop[i]
        new_pop = [best_h]
        i = 1
        while i < pop_size:
            parent1 = tournament_select(pop, fitnesses, size=tournament_size)
            parent2 = tournament_select(pop, fitnesses, size=tournament_size)
            if np.random.rand() < crossover_prob:
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()
            child = mutate(child, mutation_rate, T)
            new_pop.append(enforce_constraints(child))
            i += 1
        pop = new_pop
        T *= cooldown_rate
        gen += 1
        if gen % cooldown_count == 0:
            mutated_best = best_h.copy()
            mutate(mutated_best, mutation_rate, T)
            mutated_best = enforce_constraints(mutated_best)
            mutated_C5 = evaluate_C5(mutated_best)
            if mutated_C5 < best_C5:
                best_h = mutated_best
                best_C5 = mutated_C5
    return best_h, best_C5, n_points
