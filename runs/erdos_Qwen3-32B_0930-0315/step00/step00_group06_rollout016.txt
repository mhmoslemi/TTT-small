import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # Parameters
    n_points = 256  # higher precision
    dx = 2.0 / n_points

    # Initialize h to uniform or use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.tile(initial_h_values, n_points // len(initial_h_values))[:n_points]
    else:
        h_init = np.zeros(n_points)
        h_init[n_points//2 - 64 : n_points//2 + 64] = 1.0 / 64 * np.linspace(0, 1, 128)
    h = h_init.astype(np.float64)  # constraint: sum(h) == n_points / 2
    h *= (n_points / 2) / np.sum(h)  # ensure sum is correct

    h_opt = cp.Variable(n_points)
    constraints = [
        0 <= h_opt,
        h_opt <= 1,
        cp.sum(h_opt) == n_points / 2
    ]

    best_c5 = float('inf')
    best_h = h.copy()

    def compute_c5(h_val):
        overlaps = np.correlate(h_val, 1 - h_val, mode='full') * dx
        return np.max(overlaps)

    for iter_ in range(1000):
        if time.time() - start > budget_s - 5:
            break

        # Evaluate current h
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        c5 = np.max(overlaps)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Identify worst shift
        k_max = int(np.argmax(overlaps))
        k_index = k_max

        # Construct objective
        obj = cp.Maximize(cp.sum(cp.multiply(h_opt[k_index:], h_opt[:-k_index])))
        problem = cp.Problem(obj, constraints)
        problem.solve(solver=cp.ECOS, verbose=False)

        if problem.status in ["optimal", "optimal_inaccurate"]:
            h_update = np.clip(h_opt.value, 0, 1)
            h_update *= (n_points / 2) / np.sum(h_update)
            mu = 0.1  # line search
            h = mu * h_update + (1 - mu) * h

    best_c5 = compute_c5(best_h)

    return (best_h.tolist(), best_c5, n_points)
