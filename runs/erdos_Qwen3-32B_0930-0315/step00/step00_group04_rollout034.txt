import numpy as np
import time
import math

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    # Use provided initial_h_values
    if 'initial_h_values' in globals():
        h = np.copy(initial_h_values)
        n_points = len(h)
    else:
        n_points = 42
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2

    start_time = time.time()
    best_h = np.copy(h)
    best_c5 = float("inf")
    dx = 2.0 / n_points

    # Optimization loop
    learning_rate = 0.01
    while time.time() - start_time < budget_s:
        # Compute overlap integral
        window = 1 - h
        corr = np.correlate(h, window, mode="full") * dx
        max_idx = np.argmax(corr)

        # Get the shifted window corresponding to the most problematic overlap
        shift = max_idx - (n_points - 1)
        if shift < 0:
            pad = -shift
            window_padded = np.concatenate((window[-shift:], window[:-shift]))
        elif shift > 0:
            pad = shift
            window_padded = np.concatenate((window[shift:], window[:shift]))
        else:
            window_padded = window

        # Gradient: - window_shifted (since we minimize)
        grad = -window_padded

        # Update h with projective gradient descent
        h += learning_rate * grad

        # Projection onto [0, 1]^n and satisfying the integral constraint
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / h.sum()

        # Update best solution if we found a better one
        correlation = np.correlate(h, 1 - h, mode="full") * dx
        c5 = correlation.max()
        if c5 < best_c5:
            best_h = np.copy(h)
            best_c5 = c5

        # Learning rate decay
        if math.isclose((time.time() - start_time), budget_s - 100, abs_tol=2.0):
            learning_rate *= 0.95

    return (best_h, best_c5, n_points)
