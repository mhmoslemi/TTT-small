import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize, Bounds


def run(seed=42, budget_s=1100, **kwargs):
    # Use the global pre-imported `initial_h_values`
    h = np.copy(initial_h_values)
    n_points = h.size
    h = h.astype(np.float64)
    dx = 2.0 / n_points

    # Simple time tracking
    start_time = time.time()

    # Projection operator to keep h in [0,1] and maintain the integral constraint
    def project_h(h):
        h = np.clip(h, 0.0, 1.0)
        total = h.sum()
        if abs(total - n_points / 2) > 1e-6:
            h = cp.Variable(n_points, nonneg=True)
            constraint = [h <= 1, h.sum() == n_points / 2]
            prob = cp.Problem(cp.Minimize(cp.sum_squares(h - np.clip(h, 0, 1)))), constraint)
            if "ECOS" in cp.installed_solvers():
                prob.solve(solver=cp.ECOS)
            else:
                prob.solve()
            h = np.array(h.value).flatten()
        return h

    # Objective overlap calculation
    def compute_c5(h):
        h = np.concatenate([h, np.zeros(n_points - 1)])
        rect = h.cumsum()
        conv = rect[None, :] @ rect[:, None]
        return (conv - 2 * h @ rect[None, :])[:, n_points:] * dx

    # Optimization loop
    best_h = np.copy(h)
    best_c5 = compute_c5(best_h).max()

    h = project_h(h)

    max_iter = 10000
    iter = 0
    while time.time() - start_time < budget_s - 0.1:
        iter += 1
        # Compute overlap values and find the worst shift
        overlaps = compute_c5(h)
        k_max = np.argmax(overlaps)
        overlap_val = overlaps[k_max]

        if overlap_val < best_c5 - 1e-6:
            best_h = np.copy(h)
            best_c5 = overlap_val

        # Compute gradient for the worst shift
        gradient = np.zeros_like(h)
        for i in range(n_points - k_max):
            gradient[i] += 1 - h[i + k_max]
            gradient[i + k_max] -= h[i]
        grad_multiplier = 1.0 / np.linalg.norm(gradient)
        learning_rate = 1e-2 * grad_multiplier

        # Update h
        h -= learning_rate * gradient
        h = project_h(h)

        # Occasionally perturb to escape local optima
        if iter % 100 == 0:
            noise = np.random.randn(n_points) * 0.01
            h += noise
            h = project_h(h)

        # Early stopping if no improvement
        if iter % 100 == 0:
            current_c5 = compute_c5(h).max()
            if current_c5 >= best_c5 - 1e-8:
                break

    final_c5 = compute_c5(best_h).max()
    return (best_h, final_c5, n_points)
