import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Use initial_h_values from parent construction directly
    h = initial_h_values
    n_points = len(h)
    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5_bound = evaluate_erdos_solution(h)

    start_time = time.time()
    max_iter = 200
    iter_no_improv = 0
    max_no_improv = 20

    for it in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Compute overlaps at all lags
        h_vec = h.astype(np.float64)
        ones_minus_h = 1.0 - h_vec
        corr = np.correlate(h_vec, ones_minus_h, mode='full') * dx
        lag_mask = (np.arange(len(corr)) >= n_points) & (np.arange(len(corr)) < 2 * n_points)
        worst_lag = np.arange(len(corr))[lag_mask][np.argmax(corr[lag_mask])]
        k = worst_lag - n_points + 1

        # Gradient at worst lag k
        grad = np.zeros(n_points)
        for i in range(n_points):
            grad[i] += 1.0 - h[(i + k) % n_points]
            grad[i] -= h[(i - k) % n_points]
        grad *= dx

        # LP variable
        delta_h = cp.Variable(n_points)

        # Objective: minimize overlap at worst lag k
        obj = cp.Minimize(corr[worst_lag] + grad @ delta_h)

        # Constraints
        constraints = [
            delta_h >= -h,
            delta_h <= 1 - h,
            cp.sum(delta_h) == 0,
        ]

        # Solve
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            # Infeasible or time out, do not update
            iter_no_improv += 1
            continue

        # Update h if feasible
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_new = h + delta_h.value
            h_new = np.clip(h_new, 0.0, 1.0)
            if abs(np.sum(h_new) - n_points / 2) > 1e-6:
                weight_scale = (n_points / 2) / np.sum(h_new)
                h_new *= weight_scale
                h_new = np.clip(h_new, 0.0, 1.0)
            h = h_new

            # Evaluate C₅
            c5_bound = evaluate_erdos_solution(h)
            if c5_bound < best_c5_bound:
                best_h = h.copy()
                best_c5_bound = c5_bound
                iter_no_improv = 0
            else:
                iter_no_improv += 1

        else:
            iter_no_improv += 1

        if iter_no_improv >= max_no_improv:
            break

    # Feasibility check & clean up
    best_h = np.clip(best_h, 0.0, 1.0)
    total_mass = np.sum(best_h)
    correction = (n_points / 2.0 - total_mass) / n_points
    best_h += correction
    best_h = np.clip(best_h, 0.0, 1.0)

    return best_h.tolist(), best_c5_bound, n_points
