import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    c5_best = float('inf')
    h_best = None
    n_points = 160
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = initial_h_values[:n_points]
        h = h / h.sum() * (n_points / 2)
        h = np.clip(h, 0, 1)
    else:
        # Create a plateau initialization
        h = np.zeros(n_points)
        center = n_points // 2
        for i in range(n_points):
            dist = abs(i - center)
            h[i] = 1.0 if dist <= 20 else 0.5 if dist <= 30 else 0.0
        h_sum = h.sum()
        h = h / h_sum * (n_points / 2)

    def optimize_h(h):
        nonlocal dx, n_points
        t_start = time.time()
        rate = 0.01
        best_h = h.copy()
        best_c5 = evaluate_c5(h, dx)
        niters = 0
        timeout_margin = 5

        while time.time() - t_start < budget_s - timeout_margin:
            niters += 1
            lag, c5 = find_kmax(h, dx)
            gradient = np.zeros(n_points)
            for i in range(n_points):
                j1 = (i - lag) % n_points
                j2 = (i + lag) % n_points
                gradient[i] = (1 - h[j1]) - h[j2]
            h = np.clip(h - rate * gradient, 0, 1)
            h_sum = h.sum()
            h = h / h_sum * (n_points / 2)  # Projection onto sum constraint

            new_c5 = evaluate_c5(h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()

            if (new_c5 > best_c5 - 1e-6) or (abs(best_c5 - new_c5) < 1e-6):
                rate *= 0.9  # reduce learning rate if no improvement
            if rate < 1e-5:
                break
            if np.isnan(h).any():
                break
        return best_h, best_c5

    def find_kmax(h, dx):
        full_corr = correlate(h, 1 - h, mode='full')
        lag = (np.arange(len(full_corr)) - len(h) + 1) // 1  # integer lags, sampled at dx
        val_at_lag = full_corr * dx
        return lag[np.argmax(val_at_lag)], max(val_at_lag)

    def evaluate_c5(h, dx):
        full_corr = correlate(h, 1 - h, mode='full') * dx
        return max(full_corr)

    h_opt, c5_opt = optimize_h(h)

    # Validate and return
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt / h_opt.sum() * (n_points / 2)
    return h_opt, c5_opt, n_points
