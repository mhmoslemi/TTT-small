import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    
    # Use the provided initial_h_values if available
    try:
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
        dx = 2.0 / n_points
        vol = n_points / 2
    except NameError:
        # No initial_h_values provided, initialize randomly
        n_points = 47  # medium-sized, as suggested in the task
        dx = 2.0 / n_points
        vol = n_points / 2
        np.random.seed(seed)
        h = np.random.uniform(0, 1, n_points)
        h *= vol / h.sum()
    
    learning_rate = 0.1
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    while time.time() - start_time < budget_s:
        # Evaluate the current h
        c5 = evaluate_erdos_solution(h)
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            learning_rate *= 1.01  # increase learning rate slightly for faster progress
        else:
            learning_rate *= 0.95  # decrease learning rate slightly
        
        # Compute the correlation with itself shifted
        cor = correlate(h, 1 - h, mode="full")
        cor *= dx
        k_max = np.argmax(cor)
        
        # Compute gradient
        grad = np.zeros(n_points)
        if k_max < n_points:
            grad[:n_points - k_max] = 1 - h[k_max:]
            grad[:n_points - k_max] -= h[:n_points - k_max]
        else:
            shift = k_max - n_points + 1
            grad[shift:] = 1 - h[:n_points - shift]
            grad[shift:] -= h[shift:]
        
        # Apply gradient step
        h -= learning_rate * grad
        
        # Project onto [0, 1]
        h = np.clip(h, 0.0, 1.0)
        
        # Rescale to maintain volume constraint
        h *= vol / h.sum()
        
        # Early stopping if converging
        if best_c5 < 0.38085:
            break
    
    # Return the best solution found
    return (best_h.tolist(), best_c5, n_points)
