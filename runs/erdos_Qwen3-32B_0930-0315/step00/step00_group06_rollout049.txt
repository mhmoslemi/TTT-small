import numpy as np
import cvxpy as cp
import time


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Define basic parameters
    n_points = 256
    dx = 2.0 / n_points
    total_mass = n_points / 2.0

    # Copy from initial_h_values, or initialize uniformly
    if 'initial_h_values' in globals():
        h_init = initial_h_values[:]
        h = np.array(h_init, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        h[np.random.choice(n_points, size=int(n_points / 2), replace=False)] = 1.0

    # Normalize h to ensure the integral is correct
    while abs(np.sum(h) - total_mass) > 1e-6:
        rescale_factor = total_mass / np.sum(h)
        h = h * rescale_factor
        h = np.clip(h, 0.0, 1.0)

    max_iter = 1000
    best_C5 = float('inf')
    best_h = h.copy()

    start_time = time.time()
    for it in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

        # Compute correlations
        corr = np.correlate(h, 1 - h, mode='full')
        C5 = np.max(corr) * dx

        # Update best result
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()

        # Identify the worst shift k with max overlap
        overlapping = np.correlate(h, 1 - h, mode='full') * dx
        k_max = np.argmax(overlapping)

        # Truncate the full convolution to only relevant shifts
        k = k_max
        if k > n_points:
            continue

        # Compute the gradient w.r.t h
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if i - k >= 0:
                gradient[i] += 1.0 - h[i - k]
            if i + k < n_points:
                gradient[i] += h[i + k]
        gradient *= (2 * h - 1)
        gradient *= (2 * h - 1)

        # Gradient descent step with clipping and projection
        step_size = 0.01
        h -= step_size * gradient

        # Projection step to [0,1]^n
        h = np.clip(h, 0.0, 1.0)

        # Normalize mass constraint
        current_sum = np.sum(h)
        if abs(current_sum - total_mass) > 1e-5:
            h = h * (total_mass / current_sum)

    return (best_h.tolist(), best_C5, n_points)
