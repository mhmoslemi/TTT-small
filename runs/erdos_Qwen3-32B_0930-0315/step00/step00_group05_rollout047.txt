import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **_):
    start_time = time.time()
    np.random.seed(seed)

    dx = 2.0 / 70
    initial_h, n_points = initial_h_values, 70
    h = np.array(initial_h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h, lambda_penalty=1e5):
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if not np.isclose(h_sum, n_points / 2, atol=1e-6):
            penalty = lambda_penalty * (h_sum - n_points / 2)**2
        else:
            penalty = 0.0
        corr = correlate(h, 1 - h, mode='full') * dx
        return np.max(corr) + 2000 * penalty

    n_points = len(h)
    dx = 2.0 / n_points

    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s:
        res = minimize(objective, h, bounds=bounds, jac=None, method='L-BFGS-B', options={'maxiter': 30})
        h = np.clip(res.x, 0, 1)
        h /= h.sum() * 2.0
        h = np.clip(h, 0, 1)

        # Perturb h at regular intervals
        if iteration % 50 == 0 and iteration > 0:
            perturbation = 0.01 * np.random.randn(n_points)
            h = np.clip(h + perturbation, 0, 1)
            h /= h.sum() * 2.0
            h = np.clip(h, 0, 1)

        # Compute final overlap value
        corr = correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)

        # Update best if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.copy(h)

        # Subgrid refinement if progress is minimal
        if iteration > 100 and best_c5 < 0.38085 and time.time() - start_time < budget_s - 500:
            new_n_points = 140
            new_h = np.interp(np.linspace(0, 2, new_n_points), np.linspace(0, 2, n_points), best_h)
            new_h /= new_h.sum() * 2.0
            res = minimize(
                lambda x: objective(x, lambda_penalty=1e5),
                x0=new_h,
                bounds=[(0, 1)] * new_n_points,
                method='L-BFGS-B',
            )
            new_h = np.clip(res.x, 0, 1)
            new_h /= new_h.sum() * 2.0
            new_h = np.clip(new_h, 0, 1)

            h_out = new_h
            corr = correlate(h_out, 1 - h_out, mode='full') * (2.0 / new_n_points)
            best_h = h_out
            best_c5 = np.max(corr)
            n_points = new_n_points
            dx = 2.0 / n_points

        iteration += 1

    return best_h.tolist(), best_c5, n_points
