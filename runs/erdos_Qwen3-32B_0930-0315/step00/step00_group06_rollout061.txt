import numpy as np
import time
import scipy.signal
import cvxpy as cp
from evaluate_erdos_solution import evaluate_erdos_solution
import math

def run(seed=42, budget_s=1100, verbose=False, **kwargs):
    import numpy as np
    import scipy.signal
    try:
        initial_h = initial_h_values
        n_points = len(initial_h)
    except NameError:
        n_points = 256
        np.random.seed(seed)
        h = np.random.choice([0, 1.0], size=n_points, p=[0.5, 0.5])
        h *= 2.0 / h.sum()
        initial_h = h

    np.random.seed(seed)
    h_best = initial_h.copy()
    c5_best = evaluate_erdos_solution(h_best)

    start_time = time.time()

    while time.time() - start_time < budget_s:
        h = h_best.copy()
        dx = 2.0 / n_points

        # Helper: compute overlap for all lags
        def compute_c5(h):
            if h.ndim == 1:
                h = h.reshape(-1)
            cor = scipy.signal.correlate(h, 1 - h, mode='full')
            return np.max(cor) * dx

        c5 = compute_c5(h)

        # Find k with max overlap
        if h.ndim == 1:
            cor = scipy.signal.correlate(h, 1 - h, mode='full')
        else:
            cor = scipy.signal.correlate(h.flatten(), 1 - h.flatten(), mode='full')
        k_max = np.argmax(cor)

        # Phase 2: Magnitude Optimization (Convex)
        h_var = cp.Variable(n_points, nonneg=True)
        corr_var = 0

        for i in range(n_points):
            if 0 <= k_max - i < n_points:
                corr_var += h_var[i] * (1 - h_var[k_max - i])
        obj = cp.Minimize(corr_var * dx)
        constraints = [
            h_var <= 1,
            h_var >= 0,
            cp.sum(h_var) == n_points / 2,
        ]
        prob = cp.Problem(obj, constraints)

        for solver in [cp.HiGHS, cp.CVXOPT, cp.GLPK]:
            try:
                prob.solve(solver=solver, verbose=False)
                break
            except Exception as e:
                if verbose:
                    print(f"Solver {solver} failed. {e}")

        h_candidate = np.clip(np.array(h_var.value), 0, 1).ravel()

        # Normalize due to possible inaccuracies
        h_candidate *= (n_points / 2) / np.clip(np.sum(h_candidate), 1e-5, None)

        h_candidate = np.clip(h_candidate, 0.0, 1.0)

        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < c5_best:
            h_best = h_candidate
            c5_best = c5_candidate
            if verbose:
                print(f"Improved C₅ = {c5_best} with h size {n_points}")
                print(f"Elapsed: {time.time() - start_time} of {budget_s}")
            search_freq = min(200, math.ceil(1.5 * n_points))
        elif np.random.rand() < 0.005:
            # Perturbation to escape local minima: flip few pixels
            mask = np.random.rand(n_points) < 0.05
            h_candidate[mask] = 1 - h_candidate[mask]
            h_candidate *= (n_points / 2) / np.sum(h_candidate)
            h_candidate = np.clip(h_candidate, 0.0, 1.0)

    return h_best.tolist(), c5_best, n_points
