import numpy as np
import cvxpy as cp
import time
from math import floor

def run(seed=42, budget_s=1100):
    try:
        np.random.seed(seed)
        start_time = time.time()
        n_points = 256  # Use higher resolution than parent
        dx = 2.0 / n_points

        # Use parent solution as initial guess if available
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            n_points = len(h)
            scale = (n_points / 2) / np.sum(h)
            h = scale * h
        else:
            # Default initialization: uniform blob
            h = np.zeros(n_points)
            mid = n_points // 2
            w = 32
            h[mid - w:mid + w] = 1.0 / (2 * w)

        best_c5 = float('inf')
        best_h = None
        best_iter = 0
        learning_rate = 0.1
        proj_lr = 0.02

        while time.time() - start_time < budget_s:
            # Compute overlaps
            ones_h = 1 - h
            corr = np.correlate(h, ones_h, mode='full') * dx
            valid_corr = corr[n_points - 1:2 * n_points - 1]
            overlap = valid_corr
            ac_overlap = np.correlate(h, h, mode='full')[n_points - 1:2 * n_points - 1] * dx
            obj_values = overlap

            max_idx = np.argmax(obj_values)
            max_k = max_idx
            c5_val = obj_values[max_idx]
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
                best_iter = time.time() - start_time

            # Compute gradient at max_k
            grad = np.zeros_like(h)
            for i in range(n_points):
                j = (i + max_k) % n_points
                grad[i] -= ones_h[j]
                grad[i] += h[i] * 2 * (1 - 2 * h[i])  # Derivative term from projection
            grad = dx * grad

            # Update step
            h -= learning_rate * grad
            # Project to [0,1] constraint
            h = np.clip(h, 0, 1)
            # Project to integral constraint: sum(h) = n_points / 2
            s = np.sum(h)
            if abs(s) > 1e-8:
                h *= (n_points / 2) / s

            # Decay learning rate slowly
            t = time.time() - start_time
            fraction_done = t / budget_s
            learning_rate = 0.1 * (1 - fraction_done)

            # Perturb occasionally to escape local optima
            if np.random.random() < 0.002:
                perturbation = np.random.uniform(-0.05, 0.05, n_points)
                h += perturbation
                h = np.clip(h, 0, 1)
                s = np.sum(h)
                if abs(s) > 1e-8:
                    h *= (n_points / 2) / s

        return (best_h.tolist(), best_c5, n_points)
    except Exception as e:
        return (None, float('inf'), n_points)
