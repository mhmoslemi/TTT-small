import numpy as np
import cvxpy as cp
from scipy.interpolate import CubicSpline
from sklearn.metrics import cohen_kappa_score
import time

def fitness(h):
    n = len(h)
    dx = 2.0 / n
    ones = 1 - h
    corr = np.correlate(h, ones, mode='full') * dx
    return np.max(corr)

def repair(h, tol=1e-6):
    h = np.clip(h, 0, 1)
    integral = np.sum(h) * (2.0 / len(h))
    target = 1.0
    if abs(integral - target) > tol:
        scale = target / integral
        h *= scale
        h = np.clip(h, 0, 1)
    return h

def mutate(h, mutation_rate=0.2, noise_scale=0.1):
    mask = np.random.rand(len(h)) < mutation_rate
    h[mask] += np.random.randn(sum(mask)) * noise_scale
    return repair(h)

def crossover(h1, h2):
    n = len(h1)
    point = np.random.randint(1, n)
    return repair(np.concatenate([h1[:point], h2[point:]])), repair(np.concatenate([h2[:point], h1[point:]]))

def evaluate_population(population):
    return [(h, fitness(h)) for h in population]

def fitness_gain(population):
    return max(c for _, c in population) - min(c for _, c in population)

def gradient_step(h, lr=0.01):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_idx = np.argmax(corr)
    
    if max_idx >= n:
        h_shifted = np.concatenate([1 - h[max_idx - n:], np.zeros(max_idx - n)])
    else:
        h_shifted = np.concatenate([1 - h[:max_idx], np.zeros(n - max_idx)])
    
    grad = -h_shifted
    h_new = h - lr * grad
    return repair(h_new)

def coarse_to_fine(initial_h, coarse_n, fine_n):
    if initial_h is None:
        h_coarse = np.random.rand(coarse_n)
    else:
        h_coarse = initial_h[::len(initial_h) // coarse_n]
    h_coarse = repair(h_coarse)
    
    x = np.linspace(0, 2, coarse_n)
    cs = CubicSpline(x, h_coarse)
    x_new = np.linspace(0, 2, fine_n)
    h_fine = cs(x_new)
    return repair(h_fine)

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    
    start_time = time.time()

    coarse_n = 14
    fine_factor = 3
    final_n = coarse_n * fine_factor
    population_size = 20
    generations = 200
    learning_rate = 0.01
    grad_steps = 200
    mutation_rate = 0.1
    best_c5 = float('inf')
    best_h = None

    if 'initial_h_values' in globals():
        refined_initial = coarse_to_fine(initial_h_values, coarse_n, coarse_n)
        population = [refined_initial + np.random.randn(final_n) * 0.01 for _ in range(population_size)]
    else:
        # No parent construction; create a random starting vector algorithmically
        h0 = np.random.rand(coarse_n)
        h0 = repair(h0)
        cs = CubicSpline(np.linspace(0, 2, coarse_n), h0)
        x = np.linspace(0, 2, final_n)
        h0 = cs(x)
        h0 = repair(h0)
        population = [h0 + np.random.randn(final_n) * 0.01 for _ in range(population_size)]

    population = evaluate_population(population)
    population.sort(key=lambda x: x[1])

    for gen in range(generations):
        delta_t = time.time() - start_time
        if delta_t > budget_s - 5:  # Early return if time runs out
            break
        next_gen = []
        top_half = population[:max(1, population_size // 2)]
        while len(next_gen) < population_size:
            p1, p2 = np.random.choice(len(top_half), 2, replace=False)
            h1, c1 = top_half[p1]
            h2, c2 = top_half[p2]
            child1, child2 = crossover(h1, h2)
            if np.random.rand() < 0.1:
                child1 = mutate(child1, mutation_rate)
            if np.random.rand() < 0.1:
                child2 = mutate(child2, mutation_rate)
            next_gen.append(child1)
            next_gen.append(child2)
        population = evaluate_population(next_gen)
        population.sort(key=lambda x: x[1])
        
        best_c5_gen = population[0][1]
        if best_c5_gen < best_c5:
            best_c5 = best_c5_gen
            best_h = population[0][0]
            
        # Exit early if solution is already good enough
        if best_c5 <= 0.38086:
            break

    # Gradient descent refinement
    for _ in range(grad_steps):
        if time.time() - start_time > budget_s - 5:
            break
        best_h = gradient_step(best_h, lr=learning_rate)
        current_c5 = fitness(best_h)
        if best_c5 > current_c5:
            best_c5 = current_c5

    best_h = repair(best_h)

    return best_h, best_c5, len(best_h)
