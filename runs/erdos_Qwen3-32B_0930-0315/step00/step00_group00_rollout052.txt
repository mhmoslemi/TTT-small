import numpy as np
import time
import random
from scipy.signal import correlate

def compute_overlap_integral_correlation(h, dx):
    ones_h = np.clip(1 - h, 0, 1)
    corr = correlate(h, ones_h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def clip_and_normalize(h, n_points):
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if abs(total - n_points / 2) > 1e-8:
        delta = n_points / 2 - total
        idx = np.argsort(h)
        while abs(delta) > 0:
            i = idx[0] if delta > 0 else idx[-1]
            add = min(delta, 1 - h[i]) if delta > 0 else max(delta, -h[i])
            h[i] += add
            delta -= add
            idx = np.argsort(h)
    return h

def random_perturbation_neighbor(h, n_points, size=1):
    h = h.copy()
    indices = np.random.choice(n_points, min(size, n_points), replace=False)
    step = (1 - h[indices]) * 2 * np.random.rand(len(indices)) - h[indices]
    h[indices] += step
    return clip_and_normalize(h, n_points)

def shift_neighbor(h, n_points):
    shift = np.random.randint(1, n_points)
    h = np.concatenate([h[shift:], h[:shift]])
    return clip_and_normalize(h, n_points)

def gen_neighbor(h, n_points):
    if random.random() < 0.5:
        return random_perturbation_neighbor(h, n_points, size=2)
    else:
        return shift_neighbor(h, n_points)

    dx = 2.0 / n_points
    best_h = clip_and_normalize(h, n_points)
    best_c5 = compute_overlap_integral_correlation(best_h, dx)

    max_iter = 60000  # estimated to be well within time budget
    tabu_size = 15
    tabu_list = []
    start_time = time.time()

    for iter in range(max_iter):
        if time.time() - start_time >= budget_s:
            break

        # Generate neighbors
        neighbors = [gen_neighbor(best_h, n_points) for _ in range(20)]

        # Evaluate and select best non-tabu neighbor
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        for neighbor in neighbors:
            if not any(np.allclose(neighbor, t, atol=1e-3) for t in tabu_list):
                neighbor_c5 = compute_overlap_integral_correlation(neighbor, dx)
                if neighbor_c5 < best_neighbor_c5:
                    best_neighbor_c5 = neighbor_c5
                    best_neighbor = neighbor

        if best_neighbor is None:
            continue  # all neighbors are tabu; skip to next iteration

        # Update solution if improved
        if best_neighbor_c5 < best_c5:
            best_c5 = best_neighbor_c5
            best_h = best_neighbor

        # Update tabu list
        tabu_list.append(best_neighbor)
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)

    return best_h, best_c5, n_points
