import numpy as np
import time
from scipy.optimize import dual_annealing


def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifts = np.arange(-n_points + 1, n_points)
    corrs = np.zeros_like(shifts, dtype=np.float64)
    for k in range(len(shifts)):
        idx_lag = shifts[k]
        if idx_lag < 0:
            a = h[-idx_lag:]
            b = 1 - h[:idx_lag]
        elif idx_lag > 0:
            a = h[:n_points - idx_lag]
            b = 1 - h[idx_lag:]
        else:
            a = h
            b = 1 - h
        corrs[k] = np.dot(a, b) * dx
    return max(corrs)


def project_simplex(v):
    v_sorted = np.sort(v)[::-1]
    slopes = np.cumsum(v_sorted) - 1
    idx = np.arange(len(slopes))
    rho = idx[slopes > 0][-1] if np.any(slopes > 0) else 0
    threshold = (np.sum(v_sorted[: rho + 1]) - 1) / (rho + 1)
    return np.maximum(v - threshold, 0)
