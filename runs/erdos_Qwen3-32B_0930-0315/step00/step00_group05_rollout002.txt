import numpy as np
import scipy.signal
import time
from cvxpy import Variable, Problem, Max, Min, sum as cvp_sum

def run(seed=42, budget_s=1100):
    # Setup
    np.random.seed(seed)
    start_time = time.time()

    # Predefined initial solution
    initial_h = initial_h_values  # Assume it's normalized and n=70
    h_best = initial_h.copy()
    n_points = len(h_best)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def compute_c5(h):
        h_bin = h if isinstance(h, np.ndarray) else np.array(h)
        h_bin = h_bin.clip(0, 1)
        if np.isclose(h_bin.sum(), target_sum):
            corr = scipy.signal.correlate(h_bin, 1 - h_bin, mode="full")
            c5 = np.max(corr * dx)
        else:
            raise ValueError("Constraint sum(h) = 35 not satisfied")
        return c5

    def smooth_crossentropy_objective(h):
        h = h.clip(0, 1e-6, 1 - 1e-6)
        h = np.log(h / (1 - h))
        corr = scipy.signal.correlate(h, h - 1, mode='full')
        return np.max(corr * dx)

    def step_update(h, lr=0.01, eps=1e-10):
        hh = h.copy()
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - h.size + 1
        if 0 <= lag < h.size:
            hh[lag] += 0.5 * lr
        elif -h.size < lag < 0:
            hh[:lag] -= 0.5 * lr
            hh[lag:] += 0.5 * lr
        else:
            pass  # No valid lag found
        hh = np.clip(hh, 0, 1)
        hh /= hh.sum() * 2.0
        return hh

    def project_to_feasible(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = h_clipped.sum()
        scaling = target_sum / h_sum if h_sum != 0 else 1
        return h_clipped * scaling

    best_c5 = compute_c5(h_best)
    best_h = h_best.copy()
    print(f"Initial C₅: {best_c5:.6f}")

    while time.time() - start_time < budget_s:
        h = project_to_feasible(best_h + 0.1 * np.random.randn(*best_h.shape))
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            print(f"Improved: C₅ = {best_c5:.6f}")
    return best_h.tolist(), best_c5, n_points

if __name__ == "__main__":
    run()
