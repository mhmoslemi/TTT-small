import numpy as np
import time

def evaluate_erdos_solution(h_vals, n_points):
    h = np.array(h_vals)
    dx = 2.0 / n_points
    max_overlap = -np.inf
    for k in range(-n_points + 1, n_points):
        shifted = np.roll(h, k)
        shifted = np.where(np.arange(n_points) + k >= 0,
                           np.where(np.arange(n_points) + k < n_points,
                                    shifted, 0), 0)
        overlap = np.sum(h * (1 - shifted)) * dx
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap
