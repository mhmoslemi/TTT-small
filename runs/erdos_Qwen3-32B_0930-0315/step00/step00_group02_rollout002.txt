import numpy as np
import math
import time
import cvxpy as cp

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    h = np.array(h) if not isinstance(h, np.ndarray) else h
    shifted_h = 1 - h
    corr_all = np.correlate(h, shifted_h, mode="full")
    max_corr = np.max(corr_all) * dx
    return max_corr

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    start_time = time.time()

    n_points = 160
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values) / np.sum(initial_h_values) * (n_points / 2)
        h = h[:n_points]
        h = np.pad(h, (0, n_points - len(h)), 'constant')
    else:
        h = np.zeros(n_points)
        h[n_points // 4: 3 * n_points // 4] = 0.5
        h = h / np.sum(h) * (n_points / 2)

    h = np.clip(h, 0, 1)

    def compute_corr(h):
        shifted_h = 1 - h
        corr_all = np.correlate(h, shifted_h, mode="full")
        return np.max(corr_all)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)

    def gradient_descent_step(h):
        corr = np.correlate(h, 1 - h, mode="full") * dx
        max_idx = np.argmax(corr)
        grad = np.zeros(n_points)
        for i in range(n_points):
            k = max_idx - i
            if k < n_points:
                grad[i] += h[i] - h[i] ** 2 if k < n_points else 0
            if k >= -n_points and k < 0:
                idx = (i + k) % n_points
                grad[i] -= h[i] * (1 - h[idx])
        return grad

    def project(feasible_point):
        feasible_point.flags.writeable = True
        np.clip(feasible_point, 0, 1, out=feasible_point)
        rescale_delta = (n_points / 2) / np.sum(feasible_point)
        feasible_point *= rescale_delta
        return feasible_point

    phase1_duration = 500
    phase2_duration = 450
    start_phase1 = time.time()
    while time.time() - start_phase1 < phase1_duration:
        grad = gradient_descent_step(h)
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0:
            break
        alpha = min(0.1 / grad_norm, 0.01)
        step = h - alpha * grad
        h = project(step)
        c5 = evaluate_erdos_solution(h, n_points)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        if time.time() - start_phase1 > phase1_duration:
            break

    start_phase2 = time.time()
    h_phase2 = cp.Variable(n_points)
    obj = cp.Minimize(cp.Max(
        [cp.sum([h_phase2[i] * (1 - h_phase2[(i + k) % n_points]) for i in range(n_points)]) for k in range(n_points)]
    ))
    constraints = [
        cp.sum(h_phase2) == n_points / 2,
        0 <= h_phase2,
        h_phase2 <= 1
    ]
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, max_iters=200, verbose=False)
    if prob.status == 'optimal':
        h_opt = np.array(h_phase2.value).flatten()
        h_opt = project(h_opt)
        c5_opt = evaluate_erdos_solution(h_opt, n_points)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
        while time.time() - start_phase2 < phase2_duration:
            perturbed = best_h + np.random.laplace(0, 0.03, n_points)
            perturbed = project(perturbed)
            c5_pert = evaluate_erdos_solution(perturbed, n_points)
            if c5_pert < best_c5:
                best_h = perturbed
                best_c5 = c5_pert

    if time.time() - start_time < budget_s:
        h = best_h.copy()

    h_values = h.tolist()
    c5_bound = evaluate_erdos_solution(h, n_points)
    return h_values, c5_bound, n_points
