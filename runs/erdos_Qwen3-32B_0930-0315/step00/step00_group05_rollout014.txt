import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h = h / n_points * n_points
    corr = correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def objective(h_vec, n_points, penalty_coeff=1e4):
    h = h_vec.copy()
    # Ensure box constraints are respected
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    target = n_points / 2.0
    c5 = compute_c5(h, n_points)
    # Add penalization if the integral constraint is violated
    penalty = penalty_coeff * (sum_h - target)**2
    return c5 + penalty

def perturb(h, scale=0.02):
    h_perturbed = h.copy()
    noise = np.random.normal(0, scale, size=h.shape)
    h_perturbed += noise
    h_perturbed = np.clip(h_perturbed, 0, 1)
    # Rescale to maintain the sum constraint
    h_perturbed = h_perturbed / np.sum(h_perturbed) * (h.shape[0] / 2.0)
    return h_perturbed

def optimize_step(h_flat, bounds, n_points, method='L-BFGS-B', penalty_coeff=1e4):

        bounds=bounds,
    )
    h_opt = np.clip(result.x, 0, 1)
    h_opt = h_opt / h_opt.sum() * (n_points / 2.0)
    return h_opt
