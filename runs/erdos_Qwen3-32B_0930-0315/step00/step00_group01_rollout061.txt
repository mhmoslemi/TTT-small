import numpy as np
import time

def run(seed=42, budget_s=1100, n_points=47, learning_rate=0.2, n_warmup_steps=100, n_steps=2000, verbose=False):
    np.random.seed(seed)
    t_start = time.time()
    dx = 2.0 / n_points
    h = initial_h_values.copy() if 'initial_h_values' in globals() else np.random.rand(n_points)
    
    # Enforce constraints
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2
    best_h = h.copy()
    best_c5 = float('inf')
    
    for step in range(n_steps):
        if time.time() - t_start >= budget_s:
            break

        # Compute overlap
        overlap = np.correlate(h, 1 - h, mode='full')
        overlap_vals = overlap[n_points - 1:] * dx

        # Find worst lag k_max
        k_max = np.argmax(overlap_vals)
        max_overlap = overlap_vals[k_max]

        # Compute gradient for k_max
        grad = np.zeros(n_points)
        for i in range(n_points - k_max):
            grad[i] += (1 - h[i + k_max]) if max_overlap > 0 else 0
        for i in range(k_max, n_points):
            grad[i] -= h[i - k_max] if max_overlap > 0 else 0

        # Step and project
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

        # Track best
        overlap = np.correlate(h, 1 - h, mode='full')
        overlap_vals = overlap[n_points - 1:] * dx
        c5 = np.max(overlap_vals)
        if c5 < best_c5:
            best_c5 = c5
            best_h[:] = h.copy()

        # Occasional perturbation
        if step % 100 == 0 and step > n_warmup_steps:
            noise = (np.random.rand(n_points) - 0.5) * 0.1
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * n_points / 2

        if step % 100 == 0 and verbose:
            print(f"Step {step}, C₅={best_c5:.7f}, time={time.time() - t_start:.2f}s")

    # Validate solution
    h = best_h.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    overlap = np.correlate(h, 1 - h, mode='full')
    overlap_vals = overlap[n_points - 1:] * dx
    c5_bound = np.max(overlap_vals)

    return (h, c5_bound, n_points)
