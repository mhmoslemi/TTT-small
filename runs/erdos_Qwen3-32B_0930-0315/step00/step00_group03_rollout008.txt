import numpy as np
import time
import random
from scipy.optimize import minimize
from math import floor

def objective(h, n_points, dx):
    h_array = np.clip(h, 0, 1)
    corr = np.correlate(h_array, 1 - h_array, mode='full')
    return np.max(corr) * dx

def repair_feasibility(h, n_points):
    # Normalize to satisfy sum constraint
    h_array = np.clip(h, 0, 1)
    s = n_points / 2.0
    current_sum = np.sum(h_array)
    h_array = np.clip(h_array * (s / current_sum), 0, 1)
    return h_array

def crossover(parent1, parent2, n_points):
    crossover_point = np.random.randint(1, n_points - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return [
        repair_feasibility(child1, n_points),
        repair_feasibility(child2, n_points),
    ]

def mutate(gene, n_points, mutation_rate=0.15):
    gene = gene.copy()
    locations = np.random.choice(n_points, size=int(n_points * 0.05), replace=False)
    gene[locations] += np.random.normal(0, 0.1, size=len(locations))
    gene = np.clip(gene, 0, 1)
    # Normalize
    return repair_feasibility(gene, n_points)

def tournament_selection(population, fitness, tournament_size=5):
    contestant_indices = np.random.choice(len(population), tournament_size)
    contestant_fitness = [fitness[i] for i in contestant_indices]
    winner_index = contestant_indices[np.argmax(contestant_fitness)]
    return population[winner_index]

def run(seed=42, budget_s=1100, population_size=50, max_generations=1000):
    np.random.seed(seed)
    random.seed(seed)
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_values = initial_h_values
        n_points = len(h_values)
    else:
        # Create initial random configuration
        h_values = np.random.rand(70)
        h_values = repair_feasibility(h_values, 70)
        n_points = 70

    dx = 2.0 / n_points
    start_time = time.time()

    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(initial_h_values)
        # Add perturbed versions
        for i in range(population_size - 1):
            perturbation = np.random.normal(0, 0.1, n_points)
            candidate = np.clip(initial_h_values + perturbation, 0, 1)
            candidate = repair_feasibility(candidate, n_points)
            population.append(candidate)
    else:
        for i in range(population_size):
            h = np.random.rand(n_points)
            h = repair_feasibility(h, n_points)
            population.append(h)

    best_h = None
    best_c5 = float('inf')

    gen = 0
    while time.time() - start_time < budget_s and gen < max_generations:
        gen += 1
        fitness = []
        for h in population:
            c5 = objective(h, n_points, dx)
            fitness.append(-c5)  # Negative since we are minimizing

        # Selection and breeding
        new_population = []
        for _ in range(population_size // 2):
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            if parent1 is not None and parent2 is not None:
                child1, child2 = crossover(parent1, parent2, n_points)
                new_population.extend([mutate(child1, n_points), mutate(child2, n_points)])
        
        # Replace with top half of old population
        top_indices = np.argsort(fitness)[::-1][:population_size // 2]
        new_population += [population[i] for i in top_indices]
        
        population = new_population

        # Update best solution found
        current_best_h = None
        current_best_c5 = float('inf')
        for h in population:
            c5 = objective(h, n_points, dx)
            if c5 < current_best_c5:
                current_best_c5 = c5
                current_best_h = h

        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h

    return best_h, best_c5, n_points
