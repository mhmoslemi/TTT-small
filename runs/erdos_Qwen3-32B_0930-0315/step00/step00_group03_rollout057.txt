import numpy as np
import time
from numpy.random import default_rng
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    rng = default_rng(seed)
    t_start = time.time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    pop_size = 50
    max_gen = 2000
    mut_prob = 0.15
    noise_std = 0.1
    perturb_range = 0.1
    tournament_size = 5
    
    # Use provided initial_h_values
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values).copy()
        assert np.isclose(np.sum(h_initial), target_sum)
    else:
        raise ValueError("initial_h_values not provided")
    
    # Generate initial population with perturbations
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            h = h_initial.copy()
        else:
            h = rng.random(n_points)
        h = h + rng.uniform(-perturb_range, perturb_range, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * target_sum
        population.append(h)
    
    def evaluate(h):
        if not np.all(np.isfinite(h)):
            return float('inf')
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * target_sum
        c = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(c) * dx
        return c5
    
    best_c5 = float('inf')
    best_h = None
    
    for gen in range(max_gen):
        if time.time() - t_start > budget_s:
            break
        # Evaluate population
        fitness = np.array([evaluate(h) for h in population])
        idx = np.argmin(fitness)
        if fitness[idx] < best_c5:
            best_c5 = fitness[idx]
            best_h = population[idx].copy()
        
        # Tournament selection
        parents = []
        for _ in range(pop_size // 2):
            candidates = rng.integers(0, pop_size, tournament_size)
            best_idx = np.argmin(fitness[candidates])
            parents.append(population[candidates[best_idx]])
        
        # Crossover and mutation to create new population
        new_population = []
        for i in range(0, len(parents), 2):
            if i+1 < len(parents):
                p1 = parents[i]
                p2 = parents[i+1]
                
                # Crossover: random split
                c = rng.integers(1, n_points-1)
                o1 = np.concatenate([p1[:c], p2[c:]])
                o2 = np.concatenate([p2[:c], p1[c:]])
                
                for o in [o1, o2]:
                    # Feasibility repair
                    o = np.clip(o, 0, 1)
                    s = target_sum / np.sum(o)
                    s = np.clip(s, 0.9, 1.1)
                    o = np.clip(o * s, 0, 1)
                    
                    # Mutation
                    if rng.random() < mut_prob:
                        n = o.shape[0]
                        m = max(1, int(0.05 * n))
                        mut_idx = rng.integers(0, n, m)
                        noise = rng.normal(0, noise_std, m)
                        o[mut_idx] += noise
                        o = np.clip(o, 0, 1)
                        s = target_sum / np.sum(o)
                        s = np.clip(s, 0.9, 1.1)
                        o = np.clip(o * s, 0, 1)
                    new_population.append(o)
        
        # Elitism: keep the best solution
        new_population.append(best_h.copy())
        population = new_population[:pop_size]
    
    # Final evaluation
    best_c5 = evaluate(best_h)
    
    # Validation
    assert np.all(np.isclose(np.sum(best_h), target_sum)), f"Sum failed: {np.sum(best_h)=}, target {target_sum}"
    assert np.all(np.clip(best_h, 0, 1) == best_h), "Out-of-bounds values"
    
    return (best_h, best_c5, n_points)
