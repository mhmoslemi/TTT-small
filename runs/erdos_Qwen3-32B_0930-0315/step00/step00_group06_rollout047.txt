import numpy as np
import time
import scipy.signal
import cvxpy as cp

def run(seed=42, budget_s=1100, n_points=256):
    np.random.seed(seed)

    # Load or initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values
        if h.shape[0] != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, h.shape[0]), h)
        h = h * (n_points / 2) / h.sum()
    else:
        h = np.random.choice([1.0, 0.0], size=n_points)
        h *= (n_points / 2) / h.sum()

    dx = 2.0 / n_points

    # Evaluator for C5
    def compute_c5(h):
        h_1mh = 1 - h
        corr = np.correlate(h, h_1mh, mode='full') * dx
        return np.max(corr)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    # Optimization loop - alternating position and magnitude updates
    for step in range(1_000_000):  # Large iteration limit
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Promote early exit if converged
        if step % 1000 == 0:
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                print(f"Step {step}, C5: {best_c5:.8f}")

        # Compute max overlap interval
        h_1mh = 1 - h
        corr = scipy.signal.correlate(h, h_1mh, mode='full') * dx
        max_idx = np.argmax(np.abs(corr))
        k = max_idx - (n_points - 1)

        # Phase 1: Position Adjustment (we use magnitude shifts instead of strict position moves)
        dh = -0.1 * (np.roll(h, -k) * h_1mh)
        h += dh
        h = np.clip(h, 0, 1)

        # Magnitude Adjust with convex opt
        if step % 100 == 0:
            h_var = cp.Variable(n_points)
            objective = cp.Minimize(cp.scalar_product(h_var, np.roll(h_1mh, k)))
            constraints = [
                0 <= h_var, h_var <= 1,
                cp.sum(h_var) == n_points / 2
            ]
            problem = cp.Problem(objective, constraints)
            problem.solve(solver=cp.PDLP)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h = h_var.value.copy()
                h = np.clip(h, 0, 1)

        # Normalize for constraints
        h_sum = np.sum(h)
        if not np.isclose(h_sum, n_points / 2):
            h *= (n_points / 2) / h_sum

        # Perturb h occasionally
        if step % 1000 == 0:
            h += 0.01 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h *= (n_points / 2) / h.sum()

    final_c5 = compute_c5(best_h)
    return best_h.tolist(), final_c5, n_points
