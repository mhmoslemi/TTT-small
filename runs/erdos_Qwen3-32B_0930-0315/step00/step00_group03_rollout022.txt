import numpy as np
import cvxpy as cp
import math
import time


def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points

    h = h0.copy()
    best_h = h.copy()
    best_c5 = np.inf
    last_improve = 0
    iteration = 0
    start_time = time.time()

    def project(h_candidate):
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate + (n_points / 2 - np.sum(h_candidate)) / n_points
        h_candidate = np.clip(h_candidate, 0, 1)
        return h_candidate

    while time.time() - start_time < budget_s:
        # Compute correlation
        h_neg = 1 - h
        correl = np.correlate(h, h_neg, mode='full') * dx
        k_max = np.argmax(correl)
        c5 = correl[k_max]

        # Update the best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improve = iteration

        # Stopping condition
        if iteration > last_improve + 100:
            break

        # Learning rate decay
        t = iteration
        learning_rate = 1.0 / (1 + t / 1000)

        # Compute gradient from current max k
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] += (1 - h[i + k_max])
            if i - k_max >= 0:
                grad[i] -= h[i - k_max]

        # Gradient descent step
        h = h - learning_rate * grad
        h = project(h)

        # Enforce average sum constraint precisely
        delta_sum = (n_points / 2 - np.sum(h)).clip(-1e-6, 1e-6)
        h += delta_sum / n_points
        h = np.clip(h, 0.0, 1.0)

        iteration += 1

    # Final check of best solution
    best_h = project(best_h)
    correl = np.correlate(best_h, 1 - best_h, mode='full') * dx
    best_c5 = np.max(correl)

    return (best_h, best_c5, n_points)
