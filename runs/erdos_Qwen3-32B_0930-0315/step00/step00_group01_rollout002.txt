import numpy as np
import time
import cvxpy as cp
import scipy.signal

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    # Use machine-precision version of initial_h_values to keep full resolution
    initial_h_values = initial_h_values.copy()
    n_points = len(initial_h_values)
    n_points_orig = n_points

    # Define optimization problem size
    max_outer_iter = int(budget_s * 0.8 // 0.5)  # Conservative estimate of update budget
    verbose_outer = False
    patience = 50  # Early stopping if no improvement for N iterations
    best_c5 = float('inf')
    best_h = initial_h_values.copy()

    def evaluate_c5(h_vec):
        h = h_vec.reshape(-1)
        # Pad 1-h for correlation
        pad_1_h = np.concatenate([1 - h, 1 - h])
        corr = scipy.signal.correlate(h, pad_1_h, mode='valid')
        return (corr * (2.0 / n_points)).max()

    # Project h onto feasible set [0, 1]^n with sum(h) = n_points / 2
    def project(bad_h):
        # First clip to [0, 1]
        normalized = np.clip(bad_h, 0, 1)
        # If sum not equal to target, scale between 0 and 1 to achieve it
        target_sum = n_points / 2
        current_sum = np.sum(normalized)
        if abs(current_sum - target_sum) > 1e-4:
            # Map [0, 1]^n to functions with exact integral by scaling
            normalized = (normalized * (target_sum / current_sum))
            # Clip again in case scaling pushes past [0, 1] limits
            normalized = np.clip(normalized, 0, 1)
        return normalized

    # Initial evaluation
    best_c5 = evaluate_c5(best_h)
    h = best_h.copy()

    for outer in range(max_outer_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Finite differences gradient estimate
        step_size = 1e-5
        grad = np.zeros_like(h)
        for i in range(n_points):
            x_pos = h.copy()
            x_pos[i] += step_size
            x_pos = project(x_pos)
            loss_pos = evaluate_c5(x_pos)

            x_neg = h.copy()
            x_neg[i] -= step_size
            x_neg = project(x_neg)
            loss_neg = evaluate_c5(x_neg)

            grad[i] = (loss_pos - loss_neg) / (2 * step_size)

        # Descend along negative gradient
        learning_rate = 0.01
        h_trial = h - learning_rate * grad
        h_trial = project(h_trial)
        c5_trial = evaluate_c5(h_trial)

        # If improved, update
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            if verbose_outer:
                print(f"Iter {outer}: C5={best_c5}")
        h[:] = h_trial[:]
        # Stop if no progress for patience iterations
        if (outer % patience) == (patience - 1) and best_c5 == float('inf'):
            break

    return (best_h.tolist(), best_c5, n_points_orig)
