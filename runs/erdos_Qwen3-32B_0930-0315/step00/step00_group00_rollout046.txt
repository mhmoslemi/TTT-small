import numpy as np
import time
import cvxpy as cp
from numpy.fft import fft, ifft, fftshift

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    def preempt(dt=budget_s):
        """
        Returns a closure that can be checked for time limits
        """
        start = time.time()
        def is_time():
            return time.time() - start >= budget_s
        return is_time

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[22] = 0.5
        h[23] = 0.5
        while np.sum(h) < n_points / 2:
            h += 1.0 / n_points

    n_points = len(h)
    dx = 2.0 / n_points
    window = n_points * 2  # Total number of lags to consider

    def compute_c5(h):
        h_flat = h.ravel()
        c5_vals = np.correlate(h_flat, 1.0 - h_flat, mode='full')
        return np.max(c5_vals) * dx

    best_c5 = compute_c5(h)
    best_h = h.copy()

    def optimize_region(h_current, idxs, step_size, patience, refine=False):
        h_loc = h_current.copy()
        n = len(h_loc)
        sum_fix = np.sum(h_loc)

        def score(x):
            h_test = h_loc.copy()
            for i, v in zip(idxs, x):
                h_test[i] = v
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test / np.sum(h_test) * n / 2
            return compute_c5(h_test)

        x0 = [h_loc[i] for i in idxs]
        x = cp.Variable(len(x0))

        constraints = [0 <= x, x <= 1]
        objective = cp.Minimize(score(x))
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.ECOS, max_iters=50)

        if problem.status in ['optimal', 'optimal_inaccurate']:
            x_val = x.value
            for i, v in zip(idxs, x_val):
                h_loc[i] = v
            h_loc = np.clip(h_loc, 0, 1)
            h_loc = h_loc / np.sum(h_loc) * n / 2
            return h_loc, compute_c5(h_loc)
        else:
            return h_current, best_c5

    is_time = preempt(budget_s)

    while not is_time():
        grad_vals = np.zeros(n_points)
        c5_vals = np.correlate(h, 1 - h, mode="full") * dx
        max_c5 = np.max(c5_vals)
        max_lags = np.where(np.isclose(c5_vals, max_c5))[0]

        for k in max_lags:
            if k < n_points:
                overlap = np.minimum(h, 1 - h[k:k+n_points])
                grad = -2 * overlap
                grad_vals += grad

        grad_vals[~np.isfinite(grad_vals)] = 0
        grad_norm = np.linalg.norm(grad_vals)
        if grad_norm != 0:
            grad_vals = grad_vals / grad_norm
        else:
            grad_vals = np.zeros_like(grad_vals)

        # Select descending positions by gradient magnitude
        positions = np.argsort(-np.abs(grad_vals))
        high_impact = positions[: int(n_points * 0.2)]
        low_impact = positions[int(n_points * 0.2):]

        for idxs in [np.array([p]) for p in high_impact]:
            new_h, new_c5 = optimize_region(h, idxs, step_size=1.0, patience=100, refine=True)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5

        h = best_h.copy()

        for idxs in [np.array([p]) for p in low_impact]:
            new_h, new_c5 = optimize_region(h, idxs, step_size=0.5, patience=5)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5

    c5_vals = np.correlate(best_h, 1 - best_h, mode="full") * dx
    final_c5 = np.max(c5_vals)
    return best_h, final_c5, n_points
