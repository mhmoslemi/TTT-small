import numpy as np
import scipy.signal
import time

def evaluate_erdos_solution(h_values):
    h = np.array(h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    kernel = 1.0 - h
    corr = scipy.signal.correlate(h, kernel, mode="full") * dx
    return np.max(corr)

initial_h_values = initial_h_values if 'initial_h_values' in globals() else np.zeros(45, dtype=float)

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    n_points = 160
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    h = h * n_points / 2 / np.sum(h)  # Normalize to area = 1

    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    last_improve_time = start_time

    dx = 2.0 / n_points
    while time.time() - start_time < budget_s:
        # Correlation
        corr = scipy.signal.correlate(h, 1 - h, mode='full') * dx
        k_max = np.argmax(corr)
        k = k_max - n_points + 1

        # Evaluate gradient: C = sum_i h[i] (1 - h[i + k]) * dx
        # dC/dh[i] = (1 - h[i + k]) * dx - h[i - k] * dx if i - k >=0 and i + k < len(h)
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k < 0 or i + k >= n_points:
                continue
            grad[i] += (1 - h[(i + k) % n_points])
            grad[(i - k) % n_points] -= h[i]

        # Barzilai-Borwein step length guessing (adaptively choosing step size)
        # Use different scheme for first step
        if 'prev_h' not in locals():
            prev_h = h.copy()
            alpha = 1e-2
        else:
            delta_h = h - prev_h
            delta_grad = grad - prev_grad
            dot_dh = np.dot(delta_h, delta_h)
            dot_g = np.dot(delta_h, delta_grad)
            alpha = dot_dh / dot_g if dot_g != 0 else 1.0
            alpha = min(max(alpha, 1e-4), 10.0)
            prev_grad = grad.copy()
            prev_h = h.copy()

        # Take step
        h -= alpha * grad

        # Projection: clip to [0, 1]
        np.clip(h, 0, 1, out=h)

        # Enforce sum(h) = n_points / 2 (normalized total area = 1)
        total = np.sum(h)
        scale_factor = n_points / 2 / total
        h *= scale_factor

        # Periodic perturbation to avoid local minima
        if time.time() - last_improve_time > 20:
            h += 0.01 * np.random.randn(n_points)
            np.clip(h, 0, 1, out=h)
            h *= n_points / 2 / np.sum(h)

        # Evaluate current C and compare with best
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improve_time = time.time()

        if time.time() - start_time + 0.1 > budget_s:
            break

    return best_h.tolist(), best_c5, n_points
