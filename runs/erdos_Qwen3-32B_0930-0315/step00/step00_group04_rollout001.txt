def optimize_step_cd(h, dx):
    # Coordinate descent for non-convex step minimization
    h_new = h.copy()
    corr = np.correlate(h_new, 1 - h_new, mode="full") * dx
    k_max = np.argmax(corr)

    for i in range(len(h_new)):
        if i >= len(h_new) or i + k_max >= len(h_new):
            continue
        old_value = h_new[i]
        for val in [0.0, 1.0, (old_value + 0.5)]:
            h_test = h_new.copy()
            h_test[i] = val
            h_test = h_test / np.sum(h_test) * len(h_test) / 2
            h_test = np.clip(h_test, 0, 1)
            c_test = np.correlate(h_test, 1 - h_test, mode="full") * dx
            if np.max(c_test) < np.max(corr):
                h_new[i] = val
                corr = c_test
                break
    return h_new


def optimize_step_slsqp(h, dx, n_points):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx

    def objective(x):
        h_test = x.reshape(n)
        corr_test = np.correlate(h_test, 1 - h_test, mode="full") * dx
        return np.max(corr_test)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    def constraint_lower(x):
        return x

    def constraint_upper(x):
        return 1 - x

    constraints = (
        {'type': 'eq', 'fun': constraint_sum},
    )
    bounds = [(0, 1)] * n
    result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8, options={'ftol': 1e-8})
    return np.clip(result.x, 0, 1)
