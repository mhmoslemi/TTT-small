```python
import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp


def evaluate_erdos_solution(h, n_points):
    """Given a step function h in [0,1]^n with discrete dx = 2.0/n_points, returns the C₅ overlap value."""
    h = np.asarray(h)
    assert h.shape == (n_points,)
    assert np.isclose(np.sum(h), n_points / 2, rtol=1e-5, atol=1e-6)
    dx = 2.0 / n_points
    c5 = max(correlate(h, (1 - h), mode="full") * dx)
    return c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    start_time = time.time()
    elapsed_time = lambda: time.time() - start_time

    # Load initial_h_values if available
    initial_h_values = locals().get('initial_h_values', None)

    if initial_h_values is not None:
        h0 = initial_h_values
        n_points = len(h0)
    else:
        n_points = 160
        plateau_width = int(n_points / 2)
        h0 = np.array([0.0] * ((n_points - plateau_width) // 2) + [1.0] * plateau_width +
                      [0.0] * ((n_points - plateau_width) // 2))

    h = h0.copy().astype(float)
    h /= h.sum() * 2.0 / n_points  # normalize to conserve the total integral

    # Optimization hyperparams
    max_iter = 2000
    learning_rate_start = 0.01
    decay_rate = 0.995

    # Update h so that sum(h) == n_points / 2
    h_sum = h.sum()
    if not np.isclose(h_sum, n_points / 2):
        scale_factor = (n_points / 2) / h_sum
        h *= scale_factor
        assert np.isclose(h.sum(), n_points / 2), (h.sum(), n_points / 2)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)

    def grad_step(h):
        n = len(h)
        dx = 2.0 / n
        ones_minus_h = 1.0 - h

        corrs = np.zeros(2*n - 1)

        for k in range(n-1):
            i1 = max(0, k - n + 1)
            i2 = min(n-1, n - 1 - k)
            for i in range(i1, i2+1):
                j = n - 1 - i - k
                corr = h[i] * ones_minus_h[j]
                corrs[n + k] += corr

        for k in range(n+1, 2*n - 1):
            i1 = max(0, n - k)
            i2 = min(n-1, n - 1 - (k - n))
            for i in range(i1, i2+1):
                j = n - 1 - i - (k - n)
                corr = h[i] * ones_minus_h[j]
                corrs[k] += corr

        max_idx = np.argmax(corrs)
        grad = np.zeros(n)
        for i in range(n):
            for k in range(n + 1):
                idx = n + k
                if idx == max_idx:
                    if i + k < n:
                        grad[i] += (1 - h[i + k])
                idx = n - 1 - k
                if idx == max_idx:
                    if i - k >= 0:
                        grad[i] -= h[i - k]

        # Sign flip (we *want* to reduce max overlap)
        grad *= -1
        grad *= dx
        return grad / n  # normalize

    # Start optimization loop
    learning_rate = learning_rate_start
    for it in range(max_iter):
        grad = grad_step(h)
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad /= grad_norm  # Gradient normalization

        h_proposed = h - learning_rate * grad
        h_proposed = np.clip(h_proposed, 0, 1)
        h_proposed_sum = h_proposed.sum()
        if not np.isclose(h_proposed_sum, n_points / 2, rtol=1e-5):
            scale_factor = (n_points / 2) / h_proposed_sum
            h_proposed *= scale_factor

        # Check for better solution
        c5_proposed = evaluate_erdos_solution(h_proposed, n_points)
        c5_current = evaluate_erdos_solution(h, n_points)

        if c5_proposed < c5_current:
            h = h_proposed.copy()
            best_h[:] = h_proposed.copy()
            best_c5 = c5_proposed
        elif np.random.rand() < 0.001 and it % 50 == 0:  # Exploration with randomness
            h = np.random.rand(n_points)
            h_sum = h.sum()
            h *= (n_points / 2) / h_sum
            h = np.clip(h, 0, 1)
            grad_norm = 0  # Reset

        learning_rate *= decay_rate
        if grad_norm < 1e-5 or elapsed_time() >= budget_s:
            break

    # Final result validation
    h_sum = best_h.sum()
    if not np.isclose(h_sum, n_points / 2, rtol=1e-5):
        scale_factor = (n_points / 2) / h_sum
        best_h *= scale_factor
        assert np.isclose(best_h.sum(), n_points / 2), (best_h.sum(), n_points / 2)

    best_h = np.clip(best_h, 0, 1)
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    return (best_h, best_c5, n_points)
```
