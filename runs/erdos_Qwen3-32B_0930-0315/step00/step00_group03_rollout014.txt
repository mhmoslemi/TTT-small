import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, n_points=70):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else n_points
    h = initial_h_values.copy() if 'initial_h_values' in globals() else 0.5 * np.ones(n_points)
    h = np.clip(h, 0, 1)
    h *= n_points / 2 / np.sum(h)

    best_h = h.copy()
    best_c5_bound = evaluate_erdos_solution(h)
    dx = 2.0 / n_points
    max_iter = int(budget_s * 10)
    eta = 0.01
    decay = 0.999
    noise_scale = 0.001

    for itr in range(max_iter):
        if time.time() - start_time >= budget_s - 1e-6:
            break
        k_max, c5 = find_worst_shift(h, dx)
        g = compute_gradient(h, k_max)
        g = (g - g.mean()) / (g.std() + 1e-8)

        # Line search
        for ls_step in range(10):
            if time.time() - start_time >= budget_s - 1e-6:
                break
            h_candidate = h - eta * g
            h_candidate = project(h_candidate, n_points)
            candidate_c5 = evaluate_erdos_solution(h_candidate)
            if candidate_c5 < best_c5_bound:
                h = h_candidate
                best_c5_bound = candidate_c5
                best_h = h.copy()
                break
            else:
                eta *= 0.5

        if np.abs(best_c5_bound - evaluate_erdos_solution(h)) < 1e-6:
            h += noise_scale * np.random.randn(n_points)
            h = project(h, n_points)

        eta *= decay

    return best_h.tolist(), best_c5_bound, n_points

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1 - h, mode="full")
    return np.max(corrs) * dx

def find_worst_shift(h, dx):
    n = len(h)
    corrs = np.correlate(h, 1 - h, mode="full") * dx
    k_max = np.argmax(corrs)
    return k_max, corrs[k_max]

def compute_gradient(h, k):
    n = len(h)
    g = np.zeros(n)
    for i in range(n):
        if i + k < n:
            g[i] += (1 - h[i + k])
        if i - k >= 0:
            g[i] -= h[i - k]
    return g

def project(h, n_points):
    x = cp.Variable(n_points)
    obj = cp.Minimize(cp.sum_squares(x - h))
    constraints = [x >= 0, x <= 1, cp.sum(x) == n_points / 2]
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS)
    return x.value
