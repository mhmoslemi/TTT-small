import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use a fixed discretization size
    n_points = 70
    dx = 2.0 / n_points

    # Get initial h from external input
    h = np.copy(initial_h_values)

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h)

    MAX_ITER = 300
    iter = 0
    while iter < MAX_ITER and (time.time() - start_time) < budget_s:
        iter += 1

        # Optimization variables
        h_opt = cp.Variable(n_points, nonneg=True)
        constraints = [
            cp.sum(h_opt) == n_points / 2,
            h_opt <= 1,
        ]

        # Strategy: Minimize the worst-case correlation
        max_corr = 0.0
        max_k = 0
        for k in range(-n_points, n_points):
            if k != 0:
                A = np.zeros((n_points, n_points))
                A[np.arange(n_points), np.roll(np.arange(n_points), -k)] = 1.0
                h_shift = A @ h_opt
                corr_val = cp.sum(h_opt * (1 - h_shift))
                if corr_val.value is None:
                    corr_val.value = 0.0
                if corr_val.value > max_corr:
                    max_corr = corr_val.value
                    max_k = k

        # Goal: reduce the worst-case overlap
        objective = cp.Minimize(max_corr)
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False)

        if prob.status != cp.OPTIMAL:
            # Fallback to random perturbation if solve fails
            h = np.copy(best_h)
            perturbation = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + perturbation, 0, 1)
            h = h * (n_points / 2) / h.sum()
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_h = np.copy(h)
                best_c5 = c5
            continue

        h_new = np.clip(h_opt.value, 0, 1)
        h_new = h_new * (n_points / 2) / h_new.sum()

        c5_new = evaluate_erdos_solution(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            h = h_new  # Continue local search from improved point

    return best_h, best_c5, n_points
