import numpy as np
import scipy.optimize
import time


def evaluate(h, n_points):
    dx = 2.0 / n_points
    h = h.reshape((-1,))
    values = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(values)


def project_constraints(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    h = h + np.full_like(h, (n_points / 2 - total) / n_points)
    h = np.clip(h, 0.0, 1.0)
    return h

    h = project_constraints(h, n_points)
