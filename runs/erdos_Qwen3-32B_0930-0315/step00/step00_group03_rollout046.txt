import numpy as np
import cvxpy as cp
import time
import math

def run(seed=42, budget_s=1100, **kwargs):
    timer = time.time

    np.random.seed(seed)

    # Utilize the initial_h_values if provided
    global initial_h_values
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 70
    h = initial_h_values if 'initial_h_values' in globals() else 0.5 * np.ones(n_points)
    h = h.copy()

    dx = 2.0 / n_points

    # Enforce the constraints
    h = np.clip(h, 0, 1)
    h *= n_points / 2 / h.sum()

    def objective(h):
        max_overlap = 0
        for k in range(-n_points + 1, n_points):
            if -k <= n_points:
                shifted = np.roll(h, k)
                overlap = np.sum(h * (1 - shifted)) * dx
                max_overlap = max(max_overlap, overlap)
        return max_overlap

    def get_gradient(h):
        max_overlap = 0
        worst_k = 0
        for k in range(-n_points + 1, n_points):
            if -k <= n_points:
                shifted = np.roll(h, k)
                overlap = np.sum(h * (1 - shifted)) * dx
                if overlap > max_overlap:
                    max_overlap = overlap
                    worst_k = k
        # Compute gradient at worst k
        shifted = np.roll(h, worst_k)
        grad = - (1 - shifted) * dx
        if worst_k <= n_points and worst_k >= 0:
            grad[worst_k:] += h[:n_points - worst_k] * dx
        elif worst_k < 0:
            grad[:n_points + worst_k] += h[-worst_k:] * dx
        return grad

    best_h = h.copy()
    best_obj = objective(h)
    start_time = timer()
    iter_since_best = 0
    eta = 0.01

    while timer() - start_time < budget_s:
        # Gradient-based step
        grad = get_gradient(h)
        grad /= max(np.linalg.norm(grad), 1e-8)  # Normalize for step size

        # Line search
        h_candidate = h - eta * grad
        h_proj = project_onto_box_and_sum(h_candidate)
        obj = objective(h_proj)

        if obj < best_obj:
            h = h_proj
            best_obj = obj
            best_h[:] = h_proj[:]  # In-place copy
            iter_since_best = 0
        else:
            iter_since_best += 1

        # Learning rate adaptation
        if iter_since_best >= 100:
            eta *= 0.5
            iter_since_best = 0

        # Projection only if infeasible steps happen
        if not np.all(np.isfinite(h)):
            h = project_onto_box_and_sum(h)
    return best_h.tolist(), best_obj, n_points

def project_onto_box(h):
    return np.clip(h, 0, 1)

def project_onto_sum(h, target_sum):
    total = h.sum()
    if total <= 1e-8:
        return np.zeros_like(h)
    return h * (target_sum / total)

def project_onto_box_and_sum(h):
    # First project to box
    h = project_onto_box(h)
    # Then project to sum constraint
    h = project_onto_sum(h, n_points / 2)
    return h
