```python
import time
import numpy as np
import cvxpy as cp
from scipy.signal import correlate


def run(seed=42, budget_s=1100, **kwargs):
    # Set random seed
    np.random.seed(seed)

    # Load initial solution if provided
    initial_given = 'initial_h_values' in globals()
    if initial_given:
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Initialize with a naive pattern
        n_points = 64
        h = np.zeros(n_points)
        h[:28] = 1.0
        h = h / h.sum() * n_points / 2

    # Normalize to satisfy the constraint: sum(h) = n_points / 2
    dx = 2.0 / n_points
    h = h / h.sum() * (n_points / 2)

    # Objective function to compute C5
    def compute_c5(h, n_points):
        dx = 2.0 / n_points
        h = h.copy()
        h = np.clip(h, 0, 1)
        corr = correlate(h, 1 - h, mode="full")
        C5 = np.max(corr * dx)
        return C5

    # Function to project h into the feasible domain
    def project(h):
        h = np.clip(h, 0, 1)
        S = h.sum()
        if S <= 0:
            h = h / S
        else:
            h = h / (2 * S) + (1 - S / n_points) / (2 * S)
        return h

    start_time = time.time()

    max_iter = 5000
    for iter_ in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        # Evaluate C5
        dx = 2.0 / n_points
        h_np = np.clip(h, 0, 1)
        corr = correlate(h_np, 1 - h_np, mode="full")
        max_corr_idx = np.argmax(np.abs(corr))
        current_C5 = np.max(corr * dx)

        # Optimization problem
        def optimize_h_fast():
            h_var = cp.Variable(n_points)
            obj = cp.Minimize(corr[max_corr_idx] * dx)
            constraints = [cp.sum(h_var) == n_points / 2, h_var >= 0, h_var <= 1]
            prob = cp.Problem(obj, constraints)
            try:
                prob.solve(solver=cp.OSQP, warm_start=True, max_iter=1000, verbose=False)
                new_h = np.array(h_var.value).flatten()
                return new_h
            except:
                return h.copy()

        new_h_opt = optimize_h_fast()
        if new_h_opt is None:
            return h, current_C5, n_points

        # Project to feasible domain
        new_h = project(new_h_opt)

        # Update h if improved
        new_C5 = compute_c5(new_h, n_points)
        if new_C5 < current_C5:
            h = new_h
        elif np.random.rand() < 0.1:
            h = new_h  # occasional random step

    return h, compute_c5(h, n_points), n_points
```
