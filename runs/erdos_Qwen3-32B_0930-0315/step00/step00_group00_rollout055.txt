import numpy as np
import time
from functools import partial
from scipy.signal import correlate

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    h1 = np.concatenate((h, h))
    max_overlap = 0.0
    for k in range(1, len(h1) - 1):
        shifted = np.roll(h1, k)
        overlap = np.sum(h1 * (1 - shifted)) * dx
        max_overlap = max(max_overlap, overlap)
    return max_overlap

def perturb_h(h, n_points, n_perturb=1):
    h = np.copy(h)
    indices = np.random.choice(n_points, size=n_perturb, replace=False)
    for i in indices:
        h[i] = np.random.uniform(0, 1)
    # Project to constraint
    alpha = np.sum(h) - n_points / 2.0
    h -= alpha / n_points
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=1100, tabu_size=20, neighbors_per_iter=20, n_points=45):

    # Normalize to satisfy sum constraint
    h_best = np.clip(h_best, 0, 1)
    h_best_sum_orig = np.sum(h_best)
    h_best = h_best / h_best_sum_orig * (n_points / 2)
    
    # Tabu memory stores hashes of recent h values
    tabu_memory = set()
    
    # Initial C₅ value
    c5_bound_best = evaluate_c5(h_best, n_points)
    
    # Try to improve
    while time.time() - start_time < budget_s:
        # Generate neighbors
        neighbors = []
        for _ in range(neighbors_per_iter):
            neighbor = perturb_h(h_best, n_points)
            str_neighbor = str(neighbor)
            if str_neighbor not in tabu_memory:
                neighbors.append((str_neighbor, neighbor))
                if len(tabu_memory) >= tabu_size:
                    tabu_memory.pop()
                tabu_memory.add(str_neighbor)
        
        # Evaluate and select best candidate
        for str_neighbor, candidate in neighbors:
            if time.time() - start_time >= budget_s:
                break
            c5_current = evaluate_c5(candidate, n_points)
            if c5_current < c5_bound_best:
                h_best = candidate
                c5_bound_best = c5_current
                
    return h_best, c5_bound_best, n_points
