import numpy as np
import time
from scipy.signal import correlate

def evaluate_current_h(h, n_points):
    dx = 2.0 / n_points
    ones_h = 1.0 - h
    corr = correlate(h, ones_h, mode='full') * dx
    return np.max(corr)

def repair_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    total = h.sum()
    scale = (n_points / 2.0) / total
    return h * scale

def generate_population(n_points, population_size):
    population = []
    for _ in range(population_size):
        h = np.random.uniform(0, 1, n_points)
        h = repair_feasibility(h, n_points)
        population.append(h)
    return population

def tournament_selection(population, fitness, tournament_size=5):
    indices = np.random.choice(len(population), tournament_size, replace=False)
    best_index = indices[np.argmin(fitness[indices])]
    return population[best_index]

def crossover(parent1, parent2):
    crossover_idx = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate((parent1[:crossover_idx], parent2[crossover_idx:]))
    child2 = np.concatenate((parent2[:crossover_idx], parent1[crossover_idx:]))
    return child1, child2

def mutate(h, mutation_rate=0.15):
    mask = np.random.rand(*h.shape) < mutation_rate
    h[mask] += np.random.normal(0, 0.1, h[mask].shape)
    return repair_feasibility(h, len(h))

def run(seed=42, budget_s=1100, initial_population_size=20, initial_n_points=14):
    np.random.seed(seed)
    start_time = time.time()

    n_points = initial_n_points

    best_h = h
    best_score = evaluate_current_h(best_h, n_points)
    best_n_points = n_points
    
    t0 = time.time()
    
    while time.time() - t0 < 0.8 * budget_s and n_points < 128:
        population = generate_population(n_points, initial_population_size)
        scores = [evaluate_current_h(ind, n_points) for ind in population]
        
        # Genetic algorithm loop
        for iter in range(1000):
            if time.time() - t0 > 0.8 * budget_s:
                break
            
            new_population = population[:]
            
            for _ in range(initial_population_size // 2):
                parent1 = tournament_selection(population, scores)
                parent2 = tournament_selection(population, scores)
                child1, child2 = crossover(parent1, parent2)
                child1 = mutate(child1)
                child2 = mutate(child2)
                new_population.extend([child1, child2])
                
            # Evaluate and keep only the best individuals
            new_population = new_population[:initial_population_size]
            new_scores = [evaluate_current_h(ind, n_points) for ind in new_population]
            scores = new_scores
            population = new_population
            
            current_best = min(scores)
            current_best_idx = np.argmin(scores)
            if current_best < best_score:
                best_score = current_best
                best_h = population[current_best_idx]
        
        # Gradient descent refinement
        learning_rate = 0.01
        for step in range(50):
            if time.time() - t0 > 0.8 * budget_s:
                break
            h_corr = correlate(best_h, 1 - best_h, mode='full')
            lag = np.argmax(h_corr)
            if lag < n_points:
                gradient_direction = 1 - np.roll(best_h, lag - n_points + 1)
                best_h += learning_rate * gradient_direction
                best_h = repair_feasibility(best_h, n_points)
                new_score = evaluate_current_h(best_h, n_points)

        # Resolution doubling
        if n_points < 128 and best_score < 0.6:

            x = np.linspace(0, 2, n_points)

            best_h = interp_func(x_new)
