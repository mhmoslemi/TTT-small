import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = initial_h.copy()
    max_steps = 100000
    step = 0
    improvement_window = 100
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    def compute_c5(h, dx):
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr * dx)
        return c5

    while time.time() - start_time < budget_s and step < max_steps:
        step += 1

        if step % improvement_window == 0:
            # Perturb slightly if no improvement
            h = best_h.copy() + np.random.normal(0, 0.01, size=n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

        # Evaluate current C5
        curr_h = h.copy()
        c5 = compute_c5(curr_h, dx)
        if c5 < best_c5:
            best_h = curr_h.copy()
            best_c5 = c5

        # Identify dominant overlap
        corr = correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)

        # Gradient-based adjustment for h
        grad = np.zeros(n_points)
        for i in range(n_points):
            for j in range(i, i + n_points):
                if j >= len(corr):
                    break
                if j - i == max_idx:
                    grad[i] += 1 if (1 - h[i]) else 0  # Gradient sign from h(x)(1 - h(x + k))
                    break

        # Normalize gradient
        grad_shape = np.sqrt(np.sum((grad) ** 2))
        if grad_shape > 0:
            grad = grad / grad_shape

        h += 0.002 * grad
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)

    return best_h, best_c5, n_points
