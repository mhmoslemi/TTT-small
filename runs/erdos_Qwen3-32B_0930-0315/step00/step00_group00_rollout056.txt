import numpy as np
import time
from scipy.signal import correlate


def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Ensure the sum constraint is exactly satisfied
    def project(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if abs(current_sum - n_points / 2) > 1e-6:
            h_normalized = h_clipped * (n_points / 2) / current_sum
            return np.clip(h_normalized, 0, 1)
        return h_clipped

    # Objective function: compute C5
    def compute_c5(h):
        h_corr = np.correlate(h, 1 - h, mode="full")
        return np.max(h_corr) * dx

    # Initialize population
    pop_size = 75
    pop = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = project(h_rand)
        pop.append(h_rand)

    best_h = np.array(pop[0])
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    tournament_size = 8
    crossover_rate = 0.7
    mutation_rate = 0.05
    temp_schedule = {
        'T0': 100.0,
        'cooling': 0.95,
        'sa_interval': 50,
        'sa_mutate_rate': 0.1
    }
    T = temp_schedule['T0']

    gen = 0
    while time.time() - start_time < budget_s - 1e-3:
        # Evaluate all
        pop_c5 = [compute_c5(h) for h in pop]
        valid = [h for h in pop if (0 <= h).all() and (h <= 1).all()]
        pop = valid

        # Track best
        for i, c5 in enumerate(pop_c5):
            if c5 < best_c5:
                best_c5 = c5
                best_h = pop[i].copy()

        # Selection and breeding
        parents_idx = np.argsort(pop_c5)[:tournament_size]
        parents = [pop[i] for i in parents_idx]
        offspring = []

        for _ in range(pop_size - len(parents)):
            if np.random.rand() < crossover_rate:
                # Single-point crossover
                p1, p2 = np.random.choice(len(parents), size=2, replace=False)
                cross_idx = np.random.randint(1, n_points)
                child = np.concatenate([parents[p1][:cross_idx], parents[p2][cross_idx:]])
            else:
                parent = np.random.choice(parents)
                child = parent.copy()

            # Mutation with temperature-dependent probability
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        delta = np.random.uniform(-0.2, 0.2)
                        child[i] += delta
                        child[i] = np.clip(child[i], 0, 1)

            # Feasibility projection
            child = project(child)
            offspring.append(child)

        # Replace worst part of population
        pop = parents + offspring
        pop_c5 = [compute_c5(h) for h in pop]
        keep_idx = np.argsort(pop_c5)[:pop_size]
        pop = [pop[i] for i in keep_idx]

        # SA step
        if gen % temp_schedule['sa_interval'] == 0 and gen > 0:
            current_h = best_h.copy()
            current_c5 = best_c5
            # Perturb best_h
            new_h = current_h.copy()
            if np.random.rand() < temp_schedule['sa_mutate_rate']:
                mut_idx = np.random.randint(n_points)
                delta = np.random.uniform(-0.5, 0.5)
                new_h[mut_idx] += delta
                new_h = project(new_h)
            new_c5 = compute_c5(new_h)
            delta_c5 = new_c5 - current_c5
            if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
                best_h = new_h
                best_c5 = new_c5
            T *= temp_schedule['cooling']
        gen += 1

    return best_h.tolist(), best_c5, n_points
