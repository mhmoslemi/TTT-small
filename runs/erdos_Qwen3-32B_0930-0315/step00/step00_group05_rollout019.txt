import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    def evaluate(h, dx):
        # Compute C_5: max correlation * dx
        n = len(h)
        h = np.concatenate([h, h])  # for mode="full" emulation
        corrs = np.correlate(h, 1 - h, mode="full")[n - 1:2 * n - 1]
        c5 = corrs * dx
        return np.max(c5)

    def project(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0 or s == len(h):
            return h
        h = h * (len(h) / 2) / s
        h = np.clip(h, 0, 1)
        return h

    def optimize_h(h0, dx, budget_s_sub):
        h = cp.Variable(len(h0))
        dx = cp.Parameter(value=dx)
        constr = [
            cp.sum(h) == len(h0) / 2,
            0 <= h,
            h <= 1
        ]
        obj = cp.Minimize(calculate_corr_obj(h, dx))
        prob = cp.Problem(obj, constr)
        prob.solve(solver=cp.SCIPY, max_iter=1000, time_limit=(budget_s_sub - time.time() + t_start), verbose=False)
        if h.value is None:
            return h0
        h_proj = np.clip(h.value, 0, 1)
        h_proj = project(h_proj)
        return h_proj

    def calculate_corr_obj(h_var, dx_var):
        # Use a linear approximation to the max correlation
        n = len(h_var)
        shifts = [5, 6]  # Lags with highest influence based typical test
        corr_values = []
        for k in shifts:
            offset = k
            if offset >= n or offset < 0:
                continue
            slice_a = h_var[offset:]
            slice_b = 1 - h_var[:-offset] if offset > 0 else 1 - h_var[offset:]
            corr = cp.sum(slice_a * slice_b) * dx_var
            corr_values.append(corr)
        return cp.Maximize(cp.Minimize(cp.hstack(corr_values)))

    best_c5 = evaluate(best_h, dx)

    t_start = time.time()
    while time.time() - t_start < budget_s:

        c5 = evaluate(h_opt, dx)
