import numpy as np
import math
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 256
    dx = 2.0 / n_points

    # Use parent solution if available
    if 'initial_h_values' in globals():
        h = np.copy(initial_h_values)
    else:
        # Alternative initialization: random binary scaled to mean 0.5
        h = np.random.choice([0, 1], size=n_points)
        h = h * n_points / (2 * h.sum())

    best_h = np.copy(h)
    best_c5 = float('inf')

    max_steps = 1_000_000
    no_improvement_steps = 0
    max_no_improvement = 1_000

    perturb_counter = 0
    perturb_interval = 100
    perturb_strength = 0.05

    while time.time() - start_time < budget_s and no_improvement_steps < max_no_improvement:
        # Step 1: Compute current C5 value
        one_minus_h = 1.0 - h
        conv = correlate(h, one_minus_h, mode='full') * dx
        current_c5 = np.max(conv)

        # Step 2: Check if we found a new best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h[:]
            no_improvement_steps = 0
        else:
            no_improvement_steps += 1

        # Step 3: Find the lag with maximum overlap
        k_max = np.argmax(conv)
        k_max = (k_max - n_points + 1)  # shift to -n_points + 1 to n_points - 1

        # Step 4: Position optimization (open-loop shift)
        if -n_points < k_max < n_points:
            # Compute contributions of each h[i] to overlap at k_max
            indices = np.arange(n_points)
            from_idx = np.clip(indices - k_max, 0, n_points - 1)
            overlap_contrib = h * one_minus_h[from_idx]

            # Compute pseudo-gradient
            grad = one_minus_h[from_idx] - h[from_idx + k_max] if k_max > 0 else - h[from_idx]

            # Update positions with gradient step
            step_size = 0.1 / (1 + 0.001 * perturb_counter)
            h += step_size * grad

            # Project to feasible set and normalize
            h = np.clip(h, 0, 1)
            h *= n_points / (2 * h.sum())

        # Step 5: Magnitude optimization (convex near k_max)
        try:
            # Create convex problem for adjustment near k_max
            h_vars = cp.Variable(n_points)
            obj = cp.Minimize(2.0 / n_points * cp.norm(
                cp.multiply(h_vars, 1 - cp.roll(h_vars, k_max)), p=np.inf)
            )
            constraints = [cp.sum(h_vars) == n_points / 2,
                           h_vars >= 0,
                           h_vars <= 1]
            prob = cp.Problem(obj, constraints)
            result = prob.solve(solver=cp.SCIP, verbose=False)

            if prob.status == 'optimal':
                h = h_vars.value[:]
                h = np.clip(h, 0, 1)
                h *= n_points / (2 * h.sum())
        except:
            pass

        # Step 6: Perturb and re-normalize
        if perturb_counter % perturb_interval == 0:
            h += np.random.normal(0, perturb_strength, h.shape)
            h = np.clip(h, 0, 1)
            h *= n_points / (2 * h.sum())
            perturb_counter = 0

        perturb_counter += 1

    # Final validation
    sum_h = np.sum(h)
    h = np.clip(h, 0, 1)
    h *= n_points / (2 * h.sum())
    conv = correlate(h, 1 - h, mode='full') * dx
    final_c5 = np.max(conv)

    return best_h, best_c5, n_points
