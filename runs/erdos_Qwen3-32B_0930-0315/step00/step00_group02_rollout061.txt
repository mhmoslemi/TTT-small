import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 160
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(n_points)
        mid = int(n_points / 2)
        h[:mid] = 1.0
        h[mid:] = 0.0

    # Enforce normalization: sum(h) == n_points / 2
    h = h * (n_points / 2.0) / h.sum()

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    def compute_c5(h):
        h = h.flatten()
        c = scipy.signal.correlate(h, 1 - h, mode='full')
        return np.max(c) * dx

    start_time = time.time()
    last_improvement_time = start_time
    patience = 0.01

    perturb_period = 500
    perturb_count = 0
    step_size = 0.1

    while time.time() - start_time < budget_s:

        # Compute the overlap at each shift
        c = scipy.signal.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(c)
        k_max = max_idx - len(h) + 1

        # Prevent modulo wrap-around computation error
        k_max = max(0, min(k_max, len(h) - 1))

        # Compute gradient at current k_max
        grad = np.zeros_like(h)
        for i in range(len(h)):
            j1 = (i + k_max) % len(h)
            j2 = (i - k_max) % len(h)
            grad[i] = -(1 - h[j1]) + h[j2]

        # Gradient update with adaptive step size
        h = h - step_size * grad

        # Project back into [0, 1] and normalize
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / h.sum()

        # Evaluate the new C₅
        current_c5 = compute_c5(h)

        # Check for improvement
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            last_improvement_time = time.time()
            print(f'Improved C5 to {best_c5:.8f} at {time.time() - start_time:.2f}s')
            step_size *= 1.01
        elif time.time() - last_improvement_time > patience:
            break

        # Perturb occasionally
        if perturb_count % perturb_perturb_period == 0:
            h_perturb_mask = np.random.rand(len(h)) < 0.1
            h[h_perturb_mask] += np.random.normal(0, 0.01, size=len(h))[h_perturb_mask]
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2.0) / h.sum()
            perturb_count = 0

        perturb_count += 1

    # Final evaluation to ensure correctness
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h.tolist(), final_c5, n_points
