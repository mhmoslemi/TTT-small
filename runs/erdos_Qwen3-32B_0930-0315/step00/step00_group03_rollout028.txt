import numpy as np
import cvxpy as cp
import time
from numpy.fft import fft, ifft, fftshift

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    timeout = budget_s - 1.0  # Leave 1s for final check and exit

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = np.array(initial_h_values, dtype=float)
        n_half = n_points / 2
        if abs(np.sum(h) - n_half) > 1e-6:
            raise ValueError("Initial h is not feasible!")
    else:
        n_points = 70  # use same n as parent if missing
        h = 0.5 * np.ones(n_points)
        n_half = n_points / 2

    # Helper: compute C5
    def compute_c5(h):
        dx = 2.0 / n_points
        h1 = np.fft.fft(h)
        h2 = np.fft.fft(1 - h)
        corrs = np.fft.ifft(h1 * h2).real * dx
        return np.max(corrs)

    # Helper: compute max-k and projective search index
    def find_worst_k(h):
        h_fft = np.fft.fft(h)
        corrs_fft = h_fft * np.conj(np.fft.fft(1 - h))
        corrs = np.fft.ifft(corrs_fft).real * (2.0 / n_points)
        k = np.argmax(np.abs(corrs))
        return k

    # Gradient for a given shift k
    def compute_gradient(h, k):
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full')
        grads = np.zeros_like(h)
        max_index = np.argmax(corr)
        if not 0 <= max_index < n:
            k = max_index - n
        hcopy = h.copy()
        rhs = (1 - h[(np.arange(n) + k) % n]) * dx
        lhs = h[(-k + np.arange(n)) % n] * dx
        grads[(np.arange(n) + k) % n] = rhs
        grads[(-k + np.arange(n)) % n] -= lhs
        return grads

    # Feasibility projection onto normed simplex with [0,1] box
    def project_to_simplex(h):
        x = np.clip(h, 0, 1)
        diff = np.sum(x) - n_half
        if abs(diff) < 1e-6:
            return x
        solver = cp.ECOS
        v = cp.Variable(n_points)
        obj = cp.Minimize(cp.sum_squares(v - x))
        constraints = [0 <= v, v <= 1, cp.sum(v) == n_half]
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=solver)
        return np.array(v.value)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    learning_rate = 0.01
    max_iter = 1000
    no_improvement = 0
    for it in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= timeout:
            break
        k = find_worst_k(h)
        grad = compute_gradient(h, k)
        grad_len = np.linalg.norm(grad)
        if grad_len == 0:
            break  # no direction to improve
        grad = grad / grad_len
        h_candidate = h - learning_rate * grad
        h_projected = project_to_simplex(h_candidate)
        c5_new = compute_c5(h_projected)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_projected.copy()
            no_improvement = 0
        else:
            no_improvement += 1
        if no_improvement > 20:
            learning_rate *= 0.8
        if learning_rate < 1e-4:
            learning_rate = 1e-4
        h[:] = h_projected[:]
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
