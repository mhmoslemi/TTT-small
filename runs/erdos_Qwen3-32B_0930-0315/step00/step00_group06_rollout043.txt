import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    
    # Config
    n_points = 256
    np.random.seed(seed)

    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values[:n_points]  # Ensure it fits the resolution
    else:
        h = np.clip(np.random.rand(n_points), 0, 1)  # Random initial h in [0, 1]
    h /= h.sum() * 2  # Normalize to ensure sum(h) * dx = 1 ⇒ sum(h) == 128 for n=256

    best_h = h.copy()
    best_c5 = float('inf')
    dx = 2.0 / n_points
    learning_rate = 0.1
    lr_decay = 0.999  # exponential learning rate decay
    perturb_interval = 100  # iterations between random perturbation
    perturb_strength = 1e-3

    while time.time() - start_time < budget_s:
        corr = correlate(h, 1 - h, mode="full")
        overlap = corr * dx
        c5 = overlap.max()
        if c5 <= best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Gradient step
        shifts = np.arange(1, n_points)
        max_index = np.argmin(correlate(h, h, mode="full")[n_points - 1 : n_points + 1]) + n_points - 1
        target_corr = correlate(h, 1 - h, mode="same")
        
        # Project gradient from overlap to h
        grad = np.zeros_like(h)
        grad[:n_points - (max_index - n_points + 1)] = -(1 - h[(max_index - n_points + 1):])
        grad[(max_index - n_points + 1):] = -h[:(n_points - (max_index - n_points + 1))]

        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        
        # Normalize to volume constraint
        h /= h.sum() * 2
        
        # Learning rate decay
        learning_rate *= lr_decay
        
        # Periodic random perturbation
        if (time.time() - start_time) % perturb_interval < 1e-3:
            perturbation = np.clip((np.random.rand(n_points) - 0.5) * perturb_strength, -0.01, 0.01)
            h += perturbation
            h = np.clip(h, 0, 1)
            h /= h.sum() * 2

    return best_h.tolist(), best_c5, n_points
