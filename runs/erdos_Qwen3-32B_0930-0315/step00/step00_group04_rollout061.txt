import numpy as np
import math
import time
import copy
from scipy.interpolate import CubicSpline
from scipy.signal import correlate

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    h1 = h.copy()
    h1[h1 < 0] = 0
    h1[h1 > 1] = 1
    integrals = correlate(h1, 1 - h1, mode='full') * dx
    return np.max(integrals)

def repair_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum - n_points / 2) > 1e-8:
        h += (n_points / 2 - h_sum) / n_points
        h = np.clip(h, 0, 1)
    return h

def crossover(parent1, parent2):
    idx = np.random.randint(2, len(parent1) - 2)
    child1 = np.concatenate([parent1[:idx], parent2[idx:]])
    child2 = np.concatenate([parent2[:idx], parent1[idx:]])
    return repair_feasibility(child1, len(parent1)), repair_feasibility(child2, len(parent2))

def mutate(h, mut_rate=0.1, amount=0.2):
    h = h.copy()
    n_points = len(h)
    indices = np.random.choice(n_points, int(n_points * mut_rate))
    h[indices] += np.random.randn(len(indices)) * amount
    return repair_feasibility(h, n_points)

def selection(population, k=5):
    return np.argsort(population)[::-1][:k]

def downsample(h, factor=2):
    n = len(h)
    new_len = n // factor
    h = h[:new_len*factor].reshape(new_len, factor)
    return np.mean(h, axis=1)

def interpolate_h(h, new_len):
    cs = CubicSpline(np.linspace(0, 2, len(h)), h)
    x_new = np.linspace(0, 2, new_len)
    return np.clip(cs(x_new), 0, 1)

def gradient_step(h, learning_rate=0.1, n_points=None):
    if n_points is None:
        n_points = len(h)
    h = h.copy()
    dx = 2.0 / n_points
    integrals = correlate(h, 1 - h, mode='full') * dx
    max_idx = np.argmax(integrals)
    
    # Find shifted h where overlap happens.
    shifted_h = np.zeros(n_points)
    if 0 <= max_idx < n_points:
        shifted_h[:n_points - max_idx] = h[max_idx:]
    elif max_idx < 2 * n_points:
        shift = max_idx - n_points + 1
        shifted_h[shift:] = h[:n_points - shift]
    
    gradient = (1 - shifted_h)
    
    h -= learning_rate * gradient
    return repair_feasibility(h, n_points)

    pop = [h + np.random.randn(n) * 0.1 for _ in range(population_size)]
    pop = [repair_feasibility(x, n) for x in pop]
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, n)
    
    # Adaptive configuration
    mutation_rate = 0.2
    generations = 0

    while time.time() - start_time < budget_s * 0.7 and generations < 100:
        counter += 1
        generations += 1
        evals = [evaluate_c5(ind, n) for ind in pop]
        
        idxs = selection(evals)
        survivors = np.take(pop, idxs)

            parent1, parent2 = np.random.choice(survivors, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            child1_m = mutate(child1, mut_rate=mutation_rate)
