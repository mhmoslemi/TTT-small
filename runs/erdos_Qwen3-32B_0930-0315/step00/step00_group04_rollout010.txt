import numpy as np
import time

def run(seed=42, budget_s=1100):
    import numpy as np
    import time
    from scipy.interpolate import CubicSpline

    np.random.seed(seed)

    def compute_overlap(h, dx):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    def repair_feasibility(h, n_points):
        h = np.clip(h, 0, 1)
        h = h / h.sum() * (n_points / 2)
        return h

    def simulate_annealing(h, T, alpha, max_steps, update_period):
        best_h = h.copy()
        best_c5 = compute_overlap(best_h, 2.0 / len(h))
        steps_since_update = 0
        for step in range(max_steps):
            if time.time() - start_time > budget_s:
                break
            h_proposal = h + np.random.normal(0, T, h.shape)
            h_proposal = repair_feasibility(h_proposal, len(h))
            c5_proposal = compute_overlap(h_proposal, 2.0 / len(h))
            if c5_proposal < best_c5 or np.random.rand() < np.exp((best_c5 - c5_proposal) / T):
                best_h, best_c5 = h_proposal, c5_proposal
                h = h_proposal
                steps_since_update = 0
            else:
                steps_since_update += 1
                if steps_since_update > update_period:
                    T /= 4
                    steps_since_update = 0
            T = max(1e-4, T * alpha)
        return best_h, best_c5

    def refine_with_grid_search(h, n_steps=100):
        h = h.copy()
        dx = 2.0 / len(h)
        best_c5 = compute_overlap(h, dx)
        for step in range(n_steps):
            if time.time() - start_time > budget_s:
                break
            # Compute maximum overlap and its position
            corr = np.correlate(h, 1 - h, mode='full') * dx
            best_k = np.argmax(corr)

            # Determine shifted alignment and gradient
            h_shifted = np.roll(h, len(h) - 1 - best_k)  # aligning for coefficient k
            updated_h = h - (1 - h_shifted)[:len(h)] * 0.05
            updated_h = repair_feasibility(updated_h, len(h))
            c5 = compute_overlap(updated_h, dx)
            if c5 < best_c5:
                best_c5 = c5
                h = updated_h
        return h, best_c5

    def refine_with_cubic_interpolation(h):
        spl = CubicSpline(np.linspace(0, 2, len(h)), h)
        new_n = len(h) * 2
        x_new = np.linspace(0, 2, new_n)
        h_new = spl(x_new)
        h_new = repair_feasibility(h_new, new_n)
        return h_new

    # Initialize with the given solution if provided, otherwise use a random uniform distribution
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_values = initial_h_values
    else:
        n_points = 14
        h_values = np.random.uniform(0, 1, n_points)
        h_values = repair_feasibility(h_values, n_points)

    h_values = h_values.copy()
    start_time = time.time()

    h_coarse, c5_coarse = simulate_annealing(h_values, T=0.1, alpha=0.995, max_steps=500, update_period=50)
    h_coarse, c5_coarse = refine_with_grid_search(h_coarse, n_steps=200)

    if time.time() - start_time < budget_s * 0.8:
        h_fine = refine_with_cubic_interpolation(h_coarse)
        if len(h_fine) < 1000:
            h_fine, c5_fine = simulate_annealing(h_fine, T=0.05, alpha=0.995, max_steps=1000, update_period=50)
            h_fine, c5_fine = refine_with_grid_search(h_fine, n_steps=200)
            return h_fine, c5_fine, len(h_fine)
    return h_coarse, c5_coarse, len(h_coarse)
