import numpy as np

def project_simplex(v, s):
    u = np.sort(v)[::-1]
    cumulative_sum = np.cumsum(u)
    ind = np.arange(1, len(u) + 1)
    ind = ind * u
    t = (cumulative_sum - s) / ind
    t = t[t >= 0]
    if len(t) == 0:
        return np.clip(v, 0, 1)
    max_k = np.max(t)
    H = np.sum(u >= max_k)
    return np.clip(v - max_k, 0, 1)


def compute_overlap(h, dx):
    n = len(h)
    max_k = n
    overlap = np.zeros(n * 2 - 1)
    for k in range(max_k):
        valid = (k < n)
        if not valid:
            break
        indices0 = np.arange(n - k)
        indices1 = indices0 + k
        overlap[k] = np.sum(h[indices0] * (1 - h[indices1])) * dx
    for k in range(max_k - 1, 0, -1):
        indices0 = np.arange(n - k)
        indices1 = indices0 + k
        overlap[n + k - 1] = np.sum(h[indices1] * (1 - h[indices0])) * dx
    return overlap


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Start with the provided initial_h_values
    h = np.array(initial_h_values)

    # Scale to satisfy the integral constraint ∑h = n_points / 2
    n_points = len(h)
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)

    # Normalize to ensure even rounding in next step
    h = project_simplex(h, target_sum)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution((best_h, len(best_h)))

    start_time = time.time()
    budget_used = 0

    dx = 2.0 / n_points

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        return evaluate_erdos_solution((h, n_points))

    while time.time() - start_time < budget_s:
        # Compute overlaps for all possible shifts
        overlap = compute_overlap(h, dx)
        max_overlap_idx = np.argmax(overlap)
        max_overlap = overlap[max_overlap_idx]

        # Early stopping if we've made no improvement
        if max_overlap == best_c5:
            break

        # Compute gradient for the worst lag
        k = max_overlap_idx - n_points + 1
        if k >= 0 and k < n_points:
            grad = np.zeros_like(h)
            for i in range(n_points - k):
                grad[i] += 1 - h[i + k]
                grad[i + k] -= h[i]
        elif k < 0:
            # Handle k < 0
            k = abs(k)
            for i in range(n_points - k):
                grad[i + k] += 1 - h[i]
                grad[i] -= h[i + k]

        # Gradient-based optimization with projection
        learning_rate = 0.05
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = project_simplex(h, np.sum(best_h))  # Maintain total integral

        # Check for improvement
        c5 = evaluate_erdos_solution((h, n_points))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        budget_used = time.time() - start_time

    h_values = best_h
    c5_bound = evaluate_erdos_solution((h_values, len(h_values)))
    return h_values, c5_bound, len(h_values)
