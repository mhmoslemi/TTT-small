import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap_integral(h, dx):
    h = np.asarray(h)
    corr = correlate(h, 1 - h, mode='full')
    C5 = np.max(corr)
    return C5 * dx

def project_lagrange(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = h.sum()
    delta = (n_points / 2) - current_sum
    h += delta / n_points
    h = np.clip(h, 0, 1)
    return h

    iters_since_improvement = 0

    while time.time() - start_time < budget_s:
        # Compute current C5 and the lag with max overlap
        h_ext = np.concatenate([h, np.zeros(n_points - 1)])
        anti_h = 1 - h
        anti_h_ext = np.concatenate([anti_h, np.zeros(n_points - 1)])
        corr = correlate(h_ext, anti_h_ext, mode='full')
        C5 = np.max(corr) * dx
        max_lag = np.argmax(corr)

        # Compute gradient
        grad = np.zeros(n_points)

            if i + max_lag < n_points:
                grad[i] += (1 - h[i + max_lag])
            if i >= max_lag:
                grad[i] -= h[i - max_lag]

        grad *= dx  # scale properly with dx
        h -= alpha * grad / n_points  # add gradient update

        h = project_lagrange(h, n_points)

        # Perturbation every 50 iterations to escape local minima
        if (iters_since_improvement % 50 == 0):
            noise = np.random.normal(0, 0.01, n_points)
            h += noise
            h = project_lagrange(h, n_points)

        # Evaluate and check for improvement
        c5 = evaluate_overlap_integral(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            iters_since_improvement = 0
            alpha = 0.1  # reset learning rate if improvement is seen
        else:
            iters_since_improvement += 1
            if iters_since_improvement > 100:
