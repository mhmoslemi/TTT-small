import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Use 160 samples as a reasonable balance between accuracy and speed
    n_points = 160
    dx = 2.0 / n_points
    
    # Use provided initial_h_values if available, otherwise create new random one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy().reshape(-1)
        n_points = len(h)
        if h.size > 0 and abs(np.sum(h) - n_points / 2) > 1e-8:
            scale_factor = (n_points / 2) / np.sum(h)
            h *= scale_factor
    else:
        # Initialize with a balanced random starting point
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Scale to meet the integral constraint
        
    initial_val = h.copy()
    
    # Padded 1 - h for correlation
    def compute_c5(h):
        onesMinusH = 1 - h
        corr = correlate(h, onesMinusH, mode='full') * dx
        return np.max(corr)
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    best_time = time.time()
    
    # Gradient descent parameters and settings
    alpha = 0.01
    max_iter = 10000
    no_improvement_threshold = 100
    no_improvement_count = 0
    tic = time.time()
    
    for iter_idx in range(max_iter):
        # Calculate correlation with (1 - h)
        onesMinusH = 1.0 - h
        corr = correlate(h, onesMinusH, mode='full')
        corr = np.roll(corr, -n_points)  # Move index 0 to center (for h[0] to h[k] shift)
        
        # Find the lag with the maximum overlap
        max_corr = np.max(corr)
        max_idx = np.argmax(corr)
        
        # For the chosen lag, compute gradient
        k = max_idx - n_points
        k = k % n_points  # In Full mode, index = n_points + k is the shift of h by k
        
        grad_h = np.zeros(n_points)
        for i in range(n_points):
            j1 = (i + k) % n_points
            j2 = (i - k + n_points) % n_points
            grad_h[i] = (1 - h[j1]) - h[j2]
        
        # Update h
        h = h - alpha * grad_h
        
        # Clip to [0, 1] and project onto the constraint space (sum = n / 2)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute new C5 and check improvement
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            best_time = time.time()
            no_improvement_count = 0
            # Adaptive learning rate increase if improving
            alpha = min(alpha * 1.05, 0.1)
        else:
            no_improvement_count += 1
            # Reduce learning rate if no improvement
            alpha *= 0.99
        
        # Perturbation to escape local optima
        if iter_idx % 500 == 0 and iter_idx > 0:
            perturbation = np.random.randn(n_points) * 0.02
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            alpha = 0.01
        
        # Check time remaining
        if time.time() - tic > budget_s * 0.98:
            break
    
    # Final check of constraint
    h = best_h
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    return h.tolist(), best_c5, n_points
