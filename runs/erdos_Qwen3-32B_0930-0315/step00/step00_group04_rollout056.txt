import numpy as np
import time
from scipy.ndimage import zoom
import evaluate_erdos_solution as evaluator

def compute_overlap(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def repair_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if abs(sum_h) > 0:
        h = h * (n_points / 2) / sum_h
    else:
        h = np.ones(n_points) * 0.5
    return h

def simulated_annealing(h, T=1.0, alpha=0.99, steps=1000, n_points=None):
    if n_points is None:
        n_points = h.size
    best_h = h.copy()
    best_C = compute_overlap(best_h, n_points)
    for _ in range(steps):
        h_candidate = h + np.random.normal(0, T, n_points)
        h_candidate = repair_feasibility(h_candidate, n_points)
        C_candidate = compute_overlap(h_candidate, n_points)
        if C_candidate < best_C or np.random.rand() < np.exp(-(C_candidate - best_C) / T):
            best_h, best_C = h_candidate, C_candidate
        else:
            h = best_h.copy()
        h = repair_feasibility(h, n_points)
        T *= alpha
    return best_h, best_C

def gradient_descent_step(h, lr=0.01, n_points=None):
    if n_points is None:
        n_points = h.size
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full') * dx
    k_max = np.argmax(corr)
    shift = k_max - (n_points - 1)
    h_shifted = np.roll(h, -shift)
    gradient = (1 - h_shifted)[:(n_points + k_max + 1) - n_points]
    gradient = np.pad(gradient, (0, n_points - len(gradient)), 'constant', constant_values=(0, 0))
    h -= lr * gradient
    h = repair_feasibility(h, n_points)
    return h

def interpolate_h(h_coarse):
    n_points = h_coarse.size * 2
    return zoom(h_coarse, 2, order=3, mode='nearest')

def optimize_h(initial_h, n_points=None, max_time=1100):
    start_time = time.time()
    if n_points is None:
        n_points = initial_h.size
    h_best = repair_feasibility(initial_h, n_points)
    C_best = compute_overlap(h_best, n_points)

    # Phase 1: Simulated Annealing (Global Optimization)
    h_sa, C_sa = simulated_annealing(h_best, T=0.5, alpha=0.95, steps=500, n_points=n_points)
    if C_sa < C_best:
        h_best, C_best = h_sa, C_sa

    # Phase 2: Gradient Descent (Local Refinement)
    for _ in range(1000):
        if time.time() - start_time >= max_time:
            break
        h_new = gradient_descent_step(h_best, lr=0.01, n_points=n_points)
        C_new = compute_overlap(h_new, n_points)
        if C_new < C_best:
            h_best, C_best = h_new, C_new
            h_best = repair_feasibility(h_best, n_points)

    return h_best, C_best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # Start with given or synthetic initial h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_coarse = zoom(h_init, 1 / 3, order=3)[[0, 2, 4, 6, 8, 10, 12]]
        h_coarse = repair_feasibility(h_coarse, n_points=h_coarse.size)
    else:
        h_coarse = np.random.rand(14)
        h_coarse = repair_feasibility(h_coarse, n_points=14)

    # Coarse-resolution optimization
    h1, c1 = optimize_h(h_coarse, n_points=14, max_time=budget_s * 0.25)

    # Mid-resolution optimization
    h2 = interpolate_h(h1)
    h2, c2 = optimize_h(h2, n_points=28, max_time=budget_s * 0.5)

    # Final-resolution optimization
    h3 = interpolate_h(h2)
    h3, c3 = optimize_h(h3, n_points=56, max_time=budget_s * 0.25)

    best_idx = np.argmin([c1, c2, c3])
    h_best = [h1, h2, h3][best_idx]
    c_best = [c1, c2, c3][best_idx]

    while time.time() - start_time < budget_s * 0.05:
        h = h_best + np.random.normal(0, 0.05, h_best.shape)
        h = repair_feasibility(h, n_points=h_best.size)
        c = compute_overlap(h, h_best.size)
        if c < c_best:
            h_best, c_best = h, c
        # Early breaking with small increments
        break

    final_c5 = compute_overlap(h_best, h_best.size)
    return h_best, final_c5, h_best.size
