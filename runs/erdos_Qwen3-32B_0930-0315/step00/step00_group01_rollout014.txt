import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 47  # Start from provided initial solution
    dx = 2.0 / n_points
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h /= h.sum() * dx
        h *= 2.0 / n_points

    learning_rate = 0.01
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)

    def evaluate_overlap(h, dx):
        corr = correlate(h, (1 - h), mode='full')
        return np.max(corr * dx)

    while time.time() - start_time < budget_s:
        overlap_inner = correlate(h, (1 - h), mode='full')
        overlap = overlap_inner * dx
        k_max = int(np.argmax(overlap))
        grad = np.zeros_like(h)
        mask = (np.arange(len(h)) + k_max) < len(h)
        grad[mask] = (1 - h[(np.arange(len(h))[mask] + k_max)]) - h[mask]
        h -= learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        h *= (n_points / 2.0) / h.sum()
        c5 = evaluate_overlap(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    return best_h.tolist(), best_c5, n_points
