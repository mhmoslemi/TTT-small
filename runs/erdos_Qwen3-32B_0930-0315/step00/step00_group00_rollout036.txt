import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, n_points=45, tabu_size=15, neighbors_per_iter=20):
    np.random.seed(seed)
    dx = 2.0 / n_points
    h = np.copy(initial_h_values) if 'initial_h_values' in globals() else np.random.randint(0, 2, n_points).astype(float)
    h = h / h.sum() * (n_points / 2)
    h = np.clip(h, 0.0, 1.0)

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h)

    start_time = time.time()
    tabu_list = [np.copy(h)]

    while time.time() - start_time < budget_s:
        neighbors = []

        for _ in range(neighbors_per_iter):
            nh = np.copy(h)

            # Mutation 1: perturb 1-5 positions
            indices = np.random.choice(n_points, size=np.random.randint(1, 6), replace=False)
            shift = np.random.uniform(-0.1, 0.1, indices.shape)
            nh[indices] += shift
            nh = np.clip(nh, 0.0, 1.0)
            nh = _project_to_sum(nh, n_points / 2)

            # Mutation 2: small index shift
            nh = np.roll(h, np.random.randint(-1, 2))

            neighbors.append(nh)

        for nh in neighbors:
            if not any(np.allclose(nh, t, atol=1e-6) for t in tabu_list[-tabu_size:]):
                c5 = evaluate_erdos_solution(nh)
                if c5 < best_c5:
                    best_h, best_c5 = np.copy(nh), c5
                    h = np.copy(nh)
                tabu_list.append(np.copy(h))
                if len(tabu_list) > tabu_size:
                    tabu_list.pop(0)
                break  # Perturb and move on

        if abs(best_c5 - evaluate_erdos_solution(best_h)) < 1e-9:
            break  # Exit if no progress

    return best_h, best_c5, n_points

def _project_to_sum(a, target_sum):
    a_sum = a.sum()
    if a_sum == 0:
        return a  # Unlikely
    scale = target_sum / a_sum
    a = a * scale
    a = np.clip(a, 0.0, 1.0)
    return a
