import numpy as np
import time
import cvxpy as cp
from scipy.ndimage import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    # Load initial_h_values if it exists
    try:
        n_points = len(initial_h_values)
        h = initial_h_values.copy()
    except:
        # Fall back to uniform if no initial solution is provided
        n_points = 80
        h = np.full(n_points, 0.5)

    start_time = time.time()
    best_h = h.copy()
    best_c5 = float('inf')

    # Grid spacing
    dx = 2.0 / n_points

    # Ensure the integral constraint is satisfied
    def project(h):
        h = np.clip(h, 0, 1)
        mean_h = (n_points / 2) / np.sum(h)
        return h * mean_h

    def compute_c5(h):
        h_vals = project(h)
        one_minus_h = 1 - h_vals
        corr = correlate(h_vals, one_minus_h, mode='full') * dx
        return np.max(corr)

    # Main gradient-based descent loop
    learning_rate = 0.1
    max_iter = 10000
    patience = 200
    no_improve_count = 0
    h = project(h)

    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        # Compute C5 and worst lag
        h_vals = project(h)
        one_minus_h = 1 - h_vals
        corr = correlate(h_vals, one_minus_h, mode='full') * dx
        worst_lag = np.argmax(corr)
        worst_c5 = corr[worst_lag]
        
        if worst_c5 < best_c5:
            best_c5 = worst_c5
            best_h = h_vals.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Short-circuit if not improving
        if no_improve_count > patience:
            if learning_rate > 1e-4:
                learning_rate *= 0.9
                no_improve_count = 0
            else:
                break

        # Compute the gradient w.r.t h at the worst overlap
        # Express it as a matrix multiplication with J(x)
        def grad_J(x_k):
            J_k = np.zeros(n_points)
            J_k[0:max(x_k, 0)] = -np.roll(x_k, -x_k).copy()[max(x_k, 0):]
            J_k[n_points + x_k:] = -x_k[:n_points - x_k] if x_k < 0 else 0
            return J_k

        grad = grad_J(worst_lag - n_points)
        unit_dir = -grad / (np.linalg.norm(grad) + 1e-10)
        
        h_candidate = h_vals + learning_rate * unit_dir
        h_candidate = project(h_candidate)
        
        h = h_candidate.copy()

    # Last attempt with convex optimization to refine the best H found so far
    try:
        # Use projection into [0,1]^n with mean constraint
        x = cp.Variable(n_points)
        objective = cp.Minimize(0)
        constraints = [
            x >= 0,
            x <= 1,
            cp.sum(x) == n_points / 2
        ]
        problem = cp.Problem(objective, constraints)
        
        # Refined projection to feasible region
        def convex_project(h_vec):
            x.value = h_vec
            problem.solve(solver=cp.ECOS, verbose=False)
            return x.value

        best_h = convex_project(best_h)

    except Exception:
        # Use simple projection if convex is not available
        best_h = project(best_h)

    c5_final = compute_c5(best_h)
    return (best_h.tolist(), c5_final, n_points)
