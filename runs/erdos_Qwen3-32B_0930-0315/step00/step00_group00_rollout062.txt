import numpy as np
import time
from scipy.signal import correlate
from typing import Tuple, List

def run(seed=42, budget_s=1100, **kwargs):

    np.random.seed(seed)

    # Initial data
    h = initial_h_values.copy()
    n_points = h.shape[0]

    # Precompute dx
    dx = 2.0 / n_points

    # Enforce integral constraint exactly
    def project_h(h):
        h_clipped = np.clip(h, 0, 1).copy()
        total = np.sum(h_clipped)
        if total == 0:
            return h.copy()
        scale = (n_points / 2.0) / total
        h_clipped = np.clip(h_clipped * scale, 0, 1)
        # Enforce the exact integral constraint
        while np.sum(h_clipped) < n_points / 2.0:
            h_clipped = np.clip(h_clipped + 1e-4, 0, 1)
        h_clipped = np.clip(h_clipped, 0, 1).copy()
        return h_clipped

    # Calculate C₅ from current h
    def evaluate(h):
        cor = correlate(h, 1 - h, mode='full')
        c5 = max(cor) * dx
        return c5

    # Main search with dual grid optimization - coarse grid first, then refine
    # Initialize best result
    best_h = h.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    # Parameters
    n_iterations = 2000
    n_candidates = 1000
    convergence_window = 10
    converge_threshold = 1e-5

    # Track progress to detect convergence
    best_c5_history = [best_c5] * (convergence_window + 1)

    for iter in range(n_iterations):
        if time.time() - start_time > budget_s:
            break

        # Determine the best current overlaps (k shifts)
        hcorr = correlate(h, 1 - h, mode='full') * dx
        worst_k = np.argmax(hcorr)

        # Estimate redundancy scores R(i): how much h[i] affects the max corr
        corr_columns = np.array([correlate(h[[i]], 1 - h[[i]], mode='full')[worst_k] for i in range(n_points)])
        redundancy_weights = np.abs(corr_columns) + 1e-4

        # Normalize to get probabilities for random sampling
        prob = redundancy_weights / redundancy_weights.sum()

        # Generate candidates by perturbing h at important indices
        candidates = []
        for _ in range(n_candidates):
            i = np.random.choice(n_points, p=prob)
            delta = (np.random.rand() - 0.5) * 0.5
            new_h = h.copy()
            new_h[i] = np.clip(new_h[i] + delta, 0, 1)
            new_h = project_h(new_h)
            candidates.append(new_h)

        # Evaluate candidates
        local_best_c5 = best_c5
        local_best_h = h.copy()
        for candidate in candidates:
            c5 = evaluate(candidate)
            if c5 < local_best_c5:
                local_best_c5 = c5
                local_best_h = candidate

        # Update if improved
        if local_best_c5 < best_c5:
            best_c5 = local_best_c5
            best_h = local_best_h
            best_c5_history = [best_c5] * convergence_window + [best_c5]
        else:
            best_c5_history.append(best_c5)
            best_c5_history = best_c5_history[-convergence_window:]

        # Check convergence
        if iter > convergence_window:
            c5_diff = abs(best_c5 - min(best_c5_history[-convergence_window:]))
            if c5_diff < converge_threshold:
                break

    h = project_h(best_h)
    c5 = evaluate(h)
    return h, c5, n_points
