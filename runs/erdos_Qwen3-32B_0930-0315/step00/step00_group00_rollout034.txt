import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Load initial h
    h = initial_h_values[:]
    h = h.astype(np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def project_simplex(v, s=target_sum):
        v = np.sort(v)[::-1]
        cumulative = v.cumsum()
        k = np.argmax(v[k] > (cumulative[k] - s) / (k + 1) for k in range(n_points))
        t = max((cumulative[k] - s) / (k + 1), 0)
        return np.clip(v - t, 0, 1)

    def compute_corr(h):
        return np.correlate(h, 1 - h, mode="full") * dx

    def evaluate(h):
        corr = compute_corr(h)
        return np.max(corr)

    def compute_gradients(h, critical_lag):
        grad = np.zeros_like(h)
        corr = compute_corr(h)
        max_val = corr[critical_lag]
        mask = np.isclose(corr, max_val)

        for lag in np.where(mask)[0]:
            start = 0
            end = n_points
            if 0 <= lag < n_points:
                idx = (np.arange(n_points) + lag) % n_points
                grad[lag != 0] -= 2 * h * h[idx]
                grad[lag != 0] += 1 - h[idx]
            elif lag >= n_points:
                idx = lag - n_points
                grad[idx:] -= 2 * h[idx:] * h[:n_points - idx]
                grad[idx:] += 1 - h[:n_points - idx]
            elif lag < 0:
                idx = -lag
                grad[:n_points - idx] -= 2 * h[:n_points - idx] * h[idx:]
                grad[:n_points - idx] += 1 - h[idx:]

        return grad * dx

    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = start_time
    alpha = 0.01
    patience = 500
    counter = 0

    while time.time() - t0 < budget_s:
        corr = compute_corr(h)
        max_idx = np.argmax(corr)
        grad = compute_gradients(h, max_idx)

        h_flat = h.flatten()
        h -= alpha * grad
        h = project_simplex(h)
        h = np.ravel(h)

        curr_c5 = evaluate(h)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
            alpha = max(alpha * 0.995, 1e-5)
            counter = 0
        else:
            counter += 1
            if counter > patience:
                alpha *= 0.5
                counter = 0

    c5_bound = best_c5
    h_values = best_h
    return h_values, c5_bound, n_points
