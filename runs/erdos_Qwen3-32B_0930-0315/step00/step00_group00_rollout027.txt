import numpy as np
import time
import random
from scipy.signal import correlate
import copy

def run(seed=42, budget_s=1100, **kwargs):
    max_time = budget_s
    n_points_start = 45  # Start with parent construction

    if 'initial_h_values' not in globals():
        # Generate an initial valid solution if parent construction is not available
        h = np.zeros(n_points_start)
        n_active = n_points_start // 2
        idx = np.random.choice(n_points_start, size=n_active, replace=False)
        h[idx] = 1.0
        np.random.shuffle(h)
        initial_h_values = h

    h_best = initial_h_values.copy()
    n_points = len(h_best)
    dx = 2.0 / n_points

    # Objective function: evaluates and returns C_5
    def evaluate(h):
        one_minus_h = 1.0 - h
        corr = correlate(h, one_minus_h, mode='full') * dx
        return np.max(corr)

    # Feasible projection: clip and normalize
    def project(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        total = np.sum(h_clipped)
        scale_factor = (n_points / 2.0) / total
        return h_clipped * scale_factor

    def mutation(h, sigma, temp):
        h_mut = h.copy()
        idxs = np.arange(n_points)
        np.random.shuffle(idxs)
        for i in range(n_points):
            if np.random.rand() < 0.05:
                h_mut[idxs[i]] += np.random.normal(0.0, sigma)
        return project(h_mut)

    def crossover(parent1, parent2):
        point = np.random.randint(1, n_points - 1)
        child = np.concatenate((parent1[:point], parent2[point:]))
        return project(child)

    def tournament_selection(population, fitness, tournament_size):
        assert tournament_size <= len(population)
        indices = np.random.choice(len(population), tournament_size, replace=False)
        best_idx = indices[0]
        for idx in indices[1:]:
            if fitness[idx] < fitness[best_idx]:
                best_idx = idx
        return population[best_idx]

    # Initialization
    population = [h_best.copy()]
    for _ in range(74):  # population size 75 total
        h = np.random.rand(n_points)  # initial
        h = project(h)
        population.append(h)
    population = np.array(population)

    start_time = time.time()
    best_c5 = evaluate(h_best)
    temp = 100.0
    cooling_rate = 0.95
    sa_interval = 50
    generations = 0

    while time.time() - start_time < max_time:
        generations += 1
        new_population = []
        for _ in range(75):  # population size
            # Selection
            parent1 = tournament_selection(population, [evaluate(p) for p in population], 8)
            parent2 = tournament_selection(population, [evaluate(p) for p in population], 8)

            # Crossover
            if np.random.rand() < 0.7:
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()

            # Mutation + Simulated Annealing
            sigma = 0.1 * (1.0 - generations / 1000.0)
            if np.random.rand() < 0.05 or generations % sa_interval == 0:
                child = mutation(child, sigma, temp)

            # Ensure feasibility
            child = project(child)

            new_population.append(child)

        # Replace worst in population
        new_population = np.array(new_population)
        population = np.concatenate((population, new_population))
        population_fitness = [evaluate(p) for p in population]
        idxs = np.argsort(population_fitness)
        population = population[idxs[:75]]
        new_best_c5 = evaluate(population[0])
        if new_best_c5 < best_c5:
            best_c5 = new_best_c5
            h_best = population[0].copy()

        if generations % sa_interval == 0:
            temp *= cooling_rate

    # Final projection and validation
    h_best = project(h_best)
    c5_bound = evaluate(h_best)
    return h_best, c5_bound, n_points
