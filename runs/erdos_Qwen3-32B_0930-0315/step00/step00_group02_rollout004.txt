import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    best_c5 = compute_c5(h)
    best_h = h.copy()
    learning_rate = 0.01
    stagnation_counter = 0

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total != n_points / 2:
            scale = (n_points / 2) / total
            h = h * scale
        h = np.clip(h, 0, 1)
        return h

    iters = 0
    target_update_time = 50  # seconds between param updates
    last_update = start_time

    while time.time() - start_time < budget_s:
        iters += 1
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1

        # Adapt learning rate
        if stagnation_counter >= 100:
            learning_rate *= 0.5
            stagnation_counter = 0

        # Find k_max with max overlap
        corr = correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        if k_max >= n_points:
            k_max = k_max - n_points

        # Compute gradient
        gradient = np.zeros(n_points)
        for j in range(n_points):
            j_forward = (j + k_max) % n_points
            j_backward = (j - k_max) % n_points
            gradient[j] = (1 - h[j_forward]) - h[j_backward]

        # Gradient descent step
        h -= learning_rate * gradient
        h = project(h)

        # Enforce some regularity
        h[0] = max(min(h[0], 0.05), 1 - 0.05)  # prevent endpoints from being too high
        h[-1] = max(min(h[-1], 0.05), 1 - 0.05)

        # Update learning schedule
        if time.time() - last_update > target_update_time:
            if iters % 100 == 0:
                learning_rate *= 0.95
            last_update = time.time()

        # Include early stopping
        if stagnation_counter > 500 or iters > 20000:
            break

    return best_h, best_c5, n_points
