def run(seed=42, budget_s=1100, patience=0.995, lr=0.1, n_points=80):
    import evaluate_erdos_solution

    np.random.seed(seed)

    best_c5 = float('inf')
    best_h = None
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    prev_update_time = time.time()

    def compute_overlap(h):
        h = h * dx
        corrs = np.correlate(h, 1 - h, mode='full')
        lags = np.arange(-n_points + 1, n_points, 1)
        valid_lags = (h.size * 2) // 3  # Use central fraction to avoid edge artifacts
        return np.max(corrs[valid_lags:-valid_lags] * dx)

    def objective_func(vec):
        h = vec
        corrs = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corrs)

    def solve_convex_for_k(h0, k, n_points=n_points):
        h = cp.Variable(n_points)
        corr = 0
        for i in range(n_points):
            j = (i + k) % n_points
            corr += h[i] * (1 - h[j])
        corr *= dx
        prob = cp.Problem(
            cp.Minimize(corr),
            [
                0 <= h,
                h <= 1,
                cp.sum(h) == n_points / 2
            ]
        )
        prob.solve(solver=cp.MOSEK, verbose=False, epsabs=SOLVER_TOL, epsrel=SOLVER_TOL)
        if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h0.copy()
        return h.value

    current_h = h.copy()
    for iter in range(MAX_ITER):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Correlate to find worst lag
        h_correlated = np.correlate(current_h, 1 - current_h, mode='full') * dx
        lags = np.arange(h_correlated.size)
        mask = (lags >= n_points - 20) & (lags <= n_points + 20)
        worst_lag = lags[np.argmax(h_correlated[mask])]

        # Solve convex problem targeting worst lag
        new_h = solve_convex_for_k(current_h, worst_lag - n_points)

        # Validate feasibility
        new_h = np.clip(new_h, 0, 1)
        total_mass = np.sum(new_h)
        scale_factor = n_points / 2 / total_mass
        new_h = new_h * scale_factor

        # Update if improvement
        current_c5 = compute_overlap(new_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = new_h.copy()

        current_h = new_h.copy()
        prev_update_time = time.time()

    # Final evaluation
    c5_bound = compute_overlap(best_h)
    h_values = best_h

    return h_values, c5_bound, n_points
