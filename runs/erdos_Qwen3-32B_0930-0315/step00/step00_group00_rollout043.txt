import numpy as np
import time
from numpy.random import rand, choice, uniform, seed
import evaluate_erdos_solution


def evaluate(h, n_points):
    # dx = 2.0 / n_points
    # Correlation equivalent using fast convolution
    corr = np.correlate(h, 1 - h, mode='full')
    return corr * (2.0 / n_points)


def invalidate(h, n_points):
    h = np.clip(h, 0, 1)
    h /= h.sum() * 2.0 / n_points  # ensure sum constraint
    return h


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)

    pop_size = 75
    n_gen = 0
    n_gen_limit = 2500
    tournament_size = max(2, int(pop_size * 0.1))
    mutation_rate = 0.05
    T = 100.0
    cool_rate = 0.95
    sa_interval = 50
    keep_top = 5

    # Generate initial population
    population = [initial_h]
    while len(population) < pop_size:
        h = uniform(size=n_points)
        h = invalidate(h, n_points)
        population.append(h)

    best_c5 = float('inf')
    best_h = None

    while time.time() - start_time < budget_s and n_gen < n_gen_limit:
        n_gen += 1
        population_c5 = []

        for h in population:
            c5 = evaluate(h, n_points).max()
            population_c5.append(c5)

        top_idx = np.argsort(population_c5)[:keep_top]
        survivors = [population[i] for i in top_idx]

        # Selection and generate new offspring
        parents = []
        for _ in range(pop_size // 2):
            choices = choice(pop_size, tournament_size, replace=False)
            parent_idx = np.argmin([population_c5[i] for i in choices])
            parents.append(population[choices[parent_idx]])

        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                p1, p2 = parents[i], parents[i+1]
                # Single-point crossover
                cx_point = np.random.randint(1, n_points)
                child = np.concatenate([p1[:cx_point], p2[cx_point:]])
                # Mutation
                mut = rand(n_points) < mutation_rate
                if n_gen % sa_interval == 0:
                    mut = mut | (rand(n_points) < 0.1)
                child += mut * uniform(-0.1, 0.1, n_points)
                child = invalidate(child, n_points)
                offspring.append(child)

        offspring_c5 = [evaluate(h, n_points).max() for h in offspring]
        survivors.extend(offspring)

        population = survivors

        # Simulated Annealing step
        if n_gen % sa_interval == 0 and n_gen > 0:
            best_idx = np.argmin(population_c5)
            sa_mut = population[best_idx] + uniform(-0.1, 0.1, n_points)
            sa_mut = invalidate(sa_mut, n_points)
            sa_c5 = evaluate(sa_mut, n_points).max()
            delta = sa_c5 - best_c5
            if delta < 0 or (T > 1e-6 and np.exp(-delta / T) > rand()):
                population.append(sa_mut)
            T *= cool_rate

        # Update best value
        for h in population:
            c5 = evaluate(h, n_points).max()
            if c5 < best_c5:
                best_c5 = c5
                best_h = h

    if best_h is None and len(population) > 0:
        best_h = population[0]
        best_c5 = evaluate(best_h, n_points).max()

    return best_h, best_c5, n_points
