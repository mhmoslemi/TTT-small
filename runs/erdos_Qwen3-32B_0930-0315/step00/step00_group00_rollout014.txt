import numpy as np
import cvxpy as cp
import time

def run(budget_s=1100):
    n_points = 45
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h /= h.sum() * dx  # Normalize if sum not exact
    xs = np.linspace(0, 2, n_points)
    inner_budget = time.time()

    def compute_c5(h, dx):
        correlation = np.correlate(h, 1 - h, mode="full")
        return max(correlation) * dx

    # Main optimization loop
    C5 = float("inf")
    best_h = h.copy()
    best_C5 = compute_c5(best_h, dx)

    # Initialize time-based progress tracking
    start_time = time.time()
    while time.time() - start_time < budget_s:
        try:
            # CVXPY constraints for equality-controlled optimization
            x = cp.Variable(n_points)
            constraints = [
                x >= 0,
                x <= 1,
                cp.sum(x) == n_points / 2
            ]
            # Iteratively fix factors contributing to C₅ and minimize local overlap
            for k in [5, 25]:  # Targeting shifts in overlap
                np_conv = np.correlate(h, 1 - h, mode="full")
                max_idx = np.argmax(np_conv)
                window = np_conv[max_idx - k:max_idx + k]
                window_idx = np.argmin(window)
                contrib = max_idx - k + window_idx
                if abs(contrib) < n_points + 1:

                    def loss_func(x_flat):
                        x_arr = x_flat.reshape(-1)
                        shifted = np.zeros_like(x_arr)
                        shifted[(n_points - 1 - contrib):] = x_arr[:-max(0, n_points - 1 - contrib)]
                        x_star = x_arr.copy()
                        x_star[(n_points - 1 - contrib):] += (x_arr[:(n_points - 1 - contrib)])

                        corr_vec = np.correlate(x_arr, 1 - x_arr, mode="full")
                        return corr_vec[contrib]  # Main term to penalize

                    result = cp.Problem(cp.Minimize(loss_func(x)), constraints)
                    result.solve(solver=cp.OSQP, max_iter=5000)
                    x_np = x.value
                    if x_np is None or not np.isfinite(x_np).all():
                        continue
                    h = x_np / h.sum() * n_points / 2  # Enforce normalization
            C5 = compute_c5(h, dx)
            if C5 < best_C5:
                best_h = h.copy()
                best_C5 = C5
        except Exception:
            # Graceful fallback to original when problems arise
            h = best_h.copy()
        if abs(C5 - best_C5) < 1e-7:
            break  # Escape plateauing

    return best_h.tolist(), best_C5, n_points
