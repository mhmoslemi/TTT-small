import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution
import cvxpy as cp


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    start = time.time()
    best_c5 = np.inf
    best_h = None
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 47
    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else np.zeros(n_points, dtype=np.float64)
    h = h / h.sum() * n_points / 2
    h[h > 1.0] = 1.0

    dx = 2.0 / n_points
    obj_eval = evaluate_erdos_solution(h, n_points)
    best_c5 = obj_eval['C5']
    best_h = h.copy()

    # Gradient parameters
    learning_rate = 0.005
    patience = 100
    wait = 0

    while (time.time() - start) < budget_s:
        # Compute overlaps
        overlaps = np.zeros(n_points)
        for k in range(1, n_points):
            valid_indices = np.arange(n_points - k)
            overlaps[k] = np.sum(h[valid_indices] * (1.0 - h[valid_indices + k])) * dx

        # Find the worst shift
        k_max = np.argmax(overlaps[1:]) + 1

        # Compute gradient for the worst overlap
        grad = np.zeros(n_points)
        for i in range(n_points - k_max):
            grad[i] += (1.0 - h[i + k_max])
            grad[i + k_max] -= h[i]

        # Update h
        h = h - learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        h = h / h.sum() * n_points / 2

        # Enforce symmetry manually (optional)
        # h[:n_points//2] = h[n_points - n_points//2:] = (h[:n_points//2] + h[n_points - n_points//2:]) / 2

        # Evaluate new objective
        obj_eval = evaluate_erdos_solution(h, n_points)
        curr_c5 = obj_eval["C5"]

        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
            wait = 0
        else:
            wait += 1

        if wait >= patience:
            learning_rate *= 0.95  # decay rate
            wait = 0

        if learning_rate < 1e-6:
            break

    return best_h, best_c5, n_points
