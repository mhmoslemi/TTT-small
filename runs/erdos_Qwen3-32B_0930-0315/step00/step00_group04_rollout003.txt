import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    # Compute cross-correlation between h and 1 - h
    corr = correlate(h, 1 - h, mode="full") * dx
    # Return the max value of the cross-correlation
    return np.max(corr)

def project(h, n_points):
    # Project h onto [0,1] and rescale to satisfy sum constraint
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if s == 0:
        h = np.zeros(n_points)
    else:
        h *= (n_points / 2) / s
    return h

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 42
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Uniform initialization satisfying the sum constraint
        h = np.ones(n_points) / n_points * (n_points / 2)

    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    learning_rate = 0.01

    iter_count = 0
    while time.time() - start_time < budget_s:
        # Compute current C₅ and identify the position with maximum overlap
        corr = correlate(h, 1 - h, mode="full") * dx
        k_max = np.argmax(corr)
        if k_max == 0:
            shift = 0
        elif k_max >= n_points:
            shift = k_max - n_points + 1
        else:
            shift = 0  # Only first n_points-1 shifts valid in this mode

        # Compute shifted 1 - h
        if shift > 0:
            shifted = np.roll(1 - h, shift)
            shifted[-shift:] = 0.0  # Fill rollover with zeros
        else:
            shifted = 1 - h

        # Compute updated h using gradient descent
        h -= learning_rate * shifted
        h = project(h, n_points)

        # Evaluate new C₅
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Adjust learning rate periodically
        if iter_count % 100 == 0:
            learning_rate *= 0.95

        iter_count += 1

    return best_h, best_c5, n_points
