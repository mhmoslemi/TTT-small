import numpy as np
import cvxpy as cp
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 80 if 'initial_h_values' in globals() else 80
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.ones(n_points, dtype=np.float64) / 2  # uniform initialization

    h = h.copy()
    best_h = h.copy()

    for i in range(100):
        # Compute cross-correlation
        ones_minus_h = 1 - h
        corrs = np.correlate(h, ones_minus_h, mode='full') * dx
        k_bad = int(np.argmax(np.abs(corrs[n_points:]))) + 1  # worst lag

        # CVXPY problem: minimize overlap at k_bad
        h_var = cp.Variable(n_points, nonneg=True)
        shift = h.copy()
        shifted = np.roll(shift, -k_bad)
        constraint_sum = cp.sum(h_var) == n_points / 2
        constraints = [0 <= h_var, h_var <= 1, constraint_sum]
        obj = cp.Minimize(cp.sum(h_var * (1 - shifted)))
        problem = cp.Problem(obj, constraints)
        problem.solve(solver=cp.ECOS, verbose=False)

        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_candidate = np.array(h_var.value)
        else:
            continue

        # Ensure feasible and project to valid range
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)

        # Evaluate the new h_candidate
        corrs = np.correlate(h_candidate, 1 - h_candidate, mode='full') * dx
        c5 = np.max(corrs)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()

        h = h_candidate
        if time.time() - start_time > budget_s - 1.0:
            break
        if best_c5 < 0.380876:
            break

    c5_bound = best_c5
    h_values = best_h
    return (h_values, c5_bound, n_points)
