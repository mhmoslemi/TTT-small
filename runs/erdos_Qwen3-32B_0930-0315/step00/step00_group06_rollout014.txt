import numpy as np
import time
from functools import partial
import cvxpy as cp

def run(seed=42, budget_s=1100):
    start_time = time.time()

    # Use parent solution if available, otherwise use random initialization
    global initial_h_values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
        h *= n_points / 2.0 / np.sum(h)
    else:
        n_points = 256
        np.random.seed(seed)
        h = np.random.choice([0.0, 1.0], size=n_points)
        h *= n_points / 2.0 / np.sum(h)

    dx = 2.0 / n_points

    def compute_c5(h):
        h_flat = h.reshape(-1)
        corr = np.correlate(h_flat, 1.0 - h_flat, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Optimization parameters
    max_steps = 1000000
    step_size = 0.05
    perturb_interval = 100
    max_no_improvement = 1000

    last_improvement = 0
    step = 0

    # Constraint function to project h to valid domain
    def project(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h_clipped)
        scale_factor = (n_points / 2.0) / h_sum
        return h_clipped * scale_factor

    def perturb_h(h):
        pert = np.random.normal(0, 0.05, size=h.shape)
        pert = project(h + pert)
        return pert

    while time.time() - start_time < budget_s and step < max_steps and last_improvement < max_no_improvement:
        step += 1

        # Compute current h's C₅
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = np.argmax(corr)

        # Estimate gradient at k_max
        h_shifted = np.roll(h, -k_max)
        h_shifted = h_shifted[-k_max:2 * n_points - k_max]

        grad = - (h_shifted - 2 * h_shifted * h)
        grad = np.concatenate([np.zeros(k_max), grad])

        # Update h along gradient
        h = project(h + step_size * grad)

        # Compute new C₅
        c5 = compute_c5(h)

        # Check if improved
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1

        # Periodic perturbation to escape local optima
        if step % perturb_interval == 0 and step > 0:
            h = project(perturb_h(h))
            last_improvement = 0

    # Do one final convex optimization with CVXPY to polish solution
    try:
        h_var = cp.Variable(n_points)
        corr_max = cp.Maximize(h_var @ (1 - np.roll(h_var, k_max)))
        constraint = [cp.sum(h_var) == n_points / 2.0, 0 <= h_var, h_var <= 1]
        prob = cp.Problem(corr_max, constraint)
        prob.solve(solver=cp.SCIPY)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_cvx = np.array(h_var.value).flatten()
            h_cvx = project(h_cvx)
            best_h = h_cvx if compute_c5(h_cvx) < best_c5 else best_h
    except:
        pass

    return best_h, best_c5, n_points
