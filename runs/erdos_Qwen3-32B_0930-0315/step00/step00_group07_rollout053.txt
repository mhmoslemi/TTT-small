import numpy as np
import time
import cvxpy as cp
from numpy.fft import fft, ifft

def evaluate_lag(h, k, dx):
    # Compute overlap for lag k
    h_k = np.roll(h, k)
    return np.sum(h * (1 - h_k)) * dx

def evaluate_C5(h, dx):
    # Use fast convolution via FFT to compute all lags efficiently
    corr = np.fft.ifft(np.fft.fft(h) * np.conj(np.fft.fft(1 - h))).real
    # Only lags in [-n, n] are relevant for [0,2] domain
    n = len(h)
    C5_values = corr[n:] * dx
    return np.max(C5_values)

def run(seed=42, budget_s=1100, **kwargs):

    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)

    MAX_ITER = 100
    for iter_idx in range(MAX_ITER):
        # Early stopping condition
        if time.time() - start_time > budget_s:
            break

        # Compute lag overlaps
        corr = np.fft.ifft(np.fft.fft(h) * np.conj(np.fft.fft(1 - h))).real
        lags = corr[n:] * dx
        k_worst = np.argmax(lags)

        # Formulate the convex problem to minimize the worst lag
        h_var = cp.Variable(n_points, nonneg=True)
        objective = cp.Minimize(h_var @ (1 - np.roll(h, k_worst)))

        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var <= 1
        ]
        problem = cp.Problem(objective, constraints)

        # Solve with ECOS to be robust to budget limit (as MOSEK may be slower)
        problem.solve(solver=cp.ECOS, verbose=False)

        if problem.status != cp.OPTIMAL:
            continue  # Skip this iteration if solver failed

        # Extract and clip the optimized h
        h_new = np.clip(h_var.value, 0, 1)

        # Check feasibility
        if not np.allclose(np.sum(h_new), n_points / 2, atol=1e-8):
            continue  # Skip this update

        C5_new = evaluate_C5(h_new, dx)

            best_C5 = C5_new
            best_h = h_new.copy()
            print(f"Iter {iter_idx}: New best C5 = {best_C5:.7f}")

    return best_h, best_C5, n_points
