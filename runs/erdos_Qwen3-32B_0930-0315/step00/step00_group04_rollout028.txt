import numpy as np
import time
from scipy.ndimage import map_coordinates
from functools import partial

np.random.seed(42)

def evaluate(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def repair_feasibility(h, dx):
    h = np.clip(h, 0, 1)
    total = np.sum(h) * dx
    scale = 1.0 / total
    return h * scale

def next_best_dict(n_points, budget_s):
    return {'n': n_points, 'max_steps': int(100 * n_points), 'pop_size': 20, 'elite_frac': 0.25}

def evolve_step(pop, dx, survive_rate=0.5, mutate_std=0.1, interpolate=False):
    n = len(pop[0])
    fitness = np.array([evaluate(ind, dx) for ind in pop])
    surviving = fitness.argsort()[:int(len(pop) * survive_rate)]
    survivors = [pop[i] for i in surviving]
    next_gen = survivors.copy()
    
    while len(next_gen) < len(pop):
        a, b = pop[np.random.choice(len(pop), 2)]
        cut = np.random.randint(1, n)
        child = np.concatenate([a[:cut], b[cut:]])
        if np.random.rand() < 0.2:
            noise = np.random.normal(0, mutate_std, n)
        else:
            noise = np.zeros(n)
        child = repair_feasibility(child + noise, dx)
        next_gen.append(child)
    
    if interpolate and len(next_gen) < len(pop):
        for i in range(0, min(4, len(pop) - len(next_gen))):
            a, b = pop[np.random.choice(len(pop), 2)]
            # Use linear interpolation
            alpha = np.random.rand()
            lin_comb = alpha * a + (1 - alpha) * b
            next_gen.append(repair_feasibility(lin_comb, dx))
    
    return next_gen

def converging_gd_step(h, dx, lr=0.01, steps=10):
    improved = h.copy()
    for _ in range(steps):
        H = 1 - np.roll(h, shift=0)
        for k in range(2 * len(h) - 1):
            shifted = np.roll(h, shift=(k - len(h) + 1))
            dK = dx * (1 - shifted)
            if k == np.argmax(np.correlate(h, 1 - h, 'full')):
                improved -= lr * dK
                break
        improved = repair_feasibility(improved, dx)
    return improved

def evolve_and_descend(h0, dx, budget_s, n_points, seed=None):
    if seed is not None:
        np.random.seed(seed)

    start_time = time.time()
    best_h = repair_feasibility(h0, dx)
    best_c5 = evaluate(best_h, dx)
    print(f"[init] C₅ = {best_c5:.6f}")

    ev_budget = int(budget_s * 0.7)
    gd_budget = budget_s - ev_budget

    pop_size = 20
    steps_ev = 100
    pop = [repair_feasibility(np.random.uniform(0.25, 0.75, n_points), dx) for _ in range(pop_size)]
    
    for i in range(steps_ev):
        pop = evolve_step(pop, dx, survive_rate=0.4, mutate_std=0.1)
        best_candidate = repair_feasibility(pop[0], dx)
        c5_candidate = evaluate(best_candidate, dx)
        if c5_candidate < best_c5:
            best_h = best_candidate
            best_c5 = c5_candidate
            print(f"[ev] Step {i}, C₅ = {best_c5:.6f}")
        if time.time() - start_time > ev_budget:
            break

    # Refine with gradient descent
    steps_remaining = int(gd_budget // 0.01)
    for _ in range(steps_remaining):
        best_h = converging_gd_step(best_h, dx)
        best_c5 = evaluate(best_h, dx)
        print(f"[gd] C₅ = {best_c5:.6f}")
        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5

def run(budget_s=1100, initial_n=14, modify=True):
    if 'initial_h_values' in globals():
        base_h = initial_h_values
        base_n = len(base_h)
        if base_n != initial_n:
            print(f"Resizing initial_h_values from {base_n} to {initial_n}")
            base_h = np.interp(np.linspace(0, 2, initial_n), np.linspace(0, 2, base_n), base_h)
    else:
        base_h = np.random.rand(initial_n)
        base_h = repair_feasibility(base_h, 2.0 / initial_n)
    
    h, c5 = evolve_and_descend(base_h, 2.0 / initial_n, budget_s, initial_n)
    return h, c5, initial_n
