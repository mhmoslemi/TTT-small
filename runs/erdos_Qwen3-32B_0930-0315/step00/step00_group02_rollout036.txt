import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    n_points = len(initial_h_values)
    h = initial_h_values.copy()

    h = np.clip(h / h.sum() * n_points / 2, 0, 1)

    best_h = h.copy()
    best_c5 = initial_c5
    last_improvement = time.time()
    learning_rate = 0.01
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Compute cross-correlation
        ones_h = np.ones_like(h)
        corr = correlate(h, 1 - h, mode="full") / n_points
        c5 = np.max(corr)
        k_max = int(np.argmax(corr))
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = time.time()

        # Break if no improvement for long time
        if time.time() - last_improvement > 90:
            learning_rate *= 0.9  # decay the learning rate
            if learning_rate < 1e-5:
                break

        # Compute gradient based on the most problematic shift (k_max)
        grad = np.zeros_like(h, dtype=np.float64)
        for i in range(n_points):
            k_orbit = (i + k_max) % n_points
            grad[i] = (1 - h[k_orbit]) + (1 - h[(i - k_max) % n_points])
        grad /= 2

        # Gradient update
        h -= learning_rate * grad

        # Project to the feasible region [0, 1]^n with mean 0.5
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum < 1e-9:
            h[0] = 1e-9  # avoid zero division
            h_sum = h.sum()
        scale_factor = (n_points / 2) / h_sum
        h = np.clip(h * scale_factor, 0, 1)

        # Safeguard: ensure the integral constraint is satisfied up to machine precision
        h = h / h.sum() * (n_points / 2)
        h = np.clip(h, 0, 1)

    return best_h, best_c5, n_points
