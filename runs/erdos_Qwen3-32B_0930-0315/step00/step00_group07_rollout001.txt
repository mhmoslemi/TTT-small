import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 80
    dx = 2.0 / n_points
    h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    h = h.astype(float)
    h /= h.sum() * dx
    h *= n_points / 2
    h = np.clip(h, 0, 1)

    def compute_c5(h):
        c = correlate(h, 1 - h, mode='full')
        return np.max(c * dx)

    t0 = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    while time.time() - t0 < budget_s:
        # Evaluate current C5
        c5 = compute_c5(h)

        # If we find a better c5, update the best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            print(f"New best C5: {best_c5:.7f}")

        # Identify the worst lag k where overlap is max
        c = correlate(h, 1 - h, mode='full')
        k_bad = np.argmax(c)

        if k_bad < n_points:
            # Left-sided overlap
            start = max(0, k_bad - n_points + 1)
            end = k_bad
        else:
            # Right-sided overlap
            start = k_bad - n_points + 1
            end = n_points

        # Objective: minimize dz = integral h(x)(1 - h(x + k_bad)) dx using h optimization
        h_var = cp.Variable(n_points)
        overlap = 0
        for i in range(start, end):
            overlap += h_var[i] * (1 - h_var[i + (k_bad - start)]) * dx

        constraint = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
        prob = cp.Problem(cp.Minimize(overlap), constraint)
        prob.solve('ECOS', max_itrs=100, feastol=1e-4)

        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h = h_var.value
            h = np.clip(h, 0, 1)

        if time.time() - t0 >= budget_s:
            break

    return best_h.tolist(), best_c5, n_points
