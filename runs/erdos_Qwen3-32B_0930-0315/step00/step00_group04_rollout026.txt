import numpy as np
import time
import cvxpy as cp
import math

def evaluate_overlap(h, dx):
    full_corr = np.correlate(h, 1 - h, mode='full')
    return np.max(full_corr * dx)

def repair_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h = h / h.sum() * (n_points / 2)
    return h

def optimize_erdos(n_points, initial_h, max_steps, initial_T, cooling_rate, lr):
    dx = 2.0 / n_points
    h = initial_h.copy()
    h = repair_feasibility(h, n_points)
    
    best_h = h.copy()
    best_C = evaluate_overlap(h, dx)
    
    t_start = time.time()
    
    for step in range(max_steps):
        T = initial_T * (cooling_rate ** step)
        
        # Perturb h
        perturbation = np.random.normal(0, 0.1, size=n_points)
        h_candidate = h + perturbation
        h_candidate = repair_feasibility(h_candidate, n_points)
        
        C_candidate = evaluate_overlap(h_candidate, dx)
        
        if C_candidate < best_C or np.random.rand() < np.exp((best_C - C_candidate) / max(T, 1e-6)):
            h = h_candidate
            if C_candidate < best_C:
                best_h = h_candidate
                best_C = C_candidate
        
        # Gradient step
        full_corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(full_corr)
        shift = k_max - (n_points - 1)
        h_shifted = np.roll(h, shift)
        gradient = (1 - h_shifted)
        
        h = h - lr * gradient[:n_points]
        h = repair_feasibility(h, n_points)
        
        if time.time() - t_start >= 1100:
            break
    
    return best_h, best_C

def cubic_interpolation(old_array, new_length):
    interpolator = np.interp(np.linspace(0, len(old_array) - 1, new_length), 
                             np.arange(len(old_array)), old_array)
    return interpolator

def downsample(old_array, new_length):
    return np.interp(np.linspace(0, len(old_array) - 1, new_length), 
                     np.arange(len(old_array)), old_array)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    if 'initial_h_values' in globals():
        h_start = initial_h_values
        n_start = len(h_start)
        sample_times = [n_start, 2*n_start, 4*n_start]
    else:
        h_start = np.random.rand(14)
        h_start = h_start / h_start.sum() * (14 / 2)
        sample_times = [14, 28, 56]
    
    h_best = h_start.copy()
    C5_best = evaluate_overlap(h_best, 2.0 / len(h_best))
    
    t_start = time.time()
    for i, n_points in enumerate(sample_times):
        if time.time() - t_start >= budget_s:
            break
            
        fraction = (i+1) / len(sample_times)
        remaining_time = budget_s - (time.time() - t_start)
        steps = int(remaining_time * 0.3)
        steps = max(steps, 100)
        
        if i > 0:
            h = cubic_interpolation(h_best, n_points)
            h = repair_feasibility(h, n_points)
        else:
            h = h_best
        
        h, C5 = optimize_erdos(n_points, h, steps, 1.0, 0.995, 0.01)
        
        if C5 < C5_best:
            h_best = h
            C5_best = C5
    
    return h_best, C5_best, len(h_best)
