import time
import numpy as np
import cvxpy as cp
from numpy import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70 if 'initial_h_values' in globals() else 100
    n_points = max(2, min(1000, n_points))
    dx = 2.0 / n_points

    # Initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = 0.5 * np.ones(n_points, dtype=np.float64)

    # Feasibility check and projection
    def project(h):
        x = cp.Variable(n_points)
        obj = cp.Minimize(cp.sum_squares(x - h))
        constraints = [0 <= x, x <= 1, cp.sum(x) == n_points / 2.0]
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        return x.value

    h = project(h)

    # Gradient of objective for max-k
    def compute_C5_and_gradient(h_vec):
        h_flat = h_vec.flatten()
        h1 = h_flat
        h2 = 1 - h_flat
        corrs = correlate(h1, h2, mode='full') * dx
        kmax = np.argmax(np.abs(corrs))
        C5_val = corrs[kmax]
        n = len(h1)
        G = np.zeros(n)
        idxs = np.arange(n)
        for i in range(n):
            shift = kmax
            if i + shift < n and i + shift >= 0:
                G[i] += 1 - h1[(i + shift) % n]
            if i - shift >= 0:
                G[i] -= h1[i - shift]
        grad = G / max(1e-8, np.linalg.norm(G))
        return C5_val, grad

    best_C5 = float('inf')
    best_h = h.copy()
    h_history = []
    iter_history = []

    # Optimization loop
    t0 = time.time()
    learn_rate = 0.01
    max_iter = 5000
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        C5_val, grad = compute_C5_and_gradient(h)
        if C5_val < best_C5:
            best_C5 = C5_val
            best_h = h.copy()
        # Learning rate control
        eta = learn_rate / (1 + iter / 1000.0)
        h_candidate = h - eta * grad
        h_candidate = project(h_candidate)
        h = h_candidate
        # Check for stagnation
        if iter % 50 == 0:
            h_history.append(h.copy())
            iter_history.append(best_C5)

    # Final cleanup to ensure small improvement is recorded
    final_C5, _ = compute_C5_and_gradient(best_h)

    return best_h, final_C5, n_points
