import numpy as np
import scipy.signal
import time
import cvxpy as cp
from collections import deque

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use the initial_h_values if provided, otherwise use a flat distribution
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n_points = len(h)
    else:
        # Default construction: uniform distribution
        n_points = 45
        h = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    assert np.isclose(np.sum(h) * dx, 1), "Initial h does not satisfy integral constraint"

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    search_budget = 1000  # grid candidates per iteration
    max_iters = 10000
    tolerance = 1e-4
    prev_c5 = np.inf
    no_improvement = 0
    k_max_idx = 0

    def compute_c5(h, dx):
        h_flat = h.flatten()
        one_h = 1.0 - h_flat
        corr = scipy.signal.correlate(h_flat, one_h, mode="full", method="direct")
        c5 = np.max(corr) * dx
        return c5

    def clip_and_restore_sum_even(h, total_sum):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = h_clipped.sum()
        if np.isclose(sum_clipped, 0):
            # fallback to uniform
            h_clipped[:] = 0.5 * total_sum / n_points
        else:
            scale = total_sum / sum_clipped
            h_clipped *= scale
        return h_clipped

    def perturb_candidate(h_, dx):
        h_ = h_.copy()
        R = scipy.signal.correlate(h_, 1 - h_, mode='same', method='direct')
        # non-linear redundancy measure
        R_weights = R / (1 + R + 1e-10)
        regions = np.argsort(R_weights)[::-1]  # sort by influence
        top_n = max(1, int(0.15 * n_points))  # 15% high influence region
        change_mask = np.zeros(n_points, dtype=bool)
        change_mask[regions[:top_n]] = True
        high_step = dx / 20
        low_step = dx / 7
        for i in np.where(change_mask)[0]:
            step = high_step if R[i] > 0.3 * np.max(R) else low_step
            candidates = [h[i],
                          np.clip(h[i] + step, 0, 1),
                          np.clip(h[i] - step, 0, 1)]
            trial_h = h_.copy()
            trial_h[i] = candidates[np.argmin([compute_c5(h_.copy(), dx) for _ in candidates])]
            candidate = clip_and_restore_sum_even(trial_h, n_points / 2)
            if compute_c5(candidate, dx) < compute_c5(h_, dx):
                h_ = candidate.copy()
        return h_

    for it in range(max_iters):
        if time.time() - start_time > budget_s:
            break

        improved = False
        for _ in range(search_budget):
            trial_h = h.copy()
            trial_h = perturb_candidate(trial_h, dx)
            trial_c5 = compute_c5(trial_h, dx)
            if trial_c5 < best_c5:
                best_h[:] = trial_h[:]
                best_c5 = trial_c5
                improved = True
                no_improvement = 0
                break
            if time.time() - start_time > budget_s:
                break

        if not improved:
            no_improvement += 1
            if no_improvement > 15:
                break

        if np.abs(prev_c5 - best_c5) < tolerance:
            no_improvement += 1
        else:
            prev_c5 = best_c5

    # Run one final convex refinement
    try:
        x = cp.Variable(n_points)
        ones_x = 1 - x
        corr = scipy.signal.correlate(h, ones_x.value, mode='full') if hasattr(ones_x, 'value') else np.zeros(2*n_points -1)
        constraints = [0 <= x, x <= 1, cp.sum(x) == n_points / 2]
        obj = cp.Minimize(cp.max(scipy.signal.correlate(x, ones_x, mode='full')) * dx)
        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver='GUROBI', verbose=False)
        if result is not None and not np.isinf(result):
            candidate = np.clip(x.value, 0, 1)
            final_candidate = clip_and_restore_sum_even(candidate, n_points / 2)
            final_c5 = compute_c5(final_candidate, dx)
            if final_c5 < best_c5:
                best_h[:] = final_candidate[:]
                best_c5 = final_c5
    except Exception:
        pass

    return (best_h.tolist(), best_c5, n_points)
