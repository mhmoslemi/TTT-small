import numpy as np
import math
import time
import cvxpy as cp
import scipy.optimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    try:
        n_points = 160
        h = np.array(initial_h_values)
        h = h.astype(float) / h.sum() * n_points / 2
    except:
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = h / h.sum() * (n_points / 2)
    h = np.clip(h, 0.0, 1.0)

    dx = 2.0 / n_points
    start_time = time.time()
    best_h = h.copy()
    best_c5 = float('inf')
    learning_rate = 0.01
    patience = 100
    no_improvement_counter = 0

    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5, corr

    best_c5, _ = compute_c5(h)

    while time.time() - start_time < budget_s:
        h = np.clip(h, 0.0, 1.0)
        h = h / h.sum() * (n_points / 2)

        c5, corr = compute_c5(h)
        k_max = np.argmax(corr)

        grad = np.zeros_like(h)
        for i in range(n_points):
            if (i + k_max) < n_points:
                grad[i] += (1.0 - h[(i + k_max) % n_points])
            else:
                grad[i] += (1.0 - h[(i + k_max - n_points) % n_points])
            if (i - k_max) >= 0:
                grad[i] -= h[(i - k_max) % n_points]
            else:
                grad[i] -= h[(i - k_max + n_points) % n_points]

        h = h - learning_rate * grad

        h = np.clip(h, 0.0, 1.0)
        h = h / h.sum() * (n_points / 2)

        new_c5, _ = compute_c5(h)

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            no_improvement_counter = 0
            learning_rate = min(learning_rate * 1.05, 0.05)
        else:
            no_improvement_counter += 1
            if no_improvement_counter >= patience:
                learning_rate = max(learning_rate * 0.95, 1e-4)

        if np.random.rand() < 0.005:
            perturbation = np.random.normal(0, 0.01, size=h.shape)
            h = h + perturbation
            h = np.clip(h, 0.0, 1.0)
            h = h / h.sum() * (n_points / 2)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h / best_h.sum() * (n_points / 2)
    best_c5_final, _ = compute_c5(best_h)
    return (best_h.tolist(), best_c5_final, n_points)
