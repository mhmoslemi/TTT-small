import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 47
    dx = 2.0 / n_points
    h = initial_h_values.copy()

    def project_h(h):
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / np.sum(h)
        return h

    h = project_h(h)

    def compute_c5(h):
        h1 = 1 - h
        corr = np.correlate(h, h1, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)

    def gradient_step(h, k_max):
        grad = np.zeros_like(h)
        for i in range(n_points - k_max):
            grad[i] -= 1 - h1[i + k_max]
            grad[i + k_max] += h[i]
        grad = grad / np.linalg.norm(grad + 1e-6)
        return grad

    h1 = 1 - h
    corr = np.correlate(h, h1, mode='full')
    k_max = np.argmax(corr)
    learning_rate = 0.02
    decay_rate = 0.999

    while time.time() - start_time < budget_s and abs(k_max - (n_points - 1)) > 2:
        grad = np.zeros_like(h)
        i_max = None
        max_corr = -1e9
        # Use one key k instead of scanning, since scanning takes O(n^2) time
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        hk = h[n_points - k_max:] if k_max < n_points else h[:n_points]
        for i in range(len(hk)):
            grad[i] -= (1 - h)[i + k_max] if i + k_max < n_points else (1 - h)[i + k_max - n_points]
            if k_max < n_points:
                grad[i + k_max] += h[i]
            else:
                grad[i + k_max - n_points] += h[i]
        grad = (grad / np.linalg.norm(grad + 1e-6))
        # Gradient update with line search
        for lr in [1.0, 0.5, 0.2, 0.05, 0.02]:
            candidate = h - lr * grad
            candidate = project_h(candidate)
            c5 = compute_c5(candidate)
            if c5 < best_c5:
                learning_rate = lr
                h[:] = candidate
                best_c5 = c5
                k_max = np.argmax(np.correlate(h, 1 - h, mode='full'))
                break
        learning_rate *= decay_rate

        if compute_c5(h) < best_c5:
            best_c5 = compute_c5(h)
            best_h = h.copy()

    h = project_h(h)
    return (h, best_c5, n_points)
