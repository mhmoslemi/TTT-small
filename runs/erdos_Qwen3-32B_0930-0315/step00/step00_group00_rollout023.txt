import time
import numpy as np
from numpy.lib import corrgram
import cvxpy as cp
import evaluate_erdos_solution


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use initial_h_values if available, else default to n=45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
        n_points = len(h)
    else:
        n_points = 45
        h = np.random.randint(0, 2, n_points).astype(float)
        h *= 2  # Binary-like but scaled to match [0,2] range before final clipping
        h = np.clip(h / 2, 0, 1)  # Final step to [0,1]
        # Normalization to meet sum constraint
        area = 2.0 / n_points
        h *= n_points / 2 / h.sum()
        h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = evaluate(h)
    best_n_points = n_points
    tabu_memory = set()
    tabu_size = 20
    max_neighbors = 20
    iteration = 0

    def evaluate(h_eval):
        corr = np.correlate(h_eval, 1 - h_eval, mode="full")
        corr *= 2.0 / n_points
        return np.max(corr)

    def h_to_tuple(h_t):
        return tuple(h_t.round(6).astype(float))

    while time.time() - start_time < budget_s:
        neighbor_list = []
        indices = np.random.choice(n_points, size=max_neighbors, replace=False)

        for i in indices:
            for perturbation in [0.0, -0.1, 0.1, -0.2, 0.2]:
                neighbor = h.copy()
                neighbor[i] += perturbation
                neighbor[neighbor < 0] = 0
                neighbor[neighbor > 1] = 1
                # Projection to sum(h) = n_points / 2
                area = 2.0 / n_points
                neighbor_sum = neighbor.sum()
                if np.abs(neighbor_sum - n_points / 2) > 1e-3:
                    scale = (n_points / 2) / neighbor_sum
                    neighbor *= scale
                neighbor = np.clip(neighbor, 0, 1)
                neighbor_tuple = h_to_tuple(neighbor)
                if neighbor_tuple not in tabu_memory:
                    neighbor_list.append(neighbor)

        # Add a couple of deep random perturbations per iteration
        if np.random.rand() < 0.05:
            neighbor = h.copy()
            idxs = np.random.choice(n_points, 5, replace=False)
            neighbor[idxs] = np.random.rand(5)
            neighbor = np.clip(neighbor, 0, 1)
            area = 2.0 / n_points
            neighbor_sum = neighbor.sum()
            if np.abs(neighbor_sum - n_points / 2) > 1e-3:
                scale = (n_points / 2) / neighbor_sum
                neighbor *= scale
            neighbor_tuple = h_to_tuple(neighbor)
            if neighbor_tuple not in tabu_memory:
                neighbor_list.append(neighbor)

        current_neighbor_evaluations = []
        for neighbor in neighbor_list:
            c5 = evaluate(neighbor)
            current_neighbor_evaluations.append((c5, neighbor))

        if current_neighbor_evaluations:
            # Try to find any improvement
            current_neighbor_evaluations.sort()
            c5, neighbor = current_neighbor_evaluations[0]
            if c5 < best_c5:
                h = neighbor.copy()
                best_c5 = c5
                best_h = h.copy()
                tabu_memory.add(h_to_tuple(h))
                if len(tabu_memory) > tabu_size:
                    tabu_memory.pop()

        iteration += 1

    return (best_h.tolist(), best_c5, best_n_points)
