import scipy.signal
import time

        h[n_points//2:] = 0.0
    h = h * (n_points / 2) / np.sum(h)  # Ensure correct normalization

    dx = 2.0 / n_points
    def objective(h):
        h_tiled = np.tile(h, 2)
        overlap = scipy.signal.correlate(h_tiled, 1 - h_tiled, mode='same')[:n_points] * dx
        return np.max(overlap)

    best_h = h.copy()
    best_score = objective(h)

    # Optimization parameters
    lr = 0.1  # learning rate
    n_iterations = max(1000, int(budget_s // (n_points * 0.1)))
    
    start_time = time.time()

    for iteration in range(n_iterations):
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        for i in range(n_points):
            e = np.zeros_like(h)
            e[i] = 1e-5
            perturbed_h = h + e
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)

            perturbed_score = objective(perturbed_h)
            grad[i] = (perturbed_score - best_score) / 1e-5

        # Gradient descent step
        h = h - lr * grad

        # Projection onto feasible region: clamp to [0, 1]
        h = np.clip(h, 0, 1)

        # Ensure integral constraint
        h = h * (n_points / 2) / np.sum(h + 1e-8)

        score = objective(h)

        # Update best solution if we find a better one
        if score < best_score:
            best_h = h.copy()
            best_score = score

        # Track progress and time
        if iteration % 100 == 0:
            print(f"Iteration {iteration}, C5 = {best_score:.8f}")

        # Check time budget
        if time.time() - start_time > budget_s:
            break

    return (best_h.tolist(), best_score, n_points)
