import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp
from copy import deepcopy

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    h_vals = np.clip(h, 0, 1)
    if not np.isclose(np.sum(h_vals), n_points / 2, atol=1e-6):
        s = (n_points / 2) / np.sum(h_vals)
        h_vals = np.clip(h_vals * s, 0, 1)
    inv_h = 1 - h_vals
    corr = correlate(h_vals, inv_h, mode='full')
    c5 = np.max(corr) * dx
    return {'C5': c5, 'h': h_vals}

def run(seed=42, budget_s=1100, population_size=50, mutation_rate=0.15, tournament_size=5):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        best_h = np.copy(initial_h_values)
    else:
        n_points = 70
        best_h = np.random.uniform(0.0, 1.0, n_points)
        # Normalize to satisfy the integral constraint
        s = (n_points / 2.0) / np.sum(best_h)
        best_h = np.clip(best_h * s, 0, 1)

    best_result = evaluate_erdos_solution(best_h, n_points)
    best_c5 = best_result['C5']
    best_h_values = best_result['h'].copy()

    pop = [np.copy(best_h) for _ in range(population_size // 2)]
    for i in range(population_size // 2):
        perturbation = np.random.normal(0, 0.1, n_points)
        new_h = np.clip(np.copy(best_h) + perturbation, 0, 1)
        s = (n_points / 2.0) / np.sum(new_h)
        new_h = np.clip(new_h * s, 0, 1)
        pop.append(new_h)

    # Main GA loop
    while time.time() - start_time < budget_s:
        offspring = []
        for _ in range(population_size // 2):
            parent1, parent2 = tournament_selection(pop, tournament_size)
            child1, child2 = crossover(parent1, parent2, n_points)
            child1 = mutate(child1, mutation_rate, n_points)
            child2 = mutate(child2, mutation_rate, n_points)
            offspring.extend([child1, child2])

        new_pop = pop + offspring
        new_pop = [evaluate_erdos_solution(h, n_points) for h in new_pop]
        new_pop = sorted(new_pop, key=lambda x: x['C5'])[:population_size]

        pop = [entry['h'] for entry in new_pop]
        current_best = new_pop[0]
        if current_best['C5'] < best_c5:
            best_c5 = current_best['C5']
            best_h_values = current_best['h'].copy()

        # Early stopping if time is tight
        if time.time() - start_time + 10 > budget_s:
            break

    return (best_h_values, best_c5, n_points)

def tournament_selection(population, tournament_size):
    indices = np.random.choice(len(population), tournament_size, replace=False)
    if len(population) == 0 or not any(hasattr(x, '__len__') for x in population):
        return [], []
    tournament = [population[i] for i in indices]
    return min(tournament, key=lambda x: x['C5']), min(tournament, key=lambda x: x['C5'])

def crossover(parent1, parent2, n_points):
    point = np.random.randint(1, n_points - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])

    def repair(h, n_points):
        s = (n_points / 2) / np.sum(h)
        s = np.clip(s, 0.8, 1.2)
        h_rep = np.clip(h * s, 0, 1)
        if not np.isclose(np.sum(h_rep), n_points / 2, atol=1e-1):
            surplus = np.sum(h_rep) - n_points / 2
            sorted_indices = np.argsort(h_rep)
            if surplus > 0:
                h_rep[sorted_indices[:int(surplus * 10)]] = 0
            else:
                h_rep[sorted_indices[-int(-surplus * 10):]] = 1
        return h_rep

    child1 = repair(child1, n_points)
    child2 = repair(child2, n_points)
    return child1, child2

def mutate(individual, mutation_rate, n_points):
    if np.random.rand() < mutation_rate:
        positions = np.random.choice(n_points, size=int(n_points * 0.05), replace=False)
        noise = np.random.normal(0, 0.1, size=positions.size)
        individual[positions] += noise
        individual = np.clip(individual, 0, 1)
        surplus = np.sum(individual) - n_points / 2
        if abs(surplus) > 1e-5:
            med_idx = np.argsort(individual)
            if surplus > 0:
                individual[med_idx[:int(surplus * 10)]] = 0
            else:
                individual[med_idx[-int(-surplus * 10):]] = 1
    return individual
