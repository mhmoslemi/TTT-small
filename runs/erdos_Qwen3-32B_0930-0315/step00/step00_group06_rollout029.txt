import numpy as np
import cvxpy as cp
import time
from math import sqrt
from evaluate_erdos_solution import evaluate_erdos_solution, initial_h_values

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Configuration
    max_size = 256  # N-point discretization, powers of 2 yield nice spectrum properties
    n_points = 256
    dx = 2.0 / n_points

    # Initialize from parent solution if provided
    if 'initial_h_values' in globals():
        h = initial_h_values
        h = h[:n_points]
    else:
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2
        h = np.clip(h, 0, 1)

    h = h.astype(np.float64)
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()

    # Optimization loop
    step = 0
    max_steps = 100000
    while time.time() - start_time < budget_s and step < max_steps:
        if step % 100 == 0:
            print(f"Step {step}, best_c5={best_c5:.6f}")

        # Computeoverlap integrals
        corr = np.correlate(h, 1 - h, mode='full')
        corr = corr * dx

        # Select worst lag
        k_max = np.argmax(corr)
        c5_candidate = corr[k_max]

        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h.copy()
        corr_k = corr[k_max]

        # Magnitude adjustment: convex optimization at fixed k_max
        x = cp.Variable(n_points)
        obj = cp.Minimize(corr_k)
        # Extend h for convolution aliasing
        h_flip = np.roll(h, k_max - n_points)
        lamda = cp.Parameter(nonneg=True)
        penalty = cp.norm(2 * x - h - h_flip, 2)**2
        obj = cp.Minimize(lamda * penalty)

        constraints = [0 <= x, x <= 1, x.sum() == n_points / 2.0]
        problem = cp.Problem(obj, constraints)
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_proposed = x.value
                if h_proposed is not None:
                    h = np.clip(h_proposed, 0, 1)
                    h = h / h.sum() * n_points / 2.0
        except Exception as e:
            pass

        # Position perturbation
        if step % 100 == 0:
            perturb = np.random.randn(n_points) * 0.05
            h = np.clip(h + perturb, 0, 1)
            h = h / h.sum() * n_points / 2.0

        step += 1

    if time.time() - start_time >= budget_s:
        print("Budget exhausted at step", step)

    c5_bound = evaluate_erdos_solution(best_h)
    print(f"Final best C5 bound: {c5_bound}")
    return best_h, c5_bound, n_points
