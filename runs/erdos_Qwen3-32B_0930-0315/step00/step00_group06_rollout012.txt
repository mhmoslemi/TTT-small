import numpy as np
import scipy.signal
import time
import cvxpy as cp


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.perf_counter()
    np.random.seed(seed)

    # Use 256 points for a better resolution
    n_points = 256
    dx = 2.0 / n_points

    # Initialize h using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values[:n_points], dtype=np.float64)
    else:
        # Default initialization: uniform distribution
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 0.0
        h[mid:] = 1.0
    h = h * (n_points / 2) / h.sum()  # Normalize

    best_h = h.copy()
    best_c5 = float('inf')

    # Optimization loop
    while time.perf_counter() - start_time < budget_s:
        # Compute correlation
        ones_minus_h = 1 - h
        corr = scipy.signal.correlate(h, ones_minus_h, mode='full')
        # Compute the indices that correspond to the actual lags [0, 2.0]
        valid_corr_indices = np.arange(n_points)
        c5 = (2.0 / n_points) * np.max(corr[valid_corr_indices])

        # Keep track of the best solution found
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

        # Select the lag with the highest overlap
        k_max = np.argmin(corr[valid_corr_indices])  # Minimize negative overlap
        if k_max < 0 or k_max >= n_points:
            continue

        # Objective function value to minimize
        obj_value = h @ (1 - np.roll(h, k_max)) * dx

        # Set up a convex optimization problem to update h while satisfying constraints
        hp = cp.Variable(n_points)
        constraints = [
            hp <= 1,
            hp >= 0,
            cp.sum(hp) == n_points / 2  # ∫₀² h(x) dx = 1
        ]
        obj = cp.Minimize(hp @ (1 - np.roll(hp, k_max)) * dx)
        prob = cp.Problem(obj, constraints)

        # Use a convex solver that's likely to be available
        try:
            prob.solve(solver=cp.CVXOPT, max_iters=100)
            if prob.status == "optimal":
                h = hp.value
            else:
                pass  # Skip if not optimal
        except Exception:
            pass

        # Check for early stopping if no improvement
        if time.perf_counter() - start_time > budget_s:
            break

        # Perturb h slightly to avoid getting stuck in local minima
        if np.random.rand() < 0.05:
            h += np.random.normal(scale=0.01, size=n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / h.sum()

    # Validate final h satisfies constraints
    h = best_h
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / h.sum()

    # Re-calculate final C5
    ones_minus_h = 1 - h
    corr = scipy.signal.correlate(h, ones_minus_h, mode='full')
    valid_corr_indices = np.arange(n_points)
    final_c5 = (2.0 / n_points) * np.max(corr[valid_corr_indices])

    return (h, final_c5, n_points)
