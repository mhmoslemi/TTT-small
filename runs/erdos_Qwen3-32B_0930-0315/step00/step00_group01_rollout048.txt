import time
import numpy as np
import scipy.optimize

def run(seed=42, budget_s=1100, **kwargs):
    # Check if initial_h_values is available
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        n_points = 94
        h = np.zeros(n_points, dtype=np.float64)
        h[:n_points // 2] = 1.0  # initial guess with 1s in first half

    max_iter = int(1e6)  # Very large to allow for iterative adjustments
    dx = 2.0 / n_points
    target_sum = int(n_points / 2)

    def objective(h_vec):
        h_vec = np.clip(h_vec, 0.0, 1.0)
        h_vec_sum = h_vec.sum()
        if abs(h_vec_sum - target_sum) > 1e-6:
            alpha = target_sum / h_vec_sum
            h_vec = np.clip(h_vec * alpha, 0.0, 1.0)

        # Compute all overlaps
        corr = np.zeros(n_points * 2 - 1)
        for k in range(-n_points + 1, n_points):
            indices = np.arange(min(n_points + k, n_points))
            i1 = indices
            i2 = indices - k
            mask = (i2 >= 0) & (i2 < n_points)
            i1, i2 = i1[mask], i2[mask]
            corr_result = ((h_vec[i1] * (1.0 - h_vec[i2])).sum() * dx)
            corr[k + n_points - 1] = corr_result
        return np.max(corr)

    # Define the gradient-based optimization function
    t0 = time.time()
    best_c5 = float('inf')
    best_h = h.copy()

    # Define projected gradient step update
    def project(h_vec):
        h_vec = np.clip(h_vec, 0.0, 1.0)
        h_sum = h_vec.sum()
        if h_sum == 0 or h_sum == n_points:
            return h_vec * 0.5
        return h_vec * (target_sum / h_sum)

    def gradient_descent_step(h_vec, learning_rate):
        # Analytical gradient estimate
        h_vec = np.clip(h_vec, 0.0, 1.0)
        h_sum = h_vec.sum()
        h_vec = h_vec * (target_sum / h_sum)

        grad = np.zeros_like(h_vec)
        for i in range(n_points):
            contribs = []
            for k in range(n_points):
                lhs_index = i - k
                if 0 <= lhs_index < n_points:
                    contribs.append((1.0 - h_vec[k]) * h_vec[lhs_index])
            contribs = np.array(contribs)
            max_idx = np.argmax(contribs)
            max_k = np.arange(n_points)[max_idx]
            grad[i] += contribs[max_idx]
        grad = -2 * grad * (h_sum / target_sum)
        grad = grad * (2.0 / n_points)
        return h_vec - learning_rate * grad

    # Run optimization
    learning_rates = [0.5, 0.3, 0.1, 0.05, 0.02]
    learning_rate_idx = 0
    learning_rate = learning_rates[learning_rate_idx]

    prev_c5 = float('inf')
    iteration = 0
    while time.time() - t0 < budget_s:
        iteration += 1
        # Step update
        h = project(h - learning_rate * gradient_descent_step(h, learning_rate))
        # Evaluate
        c5 = objective(h)
        # Track best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            prev_c5 = c5
        else:
            # Reduce learning rate if no improvement
            if iteration % 100 == 0 and best_c5 >= prev_c5:
                print("Reducing learning rate")
                learning_rate_idx = min(learning_rate_idx + 1, len(learning_rates) - 1)
                learning_rate = learning_rates[learning_rate_idx]
                prev_c5 = c5
        params = {
            'iteration': iteration,
            'learning_rate': learning_rate,
            'c5': best_c5,
            'time_remaining': budget_s - (time.time() - t0)
        }
        if iteration % 100 == 0:
            print(f"Iteration: {params['iteration']}, C5: {params['c5']:.6f}, Time remaining: {params['time_remaining']:.2f}s")

    return (best_h.round(7).tolist(), float(best_c5), n_points)
