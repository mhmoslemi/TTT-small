import numpy as np
import time
import math

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    c = np.correlate(h_values, 1.0 - h_values, mode="full") * dx
    c5 = np.max(c)
    return c5, c

def run(seed=42, budget_s=1100, max_iterations=10000):
    t_start = time.time()
    np.random.seed(seed)
    
    # Load initial solution if available
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        n_points = 42
        h = np.full(n_points, 1.0 / n_points * n_points / 2, dtype=np.float64)
    
    # Feasibility projection (enforce sum constraint)
    def project(h):
        # Project into [0,1]
        h_clipped = np.clip(h, 0, 1)
        # Normalize to preserve total integral
        total = h_clipped.sum()
        if not np.isclose(total, 0.0):
            h_clipped *= n_points / 2 / total
        return h_clipped
    
    # Optimization loop
    learning_rate = 0.01
    iteration = 0
    max_iteration = max_iterations
    
    while time.time() - t_start < budget_s:
        # Evaluate current solution
        c5, cors = evaluate_erdos_solution(h, n_points)
        
        # Find the worst k with maximum overlap
        dx = 2.0 / n_points
        k_max = np.argmax(cors)
        
        # Calculate gradient at worst overlap
        # Get shifted version of h for that correlation
        center = k_max
        if center < n_points:
            h_shifted = np.concatenate([h[center:], np.zeros(n_points - center)])
        else:
            h_shifted = np.zeros(n_points)
            if center - n_points < n_points:
                h_shifted[:(center - n_points)] = h[(center - n_points):]
        
        # Compute true gradient vector = 1 - h_shifted
        grad = (1.0 - h_shifted)
        
        # Update
        h_delta = -learning_rate * grad
        h += h_delta
        h = project(h)
        
        # Reduce learning rate
        iteration += 1
        if iteration % 100 == 0:
            learning_rate *= 0.95
        
        if iteration % 1000 == 0:
            c5_new, _ = evaluate_erdos_solution(h, n_points)
            print(f"Iteration {iteration}: C5 = {c5_new:.6f}, lr = {learning_rate:.4f}")
    
    c5_final, _ = evaluate_erdos_solution(h, n_points)
    return h.tolist(), c5_final, n_points
