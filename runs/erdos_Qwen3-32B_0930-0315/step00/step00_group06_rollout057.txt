import numpy as np
import time
from scipy.signal import correlate

def project(h, n_points):
    h = np.clip(h, 0, 1)
    scale = (n_points / 2) / h.sum()
    h *= scale
    return h

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode="full")
    corr *= dx
    return np.max(corr)

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    n_points = 256
    h = np.array(initial_h_values['h'], copy=True)
    h = np.tile(h, n_points // len(h))[:n_points]
    h = project(h, n_points)

    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    n_iter = 0

    learning_rate = 0.1
    decay = 0.9999
    perturb_interval = 100
    perturb_amount = 0.05

    while time.time() - start_time < budget_s:
        n_iter += 1

        corr = correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        k = idx - n_points + 1

        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + k >= 0 and i + k < n_points:
                grad[i] = h[i + k]

        h -= learning_rate * grad
        h = project(h, n_points)

        current_c5 = compute_c5(h, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        if n_iter % perturb_interval == 0:
            h += perturb_amount * np.random.randn(n_points)
            h = project(h, n_points)

        learning_rate *= decay

    return best_h, best_c5, n_points
