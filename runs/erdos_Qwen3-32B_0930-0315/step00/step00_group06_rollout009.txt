import numpy as np
import time
from numpy import correlate, sum, clip
np.random.seed(42)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    # Parameters
    n_points = 256  # Increased resolution
    dx = 2.0 / n_points
    base_learning_rate = 0.05
    perturb_iter = 500

    # Initial solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values['h'])
        while len(h) < n_points:
            h = np.concatenate((h, h))
        h = h[:n_points]
    else:
        h = np.random.rand(n_points)
    
    h *= (n_points / 2.0) / sum(h)  # Normalize to satisfy the integral constraint

    h = clip(h, 0.0, 1.0)

    def project(h):
        total = sum(h)
        h = clip(h, 0.0, 1.0)
        h *= (n_points / 2.0) / total
        return h

    def evaluate_c5(h):
        h1 = h
        h2 = 1 - h
        corr = correlate(h1, h2, mode='full')
        corr *= dx
        return np.max(corr)

    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    patience = 0
    patience_limit = 20

    while time.time() - t0 < budget_s:
        # Compute overlap integrals for all k
        h1 = h
        h2 = 1 - h
        corr = correlate(h1, h2, mode='full') * dx
        max_idx = np.argmax(corr)

        # Compute gradient w.r.t. max overlap
        grad = np.zeros_like(h)
        grad_len = h.size
        grad[max_idx:max_idx + grad_len] += h

        grad[max_idx:max_idx + grad_len] *= -1.0
        grad[max_idx:max_idx + grad_len] += h

        grad = np.roll(grad, -grad_len)

        learning_rate = base_learning_rate * max(0.5, (time.time() - t0) / budget_s)

        h = h - learning_rate * (grad / np.linalg.norm(grad + 1e-8))

        h = project(h)

        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

        if patience > patience_limit:
            h = (h + 0.05 * np.random.randn(n_points)) / 2.0
            h = project(h)
            patience = 0

    return (best_h.tolist(), best_c5, n_points)
