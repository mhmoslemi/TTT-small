import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    st = time.time()
    dx = 2.0 / 80
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

        h = np.ones(n_points) / 2

    best_c5 = 0
    for i in range(n_points):
        valid = h[i] == h[i]
        if not valid:
            h[:] = 0.5

    # Normalize h to match the integral constraint
    h *= n_points / 2
    h = np.clip(h, 0, 1)
    h /= h.sum()
    h *= n_points / 2
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    while time.time() - st < budget_s - 10:
        # Find worst lag
        h_normalized = h / n_points * 2
        expected_corr = np.correlate(h_normalized, 1 - h_normalized, mode="full")
        C = (expected_corr * dx).tolist()
        C = C[:n_points-1] + [0] + C[n_points:]

        worst_k = np.argmax(C)

        curr_h = cp.Parameter(n_points, nonneg=True)
        curr_h.value = h.copy()
        delta = cp.Variable(n_points)

        obj = cp.Minimize(
            cp.sum(
                curr_h.value + delta[
                    (np.arange(n_points) + n_points - worst_k) % n_points
                ] * (1 - (curr_h.value + delta))
            ) * dx
        )

        constraints = [
            cp.sum(delta) == 0,
            0 <= curr_h.value + delta,
            curr_h.value + delta <= 1,
            cp.sum(curr_h.value + delta) == n_points / 2,
        ]

        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver=cp.ECOS)

        if result is not None:
            h += delta.value
            h = np.clip(h, 0, 1)
            h = (h / h.sum()) * (n_points / 2)
            h = np.clip(h, 0, 1)
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    return best_h, best_c5, n_points
