import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    dx = 2.0 / 70
    h = np.array(initial_h_values, copy=True)
    n_points = len(h)
    assert np.isclose(np.sum(h), n_points / 2)

    learning_rate = 0.1
    iters = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)

    while time.time() - start_time < budget_s:
        iters += 1

        # Compute correlation and find maximum index
        R = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(R[int(n_points / 2 - 1):int(3 * n_points / 2)])
        k = max_idx + int(n_points / 2 - 1)
        overlap = R[k] * dx

        # Compute gradient w.r.t. h at positions i and i + k
        grad = np.zeros_like(h, dtype=float)
        for i in range(n_points):
            if 0 <= i + k < n_points:
                grad[i] += (1 - h[i + k])
                grad[i + k] -= h[i]

        # Gradient step
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)

        # Current C5
        current_c5 = evaluate_c5(h, dx)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            learning_rate *= 0.99

        # Perturb every 50 iterations to escape local optima
        if iters % 50 == 0:
            noise = np.random.randn(n_points) * 0.01
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            h = np.clip(h, 0, 1)

        # Stop if learning becomes too small
        if learning_rate < 1e-4 and iters % 100 == 0:
            break

    return (best_h, best_c5, n_points)

def evaluate_c5(h, dx):
    R = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(R[int(len(h)/2-1):int(3*len(h)/2)]) * dx
    return max_overlap
