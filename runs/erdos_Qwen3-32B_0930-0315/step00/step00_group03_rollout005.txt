import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(70, 0.5)
    dx = 2 / len(h)
    np.testing.assert_approx_equal(np.sum(h), len(h) / 2, 1e-6)
    best_h = h.copy()
    best_c5 = float('inf')
    n_points = len(h)
    iteration = 0
    elapsed = 0.0
    start_time = time.time()

    while elapsed < budget_s:
        iteration += 1

        # Compute all overlaps via cross-correlation
        ones_minus_h = 1 - h
        overlaps = np.correlate(h, ones_minus_h, mode='full') * dx
        max_index = np.argmax(np.abs(overlaps))
        k = max_index - (len(h) - 1)
        current_c5 = np.max(overlaps)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Compute gradient for current worst shift k
        G = np.zeros_like(h)
        for i in range(len(h)):
            if i + k < len(h):
                G[i] += (1 - h[i + k])
            if i - k >= 0 and i - k < len(h):
                G[i] -= h[i - k]
        learning_rate = 0.01 / (1 + iteration / 1000)
        h_update = h - learning_rate * G

        # Project back to feasible region
        h_var = cp.Variable(shape=h_update.shape)
        obj = cp.Minimize(cp.sum_squares(h_var - h_update))
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS)
        if prob.status == cp.OPTIMAL:
            h = h_var.value
        else:
            print("Projection not optimal, skipping iterate")

        elapsed = time.time() - start_time

        # Convergence check
        if iteration % 100 == 0 or elapsed > budget_s - 0.1:
            # Enforce constraint explicitly (round and project)
            h = h - h.mean() + 0.5
            h = np.clip(h, 0, 1)
            h -= (np.sum(h) - n_points / 2) / n_points
            h = np.clip(h, 0, 1)

        if iteration % 2000 == 0 and iteration > 0:
            break

    # Final projection
    h_var = cp.Variable(shape=h.shape)
    obj = cp.Minimize(cp.sum_squares(h_var - h))
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2
    ]
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS)
    if prob.status == cp.OPTIMAL:
        h = h_var.value

    # Final validation
    h = np.clip(h, 0, 1)
    h -= (np.sum(h) - n_points / 2) / n_points
    h = np.clip(h, 0, 1)
    ones_minus_h = 1 - h
    overlaps = np.correlate(h, ones_minus_h, mode='full') * dx
    best_c5 = np.max(overlaps)

    return (h, best_c5, n_points)
