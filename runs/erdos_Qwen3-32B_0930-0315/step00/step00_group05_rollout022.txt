import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    h = np.copy(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    start_time = time.time()
    best_h = None
    best_c5 = float("inf")

    h = np.clip(h, 0, 1)
    h *= target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    h *= target_sum / np.sum(h)

    learning_rate = 0.1
    iteration = 0

    while time.time() - start_time < budget_s:
        iteration += 1

        # Compute the cross-correlation between h and (1 - h)
        h_extended = np.concatenate([h, h])
        corr = np.zeros(n_points)
        for k in range(n_points):
            mask = (np.arange(n_points + k) >= k) & (np.arange(n_points + k) < n_points + k)
            inner = h_extended[k:k + n_points] * (1 - h)
            overlap = np.sum(inner[mask])
            corr[k] = overlap * dx

        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]

        if max_corr < best_c5:
            best_c5 = max_corr
            best_h = np.copy(h)

        # Gradient computation
        grad = np.zeros(n_points)
        grad_k = max_idx
        for j in range(n_points):
            try:
                if j + grad_k < n_points:
                    grad[j] += (1 - h[(j + grad_k) % n_points]) * dx
                    grad[j] -= h[j] * dx
            except IndexError:
                pass

        # Gradient step
        h -= learning_rate * grad

        # Clip and project onto feasible set
        h = np.clip(h, 0, 1)
        h *= target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        h *= target_sum / np.sum(h)

        # Reduce learning rate every 100 iterations
        if iteration % 100 == 0:
            learning_rate *= 0.95

        # Perturbation every 50 iterations
        if iteration % 50 == 0:
            noise = np.random.normal(0, 0.01, size=n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h *= target_sum / np.sum(h)

    return best_h, best_c5, n_points
