import numpy as np
import time
import cvxpy as cp

def evaluate_c5(h):
    dx = 2.0 / len(h)
    h = h.astype(np.float64)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr * dx)
    return c5

def clip_and_project(h):
    h = np.clip(h, 0.0, 1.0)
    desired_sum = len(h) / 2.0
    total = np.sum(h)
    scaling = desired_sum / total
    h *= scaling
    return h
