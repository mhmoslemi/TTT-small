import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points

    # Ensure the initial guess integrates to 1
    h_sum = np.sum(h) * dx
    if not np.isclose(h_sum, 1.0):
        scale = 1.0 / h_sum
        h = h * scale
    h[h < 0] = 0
    h[h > 1] = 1

    # Constants for optimization
    max_iter = 100000
    learning_rate = 0.005
    tolerance = 1e-6

    best_h = h.copy()
    best_c5 = compute_c5(h, n_points, dx)
    no_improvement_count = 0
    start_time = time.time()

    for it in range(max_iter):
        if time.time() - start_time >= budget_s:
            break

        # Compute cross-correlation of h with (1 - h) on [0, n_points * 2 - 1]
        cross_corr = correlate(h, 1 - h, mode='full')
        cross_corr *= dx  # Scale by dx for integral

        # Take absolute indices but get circular motif via views (no wrap)
        k = np.argmin(cross_corr)  # Pointing to "most favorable" shift
        k = k - (n_points - 1) if k >= n_points else k

        # Gradient computation for shift k
        grad = np.zeros_like(h)
        for i in range(n_points):
            left = (i - k) % n_points
            right = (i + k) % n_points
            grad[i] = (1 - h[left]) - h[right]

        # Gradient descent step
        h_flat = h.flatten()
        h_flat -= learning_rate * grad

        # Projection back to feasible set
        h_flat[h_flat < 0] = 0
        h_flat[h_flat > 1] = 1
        h = h_flat.reshape(-1)

        # Repair constraint to have mean 1 (given dx = 2.0 / n_points)
        h_sum = np.sum(h) * dx  # For efficiency
        if not np.isclose(h_sum, 1.0):
            scale = 1.0 / h_sum
            h = h * scale
        h[h < 0] = 0
        h[h > 1] = 1

        # Objective eval
        cur_c5 = compute_c5(h, n_points, dx)
        if cur_c5 < best_c5:
            best_h[:] = h[:]
            best_c5 = cur_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 100:
                print("Adapting learning rate...")
                learning_rate *= 0.9
                no_improvement_count = 0

        if abs(cur_c5 - best_c5) < tolerance:
            print("Converged.")
            break

    return best_h.tolist(), best_c5, n_points

def compute_c5(h, n_points, dx):
    # Evaluate full cross-correlation h * (1 - h), unwrap to all shifts
    cross_corrs = []
    h_fft = np.fft.fft(h)
    f1 = np.fft.fft(1 - np.roll(h, i) for i in range(n_points * 2))  # Precompute for all possible shifts
    cross_corrs = [np.abs(np.fft.ifft(h_fft * np.conj(f)).real) for f in f1]
    max_c = float(np.max(cross_corrs) * dx)
    return max_c
