import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 47  # Start with the given parent construction (initial_h_values)

    if 'initial_h_values' not in globals():
        raise ValueError("initial_h_values not provided")

    # Use parent as initialization
    h = np.array(initial_h_values).copy()
    h = np.clip(h, 0, 1)
    h = h / h.sum() * n_points / 2

    dx = 2.0 / n_points
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_C5(h, dx)
    
    iter = 0
    patience = 0
    learning_rate = 0.05

    while time.time() - start_time < budget_s and patience < 50:
        h_prev = h.copy()
        # Compute overlap for all lags
        overlap = np.correlate(h, 1 - h, mode="full")
        overlap = overlap * dx
        max_index = np.argmax(np.abs(overlap))
        
        # Compute worst lag k
        k = max_index - n_points + 1 if max_index >= n_points else max_index
        max_K = min(overlap.shape[0], 2 * n_points - 1)
        for k_idx in range(n_points, max_K):
            overlap[k_idx] = 0  # Disregard invalid lags
        for k_idx in range(0, n_points):
            overlap[k_idx] = 0  # Disregard invalid lags
        k = np.argmax(overlap)
        lag = k - n_points + 1 if k >= n_points else k

        # Compute gradient for the worst lag
        grad = np.zeros_like(h)
        valid_indices = np.arange(0, n_points - lag)
        grad[valid_indices] = (1 - h[valid_indices + lag]) - h[valid_indices]
        if lag < 0:
            grad[-lag:] = (1 - h[: -lag]) - h[-lag:]
            valid_indices = np.arange(-lag, n_points)
            grad[valid_indices] = (1 - h[valid_indices + lag]) - h[valid_indices]

        # Update h
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h / h.sum() * n_points / 2
        
        # Compute new C5
        current_c5 = evaluate_C5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
            if iter % 10 == 0:
                print(f"{iter}: {best_c5}")
        else:
            patience += 1

        iter += 1

    return (best_h.tolist(), best_c5, n_points)

def evaluate_C5(h, dx):
    overlap = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)
