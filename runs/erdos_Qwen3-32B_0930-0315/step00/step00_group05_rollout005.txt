import numpy as np
import time
import scipy.signal
from math import floor

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    one_minus_h = 1.0 - h
    corr = scipy.signal.correlate(h, one_minus_h, mode='full')
    corr *= dx
    # Exclude the central point (zero shift)
    max_c5 = np.max(np.concatenate([corr[:n_points-1], corr[n_points+1:]]))
    return max_c5
