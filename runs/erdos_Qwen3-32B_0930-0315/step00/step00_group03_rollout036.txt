import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    # Set seed and use parent initialization if available
    np.random.seed(seed)
    n_points = 70
    if 'initial_h_values' in globals():
        h = np.array(globals()['initial_h_values'], dtype=np.float64).copy()
    else:
        h = np.zeros(n_points, dtype=np.float64)
        idx = np.random.choice(n_points, int(n_points/2), replace=False)
        h[idx] = 1.0

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)

    # Setup parameters
    lr = 0.01
    max_iter = int(budget_s * 1.0 / (0.002 + 0.01))
    no_improve_counter = 0
    start_time = time.time()

    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        # Evaluate overlaps and find max violating shift
        c5, k = evaluate_c5_and_argmax(h, dx)
        if c5 < best_c5 - 1e-8:
            best_c5 = c5
            best_h = h.copy()
            no_improve_counter = 0
        elif no_improve_counter >= 100:
            break
        else:
            no_improve_counter += 1

        # Compute gradient for violating shift k
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k < n_points:
                grad[i] += (1.0 - h[i + k])
            if i - k >= 0:
                grad[i] -= h[i - k]
        grad *= dx

        # Gradient step
        h_candidate = h - lr * grad

        # Project onto constraint: 0 ≤ h ≤ 1, sum(h) = n_points / 2
        h_proj = project_onto_feasible(h_candidate, n_points)

        # Update h for next iteration
        h = h_proj
        lr = 0.01 / (1 + iter / 1000)

    # Final validation and formatting
    if not np.allclose(h.sum(), n_points / 2, atol=1e-6):
        h = best_h
    c5, _ = evaluate_c5_and_argmax(h, dx)
    return h.tolist(), c5, n_points


def project_onto_feasible(h_candidate, n_points):
    h = cp.Variable(n_points)
    constraints = [0 <= h, h <= 1]
    constraints.append(h.sum() == n_points / 2)
    objective = cp.Minimize(cp.sum_squares(h - h_candidate))
    problem = cp.Problem(objective, constraints)
    try:
        problem.solve()
        return h.value
    except Exception:
        return project_onto_feasible_exact(h_candidate)


def project_onto_feasible_exact(h_candidate):
    idx = np.argsort(h_candidate)
    values = h_candidate[idx]
    n = len(values)
    target_sum = n / 2

    # Find threshold using binary search
    low, high = 0.0, 1.0
    eps = 1e-9
    sol = None

    while high - low > eps:
        mid = (low + high) / 2
        set_high = values >= mid
        result = np.sum(np.clip(values, 0.0, 1.0))
        if abs(result - target_sum) < 1e-9:
            sol = mid
            break
        elif result < target_sum:
            high = mid
        else:
            low = mid

    if sol is None:
        sol = (low + high) / 2

    h_proj = np.clip(values, 0.0, 1.0)
    h_proj[values < sol] = 0.0
    h_proj[values >= sol] = 1.0

    return h_proj[np.argsort(idx)]


def evaluate_c5(h, dx):
    values = h
    ones_minus_h = 1 - h
    result = np.correlate(values, ones_minus_h, mode='full') * dx
    return -np.min(result)


def evaluate_c5_and_argmax(h, dx):
    values = h
    ones_minus_h = 1 - h
    result = np.correlate(values, ones_minus_h, mode='full') * dx
    max_idx = np.argmin(result)
    if max_idx >= len(h):
        max_idx -= len(h)
    return -result[np.argmin(result)], max_idx
