import numpy as np
import time
import evaluate_erdos_solution as ees

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Use the initial construction
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points

    # Normalize h to satisfy the integral constraint
    h *= (n_points / 2.0) / h.sum()

    # Optimization parameters
    learning_rate = 0.01
    decay_rate = 0.9995
    max_steps = 100000
    no_improve_tolerance = 100
    steps_without_improvement = 0
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()

    for step in range(max_steps):
        # Time budget check
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break

        # Compute overlap for all lags
        overlap = ees.evaluate_erdos_solution(h)

        # Find the lag with maximum overlap
        k_max = np.argmax(overlap)
        C5 = overlap[k_max]

        # Compute subgradient for the max overlap
        grad = np.zeros_like(h)
        for i in range(n_points):
            offset = k_max
            if 0 <= i + offset < n_points:
                grad[i] += h[i + offset]
            if 0 <= i - offset < n_points:
                grad[i] += h[i - offset]
        grad -= h

        # Gradient descent step
        h -= learning_rate * grad

        # Project back into [0, 1]
        h = np.clip(h, 0, 1)

        # Renormalize to satisfy the integral constraint
        h *= (n_points / 2.0) / h.sum()

        # Check improvement
        current_c5 = overlap[k_max]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            steps_without_improvement = 0
        else:
            steps_without_improvement += 1

        # Adaptive learning rate
        learning_rate *= decay_rate

        # Perturbation for exploration after no improvement
        if steps_without_improvement >= no_improve_tolerance:
            h += 0.01 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h *= (n_points / 2.0) / h.sum()
            steps_without_improvement = 0

    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap * dx)
