import numpy as np
import time
import random

from numpy.random import default_rng

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    total = h.sum()
    scale = n_points / 2 / total
    h = h * scale
    h = np.clip(h, 0, 1)
    h = h / h.sum() * (n_points / 2)
    h = np.clip(h, 0, 1)
    return h

def mutate(h, mutation_rate=0.15, noise_std=0.1):
    h = h.copy()
    n_points = len(h)
    indices = np.random.choice(n_points, size=int(n_points * 0.05), replace=False)
    h[indices] += np.random.normal(0, noise_std, size=len(indices))
    h = np.clip(h, 0, 1)
    h = repair_h(h, n_points)
    return h

def crossover(parent1, parent2, crossover_point, n_points):
    child = np.where(np.arange(n_points) < crossover_point, parent1, parent2)
    child = repair_h(child, n_points)
    return child

def tournament_selection(population, fitness, tournament_size=5):
    indices = np.random.choice(len(population), size=tournament_size, replace=False)
    selected_idx = indices[np.argmax(fitness[indices])]
    return population[selected_idx], fitness[selected_idx]
