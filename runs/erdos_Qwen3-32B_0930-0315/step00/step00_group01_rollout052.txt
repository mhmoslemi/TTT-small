import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 128
    dx = 2.0 / n_points

    # Use initial_h_values provided by the context if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=np.float64)
        if h_init.size != n_points:
            raise ValueError("initial_h_values size does not match n_points")
        norm_sum = h_init.sum()
        if abs(norm_sum - n_points / 2) > 1e-6:
            h_init *= (n_points / 2) / norm_sum
        h_values = h_init.copy()
    else:
        h_values = np.random.rand(n_points)
        h_values = np.clip(h_values, 0, 1)
        h_values *= (n_points / 2) / (h_values.sum())

    h = cp.Variable(n_points)
    objective = cp.Minimize(cp.Constant(1e6) * cp.sum_squares(h - h_values) + cp.Constant(1.0))
    constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2]
    all_correls = []
    worst_corr = cp.Parameter(nonneg=True)

    def cross_correlate_t(h_, k):
        # k is offset (positive or negative)
        if k == 0:
            return cp.Constant(0)  # h(x)(1 - h(x)) integrates to 0 over symmetrical functions
        valid_indices = np.s_[n_points - 1 - k:] if k > 0 else np.s_[:n_points + k]
        if k > 0:
            lhs = h[:n_points - k]
            rhs = 1.0 - h[k:]
        else:
            lhs = 1.0 - h[n_points + k:]
            rhs = h[-k:]
        return cp.Sum(lhs * rhs)

    max_corr = 0.0
    best_h = h_values.copy()

    try:
        while time.time() - start_time < budget_s:
            corr_list = []
            for k in range(-n_points + 1, n_points):
                corr = cross_correlate_t(h, k)
                corr_list.append(corr)

            corr_list = [corr for corr in corr_list if corr is not None]
            max_corr_var = cp.Max(corr_list)
            constraints.append(max_corr_var <= worst_corr)
            objective = cp.Minimize(worst_corr)
            problem = cp.Problem(objective, constraints)

            # Prioritize fast solvers with no cost license
            solver = cp.CVXOPT
            problem.solve(solver=solver, max_iter=50, verbose=False)

            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = np.clip(h.value, 0, 1)
                h_sol /= h_sol.sum() * dx * 2.0
                h_sol *= (n_points / 2.0) / h_sol.sum()

                h_values = h_sol
                temp_corr = []
                for k in range(-n_points + 1, n_points):
                    if k == 0:
                        corr = 0.0
                    elif k > 0:
                        lhs = h_values[:n_points - k]
                        rhs = 1.0 - h_values[k:]
                    else:
                        lhs = 1.0 - h_values[-k:]
                        rhs = h_values[n_points + k:]
                    corr = (lhs * rhs).sum() * dx
                    temp_corr.append(corr)
                c5_bound = max(temp_corr)
                if c5_bound < max_corr:
                    best_h = h_values.copy()
                    max_corr = c5_bound

            else:
                break

        c5_bound = max_corr

    except Exception:
        pass

    c5_bound = max(np.correlate(h_values, 1 - h_values, mode='full')) * dx
    return (best_h, c5_bound, n_points)
