import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    t0 = time.time()
    best_c5 = np.inf
    best_h = None
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Start from initial_h_values
    h = np.copy(initial_h_values)

    # Gradient descent parameters
    alpha = 0.01  # initial learning rate
    max_iterations = 10_000
    stagnation_patience = 500
    stagnation_counter = 0
    best_h_update = np.copy(h)

    def compute_c5(h):
        c_cor = correlate(h, 1 - h, mode='full')
        c_cor *= dx
        return np.max(c_cor)

    def project(h, target_sum):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-5:
            scaling = target_sum / current_sum
            h = h * scaling
            h = np.clip(h, 0, 1)
        return h

    # Main optimization loop
    for it in range(max_iterations):
        elapsed = time.time() - t0
        if elapsed >= budget_s:
            break

        h_prev = np.copy(h)
        c_corr = correlate(h, 1 - h, mode='full')
        c_corr *= dx
        k_max = np.argmax(c_corr)

        # Compute gradient for the max shift index
        grad = np.zeros_like(h)
        for i in range(n_points):
            i1 = (i + k_max) % n_points
            i2 = (i - k_max + n_points) % n_points
            grad[i] = (1 - h[i1]) - (1 - h[i2])

        h -= alpha * grad
        h = project(h, target_sum)

        # Evaluate success after update
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.copy(h)
            alpha = max(0.001, alpha / 1.05)  # reduce learning rate on success
            stagnation_counter = 0
        else:
            h = np.copy(best_h_update)
            alpha = min(0.1, alpha * 1.2)  # increase learning rate on failure
            stagnation_counter += 1

        # Track progress
        if stagnation_counter >= stagnation_patience:
            h = project(h + np.random.normal(0, 0.05, size=n_points), target_sum)
            stagnation_counter = 0

        # Update tracking
        if abs(np.sum(h) - target_sum) > 1e-5:
            h = project(h, target_sum)

        # Periodic updates to best_h_update
        if it % 100 == 0:
            best_h_update = np.copy(h)

    # Final quality check
    best_h = project(best_h, target_sum)
    final_c5 = compute_c5(best_h)
    n_points = len(best_h)

    return best_h.tolist(), final_c5, n_points
