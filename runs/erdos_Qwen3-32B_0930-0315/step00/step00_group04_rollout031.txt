import numpy as np
import time
import cvxpy as cp
from scipy.interpolate import CubicSpline

def repair_constraints(h, n_points):
    h = np.clip(h, 0, 1)
    h *= n_points / (2 * np.sum(h))  # rescale to ensure ∫h(x)dx = 1
    return h

def compute_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    corrs = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corrs)

def evaluate_population(population, n_points):
    return [compute_c5(ind) for ind in population]

def select_tournament(population, scores, size=5):
    indices = np.random.choice(len(population), size, replace=False)
    selected = [population[i] for i in indices]
    scores_selected = [scores[i] for i in indices]
    best_idx = np.argmin(scores_selected)
    return selected[best_idx]

def crossover(parent1, parent2):
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return repair_constraints(child1, n_points), repair_constraints(child2, n_points)

def mutate(h, mutation_rate=0.2, n_points=None):
    if n_points is None:
        n_points = len(h)
    mask = np.random.rand(n_points) < mutation_rate
    perturbation = np.random.randn(n_points) * 0.1
    h_new = h + mask * perturbation
    return repair_constraints(h_new, n_points)

def gradient_descent(h, learning_rate=0.005, iterations=200):
    h = np.copy(h)
    n_points = len(h)
    dx = 2.0 / n_points
    for _ in range(iterations):
        # Compute correlation to find which shift gives max overlap
        corrs = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corrs)
        # Compute backward shift
        overlap_idx = max_idx - n_points + 1
        # Gradient is -(1 - h) shifted at max overlap point
        grad = np.zeros_like(h)
        if overlap_idx >= 0 and overlap_idx < n_points:
            grad[overlap_idx] = -(1 - h[overlap_idx])
        h += learning_rate * grad
        h = repair_constraints(h, n_points)
    return h

def double_resolution(h_low):
    n_points = len(h_low)
    x = np.linspace(0, 2, n_points)
    cs = CubicSpline(x, h_low)
    x_high = np.linspace(0, 2, 2 * n_points)
    h_high = cs(x_high)
    h_high = repair_constraints(h_high, 2 * n_points)
    return h_high

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use initial_h_values if available, otherwise create a basic uniform solution
    if 'initial_h_values' in globals():
        h_start = initial_h_values
    else:
        n_start = 14
        h_start = np.full(n_start, .5)
        h_start = repair_constraints(h_start, n_start)

    n_points_list = []
    res = h_start
    best_c5 = compute_c5(res)
    n_points = len(h_start)
    best_h = res
    n_points_list.append(n_points)
    resolution_steps = 0

    while time.time() - start_time < budget_s * 0.85:
        # Genetic Algorithm Phase
        population_size = 20
        generations = 50
        population = []
        for _ in range(population_size):
            if _ == 0:
                ind = np.copy(best_h)
            else:
                ind = mutate(np.copy(best_h), mutation_rate=0.2, n_points=n_points)
            ind = gradient_descent(ind, learning_rate=0.005)
            population.append(ind)

        for gen in range(generations):
            scores = evaluate_population(population, n_points)
            new_population = []
            for _ in range(population_size // 2):
                parent1 = select_tournament(population, scores)
                parent2 = select_tournament(population, scores)
                child1, child2 = crossover(parent1, parent2)
                child1 = mutate(child1, mutation_rate=0.2, n_points=n_points)
                child2 = mutate(child2, mutation_rate=0.2, n_points=n_points)
                new_population.extend([child1, child2])
            population = [select_tournament(population + new_population, evaluate_population(population + new_population, n_points)) for _ in range(population_size)]
            scores = evaluate_population(population, n_points)
            best_idx = np.argmin(scores)
            curr_best_c5 = scores[best_idx]
            if curr_best_c5 < best_c5:
                best_c5 = curr_best_c5
                best_h = population[best_idx]

        # Gradient Descent Phase
        best_h = gradient_descent(best_h, learning_rate=0.001, iterations=100)

        if best_c5 < compute_c5(res) and time.time() - start_time < budget_s * 0.9:
            res = np.copy(best_h)
        # Break condition
        if resolution_steps == 0 and best_c5 < 0.38085:
            break  # Already reached target, exit early

        # Double resolution (if time allows)
        if time.time() - start_time < budget_s * 0.8:
            best_h = double_resolution(best_h)
            n_points = len(best_h)
            n_points_list.append(n_points)
            resolution_steps += 1

        if resolution_steps >= 2:
            break  # Max two resolution doublings

    # Final refinement
    best_h = gradient_descent(best_h, learning_rate=0.001, iterations=200)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_points)
