import numpy as np
import cvxpy as cp
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time()

    # Import initial_h_values provided by the harness
    from __main__ import initial_h_values

    # Use existing n_points from initial_h_values
    h0 = initial_h_values  # Warm start
    n_points = len(h0)
    dx = 2.0 / n_points

    # CVX variables and constraints
    h = cp.Variable(n_points, nonneg=True)
    t = cp.Variable()

    constraints = [
        cp.sum(h) == n_points / 2,
        h <= 1,
        h >= 0
    ]

    max_corr = 0.0
    for k in range(-n_points + 1, n_points - 1):
        # Compute convolution indices
        valid = []  # (i, j) where h[i] * (1 - h[j]) is valid
        for i in range(n_points):
            j = (i + k + n_points) % n_points
            if j != i or k >= 0:
                valid.append((i, j))
        if not valid:
            continue  # Skip invalid shifts

        # Build the cross-correlation expression
        corr_expr = sum(h[i] * (1 - h[j]) for (i, j) in valid) * dx

        # Add constraint corr_expr <= t
        constraints.append(corr_expr <= t)

    # Solve the convex relaxation
    prob = cp.Problem(cp.Minimize(t), constraints)
    result = prob.solve(solver=cp.CVXOPT, verbose=False)
    h_sol = np.clip(np.array(h.value).flatten(), 0, 1)

    # Re-evaluate final solution using harness
    def evaluate(h):
        n = h.size
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n)

    h_sol = h_sol / np.sum(h_sol) * n_points / 2  # Project back to constraint
    h_sol = np.clip(h_sol, 0, 1)
    c5_bound = evaluate(h_sol)

    # Ensure we return within budget
    elapsed = time() - start_time
    if elapsed < budget_s / 3:
        # Do a second warm-started solve for better local search
        h2 = cp.Variable(n_points, nonneg=True)
        t2 = cp.Variable()

        constraints2 = [
            cp.sum(h2) == n_points / 2,
            h2 <= 1,
            h2 >= 0
        ]

        for k in range(-n_points + 1, n_points - 1):
            valid = []
            for i in range(n_points):
                j = (i + k + n_points) % n_points
                if j != i or k >= 0:
                    valid.append((i, j))
            if not valid:
                continue
            corr_expr2 = sum(h2[i] * (1 - h2[j]) for (i, j) in valid) * dx
            constraints2.append(corr_expr2 <= t2)

        prob2 = cp.Problem(cp.Minimize(t2), constraints2)
        result2 = prob2.solve(solver=cp.CVXOPT, verbose=False)
        h_sol2 = np.clip(np.array(h2.value).flatten(), 0, 1)
        h_sol2 = h_sol2 / np.sum(h_sol2) * n_points / 2
        c5_bound2 = evaluate(h_sol2)

        if c5_bound2 < c5_bound:
            h_sol, c5_bound = h_sol2, c5_bound2

    return h_sol.tolist(), c5_bound.tolist(), n_points
