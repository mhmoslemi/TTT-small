import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 47  # Start with initial h_values' resolution

    # Load or construct initial h
    if 'initial_h_values' in globals():
        h = np.copy(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = (h / h.sum()) * (n_points / 2)
        h = np.clip(h, 0, 1)

    h = cp.Parameter(n_points, nonneg=True)
    t = cp.Variable()

    dx = 2.0 / n_points
    obj = cp.Minimize(t)
    constraints = []

    # Constraint: sum(h) == n_points / 2
    constraints.append(cp.sum(h) == n_points / 2)
    # Bound: h in [0, 1]
    constraints.append(0 <= h)
    constraints.append(h <= 1)

    # Construct per-k correlation constraints
    for k in range(-n_points + 1, n_points):
        # Compute valid indices for cross-correlation at lag k
        if k < 0:
            # reverse offset when k is negative
            offset = -k
            idx_start = offset
            idx_end = n_points
            h1 = h[idx_start:idx_end]
            h2 = h[0:n_points + k]
        elif k == 0:
            h1 = h[1:]
            h2 = h[0:-1]
        else:
            idx_start = 0
            idx_end = n_points - k
            h1 = h[idx_start:idx_end]
            h2 = h[k:n_points]

        # Construct the k-th correlation constraint
        corr = cp.sum(h1 * (1 - h2))
        constraints.append(corr <= t * (n_points / 2))

    prob = cp.Problem(obj, constraints)

    try:
        # Warm start if initial_h_values is given
        if 'initial_h_values' in globals():
            h.value = initial_h_values.copy()
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
    except:
        pass

    # Post-process h to maintain constraints and improve precision
    h_opt = np.clip(h.value, 0, 1)
    h_opt = h_opt / h_opt.sum() * (n_points / 2)
    h_opt = np.clip(h_opt, 0, 1)

    # Compute C5 using correlation approach
    corr = np.correlate(h_opt, 1 - h_opt, mode='full') * dx

    # Final bounds
    c5_bound = np.max(corr)

    # Return the best solution found
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        print("Time budget not fully used. Consider increasing resolution or trying random variants.")
    return (h_opt.tolist(), c5_bound, n_points)
