import numpy as np
import cvxpy as cp
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time()

    # Read input
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Initialize h
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = np.inf

    def compute_c5(h):
        # Compute overlap integral for all lags
        corr = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(corr)
        worst_k = int(np.argmax(corr))
        return c5, worst_k

    # Main optimization loop
    iteration = 0
    while time() - start_time < budget_s:
        iteration += 1

        # Compute current C5 value and worst lag
        current_c5, worst_k = compute_c5(h)

        # Check if we found a better solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Stop if we meet or beat the target
        if best_c5 <= 0.380876:
            return best_h, best_c5, n_points

        # Stop early if too little improvement remains
        if iteration > 100:
            prev_c5 = best_c5
            best_c5, _ = compute_c5(best_h)
            if np.abs(prev_c5 - best_c5) < 1e-5:
                return best_h, best_c5, n_points

        # Set up convex optimization problem to improve worst lag
        h_var = cp.Variable(n_points)
        shifted_h = np.roll(h, -worst_k)
        cost = cp.sum(cp.multiply(h_var, 1 - shifted_h)) * dx

        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]

        prob = cp.Problem(cp.Minimize(cost), constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                raise RuntimeError("Solver did not converge")
            h = h_var.value
        except:
            # Fallback if solver fails
            h = best_h.copy()

        # Clip to numerical precision
        h = np.clip(h, 0, 1)

    # Final evaluation
    final_c5, _ = compute_c5(best_h)
    return best_h, final_c5, n_points
