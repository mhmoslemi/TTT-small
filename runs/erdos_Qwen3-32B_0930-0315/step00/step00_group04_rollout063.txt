import numpy as np
from evaluate_erdos_solution import evaluate_erdos_solution
import time
import copy

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    remaining_time = lambda: max(0, budget_s - (time.time() - start_time))

    if 'initial_h_values' in globals():
        n_initial = len(initial_h_values)
        n_points = max(14, n_initial // 2)  # use a coarser resolution to start
        base_h = np.mean(np.reshape(initial_h_values[:n_points * 2], (n_points, 2)), axis=1)
    else:
        n_points = 14
        base_h = np.zeros(n_points)
        base_h[:n_points // 2] = 0.8
        base_h[n_points // 2:] = 0.2
        base_h = base_h / np.sum(base_h) * n_points / 2  # normalize to meet constraint

    best_h = copy.deepcopy(base_h)
    best_c5 = float('inf')

    def repair_individual(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2  # maintain integral constraint
        return h

    def evaluate(h, n_points):
        dx = 2.0 / n_points
        h_arr = h * dx
        overlap = np.correlate(h_arr, 1 - h_arr, mode='full') * dx
        return np.max(overlap)

    # Genetic Algorithm stage
    population_size = 20
    population = [copy.deepcopy(base_h) for _ in range(population_size)]

    for gen in range(500):
        if remaining_time() < 2:
            break

        # Evaluate population
        scores = [(i, evaluate(h, n_points)) for i, h in enumerate(population)]

        # Sort by score
        scores.sort(key=lambda x: x[1])
        for i, s in scores:
            if s < best_c5:
                best_c5 = s
                best_h = copy.deepcopy(population[i])

        # Select elites (top 20%)
        elite_size = max(2, int(0.2 * population_size))
        elites = [population[i] for i, _ in scores[:elite_size]]

        # Generate new population
        new_population = elites[:]
        while len(new_population) < population_size:
            # Crossover
            if np.random.rand() < 0.8:
                p1, p2 = np.random.choice(elite_size, 2, replace=False)
                parent1 = population[scores[p1][0]]
                parent2 = population[scores[p2][0]]
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                new_population.extend([repair_individual(child1), repair_individual(child2)])
            else:
                # Mutation
                idx = np.random.randint(population_size)
                child = population[idx].copy()
                mutation_mask = np.random.rand(n_points) < np.random.uniform(0.05, 0.2)
                child[mutation_mask] += np.random.randn(np.sum(mutation_mask)) * 0.1
                new_population.append(repair_individual(child))

        # Keep population size fixed
        population = new_population[:population_size]

    # Refine with gradient descent
    h_gd = repair_individual(best_h)
    learning_rate = 0.01
    for _ in range(200):
        if remaining_time() < 1:
            break

        # Compute gradient
        dx = 2.0 / n_points
        for k in range(len(np.correlate(h_gd, 1 - h_gd, mode='full'))):
            overlap = np.correlate(h_gd, 1 - h_gd, mode='full')
            current_max_idx = np.argmax(overlap)
            gradient = np.zeros(n_points)
            lag = current_max_idx - n_points + 1
            if 0 <= lag < n_points:
                gradient[lag] = 1.0
            h_gd -= learning_rate * gradient
            h_gd = repair_individual(h_gd)

        # Evaluate refined solution
        c5 = evaluate(h_gd, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_gd

    # Discretize resolution and compute final result
    final_h = best_h
    final_n = n_points * 2
    final_h = np.interp(np.linspace(0, 1, final_n), np.linspace(0, 1, n_points), final_h)
    final_h = repair_individual(final_h)
    final_c5 = evaluate(final_h, final_n)

    return (final_h, final_c5, final_n)
