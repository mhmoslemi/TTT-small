        h = np.clip(h, 0, 1)

def compute_c5(h, n, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx


def compute_gradient(h, n):
    corr = np.correlate(h, 1 - h, mode="full")
    kmax = np.argmax(corr)

    # Compute gradient based on kmax overlap
    gradient = np.zeros(n)
    for i in range(n):
        iwr = i % n
        kmxiwr = (i - kmax) % n
        gradient[i] = (1 - h[(i + kmax) % n]) - h[i - kmax] if i >= kmax else (1 - h[(i + kmax) % n])

    gradient -= h * dx
    return gradient


def adaptive_step_size(h, grad, n):
    # Simple BB step size for direction of descent
    num = np.sum((h * grad)[:, np.newaxis] * grad, axis=1)
    denom = np.sum((h * grad)[:, np.newaxis] * grad[:, np.newaxis], axis=(1, 2))
    return num / (denom + 1e-12)
