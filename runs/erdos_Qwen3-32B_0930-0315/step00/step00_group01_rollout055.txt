import numpy as np
import cvxpy as cp
import math
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(budget_s=1100, seed=42):
    np.random.seed(seed)
    start_time = time.time()
    initial_h_values = np.array(initial_h_values if 'initial_h_values' in globals() else [])
    if initial_h_values.size > 0:
        h_version = False
        n_points = initial_h_values.shape[0]
        h_start = initial_h_values.copy()
    else:
        h_version = True
        n_points = 47  # start with a reasonable size
        h_start = np.full(n_points, n_points / 2 / n_points)

    # Precompute dx once
    dx = 2.0 / n_points

    # Warm start
    h = cp.Variable(n_points, nonneg=True)
    t = cp.Variable()

    lam = cp.Variable(n_points, nonneg=True)
    eps = 1e-6

    # Constraints for probabilities in [0, 1]
    cons = [
        h <= 1 + eps,
        h >= -eps,
    ]

    # Constraint on the integral
    cons += [cp.sum(h) == n_points / 2]

    # Constraints for correlation decomposition
    correlations = np.zeros(n_points)
    max_k = n_points - 1
    for k in range(-max_k, max_k + 1):
        if k == 0:
            continue
        overlap_indices = np.arange(max(0, -k), n_points - abs(k))
        indices1 = overlap_indices
        indices2 = indices1 + k if k > 0 else indices2 = indices1 + k
        indices2 = indices2 % n_points

        k_corr = cp.sum(h[indices1] * (1 - h[indices2]))
        cons += [k_corr <= t / dx]

    objective = cp.Minimize(t)

    prob = cp.Problem(objective, cons)
    prob.solve(solver=cp.MOSEK, verbose=False)

    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_vals = h.value
        h_vals = np.clip(h_vals, 0, 1)
        h_vals /= np.sum(h_vals) * dx
        h_vals *= n_points / 2 / np.sum(h_vals)
        h_vals = np.clip(h_vals, 0, 1)

        c5_bound = evaluate_erdos_solution(h_vals)
    else:
        h_vals = h_start
        c5_bound = evaluate_erdos_solution(h_vals)

    return h_vals, c5_bound, n_points
