import numpy as np
import cvxpy as cp
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    T0 = time()

    dx = 2. / 80
    n_points = 80
    h = initial_h_values.copy()

    # Project h exactly onto constraint: sum(h) == n_points / 2
    sum_h = np.sum(h)
    scaling = (n_points / 2.0) / sum_h
    h *= scaling
    h = np.clip(h, 0, 1)
    assert np.all((h >= 0) & (h <= 1)) and abs(np.sum(h) - n_points / 2.0) < 1e-6

    best_h = h.copy()
    best_c5 = np.inf

    def evaluate_c5(h):
        dh = 2.0 / h.size
        outer = np.correlate(h, h, mode="full")
        length = h.size * dh
        outer = outer[(h.size):]  # only overlapping parts are considered
        outer = outer * dh  # normalize by dx
        c5 = np.max(outer)
        return c5

    best_c5 = evaluate_c5(best_h)

    max_t = 5
    counter = 0
    prev_best = best_c5
    while time() - T0 < budget_s:
        counter += 1

        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            print(f"Iter {counter}, current improvement: {best_c5}")
            if best_c5 < 0.380876:
                print("Target bound achieved.")
                break

        # Find the worst overlap index
        outer = np.correlate(h, h, mode="full")
        outer = outer[(h.size):]  # only overlapping parts are considered
        outer = outer * dx
        worst_idx = np.argmax(outer)
        k = worst_idx + 1  # since we start after h.size

        assert 1 <= k <= h.size - 1

        idxs = np.arange(n_points)
        shifted = (idxs + k) % n_points

        # Create convex subproblem variables
        delta = cp.Variable(n_points)

        # Linearized objective for current h and lag k
        linearized_obj = cp.sum((delta - h[shifted] * delta - h * delta[shifted]) * dx)

        # Constraints
        constraints = [
            h + delta >= 0,
            h + delta <= 1,
            cp.sum(delta) == 0
        ]

        objective = cp.Minimize(linearized_obj)
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.ECOS, max_iters=50, reltol=1e-6)

        if problem.status != 'optimal':
            print('Convex subproblem failed: reverting')
            break

        delta_vals = delta.value

        # Update h
        step = 0.05  # conservative step to avoid constraint violation
        h += step * delta_vals

        # Project back to [0, 1]
        h[h < 0] = 0
        h[h > 1] = 1

        # Ensure the integral remains exactly 1
        scaling = (n_points / 2.0) / np.sum(h)
        h = h * scaling
        h = np.clip(h, 0, 1)

        if abs(evaluate_c5(h) - prev_best) < 1e-6:
            print(f"Early stopping - no improvement for step {counter}")
            if counter > 20:
                break

    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, n_points
