import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx):
    h = np.array(h)
    c = correlate(h, 1 - h, mode="full")
    return np.max(c) * dx

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start = time.time()
    max_time = budget_s * 0.98  # Leave small buffer for final steps

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
        actual_sum = np.sum(h)
        scaling = (n_points / 2) / actual_sum
        h *= scaling
    else:
        n_points = 160
        h = np.ones(n_points, dtype=np.float64) * 0.5
        h[:n_points//4] = 0.75
        h[n_points//4 * 3:] = 0.25

    dx = 2.0 / n_points

    alpha = 0.01
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    best_time = time.time()
    
    try:
        t0 = time.time()
        tmax = t0 + max_time
        niters_since_improvement = 0
        
        while time.time() < tmax:
            # Compute all cross overlaps
            corr = correlate(h, 1 - h, mode="full")
            
            # Find maximum overlap value and location
            max_idx = np.argmax(corr)
            current_c5 = corr[max_idx] * dx
            
            # If we've found a better solution
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h[:] = h[:]
                best_time = time.time()
                niters_since_improvement = 0
            else:
                niters_since_improvement += 1

            # Compute the Lagrangian gradient at max_idx with circular offset
            k = max_idx - (n_points - 1)
            if k >= 0:
                grad = np.zeros_like(h)
                # Lag is positive, need to minimize this overlap
                # For h[i]*(1 - h[i + k]) term, derivative is (1 - h[i + k]) - h[i]
                offset = -k
                grad[:] = (1 - np.roll(h, offset)) - h
            else:
                # For h[i]*(1 - h[i - k']) with k' = -k
                # Derivative is (1 - h[i]) - h[i - k']
                offset = -k
                grad[:] = (1 - h) - np.roll(h, offset)

            # Apply learning rate with adaptive scaling
            alpha = 0.01 / (1 + np.log10(min(100, time.time() - t0)))
            h -= alpha * grad

            # Projection to [0,1]
            np.clip(h, 0.0, 1.0, out=h)

            # Renormalize to expected value
            s = np.sum(h)
            h *= (n_points / 2.0) / s

            # Occasionally perturb to escape local optima
            if (time.time() - best_time) > 60 and np.random.rand() < 0.05:
                perturbation = np.random.normal(0, 0.05, size=n_points)
                h = h - perturbation
                np.clip(h, 0.0, 1.0, out=h)

            # Termination condition to prevent eternal stagnation
            if niters_since_improvement > 200 and (time.time() - best_time) > 180:
                break

        return (best_h, best_c5, n_points)
        
    except:
        return (best_h, best_c5, n_points)
