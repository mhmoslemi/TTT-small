import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    n_points = 256
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points, endpoint=False)

    # Initialize h based on duplicating or spreading the initial solution
    if 'initial_h_values' in globals():
        h_init = np.tile(initial_h_values, (n_points // 95, 1)).flatten()[:n_points]
    else:
        h_init = np.zeros(n_points)
        h_init[:n_points // 2] = 1.0  # 50% interval set to 1

    # Normalize h to meet the integral constraint
    h_init = h_init / np.sum(h_init) * (n_points / 2)
    h_init = np.clip(h_init, 0.0, 1.0)

    h = h_init.copy()
    start_time = time.time()
    max_iter = 1000
    alpha = 0.1  # Step size
    for _ in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Compute all correlations
        h_flat = h.flatten()
        corr = np.correlate(h_flat, 1 - h_flat, mode='full') * dx
        c5 = np.max(corr)

        # Gradient ascent on C₅
        grad = np.zeros_like(h_flat)
        for k in range(1, n_points):
            overlap = corr[n_points - 1 + k]
            if overlap == c5:
                grad += -2 * (1 - 2 * h_flat) * (1 - h_flat if k < n_points else h_flat)
                break

        # Update h
        h_flat -= alpha * grad
        h_flat = np.clip(h_flat, 0.0, 1.0)
        h_flat = h_flat / np.sum(h_flat) * (n_points / 2)
        h = h_flat.copy()

    # Final evaluation
    h_final = h
    h_final = h_final / np.sum(h_final) * (n_points / 2)
    h_final = np.clip(h_final, 0.0, 1.0)
    corr = np.correlate(h_final, 1 - h_final, mode='full') * dx
    c5_bound = np.max(corr)

    return h_final, c5_bound, n_points
