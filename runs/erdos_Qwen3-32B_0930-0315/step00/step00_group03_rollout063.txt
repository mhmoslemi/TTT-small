import numpy as np
import cvxpy as cp
from time import time

def run(seed=42, budget_s=1100, n_points=70):
    np.random.seed(seed)
    start_time = time()

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        h = h[:n_points]
        while len(h) < n_points:
            h = np.append(h, 0.0)
        if len(h) > n_points:
            h = h[:n_points]
    else:
        h = 0.5 * np.ones(n_points, dtype=np.float64)

    # Enforce initial feasibility
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-8:
        scale = (n_points / 2) / sum_h
        h *= scale
    h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    max_iter = 1000
    learning_rate = 0.01
    decay = 0.99
    line_search_factor = 0.5
    line_search_steps = 5

    for iter in range(max_iter):
        elapsed = time() - start_time
        if elapsed >= budget_s:
            break

        # Compute C₅ and identify worst shift k
        c5, k = compute_worst_shift(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

        # Compute gradient for shift k
        G = np.zeros_like(h)
        dx = 2.0 / n_points
        h_k1 = np.roll(h, -k)
        h_k2 = np.roll(h, +k)
        G = - (1 - h_k1)
        G[k:] += h_k2[:n_points - k]

        # Normalize gradient
        grad_norm = np.sqrt(np.mean(G ** 2))
        if grad_norm == 0:
            continue
        G /= grad_norm

        # Line search
        step_size = learning_rate
        for step in range(line_search_steps):
            prop_h = h - step_size * G
            prop_h = project(prop_h, n_points)
            prop_c5, _ = compute_worst_shift(prop_h)
            if prop_c5 < c5:
                h = prop_h
                break
            else:
                step_size *= line_search_factor

        # Decay learning rate
        learning_rate *= decay

    # Final check and cleanup
    h = project(h, n_points)
    return best_h, compute_c5(h), n_points


def compute_worst_shift(h):
    n_points = len(h)
    dx = 2.0 / n_points
    corr = np.zeros(2 * n_points - 1)
    for k in range(n_points):
        overlap = np.sum(h * (1 - np.roll(h, -k))) * dx
        corr[k] = overlap
        overlap = np.sum(h * (1 - np.roll(h, k))) * dx
        corr[n_points + k - 1] = overlap
    k_worst = np.argmax(corr)
    value = corr[k_worst]
    k_val = k_worst if k_worst < n_points else 2 * n_points - 1 - k_worst
    return value, k_val


def compute_c5(h):
    max_overlap, _ = compute_worst_shift(h)
    return max_overlap


def project(h, n_points):
    # Feasible region: 0 <= h[i] <= 1, sum(h) = n_points / 2
    # Projection via convex optimization to satisfy constraints
    h_proj = cp.Variable(n_points)
    constraints = [
        h_proj >= 0,
        h_proj <= 1,
        cp.sum(h_proj) == n_points / 2
    ]
    objective = cp.Minimize(cp.sum_squares(h_proj - h))
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.ECOS, max_iters=100)
    return np.clip(np.array(h_proj.value), 0.0, 1.0)
