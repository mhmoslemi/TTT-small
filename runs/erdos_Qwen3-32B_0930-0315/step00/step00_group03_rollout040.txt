import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution
import copy

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Parameters
    n_points = 70
    population_size = 50
    generations = int(1e6)
    mutation_rate = 0.15
    tournament_size = 5
    dx = 2.0 / n_points

    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        population = [initial_h.copy() for _ in range(population_size // 2)]
    else:
        population = [np.random.rand(n_points) for _ in range(population_size // 2)]

    # Fill remaining half with perturbed or random
    for _ in range(population_size // 2, population_size):
        if 'initial_h_values' in globals():
            # Perturbation preserving sum constraint
            h = initial_h + np.random.normal(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
            h = h / h.sum() * (n_points / 2)
        else:
            h = np.random.rand(n_points)
        # Enforce constraint
        h = h / h.sum() * (n_points / 2)
        population.append(h)

    best_h = None
    best_c5 = float('inf')

    def evaluate_individual(h):
        h_float = h.astype(float)
        corr = np.correlate(h_float, 1.0 - h_float, mode="full")
        return np.max(corr * dx)

    def repair_individual(h):
        h = np.clip(h, 0, 1)
        s = (n_points / 2) / np.clip(h.sum(), 1e-6, n_points)
        h = np.clip(h * s, 0, 1)
        return h

    generation = 0
    while time.time() - start_time < budget_s and generation < generations:
        generation += 1

        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_individual(h)
            fitness.append((h, c5))

        # Track best
        for h, c5 in fitness:
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        # Selection: Tournament
        new_population = []
        for _ in range(population_size // 2):
            selected = np.random.choice(fitness, tournament_size, replace=False)
            best_in_tournament = min(selected, key=lambda x: x[1])
            new_population.append(best_in_tournament[0])

        # Crossover
        for i in range(0, population_size // 2 - 1, 2):
            if i + 1 >= population_size // 2:
                break
            parent1 = new_population[i]
            parent2 = new_population[i + 1]
            cx_point = np.random.randint(1, n_points - 1)
            child1 = np.hstack([parent1[:cx_point], parent2[cx_point:]])
            child2 = np.hstack([parent2[:cx_point], parent1[cx_point:]])
            child1 = repair_individual(child1)
            child2 = repair_individual(child2)
            new_population.extend([child1, child2])

        # Mutation
        for i in range(len(new_population)):
            if np.random.rand() < mutation_rate:
                mut_h = new_population[i].copy()
                mut_indices = np.random.choice(n_points, size=int(n_points * 0.05), replace=False)
                mut_h[mut_indices] += np.random.normal(0, 0.1, len(mut_indices))
                mut_h = np.clip(mut_h, 0, 1)
                mut_h = repair_individual(mut_h)
                new_population[i] = mut_h

        population = new_population

    # Final validation
    best_h_sum = np.sum(best_h)
    assert abs(best_h_sum - n_points / 2) < 1e-6, f"Invalid total sum: {best_h_sum}"
    assert np.all((best_h >= 0) & (best_h <= 1)), f"Invalid range in h: {np.min(best_h)}-{np.max(best_h)}"

    c5_final = evaluate_individual(best_h)
    return best_h.tolist(), c5_final, n_points
