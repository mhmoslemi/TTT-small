import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else np.ones(45, dtype=np.float64) / 2.25
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = np.copy(h)
    best_c5 = compute_c5(best_h, dx)

    start_time = time.time()

    learning_rate = 0.01
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        c5_val = compute_c5(h, dx)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = np.copy(h)

        # Find k_max - the index where the max overlap is
        correlations = compute_correlations(h, dx)
        k_max = int(np.argmax(correlations))

        # Compute gradient
        gradient = np.zeros_like(h)
        for j in range(n_points):
            idx_left = (j - k_max) % n_points
            idx_right = (j + k_max) % n_points
            grad_j = 1 - h[idx_left] - h[idx_right]
            gradient[j] = grad_j

        # Gradient descent step
        h += learning_rate * gradient

        # Projection onto feasible set: clamp to [0, 1] and normalize to sum = n_points / 2
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        h *= (n_points / 2) / total

        # Optional: adjust learning rate if stalled
        if iteration % 500 == 0:
            learning_rate *= 0.95

    return best_h, best_c5, n_points

def compute_c5(h, dx):
    correlations = compute_correlations(h, dx)
    return np.max(correlations)

def compute_correlations(h, dx):
    h_flipped = 1 - h
    correlate_vals = np.correlate(h, h_flipped, mode="full")
    return correlate_vals * dx
