import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_array):
    n_points = h_array.size
    dx = 2.0 / n_points
    # Compute the overlap integrals for all lags k
    ones_h = 1.0 - h_array
    corr = np.correlate(h_array, ones_h, mode="full")
    max_corr = np.max(corr)
    c5 = max_corr * dx
    return h_array, c5

                delta = np.zeros_like(h_current)

                delta[i] = max(0, min(1, h_current[i] + delta[i]))

                if np.abs(np.sum(h_next) - n_points / 2) > 0.5:  # adjust if mass constraint violated
                    h_next = np.clip((h_next * 0.95 + 0.05) * 2, 0, 1)  # soft rescaling
                _, c5_next = evaluate_erdos_solution(h_next)
                g_current.append(c5_next)
            improvements = np.array([x < best_c5 for x in g_current])
            if np.any(improvements):
                idx = np.argmax(improvements)
                h_current[idx] = np.random.uniform(0, 1)
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points / 2) / h_current.sum()  # mass normalization
                _, c5_current = evaluate_erdos_solution(h_current)
                if c5_current < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_current
            else:
                break
        # Occasionally randomize escape from local minima
        if np.random.rand() < 0.2:
            h_random = np.random.rand(n_points)

            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current

    final_h, final_c5 = evaluate_erdos_solution(best_h)
    return (final_h, final_c5, n_points)
