import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Normalize initial_h to satisfy the integral constraint
    h = initial_h.copy()
    h = h / h.sum() * n_points / 2

    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    num_iter = 0
    learn_rate = 0.1
    min_learn_rate = 0.001
    decay_factor = 0.9995

    while True:
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 0.5:
            break

        # Compute overlap integral for all k
        corr = correlate(h, 1 - h, mode='full') * dx
        central_indices = np.arange(n_points + 1)
        overlaps = corr[corr.size - n_points - 1:corr.size - 1]
        c5 = np.max(overlaps)

        # Choose k with max overlap
        idx_max_c5 = np.argmax(overlaps)
        k = idx_max_c5

        # Construct gradient based on overlap for k
        grad = np.zeros_like(h)
        grad[:n_points - k] = -(1 - h[k:n_points])
        grad[n_points - k:] = h[:k]
        grad = grad / n_points

        # Gradient step and feasibility projection
        h = h - learn_rate * grad
        h = np.clip(h, 0.0, 1.0)
        h = h / h.sum() * n_points / 2

        # Adapt learning rate
        learn_rate = max(learn_rate * decay_factor, min_learn_rate)

        # Periodically perturb to escape local minima
        if num_iter % 50 == 0 and num_iter > 0:
            h += np.random.normal(0.0, 0.05, h.shape)
            h = np.clip(h, 0.0, 1.0)
            h = h / h.sum() * n_points / 2

        # Update best solution
        c5_new = evaluate_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

        num_iter += 1

    h_values = best_h
    return h_values.tolist(), best_c5, n_points

def evaluate_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode='full') * dx
    central_indices = np.arange(n_points + 1)
    overlaps = corr[corr.size - n_points - 1:corr.size - 1]
    c5 = np.max(overlaps)
    return c5
