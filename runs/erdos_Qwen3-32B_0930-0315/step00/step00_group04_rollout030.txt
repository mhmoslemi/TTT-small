import numpy as np
import time
from scipy.signal import correlate
from scipy.interpolate import interp1d

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h = np.array(h)
    one_minus_h = 1 - h
    corrs = correlate(h, one_minus_h, mode='full') * dx
    return np.max(corrs)

def repair_feasibility(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if abs(total - n_points / 2) > 1e-6:
        h *= (n_points / 2) / total
    return h

def initial_population(n_points, size=20, seed=42):
    np.random.seed(seed)
    pop = []
    for _ in range(size):
        h = np.random.rand(n_points)
        h = repair_feasibility(h, n_points)
        pop.append(h)
    return pop

def tournament_selection(pop, fitnesses, tournament_size=5):
    idxs = np.random.randint(0, len(pop), tournament_size)
    best_idx = idxs[np.argmin(fitnesses[idxs])]
    return pop[best_idx], fitnesses[best_idx]

def crossover(parent1, parent2):
    cut = np.random.randint(1, len(parent1) - 1)
    child = np.concatenate([parent1[:cut], parent2[cut:]])
    return child

def mutate(h, mut_rate=0.2, noise_scale=0.1):
    perturbation = np.random.randn(len(h)) * noise_scale
    mutation_mask = np.random.rand(len(h)) < mut_rate
    mutated_h = h + perturbation * mutation_mask
    return mutated_h

def gradient_update(h, learning_rate=0.01, dx=1.0):
    h = np.array(h)
    one_minus_h = 1 - h
    corrs = correlate(h, one_minus_h, mode='full') * dx
    max_idx = np.argmax(corrs)
    n = len(h)
    k = max_idx - (n - 1)
    shifted_h = np.roll(one_minus_h, -k) if k < 0 else np.roll(one_minus_h, k)
    grad = -shifted_h  # Negative to decrease overlap
    h_update = h + learning_rate * grad
    h_update = repair_feasibility(h_update, len(h))
    return h_update

def coarse_to_fine_refinement(h_coarse, target_n_points):
    f = interp1d(np.linspace(0, 2, len(h_coarse)), h_coarse, kind='cubic')
    x_fine = np.linspace(0, 2, target_n_points)
    h_fine = f(x_fine)
    return repair_feasibility(h_fine, target_n_points)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    # Initialize with coarse n_points = 14
    best_n_points = 14
    best_h = None
    best_c5 = float('inf')

    # Use initial_h_values if available, otherwise create a random one
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.resize(h_init, (1,))
        h_init_coarse = repair_feasibility(h_init[0], best_n_points)
    else:
        h_init_coarse = np.random.rand(best_n_points)
        h_init_coarse = repair_feasibility(h_init_coarse, best_n_points)

    population = initial_population(best_n_points, seed=seed)
    population_c5 = [compute_c5(p, best_n_points) for p in population]

    population[-1] = h_init_coarse
    population_c5[-1] = compute_c5(h_init_coarse, best_n_points)

    best_h = population[np.argmin(population_c5)]
    best_c5 = min(population_c5)

    ga_time_budget = budget_s * 0.6
    gd_time_budget = budget_s * 0.4

    # Genetic Algorithm Phase
    while time.time() - start_time < ga_time_budget:
        new_population = []

        while len(new_population) < len(population):
            p1, p1_c5 = tournament_selection(population, population_c5)
            p2, p2_c5 = tournament_selection(population, population_c5)

            if np.random.rand() < 0.8:
                child = crossover(p1, p2)
                if np.random.rand() < 0.2:
                    child = mutate(child)
                child = repair_feasibility(child, best_n_points)
                child_c5 = compute_c5(child, best_n_points)
                new_population.append(child)
                population_c5.append(child_c5)

        population = new_population[:len(population)]
        population_c5 = population_c5[-len(population):]

        current_best_index = np.argmin(population_c5)
        current_best_c5 = population_c5[current_best_index]

        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_index]

    # Gradient Descent Refinement
    while time.time() - start_time < budget_s:
        best_h = gradient_update(best_h)
        current_c5 = compute_c5(best_h, best_n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5

    # Coarse-to-Fine Refinement
    max_possible_res = 256
    target_n_points = min(max_possible_res, max(64, best_n_points * 2))
    best_h = coarse_to_fine_refinement(best_h, target_n_points)
    best_c5 = compute_c5(best_h, target_n_points)

    return best_h, best_c5, target_n_points
