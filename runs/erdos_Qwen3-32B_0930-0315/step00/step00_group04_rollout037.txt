import numpy as np
import cvxpy as cp
import time
import evaluate_erdos_solution as evaluator
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Scale initial_h to satisfy the constraint: integral over [0,2] is 1
    initial_sum = np.sum(initial_h) * dx
    if not np.isclose(initial_sum, 1.0, rtol=1e-6):
        initial_h = initial_h * (1.0 / initial_sum)

    h = cp.Variable(n_points)
    overlap = []
    for k in range(n_points):
        offset = k
        mask = np.arange(n_points) + offset
        in_bounds = (mask >= 0) & (mask < n_points)
        shifted = np.zeros(n_points)
        shifted[mask[in_bounds]] = 1.0 - initial_h[mask[in_bounds]]
        overlap_k = cp.sum(h * shifted)
        overlap.append(overlap_k)
    for k in range(n_points):
        offset = -k
        mask = np.arange(n_points) + offset
        in_bounds = (mask >= 0) & (mask < n_points)
        shifted = np.zeros(n_points)
        shifted[mask[in_bounds]] = 1.0 - initial_h[mask[in_bounds]]
        overlap_k = cp.sum(h * shifted)
        overlap.append(overlap_k)

    # Objective is to minimize the maximum of the overlaps
    obj = cp.Minimize(cp.max(overlap))

    # Constraints:
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2
    ]

    problem = cp.Problem(obj, constraints)
    problem.solve()

    h_values = np.clip(h.value, 0, 1)
    if not np.isclose(np.sum(h_values) * dx, 1.0, atol=1e-6):
        h_values = h_values * (1.0 / (np.sum(h_values) * dx))

    c5_bound = evaluator.evaluate_erdos_solution(h_values)

    elapsed = time.time() - start_time
    remaining = budget_s - elapsed

    # If time remains, do local optimization with CVXOPT
    if remaining > 0:
        # Use an iterative barrier method to refine solution
        h_bar = cp.Variable(n_points)
        mu = 1e3  # initial barrier parameter
        mu_decay = 0.95  # rate to decay barrier
        max_iter = 200
        it = 0
        best_h = h_values.copy()

        bar = [{'overlap': overlap}] * n_points
        bar_obj = cp.Minimize(mu * cp.sum(cp.log(h_bar) + cp.log(1 - h_bar)) + cp.max(overlap))

        while it < max_iter and time.time() - start_time < budget_s:
            prob = cp.Problem(bar_obj, constraints)
            try:
                prob.solve(solver='ECOS')
                if prob.status == 'optimal':
                    candidate = np.clip(h_bar.value, 0, 1)
                    if np.sum(candidate) * dx != 1.0:
                        candidate = candidate * (1.0 / np.sum(candidate) * n_points / 2)
                    c5_candidate = evaluator.evaluate_erdos_solution(candidate)
                    if c5_candidate < c5_bound:
                        best_h = candidate
                        c5_bound = c5_candidate
            except Exception:
                pass
            mu *= mu_decay
            it += 1

        h_values = best_h

    elapsed = time.time() - start_time
    return h_values, c5_bound, n_points
