import numpy as np
import cvxpy as cp
from time import perf_counter
import evaluate_erdos_solution

TOL = 1e-6
MAX_STEPS = 100000

def run(seed=42, budget_s=(1100.0 / 20), **kwargs):
    np.random.seed(seed)
    
    # Load or initialize
    if "initial_h_values" in globals():
        h = np.copy(initial_h_values)
        dx = 2.0 / len(h)
        n_points = len(h)
        assert np.isclose(np.sum(h), n_points / 2, atol=1e-6)
    else:
        n_points = 256
        h = np.zeros(n_points)
        idx = np.random.choice(n_points, size=int(n_points / 2), replace=False)
        h[idx] = 1.0
        dx = 2.0 / n_points

    shape = h.shape
    assert 0 <= h.min() and h.max() <= 1
    assert np.isclose(np.sum(h), n_points / 2, atol=1e-6)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    def compute_overlap_integral(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = corr * dx  # Scaling with dx
        return np.max(c5)

    t0 = perf_counter()
    step = 0
    while perf_counter() - t0 < budget_s and step < MAX_STEPS:
        step += 1
        # Find worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = abs(corr).argmax()
        over = corr[k_max] * dx  # scale by dx
        worst_idx = k_max

        # Adjust magnitudes via convex opt (Phase 1)
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(h_var * (1 - np.roll(h, worst_idx))))
        constraints = [
            cp.sum(h_var) == n_points / 2,  # Constraint 2
            0 <= h_var,
            h_var <= 1
        ]
        prob = cp.Problem(objective, constraints)
        result = prob.solve(solver=cp.CLARABEL, verbose=False)
        if result is not None:
            h_opt = h_var.value
            if h_opt is not None:
                h[:] = h_opt[:]
        
        # Renormalize to ensure sum constraint
        h /= np.sum(h) * dx  # dx = 2 / n_points
        h *= n_points / 2
        h = np.clip(h, 0, 1)

        c5 = compute_overlap_integral(h)
        if c5 < best_c5:
            best_h[:] = h.copy()
            best_c5 = c5

    c5_bound = evaluate_erdos_solution(best_h)
    return best_h, c5_bound, n_points
