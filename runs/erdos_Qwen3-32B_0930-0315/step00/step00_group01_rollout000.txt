import numpy as np
import time
import cvxpy as cp

def compute_c5(h_vec, n_points):
    h = h_vec.copy()
    dx = 2.0 / n_points
    correlate = np.correlate(h, 1 - h, mode="full")
    valid_correlate = correlate[:n_points]
    return np.max(valid_correlate) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    elapsed = 0
    tol = 1e-8  # Optimization tolerance

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 95  # Starting with an odd number for symmetry
        h = np.array([1. if i < n_points // 2 else 0. for i in range(n_points)])
    
    # Normalize to ensure the sum constraint is satisfied
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    
    # Project h onto feasible region
    def project(h):
        h_clamped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clamped)
        if h_sum == 0 or h_sum == n_points:
            return h_clamped
        return h_clamped + (n_points / 2 - h_sum) / (n_points - np.sum(1 - h_clamped)) * (1 - h_clamped)

    # Optimization with simple projected gradient descent
    grad_steps = 0
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n_points)

    while elapsed < budget_s:
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_perturbed = np.zeros_like(h)
            h_perturbed[i] = 0.01
            h1 = h + h_perturbed
            h1 = project(h1)
            h2 = h - h_perturbed
            h2 = project(h2)
            grad[i] = (compute_c5(h1, n_points) - compute_c5(h2, n_points)) / (2 * 0.01)

        # Projected gradient descent
        step_size = 0.001
        h_proposed = project(h - step_size * grad)

        # Armijo line search
        for _ in range(10):
            c5_proposed = compute_c5(h_proposed, n_points)
            if c5_proposed < best_c5:
                h = h_proposed
                best_h = h.copy()
                best_c5 = c5_proposed
                grad_steps += 1
                break
            step_size *= 0.8
            h_proposed = project(h - step_size * grad)

        elapsed = time.time() - start

    return best_h, best_c5, n_points
