import numpy as np
import cvxpy as cp
import time
import scipy.signal as ss

def run(seed=42, budget_s=1100, verbose=False):
    # Parameters
    np.random.seed(seed)
    initial_n = 70
    if 'initial_h_values' in globals():
        n_points = initial_n
        h0 = initial_h_values
    else:
        # fallback initial h
        n_points = initial_n
        h0 = np.zeros(n_points)
        h0[0::4] = 1.0
        h0[1::4] = 1.0
        h0[2::4] = 0.0
        h0[3::4] = 0.0
    h0 = h0 / h0.sum() * n_points / 2
    h0 = np.clip(h0, 0, 1)

    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0)
    t0 = time.time()
    max_time = budget_s * 0.8
    max_inner_steps = 20

    while time.time() - t0 < max_time:
        for _ in range(max_inner_steps):
            h = cp.Variable(n_points)
            obj = cp.Minimize(cp.norm(cp.conv(h, (1 - h)[::-1])[1:], p=float('inf')))
            constr = [h >= 0, h <= 1]
            constr += [cp.sum(h) == n_points / 2]
            problem = cp.Problem(obj, constr)
            problem.solve(solver=cp.OSQP, max_iter=50000)
            if h.value is None:
                h = best_h
            else:
                h = np.clip(h.value, 0, 1)
                # project to exact integral
                h_miss = cp.sum(h) - n_points / 2
                h = h - cp.Parameter(n_points, value=h_miss / n_points)
                h = cp.clip(h, 0, 1)
                h = h.value

                c5 = evaluate_erdos_solution(h)
                if c5 < best_c5:
                    best_h = h
                    best_c5 = c5
                    if verbose:
                        print(f"New best C5: {best_c5:.6f}")
                if time.time() - t0 > max_time:
                    break

    return (best_h, best_c5, n_points)
