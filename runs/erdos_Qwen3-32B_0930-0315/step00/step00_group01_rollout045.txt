import numpy as np
import time

from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    n_points = 47  # Start with initial_h_values

    if 'initial_h_values' in globals():
        h = np.copy(initial_h_values)
    else:
        h = np.random.rand(n_points)  # If no initial_h_values
        h = np.clip(h, 0, 1)  # Clamp to [0, 1]
        h *= n_points / 2 / h.sum()  # Normalize

    dx = 2.0 / n_points

    # Ensure sum constraint is met accurately
    h = np.clip(h, 0, 1)
    h *= n_points / 2 / h.sum()

    learning_rate = 0.01
    patience = 3
    no_improve_count = 0

    while time.time() - start_time < budget_s - 5:
        overlap = correlate(h, 1 - h, mode='full') * dx
        c5 = np.max(overlap)

        if c5 < best_c5:
            best_c5 = c5
            best_h = np.copy(h)
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count >= patience:
            learning_rate = max(learning_rate * 0.8, 1e-4)
            no_improve_count = 0

        # Find the worst k
        max_k = np.argmax(overlap)

        # Build gradient
        grad = np.zeros_like(h)
        valid_indices = np.arange(0, n_points - max_k)
        grad[valid_indices] += (1 - h[valid_indices + max_k])
        grad[valid_indices] -= h[valid_indices]

        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        # Project back to constraint: sum(h) == n_points / 2
        s = h.sum()
        if s != 0:
            h *= (n_points / 2) / s

        if abs(h.sum() - n_points / 2) > 1e-8:
            raise RuntimeError("Mass constraint violated")

    return (best_h.tolist(), best_c5, n_points)
