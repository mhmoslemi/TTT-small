import cvxpy as cp
import time

def evaluate_erdos_c5(h):
    dx = 2.0 / len(h)
    not_h = 1 - h
    corr = np.correlate(h, not_h, mode="full")
    c5 = np.max(corr) * dx
    return c5

    start_time = time.time()
    max_iters = 20  # Conservatively assume 50s per iteration
    early_stop_threshold = 1e-5

    for iter in range(max_iters):
        if time.time() - start_time > budget_s:
            break

        # Compute all cross terms using np.correlate
        dx = 2.0 / n_points
        not_h = 1 - h
        corr = np.correlate(h, not_h, mode="full")
        c5 = np.max(corr) * dx
        k_worst = int(np.argmax(corr))

        # Use CVXPY to minimize the worst overlap
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(corr[k_worst] * dx)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
        ]
        prob = cp.Problem(objective, constraints)

        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            h = np.clip(h_var.value, 0, 1).ravel()

            # Project back to ensure sum(h) == n_points / 2
            h_sum = np.sum(h)
            if abs(h_sum - n_points / 2) > 1e-6:
                scale = (n_points / 2) / h_sum
                h = np.clip(h * scale, 0, 1)

        except Exception:
            # Fall back to best solution and break if solver fails
            h = best_h.copy()
            break

        # Track progress
        current_c5 = evaluate_erdos_c5(h)
        if current_c5 < best_c5 - 1e-8:
            best_c5 = current_c5
            best_h = h.copy()
        elif best_c5 - current_c5 < early_stop_threshold:
            break  # No meaningful improvement

    return (best_h.tolist(), best_c5, n_points)
