import numpy as np
import time
import cvxpy as cp
import scipy.signal


def evaluate_erdos_c5(h, n_points):
    # Pad h and 1 - h to allow for full convolution
    dx = 2.0 / n_points
    h_padded = np.concatenate((h, np.zeros(n_points - 1)))
    one_minus_h = 1.0 - h
    one_minus_h_padded = np.concatenate((one_minus_h, np.zeros(n_points - 1)))
    # Full convolution: C5 = max(np.correlate(h, 1-h, mode="full")) * dx
    conv = scipy.signal.fftconvolve(h_padded, one_minus_h_padded, mode="full")
    C5 = np.max(conv[n_points - 1: 2 * n_points - 1]) * dx
    return C5


def randomized_rounding(h, n_points):
    # Adjust h to satisfy sum(h[i]) == n_points / 2 with values in [0, 1]
    sum_h = np.sum(h)
    scale = (n_points / 2.) / sum_h
    h_scaled = np.clip(h * scale, 0., 1.)
    # Correct any violations in sum via randomized rounding
    diff = (n_points / 2.) - np.sum(h_scaled)
    indices = np.flatnonzero(h_scaled > 0)  # Avoid division by 0
    if indices.size > 0:
        weights = h_scaled[indices]
        weights = weights / np.sum(weights)
        num_increments = int(np.round(abs(diff)))
        for _ in range(num_increments):
            i = np.random.choice(indices, p=weights)
            change = np.sign(diff)
            h_scaled[i] = min(max(h_scaled[i] + abs(change), 0.), 1.)
    return h_scaled


def run(seed=42, budget_s=1100, sample_rate=45):
    np.random.seed(seed)
    start_time = time.time()
    dx = 2.0 / sample_rate
    budget_remaining = budget_s

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(sample_rate)
        h /= np.sum(h) * dx
        h *= 2.0 / sample_rate

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_c5(h, sample_rate)

    max_evals = 3000
    progress_threshold = 1e-4
    no_progress_counter = 0
    eval_count = 0

    # Begin search
    while time.time() - start_time < budget_s and no_progress_counter < 100:
        indices = np.random.permutation(sample_rate)
        for i in indices:
            if eval_count >= max_evals or time.time() - start_time >= budget_s:
                break
            # Test a small perturbation at index i
            for trial_h in [0.0, 0.25, 0.5, 0.75, 1.0]:
                candidate = np.copy(best_h)
                candidate[i] = trial_h
                candidate = randomized_rounding(candidate, sample_rate)
                c5 = evaluate_erdos_c5(candidate, sample_rate)
                if c5 < best_c5 - 1e-5:
                    best_h[:] = candidate
                    best_c5 = c5
                    no_progress_counter = 0
                    eval_count += 1
                    break
            else:
                no_progress_counter += 1
                continue
            break

    return best_h.tolist(), best_c5, sample_rate
