```python
import numpy as np
import time
import math
import random
import cvxpy as cp
from scipy.signal import correlate
from numpy.random import default_rng

rng = default_rng(seed=42)

def evaluate_candidate(h, n_points):
    dx = 2.0 / n_points
    one_minus_h = 1 - h
    corr = correlate(one_minus_h, h, mode="full")
    dx_corr = corr * dx
    return np.max(dx_corr)

def project_to_feasible(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if not np.isclose(total, n_points / 2, atol=1e-6):
        scale = (n_points / 2) / total if total != 0 else 1.0
        h = np.clip(h * scale, 0.0, 1.0)
    return h

def tournament_selection(population, c5_values, tournament_size):
    indices = np.random.choice(len(population), tournament_size)
    fitnesses = c5_values[indices]
    best_idx = indices[np.argmin(fitnesses)]
    return population[best_idx]

def crossover(parent1, parent2, crossover_prob=0.7):
    if np.random.rand() > crossover_prob:
        return parent1
    n = len(parent1)
    c = np.random.randint(1, n)
    child = np.concatenate((parent1[:c], parent2[c:]))
    return child

def mutate(h, n_points, temp=1.0, mutation_prob=0.02):
    if np.random.rand() > mutation_prob:
        return h
    n = len(h)
    mask = rng.integers(0, n, int(n * mutation_prob))
    step = 0.01 * temp * (2 * rng.random(len(mask)) - 1)
    h_mut = h.copy()
    h_mut[mask] += step
    h_mut = project_to_feasible(h_mut, n_points)
    return h_mut

def compute_temperature(t0, cooling_rate, gen):
    return t0 * (cooling_rate ** gen)

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():
        n_points = len(globals()['initial_h_values'])
        h_start = globals()['initial_h_values'].copy()
    else:
        n_points = 45
        h_start = 0.5 * np.ones(n_points)

    population_size = 75
    n_gen = 0
    max_gen = 1000
    t0 = 100
    cooling_rate = 0.95
    tournament_size = int(0.1 * population_size)

    population = []
    for i in range(population_size):
        if i == 0 and 'initial_h_values' in globals():
            population.append(project_to_feasible(h_start.copy(), n_points))
        else:
            h = rng.uniform(0.0, 1.0, n_points).copy()
            h = project_to_feasible(h, n_points)
            population.append(h)

    best_h = h_start.copy()
    best_c5 = evaluate_candidate(best_h, n_points)
    start_time = time.time()

    while True:
        if time.time() - start_time > budget_s:
            break
        c5_values = np.array([evaluate_candidate(p, n_points) for p in population])
        idx = np.argmin(c5_values)
        current_best = population[idx].copy()
        current_c5 = c5_values[idx]

        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5

        next_population = [current_best]
        for _ in range(population_size - 1):
            parent1 = tournament_selection(population, c5_values, tournament_size)
            parent2 = tournament_selection(population, c5_values, tournament_size)
            child = crossover(parent1, parent2)
            child = mutate(child, n_points, temp=cooling_rate**n_gen)
            next_population.append(child)

        if n_gen % 50 == 0 and n_gen > 0:
            parent = population[np.argmin(c5_values)].copy()
            parent_c5 = evaluate_candidate(parent, n_points)
            child = mutate(parent, n_points, temp=compute_temperature(t0, cooling_rate, n_gen))
            child_c5 = evaluate_candidate(child, n_points)
            if child_c5 < parent_c5:
                parent[:] = child
            else:
                delta = child_c5 - parent_c5
                T = compute_temperature(t0, cooling_rate, n_gen)
                if np.random.rand() < math.exp(-delta / T):
                    parent[:] = child

        population = next_population
        n_gen += 1

    return best_h, best_c5, n_points
```
