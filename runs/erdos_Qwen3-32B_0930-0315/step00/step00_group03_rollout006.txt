import numpy as np
import cvxpy as cp
import time
from collections import deque

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    dx = 2.0 / 70
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.zeros(70) + 1/70 * 35
    n_points = 70
    h = np.clip(initial_h, 0.0, 1.0)
    h_sum = np.sum(h)
    h = h / h_sum * (n_points / 2)
    h_best = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    no_improvement_counter = 0
    max_no_improvement = 100
    learning_rate = 0.01
    
    while time.time() - start_time < budget_s:
        # Get shift that gives the worst (maximum) overlap
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        positive_indices = np.arange(n_points)
        worst_k = n_points - 1 - np.argmin(positive_indices[positive_indices < overlaps.shape[0]] * overlaps[-n_points:])
        
        # Compute gradient for the worst shift k
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + worst_k < n_points:
                grad[i] += (1 - h[i + worst_k])
            if i - worst_k >= 0:
                grad[i] -= h[i - worst_k]
        
        # Update h
        h_candidate = h - learning_rate * grad
        
        # Project to feasible region
        h_proj = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum_squares(h_proj - h_candidate))
        constraints = [
            cp.sum(h_proj) == n_points / 2,
            0 <= h_proj,
            h_proj <= 1
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.GUROBI, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h = h_proj.value
        else:
            h = h
            learning_rate *= 0.5
        
        # Clamp small numeric errors
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        h = h / h_sum * (n_points / 2)
        
        # Check if we improved
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            h_best = h.copy()
            no_improvement_counter = 0
            learning_rate = min(learning_rate * 1.1, 0.1)
        else:
            learning_rate = max(learning_rate * 0.98, 0.001)
            no_improvement_counter += 1
        
        if no_improvement_counter > max_no_improvement:
            break
    
    return (h_best.tolist(), best_c5, n_points)
