import numpy as np
import cvxpy as cp

from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    h = h_values
    n_points = len(h)
    dx = 2.0 / n_points
    a = correlate(h, [1] * n_points, mode="same")
    b = correlate(h, 1 - h, mode="same")
    c5_bound = np.max(b)
    return h, c5_bound

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    j = 0

    def compute_c5_and_gradient(h):
        n_points = len(h)
        dx = 2.0 / n_points

        # Compute for all k
        overlap = correlate(h, 1 - h, mode="full") * dx

        k_values = np.arange(n_points)
        valid_overlap = overlap[n_points - 1 : 2 * n_points - 1]  # only k ∈ [0, n-1]
        k_max = n_points - 1 + np.argmax(valid_overlap)

        h_cpy = h[k_max - n_points + 1 : k_max + 1]
        grad = np.zeros_like(h)
        grad[k_max - n_points + 1 : k_max] += 1 - h_cpy[1:]
        grad[k_max - n_points + 1 : k_max] -= 0  # we only consider overlaps at k
        grad[k_max - n_points + 2 : k_max + 1] += -h_cpy[:-1]

        return valid_overlap.max(), grad

    def project(h):
        # Clamp values to [0,1]
        h_clamped = np.clip(h, 0.0, 1.0)
        # Rescale to satisfy sum constraint
        total = h_clamped.sum()
        if abs(total - n_points / 2) > 1e-6:
            h_clamped = h_clamped * (n_points / 2) / total
        return h_clamped

    # Use initial_h_values if available
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = np.array(initial_h_values)
        h = (h / h.sum()) * (len(h) / 2)
        h = np.clip(h, 0.0, 1.0)
        n_points = len(h)
    else:
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points // 2] = 1.0

    h = project(h)
    current_c5, _ = compute_c5_and_gradient(h)

    learning_rate = 0.1
    decay_rate = 0.95
    best_h = h.copy()
    best_c5 = current_c5

    while time.time() - start_time < budget_s:
        current_c5, grad = compute_c5_and_gradient(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Update
        h -= learning_rate * grad
        h = project(h)
        # Decay learning rate
        learning_rate *= decay_rate

        # Perturbation to escape local minima
        if j % 50 == 0 and j > 0:
            eps = 0.01
            h_perturb = h + eps * np.random.randn(n_points)
            h_perturb = project(h_perturb)
            h = h_perturb

        j += 1

    return best_h, best_c5, len(best_h)
