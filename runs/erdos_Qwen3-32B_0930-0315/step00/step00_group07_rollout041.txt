import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h_values):
    import numpy as np
    h = np.array(h_values)
    n_points = h.shape[0]
    dx = 2.0 / n_points
    v1_h = h * dx
    v1_1mh = (1 - h) * dx

    # Compute (v1_h * v1_1mh) of cross-correlation at all lags
    corr = np.correlate(v1_h, v1_1mh, mode='full')
    return float(np.max(corr))

    pass

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    h = initial_h_values()
    n_points = len(h)

    # Ensure sum(h) = n_points / 2
    h_sum = np.sum(h)
    if not np.isclose(h_sum, n_points / 2, atol=1e-8):
        h = (h / h_sum) * (n_points / 2)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    max_iter = 500
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break

        # Compute overlap and find worst lag
        dx = 2.0 / n_points
        h_vals = h * dx
        one_minus_h_vals = (1 - h) * dx
        corr = np.correlate(h_vals, one_minus_h_vals, mode='full')
        worst_k = np.argmax(corr)
        worst_c5 = corr[worst_k]

        if worst_c5 < best_c5:
            best_c5 = worst_c5
            best_h = h.copy()
            continue

        # Compute gradient of S_k at current h
        grad = np.zeros(n_points, dtype=np.float64)
        k = worst_k
        for i in range(n_points):
            j_p = (i + k) % n_points
            j_n = (i - k) % n_points
            grad[i] = (1 - h[j_p] - h[j_n]) * dx

        # LP variables and constraints
        delta = cp.Variable(n_points)
        constraints = [
            0 <= h + delta,
            h + delta <= 1,
            cp.sum(delta) == 0,
        ]
        objective = cp.Minimize(cp.sum(cp.multiply(grad, delta)))
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.ECOS, verbose=False)

        if problem.status != 'optimal':
            continue  # Skip if infeasible or unconverged

        step = -delta.value
        h = h + step

        # Ensure sum(h) is exactly n_points / 2
        h = h - (np.sum(h) - (n_points / 2)) / n_points

        # Ensure [0,1] bounds
        h[:] = np.clip(h, 0, 1)

        curr_c5 = evaluate_erdos_solution(h)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()

    return best_h, best_c5, n_points
