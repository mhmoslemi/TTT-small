import numpy as np
import time
import random

def run(seed=42, budget_s=1100, pop_size=75, n_points=45, max_gen=1000, temp_start=100, mutation_rate=0.05, crossover_prob=0.7, cool_rate=0.95, cool_every=50):
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()

    if 'initial_h_values' in globals():
        best_h = initial_h_values
    else:
        h = np.zeros(n_points)
        h[:n_points // 2 + 1] = 1.0
        h = np.sort(h)
        h = np.clip(h, 0, 1)
        subset = np.random.choice(n_points, size=n_points // 2, replace=False)
        h[subset] = 1.0
        h[~subset] = 0.0
        best_h = h

    population = [np.sort(np.random.permutation(best_h)) for _ in range(pop_size // 2)]
    population += [best_h for _ in range(pop_size // 2)]

    best_c5 = float('inf')
    best_h = None

    def evaluate_c5(h_vec):
        h = np.sort(h_vec)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        c5 = evaluate_erdos_solution({"h_values": h, "n_points": n_points})
        return c5, h

    for gen in range(max_gen):
        genotype_population = []
        for indiv in population:
            genotype_population.append(indiv / np.sum(indiv) * (n_points / 2))

        population = []
        for _ in range(pop_size):
            idx1 = np.argmin([evaluate_c5(g)[0] for g in genotype_population])
            p1 = genotype_population.pop(idx1)
            if genotype_population:
                idx2 = random.randint(0, len(genotype_population) - 1)
                p2 = genotype_population[idx2]
            else:
                p2 = p1

            if random.random() < crossover_prob and genotype_population:
                cross_point = random.randint(1, n_points - 1)
                child = np.concatenate([p1[:cross_point], p2[cross_point:]])
            else:
                child = p1

            # Mutation
            if random.random() < mutation_rate:
                mut_indices = random.choices(range(n_points), k=int(mutation_rate * n_points))
                child[mut_indices] += np.random.uniform(-0.1, 0.1, size=len(mut_indices))
                child = np.clip(child, 0, 1)

            child = np.clip(child, 0, 1)
            child /= np.sum(child) * (n_points / 2)
            population.append(child)

        if gen % cool_every == 0 and gen > 0:
            temp = temp_start * (cool_rate ** (gen // cool_every))
            c5_values = [evaluate_c5(p)[0] for p in population]
            best_idx = np.argmin(c5_values)
            best_p = population[best_idx]

            new_p = best_p + np.random.uniform(-0.1, 0.1, size=n_points)
            new_p = np.clip(new_p, 0, 1)
            new_p /= np.sum(new_p) * (n_points / 2)
            new_c5, _ = evaluate_c5(new_p)

            if new_c5 < c5_values[best_idx]:
                population[best_idx] = new_p
            else:
                if np.random.rand() < np.exp(-(c5_values[best_idx] - new_c5) / max(temp, 1e-6)):
                    population[best_idx] = new_p

        # Update best
        c5_values = [evaluate_c5(p)[0] for p in population]
        current_idx = np.argmin(c5_values)
        current_c5, current_h = evaluate_c5(population[current_idx])
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5, n_points
