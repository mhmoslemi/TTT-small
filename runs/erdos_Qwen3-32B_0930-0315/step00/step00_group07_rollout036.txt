import numpy as np
import cvxpy as cp
import time
import warnings


def run(seed=42, budget_s=1100, *args, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 80  # Use the provided resolution from initial_h_values
    dx = 2.0 / n_points

    # Use the provided initial_h_values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default: uniform distribution (if initial_h_values not available)
        h = np.full(n_points, 1 / n_points * 2)

    # Verify constraint: sum(h) == n_points / 2
    def apply_sum_constraint(h):
        expected = n_points / 2
        actual = np.sum(h)
        if abs(actual - expected) > 1e-6:
            h *= expected / actual
        return h

    h = apply_sum_constraint(h)
    h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = float('inf')
    verbose_interval = 10
    iteration = 0

    while time.time() - start_time < budget_s - 1:
        corr = np.correlate(h, 1 - h, mode='full')
        # Correlation result includes all shifts from -n+1 to n-1
        idx = np.arange(len(corr))
        valid_shifts = idx[np.logical_and(idx >= n_points, idx < 2 * n_points - n_points)]
        valid_corr = corr[valid_shifts]
        worst_k_idx = int(n_points / 2 - valid_corr.argmin())

        # Convex optimization step: minimize at worst_k_idx
        h_vec = cp.Variable(n_points)
        objective = cp.Minimize(
            cp.sum(
                h_vec * (1 - cp.roll(h_vec, shift=worst_k_idx))
            ) * dx
        )
        constraints = [
            0 <= h_vec,
            h_vec <= 1,
            cp.sum(h_vec) == n_points / 2,
        ]
        problem = cp.Problem(objective, constraints)
        try:
            problem.solve(solver=cp.MOSEK, verbose=False, mosek_params={
                'MSK_DPAR_INTPNT_TOL_MFEAS': 1e-8,
                'MSK_DPAR_INTPNT_TOL_DFEAS': 1e-8,
                'MSK_DPAR_INTPNT_TOL_REL_GAP': 1e-8,
                'MSK_IPAR_INTPNT_OUTPUT_LEVEL': 0,
            })
            if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                warnings.warn("CP problem not OPTIMAL. Using previous best.")
                continue
            h_opt = h_vec.value
        except Exception as e:
            warnings.warn("CP solve failed. Using previous best.")
            continue

        h = h_opt
        h = apply_sum_constraint(h)
        h = np.clip(h, 0.0, 1.0)

        corr_best = np.correlate(h, 1 - h, mode='full')
        idx = np.arange(len(corr_best))
        valid_shifts_best = idx[np.logical_and(idx >= n_points, idx < 2 * n_points - n_points)]
        valid_corr_best = corr_best[valid_shifts_best]
        c5 = np.max(valid_corr_best * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            print(f"Iteration {iteration}: New best C5 = {best_c5:.6f}")
        iteration += 1

        # Termination criteria
        if iteration % verbose_interval == 0:
            elapsed = time.time() - start_time
            print(f"Iteration {iteration}, elapsed {elapsed:.2f}s, best C5 = {best_c5:.6f}")

        # If sufficient progress is absent, terminate
        if best_c5 < 1e-8:
            break

    # Final evaluation and validation
    best_h = apply_sum_constraint(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    idx = np.arange(len(corr_final))
    valid_shifts_final = idx[np.logical_and(idx >= n_points, idx < 2 * n_points - n_points)]
    c5_values = corr_final[valid_shifts_final] * dx
    final_c5 = np.max(c5_values)

    return (best_h, final_c5, n_points)
