import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Start from the pre-imported initial_h_values provided by the parent
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    assert np.all(h >= 0) and np.all(h <= 1), "Initial h not in [0, 1]"
    assert np.allclose(np.sum(h) * dx, 1.0), "Integral of initial h is not 1"

    def evaluate_c5(h_vec):
        # Compute overlap integrals at all lags via convolution
        overlap = correlate(h_vec, 1.0 - h_vec, mode="full")
        return np.max(overlap * dx)

    best_c5 = evaluate_c5(h)
    best_h = h.copy()

    print(f"Initial C5: {best_c5:.7f}")

    start_time = time.time()

    # Set up for projected gradient descent
    h_var = cp.Variable(n_points)
    constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2]
    obj = cp.Minimize(cp.Max(cp.Conv(conv_matrix(h_var, n_points)) * dx))
    problem = cp.Problem(obj, constraints)

    steps = 0
    time_budget = budget_s - 20  # Leave 20 sec buffer to finish

    while time.time() - start_time < time_budget:
        # Reconstruct objective
        obj = cp.Minimize(cp.Max(cp.Conv(conv_matrix(h_var, n_points)) * dx))
        problem = cp.Problem(obj, constraints)

        try:
            problem.solve(solver=cp.SCIP, verbose=False)
        except:
            pass

        if h_var.value is not None:
            new_h = np.clip(np.ravel(h_var.value), 0, 1)
            new_c5 = evaluate_c5(new_h)

            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h

        h = best_h.copy()
        steps += 1

        if steps % 20 == 0:
            print(f"Step: {steps}, best C5: {best_c5:.7f}, time: {time.time()-start_time:.1f}s")

    # Final clean-up projection
    best_h = np.clip(best_h, 0, 1)
    best_h *= n_points / 2 / best_h.sum()

    best_c5 = evaluate_c5(best_h)
    print(f"\nFinal C5 bound: {best_c5:.8f}")
    return best_h, best_c5, n_points


def conv_matrix(h, n_points):
    # Helper matrix to support CVXPY convolution objective
    import scipy.sparse as sp

    # Construct sparse convolution matrix using toeplitz format
    h_vals = cp.Constant(h)
    matrix = sp.lil_matrix((2*n_points - 1, n_points), dtype=np.float64)

    k = 0
    for i in range(n_points):
        if i < n_points - k:
            matrix[k, i] = h_vals[i + k]

    for k in range(1, n_points - 1):
        for i in range(n_points):
            if i < n_points - k:
                matrix[k, i] = h_vals[i + k]

    for k in range(n_points - 1, 2*n_points - 1):
        for i in range(n_points):
            if i < n_points and i + k < n_points:
                matrix[k, i] = h_vals[i + k]

    return matrix
