import numpy as np
import time
import cvxpy as cp
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    h = np.copy(initial_h_values)
    n_points = h.size
    dx = 2.0 / n_points

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h)

    alpha = 0.1
    it = 0
    update_interval = 10
    perturb_interval = 50
    cool_alpha = 0.5

    while time.time() - t_start < budget_s:
        it += 1

        # Evaluate h and get C5
        c5 = evaluate_erdos_solution(h)

        if c5 < best_c5:
            best_h = np.copy(h)
            best_c5 = c5

        # Cross-correlation for gradients
        h_vals = h
        inv_h_vals = 1 - h_vals
        corr = np.correlate(h_vals, inv_h_vals, mode="full")
        corr *= dx
        k_max = np.argmax(corr)

        # Gradient step for h
        grad = np.zeros_like(h_vals)
        for i in range(n_points):
            if 0 <= i + k_max < n_points:
                grad[i] += inv_h_vals[i + k_max]
            if 0 <= i - k_max < n_points:
                grad[i] += inv_h_vals[i - k_max]

        h_vals -= alpha * grad
        h_vals = np.clip(h_vals, 0, 1)

        # Project to feasible set
        sum_h = np.sum(h_vals)
        scale = (n_points / 2.0) / sum_h
        h_vals *= scale
        h_vals = np.clip(h_vals, 0, 1)

        h = h_vals

        # Perturbation schedule
        if it % perturb_interval == 0:
            noise = np.random.normal(0, 0.01, size=h.shape)
            h_vals += noise
            h_vals = np.clip(h_vals, 0, 1)
            sum_h = np.sum(h_vals)
            scale = (n_points / 2.0) / sum_h
            h_vals *= scale
            h_vals = np.clip(h_vals, 0, 1)
            h = h_vals

        # Learning rate decrease
        if it % 100 == 0:
            alpha *= cool_alpha

    return (best_h, best_c5, n_points)
