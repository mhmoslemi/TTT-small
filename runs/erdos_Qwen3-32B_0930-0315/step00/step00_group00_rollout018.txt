import numpy as np
import time

# Helper: compute C5 given h
def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h_padded = np.concatenate([h, np.zeros(n_points - 1)])
    one_minus_h_padded = np.concatenate([1 - h, np.zeros(n_points - 1)])
    corr = np.correlate(h_padded, one_minus_h_padded, mode='full')
    c5 = dx * np.max(corr[n_points - 1 : 2 * n_points - 1])
    return c5

# Projection: enforce 0 ≤ h[i] ≤ 1 and sum(h) = n / 2
def project(h, n_points):
    h = np.clip(h, 0, 1)
    diff = np.sum(h) - n_points / 2
    if abs(diff) > 1e-7:
        weights = np.random.rand(*h.shape)
        indices = np.argsort(weights)
        num_adjust = int(abs(diff + 1e-6))
        step = np.sign(-diff) / num_adjust if num_adjust > 0 else 0
        h[indices[:num_adjust]] += step
        h = np.clip(h, 0, 1)
    return h

# Gradient descent step for a given h, avoiding syntax errors
def gradient_step(h, n_points):
    dx = 2.0 / n_points
    h_padded = np.concatenate([h, np.zeros(n_points - 1)])
    one_minus_h_padded = np.concatenate([1 - h, np.zeros(n_points - 1)])
    corr = np.correlate(h_padded, one_minus_h_padded, mode="full")
    max_idx = np.argmax(corr[n_points - 1 : 2 * n_points - 1])
    k_star = max_idx + n_points - 1
    grad = np.zeros(n_points)
    for i in range(n_points - max_idx):
        grad[i] += h[i] - h[i + k_star]
        grad[i + k_star] += -h[i] + h[i + k_star]
    grad *= -dx
    return grad
