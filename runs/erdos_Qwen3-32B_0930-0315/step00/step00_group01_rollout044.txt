import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Normalize to satisfy the constraint sum(h) = n_points / 2
    def project_constraints(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        h *= (n_points / 2) / total
        return h
    
    h = project_constraints(h)

    lr = 0.02
    patience = 200
    no_improvement = 0
    
    while time.time() - start_time < budget_s:
        dx = 2.0 / n_points
        ones = np.ones_like(h)
        
        # Compute all overlaps via cross-correlation
        overlap = np.correlate(h, ones - h, mode="full")
        overlap = overlap * dx
        
        # Find the worst overlap
        k_max = np.argmax(overlap)
        
        # Compute gradient only from the worst overlap
        grad = np.zeros_like(h)
        if k_max < n_points:
            grad[k_max:] -= h[:-k_max]
            grad[:-k_max] += (1 - h[k_max:])
        else:
            k_max = k_max - n_points
            grad[:-k_max] -= h[k_max:]
            grad[k_max:] += (1 - h[:-k_max])

        h = h - lr * grad
        h = project_constraints(h)
        
        # Compute current C5
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1

        # Adapt learning rate and perturb if no progress
        if no_improvement >= patience:
            lr *= 0.9
            noise = np.random.randn(n_points) * 0.01
            h = project_constraints(h + noise)
            no_improvement = 0

    return (best_h.tolist(), best_c5, n_points)
