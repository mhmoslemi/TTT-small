import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, verbose=False):
    np.random.seed(seed)
    start = time.perf_counter()
    
    try:
        # Use downsampled initial construction if available
        if 'initial_h_values' in globals() and len(initial_h_values) > 0:
            n_start = 14  # Coarse starting resolution
            h = initial_h_values[:n_start]
            h = h / h.sum() * (n_start / 2)
        else:
            # fallback: random initialization if no parent
            n_start = 14
            h = np.random.rand(n_start)
            h = h / h.sum() * (n_start / 2)
    except:
        n_start = 14
        h = np.random.rand(n_start)
        h = h / np.sum(h) * (n_start / 2)
    
    best_h = h.copy()
    best_C = compute_erdos_overlap(h)
    
    for n in [14, 28, 56]:
        if time.perf_counter() - start > budget_s:
            break
            
        # Resample / interpolate existing solution at new resolution
        new_h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(h)), h)
        new_h = new_h / new_h.sum() * (n / 2)
        
        # Simulated annealing
        T = 1.0
        n_sa_steps = 500
        h_sa = new_h.copy()
        current_C = compute_erdos_overlap(h_sa)
        
        for step in range(n_sa_steps):
            if time.perf_counter() - start > budget_s:
                break
            
            # Perturb
            mask = np.random.rand(n) < 0.1
            h_candidate = h_sa.copy()
            h_candidate[mask] += np.random.normal(0, 0.1, size=mask.sum())
            h_candidate = repair_feasibility(h_candidate, n)
            
            # Evaluate
            C_new = compute_erdos_overlap(h_candidate)
            delta = C_new - current_C
            
            # Accept condition
            if delta < 0 or np.exp(-delta / T) > np.random.rand():
                h_sa = h_candidate
                current_C = C_new
            
            # Track best
            if current_C < best_C:
                best_h = h_sa.copy()
                best_C = current_C
            
            # Anneal
            T = max(1e-6, T * 0.99)
        
        # Gradient descent refinement
        h_gd = h_sa.copy()
        dx = 2.0 / n
        n_gd_steps = 100
        
        for step in range(n_gd_steps):
            if time.perf_counter() - start > budget_s:
                break
            
            C_vals = np.correlate(h_gd, 1 - h_gd, mode="full") * dx
            k_max = np.argmax(C_vals)
            
            # Align gradient with the maximizing shift
            shift = k_max - (n - 1)
            h_shifted = np.roll(h_gd, shift % n)
            gradient = 1 - h_shifted[-n:]
            
            # Update
            h_gd -= 0.001 * gradient
            
            # Feasibility fix
            h_gd = np.clip(h_gd, 0, 1)
            h_gd = h_gd / h_gd.sum() * (n / 2)
            
            # Track global best
            C_new = compute_erdos_overlap(h_gd)
            if C_new < best_C:
                best_h = h_gd.copy()
                best_C = C_new
    
    # Final validation
    final_C = compute_erdos_overlap(best_h)
    return (best_h, final_C, len(best_h))

def compute_erdos_overlap(h):
    dx = 2.0 / len(h)
    C_vals = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(C_vals)

def repair_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    return h
