import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    n_points = 256
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        # If no initial_h_values, construct random baseline
        np.random.seed(seed)
        h = np.random.rand(n_points)
        h *= (n_points / 2) / np.sum(h)

    h = h.astype(np.float64)
    best_h = h.copy()
    best_c5 = float('inf')

    # Projection step to ensure feasibility constraints
    def project(h):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if s != 0:
            h *= (n_points / 2) / s
        return h

    # Objective evaluation
    def compute_c5(h):
        h_1m = 1 - h
        corr = np.correlate(h, h_1m, mode='full') * dx
        c5 = np.max(corr)
        return c5

    # Gradient ascent for AC(k) max at k = k_max
    def gradient_at_k(h, k):
        h_1m = 1 - h
        k = k % n_points
        grad = np.roll(h_1m, -k) * dx
        grad = grad - 2 * h * np.roll(h_1m, -k) * dx
        return grad

    max_iter = 1100 * 1000  # ~1e6 iterations
    lr = 0.1
    iters = 0
    patience = 0
    while time.time() - start_time < budget_s:
        iters += 1
        # Compute corr
        h_1m = 1.0 - h
        corr = np.correlate(h, h_1m, mode='full') * dx
        c5 = np.max(corr)
        k_max = np.argmax(corr)

        # Compute ascent gradient at k_max
        grad = gradient_at_k(h, k_max)

        # Update step
        h -= lr * grad

        # Project back to feasible set
        h = project(h)

        # Check improvement
        h_c5 = compute_c5(h)
        if h_c5 < best_c5:
            best_c5 = h_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            lr *= 0.99**patience  # Learning rate decay

        # Stop if converged
        if patience > 100 and lr < 1e-4:
            break

    # Return the best so far
    return best_h, best_c5, n_points
