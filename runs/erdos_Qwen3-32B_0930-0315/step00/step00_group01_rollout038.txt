import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Compute the overlap constant C₅"""
    n_points = len(h)
    max_overlap = 0.0
    for k in range(-n_points + 1, n_points):
        corr = correlate(h, 1 - h, mode='full')
        overlap = corr[k] * dx
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap

def run(seed=42, budget_s=1100, verbose=False):
    np.random.seed(seed)
    start_time = time.time()

    # Use initial_h_values as the default construction if available
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_np = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        n_points = 94
        h_np = np.random.rand(n_points)
        h_np = h_np / h_np.sum() * n_points / 2
        h_np = np.clip(h_np, 0.0, 1.0)

    dx = 2.0 / n_points
    h_opt = h_np.copy()
    c5_opt = evaluate_c5(h_opt, dx)
    best_time = time.time() - start_time

    # SDP Optimization
    h_var = cp.Variable(n_points, nonneg=True)
    t = cp.Variable()

    constraints = [
        h_var <= 1,
        h_var >= 0,
        cp.sum(h_var) == n_points / 2.0
    ]

    overlap_constraints = []
    for k in range(n_points):
        a = h_var
        b = (1 - cp.roll(h_var, k)) * cp.Symmetric(n_points) # Exploit symmetry under approximation
        overlap = cp.sum(cp.multiply(a, b)) * dx <= t
        overlap_constraints.append(overlap)

    objective = cp.Minimize(t)
    problem = cp.Problem(objective, constraints + overlap_constraints)

    # Try multiple solver backends, fall back to generic if not available
    solvers = ['ECOS', 'SCIP', 'GUROBI']  # Add more solvers depending on availability
    best_h = h_opt
    best_c5 = c5_opt

    for solver in solvers:
        try:
            problem.solve(solver=solver, verbose=False, max_iters=150)
            h_solved = h_var.value
            if h_solved is not None:
                h_solved = np.clip(h_solved, 0, 1)
                current_c5 = evaluate_c5(h_solved, dx)
                if current_c5 < best_c5:
                    best_h = h_solved
                    best_c5 = current_c5
                if verbose:
                    print(f"Solver {solver} completed in {time.time() - start_time:.2f}s. Current C₅: {best_c5:.7f}")
        except Exception as e:
            if verbose:
                print(f"Solver {solver} failed: {e}")
        if time.time() - start_time > budget_s:
            break

    # Ensure time budget not exceeded and verify final h
    h_final = best_h
    h_final = np.clip(h_final, 0, 1)
    h_final_sum = np.sum(h_final)
    target_sum = n_points / 2.0
    if abs(h_final_sum - target_sum) > 1e-6:  # Enforce constraint if distorted
        scale_factor = target_sum / h_final_sum
        h_final = np.clip(h_final * scale_factor, 0, 1)

    final_c5 = evaluate_c5(h_final, dx)

    return (h_final.tolist(), final_c5, n_points)
