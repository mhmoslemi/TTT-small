import numpy as np
import cvxpy as cp
import time
from math import floor

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Discrete step function
    n_points = 80
    dx = 2.0 / n_points

    # Load initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
        dx = 2.0 / n_points
        assert np.allclose(np.sum(h), n_points / 2), "Initial h does not satisfy sum constraint"
        assert np.all((0 <= h) & (h <= 1)), "Initial h out of [0,1]"
    else:
        # Fallback initialization
        h = np.zeros(n_points)
        n_eff = int(n_points / 2)
        h[:n_eff] = 1.0  # Square pulse
        np.random.shuffle(h)

    best_h = h.copy()
    best_c5 = float("inf")
    tolerance = 1e-6
    max_iter = 1000
    min_update = 1e-7

    def evaluate_c5(h):
        h_1d = h.reshape(1, -1)
        ones_minus_h = 1 - h
        corrs = np.correlate(h, ones_minus_h, mode="full") * dx
        return np.max(corrs)

    best_c5 = evaluate_c5(h)
    print(f"Initial C5: {best_c5:.6f}")

    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1.0:
            break
        # Determine worst shift k
        h_1d = h.reshape(1, -1)
        ones_minus_h = 1 - h
        corrs = np.correlate(h, ones_minus_h, mode="full") * dx
        worst_k = np.argmax(corrs)

        # Formulate convex optimization problem for worst_k
        delta = cp.Variable(n_points)
        objective = cp.Minimize(corrs[worst_k] - dx * cp.sum(delta[worst_k:] * h[:n_points - worst_k] + delta[:n_points - worst_k] * h[worst_k:] + delta * delta[worst_k:]))

        constraints = [
            h + delta >= 0,
            h + delta <= 1,
            cp.sum(delta) == 0,
            cp.pnorm(delta, p=2) <= 0.05 * n_points  # bound the change size
        ]

        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.ECOS, verbose=False)

        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            delta_sol = delta.value
            h += delta_sol
            h = np.clip(h, 0, 1)
            # Enforce integral exactly
            sum_h = np.sum(h)
            multiplier = n_points / 2 / sum_h
            h *= multiplier
            h = np.clip(h, 0, 1)
            curr_c5 = evaluate_c5(h)

            if curr_c5 < best_c5 - tolerance:
                best_c5 = curr_c5
                best_h = h.copy()
                print(f"Iteration {it}, C5: {curr_c5:.6f}")
            elif abs(best_c5 - curr_c5) < min_update:
                print(f"Converged at {curr_c5:.6f}")
                break
        else:
            print(f"Iteration {it}, failed to solve for k={worst_k}: {problem.status}")
            continue

    best_h = np.clip(best_h, 0, 1)
    best_h *= (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    final_c5 = evaluate_c5(best_h)
    print(f"Final C5: {final_c5:.6f}")
    return best_h.tolist(), final_c5, n_points
