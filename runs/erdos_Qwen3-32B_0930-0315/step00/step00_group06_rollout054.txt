import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h_array = np.array(h_values, dtype=np.float64)
    one_minus_h = 1.0 - h_array
    # Compute overlap for all lags using correlation
    corr = correlate(h_array, one_minus_h, mode="full")
    # Multiply by dx to get the integral
    c5_bound = np.max(corr * dx)
    # Feasibility check
    if not (np.all(0 <= h_array) and np.all(h_array <= 1)):
        raise ValueError("Solution violates domain constraints [0, 1]")
    if abs(np.sum(h_array) * dx - 1.0) > 1e-6:
        raise ValueError(f"Solution violates mean constraint: {np.sum(h_array) * dx}")
    return h_values, c5_bound

def run(seed=42, budget_s=1100, n_points=95, learning_rate=0.05):
    t0 = time.time()
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        # Random initialization ensuring the sum constraint
        h = np.random.rand(n_points)
        h = h / h.sum() * n_points / 2
    h = h.astype(np.float64, copy=False)
    last_improvement = 0
    history = []
    
    # Main optimization loop
    while time.time() - t0 < budget_s:
        dx = 2.0 / n_points
        one_minus_h = 1.0 - h

        # Compute full correlation
        corr = correlate(h, one_minus_h, mode='full')

        # Find mode (lag) with maximum value
        k_max = np.argmax(corr)
        
        # Calculate effective lag in terms of n_points
        k_lag = k_max - n_points + 1
        if k_lag >= 0:
            # Positive shift
            left = max(k_lag, 0)
            right = n_points - k_lag
            mask = np.zeros_like(h)
            mask[left:right] = 1.0 - h[left:right]
        else:
            # Negative shift
            left = n_points - k_lag
            right = n_points
            mask = np.zeros_like(h)
            mask[right:] = 1.0 - h[right:]
            mask[0:left] = -1.0 * h[0:left]

        # Gradient ascent step
        grad = mask * (2 * h - 1) / (n_points)

        # Update h
        h += grad * learning_rate

        # Enforce constraints
        h = np.clip(h, 0.0, 1.0)
        h = h / h.sum() * n_points / 2  # renormalize to ensure mean constraint

        # Evaluate and track progress
        try:
            _, c5_candidate = evaluate_erdos_solution(h)
            history.append(c5_candidate)
            if isinstance(history[-100:], list) and len(history) >= 100 and abs(history[-1] - history[-100]) < 1e-8:
                # Terminate if no improvement
                break
        except ValueError:
            pass

    h_values = h.tolist()
    try:
        h_values, c5_bound = evaluate_erdos_solution(h_values)
        return h_values, c5_bound, n_points
    except Exception:
        return h.tolist(), np.inf, n_points
