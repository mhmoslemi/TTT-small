import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    n_points = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    assert len(corr) == 2 * n_points - 1
    return max(corr[i] * dx for i in range(n_points - 1))

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 80
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)

    h = h.copy()
    h = h / np.sum(h) * (n_points / 2.0)
    h = np.clip(h, 0, 1)

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr_idx = np.argmax(corr[:n_points - 1])
        k = max_corr_idx

        delta = cp.Variable(n_points)
        constraints = [
            h + delta >= 0,
            h + delta <= 1,
            cp.sum(delta) == 0
        ]

        # Linearized objective: minimize overlap for lag k
        def get_overlap():
            return cp.sum((h + delta) * (1 - h[(np.arange(n_points) + k) % n_points])) * dx

        objective = cp.Minimize(get_overlap())
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)

        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            delta_val = delta.value
            step_size = 0.1
            h += step_size * delta_val
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2.0)
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            break

        if time.time() - start_time > budget_s:
            break

    best_c5_bound = best_c5
    return best_h.tolist(), best_c5_bound, int(n_points)
