import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 100
    dx = 2.0 / n_points
    total = n_points / 2  # Required sum of h values

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        # Rescale to ensure sum(h) == total if required
        if np.sum(h) != total:
            h = (h / np.sum(h)) * total
    else:
        # Construct an initial binary pattern: symmetric with n_points/2 ones
        h = np.zeros(n_points)
        h[:int(n_points/2)] = 1.0

    # Projection function: ensure h stays within [0, 1] and sum(h) == total
    def project(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) == 0 or np.sum(x) == n_points:
            return x  # Avoid division by zero
        scaling = total / np.sum(x)
        return np.clip(x * scaling, 0, 1)

    # Objective function: compute C_5
    def compute_bound(h):
        corr = np.zeros(n_points + n_points - 1)
        for k in range(1, n_points):
            val = np.sum(h[:n_points - k] * (1 - h[k:]))
            corr[k] = val
        return (2.0 / n_points) * np.max(corr)

    # Gradient using finite differences (centered)
    def gradient(h):
        g = np.zeros_like(h)
        eps = 1e-6
        for i in range(n_points):
            e = np.zeros_like(h)
            e[i] = eps
            h_plus = h + e
            h_minus = h - e
            g[i] = (compute_bound(h_plus) - compute_bound(h_minus)) / (2 * eps)
        return g

    # Optimize using projected gradient descent
    max_iterations = int(budget_s * 10)  # Each iteration is ~0.1s in practice
    tolerance = 1e-5
    learning_rate = 0.02  # Initial step size

    start = time.time()
    best_h = project(h.copy())
    best_c5 = compute_bound(best_h)

    for iter in range(max_iterations):
        if time.time() - start > budget_s:
            break
        g = gradient(best_h)
        h_new = best_h - learning_rate * g
        h_new = project(h_new)
        c5 = compute_bound(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new
        if np.linalg.norm(g) < tolerance:
            break

    return (best_h.tolist(), best_c5, n_points)
