import numpy as np
from time import time
import evaluate_erdos_solution

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time()
    n_points = 70
    dx = 2.0 / n_points

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[::2] = 1.0
        h *= (n_points / 2.0) / np.sum(h)

    h = np.clip(h, 0, 1)
    h *= (n_points / 2.0) / np.sum(h)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution.evaluate_erdos_solution(h.tolist(), n_points)[0]

    # Optimization loop
    step_size = 0.1
    alpha_decay = 0.8
    max_iter = 1000
    perturb_interval = 50
    no_improvement_counter = 0
    max_no_improvement = 50

    for i in range(max_iter):
        if time() - start_time > budget_s - 1.0:
            break

        # Compute full cross-correlation between h and (1 - h)
        h_np = np.array(h, dtype=np.float64)
        ones_minus_h = 1 - h_np
        corr = np.correlate(h_np, ones_minus_h, mode='full') * dx

        # Extract valid indices
        valid_lags = (np.argmax(corr.shape) >= n_points)
        valid_corr = corr[n_points:-n_points + 1]
        maxlag = np.argmax(valid_corr)
        k = maxlag

        # Gradient for the worst-case lag k
        grad = np.zeros_like(h)
        for m in range(n_points):
            if m + k < n_points:
                grad[m] -= (1 - h[m + k]) * dx
            if m >= k:
                grad[m] += h[m - k] * dx

        # Update h
        h -= step_size * grad
        h = np.clip(h, 0, 1)
        h *= (n_points / 2.0) / np.sum(h)

        # Perturb occasionally
        if i % perturb_interval == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + noise, 0, 1)
            h *= (n_points / 2.0) / np.sum(h)

        # Evaluate and track best h
        curr_c5 = evaluate_erdos_solution.evaluate_erdos_solution(h.tolist(), n_points)[0]
        if curr_c5 < best_c5:
            best_h = h.copy()
            best_c5 = curr_c5
            no_improvement_counter = 0
        else:
            no_improvement_counter += 1
            if no_improvement_counter >= max_no_improvement:
                step_size *= alpha_decay
                no_improvement_counter = 0

    return best_h.tolist(), best_c5, n_points
