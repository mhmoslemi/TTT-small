import time
import numpy as np
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    total_mass = sum(h) * dx
    while total_mass < 1.0 - 1e-8 or total_mass > 1.0 + 1e-8:
        scale_factor = 1.0 / (total_mass * n_points / 2)
        h *= scale_factor
        h = np.clip(h, 0.0, 1.0)
        total_mass = sum(h) * dx

    best_h = h.copy()
    best_c5 = float('inf')

    alpha = 0.1
    perturb_interval = 50
    update_counter = 0
    tolerance = 1e-5
    patience = 50
    no_improvement_count = 0

    while time.time() - start_time < budget_s - 1:
        h_vec = best_h
        h1 = 1 - h_vec

        # Compute cross-correlation for all lags
        corr = correlate(h_vec, h1, mode='full')

        max_index = np.argmax(corr)
        max_lag = max_index - n_points + 1
        if max_index < n_points:
            # Reflect if outside [0,2]
            shifted_h = h1[-max_lag:]
            shift = -max_lag
        else:
            shifted_h = h1[max_index - n_points + 1 - (2 * n_points - max_index):]
            shift = max_index - n_points

        # Compute gradient contribution
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + shift < n_points:
                grad[i] += 1.0 - h1[i + shift]
            if 0 <= i + n_points - shift < n_points:
                grad[i] += h1[i + n_points - shift]
        grad = grad / n_points

        update = -grad
        h_update = best_h + alpha * update

        # Enforce bounds [0,1]
        h_update = np.clip(h_update, 0.0, 1.0)

        # Normalize sum(h) ≈ n_points / 2
        total = h_update.sum()
        scaling = (n_points / 2) / total
        h_update *= scaling
        h_update = np.clip(h_update, 0.0, 1.0)

        # Evaluate new solution
        h1_update = 1 - h_update
        corr_update = correlate(h_update, h1_update, mode='full')
        new_c5 = np.max(corr_update) * dx

        # Accept better solution
        if new_c5 < best_c5 and abs(new_c5 - best_c5) > tolerance:
            best_c5 = new_c5
            best_h = h_update.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        # Reduce learning rate if not improving
        update_counter += 1
        if no_improvement_count > patience:
            alpha *= 0.5
            no_improvement_count = 0

        # Add perturbation
        if update_counter % perturb_interval == 0:
            noise = np.random.normal(0.0, 0.01, size=n_points)
            best_h += noise
            best_h = np.clip(best_h, 0.0, 1.0)
            total = best_h.sum()
            scaling = (n_points / 2) / total
            best_h *= scaling
            best_h = np.clip(best_h, 0.0, 1.0)

    return (best_h.tolist(), best_c5, n_points)
