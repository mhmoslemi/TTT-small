import numpy as np
import time
import scipy.signal
import cvxpy as cp

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')

    n = 160
    dx = 2.0 / n
    initial_sum = n / 2.0

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, copy=True)
        n = len(h)
        assert np.isclose(np.sum(h), n / 2)
    else:
        h = np.zeros(n)
        h[n//4:-n//4] = 1.0  # plateau initialization
        h = h.astype(float)

    # Normalize to ensure sum constraint
    h *= (initial_sum / np.sum(h))

    def compute_c5(h):
        dx = 2.0 / len(h)
        _, c5, _ = evaluate_erdos_solution(h, n_points=len(h))
        return c5

    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Phase 1: Gradient descent
    learning_rate = 0.01
    n_iter_phase1 = 1000
    for i in range(n_iter_phase1):
        if time.time() - start_time > budget_s - 10:  # Leave 10 seconds for final checks
            break

        # Compute gradient
        corr = scipy.signal.correlate(h, 1 - h, mode="full")
        filt = scipy.signal.correlate(np.arange(n), 1 - h, mode="full")
        dg = np.zeros(n)
        for k in range(n):
            for j in range(n):
                if (j - k + n) % n < n:
                    dg[j] += np.clip((j - k + n) % n - n // 2, 0, n // 2)

        # Use FFT-based correlation for accurate gradient estimation
        h_adj = h.copy()
        h_adj[:-1] = h[1:]
        grad = scipy.signal.correlate(h, h_adj, mode="full") - h * h

        # Normalize and use for descent step
        grad = grad[n-1: 2*n-1]  # keep relevant part
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h *= (initial_sum / np.sum(h))

        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        learning_rate = max(0.001, learning_rate * 0.998)  # Decrease learning rate

    # Phase 2: Refine with CVXPY
    x = cp.Variable(n)
    objective = cp.Minimize(dx * cp.max(cp.conv(h, 1 - h, mode="full")))
    constraints = [0 <= x, x <= 1, cp.sum(x) == initial_sum]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.GUROBI if 'GUROBI' in cp.installed_solvers() else cp.ECOS)

    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h = x.value
        h = np.clip(h, 0, 1)
        h *= (initial_sum / np.sum(h))

    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h

    # One final check for improvement
    while time.time() - start_time < budget_s - 1:
        h = best_h + np.random.normal(0, 0.01, n)
        h = np.clip(h, 0, 1)
        h *= (initial_sum / np.sum(h))
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n)
