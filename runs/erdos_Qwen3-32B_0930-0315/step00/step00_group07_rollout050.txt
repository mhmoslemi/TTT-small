import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_C5(h, dx):
    """Compute C5: max_k correlate(h, 1 - h, mode='full') * dx"""
    h = np.array(h)
    k_all = range(-len(h) + 1, len(h))
    shift_vals = np.array([correlate(h, (1 - h).shift(i).fillna(0), mode='same') for i in k_all])
    return np.max(shift_vals) * dx


def run(seed=42, budget_s=1100, n_points=80):
    np.random.seed(seed)

    start_time = time.time()
    h_init = np.array(initial_h_values)
    h = h_init.copy()
    h = h / np.sum(h) * (n_points / 2)

    dx = 2.0 / n_points
    best_C5 = 0.38085  # current best upper bound
    best_h = h.copy()

    def step(h, rate=1e-1, steps=20):
        nonlocal best_C5, best_h
        for _ in range(steps):
            h_sig = 1 - h
            corr = correlate(h, h_sig, mode='full')
            max_idx = np.argmax(np.abs(corr))
            sign = np.sign(corr[max_idx])
            grad = np.zeros_like(h)
            for i in range(n_points):
                if 0 <= max_idx - i < len(h):
                    grad[i] += 2 * h[i] * sign * h[max_idx - i] - sign * (1 - 2 * h[i])
            grad[grad > 0] *= 1
            grad[grad < 0] *= -1
            h_corr = h + rate * grad
            h_corr = np.clip(h_corr, 0, 1)
            h_corr = h_corr / np.sum(h_corr) * (n_points / 2)
            C5 = np.max(correlate(h_corr, 1 - h_corr, mode='full')) * dx
            if C5 < best_C5:
                best_C5 = C5
                best_h = h_corr
        return best_h

    while True:
        h = step(h)
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break

    return (best_h.tolist(), best_C5, n_points)
