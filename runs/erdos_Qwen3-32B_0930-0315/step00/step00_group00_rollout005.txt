import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    timeout = budget_s

    def compute_c5(h, dx):
        h = np.array(h, dtype=float)
        ones_h = 1.0 - h
        corr_vals = np.correlate(h, ones_h, mode='full')
        return np.max(corr_vals) * dx

    def normalize_h_sum(h, n_points):
        h = np.clip(h, 0, 1)
        desired_sum = n_points / 2
        current_sum = h.sum()
        if abs(current_sum - desired_sum) > 1e-8:
            h = h * (desired_sum / current_sum)
        h = np.clip(np.round(h, 5), 0, 1)
        return h

    def perturb_neighborhood_h(h, n_perturb=5):
        h = h.copy()
        idx = np.random.choice(len(h), size=n_perturb, replace=False)
        delta = (np.random.rand(n_perturb) - 0.5) * 0.5
        h[idx] += delta
        return normalize_h_sum(h, len(h))

    def generate_neighbors(h, n_neighbors=20):
        n_points = len(h)
        neighbors = [h]
        for _ in range(n_neighbors):
            new_h = perturb_neighborhood_h(h)
            neighbors.append(new_h)
        return neighbors

    def tabu_search(h_init, n_points, iter_limit=10000, tabu_size=10, stop_on_timeout=True):
        best_h = h_init
        best_c5 = compute_c5(best_h, 2.0 / n_points)
        tabu_list = []
        iter_count = 0
        while time.time() - start_time < timeout and iter_count < iter_limit:
            iter_count += 1
            neighbors = generate_neighbors(best_h)
            valid_neighbors = []
            for nh in neighbors:
                if np.allclose(np.sum(nh), n_points / 2):
                    valid_neighbors.append(nh)

            best_neighbor = None
            best_neighbor_c5 = float('inf')

            for nh in valid_neighbors:
                if len(tabu_list) > 0 and any(np.allclose(nh, t, atol=1e-2) for t in tabu_list):
                    continue
                c5 = compute_c5(nh, 2.0 / n_points)
                if c5 < best_neighbor_c5:
                    best_neighbor_c5 = c5
                    best_neighbor = nh

            if best_neighbor is None:
                continue

            if best_neighbor_c5 < best_c5:
                best_h = best_neighbor
                best_c5 = best_neighbor_c5
            else:
                pass

            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
            tabu_list.append(best_h.copy())

        return best_h, best_c5

    def prepare_input(n_points):
        if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
            h = initial_h_values.copy()
        else:
            h = np.random.rand(n_points)
            h = normalize_h_sum(h, n_points)
        h = np.clip(h, 0.0, 1.0)
        return h

    n_points = 45
    h = prepare_input(n_points)
    h_opt, c5_bound = tabu_search(h, n_points)
    return h_opt, c5_bound, n_points
