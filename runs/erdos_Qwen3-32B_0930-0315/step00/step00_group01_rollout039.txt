import numpy as np
import cvxpy as cp
from time import time
from itertools import chain


def run(seed=42, budget_s=1100, n_points=47):
    np.random.seed(seed)

    start = time()
    dx = 2.0 / n_points

    # Warm start: Normalize to exact integral constraint
    def normalize_h(h):
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (n_points / 2) / h_sum
        return np.clip(h, 0, 1)

    # Build initial guess
    if h_init is not None:
        h = normalize_h(h_init)
    else:
        h = np.random.rand(n_points)
        h = normalize_h(h)

    # Precompute correlation positions for fast evaluation
    def evaluate_c5(h, n_points=n_points, dx=dx):
        full_corr = np.correlate(h, 1.0 - h, mode="full")
        C5 = full_corr * dx
        return np.max(C5)

    # Optimization via nonlinear constrained optimization
    h_var = cp.Variable(n_points, nonneg=True)
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]

    t = cp.Variable()
    t_param = cp.Parameter(value=0)
    t_obj = cp.Minimize(t)

    # Formulate the max correlation as maximum over batches
    def vector_correlation(h_vec, dx=dx):
        h = h_vec.value
        full_corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(full_corr * dx)
        return max_corr

    # Optimization loop via SCP (successive convex programming)
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    iters = 0
    max_iters = 30  # Hard constraint: limit SCP iterations

    while time() - start < budget_s and iters < max_iters:
        # Approximate the non-convex max(C5) with a first-order Taylor expansion
        h0 = best_h
        expr = vector_correlation(h_var)

        # Approximate using perturbation and backward difference
        perturbation = 1e-4 * np.random.randn(n_points)
        perturbation = normalize_h(h0 + perturbation) - h0

        grad = (vector_correlation(h_var + cp.multiply(perturbation, 1e-4)) - expr) / 1e-4

        constraints.append(t - grad @ (h_var - h0) >= 0)
        prob = cp.Problem(t_obj, constraints)

        try:
            result = prob.solve(solver=cp.OSQP, verbose=False)
            if result is not None and result < best_c5:
                best_h = h_var.value.copy()
                best_c5 = result
        except:
            pass  # Skip infeasible steps

        iters += 1

    best_h = normalize_h(best_h)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, n_points
