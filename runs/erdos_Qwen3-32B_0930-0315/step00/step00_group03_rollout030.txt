import numpy as np
import cvxpy as cp
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # Use parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        h = np.random.rand(70)
        h = h / h.sum() * (70 / 2)
        n_points = 70

    # Parameters
    dx = 2.0 / n_points

    # Projection function to enforce constraints
    def project(h):
        h_var = cp.Variable(n_points)
        obj = cp.Minimize(cp.sum_squares(h_var - h))
        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if not prob.status == "optimal":
                raise Exception("Projection failed")
        except:
            return h
        return np.clip(h_var.value, 0, 1)

    # Evaluate C5
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    best_h = h.copy()
    best_c5 = compute_c5(h)

    eta = 0.01
    iter_count = 0
    MAX_ITER = 1000
    NO_IMPROVE_LIMIT = 20

    while time.time() - start_time < budget_s and iter_count < MAX_ITER:
        iter_count += 1

        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)

        # Get the shift index k that gives largest overlap
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)

        # Compute gradient for worst-performing alignment
        G = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if j >= 0 and j < n_points:
                G[i] = (1 - h[j])  # Derivative w.r.t h[i]
            if j >= 0:
                G[j] -= h[i]  # Derivative w.r.t h[j] from -h[i]*(1-h[j])

        # Normalize gradient
        norm = np.sqrt(np.sum(G**2))
        if norm == 0:
            continue
        G /= norm

        # Line search with backtracking
        trial_eta = eta
        for _ in range(10):  # Max 10 line search steps
            h_candidate = h - trial_eta * G
            h_candidate = project(h_candidate)
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < c5:
                break
            trial_eta *= 0.5
        else:
            h = best_h.copy()
            continue

        h = h_candidate
        current_c5 = c5_candidate

        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        if (np.isnan(best_c5) or best_c5 > 1.0 or best_c5 < 0.0):
            h = best_h.copy()

    return (best_h.tolist(), best_c5, n_points)
