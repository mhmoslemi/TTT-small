import numpy as np
import time
from scipy.interpolate import CubicSpline
import cvxpy as cp

def evaluate_and_bound(h, n_points):
    dx = 2.0 / n_points
    ones_minus_h = 1.0 - h
    correl = np.correlate(h, ones_minus_h, mode="full")
    C5 = np.max(correl) * dx
    return C5

def repair(feasibility, h, n_points):
    h_clipped = np.clip(h, 0, 1)
    integral = np.sum(h_clipped)
    if abs(integral - n_points / 2) > 1e-6:
        h_clipped *= (n_points / 2) / integral
    return h_clipped

def evolve_seed(h, n_points, seed=None):
    np.random.seed(hash(time.time()) + seed if seed is not None else None)
    indices = np.random.choice(n_points, size=2, replace=False)
    delta = np.random.normal(0, 0.1, size=n_points)
    new_h = h.copy() + delta
    return repair(feasibility, new_h, n_points)

def evolve_population(pop, n_points, mutation_rate=0.4, seed=None):
    np.random.seed(seed)
    np.random.shuffle(pop)
    new_pop = pop.copy()
    for i in range(len(pop)):
        if np.random.rand() < mutation_rate:
            new_pop[i] = evolve_seed(pop[i], n_points, seed=i)
    return np.array(new_pop)

def evolve(seed=42, budget_s=1100, n_points=14, population_size=40, generations=200, learning_rate=0.02):
    feasibility = lambda h: np.all((h >= 0) & (h <= 1)) and abs(np.sum(h) - n_points / 2) < 1e-6
    
    initial = (initial_h_values if locals().get("initial_h_values") is not None else
               (np.random.rand(n_points) * 0.8 + 0.1))
    initial = repair(feasibility, initial, n_points)
    
    population = []
    for _ in range(population_size):
        h = evolve_seed(initial, n_points, seed=seed)
        if feasibility(h):
            population.append(h)
    population = np.array(population)

    best_h = None
    best_C5 = np.inf
    start_time = time.time()
    
    # Genetic Algorithm Main Loop
    for gen in range(generations):
        gen_start = time.time()
        population_sorted = [h for h in population]
        population_sorted.sort(key=lambda h: evaluate_and_bound(h, n_points))
        population = np.array(population_sorted[:population_size // 2])
        if feasibility(population[0]) and evaluate_and_bound(population[0], n_points) < best_C5:
            best_h = population[0].copy()
            best_C5 = evaluate_and_bound(best_h, n_points)
        population = evolve_population(population, n_points, mutation_rate=0.2, seed=seed)
        if time.time() - start_time > budget_s:
            break
        elapsed = time.time() - gen_start
        if budget_s - (time.time() - start_time) < 200:
            break

    # Gradient Descent Refinement
    if best_h is not None and feasibility(best_h):
        for _ in range(100):
            grad = np.correlate(1.0 - np.roll(best_h, -1), best_h, mode='same') - best_h * (1 - best_h)
            grad = np.clip(grad, -1, 1)
            best_h -= learning_rate * grad
            best_h = repair(feasibility, best_h, n_points)
            if evaluate_and_bound(best_h, n_points) < best_C5:
                best_C5 = evaluate_and_bound(best_h, n_points)
            if time.time() - start_time > budget_s:
                break

    # Double the resolution
    if best_h is not None and feasibility(best_h):
        cs = CubicSpline(np.linspace(0, 2, n_points), best_h, bc_type='clamped')
        x_fine = np.linspace(0, 2, n_points * 2)
        h_fine = cs(x_fine)
        h_fine = repair(feasibility, h_fine, n_points * 2)
        C5_fine = evaluate_and_bound(h_fine, n_points * 2)
        if C5_fine < best_C5:
            best_C5 = C5_fine

    return best_h, best_C5, n_points

def run(seed=42, budget_s=1100):
    h_best, c5_bound, n_points = evolve(seed=seed, budget_s=budget_s)
    return h_best, c5_bound, n_points
