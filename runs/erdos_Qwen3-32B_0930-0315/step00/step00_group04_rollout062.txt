import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_objective(h, n_points):
    dx = 2.0 / n_points
    C = correlate(h, 1 - h, mode='full') * dx
    return np.max(C)

def repair_h(h):
    h_clipped = np.clip(h, 0.0, 1.0)
    h_clipped = h_clipped * (h.sum() / h_clipped.sum())  # Rescale to maintain ∫h(x)dx=1
    return h_clipped

def random_h(n_points):
    h = np.random.uniform(low=0.0, high=1.0, size=n_points)
    h = h * (n_points / 2.0) / h.sum()  # Scale to satisfy the integral constraint
    h = repair_h(h)
    return h

def initialize_population(pop_size, n_points):
    return [random_h(n_points) for _ in range(pop_size)]

def mutate(h, mutation_rate):
    h_mut = h.copy()
    indices = np.random.choice(len(h), size=int(len(h) * mutation_rate), replace=False)
    h_mut[indices] += np.random.normal(0.0, 0.1, size=len(indices))
    return repair_h(h_mut)

def crossover(h1, h2):
    idx = np.random.randint(len(h1))
    h_child = np.concatenate((h1[:idx], h2[idx:]))
    return repair_h(h_child)

def tournament_selection(population, fitnesses, k=5):
    candidates = np.random.choice(len(population), size=k, replace=False)
    best_idx = min(candidates, key=lambda i: fitnesses[i])
    return population[best_idx]

def genetic_algorithm(population_size=20, generations=200, n_points=14, mutation_rate=0.2, crossover_rate=0.8, time_budget=500):
    start_time = time.time()
    population = initialize_population(population_size, n_points)
    fitnesses = [evaluate_objective(h, n_points) for h in population]

    best_h = population[np.argmin(fitnesses)]
    best_c5 = min(fitnesses)

    for gen in range(generations):
        if time.time() - start_time > time_budget:
            break
        new_population = []
        for _ in range(population_size):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            if np.random.rand() < crossover_rate:
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()
            if np.random.rand() < mutation_rate:
                child = mutate(child, mutation_rate)
            new_population.append(child)
        population = new_population
        fitnesses = [evaluate_objective(h, n_points) for h in population]
        current_best_idx = np.argmin(fitnesses)
        current_best_c5 = fitnesses[current_best_idx]
        if current_best_c5 < best_c5:
            best_h = population[current_best_idx]
            best_c5 = current_best_c5
    return best_h, best_c5

def gradient_descent(h, learning_rate=0.01, iterations=100, time_budget=600):
    n_points = len(h)
    best_h = h.copy()
    best_c5 = evaluate_objective(best_h, n_points)
    start_time = time.time()

    for _ in range(iterations):
        if time.time() - start_time > time_budget:
            break
        C = correlate(h, 1 - h, mode='full') * (2.0 / n_points)
        max_idx = np.argmax(C)
        left = max(0, max_idx - n_points + 1)
        h_shifted = h[(max_idx - (n_points - 1)):] if left <= 0 else h[left:max_idx + 1]
        grad = (1 - h_shifted)
        grad = np.pad(grad, (0, n_points - len(grad)), 'constant', constant_values=0.0)
        h -= learning_rate * grad
        h = repair_h(h)
        c5 = evaluate_objective(h, n_points)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    return best_h, best_c5

def run(seed=42, budget_s=1100, pop_size=20, generations=200, n_points=14):
    np.random.seed(seed)

    ga_h, ga_c5 = genetic_algorithm(pop_size=pop_size, generations=generations, n_points=n_points, time_budget=600)
    gd_h, gd_c5 = gradient_descent(ga_h, time_budget=budget_s - 600)

    # Final check and feasibility repair
    h_final = repair_h(gd_h)
    c5_final = evaluate_objective(h_final, n_points)

    return h_final, c5_final, n_points
