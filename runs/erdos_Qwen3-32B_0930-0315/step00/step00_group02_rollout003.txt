import numpy as np
import cvxpy as cp
from time import time


def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time()
    dx_re = 2.0

    # Parameters
    n_points = 160
    h_init = np.zeros(n_points)
    mid = n_points // 2
    plateau_width = 80
    h_init[mid - plateau_width // 2 : mid + plateau_width // 2] = 1.0 / dx_re
    h_init *= dx_re  # Scale to have integrated value equal to 1
    h = h_init.copy()

    # Normalize h to satisfy the constraint sum(h) == n_points / 2
    h = h / h.sum() * (n_points / 2)

    # Phase 1: Gradient descent with backtracking line search
    max_iter = 500
    learning_rate = 5e-2
    for iter in range(max_iter):
        if time() - start_time >= budget_s * 0.8:
            break  # Early termination

        # Compute the current overlap integrals
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * (2.0 / n_points))
        i_k = np.argmax(corr)

        # Compute gradient: gradient[i_j] = sum_k A[k][i_j] * (1 - h[(i_j + k) % n])
        grad = np.zeros_like(h)
        k = i_k - n_points // 2
        for i in range(n_points):
            idx = (i + k) % n_points
            grad[i] = (1 - h[idx]) if 0 <= idx < n_points else 0

        # Apply gradient step
        h -= learning_rate * grad

        # Apply constraints: 0 ≤ h[i] ≤ 1 and sum(h) == n_points / 2
        h = np.clip(h, 0, 1)
        correction = (n_points / 2) - h.sum()
        h += correction / n_points  # Small correction to preserve constraint

        if iter % 50 == 0:
            corr = np.correlate(h, 1 - h, mode="full")
            print(f"Iter {iter}, C5: {np.max(corr) * (2.0 / n_points):.6f}")

    # Start phase 2: Sequential Convex Programming (SCP) refinement
    final_h = cp.Variable(n_points)
    constraints = [0 <= final_h, final_h <= 1, cp.sum(final_h) == n_points / 2]

    # Use best result from gradient descent as starting point
    initial_h = h

    max_correlation = cp.Variable()
    epsilon = 0.01

    # We approximate max_k corr_k(h) using evaluations at k where it matters
    corr_vals = []
    for k in range(-n_points + 1, n_points):
        idx = (np.arange(n_points) + k) % n_points
        corr = cp.sum(final_h * (1 - final_h[idx]))
        corr_vals.append(corr)

    correlations = cp.hstack(corr_vals)
    max_corr = cp.Max(cp.hstack(corr_vals))

    # Use a constrained problem with convex approximation
    model = cp.Minimize(max_corr)
    scp_prob = cp.Problem(model, constraints)

    solution_radius = 1e-6
    for _ in range(5):
        if time() - start_time >= budget_s * 0.9:
            break  # Early termination

        # Solve for nearby refinement
        theta = (final_h - initial_h) / solution_radius
        scp_prob.solve(warm_start=True, solver=cp.GLPK_MI, verbose=False)

        if scp_prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_update = final_h.value
            h_update = np.clip(h_update, 0, 1)
            h_update = h_update / np.sum(h_update) * (n_points / 2)
            corr = np.correlate(h_update, 1 - h_update, mode="full")
            new_c5 = np.max(corr * (2.0 / n_points))

            if abs(new_c5 - np.max(corr_vals)) < 1e-6:
                break
        else:
            print("No progress in SCP refinement, breaking loop.")
            break

        initial_h = h_update

    # Final evaluation
    corr = np.correlate(h, 1 - h, mode="full")
    c5_bound = np.max(corr) * (2.0 / n_points)
    return (h, c5_bound, n_points)
