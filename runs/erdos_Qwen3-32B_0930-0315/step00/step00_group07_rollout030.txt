import numpy as np
import cvxpy as cp
import time

def compute_C5(h, dx):
    h_arr = np.array(h)
    n = len(h)
    corrs = np.correlate(h_arr, 1 - h_arr, mode='full')
    max_C5 = np.max(corrs[n//2 - 1:n//2 + 1]) * dx
    return max_C5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s
    best_h = None
    best_C5 = float('inf')

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
        h = initial_h.copy()
        assert np.isclose(np.sum(h), n_points / 2)
    else:
        n_points = 80
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = compute_C5(best_h, dx)

    while time.time() - start_time < max_time:
        # Step 1: Compute overlaps and find worst lag
        h_vec = h
        one_minus_h = 1 - h_vec
        corrs = np.correlate(h_vec, one_minus_h, mode='full')
        max_idx = np.argmin(corrs[n_points-5:n_points+5]) + n_points -5  # Worst C_k (since minimize max)
        k = max_idx - n_points
        corr_k = corrs[max_idx]

        # Step 2: Form convex subproblem for lag k
        h_vars = cp.Variable(n_points)
        delta_h = h_vars - h_vec

        obj = cp.sum(h_vec[k:] * (1 - h_vec[:n_points - k]) + delta_h[k:] * (1 - h_vec[:n_points - k]) + h_vec[k:] * (-delta_h[:n_points - k]) - delta_h[k:] * delta_h[:n_points - k]) * dx
        obj = cp.Minimize(obj)

        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2,
        ]
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000)
            if prob.status != 'optimal':
                continue
            new_h = np.clip(h_vars.value, 0, 1)
            new_h = new_h / np.sum(new_h) * (n_points / 2)
            new_C5 = compute_C5(new_h, dx)

            if new_C5 < best_C5:
                best_h = new_h.copy()
                best_C5 = new_C5

            h = new_h.copy()

        except Exception as e:
            print("Warning: subproblem failed", e)

        # Early stopping if nearly converged
        if best_C5 < 1e-6 or time.time() - start_time > max_time:
            break

    return best_h, best_C5, n_points
