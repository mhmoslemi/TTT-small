import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use the provided initial_h_values if available
    try:
        h = initial_h_values
        n_points = len(h)
    except NameError:
        n_points = 95  # default from the classic setup
        h = np.random.rand(n_points)
        h = h / h.sum() * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')
    no_improve_steps = 0
    max_steps = 1_000_000
    step = 0

    def evaluate_c5(h_vals):
        n_total = len(h_vals)
        h_flat = h_vals.flatten()
        h_1m = 1.0 - h_flat

        overlap = correlate(h_flat, h_1m, mode='full')
        overlap = overlap * dx
        return np.max(overlap)

    while time.time() - start_time < budget_s and no_improve_steps < 200 and step < max_steps:
        step += 1

        # Compute the overlap at all lags
        n = len(h)
        h1 = np.concatenate([h, np.zeros(n - 1)])
        h2 = np.concatenate([1 - h, np.zeros(n - 1)])
        correlations = correlate(h1, h2)
        correlations = correlations * dx

        # Find the lag with maximum overlap
        k_max = np.argmax(correlations)

        # Compute one-step gradient at that lag
        grad = np.zeros_like(h)
        for i in range(n):
            for j in range(n):
                if i + k_max < n:
                    grad[i] += h[j] * (1 - h[i + k_max])  # Cross term
                if j == i + k_max and j < n:
                    grad[i] += h[i] * (-1 + 2 * h[i + k_max])  # Quadratic term

        # Normalize gradient and take step
        grad = grad - np.mean(grad)  # Center gradient direction
        grad /= (np.linalg.norm(grad) + 1e-8)
        lr = 0.01 * (1.0 / (1.0 + step ** 0.25))  # decreasing learning rate

        h = h + lr * grad
        h = np.clip(h, 0, 1)  # Projection step
        h = h / h.sum() * (n_points / 2)  # Renormalize

        # Occasionally perturb for escape
        if np.random.random() < 0.01:
            perturbation = np.random.randn(n_points) * 0.05
            h = np.clip(h + perturbation, 0, 1)
            h = h / h.sum() * (n_points / 2)

        # Evaluate current
        current_c5 = evaluate_c5(h)

        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve_steps = 0
        else:
            no_improve_steps += 1

    print(f"Best C₅ value found: {best_c5:.8f}")
    return best_h, best_c5, len(best_h)
