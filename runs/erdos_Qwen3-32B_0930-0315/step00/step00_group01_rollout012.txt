import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = 47  # Start with initial_h_values (n=47)
    np.random.seed(seed)
    dx = 2.0 / n_points

    # Load the initial solution if it exists
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, copy=True).astype(float)
        # Normalize sum(h) to match constraint
        h *= (n_points / 2 / np.sum(h))
        h = np.clip(h, 0.0, 1.0)
    else:
        # Random initialization as a safeguard
        h = np.random.rand(n_points)
        h = np.clip(h, 0.0, 1.0)
        h *= (n_points / 2 / np.sum(h))

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    learning_rate = 0.1
    decay_rate = 0.999
    perturb_interval = 100
    perturb_strength = 0.01

    while time.time() - start_time < budget_s:
        # Compute overlaps
        # Compute (1 - h)
        one_minus_h = 1.0 - h
        # Use NumPy correlate for all lags
        corr = correlate(h, one_minus_h, mode="full")
        # Only consider lags that correspond to k ∈ [0, n_points - 1]
        val = corr[n_points - 1:] * dx
        k_max = np.argmax(val)
        c5 = val[k_max]
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

        # Gradient for k_max: grad[i] = (1 - h[i + k_max]) - h[i]
        grad = np.zeros_like(h)
        for i in range(n_points - k_max):
            grad[i] += 1 - h[i + k_max]
        for i in range(k_max, n_points):
            grad[i] -= h[i - k_max]
        
        # Gradient ascent step to reduce overlap
        h -= learning_rate * grad
        h = np.clip(h, 0.0, 1.0)

        # Project back onto feasible set: maintain mean
        h /= np.sum(h) * 2.0 / n_points

        # Anneal learning rate
        learning_rate *= decay_rate

        # Add small perturbations to prevent getting stuck
        if (len(h) % perturb_interval) == 0:
            perturb = np.random.rand(*h.shape) * perturb_strength
            h += perturb
            h = np.clip(h, 0.0, 1.0)
            h /= np.sum(h) * 2.0 / n_points

    return best_h, best_c5, n_points
