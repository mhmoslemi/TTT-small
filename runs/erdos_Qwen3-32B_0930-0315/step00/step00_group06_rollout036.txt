import numpy as np
from scipy.signal import correlate
import time


def evaluate_C5(h, dx):
    h_1_minus_h = 1 - h
    corr = correlate(h, h_1_minus_h, mode='full')
    corr *= dx
    max_corr = np.max(corr[np.abs(np.arange(len(corr)) - h.size + 1) <= len(h)])
    return max_corr


def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum == 0 or current_sum == len(h):
        return np.zeros_like(h)
    return h / current_sum * target_sum


def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    initial_n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)

    target_sum = initial_n_points / 2
    h = normalize_h(h, target_sum)

    dx = 2.0 / initial_n_points
    best_h = h.copy()
    best_C5 = evaluate_C5(best_h, dx)

    start_time = time.time()
    max_iter = int(1e6)
    step = 0
    no_improve_steps = 0
    max_no_improve = 100

    while time.time() - start_time < budget_s and no_improve_steps < max_no_improve and step < max_iter:
        # Compute overlap correlations
        h_flip = 1 - h
        corr = correlate(h, h_flip, mode='full')

        # Only consider valid lags
        center = h.size - 1
        valid_lags = np.arange(len(corr))
        in_range = np.abs(valid_lags - center) <= h.size
        valid_corr = corr[in_range]
        k_max = np.argmax(valid_corr)

        k_max_index = valid_lags[k_max]
        begin = max(0, k_max_index - (h.size - 1))
        end = min(h.size, k_max_index + 1)

        # Gradient update (simplified)
        h_grad = h_flip[begin:end] * np.arange(end - begin) / (end - begin) if end > begin else [0.0]
        h[begin:end] += 0.001 * (h_grad - 0.5 * h[begin:end])

        # Projection and normalization
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)

        # Evaluate new h
        new_C5 = evaluate_C5(h, dx)
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h.copy()
            no_improve_steps = 0
        else:
            no_improve_steps += 1

        step += 1

    return best_h, best_C5, len(best_h)
