import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    start_time = time.time()
    initial = initial_h_values
    n_points = len(initial)
    dx = 2.0 / n_points

    # Ensure initial is feasible (clipped and satisfies sum constraint)
    h = np.clip(initial, 0, 1).copy()
    s = h.sum()
    h = h * (n_points / 2.0) / s  # scale to meet the sum constraint
    h = np.clip(h, 0, 1)  # re-clip after scaling

    best_h = h.copy()
    best_c5 = float("inf")
    patience = 0
    alpha = 0.1
    max_iterations = 1000
    iteration = 0
    improvement_threshold = 1e-5

    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c5 = corr.max()
        return c5

    def update_h(h, grad, alpha):
        h -= alpha * grad
        h = np.clip(h, 0, 1)
        s = h.sum()
        if abs(s - n_points / 2.0) > 1e-4:
            h = h * (n_points / 2.0) / s
        return h

    def add_noise(h):
        noise = np.random.normal(0, 0.01, h.shape)
        h = h + noise
        return np.clip(h * (n_points / 2.0) / h.sum(), 0, 1)

    while time.time() - start_time < budget_s:
        if iteration >= max_iterations:
            break

        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        k_max = np.argmax(corr)

        grad = np.zeros_like(h)
        for m in range(n_points):
            if m + k_max < n_points:
                grad[m] += (1 - h[m + k_max]) * dx
            if m >= k_max:
                grad[m] -= h[m - k_max] * dx

        h = update_h(h, grad, alpha)

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

        # Reduce step size if no improvement
        if patience > 50:
            alpha *= 0.5
            patience = 0

        # Add random noise every 50 iterations
        if iteration % 50 == 0 and iteration > 0:
            h = add_noise(h)

        if alpha < 1e-4:
            break

        iteration += 1

    return best_h.tolist(), best_c5, n_points
