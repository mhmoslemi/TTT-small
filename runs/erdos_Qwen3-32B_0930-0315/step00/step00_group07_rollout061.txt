import numpy as np
import time
import cvxpy as cp
import evaluate_erdos_solution

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        assert 0 <= h.min() <= h.max() <= 1, "Initial h invalid"
        assert abs(h.sum() - n_points / 2) < 1e-8, "Initial sum mismatch"
    else:
        n_points = 80
        h = np.zeros(n_points)
        h[:] = 0.5
    dx = 2.0 / n_points

    best_h = h.copy()
    best_C5 = np.inf

    def evaluate_C5(h):
        h_corr = np.correlate(h, 1 - h, mode="full")
        h_corr = h_corr[n_points - 1: 2 * n_points - 1]  # consider relevant part
        C5 = np.max(h_corr) * dx
        return C5

    C5 = evaluate_C5(h)
    best_C5 = C5
    best_h[:] = h[:]

    while time.time() - start_time < budget_s:
        overlap = np.correlate(h, 1 - h, mode="full")
        start = n_points - 1
        end = 2 * n_points - 1
        overlap = overlap[start:end]
        Cks = overlap * dx
        k_max = np.argmax(Cks)
        k = k_max

        # Build 1-subproblem for k
        n = n_points
        h_var = cp.Variable(n)

        k_roll = k

        # Convex subproblem: objective linearized around h
        obj = 0
        for i in range(n):
            j = (i + k_roll) % n
            obj += h[i] * (1 - h[j]) - h[i] * h[j]
        obj = -obj * dx

        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n / 2
        ]

        problem = cp.Problem(cp.Minimize(obj), constraints)
        problem.solve(solver=cp.ECOS, verbose=False)

        if problem.status in ['optimal', 'optimal_inaccurate']:
            dh = h_var.value - h
            # Update h
            h += dh
            h = np.clip(h, 0, 1)
            # repairing
            scaling = (n / 2) / (h.sum() + 1e-12)
            h *= scaling
            h = np.clip(h, 0, 1)  # final clip

        current_C5 = evaluate_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h[:] = h[:]

        # Stop if small improvements
        if time.time() - start_time > budget_s:
            break

    return (best_h.tolist(), best_C5, n_points)
