import numpy as np
import cvxpy as cp
import time
from math import floor
import scipy.optimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    h_init = initial_h_values
    n_points = h_init.shape[0]
    h = h_init.copy()
    dx = 2.0 / n_points

    # Precompute all shifts of h for 1-h correlations
    shifts = []
    for k in range(-n_points + 1, n_points):
        shifted = np.roll(h, -k)
        shifted[k:] = h[:n_points - k]
        shifted[:k] = 0.0
        shifts.append(shifted)

    def compute_objective(h):
        denom = np.sum(h)
        h_normalized = h / denom * n_points / 2
        corr_term = np.correlate(h_normalized, 1 - h_normalized, mode='full') * dx
        return np.max(corr_term)

    def compute_derivative(h):
        denom = np.sum(h)
        h_normalized = h / denom * n_points / 2
        corr = np.correlate(h_normalized, 1 - h_normalized, mode='full') * dx
        k_max = np.argmax(corr)
        h_shifted = np.roll(1 - h_normalized, -k_max)
        h_shifted[k_max:] = 1 - h_normalized[:n_points - k_max]
        h_shifted[:k_max] = 0.0
        grad = -2 * (h_shifted - h_normalized)
        return grad

    def project(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    best_h = h.copy()
    best_c5 = compute_objective(h)
    patience = 0
    lr = 0.1
    t = 0

    while time.time() - start_time < budget_s:
        t += 1

        grad = compute_derivative(h)
        h_update = h - lr * grad
        h = project(h_update)

        c5 = compute_objective(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            patience = 0
        else:
            patience += 1

        if patience > 50:
            break

        if t % 10 == 0:
            lr *= 0.98  # Learning rate decay

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)

    return (best_h, best_c5, n_points)
