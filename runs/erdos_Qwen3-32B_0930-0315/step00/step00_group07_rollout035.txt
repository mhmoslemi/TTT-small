import numpy as np
import time
import scipy.optimize
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial_h_values from the parent construction (should be size 80)
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, copy=True, dtype=np.float64)
    dx = 2.0 / n_points
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    def objective(x):
        h_best = x.reshape(h.shape)
        c = np.correlate(h_best, 1 - h_best, mode='full')
        return np.max(c * dx)
    
    # Projection to [0, 1]^n and conserve constraint
    def project(x):
        x = np.clip(x, 0, 1)
        x_sum = np.sum(x)
        scale = n_points / 2 / x_sum
        x = x * scale
        return x
    
    def callback(x):
        nonlocal best_h, best_c5, evaluations
        evaluations += 1
        c5 = objective(x)
        if c5 < best_c5:
            best_h = np.copy(x)
            best_c5 = c5
        if time.time() - start_time > budget_s - 10:
            print(f"Time left for final sync: {(budget_s - (time.time() - start_time)):.1f}s")
        return

    # Bound constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial point
    x0 = initial_h_values.copy()
    
    # Optimization loop
    evaluations = 0
    best_h = x0.copy()
    best_c5 = objective(x0)
    
    # Adaptive sweeping: optimize for max lag with gradient steps in feasible region
    while time.time() - start_time < budget_s - 50:
        # Compute current overlaps
        cx = np.correlate(best_h, 1 - best_h, mode='full')
        cx_scaled = cx * dx
        worst_lag = np.argmax(cx_scaled)
        hc = best_h
        upper = max(best_h[worst_lag % n_points], (1 - best_h[(2 * n_points - worst_lag) % n_points]))
        lower = min(best_h[worst_lag % n_points], (1 - best_h[(2 * n_points - worst_lag) % n_points]))
        adjustment = upper - lower

        # Perturb in direction of decreasing worst-case overlap
        if adjustment > 1e-5:
            grads = np.zeros(n_points)
            grads[worst_lag % n_points] = -1
            grads[(2 * n_points - worst_lag) % n_points] = 1
            grads = grads / np.linalg.norm(grads)
            learning_rate = 0.01
            perturbation = learning_rate * grads

            hc_perturbed = np.clip(hc + perturbation, 0, 1)
            hc_corrected = (hc_perturbed * n_points / 2) / np.sum(hc_perturbed)
            hc = project(hc_corrected)
        else:
            break

        c5_candidate = objective(hc)
        if c5_candidate < best_c5:
            best_h = hc.copy()
            best_c5 = c5_candidate

    print(f"Best found C5 bound: {best_c5:.6f}")
    return (best_h, best_c5, n_points)
