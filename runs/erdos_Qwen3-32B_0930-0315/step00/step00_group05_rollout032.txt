import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_best = np.array(initial_h_values)
    else:
        n_points = 70
        h_best = np.zeros(n_points)
        h_best[:35] = 1.0

    def compute_c5(h):
        dx = 2.0 / n_points
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial C5 value
    C5_best = compute_c5(h_best)

    def project(h):
        # Clip values to [0, 1]
        h = np.clip(h, 0, 1)
        # Ensure the sum constraint
        current_sum = np.sum(h)
        desired_sum = n_points / 2
        scale = desired_sum / current_sum
        h = h * scale
        return h

    # Optimization parameters
    learning_rate = 0.01
    iterations = 0
    improved = True

    while time.time() - start_time < budget_s and improved:
        iterations += 1
        improved = False
        h = cp.Variable(n_points)
        objective = cp.Minimize(cp.norm(cp.multiply(h, 1 - cp.roll(h, shift=30)) * (2.0/n_points), p=float('inf')))
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.SCS)
        h_sol = np.array(h.value).flatten()
        h_sol = project(h_sol)
        C5 = compute_c5(h_sol)

        if C5 < C5_best:
            h_best = h_sol
            C5_best = C5
            improved = True
            learning_rate *= 0.95

    return h_best.tolist(), C5_best, n_points
