```python
import numpy as np
import cvxpy as cp
import time
import numpy.random as npr

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    diff = 1.0 - h
    corr = np.correlate(h, diff, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100, n_points=80):
    start_time = time.time()
    max_iter = 100
    tolerance = 1e-5
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        assert np.allclose(np.sum(h), n_points / 2), "Initial h must satisfy area constraint"
    else:
        h = np.ones(n_points, dtype=np.float64) * 0.5
        h[::2] = 0.6
        h[1::2] = 0.4
        h = h * (n_points / 2 / np.sum(h))

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)

    h_var = cp.Variable(n_points)
    objective = cp.Minimize(0)
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2
    ]
    problem = cp.Problem(objective, constraints)

    while time.time() - start_time < budget_s:
        # Compute overlap at all lags
        dx = 2.0 / n_points
        diff = 1 - h
        corr = np.correlate(h, diff, mode='full') * dx
        max_c5 = np.max(corr)

        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h[:] = h.copy()

        # Early stopping condition
        if best_c5 < 0.380876:
            break

        k = np.argmax(corr)

        # Formulate objective for this lag k
        shifted_h = np.zeros(n_points)
        shifted_h[k:] = h[:n_points - k]
        objective.objective = cp.Minimize(cp.sum(h_var * (1 - np.roll(h_var, k))))
        problem.solve(solver=cp.ECOS, verbose=False)

        if problem.status != cp.OPTIMAL:
            problem.solve(solver=cp.CVXOPT, verbose=False)
            if problem.status != cp.OPTIMAL:
                continue

        h_proposed = h_var.value
        h_proposed = np.clip(h_proposed, 0, 1)
        h_proposed = h_proposed * (n_points / 2 / np.sum(h_proposed))

        h = h_proposed

        if max_c5 - evaluate_erdos_solution(h, n_points) < tolerance:
            break

    return (best_h.tolist(), best_c5, n_points)
```
