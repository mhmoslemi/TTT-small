import numpy as np
import time
import cvxpy as cp
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial = initial_h_values
    best_h = np.copy(initial)
    n_points = len(initial)
    dx = 2.0 / n_points
    best_c5 = np.inf
    tic = time.time()

    def compute_overlap(h):
        overlaps = np.zeros(n_points - 1)
        for k in range(1, n_points):
            idx = np.arange(n_points - k)
            overlaps[k - 1] = np.sum(h[idx] * (1 - h[idx + k])) * dx
        return max(overlaps)

    best_c5 = compute_overlap(best_h)

    h = np.copy(best_h)
    learning_rate = 0.01
    n_steps = 1000
    for step in range(n_steps):
        if time.time() - tic > budget_s:
            break
        # compute overlaps
        overlaps = np.zeros(n_points - 1)
        for k in range(1, n_points):
            idx = np.arange(n_points - k)
            overlaps[k - 1] = np.sum(h[idx] * (1 - h[idx + k])) * dx
        k_max = np.argmax(overlaps)

        # compute gradient for k_max
        grad = np.zeros(n_points)
        idx = np.arange(n_points - k_max)
        grad[idx] += (1 - h[idx + k_max])
        grad[idx + k_max] -= h[idx]
        grad *= dx

        # update step and constraints
        step_size = 0.999
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        excess = np.sum(h) - n_points / 2
        if abs(excess) > 1e-6:
            h -= excess / n_points

        current_c5 = compute_overlap(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.copy(h)
            learning_rate *= 1.01  # slight learning rate increase
        else:
            learning_rate *= 0.9  # schedule adaptation

    return (best_h, best_c5, n_points)
