```python
import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate


def evaluate_c5(h, dx):
    n_points = len(h)
    h1h = 1 - h
    overlaps = correlate(h, h1h, mode='full')
    overlaps = overlaps[n_points - 1 : 3 * n_points - 1]
    return np.max(overlaps * dx)


def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    max_iter = 2000
    tolerance = 1e-6
    n_points = 80
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        assert len(h) == n_points and (0 <= h).all() and (h <= 1).all()
        assert abs(h.sum() - n_points / 2) < 1e-8
    else:
        h = np.ones(n_points) / 2

    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)

    start_time = time.time()

    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        h1h = 1 - h
        overlaps = correlate(h, h1h, mode='full')
        overlaps = overlaps[n_points - 1 : 3 * n_points - 1]
        lag_index = np.argmax(overlaps) - n_points + 1
        k = (lag_index + n_points) % n_points

        # Shift indices to compute overlap at lag k
        offset = k
        indices = np.arange(n_points)
        i1 = indices
        i2 = (indices + offset) % n_points

        # Define variables
        dh = cp.Variable(n_points)
        obj = cp.Minimize(cp.sum((h[i1] + dh[i1]) * (1 - (h[i2] + dh[i2])) * dx))

        # Constraints
        constraints = [0 <= dh, dh <= 1 - h]
        constraints += [cp.sum(dh) == 0]  # Keep the sum fixed

        problem = cp.Problem(obj, constraints)
        problem.solve(solver=cp.ECOS, verbose=False)

        if problem.status != cp.OPTIMAL:
            continue  # Skip non-converged updates

        h += dh.value
        h = np.clip(h, 0, 1)
        h /= h.sum() * dx / 1.0  # Normalize to adjust back to ∫h = 1

        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5 - 1e-6:
            best_c5 = current_c5
            best_h = h.copy()
        elif abs(current_c5 - best_c5) < tolerance:
            # No significant improvement, terminate early
            if time.time() - start_time > budget_s - 10:
                break

    return best_h, best_c5, n_points
```
