import numpy as np
import time
import cvxpy as cp


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use the initial_h_values provided in the runtime context
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    # Make a GPU/acceleration-agnostic copy
    h = np.ascontiguousarray(h, dtype=np.float64)

    best_h = h.copy()
    best_c5_bound = np.inf

    # Gradient descent params
    lr = 0.01
    lr_decay = 0.95
    lr_decay_interval = 100
    iter_t0 = time.time()

    def project_feasible(h):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if not np.isclose(s, n_points / 2):
            h = h * (n_points / 2) / s
        return h

    def compute_c5(h):
        # Compute (1 - h) reversed for full convolution/correlation
        h_rev = 1 - h[::-1]
        corr_full = np.convolve(h, h_rev, mode="full")
        corr_full *= dx
        return np.max(corr_full)

    best_c5_bound = compute_c5(h)
    while time.time() - start_time < budget_s:
        # Compute cross-correlation max (forward direction only for equivalent h ⋆ (1 - h))
        h_rev = 1 - h
        corr_full = np.convolve(h, h_rev, mode="full")
        corr_full *= dx

        k_max = np.argmax(corr_full)
        window_half = h.size

        if k_max < window_half:
            h_shifted = h[k_max:]
        else:
            h_shifted = np.concatenate([h[k_max - window_half:], np.zeros(k_max - window_half)])

        grad = 1 - h_shifted
        h = project_feasible(h - lr * grad)

        curr_c5 = compute_c5(h)

        if curr_c5 < best_c5_bound:
            best_c5_bound = curr_c5
            best_h = h.copy()

        # Learning rate decay
        if (time.time() - iter_t0) > lr_decay_interval:
            lr *= lr_decay
            iter_t0 = time.time()

    return (best_h.tolist(), best_c5_bound, n_points)
