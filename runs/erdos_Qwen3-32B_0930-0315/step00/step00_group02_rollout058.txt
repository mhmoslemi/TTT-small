import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 160
        h = np.zeros(n_points, dtype=np.float64)
        mid = n_points // 2
        h[:mid] = 1.0
        h[mid:] = 0.0

    dx = 2.0 / n_points

    def scale_to_one(h):
        h_sum = h.sum()
        scale = (n_points / 2) / h_sum
        h *= scale
        np.clip(h, 0.0, 1.0, out=h)
        return h

    h = scale_to_one(h.copy())

    start_time = time.time()
    alpha = 0.01
    best_h = h.copy()
    best_c5 = float('inf')
    no_progress_steps = 0

    while time.time() - start_time < budget_s:
        # Compute correlation between h and 1 - h
        conv_full = correlate(h, 1 - h, mode='full')

        # Compute relevant slices with alignment correction
        middle_index = n_points
        conv = conv_full[middle_index - n_points + 1 : middle_index]
        conv *= dx

        # Find maximum overlap
        c5 = np.max(conv)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_progress_steps = 0
            alpha *= 1.05
        else:
            no_progress_steps += 1
            alpha *= 0.95

        # Stop if no progress
        if no_progress_steps > 100:
            break

        # Gradient update
        max_overlap_index = np.argmax(conv)
        shift = n_points - max_overlap_index  # Shifting for circular behavior

        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            j_reflect = (i + shift) % n_points
            grad[i] = (1 - h[j_reflect]) - h[i]

        h -= alpha * grad
        h = scale_to_one(h)

        # Occasionally perturb to escape local minima
        if np.random.rand() < 0.002 and time.time() - start_time < budget_s - 5:
            delta = np.random.rand(n_points) * 0.05
            h += delta
            h = scale_to_one(h)

    return list(best_h), best_c5, n_points
