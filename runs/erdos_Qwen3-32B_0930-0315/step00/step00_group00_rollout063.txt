import numpy as np

def evaluate_c5(h, dx):
    h_array = h.reshape(-1)
    h_flattened = h_array
    cross_corr = np.zeros(2 * h_flattened.shape[0])
    for k, hk in enumerate(h_flattened):
        shifted = np.roll(h_flattened, -k)
        shifted = np.where(np.roll(h_flattened, -k) < 0, 0, shifted)
        shifted = shifted[-k:] if k > 0 else shifted
        product = h_flattened * (1 - shifted)
        cross_corr[k] = np.sum(product) * dx
        product = h_flattened * (1 - shifted) if k < h_flattened.shape[0] else product
        if k < h_flattened.shape[0]:
            cross_corr[k + h_flattened.shape[0]] = np.sum(h_flattened[k:] * (1 - shifted[:h_flattened.shape[0] - k])) * dx
    return np.max(cross_corr)

def project_h(h, dx):
    h = np.clip(h, 0, 1)
    while not np.isclose(np.sum(h) * dx, 1.0, atol=1e-8):
        current_sum = h.sum()
        desired_sum = (1.0 / dx)
        h[h > 0] *= (desired_sum / current_sum)
    return h

def optimize_initial(initial_h, n_points):
    if not hasattr(optimize_initial, "called"):
        optimize_initial.called = True
        h0 = initial_h.copy()
        h0 = project_h(h0, 2.0 / n_points)
        print("Initial h projection done.")
        return h0
    else:
        raise Exception("optimize_initial already called with initial h.")

def run(seed=42, budget_s=1100, max_iter=10000, early_stop=1e-6, verbose=False):

    time_start = time.time()
    dx = 2.0 / 45

        corr = np.correlate(h_current, 1 - h_current, mode='full')

        ### Focus neighborhood
        neighborhood = 6  # Adjusted based on initial response scale

                h_candidate = h_current.copy()
                h_candidate[i] = max(0.0, min(1.0, h_candidate[i] + delta))

                C5_candidate = evaluate_c5(h_candidate, dx)

                    best_C5 = C5_candidate
                    if verbose:
