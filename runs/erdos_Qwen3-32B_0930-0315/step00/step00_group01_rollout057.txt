import time

    h = np.array(h_array, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points
    overlaps = np.correlate(h, 1 - h, mode="full")

    learning_rate = 0.2
    learning_decay = 0.9999
    perturbation_interval = 100
    max_iter = 100000

    def project_to_constraints(h):
        h_clipped = np.clip(h, 0, 1)
        total_clipped = np.sum(h_clipped)
        if total_clipped > 1e-8:
            h_clipped *= (n_points / 2) / total_clipped
        return h_clipped

    for iter in range(max_iter):
        overlaps = np.correlate(h, 1 - h, mode="full")
        overlaps = (overlaps[1:] * dx).tolist()
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]

        if max_overlap < best_c5:
            best_h[:] = h.copy()
            best_c5 = max_overlap

        if iter % 100 == 0:
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break

        # Compute gradient for k_max
        grad = np.zeros(n_points)
        for i in range(n_points - k_max):
            grad[i] += (1 - h[i + k_max]) / 100  # Scaling heuristically
            grad[i + k_max] -= h[i] / 100

        # Gradient step
        h -= learning_rate * grad
        h = project_to_constraints(h)

        # Decay learning rate
        learning_rate *= learning_decay

        # Perturbation
        if iter % perturbation_interval == 0:
            perturb = np.random.randn(n_points) * 0.05
            h += perturb
            h = project_to_constraints(h)

    elapsed_time = time.time() - start_time
    print(f"Optimization took {elapsed_time:.2f}s, best C5: {best_c5:.6f}")
    return (best_h.tolist(), best_c5, n_points)
