import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    # Compute cross-correlation between h and (1 - h)
    overlap = correlate(h, 1 - h, mode='full')
    # Only consider valid k where overlap[i] corresponds to a shift
    # Norm constant: dx * dx. Multiply total by 1/dx to scale back correctly
    return overlap * dx

def find_worst_lag(h, dx):
    # Compute the worst maximum overlap
    overlap = evaluate_overlap(h, dx)
    # Only consider lags in the valid range [-len(h)+1, len(h)-1]
    mid = len(h) - 1
    worst_lag = np.argmin(overlap[mid:])  # Max occurs at smallest index (best), but take minimal
    worst_value = overlap[mid + worst_lag]
    return worst_lag, worst_value

def gradient_step(h, dx, k, alpha):
    n = len(h)
    grad = np.zeros(n)
    # contribution from h[i] * (1 - h[i + k]) is (1 - h[i + k])
    for i in range(n - k):
        grad[i] += (1 - h[i + k]) * dx

    # contribution from h[i - k] * (1 - h[i]) is -h[i - k]
    for i in range(k, n):
        grad[i] += -h[i - k] * dx

    h_new = h - alpha * grad
    # Project to [0, 1]
    h_new = np.clip(h_new, 0, 1)
    # Normalize to maintain sum constraint
    total = np.sum(h_new)
    if total > 0:
        h_new *= n / 2 / total
    else:
        h_new[0] = 1  # Edge case to avoid zero array

    return h_new

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    global initial_h_values
    # Discretization and initial setup
    n_points = 70
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        assert h.shape == (n_points,)
    else:
        # Start with uniform profile
        h = np.ones(n_points) / 2
    h = np.copy(h)

    start_time = time.time()
    dx = 2.0 / n_points

    patience = 0
    best_c5 = np.inf
    best_h = h.copy()

    # Main optimization loop
    iteration = 0
    step_size = 0.1
    while time.time() - start_time < budget_s:
        iteration += 1

        # Compute worst overlap and its index
        k_worst, c5 = find_worst_lag(h, dx)

        # Update with gradient step
        h_new = gradient_step(h, dx, k_worst, step_size)

        # Apply projection
        h_new = np.clip(h_new, 0, 1)
        total = np.sum(h_new)
        if total > 0:
            h_new *= n_points / 2 / total
        else:
            h_new[0] = 1  # Edge case
            h_new *= n_points / 2 / np.sum(h_new)

        # Check for improvement
        new_c5 = c5
        k_worst_new, new_c5 = find_worst_lag(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1

        if patience > 20:
            # Reduce step size when improvement stalls and add perturbation
            step_size *= 0.9
            if step_size < 1e-3:
                break  # Converged
            # Simple perturbation
            perturbation = 0.01 * np.random.randn(n_points)
            h_new += perturbation
            # Normalize again
            h_new = np.clip(h_new, 0, 1)
            total = np.sum(h_new)
            if total > 0:
                h_new *= n_points / 2 / total
            patience = 0

        h = h_new

    # Return final solution
    return (best_h.tolist(), best_c5, n_points)
