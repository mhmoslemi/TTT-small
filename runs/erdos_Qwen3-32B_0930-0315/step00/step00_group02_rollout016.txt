import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    one_minus_h = 1 - h
    corr = correlate(h, one_minus_h, mode='full')
    corr = corr[n_points-1 : 2*n_points-1]
    return np.max(corr) * dx

def project_onto_constraints(h):
    h = np.clip(h, 0, 1)
    mean_h = (np.sum(h) / n_points)
    target_sum = n_points / 2
    scaling = target_sum / np.sum(h)
    h = h * scaling
    h = np.clip(h, 0, 1)
    return h

def gradient_descent_step(h, learning_rate, n_points):
    gradient = np.zeros_like(h)

    dx = 2.0 / n_points
    one_minus_h = 1 - h
    corr = correlate(h, one_minus_h, mode='full')
    top_k = np.argmax(corr)
    k_index = top_k - n_points + 1
    for i in range(n_points):
        grad_i = one_minus_h[(i - k_index) % n_points]
        gradient[i] += grad_i

    h_new = h - learning_rate * gradient
    return project_onto_constraints(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use 160 points for balance between precision and computation
    n_points = 160
    dx = 2.0 / n_points

    # Initialize h with plateau shape - zero at edges and constant in the middle
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        plateau_width = n_points // 2
        h = np.zeros(n_points)
        idx = np.arange(n_points)
        center = n_points // 2
        mask = (idx >= center - plateau_width // 2) & (idx < center + plateau_width // 2)
        h[mask] = 1.0 / np.count_nonzero(mask)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, n_points)

    # Phase 1: Gradient descent
    max_epochs = 1000
    learning_rate = 1e-1
    for epoch in range(max_epochs):
        if time.time() - start_time > budget_s * 0.8:
            break

        h = gradient_descent_step(h, learning_rate, n_points)
        c5 = compute_c5(h, n_points)

        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

        # Adapt learning rate (backtracking line search)
        if (epoch > 0 and epoch % 50 == 0):
            learning_rate = max(learning_rate / 2, 1e-4)

        # Early stopping if small improvement
        if (epoch > 10 and best_c5 - c5 < 1e-5):
            break

    # Phase 2: Refinement with cvxpy
    h_var = cp.Variable(shape=n_points)
    objective = cp.Maximize(0)  # Constrained optimization
    constraints = [
        cp.sum(h_var) == n_points / 2,
        0 <= h_var,
        h_var <= 1
    ]
    max_val = cp.Constant(0)
    for k in range(n_points - 1):
        h_copy = np.roll(h, -k)
        overlaps = h_var * (1 - h_copy)
        overlap_sum = cp.sum(overlaps) * dx
        max_val = cp.maximum(max_val, overlap_sum)

    # Use as a feasibility problem since global minimization is nonconvex
    objective = cp.Minimize(max_val)
    problem = cp.Problem(objective, constraints)
    problem.solve(solver='ECOS', verbose=False)

    # Get our final result
    if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        final_h = h_var.value.reshape(-1)
    else:
        final_h = best_h
        final_c5 = best_c5

    end_time = time.time()
    remaining_time = budget_s - (end_time - start_time)
    if remaining_time > 0:
        # Try a few random restarts if time allows
        for _ in range(2 if remaining_time > 30 else 0):
            if time.time() - start_time > budget_s - 5:
                break
            h = np.random.rand(n_points)
            h = project_onto_constraints(h)
            c5 = compute_c5(h, n_points)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

    c5_bound = compute_c5(best_h, n_points)
    return (best_h, c5_bound, n_points)
