import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    n_points = 70
    dx = 2.0 / n_points

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, copy=True)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h_scale = (n_points / 2.0) / h_sum
        h *= h_scale
    else:
        # Else use uniform distribution
        h = np.full(n_points, fill_value=0.5)

    alpha = 0.1
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    patience = 0
    improvement_threshold = 1e-7

    while time.time() - t0 < budget_s - 0.5:
        # Compute all overlaps
        corr = correlate(h, 1 - h, mode="full")
        valid_corr = corr * dx
        k_max = np.argmax(valid_corr[n_points - 1 : 2 * n_points - 1]) + n_points - 1 - 1

        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] += (1 - h[i + k_max])
            if i >= k_max:
                grad[i] -= h[i - k_max]
        grad *= dx

        # Update h
        h -= alpha * grad

        # Projection step
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h *= (n_points / 2.0) / h_sum

        # Perturb occasionally to escape local optima
        if np.random.rand() < 0.02:
            perturbation = np.random.normal(loc=0, scale=0.01, size=n_points)
            h += perturbation
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h *= (n_points / 2.0) / h_sum

        # Check for improvement
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5 - improvement_threshold:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
            if alpha > 1e-5:
                alpha = max(alpha * 0.5, 1e-5)
        else:
            patience += 1
            if patience > 50:
                h = best_h.copy()
                alpha = max(alpha * 0.95, 1e-5)
                patience = 0

    return best_h.tolist(), best_c5, n_points

def evaluate_c5(h, dx):
    h_full = h.copy()
    corr = correlate(h_full, 1 - h_full, mode="full")
    valid_corr = corr * dx
    c5 = np.max(valid_corr)
    return c5
