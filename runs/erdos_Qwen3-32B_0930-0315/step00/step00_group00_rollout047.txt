import numpy as np

import time

def project(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    s = np.sum(h_clipped)
    if not np.isclose(s, n_points / 2):
        h_clipped = h_clipped * (n_points / 2) / s
    return h_clipped

def compute_c5(h):
    dx = 2.0 / len(h)
    circ_cross = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(circ_cross)
    c5 = max_overlap * dx
    return c5
