import numpy as np
import math
from scipy.interpolate import CubicSpline
import time


def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if not np.all((0 <= h) & (h <= 1)):
        raise ValueError("h must be in [0, 1]")
    if np.abs(np.sum(h) - target_sum) > 1e-6:
        raise ValueError("Integral constraint violated")
    one_minus_h = 1.0 - h
    correlations = np.correlate(h, one_minus_h, mode="full") * dx
    return np.max(correlations)


def repair_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    # Rescale if needed
    if not np.isclose(current_sum, target_sum, atol=1e-6):
        scaling_factor = target_sum / current_sum
        shift = (1.0 - scaling_factor) * h
        h = scaling_factor * h
        h = np.clip(h, 0, 1)
    return h


def mutate(h, mutation_rate=0.2, seed=None):
    if seed is not None:
        np.random.seed(seed)
    step = np.random.normal(loc=0, scale=mutation_rate, size=len(h))
    new_h = h + step
    return repair_feasibility(new_h, len(h))


def crossover(parent1, parent2, seed=None):
    if seed is not None:
        np.random.seed(seed)
    n = len(parent1)
    point = np.random.randint(1, n)
    child = np.concatenate((parent1[:point], parent2[point:]))
    return repair_feasibility(child, n)


def tournament_selection(population, size=5):
    indices = np.random.choice(len(population), size=size)
    best_idx = np.argmin([p[1] for p in [population[i] for i in indices]])
    return population[indices[best_idx]]


def step_away(h, grad, lr, n_points):
    new_h = h - lr * grad
    return repair_feasibility(new_h, n_points)


def compute_gradient(h, k):
    n_points = len(h)
    dx = 2.0 / n_points
    # Find the lag idx where the max occurs
    correlations = np.correlate(h, 1 - h, mode="full") * dx
    max_idx = np.argmax(correlations)
    # Extract the integrand at that lag
    if max_idx < n_points:
        shift = max_idx
        one_minus_h_shifted = 1 - np.roll(h, shift)
    else:
        shift = n_points - (len(correlations) - max_idx)
        one_minus_h_shifted = 1 - np.roll(h, -shift)
    return one_minus_h_shifted
