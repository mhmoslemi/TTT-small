import numpy as np
import cvxpy as cp
import time
from numpy.random import RandomState


def run(seed=42, budget_s=1100, **kwargs):
    rs = RandomState(seed)
    np.random.seed(seed)
    start_time = time.time()

    # Initialize using the provided solution if available
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = initial_h_values.copy()
    else:
        n_points = 51  # default sample size
        h_init = np.zeros(n_points)
        idx = rs.choice(n_points, size=int(n_points / 2), replace=False)
        h_init[idx] = 1

    best_h = h_init
    best_c5 = evaluate_erdos_solution(best_h)
    dx = 2.0 / n_points

    # Formulate the SDP relaxation
    h = cp.Variable(n_points, nonneg=True)
    t = cp.Variable()

    constraints = [
        cp.sum(h) == n_points / 2,
        h <= 1
    ]

    max_corr = -float('inf')
    for k in range(1, n_points):
        h_k = np.roll(h, -k)
        corr = cp.sum(h * (1 - h_k[k:])) * dx
        constraints.append(corr <= t)
        max_corr = cp.Max(max_corr, corr)

    # Objective: minimize the maximum overlap integral
    obj = cp.Minimize(max_corr)

    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.CVXOPT, verbose=False)
    elapsed = time.time() - start_time
    if prob.status in ["infeasible", "unbounded"]:
        print("SDP infeasible/unbounded, using best so far")
    else:
        best_h_vals = np.clip(h.value, 0, 1)
        best_h_vals = best_h_vals / best_h_vals.sum() * n_points / 2
        best_h = best_h_vals
        best_c5 = evaluate_erdos_solution(best_h)
        print(f"SDP solved in {elapsed:.2f}s - C₅: {best_c5:.6f}")

    # Use remaining time for local refinement
    while elapsed < budget_s:
        opt_h = np.random.randn(n_points)
        opt_h /= np.linalg.norm(opt_h)
        opt_signs = np.sign(opt_h)
        opt_h *= n_points / 2 / opt_h.sum()  # enforces sum constraint
        opt_h = np.clip(opt_h, 0, 1)
        eval_c5 = evaluate_erdos_solution(opt_h)
        if eval_c5 < best_c5:
            best_h, best_c5 = opt_h, eval_c5
            print(f"New best C₅: {best_c5:.6f}")
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    return best_h, best_c5, n_points
