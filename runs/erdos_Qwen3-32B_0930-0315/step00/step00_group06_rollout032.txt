import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    # Initial setup
    dx = 2.0 / 95
    n_points = 95
    initial_h = initial_h_values
    np.random.seed(seed)

    def compute_overlap(h):
        h = h.astype(float)
        overlap = np.correlate(h, 1 - h, mode="full")
        overlap = overlap[n_points:]  # consider only relevant overlaps
        overlap *= dx
        return np.max(overlap)

    def normalize_h(h):
        total = np.sum(h)
        scale = n_points / 2 / total
        h *= scale
        h[h > 1.0] = 1.0
        h[h < 0.0] = 0.0
        return h

    current_h = np.copy(initial_h)
    best_h = current_h.copy()
    best_c5 = compute_overlap(current_h)

    start_time = time.time()
    max_iter = 1000

    for it in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        # Gradient-style direction towards minimizing max overlap
        outer = np.outer(current_h, 1 - np.roll(current_h, 1))
        grad = -outer[n_points:]  # negative of the current derivative signal
        grad = np.sum(grad, axis=1)

        direction = -grad
        step_size = 0.01 * dx  # small step to stay within [0, 1]

        # Update h with numerical gradient step
        current_h += step_size * direction
        current_h = normalize_h(current_h)

        current_c5 = compute_overlap(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()

        # Perturbation to escape local minima
        if it > 200 and np.random.rand() < 0.01:
            perturbation = np.random.rand(n_points) * 0.1
            perturbation = normalize_h(perturbation)
            current_h = 0.8 * current_h + 0.2 * perturbation
            current_h = normalize_h(current_h)

    return best_h, best_c5, n_points
