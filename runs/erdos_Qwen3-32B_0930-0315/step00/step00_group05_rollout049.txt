import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    start_time = time.time()

    # Get initial solution from parent, or use random if not available
    try:
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    except NameError:
        n_points = 70
        h_init = np.zeros(n_points)

    def evaluate_c5(h):
        dx = 2.0 / n_points
        h = h / h.std()
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    def project_h(h):
        """Enforce constraints by clipping and rescaling"""
        h = np.clip(h, 0, 1)
        total = h.sum()
        if abs(total - (n_points / 2)) > 1e-8:
            h = h * (n_points / 2) / total
        return h

    best_h = project_h(h_init)
    best_c5 = evaluate_c5(best_h)

    # Optimization loop
    for iter_num in range(1000):
        t1 = time.time()
        if t1 - start_time > budget_s - 1.0:
            break

        # Create optimization problem
        h = cp.Variable(n_points)
        obj = cp.Minimize(cp.norm(h - best_h, 2))  # Small step toward better solution
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2,
        ]
        problem = cp.Problem(obj, constraints)

        try:
            problem.solve(solver=cp.CVXOPT, verbose=False)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = np.array(h.value).flatten()
        except:
            continue

        h_proposal = project_h(h_new)
        c5_proposal = evaluate_c5(h_proposal)

        if c5_proposal < best_c5:
            best_h = h_proposal.copy()
            best_c5 = c5_proposal

    return (best_h, best_c5, n_points)
