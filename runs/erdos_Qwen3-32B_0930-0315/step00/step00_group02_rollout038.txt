import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    targets = 1.0 - h
    target_sum = n_points / 2.0

    # Initialize some tracking
    best_h = h.copy()
    best_c5 = np.inf
    iteration = 0
    lr = 0.01  # Initial learning rate

    max_iterations = 10000
    while time.time() - start_time < budget_s and iteration < max_iterations:
        # Calculate the cross-correlation with targets
        corr = correlate(h, targets, mode='full') * dx
        max_value_index = np.argmax(np.abs(corr))
        k_max = max_value_index - h.size + 1

        # Compute gradient at current k_max
        grad = np.zeros(n_points)
        for i in range(n_points):
            idx = (i + k_max) % n_points
            grad[i] = 1 - h[idx]

        grad[range(n_points)] += 1 - h[range(n_points)]

        # Gradient step
        h -= lr * grad

        # Projection onto feasible region [0,1]^n with fixed sum
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum != 0:
            h *= (target_sum / h_sum)

        # Compute new C₅ bound
        corr = correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(corr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Learning rate decay
        if iteration % 100 == 0:
            lr *= 0.98

        iteration += 1

    return best_h.tolist(), best_c5, n_points
