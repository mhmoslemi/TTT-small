import numpy as np
import scipy.signal
import time

from scipy.optimize import check_grad

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    start_time = time.time()
    max_time = budget_s

    n_points = 160
    initial = initial_h_values if 'initial_h_values' in globals() else (0.5 * np.ones(n_points))
    h = np.array(initial)

    # Normalize to sum to n_points / 2
    h = h * (n_points / 2.0 / h.sum())
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    def compute_c5(h):
        dx = 2.0 / n_points
        hh = (np.correlate(h, 1 - h, mode='full') * dx)
        return hh.max()

    def scale_h_to_integral(h):
        total = h.sum()
        target = n_points / 2.0
        h *= target / total
        h = np.clip(h, 0, 1)
        return h

    lr = 0.01
    early_stop = 0

    while time.time() - start_time < max_time:
        dx = 2.0 / n_points
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        max_index = corr.argmax()
        if max_index >= len(h):
            k = max_index - len(h)
        else:
            k = len(h) - max_index

        grad = np.zeros_like(h)
        for i in range(n_points):
            left = (i - k) % n_points
            right = (i + k) % n_points
            grad[i] = (1 - h[right]) - h[left]

        # Gradient ascent to reduce max overlap
        h -= lr * grad

        # Project back to feasible set
        h = np.clip(h, 0, 1)
        h = scale_h_to_integral(h)

        current_c5 = compute_c5(h)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h[:]
            early_stop = 0

        # Reduce learning rate if stalled
        early_stop += 1
        if early_stop >= 100:
            lr *= 0.95
            early_stop = 0

        if lr < 5e-4:
            break

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
