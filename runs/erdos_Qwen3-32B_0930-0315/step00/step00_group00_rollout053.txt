import numpy as np
import time
from scipy.signal import correlate

def evaluate_correlation_c5(h, n_points):
    dx = 2.0 / n_points
    ones_minus_h = 1.0 - h
    corr = correlate(h, ones_minus_h, mode='full')
    c5 = np.max(corr) * dx
    return c5

def build_neighbor(h, n_points, tabu_memory):
    h_list = h.tolist()
    candidate = h.copy()

    # Randomly perturb 1-5 elements
    n_perturb = np.random.randint(1, 6)
    idxs = np.random.choice(n_points, size=n_perturb, replace=False)
    for i in idxs:
        candidate[i] = np.random.rand()
    candidate = np.clip(candidate, 0, 1)
    candidate = (candidate / np.sum(candidate)) * (n_points / 2)
    h_tuple = tuple(candidate)

    # Wrap + shift for another possibility
    for shift in np.random.choice([1, 2, -1, -2], 1):
        candidate = np.roll(h, shift)
        candidate = np.clip(candidate, 0, 1)
        candidate = (candidate / np.sum(candidate)) * (n_points / 2)
        h_tuple = tuple(candidate)
        if h_tuple not in tabu_memory:
            return candidate, h_tuple
    if h_tuple not in tabu_memory:
        return candidate, h_tuple
    
    # If all else fails, return the perturbed one
    return candidate, h_tuple

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        # Initialize h with 50% total mass, binary type
        h = np.zeros(n_points)
        h[np.random.choice(n_points, size=n_points // 2, replace=False)] = 1.0
        h = h / np.sum(h) * (n_points / 2)  # Ensure exact constraint

    h = np.clip(h, 0, 1)
    best_h = h.copy()
    current_c5 = evaluate_correlation_c5(h, n_points)
    best_c5 = current_c5

    tabu_size = 15
    tabu_memory = set([tuple(h)])

    while time.time() - start_time < budget_s:
        neighbor, neighbor_tuple = build_neighbor(h, n_points, tabu_memory)

        candidate_c5 = evaluate_correlation_c5(neighbor, n_points)
        if candidate_c5 < best_c5:
            best_h = neighbor.copy()
            best_c5 = candidate_c5
            h = neighbor.copy()
            tabu_memory.add(neighbor_tuple)
            if len(tabu_memory) > tabu_size:
                tabu_memory.pop()
        elif candidate_c5 == best_c5:
            # Accept equally good candidate to avoid getting stuck
            h = neighbor.copy()
            tabu_memory.add(neighbor_tuple)
            if len(tabu_memory) > tabu_size:
                tabu_memory.pop()

    return best_h, best_c5, n_points
