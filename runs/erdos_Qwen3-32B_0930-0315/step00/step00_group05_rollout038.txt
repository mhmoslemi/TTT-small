import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    dx = 2.0 / len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)

    def project(h):
        h = np.clip(h, 0.0, 1.0)
        total = h.sum()
        scale = (len(h) / 2) / total
        h *= scale
        return h

    h = project(h)
    n_points = len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    best_time = time.time()

    step_size = 0.1
    tol = 1e-8
    it = 0
    warmup_steps = 1000

    def evaluate_c5(h, dx):
        one_minus_h = 1.0 - h
        result = scipy.signal.correlate(h, one_minus_h, mode='full')
        result = result * dx
        return float(np.max(result))

    while time.time() - start_time < budget_s - 0.1:
        it += 1
        if it % 100 == 0:
            print(f"Iteration {it}, time {time.time() - start_time}")

        t0 = time.time()
        one_minus_h = 1.0 - h
        result = scipy.signal.correlate(h, one_minus_h, mode='full')
        max_idx = np.argmax(result)
        max_val = result[max_idx]
        lag = max_idx - (n_points - 1)

        # Compute gradient
        grad = np.zeros_like(h)
        mask_lag = np.arange(n_points) + lag < n_points
        idxs = np.arange(n_points)[mask_lag]
        grad[idxs] += one_minus_h[mask_lag]
        grad += -2 * h

        h -= step_size * grad
        h = project(h)

        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            best_time = time.time()

        # Every 50 steps, adjust step size and add noise
        if it % 50 == 0:
            if it < warmup_steps:
                step_size = min(0.2, 0.1 * (1 + it / 100))
            else:
                step_size *= 0.95
            if it % 500 == 0:
                h = (h + 0.01 * np.random.default_rng(seed + it).normal(size=n_points)) / 1.0
                h = project(h)

        if time.time() - best_time > 50 and it > warmup_steps:
            h = (best_h + 0.02 * np.random.default_rng(seed + it).normal(size=n_points)) / 1.0
            h = project(h)
            best_time = time.time()

        # Break early when improvement stalls
        if max_val < best_c5 + 1e-4:
            break

    return best_h.tolist(), best_c5, n_points
