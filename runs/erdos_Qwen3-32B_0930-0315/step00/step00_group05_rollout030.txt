import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
import math

def evaluate_Ck(h, dx):
    h = h / h.sum() * h.size * dx
    ones_h = 1.0 - h
    result = correlate(h, ones_h, mode='full') * dx
    return np.max(result)

def compute_objective(h, dx, scale_penalty):
    constraint_penalty = (np.sum(h) - h.size / 2) ** 2
    Ck = evaluate_Ck(h, dx)
    return Ck + scale_penalty * constraint_penalty

def compute_gradient(h, dx, scale_penalty):
    n = len(h)
    Ck = evaluate_Ck(h, dx)
    _, imax = np.where(correlate(h, (1 - h), mode='full') * dx == Ck)[0][0]
    lag = imax  # Find the lag index where Ck is maximized
    grad = np.zeros_like(h)

    for i in range(n):
        for j in range(n):
            if i + lag < n:
                grad[i] += (1 - h[i + lag]) * dx
            if j - lag >= 0:
                grad[j] -= h[j - lag] * dx

    grad += 2 * scale_penalty * (np.sum(h) - h.size / 2)
    return grad

def perturb(h, scale=0.02):
    noise = np.random.uniform(-scale, scale, size=h.shape)
    h = np.clip(h + noise, 0.0, 1.0)
    h *= h.size / 2 / np.sum(h)
    return h

def optimize_C5(h0, n_points, dx, budget, maxiters=500, perturb_freq=50, scale_penalty=1e2):
    h_opt = h0.copy()
    t0 = time.time()
    start_time = time.time()

    def callback(x):
        nonlocal h_opt, t0
        C5 = evaluate_Ck(x, dx)
        if C5 < evaluate_Ck(h_opt, dx):
            h_opt = x.copy()
        if time.time() - start_time > budget - 5:  # Leave 5 sec padding
            return False
        return True

    # Adjust penalty scale based on initial violation
    penalty_scale = scale_penalty * (np.abs(np.sum(h_opt) - n_points / 2) > 1e-3)

    # L-BFGS-B optimization
    result = minimize(
        fun=lambda h: compute_objective(h, dx, penalty_scale),
        x0=h_opt,
        method="L-BFGS-B",
        bounds=[(0, 1)] * n_points,
        jac=lambda h: compute_gradient(h, dx, penalty_scale),
        callback=callback,
        options={"maxiter": maxiters, "ftol": 1e-8}
    )

    best_C5 = evaluate_Ck(h_opt, dx)

    # Iterative refinement with perturbations
    perturbed_time = 0.0
    perturb_idx = 0
    while time.time() - start_time + max(0, time.time() - t0) < budget:
        if perturb_idx % perturb_freq == 0 and perturb_idx > 0:
            h_opt = perturb(h_opt)
            t0 = time.time()

            method="L-BFGS-B",
            bounds=[(0, 1)] * n_points,
            jac=lambda h: compute_gradient(h, dx, penalty_scale),
            callback=callback,
            options={"maxiter": maxiters // 4, "ftol": 1e-8}
        )

        best_C5 = min(best_C5, evaluate_Ck(h_opt, dx))
        perturb_idx += 1

        if time.time() - start_time > budget - 5:
            break

    return h_opt, best_C5
