import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    n_points = 45 if 'initial_h_values' in globals() else 90
    dx = 2.0 / n_points
    dx_corr = dx  # For correlation-based C5 evaluation

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        # Normalize to unitary area in [0,2]
        initial_h *= (n_points / 2.0) / initial_h_sum
        # Clip to avoid numerical issues
        initial_h = np.clip(initial_h, 0, 1)
        best_h = initial_h
    else:
        best_h = np.random.rand(n_points)
        best_h = (best_h / best_h.sum()) * (n_points / 2.0)
        best_h = np.clip(best_h, 0, 1)

    best_c5 = evaluate_erdos_solution(best_h)

    current_time = time.time()
    start_time = current_time
    end_time = start_time + budget_s
    generations = 0
    max_generations = 1000
    population_size = 75
    tournament_size = int(0.1 * population_size)
    mutation_rate = 0.10
    sa_mutation_rate = 0.05
    crossover_prob = 0.70
    temp = 100
    temp_decay = 0.95
    sa_step = 50
    sa_interval = 50

    # Initialize population
    if 'initial_h_values' in globals():
        population = [initial_h]
    else:
        population = []

    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = (h / h.sum()) * (n_points / 2.0)
        h = np.clip(h, 0, 1)
        population.append(h)

    def evaluate_individual(h):
        return evaluate_erdos_solution(h)

    def tournament_selection(pop, scores, size):
        indices = np.random.choice(len(pop), size, replace=False)
        winner_idx = indices[np.argmin(scores[indices])]
        return pop[winner_idx]

    def project_to_simplex(v, s=1):
        # Adjust to ensure sum(h) = n_points / 2 and 0 <= h[i] <= 1
        v = np.clip(v, 0, 1)
        if np.sum(v) == 0:
            v[:] = 1e-6
        old_sum = np.sum(v)
        scaling_factor = s * 2 / (n_points * old_sum)
        v *= scaling_factor
        v = np.clip(v, 0, 1)
        return v

    def crossover(a, b):
        if np.random.rand() < crossover_prob:
            index = np.random.randint(1, len(a) - 1)
            offspring = np.concatenate([a[:index], b[index:]])
            return project_to_simplex(offspring)
        else:
            return (a + b) / 2  # Blend

    def mutate(hlinkedin, rate, use_sa=False):
        mut = hlinkedin.copy()
        num_muts = max(1, int(rate * len(mut)))
        indices = np.random.choice(len(mut), num_muts, replace=False)
        delta = np.random.uniform(-0.2, 0.2, num_muts)  # polarization limits to prevent spikes

        if use_sa:
            old_c5 = evaluate_individual(mut)
            mut[indices] += delta
            mut = project_to_simplex(mut)
            new_c5 = evaluate_individual(mut)
            if new_c5 > old_c5:
                if np.random.random() < np.exp((old_c5 - new_c5) / temp):
                    mut[indices] -= delta
        else:
            mut[indices] += delta
        return project_to_simplex(mut)

    # Evaluate initial population
    population = np.array(population)
    pop_scores = np.apply_along_axis(evaluate_individual, 1, population)

    while time.time() < end_time and generations < max_generations:
        generations += 1

        # Selection and reproduction
        new_population = []
        for _ in range(population_size):
            parent1 = tournament_selection(population, pop_scores, tournament_size)
            parent2 = tournament_selection(population, pop_scores, tournament_size)
            child = crossover(parent1, parent2)
            if np.random.rand() < mutation_rate:
                child = mutate(child, mutation_rate)
            new_population.append(child)

        new_population = np.array(new_population)
        # Evaluate new population
        new_scores = np.apply_along_axis(evaluate_individual, 1, new_population)

        # Merge and select top
        combined_pop = np.vstack((population, new_population))
        combined_scores = np.concatenate((pop_scores, new_scores))
        idx = np.argsort(combined_scores)
        population = combined_pop[idx[:population_size]]
        pop_scores = combined_scores[idx[:population_size]]

        # Simulated Annealing cycle
        if generations % sa_interval == 0 and generations > 0:
            best_idx = np.argmin(pop_scores)
            best_individual = population[best_idx]
            mutated_h = mutate(best_individual, mutation_rate, use_sa=True)
            temp *= temp_decay  # Lower the temperature

            # Replace worst if mutated is better
            new_c5_mutated = evaluate_individual(mutated_h)
            if new_c5_mutated < best_c5:
                population[-1] = mutated_h
                pop_scores[-1] = new_c5_mutated

        # Update best so far
        best_idx = np.argmin(pop_scores)
        if pop_scores[best_idx] < best_c5:
            best_h = population[best_idx]
            best_c5 = pop_scores[best_idx]

        # Early exit if we've already matched or beat record
        if best_c5 <= 0.38085:
            break

    return best_h.tolist(), best_c5, n_points
