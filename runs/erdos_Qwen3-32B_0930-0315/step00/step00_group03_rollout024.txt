import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70  # initial setup from parent construction (if available), small for faster eval
    dx = 2.0 / n_points

    # Check for initial_h_values from parent construction
    if 'initial_h_values' in globals():
        h_init = initial_h_values
    else:
        h_init = np.random.rand(n_points)
        h_init *= n_points / 2 / h_init.sum()

    def evaluate_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Population settings
    pop_size = 50
    population = [h_init * 1.0 for _ in range(pop_size)]

    def mutate(h, mutation_rate=0.15):
        h = h.copy()
        indices = np.random.choice(n_points, size=int(n_points * 0.05), replace=False)
        for i in indices:
            h[i] += np.random.normal(0, 0.1)
            h[i] = np.clip(h[i], 0, 1)
        scale = n_points / 2 / h.sum()
        h = np.clip(h * scale, 0, 1)
        return h

    def crossover(h1, h2):
        i = np.random.randint(n_points)
        child1 = np.concatenate([h1[:i], h2[i:]]) * 1.0
        child2 = np.concatenate([h2[:i], h1[i:]]) * 1.0
        for child in [child1, child2]:
            scale = n_points / 2 / child.sum()
            child *= scale
            child = np.clip(child, 0, 1)
        return child1, child2

    def tournament_selection(pop, k=5):
        candidates = np.random.choice(pop, size=k, replace=False)
        return max(candidates, key=evaluate_c5)

    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    while time.time() - start_time < budget_s:
        offspring = []
        for _ in range(pop_size // 2):
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            c1, c2 = crossover(parent1, parent2)
            o1 = mutate(c1)
            o2 = mutate(c2)
            offspring.extend([o1, o2])

        population = population + offspring
        population = sorted(population, key=evaluate_c5)[:pop_size]

        current_best = min(population, key=evaluate_c5)
        current_c5 = evaluate_c5(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5

    # Final check on constraints
    best_h = np.clip(best_h, 0, 1)
    best_h *= n_points / 2 / best_h.sum()

    return (best_h, best_c5, n_points)
