import numpy as np
import time
from numpy.random import default_rng
import evaluate_erdos_solution as ees
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    rng = default_rng(seed)
    
    # Initial construction
    initial_h = ees.initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def fitness(h):
        h_array = np.array(h)
        ones_minus_h = 1 - h_array
        corr = np.correlate(h_array, ones_minus_h, mode='full')
        c5 = np.max(corr) * dx
        return c5  # lower is better

    def constrain_sum(h, target_sum):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if abs(current_sum - target_sum) > 1e-6:
            scale_factor = target_sum / current_sum
            h_scaled = h_clipped * scale_factor
            h_scaled = np.clip(h_scaled, 0, 1)
            # residual
            excess = target_sum - np.sum(h_scaled)
            if abs(excess) > 0.0:
                indices = np.argsort(h_scaled)
                if excess < 0:
                    h_scaled[indices[:int(-excess)+1]] = 0
                else:
                    h_scaled[indices[-int(excess)-1:-1]] = 1
        return h_scaled
    
    population_size = 50
    pm = 0.15  # Mutation rate
    tournament_size = 5
    generations = int(budget_s / 1000)
    
    def tournament_selection(population, fitnesses):
        indices = np.random.choice(len(population), tournament_size)
        return population[indices[np.argmin(fitnesses[indices])]]
    
    def crossover(parent1, parent2):
        point = rng.integers(1, len(parent1) - 1)
        child1 = np.concatenate((parent1[:point], parent2[point:]))
        child2 = np.concatenate((parent2[:point], parent1[point:]))
        child1 = constrain_sum(child1, n_points / 2)
        child2 = constrain_sum(child2, n_points / 2)
        return child1, child2
    
    def mutate(individual):
        individual = individual.copy()
        prob = rng.random(len(individual)) < pm
        indices = np.where(prob)[0]
        if len(indices) > 0:
            individual[indices] += rng.normal(0, 0.1, size=len(indices))
            individual = np.clip(individual, 0, 1)
        individual = constrain_sum(individual, n_points / 2)
        return individual
    
    # Initialize population
    population = []
    population.append(initial_h)
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = constrain_sum(h, n_points / 2)
        population.append(h)
    
    # GA loop
    best_h = initial_h
    best_c5 = fitness(best_h)
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        offspring = []
        for _ in range(population_size // 2):
            parent1 = tournament_selection(population, [fitness(ind) for ind in population])
            parent2 = tournament_selection(population, [fitness(ind) for ind in population])
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            offspring.extend([child1, child2])
        
        population = population + offspring
        scores = [fitness(ind) for ind in population]
        joint = list(zip(population, scores))
        joint.sort(key=lambda x: x[1])
        population = [x[0] for x in joint[:population_size]]
        
        if joint[0][1] < best_c5:
            best_h = joint[0][0]
            best_c5 = joint[0][1]
        
        if gen % 100 == 0:
            pass
    
    # Final validation
    best_h = constrain_sum(best_h, n_points / 2)
    assert np.all((best_h >= -1e-6) & (best_h <= 1 + 1e-6))
    assert abs(np.sum(best_h) - n_points / 2) < 1e-6
    
    return best_h, best_c5, n_points
