import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

# --------------------
# Helper functions
# --------------------

def compute_c5(h, n_points):
    # Discretization step
    dx = 2.0 / n_points
    h1 = np.concatenate((h, np.zeros_like(h)))
    h2 = 1 - np.roll(h1, shift=1)
    corr = correlate(h1, h2, mode='full')
    max_corr = max(corr[:n_points]) * dx
    return max_corr

# --------------------
# SDP-based solver
# --------------------

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    # Use initial construction from parent process if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        n_points = len(initial_h_values)
        h_init = initial_h_values / np.sum(initial_h_values) * n_points / 2
    else:
        n_points = 79  # reasonable choice to balance accuracy and speed
        h_init = np.zeros(n_points)
        h_init[::2] = 1.0
        h_init /= np.sum(h_init) * 1.0
        h_init *= n_points / 2

    best_h = h_init.copy()
    best_c5 = float('inf')

    def optimize_with_sdp(h_start):
        nonlocal best_h, best_c5

        h = cp.Variable(n_points, nonneg=True)
        t = cp.Variable()

        # Vector of h(x+k) for each k
        def get_correlations(h_vec):
            overlaps = []
            for k in range(1, n_points):
                h_k = np.roll(h_vec, shift=-k)
                h_k = h_k[:n_points - k]
                overlap = cp.sum(h_vec[:n_points - k] * h_k)
                overlaps.append(overlap * (2.0 / n_points))
            return overlaps

        # Sample constraints: one constraint per k roll
        corr_constraints = []
        for k in range(1, n_points):
            roll_length = n_points - k
            h_k = h[:roll_length]
            h_k_k = h[k:k+roll_length]
            corr_constraints.append(cp.sum(h_k * (1 - h_k_k)) * (2.0 / n_points) <= t)

        # Global constraints on h
        constraints = corr_constraints + [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1
        ]
        objective = cp.Minimize(t)

        prob = cp.Problem(objective, constraints)
        try:
            result = prob.solve(warm_start=True, solver=cp.CVXOPT, verbose=False)
            h_sol = h.value
            t_opt = t.value
            if h_sol is None:
                return
            # Clamp to bounds
            h_sol = np.clip(h_sol, 0, 1)
            if abs(np.sum(h_sol) - n_points / 2) > 1e-6:
                h_sol /= np.sum(h_sol) * 2.0
                h_sol *= n_points / 2

            c5 = compute_c5(h_sol, n_points)
            if c5 < best_c5:
                best_h = h_sol.copy()
                best_c5 = c5
        except Exception as e:
            pass  # Skip failed runs

    # Perform a warm start with initial_h_values
    optimize_with_sdp(best_h)

    # Time remaining for enhanced search
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed

    # Sample-based search: RNG over small perturbations (10 samples max)
    if remaining > 20:
        for _ in range(min(10, int(remaining / 30))):
            h = best_h.copy()
            mask = np.random.rand(n_points) < 0.3
            h[mask] = np.random.rand(np.sum(mask)) * 0.5 + 0.5
            h = np.clip(h, 0, 1)
            if abs(np.sum(h) - n_points / 2) > 1e-6:
                h /= np.sum(h) * 2.0
                h *= n_points / 2
            optimize_with_sdp(h)
            if time.time() - start_time > budget_s:
                break

    # Final build and return
    c5_final = compute_c5(best_h, n_points)
    return (best_h, c5_final, n_points)
