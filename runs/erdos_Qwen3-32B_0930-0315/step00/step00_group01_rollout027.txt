def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    n_points = 47  # Start with provided construction
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():

    else:

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)

    learning_rate = 0.005
    patience_counter = 0
    max_patience = 1000
    min_c5_improvement = 0.00001  # Smallest acceptable improvement

    while time.time() - start_time < budget_s:
        overlaps = np.correlate(h, 1 - h, mode="full") * dx

        # Find the worst-performing (max) k
        k_max = np.argmax(overlaps)

        # Compute gradient w.r.t. h for this k_max
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] += 1 - h[i + k_max]
                grad[i + k_max] -= h[i]

        # Update h with projection
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)  # Projection to [0,1]
        # Project back into feasibility constraint: sum(h) == n_points / 2
        res = h.sum()
        if abs(res - n_points / 2) > 1e-8:
            scale = (n_points / 2) / res
            h = h * scale
            h = np.clip(h, 0, 1)  # After scaling, clip again

        # Evaluate the new h
        curr_c5 = evaluate_erdos_solution(h, n_points)
        if curr_c5 < best_c5 - min_c5_improvement:
            best_h = h.copy()
            best_c5 = curr_c5
            patience_counter = 0
        else:
            patience_counter += 1

        # Early stop if no progress for long enough
        if patience_counter > max_patience:
            break

        # Occasionally perturb to escape local optima
        if np.random.rand() < 0.01:
            h += 0.01 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = h * ((n_points / 2) / h.sum())

    return best_h, best_c5, n_points
