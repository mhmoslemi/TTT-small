import numpy as np
import time
import scipy.optimize
import cvxpy as cp


def compute_overlap(h, dx):
    n_points = len(h)
    max_lag = n_points - 1
    full_corr = np.correlate(h, 1 - h, mode="full")
    full_corr = full_corr[n_points - 1 : 2 * n_points - 1]
    c5_bound = np.max(full_corr * dx)
    return c5_bound


def project_to_constraints(h):
    h = np.clip(h, 0, 1)
    h *= h.size / (2 * h.sum())
    return h


def gradient(h, k, dx):
    n_points = len(h)
    grad = np.zeros(n_points)
    for i in range(n_points - k):
        grad[i] += 1 - h[i + k]
        grad[i + k] -= h[i]
    return grad * dx


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    h = project_to_constraints(initial_h_values.copy())
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_overlap(best_h, dx)
    learning_rate = 0.1
    patience = 500
    no_progress = 0

    while time.time() - start_time < budget_s - 1:
        # Compute overlaps
        overlaps = np.correlate(h, 1 - h, mode="full")[n_points - 1 : 2 * n_points - 1] * dx
        worst_k = np.argmax(overlaps)

        # Compute gradient
        grad = gradient(h, worst_k, dx)

        # Update h
        h -= learning_rate * grad
        h = project_to_constraints(h)

        # Check improvement
        new_c5 = compute_overlap(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            no_progress = 0
            learning_rate = min(learning_rate * 1.05, 0.5)
        else:
            no_progress += 1
            if no_progress > patience:
                # Perturb slightly to escape local optima
                h += np.random.normal(0, 0.05, h.shape)
                h = project_to_constraints(h)
                no_progress = 0

        # Reduce learning rate slowly
        if new_c5 >= best_c5:
            learning_rate *= 0.99

    # Final projection
    best_h = project_to_constraints(best_h)
    final_c5 = compute_overlap(best_h, dx)
    return best_h, final_c5, n_points
