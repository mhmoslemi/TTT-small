import numpy as np
import time
from numpy.random import default_rng
from functools import partial
import evaluate_erdos_solution

def evaluate_c5(h_vals, n_points):
    h = h_vals.copy()
    dx = 2.0 / n_points
    h_sum = np.sum(h)
    # Enforce sum constraint
    if not np.isclose(h_sum, n_points / 2, atol=1e-6):
        h = np.clip(h, 0, 1)
        if h_sum == 0:
            h = np.zeros(n_points)
            h[0] = n_points / 2
        else:
            h = (h / h_sum) * (n_points / 2)
    H = 1 - h
    c5 = np.max(np.abs(np.correlate(h, H, mode='full'))) * dx
    return c5

def project_to_feasible(h_vals, n_points):
    h = h_vals.copy()
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum == 0:
        h = np.zeros(n_points)
        h[0] = n_points / 2
    else:
        h = (h / h_sum) * (n_points / 2)
    return h

def tournament_selection(population, fitness, tournament_size):
    indices = np.random.choice(len(population), tournament_size)
    idx = np.argmin(fitness[indices])
    return indices[idx]

def crossover(parent1, parent2):
    n_points = len(parent1)
    idx = np.random.randint(1, n_points - 1)
    child = np.concatenate([parent1[:idx], parent2[idx:]])
    return project_to_feasible(child, n_points)

def mutate(h_vals, n_points, temp):
    h = h_vals.copy()
    noise = np.random.randn(n_points) * temp
    h += noise
    h = h * np.random.uniform(0, 1, n_points)
    return project_to_feasible(h, n_points)

def run(seed=42, budget_s=1100, **kwargs):
    rng = default_rng(seed)
    n_points = 45
    pop_size = 75
    tournament_size = int(0.1 * pop_size)
    
    if 'initial_h_values' in globals():
        init = [globals()['initial_h_values']]
    else:
        h_start = np.zeros(n_points)
        h_start[0] = n_points / 2
        init = [h_start]

    population = np.array(init)
    
    # Initialize population with random feasible solutions
    while len(population) < pop_size:
        h = rng.uniform(0, 1, n_points)
        h = project_to_feasible(h, n_points)
        if not any(np.allclose(h, p, atol=1e-6) for p in population):
            population = np.vstack([population, h])

    start_time = time.time()
    best_h = population[0].copy()

    T = 100
    cooling_rate = 0.95
    temp = T
    sa_interval = 50
    generations = 0

    best_obj = evaluate_c5(best_h, n_points)

    while time.time() - start_time < budget_s - 0.1:
        for _ in range(pop_size):
            # Evaluate population
            fitness = np.array([evaluate_c5(p, n_points) for p in population])

            # Selection and Crossover
            parent1_idx = tournament_selection(population, fitness, tournament_size)
            parent2_idx = tournament_selection(population, fitness, tournament_size)
            parent1, parent2 = population[parent1_idx], population[parent2_idx]
            if rng.rand() < 0.7:
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()

            # Mutation
            if rng.rand() < 0.05:
                child = mutate(child, n_points, temp)

            # Feasibility projection
            child = project_to_feasible(child, n_points)
            child_obj = evaluate_c5(child, n_points)

            # Replace worst in population
            worst_idx = np.argmax(fitness)
            population[worst_idx] = child

            # Track best
            if child_obj < best_obj:
                best_h = child
                best_obj = child_obj

        # Simulated Annealing Step
        if generations % sa_interval == 0:
            new_h = mutate(best_h, n_points, temp / 2)
            new_obj = evaluate_c5(new_h, n_points)
            if new_obj < best_obj or np.exp((best_obj - new_obj) / temp) > rng.rand():
                best_h = new_h
                best_obj = new_obj

        temp *= cooling_rate
        generations += 1

    return best_h, best_obj, n_points
