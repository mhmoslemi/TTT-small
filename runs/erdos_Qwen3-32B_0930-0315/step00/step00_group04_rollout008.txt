import numpy as np
import time
from scipy.interpolate import interp1d

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    n_points_list = [14, 28, 56]

    # Use initial_h_values from parent if available
    if 'initial_h_values' in globals():
        parent_h = np.array(initial_h_values)
        n_points = parent_h.size
    else:
        # Generate starting guess
        def initial_guess(n_points):
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * (n_points / 2)
            return h
        n_points = 14

    for n_current in n_points_list:
        if best_h is None:
            # Start with initial guess
            h = initial_guess(n_current) if "initial_h_values" not in globals() else (
                interp1d(np.linspace(0, 2, len(initial_h_values)), initial_h_values, kind="cubic")(
                    np.linspace(0, 2, n_current))
            )
        else:
            # Refine by upsampling
            h = interp1d(np.linspace(0, 2, best_h.size), best_h, kind="cubic")(np.linspace(0, 2, n_current))

        # Repair to ensure constraints
        def repair_feasibility(h, n_points_required):
            h = np.clip(h, 0, 1)
            h = h / h.sum() * (n_points_required / 2)
            return h

        h = repair_feasibility(h, n_current)
        dx = 2.0 / n_current

        def compute_overlap(h):
            c_vals = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(c_vals), np.argmax(c_vals)

        # Simulated Annealing
        T = 1.0
        alpha = 0.99
        sa_steps = 1000 if n_current < 56 else 800
        for sa_step in range(sa_steps):
            if time.time() - start_time >= budget_s:
                break

            h_candidate = h + np.random.normal(0, T, h.shape)
            h_candidate = repair_feasibility(h_candidate, n_current)
            c_current, _ = compute_overlap(h)

            c_candidate, _ = compute_overlap(h_candidate)
            delta = c_candidate - c_current

            if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-8)):
                h = h_candidate
                c_current = c_candidate

            if c_current < best_c5:
                best_c5 = c_current
                best_h = h.copy()

            T *= alpha

        # Gradient Descent
        grad_steps = 1000 if n_current < 56 else 600
        learning_rate = 0.01
        for _ in range(grad_steps):
            if time.time() - start_time >= budget_s:
                break

            c_vals = np.correlate(h, 1 - h, mode='full') * dx
            k_max = np.argmax(c_vals)

            shift = k_max - (n_current - 1)
            h_shifted = np.roll(h, shift)
            gradient = (1 - h_shifted)

            h -= learning_rate * gradient
            h = repair_feasibility(h, n_current)

            c_current, _ = compute_overlap(h)
            if c_current < best_c5:
                best_c5 = c_current
                best_h = h.copy()

    return (best_h, best_c5, best_h.size)
