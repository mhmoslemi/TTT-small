import numpy as np
from time import perf_counter

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    # Use the same number of points as the initial construction if available
    if 'initial_h_values' in globals():
        h = np.copy(initial_h_values)
        n_points = len(h)
    else:
        n_points = 47  # use the same number as in the parent if not provided
        h = np.random.rand(n_points)

    dx = 2.0 / n_points
    learning_rate = 0.02
    cooling_rate = 0.9999
    min_lr = 1e-4

    # Normalize h to satisfy the constraint ∑h = n_points / 2
    def project(h):
        h[h > 1.0] = 1.0
        h[h < 0.0] = 0.0
        h *= (n_points / 2) / h.sum()
        return h

    h = project(h)

    start_time = perf_counter()
    best_h = np.copy(h)
    best_c5 = np.inf

    while perf_counter() - start_time < budget_s:
        # Compute cross-correlation of h with (1-h)
        corr = np.correlate(h, 1 - h, mode='full')
        overlap = corr * dx

        # Find the lag k with the maximum overlap
        k_max = np.argmax(overlap)
        
        if k_max >= n_points:
            k_max = n_points * 2 - 1 - k_max

        # Compute gradient for this bad shift
        grad = np.zeros_like(h)
        for i in range(n_points - k_max):
            grad[i] += 1 - h[i + k_max]
        for i in range(n_points - k_max, n_points):
            grad[i] -= h[i - (n_points - k_max)]

        # Take a gradient step
        h -= learning_rate * grad
        h = project(h)

        # Evaluate the result
        corr = np.correlate(h, 1 - h, mode='full')
        overlap = corr * dx
        c5 = np.max(overlap)

        # Update learning rate and best solution
        learning_rate *= cooling_rate
        learning_rate = np.max([learning_rate, min_lr])

        if c5 < best_c5:
            best_h = np.copy(h)
            best_c5 = c5

    return (best_h, best_c5, n_points)
