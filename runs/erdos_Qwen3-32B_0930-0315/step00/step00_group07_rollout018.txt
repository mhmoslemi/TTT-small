import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h)

    # Ensure h is feasible
    dx = 2.0 / n_points
    h *= (n_points / 2.0) / np.sum(h)
    h = np.clip(h, 0.0, 1.0)

    def objective(h_vec):
        h = np.clip(h_vec.reshape(n_points), 0.0, 1.0)
        corr = correlate(h, 1.0 - h, mode='full')
        max_overlap = np.max(corr * dx)
        return max_overlap

    def constraint_sum_h(h_vec):
        h = h_vec.reshape(n_points)
        return np.sum(h) - n_points / 2

    constraints = (
        {'type': 'eq', 'fun': constraint_sum_h},
    )
    bounds = [(0.0, 1.0)] * n_points

    best_h = h.copy()
    best_c5 = objective(h)
    best_c5_time = -1.0

    # Run gradient-based optimization for a fixed amount of time
    while time.time() - start_time < budget_s * 0.75:
        res = minimize(
            objective,
            x0=best_h.copy().ravel(),
            bounds=bounds,
            constraints=constraints,
            method='SLSQP'
        )
        new_h = np.clip(res.x.reshape(n_points), 0.0, 1.0)
        new_c5 = objective(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_c5_time = time.time() - start_time
            best_h = new_h

    print(f"Best C5: {best_c5} at {best_c5_time}s")

    # Ensure feasibility one last time
    h = best_h.copy()
    h *= (n_points / 2.0) / np.sum(h)
    h = np.clip(h, 0.0, 1.0)
