import numpy as np
import time
from numpy.random import default_rng
import evaluate_erdos_solution


def run(seed=42, budget_s=1100, **kwargs):
    rng = default_rng(seed)
    if "initial_h_values" in globals():
        h_init = initial_h_values
        n_points = len(h_init)
        dx = 2.0 / n_points
    else:
        n_points = 70
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0
        h_init = h_init + 0.1 * rng.standard_normal(h_init.shape)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init / h_init.sum() * n_points / 2
        h_init = np.round(h_init, 3)

    start_time = time.time()
    dx = 2.0 / n_points

    popsize = 50
    mutation_rate = 0.15
    n_best = popsize // 2
    population = [h_init + 0.1 * (np.random.rand(n_points) - 0.5) for _ in range(popsize)]
    for h in population:
        h = np.clip(h, 0, 1)
        h = h / h.sum() * n_points / 2

    def evaluate(h):
        hcorr = evaluate_erdos_solution(h)
        return float(np.max(hcorr))  # lower is better

    def repair(h):
        repair_delta = n_points / 2 - h.sum()
        if abs(repair_delta) < 1e-6:
            return h
        h_new = h.copy()
        if repair_delta > 0:
            idx = np.argpartition(h, -int(repair_delta), axis=0)[-int(repair_delta):]
            h_new[idx] = np.clip(h_new[idx] + 1, 0, 1)
        else:
            idx = np.argpartition(h, int(-repair_delta), axis=0)[:int(-repair_delta)]
            h_new[idx] = np.clip(h_new[idx] - 1, 0, 1)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / h_new.sum() * n_points / 2
        return h_new

    best_h = h_init
    best_fitness = evaluate(best_h)
    while time.time() - start_time < budget_s:
        scores = [(evaluate(h), h) for h in population if h.shape[0] == n_points]
        scores.sort()
        population = [h for _, h in scores[:n_best]]

        while len(population) < popsize:
            parent1 = rng.choice(population)
            parent2 = rng.choice(population)
            crossover_point = rng.integers(1, n_points, size=1)[0]
            child1 = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            child2 = np.concatenate((parent2[:crossover_point], parent1[crossover_point:]))

            for c in [child1, child2]:
                c = np.clip(c, 0, 1)
                c = c / c.sum() * n_points / 2
                if rng.random() < mutation_rate:
                    noise_indices = rng.choice(n_points, size=max(1, int(n_points * 0.05)), replace=False)
                    mutation = 0.1 * (rng.standard_normal(len(noise_indices)) - 0.5)
                    c[noise_indices] = np.clip(c[noise_indices] + mutation, 0, 1)
                    c = c / c.sum() * n_points / 2
                c = repair(c)
                population.append(c)

        for score, h in scores[:n_best]:
            if score < best_fitness:
                best_h = h
                best_fitness = score

    h_final = repair(best_h)
    c5_bound = evaluate(h_final)
    return h_final, c5_bound, n_points
