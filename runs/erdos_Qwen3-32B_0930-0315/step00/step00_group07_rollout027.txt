import numpy as np
import cvxpy as cp
import time


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Initial configuration
    h = np.array(initial_h_values, dtype=np.float64)  # Best known h
    n_points = len(h)
    dx = 2.0 / n_points

    def evaluate_c5(h_values):
        h = h_values
        diffs = h[:, np.newaxis] - h[np.newaxis, :]
        return np.max(np.trapz(diffs * (1 - diffs), dx=dx, axis=1))

    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()
    max_iter = 200

    for iter_num in range(max_iter):
        current_time = time.time()
        if current_time - start_time >= budget_s:
            break

        # Step 1: Compute overlap for all lags using convolution-like operation
        h_ext = np.concatenate([h, h, h])
        overlap = np.zeros(n_points * 3)
        for k in range(1, n_points):
            # The k-th lag corresponds to the shifted correlations
            shifted = h_ext[k : k + n_points]
            corr = h * (1 - shifted)
            overlap[k] = np.sum(corr) * dx

        worst_k = np.argmax(overlap)

        # Step 2: Compute the gradient of overlap at this lag
        def sign(x):
            return (x > 0) - (x < 0)

        grad = np.zeros(n_points)
        for i in range(n_points):
            i1 = (i + worst_k) % n_points
            i2 = (i - worst_k) % n_points
            grad[i] = 1.0 - h[i1] - h[i2]

        grad *= dx

        # Step 3: Solve linear program for update
        delta = cp.Variable(n_points)
        obj = cp.Minimize(overlap[worst_k] + grad @ delta)
        constraints = [
            0 <= h + delta,
            h + delta <= 1,
            cp.sum(delta) == 0,
        ]
        problem = cp.Problem(obj, constraints)
        problem.solve(solver=cp.ECOS, verbose=False)

        # Apply update if feasible
        if problem.status != "optimal":
            continue

        h += delta.value
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    # Ensure final h satisfies constraints
    h = best_h
    h[h < 0] = 0.0
    h[h > 1] = 1.0
    h *= (n_points / 2.0) / h.sum()

    return h.tolist(), best_c5, n_points
