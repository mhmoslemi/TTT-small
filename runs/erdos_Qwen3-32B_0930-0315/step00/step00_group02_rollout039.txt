import numpy as np
import math
from scipy.signal import correlate
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s

    # Extract and wrap initial_h_values as a numpy array if it exists
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        # Starting construction: plateau of 0.5
        n_points = 45
        h = np.full(n_points, 0.5, dtype=np.float64)

    # Enforce mass constraint exactly
    total_mass = 2 * (n_points // 1)  # should be 2
    h *= total_mass / (2 * h.sum())

    dx = 2.0 / n_points

    def project_to_mass_one(h):
        # Clip values to [0,1]
        h_clipped = np.clip(h, 0, 1)
        # Normalize to have sum = n_points/2 = total mass 1
        total = h_clipped.sum()
        if np.isclose(total, 0):
            return h
        h_scaled = h_clipped * (n_points / 2) / total
        return np.clip(h_scaled, 0, 1)

    def evaluate(h):
        h_one = np.ones_like(h)
        h_comp = h_one - h
        corr = correlate(h, h_comp, mode='full')
        return np.max(corr) * dx

    best_c5 = evaluate(h)
    best_h = np.copy(h)

    # Learning parameters
    learning_rate = 0.01
    max_iterations = 100000
    iteration = 0
    last_improvement = 0

    try:
        while iteration < max_iterations and time.time() - start_time < max_time:
            iteration += 1

            # Compute current value of overlap correlation
            c5 = evaluate(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = np.copy(h)
                last_improvement = 0
            else:
                last_improvement += 1

            # If no improvement for 100 iterations, decay learning rate
            if last_improvement >= 100:
                learning_rate *= 0.9
                last_improvement = 0
                if learning_rate < 1e-6:
                    break

            # Find the k that gives the maximum overlap
            h_one = np.ones_like(h)
            h_comp = h_one - h
            corr = correlate(h, h_comp, mode='full')
            k_max = np.argmax(corr)

            # Compute gradient for shift k_max
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_idx = (i - k_max) % n_points
                grad[i] += 1 - h[h_idx]
                h_idx = (i + k_max) % n_points
                grad[i] -= 1 - h[h_idx]
            grad *= -1

            # Gradient step
            h = project_to_mass_one(h + grad * learning_rate)

        # Final evaluation and constraint check
        final_c5 = evaluate(project_to_mass_one(h))
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h.copy()

        # Ensure the final h is valid
        best_h = project_to_mass_one(best_h)

        return (best_h.tolist(), best_c5, n_points)

    except Exception as e:
        print(f"Error encountered: {e}")
        return (best_h.tolist(), best_c5, n_points)
