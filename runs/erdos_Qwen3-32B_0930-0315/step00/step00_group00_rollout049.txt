import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    vals = np.correlate(h, 1 - h, mode='full')
    c5_val = float(np.max(vals) * dx)
    return c5_val, vals

def project_simplex(v, z=1.0, axis=None):
    if axis is None:
        v = np.reshape(v, -1)
        p = _project_simplex_nd(v, z)
        return p.reshape(v.shape)
    else:
        return np.apply_along_axis(lambda x: _project_simplex_nd(x, z), axis=axis, arr=v)

def _project_simplex_nd(v, z=1.0):
    u = np.sort(v, axis=-1)[...,::-1]
    u = (u.cumsum(axis=-1) - z) / np.arange(1, u.shape[-1]+1, dtype=np.float64)
    u = u[...,np.newaxis]
    indicator = (u > v[...,np.newaxis])
    mu = np.take_along_axis(u, np.argmax(indicator, -1)[...,np.newaxis], -1)[...,0]
    return np.maximum(v[...,np.newaxis] - mu, 0).sum(-1)
