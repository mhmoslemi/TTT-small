        corr = correlate(h, (1 - h), mode="full") * dx

        # Valid lags from 1 to n_points-1

    bounds = [(0.0, 1.0) for _ in h]

    def perturb_func():
        while True:

    best_c5 = obj(best_h)
    step = 0
    perturb = perturb_func()

    # Optimization loop with limited time budget

        def objective_with_penalty(x):

            objective_with_penalty,

            method="L-BFGS-B",
