import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    h = np.array(h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    ones_h = 1 - h
    corrs = correlate(h, ones_h, mode='full')
    c5 = np.max(corrs) * dx
    return float(c5)

    init_h = init_h / np.sum(init_h) * len(init_h) / 2

    n_points = 70  # Start with the initial discretization
    best_h = init_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    max_iter = 200
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Convex optimization attempt to minimize max overlap
        h = cp.Variable(n_points, nonneg=True)
        constraints = [cp.sum(h) == n_points / 2, h <= cp.Constant(1), h >= 0]
        
        # Construct the target objective as minimizing max overlap
        corr_values = []
        for k in range(1, n_points):
            left = h[:n_points - k]
            right = 1 - h[k:]
            corr_values.append(cp.sum(left * right))
        for k in range(1, n_points):
            left = h[k:]
            right = 1 - h[:n_points - k]
            corr_values.append(cp.sum(left * right))
        
        obj = cp.Minimize(cp.Max(corr_values))
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=150, verbose=False)
            if h.value is not None:
                new_h = np.clip(h.value, 0, 1)
                # Rescale to ensure constraint is satisfied
                total = np.sum(new_h)
                if abs(total - n_points / 2) > 1e-6:
                    new_h = new_h / total * (n_points / 2)
                new_c5 = evaluate_erdos_solution(new_h)
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
        except Exception:
            pass

        # Add random perturbation to escape local minima
        perturb = np.random.normal(0, 0.05, size=n_points)
        perturbed_h = np.clip(best_h + perturb, 0, 1)
        perturbed_h = (perturbed_h / np.sum(perturbed_h)) * (n_points / 2)
        perturbed_c5 = evaluate_erdos_solution(perturbed_h)
        if perturbed_c5 < best_c5:
            best_h = perturbed_h.copy()
            best_c5 = perturbed_c5

    return (best_h.tolist(), best_c5, n_points)
