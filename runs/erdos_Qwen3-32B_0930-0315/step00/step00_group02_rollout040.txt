```python
import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Discretization
    n_points = 160
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        h /= h.sum() * dx * 2
    else:
        # Construct a plateau-like starting guess
        x = np.linspace(0, 2, n_points)
        h = np.clip(0.5 + np.sin(2 * np.pi * x), 0, 1)
        h /= h.sum() * dx * 2
    h *= target_sum / h.sum()
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = float('inf')
    alpha = 0.01
    start = time.time()
    patience = 0  # Counter to break if no improvement for 100 iterations
    max_patience = 100

    while time.time() - start < budget_s:
        # Compute overlap (correlation)
        corr = correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        k = idx - n_points + 1

        # Compose gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_k = (i + k) % n_points
            grad[i] = (1 - h[i_k]) - h[i]  # See explanation of gradient (cost decrease from h[i], h[i-k])

        # Gradient update
        h = h - alpha * grad

        # Projection to [0,1]
        h = np.clip(h, 0, 1)

        # Constrain to sum(h) == n_points / 2
        s = h.sum()
        if abs(s - target_sum) > 1e-2:  # Small threshold to avoid excessive corrections
            h *= target_sum / s

        # Evaluate C5
        corr = correlate(h, 1 - h, mode='full') * dx
        c5 = np.max(corr)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            alpha *= 0.5
            patience = 0

        # Random perturbation to escape local minima
        if np.random.rand() < 0.001 and (time.time() - start) < budget_s - 10:
            perturbation = np.random.normal(0, 0.02, n_points)
            h += perturbation
            h = np.clip(h, 0, 1)
            h *= target_sum / h.sum()

    return best_h, best_c5, n_points
```
