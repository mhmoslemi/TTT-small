import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()

    max_iter = 1000
    no_improvement_counter = 0

    for iter_idx in range(max_iter):
        # Compute all overlaps
        h_one_minus_h = 1.0 - h
        corr = np.correlate(h, h_one_minus_h, mode='full') * dx
        curr_max_index = np.argmax(np.abs(corr))
        if curr_max_index >= n_points:
            k = curr_max_index - n_points
        else:
            k = curr_max_index

        # Compute gradient for this k
        G = np.zeros(n_points)
        for i in range(n_points):
            if (i + k) < n_points:
                G[i] += (1 - h[i + k])
            # Avoid wrapping across the edge (since function is zero outside of [0, 2])
            j = i + k
            if j < n_points:
                G[j] -= h[i]

        G = G / np.linalg.norm(G)  # normalize gradient

        # Learning rate with schedule
        alpha = 0.1 / (1 + iter_idx / 500)
        h_candidate = h - alpha * G

        # Projection step: minimize ||h_new - h_candidate|| s.t. h ∈ [0,1]^n, sum(h_new) = n_points / 2
        h_proj = cp.Variable(n_points)
        objective = cp.Minimize(cp.norm(h_proj - h_candidate))
        constraints = [0 <= h_proj, h_proj <= 1, cp.sum(h_proj) == n_points / 2]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.GUROBI, verbose=False)
        except:
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except:
                prob.solve(solver=cp.OSQP, verbose=False)

        h = h_proj.value

        # Evaluate new C₅
        h_one_minus_h = 1.0 - h
        corr = np.correlate(h, h_one_minus_h, mode='full') * dx
        c5 = np.max(np.abs(corr))

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_counter = 0
        else:
            no_improvement_counter += 1

        # Check time budget
        if time.time() - start_time > budget_s:
            break

        if no_improvement_counter > 100:
            break

    h_values = best_h
    c5_bound = best_c5
    n_points = len(h_values)
    return (h_values, c5_bound, n_points)
