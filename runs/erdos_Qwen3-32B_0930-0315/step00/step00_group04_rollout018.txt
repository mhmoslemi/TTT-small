import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    # Use the initial_h_values if available.
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 42
        # Seed-based initialization with sum constraint
        rng = np.random.RandomState(seed)
        h = rng.rand(n_points)
        h /= h.sum() * 2
        h *= n_points / 2

    # Configuration
    max_iter = 1000
    start_time = time.time()
    best_h = None
    best_c5 = np.inf
    learning_rate = 0.01
    lr_decay_steps = 100
    lr_decay_rate = 0.95

    def evaluate(h_curr):
        dx = 2.0 / len(h_curr)
        ones_minus_h = 1.0 - h_curr
        corr = correlate(h_curr, ones_minus_h, mode='full')
        corr *= dx  # Apply dx to correlation result
        c5 = np.max(corr)
        return c5

    def project_to_feasible(h_curr, sum_const=n_points / 2):
        h_curr = np.clip(h_curr, 0.0, 1.0)
        h_curr *= (sum_const / h_curr.sum())
        return h_curr

    for step in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Compute gradient
        ones_minus_h = 1.0 - h
        corr = correlate(h, ones_minus_h, mode='full')
        corr *= 2.0 / len(h)  # dx
        k_max = np.argmax(corr)

        # Find shift for k_max and compute (1 - h_shifted)
        k_start = max(0, k_max)
        k_end = min(len(h), k_max + len(h))
        h_shifted = np.zeros_like(h)
        h_shifted[k_start - k_max : k_end - k_max] = ones_minus_h[k_start - k_max : k_end - k_max]
        grad = h_shifted

        # Gradient update
        h = h - learning_rate * grad
        h = project_to_feasible(h)

        # Learning rate schedule
        if step > 0 and step % lr_decay_steps == 0:
            learning_rate *= lr_decay_rate

    return best_h.tolist(), best_c5, n_points
