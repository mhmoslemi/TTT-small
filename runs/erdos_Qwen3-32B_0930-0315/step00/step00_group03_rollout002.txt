import numpy as np
import time
from scipy.optimize import minimize
import scipy.signal


def compute_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h_vec = h.reshape(-1)
    one_minus_h = 1 - h_vec
    corr = scipy.signal.correlate(h_vec, one_minus_h, mode='full', method='direct')
    return np.max(corr) * dx


def project_to_feasible(h_vec, n_points):
    # Enforce bounds
    h_proj = np.clip(h_vec, 0, 1)
    # Enforce sum constraint
    current_sum = np.sum(h_proj)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        diff = (current_sum - target_sum) / current_sum
        h_proj = h_proj * (1 - diff)
        h_proj = np.clip(h_proj, 0, 1)
    return h_proj


def objective_func(h_vec):
    h = h_vec
    return compute_c5(h)
