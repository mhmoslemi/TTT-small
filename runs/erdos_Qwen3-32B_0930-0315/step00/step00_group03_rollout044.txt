import numpy as np
import cvxpy as cp
import time
import os

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Ensure using default budget if called without arguments
    actual_budget_s = min(budget_s, 1100)
    start_time = time.time()

    n_points = 70  # An efficient size
    initial_h = np.array(initial_h_values, copy=True) if "initial_h_values" in globals() else 0.5 * np.ones(n_points)
    initial_h /= np.sum(initial_h) * 2.0  # Normalize to 1 in box [0,2]

    best_h = initial_h.copy()
    best_c5 = evaluate_point(n_points, best_h)
    dx = 2.0 / n_points
    max_iter = 1 << 18  # 262k max iterations as hard limit
    lr = 0.01
    iter_no_improve = 0
    MAX_NO_IMPROVE = 200  # Break if no improvement for a while

    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= actual_budget_s:
            break

        h = best_h.copy()
        # Compute integral for all k
        corr = np.correlate(h, 1 - h, mode="full") * dx

        max_idx = np.argmax(corr)
        k_opt = max_idx - (n_points - 1)

        # Compute gradient for shift k_opt
        def compute_gradient(h, k):
            n = len(h)
            grad = np.zeros(n)
            for i in range(n):
                j = (i - k) % n
                if i - k >= 0:
                    grad[i] -= h[j]
                else:
                    j -= n
                    if j >= 0 and j < n:
                        grad[i] -= h[j]
                grad[i] += (1 - h[j])
            return grad

        G = compute_gradient(h, k_opt)

        # Normalize gradient
        G /= max(1.0, np.linalg.norm(G))

        # Line search
        alpha = lr
        improved = False
        for _ in range(10):  # max backtracking steps
            h_candidate = h - alpha * G
            # Project to feasible region
            h_proj = project(h_candidate, n_points)
            if h_proj is None:
                break
            new_c5 = evaluate_point(n_points, h_proj)
            if new_c5 < best_c5:
                best_h[:] = h_proj
                best_c5 = new_c5
                improved = True
                break
            alpha /= 2.0
        if not improved:
            iter_no_improve += 1
            if iter_no_improve >= MAX_NO_IMPROVE:
                break
        else:
            iter_no_improve = 0

    return (best_h, best_c5, n_points)

def evaluate_point(n_points, h):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def project(h_candidate, n_points):
    h = cp.Variable(n_points)
    objective = cp.Minimize(cp.sum_squares(h - h_candidate))
    constraints = [cp.sum(h) == n_points / 2., h >= 0, h <= 1]
    prob = cp.Problem(objective, constraints)
    try:
        result = prob.solve(solver=cp.ECOS, verbose=False)
        if h.value is None:
            return None
        return np.array(h.value).flatten()
    except cp.SolverError:
        return None
