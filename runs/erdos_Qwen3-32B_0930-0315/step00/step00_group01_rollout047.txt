import numpy as np
from scipy.signal import correlate
import time
import copy

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        n_points = 94  # using n_points larger than 1x parent for better approximation
        h = np.random.rand(n_points)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Constraint: rescale to have correct integral
    h = np.clip(h, 0, 1)
    h = h / h.sum() * target_sum

    learning_rate = 0.01
    best_h = copy.copy(h)
    best_c5 = evaluate_erdos_solution(h)

    def compute_overlap(h):
        dx = 2.0 / len(h)
        h_flipped = 1 - h
        corr = correlate(h, h_flipped, mode='full')
        c5 = np.max(corr * dx)
        return c5

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.1:  # leave 0.1s buffer for clean exit
            break

        # Compute overlap for all lags
        overlap = []
        for k in range(1, n_points):
            if k >= n_points:
                break
            h_k = h[k:]
            overlap_k = np.sum(h[:-k] * (1 - h_k)) * dx
            overlap.append(overlap_k)

        # Compute negative for maximizing
        overlap = np.array(overlap)
        k_max = np.argmax(overlap)
        worst_k = k_max + 1

        # Compute gradient for worst_k
        grad = np.zeros_like(h)
        i_valid = np.arange(h.shape[0] - worst_k)
        grad[i_valid] -= (1 - h[i_valid + worst_k])
        grad[i_valid] += h[i_valid]

        # Normalize gradient and update
        h -= learning_rate * grad

        # Clamp to [0, 1] and normalize sum
        h = np.clip(h, 0.0, 1.0)
        h = h / h.sum() * target_sum

        # Periodic perturbation
        if np.random.rand() < 0.01:
            noise = 0.05 * np.random.randn(n_points)
            h = np.clip(h + noise, 0.0, 1.0)
            h = h / h.sum() * target_sum

        # Check progress
        current_c5 = compute_overlap(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = copy.copy(h)

        # Reduce learning rate
        if current_c5 < 0.38088:
            learning_rate *= 0.995  # slower decay when below threshold

    # Final projection onto constraints
    h = np.clip(best_h, 0.0, 1.0)
    h = h / h.sum() * target_sum

    c5_bound = compute_overlap(h)
    return (h.tolist(), c5_bound, n_points)
