import numpy as np
import time
from scipy.signal import correlate
import copy

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    # Normalize h to sum to 1, scaled by area dx
    h = np.clip(h, 0, 1)
    h = (h * n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)  # clip again after scaling

    # Compute 1 - h
    one_minus_h = 1 - h
    # Full convolution computes cross-correlation of h and one_minus_h
    corr = correlate(h, one_minus_h, mode="full")
    # dx is the x-step size, so multiply for integral
    C5 = np.max(corr * dx)
    return C5

def enforce_feasibility(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    if current_sum < 1e-6:
        h[:] = 0
        h[0] = target_sum
    else:
        h[:] = (h / current_sum) * target_sum
        # Avoid under or overflow due to rounding
        residual = target_sum - np.sum(h)
        if abs(residual) > 1e-8:
            h[-1] += residual
        h[:] = np.clip(h, 0.0, 1.0)
    return h

def ga_sa_search(initial_h_values, budget_s=1100):
    start_time = time.time()
    n_points = len(initial_h_values)
    pop_size = 75
    tournament_size = max(1, int(0.1 * pop_size))
    mutation_rate = 0.05
    crossover_rate = 0.7

    if initial_h_values is not None:
        population = [copy.deepcopy(initial_h_values) for _ in range(5)] + [np.random.rand(n_points) for _ in range(pop_size - 5)]
    else:
        population = [np.random.rand(n_points) for _ in range(pop_size)]

    population = [enforce_feasibility(copy.deepcopy(x), n_points) for x in population]
    
    best_h = copy.deepcopy(min(population, key=lambda h: evaluate_c5(h)))
    best_c5 = evaluate_c5(best_h)
    
    T = 100.0
    cooling_rate = 0.95
    sa_interval = 50

    generation = 0

    while time.time() - start_time < budget_s:
        scores = [(h, evaluate_c5(h)) for h in population]
        scores.sort(key=lambda x: x[1])
        population = [h for h, _ in scores]
        
        # Update best
        current_best = scores[0]
        if current_best[1] < best_c5:
            best_h = copy.deepcopy(current_best[0])
            best_c5 = current_best[1]
        
        # Tournament selection
        parents = []
        for _ in range(int(pop_size * 2)):
            candidates = np.random.choice(pop_size, size=tournament_size, replace=False)
            winner = min(candidates, key=lambda i: scores[i][1])
            parents.append(population[winner])

        offspring = []

        for i in range(0, len(parents), 2):
            if i + 1 >= len(parents):
                offspring.append(parents[i].copy())
                continue

            p1, p2 = parents[i], parents[i + 1]
            if np.random.rand() < crossover_rate:
                cross_idx = np.random.randint(len(p1) + 1)
                child1 = np.concatenate((p1[:cross_idx], p2[cross_idx:])).copy()
                child2 = np.concatenate((p2[:cross_idx], p1[cross_idx:])).copy()
            else:
                child1 = p1.copy()
                child2 = p2.copy()

            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    noise = 0.2 * np.random.randn(n_points)
                    child += noise
                    child = enforce_feasibility(child, n_points)
                offspring.append(child)

        # Simulated Annealing
        if generation % sa_interval == 0:
            sa_best = best_h.copy()
            sa_new = sa_best + 0.1 * np.random.randn(n_points)
            sa_new = enforce_feasibility(sa_new, n_points)
            new_score = evaluate_c5(sa_new)
            if new_score < best_c5 or np.random.rand() < np.exp((best_c5 - new_score) / T):
                best_h = sa_new
                best_c5 = new_score
            T *= cooling_rate

        # Replace weak half with offspring
        population[-len(offspring):] = [enforce_feasibility(offs, n_points) for offs in offspring]
        generation += 1

    return best_h, best_c5, n_points

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    if initial_h_values is not None:
        n_points = len(initial_h_values)
    else:
        n_points = 45
        initial_h_values = np.random.rand(n_points)
        initial_h_values = enforce_feasibility(initial_h_values, n_points)

    h_values, c5_bound, n = ga_sa_search(initial_h_values, budget_s=budget_s)
    return h_values, c5_bound, n
