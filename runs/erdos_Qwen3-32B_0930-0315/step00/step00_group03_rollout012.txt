```python
import numpy as np
import cvxpy as cp
import time


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.perf_counter()

    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if "initial_h_values" in globals() and len(initial_h_values) == n_points:
        h = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        h = 0.5 * np.ones(n_points)
        h = h / np.sum(h) * target_sum

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    eta0 = 0.01
    max_iterations = 5000
    line_search_reduction = 0.5
    line_search_max_steps = 10
    convergence_threshold = 1e-6
    max_no_improvement = 100

    def compute_c5(h, dx):
        h_flat = h * np.ones((n_points, 1))
        shifts = np.arange(-n_points + 1, n_points)
        cross = np.sum(h_flat * (1 - h_flat[np.newaxis, :] * np.tile(np.eye(n_points), (1, 1))), axis=(1, 2))
        c5 = np.max(cross * dx)
        return c5

    def compute_gradient(h, k, dx):
        G = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k) % n_points
            G[i] += 1.0 - h[j]
        for j in range(n_points):
            if (j - k) % n_points < n_points:
                G[j] -= h[(j - k) % n_points]
        return G

    def project(h_candidate):
        hvar = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum_squares(hvar - h_candidate))
        constraints = [
            0 <= hvar,
            hvar <= 1,
            cp.sum(hvar) == target_sum,
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        return np.clip(hvar.value, 0, 1)

    no_improvement = 0
    it = 0

    while time.perf_counter() - start_time < budget_s - 0.1 and it < max_iterations:
        c5_current = compute_c5(h, dx)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h[:] = h[:]
            no_improvement = 0
        else:
            no_improvement += 1

        if no_improvement >= max_no_improvement:
            break

        # Find the shift k that gives the maximum overlap
        k_values = np.arange(-n_points + 1, n_points)
        h_ext = np.concatenate([h, h, h])
        shift_vals = []
        for k in k_values:
            val = np.sum(h_xt[n_points - k : n_points + (n_points - k)] * (1 - h_xt[k : k + n_points]))
            shift_vals.append(val)
        k_worst = np.argmin(shift_vals)

        # Compute the gradient for this k
        G = compute_gradient(h, k_worst, dx)

        if np.max(np.abs(G)) < convergence_threshold:
            break

        # Line search
        eta = eta0

        for _ in range(line_search_max_steps):
            h_candidate = h - eta * G
            h_proj = project(h_candidate)
            c5_candidate = compute_c5(h_proj, dx)
            if c5_candidate < c5_current:
                break
            else:
                eta *= line_search_reduction

        h[:] = project(h - eta * G)
        it += 1

    return best_h, best_c5, n_points
```
