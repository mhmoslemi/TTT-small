import numpy as np
import cvxpy as cp
import time
import math
from scipy.signal import correlate

def run(seed=42, budget_s=1100, n_points=None):
    np.random.seed(seed)
    start_time = time.time()

    if n_points is None:
        n_points = 80  # Default to 80 points

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        assert h.size == n_points, "initial_h_values is wrong length"
        assert abs(np.sum(h) - n_points / 2) < 1e-8, "Initial h has invalid sum"
    else:
        # Basic initialization: uniform distribution
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    prev_best = None
    delta_t = 1e-3

    while True:
        # Compute all overlaps
        h1 = h - h**2
        corr = correlate(h, h1, mode="full")
        corr *= dx
        overlap = np.max(corr)
        if overlap < best_c5:
            best_c5 = overlap
            best_h[:] = h[:]
            if prev_best is not None and abs(prev_best - best_c5) < 1e-6:
                break
            prev_best = best_c5
        else:
            prev_best = None

        # Find the worst lag
        i_worst = np.argmax(corr)
        lag = i_worst - n_points + 1

        # Build gradient dS/dh[k]
        grad = np.zeros_like(h)
        for i in range(n_points):
            j_ahead = (i + lag) % n_points
            j_behind = (i - lag) % n_points
            grad[i] = dx * (1.0 - h[j_ahead] - h[j_behind])

        # Solve the LP: minimize delta · grad
        # Subject to: h_new = h + delta
        #             0 <= h_new <= 1
        #             sum(delta) = 0
        delta = cp.Variable(n_points)
        objective = cp.Minimize(grad @ delta)
        constraints = [
            0 <= h + delta,
            h + delta <= 1,
            cp.sum(delta) == 0
        ]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h += delta.value
                h = np.clip(h, 0, 1)
        except Exception:
            pass  # Skip invalid steps

        # Check elapsed time
        if time.time() - start_time > budget_s:
            break

    c5_bound = best_c5
    return (best_h, c5_bound, n_points)
