import numpy as np
import time
import scipy.signal

def compute_overlap(h, dx):
    h = h.astype(np.float64)
    corr = scipy.signal.correlate(h, 1 - h, mode='full', method='auto') * dx
    return np.max(corr)

def repair_feasibility(h, n_points):
    h = np.clip(h, 0, 1).astype(np.float64)
    h = h / h.sum() * (n_points / 2)
    return h

def simulated_annealing(h, dx, steps, initial_temp, cooling_rate):
    best_h = h.copy()
    best_c5 = compute_overlap(best_h, dx)
    T = initial_temp
    for step in range(steps):
        h_proposal = h + np.random.normal(0, T / 10, h.shape)
        h_proposal = repair_feasibility(h_proposal, h.size)
        c5_proposal = compute_overlap(h_proposal, dx)
        if c5_proposal < best_c5 or np.random.rand() < np.exp((best_c5 - c5_proposal) / T):
            h = h_proposal
            if c5_proposal < best_c5:
                best_h, best_c5 = h_proposal, c5_proposal
        T *= cooling_rate
    return best_h, best_c5

def gradient_update(h, dx, lr):
    corr = scipy.signal.correlate(h, 1 - h, mode='full', method='auto') * dx
    k_max = np.argmax(corr)
    shift = k_max - (h.size - 1)
    h_shifted = np.roll(h, shift)
    gradient = (1 - h_shifted)[np.arange(h.size)]
    h -= lr * gradient
    h = repair_feasibility(h, h.size)
    return h

def interpolate_h(h, new_n_points):
    x = np.linspace(0, 2, h.size)
    x_new = np.linspace(0, 2, new_n_points)
    h_resampled = np.interp(x_new, x, h)
    return repair_feasibility(h_resampled, new_n_points)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    if 'initial_h_values' in globals():
        h = interpolate_h(np.array(initial_h_values), 14)
    else:
        h = np.random.uniform(0, 1, 14)
        h = repair_feasibility(h, 14)
    current_n_points = 14
    current_dx = 2.0 / current_n_points
    
    # Optimization phases
    h_best = h.copy()
    c5_best = compute_overlap(h_best, current_dx)
    
    while time.time() - start_time < budget_s * 0.6 and current_n_points <= 56:
        h_coarse, c5_coarse = simulated_annealing(h_best, current_dx, 
                                                 steps=500, 
                                                 initial_temp=1.0, 
                                                 cooling_rate=0.99)
        
        # Local refinement
        for _ in range(30):
            h_coarse = gradient_update(h_coarse, current_dx, lr=0.1)
            c5_coarse_new = compute_overlap(h_coarse, current_dx)
            if c5_coarse_new < c5_coarse:
                c5_coarse = c5_coarse_new
            else:
                break

        # Update best
        if c5_coarse < c5_best:
            h_best = h_coarse
            c5_best = c5_coarse
            
        # Refine resolution
        if time.time() - start_time < budget_s and current_n_points < 56:
            h_best = interpolate_h(h_best, current_n_points * 2)
            current_n_points *= 2
            current_dx = 2.0 / current_n_points

    end_time = time.time()
    
    # Final cleanup and validation
    h_best = repair_feasibility(h_best, current_n_points)
    c5_bound = compute_overlap(h_best, current_dx)
    
    return h_best.tolist(), c5_bound, current_n_points
