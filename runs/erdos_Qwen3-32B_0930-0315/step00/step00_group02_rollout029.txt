from time import time

def run(seed=42, budget_s=1100, n_points=160, **kwargs):

    t0 = time()

    best_h = h.copy()
    best_c5 = np.inf
    prev_best = best_c5
    no_improvement_count = 0
    MAX_NO_IMPROVEMENT = 1000
    MAX_ITER = 20000

    for i in range(MAX_ITER):
        if time() - t0 > budget_s:
            break
        f = 1 - h
        corr = np.correlate(h, f, mode='full') * dx
        # Only valid shifts k in [0, n_points]
        corr = np.concatenate((corr[n_points:], corr[:n_points + 1]))
        k_max = np.argmin(corr)  # Since we are minimizing C5, we want to reduce the worst (max) overlap

        grad = np.zeros(n_points)
        for j in range(n_points):
            left = (j - k_max) % n_points
            right = (j + k_max) % n_points
            grad[j] = (1 - h[right]) - h[left]

        # Barzilai-Borwein step size adapted
        if i > 0:
            y = h - prev_h
            s = grad - prev_grad
            alpha = np.dot(y, y) / np.dot(y, s)
        else:
            alpha = 0.1  # Initial step size

        prev_h = h.copy()
        prev_grad = grad.copy()
        h -= alpha * grad

        # Projection step: keep h in [0, 1] and sum(h) = n_points / 2
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)

        # Evaluation
        f = 1 - h
        corr = np.correlate(h, f, mode='full') * dx
        c5 = np.max(corr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        # Add perturbation every 200 iterations to escape local optima
        if no_improvement_count >= 500 and np.random.rand() < 0.2:
            perturbation = np.random.normal(0, 0.1, size=n_points)
            h += perturbation
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)

    f = 1 - best_h
    corr = np.correlate(best_h, f, mode='full') * dx
    c5_bound = np.max(corr)

    # Return the final values
    return best_h.tolist(), c5_bound, n_points
