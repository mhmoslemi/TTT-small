import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h = np.copy(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points

    # Ensure normalization
    h = h / h.sum() * n_points / 2
    h = np.clip(h, 0, 1)

    kt = time.time()
    dt = 0.1
    alpha = 0.1
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()

    while time.time() - kt < budget_s:
        # Compute shifted correlations
        ones_h = 1.0 - h
        corr = correlate(h, ones_h, mode='full')

        # Indicator function for max overlap
        max_index = np.argmax(corr)

        # Determine the effective shift
        if max_index < n_points:
            k = max_index
        else:
            k = max_index - 2 * n_points

        # Compute gradient of overlap with shift k
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + k < n_points:
                grad[i] += ones_h[i + k]
            if i - k >= 0:
                grad[i] -= ones_h[i]

        # Gradient descent step
        h -= alpha * grad
        h = np.clip(h, 0, 1)

        # Project back to feasible space (maintain mean)
        h = h / h.sum() * n_points / 2
        h = np.clip(h, 0, 1)

        # Perturbation every 50 steps
        if (time.time() - kt) % 50 < 1.0 and np.random.rand() < 0.1:
            h += np.random.normal(0.0, 0.01, h.shape)
            h = np.clip(h, 0, 1)
            h = h / h.sum() * n_points / 2
            h = np.clip(h, 0, 1)

        # Evaluate C5
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    return (best_h.tolist(), best_c5, n_points)
