    start_time = time.time()
    best_h = h.copy()
    best_c5 = None
    iter_count = 0
    no_improvement_counter = 0
    patience = 500  # iterations of no improvement before perturbation
    history_length = 100
    history = []

    while time.time() - start_time < budget_s:
        corrs = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(corrs)
        if best_c5 is None or c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_counter = 0
        else:
            no_improvement_counter += 1

        if best_c5 is None:
            best_c5 = c5

        # Choose the worst shift
        k_max = np.argmax(corrs)
        # Compute gradient: move h[i] to reduce overlap at k_max
        grad = np.zeros_like(h)
        for i in range(n_points):
            i_plus_k = (i + k_max) % n_points
            grad[i] = (1 - h[i_plus_k]) - h[i]  # derivative w.r.t. h[i] at k_max

        # Barzilai-Borwein step size approximation
        if iter_count > 0:
            s = h - last_h
            y = grad - last_grad
            dot_sy = np.dot(s, y)
            step_size = dot_sy / (np.dot(s, s) + 1e-8)  # Compute alpha
        else:
            step_size = 0.01

        h_new = h - step_size * grad

        # Project h_new to constraints: [0, 1] and sum(h) = n_points / 2
        h_new = np.clip(h_new, 0, 1)
        h_sum = h_new.sum()
        if h_sum > 0 and h_sum < n_points:
            h_new /= h_sum * 2 / n_points  # scale to sum to n_points / 2

        h = h_new
        iter_count += 1
        last_h = h.copy()
        last_grad = grad.copy()

        # Perturbation if stuck
        if no_improvement_counter >= patience:
            perturbation_indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            h[perturbation_indices] = np.random.rand(len(perturbation_indices))

            h_sum = h.sum()
