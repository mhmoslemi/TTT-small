import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Load initial values if available, else construct a simple plateau
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 45
        h = np.zeros(n_points, dtype=np.float64)
        h[:n_points // 2] = 1.0
        h[n_points // 2:] = 0.0
        # Adjust slightly to meet the integral constraint exactly
        h += (1.0 - h.mean() * n_points) / n_points

    # Set the number of points to use
    n_points = len(h)

    # Make dx float64 to avoid precision issues
    dx = 2.0 / n_points

    def project(h):
        # Clip values to stay within bounds
        h_clipped = np.clip(h, 0, 1)
        # Project to have the exact integral constraint
        total = h_clipped.sum()
        correct_total = n_points / 2.0
        if abs(total - correct_total) > 1e-8:
            # Small correction scale
            correction = (correct_total - total) / n_points
            h_correction = h_clipped + correction
            # Ensure that the correction does not push any value beyond [0,1]
            h_correction = np.clip(h_correction, 0, 1)
            # Final adjustment if needed
            final_total = h_correction.sum()
            scale_factor = correct_total / final_total if final_total > 0 else 1.0
            h_projected = h_correction * scale_factor
            # Finalclip to [0,1]
            h_projected = np.clip(h_projected, 0, 1)
        else:
            h_projected = h_clipped
        return h_projected

    start_time = time.time()
    best_h = h.copy()
    best_c5 = float('inf')

    # Optimization parameters
    learning_rate = 0.01
    alpha = 0.8

    # Iterate until time budget is exhausted
    while time.time() - start_time < budget_s:
        # Compute all correlations (full mode)
        correlations = scipy.signal.correlate(h, (1.0 - h), mode='full') * dx

        # Clip circular overlap (only consider k in range [0, len(h)-1])
        indices = np.arange(len(correlations))
        margins = np.arange(n_points)
        left_mask = (indices % n_points == 0)
        right_mask = (indices % n_points == (n_points - 1))
        k_mask = left_mask & right_mask
        k_indices = indices[k_mask]
        k_values = [(int(i) // n_points) for i in k_indices]
        k_unique = list(set(k_values))
        k_adjusted = [k for k in k_unique if 0 <= k < n_points]

        C = []
        for k in k_adjusted:
            cross_corr = scipy.signal.correlate(h, (1.0 - np.roll(h, k)), mode='valid')
            C_k = cross_corr.sum() * dx
            C.append((C_k, k))

        C_k, k_max = max(C, key=lambda x: x[0])

        # Start of gradient calculation
        grad = np.zeros(n_points)
        for i in range(n_points):
            idx_forward = (i + k_max) % n_points
            idx_backward = (i - k_max) % n_points
            grad[i] = 1.0 - h[idx_forward] - h[idx_backward]

        # Gradient ascent step
        h_grad = h + learning_rate * grad

        # Projection step to ensure constraints
        h_projected = project(h_grad)

        # Update h
        h = h_projected

        # Track the best solution
        correlations_all = scipy.signal.correlate(h, (1.0 - h), mode='full') * dx
        current_c5 = correlations_all.max()

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            learning_rate = min(learning_rate * alpha, 1.0)

        # Learning rate safety
        learning_rate = min(learning_rate, 0.1)

        # Check mapping invariants
        assert np.all(h >= 0.0), "Negative sampling value encountered"
        assert np.all(h <= 1.0), "Sampling value greater than 1 occurred"
        assert abs(h.sum() - (n_points / 2.0)) < 1e-6, "Integral constraint violated"

    # Final validation of best_h to ensure constraints
    assert np.all(best_h >= 0) and np.all(best_h <= 1), "Output invalid in bound"
    assert abs(best_h.sum() - (n_points / 2)) < 1e-6, "Integral constraint not met"

    return (best_h.tolist(), best_c5, n_points)
