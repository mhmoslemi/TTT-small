import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    h = initial_h.copy().astype(np.float64)

    start_time = time.time()

    # Project h to maintain the constraint: sum(h) == n_points / 2
    def project_h(h):
        total = np.sum(h)
        correction = n_points / 2 - total
        h += correction / n_points
        np.clip(h, 0, 1, out=h)

    # Compute overlap using convolution-like formula
    def compute_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    best_c5 = np.inf
    best_h = h.copy()

    max_iterations = 1000
    for i in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        corr = np.correlate(h, 1 - h, mode='full')
        corr *= dx
        k_max = np.argmax(corr)
        target_overlap = corr[k_max]

        # Compute gradient-like direction to reduce the maximum overlap
        grad = np.zeros_like(h)
        if k_max < n_points:
            grad[k_max:] -= (1 - h[:len(grad) - k_max]) * 1.0
            grad[: len(grad) - k_max] -= h[k_max:] * 1.0
        else:
            grad[:n_points] -= (1 - h[-k_max + n_points:]) * 1.0
            grad[-k_max + n_points:] -= h[:n_points] * 1.0

        # Normalize gradient
        grad_norm = np.linalg.norm(grad) + 1e-10
        grad /= grad_norm

        # Take step in gradient direction
        step_size = 1.0
        for _ in range(3):
            candidate = h.copy()
            candidate += step_size * grad
            project_h(candidate)
            candidate_c5 = compute_c5(candidate, dx)
            if candidate_c5 < best_c5:
                h[:] = candidate[:]
                best_c5 = candidate_c5
                best_h[:] = h[:]
                break
            step_size *= 0.5

    return best_h, best_c5, n_points
