import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points
    time_start = time.time()

    max_iter = 10000
    alpha = 0.1
    patience = 50
    best_c5 = float('inf')
    best_h = h.copy()
    current_iter = 0

    for it in range(max_iter):
        elapsed = time.time() - time_start
        if elapsed >= budget_s:
            break

        # Compute correlation (overlap integral over all k)
        corr = correlate(h, 1 - h, mode='full')
        corr = corr * dx
        c5 = np.max(corr)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            # Reduce step size if we have improved
            alpha = max(alpha * 0.95, 0.01)

        # Find the lag k where the max of the overlap occurs
        k_max = np.argmax(corr)
        lag = k_max - (n_points - 1)

        # Compute the gradient w.r.t. C_5 at lag k
        dh = np.zeros_like(h)
        h_k = h[lag:]  # h[i]
        h_kj = h[:-lag]  # h[i + k]
        dh[lag:] += (1 - h_kj)
        dh[:-lag] -= (1 - h_kj)  # Derivative from both terms in h[i](1 - h[i+k])

        # Gradient descent update
        h = h - alpha * dh

        # Clip to stay in [0, 1]
        h_clipped = np.clip(h, 0.0, 1.0)

        # Rescale to maintain sum constraint: sum(h) == n_points / 2
        s = np.sum(h_clipped)
        if s != 0:
            h = h_clipped * (n_points / 2) / s
        else:
            h = h_clipped  # No-op if sum is zero

        # Perturb every few steps to avoid local minima
        if it % patience == 0:
            perturbation = np.random.uniform(-0.01, 0.01, size=h.shape)
            h += perturbation
            h = np.clip(h, 0, 1)
            s = np.sum(h)
            if s != 0:
                h = h * (n_points / 2) / s

    return best_h.tolist(), best_c5, n_points
