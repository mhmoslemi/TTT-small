import numpy as np
import math
import time

def run(seed=42, budget_s=1100, n_points=47):
    # Use the initial_h_values as starting point if available
    try:
        h = initial_h_values.copy()
    except NameError:
        # Default uniform initialization
        h = np.zeros(n_points)
        h[:n_points // 2] = 1

    h = np.clip(h, 0, 1)
    total = np.sum(h)
    scale_factor = n_points / 2 / total
    h = h * scale_factor

    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float("inf")
    best_h = h.copy()

    dx = 2.0 / n_points
    while time.time() - start_time < budget_s:
        # Compute overlaps for all valid lags
        overlaps = []
        for k in range(1, n_points):
            overlap = np.sum([h[i] * (1 - h[i + k]) for i in range(n_points - k)]) * dx
            overlaps.append(overlap)
        overlaps = np.array(overlaps)
        if len(overlaps) == 0:
            break
        k_max = np.argmax(overlaps)
        current_c5 = overlaps[k_max]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Compute gradient of the max overlap with respect to h
        k = k_max + 1  # +1 because indexing skipped 0
        grad = np.zeros_like(h)
        for i in range(n_points - k):
            grad[i] += (1 - h[i + k])
            grad[i + k] -= h[i]

        # Gradient descent with learning rate decay
        t = time.time() - start_time
        learning_rate = 0.05 * max(0.1, 1 - t / budget_s)  # Decaying learning rate
        h -= learning_rate * grad

        # Projection to ensure 0 ≤ h ≤ 1 and ∑h = n_points / 2
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - n_points / 2) > 1e-8:
            scale_factor = (n_points / 2) / total
            h = h * scale_factor

        # Optional: Perturbation to escape local optima
        if np.random.rand() < 0.02:
            h += np.random.uniform(-0.02, 0.02, size=h.shape)
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            if abs(total - n_points / 2) > 1e-8:
                scale_factor = (n_points / 2) / total
                h = h * scale_factor

    # Re-evaluate best C₅ with final best_h
    overlaps = []
    dx = 2.0 / n_points
    for k in range(1, n_points):
        overlap = np.sum([best_h[i] * (1 - best_h[i + k]) for i in range(n_points - k)]) * dx
        overlaps.append(overlap)
    final_c5 = max(overlaps)

    return (best_h, final_c5, n_points)
