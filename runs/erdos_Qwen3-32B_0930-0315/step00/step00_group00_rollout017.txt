import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    n_points = 45
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = h * (n_points / h.sum())  # Ensure sum(h) == n_points / 2
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()
    max_time = budget_s

    learning_rate = 1.0
    learning_decay = 0.999
    perturb_rate = 0.1
    perturb_decay = 0.995

    while time.time() - start_time < max_time:
        # Evaluate current C5 value
        c5 = evaluate_erdos_solution(h)

        if c5 < best_c5:
            best_h[:] = h
            best_c5 = c5

        # Compute cross-correlation for each lag
        n = len(h)
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        lag = idx - (n - 1)

        # Compute gradient: shift critical lag's contribution
        grad = np.zeros_like(h)
        k_lag = int(lag)

        for i in range(n):
            i_shifted = i + k_lag
            if 0 <= i_shifted < n:
                grad[i] = 0.5 * (1 - 2 * h[i_shifted]) * (1 - 2 * h[i])
            elif i_shifted < 0:
                if 0 <= i < n:
                    grad[i] -= 2 * h[i] * h[i - k_lag]
                if 0 <= (i - k_lag) < n:
                    grad[i - k_lag] -= 2 * h[i] * h[i - k_lag]
            else:
                if i + k_lag < n:
                    grad[i + k_lag] -= 2 * h[i] * h[i + k_lag]
                if i < n:
                    grad[i] -= 2 * h[i] * h[min(i + k_lag, n - 1)]

        grad *= dx
        h -= learning_rate * grad

        # Projection to [0, 1]
        h = np.clip(h, 0, 1)

        # Projection to the simplex: sum(h) == n_points / 2 with item in [0, 1]
        h += (n_points / 2 - h.sum()) / n_points
        h = np.clip(h, 0, 1)
        h /= h.sum() * dx * 2

        # Learning rate and perturbation decay
        learning_rate *= learning_decay
        perturb_rate *= perturb_decay

        # Random perturbation to escape local optima
        perturbation = np.random.uniform(-perturb_rate, perturb_rate, size=n)
        h = np.clip(h + perturbation, 0, 1)
        h /= h.sum() * dx * 2

    return best_h.tolist(), best_c5, n_points
