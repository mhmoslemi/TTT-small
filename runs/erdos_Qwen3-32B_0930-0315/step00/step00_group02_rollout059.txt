import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy().astype(float)

    def update_and_project(h, dx):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return h_clipped
        scaling = (n_points / 2) / total
        return h_clipped * scaling

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    def compute_c5(h):
        c = correlate(h, 1.0 - h, mode='full') * dx
        return np.max(c), c

    max_iter = 10000
    patience = 50
    no_improvement = 0
    lr = 0.01
    best_time = time.time()

    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        _, corr = compute_c5(h)
        max_idx = np.argmax(corr)
        max_k = max_idx

        # Compute gradient for max_k
        grad = np.zeros(n_points)
        h_fft = h.copy()
        for i in range(n_points):
            mod_idx = (i + max_k) % n_points
            mod_idx_prev = (i - max_k + n_points) % n_points
            grad[i] = (1 - h[mod_idx]) - h[mod_idx_prev]

        h -= lr * grad
        h = update_and_project(h, dx)

        current_c5, _ = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h
            no_improvement = 0
            best_time = time.time()
        else:
            no_improvement += 1
            if no_improvement >= patience:
                lr *= 0.5
                no_improvement = 0

        # Introduce small perturbation and reset counter
        if iteration > 500 and np.random.rand() < 0.01:
            h += np.random.uniform(-0.02, 0.02, size=h.shape)
            h = update_and_project(h, dx)

    # Final sanity check
    assert np.all(h <= 1 + 1e-6) and np.all(h >= -1e-6), "h out of bounds"
    h = update_and_project(h, dx)
    final_c5, _ = compute_c5(h)

    return (h, final_c5, n_points)
