import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)

    dx = 2.0 / n_points
    best_h = initial_h.copy()
    best_C5 = estimate_C5(best_h, n_points, dx)

    start_time = time.time()
    best_time = start_time
    max_iter = 100000
    learning_rate = 0.005
    patience = 100
    tolerance = 1e-6
    iters_since_improve = 0

    def estimate_C5(h, n, dx):
        # Compute all cross correlations (1 - h) shifted by k
        h = h.astype(np.float64)
        one_minus_h = 1.0 - h
        full_correlate = np.zeros(2 * n, dtype=np.float64)
        for k in range(1, n):
            shifted = np.roll(one_minus_h, -k)
            correl = np.dot(h, shifted)
            full_correlate[k] = correl
            full_correlate[-k] = correl
        return np.max(full_correlate) * dx

    def objective(h):
        return estimate_C5(h, n_points, dx)

    def gradient(h, k):
        # Gradient for a specific k shift
        h = h.astype(np.float64)
        shifted = np.roll(1.0 - h, -k)
        pos = np.roll(1.0 - h, -k)
        neg = np.roll(1.0 - h, k)
        grad = -pos - neg
        return grad

    def project(h):
        # Ensure h[i] ∈ [0, 1] and ∑h = n / 2
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        scale = (n_points / 2.0) / current_sum
        return h_clipped * scale

    h = initial_h.copy()
    best_h = h.copy()
    best_C5 = objective(h)

    for it in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        if (time.time() - best_time) > 60:
            # Reanneal if stuck
            h = (np.random.rand(n_points) * 0.1 + 0.45)
            h = project(h)
            best_time = time.time()
            best_h = h.copy()
            best_C5 = objective(h)
            learning_rate = 0.005
            iters_since_improve = 0

        # Sample 10 random k values to approximate the gradient
        k_list = np.random.randint(1, n_points, size=10)
        avg_grad = np.zeros(n_points)
        for k in k_list:
            avg_grad += gradient(h, k)
        avg_grad /= len(k_list)

        # Gradient step
        h = h - learning_rate * avg_grad

        # Project back to feasible region
        h = project(h)

        # Compute current C5
        curr_C5 = objective(h)

        # Track best
        if curr_C5 < best_C5 - 1e-6:
            best_h = h.copy()
            best_C5 = curr_C5
            best_time = time.time()
            iters_since_improve = 0
            learning_rate *= 2.0  # Rapid improvement: increase learning rate
        else:
            iters_since_improve += 1
            if iters_since_improve > patience:
                learning_rate *= 0.8  # Stagnant: reduce learning rate
                iters_since_improve = 0

        # Early convergence condition
        if learning_rate < 1e-6 or (it % 100 == 0 and abs(best_C5 - objective(best_h)) < tolerance):
            break

    # Return best solution
    return best_h, best_C5, n_points
