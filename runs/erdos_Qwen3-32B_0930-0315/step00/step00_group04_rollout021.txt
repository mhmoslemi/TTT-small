import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    dx = 2.0 / (len(initial_h_values) if 'initial_h_values' in globals() else 42)
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 42
    h0 = np.array(initial_h_values) if 'initial_h_values' in globals() else np.zeros(n_points)

    # Ensure initial h is in [0, 1] and satisfies constraint
    h0 = np.clip(h0, 0, 1)
    h0 = h0 / np.sum(h0) * (n_points / 2)
    h0 = np.clip(h0, 0, 1)

    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0)

    for mu in np.logspace(4, -5, 100):
        if time.time() - start_time > budget_s:
            break
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]

        overlap = cp.multiply(h, 1 - cp.roll(h, 1)) * dx
        for k in range(2, n_points):
            overlap += cp.multiply(h, 1 - cp.roll(h, k)) * dx
        max_overlap = cp.maximum(overlap)

        barrier = mu * cp.sum(cp.log(h) + cp.log(1 - h))
        objective = cp.Minimize(max_overlap + barrier)

        prob = cp.Problem(objective, constraints)
        result = prob.solve(solver=cp.CVXOPT, verbose=False)

        if h.value is not None:
            h_vals = h.value.astype(float)
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals / h_vals.sum() * (n_points / 2)
            current_c5 = evaluate_erdos_solution(h_vals)
            if current_c5 < best_c5:
                best_h = h_vals
                best_c5 = current_c5

    return (best_h, best_c5, n_points)
