def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    h_normalized = h / h_sum * (n_points / 2)
    # Evaluate overlap across all offsets
    padded_h = np.pad(h_normalized, (0, n_points - 1), mode='constant')
    padded_1m_h = np.pad(1 - h_normalized, (n_points - 1, 0), mode='constant')
    full_corr = np.correlate(padded_h, padded_1m_h, mode='valid')
    c5 = full_corr * dx
    return float(c5.max())

def project_h(h):
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if np.isclose(h_sum, 0):
        return h
    h = h / h_sum * (len(h) / 2)
    return h
