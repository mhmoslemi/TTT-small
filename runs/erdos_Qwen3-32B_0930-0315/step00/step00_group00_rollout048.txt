import numpy as np
import time
import cvxpy as cp


def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    result = np.correlate(h, 1 - h, mode='full')
    return np.max(result * dx)


def project_simplex(v, z=1.0, axis=None):
    v = np.asarray(v)
    if axis is None:
        axis = np.ndim(v) - 1

    # Make sure v is at least 1-dimensional
    if axis != np.ndim(v) - 1:
        axes = list(range(np.ndim(v)))
        axes.append(axes.pop(axis))
        v = np.transpose(v, axes)

    n = v.shape[axis]
    irepeat = np.ndim(v) * [1]
    irepeat[axis] = n
    zrepmat = np.reshape(z, irepeat)

    def _project(v):
        v = np.reshape(v, (-1, n))
        result = np.zeros_like(v)
        for i in range(v.shape[0]):
            v_i = v[i, :]
            u = np.sort(v_i, axis=0)[::-1]
            tmp_sum = np.cumsum(u, axis=0)
            tmp_ind = np.arange(1, n + 1)
            mask = tmp_sum - zrepmat[0] * tmp_ind >= 0
            if np.sum(mask) == 0:
                rho = n
            else:
                rho = np.min(np.where(mask))
            theta = (np.sum(u[:rho]) - z) / rho
            result[i, :] = (v_i - theta).clip(min=0)
        return result

    result = _project(v)
    if axis != np.ndim(v) - 1:
        result = np.transpose(result, axes)
    return result

        n_points = len(initial_h)
    except:
        n_points = 45
        np.random.seed(seed)
        h = np.zeros(n_points)
        h[: n_points // 2] = 1.0
        np.random.shuffle(h)
        initial_h = h

    h = project_simplex(initial_h, z=n_points / 2)

    start_time = time.time()
    max_iterations = 5000
    best_h = h.copy()
    best_c5 = evaluate(h)
    min_lr = 1e-6
    lr = 0.02
    wd_step = 100
    wd_iter = 2000
    patience = 50
    best_iter = 0

    for i in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full")
        peak_indices = np.where(np.isclose(np.abs(corr), np.max(np.abs(corr))))[0]

        grad = np.zeros_like(h)
        for k in peak_indices:
            for i in range(n_points):
                a = i
                b = (i + k) % n_points
                val = (1 - 2 * h[b]) if h[a] == 1 else 0
                grad[i] += val * dx

        h = h - lr * grad
        h = project_simplex(h, z=n_points / 2)

        new_c5 = evaluate(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            best_iter = i
        elif i - best_iter > patience:
            lr = max(lr / 2, min_lr)

        if i % 100 == 0:
            print(f"Iter {i}, LR {lr:.4f}, C5: {best_c5:.8f}", flush=True)

        if i > wd_iter and (i % wd_step) == 0:
            noise = np.random.uniform(
                -lr / 2, lr / 2, size=h.shape)
            h = h + noise
            h = project_simplex(h, z=n_points / 2)

    best_c5 = evaluate(best_h)
    return best_h.tolist(), best_c5, n_points
