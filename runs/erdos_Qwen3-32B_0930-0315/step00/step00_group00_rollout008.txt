import numpy as np
import time
import random
from math import exp

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time.time()
    
    h_hints = initial_h_values
    n_points = len(h_hints)
    dx = 2.0 / n_points

    def evaluate(h):
        h = np.array(h, dtype=float)
        h[h < 0] = 0
        h[h > 1] = 1
        h_sum = h.sum()
        if h_sum > 1e-6:
            h *= (n_points / 2) / h_sum
        else:
            h[:] = 0.5
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = corr.max() * dx
        return h, c5

    # Initialize population
    population = [h_hints.copy()]
    for _ in range(75 - 1):  # 75 total population
        h = np.random.rand(n_points)
        h, c5 = evaluate(h)
        population.append(h)

    best_h, best_c5 = evaluate(population[0])
    for h in population:
        _, c5 = evaluate(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    tournament_size = max(1, int(0.1 * len(population)))
    mutation_rate = 0.05
    crossover_rate = 0.7
    temp = 100.0  # initial SA temperature
    cooling = 0.95  # SA cooling factor
    sa_interval = 50

    generation = 0
    while time.time() - start_time < budget_s:
        generation += 1
        new_population = population.copy()

        for i in range(len(population)):
            # Tournament selection
            players = random.sample(population, tournament_size)
            parent1 = min(players, key=lambda h: np.correlate(h, 1 - h, mode='full').max() * dx)
            players2 = random.sample(population, tournament_size)
            parent2 = min(players2, key=lambda h: np.correlate(h, 1 - h, mode='full').max() * dx)

            # Crossover
            if random.random() < crossover_rate:
                cx_point = random.randint(1, n_points - 1)
                child = np.concatenate((parent1[:cx_point], parent2[cx_point:]))
            else:
                child = parent1.copy()

            # Mutation (GA + SA hybrid)
            if random.random() < mutation_rate:
                mutation = np.random.randn(n_points) * 0.1
                child += mutation
            else:
                # SA-style mutation: perturb best_h
                child = best_h.copy() + np.random.randn(n_points) * (temp / 100.0)

            # Clip and normalize
            child, c5 = evaluate(child)
            new_population.append(child)

        # Truncate population
        population = sorted(new_population, key=lambda h: np.correlate(h, 1 - h, mode='full').max() * dx)
        population = population[: len(population) // 2]  # Elitism + survival

        # SA step
        if generation % sa_interval == 0:
            new_h = best_h.copy() + np.random.randn(n_points) * (temp / 100.0)
            new_h, new_c5 = evaluate(new_h)
            delta = new_c5 - best_c5
            if delta < 0:
                best_h = new_h.copy()
                best_c5 = new_c5
            elif random.random() < exp(-delta / temp):
                best_h = new_h.copy()
                best_c5 = new_c5
            temp *= cooling

        # Update best_h
        for h in population:
            h_evaled, c5 = evaluate(h)
            if c5 < best_c5:
                best_h = h_evaled.copy()
                best_c5 = c5

    return best_h.tolist(), best_c5, n_points
