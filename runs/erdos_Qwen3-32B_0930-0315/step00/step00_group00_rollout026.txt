import numpy as np
import time
import random
from cvxpy import *  # for feasibility projection if needed

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    c5_vals = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(c5_vals)

def clip_and_normalize(h, n_points):
    # Ensure values in [0,1]
    h_clipped = np.clip(h, 0, 1)
    # Enforce integrality: sum(h) = n_points / 2
    h_sum = np.sum(h_clipped)
    if abs(h_sum - n_points / 2) > 1e-6:
        h_clipped = h_clipped * (n_points / 2) / h_sum
    return h_clipped

def tournament_selection(population, fitness, tournament_size):
    candidates = random.sample(list(zip(population, fitness)), tournament_size)
    return min(candidates, key=lambda x: x[1])[0]  # lower C₅ is better

def single_point_crossover(parent1, parent2):
    crossover_point = random.randint(1, len(parent1)-1)
    child1 = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
    child2 = np.concatenate((parent2[:crossover_point], parent1[crossover_point:]))
    return clip_and_normalize(child1, len(parent1)), clip_and_normalize(child2, len(parent2))

def mutate(h, mutation_rate, temperature):
    new_h = h.copy()
    n_points = len(h)
    perturbation = np.random.normal(0, 0.1, n_points)
    prob = 1 / (1 + np.exp(-temperature))
    for i in range(n_points):
        if random.random() < mutation_rate * prob:
            new_h[i] += perturbation[i]
    return clip_and_normalize(new_h, n_points)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    # Hyperparameters
    n_points = 45
    pop_size = 75
    tournament_size = max(2, int(0.1 * pop_size))
    mutation_rate = 0.05
    crossover_rate = 0.7
    generations = 500
    initial_temperature = 100.0
    cooling_rate = 0.95
    sa_interval = 50
    start_time = time.time()

    # Population initialization
    population = []
    if 'initial_h_values' in globals():
        population.append(initial_h_values)
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = clip_and_normalize(h, n_points)
        population.append(h)

    best_solution = None
    best_c5 = float('inf')
    temperature = initial_temperature

    generation = 0
    while time.time() - start_time < budget_s and generation < generations:
        # Evaluate current population
        fitness = [evaluate_c5(h, n_points) for h in population]
        # Find best solution
        current_best = min(zip(population, fitness), key=lambda x: x[1])[0]
        current_best_c5 = min(fitness)
        if current_best_c5 < best_c5:
            best_solution = current_best
            best_c5 = current_best_c5

        # Generate new population
        new_population = []
        for _ in range(pop_size // 2):
            # Selection
            parent1 = tournament_selection(population, fitness, tournament_size)
            parent2 = tournament_selection(population, fitness, tournament_size)

            # Crossover
            if random.random() < crossover_rate:
                child1, child2 = single_point_crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()

            # Mutation
            child1 = mutate(child1, mutation_rate, temperature)
            child2 = mutate(child2, mutation_rate, temperature)

            new_population.extend([child1, child2])

        # Simulated Annealing step
        if generation % sa_interval == 0:
            best_idx = np.argmin(fitness)
            mutated_best = mutate(population[best_idx], mutation_rate, temperature)
            c5_old = evaluate_c5(population[best_idx], n_points)
            c5_new = evaluate_c5(mutated_best, n_points)

            if c5_new < c5_old or random.random() < np.exp(-(c5_new - c5_old) / temperature):
                population[best_idx] = mutated_best

        # Update temperature
        temperature *= cooling_rate

        # Replace old population
        population = new_population
        generation += 1

    # Final evaluation
    best_c5 = evaluate_c5(best_solution, n_points) if best_solution is not None else float('inf')
    return (best_solution.tolist(), best_c5, n_points)
