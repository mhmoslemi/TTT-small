import numpy as np
import time
import scipy.signal

dx = None
n_points = None

def compute_overlap(h):
    global dx
    n = len(h)
    cross_corr = scipy.signal.correlate(h, 1 - h, mode='full')
    return np.max(cross_corr) * dx

def project_simplex(v):
    n = len(v)
    v = np.clip(v, 0, 1)
    s = np.sum(v)
    scale = n / (2 * n / s - 1) if s < n / 2 else n / (2 * n / s)
    return v * scale

    patience_limit = 500

    while time.time() - start_time < budget_s and iteration < max_iterations:
        iteration += 1

        idxs = np.arange(n_points)

        max_c = -float('inf')
        k_max = 0
        for k in range(2 * n_points):
            shifted = np.roll(idxs, -k)
            shifted = shifted[:-n_points] if k < n_points else shifted[n_points:]

            base_slice = idxs[:n_points] if k < n_points else idxs[-k:]
            comp_slice = shifted[:len(base_slice)]

            overlap = np.dot(current_h[base_slice], 1 - current_h[comp_slice])
            if overlap > max_c:
                max_c = overlap
                k_max = k

        # Compute gradient
        shift_true = k_max
        if shift_true < n_points:
            grad_idx = np.arange(n_points)
            grad_true = 1 - current_h[(grad_idx + shift_true) % n_points]
            grad = grad_true - 2 * current_h
        else:
            shift_true -= n_points
            grad_idx = np.arange(n_points)
            grad_true = current_h[(grad_idx - shift_true) % n_points]
            grad = 2 * current_h - grad_true + np.zeros_like(current_h)

        # Gradient descent step
        current_h -= learning_rate * grad
        current_h = project_simplex(current_h)

        # Update best solution
        current_c5 = compute_overlap(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            best_time = time.time()
            patience = 0
        else:
            patience += 1

        if patience >= patience_limit:
            print("Reducing learning rate")
            learning_rate *= 0.2
            patience = 0

        if learning_rate < 1e-6:
            break

    h_values = best_h.tolist()
    c5_bound = best_c5

    return h_values, c5_bound, n_points
