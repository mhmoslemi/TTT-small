import numpy as np

def run(seed=42, budget_s=120, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial_h_values if available
    try:

        n_points = initial_h.size
    except:
        n_points = 95  # Starting size; can be adjusted
        initial_h = np.random.rand(n_points)
        initial_h = (initial_h / initial_h.sum()) * (n_points / 2)
        initial_h = np.clip(initial_h, 0, 1)

    h = np.copy(initial_h)
    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h)

    step = 0
    max_steps = 100000
    learn_rate = 0.01
    no_improvement_patience = 100
    no_improvement_counter = 0

    while time.time() - start_time < budget_s and step < max_steps and no_improvement_counter < no_improvement_patience:
        step += 1

        # Compute overlaps
        lagged_overlaps = np.correlate(h, 1 - h, mode='full')
        lagged_overlaps *= 2.0 / n_points  # dx = 2.0 / n_points
        current_max_index = np.argmax(lagged_overlaps)
        current_max_value = lagged_overlaps[current_max_index]

        # Skip the self-overlap (k=0) if it's not the max, since it is "invalid" per the problem
        if current_max_index == n_points - 1:
            h = best_h
            no_improvement_counter += 1
            continue

        # Calculate gradient
        k = current_max_index
        gradient = (1 - h[k:] - h[:-k]) if k > 0 else (1 - h)

        # Gradient ascent on h to lower the max overlap (i.e., minimize current_max_value)
        h += learn_rate * (gradient / n_points)
        h = np.clip(h, 0, 1)

        # Renormalize h to satisfy the constraint ∫ h = 1
        h = (h / h.sum()) * (n_points / 2)

        # Evaluate new overlap
        new_c5 = evaluate_erdos_solution(h)

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = np.copy(h)
            no_improvement_counter = 0
        else:
            no_improvement_counter += 1

        # Later learning steps: reduce step size slightly
        if step > max_steps * 0.7:
            learn_rate *= 0.99

    return (best_h.tolist(), best_c5, n_points)
