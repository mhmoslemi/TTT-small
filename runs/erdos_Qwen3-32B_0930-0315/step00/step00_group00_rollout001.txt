import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points

    def project_simplex(v, s):
        # Fix sum to n_points / 2 and clamp [0, 1]
        v = np.clip(v, 0, 1)
        total = np.sum(v)
        if np.abs(total - s) > 1e-8:
            v = s * v / total
        return v

    def evaluate_C5(h):
        h1 = 1.0 - h
        correl = np.correlate(h, h1, mode='full')
        correl *= dx
        return np.max(correl)

    best_h = np.copy(h)
    best_c5 = evaluate_C5(best_h)
    t_start = time.time()

    it = 0
    learning_rate = 0.2
    decay_rate = 0.9999
    while True:
        # Exit condition: time budget
        if time.time() - t_start >= budget_s:
            break

        if it % 100 == 0:
            print(f"{it}: C5 = {best_c5} at step {it}", flush=True)

        # Current attempt
        h = np.copy(best_h)

        # Try a few steps of gradient descent
        for s in range(10):
            # Watch time
            if time.time() - t_start >= budget_s - 0.5:
                break
            h1 = 1.0 - h
            correl = np.correlate(h, h1, mode='same')  # only full vs same

            # Find maximum overlap index
            i_k = int(n_points / 2)
            max_val = -1
            max_k = 0
            for k in range(len(correl)):
                val = correl[k] * dx
                if val > max_val:
                    max_val = val
                    max_k = k

            # Compute gradient for maximum correlation lag
            grad = np.zeros_like(h)
            for i in range(n_points):
                for di in [0, 1, -1]:  # use nearest neighbors
                    j = (i + di) % n_points
                    grad[j] += h[i] if i != j else (h[j] - (1 - h[j]))

            # Gradient update
            h = np.clip(h - learning_rate * grad, 0, 1)
            h = project_simplex(h, n_points / 2)

            # Evaluation
            cur_c5 = evaluate_C5(h)
            if cur_c5 < best_c5:
                best_h = np.copy(h)
                best_c5 = cur_c5

            learning_rate *= 0.99  # optional linear decay

            it += 1

    return (best_h.tolist(), best_c5, n_points)
