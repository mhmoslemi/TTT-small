from scipy.signal import correlate

def evaluate_erdos_solution(h):
    dx = 2.0 / len(h)
    h_array = np.array(h) * dx
    overlap = correlate(h_array, 1 - h_array, mode='full')
    C5 = np.max(overlap) / dx  # normalize by dx to recover the original integrand max
    return C5
