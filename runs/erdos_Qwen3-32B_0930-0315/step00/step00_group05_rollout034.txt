    def project_to_simplex(v, c=1.0):
        v = np.clip(v, 0, 1)
        s = max(np.sum(v), c)
        if s < c:
            v += (c - s) / len(v)
        return v

        dx = 2.0 / n

            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap * dx

            h >= 0,
