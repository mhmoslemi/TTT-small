import numpy as np
import time
import cvxpy as cp

def correlate_full(h1, h2):
    n = len(h1)
    result = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        start1 = max(0, k - n + 1)
        start2 = max(0, n - k)
        end1 = min(n, k + 1)
        end2 = min(n, 2 * n - k)
        indices1 = np.arange(start1, end1)
        indices2 = np.arange(start2, end2)
        val = np.dot(h1[indices1], h2[indices2])
        result[k] = val
    return result

        h_vec = cp.Variable(n, pos=True)
        constraints = [h_vec >= 0, h_vec <= 1, cp.sum(h_vec) == n / 2.0]
        corr = correlate_full(h_vec, 1 - h_vec)
        max_overlap = cp.Max(corr * dx)
        barrier_regularizer = mu * cp.sum(cp.log(h_vec) + cp.log(1 - h_vec))
        objective = max_overlap + barrier_regularizer
        problem = cp.Problem(cp.Minimize(objective), constraints)
        problem.solve(solver=cp.ECOS)

        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_vals = h_vec.value
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals * (n / 2.0) / h_vals.sum()
            corr_val = correlate_full(h_vals, 1 - h_vals)
            c5_current = np.max(corr_val * dx)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_vals
