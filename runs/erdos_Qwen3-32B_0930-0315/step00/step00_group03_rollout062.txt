import numpy as np
import cvxpy as cp
import time
from math import isfinite

def np_correlate_max(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return corr * dx

def compute_c5(h):
    return np.max(np_correlate_max(h))

def project_to_simplex(v, dx):
    n = len(v)
    h = cp.Variable(n)
    constraints = [0 <= h, h <= 1, cp.sum(h) == n / 2]
    obj = cp.Minimize(cp.sum_squares(h - v))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=100)
        if not isfinite(prob.value):
            return v
        return np.array(h.value)
    except Exception:
        return v

def gradient(h, k):
    n = len(h)
    G = np.zeros_like(h)
    for i in range(n):
        j = (i - k + n) % n
        if j < n:
            G[j] += 1 - h[i]
            G[i] -= h[j]
    return G

def line_search(h, G, f, lr=0.01, max_steps=10, decay=0.5):
    best_h = h.copy()
    best_val = compute_c5(best_h)
    for _ in range(max_steps):
        h_candidate = h - lr * G
        h_projected = project_to_simplex(h_candidate, 2.0 / len(h))
        val = compute_c5(h_projected)
        if val < best_val:
            return h_projected, val
        lr *= decay
        if np.isclose(lr, 0):
            break
    return best_h, best_val

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    if 'initial_h_values' in globals():

    h = project_to_simplex(h, 2.0 / n_points)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    tolerance = 1e-6
    max_iter = 3000
    for iter_idx in range(max_iter):
        c5_vals = np_correlate_max(h)
        max_idx = np.argmax(np.abs(c5_vals))
        k = max_idx - (n_points - 1)
        G = gradient(h, k)
        if np.allclose(G, 0):
            break

        unit_g = G / (np.linalg.norm(G) + 1e-12)
        h_projected, new_c5 = line_search(h, unit_g, compute_c5)

        if new_c5 < best_c5:
            best_h = h_projected.copy()
            best_c5 = new_c5

        h = h_projected

        if time.time() - start_time > budget_s:
            break

        if iter_idx % 50 == 0:
            print(f"Iter {iter_idx} C₅: {best_c5:.8f}, elasped time: {time.time() - start_time:.2f}s")

    return (best_h.tolist(), best_c5, n_points)
