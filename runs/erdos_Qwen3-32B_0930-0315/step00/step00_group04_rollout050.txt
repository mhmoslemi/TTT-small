import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    def evaluate(h, n_points):
        dx = 2.0 / n_points
        # Compute cross-correlation
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        return np.max(c)

    def project(h, n_points):
        # Clip to [0, 1]
        h = np.clip(h, 0.0, 1.0)
        # Enforce sum(h) = n_points / 2 (integral over [0,2] is 1)
        s = h.sum()
        h = h * (n_points / 2.0) / s
        return h

    global initial_h_values
    h = np.array(initial_h_values)  # Initial guess with n_points = 42
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate(h, n)

    # Optimization loop
    max_iter = 1000
    start_time = time.time()
    lr = 0.01
    lr_decay = 0.95
    lr_update_freq = 100

    for iter_idx in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

        c = np.correlate(h, 1.0 - h, mode='full') * dx
        k_max = np.argmax(c)

        # Compute gradient direction (simplified approximation)
        gradient = np.zeros_like(h)
        if 0 <= k_max < n:
            gradient = (1.0 - h)
        elif n <= k_max < 2 * n - 1:
            gradient = (1.0 - h[-(k_max - n + 1):])  # Shift right by k_max - n + 1
        elif k_max >= 2 * n:
            gradient = (1.0 - h Citizen h[:k_max - 2 * n + 1])  # Shift left by k_max - 2*n + 1

        h -= lr * gradient
        h = project(h, n)

        current_c5 = evaluate(h, n)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # LR scheduling
        if iter_idx % lr_update_freq == 0:
            lr *= lr_decay

    return (best_h.tolist(), best_c5, n)
