import numpy as np
import time
import evaluate_erdos_solution as evaluator

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 160  # Number of sample points (chosen for accuracy and speed tradeoff)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values if len(initial_h_values) == n_points else np.tile(initial_h_values, n_points // len(initial_h_values) + 1)[:n_points]
    else:
        h = np.linspace(0.5, 0.5, n_points)
    
    # Enforce sum constraint
    h = h / h.sum() * target_sum
    
    # Optimization parameters
    alpha = 0.01
    max_iter = 5000
    stagnation_limit = 100
    perturb_iter = 500
    best_h = h.copy()
    best_c5 = evaluator.evaluate_erdos_solution(best_h)[0]
    last_improvement = 0

    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break

        # Compute correlation to find max overlap
        correl = np.correlate(h, 1 - h, mode="full")
        max_overlap_idx = correl.argmax()
        k_max = max_overlap_idx - n_points + 1

        # Gradient of C5 w.r.t. h at worst lag
        gradient = np.zeros(n_points)
        for j in range(n_points):
            idx1 = (j + k_max) % n_points
            idx2 = (j - k_max) % n_points
            gradient[j] = (1 - h[idx1]) - h[idx2]

        # Gradient update
        h = h - alpha * gradient

        # Project to feasible set: [0,1] and sum = target_sum
        h = np.clip(h, 0, 1)
        h = h / h.sum() * target_sum

        # Evaluate current C5 bound
        c5 = evaluator.evaluate_erdos_solution(h)[0]

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = i
        elif i - last_improvement > stagnation_limit:
            # Introduce random perturbation to escape local minima
            perturbation = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + perturbation, 0, 1)
            h = h / h.sum() * target_sum
            last_improvement = i

    return best_h, best_c5, n_points
