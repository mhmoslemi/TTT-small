import numpy as np
import time
import random
from evaluate_erdos_solution import evaluate_erdos_solution
import scipy.interpolate
from scipy.signal import correlate

def compute_c5(h, dx):
    n_points = len(h)
    corr = correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def mutate_and_project(h, dx, mutation_rate=0.1):
    n_points = len(h)
    h_mut = h.copy()
    indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
    h_mut[indices] += np.random.uniform(-0.5, 0.5, size=len(indices))
    h_mut = np.clip(h_mut, 0, 1)
    rescale_factor = (n_points / 2) / np.sum(h_mut)
    h_mut *= rescale_factor
    return h_mut

def tournament_selection(population, k=5):
    candidates = random.choices(population, k=k)
    return min(candidates, key=lambda x: x[1])

def crossover(parent1, parent2):
    n_points = len(parent1)
    split = np.random.randint(1, n_points - 1)
    child = np.concatenate((parent1[:split], parent2[split:]))
    rescale_factor = (n_points / 2) / np.sum(child)
    return child * rescale_factor

def gradient_step(h, dx, learning_rate=0.01):
    n_points = len(h)
    h_gd = h.copy()
    C = correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(C)
    k = max_idx - n_points + 1
    if k < n_points:  # Only valid shift indices
        h_shifted = np.concatenate((np.zeros(k), h[:n_points - k]))
        gradient = (1 - h_shifted) * learning_rate * dx
        h_gd += gradient
        h_gd = np.clip(h_gd, 0, 1)
        rescale_factor = (n_points / 2) / np.sum(h_gd)
        h_gd *= rescale_factor
    return h_gd

def upscale_h(h, factor=2):
    n_points = len(h)
    x = np.linspace(0, 2, n_points)
    f = scipy.interpolate.CubicSpline(x, h, bc_type='natural', extrapolate=False)
    x_fine = np.linspace(0, 2, n_points * factor)
    h_fine = f(x_fine)
    h_fine = np.clip(h_fine, 0, 1)
    rescale_factor = (len(h_fine) / 2) / np.sum(h_fine)
    h_fine *= rescale_factor
    return h_fine

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        n0 = len(h_initial)
    else:
        n0 = 14
        h_initial = np.random.uniform(0.2, 0.8, size=n0)
        h_initial = h_initial / h_initial.sum() * n0 / 2

    n_points_list = [14, 28, 56]
    best_h = h_initial
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))

    for n_points in n_points_list:
        dx = 2.0 / n_points
        population_size = 20
        population = []

        # Warmstart from last best_h if available
        if n_points != len(best_h):
            if 'initial_h_values' in globals() and n_points == len(h_initial):
                h_base = h_initial
            else:
                h_base = upscale_h(best_h, factor=n_points // len(best_h))
        else:
            h_base = best_h

        # Initialize population
        for _ in range(population_size):
            h = mutate_and_project(h_base, dx)
            c5 = compute_c5(h, dx)
            population.append((h, c5))

        ga_time_budget = int(0.7 * (budget_s - (time.time() - start_time)))
        t0 = time.time()
        while time.time() - t0 < ga_time_budget:
            new_population = []

            # Selection, crossover, and mutation
            for _ in range(population_size):
                parent1 = tournament_selection(population)
                parent2 = tournament_selection(population)
                child = crossover(parent1[0], parent2[0])
                child = mutate_and_project(child, dx)
                c5_child = compute_c5(child, dx)
                new_population.append((child, c5_child))

            # Replace population with new_population
            population = sorted(population + new_population, key=lambda x: x[1])[:population_size]

            # Track progress
            current_best = min(population, key=lambda x: x[1])
            if current_best[1] < best_c5:
                best_h = current_best[0]
                best_c5 = current_best[1]

        # Apply gradient descent
        for _ in range(100 + n_points // 14):
            if time.time() - start_time > budget_s:
                break
            best_h = gradient_step(best_h, dx)
            best_c5 = compute_c5(best_h, dx)

        # Final repair and evaluation
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h / best_h.sum() * n_points / 2
        final_c5 = compute_c5(best_h, dx)

    elapsed = time.time() - start_time
    return (best_h.tolist(), final_c5, len(best_h))
