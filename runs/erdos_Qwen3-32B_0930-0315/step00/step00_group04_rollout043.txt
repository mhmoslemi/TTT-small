import numpy as np
import time
import math

def evaluate_erdos_solution(h):
    dx = 2.0 / len(h)
    C = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(C)

def repair_feasibility(h, n_points):
    h_clipped = np.clip(h, 0.0, 1.0)
    h_scaled = h_clipped * (n_points / 2.0) / np.sum(h_clipped)
    return h_scaled

def simulated_annealing(h, n_points, dx, max_steps=1000, initial_temp=1.0, cooling_rate=0.99):
    h_best = h.copy()
    c5_best = evaluate_erdos_solution(h_best)
    temp = initial_temp

    for _ in range(max_steps):
        h_new = h + np.random.normal(0, 0.1, size=n_points)
        h_new = repair_feasibility(h_new, n_points)
        c5_new = evaluate_erdos_solution(h_new)

        if c5_new < c5_best or np.random.rand() < np.exp((c5_best - c5_new) / max(temp, 1e-6)):
            h_best, c5_best = h_new, c5_new
            h = h_new

        temp *= cooling_rate
        if temp < 1e-6:
            break

    return h_best, c5_best

def interpolate_h(h, resolution):
    h_interpolated = np.interp(np.linspace(0, 2, resolution), np.linspace(0, 2, len(h)), h)
    h_interpolated = repair_feasibility(h_interpolated, resolution)
    return h_interpolated

def run(seed=42, budget_s=1100):
    time_start = time.time()
    np.random.seed(seed)

    # Initial conditions from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
        n_points = max(14, n_points)
    else:
        # Start with a random h with uniform distribution
        h = np.ones(14) / 14 * 7
        n_points = 14

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    # Annealing + refinement loop
    n_points = 14
    dx = 2.0 / n_points

    while time.time() - time_start < budget_s and n_points <= 256:
        # Simulated Annealing Phase
        h_current, c5_current = simulated_annealing(h, n_points, dx, max_steps=300, initial_temp=1.0)

        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            print(f"Improved to C₅ = {best_c5:.6f} at resolution {n_points}")

        # Interpolate to next resolution
        n_points = min(2 * n_points, 256)
        h = interpolate_h(best_h, n_points)
        dx = 2.0 / n_points

    # Final clean-up
    if time.time() - time_start < budget_s:
        # Extra optimization
        steps_remaining = int((budget_s - (time.time() - time_start)) * 100)
        h = simulated_annealing(best_h, n_points, dx, max_steps=steps_remaining, initial_temp=0.5)[0]
        best_c5 = evaluate_erdos_solution(h)
        best_h = h.copy()

    return best_h.tolist(), best_c5, n_points
