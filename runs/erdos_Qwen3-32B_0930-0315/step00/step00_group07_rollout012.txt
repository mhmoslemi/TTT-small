import numpy as np
import cvxpy as cp
import math
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    dx = 2.0 / 80
    n_points = 80
    max_iter = 500
    patience = 100
    if 'initial_h_values' in globals():
        h = initial_h_values
        # Validate constraints
        assert np.all(h >= 0) and np.all(h <= 1), "initial h is out of bounds"
        assert np.isclose(np.sum(h), n_points / 2), "initial h violates integral constraint"
    else:
        h = np.full(n_points, 0.5)  # Uniform initial guess
    best_h = h.copy()
    best_c5 = float('inf')

    start = time.time()

    tolerance = 1e-7
    prev = float('inf')

    while time.time() - start < budget_s:
        shift = int(5.0 / dx)  # fixed shift for C₅, since k = 5
        shifted = np.roll(h, -shift)
        C_k = np.mean(h * (1 - shifted))  # dx = 2.0 / n_points; mean = * dx

        # Formulate convex subproblem for fixed lag k
        h_opt = cp.Variable(n_points)
        objective = cp.Minimize(np.mean(h_opt * (1 - np.roll(h_opt, -shift))))
        constraints = [
            0 <= h_opt, h_opt <= 1,
            cp.sum(h_opt) == n_points / 2
        ]
        problem = cp.Problem(objective, constraints)

        try:
            problem.solve(solver=cp.ECOS, max_iters=10000)
            h_candidate = np.clip(h_opt.value, 0.0, 1.0)
        except cp.SolverError:
            h_candidate = h

        # Line search & backtracking
        for alpha in np.geomspace(1.0, 0.001, 20):
            h_eval = (1 - alpha) * h + alpha * h_candidate
            h_eval = np.clip(h_eval, 0.0, 1.0)
            h_eval -= (np.sum(h_eval) - n_points / 2) / n_points  # adjust to maintain constraint

            # Enforce constraint exactly
            excess = np.sum(h_eval) - n_points / 2
            h_eval -= excess / n_points

            # Evaluate this candidate
            shifted = np.roll(h_eval, -shift)
            C_k_eval = np.mean(h_eval * (1 - shifted))

            if C_k_eval < best_c5:
                best_h = h_eval.copy()
                best_c5 = C_k_eval
                h = h_eval.copy()
                break
            h = h_eval.copy()

        # Check for convergence
        if best_c5 > prev - tolerance:
            patience -= 1
            if patience <= 0:
                break
        else:
            patience = 100
            prev = best_c5

    # Final validation
    h = best_h
    h = np.clip(h, 0.0, 1.0)
    excess = np.sum(h) - n_points / 2
    h -= excess / n_points
    shifted = np.roll(h, -shift)
    final_c5 = np.mean(h * (1 - shifted))

    return h, final_c5, n_points
