import numpy as np
import math
import time
from scipy.signal import correlate


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values
    n_points = len(h)
    dx = 2.0 / n_points
    h = h.astype(float)
    h *= n_points / 2.0 / h.sum()  # Normalize to satisfy the integral constraint

    start_time = time.perf_counter()
    best_h = h.copy()
    best_c5 = float('inf')
    patience_counter = 0
    improvement_threshold = 1e-5
    early_stop_limit = 2000
    k_max_history = []

    while time.perf_counter() - start_time < budget_s:
        overlap = correlate(h, 1 - h, mode="full")
        overlap = overlap[n_points:]  # Only consider k >= 1 (positive overlaps)
        k_max = np.argmax(overlap)
        k_max = min(k_max, n_points - k_max - 1)  # Choose nearest shift

        # Compute gradient for the dominant shifting constraint
        grad = np.zeros(n_points)
        for i in range(n_points):
            grad[i] = (1 - h[(i + k_max) % n_points]) - h[(i - k_max) % n_points]

        grad *= dx  # Account for dx in the integral

        # Adaptive step size (Barzilai-Borwein approximation)
        if 'h_prev' in locals():
            s = h - h_prev
            y = grad - grad_prev
            alpha = np.dot(s, y) / np.dot(y, y)
            alpha = np.clip(alpha, 0.01, 1e4)
        else:
            alpha = 0.1

        h_prev = h.copy()
        grad_prev = grad.copy()

        # Gradient update
        h -= alpha * grad

        # Projection onto [0,1]
        with np.errstate(invalid='ignore'):
            h = np.clip(h, 0, 1)

        # Enforce integral constraint
        h *= n_points / 2.0 / h.sum()

        # Evaluate
        overlap_all = correlate(h, 1 - h, mode="full") * dx
        overlap_all = overlap_all[n_points:]
        current_c5 = np.max(overlap_all)

        if current_c5 < best_c5 - improvement_threshold:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1

        if patience_counter >= 100:
            # Perturbation to help escape local optima
            indices = np.random.choice(n_points, size=np.random.randint(5, 15), replace=False)
            h[indices] = np.random.uniform(0, 1, len(indices))
            h *= n_points / 2.0 / h.sum()
            patience_counter = 0

        if patience_counter >= early_stop_limit:
            break

    overlap_all = correlate(best_h, 1 - best_h, mode="full") * dx
    overlap_all = overlap_all[len(best_h):]
    final_c5 = np.max(overlap_all)
    return best_h, final_c5, n_points
