import numpy as np
import cvxpy as cp
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Use initial construction provided by the harness
    h = initial_h_values
    n_points = len(h)
    dx = 2.0 / n_points

    # Ensure h is feasible
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5, _ = evaluate_erdos_solution(best_h)
    best_time = time.time()

    h_var = cp.Variable(n_points)
    c_var = cp.Parameter(n_points)

    constraints = [
        0 <= h_var,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2
    ]

    objective = cp.Minimize(cp.sum(cp.multiply(h_var, 1 - cp.roll(c_var, shift=0))))
    prob = cp.Problem(objective, constraints)

    start_time = time.time()
    while True:
        current_time = time.time()
        if current_time - start_time > budget_s:
            break

        # Compute all lags
        current_c5, corr_values = evaluate_erdos_solution(h)
        max_lag_idx = np.argmax(corr_values)

        # Set the worst lag as the rolling pattern
        c_var.value = 1 - np.roll(h, shift=max_lag_idx % n_points)

        # Solve the convex problem
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_candidate = h_var.value
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = h_candidate / np.sum(h_candidate) * (n_points / 2)
                h_candidate = np.clip(h_candidate, 0, 1)
                c5_candidate, _ = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
                    best_time = current_time

                h = h_candidate.copy()
        except Exception as e:
            print(f"Warning: Solver error - {e}, reverting to best solution")
            h = best_h.copy()

        # Early stopping on minimal improvement (commented out for safety)
        # if current_time - best_time > 10 and best_c5 < 0.380876:
        #     break

    return (best_h.tolist(), best_c5, n_points)
