import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the provided initial_h_values if available
    if "initial_h_values" in globals():
        n_points = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        # Fallback to a uniform distribution if no initial solution is provided
        n_points = 47
        h0 = np.linspace(0, 1, n_points)  # Dummy fallback, random is better
        h0 = np.clip(h0, 0, 1)
        h0 = h0 / h0.sum() * (n_points / 2)  # Normalize to the required integral
    
    # Define variables
    h = cp.Variable(n_points, nonneg=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.sum(h) == n_points / 2,  # Integral constraint
        h >= 0, h <= 1  # Bounds
    ]
    
    dx = 2.0 / n_points

    for k in range(-n_points + 1, n_points):
        overlap = 0
        for i in range(max(0, -k), min(n_points, n_points - k)):
            overlap += h[i] * (1 - h[i + k]) * dx
        constraints.append(overlap <= t)

    # Form and solve the problem
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    # Use a faster solver if possible, ensure PDLP is available (cvxpy 1.2+)
    try:
        result = prob.solve(solver=cp.PDLP, verbose=False)
    except:
        result = prob.solve(solver=cp.CVXOPT, verbose=False)
    
    # Extract result
    h_values = np.clip(np.array(h.value).flatten(), 0, 1)
    h_values_sum = np.sum(h_values)
    
    # Enforce the sum constraint in floating-point errors
    if abs(h_values_sum - n_points / 2) > 1e-6:
        h_values = h_values / h_values_sum * (n_points / 2)

    h_values = h_values.tolist()
    c5_bound = t.value if t.value is not None else np.inf
    elapsed_time = time.time() - start_time
    
    if elapsed_time >= budget_s:
        print("Reached time budget limit and returned best result so far.")
    
    return h_values, c5_bound, n_points
