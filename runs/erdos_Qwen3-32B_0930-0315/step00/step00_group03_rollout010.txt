import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # 1. Initialize n_points from the parent construction if available
    initial_h = initial_h_values() if "initial_h_values" in globals() else None

    if initial_h is not None:
        n_points = len(initial_h)
    else:
        n_points = 70  # default to matching known parent

    dx = 2.0 / n_points
    target_sum = n_points // 2  # ∫ h(x) dx = 1 implies sum(h) == n_points / 2

    pop_size = 50
    generations = 0

    # 2. Initialize population
    population = []

    if initial_h is not None:
        population.append(initial_h.copy())
        # Perturb small variations from initial_h
        for _ in range(pop_size - 1):
            h = initial_h.copy()
            idxs = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
            h[idxs] += (np.random.rand(len(idxs)) - 0.5) * 0.2
            h = np.clip(h, 0, 1)
            scale = target_sum / np.sum(h)
            h = np.clip(h * scale, 0, 1)
            population.append(h)
    else:
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            scale = target_sum / np.sum(h)
            h = np.clip(h * scale, 0, 1)
            population.append(h)

    t0 = time.time()
    best_h = None
    best_c5 = float('inf')

    while time.time() - t0 < budget_s:
        next_gen = []

        for i in range(0, len(population), 2):
            if i + 1 >= len(population):
                break

            p1, p2 = population[i], population[i+1]

            # Crossover
            cxpt = random.randint(0, n_points - 1)
            c1 = np.concatenate((p1[:cxpt], p2[cxpt:]))
            c2 = np.concatenate((p2[:cxpt], p1[cxpt:]))

            # Repair: normalize to sum constraint
            for child in [c1, c2]:
                s = target_sum / np.sum(child)
                child = np.clip(child * s, 0, 1)
                next_gen.append(child)

        # Mutation
        for h in next_gen:
            if random.random() < 0.15:
                idxs = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                h[idxs] += (np.random.rand(len(idxs)) - 0.5) * 0.1
                h = np.clip(h, 0, 1)
                s = target_sum / np.sum(h)
                h = np.clip(h * s, 0, 1)

        # Evaluate population and keep best
        population = population + next_gen
        scores = []
        for h_val in population:
            corr = np.correlate(h_val, 1 - h_val, mode="full")
            c5 = np.max(corr) * dx
            scores.append((h_val, c5))

        # Select top population
        scores.sort(key=lambda x: x[1])
        population = [h for h, _ in scores[:pop_size]]

        # Track best solution
        best_so_far = scores[0]
        if best_so_far[1] < best_c5:
            best_c5 = best_so_far[1]
            best_h = best_so_far[0].copy()

        generations += 1

    if best_h is None:
        best_h = population[np.argmin([np.max(np.correlate(h, 1 - h, mode="full")) * dx for h in population])].copy()
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx

    # Final check of constraints
    assert np.all((best_h >= 0) & (best_h <= 1)), "Solution out of bounds [0,1]"
    assert np.abs(np.sum(best_h) - target_sum) < 1e-5, "Integral constraint violated"

    return (best_h, best_c5, n_points)
