import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    h = np.copy(initial_h_values)
    n_points = h.shape[0]
    dx = 2.0 / n_points

    # Ensure feasibility
    h = np.clip(h, 0, 1)
    h *= n_points / 2 / h.sum()

    alpha = 0.1
    max_iter = 10000
    perturb_interval = 50
    decay_factor = 0.5
    best_h = np.copy(h)
    best_c5 = np.inf
    no_improve_count = 0
    start_time = time.time()

    for it in range(max_iter):
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break

        h = np.clip(h, 0, 1)
        h *= n_points / 2 / h.sum()

        # Compute correlation and identify worst k
        corr = correlate(h, 1 - h, mode='full')
        resp = corr * dx
        worst_k = np.argmax(resp)
        max_c5 = resp[worst_k]

        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = np.copy(h)
            no_improve_count = 0
        else:
            no_improve_count += 1

        # When no improvement for 10 iterations, decay step size
        if no_improve_count >= 10:
            alpha *= decay_factor

        # Compute gradient from the worst overlap
        grad = np.zeros_like(h)

        for i in range(n_points):
            if 0 <= i + worst_k < n_points:
                grad[i] = (1 - h[i + worst_k]) * dx
            if 0 <= i - worst_k < n_points:
                grad[i] -= h[i - worst_k] * dx

        # Gradient step
        h -= alpha * grad
        h = np.clip(h, 0, 1)
        h *= n_points / 2 / h.sum()

        if it % perturb_interval == 0:
            # Random perturbation to escape local minima
            perturb = np.random.normal(0, 0.01, size=h.shape)
            h += perturb
            h = np.clip(h, 0, 1)
            h *= n_points / 2 / h.sum()

    return (best_h, best_c5, n_points)
