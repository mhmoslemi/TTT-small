import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h = np.copy(initial_h_values)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()
    learning_rate = 0.05
    n_decay = 100
    perturb_freq = 200

    def evaluate_h(_h):
        _h = np.clip(_h, 0, 1)
        _h = (_h * (n_points / 2)) / _h.sum() if _h.sum() != 0 else _h
        return evaluate_erdos_solution(_h), _h

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        c5 = np.max(corr) * dx

        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

        # Gradient update
        grad = np.zeros(n_points)
        for lag in [np.argmax(corr)]:
            for i in range(max(0, lag - n_points + 1), min(n_points, lag + 1)):
                grad[i] += (1 - h[i + n_points - lag]) if (i + n_points - lag) < n_points else 0
                grad[i] -= h[i] if (i + n_points - lag) < n_points else 0

        h -= learning_rate * grad

        # Projection onto feasibility set
        h = np.clip(h, 0, 1)
        if np.abs(h.sum() - n_points / 2) > 1e-6:
            h = h * (n_points / 2) / h.sum()

        # Perturbation
        if np.random.rand() < 0.02 and time.time() - start_time < budget_s:
            h += np.random.laplace(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / h.sum()

        learning_rate *= 0.998

    return best_h, best_c5, n_points
