import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution, initial_h_values


def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    # Start with the existing initial_h_values
    h = np.copy(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points

    best_h = np.copy(h)
    best_c5 = float("inf")

    # Precompute domain for valid k
    K_domain = np.arange(n_points)

    # Optimization parameters
    T_max = int(budget_s / 0.1)  # Conservative estimate of iteration budget
    T_stall_limit = 200
    t_stall = 0
    lr = 0.01
    lr_decay = 0.95
    use_initial_h = h is not None

    start_time = time.time()

    for t in range(1, T_max + 1):
        # Compute current value of C_5
        with np.errstate(all="ignore"):
            c5 = evaluate_erdos_solution(h)

        if c5 < best_c5:
            best_h = np.copy(h)
            best_c5 = c5
            t_stall = 0
            lr *= lr_decay  # Optional: increase reward period
        elif t_stall >= T_stall_limit:
            lr *= lr_decay  # Reset learning rate on prolonged stall
            t_stall = 0
        else:
            t_stall += 1

        if time.time() - start_time > budget_s - 1.0:
            break

        # Get max convolution index (k_max)
        with np.errstate(all="ignore"):
            corr = np.correlate(h, 1 - h, mode="full")
            corr = corr[n_points - 1 : 2 * n_points - 1]
            k_max = np.argmax(corr)

        # Gradient dC5/dh[i]
        def grad(i, k):
            prev_idx = (i - k) % n_points
            next_idx = (i + k) % n_points
            return 1.0 - h[prev_idx] - h[next_idx]

        grad_vals = np.array([grad(i, k_max) for i in range(n_points)])
        h_grad = grad_vals

        # Gradient ascent to minimize the maximum convolution
        h_update = h - lr * h_grad

        # Clip to [0, 1]
        h_clipped = np.clip(h_update, 0, 1)

        # Rescale to satisfy the sum constraint
        sum_h = np.sum(h_clipped)
        scale = (n_points / 2) / sum_h
        h = h_clipped * scale

    # Return best solution
    return best_h, best_c5, n_points
