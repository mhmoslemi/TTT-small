import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h - 0.5, 2 * h - 1, mode="full")
    C5 = np.max(np.abs(corr * dx))
    assert np.allclose(np.sum(h) * dx, 1.0, atol=1e-8), "Constraint violated: integral not 1"
    return C5

def run(seed=42, budget_s=1100, n_points=45):

    np.random.seed(seed)

    h = initial_h_values
    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(best_h, n_points)

    start_time = time.time()
    t0 = time.time()

    def time_left():
        return budget_s - (time.time() - t0)

    n_neighbors = 5
    refinement_steps = 500

    while time_left() > 0 and refinement_steps > 0:
        refinement_steps -= 1

        # Evaluate current C5
        dx = 2.0 / n_points
        corr = correlate(h - 0.5, 2 * h - 1, mode="full")
        c5 = np.max(np.abs(corr * dx))

        # Identify the dominant k
        max_idx = np.argmax(np.abs(corr))
        if max_idx > n_points:
            k = max_idx - n_points
        else:
            k = max_idx

        # Compute redundancy heuristic
        R = np.zeros(n_points)
        for i in range(n_points):
            neighborhood = [j for j in range(max(i - n_neighbors, 0), min(i + n_neighbors, n_points))]
            for j in neighborhood:
                R[i] += np.abs(corr[j + k] if j + k < len(corr) else 0)

        # Split into high/low influence regions
        threshold = np.quantile(R, 0.75)
        high_influence = np.where(R > threshold)[0]

        # Optimize high-influence regions with small steps
        for i in high_influence:
            for dh in [-0.05, 0.05, 0.01, -0.01, 0.025, -0.025]:
                trial_h = np.copy(h)
                trial_h[i] += dh
                idx = np.where(trial_h < 0)[0]
                trial_h[idx] = 0.0
                idx = np.where(trial_h > 1)[0]
                trial_h[idx] = 1.0

                # Enforce budget constraint
                delta = trial_h.sum() - n_points / 2
                trial_h += -delta / n_points
                idx = np.where(trial_h < 0)[0]
                trial_h[idx] = 0.0
                idx = np.where(trial_h > 1)[0]
                trial_h[idx] = 1.0

                trial_c5 = evaluate_erdos_solution(trial_h, n_points)

                if trial_c5 < best_c5:
                    best_h[:] = trial_h[:]
                    best_c5 = trial_c5
                    h[:] = trial_h[:]
                    break
            if time_left() <= 0:
                break

        # Quadratic programming attempt
        if time_left() > 0:
            x = cp.Variable(n_points)
            objective = cp.MaxEntr(np.abs(np.correlate(x, 1 - x, mode="full")))
            constraints = [x >= 0, x <= 1, cp.sum(x) == n_points / 2]
            prob = cp.Problem(cp.Minimize(objective), constraints)
            try:
                prob.solve(optimizer='ECOS', max_itrs=50, verbose=False)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    qp_h = np.minimum(np.maximum(np.array(x.value), 0), 1)
                    qp_c5 = evaluate_erdos_solution(qp_h, n_points)
                    if qp_c5 < best_c5:
                        best_h[:] = qp_h[:]
                        best_c5 = qp_c5
            except Exception:
                pass

        if best_c5 < evaluate_erdos_solution(h, n_points):
            h[:] = best_h[:]
            print(f"Updated to {best_c5}")

    return best_h, best_c5, n_points
