import numpy as np
import cvxpy as cp
import time
import evaluate_erdos_solution


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    st_time = time.time()
    n_points = 80
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    while time.time() - st_time < budget_s * 0.95:  # leave time for final evaluation
        # Compute overlap for all lags
        h_float = h.astype(np.float64)
        overlaps = np.correlate(h_float, 1.0 - h_float, mode='full') * dx
        max_idx = np.argmax(np.abs(overlaps))
        k = max_idx - n_points + 1
        max_overlap = overlaps[max_idx]

        # Compute gradient
        grad = np.zeros(n_points)
        kl = abs(k)
        for i in range(n_points):
            r1 = (i + kl) % n_points
            r2 = (i - kl) % n_points
            grad[i] = 1.0 - h[r1] - h[r2]
        grad = grad * dx

        # Solve LP
        delta = cp.Variable(n_points)
        objective = cp.Minimize(np.dot(grad, delta))
        constraints = [
            h + delta >= 0,
            h + delta <= 1,
            cp.sum(delta) == 0
        ]
        prob = cp.Problem(objective, constraints)
        try:
            result = prob.solve(solver=cp.ECOS)
        except:
            continue

        if result is None or prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            continue

        new_h = (h + delta.value).copy()
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        correction = (n_points / 2.0 - new_h_sum) / n_points
        new_h += correction
        new_h = np.clip(new_h, 0, 1)

        # Recompute sum to enforce constraint
        assert np.isclose(np.sum(new_h), n_points / 2), np.sum(new_h)
        h = new_h

        # Evaluate improvement
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    return best_h.tolist(), best_c5, n_points
