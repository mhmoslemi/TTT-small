import numpy as np
import scipy
import time
import cvxpy as cp
from collections import deque

def evaluate_C5(h, dx):
    """Compute the maximum overlap integral (C₅) for a given step function h"""
    n_points = len(h)
    circ_h = np.concatenate([h, h])
    max_overlap = 0.0
    for k in range(1, n_points):
        shifted = circ_h[k:k + n_points]
        overlap = np.sum(h * (1 - shifted))
        max_overlap = max(max_overlap, overlap * dx)
    return max_overlap

def compute_overlap_integral(h, dx):
    """Compute overlap integral using circular convolution"""
    n_points = len(h)
    h_double = np.concatenate([h, h])
    c = np.correlate(h, 1 - h, mode='full')
    c = c[: len(h_double)] * dx
    return c

def compute_gradient(h, dx):
    """Compute the gradient of C₅ w.r.t. h"""
    n_points = len(h)
    overlap = compute_overlap_integral(h, dx)
    max_idx = np.argmax(overlap[:n_points])
    
    grad = np.zeros(n_points)
    h_double = np.concatenate([h, h])
    
    for i in range(n_points):
        # First term: h[i] * (1 - h[i + k])
        grad[i] += dx * (1.0 - h[(i + max_idx) % n_points])
        # Second term: -h[i + k] * h[i]
        grad[i] -= dx * h[(i + max_idx) % n_points]
    
    return grad

def project_feasibility(h, dx, target_sum=None):
    """Project h onto the feasible set [0,1]^n with ∑h = n/2"""
    h = np.clip(h, 0, 1)
    if target_sum is None:
        target_sum = len(h) / 2
    current_sum = np.sum(h)
    scaling = target_sum / current_sum if current_sum != 0 else 1.0
    return np.clip(h * scaling, 0, 1)

def gradient_descent_phase(h, dx, n_iter=2000):
    """Phase 1: Gradient descent with backtracking line search"""
    n_points = len(h)
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    h = project_feasibility(h, dx)
    
    for i in range(n_iter):
        grad = compute_gradient(h, dx)
        step_size = 0.01
        
        for _ in range(5):
            # Line search
            new_h = h - step_size * grad
            new_h = project_feasibility(new_h, dx)
            new_C5 = evaluate_C5(new_h, dx)
            
            if new_C5 < best_C5:
                h = new_h.copy()
                best_C5 = new_C5
                best_h = h.copy()
                print(f"Iteration {i}: C5={best_C5:.8f}")
                break
        
        step_size *= 0.95  # Reduce step size if no progress
    
    return best_h, best_C5

def refine_with_cvxpy(h, dx):
    """Phase 2: Refinement using quadratic programming"""
    n_points = len(h)
    h_var = cp.Variable(n_points, nonneg=True)
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2
    ]
    
    # Objective: Minimize C₅
    overlap = compute_overlap_integral(h, dx)
    max_idx = np.argmax(overlap[:n_points])
    
    objective = cp.Minimize(
        dx * cp.sum(h_var * (1 - h_var[max_idx % n_points : (max_idx + n_points) % n_points]))
    )
    
    problem = cp.Problem(objective, constraints)
    
    try:
        # First try with ECOS solver
        problem.solve(solver=cp.ECOS)
    except:
        # Fallback to SCS solver
        problem.solve(solver=cp.SCS)
    
    if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return np.clip(h_var.value, 0, 1)
    return h

def add_noise(h, noise_level=0.0001):
    """Add small random perturbation to escape local minima"""
    n_points = len(h)
    noise = np.random.normal(0, noise_level, size=n_points)
    h = h + noise
    return project_feasibility(h, dx)

def smooth_window(h, window_size=5):
    """Apply moving average smoothing to regularize solution"""
    n_points = len(h)
    weights = np.hanning(window_size)
    weights /= weights.sum()
    h_smoothed = np.convolve(h, weights, mode='same')
    return project_feasibility(h_smoothed, dx)

def run(seed=42, budget_s=1100, n_points=160):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    if 'initial_h_values' in globals():
        # Use the provided initial solution
        h = initial_h_values.copy()
        h = np.resize(h, n_points)  # Ensure correct size
        h = h / np.sum(h) * n_points / 2  # Normalize to have sum = n/2
    else:
        # Construct basic initial solution: plateau shape
        h = np.zeros(n_points)
        plateau_width = n_points // 2
        h[n_points//2 - plateau_width//2 : n_points//2 + plateau_width//2] = 1.0
        h = h / np.sum(h) * n_points / 2
    
    # Normalize to have sum = n/2
    h = h / np.sum(h) * n_points / 2
    
    # Compute dx for all calculations
    dx = 2.0 / n_points
    
    # Start with basic projection
    h = project_feasibility(h, dx)
    start_time = time.time()
    
    # Phase 1: Gradient descent optimization
    h, best_C5 = gradient_descent_phase(h, dx, n_iter=2000)
    
    # Track progress
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    # Time remaining for phase 2
    time_used = time.time() - start_time
    remaining = budget_s * 0.7 - time_used
    
    while remaining > 5 and remaining > 0:
        # Refinement with CVXPY
        h = refine_with_cvxpy(h, dx)
        h = project_feasibility(h, dx)
        current_C5 = evaluate_C5(h, dx)
        
        if current_C5 < best_C5:
            best_h = h.copy()
            best_C5 = current_C5
            
        print(f"CVXPY step: C₅ = {best_C5:.8f}")
        
        # Add noise and smooth for exploration
        h = add_noise(h)
        h = smooth_window(h)
        h = project_feasibility(h, dx)
        
        # Update time tracking
        time_used = time.time() - start_time
        remaining = budget_s * 0.7 - time_used
    
    # Final evaluation
    h = best_h
    h = project_feasibility(h, dx)
    c5_bound = evaluate_C5(h, dx)
    
    return h.tolist(), c5_bound, n_points
