import numpy as np
import cvxpy as cp
from time import perf_counter
import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Use initial_h_values provided by the parent if available
    global initial_h_values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
        assert all(0 <= x <= 1 for x in h), "Invalid initial values"
        assert np.isclose(np.sum(h), n_points / 2), "Initial h violates the integral constraint"
    else:
        # Initialize with 80 uniform samples and sum=1
        n_points = 80
        h = np.zeros(n_points)
        h[::2] = 1
        h = h / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = perf_counter()

    while perf_counter() - start_time < budget_s:
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        max_k = np.argmax(overlaps)

        # Formulate and solve subproblem: balance inner convolution
        k = max_k
        h_var = cp.Variable(n_points)
        constraints = [0 <= h_var, h_var <= 1]
        constraints.append(cp.sum(h_var) == n_points / 2)
        objective = cp.sum(h_var * (1 - cp.roll(h_var, -k))) * dx
        problem = cp.Problem(cp.Minimize(objective), constraints)

        warm_start = h.copy()

        try:
            problem.solve(solver=cp.ECOS, warm_start=True)
            h_thickness = h_var.value
            if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                raise Exception("Suboptimal solution; re-try")
            delta = h_thickness - h
            h = np.clip(h + np.minimum(delta, 0.05), 0, 1)
            h = h / np.sum(h) * (n_points / 2)

            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

        except:
            # If the subproblem is infeasible, skip the update and project
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)

    # Final validation
    assert np.allclose(np.sum(h), n_points / 2), "Integral constraint violated"
    assert np.all(0 <= h) and np.all(h <= 1), "Invalid bounds on h"
    final_c5 = evaluate_erdos_solution(best_h)

    return (best_h.tolist(), final_c5, n_points)
