import numpy as np
import time
from numpy.random import default_rng
import cvxpy as cp


def evaluate(h, dx):
    corrs = np.correlate(h, 1 - h, mode="full")[len(h):-len(h)]
    return np.max(corrs * dx)


def project(h, dx):
    # Enforce sum constraint
    mass = np.sum(h)
    desired_mass = h.shape[0] * 0.5
    diff = desired_mass - mass
    h = h + diff / h.shape[0]
    h = np.clip(h, 0, 1)
    return h


def generate_neighbors(h, n_neighbors=20, max_shift=5):
    n_points = len(h)
    neighbors = []
    rng = default_rng()
    indices = np.arange(n_points)
    for _ in range(n_neighbors):
        if rng.random() < 0.6:
            # 1-norm perturbation
            perm = indices[rng.permutation(n_points)]
            amount = 0
            for i in perm[:np.random.randint(1, max_shift + 1)]:
                delta = (rng.random() - rng.random()) * h[i]
                h2 = h.copy()
                h2[i] += delta
                amount += delta
                h2 = np.clip(h2, 0, 1)
            h2 = project(h2, 2.0 / n_points)
            neighbors.append(h2)
        else:
            # Shift pattern
            h2 = h.copy()
            shift = int(2 * (2 * rng.random() - 1) * n_points)
            h2 = np.roll(h2, shift)
            h2 = project(h2, 2.0 / n_points)
            neighbors.append(h2)
    return neighbors


def run(seed=42, budget_s=1100, n_points=45, tabu_size=20):
    np.random.seed(seed)

    t_start = time.time()

    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        h = np.full(n_points, 0.5)
        h = project(h, 2.0 / n_points)

    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate(best_h, 2.0 / n_points)
    tabu_memory = [h.tolist()]

    while time.time() - t_start < budget_s:
        neighbors = generate_neighbors(h)
        improved = False
        candidates = []

        for nh in neighbors:
            # Skip tabu
            if nh.tolist() in tabu_memory:
                continue
            nh = nh.copy()
            nh_c5 = evaluate(nh, 2.0 / n_points)
            candidates.append((nh, nh_c5))

        if not candidates:
            continue

        # Sort candidates by C₅
        candidates.sort(key=lambda x: x[1])
        best_neighbor, best_c5_candidate = candidates[0]
        if best_c5_candidate < best_c5:
            best_c5 = best_c5_candidate
            best_h = best_neighbor
            h = best_neighbor
            improved = True
        else:
            # Random exploration
            if not improved and time.time() - t_start < budget_s:
                h = np.random.rand(n_points)
                h = project(h, 2.0 / n_points)

        if improved:
            tabu_memory.append(h.tolist())
            if len(tabu_memory) > tabu_size:
                tabu_memory.pop(0)

    return best_h, best_c5, n_points
