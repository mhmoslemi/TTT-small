import numpy as np
import time
import cvxpy as cp
from numpy.fft import fft, ifft, fftshift

def run(seed=42, budget_s=1100, n_points=80):
    np.random.seed(seed)
    start_time = time.time()

    dx = 2.0 / n_points
    h = initial_h_values if 'initial_h_values' in globals() else 0.5 * np.ones(n_points)
    h = h.astype(float)
    h /= h.sum() / (n_points / 2)  # normalize to satisfy integral constraint

    def compute_c5(h):
        h1 = h.copy()
        h1[(h1 < 1e-6)] = 1e-6  # avoid numerical issues
        h1[(h1 > 1 - 1e-6)] = 1 - 1e-6
        corr = np.correlate(h1, 1 - h1, mode="full")
        corr = corr[n_points - 1: 2 * n_points - 1]
        return np.max(corr) * dx

    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h / h.sum() * (n_points / 2)
        return h

    def gradient_direction(h, lags, eps=1e-8, dir_eps=1e-6):
        grad = np.zeros_like(h)
        for i in range(n_points):
            for sgn in [1.0, -1.0]:
                hp = h.copy()
                hp[i] += dir_eps * sgn
                hp = project_h(hp)
                c0 = compute_c5(h)
                c1 = compute_c5(hp)
                grad[i] += (c1 - c0) / dir_eps * sgn
        return grad

    best_h = h.copy()
    best_c5 = compute_c5(h)
    tol = 1e-6
    max_iter = 500
    step_size = 1e-2
    early_stop_count = 0
    best_iter = 0

    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        lags = np.arange(-n_points + 1, n_points)
        g = gradient_direction(h, lags)
        anti_g = -g
        dir_norm = np.sqrt((anti_g ** 2).sum())
        if dir_norm < 1e-8:
            early_stop_count += 1
            if early_stop_count >= 10:
                break
            continue
        else:
            early_stop_count = 0
        anti_g = anti_g / dir_norm

        h_update = h + step_size * anti_g
        h_update = project_h(h_update)

        c5_new = compute_c5(h_update)
        if c5_new < best_c5 - 1e-8:
            h = h_update.copy()
            best_c5 = c5_new
            best_h = h.copy()
            step_size *= 1.1
            best_iter = iter
        else:
            step_size *= 0.95

        if iter - best_iter > 100 or step_size < 1e-6:
            # Add convex optimization refinement
            x = cp.Variable(n_points)
            obj = cp.Minimax([cp.Conv(cp.multiply(x, 1 - cp.roll(x, k)), dx) for k in range(n_points)])
            constraints = [
                x >= 0,
                x <= 1,
                cp.sum(x) == n_points / 2
            ]
            prob = cp.Problem(cp.Minimize(obj), constraints)
            result = prob.solve(solver="GUROBI" if "GUROBI" in cp.installed_solvers() else "ECOS")
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h = x.value.copy()
                h = project_h(h)
                best_c5 = compute_c5(h)
                best_h = h.copy()

    return (best_h.tolist(), best_c5, n_points)
