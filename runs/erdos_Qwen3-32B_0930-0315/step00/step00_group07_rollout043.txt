        corr = np.concatenate([
            corr[n_points - 1:],  # positive lags
            corr[:n_points]       # negative lags
        ])
        worst_k = np.argmax(corr)

        # Compute the gradient for worst_k
        k = worst_k - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j1 = (i + k) % n_points
            j2 = (i - k) % n_points
            grad[i] = dx * (1 - h[j1] - h[j2])

        # Setup and solve the LP
        delta = cp.Variable(n_points)
        obj = cp.Minimize(grad @ delta)
        constraints = [
            delta >= -h,
            delta <= 1 - h,
            cp.sum(delta) == 0,
        ]
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS)

        # Update h
        h += delta.value
        h = np.clip(h, 0, 1)

        # Project back to exact mean constraint
        h_sum = h.sum()
        err = h_sum - n_points / 2
        if abs(err) > 1e-6:
            h -= err / n_points

        # Evaluate and update
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:

            best_h = h.copy()
