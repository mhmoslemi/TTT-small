import numpy as np
import scipy.optimize as opt
from time import perf_counter

dx = 2.0 / len(initial_h_values)
sum_h = len(initial_h_values) / 2.0

def compute_c5(h):
    h = h.reshape(-1)
    corr = np.correlate(h, 1 - h, mode="full")
    corr = corr * dx
    return np.max(corr)

def objective_function(h):
    return compute_c5(h)

def constraint_sum_h(x):
    return x.sum() - sum_h

def bounds_h(n):
    return [(0.0, 1.0) for _ in range(n)]

def project_to_feasible_region(x):
    x = np.clip(x, 0, 1)
    x = x - (x.sum() - sum_h) / len(x)
    return np.clip(x, 0, 1)

def run(seed=42, budget_s=1100):
    h = np.array(initial_h_values)
    start_time = perf_counter()
    h = h.copy()
    n_points = len(h)

    def callback(x):
        nonlocal start_time, h
        h[:] = x
        elapsed = perf_counter() - start_time
        if elapsed >= budget_s:
            return True
        return False

    result = opt.minimize(
        objective_function,
        h,
        method='L-BFGS-B',
        bounds=bounds_h(n_points),
        callback=callback,
        options={'ftol': 1e-9, 'maxiter': 500, 'maxcor': 50, 'eps': 1e-4}
    )

    h = project_to_feasible_region(result.x)
    c5_bound = compute_c5(h)

    return h.tolist(), c5_bound, n_points
