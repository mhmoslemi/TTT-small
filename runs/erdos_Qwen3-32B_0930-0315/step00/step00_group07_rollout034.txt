import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_lag(h, dx):
    """Compute overlap integrals for all lags and return the max."""
    h = np.r_[h, np.zeros_like(h)]  # Extend for mode='full'
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr[1:-1] * dx)

def feasibility_check(h, n_points):
    """Ensure values are in [0, 1] and the integral is 1."""
    h = h.clip(0, 1)
    h_sum = h.sum()
    h *= n_points / (2 * h_sum)  # Normalize to sum to n_points / 2
    return h

def objective(x, n_points, dx):
    """Objective function for optimization."""
    h = feasibility_check(x, n_points)
    return evaluate_lag(h, dx)

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    initial = np.copy(initial_h_values) if "initial_h_values" in globals() else np.full(80, 0.5)
    n_points = len(initial)
    dx = 2.0 / n_points

    def callback(x):
        nonlocal best_h, best_c5, start_time, best_iter
        h = feasibility_check(x, n_points)
        if h.shape[0] == n_points:
            c5 = evaluate_lag(h, dx)
            if c5 < best_c5:
                best_h[:] = h
                best_c5 = c5
                best_iter = 0
            else:
                best_iter += 1
            print(f"C₅: {c5:.6f} at step {step}, improvement steps: {best_iter}")
        return best_iter > 50 or time() - start_time > budget_s - 10

    best_h = np.copy(initial)
    best_c5 = evaluate_lag(best_h, dx)
    best_iter = 0
    step = 0
    start_time = time()

    while time() - start_time < budget_s:
        res = minimize(
            objective,
            x0=best_h,
            args=(n_points, dx),
            method='BFGS',
            jac=False,
            options={'maxiter': 100},
            callback=callback
        )
        best_h = feasibility_check(res.x, n_points)
        best_c5 = evaluate_lag(best_h, dx)
        print(f"Completed step {step}, current best C₅: {best_c5:.6f}")
        step += 1

    return best_h, best_c5, n_points
