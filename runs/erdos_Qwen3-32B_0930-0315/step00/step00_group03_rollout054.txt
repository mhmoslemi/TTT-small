import numpy as np
import time
import cvxpy as cp


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points
    total_mass = int(n_points / 2)

    h = np.copy(h_init)
    best_h = np.copy(h)
    best_c5 = float('inf')

    # Projection constraint: maintain sum(h) == total_mass, and 0 <= h <= 1
    def project_h(h_candidate, total_mass):
        h_var = cp.Variable(h_candidate.shape)
        objective = cp.Minimize(cp.sum_squares(h_var - h_candidate))
        constraints = [
            cp.sum(h_var) == total_mass,
            h_var >= 0,
            h_var <= 1
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        return h_var.value

    # Gradient step + projection
    num_iter = 0
    max_iter = 1000
    no_progress_counter = 0
    best_lr = 0.01
    while time.time() - start_time < budget_s and num_iter < max_iter:
        # Compute overlap
        h1 = np.concatenate([h, h])
        ones_minus_h = 1.0 - h1
        corr = np.correlate(h1, ones_minus_h, mode='full') * dx

        # Take only k in [0, n)
        k_values = np.arange(n_points)
        c5 = np.max(corr[k_values])

        if c5 < best_c5:
            best_c5 = c5
            best_h = np.copy(h)
            no_progress_counter = 0
        else:
            no_progress_counter += 1

        if no_progress_counter > 100:
            break  # Exit if no improvement

        # Find worst shift k
        worst_k = k_values[np.argmax(corr[k_values])]

        # Compute gradient of integrand for worst_k
        grad = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            if (i + worst_k) < n_points:
                grad[i] += (1 - h[i + worst_k])
            if (i - worst_k) >= 0:
                grad[i] -= h[i - worst_k]

        # Gradient descent step
        learning_rate = best_lr / (1 + num_iter / 1000)
        h_candidate = h - learning_rate * grad
        h = project_h(h_candidate, total_mass)

        num_iter += 1

    # Final validation and evaluation of best_h
    h = best_h
    corr = np.correlate(np.concatenate([h, h]), 1 - np.concatenate([h, h]), mode='full') * dx
    final_c5 = np.max(corr[:n_points])

    return (h, final_c5, n_points)
