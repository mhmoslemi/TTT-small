import numpy as np
import time
import cvxpy as cp
from numpy import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Use a higher resolution for better solution search
    n_points = 256
    dx = 2.0 / n_points

    # Optionally use the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Fallback to an occupancy-1 random initialization
        h = np.random.randn(n_points)
        h = (h - h.min()) / (h.max() - h.min())
        h = np.clip(h, 0, 1)

    # Normalize h to have total area = 1
    h *= n_points / 2.0 / h.sum()

    start_time = time.time()
    best_h = h.copy()
    best_C5 = float('inf')

    # Optimization parameters
    lr = 0.1
    decay_rate = 0.999
    best_h_update_count = 0
    perturb_interval = 1000
    perturb_strength = 0.01

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Compute the overlap integrals
        h_array = np.concatenate([h / np.sqrt(dx), np.zeros(n_points // 2)])
        corrs = correlate(h_array, h_array, mode='full')
        corrs = corrs[n_points - 1 : 2 * n_points - 1]

        # Compute C₅ as max overlap (1 - AC(k))
        # C5 = max(1 - AC(k)) = max(overlap) = max(overlap integral)
        current_C5 = np.max(np.concatenate([
            corrs[: n_points], 
            1 - h.sum() * dx - corrs]))

        # Update best solution
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            best_h_update_count = 0
        else:
            best_h_update_count += 1

        # Early stopping if no improvement for a while
        if best_h_update_count > 2000:
            h = best_h.copy()
            best_h_update_count = 0

        # Perturb every few iterations to escape local minima
        if best_h_update_count % perturb_interval == 0:
            noise = np.random.randn(n_points)
            noise -= noise.min()
            noise /= noise.max()
            h += lr * perturb_strength * (2 * noise - 1)
            h = np.clip(h, 0, 1)
            h *= n_points / 2.0 / h.sum()

        # Compute the gradient for the worst-case shift
        max_index = np.argmax(np.concatenate([
            corrs[: n_points], 
            1 - h.sum() * dx - corrs]))
        k = max_index if max_index < n_points else (2 * max_index - 2 * n_points)
        k = np.clip(k, 0, n_points - 1)

        left_h = h[k:] / np.sqrt(dx)
        right_h = h[: n_points - k] / np.sqrt(dx)
        gradient = np.zeros_like(h)
        gradient[k:] += right_h
        gradient[: n_points - k] += left_h
        gradient *= -1.0

        # Gradient update
        h += lr * gradient
        h = np.clip(h, 0, 1)
        h *= n_points / 2.0 / h.sum()

        # Learning rate decay
        lr = max(0.001, lr * decay_rate)

    return best_h, best_C5 * dx, n_points
