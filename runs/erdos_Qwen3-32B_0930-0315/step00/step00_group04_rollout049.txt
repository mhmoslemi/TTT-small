import numpy as np
import math
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)

    # Load initial guess if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 42
        h = np.zeros(n_points)
        h[:n_points // 2] = 1.0

    # Ensure constraint sum(h) == n_points / 2
    n_points = len(h)
    dx = 2.0 / n_points
    def project(h):
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Normalize to maintain constraint sum(h) == n_points / 2
        total = np.sum(h)
        if total > 0:  # Avoid division by zero
            h *= (n_points / 2) / total
        else:
            h[h > 0] = 1
        return h

    def evaluate_c5(h):
        # Compute overlap function
        for k in range(1, n_points):  # Only consider k in 1 to n_points-1
            h_shifted = np.concatenate([h[k:], np.zeros(k)])
            overlap = np.sum(h * (1 - h_shifted)) * dx
        h_shifted = np.concatenate([h[1:], np.zeros(1)])
        overlap_1 = np.sum(h * (1 - h_shifted)) * dx

        for k in range(2, n_points):  # Only consider k in 1 to n_points-1
            h_shifted = np.concatenate([h[k:], np.zeros(k)])
            overlap_k = np.sum(h * (1 - h_shifted)) * dx
            if overlap_k > overlap_1:
                overlap_1 = overlap_k
        return overlap_1

    # Start time
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)

    learning_rate = 0.01
    lr_decay = 0.95
    update_counts = 0

    while time.time() - start_time < budget_s - 0.1:
        h = project(h)

        # Compute gradients
        max_overlap = -np.inf
        k_max = 0
        for k in range(1, n_points):
            h_shifted = np.concatenate([h[k:], np.zeros(k)])
            overlap = np.sum(h * (1 - h_shifted)) * dx
            if overlap > max_overlap:
                max_overlap = overlap
                k_max = k
        h_shifted = np.concatenate([h[k_max:], np.zeros(k_max)])

        # Gradient is (1 - h_shifted) for this k_max
        grad = (1 - h_shifted)

        # Update h
        h -= learning_rate * grad
        h = project(h)

        # Re-evaluate best_c5
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        learning_rate *= lr_decay
        update_counts += 1

    return (best_h.tolist(), best_c5, n_points)
