import numpy as np
import time
import cvxpy as cp
import sys

def evaluate_erdos_solution(h_values, n_points):
    h = np.array(h_values)
    dx = 2.0 / n_points
    c = np.correlate(h, 1 - h, mode="full")
    c5_bound = np.max(c) * dx
    return float(c5_bound)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 160
        h = np.clip(0.5 + 0.3 * np.sin(np.pi * np.linspace(0, 2, n_points)), 0, 1)
        h *= 0.5 / np.mean(h)

    alpha = 0.01
    max_iter = 10000
    patience = 100
    best_h = h.copy()
    best_c5 = float('inf')
    last_improvement = 0

    for it in range(max_iter):
        if time.time() - start_time >= budget_s:
            break

        c = np.correlate(h, 1 - h, mode="full")
        c5_idx = np.argmax(c)
        k_max = c5_idx - (n_points - 1)
        if k_max < 0:
            k_max += 2 * n_points

        grad = np.zeros_like(h)
        for j in range(n_points):
            idx_kp = (j + k_max) % n_points
            idx_km = (j - k_max) % n_points
            grad[j] = (1 - h[idx_kp]) - h[idx_km]

        h -= alpha * grad
        # Projection onto [0,1]
        h = np.clip(h, 0, 1)
        # Projection onto sum constraint
        current_sum = h.sum()
        h *= (n_points / 2) / current_sum

        current_c5 = evaluate_erdos_solution(h, n_points)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            last_improvement = it
        elif it - last_improvement > patience:
            # Break if no improvement found for patience iterations
            break

    return best_h, best_c5, n_points
