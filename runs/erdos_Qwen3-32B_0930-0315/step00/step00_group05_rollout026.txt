import numpy as np
import time
from scipy.signal import correlate

start_time = time.time()

def run(seed=42, budget_s=1100, **kwargs):
    global start_time

    np.random.seed(seed)
    dx = 2.0 / 70
    target_sum = 70 / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(70)
        h = np.clip(h, 0, 1)
        h *= target_sum / np.sum(h)

    best_h = h.copy()
    best_c5 = float('inf')
    n_points = 70
    it = 0
    alpha = 0.1
    patience_alpha = 0

    while time.time() - start_time < budget_s:
        it += 1

        # Compute convolve of h and 1 - h for all lags
        ones_minus_h = 1 - h
        corr = correlate(h, ones_minus_h, mode='full')
        c5 = np.max(corr * dx)

        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h[:] = h.copy()
            patience_alpha = 0
        else:
            patience_alpha += 1
            if patience_alpha >= 50:
                alpha *= 0.5
                patience_alpha = 0

        # Find the lag with max correlation
        k_max = np.argmax(corr)

        # Compute gradient
        grad = np.zeros_like(h)
        valid_indices_forward = np.where(k_max <= np.arange(70, dtype=int))
        grad[valid_indices_forward] += ones_minus_h[valid_indices_forward[0] - k_max]

        valid_indices_backward = np.where(k_max <= 70 - np.arange(70, dtype=int))
        grad[valid_indices_backward] -= h[valid_indices_backward[0]]

        # Update h
        h -= alpha * grad

        # Project to [0, 1]
        h = np.clip(h, 0, 1)

        # Keep average value of h fixed
        h *= target_sum / np.sum(h)

        # Add small perturbation every 50 iterations to escape local minima
        if it % 50 == 0:
            h += np.random.normal(0, 0.01, h.shape)
            h = np.clip(h, 0, 1)
            h *= target_sum / np.sum(h)

    return best_h, best_c5, n_points
