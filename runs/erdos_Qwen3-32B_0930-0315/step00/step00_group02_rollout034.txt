import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def build_initial_h(n):
    center = n // 2
    width = n // 2 - 10
    h = np.zeros(n)
    h[center - width : center + width] = 1.0
    h = np.clip(h * n / h.sum(), 0, 1) if h.sum() != 0 else 0
    return h

def project_h(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    h_sum = h_clipped.sum()
    target = n_points / 2
    scale = target / h_sum if h_sum != 0 else 0
    h_scaled = h_clipped * scale
    diff = target - h_scaled.sum()
    idx = np.argpartition(h_scaled, -int(np.ceil(abs(diff))))
    if diff > 0:
        h_scaled[idx[-int(np.ceil(abs(diff)))]] += diff / int(np.ceil(abs(diff)))
    elif diff < 0:
        h_scaled[idx[:int(np.ceil(abs(diff)))]] += diff / int(np.ceil(abs(diff)))
    return np.clip(h_scaled, 0, 1)

def compute_overlap_integrals(h, dx):
    h = np.array(h)
    h1 = h[None, :]
    h2 = h[:, None]
    h_stack = np.flip(h2, axis=0)
    overlaps = (h1 * (1 - h_stack[1:-1])).sum(axis=1) * dx
    return overlaps

def gradient_descent(h, n_points, dx):
    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_overlap_integrals(h, dx).max()
    max_iter = 500
    for _ in range(max_iter):
        if time.time() - start_time > 1100:
            break
        # Compute gradient estimate via finite differences
        e = 1e-4
        f0 = compute_overlap_integrals(h, dx).max()
        grad = np.zeros(n_points)
        for i in range(n_points):
            h[i] += e
            fi = compute_overlap_integrals(h, dx).max()
            grad[i] = (fi - f0) / e
            h[i] -= e  # restore

        # Gradient step with backtracking line search
        learning_rate = 0.1
        h_candidate = h - learning_rate * grad
        h_candidate = project_h(h_candidate, n_points)

        if compute_overlap_integrals(h_candidate, dx).max() < f0:
            h[:] = h_candidate[:]
            f0 = compute_overlap_integrals(h, dx).max()
            if compute_overlap_integrals(h, dx).max() < compute_overlap_integrals(best_h, dx).max():
                best_h[:] = h[:]
                best_c5 = compute_overlap_integrals(h, dx).max()

        # Add noise to avoid local minima
        if np.random.rand() < 0.1:
            noise = np.random.normal(0, 0.1, n_points)
            h += noise
            h = project_h(h, n_points)

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 160
    h = build_initial_h(n_points)
    dx = 2.0 / n_points

    start_time = time.time()
    h, c5_bound = gradient_descent(h, n_points, dx)
    elapsed = time.time() - start_time

    return (h.tolist(), c5_bound, n_points)
