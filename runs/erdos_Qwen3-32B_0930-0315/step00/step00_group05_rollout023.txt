import numpy as np
import time
import scipy.signal

dx = 2.0 / 70  # Fixed for initial_h_values

def evaluate_c5(h):
    h = np.asarray(h)
    n_points = len(h)
    result = scipy.signal.correlate(h, 1 - h, mode='full')
    middle = result[n_points - 1 : 2 * n_points - 1]
    c5 = np.max(middle * dx)
    return c5, result[n_points - 1 + np.argmax(middle)] * dx

def project_feasible(h, n_points):
    h_proj = np.clip(h, 0, 1)
    total = np.sum(h_proj)
    scale = n_points / 2 / total
    h_proj *= scale
    return h_proj

    idxs = np.arange(n_points)

    for i in range(n_points):
        j = (i + shift) % n_points

    return grad
