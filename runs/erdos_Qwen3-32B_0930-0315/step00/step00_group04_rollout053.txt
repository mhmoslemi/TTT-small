import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 42  # Initial resolution
    dx = 2.0 / n_points

    # Use the provided initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Start with uniform distribution
        h = np.ones(n_points) / n_points * n_points / 2  # Feasible densifier

    h = h / np.sum(h) * n_points / 2  # Enforce exact normalization
    h = np.clip(h, 1e-6, 1 - 1e-6)  # Enforce strict interiority

    best_h = h.copy()
    best_c5 = float('inf')
    
    max_iterations = 1000
    mu = 1e4  # Initial barrier parameter
    decay_factor = 0.1  # Decrease factor for mu
    decay_period = 50  # Every N iterations
    learning_rate_decay = 0.5
    learning_rate = 1.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    for iteration in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if iteration % 10 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                print(f"Iter {iteration}: c5={c5:.10f} (new best)")

        # Create CVXPY problem
        h_var = cp.Variable(n_points)
        h_ref = h.copy()

        # Define the correlation constraint at current peak
        corr = np.correlate(h_ref, 1 - h_ref, mode='full') * dx
        k_max = np.argmax(corr)

        # Valid object indices for correlation at k_max
        from_indices = max(k_max, 0)
        to_indices = min(k_max + n_points, n_points)

        corr_term = cp.sum( h_var[from_indices:n_points] * (1 - h_var[:to_indices]) ) * dx

        # Barrier terms
        bar1 = cp.sum(cp.log(h_var))
        bar2 = cp.sum(cp.log(1 - h_var))
        bar3 = cp.log(cp.sum(h_var) * dx - 1 + 1e-10)

        objective = corr_term + mu * (bar1 + bar2 + bar3)

        constraints = [
            0 < h_var, h_var < 1
        ]

        problem = cp.Problem(cp.Minimize(objective), constraints)
        problem.solve(solver=cp.ECOS, verbose=False)

        if problem.status != 'optimal':
            h = h_ref + 0.1 * np.random.randn(n_points)
            h = np.clip(h, 1e-6, 1 - 1e-6)
            h = h / np.sum(h) * n_points / 2
            continue

        h = h_var.value
        h = np.clip(h, 1e-6, 1 - 1e-6)
        h = h / np.sum(h) * n_points / 2

        if iteration % decay_period == 0:
            mu *= decay_factor
            learning_rate *= learning_rate_decay

    # Final check with evaluator
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    best_c5 = compute_c5(best_h)
    return (best_h.tolist(), best_c5, n_points)
