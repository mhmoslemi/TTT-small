import numpy as np
import cvxpy as cp
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    start = time()
    np.random.seed(seed)

    # Initial setup
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    print(f"Initial C5: {best_c5:.6f}")

    # Reserve safety buffer
    budget_s *= 0.98

    def compute_overlaps(h):
        return np.correlate(h, 1 - h, mode="full") * dx

    while time() - start < budget_s:
        overlaps = compute_overlaps(h)
        k_worst = int(np.argmax(overlaps[n_points:]))

        # Build gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            grad[i] = 1.0 - h[(i + k_worst) % n_points] - h[(i - k_worst) % n_points]
        grad *= dx

        # Normalize to step within sum=0 constraint
        grad -= np.mean(grad)

        # LP variables
        delta = cp.Variable(n_points)
        constraints = [
            h + delta >= 0,
            h + delta <= 1,
            cp.sum(delta) == 0
        ]
        cost = cp.Maximize(cp.sum(grad * delta))
        problem = cp.Problem(problem=cost, constraints=constraints)
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
        except:
            problem.solve(solver=cp.GLPK, verbose=False)

        if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            continue

        delta_sol = delta.value
        h += delta_sol
        h = np.clip(h, 0, 1)  # Enforce feasibility explicitly

        # Enforce exact mean constraint
        current_sum = np.sum(h)
        target_sum = n_points / 2.0
        correction = (target_sum - current_sum) / n_points
        h += correction
        h = np.clip(h, 0, 1)

        # Evaluate
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            print(f"Improved C5: {best_c5:.6f}")

    return (best_h, best_c5, n_points)
