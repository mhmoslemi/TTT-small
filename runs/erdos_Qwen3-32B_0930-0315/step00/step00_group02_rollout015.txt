        raise ValueError("Invalid h shape")

    if not np.all((h >= 0) & (h <= 1)):
        return float('inf')

        return float('inf')

    h_circular = np.concatenate([h, h])
    overlaps = correlate(h_circular, 1.0 - h_circular, mode='valid') * dx
    best_C5 = np.round(np.max(overlaps), precision)

    return best_C5

def box_projection(h, n_points, s=2):
    """Project back into feasible box region with normalization."""
    h = np.clip(h, 0, 1)
    avg = np.sum(h) / n_points
    h = h + (s / n_points) * (0.5 - avg)
    h = np.clip(h, 0, 1)
    return h
