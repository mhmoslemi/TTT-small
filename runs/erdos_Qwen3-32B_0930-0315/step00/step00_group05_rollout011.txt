import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):  # stub, the real one is pre-imported
    dx = 2.0 / len(h)
    correl = correlate(h, 1 - h, mode='full')
    indices = np.arange(len(correl))
    mask = (indices >= len(h)) & (indices < 2 * len(h) - len(h) + 1)
    return np.max(correl[mask] * dx)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    n_points = 70
    dx = 2.0 / n_points

    if 'initial_h_values' in kwargs:
        h = np.copy(kwargs['initial_h_values'].astype(np.float64))
    else:
        h = np.zeros(n_points)
        h[:n_points // 2] = 1.0

    h = np.clip(h, 0.0, 1.0)
    h_sum = np.sum(h)
    missing = n_points / 2.0 - h_sum
    h += missing / n_points
    h = np.clip(h, 0.0, 1.0)

    start_time = time.time()
    alpha = 0.1
    decay = 0.8
    perturb_every = 50
    improvement_tol = 1e-6
    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h)
    step = 0

    while time.time() - start_time < budget_s:
        step += 1

        correl = correlate(h, 1 - h, mode="full")
        indices = np.arange(len(correl))
        valid = (indices >= n_points) & (indices < len(h) * 2 - n_points)
        correl_valid = correl[valid]
        k_max = np.argmax(correl_valid) + n_points

        grad = np.zeros_like(h)
        # Forward contributions
        grad[:n_points - k_max] -= (1 - h[k_max: n_points])
        # Backward contributions
        grad[k_max:] += h[:n_points - k_max]
        grad *= dx

        h -= alpha * grad
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        missing = n_points / 2.0 - h_sum
        h += missing / n_points
        h = np.clip(h, 0.0, 1.0)

        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5 - 1e-7:
            best_c5 = current_c5
            best_h = np.copy(h)
            alpha = min(alpha * decay, 0.01)
        elif step % 200 == 0:
            alpha = min(alpha * decay, 0.01)

        if step % perturb_every == 0:
            h += 0.01 * np.random.randn(*h.shape)
            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            missing = n_points / 2.0 - h_sum
            h += missing / n_points
            h = np.clip(h, 0.0, 1.0)

        if np.abs(current_c5 - best_c5) < improvement_tol:
            alpha = max(alpha * decay, 1e-3)

    return (best_h.tolist(), best_c5, n_points)
