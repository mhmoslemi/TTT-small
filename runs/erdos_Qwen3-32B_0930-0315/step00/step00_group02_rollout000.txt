```python
import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_C5(h, dx):
    ones_h = 1 - h
    corr = correlate(h, ones_h, mode="full")
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_constraints(h, n_points):
    h_clamped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clamped)
    scale_factor = n_points / 2 / current_sum
    h_scaled = h_clamped * scale_factor
    return h_scaled

def gradient_C5(h, dx):
    n_points = len(h)
    ones_h = 1 - h
    corr = correlate(ones_h, h, mode="full")
    max_index = np.argmax(corr)
    
    k = max_index - n_points + 1
    if k < 0 or k >= n_points:
        return np.zeros(n_points)

    grad = np.zeros(n_points)
    grad = h * ones_h[(k - 1 + np.arange(n_points)) % n_points]
    return -grad * dx

def gradient_descent_phase1(h, dx, max_iter=200, learning_rate=0.03):
    h = project_to_constraints(h, len(h))
    for _ in range(max_iter):
        grad = gradient_C5(h, dx)
        h -= learning_rate * grad
        h = project_to_constraints(h, len(h))
    return h

def cvxpy_phase2(h, dx):
    n = len(h)
    h_var = cp.Variable(n)

    objective = cp.Minimize(
        cp.Max(
            [cp.sum(h_var * (1 - h_var[(k - 1 + np.arange(n)) % n])) * dx for k in range(1, n)]
        )
    )

    constraints = [
        0 <= h_var,
        h_var <= 1,
        cp.sum(h_var) == n / 2
    ]

    prob = cp.Problem(objective, constraints)
    prob.solve(max_iter=1000, verbose=False)

    return np.clip(h_var.value, 0, 1)

def optimize_erdos(h_initial, dx, budget_s=1100):
    start_time = time.time()
    h = h_initial.copy()

    n_points = len(h_initial)
    h = gradient_descent_phase1(h, dx)

    h = cvxpy_phase2(h, dx)
    h = project_to_constraints(h, n_points)
    elapsed = time.time() - start_time
    C5 = evaluate_C5(h, dx)

    return h, C5, n_points

def run(budget_s=1100):
    n_points = 160
    dx = 2.0 / n_points
    h_mid = (n_points // 2)
    plateau_width = 80
    h_initial = np.zeros(n_points)
    h_initial[h_mid - plateau_width//2 : h_mid + plateau_width//2] = 0.5

    h_feasible = project_to_constraints(h_initial, n_points)
    h_opt, c5_bound, _ = optimize_erdos(h_feasible, dx, budget_s)

    return h_opt, c5_bound, n_points
```
