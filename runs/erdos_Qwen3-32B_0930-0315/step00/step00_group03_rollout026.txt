import numpy as np
import time
from functools import partial
import scipy.optimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    time_start = time.time()
    
    # Use provided initial_h_values
    initial_h = np.copy(initial_h_values)
    n_points = len(initial_h)

    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def compute_c5(h_vec):
        h_vec = np.asarray(h_vec)
        h = h_vec / h_vec.sum() * target_sum
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def repair_sum(h_vec, eps=1e-6):
        h_sum = h_vec.sum()
        if abs(h_sum - target_sum) < eps:
            return h_vec
        h_vec = h_vec.copy()
        h_vec += (target_sum - h_sum) / n_points
        h_vec = np.clip(h_vec, 0, 1)
        return h_vec

    def tournament_selection(population, fitness, size=5):
        contestants = np.random.choice(len(population), size)
        winner_idx = np.argmin([fitness[i] for i in contestants])
        return population[contestants[winner_idx]], fitness[contestants[winner_idx]]

    def crossover(parent1, parent2):
        point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        child1 = repair_sum(child1)
        child2 = repair_sum(child2)
        return child1, child2

    def mutate(h_vec, mutation_rate=0.15, noise_scale=0.1):
        if np.random.rand() < mutation_rate:
            mask = np.random.rand(n_points) < 0.05
            h_vec = h_vec.copy()
            h_vec[mask] += np.random.normal(0, noise_scale, n_points)[mask]
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = repair_sum(h_vec)
        return h_vec

    population_size = 50
    population = [initial_h.copy() for _ in range(population_size // 2)]
    for _ in range(population_size // 2):
        h = np.random.rand(n_points)
        h = repair_sum(h)
        population.append(h)

    best_h = None
    best_c5 = float('inf')

    while time.time() - time_start < budget_s:
        fitness = [compute_c5(h) for h in population]
        next_gen = []

        for _ in range(population_size // 2):
            p1, _ = tournament_selection(population, fitness)
            p2, _ = tournament_selection(population, fitness)
            c1, c2 = crossover(p1, p2)
            c1 = mutate(c1)
            c2 = mutate(c2)
            next_gen.extend([c1, c2])

        population = sorted(population + next_gen, key=lambda x: compute_c5(x))[:population_size]
        
        current_c5 = min(fitness)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[0].copy()
        
        # Early stopping if low bound is met
        if best_c5 < 0.380876:
            break

    return best_h, best_c5, n_points
