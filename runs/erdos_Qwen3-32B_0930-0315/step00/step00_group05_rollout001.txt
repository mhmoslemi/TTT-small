import time
import numpy as np
import cvxpy as cp
from evaluate_erdos_solution import evaluate_erdos_solution


def run(seed=42, budget_s=1100):
    start_time = time.perf_counter()

    # Hyperparameters
    n_points = 70
    max_iter = 1000
    display_interval = 100
    dx = 2.0 / n_points

    # Initialize
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Uniform initialization satisfying mass constraint
        h = np.ones(n_points, dtype=np.float64) / 2

    def project_h(h):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if s == 0:
            h[:] = 0.5
        else:
            h *= (n_points / 2) / s
        return h

    def compute_overlap(h):
        h_inactive = 1.0 - h
        correlations = np.correlate(h, h_inactive, mode="full")
        c5 = correlations * dx
        c5_bound = np.max(c5)
        return c5_bound, c5

    best_h = h.copy()
    best_c5, _ = compute_overlap(best_h)

    for i in range(max_iter):
        elapsed_time = time.perf_counter() - start_time
        if elapsed_time >= budget_s:
            break

        # Solve convex relaxation
        h_cpx = cp.Variable(n_points, nonneg=True)
        overlap = cp.Parameter(n_points + n_points - 1)
        objective = cp.Minimize(cp.norm(cp.multiply(overlap, dx), p=np.inf))
        constraints = [h_cpx >= 0, h_cpx <= 1, cp.sum(h_cpx) == n_points / 2.0]
        prob = cp.Problem(objective, constraints)
        
        sampled_overlap, _ = compute_overlap(h)
        overlap.value = np.correlate(h, 1 - h, mode='full')
        
        try:
            result = prob.solve(solver='ECOS', verbose=False)
            if h_cpx.value is not None:
                h = project_h(h_cpx.value)
                current_c5, _ = compute_overlap(h)
                
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
            else:
                pass  # fallback to gradient descent step
        except:
            pass  # solver failed, continue with gradient descent

        if i % display_interval == 0 and i > 0:
            print(f"Iter {i}: C5 = {best_c5:.7f}")

    result_c5, _ = compute_overlap(best_h)
    return best_h, result_c5, n_points
