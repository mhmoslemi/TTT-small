import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = np.copy(initial_h_values)
    best_c5, _ = evaluate_erdos_solution(best_h)
    n_points = len(best_h)
    dx = 2.0 / n_points
    max_iter = 0
    h = np.copy(initial_h_values)
    learning_rate = 0.01
    lr_decay = 0.95
    lr_decay_iter = 100
    best_c5 = 1.0
    best_h = np.copy(h)
    iter_start = 0
    
    while time.time() - start_time < budget_s - 5:
        h = np.copy(best_h)
        iter_start = 0
        for iter in range(iter_start, 500):
            if time.time() - start_time >= budget_s - 5:
                break
            corr = np.correlate(h, 1 - h, mode='full') * dx
            k_max = np.argmax(corr)
            shift = k_max - n_points + 1
            if shift >= 0:
                h_shifted = np.roll(h, shift)
                h_shifted[shift:] -= h[:n_points - shift]
            else:
                h_shifted = np.roll(h, shift)
                h_shifted[:n_points + shift] -= h[-shift:]

            grad = 1 - h_shifted
            h_new = h - learning_rate * grad

            # Project onto feasible region [0, 1]
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)

            # Evaluate candidate update
            new_c5, _ = evaluate_erdos_solution(h_new)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = np.copy(h_new)
                print(f"Improved bound: {best_c5:.6f} at iter {iter}")
            else:
                learning_rate *= 0.9  # reduce learning rate on plateau

            if iter % lr_decay_iter == 0 and iter > 0:
                learning_rate *= lr_decay
                if learning_rate < 1e-4:
                    break

        print(f"Time: {time.time()-start_time:.1f} s | C₅ bound: {best_c5:.6f}")
        if best_c5 < 0.38085:
            break

    return best_h, best_c5, n_points
