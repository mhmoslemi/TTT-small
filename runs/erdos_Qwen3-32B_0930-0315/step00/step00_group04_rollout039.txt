import time
import numpy as np
from scipy import signal
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    n_points = 42
    dx = 2.0 / n_points
    h = initial_h_values
    assert h.shape[0] == n_points
    assert np.allclose(np.sum(h) * dx, 1.0)
    
    best_h = np.copy(h)
    best_c5 = evaluate_correlation(h, dx)
    
    max_iter = 200
    mu = 1e4
    num_iterations = 0
    
    while time.time() - t0 < budget_s and num_iterations < max_iter:
        h_safe = np.clip(h, 1e-6, 1 - 1e-6)
        
        # CVXPY formulation
        H = cp.Variable(n_points, pos=True)
        obj = mu * cp.sum(cp.log(H[cp.newaxis, :] * (1 - H[:, cp.newaxis])))
        
        C = signal.correlate(H, 1 - H, mode="full") * dx
        obj += cp.max(C)
        
        constraints = [
            cp.sum(H) == n_points / 2,
            H <= 1,
            H >= 0
        ]
        
        problem = cp.Problem(cp.Minimize(obj), constraints)
        result = problem.solve(solver=cp.CVXOPT, verbose=False)
        
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h = H.value
            c5 = evaluate_correlation(h, dx)
            if c5 < best_c5:
                best_h = np.copy(h)
                best_c5 = c5
                print(f"Iter: {num_iterations}, C5: {best_c5:.7f}")
        
        mu *= 0.99
        num_iterations += 1
    
    h = np.clip(best_h, 0, 1)
    h = h / np.sum(h) * n_points / 2  # Normalization to enforce constraint
    return h, best_c5, n_points

def evaluate_correlation(h, dx):
    corr = signal.correlate(h, 1 - h, mode="full")
    corr *= dx
    return np.max(corr)
