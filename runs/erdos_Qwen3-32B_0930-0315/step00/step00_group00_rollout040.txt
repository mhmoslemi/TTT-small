import numpy as np
import time
import random

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    h = h * dx
    from numpy import correlate
    correlations = correlate(h, 1 - h, mode="full")
    return np.max(correlations) * dx

def feasible_project(h, n_points, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    scale_factor = target_sum / current_sum
    h = h * scale_factor
    h = np.clip(h, 0, 1)
    return h

def tournament_select(population, scores, tournament_size):
    indices = np.random.choice(len(population), tournament_size, replace=False)
    winner_idx = np.argmin([scores[i] for i in indices])
    return population[indices[winner_idx]]

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    target_sum = n_points / 2.0

    population_size = 75
    tournament_size = max(2, int(0.1 * population_size))
    crossover_prob = 0.7
    mutation_prob = 0.05
    sa_interval = 50
    cooling_rate = 0.95
    initial_temperature = 100
    generations_without_improvement = 0
    max_stall_generations = 500
    time_limit = budget_s

    population = [feasible_project(np.random.rand(n_points), n_points, target_sum) for _ in range(population_size)]
    if 'initial_h_values' in globals():
        population[0] = initial_h

    best_idx = np.argmin([evaluate_c5(h, n_points) for h in population])
    best_h = population[best_idx]
    best_c5 = evaluate_c5(best_h, n_points)

    start_time = time.time()
    gen = 0
    temperature = initial_temperature
    while time.time() - start_time < time_limit:
        gen += 1

        # Evaluate C₅ for all in population
        scores = [evaluate_c5(h, n_points) for h in population]

        # Keep track of current best
        current_best_idx = np.argmin(scores)
        current_best_solution = population[current_best_idx]
        current_best_score = scores[current_best_idx]
        if current_best_score < best_c5:
            best_h = current_best_solution
            best_c5 = current_best_score
            generations_without_improvement = 0
        else:
            generations_without_improvement += 1

        # Stop if no improvement for too long
        if generations_without_improvement >= max_stall_generations:
            break

        # Stop if temperature too low
        if temperature < 1e-6:
            break

        # Genetic Algorithm Step
        next_population = []
        for _ in range(population_size):
            parent1 = tournament_select(population, scores, tournament_size)
            parent2 = tournament_select(population, scores, tournament_size)

            # Crossover
            if random.random() < crossover_prob:
                crossover_point = random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent1.copy()

            # Mutation
            if random.random() < mutation_prob:
                idx = random.randint(0, n_points - 1)
                mutation_amount = random.uniform(-0.1, 0.1)
                child[idx] = np.clip(child[idx] + mutation_amount, 0, 1)
                # Ensure constraint
                child = feasible_project(child, n_points, target_sum)

            next_population.append(child)

        # Simulated Annealing Step every 'sa_interval' generations
        if gen % sa_interval == 0:
            # Mutate best solution
            mutated = best_h.copy()
            idx = random.randint(0, n_points - 1)
            mutated[idx] = np.clip(mutated[idx] + random.uniform(-0.2, 0.2), 0, 1)
            mutated = feasible_project(mutated, n_points, target_sum)
            mutated_score = evaluate_c5(mutated, n_points)

            if mutated_score < current_best_score:
                population[0] = mutated
            else:
                delta = current_best_score - mutated_score
                if random.random() < np.exp(delta / temperature):
                    population[0] = mutated

            temperature *= cooling_rate

        population = next_population

    return (best_h, best_c5, n_points)
