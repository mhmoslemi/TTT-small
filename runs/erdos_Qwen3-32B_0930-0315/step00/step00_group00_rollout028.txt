import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(budget_s=1100, seed=42, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    h0 = np.array(initial_h_values)
    n_points = len(h0)
    dx = 2.0 / n_points
    total = n_points / 2.0  # ∫ h(x) dx = 1 => sum h[i] = n_points / 2

    h = cp.Variable(n_points, nonneg=True)
    constraints = [0 <= h, h <= 1, cp.sum(h) == total]
    h_full = h0.copy()

    def compute_c5(hv):
        hv_1h = 1 - hv
        corr = correlate(hv, hv_1h, mode='full')
        corr = corr[:len(corr)//2]  # Only symmetric overlaps from k=1 to k=n-1
        return np.max(corr * dx)

    best_c5 = compute_c5(h_full)
    best_h = h_full.copy()

    while time.time() - start_time < budget_s:
        idxs = np.random.permutation(n_points)
        feasible = True
        for idx in idxs:
            wp = 1 - h_full[idx]
            val_range = (0, 1)
            if feasible:
                new_h = h_full.copy()
                for v in [0.0, 1.0]:
                    if new_h[idx] == v:
                        continue
                    new_h[idx] = v
                    if np.sum(new_h) <= total + 1e-8 and np.sum(new_h) >= total - 1e-8:
                        c5 = compute_c5(new_h)
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = new_h.copy()
                            feasible = True
                            break
                else:
                    feasible = False
            else:
                break

        if not feasible:
            # Sequential local optimization
            step = 0.01
            for idx in range(n_points):
                for delta in [-1, 0, 1]:
                    delta_val = h_full[idx] + step * delta
                    proposed_h = h_full.copy()
                    proposed_h[idx] = np.clip(delta_val, 0.0, 1.0)
                    if np.abs(np.sum(proposed_h) - total) < 1e-8:
                        c5 = compute_c5(proposed_h)
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = proposed_h.copy()
                            break

        time_left = budget_s - (time.time() - start_time)
        if time_left < 5:
            break

    # Final post-check
    best_h[best_h < 0] = 0
    best_h[best_h > 1] = 1
    best_h /= np.sum(best_h)
    best_h *= total  # Ensure the average is correct at end

    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_points
