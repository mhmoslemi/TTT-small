import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    n_points = 70  # Prefer small for speed
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    initial_h = np.maximum(np.minimum(initial_h, 1.0), 0.0)

    if 'initial_h_values' in globals():  # Validate feasibility
        assert np.all(initial_h >= -1e-8) and np.all(initial_h <= 1.0 + 1e-8)
        assert abs(sum(initial_h) - n_points / 2) < 1e-5, "Invalid initial_h_values!"

    dx = 2.0 / n_points
    target_sum = n_points / 2  # Because dx * sum(h) = 1

    def evaluate_c5(h):
        h_padded = np.concatenate([h, 1 - h])
        c = np.correlate(h, h_padded, mode="valid")
        return np.max(c * dx)

    def project_h(h):
        h_var = cp.Variable(n_points)
        cons = [0 <= h_var, h_var <= 1, cp.sum(h_var) == target_sum]
        prob = cp.Problem(cp.Minimize(cp.sum_squares(h_var - h)), cons)
        prob.solve(solver=cp.ECOS, verbose=False)
        return h_var.value

    best_h = initial_h.copy().astype(float)
    best_c5 = evaluate_c5(best_h)
    best_h_valid = best_h.copy()
    t_start = time.time()

    grad_buffer = []
    learning_rate = 0.01
    num_iter = 0
    max_iter = 500

    h = best_h.copy()
    h_valid = best_h_valid.copy()

    while time.time() - t_start < budget_s and num_iter < max_iter:
        num_iter += 1

        # Compute cross-correlation and C₅
        hcon = np.concatenate([h, 1 - h])
        corr = np.correlate(h, hcon, mode="valid")
        idx_max = np.argmax(corr)
        k = idx_max  # worst shift

        # Compute gradient
        grad = np.zeros(n_points)

        for i in range(n_points):
            right_idx = (i + k) % n_points  # ignore %, just mod here for symmetry
            if right_idx < n_points:
                grad[i] += (1 - h[right_idx])
            if (i - k) >= 0 and (i - k) < n_points:
                grad[i] -= h[i - k]

        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0:
            continue  # avoid div by zero
        grad = grad / grad_norm

        # Line search
        lr = learning_rate
        found = False
        for _ in range(3):
            h_candidate = h - lr * grad
            h_candidate = np.clip(h_candidate, -100, 200)  # prevent overflow
            h_candidate = project_h(h_candidate)

            c5_candidate = evaluate_c5(h_candidate)
            if c5_candidate < best_c5 and np.all(h_candidate >= -1e-8) and np.all(h_candidate <= 1.0 + 1e-8):
                h = h_candidate.copy()
                best_c5 = c5_candidate
                best_h_valid = h_candidate
                found = True
                break
            lr *= 0.5
        if not found:
            continue

    best_c5 = max(evaluate_c5(h) for _ in range(5))  # reduce timing jitter
    final_h = project_h(best_h_valid)
    final_c5 = evaluate_c5(final_h)

    return (
        final_h.clip(0, 1).tolist(),
        final_c5,
        n_points
    )
