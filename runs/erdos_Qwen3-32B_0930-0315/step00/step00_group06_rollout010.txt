import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    max_n_points = 256
    dx = 2.0 / max_n_points

    # Use starting h if available, or initialize randomized
    if 'initial_h_values' in globals() and len(initial_h_values) == max_n_points:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(max_n_points)
        h = h / h.sum() * max_n_points / 2

    h = h.clip(0, 1)
    start_time = time.time()
    best_h = np.copy(h)
    best_c5 = float('inf')

    learning_rate = 0.1
    decay = 0.999
    perturb_period = 100
    perturb_scale = 0.01

    while time.time() - start_time < budget_s:
        # Normalize to enforce constraint
        h_sum = h.sum()
        if h_sum > 0:
            h = h / h_sum
        h = h.clip(0, 1)
        h = h * (max_n_points / 2) / h.sum()  # Ensure sum(h) == max_n_points / 2

        # Compute overlaps for all lags
        corrs = np.correlate(h, 1 - h, mode="full")
        corrs = corrs * dx

        # Get max overlap (target to minimize)
        current_c5 = corrs.max()
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.copy(h)

        # Select shift k with the maximal overlap
        k_max = np.argmax(corrs)

        # Compute gradient based on k_max (from convolution perspective)
        k_center = k_max - max_n_points + 1
        grad = np.zeros(max_n_points)
        if k_center >= 0:
            grad[k_center:] -= 1.0
            grad[:max_n_points - k_center] -= 1.0
        else:
            k_center = -k_center
            grad[:max_n_points - k_center] += 1.0
            grad[k_center:] += 1.0

        # Update h (minimize corr = max(∫h(1-h(x+k))dx) => move h toward reducing max overlap)
        h = h - learning_rate * (grad - h) / max_n_points

        # Clip and renormalize
        h = h.clip(0, 1)
        h_sum = h.sum()
        if h_sum > 0:
            h = h / h_sum * (max_n_points / 2)

        # Update learning rate
        learning_rate *= decay

        # Periodic perturbation to escape local minima
        if (len(best_h) - time.time() + start_time) % perturb_period == 0:
            perturbation = np.random.randn(max_n_points) * perturb_scale
            h = h + perturbation
            h = h.clip(0, 1)
            h_sum = h.sum()
            if h_sum > 0:
                h = h / h_sum * (max_n_points / 2)

    h = np.round(h, 10)
    return (h.tolist(), best_c5, max_n_points)
