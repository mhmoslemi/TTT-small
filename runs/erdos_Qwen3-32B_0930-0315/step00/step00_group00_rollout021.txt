import numpy as np
import time
from scipy.signal import correlate
import random

def compute_c5(h, dx):
    h = np.concatenate((h, np.zeros_like(h)))
    return max(correlate(h, 1 - h, mode="full")[len(h):2 * len(h)]) * dx

def tabu_search(initial_h, max_time, n_points):
    h = initial_h.copy()
    n = len(h)
    total_energy = n * 1.0
    current_energy = np.sum(h)

    def normalize_h(h):
        h = np.clip(h, 0, 1)
        scale = total_energy / np.sum(h)
        h *= scale
        return np.clip(h, 0, 1)

    def perturb_h(h):
        new_h = h.copy()
        idxs = np.random.choice(n, size=min(5, n), replace=False)
        for i in idxs:
            new_h[i] = np.random.uniform(0, 1)
        return normalize_h(new_h)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, 2.0 / n)
    tabu_list = [tuple(best_h)]
    tabu_size = 20
    start_time = time.time()
    it = 0
    budget = max_time

    while time.time() - start_time < budget:
        neighbors = []
        for _ in range(20):
            neighbor = perturb_h(h)
            neighbor_tuple = tuple(neighbor)
            if neighbor_tuple not in tabu_list:
                neighbors.append(neighbor)
                tabu_list.append(neighbor_tuple)
                if len(tabu_list) > tabu_size:
                    tabu_list.pop(0)
        for neighbor in neighbors:
            c5 = compute_c5(neighbor, 2.0 / n)
            if c5 < best_c5:
                best_h = neighbor.copy()
                best_c5 = c5
        if neighbors:
            h = min(neighbors, key=lambda x: compute_c5(x, 2.0 / n))
        it += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, n_points=45):
    random.seed(seed)
    np.random.seed(seed)

    if "initial_h_values" in globals():

        h = h[:n_points]
    else:

        idxs = np.random.choice(n_points, size=n_points // 2, replace=False)
        h[idxs] = 1.0

    h = h / np.sum(h) * n_points / 2.0  # normalize to exact integral constraint

    best_h, best_c5 = tabu_search(h, budget_s, n_points)
    return (best_h.tolist(), best_c5, n_points)
