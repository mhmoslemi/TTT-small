```python
import numpy as np
import math
import time
from scipy.signal import correlate

def evaluate_correlation(h):
    dx = 2.0 / len(h)
    corr_overlap = correlate(h, 1 - h, mode="full")
    return np.max(corr_overlap * dx)

def project_lagrange(h):
    h_clamped = np.clip(h, 0, 1)
    s = np.sum(h_clamped)
    n = len(h)
    target_sum = n / 2
    h_scaled = h_clamped * (target_sum / s)
    return h_scaled

def compute_gradient(h, k_max):
    n = len(h)
    grad = np.zeros(n)
    if k_max >= n:
        # h[i + k] is undefined for i; only tiling into arrays can resolve this.
        # Our model ensures k_max is within [0, n-1]
        return grad
    for i in range(n - k_max):
        grad[i] += (1 - h[i + k_max])
        grad[i + k_max] -= h[i]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float).copy()
        n_points = len(h)
    else:
        n_points = 47
        np.random.seed(seed)
        h = np.random.rand(n_points)

    dx = 2.0 / n_points
    h = project_lagrange(h)

    max_iter = 1000000
    best_h = h.copy()
    best_c5 = evaluate_correlation(h)
    iteration = 0

    learning_rate = 0.1
    perturbation = 0.05

    while time.time() - start_time < budget_s:
        iteration += 1
        corr = correlate(h, 1 - h, mode="full")
        overlaps = corr * dx
        c5_index = np.argmax(np.abs(overlaps))
        k_max = int(c5_index - (len(overlaps) - 1) / 2)
        k_max = max(0, min(k_max, n_points - 1))

        # Compute gradient and update
        grad = compute_gradient(h, k_max)

        # Gradient step
        h -= learning_rate * grad
        h = np.clip(h, 0 - perturbation, 1 + perturbation)
        h = project_lagrange(h)

        # Perturb occasionally
        if iteration % 10000 == 0:
            h += perturbation * (2 * np.random.rand(n_points) - 1)
            h = np.clip(h, 0, 1)
            h = project_lagrange(h)

        # Evaluate at best
        new_c5 = evaluate_correlation(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()

        if iteration % 1000 == 0:
            print(f"Iter: {iteration}, C5: {best_c5:.6f}, time: {time.time() - start_time:.2f}s")

        if time.time() - start_time > budget_s:
            break

    return (best_h.tolist(), best_c5, n_points)
```
