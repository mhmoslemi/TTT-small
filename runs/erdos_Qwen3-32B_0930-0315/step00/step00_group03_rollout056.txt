import numpy as np
import time
from scipy.signal import correlate
import random
import copy

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def ensure_constraints(h, n_points):
    n_values = n_points / 2.0
    h_sum = np.sum(h)
    factor = n_values / h_sum if h_sum != 0 else 1.0
    h = np.clip(h * factor, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum - n_values) > 1e-6:
        excess = h_sum - n_values
        indices = np.where(h > 0.5)[0]
        h[indices] = np.clip(h[indices] - excess / len(indices), 0, 1)
    return h

def create_population(n_points, population_size, initial_h=None):
    population = []
    if initial_h is not None:
        population.append(ensure_constraints(np.copy(initial_h), n_points))
        for i in range(population_size - 1):
            h = np.copy(initial_h)
            indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
            h[indices] += 0.2 - np.random.rand(len(indices))*0.4
            h = np.clip(h, 0, 1)
            population.append(ensure_constraints(h, n_points))
    else:
        for i in range(population_size):
            h = np.random.rand(n_points)
            population.append(ensure_constraints(h, n_points))
    return np.array(population)

def tournament_selection(population, fitness, tournament_size):
    indices = np.random.choice(len(population), tournament_size, replace=False)
    best_idx = indices[0]
    for idx in indices[1:]:
        if fitness[idx] < fitness[best_idx]:  # lower C5 is better
            best_idx = idx
    return population[best_idx]

def crossover(parent1, parent2):
    point = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate((parent1[:point], parent2[point:]))
    child2 = np.concatenate((parent2[:point], parent1[point:]))
    for child in [child1, child2]:
        child = ensure_constraints(child, len(child))
    return child1, child2

def mutate(h, mutation_rate=0.15, noise_std=0.1):
    h = np.copy(h)
    indices = np.random.choice(len(h), size=int(len(h) * 0.05), replace=False)
    h[indices] += np.random.normal(0, noise_std, size=len(indices))
    h = np.clip(h, 0, 1)
    h = ensure_constraints(h, len(h))
    return h

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    initial_population_size = 50
    n_points = 70
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = None

    start_time = time.time()
    population = create_population(n_points, initial_population_size, initial_h)
    best_h = None
    best_c5 = float('inf')

    while time.time() - start_time < budget_s:
        fitness = np.array([evaluate_c5(h, n_points) for h in population])

        new_population = []

        for _ in range(initial_population_size // 2):
            p1 = tournament_selection(population, fitness, tournament_size=5)
            p2 = tournament_selection(population, fitness, tournament_size=5)
            c1, c2 = crossover(p1, p2)
            new_population.extend([mutate(c1), mutate(c1), mutate(c2), mutate(c2)])

        # Keep top 10% from previous generation to prevent losing best solutions
        top_idx = np.argsort(fitness)[:max(1, initial_population_size // 10)]
        new_population += [population[i] for i in top_idx]

        # Prune to keep population size constant
        population = new_population[:initial_population_size]
        current_fitness = [evaluate_c5(h, n_points) for h in population]
        for i in range(len(population)):
            if current_fitness[i] < best_c5:
                best_c5 = current_fitness[i]
                best_h = population[i]

    best_h = np.array(best_h).astype(float)
    best_h = ensure_constraints(best_h, n_points)
    best_c5 = evaluate_c5(best_h, n_points)

    return best_h.tolist(), best_c5, n_points
