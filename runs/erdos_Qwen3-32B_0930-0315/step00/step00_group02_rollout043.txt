import numpy as np
import time
from math import log, exp

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 160
    validate_interval = 100
    max_iter = 1000
    alpha = 0.02
    alpha_decay = 0.996
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        h = h[:n_points] / np.max(h[:n_points])
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct a plateau-shaped starting point, scaled to the constraint
        x = np.linspace(0, 2, n_points)
        h = np.where((x > 0.5) & (x < 1.5), 1, 0)
        h = h / np.sum(h) * (n_points / 2)  # Enforce the sum constraint
    h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = float('inf')

    def evaluate_c5(h_eval):
        corr = np.correlate(h_eval, 1 - h_eval, mode='full')
        return np.max(corr) * dx

    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        if i % validate_interval == 0:
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                print(f"Iteration {i}: New best C5 = {best_c5:.8f}")

        # Compute the overlap for all k
        corr = np.correlate(h, 1 - h, mode='full') * dx
        k_max = np.argmax(corr)

        # Define lag k
        k = k_max

        # Compute gradient with boundary wrapping (circular shift)
        wrap_indices_right = (np.arange(n_points) + k) % n_points
        wrap_indices_left = (np.arange(n_points) - k) % n_points

        gradient = (
            (1 - h[wrap_indices_right]) - h[wrap_indices_left]
        )
        gradient_sum = np.sum(gradient)

        # Normalize gradient (useful in gradient descent)
        norm = np.max([np.abs(gradient).max(), 1e-8])

        # Step
        h = h - alpha * gradient / norm

        # Projection 1: clip to [0, 1]
        h = np.clip(h, 0, 1)

        # Projection 2: maintain mean (sum constraint)
        expected_sum = n_points / 2.0
        h = h / np.sum(h) * expected_sum

        # Decay learning rate
        alpha *= alpha_decay

        # Optional: heavy perturbation every 500 iterations if progress is slow
        if i % 500 == 0 and i > 0:
            h += 0.05 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * expected_sum

    # Final evaluation and output
    c5_bound = evaluate_c5(best_h)
    return (best_h.tolist(), c5_bound, n_points)
