import numpy as np
import cvxpy as cp
import time


def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    # 1. Use the pre-imported `initial_h_values` as starting point
    h = initial_h_values
    n_points = len(h)
    dx = 2.0 / n_points

    # 2. Normalize h to satisfy the volume constraint
    h *= n_points / (2 * h.sum())

    # Timer
    start_time = time.time()

    best_h = h.copy()
    best_c5_bound = np.inf

    while time.time() - start_time < budget_s - 0.05:  # Leave 50ms buffer

        # 3. Evaluate current C₅
        xcorr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = xcorr.max()

        if current_c5 < best_c5_bound - 1e-6:
            best_h = h.copy()
            best_c5_bound = current_c5
        elif abs(current_c5 - best_c5_bound) < 1e-8:
            break  # No significant improvement

        # 4. Identify worst-lag to focus optimization
        k_max = np.argmax(xcorr)

        # 5. Step 1: Magnitude adjustment via convex optimization
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(h_var @ (1 - np.roll(h, k_max)) * dx)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        prob_opt = cp.Problem(objective, constraints)
        prob_opt.solve(solver=cp.GUROBI, verbose=False, TimeLimit=0.1)
        if prob_opt.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h[:] = h_var.value.copy()
        h = np.clip(h, 0, 1)

        # 6. Step 2: Small random perturbation to escape local optimum
        h += 0.01 * np.random.randn(n_points)  # Small noise
        h = np.clip(h, 0, 1)
        h *= n_points / (2 * h.sum())  # Normalize back

    # 7. Final evaluation
    xcorr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    best_c5_bound = xcorr.max()

    return best_h, best_c5_bound, n_points
