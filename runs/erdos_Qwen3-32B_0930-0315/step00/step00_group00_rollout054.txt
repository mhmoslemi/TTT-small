import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, debug=False, n_points=45, tabu_tenure=20, neighborhood_size=20):
    start_time = time.time()
    np.random.seed(seed)

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.randint(0, 2, size=n_points).astype(float)
        h[h == 0] += np.random.rand(np.sum(h == 0)) * 0.8  # Perturb some 0s
        h[h == 1] -= np.random.rand(np.sum(h == 1)) * 0.8  # Perturb some 1s
        h = np.clip(h, 0, 1)
        h /= np.sum(h)  # Normalize
        h *= n_points / 2  # Satisfy constraint

    best_h = h.copy()
    tabu_list = []

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full') * (2.0 / n_points)
        return np.max(corr)

    best_c5 = evaluate(best_h)
    elapsed = time.time() - start_time
    if debug:
        print(f"Init C₅: {best_c5}, Elapsed: {elapsed}s")

    while time.time() - start_time < budget_s:
        neighbors = []
        # Perturb neighbors
        for _ in range(neighborhood_size):
            neighbor = h.copy()
            idxs = np.random.choice(n_points, size=np.random.randint(1, 6), replace=False)
            delta = np.random.uniform(-0.2, 0.2, size=len(idxs))
            neighbor[idxs] += delta
            neighbor = np.clip(neighbor, 0, 1)
            neighbor /= np.sum(neighbor) * (2.0 / n_points)
            neighbor *= n_points / 2
            neighbors.append(neighbor)

        # Evaluate all neighbors
        for neighbor in neighbors:
            neighbor_c5 = evaluate(neighbor)
            if neighbor.tolist() not in tabu_list or neighbor_c5 < best_c5:
                if neighbor_c5 < best_c5:
                    best_h = neighbor.copy()
                    best_c5 = neighbor_c5
                tabu_list.append(neighbor.tolist())
                if len(tabu_list) > tabu_tenure:
                    tabu_list.pop(0)

        # Break early if no improvement
        elapsed = time.time() - start_time
        if debug:
            print(f"Current C₅: {best_c5}, Elapsed: {elapsed}s")
        if best_c5 <= 0.380876 and elapsed > 60:  # Early stopping if target achieved
            break

    return best_h.tolist(), best_c5, n_points
