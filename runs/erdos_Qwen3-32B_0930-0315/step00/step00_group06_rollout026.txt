import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, n_points=256):
    np.random.seed(seed)

    start_time = time.time()
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Load the initial value from the global if it exists
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Start with a random guess, normalized
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum

    # Ensure h is in [0, 1]
    h = np.clip(h, 0, 1)

    # Optimization loop
    best_c5 = np.inf
    best_h = h.copy()
    learning_rate = 0.5
    decay_rate = 0.9995
    perturb_freq = 100

    while time.time() - start_time < budget_s:
        # Compute cross-correlation of h and 1-h
        overlap = np.correlate(h, 1 - h, mode='full') * dx

        # Ensure we only look at valid shifts
        if n_points > len(overlap):
            raise ValueError("Unexpected correlation result length.")
        valid_overlap = overlap[n_points:]

        # Find the shift with maximum overlap
        min_ac = np.argmin(np.correlate(h, h, mode='full') * dx)
        k_min = min_ac % n_points

        # Compute gradient for the worst shift
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + k_min >= n_points:
                grad[i] = -h[i + k_min - n_points]
            else:
                grad[i] = -h[i + k_min]
        grad = grad * dx

        # Update step
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)

        # Normalize to satisfy sum constraint
        current_sum = np.sum(h)
        h = h * (target_sum / current_sum)

        # Periodically perturb for robustness
        if (time.time() - start_time) % perturb_freq < 1e-5 and (time.time() - start_time) > 0:
            noise = np.random.randn(n_points)
            noise = np.clip(noise, -1, 1)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))

        # Compute C5 for tracking
        overlap = np.correlate(h, 1 - h, mode='full') * dx
        valid_overlap = overlap[n_points:]
        current_c5 = np.max(valid_overlap)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Learning rate decay
        learning_rate *= decay_rate

        # Early stopping if improvement stalls
        if learning_rate < 1e-6:
            break

    # Final check
    valid_overlap = np.max(np.correlate(best_h, 1 - best_h, mode='full')[n_points:] * dx)
    return best_h, valid_overlap, n_points
