import numpy as np
import cvxpy as cp
import time
from numpy import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    # Parameters
    initial_h = np.array(initial_h_values)
    n_points = initial_h.shape[0]
    dx = 2.0 / n_points
    h = cp.Variable(n_points, nonneg=True)
    
    alpha = 1.0 / np.sum(h.value) if h.value is not None else 1.0
    h = cp.Variable(n_points, nonneg=True)
    constraint_sum = cp.sum(h) == n_points / 2
    
    # Barrier parameter
    mu = 1e4
    barrier_term = 1e-6  # Small value to avoid log(0)
    barrier_area = cp.log(barrier_term + cp.sum(h)) + cp.log(barrier_term + n_points/2 - cp.sum(h))
    
    # Compute objective
    h_var = h
    objective_overlaps = []
    for k in range(-n_points + 1, n_points):
        indices = np.arange(0, n_points, dtype=int)
        indices_k = np.clip(indices + k, 0, n_points - 1)
        overlap = cp.sum(h_var[indices] * (1 - h_var[indices_k]))
        scaled_overlap = overlap * dx
        objective_overlaps.append(scaled_overlap)

    max_overlap = cp.Max(objective_overlaps)
    objective = max_overlap + mu * barrier_area

    constraints = [h <= 1.0, constraint_sum]

    # Set up problems list for barrier method
    print_iterations = 50
    best_h = None
    best_c5 = float('inf')

    for i in range(1000):
        if time.time() - t_start > budget_s - 5:  # Reserve 5 sec to wrap up
            break

        if i % print_iterations == 0:
            print(f"Iter: {i}, mu: {mu:.2e}")

        try:
            problem = cp.Problem(cp.Minimize(objective), constraints)
            problem.solve(solver=cp.CVXOPT, verbose=False)

            if problem.status in ["optimal", "optimal_inaccurate"]:
                h_result = h.value
                # Clipping and scaling for robustness
                h_result[h_result < 0] = 0
                h_result[h_result > 1] = 1
                norm = np.sum(h_result)
                h_result = h_result * max(1, n_points / 2 / norm)

                # Evaluate measured C5
                c5_val = evaluate_erdos_solution(h_result)

                if c5_val < best_c5:
                    best_h = h_result.copy()
                    best_c5 = c5_val

            # Barrier update strategy
            mu = max(1e-8, mu * 0.95)

        except Exception as e:
            print(f"Optimization failed at iteration {i}: {e}")
    else:
        # Handle immediate timeout (i.e., optimization not used)
        best_h = initial_h
        best_c5 = evaluate_erdos_solution(best_h)

    return (best_h.tolist(), best_c5, n_points)
