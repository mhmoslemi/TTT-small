import time
import numpy as np
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    n_points = 256
    dx = 2.0 / n_points
    
    # Use initial_h_values if provided, otherwise go with uniform distribution
    if 'initial_h_values' in globals():
        # Upsample initial_h_values to 256 points using linear interpolation
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    else:
        # Construct a binary starting pattern assuming half is 1 and half is 0
        h = np.zeros(n_points, dtype=np.float64)
        h[:n_points // 2] = 1.0
    
    def normalize_h(h):
        h = np.clip(h, 0.0, 1.0)
        h = h / np.sum(h) * (n_points / 2)
        return h

    best_h = h.copy()
    best_c5 = None
    h = normalize_h(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode='same')
        return np.max(overlap) * dx

    for iteration in range(50):
        if time.time() - start_time > budget_s - 10:  # Leave some safety time
            break

        h_var = cp.Variable(n_points, nonneg=True)
        constraints = [h_var <= 1.0]
        
        overlap_k = correlate(h_var, 1 - h, mode='full') * dx
        worst_k = np.argmax(overlap_k)
        current_shift = overlap_k[worst_k]
        c5_objective = cp.Minimize(current_shift)
        
        constraints.extend([
            cp.sum(h_var) == n_points / 2,
            cp.norm(h_var - h, 2) <= 1.0
        ])

        problem = cp.Problem(c5_objective, constraints)
        try:
            problem.solve(solver=cp.OSQP)
            if problem.status in ['optimal', 'optimal_inaccurate']:
                h = np.clip(h_var.value, 0.0, 1.0)
                h = normalize_h(h)
                current_c5 = compute_c5(h)
                if best_c5 is None or current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h
        except Exception as e:
            continue  # Skip on failure

    return (best_h, best_c5, n_points)
