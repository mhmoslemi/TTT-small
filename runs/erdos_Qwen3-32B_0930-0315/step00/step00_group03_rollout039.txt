import numpy as np
import time
import cvxpy as cp


def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)
    
    # Use initial construction if provided
    initial_h = np.asarray(initial_h_values) if 'initial_h_values' in globals() else np.full(70, 0.5)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    h = cp.Parameter(n_points, nonneg=True)
    constraints_h = [h >= 0, h <= 1, cp.sum(h) == n_points / 2.0]
    
    def project_h(h_candidate):
        proj_problem = cp.Problem(cp.Minimize(0), constraints_h + [h == h_candidate])
        proj_problem.solve(solver=cp.GLPK)
        return h.value
    
    best_h = initial_h.copy()
    best_c5 = float('inf')
    no_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        h_np = best_h
        
        # Step 1: Compute overlaps
        h_flat = h_np.flatten()
        correl = np.correlate(h_flat, 1 - h_flat, mode='full') * dx
        
        # Truncate the outer estimates where indices are invalid
        min_shift = 0
        max_shift = n_points - 1
        correl = correl[min_shift:max_shift+1]
        
        # Step 2: Find max overlap
        c5 = np.max(correl)
        if c5 < best_c5:
            best_h = h_np.copy()
            best_c5 = c5
            no_improvement = 0
        else:
            no_improvement += 1
        
        if no_improvement >= 100:
            break
        
        # Choose the most constraining shift k
        k = np.argmax(correl)
        G = np.zeros(n_points)
        
        for i in range(n_points - k):
            G[i] += 1 - best_h[i + k]
        for j in range(k, n_points):
            G[j] -= best_h[j - k]
        
        # Step 3: Apply update with decaying learning rate
        lr = 0.01 / (1 + iteration / 1000)
        candidate_h = best_h - lr * G
        
        # Step 4: Project back into the constraint set
        proj_h = project_h(candidate_h)
        
        if proj_h is not None and np.isfinite(proj_h).all():
            best_h = proj_h
            iteration += 1
    
    return best_h.tolist(), best_c5, n_points
