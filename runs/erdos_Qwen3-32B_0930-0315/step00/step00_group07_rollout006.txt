import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    dx = 2.0 / 80
    n_points = 80
    h = initial_h_values[:n_points].copy()

    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    tol = 1e-5
    max_iter = 400

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break

        # Compute overlap for all lags
        h_array = h - h * h
        c = np.correlate(h_array, np.ones_like(h), mode='full') * dx
        max_c5_idx = np.argmax(c[len(h)-1:])
        k = max_c5_idx + 1

        # Shift h by k
        h_k = np.roll(h, -k)

        # Construct CVXPY minimization problem
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(h_var * (1 - cp.roll(h_var, -k))) * dx)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        prob = cp.Problem(objective, constraints)

        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
        except:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except:
                continue

        if prob.status != 'optimal':
            continue

        h_new = np.clip(np.array(h_var.value), 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        h_new = np.clip(h_new, 0, 1)

        c_new = evaluate_erdos_solution(h_new)
        h[c_new < best_c5] = h_new[c_new < best_c5]

        # Track best solution
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

        # Early stopping
        if best_c5 <= 0.380876:
            break

    return (best_h, best_c5, n_points)
