import numpy as np
import math
import time

def compute_overlap(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    corr = corr[n_points - 1:n_points + n_points - 1]  # Valid indices only
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - n_points / 2) > 1e-6:
        h *= (n_points / 2) / s
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    n_points = 70  # Start from the known construction resolution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # No parent construction given: uniform initialization
        h = np.ones(n_points) / n_points * (n_points / 2)
    best_h = h.copy()
    best_c5 = compute_overlap(h, n_points)
    alpha = 0.1
    iteration = 0
    max_iterations = 5000

    while time.time() - start < budget_s:
        iteration += 1
        if iteration >= max_iterations:
            break
        if iteration % 50 == 0:
            # Perturbation to escape local minima
            noise = np.random.normal(0, 0.01, n_points)
            h += noise
            h = project_h(h, n_points)

        # Compute overlaps for all valid lags
        corr = np.correlate(h, 1 - h, mode='full')
        valid_corr = corr[n_points - 1 : 2 * n_points - 1]
        k_max_index = np.argmax(valid_corr)
        k_max = k_max_index

        # Compute gradient for the critical lag
        grad = np.zeros(n_points)

        # Forward direction
        for i in range(n_points - k_max):
            grad[i] -= (1 - h[i + k_max])  # Derivative w.r.t. h[i]
        # Backward direction
        for i in range(k_max, n_points):
            grad[i] += h[i - k_max]  # Derivative w.r.t. h[i]

        grad *= (2.0 / n_points)  # Gradient scaled by dx

        # Update h in gradient direction
        h -= alpha * grad
        h = project_h(h, n_points)

        # Evaluate new C5 and update best
        current_c5 = compute_overlap(h, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            alpha = 0.7 * alpha  # Optional step size decay

    # Return the best solution found
    return (best_h.tolist(), best_c5, n_points)
