import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)
    
    # Use initial construction
    h0 = np.array(initial_h_values)
    
    n_points = len(h0)
    dx = 2.0 / n_points

    # Constraints
    def objective(h):
        h = h.reshape(-1)
        max_overlap = 0.0
        for k in range(-n_points + 1, n_points):
            offset = np.roll(h, k)
            overlap = np.dot(h, 1 - offset) * dx
            max_overlap = max(max_overlap, overlap)
            if time.time() - start_time > budget_s:
                return max_overlap
        return max_overlap

    # Bounds and constraints
    bounds = [(0.0, 1.0)] * n_points
    cons = [{"type": "eq", "fun": lambda h: np.sum(h) - n_points / 2}]
    
    # Optimization with L-BFGS-B
    result = scipy.optimize.minimize(
        objective,
        h0,
        bounds=bounds,
        constraints=cons,
        method="L-BFGS-B",
        options={"maxiter": int(budget_s * 100)},
    )
    
    optimized_h = result.x
    best_c5 = objective(optimized_h)
    
    return (optimized_h, best_c5, n_points)
