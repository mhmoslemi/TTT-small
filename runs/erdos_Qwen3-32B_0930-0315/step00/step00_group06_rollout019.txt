```python
import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    n_points = 256
    dx = 2.0 / n_points
    total_area = 1.0
    target_sum = int(total_area / dx + 1e-8)

    # Initialize
    initial_values = np.zeros(n_points)
    if "initial_h_values" in globals():
        initial_values = np.array(initial_h_values[:n_points])
        if initial_values.shape[0] < n_points:
            initial_values = np.pad(initial_values, (0, n_points - initial_values.shape[0]), mode='constant')
        initial_values = initial_values.astype(float)
        # normalize to sum to target_sum * dx
        s = initial_values.sum()
        if s > 1e-14:
            initial_values *= target_sum / s
    else:
        # create a binary initial distribution with area ≈ 1
        indices = np.random.choice(n_points, size=target_sum, replace=False)
        initial_values[indices] = 1.0

    h = initial_values.copy()
    h_sum = h.sum()
    h = h * (target_sum / h_sum)
    assert (h <= 1 + 1e-5).all(), "value exceeds 1"
    assert np.isclose(h.sum(), target_sum), f"Sum = {h.sum()}, target = {target_sum}"

    best_h = h.copy()
    best_c5 = np.inf

    # optimization loop
    max_iter = 1000
    for iter in range(max_iter):
        if time.time() - t_start > budget_s - 1:
            break

        # Compute correlations
        corr = correlate(h, 1.0 - h, mode='full')
        corr = corr * dx
        c5 = np.max(corr)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

        # Find the peak
        mid = n_points
        max_idx = np.argmax(corr)
        k = max_idx - mid
        h1 = h[: n_points - k]
        h2 = (1 - h[k:])
        overlap = np.sum(h1 * h2 * dx)

        # compute gradient
        grad = np.zeros_like(h)
        grad[:-k] += (1 - h[k:]) * dx + (h[:-k] - overlap / (np.sum(h1 * dx) * np.sum(h2 * dx)))
        grad[k:] -= (h[:-k] * dx + h[k:] - overlap / (np.sum(h1 * dx) * np.sum(h2 * dx)))

        # step and project
        step_size = 0.1
        h_candidate = h + step_size * grad
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate *= target_sum / h_candidate.sum()

        if np.isfinite(h_candidate).all() and np.isclose(h_candidate.sum(), target_sum):
            h = h_candidate

    return best_h.astype(float), best_c5, n_points
```
