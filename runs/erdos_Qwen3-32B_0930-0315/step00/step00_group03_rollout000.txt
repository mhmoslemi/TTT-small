import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Configuration
    start_time = time.time()
    initial_n_points = len(initial_h_values)

    best_c5 = np.inf
    best_h = None
    best_n_points = initial_n_points

    def objective(h, n_points):
        dx = 2.0 / n_points
        h_vals = h.reshape(-1)
        overlaps = np.correlate(h_vals, 1 - h_vals, mode="full") * dx
        return np.max(overlaps)

    def project_constraints(h, n_points):
        h = np.clip(h, 0, 1)
        missing = n_points / 2 - np.sum(h)
        h = h + missing / n_points
        h = np.clip(h, 0, 1)
        return h

    def callback(xk):
        nonlocal best_c5, best_h, best_n_points, start_time
        if time.time() - start_time >= budget_s:
            return True
        c5 = objective(xk, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = xk.copy()
        return False

    n_points = initial_n_points
    h0 = np.array(project_constraints(initial_h_values.copy(), n_points))
    res = opt.minimize(
        fun=lambda h: objective(h, n_points),
        x0=h0,
        method='L-BFGS-B',
        bounds=[(0, 1)] * n_points,
        callback=callback,
        options={'maxiter': 1000, 'gtol': 1e-8}
    )

    best_h = project_constraints(best_h, n_points)

    return (best_h.tolist(), best_c5, best_n_points)
