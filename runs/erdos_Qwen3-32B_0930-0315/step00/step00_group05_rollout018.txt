import numpy as np
import time
import cvxpy as cp
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, n_points_start=70, n_points_refine=300):
    np.random.seed(seed)
    start_time = time.time()

    # Clone initial_h_values to avoid mutation of input
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.zeros(n_points_start)
        h[:n_points_start // 2] = 1
        np.random.shuffle(h)

    h = h.clip(0, 1)
    h = h / h.sum() * n_points_start / 2

    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    n_points = h.shape[0]

    def correlation_contribution(h1, h2):
        return np.correlate(h1, (1 - h2), mode='full')

    def optimize_step(h, n_points, dx):
        h = cp.Variable(n_points)
        objective = cp.Minimize(cp.Max(cp.Conv(2 * h, h, mode="full")))
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.ECOS)
        return np.array(h.value)

    optimize_iterations = 200
    for it in range(optimize_iterations):
        t = time.time()
        if t - start_time > budget_s * 0.9:
            break

        # Step 1: Run convex optimization
        h_opt = optimize_step(h, n_points, dx=2 / n_points)

        # Step 2: Normalize and clip
        h_opt = h_opt.clip(0, 1)
        scale = (n_points / 2) / h_opt.sum()
        h_opt *= scale

        # Step 3: Compute new C₅
        c5_new = evaluate_erdos_solution(h_opt)

        # Step 4: Accept improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_opt.copy()

        # Perturbation step to avoid local minima
        mask = np.random.rand(n_points) < 0.05
        h = best_h.copy()
        h[mask] = np.random.rand(np.sum(mask))
        h = h.clip(0, 1)
        scale = (n_points / 2) / h.sum()
        h *= scale

    # Optionally, refine at higher resolution
    if h.shape[0] < n_points_refine and (time.time() - start_time < budget_s * 0.95):
        h_fin = np.interp(np.linspace(0, 2, n_points_refine), np.linspace(0, 2, n_points), h)
        h_fin = h_fin.clip(0, 1).copy()
        scale = (n_points_refine / 2) / h_fin.sum()
        h_fin *= scale
        c5_refin = evaluate_erdos_solution(h_fin)
        if c5_refin < best_c5:
            best_c5 = c5_refin
            best_h = h_fin
            n_points = h_fin.shape[0]

    # Ensure final solution meets the contract
    assert sum(best_h) * (2.0 / n_points) <= 1.0 + 1e-8
    assert all(0 <= v <= 1 for v in best_h)

    return (best_h, best_c5, n_points)
