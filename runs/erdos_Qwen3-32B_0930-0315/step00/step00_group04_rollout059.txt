import numpy as np
import time
from scipy.interpolate import CubicSpline
from math import exp


def compute_overlap(h):
    dx = 2.0 / h.size
    C = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(C)


def repair_feasibility(h):
    h = np.clip(h, 0, 1)
    h = h * (h.size / 2) / h.sum()
    return h


def optimize_and_refine(initial_h, budget=1100):
    best_h = None
    best_C = float('inf')
    start_time = time.time()

    n = len(initial_h)
    dx = 2.0 / n
    h = repair_feasibility(initial_h)

    # Budget allocation: log(n)
    T0 = 0.05 * (1.0 / n)
    alpha = 0.99
    steps = 1000
    lr = 0.01

    # Start with global search at full resolution
    for step in range(steps):
        if time.time() - start_time > budget:
            break

        h_candidate = h + np.random.normal(0, T0 * (alpha ** step), h.shape)
        h_candidate = repair_feasibility(h_candidate)
        C_candidate = compute_overlap(h_candidate)

        if C_candidate < best_C:
            best_h = h_candidate.copy()
            best_C = C_candidate
            h = h_candidate
            T0 *= 0.95  # Adaptive temperature if improvement occurs

    # Transition to gradient descent
    for _ in range(max(1, int(n * 100))):
        if time.time() - start_time > budget:
            break

        C = np.correlate(h, 1 - h, mode='full') * dx
        k_max = np.argmax(C)

        shift = k_max - (n - 1)
        if shift < 0:  # Handle overlap with 1-h
            h_shifted = np.roll(h, shift)
        else:
            h_shifted = np.roll(h, shift)
        gradient = (1 - h_shifted)

        h = repair_feasibility(h - lr * gradient[:n])
        C_current = compute_overlap(h)
        if C_current < best_C:
            best_h = h.copy()
            best_C = C_current

    return best_h, best_C


def run(seed=42, budget_s=1100, **kwargs):
    if not hasattr(run, 'initial_h_values'):
        n = 42
        h_random = np.random.rand(n)
        h_random = h_random / h_random.sum() * (n / 2)
        run.initial_h_values = h_random

    np.random.seed(seed)
    optimal_h, lowest_C = optimize_and_refine(run.initial_h_values, budget_s)
    n_points = len(optimal_h)

    return (optimal_h.tolist(), lowest_C, n_points)
