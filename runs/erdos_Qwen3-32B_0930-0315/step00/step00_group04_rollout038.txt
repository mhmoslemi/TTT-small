import time
import numpy as np
import cvxpy as cp
from evaluate_erdos_solution import evaluate_erdos_solution
import initial_h_values as init_h  # if exists

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = len(init_h)
    dx = 2.0 / n_points
    h_init = np.array(init_h, dtype=np.float64)
    h_init = h_init / np.sum(h_init) * (n_points / 2)
    h_init = np.clip(h_init, 0, 1)

    h = cp.Variable(n_points)
    obj_vals = []
    best_h = h_init
    best_c5 = np.inf

    def compute_c5(h_val):
        h_f = np.clip(h_val, 0, 1)
        h_f_sum = np.sum(h_f) * dx
        if not np.isclose(h_f_sum, 1, atol=1e-5):
            h_f = h_f / np.sum(h_f) * (n_points / 2)
        cor = np.correlate(h_f, 1 - h_f, mode='full') * dx
        return float(np.max(cor))

    while time.time() - start_time < budget_s - 5:
        mu = max(1e-8, 1e4 * (budget_s - time.time()) / 1100)
        cor = np.correlate(best_h, 1 - best_h, mode='full') * dx
        k_max = np.argmax(cor)
        target = cor[k_max]

        objective = cp.Maximize(
            -target
            + mu * cp.sum(cp.log(best_h))
            + mu * cp.sum(cp.log(1 - best_h))
            + mu * cp.log(1 - 1e-8 - cp.abs(np.sum(best_h) * dx - 1))
        )
        constraints = [
            1e-8 <= best_h,
            best_h <= 1 - 1e-8,
            cp.sum(best_h) * dx == 1,
        ]

        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False)

        if prob.status == cp.OPTIMAL:
            new_h = best_h.value
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h / np.sum(new_h) * (n_points / 2)
            current_c5 = compute_c5(new_h)
            if current_c5 < best_c5:
                best_h = new_h
                best_c5 = current_c5

        if 1e-8 < mu < 1e-6:
            break

    # Final validity check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
