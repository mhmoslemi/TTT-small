```python
import numpy as np
import math
import time

def run(seed=42, budget_s=1100, **kwargs):
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.copy(globals()['initial_h_values'])
        n_points = len(h)
    else:
        # Construct a basic initial vector -- uniform, normalized
        n_points = 128
        h = 0.5 * np.ones(n_points)
    
    dx = 2.0 / n_points

    # Clamp to [0, 1] and rescale to unit area
    def project(h):
        h = np.clip(h, 0, 1)
        scale = n_points / 2 / np.sum(h)
        h = h * scale
        return h

    def evaluate_C5(h):
        ones_minus_h = 1 - h
        corr = np.correlate(h, ones_minus_h, mode="full")
        return np.max(corr) * dx  # scale by dx for integration

    learning_rate = 0.01
    accuracy = 1.0e-4
    history = []
    start_time = time.time()

    # Initial projection
    h = project(h)
    best_h = np.copy(h)
    best_c5 = evaluate_C5(h)

    while time.time() - start_time < budget_s - 0.1:
        prev_h = np.copy(h)
        ones_minus_h = 1 - h

        # Compute cross-correlation manually to avoid mode="full" overhead
        corrs = []
        for k in range(1, n_points):
            corr = np.dot(h[:n_points - k], 1 - h[k:])
            corrs.append(corr)
        for k in range(1, n_points):
            corr = np.dot(h[k:], 1 - h[:n_points - k])
            corrs.append(corr)

        max_corr = max(corrs)
        best_k = np.argmax(corrs)

        # Compute gradient with best_k
        if best_k < n_points:
            grad = np.zeros(n_points)
            grad[best_k:] += (1 - h[:n_points - best_k])
            grad[:n_points - best_k] -= h[best_k:]
        else:
            best_k = best_k - n_points + 1
            grad = np.zeros(n_points)
            grad[:n_points - best_k] += (1 - h[best_k:])
            grad[best_k:] -= h[:n_points - best_k]

        # Update h using gradient descent
        h = h - learning_rate * grad
        h = project(h)

        # Compute new C5
        c5 = evaluate_C5(h)

        # Check convergence
        if abs(best_c5 - c5) < accuracy:
            break

        if c5 < best_c5:
            best_c5 = c5
            best_h = np.copy(h)
        else:
            # Perturbation if plateau
            h = best_h + 0.01 * np.random.randn(*h.shape)
            h = project(h)

    return best_h, best_c5, n_points
```
