import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Use the parent-supplied construction
    h = np.array(initial_h_values)
    n_points = h.size
    dx = 2.0 / n_points

    # Enforce constant sum constraint with precision
    scale_factor = (n_points / 2.0) / h.sum()
    h = h * scale_factor
    h = np.clip(h, 0.0, 1.0)

    start_time = time.time()
    max_duration = budget_s
    alpha = 0.1  # Initial learning rate
    perturb_interval = 50
    patience = 100  # To reduce step size if C5 stops improving
    best_c5 = float('inf')
    no_progress = 0

    while time.time() - start_time < max_duration:
        h = np.clip(h, 0.0, 1.0)

        # Compute overlap for all lags
        overlap = correlate(h, 1 - h, mode='full') * dx

        # Valid indices for -n_points < lag < n_points
        valid_overlap = overlap[n_points - 1 : 2 * n_points - 1]
        c5 = valid_overlap.max()

        if c5 < best_c5:
            best_c5 = c5
            no_progress = 0
        else:
            no_progress += 1
            if no_progress >= patience:
                alpha *= 0.5
                no_progress = 0

        # Determine worst-case lag: index of max overlap
        max_overlap_index = np.argmax(valid_overlap)
        lag = max_overlap_index

        dh = np.zeros_like(h)

        # Compute gradient
        for i in range(n_points):
            if i + lag < n_points:
                dh[i] += (1 - h[i + lag]) * dx
            if i - lag >= 0:
                dh[i] -= h[i - lag] * dx

        # Perform gradient descent
        h -= alpha * dh
        # Project back to constraint set
        h = np.clip(h, 0.0, 1.0)

        # Perturb h to avoid local optima
        if (time.time() - start_time) < max_duration and (np.random.randint(0, perturb_interval) == 0):
            noise = np.random.randn(n_points) * 0.01
            h += noise
            h = np.clip(h, 0.0, 1.0)
            scale_factor = (n_points / 2.0) / h.sum()
            h = h * scale_factor
            h = np.clip(h, 0.0, 1.0)

    # Final constraint enforcement
    h = np.clip(h, 0.0, 1.0)
    scale_factor = (n_points / 2.0) / h.sum()
    h = h * scale_factor

    # Final evaluation
    overlap = correlate(h, 1 - h, mode='full') * dx
    valid_overlap = overlap[n_points - 1 : 2 * n_points - 1]
    final_c5 = valid_overlap.max()

    return h.tolist(), final_c5, n_points
