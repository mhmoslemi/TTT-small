        h -= learning_rate * grad

        h = np.clip(h, 0, 1)

        total = np.sum(h)
        if np.isclose(total, 0):

        h *= (n_points / 2) / total

        # Learning rate decay
        if iter_outer % 100 == 0 and iter_outer > 0:
            learning_rate *= 0.95
