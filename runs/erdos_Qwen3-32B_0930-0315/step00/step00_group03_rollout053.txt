import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    dx = 0.0285714285714  # dx = 2.0 / 70
    n_points = 70
    np.random.seed(seed)

    # Use the provided initial construction if available
    try:
        h = initial_h_values.copy()
    except:
        h = np.full(n_points, 0.5)  # fallback initialization

    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()

    def compute_c5(h):
        ones_minus_h = 1 - h
        overlaps = np.correlate(h, ones_minus_h, mode="full") * dx
        return float(np.max(overlaps[n_points:2*n_points]))

    # Optimization loop
    iter_no = 0
    no_improvement = 0
    while time.time() - start_time < budget_s:
        # Compute overlaps and identify worst-case k
        ones_minus_h = 1 - h
        overlaps = np.correlate(h, ones_minus_h, mode="full") * dx
        worst_k = np.argmax(overlaps[n_points:2*n_points])  # k in [0, n)
        worst_k += n_points  # for use in full-length array

        # Compute gradient with respect to worst_k
        G = np.zeros_like(h)
        for i in range(n_points):
            if i + worst_k - n_points < n_points:
                G[i] += ones_minus_h[i + worst_k - n_points]
            if i - worst_k + n_points >= 0:
                G[i] -= h[i - worst_k + n_points]

        # Apply gradient step
        lr = 0.01 / (1 + iter_no / 1000)
        h_candidate = h - lr * G

        # Enforce constraints via projection
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum_squares(h_var - h_candidate))
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, ignore_dpp=True)
        if prob.status != cp.OPTIMAL:
            # fallback projection to convex constraints if solver fails
            h_candidate = np.clip(h_candidate, 0, 1)
            scale = n_points / 2 / np.sum(h_candidate)
            h_candidate = h_candidate * scale
        h = h_var.value if prob.status == cp.OPTIMAL else h_candidate

        # Validate rounding
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h[0] += (n_points / 2 - h_sum)  # adjust one point to balance sum
        h[0] = min(1, max(0, h[0]))

        # Evaluate and update best solution
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1

        # Terminate if plateaus
        if no_improvement > 100:
            break

        iter_no += 1

    return (best_h, best_c5, n_points)
