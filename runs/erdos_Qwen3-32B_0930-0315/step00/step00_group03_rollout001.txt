import numpy as np
import scipy.optimize
import time

def evaluate_erdos_solution(h_vec, n_points):
    dx = 2.0 / n_points
    h = h_vec.reshape(-1)
    full_corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(full_corr)

def project_h(h, n_points):
    # Enforce constraints 0 ≤ h ≤ 1
    h_projected = np.clip(h, 0, 1)
    # Enforce sum constraint
    norm = np.sum(h_projected)
    scale = n_points / 2 / norm
    h_projected *= scale
    return h_projected

def objective(h_flat, n_points):
    h = h_flat.reshape(-1)
    return evaluate_erdos_solution(h, n_points)

def optimize_h(initial, n_points, max_time=1100, verbose=False):
    start_time = time.time()
    res = scipy.optimize.minimize(
        objective,
        initial.flatten(),
        args=(n_points,),
        method='L-BFGS-B',
        options={'ftol': 1e-8, 'disp': False, 'maxiter': 10000}
    )

    # Check if time expired
    if time.time() - start_time > max_time:
        if verbose:
            print("Time budget exceeded. Returning best so far.")
        return res.x, res.fun

    if not res.success:
        if verbose:
            print("Optimization failed. Projecting...")
        h_best = res.x
    else:
        if verbose:
            print("Optimization succeeded.")
        h_best = res.x

    h_best = project_h(h_best, n_points)
    c5 = evaluate_erdos_solution(h_best, n_points)
    return h_best, c5
