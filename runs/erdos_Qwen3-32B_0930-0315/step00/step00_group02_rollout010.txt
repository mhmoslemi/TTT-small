import numpy as np
import time
from scipy.signal import correlate

def evaluate_C5(h, n_points):
    dx = 2.0 / n_points
    h_extended = np.tile(h, 5)  # Replicate h to handle all lags
    ones_minus_h = 1 - h
    corr = correlate(h_extended, ones_minus_h, mode='valid') * dx
    c5_bound = np.max(corr)
    return c5_bound

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    t0 = time.time()

    # Use initial_h_values if available, otherwise create a plateau
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        if np.isclose(np.sum(initial_h_values), n_points / 2):
            h = initial_h_values
        else:
            raise ValueError("Initial h does not satisfy sum constraint.")
    else:
        # No parent available; initialize a symmetric h with plateau-like shape
        n_points = 160
        x = np.linspace(0, 2, n_points)
        h = np.clip(0.4 + 0.1 * np.sin(np.pi * x), 0, 1)
        h /= h.sum() * 2  # Normalize to satisfy ∫h(x)dx = 1

    dx = 2.0 / n_points
    h = np.array(h)

    max_iter = 1000
    alpha = 0.01
    improvement_patience = 1000
    best_c5 = evaluate_C5(h, n_points)
    best_h = h.copy()
    current_iter = 0
    no_improvement = 0

    # Run optimization
    while time.time() - t0 < budget_s and current_iter < max_iter:
        # Compute correlation
        corr = correlate(np.tile(h, 2), (1 - h), mode='valid') * dx
        max_idx = np.argmax(corr)
        if max_idx >= n_points:
            max_idx -= n_points  # Adjust for overlapping wraparound
        k_max = max_idx

        # Gradient step: dC / dh[i] = (1 - h[i + k_max]) - h[i - k_max]
        i_plus_k = (np.arange(n_points) + k_max) % n_points
        i_minus_k = (np.arange(n_points) - k_max) % n_points
        grad = (1 - h[i_plus_k]) - h[i_minus_k]

        # Gradient descent
        h = h - alpha * grad

        # Project onto [0,1]^n
        h = np.clip(h, 0, 1)

        # Rescale to satisfy sum constraint
        current_sum = h.sum()
        h = h * (n_points / 2) / current_sum

        current_c5 = evaluate_C5(h, n_points)
        current_iter += 1

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > improvement_patience // 10:
                alpha *= 0.5  # Reduce adaptive learning rate
            elif no_improvement > improvement_patience // 2:
                h = best_h * 0.99 + np.random.randn(n_points) * 0.01  # Perturb
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / h.sum()

    return best_h.tolist(), best_c5, n_points
