import numpy as np
import scipy
import cvxpy as cp
import time
import random
from math import exp

def evaluate_candidate(h, n_points):
    dx = 2.0 / n_points
    max_corr = 0.0
    h_flipped = 1 - h
    for shift in range(-n_points + 1, n_points):
        if shift < 0:
            corr = np.sum(h[abs(shift):] * h_flipped[:n_points - abs(shift)])
        else:
            corr = np.sum(h[:n_points - shift] * h_flipped[shift:])
        max_corr = max(max_corr, corr * dx)
    return max_corr

def clip_and_normalize(h):
    h_clipped = np.clip(h, 0, 1)
    total_mass = h_clipped.sum()
    if abs(total_mass - n_points / 2) > 1e-6:
        scaling = (n_points / 2) / total_mass
        h_clipped *= scaling
    return h_clipped

def mutation(h, mutation_rate=0.05, temp=1.0):
    h_mut = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            delta = np.random.normal(0, 0.2)
            h_mut[i] += delta * temp / (1 + temp)
    return clip_and_normalize(h_mut)

def crossover(parent1, parent2):
    idx = random.randint(1, len(parent1) - 2)
    child1 = np.concatenate((parent1[:idx], parent2[idx:]))
    child2 = np.concatenate((parent2[:idx], parent1[idx:]))
    return clip_and_normalize(child1), clip_and_normalize(child2)

def tournament_selection(population, fitness, tournament_size=5):
    indices = random.sample(range(len(population)), min(tournament_size, len(population)))
    best_idx = min(indices, key=lambda i: fitness[i])
    return population[best_idx], fitness[best_idx]

def run(seed=42, budget_s=1100):
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    n_points = 45
    dx = 2.0 / n_points
    initial_population = []

    # Use the provided initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values).copy()
        h0 = clip_and_normalize(h0)
        initial_population.append(h0)

    # Add random samples to the population
    population_size = 75
    for _ in range(population_size - len(initial_population)):
        h = np.random.uniform(0, 1, n_points)
        h = clip_and_normalize(h)
        initial_population.append(h)

    population = initial_population
    fitness = [evaluate_candidate(h, n_points) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    # Simulated Annealing params
    temp = 100
    cooling_rate = 0.95
    sa_interval = 50
    gen = 0
    max_generations = 100000

    while time.time() - start_time < budget_s:
        for _ in range(10):
            # Tournament selection
            parent1, f1 = tournament_selection(population, fitness)
            parent2, f2 = tournament_selection(population, fitness)

            # Crossover
            child1, child2 = crossover(parent1, parent2)
            c1 = evaluate_candidate(child1, n_points)
            c2 = evaluate_candidate(child2, n_points)

            # Insert new children
            worst_idx = np.argmax(fitness)
            if c1 < fitness[worst_idx]:
                population[worst_idx] = child1
                fitness[worst_idx] = c1
            if c2 < fitness[worst_idx] or (c2 < fitness[np.argmin(fitness)]):
                population[worst_idx] = child2
                fitness[worst_idx] = c2

            # Mutation
            if random.random() < 0.05:
                i = random.randint(0, population_size - 1)
                mut = mutation(population[i], temp=0.5 * (temp / (1 + temp)))
                m_c5 = evaluate_candidate(mut, n_points)
                if m_c5 < fitness[i]:
                    population[i] = mut
                    fitness[i] = m_c5

        # Update best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]

        gen += 1
        if gen % sa_interval == 0 and gen > 0:
            temp *= cooling_rate

        # Termination check
        if gen % 10 == 0:
            elapsed_time = time.time() - start_time
            if elapsed_time >= budget_s:
                break

    return best_h.tolist(), best_c5, n_points
