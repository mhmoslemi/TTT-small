import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_start = np.array(initial_h_values, dtype=float)
    n_start = len(h_start)

    # Use a moderate n_points to balance accuracy and speed
    n_points = 160
    dx = 2.0 / n_points

    # Normalize the initial condition to meet the sum constraint
    h = h_start.astype(float)
    h = np.tile(h, (n_points // n_start) + 1)[:n_points]
    h *= (n_points / 2.0) / np.sum(h)

    # Ensure h is in [0, 1]
    h = np.clip(h, 0.0, 1.0)

    # Enforce the integration constraint
    sum_h = np.sum(h)
    h *= (n_points / 2.0) / sum_h

    n = n_points
    max_iter = 2500
    eps = 1e-6
    best_c5 = float('inf')
    best_h = h.copy()

    t0 = time.time()

    for iter in range(max_iter):
        if time.time() - t0 > budget_s:
            break

        # Compute the cross-correlation
        correlation = np.correlate(h, 1 - h, mode="full") * dx

        # Compute current max correlation (C₅)
        current_c5 = np.max(correlation)

        # Check if improvement
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Identify the maximum correlation shift k
        if len(correlation) == n:
            kmax = 0
        else:
            kmax = np.argmax(correlation[n:])  # Only shifts k = 1 to n-1

        # Compute gradient depending on the shift kmax
        gradient = np.zeros(n)
        for i in range(n):
            j_forward = (i + kmax + 1) % n
            j_back = (i - kmax - 1) % n
            gradient[i] = -(1 - h[j_forward]) + h[j_back]

        # Barzilai-Borwein step size estimation
        if iter > 0:
            diff = h - h_prev
            grad_diff = gradient - gradient_prev
            alpha = np.sum((diff * grad_diff)) / np.sum(grad_diff ** 2 + eps)
        else:
            alpha = 0.02  # Initial step size

        # Update h
        h = h - alpha * gradient

        # Project back onto feasible set [0,1] and enforce integral constraint
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        h *= (n_points / 2.0) / sum_h

        h_prev = h.copy()
        gradient_prev = gradient.copy()

        # Perturbation every 500 iterations if no improvement
        if iter % 500 == 0:
            if current_c5 >= best_c5 + 1e-6:
                mask = np.random.rand(n) < 0.1  # Perturb 10% of entries
                noise = np.random.randn(n) * 0.1
                h_perturbed = h + np.where(mask, noise, 0)
                h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
                sum_perturbed = np.sum(h_perturbed)
                h_perturbed *= (n_points / 2.0) / sum_perturbed
                h = h_perturbed

    # Final evaluation and constraint check
    correlation = np.correlate(best_h, 1 - best_h, mode="full") * dx
    final_c5 = np.max(correlation)
    assert np.allclose(np.sum(best_h), n_points / 2, atol=1e-6)
    assert np.all(best_h >= 0) and np.all(best_h <= 1)

    return (best_h.tolist(), final_c5, n_points)
