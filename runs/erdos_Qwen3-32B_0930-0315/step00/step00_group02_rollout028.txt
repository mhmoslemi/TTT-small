import numpy as np
import time
import cvxpy as cp
from math import floor


def evaluate_C5(h, n_points):
    dx = 2.0 / n_points
    shifted_h = np.concatenate([1 - h, 1 - h, 1 - h])  # Three copies for full convolution
    h_fft = np.fft.fft(h)
    shifted_fft = np.fft.fft(shifted_h)
    cross_corr = np.fft.ifft(h_fft * np.conj(shifted_fft)).real  # Real part for accuracy
    return np.max(cross_corr[n_points:2 * n_points]) * dx


def normalize_h(h, n_points):
    total = np.sum(h)
    excess = total - n_points / 2
    if abs(excess) < 1e-10:
        return h
    idxs = np.argsort(h)
    if excess < 0:  # Need to increase h
        h[idxs] += 1 / (n_points - len(idxs) - 1 + 1)
    else:  # Need to decrease h
        h[idxs] -= 1 / (len(idxs) + 1)
    return np.clip(h, 0, 1)

        h *= n_points / 2

    else:
        h = np.random.uniform(0, 1, n_points)
        h = normalize_h(h, n_points)

    best_h = h.copy()
    best_c5 = evaluate_C5(best_h, n_points)
    patience = 500
    counter = 0

        k_max = np.argmax(cross_corr[n_points:2 * n_points]) + n_points

            idx2 = (j - k_max) % n_points
