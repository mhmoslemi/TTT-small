import numpy as np
import cvxpy as cp
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, n_points=47, solver='ECOS'):
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
    else:
        np.random.seed(seed)
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        h_initial = np.clip(h_initial, 0, 1)

    h = cp.Variable(n_points, nonneg=True)
    objective = cp.Minimize(0)  # CVXOPT still needs this
    constraints = [np.sum(h) == n_points / 2, 0 <= h, h <= 1]
    prob = cp.Problem(objective, constraints)
    max_c5 = float('inf')
    best_h = None
    start_time = time.time()

    while time.time() - start_time < budget_s - 5:
        t = cp.Variable()
        # Form t = max correlation value
        Cs = [cp.sum(cp.multiply(h, 1 - cp.roll(h, k)[n_points + k:] for k in range(1 - n_points, n_points)))]
        for k in range(1, n_points + 1):
            term = cp.sum(h * (1 - cp.roll(h, k)[n_points:])) / n_points * 2.0
            constraints.append(term <= t)

        obj = cp.Minimize(t)
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=solver, verbose=False)
        current_h = h.value
        current_h = np.clip(current_h, 0.0, 1.0)
        current_h = current_h / np.sum(current_h) * (n_points / 2)
        current_c5 = evaluate_erdos_solution([x for x in current_h], n=n_points)
        if current_c5 < max_c5:
            max_c5 = current_c5
            best_h = current_h
        prob.constraints += constraints

    np.random.seed(seed)
    return (best_h.tolist(), max_c5, n_points)
