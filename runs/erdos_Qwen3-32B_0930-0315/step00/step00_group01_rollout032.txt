import numpy as np
import time
import cvxpy as cp
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def symmetric_clip_and_rescale(h, n_points):
    if not np.all(0 <= h) and np.all(h <= 1):
        h = np.clip(h, 0, 1)
    total = np.sum(h)
    normalization = n_points / 2.0 / total if total != 0 else 1.0
    h = h * normalization
    return h

def compute_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr[n_points - 1:2 * n_points - 1])
    return max_corr * dx

def gradient_step(h, step=0.0001):
    n_points = len(h)
    h_plus = np.clip(h + step, 0, 1)
    h_minus = np.clip(h - step, 0, 1)

    g = (compute_c5(h_plus) - compute_c5(h_minus)) / (2 * step)
    grad = np.zeros(h.shape)
    
    for k in range(n_points):
        overlap = []
        for i in range(n_points):
            if i + k < n_points:
                overlap.append(h[i] * (1 - h[i + k]))
            else:
                continue
        sum_overlap = np.sum(overlap)

        overlap_plus = []
        for i in range(n_points):
            if i + k < n_points:
                overlap_plus.append(np.gradient(h[i] * (1 - h[i + k]), h[i])[0])
            else:
                continue
        grad_contribution = np.sum(overlap_plus)

        if k < n_points:
            grad[k] += grad_contribution
        if k > 0:
            grad[n_points - k] += grad_contribution

    return g * grad

def optimize_h(h, max_iter=1000, step=0.001, tolerance=1e-6, n_points=47):
    h = np.copy(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    patience_counter = 0
    patience_limit = 20

    for i in range(max_iter):
        start = time.time()
        grad = gradient_step(h)
        h = symmetric_clip_and_rescale(h - step * grad, n_points)
        current_c5 = compute_c5(h)
        elapsed = time.time() - start

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1

        if patience_counter >= patience_limit:
            break

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    # Start from initial_h_values if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_points = h_initial.size
    else:
        n_points = 47
        h_initial = np.zeros(n_points)
        h_initial[:n_points // 2] = 1.0
        h_initial = symmetric_clip_and_rescale(h_initial, n_points)

    start_time = time.time()

    h_opt, c5_opt = optimize_h(h_initial, n_points=n_points)
    while time.time() - start_time < budget_s:
        new_h = np.random.rand(n_points)
        new_h = symmetric_clip_and_rescale(new_h, n_points)
        new_h, new_c5 = optimize_h(new_h, n_points=n_points)

        if new_c5 < c5_opt:
            h_opt = new_h.copy()
            c5_opt = new_c5

    return h_opt.astype(float), float(c5_opt), int(n_points)
