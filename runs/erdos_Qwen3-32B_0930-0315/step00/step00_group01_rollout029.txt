import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        h = initial_h_values
        n_points = len(h)
    else:
        n_points = 47  # start with the suggestion
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h / h.sum() * (n_points / 2)
    
    start_time = time.time()
    best_c5 = np.finfo(float).max
    best_h = h.copy()
    dx = 2.0 / n_points

    def compute_max_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return (corr * dx).max()

    learning_rate = 0.1
    tolerance = 1e-5
    max_iter = int(1e6)
    patience = 30
    no_improvement = 0

    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        if iter % 500 == 0 or no_improvement >= patience:
            c5 = compute_max_overlap(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improvement = 0
                print(f"Iter {iter}: Better C₅={best_c5:.8f}")
            else:
                no_improvement += 1
                learning_rate *= 0.9  # reduce step to prevent overshooting
            if no_improvement >= patience and learning_rate < 1e-3:
                initial_learning_rate = learning_rate * 2
                learning_rate = initial_learning_rate * (0.5 ** (iter // 300))

        # Compute gradient at worst k
        corr = np.correlate(h, 1 - h, mode='full')
        idx = corr.argmax()
        k = idx - (n_points - 1)
        max_corr = corr[idx]
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) - h[i]  # gradient for h[i] at worst k

        # Distribute learning rate via project gradient descent
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h / h.sum() * (n_points / 2)

    return best_h, best_c5, n_points
