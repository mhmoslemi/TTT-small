import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    dx = 2.0 / 70
    h = initial_h_values.copy()
    n_points = len(h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    t_start = time.time()
    MAX_ITER = 2000
    eta = 0.02
    eta_decay = 0.9
    no_improve_tol = 100
    no_improve_count = 0

    def compute_c5_and_k(h):
        corr = np.correlate(h, 1 - h, mode='full')
        corr *= dx
        return float(np.max(corr)), int(np.argmax(corr))

    def compute_gradient(h, k):
        n = len(h)
        grad = np.zeros(n)
        h_k_lagged = np.roll(h, k) if k <= n // 2 else np.roll(h, -(n - k))
        # For shifting left (negative lag) use zero-padding
        if k > n:
            raise ValueError("Invalid shift k >= n")
        for i in range(n):
            if k <= i < n:
                j = i - k
                grad[i] += (1 - h[j])
        for i in range(k):
            # Enforce that same index isn't double-counted
            j = i + n - k
            grad[j] -= h[i]
        return grad

    def project_h(h):
        x = cp.Variable(n_points)
        prob = cp.Problem(
            cp.Minimize(cp.sum_squares(x - h)),
            [0 <= x, x <= 1, cp.sum(x) == n_points / 2]
        )
        # Speed projection by using a lightweight solver
        prob.solve(solver=cp.ECOS, verbose=False)
        return np.nan_to_num(x.value)

    best_c5, k = compute_c5_and_k(h)
    best_h = h.copy()
    for iter in range(MAX_ITER):
        elapsed = time.time() - t_start
        if elapsed >= budget_s:
            break
        grad = compute_gradient(h, k)
        if np.allclose(grad, np.zeros_like(grad)):
            no_improve_count += 1
            continue

        grad_len = np.sqrt(np.sum(grad * grad))
        if grad_len > 0:
            grad /= grad_len

        h_candidate = h - eta * grad
        h_projected = project_h(h_candidate)
        c5_candidate, k_candidate = compute_c5_and_k(h_projected)

        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_projected.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        k = k_candidate
        h[:] = h_projected
        if no_improve_count > no_improve_tol:
            eta *= eta_decay
            no_improve_count = 0
        if iter % 10 == 0:
            pass  # Could print for monitoring if needed

    return (best_h.tolist(), best_c5, n_points)
