import numpy as np
import cvxpy as cp
from time import time
import math

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    initial_h = np.array(initial_h_values) if "initial_h_values" in globals() else np.zeros(95)
    n_points = len(initial_h)
    dx = 2.0 / n_points

    h = initial_h.copy()
    h /= np.sum(h) * dx  # normalize to [0, 1]
    h *= n_points / (2.0 * np.sum(h))  # ensure sum constraint

    max_iter = 100000
    iter_no_improve = 0
    stop_threshold = 100
    learning_rate = 0.01
    t_start = time()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    for i in range(max_iter):
        if time() - t_start >= budget_s:
            break

        overlaps = np.correlate(h, 1.0 - h, mode='full')
        overlaps *= dx
        k_max = np.argmax(overlaps)

        # Compute gradient based on the worst overlap
        G = np.zeros_like(h)
        G[(k_max - 1):] += (1.0 - h[:-k_max]) if k_max <= n_points else (1.0 - h[k_max - n_points:]) 
        G[:n_points - k_max + 2] -= h[:n_points - k_max + 1]

        # Calculate unit gradient vector
        G_norm = np.linalg.norm(G)
        if G_norm > 0:
            G /= G_norm

        h -= learning_rate * G
        h = np.clip(h, 0, 1)
        h *= n_points / (2.0 * np.sum(h))  # normalize to [0, 1], maintain sum constraint

        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            iter_no_improve = 0
        else:
            iter_no_improve += 1

        if iter_no_improve >= stop_threshold:
            # Decrease learning rate if no improvement for many steps
            learning_rate *= 0.8
            iter_no_improve = 0

    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(overlaps)
