import numpy as np
import numpy.random as npr
import time

def run(seed=42, budget_s=1100):
    n_points = 160
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        h = h[:n_points]
        h = np.resize(h, n_points)  # pad if needed, and repeat if necessary
        h = np.clip(h, 0, 1)
        sum_h = h.sum()
        if sum_h != n_points / 2:
            scale = (n_points / 2) / sum_h
            h *= scale
    else:
        plateau = np.linspace(0.6, 1.0, n_points // 2)
        h = np.concatenate((plateau, plateau[::-1]))
        h = np.clip(h, 0, 1)
        sum_h = h.sum()
        if sum_h != n_points / 2:
            scale = (n_points / 2) / sum_h
            h *= scale

    alpha = 0.01
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    np.random.seed(seed)
    iterations = 0
    min_update = 1.0
    last_improvement = 0
    max_wait = 100

    while iterations < 10**6 and time.time() - start_time < budget_s:
        iterations += 1

        # Compute overlap correlation
        correlate = np.correlate(h, 1 - h, mode="full")
        correlate *= dx
        c5 = np.amax(correlate)
        k_max = np.argmax(correlate)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = 0
        elif last_improvement >= max_wait:
            # Perturb solution if stuck
            noise = npr.normal(0.0, 0.05, size=n_points)
            h += noise
            h = np.clip(h, 0, 1)
            sum_h = h.sum()
            if sum_h != n_points / 2:
                scale = (n_points / 2) / sum_h
                h *= scale
            last_improvement = 0
        else:
            last_improvement += 1

        indices = np.arange(n_points)
        shift_k = indices - k_max
        shift_k = (shift_k + n_points) % n_points

        grad = (1 - h[shift_k]) - h[shift_k]
        h = h - alpha * grad

        h = np.clip(h, 0, 1)
        sum_h = h.sum()
        if sum_h != n_points / 2:
            scale = (n_points / 2) / sum_h
            h *= scale

    return best_h.tolist(), best_c5, n_points
