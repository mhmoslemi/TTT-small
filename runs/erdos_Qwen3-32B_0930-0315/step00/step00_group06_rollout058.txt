import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 256

    # Initialize h if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).astype(np.float64)
        n_points = len(h)
    else:
        h = np.random.rand(n_points)
    h *= n_points / 2 / h.sum()

    best_h, best_c5 = h.copy(), float('inf')

    # Optimization
    lr = 0.1
    while time.time() - start_time < budget_s:
        # Normalize
        h *= (n_points / 2) / h.sum()

        # Compute anti-correlation
        ones_h = 1 - h
        corr = correlate(h, ones_h, mode="full")
        dx = 2.0 / n_points
        overlap = corr * dx
        C5 = overlap.max()

        if C5 < best_c5:
            best_h, best_c5 = h.copy(), C5

        # Find the worst overlap shift
        kmax = np.argmax(overlap[n_points:])

        # Gradient for given kmax
        # For a value at i, it affects h[i] * ones_h[i + k] = h[i] * (1 - h[i + k])
        # Gradient w.r.t h[i] = +ones_h[i + k]
        # Gradient w.r.t h[i + k] = -h[i]

        grad = np.zeros_like(h)

        for idx in range(n_points):
            idx_k = (idx + kmax) % n_points
            grad[idx] += (1 - h[idx_k])
            grad[idx_k] -= h[idx]

        # Normalize gradient for stability
        grad = (grad - grad.mean()) / (np.std(grad) + 1e-8)

        h -= lr * grad
        
        # Project into [0, 1] and normalize again
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / h.sum()

        # Decay learning rate
        lr *= 0.999

    return best_h, best_c5, n_points
