        h_init = initial_h_values.copy()
        h_init_sum = np.sum(h_init)
        if abs(h_init_sum - target_sum) > 1e-8:
            h_init = h_init * target_sum / h_init_sum
    else:
        h_init = np.random.rand(n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init / np.sum(h_init) * target_sum

    # Optimization variables and constraints
    h = cp.Variable(n_points, nonneg=True)
    t = cp.Variable()

    constraints = [
        cp.sum(h) == target_sum,
        h >= 0,
        h <= 1,
    ]

    # Objective: minimize t such that all over lap integrals are <= t
    overlaps = []

    for k in range(n_points):
        overlap = 0.0
        for i in range(n_points - k):
            overlap += h[i] * (1 - h[i + k])
        overlaps.append(overlap)

    overlap_constr = [ov <= t for ov in overlaps]
    constraints += overlap_constr

    objective = cp.Minimize(t)

    # Set up and solve the problem
    problem = cp.Problem(objective, constraints)

    # Run the solver
    while time.time() - start_time < max_time:
