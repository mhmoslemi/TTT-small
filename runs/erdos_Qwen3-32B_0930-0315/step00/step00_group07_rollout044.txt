import numpy as np
import cvxpy as cp
import time
from math import floor

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    # Use initial_h_values as provided
    h = np.array(initial_h_values['h_values'])
    n_points = h.size
    dx = 2.0 / n_points

    # Clamp to [0, 1] and rescale to ensure sum(h) = n_points / 2
    h = np.clip(h, 0, 1)
    h /= np.sum(h)
    h *= n_points / 2
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    num_iterations = 0
    max_iter = 300  # conservative estimate based on 3-5s per iteration on average
    start_time = time.time()
    terminate = False

    while time.time() - start_time < budget_s and num_iterations < max_iter and not terminate:
        num_iterations += 1

        # Evaluate all C_k
        corr = np.correlate(h, 1 - h, mode='full') * dx

        # Max overlap lag index
        max_index = np.argmax(corr)
        k = max_index

        # Shifted h to be aligned with lag k (circular)
        shifted = np.roll(h, -k)

        # CVXPY problem to minimize the maximum correlation term at lag k
        delta_h = cp.Variable(n_points)

        objective = cp.Minimize(cp.sum((h + delta_h) * (1 - (shifted + np.roll(delta_h, -k)))) * dx)

        constraints = [
            h + delta_h >= 0,
            h + delta_h <= 1,
            cp.sum(delta_h) == 0  # maintain integral constraint
        ]

        prob = cp.Problem(objective, constraints)

        # Try solving
        try:
            result = prob.solve(solver=cp.ECOS, verbose=False)
            if result is None:
                continue
        except Exception as e:
            print(f"Skipping iteration due to optimization error: {e}")
            continue

        # Apply the update
        new_h = h + np.clip(delta_h.value, -0.05, 0.05)

        # Project back to [0, 1] and preserve total mass
        new_h = np.clip(new_h, 0, 1)
        new_h *= (n_points / 2) / np.clip(np.sum(new_h), 1e-15, None)

        c5_new = evaluate_erdos_solution(new_h)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = new_h.copy()

        # Early stopping: no improvement for 20 iterations
        if num_iterations % 20 == 0 and abs(best_c5 - evaluate_erdos_solution(h)) < 1e-6:
            terminate = True

        h = new_h.copy()

    return (best_h, best_c5, n_points)
