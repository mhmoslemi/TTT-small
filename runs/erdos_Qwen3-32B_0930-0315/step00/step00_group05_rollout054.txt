import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    # Use provided initial solution
    h = initial_h_values['h_values']
    n_points = len(h)
    dx = 2.0 / n_points

    # Normalize to ensure sum constraint
    h = h.astype(np.float64)
    h *= n_points / 2.0 / np.sum(h)

    best_h = h.copy()
    best_C5 = evaluate_C5(h)

    t_start = time.time()
    t_budget = budget_s
    alpha = 0.1
    best_alpha = alpha

    patience = 0
    max_patience = 3

    while time.time() - t_start < t_budget:
        # Compute correlation and maximum overlap index
        corr = correlate(h, 1 - h, mode='full')
        corr *= dx
        max_idx = np.argmax(corr[n_points-1: 2*n_points-1])

        # Convert from full correlate index back to shift index
        k = max_idx

        # Compute gradient contribution from max overlap
        grad = np.zeros(n_points)
        for i in range(n_points - k):
            grad[i] += (1 - h[i + k])
        for i in range(n_points - k):
            grad[i + k] -= h[i]

        # Gradient descent update with L2 projection
        h -= alpha * grad

        # Project onto feasible region [0,1] and maintain integral constraint
        h_proj = np.clip(h, 0, 1)

        # Rescale to maintain the integral constraint
        sum_h_proj = np.sum(h_proj)
        if sum_h_proj > 0:
            h_proj *= n_points / 2.0 / sum_h_proj

        # Occasionally perturb the solution to escape local minima
        if (time.time() - t_start) % 100 < 0.1 and patience < max_patience:
            perturbation = np.random.default_rng(seed).uniform(-0.05, 0.05, size=n_points)
            h_proj += perturbation
            h_proj = np.clip(h_proj, 0, 1)
            sum_h_proj = np.sum(h_proj)
            if sum_h_proj > 0:
                h_proj *= n_points / 2.0 / sum_h_proj

        h = h_proj

        # Evaluate current C5 and update best solution
        current_C5 = evaluate_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

        # Reduce learning rate if progress stalls
        alpha = max(0.001, alpha * 0.95)

    return (best_h, best_C5, n_points)

def evaluate_C5(h):
    # Compute the overlap integral C5
    n_points = len(h)
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode='full')
    corr *= dx
    max_overlap = np.max(corr[n_points-1: 2*n_points-1])
    return max_overlap
