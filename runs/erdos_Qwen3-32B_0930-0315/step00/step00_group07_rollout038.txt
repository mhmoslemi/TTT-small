import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_values, n_points):
    """Pre-imported helper function."""
    h = np.array(h_values)
    dx = 2.0 / n_points
    corr = correlate(h, 1.0 - h, mode='full')
    corr = corr * dx
    c5 = np.max(corr)
    return c5
