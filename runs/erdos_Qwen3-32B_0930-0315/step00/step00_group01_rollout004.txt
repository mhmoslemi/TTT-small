import numpy as np
import cvxpy as cp
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Get the initial construction
    h_init = initial_h_values()

    n_points = len(h_init)

    start_time = time()

    # Define variables and constraints
    h = cp.Variable(n_points, nonneg=True)

    constraints = [
        h <= 1.0,
        cp.sum(h) == n_points / 2  # integral constraint
    ]

    # Reformulate objective into a convex problem
    t = cp.Variable()

    # Precompute convolution lengths
    max_offset = n_points - 1
    offsets = np.arange(-max_offset, max_offset + 1)
    valid_offsets = offsets[(offsets >= -n_points + 1) & (offsets <= n_points - 1)]

    for k in valid_offsets:
        if k == 0:
            continue
        elif k > 0:
            lhs = h[0 : n_points - k] @ (1 - h[k : n_points])
        else:  # k < 0
            lhs = h[abs(k) : n_points] @ (1 - h[0 : n_points + k])

        constraints.append(lhs <= t * (n_points / 2))

    # Combine objective and constraints
    problem = cp.Problem(cp.Minimize(t), constraints)

    # Solve using ECOS solver
    problem.solve(solver=cp.ECOS, max_iters=100, verbose=False)

    if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_values = np.clip(h.value, 0, 1)
        h_values = h_values / h_values.sum() * (n_points / 2)
        c5_bound = evaluate_erdos_solution(h_values)
    else:
        h_values = h_init
        c5_bound = evaluate_erdos_solution(h_values)

    return h_values.tolist(), c5_bound, n_points
