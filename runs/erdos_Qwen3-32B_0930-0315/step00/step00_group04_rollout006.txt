import numpy as np
import cvxpy as cp
import time

def evaluate_h(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    corr = corr * dx
    return np.max(corr)

def clip_and_rescale(h, dx):
    h = np.clip(h, 0, 1)
    if np.sum(h) * dx < 1e-8:
        return np.where(h > 0, h / (np.sum(h) * dx + 1e-8), 0.0)
    h = h / (np.sum(h) * dx)
    h = np.clip(h, 0, 1)
    return h

def feasible_projection(h, dx):
    h = np.copy(h)
    h = np.clip(h, 0, 1)
    ratio = 1.0 / (np.sum(h) * dx)
    h *= ratio
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=1100, n_points=42, reg_weight=1.0, barrier_penalty=1e2):
    np.random.seed(seed)
    
    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        # Start with simple rectangle of width 1, centered or spread
        h = np.zeros(n_points)
        center = n_points // 2
        half_width = n_points // 4
        h[max(0, center - half_width):min(n_points, center + half_width)] = 0.5
        h = feasible_projection(h, 2.0 / n_points)

    best_h = h.copy()
    best_c5 = evaluate_h(h, 2.0 / n_points)
    
    dx = 2.0 / n_points
    
    t0 = time.time()
    
    μ = barrier_penalty
    
    while time.time() - t0 < budget_s:
        h = h[cp.Parameter(n_points)]
        objective = cp.Maximize(cp.Minimize(cp.sum(cp.pos(cp.conv(h, 1 - h)) * dx) - μ * cp.sum(cp.log(h) + cp.log(1 - h)) + cp.log(1e-8)))
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) * dx == 1,
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        h_val = h.value
        h_val = feasible_projection(h_val, dx)
        current_c5 = evaluate_h(h_val, dx)
        
        if current_c5 < best_c5:
            best_h = h_val.copy()
            best_c5 = current_c5
        
        μ *= 0.95
        if μ < 1e-8:
            break
        
        if time.time() - t0 > budget_s - 10:
            break

    return best_h, best_c5, n_points
