import time
import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import math

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 256  # Target length
    dx = 2.0 / n_points

    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        # Extend to n_points if needed using linear interpolation
        if h.size != n_points:
            x = np.linspace(0, 2, h.size)
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x, h)
            h = h * n_points / np.sum(h)
        h = h.copy()
    else:
        h = np.random.uniform(0, 1, size=n_points)
        h *= n_points / np.sum(h)

    max_iter = 2000
    patience = 100

    for it in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Normalize
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)

        # Ensure sum constraint
        h = h / np.sum(h) * (n_points / 2)

        # Compute C₅: max of cross-correlation
        anti_h = 1 - h
        corr = correlate(h, anti_h, mode='full') * dx
        c5 = np.max(corr)

        # Store best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Find k where overlap is maximum
        k_all = np.arange(-(n_points - 1), n_points)
        overlap = correlate(h, anti_h, mode='full') * dx
        k_max_idx = np.argmax(overlap)
        k_max = k_all[k_max_idx]

        # Phase 1: adjust magnitude under constraint using convex opt.
        # Objective: min equilibrium at current k_max
        # s.t. sum(h) = n_points / 2, 0<=h<=1
        var_h = cp.Variable(n_points)
        objective = cp.Maximize(var_h @ anti_h[:n_points] * dx)
        constraints = [
            cp.sum(var_h) == n_points / 2,
            0 <= var_h,
            var_h <= 1
        ]
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.GLPK, max_iter=100)

        if problem.status != 'optimal':
            h = cp.Parameter(n_points, value=h).value
        else:
            h = var_h.value

    # Final projection
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)

    # Re-calculate C₅ with final h
    anti_h = 1 - h
    corr = correlate(h, anti_h, mode='full') * dx
    final_c5 = np.max(corr)

    return h, final_c5, n_points
