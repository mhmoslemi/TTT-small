import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    # Set random seed if needed
    np.random.seed(seed)
    
    # Set resolution and initialize h
    n_points = 256
    h_init = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    if len(h_init) != n_points:
        h = np.zeros(n_points)
        h[:len(h_init)] = h_init
        h[len(h_init):] = 0.0
    else:
        h = h_init.copy()

    # Normalize to satisfy the constraint ∫ h(x) dx = 1
    h = h / h.sum() * (n_points / 2)

    # Define bounds check and normalization
    def bound_and_normalize(h):
        h = np.clip(h, 0, 1)
        total = h.sum()
        scale_factor = (n_points / 2) / total if total != 0 else 1.0
        h *= scale_factor
        return h

    # Setup variables and optimization params
    dx = 2.0 / n_points
    max_iter = int(1e6)
    learning_rate = 0.1
    start_time = time.time()
    best_h = h.copy()
    best_c5 = np.inf
    perturb_counter = 0

    # Main optimization loop
    for it in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

        # Compute overlaps using cross-correlation
        overlap = correlate(h, 1 - h, mode='full')
        overlap *= dx
        c5 = overlap.max()

        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

        if it % 100 == 0:
            perturb_counter = (perturb_counter + 1) % 10
            if perturb_counter == 0:
                # Add small random perturbation to escape local minima
                h += 0.01 * np.random.randn(*h.shape)
                h = bound_and_normalize(h)

        # Update only based on the worst overlap
        k_max = overlap.argmax()
        grad_step = np.zeros_like(h)
        if n_points <= k_max < 2 * n_points:
            k = k_max - n_points
            grad_step[:n_points - k] = 1 - 2 * h[:n_points - k]
            grad_step[k:] = h[:-k]
        elif 0 <= k_max < n_points:
            k = -k_max
            grad_step[:n_points + k] = 1 - 2 * h[:n_points + k]
            grad_step[k:] = h[:-n_points - k]
        h = h - learning_rate * grad_step
        h = bound_and_normalize(h)

    return (best_h.tolist(), float(best_c5), n_points)
