import numpy as np
import time
import scipy.signal
import copy

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values)
    h_best = initial_h_values.copy()
    best_c5 = evaluate_c5(h_best, n_points)
    tabu_memory = [h_best.copy()]
    tabu_size = 20
    neighborhood_size = 20
    max_neighbors = 20

    while time.time() - start_time < budget_s:
        n_generated = 0
        current_h = h_best.copy()
        new_h = []

        # Perturbation-based neighborhood generation
        candidates = []
        for _ in range(max_neighbors):
            proposal = current_h.copy()
            # Randomly perturb a small number of entries
            num_perturb = np.random.randint(1, 6)
            idxs = np.random.choice(n_points, size=num_perturb, replace=False)
            delta = np.random.uniform(-0.1, 0.1, size=num_perturb)
            proposal[idxs] += delta
            np.clip(proposal, 0, 1, out=proposal)

            # Project back to satisfy the constraint
            total = np.sum(proposal)
            if total > 0:
                proposal *= n_points / (2 * total)

            if not any(np.allclose(proposal, mem) for mem in tabu_memory):
                candidates.append(proposal)
            n_generated += 1

        # Evaluate all candidates
        for candidate in candidates:
            c5 = evaluate_c5(candidate, n_points)
            if c5 < best_c5:
                h_best = candidate.copy()
                best_c5 = c5
                tabu_memory.append(candidate.copy())
                break
            else:
                tabu_memory.append(candidate.copy)

        if len(tabu_memory) > tabu_size:
            tabu_memory.pop(0)

        if n_generated == 0:
            break

    return h_best, best_c5, n_points

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    h_array = np.array(h)
    one_minus_h = 1 - h_array
    corr = scipy.signal.correlate(h_array, one_minus_h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
