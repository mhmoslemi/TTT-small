import numpy as np
import scipy.optimize
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h0 = np.array(initial_h_values, copy=True)
    n_points = len(h0)
    total_mass = n_points / 2.0

    # Constraint: sum(h) = n/2.0
    def constraint_sum(h):
        return np.sum(h) - total_mass

    # Objective: minimize the maximum overlap
    def objective(h):
        h = h.reshape(-1)
        dx = 2.0 / n_points
        # Construct 1 - h
        one_h = 1.0 - h
        # Compute correlation with mode='full' by using convolution
        correlate_float = scipy.signal.correlate(h, one_h, mode="full")
        scaled_correlate = correlate_float * dx
        return np.max(scaled_correlate)

    # Constraint bounds
    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization setup
    start_time = time()

    # Project h to valid domain after each step
    def project(x):
        x = np.clip(x, 0, 1)  # Bounds [0, 1]
        # Normalize to total_mass
        current_sum = np.sum(x)
        if abs(current_sum - total_mass) > 1e-9:
            x = x * (total_mass / current_sum)
        return x

    # Main optimization loop using L-BFGS-B
    res = scipy.optimize.minimize(
        fun=lambda h: objective(h),
        x0=h0,
        method='L-BFGS-B',
        bounds=bounds,
        callback=lambda x: None if (time() - start_time) < budget_s else (print("Budget exceeded, halting."), None),
        options={"ftol": 1e-9, "gtol": 1e-6, "maxiter": int(budget_s * 100)}
    )

    # Last possible attempt to project the final result
    h_opt = project(res.x)

    # Final C5 calculation
    c5_bound = objective(h_opt)

    return (h_opt, c5_bound, n_points)
