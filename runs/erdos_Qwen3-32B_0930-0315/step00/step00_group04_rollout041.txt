import numpy as np
import time

def run(seed=42, budget_s=1100):
    import cvxpy as cp
    import scipy.interpolate

    np.random.seed(seed)
    
    # --- Feasibility repair functions ---
    def repair_feasibility(h, n_points):
        h = np.clip(h, 0, 1)
        h = h / h.sum() * (n_points / 2)
        return h

    def compute_overlap(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    # --- Main functions ---
    def optimize_resolution(h, n_points, max_iter, dx, lr_start=0.01, T_start=1.0):
        best_h = h.copy()
        best_C = compute_overlap(h, dx)
        
        t_start = time.time()
        t_end = t_start + max_iter
        T = T_start
        lr = lr_start
        while time.time() < t_end:
            for _ in range(10):
                # Gradient step
                corr = np.correlate(h, 1 - h, mode='full') * dx
                k_max = np.argmax(corr)
                shift = k_max - (n_points - 1)
                h_shifted = np.roll(h, shift)
                grad = (1 - h_shifted)[:n_points]
                h -= lr * grad
                h = repair_feasibility(h, n_points)
                
                # Evaluate and update
                C = compute_overlap(h, dx)
                if C < best_C:
                    best_h = h.copy()
                    best_C = C
                    T *= 0.99
                    lr *= 0.998
                    
                # Simulated annealing step
                if np.random.rand() < 0.2:
                    h_candidate = h + np.random.normal(0, T, n_points)
                    h_candidate = repair_feasibility(h_candidate, n_points)
                    C_candidate = compute_overlap(h_candidate, dx)
                    if C_candidate < best_C or np.random.rand() < np.exp(-(C_candidate - best_C)/T):
                        h = h_candidate
                        best_h = h_candidate.copy()
                        best_C = min(best_C, C_candidate)
            T *= 0.95
        
        return best_h, best_C

    # --- Resolution doubling function ---
    def double_resolution(h, dx, kind='cubic'):
        x = np.linspace(0, 2, h.shape[0])
        f = scipy.interpolate.CubicSpline(x, h)
        x_fine = np.linspace(0, 2, h.shape[0] * 2)
        h_fine = f(x_fine)
        h_fine = repair_feasibility(h_fine, h_fine.shape[0])
        return h_fine

    # --- Build from parent if available ---
    if 'initial_h_values' in globals():

    else:
        h = np.random.uniform(0, 1, 14)
        h = h / h.sum() * 7  # 14 samples -> should sum to 7

    # --- Main optimization loop with timing control ---
    best_h = None
    best_C = float('inf')
    
    n_points_list = [14, 28, 56]
    progress_time = 0
    
    for i, n in enumerate(n_points_list):
        start_time = time.time()
        
        if n == n_points_list[0]:
            # First resolution round
            pass
        else:
            # Interpolate solution from previous resolution
            h = double_resolution(h, 2.0 / n)
        
        dx = 2.0 / n
        h = repair_feasibility(h, n)
        
        # Allocate more time for final refinement
        time_frac = 0.7 if i < len(n_points_list) - 1 else 0.3
        alloc_time = budget_s * time_frac
        
        h, C = optimize_resolution(h, n, int(alloc_time), dx)
        
        if C < best_C:
            best_C = C
            best_h = h.copy()
            progress_time = time.time() - start_time
        
        # Guard against time budget exhaustion
        if progress_time >= budget_s - 30:
            break
    
    # Ensure we return within the budget
    return best_h.tolist(), best_C, len(best_h)
