import time

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()

    n_points = 160

    if initial is not None:

        assert np.allclose(h.sum(), n_points / 2), "Invalid initial h"
    else:

        h = np.clip(h, 0, 1)
        h *= n_points / 2 / h.sum() if np.abs(h.sum()) > 1e-10 else 1.0

    def evaluate_c5_vec(h):
        c_hs = np.correlate(h, 1 - h, mode="full")
        return c_hs * dx

    last_c5 = np.inf
    best_h = h.copy()
    best_c5 = np.inf
    t_last = time.time()
    n_iter = 0
    while time.time() - t0 < budget_s:
        n_iter += 1

        # Correlation/overlap evaluation
        corr = np.correlate(h, 1 - h, mode="full")
        corr_values_right = corr[n_points:]  # Positive shifts only
        max_idx = np.argmax(corr_values_right)
        k_max = max_idx + 1  # +1 because of how the full mode index works

        hmirrored = h.take((n_points - k_max + np.arange(n_points)) % n_points)
        grad = 1 - hmirrored - h

        # Barzilai-Borwein step size
        if n_iter > 1:
            delta_h = h - prev_h
            delta_g = grad - prev_grad
            if np.dot(delta_h, delta_g) != 0:
                alpha = np.abs(np.dot(delta_h, delta_h) / np.dot(delta_h, delta_g))
            else:
                alpha = 1e-3
        else:
            alpha = 1e-2  # Initial step size

        # Update h using gradient descent
        h = h - alpha * grad

        # Projection 1: Clamp to [0, 1]
        h = np.clip(h, 0, 1)

        # Projection 2: Ensure the sum constraint
        total = h.sum()
        desired = n_points / 2
        if abs(total - desired) > 1e-3:
            h += (desired - total) / n_points

        # Fallback to previous best if no improvement for 200 iterations
        if n_iter % 200 == 0 and abs(corr[n_points + k_max - 1] * dx) > best_c5:
            h = best_h.copy()

        # Evaluate and trace
        c5_candidate = np.max(evaluate_c5_vec(h))
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h.copy()
            last_c5 = c5_candidate
            t_last = time.time()

        # Periodic perturbation to escape local optima
        if n_iter % 500 == 0:
            perturb = np.random.uniform(-0.5, 0.5, size=n_points)
            perturb[np.abs(perturb) > 1] = np.sign(perturb[np.abs(perturb) > 1])
            h = best_h + perturb
            h = np.clip(h, 0, 1)
            h *= n_points / 2 / h.sum() if np.abs(h.sum()) > 1e-7 else 1.0

        # Save gradient state for Barzilai-Borwein in next step
        prev_h = h.copy()
        prev_grad = grad

        # Terminate if no improvement in a while
        if time.time() - t_last > 300:
            break

    return best_h, best_c5, n_points
