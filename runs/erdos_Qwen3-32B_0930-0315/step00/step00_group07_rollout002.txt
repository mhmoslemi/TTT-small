import numpy as np
import time

def evaluate_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all (h(x) * (1 - h(x + k))) overlap integrals
    hwrap = np.concatenate([h, h, h])  # extend for wraparound
    overlap = np.array([np.sum(h * (1 - hwrap[n_points + k : n_points + k + n_points])) * dx for k in range(2 * n_points)])
    c5 = np.max(overlap)
    return c5

def optimize_h(initial_h, budget_s, max_iter=1000):
    start_time = time.time()
    h = np.copy(initial_h).astype(float)
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = np.copy(h)
    best_c5 = evaluate_c5(h)

    for _iter in range(max_iter):
        # Evaluate current C5
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = np.copy(h)

        # Early stopping
        if time.time() - start_time >= budget_s:
            print("Time budget exceeded")
            break

        # Gradient step
        hwrap = np.concatenate([h, h, h])
        overlap = np.array([np.sum(h * (1 - hwrap[n_points + k : n_points + k + n_points])) for k in range(2 * n_points)])
        max_lag = np.argmax(overlap)
        grad = -hwrap[n_points + max_lag : n_points + max_lag + n_points]

        # Update h with step size
        step_size = 0.01
        h_new = h + step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)  # Normalize to maintain integral constraint

        h = h_new

    return best_h, best_c5
