import numpy as np
import scipy.optimize
import time

def correlate_bounds(h, dx):
    full_corr = np.correlate(h, 1 - h, mode="full")
    return full_corr * dx

def compute_C5(h, dx):
    full_corr = np.correlate(h, 1 - h, mode="full")
    C5 = np.max(full_corr * dx)
    return C5

def objective_with_penalty(h, n_points, target_sum, dx, penalty_weight=1e4):
    h = h.copy()
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    penalty = penalty_weight * (current_sum - target_sum) ** 2
    C5 = compute_C5(h, dx)
    return C5 + penalty

def gradient_with_penalty(h, n_points, target_sum, dx, k_max, penalty_weight=1e4):
    h = h.copy()
    gradient = np.zeros_like(h)
    # Gradient for max overlap at lag k_max
    for i in range(n_points - k_max):
        gradient[i] -= (1 - h[i + k_max]) * dx
    for j in range(k_max, n_points):
        gradient[j] -= h[j - k_max] * dx
    # Gradient for penalty term
    penalty_grad = 2 * penalty_weight * (np.sum(h) - target_sum)
    gradient += penalty_grad
    return gradient

def ensure_feasibility(h, target_sum):
    h = np.clip(h, 0, 1)
    factor = target_sum / np.sum(h)
    h *= factor
    return h
