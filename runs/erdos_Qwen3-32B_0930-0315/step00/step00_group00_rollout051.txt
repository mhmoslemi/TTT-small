import numpy as np
import time
import cvxpy as cp
import scipy.optimize as so


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 45  # default initial n (max to fit in 1k samples budget)
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Build initial binary-like h with area 1
        h = np.zeros(n_points)
        h[: n_points // 2] = 1.0
        h[n_points // 2] = n_points % 2
        h = h / h.sum() * n_points / 2

    assert np.all(h >= 0) and np.all(h <= 1)
    assert np.isclose(h.sum(), n_points / 2)

    best_h = h.copy()
    best_c5 = float('inf')

    max_iter = 10000
    lr = 0.01
    n_update_print = 200

    for iter in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        corr = np.correlate(h, 1 - h, mode='full')
        corr = corr[n_points - 1 : 2 * n_points - 1]  # only lags 0 to n-1
        corr *= dx
        c5 = np.max(corr)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        if iter % n_update_print == 0:
            print(f"{iter}: C₅ = {best_c5:.6f}")

        max_idx = np.argmax(corr)
        grad = np.zeros_like(h)

        # compute gradient for the max lag contributing to C₅
        for i in range(n_points - max_idx):
            grad[i] += 1 - h[i + max_idx]
            grad[i + max_idx] -= 1 - h[i + max_idx]

        # normalize gradient
        grad = grad / max(1, np.max(np.abs(grad)))

        # apply update
        h -= lr * grad

        # project to constraints: sum(h) = n/2, 0 <= h <=1
        h[h < 0] = 0
        h[h > 1] = 1
        old_sum = h.sum()
        delta = (n_points / 2 - h.sum()) / old_sum
        h += delta * h * (h > 0)
        h[h < 0] = 0
        h = h / h.sum() * (n_points / 2)

    return (best_h.tolist(), best_c5, int(n_points))
