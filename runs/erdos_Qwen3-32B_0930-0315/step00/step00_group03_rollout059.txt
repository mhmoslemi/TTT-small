import numpy as np
import time
import random
import cvxpy as cp

def evaluate_population(population, dx):
    scores = []
    for h in population:
        corr = np.correlate(h, 1 - h, mode='full')
        score = np.max(corr) * dx
        scores.append(score)
    return np.array(scores)

def tournament_selection(population, scores, tournament_size=5):
    indices = np.arange(len(population))
    selected_indices = []
    for _ in range(len(population)):
        tournament_indices = np.random.choice(indices, tournament_size, replace=False)
        selected_indices.append(tournament_indices[np.argmin(scores[tournament_indices])])
    return population[np.array(selected_indices)]

def crossover(parents, dx, n_points):
    offspring = []
    for i in range(0, len(parents), 2):
        if i+1 >= len(parents):
            break
        parent1, parent2 = parents[i], parents[i+1]
        crossover_point = np.random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        
        # Normalize to maintain integral requirement (np.sum(h) == n_points / 2)
        for child in [child1, child2]:
            s = (n_points / 2) / np.sum(child)
            child = np.clip(child * s, 0, 1)
            child = adaptive_feasibility_repair(child, n_points)
            offspring.append(child)
    return np.array(offspring)

def adaptive_feasibility_repair(child, n_points):
    # If we're close to constraint violation, redistribute excess to satisfy sum constraint
    ideal_sum = n_points / 2
    current_sum = np.sum(child)
    if abs(current_sum - ideal_sum) > 1e-4:
        error = (ideal_sum - current_sum)
        # Use the largest N values to redistribute error
        N = max(5, int(n_points / 20))
        indices = np.argpartition(child, -N)[-N:]
        for i in indices:
            child[i] = np.clip(child[i] + error / N, 0, 1)
        child = np.clip(child, 0, 1)
    return child

def mutate(h, mutation_rate=0.15, noise_range=0.1, n_points=140):
    h = h.copy()
    num_mutate = int(mutation_rate * n_points)
    mutate_indices = np.random.choice(n_points, num_mutate, replace=False)
    h[mutate_indices] += np.random.normal(0, noise_range, size=len(mutate_indices))
    h = np.clip(h, 0, 1)
    h = adaptive_feasibility_repair(h, n_points)
    return h

def run(seed=42, budget_s=1100, n_points=140, population_size=50):
    np.random.seed(seed)
    random.seed(seed)

    # Assume n_points=140 for compatibility with genetic operators and initial_h_values
    if 'initial_h_values' in globals():
        if initial_h_values is not None:
            # Rescale n_points if needed and use it as seed
            if len(initial_h_values) != n_points:
                initial_h_values = np.interp(
                    np.linspace(0, 2, n_points),
                    np.linspace(0, 2, len(initial_h_values)),
                    initial_h_values
                )
            # Normalize initial solution to satisfy constraint
            s = (n_points / 2) / np.sum(initial_h_values)
            h0 = np.clip(initial_h_values * s, 0, 1)
            h0 = adaptive_feasibility_repair(h0, n_points)
        else:
            # Start from random feasible vector
            h0 = np.random.rand(n_points)
            h0 = adaptive_feasibility_repair(h0, n_points)
    else:
        # Start from random feasible vector
        h0 = np.random.rand(n_points)
        h0 = adaptive_feasibility_repair(h0, n_points)

    population = []
    population.append(h0)
    # Add nearby solutions from initial to encourage exploration
    num_augment = 49  # populate to size 50
    for _ in range(num_augment):
        h = mutate(h0, mutation_rate=0.05)  # conservative mutation
        population.append(h)

    dx = 2.0 / n_points
    best_score = np.inf
    best_h = None

    start_time = time.time()
    while time.time() - start_time < budget_s - 0.1:
        # Evaluate
        scores = evaluate_population(population, dx)
        min_idx = np.argmin(scores)

        # Update best
        if scores[min_idx] < best_score:
            best_score = scores[min_idx]
            best_h = population[min_idx]

        # Selection
        parents = tournament_selection(population, scores)

        # Crossover
        offspring = crossover(parents, dx, n_points)

        # Mutation
        mutated_offspring = [mutate(ind, mutation_rate=0.15) for ind in offspring]

        # Replacement
        population = np.vstack([population, mutated_offspring])
        # Only keep the best to limit size growth
        scores = evaluate_population(population, dx)
        elite_indices = np.argsort(scores)[:population_size]
        population = population[elite_indices]

    c5_bound = best_score
    return best_h, c5_bound, n_points
