```python
import numpy as np
import cvxpy as cp
import time
from math import floor
from scipy.optimize import minimize

# Preloaded into the environment
# from erdos_helpers import evaluate_erdos_solution, initial_h_values

def run(seed=42, budget_s=1100):
    start_time = time.time()
    total_time = budget_s

    if not hasattr(evaluate_erdos_solution, '__call__'):
        raise RuntimeError("Assessment function `evaluate_erdos_solution` must be available.")

    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.zeros(45)
    n_points = len(initial_h)
    required_sum = n_points / 2
    dx = 2.0 / n_points

    if abs(np.sum(initial_h) - required_sum) > 1e-6:
        initial_h = np.clip(initial_h, 0, 1)
        _, h_opt = cp.Problem(cp.Minimize(cp.sum_squares(cp.Variable(n_points) - initial_h)),
                              [cp.sum(cp.Variable(n_points)) == required_sum,
                               cp.Variable(n_points) >= 0,
                               cp.Variable(n_points) <= 1]).solve(solver=cp.SCIPY)
        initial_h = h_opt

    def project_simplex(h):
        h = np.clip(h, 0, 1)
        diff = np.sum(h) - required_sum
        if abs(diff) < 1e-8:
            return h
        h_sorted = np.sort(h)[::-1]
        cumsum = np.cumsum(h_sorted)
        rho = np.where(cumsum - 1 >= diff)[0][0]
        theta = (cumsum[rho] - 1) / (rho + 1)
        h = np.maximum(h - theta, 0)
        return h / np.sum(h) * required_sum

    def get_overlap(h):
        h = h.ravel()
        corr = np.correlate(h, 1.0 - h, mode='full')
        lag0 = n_points
        bounds = [int(lag0 - 4), int(lag0 + 5)]
        sliced_corr = corr[bounds[0]:bounds[1]] * dx
        return np.max(sliced_corr), sliced_corr

    best_h = initial_h.copy()
    best_c5, _ = get_overlap(best_h)
    h = initial_h.copy()

    # Gradient Descent parameters
    max_iter = 3000
    it = 0
    lr = 0.005
    decay_rate = 0.998
    tolerance = 1e-4
    best_lr = lr

    while time.time() - start_time < total_time and it < max_iter:
        overlap, corr = get_overlap(h)
        lag_idx = np.argmax(corr[bounds[0]:bounds[1]]) + bounds[0]
        grad = np.zeros_like(h)

        for i in range(n_points):
            if 0 <= i + lag_idx < n_points:
                grad[i] += h[i + lag_idx] - 1.0
            if 0 <= i - lag_idx < n_points:
                grad[i] += h[i - lag_idx]

        grad *= dx
        h = project_simplex(h - lr * grad)
        new_overlap, _ = get_overlap(h)

        if new_overlap < overlap:
            # successful step
            best_lr = lr
            if new_overlap < best_c5:
                best_c5 = new_overlap
                best_h[:] = h[:]
                print(f"Tick: {it}, best_c5: {best_c5:.7f}")
        else:
            # step failed, reset
            h[:] = best_h[:]
            lr = max(lr * 0.4, 0.5 * best_lr)

        lr = max(lr * decay_rate, 1e-4)
        it += 1

        if it % 500 == 0:
            if time.time() - start_time >= total_time:
                break

    final_c5, _ = get_overlap(best_h)
    return best_h.tolist(), final_c5, n_points
```
