```python
import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate as scipy_correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    dx = 2.0 / 70
    n_points = 70

    def project_h(h):
        h = np.clip(h, 0, 1)
        h *= n_points / 2 / h.sum()
        return h

    def compute_c5(h):
        dx = 2.0 / h.size
        h_expanded = np.zeros_like(h)
        h_expanded[:h.size] = h
        one_minus_h = 1.0 - h
        corr = scipy_correlate(h_expanded, one_minus_h, mode='full')
        max_overlap = np.max(corr * dx)
        return max_overlap

    # Use the provided initialization if available
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = project_h(h)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    start_time = time.time()

    while time.time() - start_time < budget_s:
        h = project_h(h)

        var_h = cp.Variable(n_points)
        obj = cp.Minimize(compute_c5(var_h))
        constraints = [0 <= var_h, var_h <= 1, cp.sum(var_h) == n_points / 2]
        problem = cp.Problem(obj, constraints)

        try:
            problem.solve(solver=cp.CVXOPT, kktsolver="robust")
            h_update = np.array(var_h.value).flatten()
            h = h_update
        except Exception:
            break

        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_points
```
