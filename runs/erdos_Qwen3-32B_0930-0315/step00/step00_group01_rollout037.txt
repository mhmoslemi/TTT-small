import cvxpy as cp
import numpy as np
import time
from numpy import corrcoef, concatenate


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the given initial_h_values (n_points=47 by default)
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        # If initial_h_values not provided, use custom starting distribution
        n_points = 2 * 47  # double resolution to have better search grid
        h0 = np.zeros(n_points)
        idx = np.arange(n_points)
        peak = int(n_points / 2)
        widths = [3, 7, 11]
        for w in widths:
            h0 += np.exp(-0.5 * ((idx - peak) / w) ** 2)
        h0 = h0 / h0.sum() * n_points / 2  # normalize to satisfy ∫h dx = 1
        h0 = np.clip(h0, 0, 1)
    
    h0 = h0.astype(np.float64)

    best_c5 = np.inf
    best_h = h0.copy()

    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Setup optimization problem
    h = cp.Variable(n_points, nonneg=True)
    t = cp.Variable()

    constraints = [
        h <= 1,
        cp.sum(h) == n_points / 2,
    ]

    # Build a grid of all possible k-shifted overlap scored
    corr_k_all = []
    for k in range(n_points + 1):  # for lags 0 to n_points
        # Construct shifted (1 - h) vector aligned with h
        shifted = np.roll(1 - h, -k)
        overlap = cp.sum(h * cp.index(1 - h, range(k, n_points))) * dx
        constraints += [overlap <= t]
    
    # Objective is to minimize the max overlap score
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.ECOS, verbose=False)

    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_opt = h.value
        h_opt[~np.isfinite(h_opt)] = 0
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt / cp.sum(h_opt).value * (n_points / 2)
        h_opt = np.clip(h_opt, 0, 1)

        # Recompute the max overlap after solving
        h_vec = cp.Parameter(n_points)
        x_autocorr_corr = np.correlate(h_opt, 1 - h_opt, mode='full')
        corr_vals = x_autocorr_corr[np.arange(1, n_points) - 0]
        C5 = np.max(corr_vals * dx)

        if C5 < best_c5:
            best_c5 = C5
            best_h = h_opt

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (best_h, best_c5, n_points)

    # Final adjustments to meet the point constraint precisely
    h_sum = best_h.sum()
    adjust = (n_points / 2 - h_sum) / n_points
    best_h = np.clip(best_h + adjust, 0, 1)

    # Final re-evaluation of the C5 bound
    h_vec = best_h
    h1 = np.concatenate([h_vec, h_vec[0]])
    x_autocorr_corr = np.correlate(h_vec, 1 - h1, mode='full')
    corr_vals = x_autocorr_corr[np.arange(1, n_points) - 0]
    best_c5 = np.max(corr_vals * dx)

    return (best_h, best_c5, n_points)
