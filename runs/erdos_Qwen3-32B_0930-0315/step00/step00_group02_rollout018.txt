import numpy as np
from scipy.signal import correlate
import time
import evaluate_erdos_solution
from scipy.optimize import minimize
from cvxpy import Variable, Problem, Minimize, Max, sum as cp_sum

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    h = h.reshape(-1)
    rev_h = 1 - h
    corr = correlate(h, rev_h, mode="full")
    c5 = np.max(corr * dx)
    return c5

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cumsum = np.cumsum(u) - 1.0
    rho = np.sum(cumsum > 0)
    threshold = cumsum[rho - 1] / rho if rho > 0 else 0.0
    x = np.maximum(v - threshold, 0.0)
    return x

def gradient(h, dx):
    n = len(h)
    h = h.reshape(-1)
    rev_h = 1 - h
    corr = correlate(h, rev_h, mode="full")
    max_idx = np.argmax(corr)
    k_star = max_idx - n + 1
    grad = np.zeros(n)
    for i in range(n):
        if 0 <= i + k_star < n:
            grad[i] += (1 - h[i + k_star]) * dx
        if 0 <= i - k_star < n:
            grad[i] -= h[i - k_star] * dx
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    n = 160
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        # Use the parent construction if available
        h = project_simplex(np.copy(initial_h_values))
    else:
        # Default starting vector: plateau shape
        h = np.zeros(n)
        mid = n // 2
        plateau_width = n // 2
        h[mid - plateau_width // 2 : mid + plateau_width // 2] = 1 / dx

    # Discretization parameters
    dx = 2.0 / n
    best_c5 = evaluate_c5(h)
    best_h = h.copy()

    start_time = time.time()

    # Phase 1: Gradient descent with line search
    max_iter = 100
    step_size = 0.01
    patience = 5

    for _ in range(max_iter):
        # Check time budget
        if time.time() - start_time > budget_s:
            break

        # Compute gradient
        g = gradient(h, dx)

        # Line search
        alpha = step_size
        new_h = h - alpha * g
        new_h = project_simplex(new_h)

        # Projection to the simplex maintains the constraint sum(h) = n / 2
        if abs(np.sum(new_h) - n / 2) > 1e-6:
            new_h = (new_h / np.sum(new_h)) * (n / 2)
            new_h = np.clip(new_h, 0, 1)

        # Check improvement
        new_c5 = evaluate_c5(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        else:
            step_size *= 0.95  # Reduce step size if no improvement
        h = new_h.copy()

    # Fallback solution in case early stopping
    if abs(np.sum(best_h) - n / 2) > 1e-6:
        best_h = (best_h / np.sum(best_h)) * (n / 2)
        best_h = np.clip(best_h, 0, 1)

    return best_h, best_c5, n
