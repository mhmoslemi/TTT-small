import numpy as np
import cvxpy as cp
import time
from evaluate_erdos_solution import evaluate_erdos_solution  # Pre-imported helper

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use the pre-defined initial_h_values provided by the harness
    h0 = initial_h_values
    n_points = len(h0)
    dx = 2.0 / n_points

    # Ensure initial h0 is feasible
    h0_clip = np.clip(h0, 0, 1)
    h0_sum = np.sum(h0_clip)
    if h0_sum > 1e-10:  # to avoid division by zero
        h0_scaled = h0_clip * (n_points / 2) / h0_sum
    else:
        h0_scaled = np.zeros_like(h0_clip)

    h = cp.Variable(n_points, pos=True)
    h_constraints = [
        cp.sum(h) == n_points / 2,
        h <= 1.0 - 1e-8  # Strictly less than 1 to avoid log(0)
    ]

    obj_record = []
    best_h = h0_scaled
    best_c5 = float('inf')

    mu = 1e6  # High initial barrier parameter
    for iter_count in range(1000):
        if time.time() - start_time > budget_s:
            break

        print(f"Iteration {iter_count}, mu={mu}")  # Keepers for development

        # Create approximate barrier objective
        # We'll minimize the max overlap + barrier terms
        h_eval = h.value if h.value is not None else h0_scaled

        # Compute function values for one h
        hh = h_eval
        hh = hh.reshape(-1)
        ones_minus_h = 1 - hh
        corr = np.correlate(hh, ones_minus_h, mode='full') * dx
        c5 = np.max(corr)

        # Add barrier terms for strict interior feasibility
        barrier = mu * cp.sum(cp.log(h)) + mu * cp.sum(cp.log(1 - h)) + mu * cp.log(cp.sum(h) - n_points / 2 + 1e-8)

        objective = cp.Minimize(c5 + barrier)

        problem = cp.Problem(objective, h_constraints)

        try:
            problem.solve(solver=cp.CVXOPT, max_iters=100)
        except Exception as e:
            print(f'Solver failed at iter {iter_count}, try again with lower mu')
            mu *= 0.7
            continue

        if h.value is None:
            print(f'No solution at iter {iter_count}')
            continue

        h_new = h.value
        h_new = np.clip(h_new, 0 + 1e-8, 1 - 1e-8)

        # Ensure normalization
        h_new_sum = np.sum(h_new)
        if abs(h_new_sum - n_points / 2) > 1e-6:
            h_new = h_new * (n_points / 2) / h_new_sum

        # Evaluate
        new_corr = np.correlate(h_new, 1 - h_new, mode='full') * dx
        new_c5 = np.max(new_corr)

        # Update if improved
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new

        mu *= 0.999  # Slow decrease to ensure convergence

    # Final check to ensure we are under the budget
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    best_h = best_h * (n_points / 2) / best_h_sum

    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
