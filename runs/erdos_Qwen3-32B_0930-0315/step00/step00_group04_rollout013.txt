import numpy as np
import time
from numpy.fft import rfft, irfft
import cvxpy as cp
import random
from scipy.interpolate import CubicSpline

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h = np.clip(h, 0, 1)
    h = (h * dx) / h.sum() * n_points / 2  # Ensure exact integral = 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr * dx), h

def rescale_h(h):
    n = h.size
    total = h.sum() * (2.0 / n)
    h = np.clip(h, 0, 1)
    h = h * (1.0 / total) * (n / 2) * (2.0 / n)
    return h

def feasible_repair(h):
    h = np.clip(h, 0, 1)
    h = h / h.sum() * (len(h) / 2) * (2.0 / len(h))
    return h

def resample_cubic(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    cs = CubicSpline(x_old, h, bc_type='natural')
    h_new = cs(x_new)
    return feasible_repair(h_new)

def initialize_population(n, pop_size=20):
    population = []
    for _ in range(pop_size):
        h = np.random.rand(n)
        h = feasible_repair(h)
        c5, h = compute_c5(h, n)
        population.append((c5, h))
    population.sort()
    return population

def tournament_selection(population, tournament_size=5):
    selected = random.sample(population, tournament_size)
    selected.sort()
    return selected[0]

def crossover(parent1, parent2, crossover_point=None):
    n = len(parent1)
    h1, h2 = parent1.copy(), parent2.copy()
    if crossover_point is None:
        crossover_point = random.randint(1, n - 1)
    h1[crossover_point:], h2[crossover_point:] = h2[crossover_point:], h1[crossover_point:]
    return h1, h2

def mutate(h, mutation_rate=0.2):
    h_mutated = h.copy()
    mutation_mask = np.random.rand(*h.shape) < mutation_rate
    h_mutated[mutation_mask] += (np.random.rand(*h.shape[mutation_mask.shape]) - 0.5) * 0.2
    return feasible_repair(h_mutated)

def compute_gradient(h, best_k):
    n = len(h)
    h_centered = h[n - best_k:] if best_k < n else h[:best_k - n + 1]
    gradient = (1.0 - h_centered)
    full_grad = np.zeros(n)
    full_grad[n - best_k:][:] = gradient if best_k < n else full_grad[:best_k - n + 1][:] = gradient
    return full_grad

def optimize_with_gd(h, n_points):
    best_c5, h = compute_c5(h, n_points)
    for _ in range(200):
        corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n_points)
        best_k = np.argmax(corr)
        grad = compute_gradient(h, best_k)
        h = h - 0.005 * grad
        h = feasible_repair(h)
        new_c5, _ = compute_c5(h, n_points)
        if abs(new_c5 - best_c5) < 1e-6:
            break
        best_c5 = new_c5
    return best_c5, h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_n = 14
    if 'initial_h_values' in globals() and len(initial_h_values) >= initial_n:
        h0 = feasible_repair(initial_h_values[:initial_n])
    else:
        n_points = initial_n
        h0 = feasible_repair(np.random.rand(n_points))

    time_start = time.time()
    pop_size = 20
    mutation_rate = 0.2
    ga_time_percent = 0.6
    total_population = []

    n_points = initial_n
    h = h0.copy()
    population = initialize_population(n_points, pop_size)

    final_h = None
    final_c5 = float('inf')

    while time.time() - time_start < budget_s * ga_time_percent:
        new_population = []
        for _ in range(pop_size):
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            h1, h2 = crossover(parent1[1], parent2[1])
            h1 = feasible_repair(h1)
            h2 = feasible_repair(h2)
            h1 = mutate(h1, mutation_rate)
            h2 = mutate(h2, mutation_rate)
            for h_child in [h1, h2]:
                c5, h_child = compute_c5(h_child, n_points)
                new_population.append((c5, h_child))
        population += new_population
        population.sort()
        population = population[:pop_size]

        best_c5 = population[0][0]
        best_h = population[0][1]
        if best_c5 < final_c5:
            final_c5 = best_c5
            final_h = best_h.copy()

        if time.time() - time_start > budget_s * 0.95:
            break

    final_c5, final_h = optimize_with_gd(final_h, n_points)
    total_population.append((final_c5, final_h))

    while time.time() - time_start < budget_s * 0.99:
        n_points *= 2
        best_entry = min(total_population)
        h = best_entry[1]
        h = resample_cubic(h, n_points)
        c5, h = compute_c5(h, n_points)
        population = initialize_population(n_points, pop_size)

        while time.time() - time_start < budget_s:
            new_population = []
            for _ in range(pop_size):
                parent1 = tournament_selection(population)
                parent2 = tournament_selection(population)
                h1, h2 = crossover(parent1[1], parent2[1])
                h1 = feasible_repair(h1)
                h2 = feasible_repair(h2)
                h1 = mutate(h1, mutation_rate)
                h2 = mutate(h2, mutation_rate)
                for h_child in [h1, h2]:
                    c5_child, h_child = compute_c5(h_child, n_points)
                    new_population.append((c5_child, h_child))
            population += new_population
            population.sort()
            population = population[:pop_size]

            best_c5 = population[0][0]
            best_h = population[0][1]
            if best_c5 < final_c5:
                final_c5 = best_c5
                final_h = best_h.copy()

            final_c5, final_h = optimize_with_gd(best_h, n_points)
            break

    final_c5, final_h = compute_c5(final_h, len(final_h))
    return (final_h, final_c5, len(final_h))
