import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64, copy=False)
    correl = np.correlate(h, 1 - h, mode='full')
    correl = correl[n_points - 1:] * dx
    c5 = correl.max()
    # Validate the solution
    assert np.allclose(h.sum(), n_points / 2.0, atol=1e-8), ("h mass must match exactly", h.sum(), n_points / 2.0, h.sum() - n_points / 2.0)
    return c5

def run(seed=42, budget_s=1100):
    t_start = time.monotonic()

    # Sample count (initial_h_values contains 80 samples)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64).copy()

    # Precompute all overlap values
    def compute_Sk(h):
        correl = np.correlate(h, 1.0 - h, mode='full') * dx
        return correl[n_points - 1:]

    def update_direction(h, k):
        n = len(h)
        grad = np.zeros_like(h)
        idx1 = (np.arange(n) + k) % n
        idx2 = (np.arange(n) - k) % n
        grad += (1.0 - h[idx1])  # from forward direction
        grad += (1.0 - h[idx2])  # from reverse (mirrored) direction
        grad *= dx  # scale by dx
        return grad

    def feasible_update(h, grad):
        n = len(h)
        delta = cp.Variable(n)
        objective = cp.Minimize(cp.sum(grad * delta))
        constraints = [
            h + delta >= 0.0,
            h + delta <= 1.0,
            cp.sum(delta) == 0.0,
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status != cp.OPTIMAL:
            return None
        return delta.value

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    t0 = time.monotonic()
    while time.monotonic() - t_start < budget_s:
        S = compute_Sk(h)
        k = np.argmax(S)

        grad = update_direction(h, k)
        delta = feasible_update(h, grad)
        if delta is None:
            break

        step = 1.0
        while True:
            h_candidate = h - step * delta
            if np.all(0.0 <= h_candidate) and np.all(h_candidate <= 1.0):
                break
            step *= 0.5
        h[:] = h_candidate
        h[:] = np.clip(h, 0.0, 1.0)
        h[:] = h / h.sum() * (n_points / 2.0)

        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h[:] = h[:]

    return best_h.tolist(), best_c5, n_points
