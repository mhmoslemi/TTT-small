import numpy as np
import time
import copy
from scipy.optimize import minimize
import cvxpy as cp
import math

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    max_overlap = 0
    h_vec = h if n_points > len(h) else h
    correlations = np.correlate(h_vec, 1 - h_vec, mode='full')
    C5 = np.max(correlations) * dx
    return C5

def run(seed=42, budget_s=1100):
    start_time = time.time()
    if 'initial_h_values' in globals():
        h = np.array(globals()['initial_h_values'])
        n_points = len(h)
    else:
        n_points = 47
        h = np.zeros(n_points)
        h[:n_points // 2] = 1.0
    h[h > 1.0] = 1.0
    h[h < 0.0] = 0.0
    h_sum = np.sum(h)
    h *= n_points / (2.0 * h_sum)

    np.random.seed(seed)
    best_h = copy.deepcopy(h)
    best_c5 = evaluate_erdos_solution(best_h, n_points)

    def compute_overlap_gradient(h, k):
        n = len(h)
        dx = 2.0 / n
        overlap = 0
        gradient = np.zeros(n)
        for i in range(n - k):
            overlap += h[i] * (1 - h[i + k])
            gradient[i] += 1 - h[i + k]
            gradient[i + k] -= h[i]
        grad_output = gradient * dx
        return overlap * dx, grad_output

    def get_max_overlap_gradient(h):
        n = len(h)
        max_overlap = -np.inf
        max_k = -1
        max_grad = np.zeros(n)
        for k in range(1, n - 1):
            overlap, grad = compute_overlap_gradient(h, k)
            if overlap > max_overlap:
                max_overlap = overlap
                max_k = k
                max_grad = grad
        return max_overlap, max_k, max_grad

    def project_h(h):
        # Project h onto [0,1]^n and normalize to unit area
        h_clamped = np.clip(h, 0, 1)
        total_sum = np.sum(h_clamped)
        if abs(total_sum) > 1e-8:
            h_clamped *= n_points / (2.0 * total_sum)
        h_clamped = np.clip(h_clamped, 0, 1)  # re-clip due to scaling
        return h_clamped

    def compute_c5(h):
        dx = 2.0 / len(h)
        correlations = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(correlations)
        c5 = max_corr * dx
        return c5

    step_size = 1.0
    num_steps = 0
    tolerance = 1e-6

    while time.time() - start_time < budget_s:
        # Compute worst overlap gradient
        c5, k_max, grad = get_max_overlap_gradient(h)

        if k_max == -1:
            break  # no improvement possible

        # Projected gradient update
        h = project_h(h - step_size * grad)

        if compute_c5(h) < best_c5:
            best_c5 = compute_c5(h)
            best_h = h.copy()
            print(f"Improved to {best_c5:.7f} at step {num_steps}")
            step_size *= 1.05  # increase step if improvement
        else:
            # backtracking if not improving
            step_size *= 0.5

        if step_size < 1e-7:
            step_size = 1.0

        num_steps += 1

        # Perturb occasionally to avoid local minima
        if num_steps % 50 == 0:
            perturb = np.random.randn(n_points) * 0.05
            perturbed_h = project_h(best_h + perturb)
            if compute_c5(perturbed_h) < best_c5:
                h = perturbed_h
                step_size = 1.0

        if c5 < best_c5 - 1e-8 and time.time() - start_time > budget_s - 10:
            print(f"Final improvement to {c5:.7f}")

    return best_h.tolist(), best_c5, n_points
