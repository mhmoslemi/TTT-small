import numpy as np
import scipy.optimize as opt
from math import prod
from time import time

# Constants
MAX_TIME = 1100.0
EPS = 1e-6

def np_correlate_with_penalty(h, sum_h=None, dx=None):
    if sum_h is None:
        sum_h = np.sum(h)
    if dx is None:
        n = len(h)
        dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode="full")
    conv *= dx
    max_c5 = np.max(conv[1:-1])
    penalty = (sum_h - n / 2)**2
    return max_c5 + 1e4 * penalty

def objective(h):
    h = h.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if abs(sum_h) < 1e-8:
        return np.inf
    return np_correlate_with_penalty(h, sum_h=sum_h, dx=2.0 / len(h))

def constraint_sum(h, n_points):
    return np.sum(h) - n_points / 2

def grad_max_corr(h, dx=None, i_max=None):
    if dx is None:
        dx = 2.0 / len(h)
    if i_max is None:
        # Find the index of max C5
        conv = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(conv[1:-1]) + 1
    else:
        max_idx = i_max
    dh = np.zeros_like(h)
    # Forward terms: h[i]*(1 - h[i + k])
    for i in range(len(h) - max_idx):
        dh[i] += (1 - h[i + max_idx]) * dx
    # Backward terms: -(1 - h[j - k])*h[j - k]
    for j in range(max_idx, len(h)):
        dh[j] -= h[j - max_idx] * dx
    return dh
