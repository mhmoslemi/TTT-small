import numpy as np
import cvxpy as cp
import time
from numpy import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s

    h = np.array(initial_h_values).copy()
    n_points = len(h)
    dx = 2.0 / n_points
    expected_sum = n_points / 2
    assert abs(h.sum() - expected_sum) < 1e-6, "Initial h violates constraint"

    def compute_correlation(h):
        hx = h.copy()
        one_minus_h = 1 - h
        corr = np.zeros_like(h)
        for k in range(n_points * 2 - 1):
            lhs = 0
            rhs = k
            valid = slices_valid(lhs, rhs, n_points)
            if not valid:
                continue
            sum_val = np.sum(hx[valid[0]] * one_minus_h[valid[1]])
            corr[k] = sum_val * dx
        return corr

    def slices_valid(lhs, rhs, n_points):
        if rhs < 0 or lhs >= n_points:
            return None, None
        sl1 = slice(lhs, lhs + n_points - rhs - 1)
        sl2 = slice(rhs, n_points)
        return sl1, sl2

    best_c5 = -np.inf
    best_h = h.copy()
    best_corr = None

    # Optimization loop
    iteration = 0
    while time.time() - start_time < max_time:
        # Compute current C5
        corr = compute_correlation(h)

        current_c5 = corr.max()
        best_c5 = max(best_c5, current_c5)
        best_corr = corr if current_c5 > best_c5 else best_corr

        # Update h using a convex relaxation step
        delta = 0.003 * iteration / 1000  # adaptive step size
        delta = max(0.001, delta)

        # Build linear approximation of the influence for gradient direction
        influence = np.zeros_like(h)

        # Estimate how much changing h[i] would influence the maximum C5
        for i in range(n_points):
            # Try moving up a small step
            h_up = h.copy()
            h_up[i] = min(1.0, h[i] + delta)
            h_up /= h_up.sum()
            h_up *= expected_sum

            # Try moving down a small step
            h_down = h.copy()
            h_down[i] = max(0.0, h[i] - delta)
            h_down /= h_down.sum()
            h_down *= expected_sum

            corr_up = compute_correlation(h_up)
            corr_down = compute_correlation(h_down)

            influence[i] = corr_up.max() - corr_down.max()

        # Try both directions, pick the one that reduces C5 the most
        options = []
        for i in range(n_points):
            up_h = h.copy()
            up_h[i] = min(1.0, h[i] + delta * (influence[i] > 0 or np.random.rand() < 0.05))
            up_h /= up_h.sum()
            up_h *= expected_sum
            up_corr = compute_correlation(up_h)
            options.append((up_corr.max(), up_h, "up"))

            down_h = h.copy()
            down_h[i] = max(0.0, h[i] - delta * (influence[i] < 0 or np.random.rand() < 0.05))
            down_h /= down_h.sum()
            down_h *= expected_sum
            down_corr = compute_correlation(down_h)
            options.append((down_corr.max(), down_h, "down"))

        # Pick the best next point that lowers C5 the most
        options.sort()
        if options and options[0][0] < current_c5:
            h = options[0][1]

        # Occasionally use convex relaxation
        elif iteration % 100 == 0:
            # Formulate convex relaxation of the optimization
            h_var = cp.Variable(n_points)
            constraints = [
                0 <= h_var,
                h_var <= 1,
                cp.sum(h_var) == expected_sum,
            ]

            # Use a convex relaxation of the max correlation
            h_temp = cp.Parameter(n_points, value=h)
            corr_obj = []
            for k in range(n_points * 2 - 1):
                lhs = k
                rhs = 0
                sl1, sl2 = slices_valid(lhs, rhs, n_points)
                if sl1 is not None:
                    corr_k = cp.sum(h_var[sl1] * (1 - h_temp[sl2]))
                    corr_k *= dx
                    corr_obj.append(corr_k)

            obj = cp.Minimize(cp.max(corr_obj))
            prob = cp.Problem(obj, constraints)
            try:
                result = prob.solve(solver=cp.GUROBI, verbose=False)
                h_relax = np.array(h_var.value)
                h_relax /= h_relax.sum() * dx
                h_relax *= expected_sum
                h = h_relax
            except Exception as e:
                pass  # fall back to local adjustment

        iteration += 1

    # Final verification
    best_c5 = best_corr.max()
    assert abs(h.sum() - expected_sum) < 1e-6
    return (h.tolist(), best_c5, n_points)
