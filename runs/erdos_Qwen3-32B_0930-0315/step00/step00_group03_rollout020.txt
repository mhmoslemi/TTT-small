    max_no_improvement = 100
    learning_rate = 0.01

    while time.time() - start_time < budget_s:
        # Compute all correlations
        h1h = h * (1 - h)
        c5, max_k = compute_c5_and_shift(h, n_points, dx)

        # Compute gradient for the worst shift
        G = compute_gradient(h, max_k, n_points)

        # Update h
        h = h - learning_rate * G
        learning_rate *= 0.999  # learning rate decay

        # Project to feasible region
        h = project_h(h, n_points)

        # Check if improved
        current_c5 = compute_c5(h, n_points, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        if no_improvement_count >= max_no_improvement:
            break

    return best_h, best_c5, n_points

def compute_c5(h, n_points, dx):
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def compute_c5_and_shift(h, n_points, dx):
    corr = np.correlate(h, 1 - h, mode="full") * dx
    max_idx = np.argmax(corr)
    return corr[max_idx], max_idx - n_points + 1

def compute_gradient(h, k, n_points):
    n = len(h)
    G = np.zeros(n)
    for i in range(n):
        j = (i + k) % n
        if k >= 0 and j < n:
            G[i] += (1 - h[j])
        if k < 0 and i >= -k:
            G[i] -= h[i + k]
    return G

def project_h(h, n_points):
    n = len(h)
    h_var = cp.Variable(n)
    constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2]
    objective = cp.Minimize(cp.sum_squares(h_var - h))
    problem = cp.Problem(objective, constraints)
    try:
        SOLVER = cp.GUROBI if 'GUROBI' in cp.installed_solvers() else cp.ECOS
        problem.solve(solver=SOLVER, verbose=False)
        h_proj = np.clip(h_var.value, 0, 1)
    except:
        h_proj = np.clip(h, 0, 1)
        h_proj = (h_proj / np.sum(h_proj)) * (n_points / 2)
    return h_proj
