    best_c5 = np.inf

    last_improvement_time = start_time

        # Compute all correlation lags and their values
        lags = np.array([0] * len(h))
        for k in range(1, n_points):
            ln = h.shape[0] - k
            val = (h[:ln] * (1 - h[k:])).sum() * dx
            lags[k] = val
        for k in range(1, n_points):
            ln = h.shape[0] - k
            val = (h[k:] * (1 - h[:ln])).sum() * dx
            lags[-k] = val

        max_k = np.argmax(np.abs(lags))
        c5 = lags[max_k]
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement_time = time.time()

        # Get gradient direction for max overlap lag
        gradient = np.zeros(n_points)

        if max_k < n_points / 2:
            # Forward overlap
            k = int(max_k)

        # Update h with gradient descent
        h -= alpha * gradient
        # Clip to [0, 1]
        h = np.clip(h, 0.0, 1.0)

        scaling = (n_points / 2.0) / h.sum()

            h *= scaling

        # Early exit if no improvement in last 25% of time

    h = best_h.copy()

        val = (h[k:] * (1 - h[:ln])).sum() * dx
