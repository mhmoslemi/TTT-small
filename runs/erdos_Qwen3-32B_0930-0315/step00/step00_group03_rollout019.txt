import numpy as np
import scipy.optimize as opt
import time

    dx = 2.0 / n_points

    # Constraint: sum(h) = n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    # Objective function: min max overlap
    def objective(x):
        h = x.reshape(-1)
        max_overlap = -np.inf
        for k in range(1, n_points):
            rolled = np.roll(h, k)
            overlap = np.sum(h * (1 - rolled)) * dx
            max_overlap = max(max_overlap, overlap)
        return max_overlap

    # Bounds and projection
    bounds = np.array([(0.0, 1.0) for _ in range(n_points)])
    def project_to_bounds(x):
        return np.clip(x, bounds[:,0], bounds[:,1])

    def projected_objective(x):
        x = x.copy()
        x = project_to_bounds(x)
        return objective(x)

    # Optimization with L-BFGS-B
    result = opt.minimize(
        projected_objective,
        h,
        method='L-BFGS-B',
        jac=None,
        options={'maxiter': 1000, 'maxfun': 1000, 'disp': False}
    )

    best_h = result.x
    best_h = project_to_bounds(best_h)
    best_h = best_h / np.sum(best_h) * (n_points / 2)  # Re-normalize in case constraint drifted

    # Evaluate C5 from the best found h
    max_corr = -np.inf
    for k in range(1, n_points):
        rolled = np.roll(best_h, k)
        corr = np.sum(best_h * (1 - rolled)) * dx
        max_corr = max(max_corr, corr)

    best_c5 = max_corr
    time_used = time.time() - start_time

    # Safety loop: re-evaluate every 10 iterations if time allows
    while time_used < budget_s * 0.9:
        result = opt.minimize(
            projected_objective,
            best_h,
            method='L-BFGS-B',
            jac=None,
            options={'maxiter': 100, 'maxfun': 100, 'disp': False}
        )
        candidate_h = result.x
        candidate_h = project_to_bounds(candidate_h)
        candidate_h = candidate_h / np.sum(candidate_h) * (n_points / 2)

        # Compute correlation
        corr = -np.inf
        for k in range(1, n_points):
            rolled = np.roll(candidate_h, k)
            corr_k = np.sum(candidate_h * (1 - rolled)) * dx
            corr = max(corr, corr_k)

        if corr < best_c5:
            best_c5 = corr
            best_h = candidate_h

        time_used = time.time() - start_time
        if time_used > budget_s:
