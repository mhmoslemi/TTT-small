import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

# Optimized version using adaptive search regions and feasibility projection

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h = np.copy(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    assert np.isclose(np.sum(h), n_points / 2), "Inital h is not normalized"
    
    def compute_c5(h_vec):
        h_flip = 1 - h_vec
        cor = correlate(h_vec, h_flip, mode="full")
        return np.max(cor * dx)

    # Main optimization
    best_h = np.copy(h)
    best_c5 = compute_c5(best_h)
    last_improvement = 0
    patience = 100
    evals = 0
    
    # Define region influence matrix
    R = np.zeros_like(h)
    k_shifts = np.zeros_like(h)

    baseline_c5 = best_c5
    print(f"Initial C5: {best_c5:.5f}")

    while time.time() - start_time < budget_s - 50:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if elapsed > 0.95 * budget_s:
            break

        # Calculate current convergence metric
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h[:]
            last_improvement = 0
        else:
            last_improvement += 1

        # Break early if no improvement
        if last_improvement > patience:
            break

        # Redundancy measure - how much this point affects max overlap
        k_shifts = np.zeros(n_points)
        for i in range(n_points):
            for k in range(-n_points, n_points):
                if 0 <= i + k < n_points:
                    k_shifts[i] += h[i] * (1 - h[i + k])
        R[:] = k_shifts / n_points  # normalized redundancy
        rescale = 2.0 / (np.max(R) + 1e-6)
        R = np.clip(R * rescale, 0, 1)  # enforce bounds

        # Identify high redundancy regions to focus search
        high_red_idx = np.argsort(R)[-n_points//4:]
        step_sizes = np.where(np.isin(np.arange(n_points), high_red_idx), 0.005, 0.02)

        # Generate candidate updates under constraints
        h_candidate = np.copy(h)
        for i in range(n_points):
            delta = np.random.uniform(-step_sizes[i], step_sizes[i])
            h_candidate[i] += delta
            h_candidate[i] = np.clip(h_candidate[i], 0, 1)

        # Ensure the constraint remains satisfied
        current_sum = np.sum(h_candidate)
        scaling_factor = (n_points/2) / max(current_sum, 1e-6)
        h_candidate = h_candidate * scaling_factor
        h_candidate = np.clip(h_candidate, 0, 1)

        if compute_c5(h_candidate) < best_c5:
            h[:] = h_candidate[:]
            last_improvement = 0

        # Optional refinement using convex programming with random restarts
        if time.time() - start_time < budget_s - 100:
            # Focus on regions with high redundancy
            active_idx = high_red_idx
            focus = h[active_idx]
            focus_sum = np.sum(focus)
            
            # Build the convex optimization problem
            x = cp.Variable(len(active_idx))
            constraints = [0 <= x, x <= 1, cp.sum(x) == focus_sum]
            
            # Objective: minimize the max overlap in candidate regions
            overlap = correlate(focus, 1 - focus, mode="full")
            objective = cp.Minimize(overlap.max())

            prob = cp.Problem(objective, constraints)
            result = prob.solve(solver=cp.SCIP, verbose=False)
            
            # Update with solution if improved
            if result < best_c5:
                h[active_idx] = x.value
                best_c5 = result
                best_h[:] = h[:]

        evals += 1

    # Return the best solution found
    return (best_h, best_c5, n_points)
