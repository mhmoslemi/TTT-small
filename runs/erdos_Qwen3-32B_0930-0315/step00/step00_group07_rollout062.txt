import numpy as np
import cvxpy as cp
import time
from math import exp

def evaluate_overlap(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)

def solve_subproblem(n_points, h, k, dx, max_change=0.05, solver='ECOS'):
    dh = cp.Variable(n_points)
    
    # Mask indices where wraparound occurs at (i + k) mod n_points
    idx = np.arange(n_points)
    idx_k = (idx + k) % n_points
    
    # Linearized objective
    obj = cp.Minimize(sum(h[i] * (1.0 - h[j]) + h[i] * dh[j] - h[j] * dh[i] for i, j in zip(idx, idx_k)) * dx)

    constraints = [
        dh[i] >= -h[i] for i in range(n_points)
    ] + [
        dh[i] <= 1.0 - h[i] for i in range(n_points)
    ] + [cp.sum(dh) == 0]

    problem = cp.Problem(obj, constraints)
    problem.solve(solver=solver)
    
    return dh.value

def repair_h(h, n_points):
    h_clamped = np.clip(h, 0.0, 1.0)
    scale = (n_points / 2.0) / np.sum(h_clamped)
    h_scaled = h_clamped * scale
    return h_scaled
