import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    
    # Initial solution
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Feasibility: normalize to satisfy integral constraint
    h = np.clip(h, 0, 1)
    h = h / h.sum() * n_points / 2
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    start_time = time.time()
    
    max_iterations = 500
    for iter_idx in range(max_iterations):
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break

        def objective(h_vec):
            h = h_vec.reshape(n_points)
            # Compute cross-correlation between h and 1 - h
            corr_vals = correlate(h, 1 - h, mode='full')
            corr_vals = corr_vals * dx
            # Compute C₅ as max overlap
            max_corr = np.max(corr_vals)
            # Add penalty for constraint violation
            mass_error = np.sum(h) - n_points / 2
            lambda_penalty = 1.0e4
            return max_corr + lambda_penalty * mass_error ** 2

        def constraint_func(h_vec):
            h = h_vec.reshape(n_points)
            return np.sum(h) - n_points / 2

        def bounds():
            return [(0.0, 1.0)] * n_points

        # Compute current best objective and its index for gradient-based search
        result = minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds(),
            jac=None if iter_idx == 0 else appx_gradient(h, dx),
            options={'maxiter': 1, 'ftol': 1e-8}
        )

        h = result.x
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)

        # Project back to feasible region: scale to satisfy constraint
        mass_error = np.sum(h) - n_points / 2
        if abs(mass_error) > 1e-5:
            h = h / np.sum(h) * n_points / 2

        # Add perturbation every 50 iterations
        if iter_idx % 50 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h / np.sum(h) * n_points / 2

        # Evaluate and track best
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    corr = corr * dx
    return np.max(corr)

def appx_gradient(h, dx):
    def grad_func(h_flat):
        h = h_flat.reshape(-1)
        n = len(h)
        grad = np.zeros(n)
        corr = correlate(h, 1 - h, mode='full')
        corr = corr * dx
        max_idx = int(np.argmax(corr))
        if 0 < max_idx < n:
            grad[:n - max_idx] += (1 - h[max_idx:n])[::-1] * dx
            grad[max_idx:] += (-h[:n - max_idx]) * dx
        mass = np.sum(h)
        lambda_ = 1.0e4
        grad += 2 * lambda_ * (mass - n / 2) / n
        return grad
    return grad_func
