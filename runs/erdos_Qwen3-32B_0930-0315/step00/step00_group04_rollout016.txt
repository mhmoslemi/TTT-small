import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    
    # Initial setup
    h = initial_h_values
    n_points = len(h)
    dx = 2.0 / n_points

    start_time = time.time()

    # Helper functions
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total != 0:
            h = h * (n_points / 2) / total
        return h

    # Optimization loop parameters
    lr = 0.01  # Initial learning rate
    best_h, best_c5 = h.copy(), compute_c5(h)
    prev_c5 = best_c5 + 1  # Initialize with a value larger than the best

    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        if iteration % 100 == 0:
            lr *= 0.95  #Anneal learning rate

        # Compute correlation
        corr = np.correlate(h, 1 - h, mode="full")
        max_index = np.argmax(corr)

        # Extract the shifted version of 1 - h that contributed to max overlap
        k = max_index - n_points + 1  # shift amount
        if k < 0:
            h_shifted = np.concatenate((np.zeros(-k), 1 - h[:k + n_points]))
        elif k > 0:
            h_shifted = np.concatenate((1 - h[k:], np.zeros(k)))
        else:
            h_shifted = 1 - h

        # Gradient update
        grad = h_shifted
        h = h - lr * grad

        # Projection
        h = project_to_feasible(h)

        # Update best solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

        # Stop if converged
        if abs(prev_c5 - current_c5) < 1e-7:
            break
        prev_c5 = current_c5

    return (best_h, best_c5, n_points)
