import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, n_points=45):
    np.random.seed(seed)
    start_time = time.time()
    
    # Extract initial h from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.randint(0, 2, size=n_points)
        h = h.astype(np.float64)
        h *= 2  # scale to [0, 2]
        # Normalize to the constraint
        total = np.sum(h)
        h = h * (n_points / 2) / total

    h_best = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    tabu_list = []

    MAX_TABU_SIZE = 20
    NEIGHBORS_PER_ITER = 20
    MAX_STEPS = 10000000
    MAX_PERTURB = 5
    
    dx = 2.0 / n_points
    
    for step in range(MAX_STEPS):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        neighbors = []
        for _ in range(NEIGHBORS_PER_ITER):
            hh = h.copy()
            
            # Perturbation: change 1-5 random indices
            n_change = np.random.randint(1, MAX_PERTURB + 1)
            pos = np.random.choice(n_points, n_change, replace=False)
            
            for i in pos:
                delta = np.random.uniform(-0.5, 0.5)
                hh[i] = np.clip(hh[i] + delta, 0, 1)
            
            # Normalize to satisfy the constraint
            total = np.sum(hh)
            if total < 1e-10:
                continue
            hh = hh * (n_points / 2) / total
            neighbors.append(hh)
            
        # Compute C5 values for all neighbors
        valid_neighbor_indices = []
        neighbor_C5 = []
        for nn in neighbors:
            # Skip if in tabu
            skip = False
            for t in tabu_list:
                if np.allclose(nn, t):
                    skip = True
                    break
            if skip:
                continue
            valid_neighbor_indices.append(nn)
            neighbor_C5.append(evaluate_erdos_solution(nn))
            
        if valid_neighbor_indices:
            idx = np.argmin(neighbor_C5)
            h = valid_neighbor_indices[idx]
            current_C5 = neighbor_C5[idx]
            
            if current_C5 < best_C5:
                h_best = h.copy()
                best_C5 = current_C5
                
            # Add current state to tabu
            if len(tabu_list) >= MAX_TABU_SIZE:
                tabu_list.pop(0)
            tabu_list.append(h_best.copy())
        else:
            # If no valid neighbors, perturb current h
            for _ in range(2):
                idx = np.random.randint(n_points)
                delta = np.random.uniform(-0.1, 0.1)
                h[idx] = np.clip(h[idx] + delta, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return h_best, best_C5, n_points
