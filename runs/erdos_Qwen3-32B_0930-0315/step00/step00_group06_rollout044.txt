```python
import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 256
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h
    if 'initial_h_values' in globals():
        if len(initial_h_values) == 2 * n_points:
            h = np.array(initial_h_values[n_points:2*n_points])  # take second half
        else:
            h = initial_h_values[:n_points].copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h *= (target_sum / np.sum(h))

    best_h = h.copy()
    best_c5 = float('inf')

    def compute_c5(h_flat):
        h = h_flat.reshape(-1)
        corr = correlate(h, 1 - h, mode='full')
        corr = corr[n_points - 1:-n_points + 1]  # keep only relevant lags
        return float(np.max(corr * dx))

    def update_magnitudes(h_flat):
        h = h_flat.reshape(-1)
        # Objective: minimize max overlap - we instead minimizing sum over certain size of |overlaps|
        overlap_contribution = correlate(h, 1 - h, mode='full')[n_points - 1:-n_points + 1]
        k_max = np.argmax(overlap_contribution)

        h_var = cp.Variable(n_points, nonneg=True)
        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == target_sum
        ]
        obj = cp.Minimize(overlap_contribution[k_max])
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.GUROBI, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_var = np.array(h_var.value).flatten()
            return h_var
        return h_flat

    step = 0
    while time.time() - start_time < budget_s:
        h = update_magnitudes(h)
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # predefined exploration
        if step % 100 == 0 and step > 0:
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h = np.clip(h + perturbation, 0, 1)
            h *= (target_sum / np.sum(h))
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

        step += 1

    return best_h.tolist(), best_c5, n_points
```
