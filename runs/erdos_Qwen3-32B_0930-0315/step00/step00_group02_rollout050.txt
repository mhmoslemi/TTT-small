import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)
    n_points = 160  # Resolution for the step function
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h: plateau in the middle
    h = np.zeros(n_points, dtype=np.float64)
    midpoint = n_points // 2
    plateau_width = n_points // 2
    h[midpoint - plateau_width//2 : midpoint + plateau_width//2] = 0.5

    # Ensure sum constraint
    current_sum = h.sum()
    if current_sum < target_sum:
        idx = np.argsort(h)
        fill = int(target_sum - current_sum)
        h[idx[-fill:]] = 1.0
    else:
        idx = np.argsort(-h)
        fill = int(current_sum - target_sum)
        h[idx[:fill]] = 0.0

    h = h[None, :].copy()
    h_best = h.copy().flatten()
    c5_best = float('inf')

    max_epochs = 100
    noisy_epochs = 100

    for epoch in range(max_epochs):
        if time.time() - start_time > budget_s - 5:
            break

        h = h_best[None, :].copy()
        # Phase 1: Gradient descent update
        gradient = np.zeros_like(h)
        for k in range(-2 * n_points, 2 * n_points):
            shifted = np.roll(h, shift=k, axis=1)
            shifted = shifted[:, (k + n_points) % (2 * n_points) : (k + 2 * n_points) % (2 * n_points)]
            integrand = h * (1.0 - shifted)
            integral = np.sum(integrand)
            gradient += integrand

        # Normalize gradient
        grad_norm = np.sqrt(np.sum(gradient**2))
        gradient = gradient / (1e-12 + grad_norm)
        # Gradient step
        step = 0.5
        h = h - step * gradient
        h = np.clip(h, 0.0, 1.0)

        # Project onto constraint set (enforce sum constraint with box projections)
        while np.abs(h.sum() - target_sum) > 1.0:
            diff = h.sum() - target_sum
            idx = np.argsort(h, axis=None)
            if diff > 0:
                # Remove excess
                change = 1.0 if diff > 0 else -1.0
                while diff != 0 and idx[-1] >= 0:
                    h.flat[idx[-1]] = min(h.flat[idx[-1]] + change, 1.0 if change > 0 else 0.0)
                    diff -= change
                    idx = idx[:-1]
            else:
                # Add missing
                change = -diff
                idx = idx[::-1]
                while diff != 0 and idx[-1] >= 0:
                    h.flat[idx[-1]] = max(h.flat[idx[-1]] + change, 0.0 if change < 0 else 1.0)
                    diff += change
                    idx = idx[:-1]

        h = np.clip(h, 0, 1)

        # Evaluate current C5
        c5 = evaluate_erdos_solution(h[0], n_points)
        if c5 < c5_best:
            h_best = h.copy().flatten()
            c5_best = c5
            print(f"Epoch {epoch:03d} C5={c5_best:.6f}")

    for dim in range(noisy_epochs):
        if time.time() - start_time > budget_s - 1:
            break

        h = h_best.copy()
        noise = np.random.randn(n_points) * 0.05
        h += noise
        h = np.clip(h, 0, 1)
        while np.abs(h.sum() - target_sum) > 1.0:
            diff = h.sum() - target_sum
            idx = np.argsort(h)
            if diff > 0:
                h[idx[-1]] = 0.0
            else:
                h[idx[0]] = 1.0
        h = np.clip(h, 0, 1)

        c5 = evaluate_erdos_solution(h, n_points)
        if c5 < c5_best:
            h_best = h.copy()
            c5_best = c5
            print(f"Noisy {dim:03d} C5={c5_best:.6f}")

    # Final refinement using convex programming
    try:
        x = cp.Variable(n_points)
        constraints = [0 <= x, x <= 1, cp.sum(x) == target_sum]
        C_all = []
        for k in range(-2 * n_points, 2 * n_points):
            shifted = np.roll(h_best, shift=k, axis=0)
            shifted = shifted[(k + n_points) % (2 * n_points) : (k + 2 * n_points) % (2 * n_points)]
            integrand = x * (1.0 - shifted)
            C_all.append(cp.sum(integrand))
        worst_Ck = cp.Max(*C_all)
        objective = cp.Minimize(worst_Ck)
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.ECOS)
        if problem.status in cp.SOLUTION_PRESENT:
            h_best_ecos = x.value
            h_best_ecos = np.clip(np.round(h_best_ecos, 8), 0, 1)
            c5_ecos = evaluate_erdos_solution(h_best_ecos, n_points)
            print(f"ECOS: Final C5={c5_ecos:.6f}")
            if c5_ecos < c5_best:
                h_best = h_best_ecos
                c5_best = c5_ecos
    except:
        pass  # Fallback if ECOS is unavailable or if timeout

    return h_best, c5_best, n_points
