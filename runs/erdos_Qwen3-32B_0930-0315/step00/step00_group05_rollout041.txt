import numpy as np
import time
from math import floor

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.clip(h, 0, 1)
        s = h.sum()
        if s > 0:
            h *= (n_points / 2) / s
    else:
        h = np.zeros(n_points, dtype=np.float64)
        h[:n_points // 2] = 1.0

    best_h = h.copy()
    best_c5 = float('inf')
    patience = 0
    alpha = 0.1
    noise_kick = 50
    tolerance = 1e-4

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        valid_corr = corr[n_points - 1:2 * n_points - 1]
        return valid_corr.max() * dx

    while time.time() - start_time < budget_s:
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

        if patience >= 100:
            # Reduce learning rate if no improvement
            alpha *= 0.75
            patience = 0

        # Identify the worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr[n_points - 1:2 * n_points - 1])
        k = max_k + (n_points - 1)

        # Compute gradient for this k
        grad = np.zeros(n_points)
        for m in range(n_points):
            if m + k < n_points:
                grad[m] += (1 - h[m + k]) * dx
            if m >= k:
                grad[m] -= h[m - k] * dx
        grad *= dx

        # Update and project h
        h -= alpha * grad
        h = np.clip(h, 0, 1)
        s = h.sum()
        if s > 0:
            h *= (n_points / 2) / s

        # Add noise periodically
        if np.random.rand() < 0.02:
            h += np.random.uniform(-0.005, 0.005, size=n_points)
            h = np.clip(h, 0, 1)
            s = h.sum()
            if s > 0:
                h *= (n_points / 2) / s

    # Ensure returned solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h *= (n_points / 2) / best_h.sum()
    return best_h, best_c5, n_points
