import numpy as np
import cvxpy as cp
from time import time

def evaluate_starting_point(h):
    n_points = len(h)
    dx = 2.0 / n_points
    overlap = np.correlate(h, 1 - h, mode='full')
    values = overlap * dx
    c5 = np.max(values)
    return c5

def run(seed=42, budget_s=1100):
    t0 = time()
    np.random.seed(seed)

    n_points = 47  # Starting from the known initialization size
    dx = 2.0 / n_points

    # Use the parent-provided initial_h_values if available
    if 'initial_h_values' in globals():
        h_vals = initial_h_values.copy()
    else:
        # Construct a uniform initial h
        h_vals = np.full(n_points, 0.5 / dx)
        h_vals[-1] += 1.0 / dx  # Slight adjustment to ensure sum constraint
        h_vals = np.clip(h_vals * dx, 0.0, 1.0)

    best_h = h_vals
    best_c5 = evaluate_erdos_solution(n_points, h_vals)

    # SDP relaxed optimization
    def cvx_loop(h_init):
        h = cp.Variable(n_points)
        t = cp.Variable()

        # Constraints
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]

        # Rename correlate matrix as corr
        corr = np.zeros((n_points, n_points))
        for k in range(n_points):
            for i in range(max(0, n_points - k - 1)):
                corr[i, i + k] = (h[i] * (1 - h[i + k])) * dx
                corr[i + k, i] = (h[i + k] * (1 - h[i])) * dx  # symmetry

        corr_diag_sum = cp.diag(corr).sum()

        # Replace max with it's scalar approximation via introducing bound with a small tolerance
        constraints += [corr_diag_sum <= t]

        # Objective: minimize t
        objective = cp.Minimize(t)
        problem = cp.Problem(objective, constraints)

        try:
            problem.solve(solver=cp.CVXOPT, verbose=True)
            h_sol = np.round(h.value, 6)
            h_sol = np.clip(h_sol, 0.0, 1.0)

            # Enforce the sum constraint
            if np.sum(h_sol) != n_points / 2:
                adjust = (n_points / 2) - np.sum(h_sol)
                h_sol[0] += adjust / n_points  # Distribute the mismatch
                h_sol = np.clip(h_sol, 0.0, 1.0)

            return h_sol, np.round(t.value, 6)
        except:
            return None, None

    # Warm start with initial_h_values
    h_opt, t_opt = cvx_loop(best_h)

    if h_opt is not None:
        c5_opt = evaluate_erdos_solution(n_points, h_opt)
        best_h = h_opt
        best_c5 = c5_opt

    # Multi-random start search
    max_iter = 10
    for i in range(max_iter):
        if time() - t0 > budget_s - 10:
            break

        # Random construction with sum constraint
        h_rand = np.random.randint(0, 2, n_points).astype(float)
        h_rand /= h_rand.sum()
        h_rand *= n_points / 2
        h_rand = np.clip(h_rand, 0.0, 1.0)

        h_trial, t_trial = cvx_loop(h_rand)
        if h_trial is not None:
            c5_trial = evaluate_erdos_solution(n_points, h_trial)
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial

    return best_h, best_c5, n_points
