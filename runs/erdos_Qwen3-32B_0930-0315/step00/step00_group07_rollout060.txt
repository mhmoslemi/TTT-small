import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    from matplotlib import pyplot as plt  # Plotter

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
        assert np.allclose(np.sum(h), n_points / 2), "Initial h doesn't satisfy sum constraint"
    else:
        n_points = 80
        h = np.full(n_points, 0.5, dtype=np.float64)
    dx = 2.0 / n_points

    best_c5 = np.inf
    best_h = h.copy()

    def compute_c5(h_):
        h = h_.copy()
        # Zero-padding to avoid wraparound
        h_extended = np.concatenate([np.zeros_like(h), h, np.zeros_like(h)])
        c5 = np.correlate(h, 1 - h, mode="same")
        c5 = c5[n_points:2*n_points] * dx
        return np.max(c5)

    def solve_subproblem(h_, k):
        h = h_.copy()
        n = len(h)
        delta = cp.Variable(n)
        obj = cp.Minimize(sum([h[i] * (1 - h[(i + k) % n]) * dx for i in range(n)]
                             - delta * h * (1 - h[(k + np.arange(n)) % n]) * dx
                             - h * delta[(k + np.arange(n)) % n] * (1 - h[(k + np.arange(n)) % n]) * dx
                             + ...))
        obj = cp.Minimize(sum([h[i] * (1 - h[(i + k) % n]) + h[i] * (-delta[(i + k) % n] * (1 - h[(i + k) % n])) + (-delta[i]) * (1 - h[(i + k) % n]) for i in range(n)]) * dx)
        constraints = [0 <= h[i] + delta[i] for i in range(n)] + \
                      [h[i] + delta[i] <= 1 for i in range(n)] + \
                      [sum(delta) == 0]
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver='ECOS')
            return delta.value
        except:
            return np.zeros_like(h)

    def repair_h(h_):
        h = h_.copy()
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        h = h * (n_points / 2) / s
        return h

    iteration = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improvement = 0

    while time.time() - start_time < budget_s:
        iteration += 1

        c5 = compute_c5(h)
        k_max = np.argmax(c5)

        delta = solve_subproblem(h, k_max)
        if delta is not None:
            h_new = h + delta
            h = repair_h(h_new)
            c5_new = compute_c5(h)
        else:
            c5_new = c5

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1

        if no_improvement >= 20:
            break

    return (best_h.tolist(), best_c5, n_points)
