```python
import numpy as np
import scipy.optimize as opt
import math
import time

initial_h_values = np.array([0.5]*70)

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h):
        h = h.copy().clip(0, 1)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        C5 = np.max(corr)
        sum_h = h.sum()
        penalty = (sum_h - target_sum) ** 2 * 1e4
        return C5 + penalty

    def project(h):
        h = h.clip(0, 1)
        scale = target_sum / h.sum()
        h = h * scale
        h = h.clip(0, 1)
        return h

    best_h = initial_h_values.copy()
    best_C5 = objective(best_h)
    iter_count = 0

    while time.time() - start_time < budget_s:
        h0 = best_h.copy()
        op_res = opt.minimize(
            objective,
            x0=h0,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 50, 'ftol': 1e-10}
        )
        h_opt = op_res.x
        h_opt = project(h_opt)
        c5 = objective(h_opt)

        if c5 < best_C5:
            best_h[:] = h_opt[:]
            best_C5 = c5

        if iter_count % 50 == 0:
            idxs = np.random.choice(n_points, min(10, n_points), replace=False)
            pert = np.random.normal(0, 0.01, n_points)
            best_h += pert
            best_h = project(best_h)

        iter_count += 1

        if (time.time() - start_time > 600) and (n_points == 70):
            n_points = 140
            dx = 2.0 / n_points
            target_sum = n_points / 2.0
            best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, best_h.shape[0]), best_h)
            best_h = project(best_h)

    h_values = best_h
    c5_bound = best_C5
    return h_values, c5_bound, n_points
```
