import numpy as np
import time
import evaluate_erdos_solution

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float("inf")
    resolution_sequence = [14, 28, 56]
    
    for n in resolution_sequence:
        # Initialize h from initial_h_values if available
        if "initial_h_values" in globals():
            # Downsample initial_h_values to size n by averaging
            full_h = np.array(initial_h_values, dtype=np.float64)
            downsampled_h = np.linspace(0, 2, len(full_h))
            x_new = np.linspace(0, 2, n)
            h = np.interp(x_new, downsampled_h, full_h)
            h = h / h.sum() * (n / 2)
        else:
            # Random initialization
            h = np.random.uniform(0, 1, n)
            h = h / h.sum() * (n / 2)

        if time.time() - start_time >= budget_s:
            break

        # Store current time for this resolution
        t0 = time.time()

        # Phase 1 - Simulated Annealing
        T = 1.0
        alpha = 0.99
        sa_steps = 500
        for sa_step in range(sa_steps):
            if time.time() - t0 >= 300 and time.time() - start_time < budget_s - 200:
                t0 = time.time()  # Reset timer if not last resolution
                break

            # Generate candidate
            h_candidate = h + np.random.normal(0, T, h.shape)
            
            # Feasibility repair: maintain constraints
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate / h_candidate.sum() * (n / 2)
            
            # Compute objective
            dx = 2.0 / n
            kernel = 1 - h_candidate
            corr = np.correlate(h_candidate, kernel, mode="full") * dx
            c5_candidate = np.max(corr)
            
            # Acceptance rule
            if c5_candidate < best_c5 or np.random.rand() < np.exp((best_c5 - c5_candidate) / T):
                h = h_candidate
                best_c5 = c5_candidate
        
            # Cool down
            T *= alpha

        # Phase 2 - Gradient Descent for local refinement
        if time.time() - t0 < budget_s - 100:
            lr = 0.01
            grad_steps = 500
            for grad_step in range(grad_steps):
                if time.time() - t0 >= 300 and time.time() - start_time < budget_s - 200:
                    t0 = time.time()  # Reset timer if not last resolution
                    break

                # Compute cross-correlation
                kernel = 1 - h
                corr = np.correlate(h, kernel, mode="full") * dx
                k_max = np.argmax(corr)

                # Compute gradient
                shift = k_max - (n - 1)
                h_shifted = np.roll(h, shift)
                gradient = (1 - h_shifted).copy()

                # Update h
                h -= lr * gradient[:n]
                
                # Feasibility repair
                h = np.clip(h, 0, 1)
                h = h / h.sum() * (n / 2)

                # Compute current objective
                corr = np.correlate(h, 1 - h, mode="full") * dx
                c5_candidate = np.max(corr)

                # Update best solution if improved
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    # Save the interpolated version at next resolution
                    if n < max(resolution_sequence):
                        best_h = h

        # Update the best solution
        if best_c5 < evaluate_erdos_solution(evaluate_erdos_solution, h):
            best_h = h
            best_c5 = evaluate_erdos_solution(evaluate_erdos_solution, h)

    # Return best solution found
    return best_h.tolist(), best_c5, len(best_h)
