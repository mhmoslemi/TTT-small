import numpy as np
import evaluate_erdos_solution as evaluate
import time
import cvxpy as cp
import math

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    t0 = time.time()
    dx = 2.0 / 42
    h = initial_h_values / 42 * 10  # initial_h_values is already in [0,42]
    h = h.astype(np.float64)
    h /= 2.0  # rescaling for easier gradient descent

    n_points = len(h)
    best_h = h.copy()
    best_c5 = float('inf')
    last_update_time = t0
    iter_count = 0

    def evaluate_c5(h):
        h_flat = h.flatten()
        corr = np.correlate(h_flat, 1 - h_flat, mode="full")
        return np.max(corr * dx)

    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        total = h.sum()
        h = h * (n_points / 2) / total
        return h

    learning_rates = [0.01 * (0.95 ** (i // 100)) for i in range(1000)]
    l_rate_idx = 0

    while time.time() - t0 < budget_s - 1:
        iter_count += 1

        h = project_to_feasible(h)

        # compute max overlap k
        corr = np.correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)

        # gradient w.r.t. h
        grad = np.zeros_like(h)
        if 0 <= k_max - n_points < len(grad):
            grad[k_max - n_points:] = (1 - h[:len(grad) - (k_max - n_points)])
        elif k_max < n_points:
            grad[:k_max] = (1 - h[k_max:])
        else:
            grad = (1 - h)

        # apply gradients with learning rate
        lr = learning_rates[l_rate_idx % len(learning_rates)]
        h -= lr * grad * dx

        # project to feasible region
        h = project_to_feasible(h)

        c5 = evaluate_c5(h)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_update_time = time.time()

        # cooldown if stuck
        if (time.time() - last_update_time > 30 and iter_count > 100):
            print(f"Stuck. Resetting towards model {best_c5}")
            h = project_to_feasible(best_h + np.random.normal(0, 0.05, n_points))

        # update learning rate schedule
        if iter_count % 50 == 1:
            l_rate_idx += 1

        # enforce final feasible and objective consistency check
        if time.time() - t0 > budget_s - 2:
            break

    best_h = project_to_feasible(best_h)
    h_values = best_h.copy()
    return (h_values, best_c5, n_points)
