import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    n_points = 256
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
        h = np.tile(h, (n_points // len(h), 1)).flatten()[:n_points]
        h = h / h.sum() * (n_points / 2)
        h = np.clip(h, 0, 1)
    else:
        # fallback to simple starting point
        h = np.zeros(n_points)
        h[n_points // 2 - n_points // 4: n_points // 2 + n_points // 4] = 0.5

    h = h / h.sum() * (n_points / 2)
    h = np.clip(h, 0, 1)
    
    max_iterations = 1000
    for it in range(max_iterations):
        if time.time() - start_time > budget_s - 2:
            break

        # Compute all overlaps (1 - h(x)h(x+k))
        hh = np.correlate(h, h, mode='full')
        oh = np.correlate(h, 1 - h, mode='full')
        coh = np.correlate(1 - h, h, mode='full')
        ooh = np.correlate(1 - h, 1 - h, mode='full')
        # Only need to evaluate overlap = correlate(h, 1 - h, mode='full') * dx
        overlap = oh * dx
        
        # Compute objective value
        c5_bound = np.max(overlap)
        
        # Select worst shift to optimize
        k_max = np.argmax(overlap)

        # Perturb to compute gradient
        h_perturb = h.copy()
        grad_indices = np.maximum(0, np.minimum(n_points - 1, np.arange(n_points) + k_max - n_points // 2))
        grad_effect = 1.0 - 2.0 * h[grad_indices]  # gradient sign at h[i]
        grad = np.zeros_like(h)
        grad[grad_indices] = grad_effect

        # Normalize gradient
        grad = grad / np.sqrt(np.sum(grad ** 2))

        # Take step
        step_size = 1.0
        for _ in range(5):
            h_candidate = h - step_size * grad
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate / np.sum(h_candidate) * (n_points / 2)  # Re-normalize
            overlap_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full') * dx
            c5_candidate = np.max(overlap_candidate)
            if c5_candidate < c5_bound:
                h = h_candidate
                c5_bound = c5_candidate
                break
            step_size *= 0.5

    h_values = h.tolist()
    return h_values, c5_bound, n_points
