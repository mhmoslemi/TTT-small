import numpy as np
import scipy.optimize as opt
import time
import math

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h0 = np.array(initial_h_values).copy()
    dx = 2.0 / n_points
    dx_scaled = 1.0 / n_points  # correlate is unnormalized

    start = time.time()

    def objective(h):
        h = h.reshape(-1)
        max_overlap = 0.0
        h1 = h
        h2 = 1.0 - h
        for k in range(-n_points + 1, n_points):
           rolled = np.roll(h2, shift=k)
            overlap = np.dot(h1, rolled)
            max_overlap = max(max_overlap, overlap)
        return max_overlap * dx  # return scaled value for C5

    def project(h):
        h = np.clip(h, 0., 1.)
        sum_h = np.sum(h)
        if not math.isclose(sum_h, n_points / 2, abs_tol=1e-6):
            scale = (n_points / 2) / sum_h
            h = h * scale
            h = np.clip(h, 0., 1.)
        return h
    
    def bounds_and_constraints(x):
        x = project(x)
        return x

    # Optimization with constraints
    result = opt.minimize(
        objective,
        h0,
        method='L-BFGS-B',
        bounds=[(0, 1)] * n_points,
        callback=lambda x: x  # mandatory for minimize with no callback
    )

    # Retrieve best solution found within the time budget
    elapsed = time.time() - start
    while elapsed < budget_s:
        print(f"Elapsed {elapsed:.2f}s of {budget_s:.2f}s.")
        next_h = project(result.x)
        val = objective(next_h)
        print(f"Current C5 value: {val:.6f}")
        result = opt.minimize(
            objective,
            next_h,
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            callback=lambda x: x
        )
        elapsed = time.time() - start

    h_best = project(result.x)
    c5_best = objective(h_best)

    return h_best, c5_best, n_points
