import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Use the initial H if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Binary initialization with equal area
        n_points = 47
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h[mid:] = 0.0
        h *= (n_points / 2) / np.sum(h)  # Ensure area is 1

    # Check constraint: ∑h = n_points / 2
    total_area = np.sum(h)
    h = h / total_area * (n_points / 2)

    def objective(x):
        x = x.reshape(-1)
        dx = 2.0 / n_points
        h_f = x
        overlaps = []
        for k in range(1, n_points):
            if k >= len(h_f):
                valid_idx = np.where((np.arange(n_points) + k) < n_points)[0]
                overlap = np.sum(h_f[valid_idx] * (1 - h_f[valid_idx + k])) * dx
            else:
                overlap = np.sum(h_f[:n_points - k] * (1 - h_f[k:])) * dx
            overlaps.append(overlap)
        return max(overlaps)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    def constraint_bounds(x):
        return x

    cons = ({'type': 'eq', 'fun': constraint_sum},
            {'type': 'ineq', 'fun': lambda x: x - 0},
            {'type': 'ineq', 'fun': lambda x: 1 - x})
    
    h_history = [h.copy()]
    best_h = h.copy()
    best_c5 = objective(h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        result = minimize(
            lambda x: objective(x),
            x0=h,
            constraints=cons,
            bounds=[(0, 1)] * n_points,
            method='SLSQP',
            options={'maxiter': 20}
        )

        h_new = result.x
        c5_new = objective(h_new)
        h = h_new

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

        h_history.append(h_new.copy())

        if time.time() - start_time > budget_s - 5:
            break

    # Final check of constraints and re-evaluation
    best_h = np.clip(best_h / np.sum(best_h) * (n_points / 2), 0, 1)
    final_c5 = objective(best_h)
    return best_h, final_c5, n_points
