import numpy as np
import time
from scipy.interpolate import interp1d
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    # Tracking best results so far
    h_best = None
    c5_best = float('inf')
    n_points_best = 14

    # Adaptive time plan
    def elapsed():
        return time.time() - start_time

    def compute_overlap(h, dx):
        C = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(C)

    def repair_feasibility(h, n_points):
        h = np.clip(h, 0, 1)
        h = h / h.sum() * (n_points / 2)
        return h

    def simulated_annealing(h, T=1.0, alpha=0.999, budget_steps=1000):
        h_anneal = h.copy()
        C_best = compute_overlap(h_anneal, 2.0 / len(h_anneal))
        h_anneal_best = h_anneal.copy()
        for _ in range(budget_steps):
            if elapsed() >= budget_s:
                break
            perturbation = np.random.normal(0, 1, len(h_anneal))
            h_candidate = h_anneal + 0.1 * T * perturbation
            h_candidate = repair_feasibility(h_candidate, len(h_candidate))
            C_candidate = compute_overlap(h_candidate, 2.0 / len(h_candidate))

            if C_candidate < C_best - 1e-8 or np.random.rand() < np.exp((C_best - C_candidate) / T):
                h_anneal, C_best = h_candidate, C_candidate
                if C_candidate < C_best:
                    h_anneal_best = h_candidate

            T *= alpha
        return h_anneal_best, C_best

    def compute_gradient(h):
        n_points = len(h)
        dx = 2.0 / n_points
        C = np.correlate(h, 1 - h, mode='full') * dx
        k_max = np.argmax(C)
        # Compute the optimal shift index
        h_shifted = np.roll(h, k_max - (n_points - 1))
        gradient = 1 - h_shifted
        return gradient[:n_points]

    def gradient_descent(h, lr=0.05, steps=1000):
        h_gd = h.copy()
        C_best = compute_overlap(h_gd, 2.0 / len(h_gd))
        for _ in range(steps):
            if elapsed() >= budget_s:
                break
            gradient = compute_gradient(h_gd)
            h_gd -= lr * gradient
            h_gd = repair_feasibility(h_gd, len(h_gd))
            C = compute_overlap(h_gd, 2.0 / len(h_gd))
            if C < C_best:
                C_best = C
        return h_gd, C_best

    start_time = time.time()

    # Initial setup
    initial_n_points = 14
    if 'initial_h_values' in globals():
        # Use existing construction from parent process
        h = initial_h_values[:initial_n_points] if len(initial_h_values) >= initial_n_points else initial_h_values
        h = repair_feasibility(h, initial_n_points)
    else:
        h = np.random.uniform(0, 1, initial_n_points)
        h = h / h.sum() * (initial_n_points / 2)

    h, c5 = simulated_annealing(h, T=1.0, alpha=0.999, budget_steps=1000)
    h, c5 = gradient_descent(h, lr=0.05, steps=1000)

    if c5 < c5_best:
        h_best = h
        c5_best = c5
        n_points_best = initial_n_points

    # Refinement stage: double the resolution step by step
    resolutions = [28, 56]
    for n_next in resolutions:
        if elapsed() >= budget_s:
            break
        # Interpolate the current h to the next resolution
        x = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_next)
        f = interp1d(x, h_best, kind='cubic', bounds_error=False, fill_value="extrapolate")
        h_interpolated = f(x_new)
        h_interpolated = repair_feasibility(h_interpolated, n_next)

        h, c5 = simulated_annealing(h_interpolated, T=1.0, alpha=0.999, budget_steps=1000)
        h, c5 = gradient_descent(h, lr=0.05, steps=1000)

        if c5 < c5_best:
            h_best = h
            c5_best = c5
            n_points_best = n_next

        if elapsed() >= budget_s:
            break

    # Final post-processing and validation
    if h_best is None:
        h_best = h  # fallback to most recent state
        c5_best = c5
        n_points_best = len(h)

    h_best = repair_feasibility(h_best, n_points_best)
    h_best /= h_best.sum() * (n_points_best / 2)  # Final repair
    c5_best = compute_overlap(h_best, 2.0 / n_points_best)

    return h_best, c5_best, n_points_best
