import numpy as np
import cvxpy as cp
import time
from math import isclose

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Start from initial_h_values if available, else uniform h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
        # Normalize sum(h) exactly to n_points / 2
        total = np.sum(h)
        h *= (n_points / 2) / total
    else:
        n_points = 70
        h = 0.5 * np.ones(n_points)

    # Ensure feasibility explicitly
    def project_h(h):
        h_projected = cp.Variable(shape=h.shape)
        objective = cp.Minimize(cp.sum_squares(h_projected - h))
        constraints = [
            h_projected >= 0,
            h_projected <= 1,
            cp.sum(h_projected) == n_points / 2,
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS)
        if h_projected.value is None:
            h_projected.value = h.copy()
        return np.clip(h_projected.value, 0, 1)

    h = project_h(h)

    # Evaluate best C5
    def compute_C5(h):
        h = h - 1e-7  # tiny negative shift to maintain strict < 1
        corr = np.correlate(h, 1 - h, mode='full')
        # Normalize to [0,2] with weight dx = 2 / n_points
        dx = 2.0 / n_points
        return (corr * dx).max()

    best_h = h.copy()
    best_c5 = compute_C5(best_h)
    last_improvement = 0
    t_start = time.time()
    patience = 100

    while time.time() - t_start < budget_s:
        # Compute gradient
        C5 = compute_C5(h)
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr)

        # Compute grad h * (1 - h_{i+k}) and - h_{i-k} * 1
        grad = np.zeros_like(h)
        if k < n_points:
            grad -= (1 - h[k:])  # Wrapping overlap
        other_k = k - n_points
        if other_k > -n_points:
            grad[:n_points + other_k] -= h[abs(other_k):]

        # Normalize gradient
        grad_len = np.linalg.norm(grad)
        if grad_len == 0:
            break  # converged
        grad /= grad_len

        # Line search
        eta = 0.01
        new_h = h - eta * grad
        new_h = project_h(new_h)
        new_C5 = compute_C5(new_h)

        # Try backtracking for improvement
        attempts = 0
        while new_C5 > C5 + 5e-7 and attempts < 10 and eta > 1e-8:
            eta *= 0.5
            new_h = h - eta * grad
            new_h = project_h(new_h)
            new_C5 = compute_C5(new_h)
            attempts += 1

        if new_C5 < best_c5 - 1e-10:
            best_c5 = new_C5
            best_h = new_h.copy()
            last_improvement = 0
        else:
            last_improvement += 1

        h = new_h  # Always keep trying new solution
        if last_improvement >= patience:
            break  # Stall for patience iterations

    return (best_h.tolist(), best_c5, n_points)
