import numpy as np
import cvxpy as cp
import time
from math import isclose
import copy

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h_values = globals().get("initial_h_values", None)
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 70
        h = np.full(n_points, 0.5, dtype=np.float64)

    best_h = h.copy()
    best_c5 = evaluate_c5(h, n_points)

    start_time = time.time()
    learning_rate = 0.01
    decay_rate = 0.00001
    max_iter = int(budget_s * 0.8)
    h_best_list = []
    c5_best_list = []

    def evaluate_c5(h, n_points):
        dx = 2.0 / n_points
        k_overlaps = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(k_overlaps * dx)
        return c5

    for iter_idx in range(1, max_iter + 1):
        if time.time() - start_time >= budget_s:
            break

        lr = learning_rate / (1 + decay_rate * iter_idx)

        k_overlaps = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(k_overlaps)
        k = max_idx - h.size + 1

        grad = np.zeros_like(h)
        # Compute gradient for the worst k
        for i in range(h.size):
            j = (i + k) % h.size
            if j >= i:
                grad[i] += (1 - h[j])
                grad[j] -= h[i]
            else:
                grad[i] += (1 - h[j]) if j < h.size else 0
                grad[j] -= h[i] if j >= 0 else 0

        # Take step
        h_candidate = h - lr * grad

        # Projection to feasible region
        x = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum_squares(x - h_candidate))
        constraints = [
            x >= 0, x <= 1,
            cp.sum(x) == n_points / 2.0
        ]
        prob = cp.Problem(objective, constraints)
        result = prob.solve(solver=cp.OSQP, max_iter=10000)

        h = x.value
        if h is None:
            h = best_h.copy()
            continue

        current_c5 = evaluate_c5(h, n_points)

        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            h_best_list.append(copy.deepcopy(h))
            c5_best_list.append(current_c5)

    final_c5 = evaluate_c5(best_h, n_points)
    return (best_h.tolist(), final_c5, n_points)
