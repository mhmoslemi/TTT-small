import numpy as np
import math
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use the initial_h_values as the starting point
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points

    # Normalization to satisfy the constraint sum(h) = n_points / 2
    current_sum = np.sum(h)
    desired_sum = n_points / 2
    h = h * (desired_sum / current_sum)
    h = np.clip(h, 0, 1)

    best_h = np.copy(h)
    best_c5 = compute_c5(h, dx)

    update_count = 0
    no_improvement_count = 0
    max_no_improvement = 100
    learning_rate = 0.05

    while (time.time() - start_time) < budget_s:
        overlap = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap_idx = np.argmax(overlap)

        # Check for early stopping if no improvement for a while
        current_c5 = np.max(overlap)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.copy(h)
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:
                learning_rate *= 0.85
                no_improvement_count = 0

        # Compute gradient direction for negative of the overlap at max lag
        corr = np.correlate(np.sign(h), np.diagonal(1 - h)[max_overlap_idx:], mode='valid', old_behavior=False)
        grad = np.zeros_like(h)
        grad[:len(corr)] += corr
        grad = np.roll(grad, -max_overlap_idx) * (-1.0)

        # Update h in the gradient direction, project to [0, 1], and normalize
        h += learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (desired_sum / np.sum(h))

        update_count += 1

        # Early exit if progress is negligible
        if update_count % 1000 == 0:
            if math.isclose(best_c5, compute_c5(h, dx), rel_tol=1e-9, abs_tol=1e-9):
                break

    # Final validation and return
    final_h = np.copy(best_h)
    final_h = final_h * (desired_sum / np.sum(final_h))
    final_h = np.clip(final_h, 0, 1)
    final_c5 = compute_c5(final_h, dx)

    return list(final_h), final_c5, n_points

def compute_c5(h, dx):
    overlap = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(overlap)
