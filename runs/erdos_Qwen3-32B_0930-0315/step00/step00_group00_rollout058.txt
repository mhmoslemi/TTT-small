import numpy as np
import cvxpy as cp
import time
import random
from scipy.signal import correlate

# Global timer to track elapsed time
start_time = time.time()

def is_time_expired(budget_s):
    return time.time() - start_time > budget_s

def objective(h, n_points):
    dx = 2.0 / n_points
    compl = 1.0 - h
    corr = correlate(h, compl, mode="full")
    C5 = np.max(corr) * dx
    return C5

def clip_and_normalize(h, n_points):
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if total == 0:
        h[:] = 1.0 / n_points
    else:
        h *= (n_points / 2) / total
    return h

def tournament_selection(population, scores, tournament_size):
    indices = np.random.choice(len(population), tournament_size, replace=False)
    selected_index = indices[np.argmin([scores[i] for i in indices])]
    return population[selected_index]

def crossover(p1, p2, prob=0.7):
    if np.random.rand() > prob:
        return p1.copy()
    crossover_point = np.random.randint(len(p1))
    child = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
    return child

def mutate(individual, mutation_rate, temp=0.1, temperature=1.0):
    for i in range(len(individual)):
        if np.random.rand() < mutation_rate:
            noise = np.random.normal(0, temp)
            individual[i] += noise
    return clip_and_normalize(individual, len(individual))

def evaluate_population(population, n_points):
    return [objective(ind, n_points) for ind in population]

def find_best(population, scores):
    best_index = np.argmin(scores)
    return population[best_index], scores[best_index], best_index

def run(seed=42, budget_s=1100, n_points=45, pop_size=75):

        initial = clip_and_normalize(initial, n_points)

    best_h = initial.copy()
    best_c5 = objective(best_h, n_points)
    population = [initial]
    
    # Initialize remaining population
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = clip_and_normalize(h, n_points)
        population.append(h)

    scores = evaluate_population(population, n_points)
    best_h, best_c5, _ = find_best(population, scores)

    temperature = 100.0
    cooling_rate = 0.95
    gen = 0

    while not is_time_expired(budget_s):
        gen += 1
        new_population = []
        for _ in range(pop_size // 2):
            # Selection
            parent1 = tournament_selection(population, scores, int(0.1 * pop_size))
            parent2 = tournament_selection(population, scores, int(0.1 * pop_size))
            child = crossover(parent1, parent2)
            child = mutate(child, mutation_rate=0.05, temp=1.0, temperature=temperature)
            new_population.append(child)

        population = population + new_population
        scores = evaluate_population(population, n_points)

        # Trim to population size, keep better half
        indices = np.argsort(scores)
        population = [population[i] for i in indices[:pop_size]]
        scores = [scores[i] for i in indices[:pop_size]]

        # SA step
        current_best_h, current_best_c5, _ = find_best(population, scores)
        if current_best_c5 < best_c5:
            best_h = current_best_h.copy()
            best_c5 = current_best_c5

        # Annealing every 50 generations
        if gen % 50 == 0:
            # Try a random mutation of the best individual
            mutated = mutate(best_h, mutation_rate=0.2, temp=2.0, temperature=temperature)
            mutated_c5 = objective(mutated, n_points)
            if mutated_c5 < best_c5 or np.exp((best_c5 - mutated_c5) / temperature) > np.random.rand():
                best_h = mutated.copy()
                best_c5 = mutated_c5
            # Cool the temperature
            temperature *= cooling_rate

        # Periodically update best
        current_best_h, current_best_c5, _ = find_best(population, scores)
        if current_best_c5 < best_c5:
            best_h = current_best_h.copy()
            best_c5 = current_best_c5

    return best_h, best_c5, n_points
