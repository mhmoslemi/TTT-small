import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    dx = 2.0 / 70
    h0 = initial_h_values.copy()
    n_points = len(h0)

    def objective(h):
        h_corr = 1 - h
        corr = correlate(h, h_corr, mode='full') * dx
        return np.max(corr)

    def constraint(h):
        return np.sum(h) - n_points / 2

    def objective_with_penalty(h, lambda_p=1e4):
        return objective(h) + lambda_p * (constraint(h) ** 2)

    def gradient(h):
        h_corr = 1 - h
        corr_k = correlate(h, h_corr, mode='full') * dx
        k_max = np.argmax(corr_k)

        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] += (1 - h_corr[i + k_max]) * dx
            if i - k_max >= 0:
                grad[i] -= h[i - k_max] * dx
        return grad

    def objective_and_gradient(h):
        f = objective_with_penalty(h)
        g = gradient(h) + 2 * 1e4 * (constraint(h)) * np.ones_like(h)
        return f, g

    bounds = [(0, 1)] * n_points
    h_curr = np.clip(h0, 0, 1)
    h_curr = h_curr * (n_points / 2 / np.sum(h_curr))
    best_h = h_curr.copy()
    best_c5 = objective(best_h)
    last_log = time.time()
    start = time.time()

    while time.time() - start < budget_s:
        # L-BFGS-B iteration
        def obj_fg(h):
            return objective_and_gradient(h)

        res = minimize(obj_fg, x0=best_h, method='L-BFGS-B', jac=True, bounds=bounds, options={'maxiter': 5})
        if not res.success:
            break
        h_curr = res.x
        h_curr = np.clip(h_curr, 0, 1)
        h_curr = h_curr * (n_points / 2 / np.sum(h_curr))
        c5 = objective(h_curr)

        if c5 < best_c5:
            best_h = h_curr.copy()
            best_c5 = c5

        # Perturbation
        if np.random.rand() < 0.5 and time.time() - start > 300:
            h_curr += np.random.normal(0, 0.01, size=h_curr.shape)
            h_curr = np.clip(h_curr, 0, 1)
            h_curr = h_curr * (n_points / 2 / np.sum(h_curr))

        # Log status
        if time.time() - last_log > 10:
            last_log = time.time()

    return best_h.tolist(), best_c5, n_points
