import time

from numpy import correlate

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Use initial_h_values
    h = np.array(initial_h_values)
    n_points = len(h)
    h = h / h.sum() * n_points / 2  # Ensure average is 1/2

    max_iter = 10000000
    patience = 100
    learning_rate = 0.01
    best_h = h.copy()
    best_c5 = float('inf')
    no_improve_counter = 0

    for i in range(max_iter):
        current_time = time.time()
        if current_time - start_time > budget_s - 1:
            break

        # Compute the overlap for each shift
        h_flat = h.flatten()
        c = correlate(h_flat, 1 - h_flat, mode='full')
        c5 = c * (2.0 / n_points)

        # Find maximum overlap
        index_c5_max = np.argmax(c5)
        k_max = index_c5_max - n_points

        # Compute overlap gradient (numerical gradient)
        h1 = h_flat.copy()
        h2 = h_flat.copy()

        # Shift h1 left with wrap around
        h1_shifted_left = np.roll(h1, shift=-k_max)
        select_idx = (np.arange(n_points) % n_points)[(np.arange(n_points) + k_max) % n_points]
        grad1 = 1 - h1_shifted_left[select_idx]

        # Shift h2 right with wrap around
        h2_shifted_right = np.roll(h2, shift=k_max)
        grad2 = 1 - h2_shifted_right[select_idx]

        # Update h
        grad = (grad1 + grad2) / 2
        h = h - learning_rate * grad

        # Project to [0, 1] and normalize
        h = np.clip(h, 0, 1)
        h = h / h.sum() * n_points / 2

        # Evaluate current c5
        c = correlate(h, 1 - h, mode='full')
        c5 = c * (2.0 / n_points)
        current_c5 = np.max(c5)

        # Track best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve_counter = 0
        else:
            no_improve_counter += 1

        # Terminate if no improvement for patience steps
        if no_improve_counter >= patience:
            break

    return (best_h.tolist(), best_c5, n_points)
