def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Validate constraints
    if np.any(h < 0) or np.any(h > 1):
        return np.inf
    if abs(h.sum() - n_points / 2) > 1e-7:
        return np.inf
    # Compute cross-correlation of h and (1-h)
    s = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(s) * dx
    return c5

                for i in range(n_points):
                    j = (i + shift) % n_points
                    term = (1 - h[j]) if j < len(h) else 0
                    grad[i] += (term - 2 * h[i] * term)
            return grad

        h_new = h_array - learning_rate * grad

        h_new = np.clip(h_new, 0, 1)
