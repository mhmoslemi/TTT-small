import numpy as np
import cvxpy as cp
from time import time
import evaluate_erdos_solution
from math import log

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time()
    n_points = 42
    dx = 2.0 / n_points

    # Load predefined starting point
    h = np.array(initial_h_values).copy()
    h = h * (n_points / h.sum()) if h.sum() != 0 else np.ones(n_points) / n_points * (n_points / 2)
    h = np.clip(h, 0, 1)
    h *= (n_points / 2) / h.sum()

    mu = 1e4
    best_h = h.copy()
    best_c5 = float('inf')
    n_iter = 0

    for it in range(1000):
        if time() - t0 > budget_s:
            break
        n_iter += 1
        h_old = h.copy()

        # Compute the cross-correlation with (1 - h)
        corr = np.correlate(h, 1 - h, mode='same') * dx
        k_max = np.argmax(corr)
        max_corr = corr[k_max]

        # Setup convex optimization with barrier method
        h_cvx = cp.Variable(n_points)
        obj = max_corr + mu * cp.sum(cp.log(h_cvx) + cp.log(1 - h_cvx))
        obj = cp.Minimize(obj)
        constraints = [
            h_cvx >= 1e-6,
            h_cvx <= 1 - 1e-6,
            cp.sum(h_cvx) * dx == 1
        ]
        prob = cp.Problem(obj, constraints)
        try:
            result = prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                mu = max(mu * 0.5, 1e-6)
                continue
            h = h_cvx.value
            h = np.clip(h, 0, 1)
            h *= (n_points / 2) / h.sum()
            c5_current = max(np.correlate(h, 1 - h, mode='full') * dx)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h.copy()
        except:
            pass

        # Decay barrier parameter
        if it % 50 == 0:
            mu = max(mu * 0.5, 1e-8)

    h = best_h
    c5 = max(np.correlate(h, 1 - h, mode='full') * dx)
    h *= (n_points / 2) / h.sum()
    h = np.clip(h, 0, 1)
    return h.tolist(), c5, n_points
