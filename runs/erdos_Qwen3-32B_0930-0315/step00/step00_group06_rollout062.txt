import numpy as np
import cvxpy as cp
import time
from functools import partial

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None

    # Configuration
    n_points = 256
    dx = 2.0 / n_points
    total_mass = n_points // 2
    h = np.array(initial_h_values).copy()

    def compute_c5(h):
        h_flat = h.reshape(-1)
        ones_minus_h = 1.0 - h_flat
        conv_result = np.correlate(h_flat, ones_minus_h, mode='full')
        c5 = np.max(conv_result) * dx
        return c5

    def optimize_magnitudes(h_refined, k_max):
        h_cvp = cp.Variable(n_points)
        
        mass_constr = [cp.sum(h_cvp) == total_mass]
        box_constr = [0 <= h_cvp, h_cvp <= 1]
        lagged = np.roll(h_refined, k_max)
        locconstr = [h_cvp[i] >= 0, h_cvp[i] <= 1 for i in range(n_points)]

        objective = cp.Minimize(cp.sum(h_cvp * (1 - lagged)))
        constraints = mass_constr + box_constr + locconstr
        prob = cp.Problem(objective, constraints)
        
        prob.solve(solver=cp.OSQP, max_iter=10000)
        
        if h_cvp.value is not None:
            h_cvp = np.clip(h_cvp.value, 0, 1).flatten()
            return h_cvp, compute_c5(h_cvp)
        else:
            return h_refined, compute_c5(h_refined)

    best_h = np.clip(h.astype(float), 0, 1)
    best_c5 = compute_c5(best_h)

    iter_count = 0
    best_iter = 0
    smooth_count = 0

    while time.time() - start_time < budget_s - 1:
        iter_count += 1

        c5 = compute_c5(best_h)
        conv_result = np.correlate(best_h, 1 - best_h, mode='full')
        for k_idx in range(len(conv_result)):
            if conv_result[k_idx] > 0:
                k_max = k_idx - best_h.size + 1
                break

        lagged_h = np.roll(best_h, k_max)

        # Perturb with Gaussian noise every 200 steps
        if iter_count % 200 == 0:
            best_h += np.random.normal(0.0, 0.025, n_points)
            best_h = np.clip(best_h, 0, 1)
            norm_c = np.sum(best_h) / n_points
            scale_factor = total_mass / np.sum(best_h)
            best_h *= scale_factor
            best_h = np.clip(best_h, 0, 1)

        # Optimize magnitudes
        optimized_h, new_c5 = optimize_magnitudes(best_h, k_max)

        if new_c5 < best_c5:
            best_h = optimized_h
            best_c5 = new_c5
            best_iter = iter_count
            smooth_count = 0
        else:
            smooth_count += 1

        # Convergence heuristic: stop if no improvement for many steps
        if smooth_count > 2000:
            break

    return best_h, best_c5, n_points
