import numpy as np
import time

def compute_c5_and_kmax(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    kmax = np.argmax(corr)
    if kmax >= n:
        kmax -= n  # Ensure it corresponds to shift in [1 - n, n - 1]
    return c5, kmax

def project_h(h):
    h = np.clip(h, 0, 1)
    total = h.sum()
    target = n_points / 2
    if abs(total - target) < 1e-6:
        return h
    # Adjust scale to meet target sum
    alpha = target / total
    h = np.clip(h * alpha, 0, 1)

    # Iteratively clip and scale if needed
    count = 0
    while abs(h.sum() - target) > 1e-6 and count < 1000:
        h = np.clip(h, 0, 1)
        delta = h.sum() - target
        if abs(delta) > 1e-9:
            scale_factor = 1 - delta / n_points
            h = np.clip(h * scale_factor, 0, 1)
        count += 1
    return h

def optimize_h(h_start, max_iterations=10000, lr=0.01):
    h = np.copy(h_start)
    best_h_opt = h.copy()
    best_c5_opt = evaluate_erdos_solution(h)
    t0 = time.time()

    for i in range(max_iterations):
        c5, kmax_all = compute_c5_and_kmax(h)
        if c5 < 0.380876:  # Desired target
            best_h_opt = h.copy()
            break

        kmax = kmax_all

        n = len(h)
        # Compute gradient
        grad = np.zeros(n)
        for j in range(n):
            kplus = (j + kmax) % n
            kminus = (j - kmax) % n
            grad[j] = 1.0 - h[kplus] - h[kminus]

        h -= lr * grad
        h = project_h(h)
        c5_new, _ = compute_c5_and_kmax(h)
        if c5_new < best_c5_opt:
            best_c5_opt = c5_new
            best_h_opt = h.copy()

        if time.time() - t0 > budget_s - 1:
            break
        elif i % 100 == 0 or i < 100:
            # Adjust learning rate if needed
            if i > 0 and c5_new > best_c5_opt and abs(c5_new - best_c5_opt) < 1e-4:
                lr *= 0.9  # Decay learning rate on stagnation
                h = best_h_opt.copy()

    return best_h_opt, best_c5_opt

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h_opt, c5_opt = optimize_h(initial_h_values)
    h_values_numpy = np.copy(h_opt)

    # Ensure final solution meets all constraints
    assert np.all(h_opt >= 0) and np.all(h_opt <= 1), "Invalid range in h"
    assert abs(h_opt.sum() - n_points / 2) < 1e-6, "Integral of h != 1"
    assert 0 < c5_opt < 1, "Invalid overlap integral result"

    return h_values_numpy, c5_opt, n_points
