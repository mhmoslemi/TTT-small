import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 256
    dx = 2.0 / n_points
    initial_h = initial_h_values
    h = initial_h.copy().astype(float) if 'initial_h_values' in globals() else None

    # Use parent solution if available
    if h is not None:
        # Ensure normalization
        h = h * (n_points / 2) / h.sum()
    else:
        base_h = np.random.choice([1.0, 0.0], size=n_points, p=[0.5, 0.5])
        base_h = base_h * (n_points / 2) / base_h.sum()
        h = base_h

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5_value = np.max(corr) * dx
        return c5_value

    best_c5 = compute_c5(h)
    best_h = h.copy()

    n_rounds = 0
    max_no_improvement = 100
    no_improvement = 0
    perturb_freq = 100
    lr_pos = 0.01
    lr_mag = 0.1

    while time.time() - start_time < budget_s:
        n_rounds += 1
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1

        if no_improvement >= max_no_improvement:
            break

        if n_rounds % perturb_freq == 0 or n_rounds == 1:
            # Add small random perturbation to avoid local optima
            h += np.random.normal(0, 0.05, size=h.shape)
            h = np.clip(h, 0.0, 1.0)
            h *= (n_points / 2) / h.sum()

        # Phase 1: Position adjustment
        corr = np.correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        shifts = np.arange(n_points)

        # Create shifted target window from k_max
        target = np.zeros(n_points)
        max_offset = min(k_max, n_points - k_max)
        for i in range(n_points):
            shifted_i = (i + k_max) % n_points
            target[i] = 1 - h[shifted_i]

        # Compute gradient of the overlap for position (index) shifts
        dh = np.zeros(n_points)
        dh[:] = -2 * (h - h * h - 0.5 * h * h)
        dh = dh * (target - h)

        dh = np.roll(dh, -k_max)
        dh = np.roll(dh, n_points - k_max)

        h += lr_pos * dh
        h = np.clip(h, 0.0, 1.0)
        h *= (n_points / 2) / h.sum()

        # Phase 2: Magnitude adjustment via convex optimization
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(h_var * (1 - np.roll(h_var, k_max - n_points // 2))) * dx)
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2]
        prob = cp.Problem(objective, constraints)
        try:
            result = prob.solve(solver=cp.GUROBI, verbose=False)
            h = h_var.value.copy()
        except:
            # Fallback if solver fails
            pass

        h = np.clip(h, 0.0, 1.0)
        h *= (n_points / 2) / h.sum()

        if n_rounds % 100 == 0:
            c5 = compute_c5(h)
            print(f"Round {n_rounds}, c5: {c5:.6f}")

    c5 = compute_c5(best_h)
    return best_h, c5, n_points
