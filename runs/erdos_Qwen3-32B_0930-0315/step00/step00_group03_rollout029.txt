import numpy as np
import cvxpy as cp
import time
from scipy import optimize

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2.0

    def constraint_upper(h_flat):
        return 1.0 - h_flat

    def constraint_lower(h_flat):
        return h_flat

    def project(h):
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum_squares(h_var - h))
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2.0]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS)
        return h_var.value

    while time.time() - start_time < budget_s:
        k_max = None
        C5 = float('inf')
        for k in range(n_points - 1, n_points + 2):  # k from support range
            if k >= 0 and k < n_points:
                align = np.roll(1 - h, k)
                overlap = np.sum(h * align) * dx
            else:
                h_shifted = np.zeros(n_points)
                if k >= 0:
                    h_shifted[:n_points - k] = 1 - h[k:]
                else:
                    h_shifted[-k:] = 1 - h[:k]
                overlap = np.sum(h * h_shifted) * dx
            if overlap > C5:
                C5 = overlap
                k_max = k
        if C5 <= best_c5:
            best_h = h.copy()
            best_c5 = C5
        if C5 < 1e-10 or time.time() - start_time >= budget_s:
            break
        # Gradient
        grad = np.zeros(n_points)
        if k_max >= 0 and k_max < n_points:
            grad += (1 - h[(np.arange(n_points) + k_max) % n_points]) * (1 if k_max < n_points else -1)
        else:
            if k_max >= n_points:
                grad[:(n_points - k_max)] += (1 - h[k_max:])
            else:  # k_max < 0
                grad[-k_max:] += (1 - h[: -k_max])
        if k_max >= n_points:
            grad[k_max:] -= h[:(n_points - k_max)]
        elif k_max < 0:
            grad[: -k_max] -= h[-k_max:]
        grad *= dx
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad_unit = grad / grad_norm
        else:
            break
        learning_rate = max(1e-3, 1e-1 / (np.abs(grad).sum() + 1e-6))
        h_candidate = h - learning_rate * grad_unit
        h = project(h_candidate)

        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2.0)

    h = project(h)
    C5_final = np.max(np.correlate(h, 1 - h, mode='full') * dx)
    return h.flatten(), C5_final, n_points
