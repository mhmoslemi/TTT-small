import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Use the given initial construction
    h = np.copy(initial_h_values)
    n_points = h.shape[0]
    dx = 2.0 / n_points

    # Project function to ensure it meets the constraint
    def project(h):
        h_clipped = np.clip(h, 0, 1)
        s = np.sum(h_clipped)
        h_scaled = h_clipped * (n_points / 2) / s
        return h_scaled

    h = project(h)

    best_h = np.copy(h)
    best_c5 = float('inf')
    iters = 0
    alpha = 0.1  # step size

    while time.time() - start_time < budget_s:
        iters += 1
        hh = h
        h_t = np.array([1 - hh, 1 - hh])[::-1]
        correl = scipy.signal.correlate(h, h_t, mode='full')

        def valid_corr(h):
            vcorr = np.zeros_like(correl)
            for k in range(1, n_points):
                overlap = hh * (1 - np.roll(hh, k))
                vcorr[k] = np.sum(overlap) * dx
            for k in np.where(correl.shape[0] - k <= n_points)[0]:
                k_new = correl.shape[0] - k
                if 1 <= k_new < n_points:
                    overlap = hh * (1 - np.roll(hh, -k_new))
                    vcorr[k] = np.sum(overlap) * dx
            return vcorr

        c5 = valid_corr(h)
        k_max = np.argmax(c5)
        current_c5 = c5[k_max]

        if current_c5 < best_c5:
            best_h[:] = h
            best_c5 = current_c5
            alpha = max(alpha * 1.001, 0.01)

        if iters % 50 == 0:
            noise = 0.005 * np.random.randn(n_points)
            h += noise
            h = project(h)
            alpha = max(alpha * 0.5, 0.001)  # reduce step size to fine-tune

        # Compute gradient only at max C_5 lag
        k = k_max
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + k < n_points:
                grad[i] += (1 - h[i + k]) * dx
        for i in range(n_points):
            if i - k >= 0:
                grad[i] -= h[i - k] * dx

        h -= alpha * grad
        h = project(h)

        if abs(best_c5 - current_c5) < 1e-7:
            break

    return best_h, best_c5, n_points
