import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    dx = 2 / 70
    n_points = 70
    dx_const = dx
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    assert 'initial_h_values' in globals()
    h_best = initial_h.copy()
    c5_best = evaluate_erdos_solution(h_best, dx=dx_const)
    start_time = time.time()

    h = initial_h.copy()
    lr = 0.01
    max_iter = 10000
    no_improvement_counter = 0
    max_no_improvement = 100

    for iter_index in range(max_iter):
        t0 = time.time()

        # Compute overlap for all shifts
        corr = correlate(h, 1 - h, mode='full')
        corr = corr * dx_const
        # Only consider k in [1, n_points-1], k=0 is not relevant for the problem
        relevant_corr = corr[corr.shape[0] // 2 + 1: corr.shape[0] // 2 + n_points]
        current_c5 = np.max(relevant_corr)

        if current_c5 < c5_best:
            c5_best = current_c5
            h_best = h.copy()
            no_improvement_counter = 0
            print(f"Iter {iter_index}, Improved C5: {c5_best}")
        else:
            no_improvement_counter += 1

        if no_improvement_counter >= max_no_improvement:
            break

        if time.time() - start_time > budget_s:
            break

        # Find worst shift k
        k = np.argmin(relevant_corr) + 1  # Since max overlaps correspond to min C₅
        # Ensure this k is within the relevant range
        k = min(k, n_points - 1)

        # Compute gradient for shift k
        grad = np.zeros_like(h)
        for i in range(n_points):
            if k + i < n_points:
                grad[i] += 1 - h[k + i]
            if i - k >= 0:
                grad[i] -= h[i - k]
        # Normalize gradient with dx
        grad *= dx_const

        # Update h
        h = h - lr * grad

        # Constrain h in [0, 1]
        h = np.clip(h, 0, 1)

        # Project back to feasible region: sum(h) == n_points / 2
        x = cp.Variable(n_points)
        obj = cp.Minimize(cp.sum_squares(x - h))
        constraints = [0 <= x, x <= 1, cp.sum(x) == n_points / 2]
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.GUROBI if 'GUROBI' in cp.installed_solvers() else cp.SCIP)
        h = np.array(x.value)

        # Update learning rate
        lr = 0.01 / (1 + iter_index / 1000)

    # Final validation
    assert (
        abs(np.sum(h_best) - n_points / 2) < 1e-6
        and np.all((0 <= h_best) & (h_best <= 1))
    ), "Invalid h values or constraint violation."

    return h_best.tolist(), c5_best, n_points
