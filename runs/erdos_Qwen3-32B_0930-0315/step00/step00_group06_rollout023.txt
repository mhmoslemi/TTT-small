import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    # Use initial_h_values directly
    global initial_h_values
    h = np.copy(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize if necessary
    h *= (target_sum / h.sum())

    def compute_c5(h):
        ones_h = 1.0 - h
        correl = np.correlate(h, ones_h, mode="full")
        return np.max(correl) * dx

    # Optimization parameters
    np.random.seed(seed)
    start_time = time.time()
    max_steps = int(3600 * 1000 / budget_s)  # 1ms steps
    lr = 1.0
    patience = 100
    best_c5 = compute_c5(h)
    best_h = np.copy(h)
    step = 0

    while time.time() - start_time < budget_s:
        step += 1
        if step > max_steps:
            break

        # Compute max overlap lag
        correl = np.correlate(h, 1 - h, mode="full")
        max_index = np.argmax(correl)
        k_max = max_index - (n_points - 1)

        # Calculate gradient based on the largest overlap
        grad = np.zeros_like(h)
        if k_max < 0:
            window = h[-k_max:]
            grad[-k_max:] = 1 - h[-k_max:]
        elif k_max >= 0:
            window = h[:n_points - k_max]
            grad[:n_points - k_max] = 1 - h[:n_points - k_max]

        # Move in direction of gradient
        h += lr * grad
        h = np.clip(h, 0, 1)

        # Re-normalize to maintain constraint
        h *= (target_sum / h.sum())

        # Check for improvement
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.copy(h)
            lr *= 1.1
            patience = 100
        else:
            lr *= 0.97
            patience -= 1
            if patience <= 0:
                lr *= 0.5
                patience = 100

    return (best_h, best_c5, n_points)
