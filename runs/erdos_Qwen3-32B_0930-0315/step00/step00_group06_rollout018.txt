import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    
    n_points = 256
    dx = 2.0 / n_points
    h = initial_h_values
    h /= np.sum(h) * dx * 2  # Normalize to satisfy sum(h) == n_points / 2
    
    start_time = time.time()
    total_iterations = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(h, n_points, dx)
    
    while time.time() - start_time < budget_s - 1:
        total_iterations += 1
        overlaps = np.correlate(h, 1 - h, mode="full")
        current_c5 = np.max(overlaps) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Gradient descent-style update
        k_max = np.argmax(overlaps)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if k_max - i >= 0 and k_max - i < n_points:
                grad[i] = -h[k_max - i]  # Derivative of h(x)(1 - h(x+k))
            if i + k_max < n_points:
                grad[i] = - (1 - h[i + k_max])
        grad += 2 * h  # Regularization
        
        # Step update with projection to [0, 1] and normalization
        step_size = 0.01
        h = np.clip(h + step_size * grad, 0, 1)
        h *= n_points / (2 * h.sum())
        
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, n_points, dx):
    overlaps = np.correlate(h, 1 - h, mode="full")
    return np.max(overlaps) * dx
