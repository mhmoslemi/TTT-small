import numpy as np
from time import time
from math import modf

def evaluate_erdos_solution(h_values):
    h = np.array(h_values).astype(np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    h = h / np.sum(h) * (n_points / 2)
    h_clipped = np.clip(h, 0.0, 1.0)
    full_corr = np.correlate(h_clipped, 1 - h_clipped, mode="full")
    c5 = np.max(full_corr) * dx
    constraints_violated = not (np.all(0 <= h_clipped) and np.all(h_clipped <= 1))
    return {"c5": c5, "constraints_violated": constraints_violated}

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time()
    n_points = 160  # moderate resolution for balance in feasibility and efficiency
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).astype(np.float64)
        h *= (n_points / 2) / np.sum(h)  # normalize to ensure the integral = 1
    else:
        h = np.random.uniform(0.0, 1.0, n_points)
        h *= (n_points / 2) / np.sum(h)  # normalize

    best_h = h.copy()
    best_c5 = np.inf
    iters_no_improve = 0

    window = 1500  # only keep relevant overlaps; see indices around max(C) up to a limit
    eps = 1e-8  # perturbation threshold for converged gradient
    alpha = 0.5
    def normalize_h(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        h_total = np.sum(h_clipped)
        if abs(h_total - n_points / 2) > 1e-4:
            h_clipped *= (n_points / 2) / h_total
        return h_clipped

    def compute_c5(h):
        h_clipped = normalize_h(h)
        full_corr = np.correlate(h_clipped, 1 - h_clipped, mode="full")
        c5 = np.max(full_corr) * dx
        return c5

    def compute_gradient(h, k_max):
        grad = np.zeros_like(h)
        for j in range(n_points):
            # j - k_max can be negative → mod it to cyclic
            k = k_max
            j_cyc = j + n_points
            val = (1 - h[(j + k) % n_points]) - h[(j - k + n_points) % n_points]
            grad[j % n_points] += val
        return grad

    while time() - start_time < budget_s:
        full_corr = np.correlate(normalize_h(h), 1 - normalize_h(h), mode="full")
        k_max = np.argmax(full_corr)
        grad = compute_gradient(h, k_max)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < eps:
            iters_no_improve += 1
        else:
            alpha = max(alpha, 1e-3)  # dampen alternate steps if too aggressive
            h -= alpha * grad
            iters_no_improve = 0
        h = normalize_h(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        elif iters_no_improve % 500 == 0:
            # random perturbation when stuck or making no progress
            mask = np.random.rand(n_points) < 0.1  # perturb 10% of entries
            perturbation = np.random.uniform(-0.01, 0.01, size=n_points)
            h[mask] += perturbation[mask]
            h = normalize_h(h)
        iters_no_improve += 1

    h_refined = best_h
    result = evaluate_erdos_solution(h_refined)
    if result['constraints_violated']:
        return initial_h_values, 0.38085, len(initial_h_values)
    return h_refined, result['c5'], n_points
