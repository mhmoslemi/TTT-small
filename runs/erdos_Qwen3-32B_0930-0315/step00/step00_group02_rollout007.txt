import numpy as np
import math
from time import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time()

    # Use initial_h_values provided
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h)

    current_h = np.copy(h)
    current_c5 = best_c5

    # Gradient descent parameters
    learning_rate = 0.01
    patience = 100
    tolerance = 1e-6
    max_iter = 5000
    no_improvement_count = 0

    def compute_corr(h):
        corr_values = np.correlate(h, 1 - h, mode="full")
        return np.max(corr_values) * dx

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scale = n_points / 2 / total
        h *= scale
        return h

    for iter_i in range(max_iter):
        if time() - start_time >= budget_s:
            break

        corr_values = np.correlate(current_h, 1 - current_h, mode="full")
        max_corr_index = np.argmax(corr_values)
        k_max = max_corr_index - (n_points - 1)

        # Compute gradient w.r.t. the current k_max
        grad = np.zeros(n_points)
        for j in range(n_points):
            grad[j] = 1 - current_h[(j + k_max) % n_points] - current_h[(j - k_max) % n_points]
        grad *= k_max / n_points  # Normalize by shift

        # Update step
        current_h -= learning_rate * grad
        projected_h = project(current_h)
        new_corr = compute_corr(projected_h)

        if new_corr < current_c5 - 1e-4:
            # Improvement
            current_h = projected_h
            current_c5 = new_corr
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count >= patience:
                learning_rate *= 0.9  # decay learning rate
                no_improvement_count = 0

        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.copy(current_h)
            print(f"Iter {iter_i}, best C5: {best_c5:.8f} (learning rate: {learning_rate})")

    # Final projection and evaluation
    best_h = project(best_h)
    final_c5 = evaluate_erdos_solution(best_h)

    # Ensure exact constraint satisfaction
    if abs(np.sum(best_h) - n_points / 2) > 1e-8:
        scale = n_points / 2 / np.sum(best_h)
        best_h *= scale

    return (best_h.tolist(), final_c5, n_points)
