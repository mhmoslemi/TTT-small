        if c5 < best_c5:

def balance_mass(h, n_points):
    total = h.sum()
    target = n_points / 2
    to_add = max(0, target - total)
    to_remove = max(0, total - target)
    
    if abs(to_add - to_remove) < 1e-9:
        return h
        
    h = h / h.sum() * target  # soft balance
    h = np.round(h * 10) / 10  # discretize for better stability
    h = np.clip(h, 0, 1)
    
    # last resort numeric balance fixing
    while abs(h.sum() - target) > 1e-6:
        if h.sum() < target:
            h[np.argmin(h)] += abs(h.sum() - target)
        else:
            h[np.argmax(h)] -= abs(h.sum() - target)
    
    return h

def compute_gradients(h, n_points, dx):
    grads = np.zeros_like(h)

    # Cross-correlate h with (1-h)
    hh = np.zeros((n_points, n_points))
    for i in range(n_points):
        for k in range(n_points):
            if i + k < n_points:
                hh[i, i + k] = h[i] * (1 - h[i + k])

    # Find top 3 lags to compute gradients
    flat_corr = hh.sum(axis=0)
    top_lags = np.argpartition(-flat_corr, 3)[:3]
    
    for k in top_lags:
        for i in range(n_points):
            if i + k < n_points:
                grads[i] -= (1 - h[i + k]) * (2*h[i] - 1)
                grads[i + k] -= h[i] * (-1)
                
    return grads / n_points

def compute_c5(h, n_points, dx):
    result = np.correlate(h, 1 - h, mode='full')
    return np.max(result) * dx
