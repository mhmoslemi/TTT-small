import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_C5(h, n_points):
    dx = 2.0 / n_points
    h_circular = np.concatenate([h, h])
    corr = correlate(h_circular, 1 - h_circular, mode="full")[n_points:2*n_points]
    return np.max(corr * dx)

def objective(h_flat, n_points):
    h = h_flat.reshape(-1)
    return evaluate_C5(h, n_points)

def constraint_sum(h_flat, n_points):
    return np.sum(h_flat) - n_points / 2

def constraint_bounds(h_flat):
    return (0 <= h_flat) & (h_flat <= 1)

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 80
        h0 = np.zeros(n_points)
        idx = int(n_points / 2)
        h0[idx-1:idx+2] = 0.5
        h0 = h0 / np.sum(h0) * (n_points / 2)
    
    start_time = time.time()

    # Define the problem
    def wrapper_obj(h_flat):
        return objective(h_flat, n_points)
    
    constraints = (
        {'type': 'eq', 'fun': lambda h_flat: constraint_sum(h_flat, n_points)},
    )

    # Initial bounds: [0,1] for all h
    bounds = [(0, 1)] * n_points

    # Gradient descent optimization
    result = minimize(wrapper_obj, x0=h0, constraints=constraints, bounds=bounds, method='L-BFGS-B', options={'maxiter': 500, 'ftol': 1e-8})

    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        print("Budget exceeded, returning best found solution so far.")

    best_h = np.clip(result.x, 0, 1)
    final_C5 = evaluate_C5(best_h, n_points)
    return best_h, final_C5, n_points
