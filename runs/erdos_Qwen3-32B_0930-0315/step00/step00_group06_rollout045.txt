import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

initial_h_values = None  # This is consumed as pre-imported global

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    n_points = 95
    dx = 2.0 / n_points

    if initial_h_values is not None:
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.choice([0, 1], size=n_points)
        h = (h * n_points / 2) / h.sum()

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    # Main optimization loop
    start_time = time.time()
    steps = 0
    while time.time() - start_time < budget_s - 1:
        steps += 1
        # Evaluate overlaps
        ones_minus_h = 1 - h
        corr = correlate(h, ones_minus_h, mode='full') * dx
        c5_candidate = corr.max()
        k_max = np.argmax(corr)

        # Phase 1: Magnitude update with convex optimization
        h_var = cp.Variable(n_points)
        ones_minus_h_var = 1 - h_var
        overlap = h_var[k_max:] @ ones_minus_h_var[:n_points - k_max] * dx if k_max < n_points else h_var[:n_points - k_max] @ ones_minus_h_var[k_max:] * dx
        problem = cp.Problem(
            cp.Minimize(overlap),
            [cp.sum(h_var) == n_points / 2,
             0 <= h_var, h_var <= 1]
        )
        problem.solve(solver=cp.ECOS)
        h_opt = np.clip(h_var.value, 0, 1)
        if problem.status != 'optimal':
            break

        h = h_opt.copy()

        # Projection and constraint enforcement
        h = np.clip(h, 0, 1)
        h *= n_points / 2 / h.sum()

        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Perturbation to escape local minima
        if np.random.rand() < 0.01:
            idx = np.random.randint(n_points)
            h[idx] = max(0, min(1, h[idx] + (np.random.rand() - 0.5) * 0.1))
            h *= n_points / 2 / h.sum()

    return (best_h, best_c5, n_points)
