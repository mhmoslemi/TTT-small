import numpy as np
import cvxpy as cp
import time
from math import floor

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)

    # Configuration
    n_points = 80
    dx = 2.0 / n_points

    # Initialize h based on provided initial_h_values or create own
    if 'initial_h_values' in globals():

        if h_vals.ndim > 1 or h_vals.shape[0] != n_points or np.any((h_vals < 0) | (h_vals > 1)):
            h_vals = np.full(n_points, 0.5)
    else:
        h_vals = np.full(n_points, 0.5)

    best_h = h_vals.copy()
    best_c5 = evaluate_erdos_solution(h_vals)
    max_iter = 200
    radius = 0.05  # step size constraint for subproblems

    # Ensure constraints are respected
    h_vals = np.clip(h_vals, 0, 1)
    h_vals *= n_points / (2.0 * h_vals.sum() if h_vals.sum() != 0 else 1.0)

    for it in range(max_iter):
        remaining = time.time() - start_time
        if remaining >= budget_s - 0.7:
            break

        # Evaluate overlap integrals (C_k) for all lags k
        h_real = h_vals.copy()
        h_real[h_vals < 1e-10] = 0  # numerical stability
        h_complement = 1.0 - h_real
        corr = np.correlate(h_real, h_complement, mode="full") * dx
        max_c5 = corr.max()
        worst_lag = np.argmax(corr)
        k = worst_lag

        # Formulate and solve convex subproblem for lag k
        h = cp.Variable(n_points)
        obj = h - h * np.roll(h, -k)
        objective = cp.Minimize(cp.sum(obj) * dx)
        constraints = [
            cp.sum(h) == cp.sum(h_vals),  # preserve total sum
            h >= 0, h <= 1,  # box constraints
            # small update step control
            h - h_vals <= radius * n_points,
            h_vals - h <= radius * n_points
        ]
        problem = cp.Problem(objective, constraints)
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
            new_h_vals = np.clip(np.array(h.value), 0, 1)
            new_h_vals *= n_points / (2.0 * new_h_vals.sum() if new_h_vals.sum() != 0 else 1.0)
            current_c5 = evaluate_erdos_solution(new_h_vals)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_h_vals.copy()
            h_vals = new_h_vals
        except Exception as e:
            continue  # fallback to last valid h_vals if solve fails

        # Early stopping if close to target
        if best_c5 < 0.380876 + 1e-6:
            break

    return (best_h.tolist(), best_c5, n_points)
