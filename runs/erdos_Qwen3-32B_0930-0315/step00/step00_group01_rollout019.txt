import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h_values):
    n_points = len(h_values)
    h = np.array(h_values, dtype=np.float64)
    dx = 2.0 / n_points
    # Compute max correlation
    ones_minus_h = 1.0 - h
    corr_real = correlate(h, ones_minus_h, mode='full')
    c5_candidate = np.max(corr_real * dx)
    return c5_candidate

def project_to_constraints(h, n_points):
    # Clamp values to [0, 1]
    h_clamped = np.clip(h, 0.0, 1.0)
    # Rescale to satisfy the sum constraint
    h_s = h_clamped * (n_points / 2.0 / np.sum(h_clamped))
    h_s = np.clip(h_s, 0.0, 1.0)
    return h_s

def compute_finite_diff_gradient(f, x, step_size=1e-5):
    n = len(x)
    grad = np.zeros(n)
    fx = f(x)
    for i in range(n):
        x_perturbed = np.copy(x)
        x_perturbed[i] += step_size
        fx_perturbed = f(x_perturbed)
        grad[i] = (fx_perturbed - fx) / step_size
    return grad

def gradient_projection_step(h_current, grad, step_size):
    h_next = h_current - step_size * grad
    h_projected = project_to_constraints(h_next, len(h_current))
    return h_projected

def run(seed=42, budget_s=1100, lr=0.1, max_iter=1000):
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n_points = len(h)
    else:
        n_points = 47  # reasonable starting size
        h = np.zeros(n_points, dtype=np.float64)
        h[:n_points // 2] = 1.0

    tic = time.time()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h)

    for iteration in range(max_iter):
        if time.time() - tic > budget_s - 1.0:
            break

        def objective(x):
            h_test = np.array(x, dtype=np.float64)
            h_test = project_to_constraints(h_test, n_points)
            return evaluate_h(h_test)

        # Compute gradient via finite differences
        h_projected = project_to_constraints(h, n_points)
        grad = compute_finite_diff_gradient(objective, h_projected)

        # Gradient descent step with constraint projection
        h_projected = project_to_constraints(h_projected, n_points)
        h = gradient_projection_step(h_projected, grad, lr)

        # Evaluate and update best known
        current_c5 = evaluate_h(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Reduce learning rate over time
        lr *= 0.995

    return best_h.tolist(), best_c5, n_points
