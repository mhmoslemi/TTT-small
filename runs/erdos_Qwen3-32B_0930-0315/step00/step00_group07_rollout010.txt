import numpy as np
import time
import cvxpy as cp
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    # Parameters
    n_points = 80  # Can be adjusted for different resolution
    dx = 2.0 / n_points
    iteration_cap = 500  # Ensure fits within the time constraint

    # Initialize h using the parent construction when available
    if "initial_h_values" in globals():
        h = np.copy(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[:40] = 1   # uniform initialization with correct mass
        np.random.shuffle(h)

    # Normalize to correct mass
    h *= n_points / 2 / h.sum()

    best_h = np.copy(h)
    best_c5 = np.inf
    start_time = time.time()

    for _iter in range(iteration_cap):
        # Compute correlation for all lags
        correlate_result = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlate_result * dx)

        # Update if better
        if c5 < best_c5:
            best_c5 = c5
            best_h = np.copy(h)

        if time.time() - start_time > budget_s - 10:
            break

        # Identify worst lag
        n_corr = len(correlate_result)
        offset = n_corr // 2
        max_idx = np.argmax(correlate_result[(offset - 40):(offset + 40)])
        k = max_idx - 40

        # Build gradient for worst lag
        gradient = np.zeros_like(h)
        for i in range(n_points):
            j1 = (i + k) % n_points
            j2 = (i - k) % n_points
            gradient[i] = 1 - h[j1] - h[j2]
        gradient *= dx

        # Solve LP: minimize c5 improvement with step in feasible set
        delta = cp.Variable(n_points)

        objective = cp.Minimize(gradient @ delta)
        constraints = [
            h + delta >= 0,
            h + delta <= 1,
            cp.sum(delta) == 0,
        ]
        problem = cp.Problem(objective, constraints)

        try:
            problem.solve(solver=cp.GUROBI, verbose=False)
        except:
            try:
                problem.solve(solver=cp.ECOS, verbose=False)
            except:
                continue  # skip if no LP solver available or problem failed

        if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            continue

        alpha = 1.0
        delta_val = delta.value

        # Line Search / Step-size adaptation
        while True:
            next_h = h + alpha * delta_val
            if np.all((0 <= next_h) * (next_h <= 1)) and np.abs(np.sum(next_h) - n_points / 2) < 1e-6:
                break
            alpha *= 0.5
            if alpha < 1e-6:
                break

        h = h + alpha * delta_val

    # Final validation
    if not np.isclose(np.sum(h), n_points / 2):
        h = best_h

    bound = evaluate_erdos_solution(h)
    return (h, bound, n_points)
