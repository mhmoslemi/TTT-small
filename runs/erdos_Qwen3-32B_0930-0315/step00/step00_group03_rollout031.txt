```python
import numpy as np
import cvxpy as cp
import time
from math import isclose

def evaluate_c5(h, dx):
    ones_h = 1 - h
    corr = np.correlate(h, ones_h, mode='full')
    return np.max(corr * dx)

def is_feasible(h, n_points):
    return (np.all(h >= 0) and np.all(h <= 1) and isclose(np.sum(h), n_points / 2, abs_tol=1e-6))

def project_h(h, n_points):
    h_var = cp.Variable(n_points)
    objective = cp.Minimize(cp.sum_squares(h_var - h))
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS)
    if prob.status != cp.OPTIMAL:
        raise RuntimeError("Projection failed.")
    return h_var.value.astype(h.dtype)

def run(seed=42, budget_s=1100, *args, **kwargs):
    start_time = time.perf_counter()
    np.random.seed(seed)

    n_points_base = 70
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = 0.5 * np.ones(n_points_base, dtype=np.float64)

    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)

    # Gradient descent with convex projection
    iter_max = 1000
    eta = 0.01
    for iter in range(iter_max):
        elapsed = time.perf_counter() - start_time
        if elapsed >= budget_s:
            break

        # Compute overlap and find max index
        ones_h = 1 - h
        corr = np.correlate(h, ones_h, mode='full')
        max_idx = np.argmax(corr)

        # Compute gradient for this k
        k = max_idx - n_points
        G = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            if i + k < n_points:
                G[i] = -(1 - h[i + k])
            if i - k >= 0 and i - k < n_points:
                G[i] -= h[i - k]

        # Normalize gradient
        norm_G = np.linalg.norm(G)
        if norm_G > 1e-5:
            G /= norm_G

        h_candidate = h - eta * G
        h_candidate = project_h(h_candidate, n_points)
        candidate_c5 = evaluate_c5(h_candidate, dx)

        if candidate_c5 < best_c5:
            h = h_candidate
            best_c5 = candidate_c5

    return best_h, best_c5, n_points
```
