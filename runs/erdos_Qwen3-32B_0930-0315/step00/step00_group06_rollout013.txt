import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Stage 1: Use parent construction if provided, else build a base
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = initial_h_values.copy()
    else:
        n_points = 256
        base_h = np.random.rand(n_points)
        base_h = np.clip(base_h, 0, 1)
        base_h *= 2 / n_points / np.mean(base_h)
        h = base_h

    # Ensure h meets sum constraint
    h_sum = np.sum(h)
    alpha = n_points / 2 / h_sum
    h *= alpha

    # Problem setup
    dx = 2.0 / n_points
    n_reps = len(h)
    n_k = 2 * n_points + 1

    # Time budget tracking
    start_time = time.time()

    best_h = h.copy()
    best_c5 = float('inf')

    # Optimization parameters
    max_iter = 1000000
    perturb_interval = 100
    perturb_size = 0.02

    for i in range(max_iter):
        # Periodic perturbations
        if i % perturb_interval == 0:
            perturbation = np.random.uniform(-perturb_size, perturb_size, size=(len(h),))
            h = np.clip(h + perturbation, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 1e-6:
                h *= n_points / 2 / h_sum

        # Compute cross-correlation using optimized approach
        h_padded = np.zeros(4 * n_points)
        h_padded[n_points:3*n_points] = h

        # Copy h into 1-h mask space with padding to avoid wrap-around
        h2 = 1 - h_padded[n_points:3*n_points]
        
        # Compute correlation at all lags
        result = np.correlate(h2, h_padded, mode='full')
        c5 = np.max(result[n_points:n_points+n_points]) * dx

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            print(f"Iter {i}: Improved C₅ to {best_c5:.8f}")

        # Limit on time
        if time.time() - start_time > budget_s - 1:
            break

        # Convex update for weights using CVXPY
        h_var = cp.Variable(n_points)
        obj = 0
        for k in range(n_points):
            obj += h_var[k] * (1 - h[n_points - 1 - k])

        constraints = [
            cp.sum(h_var) == n_points / 2,
            0 <= h_var,
            h_var <= 1,
        ]

        problem = cp.Problem(cp.Minimize(obj), constraints)
        problem.solve(solver=cp.GUROBI, verbose=False)

        if problem.status != 'optimal':
            break

        h = h_var.value.copy()

    # Final validation
    h_sum = np.sum(h)
    h[:] = np.clip(h, 0, 1)
    if h_sum > 1e-6:
        h /= h_sum
        h *= n_points / 2
