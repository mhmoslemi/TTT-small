```python
import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Ensure sum is exactly 35 when n_points == 70
    h = h / np.sum(h) * n_points / 2
    h = np.clip(h, 0, 1)
    c5 = 0
    for k in range(-n_points + 1, n_points):
        h2 = np.roll(h, -k)
        c = np.sum(h * (1 - h2)) * dx
        if c > c5:
            c5 = c
    return c5

def initial_h(n_points):
    # approximation of initial_h_values by resampling
    h = np.zeros(n_points)
    h[0:n_points//4] = 1.0
    h[-n_points//4:] = 1.0
    h = h / np.sum(h) * n_points / 2
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    best_h = None
    best_score = float('inf')
    best_n_points = 70

    start_time = time.time()
    n_points = 70
    h = initial_h(n_points).copy()

    # Optimization loop
    n_iter = 0
    while time.time() - start_time < budget_s:
        dx = 2.0 / n_points
        h_cvx = cp.Variable(n_points, nonneg=True)
        constraints = [
            h_cvx <= 1,
            cp.sum(h_cvx) == n_points / 2
        ]
        max_index = 0
        max_corr = 0

        # Use a smooth max approximation to avoid full correlation matrix
        for k in range(-n_points + 1, n_points):
            if k >= 0 and k < n_points:
                # Only consider inner lags to avoid wrapping
                correlation = cp.sum(h_cvx * (1 - cp.roll(h_cvx, -k))) * dx
                if (correlation.value or 0) > max_corr:
                    max_corr = correlation.value or 0
                    max_index = k

        # Use a single worst k for optimization
        k = max_index
        if 0 <= k < n_points:
            objective = cp.sum(h_cvx * (1 - cp.roll(h_cvx, -k))) * dx
            prob = cp.Problem(cp.Minimize(objective), constraints)
            prob.solve(solver=cp.OSQP, verbose=False)

            if h_cvx.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
                h = h_cvx.value
                h = np.clip(h, 0, 1)
                h = h / np.sum(h) * n_points / 2

        # Evaluate current solution
        c5 = evaluate_erdos_solution(h, n_points)
        if c5 < best_score:
            best_score = c5
            best_h = h.copy()
            best_n_points = n_points

        n_iter += 1
        if n_iter % 10 == 0:
            print(f"Iteration {n_iter}, Best C5: {best_score}, n_points: {best_n_points}")

    if best_h is None:
        best_h = h

    return (best_h.tolist(), best_score, best_n_points)
```
