import numpy as np
from functools import partial
import time
import evaluate_erdos_solution

def compute_overlap(h, dx):
    h_full = np.concatenate([h, np.zeros(len(h))])
    corr = np.correlate(h_full, 1 - h_full, mode="full") * dx
    return float(np.max(corr))

def repair_feasibility(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    h = h / np.sum(h) * (n_points / 2)
    return h

def gradient_update(h, dx, lr):
    n_points = len(h)
    h_full = np.concatenate([h, np.zeros(n_points)])
    corr = np.correlate(h_full, 1 - h_full, mode="full") * dx
    k_max = int(np.argmax(corr))
    shift = k_max - (n_points - 1)
    h_shifted = np.roll(h, shift)
    grad = (1 - h_shifted) * dx
    h -= lr * grad[:n_points]
    return repair_feasibility(h, n_points)

def optimize_step(h, dx, lr, temperature):
    n_points = len(h)
    h_proposal = h + np.random.normal(0, temperature, n_points)
    h_proposal = repair_feasibility(h_proposal, n_points)
    C_proposal = compute_overlap(h_proposal, dx)
    C_current = compute_overlap(h, dx)
    delta = C_proposal - C_current
    if delta < 0 or np.exp(-delta / temperature) > np.random.rand():
        return h_proposal, C_proposal
    return h, C_current

def refine_phase(h, dx, budget, step_size, verbose=False):
    t0 = time.time()
    n_points = len(h)
    C_best = compute_overlap(h, dx)
    h_best = h.copy()
    while time.time() - t0 < budget:
        h = gradient_update(h, dx, step_size)
        C = compute_overlap(h, dx)
        if C < C_best:
            C_best = C
            h_best = h.copy()
    return h_best, C_best

def simulated_annealing_phase(h, dx, budget, temp_init, temp_decay):
    t0 = time.time()
    n_points = len(h)
    C_best = compute_overlap(h, dx)
    h_best = h.copy()
    temp = temp_init
    while time.time() - t0 < budget:
        h, C = optimize_step(h_best, dx, 0.0, temp)
        if C < C_best:
            h_best = h.copy()
            C_best = C
        temp *= temp_decay
    return h_best, C_best
