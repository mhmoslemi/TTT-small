import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    h1 = h - np.floor(h)  # Ensure h in [0,1]
    h1 = np.clip(h1, 0, 1)
    corr = correlate(h1, 1 - h1, mode="full")
    max_corr = np.max(np.abs(corr)) * dx
    return max_corr

def ensure_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    scale = (n_points / 2) / np.sum(h)
    h = h * scale
    return h

    population_size = 75
    tournament_size = max(2, int(0.10 * population_size))

    mutation_rate = 0.05
    crossover_rate = 0.7
    starting_population = [initial_h]

    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = ensure_feasible(h, n_points)
        starting_population.append(h)

    population = np.array(starting_population)
    best_c5 = np.inf
    best_h = None

    start_time = time.time()
    t = 100  # initial temperature
    cooling_rate = 0.95

    for generation in range(n_generations):
        if time.time() - start_time > budget_s:
            break
        scores = np.array([evaluate_c5(h, n_points) for h in population])
        best_idx = np.argmin(scores)
        if scores[best_idx] < best_c5:
            best_c5 = scores[best_idx]
            best_h = population[best_idx]

        # Tournament selection
        parent_indices = []
        for _ in range(population_size // 2):
            candidates = np.random.choice(population_size, tournament_size, replace=False)
            winner = np.argmin(scores[candidates])
            parent_indices.append(candidates[winner])
        parents = population[parent_indices]

        # Crossover
        children = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents) and np.random.rand() < crossover_rate:
                a, b = parents[i], parents[i + 1]
                cp = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([a[:cp], b[cp:]])
                child2 = np.concatenate([b[:cp], a[cp:]])
                children.append(ensure_feasible(child1, n_points))
                children.append(ensure_feasible(child2, n_points))
            else:
                children.append(parents[i])
                if i + 1 < len(parents):
                    children.append(parents[i + 1])
        new_population = []
        # Add top 1/3 to survive
        top_indices = np.argsort(scores)[:len(population) // 3]
        new_population.extend(population[top_indices])
        new_population.extend(children[:population_size - len(top_indices)])
        population = np.array(new_population)

        # SA-inspired mutation on best solution every 50 generations
        if generation % 50 == 0:
            h = best_h.copy()
            if np.random.rand() < mutation_rate:
                idxs = np.random.choice(n_points, size=int(mutation_rate * n_points), replace=False)
                h[idxs] += (2 * np.random.rand(len(idxs)) - 1) * 0.2
                h = ensure_feasible(h, n_points)
            new_score = evaluate_c5(h, n_points)
            if new_score < best_c5 or np.random.rand() < np.exp((best_c5 - new_score) / t):
                best_c5 = new_score
                best_h = h
            t *= cooling_rate

    return best_h, best_c5, n_points
