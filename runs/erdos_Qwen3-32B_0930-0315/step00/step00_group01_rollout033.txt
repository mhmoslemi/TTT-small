import numpy as np
import time
from scipy.signal import correlate

# Evaluate the Erdos constant C5 for a given h
def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    n = len(h)
    full_corr = correlate(h, 1.0 - h, mode='full')
    c5_values = full_corr * dx
    c5 = np.max(c5_values)
    return c5

def run(seed=42, budget_s=1100, *args, **kwargs):
    t0 = time.time()
    max_time = budget_s

    # Use the initial_h_values if provided, fallback to a simple initialization
    if 'initial_h_values' in globals():
        print("Using initial_h_values")
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        print("Constructing initial_h_values")
        n_points = 47  # Starting with the same size as the previous construction
        np.random.seed(seed)
        h = np.zeros(n_points)
        u = np.random.permutation(n_points)
        h[u[:n_points // 2]] = 1.0  # Use 47/2 = 23.5 => 24 ones to satisfy sum(h) = 24 (as close to 23.5 as possible)
        print("Warning: initial h is not perfectly balanced")

    # Ensure sum of h is n_points/2 exactly
    current_sum = h.sum()
    target_sum = n_points / 2
    if np.isclose(current_sum, target_sum):
        pass
    else:
        # Adjust h to reach target_sum
        diff = target_sum - current_sum
        # Select one index to adjust to balance sum exactly
        idx = 0
        if diff > 0:
            h[idx] += diff
        else:
            h[idx] -= abs(diff)

    # Define constraint projection onto feasible region [0,1] and sum = n_points / 2
    def project(h):
        h = np.clip(h, 0, 1)
        sum_h = h.sum()
        if not np.isclose(sum_h, target_sum, atol=1e-4):  # Tolerance for numerical precision
            scale = target_sum / sum_h
            h = h * scale
            h = np.clip(h, 0, 1)  # After rescaling, ensure no value goes out of [0,1]
        return h

    # Compute the correlation for h and (1 - h) and return the maximum
    def compute_C5(h):
        if not (0 <= h.all() <= 1):
            return np.inf
        dx = 2.0 / n_points
        full_corr = correlate(h, 1.0 - h, mode='full')
        return np.max(full_corr * dx)

    # Gradient computation using finite differences
    def compute_gradient(h, epsilon=1e-5):
        g = np.zeros_like(h)
        base_C5 = compute_C5(h)
        for i in range(n_points):
            h_p = h.copy()
            h_p[i] += epsilon
            h_p = project(h_p)
            C5_p = compute_C5(h_p)
            grad = (C5_p - base_C5) / epsilon
            g[i] = grad
        return g

    # Optimization parameters
    learning_rate = 0.1
    iteration = 0
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    print(f"Iteration {iteration:3} - Current C5: {best_C5:.6f}")

    # Optimization loop
    while time.time() - t0 < max_time:
        iteration += 1
        # Compute gradient
        grad = compute_gradient(h)
        # Gradient descent step
        h -= learning_rate * grad
        # Project back to feasible region
        h = project(h)
        # Compute current value
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            print(f"Iteration {iteration:3} - Current C5: {best_C5:.6f}")
        if time.time() - t0 >= max_time:
            break

    return (best_h.tolist(), best_C5, n_points)
