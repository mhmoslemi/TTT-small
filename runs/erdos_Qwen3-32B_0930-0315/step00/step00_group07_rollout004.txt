import numpy as np
import cvxpy as cp
import time
import evaluate_erdos_solution
from functools import lru_cache
import sys

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_time = time.time()
    
    # Setup grid
    n_points = 80  # Consistent with initial_h_values
    dx = 2.0 / n_points

    # Validate and use initial solution
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points_initial = len(h)
        # Resize to match n_points
        if n_points != n_points_initial:
            scale_factor = n_points / n_points_initial
            h = np.interp(np.arange(n_points) / scale_factor,
                          np.arange(n_points_initial) / scale_factor,
                          h)
    except:
        h = np.ones(n_points) / 2  # Fallback to uniform if initial invalid

    # Normalize h to have average 0.5
    h *= (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_c5 = 0.0

    # Evaluation function
    @lru_cache(maxsize=256)
    def compute_c5_fast(h_tuple):
        h_vec = np.array(h_tuple)
        h_vec = np.clip(h_vec, 0.0, 1.0)
        corrs = np.correlate(h_vec, 1 - h_vec, mode="full") * dx
        return float(np.max(corrs))

    c5_current = compute_c5_fast(tuple(h))
    best_c5 = c5_current

    try:
        max_iter = 30
        for iter_idx in range(max_iter):
            if time.time() - initial_time >= budget_s:
                break

            # Save best solution
            if c5_current < best_c5:
                best_h = h.copy()
                best_c5 = c5_current

            # Identify worst lag
            h_vec = np.clip(h, 0.0, 1.0)
            corrs = np.correlate(h_vec, 1 - h_vec, mode="full") * dx
            k = np.argmax(corrs)

            # Create shifted version
            @np.vectorize
            def shift_index(i, k, n):
                return (i + k) % n

            indices = (np.arange(n_points) + k) % n_points
            h_k = h_vec[indices]

            # Formulate convex problem
            h_var = cp.Variable(n_points)
            overlaps = h_var @ (1 - h_k) * dx  # Objective
            constraint_sum = cp.sum(h_var) == n_points / 2
            constraints = [
                constraint_sum,
                0 <= h_var,
                h_var <= 1
            ]
            objective = cp.Minimize(overlaps)
            problem = cp.Problem(objective, constraints)

            # Solve with fast solver
            try:
                problem.solve(solver=cp.MOSEK, max_iter=1000, verbose=False)
            except:
                problem.solve(solver=cp.ECOS, max_iter=1000, verbose=False)

            h_new = np.clip(h_var.value, 0.0, 1.0)
            c5_new = compute_c5_fast(tuple(h_new))

            # Accept better solution
            if c5_new < c5_current:
                h = h_new
                c5_current = c5_new

    except Exception as e:
        print(f"Error during optimization: {e}", file=sys.stderr)

    finally:
        # Final validation and return
        h = np.clip(best_h, 0.0, 1.0)
        h *= (n_points / 2) / np.sum(h)
        h_values = h.tolist()
        c5_bound = compute_c5_fast(tuple(h))

    return h_values, c5_bound, n_points
