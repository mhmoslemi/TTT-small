import numpy as np
import time
import cvxpy as cp
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, n_points=45):
    np.random.seed(seed)
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy().ravel()
        h[:] = h.astype(np.float64)
    else:
        h = np.zeros(n_points)
        idx = np.random.choice(n_points, size=n_points // 2, replace=False)
        h[idx] = 1.0

    h[:] = h.clip(0, 1)

    def normalize_h(h):
        h[:] = h.clip(0, 1)
        s = np.sum(h)
        if abs(s - n_points / 2) > 1e-7:
            h[:] = h / s * (n_points / 2)
        return h

    def evaluate_c5(h, dx):
        return evaluate_erdos_solution(h)

    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    tabu_memory = set()
    tabu_size = 15
    n_neighbors = 20
    max_iter = int(1e6)
    dp = int(n_points / 5)

    for it in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        current_key = tuple(np.round(h, decimals=6).astype(np.float64).tobytes())
        tabu_memory.discard(current_key)
        if current_key in tabu_memory:
            # Perturb out of local tabu if stuck
            idx = np.random.choice(n_points, 1)
            h[idx] += np.random.uniform(-0.2, 0.2)
            h = normalize_h(h.copy())
            current_key = tuple(np.round(h, decimals=6).astype(np.float64).tobytes())
            tabu_memory.add(current_key)
            continue

        candidates = []
        for _ in range(n_neighbors):
            h_proposal = h.copy()
            idx = np.random.choice(n_points, np.random.randint(1, 6), replace=False)

            for i in idx:
                delta = np.random.uniform(-0.2, 0.2)
                h_proposal[i] += delta
                h_proposal[i] = np.clip(h_proposal[i], 0, 1)

            h_proposal = normalize_h(h_proposal)
            proposal_key = tuple(np.round(h_proposal, decimals=6).astype(np.float64).tobytes())
            if proposal_key == tuple(np.round(h, decimals=6).astype(np.float64).tobytes()):
                continue

            c5_proposal = evaluate_c5(h_proposal, dx)
            candidates.append((c5_proposal, h_proposal))

        if not candidates:
            continue

        candidates.sort()
        if candidates[0][0] < best_c5:
            best_h[:] = candidates[0][1].copy()
            best_c5 = candidates[0][0]

        try:
            h = candidates[0][1].copy()
        except:
            h[:] = best_h.copy()

        tabu_memory.add(tuple(np.round(h, decimals=6).astype(np.float64).tobytes()))
        if len(tabu_memory) > tabu_size:
            tabu_memory.popitem(last=False)

    return best_h.tolist(), best_c5, n_points
