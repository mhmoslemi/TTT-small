import numpy as np
import cvxpy as cp
import time
import warnings
from scipy.optimize import linprog
import math

def compute_c5(h, dx):
    k_vals = list(range(-h.size + 1, h.size))
    overlaps = []
    for k in k_vals:
        corr = np.sum(h * (1 - np.roll(h, k))[:h.size]) * dx
        overlaps.append(corr)
    return max(overlaps)

def gradient(h, dx, k_max):
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        j = i + k_max
        if 0 <= j < n:
            grad[i] -= (1 - h[j])
        j = i - k_max
        if 0 <= j < n:
            grad[i] -= h[j]
    return grad

def project_cvx(h, n_points):
    h_var = cp.Variable(n_points)
    constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2.0]
    objective = cp.Minimize(cp.sum_squares(h_var - h))
    with warnings.catch_warnings():
        warnings.simplefilter("ignore")
        prob = cp.Problem(objective, constraints)
    try:
        result = prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h_var.value
        else:
            print("Projection fallback to GLOP")
            prob.solve(solver=cp.GLOP, verbose=False)
            return h_var.value
    except:
        return h  # fallback to original if failed

    best_iter = -1

        niters += 1
        c5 = compute_c5(h, dx)

        # Track best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_iter = niters
            patience = 200  # Extend patience on progress
        else:
            patience -= 1
            if patience == 0:
                print("Early stopping due to no improvement")

        # Find worst shift k for gradient
        k_vals = list(range(-h.size + 1, h.size))
        overlaps = np.array([np.sum(h * (1 - np.roll(h, k))[:h.size]) for k in k_vals])
        k_max = k_vals[overlaps.argmax()]

        # Compute gradient
        grad = gradient(h, dx, k_max)

        # Gradient descent update
        step = learn_rate * grad / (1e-6 + np.norm(grad))
        h_candidate = h - step

        # Project back to feasible region
        h = project_cvx(h_candidate, n_points)
        if h is None:
            h = h_candidate  # fallback if projection failed

        # Decay learning rate slowly
        if niters % 100 == 0 and niters > 0:
            learn_rate *= 0.95

        # Break on convergence
        if niters > 1000 or time.time() - t_start + 10 > budget_s:
            break

    if best_h is not None:
