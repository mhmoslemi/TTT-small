import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def get_initial_h(n_points):
    if 'initial_h_values' in globals():
        h0 = initial_h_values.copy()
        h0.resize(n_points)
        h0[h0 < 0] = 0
        h0[h0 > 1] = 1
        return h0
    # Construct initial balanced binary sequence if no prior construction
    h0 = np.zeros(n_points)
    h0[:n_points//2] = 1
    np.random.shuffle(h0)
    return h0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    a = 1 - h
    c = correlate(h, a, mode='full')[n-1:2*n-1] * dx
    return np.max(c)

def project(h, dx):
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - 1.0) > 1e-4:
        scale_factor = 1.0 / s
        h *= scale_factor
    return h

def estimate_gradient(h, perturbation=1e-3):
    n = len(h)
    best_h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)

    # Store gradients as finite differences
    grad = np.zeros_like(h)

    direction = np.zeros_like(h)
    for i in range(n):
        direction[i] = perturbation
        new_h = h + direction
        new_h = project(new_h, dx)
        up_c5 = compute_c5(new_h)
        new_h = h - direction
        new_h = project(new_h, dx)
        down_c5 = compute_c5(new_h)
        grad[i] = (up_c5 - down_c5) / (2 * perturbation)
        direction[i] = 0

    return grad

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    # Initial sequence size (small for speed)
    initial_n_points = 64
    h = get_initial_h(initial_n_points)
    h = project(h, 2.0 / initial_n_points)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    start_time = time.time()
    last_improvement = start_time
    h = get_initial_h(initial_n_points)
    h = project(h, 2.0 / initial_n_points)

    # Try adjusting n_points based on progress
    for n_points in [96, 128]:
        start_h = get_initial_h(n_points)
        start_h = project(start_h, 2.0 / n_points)
        h = start_h.copy()
        current_c5 = compute_c5(h)
        best_c5 = min(best_c5, current_c5)
        best_h = h.copy() if current_c5 == best_c5 else best_h

        learning_rate = 0.1
        tolerance = 1e-4
        for iter in range(200):
            if time.time() - start_time > budget_s - 2:
                break

            # Compute gradient
            grad = estimate_gradient(h)

            # Apply gradient step with clipping
            h_grad_step = h - learning_rate * grad
            h = project(h_grad_step, 2.0 / n_points)

            # Track improvement
            c5_val = compute_c5(h)
            if c5_val < best_c5:
                best_h = h.copy()
                best_c5 = c5_val
                last_improvement = time.time()

            # Adaptive learning rate if no improvement
            if time.time() - last_improvement > 30:
                learning_rate *= 0.9

            # Terminate if progress is minimal
            if max(abs(grad)) < tolerance and iter > 50:
                break

    return (best_h, best_c5, len(best_h))
