import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    dx = 2.0 / 256
    n_points = 256
    h_init = np.zeros(n_points)

    # Use normalized version of provided initial construction if available
    if 'initial_h_values' in globals():
        h_init = initial_h_values
        h_init = h_init / h_init.sum() * n_points * dx
    else:
        h_init = np.zeros(n_points)
        h_init[:128] = 1.0  # Simple baseline: uniform 50% coverage

    h = np.copy(h_init)

    # Optimization loop
    start_time = time.time()
    max_iter = 5000
    for iter in range(max_iter):

        if time.time() - start_time > budget_s:
            break

        # Compute auto-correlation at all lags
        ac = np.correlate(h, h, mode="full")
        if (ac.size - n_points) > n_points:
            ac = ac[-(2 * n_points):]

        # Zero out central peak to focus only on side lags
        ac = np.roll(ac, n_points)
        ac[n_points - 1] = 0

        # Find worst autocorrelation (minimum for our max(1 - autocorr))
        ac_min_idx = np.argmin(ac)
        ac_min = ac[ac_min_idx]

        # Determine how much to move h to improve worst autocorrelation
        lag = ac_min_idx
        if lag <= n_points:
            ac_at_lag = np.correlate(h, h, mode="full")[lag]
            overlap_k = np.correlate(h, 1 - h, mode="full")[lag] * dx
            grad = np.roll(h, -lag) - h  # Derivative of overlap w.r.t. h
        else:
            lag = 2 * n_points - lag
            ac_at_lag = np.correlate(h, h, mode="full")[lag]
            overlap_k = np.correlate(h, 1 - h, mode="full")[lag] * dx
            grad = h - np.roll(h, -lag)  # Derivative of overlap w.r.t. h

        # Normalize gradient
        grad /= (dx * np.linalg.norm(grad))

        # Apply gradient step with line search
        for step_size in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = h + step_size * grad

            # Project onto [0, 1]
            h_new = np.clip(h_new, 0.0, 1.0)

            # Enforce average constraint: {h}
            h_mean = 1.0 / n_points * np.sum(h_new)
            if h_mean > 1e-8:
                h_target = n_points / 2
                h_new *= h_target / (dx * np.sum(h_new))

            # Check if we improved the worst autocorrelation
            ac_new = np.correlate(h_new, h_new, mode="full")
            if (ac_new.size - n_points) > n_points:
                ac_new = ac_new[-(2 * n_points):]
            ac_new = np.roll(ac_new, n_points)
            ac_new[n_points - 1] = 0
            ac_min_new = np.min(ac_new)
            if ac_min_new > ac_min or ac_min < 1e-6:
                h = h_new.copy()
                break  # Accept step and move to next iteration

    # Evaluate final result
    ac_final = np.correlate(h, h, mode="full")
    ac_diff = 1.0 - ac_final
    if (ac_diff.size - n_points) > n_points:
        ac_diff = ac_diff[-(2 * n_points):]
    ac_diff = np.roll(ac_diff, n_points)
    c5_bound = np.max(ac_diff[n_points:2 * n_points]) * dx

    # Normalize and scale for output
    h_values = h

    return h_values, c5_bound, n_points
