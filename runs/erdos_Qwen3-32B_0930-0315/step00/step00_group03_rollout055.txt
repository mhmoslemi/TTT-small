import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = np.array(initial_h_values)
    else:
        n_points = 70
        h = np.full(n_points, 0.5)

    initial_h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    last_update = time.time()

    def compute_overlap(h):
        dx = 2.0 / n_points
        h = h.astype(np.float64)
        rh = 1.0 - h
        corr = np.correlate(h, rh, mode="full") * dx
        return np.max(corr), corr

    max_iter = 1000
    iter_no_improvement = 0
    learning_rate = 0.01

    for it in range(max_iter):
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break

        c5, corr = compute_overlap(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_update = time.time()
            iter_no_improvement = 0
        else:
            iter_no_improvement += 1

        if iter_no_improvement >= 100:
            break

        # Find the worst (largest) overlap index k
        k = np.argmax(corr)

        # Create gradient
        G = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k) % n_points
            if j < n_points:
                G[i] += (1 - h[j])
            j = (i - k) % n_points
            if j < n_points:
                G[i] -= h[j]

        # Update h with learning rate decay
        learning_rate = 0.01 / (1 + it / 1000)
        h -= learning_rate * G

        # Projection back to the feasible region (0 <= h <= 1, sum h == n_points / 2)
        H = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum_squares(H - h))
        constraints = [
            H <= 1.0,
            H >= 0.0,
            cp.sum(H) == n_points / 2.0,
        ]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.GUROBI)
            h = H.value.copy()
        except:
            pass  # Use h from before if projection fails

    # Final evaluation of best solution
    best_c5 = evaluate_erdos_solution(best_h)
    return (best_h, best_c5, n_points)
