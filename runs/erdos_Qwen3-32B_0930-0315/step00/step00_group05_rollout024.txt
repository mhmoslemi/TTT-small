import numpy as np
import time
import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points

    # Use initial_h_values directly as per rules
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, copy=True, dtype=np.float64)
        if h.size != n_points:
            # Truncate or pad if needed to keep n_points fixed
            h = np.pad(h, (0, n_points - h.size), mode='constant')[:n_points]
    else:
        # If no initial guess is given, start with uniform
        h = np.ones(n_points, dtype=np.float64) * 0.5

    # Normalize to meet sum(h) == n_points / 2
    h = h / np.sum(h) * (n_points / 2)
    h = np.clip(h, 0.0, 1.0)

    def compute_c5(h_vals):
        cors = np.correlate(h_vals, 1.0 - h_vals, mode='full') * dx
        return np.max(cors)

    t_start = time.time()
    alpha = 0.1  # Step size
    best_c5 = compute_c5(h)
    best_h = h.copy()
    c5_trace = [best_c5]
    last_improve_time = time.time()
    patience = 200
    iter_count = 0

    while time.time() - t_start < budget_s:
        iter_count += 1

        # Compute correlation to find the max overlap lag
        h_tilde = h
        ones_minus_h = 1.0 - h_tilde
        corr = np.correlate(h_tilde, ones_minus_h, mode='full') * dx
        k_max = np.nanargmax(corr)

        # Construct gradient for the worst lag k_max
        grad = np.zeros_like(h)
        lag = k_max
        for m in range(h.size):
            if 0 <= m + lag < h.size:
                grad[m] += ones_minus_h[m + lag]
            if m - lag >= 0:
                grad[m] -= h[m - lag]

        # Gradient descent step
        h -= alpha * grad

        # Enforce constraints: 0 <= h <= 1, sum h == n_points / 2
        h = np.clip(h, 0.0, 1.0)
        h = h / h.sum() * (n_points / 2)

        # Compute current C5
        current_c5 = compute_c5(h)

        # Track best solution so far
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improve_time = time.time()
            c5_trace.append(best_c5)
        else:
            c5_trace.append(current_c5)

        # Patience check and adaptive step size
        if iter_count % 50 == 0:
            if time.time() - last_improve_time > patience:
                alpha *= 0.8  # Reduce step size
                if alpha < 1e-4:
                    break  # Early stop if too small to make progress

        # Periodic perturbation to escape local optima
        if iter_count % 50 == 0:
            perturb = np.random.randn(n_points) * 0.01
            h = h + perturb
            h = np.clip(h, 0.0, 1.0)
            h = h / h.sum() * (n_points / 2)

        # Break if time up or no improvement
        if time.time() - t_start >= budget_s:
            break

    return (best_h.tolist(), best_c5, n_points)
