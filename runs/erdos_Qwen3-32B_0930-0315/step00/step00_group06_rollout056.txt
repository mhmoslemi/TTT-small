import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use 256 points for high-resolution optimization
    n_points = 256
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available, else use uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.random.rand(n_points)
    h *= n_points / 2 / h.sum()  # Normalize to have total integral 1

    best_h = h.copy()
    best_c5 = float("inf")

    learning_rate = 0.1
    decay_rate = 0.9999
    perturb_interval = 100
    it = 0
    
    while time.time() - start_time < budget_s - 0.5:
        it += 1

        # Evaluate correlation overlaps
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        overlaps = overlaps[n_points - 1 : 3 * n_points - 1]
        k_max = np.argmin(overlaps)  # We seek to decrease max overlap
        min_overlap = overlaps[k_max]

        # Gradient update: To decrease overlap(k_max) = sum h[i] * (1 - h[i + k_max])
        # So the negative gradient is +h[i], and -1 + 2h[i + k_max] at i + k_max
        grad = np.zeros_like(h)
        grad[:n_points - k_max] += h[k_max:]
        grad[k_max:] -= h[:n_points - k_max]

        h -= learning_rate * grad

        # Project to [0, 1]
        h = np.clip(h, 0.0, 1.0)

        # Normalize to total sum = n_points / 2
        h *= (n_points / 2) / h.sum()

        # Perturb occasionally to escape local optima
        if it % perturb_interval == 0:
            noise = np.random.normal(0, 0.02, n_points)
            h = np.clip(h + noise, 0.0, 1.0)
            h *= (n_points / 2) / h.sum()

        # Update learning rate
        learning_rate *= decay_rate

        # Evaluate best so far
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        overlaps = overlaps[n_points - 1 : 3 * n_points - 1]
        current_c5 = np.max(overlaps)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Break early if converged
        if time.time() - start_time > budget_s - 0.5:
            break

    return best_h, best_c5, n_points
