import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Start from the initial construction
    n = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)

    # Normalize the initial function so that it integrates to 1
    dx = 2.0 / n
    mass = np.sum(h) * dx
    h /= mass
    h *= 1.0

    # Evaluate starting C₅
    t0 = time.time()
    def obj_and_report(h_):
        h_ = np.clip(h_, 0, 1)
        result = evaluate_erdos_solution(h_)
        return result

    best_result = obj_and_report(h)
    best_h = h.copy()
    best_c5 = best_result['C5']

    # Configurable parameters
    max_iterations = 2000
    alpha = 0.1
    alpha_decay = 0.995
    patience = 20
    tolerance = 1e-5

    # Optimization loop
    iteration = 0
    stagnation = 0
    t1 = time.time()

    while iteration < max_iterations and (time.time() - t0) < budget_s - 10:
        # Compute correlation and find worst lag
        corr = np.correlate(h, 1 - h, mode="full") * dx
        indices = np.arange(len(corr))
        mask = (indices >= n) & (indices <= 2 * n)
        worst_lag = indices[mask][np.argmax(corr[mask])]

        # Compute gradient
        grad = np.zeros(n)
        for i in range(n):
            if 0 <= i - worst_lag + n < n:
                grad[i] -= h[(i - worst_lag + n) % n]
            if n <= worst_lag + i < n + n:
                grad[i] -= 1 - h[worst_lag + i - n]

        # Project gradient step
        h_proposed = h - alpha * grad
        h_proposed = np.clip(h_proposed, 0, 1)
        prop_mass = np.sum(h_proposed) * dx
        h_proposed *= 1.0 / prop_mass

        # Line search with restart
        improved = False
        for beta in [0.95, 0.8, 0.6, 0.5, 0.3, 0.1]:
            h_mix = h * beta + h_proposed * (1 - beta)
            h_mix = np.clip(h_mix, 0, 1)
            h_mix /= np.sum(h_mix) * dx
            result = obj_and_report(h_mix)
            c5 = result['C5']
            if c5 < best_c5 - 1e-6:
                best_c5 = c5
                best_h = h_mix.copy()
                best_result = result
                h = h_mix.copy()
                improved = True
                stagnation = 0
                break

        if not improved:
            stagnation += 1
            alpha *= alpha_decay

        if stagnation >= patience:
            break

        iteration += 1

        # Progress diagnostics
        if iteration % 10 == 0:
            curr_c5 = best_result['C5']
            elapsed = time.time() - t0
            print(f"iter {iteration} | elapsed: {elapsed:.2f}s | best_c5: {curr_c5:.6f} | alpha: {alpha:.4f}")

        # Check time constraint
        if time.time() - t0 >= budget_s - 1:
            print("Time budget reached, stopping")
            break

    # Return best_h and best bound
    return np.clip(best_h, 0, 1), best_c5, len(best_h)
