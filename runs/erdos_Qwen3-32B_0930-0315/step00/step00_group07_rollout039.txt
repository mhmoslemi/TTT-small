import numpy as np
import cvxpy as cp
import time


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = 80
    dx = 2.0 / n_points

    # Start from the initial construction if available
    try:
        h = initial_h_values
    except NameError:
        # Fallback initialization: balanced uniform starting point
        h = np.zeros(n_points, dtype=float)
        step = n_points // 2
        h[:step] = 1.0
        h[step:] = 0.0

    def compute_c5(h):
        h = np.array(h)
        ones_h = 1 - h
        corrs = np.correlate(h, ones_h, mode="full")
        return np.max(corrs) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    h_var = cp.Variable(n_points)

    while time.time() - start_time < budget_s:
        # Step 1: Evaluate current h and find worst-case lag
        h_np = np.array(h)
        ones_h = 1 - h_np
        corrs = np.correlate(h_np, ones_h, mode="full") * dx
        worst_k = np.argmax(corrs)

        # Step 2: Build shifted version of h for chosen lag
        def shift(x, k):
            return np.concatenate([x[-k:], x[:-k]]) if k != 0 else x

        h_shifted = shift(h_np, worst_k)
        ones_h_shifted = 1 - h_shifted

        # Step 3: Formulate convex problem
        objective = cp.Minimize(cp.sum(h_var * ones_h_shifted))
        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]

        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.ECOS, verbose=False)

        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_new = np.clip(h_var.value, 0.0, 1.0)
            # Update solution if improved
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h = h_new  # Continue local improvement
        else:
            break  # Exit if solver fails

        # Check stagnation (optional early stopping)
        if best_c5 < 0.380876:
            break  # Goal achieved

    return best_h.tolist(), best_c5, n_points
