import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp


def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    p1 = np.arange(n_points)
    p2 = p1
    p1, p2 = np.meshgrid(p1, p2, indexing='ij')
    k = p1 - p2
    selected = np.logical_or(k == 0, k == 1, k == 2, k == 3, k == 4)
    h_flat = h.reshape(-1, 1)
    overlap = (h_flat.T * (1 - h_flat)).flatten()
    overlap = np.where(selected, overlap, -np.inf)
    max_overlap = np.max(overlap)
    return max_overlap * dx


def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    h = np.copy(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    h_best = np.copy(h)
    c5_best = evaluate_erdos_solution(h)

    start_time = time.time()

    iter = 0
    max_iters = 1000
    step_size = 0.1
    schedule_decay = 50
    decay_rate = 0.5

    while time.time() - start_time < budget_s and iter < max_iters:
        iter += 1

        if iter % schedule_decay == 0:
            step_size *= decay_rate

        hh = h.reshape(-1, 1)
        corr = correlate(hh, 1 - hh, mode="full").ravel()
        lags = np.arange(-n_points + 1, n_points)
        valid_lags = lags[0:5]  # Only k = 0,1,2,3,4
        vals = corr[(n_points - valid_lags - 1).astype(int)]
        max_loc = np.argmax(vals)
        kopt = valid_lags[max_loc]

        grad = np.zeros_like(h)
        h0 = h
        h1 = np.roll(h, shift=-kopt)
        grad += (1 - h1)
        grad += h0 * (-1 * np.roll((1 - h1).copy(), shift=kopt))

        h = h - step_size * grad
        h = np.clip(h, 0, 1)
        hsum = np.sum(h)
        h_scale = n_points / 2 / hsum
        h *= h_scale

        c5 = evaluate_erdos_solution(h)

        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()

        # Perturbation every 50 iters
        if iter % 50 == 0:
            noise = np.random.normal(0, 0.01, size=n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            hsum = np.sum(h)
            h_scale = n_points / 2 / hsum
            h *= h_scale

    return h_best, c5_best, n_points
