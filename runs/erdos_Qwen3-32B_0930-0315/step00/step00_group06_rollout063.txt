```python
import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_C5(h, dx):
    h1 = h.copy()
    h2 = 1 - h.copy()
    corr = correlate(h1, h2, mode='full')
    return (2.0 / len(h)) * np.max(corr * dx)

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    h *= n_points / 2.0 / np.sum(h)  # renormalize
    return h

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    
    # Use the initial construction, or a uniform one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
        n_points = len(h)
    else:
        h = np.full(256, 0.5)  # uniform density
        n_points = len(h)

    dx = 2.0 / n_points
    h = project_h(h, n_points)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_C5(h, dx)
    last_update = time.time()

    # Optimization settings
    max_steps = 500000
    step_size = 0.01
    PRINT_FREQ = 5000
    PERTURB_FREQ = 50000

    for i in range(max_steps):
        t = time.time()
        if t - start_time >= budget_s:
            break

        h = project_h(h, n_points)

        # Compute overlap integral and find max overlap
        h1 = h
        h2 = 1 - h
        corr = correlate(h1, h2, mode='full')
        C5 = (2.0 / n_points) * np.max(corr * dx)

        # Check for improvements
        if C5 < best_c5:
            best_c5 = C5
            best_h = h.copy()
            last_update = time.time()

        # Check for divergence; reset if no improvement in 100 steps
        if (t - last_update) > 60:
            h = project_h(np.random.uniform(size=n_points), n_points)
            last_update = t

        # Periodic small perturbation to escape local minima
        if i % PERTURB_FREQ == 0:
            perturbation = np.random.normal(0, 0.02, n_points)
            h = project_h(h + perturbation, n_points)

        # Convex optimization: Fixing the most problematic lag and optimizing for values
        if i % 1000 == 0:
            h1 = h
            h2 = 1 - h
            corr = correlate(h1, h2, mode='full')
            k_max = np.argmax(corr)
            k_start = max(k_max - 10, 0)
            k_end = min(k_max + 11, n_points)

            # Construct a windowed objective with overlapping regions
            h_var = cp.Variable(n_points, nonneg=True)
            constraints = [
                h_var <= 1,
                cp.sum(h_var) == n_points / 2
            ]
            weighted_obj = 0
            for k in range(k_start, k_end):
                shifted_h = np.roll(h_var, -k)
                overlap = cp.sum(h_var * (1 - shifted_h))
                weighted_obj += overlap

            prob = cp.Problem(cp.Minimize(weighted_obj), constraints)
            result = prob.solve(solver=cp.GUROBI, verbose=False)
            if result < best_c5 * n_points:
                h = h_var.value
                best_h = h.copy()
                h = project_h(h, n_points)
                best_c5 = evaluate_C5(h, dx)
                last_update = t

        # Gradient descent step
        h1 = h
        h2 = 1 - h
        corr = correlate(h1, h2, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        if k_max < n_points:
            overlap = h * (1 - np.roll(h, -k_max))
            dloss = 1 - 2 * h - 2 * np.roll(h, k_max)
            grad = dloss

        # Projected gradient descent
        h = project_h(h + step_size * grad, n_points)

        # Logging
        if i % PRINT_FREQ == 0:
            print(f"Step {i}: C₅ = {best_c5:.6f}, time={t - start_time:.2f}s")

    h_values = best_h
    c5_bound = best_c5
    return h_values, c5_bound, n_points
```
