import numpy as np
from scipy.optimize import minimize
from math import floor, ceil


def evaluate_solution(h, n_points):
    h = np.ascontiguousarray(h).ravel()
    h_bin = h * n_points / h.size  # restore to n_points resolution if downscaled
    dx = 2.0 / n_points
    rh = np.correlate(h_bin, 1 - h_bin, mode="full")
    c5 = np.max(rh) * dx
    return c5


def get_penalty(h, n_points):
    return np.power(np.sum(h) - n_points / 2, 2)


def objective(h, n_points, k, lambda_penalty=1e4):
    h = h * n_points / h.size  # restore to n_points resolution if downscaled
    dx = 2.0 / n_points
    n = len(h)
    lag = k
    overlap = 0.0

    for i in range(n - lag):
        overlap += h[i] * (1 - h[i + lag])

    return overlap * dx + lambda_penalty * get_penalty(h, n_points)


def compute_best_lag(h, n_points):

    rh = np.correlate(h_bin, 1 - h_bin, mode="full")
    c5 = np.max(rh) * dx
    best_lag = np.argmax(rh)
    return best_lag
