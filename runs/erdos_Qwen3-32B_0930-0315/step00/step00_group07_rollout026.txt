import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    dx = 0.025  # 2 / 80
    desired_sum = 40  # n_points / 2 = 80 / 2 = 40

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        h = np.ones(80, dtype=np.float64) / 80 * 40  # uniform [0,1], ∑h = 40

    t0 = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)

    while time.time() - t0 < budget_s:
        # Normalize to ensure sum constraint
        if not np.allclose(np.sum(h), desired_sum):
            h = h / np.sum(h) * desired_sum

        # Ensure h ∈ [0, 1]
        h = np.clip(h, 0, 1)

        overlaps = np.correlate(h, 1 - h, mode="full")[80:] * dx
        worst_lag = np.argmax(overlaps)
        worst_c5 = overlaps[worst_lag]

        if worst_c5 < best_c5:
            best_c5 = worst_c5
            best_h = h.copy()

        # Compute gradient for worst lag
        n_points = len(h)
        k = worst_lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            plus_idx = (i + k) % n_points
            minus_idx = (i - k) % n_points
            gradient[i] = 1 - h[plus_idx] - h[minus_idx]

        # Solve LP to minimize overlap
        delta = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(cp.multiply(gradient, delta) * dx))
        constraints = [
            0 <= h + delta,
            h + delta <= 1,
            cp.sum(delta) == 0,
        ]
        prob = cp.Problem(objective, constraints)

        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except cp.SolverError:
            continue

        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h += delta.value
            h = np.clip(h + delta.value, 0, 1)

    h = best_h
    c5_bound = evaluate_erdos_solution(h, dx)

    return (h.tolist(), c5_bound, len(h))
