import numpy as np
import time
from numpy.random import rand, choice
from numpy import correlate
import evaluate_erdos_solution as ert

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():
        h_initial = initial_h_values
    else:
        raise ValueError("initial_h_values is required")

    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum_h = n_points / 2

    def compute_c5(h):
        h = h.astype(float)
        c = correlate(h, 1 - h, mode="full")
        return np.max(c) * dx

    def repair_sum(h, target_sum=target_sum_h):
        s = target_sum / np.sum(h)
        h = np.clip(h * s, 0, 1)
        return h

    population_size = 50
    mutation_rate = 0.15
    tournament_size = 5
    max_generations = 100_000
    best_solution = h_initial
    best_c5 = compute_c5(best_solution)
    time_limit = start_time + budget_s
    population = [np.clip(h_initial + 0.1 * rand(n_points), 0, 1) if n_points > 50 else h_initial for _ in range(population_size)]
    for p in population:
        p[:] = repair_sum(p)

    gen = 0
    while time.time() < time_limit:
        gen += 1
        new_population = []
        # Evaluate fitness
        def fitness(ind):
            return -compute_c5(ind)
        # Selection
        selected = []
        for _ in range(population_size // 2):
            competitors = [choice(population) for _ in range(tournament_size)]
            selected.append(min(competitors, key=compute_c5))
        # Crossover
        for i in range(0, len(selected), 2):
            if i + 1 >= len(selected):
                continue
            p1, p2 = selected[i], selected[i + 1].copy()
            cx = np.random.randint(1, n_points)
            o1 = np.concatenate((p1[:cx], p2[cx:]))
            o2 = np.concatenate((p2[:cx], p1[cx:]))
            for o in [o1, o2]:
                o[:] = repair_sum(o)
                if o.shape != (n_points,):
                    o = repair_sum(np.resize(o, n_points))
                new_population.append(o)
        # Mutation
        for o in new_population:
            if np.random.rand() < mutation_rate:
                mut_indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                o[mut_indices] += np.random.normal(0, 0.1, size=len(mut_indices))
                o = np.clip(o, 0, 1)
                o = repair_sum(o)
        # Replacement
        population = population + new_population
        population = sorted(population, key=compute_c5)[:population_size]
        current_best = min(population, key=compute_c5)
        current_c5 = compute_c5(current_best)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_solution = current_best
        if gen % 100 == 0:
            print(f"Gen {gen}: C5 = {best_c5:.6f}")

    return (best_solution, best_c5, n_points)
