import numpy as np
import time
import scipy
from scipy.signal import correlate
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 160
    dx = 2.0 / n_points
    h_init = initial_h_values
    h = h_init.copy()
    n = len(h)

    def objective(h):
        h_flat = h.flatten()
        h_bar = 1 - h_flat
        corr = correlate(h_flat, h_bar, mode="full")
        C5 = np.max(corr) * dx
        return C5

    def feasible_project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scale = (n_points / 2) / total
        h *= scale
        h = np.clip(h, 0, 1)  # enforce [0, 1]
        return h

    def gradient(h):
        h_flat = h.flatten()
        h_bar = 1 - h_flat
        H = np.lib.stride_tricks.sliding_window_view(np.concatenate([h_flat, h_flat, h_flat]), 2)
        dh = (1 - 2 * h_flat) * np.sum(H[:, 0] - H[:, 1], axis=0)
        return dh

    best_h = h.copy()
    best_c5 = float('inf')

    # Phase 1: Gradient descent
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        g = gradient(h)
        alpha = 0.01
        for _ in range(10):
            h_candidate = h - alpha * g
            h_candidate = feasible_project(h_candidate)
            c5 = objective(h_candidate)
            if c5 < objective(h):
                break
            alpha *= 0.5
        h -= alpha * g
        h = feasible_project(h)

        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        if best_c5 < 0.38:
            break

    # Phase 2: Smoothing and refinement
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        d = np.random.uniform(-0.05, 0.05, size=n)
        h_candidate = h + d
        h_candidate = feasible_project(h_candidate)
        c5 = objective(h_candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()

    h_values = best_h
    c5_bound = evaluate_erdos_solution(h_values, n_points=n_points, return_c5=True)

    return h_values, c5_bound, n_points
