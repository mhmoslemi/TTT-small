target_sum = n_points / 2  # h must sum to this


def compute_c5(h, dx):
    h_conj = 1 - h
    corr = np.correlate(h, h_conj, mode="full")
    corr *= dx
    return np.max(corr)


def project_onto_box_and_sum(h):
    # Box projection
    h_clipped = np.clip(h, 0.0, 1.0)
    # Ensure sum remains same
    sum_deviation = np.sum(h_clipped) - target_sum
    if abs(sum_deviation) < 1e-9:
        return h_clipped
    # Correction: Add/subtract from all elements
    h_corrected = h_clipped - (sum_deviation / n_points)
    # Check if going out of bounds
    near_bounds = np.isclose(h_corrected, 0) | np.isclose(h_corrected, 1)
    if any(near_bounds):
        idxs = np.where(near_bounds)
        h_remaining = h_corrected.copy()
        h_remaining[idxs] = 0.0 if sum_deviation < 0 else 1.0
        h_remaining_sum = np.sum(h_remaining)
        adjustment = (target_sum - h_remaining_sum) / (n_points - len(idxs[0]))
        h_remaining[~near_bounds] += adjustment
        return np.clip(h_remaining, 0, 1)
    return h_corrected


def compute_gradient(h, dx):
    h = np.reshape(h, (n_points,))
    h_conj = 1.0 - h
    grad = np.zeros_like(h)
    c5 = compute_c5(h, dx)

    for k in range(1 - n_points, n_points):
        shift_h = np.roll(h, -k)
        corr = np.dot(h, 1 - shift_h)
        corr *= dx
        if abs(corr - c5) < 1e-9:
            # Found the max index
            grad_h = 1 - shift_h - h
            grad += grad_h
    return grad - 0.1 * h  # Small regularization
