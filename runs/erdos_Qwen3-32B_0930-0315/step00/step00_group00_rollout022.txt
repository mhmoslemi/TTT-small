import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    # Use provided initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n = len(h)
        assert np.sum(h) == n / 2, "Initial h does not satisfy the constraint"
    else:
        n = 45  # default points
        h = np.zeros(n)
        h[::2] = 1.0  # binary starting configuration
        h[-1] = 0.0  # ensure exact constraint
        h /= h.sum() * 2  # normalize to unit mass
        h *= n / 2  # exact sum constraint

    def compute_c5(h):
        dx = 2.0 / len(h)
        inv_h = 1.0 - h
        corr_values = correlate(h, inv_h, mode='full')
        corr_values = corr_values * dx
        return np.max(corr_values)

    def perturb_h(h, n_points, budget_move=0.1):
        h = h.copy()
        n_elem = np.random.randint(1, 6)  # Perturb 1-5 elements
        indices = np.random.choice(n_points, size=n_elem, replace=False)
        delta = np.random.uniform(-budget_move, budget_move, size=n_elem)
        h[indices] += delta
        h = np.clip(h, 0.0, 1.0)
        # Project to constraint sum(h) = n_points / 2
        h_sum = np.sum(h)
        scaling = (n_points / 2.0) / h_sum
        h = h * scaling
        h = np.clip(h, 0.0, 1.0)
        return h

    def shift_h(h, step):
        h = h.copy()
        h = np.roll(h, shift=step)
        h = perturb_h(h, len(h))
        return h

    # Tabu search setup
    tabu_memory = set()
    max_neighbors = 20
    tabu_size = 20
    best_c5 = float("inf")
    best_h = None
    start_time = time.time()

    while time.time() - start_time < budget_s:
        neighbors = []
        for _ in range(max_neighbors):
            # Generate diverse neighbors
            if np.random.rand() < 0.5:
                neighbor = shift_h(h, np.random.randint(2, 4))
            else:
                neighbor = perturb_h(h, len(h))
            h_tuple = tuple(neighbor.round(8))
            if h_tuple not in tabu_memory:
                neighbors.append(neighbor)
        np.random.shuffle(neighbors)

        # Evaluate neighbors
        for neighbor in neighbors:
            if time.time() - start_time >= budget_s:
                break
            c5 = compute_c5(neighbor)
            if c5 < best_c5:
                h = neighbor
                best_c5 = c5
                best_h = h
                tabu_memory.add(tuple(np.round(h, 8)))
                while len(tabu_memory) > tabu_size:
                    tabu_memory.pop()

    return (best_h.tolist(), best_c5, len(best_h))
