import scipy.optimize as optimize

def run_packing():

    num_circles = 26

    cols = int(np.ceil(num_circles / rows))

    h = 1.0 / (cols + 1)
    w = np.sqrt(3) / (rows + 1)

    centers = np.zeros((num_circles, 2))

    radii = np.full(num_circles, 1.0 / (rows + 1))

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum (maximize sum)

    def constraint_overlap(radii, centers):

                if dist < radii[i] + radii[j] - 1e-8:
                    return -1  # Constraint violation
        return 0
