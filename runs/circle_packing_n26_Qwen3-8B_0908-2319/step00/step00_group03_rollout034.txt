import numpy as np
from scipy.optimize import minimize
from typing import Tuple, Any

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initialize centers and radii with a rough guess
    # Use a regular hexagonal grid in a unit square
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    cells_per_row = int(np.sqrt(num_circles))
    cells_per_col = (num_circles + cells_per_row - 1) // cells_per_row
    row_spacing = 1.0 / (cells_per_row + 1)
    col_spacing = 1.0 / (cells_per_col + 1)
    
    for i in range(num_circles):
        row = i // cells_per_col
        col = i % cells_per_col
        x = col * col_spacing + col_spacing
        y = row * row_spacing + row_spacing
        centers[i, 0] = x
        centers[i, 1] = y
        # Initial guess for radii
        radii[i] = min(x, 1 - x, y, 1 - y) * 0.8  # 80% of minimal distance to boundary

    def objective(radii_vec):
        return -np.sum(radii_vec)

    def constraint_overlap(centers, radii_vec):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii_vec[i] + radii_vec[j] - 1e-9:
                    return -np.inf
        return 0.0

    def constraint_boundary(radii_vec):
        for i in range(num_circles):
            if radii_vec[i] < 0 or radii_vec[i] > 1:
                return -np.inf
        return 0.0

    bounds = [(0, 1) for _ in range(num_circles)]

    # Optimization
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': lambda r: constraint_overlap(centers, r)},
            {'type': 'ineq', 'fun': lambda r: constraint_boundary(r)}
        ],
        options={'maxiter': 1000}
    )

    optimal_radii = result.x
    centers_opt = centers.copy()
    sum_radii = np.sum(optimal_radii)

    return centers_opt, optimal_radii, sum_radii
