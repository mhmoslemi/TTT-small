import numpy as np
import scipy.optimize as opt
import math

def run_packing():
    # Initialize centers and radii
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Initial hexagonal packing with slight stagger for better utilization
    # Adjust spacing to fit 26 circles
    rows = 5
    cols = 6
    spacing = 0.25
    # Generate initial positions
    for i in range(rows):
        for j in range(cols):
            if i % 2 == 0:
                x = j * spacing
            else:
                x = (j + 0.5) * spacing
            y = i * spacing * math.sqrt(3)
            centers[i * cols + j] = [x, y]
    # Adjust to fit into unit square and ensure arcs within bounds
    margin = 0.1
    centers = centers / (1 + 2 * margin) * (1 - 2 * margin) + margin

    # Define objective function to maximize sum of radii
    def objective(x):
        return -np.sum(x)  # Negative to use minimization

    # Define constraints
    def constraints(x):
        # Ensure no two circles overlap
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < x[i] + x[j] - 1e-8:
                    return False
        # Ensure all circles are within the unit square

                return False
            if centers[i][1] - x[i] < 0 or centers[i][1] + x[i] > 1:
                return False
        return True

    bounds = [(0, 1) for _ in range(n)]
    result = opt.minimize(
        objective,
        x0=radii,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraints}

    sum_radii = np.sum(radii)

        return False, "NaN values present"
