import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
import matplotlib.pyplot as plt

def run_packing():

    def objective(radii, centers):
        return -np.sum(radii)  # Negative for minimization

    def constraint_non_overlap(centers, radii):
        n = centers.shape[0]
        return np.array([
            np.sqrt(np.sum((centers[i] - centers[j])**2)) - (radii[i] + radii[j]) for i in range(n) for j in range(i+1, n)

        n = centers.shape[0]
        return np.array([
            centers[i, 0] - radii[i] + 1e-12 for i in range(n)
        ]) + np.array([
            centers[i, 1] - radii[i] + 1e-12 for i in range(n)
        ]) + np.array([
            1 - (centers[i, 0] + radii[i]) + 1e-12 for i in range(n)

            1 - (centers[i, 1] + radii[i]) + 1e-12 for i in range(n)
        ])

    # Generate initial grid for hexagonal packing
    def generate_initial_positions(n):

            radii = np.full(n, 0.1)
            positions = []
            for i in range(n):
                row = i // 5
                col = i % 5
                x = col * (1 + radii[i]) + radii[i] + padding
                y = row * (np.sqrt(3) * (1 + radii[i])) + radii[i] + padding
                positions.append(np.array([x, y]))
            return np.array(positions), radii

        initial_guess = np.concatenate([positions.flatten(), radii])
        result = minimize(
