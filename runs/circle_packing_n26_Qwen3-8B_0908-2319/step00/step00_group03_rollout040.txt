import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    # Initial guess: 4 rows x 7 columns grid, slightly offset for hexagonal packing
    num_circles = 26
    grid_rows, grid_cols = 4, 7
    centers = np.zeros((num_circles, 2))
    radii = np.ones(num_circles)
    
    # Initial grid (uniform spacing)
    spacing = 1.0 / (grid_cols - 1)
    for i in range(grid_rows):
        for j in range(grid_cols):
            x = j * spacing
            y = i * spacing
            centers[i * grid_cols + j] = [x, y]
    
    # Initial guess for optimization
    initial_guess = np.hstack((centers.reshape(-1), radii))

    def objective(x):
        # Flatten x into centers and radii
        centers_flat = x[:2 * num_circles].reshape(-1, 2)
        radii_flat = x[2 * num_circles:]
        
        # Objective: Maximize sum of radii
        sum_radii = np.sum(radii_flat)
        return -sum_radii  # Minimize the negative to maximize

    def constraint_overlap(x):
        centers_flat = x[:2 * num_circles].reshape(-1, 2)
        radii_flat = x[2 * num_circles:]
        dist_matrix = squareform(pdist(centers_flat))
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = dist_matrix[i, j]
                constraints.append(dist - (radii_flat[i] + radii_flat[j]))
        return np.array(constraints)

    def constraint_bounds(x):
        centers_flat = x[:2 * num_circles].reshape(-1, 2)
        radii_flat = x[2 * num_circles:]
        constraints = []
        for i in range(num_circles):
            x, y = centers_flat[i]
            r = radii_flat[i]
            constraints.append(x - r)
            constraints.append(1 - x - r)
            constraints.append(y - r)
            constraints.append(1 - y - r)
        return np.array(constraints)

    constraints = [
        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'ineq', 'fun': constraint_bounds}
    ]

    bounds = Bounds(lb=np.zeros(2 * num_circles + num_circles),
                    ub=np.array([1.0 + 1e-12] * (2 * num_circles) + [1.0] * num_circles))

    result = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=constraints)

    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

    centers_opt = result.x[:2 * num_circles].reshape(-1, 2)
    radii_opt = result.x[2 * num_circles:]

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
