import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import warnings

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize the total sum of radii."""

    n = 26
    num_rows = int(np.ceil(np.sqrt(n)))
    num_cols = int(np.ceil(n / num_rows))

    initial_centers = np.zeros((n, 2))
    for i in range(n):
        row = i // num_cols
        col = i % num_cols

        y = row / (num_rows + 1)
        initial_centers[i] = [x, y]
    
    # Define the initial radii: start with uniform radius
    initial_radii = np.full(n, 1.0 / (np.sqrt(2) * num_rows * num_cols))

        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraints for non-overlap and boundary
    def constraint_overlap(x, centers, radii):
        # x is concatenated [centers_flat, radii_flat]

        return 0.0

        return 0.0

    # Initial guess

    # Define constraints for optimization

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
