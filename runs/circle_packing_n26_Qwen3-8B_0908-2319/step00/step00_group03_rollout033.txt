import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial guess for centers (random within square)
    centers = np.random.rand(26, 2)
    # Initial guess for radii (small values)
    radii = np.random.rand(26) * 0.05
    
    # Objective function: maximize sum of radii
    def objective(x):
        # x contains flattened centers and radii
        flat_centers = x[:26*2].reshape(26, 2)
        flat_radii = x[26*2:]
        return -np.sum(flat_radii)  # Negative for maximization

    # Constraints: non-overlap and boundary conditions
    def non_overlap_constraint(x):
        flat_centers = x[:26*2].reshape(26, 2)
        flat_radii = x[26*2:]
        for i in range(26):
            for j in range(i + 1, 26):
                dx = flat_centers[i, 0] - flat_centers[j, 0]
                dy = flat_centers[i, 1] - flat_centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < flat_radii[i] + flat_radii[j] - 1e-8:
                    return dist - flat_radii[i] - flat_radii[j] + 1e-8
        return 0.0

    def boundary_constraint(x):
        flat_centers = x[:26*2].reshape(26, 2)
        flat_radii = x[26*2:]
        for i in range(26):
            x_i, y_i = flat_centers[i]
            r_i = flat_radii[i]
            if x_i - r_i < -1e-8 or x_i + r_i > 1 + 1e-8 or y_i - r_i < -1e-8 or y_i + r_i > 1 + 1e-8:

        return 0.0

    # Set up optimization problem
    bounds = Bounds([0, 0], [1, 1])  # for centers
    bounds = Bounds([0]*26*2 + [0]*26, [1]*26*2 + [1]*26)  # for centers and radii

    x0 = np.concatenate([centers.flatten(), radii])

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
