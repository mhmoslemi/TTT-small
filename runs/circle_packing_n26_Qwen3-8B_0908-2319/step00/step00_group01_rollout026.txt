def run_packing():
    # Initialize circles in hexagonal packing
    num_circles = 26
    grid_rows = 5
    grid_cols = 5
    circle_radius = 0.04

    # Start with a simple grid-like arrangement
    centers = np.zeros((num_circles, 2))
    radii = np.full(num_circles, circle_radius)

    # Simple hexagonal grid with offset
    for i in range(grid_rows):
        for j in range(grid_cols):
            x = j * (2 * circle_radius) + (i % 2) * circle_radius
            y = i * (np.sqrt(3) * circle_radius)
            idx = i * grid_cols + j
            centers[idx] = [x, y]
            radii[idx] = circle_radius

    # We'll refine this using a more sophisticated optimization using simulated annealing
    # Reformulate the problem into a minimization of the non-overlap and boundary constraints

    # Define objective function
    def objective(x):
        # x is a vector of [x1, y1, r1, x2, y2, r2, ...]
        # Reshape to centers and radii
        centers_flat = x[:num_circles * 2].reshape(num_circles, 2)
        radii_flat = x[num_circles * 2:].reshape(num_circles)
        # Compute the sum of radii to maximize (used as negative for minimization)
        return -np.sum(radii_flat)

    # Define constraints
    def constraint_1(x):
        centers_flat = x[:num_circles * 2].reshape(num_circles, 2)
        radii_flat = x[num_circles * 2:].reshape(num_circles)
        for i in range(num_circles):
            x1, y1 = centers_flat[i]
            r1 = radii_flat[i]
            if x1 - r1 < 0 or x1 + r1 > 1 or y1 - r1 < 0 or y1 + r1 > 1:
                return 1.0
        return 0.0

    def constraint_2(x):
        centers_flat = x[:num_circles * 2].reshape(num_circles, 2)

                x2, y2 = centers_flat[j]
                r1 = radii_flat[i]
                r2 = radii_flat[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                if dist < r1 + r2 - 1e-8:

        centers.flatten(),

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
