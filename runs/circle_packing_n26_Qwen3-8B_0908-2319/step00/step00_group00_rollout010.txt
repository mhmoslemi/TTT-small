from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    grid_size = math.ceil(math.sqrt(n))

        row = i // grid_size
        col = i % grid_size
        centers[i, 0] = col / (grid_size - 1)  # column-wise placement
        centers[i, 1] = row / (grid_size - 1)  # row-wise placement

    # Define objective function: maximize sum of radii
    def objective(x):
        # x is a flat array: first n*2 elements are centers, then n elements are radii
        centers_flat = x[:n * 2].reshape(n, 2)

        # Objective: maximize sum of radii (we minimize the negative)
        return -np.sum(radii_flat)

    # Define constraint for boundary
    def boundary_constraint(x):
        centers_flat = x[:n * 2].reshape(n, 2)
        radii_flat = x[n * 2:].reshape(n)
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if (x_i - r_i < 0 or x_i + r_i > 1 or
                y_i - r_i < 0 or y_i + r_i > 1):
                return -1  # violates constraint
        return 0

    # Define constraint for non-overlapping

                dist = math.hypot(dx, dy)

                    return -1  # violates constraint
        return 0

    # Initial guess (centers and radii)
    x0 = np.concatenate([centers.flatten(), radii])

    # Constraints

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
