import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    num_circles = 26
    grid_size = int(np.ceil(np.sqrt(num_circles)))
    centers = np.zeros((num_circles, 2))

    x, y = 0.5, 0.5  # Start from center
    radius = 0.1  # Initial guess
    for i in range(num_circles):
        # Hexagonal grid approximation
        row = i // grid_size
        col = i % grid_size
        angle = 2 * np.pi / 3 * row
        x = 0.5 + 0.5 * np.cos(angle) * (col + 0.5)
        y = 0.5 + 0.5 * np.sin(angle) * (row + 0.5)
        centers[i, 0] = x
        centers[i, 1] = y

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum

    def non_overlap_constraint(radii, centers):
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                constraints.append(dist - (radii[i] + radii[j]) >= -1e-8)
        return constraints

    def boundary_constraint(radii, centers):
        constraints = []
        for i in range(num_circles):
            x, y = centers[i, 0], centers[i, 1]
            r = radii[i]
            constraints.append(x - r >= -1e-8)
            constraints.append(x + r <= 1 + 1e-8)
            constraints.append(y - r >= -1e-8)
            constraints.append(y + r <= 1 + 1e-8)
        return constraints

    # Initialize radii
    radii = np.full(num_circles, 0.05)

    # Create bounds: radii >= 0
    bounds = Bounds(0, np.inf)

    # We use a local SQP solver that supports constraints
    constraints = []
    if num_circles > 1:
        constraints += non_overlap_constraint(radii, centers)
    constraints += boundary_constraint(radii, centers)

    # Wrap constraints to pass radii and centers
    def non_overlap_con(radii):
        return non_overlap_constraint(radii, centers)

    def boundary_con(radii):
        return boundary_constraint(radii, centers)

    cons = [
        {'type': 'ineq', 'fun': non_overlap_con},
        {'type': 'ineq', 'fun': boundary_con}
    ]

    # Optimization
    result = opt.minimize(
        fun=objective,
        x0=radii,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-8,
        options={'maxiter': 1000}

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
