import scipy.optimize as opt

    # Adjust centers so they're within the unit square
    centers = np.clip(centers, 0, 1)
    radii = np.clip(radii, 0, 0.5)  # Max radius is 0.5, as circles can't overlap

    def objective(x):
        # x has shape (2*n,)
        x_centers = x[:2*n].reshape(n, 2)

        return -np.sum(x_radii)  # Minimize negative sum to maximize sum
