import numpy as np
import scipy.optimize as optimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    # Initial guess: hexagonal packing inside the unit square
    n_circles = 26
    # Number of rows in hexagonal grid
    rows = 6
    cols = 5  # 5 columns to fit 26 circles (6 rows, 5 columns gives 26 circles)

    # Calculate the spacing for a hexagonal grid
    spacing = 1 / (cols + 1)  # Horizontal spacing
    vertical_spacing = np.sqrt(3) / (2 * (rows + 1))  # Vertical spacing

    # Generate initial positions
    centers = np.zeros((n_circles, 2))
    idx = 0
    for row in range(rows):
        for col in range(cols):
            x = col * spacing + spacing / 2
            y = row * vertical_spacing + vertical_spacing / 2
            # Alternate the horizontal offset for even rows
            if row % 2 == 1:
                x += spacing / 2
            centers[idx] = [x, y]
            idx += 1

    # Initial radii: calculate based on the minimum distance between circles
    dists = np.zeros(n_circles)
    for i in range(n_circles):
        dists[i] = np.min(np.linalg.norm(centers[i] - centers[j], 2) for j in range(n_circles) if j != i)
    radii = dists / 2

    def objective(radii_flat):
        return -np.sum(radii_flat)  # Minimize negative of sum to maximize sum

    def constraint_overlap(i, j, radii_flat):
        x_i, y_i = centers[i]
        x_j, y_j = centers[j]
        dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
        return radii_flat[i] + radii_flat[j] - dist

    def constraint_boundary(i, radii_flat):
        x_i, y_i = centers[i]
        r_i = radii_flat[i]
        return x_i - r_i, x_i + r_i - 1, y_i - r_i, y_i + r_i - 1

    # Prepare constraints
    constraints = []
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: constraint_overlap(i, j, r)})
    for i in range(n_circles):
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: constraint_boundary(i, r)[0]})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: constraint_boundary(i, r)[1]})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: constraint_boundary(i, r)[2]})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: constraint_boundary(i, r)[3]})

    # Initial guess for radii
    radii_flat = radii.copy()

    # Perform optimization
    result = optimize.minimize(

        radii_flat,
        method='SLSQP',
        constraints=constraints,
        bounds=[(0, 1) for _ in range(n_circles)]
