import numpy as np
from scipy.optimize import minimize
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial guess: staggered hexagonal packing with padding
    n_circles = 26
    num_rows = 5
    num_cols = 6

    # Hexagonal grid parameters
    spacing = 0.3
    row_spacing = np.sqrt(3) * spacing

    centers = np.zeros((n_circles, 2))
    radii = np.full(n_circles, 0.01)

    for i in range(num_rows):
        for j in range(num_cols):
            idx = i * num_cols + j
            if i % 2 == 0:
                x = j * spacing
            else:
                x = (j + 0.5) * spacing
            y = i * row_spacing
            centers[idx] = [x, y]
            radii[idx] = 0.2  # Initial guess for radius

    # Objective function to minimize (negative of sum of radii)
    def objective(x: np.ndarray) -> float:
        centers_flat = x[:n_circles * 2].reshape((n_circles, 2))
        radii_flat = x[n_circles * 2:]

        if np.any(radii_flat < 0):
            return float('inf')

        sum_radii = np.sum(radii_flat)
        return -sum_radii  # Minimize negative of sum_radii is equivalent to maximizing sum_radii

    # Constraint: boundary condition
    def boundary_constraint(x: np.ndarray) -> np.ndarray:
        centers_flat = x[:n_circles * 2].reshape((n_circles, 2))
        radii_flat = x[n_circles * 2:]
        return np.concatenate([
            centers_flat[:, 0] - radii_flat,
            1 - centers_flat[:, 0] - radii_flat,
            centers_flat[:, 1] - radii_flat,
            1 - centers_flat[:, 1] - radii_flat
        ])

    # Constraint: non-overlap between circles
    def non_overlap_constraints(x: np.ndarray) -> np.ndarray:
        centers_flat = x[:n_circles * 2].reshape((n_circles, 2))
        radii_flat = x[n_circles * 2:]
        constraints = []

        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                constraint = dist - (radii_flat[i] + radii_flat[j])
                constraints.append(constraint)

        return np.array(constraints)

    # Initial guess for all variables (positions + radii)
    x0 = np.concatenate([
        centers.flatten(),
        radii
    ])

    # Bounds: positions must be within [0, 1], radii must be non-negative
    bounds = []
    for _ in range(n_circles * 2):
        bounds.append((0.0, 1.0))  # centers
    for _ in range(n_circles):
        bounds.append((0.0, 1.0))  # radii

    # Constraints
    constraints = [
        {
            'type': 'ineq',
            'fun': boundary_constraint
        },
        {
            'type': 'ineq',
            'fun': non_overlap_constraints
        }
    ]

    # Optimization
    result = minimize(
        fun=objective,
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )

    centers_final = result.x[:n_circles * 2].reshape((n_circles, 2))
    radii_final = result.x[n_circles * 2:]
    sum_radii_final = np.sum(radii_final)

    # Validate before returning
    try:
        valid, message = validate_packing(centers_final, radii_final)
        if not valid:
            raise ValueError(message)
    except Exception as e:
        raise RuntimeError(f"Validation failed after optimization: {e}")

    return centers_final, radii_final, sum_radii_final

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
