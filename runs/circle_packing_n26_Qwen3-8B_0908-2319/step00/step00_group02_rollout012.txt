import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds
from scipy.optimize import minimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def sum_radii(radii):
    return np.sum(radii)

def initial_packing(num_circles):
    # Start with a 5x5 grid with one circle removed
    grid = np.array([[i / 4, j / 4] for i in range(5) for j in range(5)])
    grid = np.delete(grid, np.random.randint(0, 25), axis=0)
    return grid, np.full(num_circles, 0.05)

def objective_function(x, num_circles):
    # x is a flattened array of [x1, y1, r1, x2, y2, r2, ..., xn, yn, rn]
    centers = x.reshape((num_circles, 2))
    radii = x[num_circles * 2 :]

    # Check constraints
    valid, msg = validate_packing(centers, radii)
    if not valid:
        return -float('inf')

    # Calculate the sum of radii
    return np.sum(radii)

def bounds(num_circles):
    # Bounds for each circle: x, y in [0, 1], r >= 0
    lower = np.zeros(2 * num_circles + num_circles)
    upper = np.ones(2 * num_circles + num_circles)
    return Bounds(lower, upper)

def run_packing():
    num_circles = 26
    x0_centers, x0_radii = initial_packing(num_circles)
