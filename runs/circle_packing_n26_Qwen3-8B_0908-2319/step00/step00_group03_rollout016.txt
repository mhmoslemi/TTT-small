import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Initial configuration: hexagonal packing
    num_circles = 26
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))
    
    # Hexagonal grid spacing
    h = np.sqrt(3) / 2  # Vertical spacing between rows
    spacing = 1.0 / (cols)  # Horizontal spacing between columns
    
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    for i in range(rows):
        for j in range(cols):
            if i % 2 == 0:
                x = j * spacing
            else:
                x = (j + 0.5) * spacing
            y = i * h
            if x + 0.5 * spacing <= 1 and y + 0.5 * h <= 1:
                centers[i * cols + j] = [x, y]
                radii[i * cols + j] = 0.5 * spacing  # Initial guess for radius

    # Refine the packing
    def objective(x):
        # Reshape x to centers and radii
        centers_flat = x[:num_circles * 2].reshape(num_circles, 2)
        radii_flat = x[num_circles * 2:].reshape(num_circles)
        
        # Objective: maximize the sum of radii
        sum_radii = np.sum(radii_flat)
        return -sum_radii  # Minimize the negative sum to maximize the sum

    def constraints(x):
        centers_flat = x[:num_circles * 2].reshape(num_circles, 2)
        radii_flat = x[num_circles * 2:].reshape(num_circles)
        
        # Constraint 1: All circles are within the unit square
        for i in range(num_circles):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return np.inf  # Penalize out-of-bound circles

        # Constraint 2: No overlapping between circles
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return np.inf  # Penalize overlapping circles

        return 0  # No constraints violated

    # Define bounds for centers and radii
    x_bounds = []
    for i in range(num_circles):
        x_bounds.extend([(0, 1), (0, 1)])  # x and y bounds for each circle
        x_bounds.extend([(0, 1)])         # radius bounds for each circle

    # Use a sequential optimization approach
    # Initial guess is our hexagonal packing
    x0 = np.concatenate([centers.reshape(-1), radii])

    # Perform optimization with constraints
    result = opt.minimize(

        x0=x0,
        method='SLSQP',

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
