import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial guess for circle centers (hexagonal packing)
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # First, place a hexagonal grid in the square with some small padding
    num_rows = int(np.ceil(np.sqrt(n)))
    num_cols = int(np.ceil(n / num_rows))
    max_r = 0.025  # small initial radius
    radii[:] = max_r
    row_height = 2 * max_r * np.sqrt(3)  # vertical distance between rows
    col_width = 2 * max_r  # horizontal distance between columns
    padding = 0.05  # padding from square boundaries

    for i in range(n):
        row = i // num_cols
        col = i % num_cols
        x = padding + col * col_width
        y = padding + row * row_height
        centers[i] = [x, y]

    # Define objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # negative for minimization

    # Define constraints
    def constraint_centers(radii):
        return np.array([radii[i] for i in range(n)])

    def constraint_radii(radii):
        return np.array([radii[i] for i in range(n)])

    # Bound on radii: nonnegative
    bounds = Bounds([0.0] * n, [np.inf] * n)

    # Function to validate packing
    def validate_packing(centers, radii):
        n = centers.shape[0]

        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"

        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]

    def optimize_radii(centers):

        res = opt.minimize(
