import numpy as np
import scipy.optimize as optimize
from scipy.optimize import constraints

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial grid-based positions
    n = 26
    grid = np.linspace(0, 1, 6)  # 5x5 grid with 6 points per side
    centers = np.zeros((n, 2))
    idx = 0
    for i in range(5):
        for j in range(5):
            centers[idx] = [grid[i], grid[j]]
            idx += 1

    # Initial radii (small for optimization)
    radii = np.full(n, 0.05)

    # Define optimization variables: positions and radii
    variables = np.concatenate([centers.flatten(), radii])

    def objective(variables):
        return -np.sum(variables[-n:])  # Minimize negative of sum of radii

    def constraint_positions(variables):
        return variables[:2*n].reshape(n, 2)

    def constraint_radii(variables):
        return variables[-n:]

    def constraint_boundary(variables):
        positions = constraint_positions(variables)
        radii = constraint_radii(variables)
        for i in range(n):
            x, y = positions[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return -1e-6
        return 0

    def constraint_overlap(variables):
        positions = constraint_positions(variables)
        radii = constraint_radii(variables)
        for i in range(n):
            for j in range(i + 1, n):
                dx = positions[i, 0] - positions[j, 0]
                dy = positions[i, 1] - positions[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-6:
                    return -1e-6
        return 0

    # Set up constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Optimization
    bounds = [(0, 1) for _ in range(2*n)] + [(0, 1) for _ in range(n)]
    result = optimize.minimize(
        objective,
        variables,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-8,
        options={'maxiter': 1000}
    )

    # Extract final results
    final_centers = result.x[:2*n].reshape(n, 2)
    final_radii = result.x[-n:]
    final_sum = np.sum(final_radii)

    # Validate the packing
    is_valid, message = validate_packing(final_centers, final_radii)
    if not is_valid:
        raise ValueError(f"Invalid packing: {message}")

    return final_centers, final_radii, final_sum

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
