import scipy.optimize as opt
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    n = 26

    # Initial guess: hexagonal grid
    # Determine the number of rows and columns in the hexagonal grid
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))

    # Compute the initial grid spacing
    spacing = 1.0 / (cols + 1)  # so circles are slightly spaced within the square

    centers = np.zeros((n, 2))
    radius_guess = 0.05  # initial radius guess

        return -np.sum(x)  # we minimize the negative sum to maximize the sum

    def constraint_center_bounds(x, centers):

        for i in range(n):
            x_i, y_i = centers[i]
            r_i = x[i]
            if (x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or
