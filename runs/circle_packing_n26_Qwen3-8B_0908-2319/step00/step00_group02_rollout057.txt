import scipy.optimize as optimize

    cols = 5
    spacing = 0.25  # Adjust spacing to fit the unit square

    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * spacing * math.sqrt(3) / 2  # Hexagonal grid vertical spacing
            centers.append([x, y])

    centers = np.array(centers)
    radii = np.ones(n_circles) * 0.05  # Small initial radius

    # Scale to ensure all circles are within [0, 1]^2 and adjust radii accordingly
    x_min, x_max = min(centers[:, 0]), max(centers[:, 0])

    scale = 1.0 / max(x_max - x_min, y_max - y_min)
    centers = centers * scale
    radii *= scale

    # Step 3: Define the objective function (maximize sum of radii)
    def objective(radii):
        return -sum(radii)  # Minimize negative sum to maximize sum
