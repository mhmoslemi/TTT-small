def run_packing():
    # Number of circles
    n = 26

    # Rows of hexagonal packing
    rows = 6
    cols_per_row = 5
    radius = 0.05  # initial guess for radius

    # Generate hexagonal grid within the unit square
    centers = []
    for i in range(rows):
        for j in range(cols_per_row):
            x = j + 0.5 * i
            y = i * np.sqrt(3) / 2

                x += 0.5
            # Scale to fit within the unit square

            centers.append([x, y])
    centers = np.array(centers)

    radii = np.full(n, radius)

    # Optimization parameters
    max_iter = 1000
    tolerance = 1e-6

    # Define objective function: maximize sum of radii
    def objective(x):
        return -np.sum(x)  # Negative because we minimize

    # Constraints: circles must not overlap

                    return dist - (x[i] + x[j]) + 1e-8
        return 0

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
