from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    """Packs 26 circles in a unit square to maximize the sum of their radii."""
    # Initial layout: hexagonal packing
    num_circles = 26
    rows = 4
    cols = 7
    # Initial hexagonal grid, offset for rows
    centers = np.zeros((num_circles, 2))
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3) / 2
            centers[i * cols + j] = [x, y]

    # Initial guess for radii: all equal
    radii = np.ones(num_circles) * 0.15

    # Define the objective function (maximize sum of radii)
    def objective(radii):
        return -np.sum(radii)  # Minimize the negative of the sum

    # Define the constraints
    def constraint_overlap(radii):
        dists = cdist(centers, centers)
        return np.array([dists[i, j] - radii[i] - radii[j] for i in range(num_circles) for j in range(i+1, num_circles)])

    def constraint_boundary(radii):
        left = centers[:, 0] - radii
        right = centers[:, 0] + radii
        bottom = centers[:, 1] - radii
        top = centers[:, 1] + radii
        return np.array([left, right, bottom, top]).flatten()

    # Bounds for radii
    bounds = [(0, 1) for _ in range(num_circles)]

    # Setup optimization
    constraints = [
        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'ineq', 'fun': constraint_boundary}
    ]
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8)

    # Extract the result
    centers = centers[np.argsort(result.success)]
    radii = result.x
    sum_radii = np.sum(radii)

    valid, message = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Invalid packing: {message}")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
