import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 128
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.zeros(n_points)
    h[::2] = 1.0
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Phase 1: Refinement with SLSQP
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-9}
    )
    best_h = result.x.copy()
    best_c5 = objective(best_h)
    
    # Phase 2: Structured perturbation with decreasing magnitude
    for perturb_size in np.logspace(-3, -1, num=5, base=10):
        if time.time() - start_time >= budget_s:
            break
        for _ in range(5):
            if time.time() - start_time >= budget_s:
                break
            perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
            perturbed_h = best_h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            
            result = opt.minimize(
                objective,
                perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-9}
            )
            
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = objective(current_h)
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
