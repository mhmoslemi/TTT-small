import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Local optimization with SLSQP
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8}
    )
    h = result.x
    best_h = h.copy()
    best_c5 = objective(h)
    
    # Metaheuristic search with simulated annealing
    def simulated_annealing():
        nonlocal best_h, best_c5
        temp = 1.0
        cooling = 0.995
        max_iter = 1000
        
        for _ in range(max_iter):
            if time.time() - start_time >= budget_s:
                break
            # Generate perturbation
            perturb = np.random.normal(0, 0.05, n_points)
            perturb = perturb / np.std(perturb) * 0.1
            perturbed_h = h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            
            # Evaluate
            current_c5 = objective(perturbed_h)
            
            # Acceptance criterion
            if current_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - current_c5) / temp):
                h = perturbed_h.copy()
                best_c5 = current_c5
                best_h = h.copy()
            
            temp *= cooling
    
    simulated_annealing()
    
    return best_h, best_c5, n_points
