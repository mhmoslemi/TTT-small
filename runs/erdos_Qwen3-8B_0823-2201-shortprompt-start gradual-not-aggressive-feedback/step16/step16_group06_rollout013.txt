        return c5

    class PSOParticle:
        def __init__(self, n_points):
            self.position = np.random.rand(n_points) * 2 - 1
            self.velocity = np.random.rand(n_points) * 0.1
            self.best_position = self.position.copy()
            self.best_value = float('inf')

            r1, r2 = np.random.rand(2, n_points)
