import scipy.optimize as opt
import time
import random

        generations = 10

        mutation_strength = 0.1

            h = h.reshape(n_points)
            h = np.clip(h, 0, 1)

            fitness_values = np.array([fitness(ind) for ind in population])

            parents_indices = np.argsort(fitness_values)[:population_size//2]

        best_c5 = genetic_c5
