            self.objective = objective
            self.constraint_sum = constraint_sum
            self.target_sum = target_sum
            self.dx = dx

            self.best_objectives = np.array([self.evaluate(p) for p in self.particles])

            self.global_best_objective = self.best_objectives[0]
        
        def evaluate(self, position):
            position = position.copy()
            position = np.clip(position, 0, 1)
            position = position * self.target_sum / np.sum(position)
            corr = np.correlate(position, 1 - position, mode="full")
            c5 = np.max(corr) * self.dx
            return c5

            r2 = np.random.rand(self.n_particles, self.n_vars)

                self.global_best_position = self.best_positions[best_index].copy()
                self.global_best_objective = self.best_objectives[best_index]

                self.update_velocities()
                self.update_positions()
                self.update_best_positions()
