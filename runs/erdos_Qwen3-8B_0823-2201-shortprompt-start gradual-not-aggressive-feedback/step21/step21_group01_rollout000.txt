import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Hybrid optimization: Simulated Annealing with SLSQP refinement
    # Start with a more diverse initial guess
    for _ in range(3):
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        perturbed_h = h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Simulated Annealing
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-9}

            best_c5 = current_c5
        
        # Early stopping if time is up
        if time.time() - start_time >= budget_s:
            break
    
    return best_h, best_c5, n_points
