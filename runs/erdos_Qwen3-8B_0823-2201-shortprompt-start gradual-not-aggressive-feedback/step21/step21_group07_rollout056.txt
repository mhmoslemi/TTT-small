import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    # Use a different initial guess with n=64 points
    n_points = 64
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Create a more diversified initial guess
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 == 0:
            initial_h_values[i] = 0.5
        elif i % 4 == 1:
            initial_h_values[i] = 1.0
        elif i % 4 == 2:
            initial_h_values[i] = 0.0
        else:
            initial_h_values[i] = 0.5
    
    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h)
    
    def objective_overlap(h_vec):
        h = h_vec.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def objective_variation(h_vec):
        h = h_vec.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        variation = np.sum(np.abs(np.gradient(h)))
        return variation
    
    def constraint_sum(h_vec):
        h = h_vec.reshape(n_points)
        return np.sum(h) - target_sum

        result = opt.minimize(
            objective_overlap,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-9}

        # Local refinement with variation minimization
        result = opt.minimize(
            objective_variation,
            h,
            method='SLSQP',
            bounds=bounds,
