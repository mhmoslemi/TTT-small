import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    start_time = time.time()
    
    # Hybrid optimization: gradient descent with random restarts
    for stage in range(3):
        # Gradient descent with adaptive learning rate
        h = h.copy()

            if time.time() - start_time >= budget_s:
                break

            h_flat = h.reshape(n_points)

            h_flat = h_flat * target_sum / np.sum(h_flat)

            c5 = np.max(corr) * dx

            # Update parameters
