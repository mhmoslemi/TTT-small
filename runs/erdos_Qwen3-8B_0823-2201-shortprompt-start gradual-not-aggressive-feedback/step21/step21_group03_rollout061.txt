import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Particle swarm optimization with gradient refinement
    particles = np.random.uniform(0, 1, (n_points, n_points))
    particles = particles * target_sum / np.sum(particles, axis=1, keepdims=True)
    velocities = np.random.uniform(-0.1, 0.1, (n_points, n_points))
    
    best_positions = particles.copy()
    best_values = np.array([objective(p) for p in best_positions])
    
    for iter in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Update velocities and positions
        inertia = 0.8 + 0.4 * np.sin(iter * np.pi / 100)
        cognitive = 1.5
        social = 1.5
        
        for i in range(n_points):
            r1, r2 = np.random.rand(2, n_points)
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (best_positions[i] - particles[i]) + social * r2 * (np.mean(best_positions, axis=0) - particles[i])
            particles[i] = np.clip(particles[i] + velocities[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
        
        # Evaluate new positions
        current_values = np.array([objective(p) for p in particles])
        improve = current_values < best_values
        
        best_positions[improve] = particles[improve]
        best_values[improve] = current_values[improve]
        
        # Update best solution
        if np.min(best_values) < best_c5:
            best_h = best_positions[np.argmin(best_values)].copy()
            best_c5 = np.min(best_values)
    
    # Local refinement with SLSQP
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-9}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
