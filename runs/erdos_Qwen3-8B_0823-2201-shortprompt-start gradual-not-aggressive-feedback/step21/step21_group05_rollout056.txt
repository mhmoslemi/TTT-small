import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Reduced from 47 to 100 for better resolution
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial guess based on uniform distribution
    initial_h = np.ones(n_points) * 0.5
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Multi-stage perturbation with decreasing scale
    for stage in range(3):
        if time.time() - start_time > budget_s:
            break
        # Perturb high-overlap regions
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        overlap_indices = np.argsort(corr[-n_points//2:])[-n_points//4:]
        perturb = np.zeros(n_points)
        perturb[overlap_indices] = np.random.uniform(-0.02, 0.02, len(overlap_indices))
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Optimize perturbed solution
        result = opt.minimize(
            objective,
            perturbed_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-9}
        )
        best_h = result.x
        current_c5 = objective(best_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
