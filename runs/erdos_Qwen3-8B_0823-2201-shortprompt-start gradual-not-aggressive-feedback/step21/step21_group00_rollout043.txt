import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = 40
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial guess: use a simple alternating pattern
    initial_h_values = np.array([0.5 if i % 2 == 0 else 0.0 for i in range(n_points)])

        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]

    h = np.array(initial_h_values, dtype=np.float64)

    best_h = h.copy()

    start_time = time.time()
    
    # Optimizer parameters

    perturb_size = 0.05
    
    for stage in range(max_iter):

        result = opt.minimize(

            bounds=bounds,

        )
        h = result.x

        h = np.clip(h, 0, 1)
