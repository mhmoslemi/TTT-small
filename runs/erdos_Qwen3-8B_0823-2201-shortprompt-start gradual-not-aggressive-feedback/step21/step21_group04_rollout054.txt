import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)

    start_time = time.time()
    
    # Try different discretization sizes
    for n in [47, 53, 59, 65, 71, 77, 83, 89, 95, 101]:
        if time.time() - start_time >= budget_s:
            break
        
        # Check if we have enough time for this discretization
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 1:
            continue

        h_new[:n_points] = h
        h_new[n_points:] = 0.5
        h_new = h_new * (n / 2) / np.sum(h_new)
        
        # Initial solution for this n
        current_h = h_new.copy()
        current_c5 = objective(current_h)
        
        # Optimization loop
        for _ in range(3):
            # Gradient-based refinement
            result = opt.minimize(
                objective,
                current_h,
                method='L-BFGS-B',
                bounds=bounds,

            result = opt.minimize(
