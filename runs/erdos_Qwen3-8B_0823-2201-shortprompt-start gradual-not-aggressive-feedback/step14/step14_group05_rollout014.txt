import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Main optimization loop
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(h)
    
    # Gradient-based refinement
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-9}
    )
    h = result.x
    current_c5 = objective(h)
    
    if current_c5 < best_c5:
        best_h = h.copy()
        best_c5 = current_c5
    
    # Simulated annealing with decreasing temperature
    temp = 1.0
    cooling_rate = 0.995
    max_iterations = 500
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break

        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Evaluate perturbed solution
        perturbed_c5 = objective(perturbed_h)
        
        # Acceptance probability

            current_c5 = perturbed_c5
            if current_c5 < best_c5:

                best_c5 = current_c5
