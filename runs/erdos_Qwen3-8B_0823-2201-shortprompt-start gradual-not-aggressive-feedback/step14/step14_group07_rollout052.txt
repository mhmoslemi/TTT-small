import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)

        corr = np.correlate(h, 1 - h, mode="full")

        return c5

        h = h.reshape(n_points)

    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)

    start_time = time.time()
    for _ in range(10):
        # Gradient-based refinement
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        h = result.x
        best_c5 = objective(h)
        
        # Greedy local search
        for _ in range(5):
            if time.time() - start_time >= budget_s:
                break
            # Find the worst position
            h_copy = h.copy()
            h_copy = h_copy * target_sum / np.sum(h_copy)
            corr = np.correlate(h_copy, 1 - h_copy, mode="full")
            worst_idx = np.argmax(corr)

            h_flip[worst_idx] = 1 - h_flip[worst_idx]

            flip_c5 = objective(h_flip)

        if time.time() - start_time >= budget_s:
