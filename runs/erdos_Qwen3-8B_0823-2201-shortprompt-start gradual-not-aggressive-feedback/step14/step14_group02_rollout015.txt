import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Smaller n_points for faster evaluation
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial construction based on known patterns
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if x < 1.0:
            initial_h[i] = 0.5
        elif x < 1.5:
            initial_h[i] = 0.75
        else:
            initial_h[i] = 0.25
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    
    # Multi-stage optimization with tighter tolerances
    start_time = time.time()
    for stage in range(3):
        # Gradient-based refinement with adaptive step size
        def local_refinement(h):
            result = opt.minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-10}
            )
            return result.x
        
        h = local_refinement(h)
        best_c5 = objective(h)
        
        # Adaptive perturbation with decreasing magnitude
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break
        for perturb_size in np.logspace(-4, -1, num=5, base=10):
            if time.time() - start_time >= budget_s:
                break
            perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
            perturbed_h = h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            
            result = opt.minimize(
                objective,
                perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 80, 'ftol': 1e-10, 'gtol': 1e-10}
            )
            
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = objective(current_h)
            
            if current_c5 < best_c5:
                h = current_h
                best_c5 = current_c5
    
    return h, best_c5, n_points
