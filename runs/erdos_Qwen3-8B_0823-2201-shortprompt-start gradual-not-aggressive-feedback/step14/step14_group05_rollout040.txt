import scipy.optimize as opt
import time

        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)

    result = opt.minimize(

        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},

    population_size = 20

        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]

            fitness = []

                corr = np.correlate(ind, 1 - ind, mode="full")
                c5 = np.max(corr) * dx
                fitness.append(c5)
            
            # Select best individuals
            population = [population[i] for i in np.argsort(fitness)[:population_size//2]]

                new_population.append(child)

        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},

    )
