import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Transform to Fourier space for better exploration
    def fourier_to_h(coeffs, n_points):
        x = np.linspace(0, 2, n_points, endpoint=False)
        h = np.zeros(n_points)
        for k in range(len(coeffs)):
            h += coeffs[k] * np.cos(k * np.pi * x)
            h += coeffs[k] * np.sin(k * np.pi * x)
        return h
    
    def h_to_fourier(h, n_points):
        x = np.linspace(0, 2, n_points, endpoint=False)
        coeffs = np.zeros(2 * n_points)
        for k in range(n_points):
            coeffs[k] = np.mean(h * np.cos(k * np.pi * x))
            coeffs[n_points + k] = np.mean(h * np.sin(k * np.pi * x))
        return coeffs
    
    def objective(coeffs):
        h = fourier_to_h(coeffs, n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = fourier_to_h(coeffs, n_points)

            objective,
