import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Main optimization loop
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(h)
    
    # Gradient-based refinement
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-9}
    )
    h = result.x
    current_c5 = objective(h)
    
    if current_c5 < best_c5:
        best_h = h.copy()
        best_c5 = current_c5
    
    # Population-based perturbation
    population_size = 10
    for _ in range(3):
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 1:
            break
        
        # Generate perturbed solutions
        perturbations = np.random.uniform(-0.05, 0.05, (population_size, n_points))
        perturbed_h_list = []
        
        for p in perturbations:
            perturbed_h = h + p
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            perturbed_h_list.append(perturbed_h)
        
        # Evaluate perturbed solutions
        candidates = []
        for ph in perturbed_h_list:
            result = opt.minimize(
                objective,
                ph,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-9}
            )
            candidates.append((result.x, objective(result.x)))
        
        # Select best candidate
        candidates.sort(key=lambda x: x[1])
        best_candidate = candidates[0]
        current_h, current_c5 = best_candidate
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
