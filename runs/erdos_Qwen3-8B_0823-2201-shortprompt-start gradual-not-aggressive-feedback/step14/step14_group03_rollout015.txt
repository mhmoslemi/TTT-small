import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    best_c5 = objective(h)
    best_h = h.copy()
    
    start_time = time.time()
    
    # Main optimization loop
    for stage in range(3):
        # Local refinement with SLSQP
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        h = result.x
        current_c5 = objective(h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Temperature schedule for simulated annealing
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break

        # Random perturbation with cooling
        for _ in range(5):
            if time.time() - start_time >= budget_s:
                break

            perturbed_h = h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            
            # Evaluate perturbed solution
            result = opt.minimize(
                objective,
                perturbed_h,

                bounds=bounds,

                best_h = perturbed_h.copy()
    
    return best_h, best_c5, n_points
