import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize with refined version of initial construction
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Best solution so far
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    # Phase 1: Local optimization with adaptive perturbation
    for _ in range(10):
        # Local refinement with SLSQP
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        best_c5 = objective(best_h)
        
        # Adaptive perturbation with decreasing magnitude
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 1:
            break
        perturb_size = max(0.005, 0.02 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Local refinement of perturbed solution
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        perturbed_h = result.x
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        current_c5 = objective(perturbed_h)
        
        if current_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = current_c5
    
    # Phase 2: Global exploration with genetic algorithm
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    def fitness(individual):
        return objective(individual)
    
    def mutate(individual):
        mutated = individual.copy()
        for i in range(len(mutated)):
            if random.random() < mutation_rate:
                mutated[i] += np.random.uniform(-0.05, 0.05)
                mutated[i] = np.clip(mutated[i], 0, 1)
                mutated[i] = mutated[i] * target_sum / np.sum(mutated)
        return mutated
    
    def crossover(parent1, parent2):
        child = parent1.copy()
        for i in range(len(child)):
            if random.random() < crossover_rate:
                child[i] = parent2[i]
        child = child * target_sum / np.sum(child)
        return child
    
    # Initialize population
    population = [np.random.uniform(0, 1, n_points) * target_sum / np.sum(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    for _ in range(generations):
        # Evaluate fitness
        fitness_values = [fitness(ind) for ind in population]
        # Select parents
        parents = [population[i] for i in np.argsort(fitness_values)[:population_size//2]]
        # Generate new population
        new_population = []
        while len(new_population) < population_size:
            parent1, parent2 = random.sample(parents, 2)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        population = new_population
    
    # Evaluate final population
    final_fitness = [fitness(ind) for ind in population]
    best_genetic = population[np.argmin(final_fitness)]
    best_c5_genetic = np.min(final_fitness)
    
    # Compare and choose best solution
    if best_c5_genetic < best_c5:
        best_h = best_genetic
        best_c5 = best_c5_genetic
    
    return best_h, best_c5, n_points
