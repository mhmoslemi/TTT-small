import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Hybrid optimization with gradient refinement and genetic search
    for stage in range(3):
        # Gradient-based refinement
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-9}
        )
        h = result.x
        current_c5 = objective(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Genetic algorithm with adaptive mutation
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break

        generations = min(10, int(remaining_time / 2))

        population = [h * target_sum / np.sum(h) for h in population]
        
        for gen in range(generations):
            # Evaluate fitness
            fitness = [objective(h) for h in population]

                mutation = np.random.normal(0, 0.1, n_points)
                child += mutation
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
                new_population.append(child)
            
            population = new_population
        
    return best_h, best_c5, n_points
