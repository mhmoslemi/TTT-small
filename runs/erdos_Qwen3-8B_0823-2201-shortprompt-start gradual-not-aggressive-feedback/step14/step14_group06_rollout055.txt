import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize with refined version of initial construction
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    # Genetic algorithm parameters
    pop_size = 20
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Generate population
        population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
        for p in population:
            p = p * target_sum / np.sum(p)
        
        # Evaluate population
        fitness = [objective(p) for p in population]
        best_idx = np.argmin(fitness)
        current_best = population[best_idx].copy()
        current_best = current_best * target_sum / np.sum(current_best)
        current_c5 = objective(current_best)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
        
        # Crossover and mutation
        for i in range(pop_size):
            if np.random.random() < crossover_rate:
                j = np.random.randint(0, pop_size)
                if i != j:
                    p1 = population[i].copy()
                    p2 = population[j].copy()
                    # Single-point crossover
                    split = np.random.randint(1, n_points-1)
                    child = np.concatenate([p1[:split], p2[split:]])
                    child = child * target_sum / np.sum(child)
                    population[i] = child
            
            if np.random.random() < mutation_rate:
                # Gaussian mutation
                mutation = np.random.normal(0, 0.05, n_points)
                population[i] += mutation
                population[i] = np.clip(population[i], 0, 1)
                population[i] = population[i] * target_sum / np.sum(population[i])
        
        # Local refinement of best solution
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,

        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
