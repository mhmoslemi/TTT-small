import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    dx = 2.0 / n_points

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    for _ in range(3):

            objective,
            h,
            method='SLSQP',

            options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-9}
        )
        h = result.x
        current_c5 = objective(h)

            best_h = h.copy()
            best_c5 = current_c5
