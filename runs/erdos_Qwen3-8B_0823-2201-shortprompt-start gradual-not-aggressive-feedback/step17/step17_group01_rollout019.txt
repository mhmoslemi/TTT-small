import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Gradient-based refinement with tighter tolerances
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-9}
    )
    h = result.x
    current_c5 = objective(h)
    
    if current_c5 < best_c5:
        best_h = h.copy()
        best_c5 = current_c5
    
    # Population-based search with random sampling and mutation
    population_size = min(10, n_points // 2)
    for _ in range(5):
        if time.time() - start_time >= budget_s:
            break
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        for p in population:
            p = p * target_sum / np.sum(p)
            result = opt.minimize(
                objective,
                p,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-9}
            )
            p = result.x
            p = np.clip(p, 0, 1)
            p = p * target_sum / np.sum(p)
            current_c5 = objective(p)
            if current_c5 < best_c5:
                best_h = p.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
