import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Multi-stage optimization with adaptive refinement and grid-based local search
    for stage in range(3):
        # Gradient-based refinement with tighter tolerances
        def local_refinement(h):
            result = opt.minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-9}
            )
            return result.x
        
        h = local_refinement(h)
        current_c5 = objective(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Adaptive grid-based refinement
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break
        
        # Focus on high-overlap regions
        h_flat = h.reshape(n_points)
        corr = np.correlate(h_flat, 1 - h_flat, mode="full")
        max_indices = np.argsort(corr[-n_points//2:])[-5:]
        
        # Refine each high-overlap region with grid search
        for idx in max_indices:
            if time.time() - start_time >= budget_s:
                break
            # Create a grid around the current value
            grid = np.linspace(0.0, 1.0, 100)
            best_perturb = h[idx]
            best_val = objective(h.copy())
            
            for p in grid:
                perturbed_h = h.copy()
                perturbed_h[idx] = p
                perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
                current_c5 = objective(perturbed_h)
                
                if current_c5 < best_val:
                    best_val = current_c5
                    best_perturb = p
            
            h[idx] = best_perturb
    
    return best_h, best_c5, n_points
