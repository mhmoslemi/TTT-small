import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Multi-stage optimization with structured perturbation
    for stage in range(5):
        # Gradient-based refinement with tighter tolerances
        def local_refinement(h):
            result = opt.minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-9}
            )
            return result.x
        
        h = local_refinement(h)
        current_c5 = objective(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Adaptive structured perturbation
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break
        
        # Define perturbation grid
        perturb_steps = 5
        perturb_range = 0.1
        perturb_values = np.linspace(-perturb_range, perturb_range, perturb_steps)
        
        for perturb in perturb_values:
            if time.time() - start_time >= budget_s:
                break
            perturb_h = h + perturb
            perturb_h = np.clip(perturb_h, 0, 1)
            perturb_h = perturb_h * target_sum / np.sum(perturb_h)
            
            result = opt.minimize(
                objective,
                perturb_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-9}
            )
            
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = objective(current_h)
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
