    iterations = 0
    max_iter = 150
    population_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    while time.time() - start_time < budget_s and iterations < max_iter:
        # Create population
        population = [h.copy() for _ in range(population_size)]
        for i in range(population_size):
            population[i] = np.random.uniform(0, 1, n_points)
            population[i] = population[i] * target_sum / np.sum(population[i])
        
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        
        # Select best individuals
        best_indices = np.argsort(fitness)[:int(population_size/2)]

        for i in range(len(selected)):
