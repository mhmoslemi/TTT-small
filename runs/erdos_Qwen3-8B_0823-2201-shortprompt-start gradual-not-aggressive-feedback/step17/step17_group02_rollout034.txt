import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    # Initial solution
    h = np.array(initial_h, dtype=np.float64)

    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Simulated annealing with adaptive cooling
    temp = 1.0
    cooling_rate = 0.995
    for _ in range(500):
        if time.time() - start_time >= budget_s:
            break
        
        # Generate random perturbation
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = h + perturb

        # Evaluate objective
        current_c5 = objective(perturbed_h)
        
        # Accept with probability based on temperature
        if current_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - current_c5) / temp):
