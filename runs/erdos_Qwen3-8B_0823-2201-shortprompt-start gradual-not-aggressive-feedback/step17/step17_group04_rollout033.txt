import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Adaptive perturbation and refinement
    for _ in range(5):
        if time.time() - start_time >= budget_s:
            break
        # Greedy refinement
        for i in range(n_points):
            if time.time() - start_time >= budget_s:
                break

            h_perturb = h.copy()
            h_perturb[i] += 0.05
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * target_sum / np.sum(h_perturb)

            if current_c5 < best_c5:

        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,

            method='SLSQP',
