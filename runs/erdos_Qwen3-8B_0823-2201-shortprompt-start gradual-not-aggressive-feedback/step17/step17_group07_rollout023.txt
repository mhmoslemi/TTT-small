import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Genetic algorithm for global exploration
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    elite_size = 2
    
    for gen in range(generations):
        if time.time() - start_time >= budget_s:
            break
        
        # Evaluate population
        population = [np.random.rand(n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        fitness = [objective(h) for h in population]
        
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elites.copy()
        for _ in range(population_size - elite_size):
            # Select parents
            parent1 = np.random.choice(elite_indices)
            parent2 = np.random.choice(elite_indices)
            parent1 = population[parent1]
            parent2 = population[parent2]
            
            # Crossover
            child = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.normal(0, 0.05)
                    child[i] = np.clip(child[i], 0, 1)
            
            # Ensure constraint
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        population = new_population
        fitness = [objective(h) for h in population]
        best_index = np.argmin(fitness)
        best_h = population[best_index]
        best_c5 = fitness[best_index]
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
