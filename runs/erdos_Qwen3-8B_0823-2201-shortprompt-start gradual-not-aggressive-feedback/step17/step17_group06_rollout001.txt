import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)

    # Main optimization loop
    start_time = time.time()

    best_c5 = objective(h)
    
    # Optimization parameters
    iterations = 0
    max_iter = 200
    perturbation_size = 0.15
    
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        
        # Generate multiple perturbations
        perturbations = []
        for _ in range(5):
            perturb = np.random.uniform(-perturbation_size, perturbation_size, n_points)

        # Try each perturbation

            perturbed_h = np.clip(perturbed_h, 0, 1)

            result = opt.minimize(
