import random

def run(seed=42, budget_s=20, **kwargs):

        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        c5 = np.max(corr) * dx

    start_time = time.time()
    best_h = h.copy()

    population_size = 50
    generations = 100
    mutation_rate = 0.1
    mutation_strength = 0.1
    
    for gen in range(generations):

        population = [h.copy() for _ in range(population_size)]

            p = p * target_sum / np.sum(p)
        
        # Evaluate population
        fitness = [objective(p) for p in population]
        
        # Select best
        best_idx = np.argmin(fitness)

        population = population + new_population
        population = np.array([p for p in population])
