import scipy.optimize as opt
import time

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)

    # Function to evaluate and return the best solution
    def evaluate_solution(h):
        h = h.reshape(n_points)

        return np.max(corr) * dx, h
    
    # Multi-start optimization with adaptive perturbations
    best_h = h.copy()
    best_c5 = evaluate_solution(best_h)[0]
    start_time = time.time()

        # Gradient-based optimization
        result = opt.minimize(
            lambda x: objective(x),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},

        )
