import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h = local_refinement(h)
    best_c5 = objective(best_h)
    
    # Evolutionary strategy with population-based search
    population_size = 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    def evolve(population):
        offspring = []
        for i in range(population_size):
            parent1 = np.random.choice(population)
            parent2 = np.random.choice(population)
            child = np.copy(parent1)
            for j in range(n_points):
                if np.random.rand() < crossover_rate:
                    child[j] = parent2[j]
            # Apply mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[j] += np.random.uniform(-0.1, 0.1)
                    child[j] = np.clip(child[j], 0, 1)
            # Enforce constraint
            child = child * target_sum / np.sum(child)
            offspring.append(child)
        return offspring
    
    # Run evolutionary search
    population = [local_refinement(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        population += evolve(population)
        population = np.array([h * target_sum / np.sum(h) for h in population])
        population = [local_refinement(h) for h in population]
        population = np.array([h * target_sum / np.sum(h) for h in population])
        population = np.array([h for h in population if np.sum(h) > 0.9 * target_sum])
        if len(population) > population_size:
            population = np.array([population[i] for i in np.argsort([objective(h) for h in population])[:population_size]])
    
    best_h = population[0]
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
