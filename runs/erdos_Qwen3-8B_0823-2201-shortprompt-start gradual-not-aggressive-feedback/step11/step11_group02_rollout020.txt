import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]

    h = np.array(initial_h_values, dtype=np.float64)

        result = opt.minimize(

            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h = local_refinement(h)
    best_c5 = objective(best_h)
    
    # Adaptive perturbation with decreasing magnitude
    while time.time() - start_time < budget_s:

        perturb_size = max(0.001, 0.05 * (remaining_time / budget_s))
        
        # Generate perturbed solutions
        perturbed_h = best_h + np.random.uniform(-perturb_size, perturb_size, n_points)

            best_c5 = current_c5
