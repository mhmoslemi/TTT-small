import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Main optimization
    for stage in range(3):
        # Coordinate descent with adaptive step size
        for i in range(n_points):
            def local_update(h):
                h = h.copy()
                h[i] = np.clip(h[i], 0, 1)
                h = h * target_sum / np.sum(h)
                corr = np.correlate(h, 1 - h, mode="full")
                return np.max(corr) * dx
            
            # Binary search for optimal value at position i
            low, high = 0, 1
            for _ in range(20):
                mid = (low + high) / 2
                h_test = h.copy()
                h_test[i] = mid
                h_test = h_test * target_sum / np.sum(h_test)
                c5_test = np.max(np.correlate(h_test, 1 - h_test, mode="full")) * dx
                if c5_test < objective(h):
                    high = mid
                else:
                    low = mid
            h[i] = high
        
        # Randomized perturbation with decreasing magnitude
        remaining_time = budget_s - (time.time() - time.time())
        for perturb_size in np.logspace(-3, -1, num=5, base=10):
            if time.time() - time.time() >= budget_s:
                break
            perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
            perturbed_h = h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            
            result = opt.minimize(
                objective,
                perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = objective(current_h)
            
            if current_c5 < objective(h):
                h = current_h
    
    return h, objective(h), n_points
