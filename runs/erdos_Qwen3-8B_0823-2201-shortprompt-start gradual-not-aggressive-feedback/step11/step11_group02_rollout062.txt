        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    h = np.array(initial_h, dtype=np.float64)

        new_population = []
        for h in population:
            if np.random.rand() < crossover_rate:
                j = np.random.choice(population)
                crossover_point = np.random.randint(n_points)
                child = np.concatenate([h[:crossover_point], j[crossover_point:]])
            else:
                child = h.copy()
            # Mutation
            if np.random.rand() < mutation_rate:
                perturb = np.random.normal(0, 0.05, n_points)
                child = np.clip(child + perturb, 0, 1)
            # Normalize
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        return new_population
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )

    start_time = time.time()
