import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Improved initialization with pattern-based refinement
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def local_refinement(h):
        # First use SLSQP for local optimization
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        h = result.x
        
        # Then use Nelder-Mead for further refinement
        result = opt.minimize(
            objective,
            h,
            method='Nelder-Mead',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'xatol': 1e-6, 'maxiter': 30}
        )
        return result.x
    
    def adaptive_perturbation(h, perturb_strength):
        perturb = np.random.uniform(-perturb_strength, perturb_strength, n_points)
        perturbed = np.clip(h + perturb, 0, 1)
        return perturbed * target_sum / np.sum(perturbed)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    # Initial refinement
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Adaptive perturbation and refinement
    while time.time() - start_time < budget_s:

            best_c5 = current_c5
