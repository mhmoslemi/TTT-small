import numpy as np
import scipy.optimize as opt
import time
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Genetic algorithm setup
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", np.ndarray, fitness=creator.FitnessMin)
    
    toolbox = base.Toolbox()
    toolbox.register("uniform", np.random.uniform, 0, 1)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.uniform, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    
    def eval_func(individual):
        individual = np.clip(individual, 0, 1)
        individual = individual * target_sum / np.sum(individual)
        corr = np.correlate(individual, 1 - individual, mode="full")
        c5 = np.max(corr) * dx
        return (c5,)
    
    toolbox.register("evaluate", eval_func)
    toolbox.register("mate", tools.cxTwoPoint)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    # Genetic algorithm refinement
    def genetic_refinement(population, n_gen=50):
        pop = population
        for gen in range(n_gen):
            offspring = algorithms.varAnd(pop, toolbox, cxpb=0.5, mutpb=0.1)
            fits = toolbox.map(toolbox.evaluate, offspring)
            offspring = list(map(lambda ind, fit: ind.fitness.values, offspring, fits))
            pop = tools.selBest(offspring, len(pop))
            if gen % 10 == 0:
                best = min(pop, key=lambda ind: ind.fitness.values)
                print(f"Generation {gen}: Best C₅ = {best.fitness.values[0]}")
        return pop
    
    start_time = time.time()
    best_h = local_refinement(h)
    best_c5 = objective(best_h)
    
    # Run genetic algorithm within budget
    population = [best_h]
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        n_gen = int(remaining_time / 1.0)
        population = genetic_refinement(population, n_gen=n_gen)
        best = min(population, key=lambda ind: ind.fitness.values)
        best_h = np.array(best)
        best_c5 = best.fitness.values[0]
    
    return best_h, best_c5, n_points
