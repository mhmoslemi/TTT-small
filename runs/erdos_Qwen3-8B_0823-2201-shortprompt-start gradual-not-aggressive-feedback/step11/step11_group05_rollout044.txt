import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize population
    population_size = 50
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    def evolve_population(population, generations=50):
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            # Select top 20% for breeding
            elite_idx = np.argsort(fitness)[:int(population_size * 0.2)]
            elites = population[elite_idx]
            
            # Crossover
            new_population = []
            for _ in range(int(population_size * 0.5)):
                parent1, parent2 = np.random.choice(elites, 2, replace=False)
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                new_population.append(child)
            
            # Mutation
            for i in range(len(new_population)):
                mutation_rate = 0.1
                mutation_size = 0.1 * (target_sum / np.sum(new_population[i]))
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        new_population[i][j] += np.random.uniform(-mutation_size, mutation_size)
                        new_population[i][j] = np.clip(new_population[i][j], 0, 1)
                # Normalize
                new_population[i] = new_population[i] * target_sum / np.sum(new_population[i])
            
            # Replace population
            population = np.vstack([elites, new_population])
            population = population[np.random.permutation(population_size)]
        
        return population
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Evolve population
    population = np.array(initial_h).reshape(1, n_points)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    population = np.vstack([population] * population_size)
    
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        population = evolve_population(population)
        current_best = population[np.argsort([objective(ind) for ind in population])[0]]
        current_c5 = objective(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
