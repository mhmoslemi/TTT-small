    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-9}
        )
        return result.x
    
    def adaptive_perturbation(h, temperature):
        perturb_size = 0.05 * temperature
        perturb = np.random.normal(0, perturb_size, n_points)
        perturbed = np.clip(h + perturb, 0, 1)

        return perturbed
    
    start_time = time.time()

    best_h = local_refinement(best_h)
