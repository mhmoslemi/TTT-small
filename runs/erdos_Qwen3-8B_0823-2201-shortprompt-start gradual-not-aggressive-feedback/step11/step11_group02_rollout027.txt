import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    def adaptive_mutation(h, mutation_rate=0.15, scale=0.05):
        perturb = np.random.normal(0, scale, n_points)
        mutated = np.clip(h + perturb, 0, 1)
        mutated = mutated * target_sum / np.sum(mutated)
        return mutated
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    # Initial refinement
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Adaptive search
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        mutation_rate = max(0.05, 0.1 * (remaining_time / budget_s))
        scale = max(0.01, 0.05 * (remaining_time / budget_s))
        
        # Generate mutated versions
        mutated_h = adaptive_mutation(best_h, mutation_rate, scale)
        mutated_c5 = objective(mutated_h)
        
        # Compare and update
        if mutated_c5 < best_c5:
            best_h = mutated_h
            best_c5 = mutated_c5
        
        # Local refinement
        refined_h = local_refinement(best_h)
        refined_c5 = objective(refined_h)
        
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
    
    return best_h, best_c5, n_points
