import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Genetic algorithm parameters
    pop_size = 20
    generations = 50
    mutation_rate = 0.1
    elite_size = 2
    
    def genetic_algorithm(population):
        population = [np.clip(ind, 0, 1) * target_sum / np.sum(np.clip(ind, 0, 1)) for ind in population]
        fitness = [objective(ind) for ind in population]
        indices = np.argsort(fitness)
        elite = [population[i] for i in indices[:elite_size]]
        
        for _ in range(generations):
            selected = np.random.choice(len(population), size=pop_size, p=np.array(fitness)/np.sum(fitness))
            offspring = []
            for i in range(pop_size):
                parent1 = population[selected[i]]
                parent2 = population[selected[np.random.randint(0, len(selected))]]
                child = np.zeros_like(parent1)
                for j in range(n_points):
                    if np.random.rand() < 0.5:
                        child[j] = parent1[j]
                    else:
                        child[j] = parent2[j]
                if np.random.rand() < mutation_rate:
                    child += np.random.normal(0, 0.05, n_points)
                child = np.clip(child, 0, 1) * target_sum / np.sum(np.clip(child, 0, 1))
                offspring.append(child)
            population = elite + offspring
            fitness = [objective(ind) for ind in population]
            indices = np.argsort(fitness)
            population = [population[i] for i in indices[:pop_size]]
        return population[0]
    
    # Initial population for genetic algorithm
    initial_population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
    best_h = genetic_algorithm(initial_population)
    
    # Local refinement
    best_h = local_refinement(best_h)
    
    # Adaptive refinement
    start_time = time.time()
    best_c5 = objective(best_h)
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1) * target_sum / np.sum(np.clip(perturbed_h, 0, 1))
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        current_h = result.x
        current_h = np.clip(current_h, 0, 1) * target_sum / np.sum(np.clip(current_h, 0, 1))
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
