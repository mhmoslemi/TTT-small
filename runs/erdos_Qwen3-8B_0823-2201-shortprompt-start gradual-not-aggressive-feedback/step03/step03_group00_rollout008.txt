import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def evolve_population(population, mutation_rate=0.1):
        new_population = []
        for h in population:
            h = h.reshape(n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            perturb = np.random.uniform(-mutation_rate, mutation_rate, n_points)
            mutated = h + perturb
            mutated = np.clip(mutated, 0, 1)
            mutated = mutated * target_sum / np.sum(mutated)
            new_population.append(mutated)
        return new_population

    population = [np.copy(initial_h) for _ in range(20)]
    best_h = np.copy(initial_h)
    best_c5 = objective(best_h)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx]
        
        # Evolve population
        population = evolve_population(population)
        
        # Local optimization on best candidate
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
            print(f"Improved bound: {best_c5}")
    
    return best_h, best_c5, n_points
