import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

        return objective(x)

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Particle Swarm Optimization
    class PSO:
        def __init__(self, n_particles, n_dimensions, bounds, objective, max_iter=100):
            self.n_particles = n_particles
            self.n_dimensions = n_dimensions
            self.bounds = bounds
            self.objective = objective
            self.max_iter = max_iter
            self.particles = np.random.uniform(self.bounds[0], self.bounds[1], (n_particles, n_dimensions))
            self.velocities = np.random.uniform(-1, 1, (n_particles, n_dimensions))
            self.personal_best = self.particles.copy()
            self.personal_best_values = np.array([self.objective(p) for p in self.particles])
            self.global_best = self.personal_best[np.argmin(self.personal_best_values)]
            self.global_best_value = np.min(self.personal_best_values)

        def update(self):
            for i in range(self.n_particles):
                r1, r2 = np.random.rand(2)
                cognitive = 1.5 * r1 * (self.personal_best[i] - self.particles[i])
                social = 1.5 * r2 * (self.global_best - self.particles[i])
                self.velocities[i] = self.velocities[i] * 0.8 + cognitive + social
                self.particles[i] = self.particles[i] + self.velocities[i]
                self.particles[i] = np.clip(self.particles[i], self.bounds[0], self.bounds[1])
                current_value = self.objective(self.particles[i])
                if current_value < self.personal_best_values[i]:
                    self.personal_best[i] = self.particles[i]
                    self.personal_best_values[i] = current_value
            self.global_best = self.personal_best[np.argmin(self.personal_best_values)]
            self.global_best_value = np.min(self.personal_best_values)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # PSO search
        pso = PSO(30, n_points, bounds, pso_objective, max_iter=20)
        for _ in range(pso.max_iter):
            pso.update()
        current_h = pso.global_best
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    return best_h, best_c5, n_points
