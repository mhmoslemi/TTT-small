import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def evolve_population(population, mutation_rate=0.1):
        new_population = []
        for h in population:
            # Mutation
            mutated = h.copy()
            for i in range(n_points):
                if random.random() < mutation_rate:
                    mutated[i] += np.random.uniform(-0.1, 0.1)
                    mutated[i] = np.clip(mutated[i], 0, 1)
            # Crossover with another random individual
            other = random.choice(population)
            crossover_point = random.randint(0, n_points)
            child = np.concatenate((mutated[:crossover_point], other[crossover_point:]))
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        return new_population

    # Initial population
    population_size = 20
    population = [np.random.uniform(0, 1, n_points) * target_sum / np.sum(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]

    best_h = initial_h.copy()
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate population
        evaluations = []
        for h in population:
            evaluations.append((objective(h), h))
        evaluations.sort()
        best_eval, best_h_candidate = evaluations[0]
        
        # Update best solution
        if best_eval < best_c5:
            best_c5 = best_eval
            best_h = best_h_candidate
        
        # Evolve population
        population = evolve_population(population)
        
        # Local optimization on best candidate
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
    
    return best_h, best_c5, n_points
