from scipy.optimize import minimize
import time
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 100
    dx = 2.0 / n_points
    h = initial_h_values.copy()[:n_points]
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def objective(x):
        h = np.clip(x, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def constraint(x):
        h = np.clip(x, 0, 1)
        return np.sum(h) - n_points / 2
    
    # Initial optimization with SLSQP
    result = minimize(objective, h, bounds=[(0, 1) for _ in range(n_points)], method='SLSQP', constraints={'type': 'eq', 'fun': constraint})
    best_h = result.x
    best_c5 = evaluate(best_h)
    
    # Genetic algorithm setup
    if time.time() - start_time < budget_s - 3:
        creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
        creator.create("Individual", list, fitness=creator.FitnessMin)
        
       toolbox = base.Toolbox()
        toolbox.register("attr_float", np.random.uniform, 0, 1)
        toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
        toolbox.register("population", tools.initRepeat, list, toolbox.individual)
        
        toolbox.register("evaluate", lambda ind: evaluate(ind))
        toolbox.register("mate", tools.cxBlend, alpha=0.5)
        toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
        toolbox.register("select", tools.selTournament, tournsize=3)
        
        population = toolbox.population(n=50)
        algorithms.eaSimple(population, toolbox, cxpb=0.5, mutpb=0.1, ngen=10,

        # Evaluate best individual
        best_ind = min(population, key=lambda ind: ind.fitness.values[0])
        best_h = np.array(best_ind)
        best_c5 = evaluate(best_h)
    
    return (best_h, best_c5, n_points)
