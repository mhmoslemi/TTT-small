import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 150  # Reduced for faster optimization
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial guess: alternating pattern
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            initial_h[i] = 1.0
        elif i % 3 == 1:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 0.0
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 20
    mutation_rate = 0.1
    elite_size = 5
    
    # Initial population
    population = np.random.rand(population_size, n_points)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    population = np.clip(population, 0, 1)
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for gen in range(generations):
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = population[elite_indices]
        
        # Select parents
        parent_indices = np.argsort(fitness)[:int(population_size * 0.6)]
        parents = population[parent_indices]
        
        # Generate offspring via crossover
        offspring = []
        for _ in range(population_size - elite_size):
            parent1 = parents[np.random.randint(len(parents))]
            parent2 = parents[np.random.randint(len(parents))]
            child = (parent1 + parent2) / 2
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            offspring.append(child)
        
        # Mutate offspring
        for i in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                mutation = np.random.uniform(-0.1, 0.1, size=n_points)
                offspring[i] += mutation
                offspring[i] = np.clip(offspring[i], 0, 1)
                offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
        
        # Combine elites and offspring
        population = np.vstack((elites, offspring))
        
        # Update best solution
        current_best = population[np.argmin(fitness)]
        current_c5 = fitness[np.argmin(fitness)]
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
        
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # Local refinement with SLSQP
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
