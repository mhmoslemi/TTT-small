import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial solution
    best_h = initial_h.copy()
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    start_time = time.time()

    # Genetic algorithm parameters
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    crossover_rate = 0.8

    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Generate new population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        for p in population:
            p = p * target_sum / np.sum(p)
        
        # Evaluate population
        population_fitness = [objective(p) for p in population]
        best_idx = np.argmin(population_fitness)
        best_candidate = population[best_idx]
        
        # Update best solution
        if population_fitness[best_idx] < best_c5:
            best_h = best_candidate.copy()
            best_c5 = population_fitness[best_idx]
        
        # Crossover and mutation
        new_population = []
        for i in range(population_size):
            if random.random() < crossover_rate:
                j = (i + random.randint(1, population_size - 1)) % population_size
                child = (population[i] + population[j]) / 2
            else:
                child = population[i].copy()
            
            # Mutation
            for k in range(n_points):
                if random.random() < mutation_rate:
                    child[k] += np.random.uniform(-0.1, 0.1)
                    child[k] = np.clip(child[k], 0, 1)
            
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        population = new_population

    # Gradient-based refinement
    result = opt.minimize(
        objective,
        best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
