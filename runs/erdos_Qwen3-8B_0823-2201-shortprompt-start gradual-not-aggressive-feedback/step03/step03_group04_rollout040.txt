import numpy as np
import random
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial guess: use provided initial_h_values
    global initial_h_values
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    def crossover(h1, h2):
        h = np.zeros_like(h1)
        for i in range(len(h)):
            if random.random() < 0.5:
                h[i] = h1[i]
            else:
                h[i] = h2[i]
        h = h * target_sum / np.sum(h)
        return h
    
    best_h = h.copy()
    best_c5 = evaluate(best_h)
    elite_pop = [best_h]
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        population = [mutate(h) for h in elite_pop]
        population += [mutate(h) for h in random.sample(elite_pop, min(5, len(elite_pop)))]
        
        for h in population:
            c5 = evaluate(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                elite_pop = [h]
            elif c5 < best_c5 * 1.1:
                elite_pop.append(h)
        
        if len(elite_pop) > 20:
            elite_pop = random.sample(elite_pop, 20)
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return best_h, best_c5, n_points
