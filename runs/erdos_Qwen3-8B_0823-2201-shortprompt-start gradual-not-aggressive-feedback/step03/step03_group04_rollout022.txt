import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial guess: alternating pattern
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            initial_h[i] = 1.0
        elif i % 3 == 1:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 0.0
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Genetic algorithm with gradient descent refinement
    population_size = 50
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    for _ in range(10):
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        
        # Select parents
        parent_indices = np.argsort(fitness)[:population_size // 2]
        parents = population[parent_indices]
        
        # Crossover
        offspring = []
        for _ in range(population_size // 2):
            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            mutation = np.random.normal(0, 0.05, n_points)
            offspring[i] += mutation
            offspring[i] = np.clip(offspring[i], 0, 1)
            offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
        
        # Combine and refine
        population = np.vstack((parents, offspring))
        population = np.unique(population, axis=0)
        population = population[:population_size]
        
        # Local refinement with gradient descent
        for ind in population:
            result = opt.minimize(
                objective,
                ind,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            if result.success:
                population = np.vstack((population, result.x.reshape(1, -1)))
                population = np.unique(population, axis=0)
                population = population[:population_size]
    
    # Select best solution
    best_h = population[np.argmin([objective(ind) for ind in population])]
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
