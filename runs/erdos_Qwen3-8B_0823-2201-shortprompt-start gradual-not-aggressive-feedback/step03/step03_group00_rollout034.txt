import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm for global search
    start_time = time.time()
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    while time.time() - start_time < budget_s:
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        for i in range(population_size):
            population[i] = population[i] * target_sum / np.sum(population[i])
        
        for _ in range(generations):
            fitness = [objective(ind) for ind in population]
            best_idx = np.argmin(fitness)
            best_individual = population[best_idx]
            
            if fitness[best_idx] < best_c5:
                best_h = best_individual
                best_c5 = fitness[best_idx]
                print(f"Improved bound: {best_c5}")
            
            # Select parents
            parents = np.random.choice(population_size, size=population_size, replace=True)
            offspring = []
            for i in range(population_size):
                parent1 = population[parents[i]]
                parent2 = population[np.random.choice(parents)]
                child = np.zeros(n_points)
                for j in range(n_points):
                    if np.random.rand() < 0.5:
                        child[j] = parent1[j]
                    else:
                        child[j] = parent2[j]
                # Mutate
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[j] += np.random.uniform(-0.1, 0.1)
                        child[j] = np.clip(child[j], 0, 1)
                child = child * target_sum / np.sum(child)
                offspring.append(child)
            
            population = offspring
        
        # Local refinement with SLSQP
        for i in range(population_size):
            result = opt.minimize(
                objective,
                population[i],
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            population[i] = result.x
            population[i] = np.clip(population[i], 0, 1)
            population[i] = population[i] * target_sum / np.sum(population[i])
    
    return best_h, best_c5, n_points
