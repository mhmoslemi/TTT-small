import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize best solution
    best_h = initial_h.copy()
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    start_time = time.time()

    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.15
    crossover_rate = 0.8

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Create population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population.append(best_h)  # Add best solution to population
        
        # Evaluate population
        population_scores = [objective(h) for h in population]
        
        # Select top performers
        top_indices = np.argsort(population_scores)[:population_size]
        top_population = [population[i] for i in top_indices]
        
        # Genetic operations
        new_population = []
        for i in range(population_size):
            if np.random.rand() < crossover_rate:
                # Crossover
                j = np.random.randint(population_size)
                child = (top_population[i] + top_population[j]) / 2
            else:
                # Mutation
                child = top_population[i] + np.random.uniform(-0.1, 0.1, n_points)
            
            # Clip and normalize
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        population = new_population
        
        # Evaluate new population
        new_scores = [objective(h) for h in population]
        best_index = np.argmin(new_scores)
        current_h = population[best_index]
        current_c5 = new_scores[best_index]
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
            print(f"Improved bound: {best_c5}")
    
    # Final refinement with SLSQP
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
