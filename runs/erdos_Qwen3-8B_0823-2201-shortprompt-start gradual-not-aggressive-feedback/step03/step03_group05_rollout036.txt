        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

        total = np.sum(h)

            return np.inf
        return np.sum(h) - n / 2

        bounds = Bounds(0, 1)
        cons = LinearConstraint(np.ones(n), lb=n/2, ub=n/2)

                                 method='SLSQP', bounds=bounds, constraints=cons,
                                 tol=1e-6, options={'maxiter': 100})
