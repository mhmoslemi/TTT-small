import scipy.optimize as opt
import time

    n_points = 200

    target_sum = n_points / 2
    
    # Start from the known initial construction
    initial_h = np.array(initial_h_values)

        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):

        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = initial_h
    best_c5 = objective(best_h)
    
    start_time = time.time()
    for _ in range(3):
        # Random restart
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = initial_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Optimizer with L-BFGS-B
        result = opt.minimize(
            objective,
            perturbed_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )

        result = opt.minimize(
