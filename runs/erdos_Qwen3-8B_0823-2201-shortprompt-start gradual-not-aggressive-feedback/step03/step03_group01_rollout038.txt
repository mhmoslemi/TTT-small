import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def genetic_search():
        population = np.random.rand(20, n_points)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        population = np.clip(population, 0, 1)
        best_h = initial_h.copy()
        best_c5 = objective(best_h)
        
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            # Evaluate population
            fitness = np.array([objective(ind) for ind in population])
            # Select top 10
            selected = population[np.argsort(fitness)[:10]]
            # Mutate
            mutated = selected + np.random.normal(0, 0.05, selected.shape)
            mutated = np.clip(mutated, 0, 1)
            mutated = mutated * target_sum / np.sum(mutated, axis=1, keepdims=True)
            # Crossover
            offspring = []
            for i in range(10):
                for j in range(i+1, 10):
                    child = np.concatenate([selected[i][:n_points//2], selected[j][n_points//2:]])
                    child = np.clip(child, 0, 1)
                    child = child * target_sum / np.sum(child)
                    offspring.append(child)
            population = np.vstack([selected, mutated, np.array(offspring)])
            population = np.random.permutation(population)
            # Update best
            for ind in population:
                c5 = objective(ind)
                if c5 < best_c5:
                    best_h = ind.copy()
                    best_c5 = c5
        return best_h, best_c5

    start_time = time.time()
    best_h, best_c5 = genetic_search()
    
    # Local refinement with L-BFGS-B
    result = opt.minimize(
        objective,
        best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
