import scipy.optimize as opt

import random

        initial_h = initial_h * target_sum / np.sum(initial_h)

            h = h.reshape(n_points)

            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5

            h = h.reshape(n_points)
            return np.sum(h) - target_sum

        result = opt.minimize(
            objective,

            method='SLSQP',

            h = h.reshape(n_points)

            c5 = np.max(corr) * dx
            return c5
