import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial guess: alternating pattern
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            initial_h[i] = 1.0
        elif i % 3 == 1:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 0.0
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial optimization with SLSQP
    result = opt.minimize(
        objective,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    # Particle swarm optimization for global search
    class PSO:
        def __init__(self, n_particles=30, n_iterations=50, w=0.5, c1=1.5, c2=1.5):
            self.n_particles = n_particles
            self.n_iterations = n_iterations
            self.w = w
            self.c1 = c1
            self.c2 = c2
        
        def optimize(self, objective, bounds):
            dimensions = len(bounds)
            particles = np.random.uniform(bounds[:, 0], bounds[:, 1], (self.n_particles, dimensions))
            velocities = np.zeros((self.n_particles, dimensions))
            personal_best_positions = particles.copy()
            personal_best_values = np.array([objective(p) for p in particles])
            global_best_position = personal_best_positions[np.argmin(personal_best_values)]
            global_best_value = np.min(personal_best_values)
            
            for _ in range(self.n_iterations):
                r1 = np.random.rand(dimensions)
                r2 = np.random.rand(dimensions)
                velocities = (self.w * velocities +
                              self.c1 * r1 * (personal_best_positions - particles) +
                              self.c2 * r2 * (global_best_position - particles))
                particles = np.clip(particles + velocities, bounds[:, 0], bounds[:, 1])
                current_values = np.array([objective(p) for p in particles])
                
                for i in range(self.n_particles):
                    if current_values[i] < personal_best_values[i]:
                        personal_best_positions[i] = particles[i].copy()
                        personal_best_values[i] = current_values[i]
                
                global_best_position = personal_best_positions[np.argmin(personal_best_values)]
                global_best_value = np.min(personal_best_values)
            
            return global_best_position, global_best_value
    
    pso = PSO(n_particles=30, n_iterations=50, w=0.5, c1=1.5, c2=1.5)
    best_pso, _ = pso.optimize(objective, bounds)
    
    # Refine with SLSQP
    result = opt.minimize(
        objective,
        best_pso,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
