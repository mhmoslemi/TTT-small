import numpy as np
import scipy.optimize as opt
import time
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize genetic algorithm
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)

    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0, 1)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)

    toolbox.register("mate", tools.cxTwoPoint)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)
    toolbox.register("evaluate", objective)

    # Initial population
    pop = toolbox.population(n=50)
    fitnesses = list(map(toolbox.evaluate, pop))
    for ind, fit in zip(pop, fitnesses):
        ind.fitness.values = fit

    # Local optimizer
    def local_optimizer(ind):
        h = np.array(ind)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x

    # Genetic algorithm parameters
    ngens = 5
    cxpb = 0.5
    mutpb = 0.1
    elitism = 0.2

    start_time = time.time()
    for gen in range(ngens):
        if time.time() - start_time > budget_s:
            break
        offspring = algorithms.varAnd(pop, toolbox, cxpb, mutpb)
        fits = list(map(toolbox.evaluate, offspring))
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = fit
        # Select the next generation
        pop = tools.selTournament(pop + offspring, len(pop))
        # Apply local optimization to top candidates
        top_candidates = sorted(pop, key=lambda x: x.fitness.values)[0:10]
        for ind in top_candidates:
            ind[:] = local_optimizer(ind)
            ind.fitness.values = objective(ind)
        # Keep best individual
        best_ind = min(pop, key=lambda x: x.fitness.values)
        print(f"Generation {gen} best C5: {best_ind.fitness.values[0]}")

    best_h = np.array(best_ind)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
