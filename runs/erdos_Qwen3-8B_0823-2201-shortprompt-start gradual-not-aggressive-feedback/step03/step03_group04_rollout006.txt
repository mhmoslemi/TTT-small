import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial guess: refined version of the previous construction
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            initial_h[i] = 1.0
        elif i % 5 == 1:
            initial_h[i] = 0.75
        elif i % 5 == 2:
            initial_h[i] = 0.5
        elif i % 5 == 3:
            initial_h[i] = 0.25
        else:
            initial_h[i] = 0.0
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # First stage: gradient-based optimization
    result = opt.minimize(
        objective,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-9}
    )
    
    # Second stage: random restarts with local refinement
    best_result = result
    elapsed = 0
    start_time = time.time()
    
    for _ in range(5):
        if time.time() - start_time > budget_s - 2:
            break
        # Random perturbation
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = result.x + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Local refinement
        refined_result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        
        if refined_result.fun < best_result.fun:
            best_result = refined_result
    
    # Final refinement with different starting point
    if time.time() - start_time < budget_s - 2:
        new_initial = np.random.uniform(0, 1, n_points)
        new_initial = new_initial * target_sum / np.sum(new_initial)
        final_result = opt.minimize(
            objective,
            new_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        if final_result.fun < best_result.fun:
            best_result = final_result
    
    best_h = best_result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
