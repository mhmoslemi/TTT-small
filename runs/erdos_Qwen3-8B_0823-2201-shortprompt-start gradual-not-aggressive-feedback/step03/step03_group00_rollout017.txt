import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def genetic_algorithm():
        population = np.random.rand(20, n_points)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        population = np.clip(population, 0, 1)
        best_h = initial_h.copy()
        best_c5 = objective(best_h)
        
        for _ in range(100):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            best_idx = np.argmin(fitness)
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
            
            # Select parents
            parents = population[np.argsort(fitness)[:10]]
            
            # Crossover and mutation
            new_population = []
            for _ in range(10):
                parent1, parent2 = random.sample(parents, 2)
                child = np.zeros(n_points)
                for i in range(n_points):
                    if random.random() < 0.5:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
                new_population.append(child)
            
            # Replace worst individuals
            population = np.vstack((population, new_population))
            population = np.sort(population, axis=0)
            population = population[-20:]
        
        return best_h, best_c5

    best_h, best_c5 = genetic_algorithm()
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Local optimization with SLSQP
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
            print(f"Improved bound: {best_c5}")
    
    return best_h, best_c5, n_points
