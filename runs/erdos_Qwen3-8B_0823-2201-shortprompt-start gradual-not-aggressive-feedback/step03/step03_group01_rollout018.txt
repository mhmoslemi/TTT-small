import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def generate_candidate(parent_h):
        h = parent_h.copy()
        idxs = np.random.choice(n_points, size=min(n_points // 2, 10), replace=False)
        h[idxs] = np.random.uniform(0, 1, size=len(idxs))
        h = h * target_sum / np.sum(h)
        return h

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Genetic step
        child_h = generate_candidate(best_h)
        child_c5 = objective(child_h)
        
        # Local optimization
        result = opt.minimize(
            objective,
            child_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 20, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    return best_h, best_c5, n_points
