import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def genetic_algorithm(starting_population_size=20):
        population = np.random.uniform(0, 1, (starting_population_size, n_points))
        population = (population * target_sum / np.sum(population, axis=1, keepdims=True))
        
        for _ in range(100):
            fitness = np.array([objective(ind) for ind in population])
            best_idx = np.argmin(fitness)
            best_h = population[best_idx].copy()
            
            # Selection
            selected = np.random.choice(len(population), size=len(population), p=fitness / np.sum(fitness))
            population = population[selected]
            
            # Crossover
            for i in range(0, len(population), 2):
                if i + 1 < len(population):
                    parent1 = population[i]
                    parent2 = population[i+1]
                    child1 = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                    child2 = np.where(np.random.rand(n_points) < 0.5, parent2, parent1)
                    population[i] = child1
                    population[i+1] = child2
            
            # Mutation
            mutation_rate = 0.1
            for i in range(len(population)):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        population[i, j] = np.clip(population[i, j] + np.random.uniform(-0.1, 0.1), 0, 1)
                        population[i, j] = population[i, j] * target_sum / np.sum(population[i])
            
            # Replace worst with best
            worst_idx = np.argmax(fitness)
            population[worst_idx] = best_h.copy()
        
        best_h = population[np.argmin([objective(ind) for ind in population])]
        return best_h

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm search
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Run genetic algorithm
        genetic_h = genetic_algorithm()
        
        # Optimization
        result = opt.minimize(
            objective,
            genetic_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
            print(f"Improved bound: {best_c5}")
    
    return best_h, best_c5, n_points
