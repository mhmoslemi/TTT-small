import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def genetic_algorithm():
        population_size = 20
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        for _ in range(100):
            fitness = [objective(ind) for ind in population]
            best_idx = np.argmin(fitness)
            best_h = population[best_idx]
            population = [best_h] + [np.random.uniform(0, 1, n_points) for _ in range(population_size - 1)]
            for ind in population[1:]:
                mutation = np.random.normal(0, 0.05, n_points)
                ind += mutation
                ind = np.clip(ind, 0, 1)
                ind = ind * target_sum / np.sum(ind)
            population = [ind for ind in population if objective(ind) < 0.5]
        return np.min(fitness), population[np.argmin(fitness)]

    start_time = time.time()
    best_c5, best_h = genetic_algorithm()
    best_h = best_h.reshape(n_points)

    while time.time() - start_time < budget_s:
        result = opt.minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        best_c5 = objective(best_h)

    return best_h, best_c5, n_points
