import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def evolve(population, fitness_func, mutation_rate=0.1):
        population = np.array(population)
        fitness = np.array([fitness_func(ind) for ind in population])
        best_idx = np.argmin(fitness)
        best = population[best_idx].copy()
        new_population = [best]
        
        for _ in range(len(population) - 1):
            parent1 = np.random.choice(population)
            parent2 = np.random.choice(population)
            child = (parent1 + parent2) / 2
            if np.random.rand() < mutation_rate:
                child += np.random.normal(0, 0.1, size=len(child))
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        return new_population

    population_size = 20
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        population = evolve(population, objective)
        best_h = population[np.argmin([objective(h) for h in population])]
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        best_c5 = objective(best_h)
        
        # Local optimization
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        best_c5 = objective(best_h)
        
        print(f"Improved bound: {best_c5}")
    
    return best_h, best_c5, n_points
