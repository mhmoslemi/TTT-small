import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Multi-stage optimization
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Genetic algorithm for global search
        population_size = min(20, n_points)
        population = np.random.uniform(0, 1, (population_size, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(5):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            
            # Select parents
            parents_indices = np.argsort(fitness)[:population_size//2]
            parents = population[parents_indices]
            
            # Crossover
            offspring = []
            for _ in range(population_size//2):
                parent1, parent2 = np.random.choice(parents, 2, replace=False)
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                offspring.append(child)
            
            # Mutation
            for i in range(len(offspring)):
                offspring[i] = np.clip(
                    offspring[i] + np.random.normal(0, 0.05, n_points),
                    0, 1
                )
                offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
            
            # Replace population
            population = np.vstack((parents, offspring))
            population = population[np.random.permutation(population.shape[0])]
            
            # Update best solution
            for ind in population:
                current_c5 = objective(ind)
                if current_c5 < best_c5:
                    best_h = ind
                    best_c5 = current_c5
                    print(f"Improved bound: {best_c5}")

        # Local refinement with SLSQP
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
            print(f"Improved bound: {best_c5}")
    
    return best_h, best_c5, n_points
