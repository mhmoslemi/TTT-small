from scipy.optimize import minimize
import time
from deap import base, creator, tools, algorithms

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    def eval_func(ind):
        h = np.array(ind)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return (c5,)
    
    def cx_ind(ind1, ind2):
        size = len(ind1)
        child1, child2 = np.copy(ind1), np.copy(ind2)
        for i in range(size):
            if np.random.random() < 0.5:
                child1[i], child2[i] = child2[i], child1[i]
        return child1, child2
    
    def mutate(ind):
        for i in range(len(ind)):
            if np.random.random() < 0.1:
                ind[i] += np.random.normal(0, 0.1)
                ind[i] = np.clip(ind[i], 0, 1)
        return ind,
    
    # Genetic algorithm parameters
    toolbox = base.Toolbox()
    toolbox.register("evaluate", eval_func)
    toolbox.register("mate", cx_ind)
    toolbox.register("mutate", mutate)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    # Initialize population
    pop = [np.random.rand(n_points) for _ in range(50)]
    for ind in pop:
        ind[:] = np.clip(ind, 0, 1)
    
    # Optimization
    for gen in range(100):
        if time.time() - start_time >= budget_s - 1:
            break
        offspring = algorithms.varAnd(pop, toolbox, cxpb=0.5, mutpb=0.1)
        fits = toolbox.map(toolbox.evaluate, offspring)
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = fit
        pop = toolbox.select(pop + offspring, len(pop))
    
    # Find best solution
    best_h = min(pop, key=lambda x: x.fitness.values[0])
    best_c5 = evaluate(best_h)
    
    # Local refinement

        h = np.clip(x, 0, 1)

        return np.sum(h) - n_points / 2
    
    result = minimize(objective, best_h, bounds=[(0, 1) for _ in range(n_points)], method='SLSQP', constraints={'type': 'eq', 'fun': constraint})
    best_h = result.x
    best_c5 = evaluate(best_h)
    
    return (best_h, best_c5, n_points)
