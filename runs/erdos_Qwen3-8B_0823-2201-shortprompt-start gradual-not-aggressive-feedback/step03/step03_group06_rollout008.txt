from scipy.signal import correlate

def run(seed=42, budget_s=20, **kwargs):

    n_points = 256
    dx = 2.0 / n_points
    population_size = 64
    generations = 100
    mutation_rate = 0.15
    best_c5 = float('inf')
    best_h = None
    
    # Initial population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)

        h = np.clip(h, 0, 1)

        c5 = np.max(corr) * dx
        return c5
    
    # Evolutionary algorithm
    for gen in range(generations):

        fitness = [objective(ind) for ind in population]

        if best_c5_current < best_c5:
            best_c5 = best_c5_current

        parents = np.random.choice(len(population), size=population_size, p=np.array(fitness)/np.sum(fitness))

        new_population = []
