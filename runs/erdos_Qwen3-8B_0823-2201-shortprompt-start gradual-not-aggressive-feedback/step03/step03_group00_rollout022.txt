import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def evaluate_individual(h):
        return objective(h)

    def crossover(parent1, parent2):
        child = np.copy(parent1)
        indices = np.random.choice(n_points, size=n_points // 2, replace=False)
        child[indices] = parent2[indices]
        return child

    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < 0.1:
                h[i] += np.random.uniform(-0.1, 0.1)
                h[i] = np.clip(h[i], 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        population = [mutate(best_h) for _ in range(10)]
        population.append(best_h)
        population = [np.clip(p, 0, 1) for p in population]
        population = [p * target_sum / np.sum(p) for p in population]
        scores = [evaluate_individual(p) for p in population]
        best_idx = np.argmin(scores)
        best_h = population[best_idx]
        best_c5 = scores[best_idx]
        print(f"Generation {generation}: Best C5 = {best_c5}")

        # Local optimization
        result = opt.minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        best_c5 = objective(best_h)

    return best_h, best_c5, n_points
