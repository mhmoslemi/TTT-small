import numpy as np
import scipy.signal as signal
import random
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 150  # Smaller for faster computation
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial guess: alternating pattern
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            initial_h[i] = 1.0
        elif i % 3 == 1:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 0.0
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def normalize(h):
        """Maintain integral constraint"""
        return h * target_sum / np.sum(h)
    
    def evaluate(h):
        """Evaluate the C₅ bound"""
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = signal.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def mutate(h, mutation_rate=0.1):
        """Apply small random perturbations"""
        h = h.copy()
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] += np.random.uniform(-0.1, 0.1)
                h[i] = np.clip(h[i], 0, 1)
        return normalize(h)
    
    def crossover(parent1, parent2):
        """Single-point crossover"""
        point = random.randint(1, len(parent1)-1)
        child = np.concatenate((parent1[:point], parent2[point:]))
        return normalize(child)
    
    population_size = 50
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    
    start_time = time.time()
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [evaluate(individual) for individual in population]
        
        # Select parents
        parents = np.argsort(fitness)[:population_size//2]
        parents = [population[i] for i in parents]
        
        # Create new population
        new_population = []
        for _ in range(population_size//2):
            parent1, parent2 = random.sample(parents, 2)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        # Replace old population
        population = new_population
        
        # Track best solution
        best_index = np.argmin(fitness)
        best_h = population[best_index]
        best_c5 = fitness[best_index]
    
    best_h = normalize(best_h)
    best_c5 = evaluate(best_h)
    
    return best_h, best_c5, n_points
