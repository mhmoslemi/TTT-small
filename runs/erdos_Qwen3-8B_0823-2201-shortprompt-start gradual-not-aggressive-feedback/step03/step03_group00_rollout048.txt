import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with a uniform distribution
    h = np.full(n_points, 0.5)
    h = np.random.permutation(h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    # Simulated annealing with local optimization
    temp = 1.0
    cooling_rate = 0.995
    
    while time.time() - start_time < budget_s:
        for _ in range(100):
            # Perturb the solution
            perturb = np.random.normal(0, 0.05, n_points)
            new_h = best_h + perturb
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h)
            
            # Local optimization
            result = opt.minimize(
                objective,
                new_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = objective(current_h)
            
            # Accept with probability based on temperature
            if current_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - current_c5)/temp):
                best_h = current_h
                best_c5 = current_c5
        
        temp *= cooling_rate
    
    return best_h, best_c5, n_points
