import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def local_optimize(h):
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x

    def genetic_step(population, mutation_rate=0.1, crossover_rate=0.8):
        new_population = []
        for _ in range(len(population)):
            parent1, parent2 = random.sample(population, 2)
            if np.random.rand() < crossover_rate:
                child = np.zeros_like(parent1)
                for i in range(len(child)):
                    if np.random.rand() < 0.5:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
            else:
                child = parent1.copy()
            for i in range(len(child)):
                if np.random.rand() < mutation_rate:
                    child[i] = np.clip(child[i] + np.random.normal(0, 0.1), 0, 1)
            new_population.append(child)
        return new_population

    population = [initial_h.copy() for _ in range(20)]
    population = [np.array(p) for p in population]
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Local optimization
        for h in population:
            h_opt = local_optimize(h)
            c5 = objective(h_opt)
            if c5 < best_c5:
                best_h = h_opt
                best_c5 = c5
        # Genetic step
        population = genetic_step(population)
        # Prune population
        population = np.array([h for h in population if np.sum(h) > target_sum - 1e-5])
        population = [np.clip(h, 0, 1) for h in population]
        population = [h * target_sum / np.sum(h) for h in population]
        population = [np.array(h) for h in population]

    return best_h, best_c5, n_points
