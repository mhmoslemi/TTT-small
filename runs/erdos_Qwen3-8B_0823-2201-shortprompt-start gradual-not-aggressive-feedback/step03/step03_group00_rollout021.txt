def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize population
    population_size = 30
    population = [np.array(initial_h) + np.random.uniform(-0.1, 0.1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    best_h = np.array(initial_h)
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = [objective(h) for h in population]
        
        # Find the best individual
        idx = np.argmin(fitness)
        current_h = population[idx]
        current_c5 = fitness[idx]
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        
        # Create new population
        new_population = []
        
        # Elitism: keep the best individual
        new_population.append(current_h)
        
        # Select parents
        parents = np.random.choice(population_size, size=population_size, replace=True)
        
        # Crossover and mutation
        for i in range(population_size - 1):
            parent1 = population[parents[i]]
            parent2 = population[parents[(i+1)%population_size]]
            
            # Crossover
            child = np.copy(parent1)
            mask = np.random.rand(n_points) < 0.5
            child[mask] = parent2[mask]
            
            # Mutation
            mutation = np.random.uniform(-0.05, 0.05, n_points)
            child += mutation
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
    
    return best_h, best_c5, n_points
