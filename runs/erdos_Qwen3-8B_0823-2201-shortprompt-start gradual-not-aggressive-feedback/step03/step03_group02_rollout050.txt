import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = [h * (n_points / 2) / np.sum(h) for h in population]
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Genetic algorithm parameters
    mutation_rate = 0.1
    crossover_rate = 0.7
    generations = 50
    
    best_h = population[0]
    best_c5 = evaluate(best_h)
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate population
        fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Selection
        parents = np.random.choice(population_size, size=population_size, p=np.array(fitness)/np.sum(fitness))
        parents = [population[i] for i in parents]
        
        # Crossover
        offspring = []
        for i in range(population_size // 2):
            if np.random.rand() < crossover_rate:
                parent1 = parents[2*i]
                parent2 = parents[2*i + 1]
                child1 = np.concatenate([parent1[:n_points//2], parent2[n_points//2:]])
                child2 = np.concatenate([parent2[:n_points//2], parent1[n_points//2:]])
                offspring.append(child1)
                offspring.append(child2)
            else:
                offspring.append(np.random.uniform(0, 1, n_points))
        
        # Mutation
        for i in range(len(offspring)):
            h = offspring[i]
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    h[j] = np.clip(h[j] + np.random.normal(0, 0.05), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            offspring[i] = h
        
        # Replace population
        population = offspring
        
    return (best_h, best_c5, n_points)
