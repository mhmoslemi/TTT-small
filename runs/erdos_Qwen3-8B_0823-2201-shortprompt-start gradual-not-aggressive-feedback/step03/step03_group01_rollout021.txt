    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

        h = genome.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = evaluate(best_h)

    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.15
    generations = 20
    time_remaining = budget_s

    start_time = time.time()

    for _ in range(generations):
        if time.time() - start_time > time_remaining:
            break
        
        # Create population
        population = [best_h.copy() for _ in range(population_size)]
        for i in range(population_size):
            if random.random() < 0.3:
                population[i] = np.random.uniform(0, 1, n_points)
        
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        best_index = np.argmin(fitness)
        current_best = population[best_index].copy()
        current_c5 = fitness[best_index]
        
        # Local optimization
        result = opt.minimize(
            objective,
