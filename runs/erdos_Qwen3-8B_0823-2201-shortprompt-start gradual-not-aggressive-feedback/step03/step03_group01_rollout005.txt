import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def genetic_algorithm():
        population_size = 50
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        
        for generation in range(200):
            if time.time() - start_time > budget_s:
                break
                
            fitness = [objective(h) for h in population]
            best_idx = np.argmin(fitness)
            best_h = population[best_idx]
            best_c5 = fitness[best_idx]
            
            if best_c5 < current_best_c5:
                current_best_h = best_h
                current_best_c5 = best_c5
                
            # Selection
            selected = np.argsort(fitness)[:population_size//2]
            selected = [population[i] for i in selected]
            
            # Crossover
            new_population = []
            for _ in range(population_size//2):
                parent1, parent2 = random.sample(selected, 2)
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                new_population.append(child)

            for i in range(len(new_population)):

                new_population[i] += mutation
