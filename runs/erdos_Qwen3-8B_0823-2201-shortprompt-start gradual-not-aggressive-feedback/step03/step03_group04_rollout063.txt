import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial population
    population_size = 50
    population = np.random.rand(population_size, n_points)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    population = np.clip(population, 0, 1)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def evaluate_individual(h):
        return objective(h)
    
    # Genetic algorithm parameters
    generations = 100
    tournament_size = 5
    mutation_rate = 0.1
    mutation_strength = 0.1
    
    start_time = time.time()
    
    for generation in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = np.array([evaluate_individual(ind) for ind in population])
        
        # Select parents using tournament selection
        indices = np.random.randint(0, population_size, (population_size, tournament_size))
        tournament_fitness = fitness[indices]
        winners = np.argmin(tournament_fitness, axis=1)
        parents = population[winners]
        
        # Crossover
        offspring = np.zeros_like(parents)
        for i in range(population_size):
            if np.random.rand() < 0.8:
                parent1 = parents[np.random.randint(0, population_size)]
                parent2 = parents[np.random.randint(0, population_size)]
                crossover_point = np.random.randint(1, n_points - 1)
                offspring[i] = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                offspring[i] = parents[i]
        
        # Mutation
        for i in range(population_size):
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    offspring[i][j] += np.random.uniform(-mutation_strength, mutation_strength)
                    offspring[i][j] = np.clip(offspring[i][j], 0, 1)
        
        # Replace population
        population = offspring
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        population = np.clip(population, 0, 1)
    
    # Find best solution
    best_idx = np.argmin(evaluate_individual(population))
    best_h = population[best_idx]
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = evaluate_individual(best_h)
    
    return best_h, best_c5, n_points
