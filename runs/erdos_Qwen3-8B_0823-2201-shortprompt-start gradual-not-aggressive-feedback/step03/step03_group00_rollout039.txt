import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def evolve_population(population, mutation_rate=0.1):
        new_population = []
        for h in population:
            mutated = h.copy()
            for i in range(len(mutated)):
                if random.random() < mutation_rate:
                    mutated[i] += np.random.uniform(-0.1, 0.1)
                    mutated[i] = np.clip(mutated[i], 0, 1)
            new_population.append(mutated)
        return new_population
    
    population = [initial_h.copy() for _ in range(10)]
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate population
        evaluated = []
        for h in population:
            h = h.copy()
            h = h * target_sum / np.sum(h)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            evaluated.append((c5, h))
        
        # Select best candidates
        evaluated.sort()
        top_candidates = [h for (c5, h) in evaluated[:5]]
        
        # Evolve new population
        population = evolve_population(top_candidates)
        
        # Refine best candidate with SLSQP
        for h in top_candidates:
            h = h.copy()
            h = h * target_sum / np.sum(h)
            result = opt.minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = objective(current_h)
            
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                print(f"Improved bound: {best_c5}")
    
    return best_h, best_c5, n_points
