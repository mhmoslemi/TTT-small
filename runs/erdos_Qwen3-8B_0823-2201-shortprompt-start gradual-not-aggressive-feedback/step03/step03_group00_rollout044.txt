        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    population_size = 20
    population = [initial_h.copy() for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        
        # Generate new population
        new_population = []
        for _ in range(population_size // 2):
            # Random selection for crossover
            idx1, idx2 = np.random.choice(range(population_size), 2, replace=False)
            parent1 = population[idx1]
            parent2 = population[idx2]
            
            # Crossover
            child1 = np.concatenate([parent1[:n_points//2], parent2[n_points//2:]])
            child2 = np.concatenate([parent2[:n_points//2], parent1[n_points//2:]])

            for i in range(n_points):

                    child1[i] += np.random.uniform(-0.1, 0.1)

            # Clip and normalize
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * target_sum / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * target_sum / np.sum(child2)
            
            new_population.append(child1)
