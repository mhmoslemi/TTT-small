import scipy.optimize as optimize
from scipy.signal import correlate

def run(seed=42, budget_s=20, **kwargs):

        x_initial = np.linspace(0, 2, len(initial_h_values))

        # Enforce integral constraint
        sum_h = np.sum(h_values)
        if not np.isclose(sum_h, n_points / 2):
            h_values = h_values * (n_points / 2) / sum_h
        
        def objective(h):
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if not np.isclose(sum_h, n_points / 2):
                h = h * (n_points / 2) / sum_h
            h = np.clip(h, 0, 1)
            one_minus_h = 1 - h
            corr = correlate(h, one_minus_h, mode="full")
            c5 = np.max(corr) * (2.0 / n_points)
            return c5
        
        def constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]

        # Initial optimization
        result = optimize.minimize(
            objective,
            h_values,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
        )
        
        best_h_current = result.x
        best_c5_current = objective(best_h_current)

                method='SLSQP',
