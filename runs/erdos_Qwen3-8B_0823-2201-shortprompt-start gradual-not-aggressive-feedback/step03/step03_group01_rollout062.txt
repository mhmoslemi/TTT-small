import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with SLSQP
    result = opt.minimize(
        objective,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 50
    time_remaining = budget_s

    # Initialize population
    population = [np.random.rand(n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]

    start_time = time.time()
    for _ in range(generations):
        if time.time() - start_time > time_remaining:
            break
        # Evaluate population
        fitness = [objective(h) for h in population]
        # Select elite
        elite_index = np.argmin(fitness)
        elite = population[elite_index]
        # Create new population
        new_population = [elite.copy() for _ in range(population_size // 2)]
        for _ in range(population_size // 2):
            if time.time() - start_time > time_remaining:
                break
            # Select parents
            parent1 = population[np.random.choice(len(population))]
            parent2 = population[np.random.choice(len(population))]
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            else:
                child = parent1.copy()
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation = np.random.uniform(-0.1, 0.1, n_points)
                child += mutation
            # Clip and normalize
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = new_population

    # Evaluate final population
    fitness = [objective(h) for h in population]
    best_index = np.argmin(fitness)
    best_h = population[best_index]
    best_c5 = fitness[best_index]

    return best_h, best_c5, n_points
