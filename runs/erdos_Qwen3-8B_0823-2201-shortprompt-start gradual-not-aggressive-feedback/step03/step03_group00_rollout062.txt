import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_c5 = objective(best_h)

    # Global search with random perturbations and SLSQP
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Random perturbation
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        
        # Optimization with penalty for constraint violation
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )

            best_h = current_h
            best_c5 = current_c5
            print(f"Improved bound: {best_c5}")
