import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def evaluate(h, n):
        dx = 2.0 / n
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.inf
        h = h * (n / 2) / total
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def constraint(h, n):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.inf
        return np.sum(h) - n / 2
    
    best_h = h.copy()
    best_c5 = evaluate(h, n_points)
    time_left = budget_s
    
    for _ in range(5):
        if time_left <= 0:
            break
        time_start = time.time()
        h_init = np.random.uniform(0, 1, n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        bounds = Bounds(0, 1)
        cons = LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2)

        time_left -= time_used
