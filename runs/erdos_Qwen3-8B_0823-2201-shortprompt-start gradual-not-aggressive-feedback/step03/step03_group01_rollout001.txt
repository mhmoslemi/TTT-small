import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    dx = 2.0 / n_points

        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    best_h = initial_h.copy()
    best_h = np.clip(best_h, 0, 1)

    best_c5 = objective(best_h)

    start_time = time.time()

    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.1
    crossover_rate = 0.8

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Create population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        
        # Evaluate population
        population_objs = []
        for h in population:
            h = h * target_sum / np.sum(h)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            population_objs.append(c5)
        
        # Select best individuals
        sorted_indices = np.argsort(population_objs)
        selected = [population[i] for i in sorted_indices[:population_size//2]]

        new_population = []

            parent1 = random.choice(selected)
            parent2 = random.choice(selected)
            child = parent1 * np.random.rand() + parent2 * (1 - np.random.rand())
            new_population.append(child)
        
        # Mutation
        for i in range(len(new_population)):

        # Combine populations
        population = selected + new_population
        
        # Update best solution

            c5 = np.max(corr) * dx

                best_h = h.copy()

    # Refine with local optimization

    return best_h, best_c5, n_points
