import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with SLSQP
    result = opt.minimize(
        objective,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm with local refinement
    start_time = time.time()
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    mutation_scale = 0.15

    while time.time() - start_time < budget_s:
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        for i in range(population_size):
            population[i] = population[i] * target_sum / np.sum(population[i])
        
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_gen = population[best_idx]
        best_c5_gen = fitness[best_idx]

        if best_c5_gen < best_c5:
            best_h = best_gen
            best_c5 = best_c5_gen

        # Evolutionary step
        for _ in range(generations):
            # Select parents
            parents = np.random.choice(population_size, size=population_size, replace=True)
            parents = [population[i] for i in parents]
            
            # Crossover
            offspring = []
            for i in range(0, population_size, 2):
                if i + 1 < population_size:
                    p1 = parents[i]
                    p2 = parents[i + 1]
                    child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
                    offspring.append(child)
                else:
                    offspring.append(parents[i])
            
            # Mutation
            for i in range(len(offspring)):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        offspring[i][j] += np.random.uniform(-mutation_scale, mutation_scale)
                        offspring[i][j] = np.clip(offspring[i][j], 0, 1)
                        offspring[i][j] = offspring[i][j] * target_sum / np.sum(offspring[i])
            
            # Replace population
            population = offspring
        
        # Local refinement
        for ind in population:
            result = opt.minimize(
                objective,
                ind,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            ind = result.x
            ind = np.clip(ind, 0, 1)
            ind = ind * target_sum / np.sum(ind)
            fitness = objective(ind)
            if fitness < best_c5:
                best_h = ind
                best_c5 = fitness

    return best_h, best_c5, n_points
