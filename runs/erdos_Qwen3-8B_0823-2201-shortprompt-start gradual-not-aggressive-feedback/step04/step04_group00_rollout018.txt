import scipy.optimize as opt
import time

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Genetic algorithm for global search
    def genetic_algorithm():
        population = np.random.rand(30, n_points)
        population = np.clip(population, 0, 1)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        best_h = population[0]
        best_c5 = objective(best_h)
        
        for _ in range(100):
            # Selection
            fitness = np.array([objective(ind) for ind in population])
            probabilities = fitness / np.sum(fitness)
            parents = np.random.choice(range(30), size=10, p=probabilities)
            
            # Crossover
            offspring = []
            for i in range(0, 10, 2):
                a, b = parents[i], parents[i+1]
                mask = np.random.rand(n_points) > 0.5
                child1 = np.where(mask, population[a], population[b])
                child2 = np.where(mask, population[b], population[a])
                offspring.append(child1)
                offspring.append(child2)
            
            # Mutation
            for i in range(len(offspring)):
                mutant = offspring[i] + 0.1 * np.random.randn(n_points)
                mutant = np.clip(mutant, 0, 1)
                mutant = mutant * target_sum / np.sum(mutant)
                offspring[i] = mutant
            
            # Replacement
            population = np.array(offspring)
            fitness = np.array([objective(ind) for ind in population])
            idx = np.argsort(fitness)
            population = population[idx[:30]]

            current_best = population[0]

            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
