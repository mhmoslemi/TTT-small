import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Genetic algorithm for global search
    def genetic_algorithm(pop_size=50, generations=50):
        population = np.random.rand(pop_size, n_points)
        population = np.clip(population, 0, 1)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        best_h = population[0]
        best_c5 = objective(best_h)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            
            # Select parents
            parents = population[np.argsort(fitness)[:pop_size//2]]
            
            # Crossover
            offspring = []
            for _ in range(pop_size//2):
                parent1, parent2 = np.random.choice(parents, 2, replace=False)
                child = (parent1 + parent2) / 2
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
                offspring.append(child)
            
            # Mutate
            for i in range(len(offspring)):
                mutation = np.random.normal(0, 0.05, n_points)
                offspring[i] += mutation
                offspring[i] = np.clip(offspring[i], 0, 1)
                offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
            
            # Replace population
            population = np.vstack((parents, offspring))
            population = np.random.permutation(population)[:pop_size]
            
            # Update best solution
            for ind in population:
                current_c5 = objective(ind)
                if current_c5 < best_c5:
                    best_h = ind
                    best_c5 = current_c5
        
        return best_h, best_c5
    
    # Local refinement with SQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Main optimization loop
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        # Perturb and refine
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
