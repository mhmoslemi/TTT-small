import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial population
    population = np.random.rand(20, n_points)
    population = np.clip(population, 0, 1)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    population = population.tolist()
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        idx = np.random.choice(n_points, size=n_points//2, replace=False)
        child[idx] = parent2[idx]
        return np.clip(child, 0, 1) * target_sum / np.sum(child)
    
    def mutate(h):
        mutated = np.copy(h)
        idx = np.random.choice(n_points, size=n_points//4, replace=False)
        mutated[idx] += np.random.uniform(-0.1, 0.1, size=len(idx))
        mutated = np.clip(mutated, 0, 1) * target_sum / np.sum(mutated)
        return mutated
    
    best_h = np.copy(initial_h)
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Genetic algorithm
        new_population = []
        for i in range(10):
            p1, p2 = np.random.choice(population, size=2, replace=False)
            child = crossover(p1, p2)
            child = mutate(child)
            new_population.append(child)
        population += new_population
        population = np.array(population)
        population = population[np.argsort([objective(p) for p in population])[:20]]
        
        # Local refinement
        for p in population:
            result = opt.minimize(
                objective,
                p,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            if objective(result.x) < best_c5:
                best_h = result.x
                best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
