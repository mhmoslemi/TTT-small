    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        return objective(h)

    population_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 100

    def evolve(population):
        parents = np.random.choice(population, size=population_size, replace=False)
        children = []
        for i in range(population_size):
            if np.random.rand() < crossover_rate:
                p1, p2 = np.random.choice(parents, size=2, replace=False)
                child = np.clip(p1 * np.random.rand() + p2 * (1 - np.random.rand()), 0, 1)
            else:
                child = np.copy(np.random.choice(parents, size=1)[0])
            child = child * target_sum / np.sum(child)
            children.append(child)
        return children

    population = [np.copy(initial_h) * target_sum / np.sum(initial_h) for _ in range(population_size)]

    best_c5 = objective(best_h)

    for gen in range(generations):

            break

        for h in population:

            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5
                print(f"Improved bound: {best_c5}")
