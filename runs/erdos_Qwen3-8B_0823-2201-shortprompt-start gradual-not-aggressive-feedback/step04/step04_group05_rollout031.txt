import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    best_h = initial_h.copy()

    best_n = n_points
    time_left = budget_s

    def evaluate(h, n):
        dx = 2.0 / n
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.inf
        h = h * (n / 2) / total
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint(h, n):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.inf
        return np.sum(h) - n / 2

    # Start with the initial construction
    n_vals = [100, 120, 140, 160, 180, 200]
    for n in n_vals:
        if time_left <= 0:
            break
        time_start = time.time()
        # Interpolate from initial_h to the new n
        x_initial = np.linspace(0, 2, len(initial_h))
        x_new = np.linspace(0, 2, n)
        h_init = np.interp(x_new, x_initial, initial_h)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n / 2) / np.sum(h_init)
        
        bounds = Bounds(0, 1)
        cons = LinearConstraint(np.ones(n), lb=n/2, ub=n/2)
        
        # Use SLSQP with tighter tolerances
        result = optimize.minimize(lambda x: evaluate(x, n), h_init, 
                                 method='SLSQP', bounds=bounds, constraints=cons,
                                 tol=1e-8, options={'maxiter': 200})
        
        if result.success:
            h_opt = result.x
            c5 = evaluate(h_opt, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n = n
        time_used = time.time() - time_start

    # Genetic algorithm refinement
    if time_left > 0:
        n_refine = 100
        time_start = time.time()

        h_refine = np.clip(h_refine, 0, 1)
        h_refine = h_refine * (n_refine / 2) / np.sum(h_refine)

        toolbox.register("population", tools.initRepeat, list, toolbox.individual)
        toolbox.register("mate", tools.cxBlend, alpha=0.5)

        toolbox.register("select", tools.selTournament, tournsize=3)
        toolbox.register("evaluate", lambda ind: evaluate(ind, n_refine))

        # Evaluate the entire population
        fits = list(map(toolbox.evaluate, pop))

            pop = tools.selBest(pop + offspring, len(pop))
            time_used = time.time() - time_start
            time_left -= time_used

        best_ind = min(pop, key=lambda ind: ind.fitness.values[0])
        if best_ind.fitness.values[0] < best_c5:
            best_c5 = best_ind.fitness.values[0]
            best_h = best_ind
            best_n = n_refine
    
    return best_h, best_c5, best_n
