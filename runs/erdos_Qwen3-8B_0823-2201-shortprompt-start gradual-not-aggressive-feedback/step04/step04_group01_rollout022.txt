import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def local_optimize(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x

    def perturb(h):
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        perturbed = h + perturb
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        return perturbed

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Local optimization
        best_h = local_optimize(best_h)
        best_c5 = objective(best_h)
        
        # Periodic perturbation and re-optimization
        if np.random.rand() < 0.2:
            best_h = perturb(best_h)
            best_h = local_optimize(best_h)
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
