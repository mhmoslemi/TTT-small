            fitness = np.array([objective(h) for h in population])

            parents = np.random.choice(len(population), size=10, p=probabilities)
            offspring = []
            for i in range(0, 10, 2):
                h1 = population[parents[i]]
                h2 = population[parents[i+1]]
                child = np.where(np.random.rand(n_points) < 0.5, h1, h2)
                offspring.append(child)
