import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Using a larger n_points than the initial construction
    initial_h = np.zeros(n_points)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    # Initial construction: evenly spaced intervals
    for i in range(n_points):
        initial_h[i] = 1 if i % 5 == 0 else 0
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    remaining_time = budget_s
    
    # Try different optimization methods and parameters
    for method in ['SLSQP', 'COBYLA', 'Nelder-Mead']:
        if remaining_time <= 0:
            break
        # Random perturbation and optimization
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        try:
            result = opt.minimize(
                objective,
                perturbed_h,
                method=method,
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = objective(current_h)
            
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                print(f"Improved bound: {best_c5}")
        except:
            pass
        
        # Check time budget
        remaining_time = budget_s - (time.time() - start_time)
    
    return best_h, best_c5, n_points
