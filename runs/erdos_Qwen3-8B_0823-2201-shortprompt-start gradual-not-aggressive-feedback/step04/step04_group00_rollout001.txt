import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize with initial construction
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Simulated annealing parameters
    temp = 1.0
    cooling_rate = 0.995
    iterations = 1000
    
    start_time = time.time()
    
    for _ in range(iterations):
        if time.time() - start_time > budget_s:
            break
        # Generate perturbation
        perturb = np.random.normal(0, 0.1, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Evaluate
        perturbed_c5 = objective(perturbed_h)
        
        # Acceptance probability
        if perturbed_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - perturbed_c5) / temp):
            best_h = perturbed_h
            best_c5 = perturbed_c5
        
        temp *= cooling_rate
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
