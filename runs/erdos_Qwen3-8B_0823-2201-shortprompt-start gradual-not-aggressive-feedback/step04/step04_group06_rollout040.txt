import numpy as np
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial population
    n_points = 200
    dx = 2.0 / n_points
    population_size = 50
    population = []
    for _ in range(population_size):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Evolutionary loop
    for generation in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Selection
        selected = np.random.choice(range(population_size), size=population_size, p=np.array(fitness)/np.sum(fitness))
        selected = [population[i] for i in selected]
        
        # Crossover
        new_population = []
        for i in range(0, population_size, 2):
            if i + 1 < population_size:
                h1 = selected[i]
                h2 = selected[i+1]
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
                child2 = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
                new_population.extend([child1, child2])
            else:
                new_population.append(selected[i])
        
        # Mutation
        for i in range(len(new_population)):
            h = new_population[i]
            mutation = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + mutation, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            new_population[i] = h
        
        population = new_population
    
    return (best_h, best_c5, n_points)
