import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values  # 47-point initial construction
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Local optimization function
    def local_optimize(h_initial):
        result = opt.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x

    # Genetic algorithm with local refinement
    def genetic_algorithm():
        population = [np.random.uniform(0, 1, n_points) for _ in range(50)]
        population = [h * target_sum / np.sum(h) for h in population]
        best_h = population[0]
        best_c5 = objective(best_h)
        
        for _ in range(100):
            new_population = []
            for h in population:
                perturb = np.random.normal(0, 0.05, n_points)
                perturbed = h + perturb
                perturbed = np.clip(perturbed, 0, 1)
                perturbed = perturbed * target_sum / np.sum(perturbed)
                new_population.append(perturbed)
            
            population = new_population
            for h in population:
                current_c5 = objective(h)
                if current_c5 < best_c5:
                    best_h = h
                    best_c5 = current_c5
        
        # Local refinement
        refined_h = local_optimize(best_h)
        refined_c5 = objective(refined_h)
        return refined_h, refined_c5

    best_h, best_c5 = genetic_algorithm()
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Local optimization
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    return best_h, best_c5, n_points
