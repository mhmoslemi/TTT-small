import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initial population
    population_size = 50
    population = np.array([initial_h.copy() for _ in range(population_size)])
    population = np.array([h * target_sum / np.sum(h) for h in population])
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = np.array([evaluate(h) for h in population])
        
        # Select parents
        parents = population[np.argsort(fitness)[:population_size//2]]
        
        # Crossover
        offspring = []
        for _ in range(population_size//2):
            parent1, parent2 = np.random.choice(parents, size=2, replace=False)
            crossover_point = np.random.randint(1, n_points-1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            for j in range(n_points):
                if np.random.rand() < 0.1:
                    offspring[i][j] += np.random.uniform(-0.1, 0.1)
                    offspring[i][j] = np.clip(offspring[i][j], 0, 1)
        
        # Create new population
        population = np.array(offspring)
        population = np.array([h * target_sum / np.sum(h) for h in population])
        
        # Update best solution
        for h in population:
            c5 = evaluate(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    
    return best_h, best_c5, n_points
