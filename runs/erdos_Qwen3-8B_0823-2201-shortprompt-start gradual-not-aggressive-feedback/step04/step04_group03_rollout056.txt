import scipy.optimize as opt
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    # Initial solution
    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    popsize = 15
    mutation = 0.1
    recombination = 0.5
    tol = 1e-4
    maxiter = 100
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        result = differential_evolution(
            objective,
            bounds,
            strategy='rand',
            popsize=popsize,
            mutation=mutation,
            recombination=recombination,
            tol=tol,
            maxiter=maxiter,
            constraints={'type': 'eq', 'fun': constraint},
            seed=seed
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
