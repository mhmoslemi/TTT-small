import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_initial = 80
    n_vals = [80, 100, 120, 140, 160, 180, 200]
    best_h = None
    best_c5 = float('inf')
    best_n = n_initial
    time_left = budget_s

    def evaluate(h, n):
        dx = 2.0 / n
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.inf
        h = h * (n / 2) / total
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint(h, n):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.inf
        return np.sum(h) - n / 2

    for n in n_vals:
        if time_left <= 0:
            break
        time_start = time.time()
        # Generate a diverse initial guess
        x = np.linspace(0, 2, n)
        h_init = np.zeros(n)
        for i in range(n):
            if i % 5 == 0:
                h_init[i] = 1.0
            elif i % 4 == 0:
                h_init[i] = 0.5
            else:
                h_init[i] = np.random.uniform(0, 1)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n / 2) / np.sum(h_init)
        
        bounds = Bounds(0, 1)
        cons = LinearConstraint(np.ones(n), lb=n/2, ub=n/2)
        
        # Use L-BFGS-B with bounds and constraint
        result = optimize.minimize(lambda x: evaluate(x, n), h_init, 
                                 method='L-BFGS-B', bounds=bounds, constraints=cons,
                                 tol=1e-8, options={'maxiter': 150})
        
        if result.success:
            h_opt = result.x
            c5 = evaluate(h_opt, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n = n
        time_used = time.time() - time_start
        time_left -= time_used

    # Add a refinement step with smaller n

                best_h = h_opt
                best_n = n_refine
        time_used = time.time() - time_start
        time_left -= time_used

    return best_h, best_c5, best_n
