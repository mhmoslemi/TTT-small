import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def fitness(h):
        c5 = evaluate(h)
        # Penalize deviations from the integral constraint
        penalty = 100 * np.abs(np.sum(h) - target_sum)
        return c5 + penalty
    
    population_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 100

    population = np.clip(population, 0, 1)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitnesses = np.array([fitness(ind) for ind in population])
        # Select parents
        parents_indices = np.argsort(fitnesses)[:int(population_size/2)]
        parents = population[parents_indices]
        # Create new population
        new_population = []
        for _ in range(population_size):
            if np.random.rand() < crossover_rate:
                # Crossover
                parent1 = np.random.choice(parents)
                parent2 = np.random.choice(parents)
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            else:

                child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = np.array(new_population)
        # Update best solution
        for ind in population:
            current_c5 = evaluate(ind)
            if current_c5 < best_c5:
                best_h = ind.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
