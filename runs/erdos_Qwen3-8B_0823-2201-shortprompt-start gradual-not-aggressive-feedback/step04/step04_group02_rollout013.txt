import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Start with initial solution
    best_h = initial_h.copy()
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Local optimization with L-BFGS-B
        result = opt.minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
