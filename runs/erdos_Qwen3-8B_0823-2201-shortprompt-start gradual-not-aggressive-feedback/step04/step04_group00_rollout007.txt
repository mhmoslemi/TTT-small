import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Start with initial guess
    h = initial_h.copy()
    h = h * target_sum / np.sum(h)
    
    def differential_evolution(population_size=20, generations=50):
        population = np.random.rand(population_size, n_points)
        population = np.clip(population, 0, 1)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        best_h = population[0]
        best_c5 = objective(best_h)
        
        for _ in range(generations):
            for i in range(population_size):
                a, b, c = np.random.choice(population_size, 3, replace=False)
                mutant = population[a] + 0.5 * (population[b] - population[c])
                mutant = np.clip(mutant, 0, 1)
                mutant = mutant * target_sum / np.sum(mutant)
                mutant_c5 = objective(mutant)
                
                if mutant_c5 < best_c5:
                    best_h = mutant
                    best_c5 = mutant_c5
                population[i] = mutant
        
        return best_h, best_c5
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h, best_c5 = differential_evolution()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Refinement loop
    while time.time() - start_time < budget_s:
        # Perturb and refine
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
