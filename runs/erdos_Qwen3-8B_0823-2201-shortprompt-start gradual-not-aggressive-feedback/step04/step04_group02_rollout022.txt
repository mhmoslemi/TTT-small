import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial solution
    best_h = np.array(initial_h)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Genetic algorithm stage
        population = np.random.uniform(0, 1, (20, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        fitness = np.array([objective(ind) for ind in population])
        best_idx = np.argmin(fitness)
        best_genetic = population[best_idx]
        
        # Combine with previous best
        combined = 0.5 * best_h + 0.5 * best_genetic
        combined = np.clip(combined, 0, 1)
        combined = combined * target_sum / np.sum(combined)
        
        # Gradient descent refinement
        result = opt.minimize(
            objective,
            combined,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
            print(f"Improved bound: {best_c5}")
    
    return best_h, best_c5, n_points
