    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial population
    population = [initial_h.copy() for _ in range(50)]
    for h in population:
        h = h * target_sum / np.sum(h)
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_best_c5 = fitness[best_idx]
        
        # Update best solution
        if current_best_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_best_c5
        
        # Generate new population
        new_population = []
        for _ in range(len(population) // 2):
            # Select parents
            parent1 = population[np.random.choice(len(population))]
            parent2 = population[np.random.choice(len(population))]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    child1[i] += np.random.uniform(-0.1, 0.1)
                    child2[i] += np.random.uniform(-0.1, 0.1)
            
            # Clip and normalize
            child1 = np.clip(child1, 0, 1) * target_sum / np.sum(child1)
            child2 = np.clip(child2, 0, 1) * target_sum / np.sum(child2)
            
            new_population.append(child1)
            new_population.append(child2)
        
        # Replace population
        population = new_population
    
    return best_h, best_c5, n_points
