import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_vals = [100, 120, 140, 160, 180, 200, 220, 240, 260, 280, 300]
    best_h = None
    best_c5 = 1.0
    best_n = 100
    time_left = budget_s

    for n in n_vals:
        if time_left <= 0:
            break
        time_start = time.time()
        # Create initial guess with random perturbation
        h_init = np.random.rand(n)
        h_init = h_init * (n / 2) / np.sum(h_init)
        
        bounds = Bounds(0, 1)
        cons = LinearConstraint(np.ones(n), lb=n/2, ub=n/2)
        
        # Use SLSQP with tighter tolerances
        result = optimize.minimize(lambda x: evaluate(x, n), h_init, 
                                 method='SLSQP', bounds=bounds, constraints=cons,
                                 tol=1e-8, options={'maxiter': 200})
        
        if result.success:
            h_opt = result.x
            c5 = evaluate(h_opt, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n = n
        time_used = time.time() - time_start
        time_left -= time_used

    # Add a refinement step with smaller n
    if time_left > 0:
        n_refine = 150
        time_start = time.time()
        x_refine = np.linspace(0, 2, n_refine)
        h_refine = np.interp(x_refine, np.linspace(0, 2, best_n), best_h)
        h_refine = np.clip(h_refine, 0, 1)
        h_refine = h_refine * (n_refine / 2) / np.sum(h_refine)
        
        bounds = Bounds(0, 1)
        cons = LinearConstraint(np.ones(n_refine), lb=n_refine/2, ub=n_refine/2)
        
        result = optimize.minimize(lambda x: evaluate(x, n_refine), h_refine, 
                                 method='SLSQP', bounds=bounds, constraints=cons,
                                 tol=1e-8, options={'maxiter': 100})
        
        if result.success:
            h_opt = result.x
            c5 = evaluate(h_opt, n_refine)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n = n_refine
        time_used = time.time() - time_start
        time_left -= time_used

    return best_h, best_c5, best_n

def evaluate(h, n):
    dx = 2.0 / n
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if total == 0:
        return np.inf
    h = h * (n / 2) / total
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr * dx)
