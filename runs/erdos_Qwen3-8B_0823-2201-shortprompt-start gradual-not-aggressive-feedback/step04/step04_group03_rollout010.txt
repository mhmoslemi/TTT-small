import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    elite_size = 5
    
    population = np.random.rand(population_size, n_points)
    population = (population * target_sum / np.sum(population, axis=1, keepdims=True)).flatten()
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate population
        fitness = np.array([objective(ind) for ind in population])
        
        # Select elite
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = population[elite_indices]
        
        # Tournament selection
        parents = []
        for _ in range(population_size - elite_size):
            tournament = np.random.choice(np.arange(population_size), size=5)
            winner = tournament[np.argmin(fitness[tournament])]
            parents.append(population[winner])
        
        # Crossover
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                parent1 = parents[i]
                parent2 = parents[i+1]
                child1 = np.concatenate([parent1[:n_points//2], parent2[n_points//2:]])
                child2 = np.concatenate([parent2[:n_points//2], parent1[n_points//2:]])
                offspring.extend([child1, child2])
        
        # Mutation
        for i in range(len(offspring)):
            offspring[i] = np.where(np.random.rand(n_points) < mutation_rate, 
                                   1 - offspring[i], offspring[i])
            offspring[i] = np.clip(offspring[i], 0, 1)
            offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
        
        # Combine population
        population = np.vstack([elite, offspring])
        population = np.unique(population, axis=0)
        
        # Update best solution
        current_best = population[np.argmin(fitness)]
        current_c5 = objective(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
            print(f"Improved bound: {best_c5}")
    
    return best_h, best_c5, n_points
