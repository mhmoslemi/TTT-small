import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 10

    # Convert to array for easier manipulation
    best_h = best_h.astype(float)
    
    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Create population
        population = [np.copy(best_h) for _ in range(population_size)]

            # Random perturbation
            perturb = np.random.uniform(-0.1, 0.1, n_points)
            population[i] = best_h + perturb
            population[i] = np.clip(population[i], 0, 1)
            population[i] = population[i] * target_sum / np.sum(population[i])
        
        # Evaluate population
        population_fitness = [objective(h) for h in population]
        
        # Select best individuals
        parents = [population[i] for i in np.argsort(population_fitness)[:int(population_size/2)]]
        
        # Crossover
        new_population = []
        for _ in range(population_size):
            if np.random.rand() < crossover_rate:
                parent1 = random.choice(parents)

                child = np.copy(parent1)

                        child[i] = parent2[i]
                new_population.append(child)
            else:
                new_population.append(random.choice(parents))
        
        # Mutation
        for i in range(len(new_population)):

                if np.random.rand() < mutation_rate:

        new_population_fitness = [objective(h) for h in new_population]
        
        # Select best solution
        best_index = np.argmin(new_population_fitness)
        best_h = new_population[best_index]
        best_c5 = new_population_fitness[best_index]

            best_h = best_h
            best_c5 = best_c5
    
    return best_h, best_c5, n_points
