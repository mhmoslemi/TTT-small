import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm parameters
    pop_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 50
    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Create population
        population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
        for i in range(pop_size):
            population[i] = population[i] * target_sum / np.sum(population[i])
        
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        
        # Select best individuals
        elite_indices = np.argsort(fitness)[:3]
        elite = [population[i] for i in elite_indices]
        
        # Crossover
        for i in range(pop_size):
            if random.random() < crossover_rate:
                parent1 = random.choice(elite)

                child = np.zeros(n_points)
