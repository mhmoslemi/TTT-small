import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        return c5

        return np.sum(h) - target_sum

    def refine_solution(h_initial):
        result = opt.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        return result.x

    best_h = np.array(initial_h)
    best_c5 = objective(best_h)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Local refinement
        refined_h = refine_solution(best_h)
        refined_c5 = objective(refined_h)
        if refined_c5 < best_c5:
            best_h = refined_h
