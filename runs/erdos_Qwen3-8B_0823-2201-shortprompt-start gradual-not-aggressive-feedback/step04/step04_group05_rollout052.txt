import scipy.optimize as optimize
import time

def run(seed=42, budget_s=20, **kwargs):

    def create_initial_h(n):

            if x[i] < 1:
                h[i] = 1 - (x[i] / 1)
            else:
                h[i] = x[i] - 1
        return h
    
    best_h = None

            h = np.clip(x, 0, 1)
            total = np.sum(h)
            if total == 0:
                return np.inf

            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr * (2.0 / n))

                return np.inf

        bounds = [(0, 1) for _ in range(n)]
        cons = {'type': 'eq', 'fun': constraint}
        
        # Use L-BFGS-B for faster convergence
