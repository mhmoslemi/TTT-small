        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)

        return c5

    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Use SLSQP with bounds and constraint
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
