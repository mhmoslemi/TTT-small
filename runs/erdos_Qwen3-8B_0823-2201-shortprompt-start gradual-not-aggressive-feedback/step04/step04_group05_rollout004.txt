        n = len(individual)
        h = individual.copy()
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.inf,
        h = h * (n / 2) / total
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * 2.0 / n),

    def mut_gaussian(individual, indpb):
        for i in range(len(individual)):
            if np.random.random() < indpb:
                individual[i] += np.random.normal(0, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        return individual,

    def cx_two_point(ind1, ind2):
        size = len(ind1)
        cx1, cx2 = np.copy(ind1), np.copy(ind2)
        p1, p2 = np.random.randint(0, size, 2)
        if p1 < p2:
            cx1[p1:p2] = ind2[p1:p2]
            cx2[p1:p2] = ind1[p1:p2]

        population = [np.random.uniform(0, 1, n) for _ in range(50)]

        toolbox = base.Toolbox()

            best_h = best_ind.copy()
