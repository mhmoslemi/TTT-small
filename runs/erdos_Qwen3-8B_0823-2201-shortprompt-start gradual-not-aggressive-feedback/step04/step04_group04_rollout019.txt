    generations = 10
    mutation_rate = 0.1

    population = np.random.rand(pop_size, n_points)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)

        fitness = np.array([objective(ind) for ind in population])

        parents_indices = np.argsort(fitness)[:int(pop_size/2)]
        parents = population[parents_indices]
        
        # Crossover
        offspring = []
        for _ in range(pop_size - len(parents)):
            parent1, parent2 = np.random.choice(parents, size=2, replace=False)
