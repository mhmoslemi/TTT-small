import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x

    def genetic_step(population):
        new_pop = []
        for h in population:
            perturb = np.random.normal(0, 0.05, n_points)
            perturbed = h + perturb
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            new_pop.append(perturbed)
        return new_pop

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    population = [best_h.copy() for _ in range(50)]
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        population = genetic_step(population)
        for h in population:
            h = local_refinement(h)
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
