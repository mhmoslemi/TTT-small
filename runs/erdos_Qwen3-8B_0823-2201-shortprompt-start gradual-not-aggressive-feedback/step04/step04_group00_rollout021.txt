import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    elite_size = 2
    generations = 50
    
    def genetic_algorithm():
        # Initialize population
        population = np.random.rand(population_size, n_points)
        population = np.clip(population, 0, 1)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            
            # Select elites
            elite_indices = np.argsort(fitness)[:elite_size]
            elites = population[elite_indices]
            
            # Create new population
            new_population = elites.copy()
            
            # Tournament selection and crossover
            for _ in range(population_size - elite_size):
                parents = np.random.choice(population_size, size=2, replace=False)
                parent1 = population[parents[0]]
                parent2 = population[parents[1]]
                
                # Crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                
                # Mutation
                mutation_mask = np.random.rand(n_points) < mutation_rate
                child[mutation_mask] = np.random.rand(np.sum(mutation_mask))
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
                
                new_population = np.vstack([new_population, child])
            
            population = new_population
            population = population[:population_size]
        
        best_h = population[np.argmin(objective(population))]
        best_c5 = objective(best_h)
        return best_h, best_c5
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Start with genetic algorithm
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()
    best_h = local_refinement(best_h)

    # Additional refinement within budget

        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,

            options={'maxiter': 50, 'ftol': 1e-8}

            best_c5 = current_c5
