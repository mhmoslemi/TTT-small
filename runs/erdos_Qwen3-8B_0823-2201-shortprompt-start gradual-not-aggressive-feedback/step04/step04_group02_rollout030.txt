import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize population
    population_size = 50
    population = [initial_h + np.random.uniform(-0.1, 0.1, n_points) for _ in range(population_size)]
    population = [np.clip(p, 0, 1) * target_sum / np.sum(np.clip(p, 0, 1)) for p in population]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        
        # Select parents using tournament selection
        parents = []
        for _ in range(population_size):
            candidates = random.sample(range(population_size), 5)
            best_idx = min(candidates, key=lambda x: fitness[x])
            parents.append(population[best_idx])
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if i + 1 < population_size:
                parent1 = parents[i]
                parent2 = parents[i+1]
                child1 = np.zeros(n_points)
                child2 = np.zeros(n_points)
                for j in range(n_points):
                    if random.random() < 0.5:
                        child1[j] = parent1[j]
                        child2[j] = parent2[j]
                    else:
                        child1[j] = parent2[j]
                        child2[j] = parent1[j]
                offspring.append(child1)
                offspring.append(child2)
        
        # Mutation
        mutated_offspring = []
        for ind in offspring:
            mutated = ind + np.random.uniform(-0.05, 0.05, n_points)
            mutated = np.clip(mutated, 0, 1) * target_sum / np.sum(np.clip(mutated, 0, 1))
            mutated_offspring.append(mutated)
        
        # Replace population
        population = mutated_offspring
        
        # Local optimization
        for ind in population:
            result = opt.minimize(
                objective,
                ind,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 20, 'ftol': 1e-8}
            )
            ind[:] = result.x
        
        # Update best solution
        for ind in population:
            current_c5 = objective(ind)
            if current_c5 < best_c5:
                best_h = ind.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
