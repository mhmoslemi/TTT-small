import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial population
    population_size = 20
    population = [np.copy(initial_h) + np.random.uniform(-0.1, 0.1, n_points) for _ in range(population_size)]
    population = [np.clip(p, 0, 1) * target_sum / np.sum(np.clip(p, 0, 1)) for p in population]
    
    best_h = np.copy(initial_h)
    best_c5 = objective(np.copy(initial_h))
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = [objective(p) for p in population]
        
        # Find the best candidate
        best_idx = np.argmin(fitness)
        current_c5 = fitness[best_idx]
        
        if current_c5 < best_c5:
            best_h = np.copy(population[best_idx])
            best_c5 = current_c5
            print(f"Improved bound: {best_c5}")
        
        # Generate new population
        new_population = []
        
        # Elite selection
        new_population.extend(population[:int(population_size/2)])
        
        # Crossover and mutation
        for _ in range(int(population_size/2)):
            parent1, parent2 = random.choices(population, k=2)
            child = (parent1 + parent2) / 2 + np.random.uniform(-0.05, 0.05, n_points)
            child = np.clip(child, 0, 1) * target_sum / np.sum(np.clip(child, 0, 1))
            new_population.append(child)
        
        population = new_population
        
        # Local optimization on best candidate
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
