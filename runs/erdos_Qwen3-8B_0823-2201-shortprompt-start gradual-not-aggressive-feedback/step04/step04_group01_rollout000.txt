import scipy.optimize as opt

import random

def run(seed=42, budget_s=20, **kwargs):

    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial solution
    best_h = initial_h.copy()
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    start_time = time.time()

    # Gradient-based optimization
    def optimize_h(h_initial):
        result = opt.minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        return result.x

    # Genetic algorithm for exploration
    def genetic_search(pop_size=50, generations=20):
        population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        best_h = population[0]
        best_c5 = objective(best_h)
        
        for _ in range(generations):
            new_population = []

                perturbed = perturbed * target_sum / np.sum(perturbed)
                new_population.append(perturbed)
            
            population = new_population

                current_c5 = objective(h)

                    best_h = h
                    best_c5 = current_c5

    # Initial genetic search
