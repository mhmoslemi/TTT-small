import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Genetic algorithm parameters
    pop_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 50
    
    # Initialize population
    def generate_population(size):
        pop = np.random.rand(size, n_points)
        pop = np.clip(pop, 0, 1)
        pop = pop * target_sum / np.sum(pop, axis=1, keepdims=True)
        return pop
    
    # Fitness function
    def fitness(h):
        return objective(h)
    
    # Selection
    def select_parents(pop, fitnesses):
        tournament = np.random.choice(len(pop), size=5, replace=False)
        best_idx = tournament[np.argmin(fitnesses[tournament])]
        return pop[best_idx].copy()
    
    # Crossover
    def crossover(parent1, parent2):
        if np.random.rand() < crossover_rate:
            point = np.random.randint(1, n_points - 1)
            child = np.concatenate((parent1[:point], parent2[point:]))
        else:
            child = parent1.copy()
        return child
    
    # Mutation
    def mutate(individual):
        individual = individual.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] += np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        return individual * target_sum / np.sum(individual)
    
    # Main genetic algorithm loop
    population = generate_population(pop_size)
    best_h = population[0]
    best_c5 = fitness(best_h)
    
    start_time = time.time()
    
    for _ in range(generations):
        fitnesses = np.array([fitness(ind) for ind in population])
        parents = [select_parents(population, fitnesses) for _ in range(pop_size)]
        offspring = []
        
        for i in range(pop_size):
            parent1 = parents[i]
            parent2 = parents[np.random.choice(range(pop_size), replace=False, size=1)[0]]
            child = crossover(parent1, parent2)
            child = mutate(child)
            offspring.append(child)
        
        population = np.array(offspring)
        population = np.array([np.clip(ind, 0, 1) * target_sum / np.sum(np.clip(ind, 0, 1)) for ind in population])
        best_h = population[np.argmin(fitnesses)]
        best_c5 = fitness(best_h)

            break
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )

        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
