import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with a coarser grid
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initial population
    population = np.random.rand(n_points, 100)
    population = np.clip(population, 0, 1)
    population = population * target_sum / np.sum(population, axis=0, keepdims=True)
    
    best_h = population[:, 0]
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = np.array([evaluate(ind) for ind in population])
        
        # Select parents
        parents = population[np.argsort(fitness)[:int(0.2 * population.shape[1])]]
        
        # Crossover and mutation
        new_population = []
        for _ in range(population.shape[1]):
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            child = np.clip(0.5 * parent1 + 0.5 * parent2 + np.random.normal(0, 0.05, n_points), 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        population = np.vstack([population, np.array(new_population)])
        population = population[np.argsort(fitness)[:population.shape[1]]]
        
        # Update best solution
        current_best = population[:, 0]
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
