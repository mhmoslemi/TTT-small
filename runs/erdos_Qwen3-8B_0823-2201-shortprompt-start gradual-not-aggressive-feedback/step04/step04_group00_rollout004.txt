import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def generate_population(size):
        pop = np.random.rand(size, n_points)
        pop = np.clip(pop, 0, 1)
        pop = pop * target_sum / np.sum(pop, axis=1, keepdims=True)
        return pop
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parent2[i]
        child = np.clip(child, 0, 1)
        child = child * target_sum / np.sum(child)
        return child
    
    def mutate(individual):
        mutated = np.copy(individual)
        for i in range(n_points):
            if np.random.rand() < 0.1:
                mutated[i] += np.random.uniform(-0.1, 0.1)
                mutated[i] = np.clip(mutated[i], 0, 1)
        mutated = mutated * target_sum / np.sum(mutated)
        return mutated
    
    population = generate_population(20)
    best_h = population[0]
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for i in range(len(population)):
            a, b = np.random.choice(len(population), 2, replace=False)
            child = crossover(population[a], population[b])
            child = mutate(child)
            new_population.append(child)
        population = new_population
    
    best_h = min(population, key=objective)
    best_c5 = objective(best_h)
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
