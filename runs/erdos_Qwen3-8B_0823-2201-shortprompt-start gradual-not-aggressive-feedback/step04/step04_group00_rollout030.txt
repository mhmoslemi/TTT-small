import scipy.optimize as opt
import time

    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Genetic algorithm for global search
    def genetic_search(population_size=50, generations=50):
        population = np.random.rand(population_size, n_points)
        population = np.clip(population, 0, 1)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            fitness = np.array([objective(ind) for ind in population])
            best_idx = np.argmin(fitness)
            best_h = population[best_idx]
            best_c5 = fitness[best_idx]
            
            # Selection and crossover
            selected = np.random.choice(population_size, size=population_size, replace=False)
            parents = population[selected]
            
            # Mutation
            mutants = np.array([np.clip(p + 0.1 * np.random.randn(n_points), 0, 1) for p in parents])
            mutants = mutants * target_sum / np.sum(mutants, axis=1, keepdims=True)
            
            # Replacement
            population = np.array([p for p in mutants])
        
        return best_h, best_c5
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Main optimization loop
    start_time = time.time()
    best_h, best_c5 = genetic_search()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:

        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,

            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
