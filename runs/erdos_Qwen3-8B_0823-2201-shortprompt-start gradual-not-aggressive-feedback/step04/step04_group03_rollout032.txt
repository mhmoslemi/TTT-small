import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def evaluate(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    population_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 50
    
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = (population * target_sum / np.sum(population, axis=1, keepdims=True))
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate population
        fitness = np.array([evaluate(ind) for ind in population])
        
        # Tournament selection
        tournament = np.random.choice(population_size, size=population_size, replace=False)
        selected = population[tournament[np.argsort(fitness[tournament])[:population_size//2]]]
        
        # Crossover
        offspring = []
        for i in range(population_size//2):
            if np.random.rand() < crossover_rate:
                parent1 = selected[i]
                parent2 = selected[i+1]
                child1 = np.concatenate((parent1[:n_points//2], parent2[n_points//2:]))
                child2 = np.concatenate((parent2[:n_points//2], parent1[n_points//2:]))
                offspring.append(child1)
                offspring.append(child2)
            else:
                offspring.append(selected[i])
                offspring.append(selected[i+1])
        
        # Mutation
        for i in range(population_size):
            if np.random.rand() < mutation_rate:
                mutation = np.random.uniform(-0.1, 0.1, n_points)
                offspring[i] += mutation
                offspring[i] = np.clip(offspring[i], 0, 1)
                offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
        
        # Replace population
        population = np.array(offspring)
        population = (population * target_sum / np.sum(population, axis=1, keepdims=True))
        
        # Update best solution
        for ind in population:
            current_c5 = evaluate(ind)
            if current_c5 < best_c5:
                best_h = ind.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
