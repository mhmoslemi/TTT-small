import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize with existing solution
    h = initial_h.copy()
    h = h * target_sum / np.sum(h)
    
    # First refinement with SLSQP
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Genetic algorithm for exploration
    def genetic_algorithm(population_size=50, generations=100, mutation_rate=0.05):
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        best_h = population[0]
        best_c5 = objective(best_h)
        
        for _ in range(generations):
            new_population = []
            for h in population:
                # Mutation
                perturb = np.random.normal(0, mutation_rate, n_points)
                perturbed = h + perturb
                perturbed = np.clip(perturbed, 0, 1)
                perturbed = perturbed * target_sum / np.sum(perturbed)
                new_population.append(perturbed)
            
            population = new_population
            for h in population:
                current_c5 = objective(h)
                if current_c5 < best_c5:
                    best_h = h
                    best_c5 = current_c5
        return best_h, best_c5
    
    best_h, best_c5 = genetic_algorithm()
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Local search with SLSQP
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
