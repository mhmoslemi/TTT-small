        fitness = np.array([objective(ind) for ind in population])
        
        # Selection
        selected_indices = np.argsort(fitness)[:int(population_size/2)]
        selected = population[selected_indices]
        
        # Crossover
        offspring = []
        for _ in range(population_size - len(selected)):
            parent1, parent2 = np.random.choice(selected, size=2, replace=False)
            child = np.random.rand(n_points)
            child = child * parent1 + (1 - child) * parent2
            child = child * target_sum / np.sum(child)
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):

        # Create new population
        population = np.vstack((selected, offspring))
        
        # Update best solution
        for ind in population:
