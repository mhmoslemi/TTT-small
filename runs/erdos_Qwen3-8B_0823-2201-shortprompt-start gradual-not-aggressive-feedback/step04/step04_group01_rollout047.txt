import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def fine_tune(h_initial):
        result = opt.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-4}
        )
        return result.x

    def local_search(h):
        best_h = h.copy()
        best_c5 = objective(best_h)
        for _ in range(50):
            perturb = np.random.normal(0, 0.02, n_points)
            perturbed = h + perturb
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            current_c5 = objective(perturbed)
            if current_c5 < best_c5:
                best_h = perturbed.copy()
                best_c5 = current_c5
        return best_h

    # Initial fine-tuning
    best_h = fine_tune(initial_h)
    best_c5 = objective(best_h)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Local search
        best_h = local_search(best_h)
        best_c5 = objective(best_h)
        
        # Global optimization
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8, 'eps': 1e-4}
        )
        best_h = result.x
        best_c5 = objective(best_h)
        
        # Random perturbation
        perturb = np.random.normal(0, 0.01, n_points)
        perturbed = best_h + perturb

        current_c5 = objective(perturbed)
        if current_c5 < best_c5:
            best_h = perturbed.copy()
            best_c5 = current_c5

    return best_h, best_c5, n_points
