def run(seed=42, budget_s=20, **kwargs):

    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    bounds = [(0, 1) for _ in range(n_points)]

    def evolve_population(population, generations=100):
        for _ in range(generations):
            # Evaluate fitness
            fitness = [objective(ind) for ind in population]
            # Select parents
            parents = np.random.choice(population, size=len(population), p=1/np.array(fitness))
            # Crossover
            offspring = []
            for _ in range(len(population)//2):
                parent1, parent2 = random.sample(parents, 2)
                child = np.zeros_like(parent1)
                for i in range(len(child)):
                    if np.random.rand() < 0.5:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
                offspring.append(child)
            # Mutate
            for i in range(len(offspring)):
                for j in range(len(offspring[i])):
                    if np.random.rand() < 0.1:
                        offspring[i][j] = np.clip(offspring[i][j] + np.random.uniform(-0.1, 0.1), 0, 1)
            # Replace population
            population = np.concatenate((population, offspring))
            population = np.random.choice(population, size=len(population), replace=False)
        return population

    population = [initial_h.copy() for _ in range(50)]

        population = evolve_population(population, generations=10)

            if current_c5 < best_c5:
                best_h = ind
                best_c5 = current_c5
                print(f"Improved bound: {best_c5}")
    
    return best_h, best_c5, n_points
