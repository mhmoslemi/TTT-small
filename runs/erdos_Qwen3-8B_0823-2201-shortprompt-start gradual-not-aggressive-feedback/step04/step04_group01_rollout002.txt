import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial solution
    h = initial_h.copy()
    h = h * target_sum / np.sum(h)
    best_h = h
    best_c5 = objective(h)

    # Genetic algorithm with local optimization
    def genetic_step(population, elites=5):
        new_population = []
        for h in population:
            # Local optimization
            result = opt.minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 20, 'ftol': 1e-8}
            )
            new_population.append(result.x)
        
        # Keep elites and add random mutations
        new_population += [np.random.uniform(0, 1, n_points) for _ in range(elites)]
        new_population = [h * target_sum / np.sum(h) for h in new_population]
        return np.unique(new_population, axis=0)[:50]

    # Main loop
    population = [h * target_sum / np.sum(h) for _ in range(50)]
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        population = genetic_step(population)
        for h in population:
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
