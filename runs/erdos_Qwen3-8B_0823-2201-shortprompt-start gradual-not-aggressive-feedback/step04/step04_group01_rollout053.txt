import random

        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    def genetic_search(pop_size=50, generations=100, mutation_rate=0.05):
        population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        best_h = population[0]
        best_c5 = objective(best_h)
        
        for _ in range(generations):

            population = new_population

    best_h, best_c5 = genetic_search()
