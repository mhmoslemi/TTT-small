import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial guess
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        # Ensure bounds and normalize
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h_vec):
        h = h_vec.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Multi-stage optimization with adaptive strategies
    for stage in range(3):
        if stage == 0:
            # Stage 1: Global exploration with adaptive perturbations
            for _ in range(20):
                if time.time() - start_time >= budget_s:
                    break
                # Adaptive perturbation with decreasing magnitude
                perturb = np.random.uniform(-0.15 * (0.95 ** _), 0.15 * (0.95 ** _), n_points)
                perturbed_h = h + perturb
                perturbed_h = np.clip(perturbed_h, 0, 1)
                perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
                
                result = opt.minimize(
                    objective,
                    perturbed_h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
                )
                
                current_h = result.x
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * target_sum / np.sum(current_h)
                current_c5 = objective(current_h)
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    h = current_h
        elif stage == 1:
            # Stage 2: Refinement with structured perturbations
            for _ in range(15):
                if time.time() - start_time >= budget_s:
                    break

                perturbed_h = h + perturb
                perturbed_h = np.clip(perturbed_h, 0, 1)
                perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
                
                result = opt.minimize(
                    objective,
                    perturbed_h,
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-9}
                )
                
                current_h = result.x
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * target_sum / np.sum(current_h)
                current_c5 = objective(current_h)
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    h = current_h
        else:
            # Stage 3: Fine-tuning with gradient descent
            for _ in range(10):
                if time.time() - start_time >= budget_s:
                    break
                result = opt.minimize(
                    objective,
                    h,
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-9}

                    best_c5 = current_c5

    return best_h, best_c5, n_points
