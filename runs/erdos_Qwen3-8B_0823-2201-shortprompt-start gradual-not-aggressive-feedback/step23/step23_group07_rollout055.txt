import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Harmonic-based initialization
    def harmonic_init(n_points):
        h = np.zeros(n_points)
        freq = 5
        phase = np.pi / 4
        for i in range(n_points):
            h[i] = 0.5 + 0.5 * np.sin(2 * np.pi * freq * i / n_points + phase)
        h = h * target_sum / np.sum(h)
        return h
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = objective(h)
    
    # Alternative harmonic initialization
    alt_h = harmonic_init(n_points)
    alt_c5 = objective(alt_h)
    if alt_c5 < best_c5:
        best_h = alt_h.copy()
        best_c5 = alt_c5
    
    start_time = time.time()
    
    # Structured optimization with adaptive learning
    for stage in range(3):
        # Coordinate descent with adaptive learning
        def local_optimize(h):
            result = opt.minimize(
                objective,
                h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-9}
            )
            return result.x
        
        h = local_optimize(h)
        current_c5 = objective(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Adaptive perturbations with structured patterns
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break
        
        for perturb_size in [0.05, 0.02, 0.01, 0.005, 0.002]:
            if time.time() - start_time >= budget_s:
                break
            
            # Structured perturbation using sine wave
            perturb = np.random.normal(0, perturb_size, n_points)
            perturb = np.clip(perturb, -0.1, 0.1)
            perturbed_h = h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            
            result = opt.minimize(
                objective,
                perturbed_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 150, 'ftol': 1e-10, 'gtol': 1e-9}
            )

            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = objective(current_h)
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
