import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Smaller n_points for faster optimization
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial construction
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < 25:
            h[i] = 1.0
        elif 25 <= i < 50:
            h[i] = 0.5
        else:
            h[i] = 0.0
    h = h * target_sum / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        # Normalize to maintain integral constraint
        h = h * target_sum / np.sum(h)
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(h)
    start_time = time.time()
    
    # Multi-stage refinement with decreasing perturbation scales
    for stage in range(3):
        # Gradient-based refinement
        result = opt.minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 150, 'ftol': 1e-12, 'gtol': 1e-9}
        )
        best_h = result.x
        best_c5 = objective(best_h)
        
        # Adaptive perturbation with structured magnitude
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break
        
        perturb_sizes = [0.05, 0.025, 0.01, 0.005, 0.0025]
        for ps in perturb_sizes:
            if time.time() - start_time >= budget_s:
                break
            perturb = np.random.uniform(-ps, ps, n_points)
            perturbed_h = best_h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            
            result = opt.minimize(
                objective,
                perturbed_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-9}
            )
            
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = objective(current_h)
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
