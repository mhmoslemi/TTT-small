import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)

    start_time = time.time()

    for stage in range(3):

            h,
            method='L-BFGS-B',
            bounds=bounds,

            options={'maxiter': 150, 'ftol': 1e-10, 'gtol': 1e-9}
        )
