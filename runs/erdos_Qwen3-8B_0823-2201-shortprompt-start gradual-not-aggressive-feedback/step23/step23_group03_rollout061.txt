import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Multi-stage optimization with adaptive perturbation
    for stage in range(3):
        # Stage 1: Global exploration with random perturbations
        if stage == 0:
            for _ in range(20):
                if time.time() - start_time >= budget_s:
                    break
                # Apply informed perturbation based on current solution
                perturb = np.random.normal(0, 0.15, n_points)
                perturbed_h = h + perturb
                perturbed_h = np.clip(perturbed_h, 0, 1)
                perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
                
                result = opt.minimize(
                    objective,
                    perturbed_h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-9}
                )
                
                current_h = result.x
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * target_sum / np.sum(current_h)
                current_c5 = objective(current_h)
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    h = current_h
        # Stage 2: Refinement with simulated annealing
        elif stage == 1:
            temp = 1.0
            for _ in range(10):
                if time.time() - start_time >= budget_s:
                    break
                # Simulated annealing with adaptive temperature
                perturb = np.random.uniform(-temp, temp, n_points)
                perturbed_h = h + perturb
                perturbed_h = np.clip(perturbed_h, 0, 1)
                perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
                
                result = opt.minimize(
                    objective,
                    perturbed_h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-9}
                )
                
                current_h = result.x
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * target_sum / np.sum(current_h)
                current_c5 = objective(current_h)
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    h = current_h
                temp *= 0.95
        # Stage 3: Fine-tuning with gradient descent
        else:
            for _ in range(10):
                if time.time() - start_time >= budget_s:
                    break
                result = opt.minimize(
                    objective,
                    h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-9}
                )
                
                current_h = result.x
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * target_sum / np.sum(current_h)
                current_c5 = objective(current_h)
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    h = current_h
    
    return best_h, best_c5, n_points
