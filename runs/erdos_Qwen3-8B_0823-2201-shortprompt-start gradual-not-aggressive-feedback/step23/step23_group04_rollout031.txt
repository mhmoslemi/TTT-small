import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Use multiple initial configurations
    initial_configs = [initial_h_values]
    for _ in range(2):
        perturb = np.random.normal(0, 0.1, n_points)
        perturb = np.clip(perturb, -0.2, 0.2)
        new_config = np.array(initial_h_values, dtype=np.float64) + perturb
        new_config = np.clip(new_config, 0, 1)
        new_config = new_config * target_sum / np.sum(new_config)
        initial_configs.append(new_config)
    
    for config in initial_configs:
        h = np.array(config, dtype=np.float64)
        h = h * target_sum / np.sum(h)
        
        # Gradient-based refinement
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-9}
        )
        h = result.x
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = objective(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Adaptive perturbation targeting high-overlap regions
        for perturb_size in [0.05, 0.02, 0.01, 0.005, 0.002]:
            if time.time() - start_time >= budget_s:
                break
            # Structured perturbation: focus on high-overlap regions
            corr = np.correlate(h, 1 - h, mode="full")
            overlap_indices = np.where(corr > best_c5 * 1.1)[0]
            perturb = np.zeros(n_points)
            if len(overlap_indices) > 0:
                perturb[overlap_indices] = np.random.normal(0, perturb_size, len(overlap_indices))
            perturb = np.clip(perturb, -0.1, 0.1)
            perturbed_h = h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            
            result = opt.minimize(
                objective,
                perturbed_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-10, 'gtol': 1e-9}
            )
            
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = objective(current_h)
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
