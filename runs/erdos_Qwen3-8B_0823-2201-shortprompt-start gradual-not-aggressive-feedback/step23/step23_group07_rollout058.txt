import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

        h = np.zeros(n_points)

            h[i] = 0.5 + 0.5 * np.sin(2 * np.pi * freq * i / n_points + phase)
        h = h * target_sum / np.sum(h)
        return h
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)

    best_h = h.copy()

    alt_c5 = objective(alt_h)

        best_h = alt_h.copy()
        best_c5 = alt_c5
