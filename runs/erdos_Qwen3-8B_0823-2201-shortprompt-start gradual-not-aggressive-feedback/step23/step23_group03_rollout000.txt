import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Hybrid optimization: differential evolution + L-BFGS-B
    # First phase: differential evolution for global exploration
    def de_objective(h_flat):
        h = h_flat.reshape(n_points)
        return objective(h)
    
    de_bounds = [(0, 1) for _ in range(n_points)]
    
    result_de = opt.differential_evolution(
        de_objective,
        de_bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-8,
        maxiter=50,
        disp=False,
        polish=True,
        polish_options={
            'method': 'L-BFGS-B',
            'bounds': bounds,
            'constraints': [{'type': 'eq', 'fun': constraint_sum}],
            'options': {'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-9}
        }
    )
    
    best_h = result_de.x
    best_c5 = objective(best_h)
    
    # Second phase: local refinement with L-BFGS-B
    for _ in range(20):
        if time.time() - start_time >= budget_s:
            break
        result = opt.minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-9}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
