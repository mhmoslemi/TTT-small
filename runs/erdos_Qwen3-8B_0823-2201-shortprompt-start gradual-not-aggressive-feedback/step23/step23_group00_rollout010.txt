import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    
    # Start with a more even initial guess
    n_points = 47
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Create an even initial guess
    h = np.linspace(0, 1, n_points)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h_vec):
        h = h_vec.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(h)
    start_time = time.time()
    
    # Multi-phase optimization with adaptive refinement
    for phase in range(3):
        # Local refinement with decreasing learning rate
        learning_rate = 0.1 if phase == 0 else 0.05 if phase == 1 else 0.02
        
        def local_refinement(h):
            result = opt.minimize(
                objective,
                h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-9, 'learning_rate': learning_rate}
            )
            return result.x
        
        h = local_refinement(h)
        current_c5 = objective(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Adaptive perturbation with decreasing magnitude
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break
        
        # Apply structured perturbations
        for perturb_size in [0.05, 0.025, 0.0125, 0.00625, 0.003125]:
            if time.time() - start_time >= budget_s:
                break
            perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
            perturbed_h = h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            
            result = opt.minimize(
                objective,
                perturbed_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 150, 'ftol': 1e-12, 'gtol': 1e-9}
            )
            
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = objective(current_h)
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
