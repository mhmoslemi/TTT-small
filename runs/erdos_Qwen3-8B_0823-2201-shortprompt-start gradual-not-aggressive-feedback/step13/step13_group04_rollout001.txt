import scipy.optimize as opt
import time

    n_points = len(initial_h)

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Phase 1: Simulated annealing for global exploration
    temp = 1.0
    cooling_rate = 0.995
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break

        perturb = np.random.uniform(-0.1, 0.1, n_points)

        current_c5 = objective(perturbed_h)
