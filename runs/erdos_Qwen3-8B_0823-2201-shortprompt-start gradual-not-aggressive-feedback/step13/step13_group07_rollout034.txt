import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    # Use a different starting point with 100 samples
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Create a better initial guess with alternating patterns
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            initial_h[i] = 1.0

            initial_h[i] = 0.5
        else:
            initial_h[i] = 0.0
    initial_h = initial_h * target_sum / np.sum(initial_h)

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum
