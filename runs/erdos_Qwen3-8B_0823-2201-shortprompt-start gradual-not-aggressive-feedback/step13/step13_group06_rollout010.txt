import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def evaluate(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    
    # Genetic algorithm parameters
    population_size = 20
    crossover_rate = 0.8
    mutation_rate = 0.1
    generations = 50
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Generate population
        population = [h.copy() for _ in range(population_size)]
        for i in range(population_size):
            if np.random.rand() < mutation_rate:
                population[i] = np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1)
        
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        
        # Select best individuals
        sorted_indices = np.argsort(fitness)
        selected = [population[i] for i in sorted_indices[:int(population_size/2)]]
        
        # Crossover
        new_population = []
        for i in range(0, population_size, 2):
            if i+1 < population_size:
                parent1 = selected[i]
                parent2 = selected[i+1]
                if np.random.rand() < crossover_rate:
                    split = np.random.randint(1, n_points-1)
                    child1 = np.concatenate((parent1[:split], parent2[split:]))
                    child2 = np.concatenate((parent2[:split], parent1[split:]))
                else:
                    child1, child2 = parent1.copy(), parent2.copy()
                new_population.extend([child1, child2])
            else:
                new_population.append(selected[i])
        
        # Mutation
        for i in range(len(new_population)):
            ind = new_population[i]
            if np.random.rand() < mutation_rate:
                ind = np.clip(ind + np.random.normal(0, 0.05, n_points), 0, 1)
            new_population[i] = ind
        
        # Update population
        population = new_population
        
        # Local optimization
        for ind in population:
            result = opt.minimize(
                evaluate,
                ind,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-10, 'gtol': 1e-10}
            )
            ind[:] = result.x
        
        # Update best solution
        for ind in population:
            current_c5 = evaluate(ind)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = ind.copy()
    
    return best_h, best_c5, n_points
