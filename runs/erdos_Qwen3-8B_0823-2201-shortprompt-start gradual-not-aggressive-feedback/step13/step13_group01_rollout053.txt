import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize with refined version of initial construction
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Hybrid optimization strategy with simulated annealing and local refinement
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    # Phase 1: Simulated annealing with decreasing temperature
    temperature = 1.0
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Generate perturbation
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        current_c5 = objective(perturbed_h)
        
        # Acceptance probability
        delta = current_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
            best_h = perturbed_h
            best_c5 = current_c5
        
        # Cool down
        temperature *= 0.995
    
    # Phase 2: Local refinement with SLSQP
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
