import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Genetic algorithm with adaptive mutation
    for gen in range(30):
        if time.time() - start_time > budget_s:
            break
        
        # Generate offspring through mutation
        offspring = []
        for _ in range(50):
            perturb = np.random.uniform(-0.1, 0.1, n_points)
            child = h + perturb
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            offspring.append(child)
        
        # Evaluate offspring
        offspring_c5 = [objective(child) for child in offspring]
        best_offspring = min(zip(offspring_c5, offspring), key=lambda x: x[0])[1]
        
        # Update best solution
        if objective(best_offspring) < best_c5:
            best_h = best_offspring.copy()
            best_c5 = objective(best_offspring)
            h = best_offspring
        
        # Gradient refinement
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8}
        )
        h = result.x
    
    return best_h, best_c5, n_points
