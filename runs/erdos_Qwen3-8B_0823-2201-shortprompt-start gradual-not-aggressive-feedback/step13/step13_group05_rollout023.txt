import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use 100 samples for better accuracy
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with a uniform distribution
    h = np.full(n_points, 1.0 / n_points) * target_sum
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):

        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]

    for stage in range(3):
        # Gradient-based refinement
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        h = result.x

            perturbed_h = h + perturb

            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
