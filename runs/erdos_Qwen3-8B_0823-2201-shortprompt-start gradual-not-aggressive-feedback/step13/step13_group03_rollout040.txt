import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    h = np.array(initial_h, dtype=np.float64)

    start_time = time.time()
    best_h = h.copy()

    # Parameters for genetic algorithm
    population_size = 10
    generations = 50
    mutation_rate = 0.1
    
    for gen in range(generations):

        population = [np.array(initial_h, dtype=np.float64) * target_sum / np.sum(initial_h) for _ in range(population_size)]

        fitness = [objective(ind) for ind in population]
        
        # Select best solution
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # If time is up, return best solution
        if time.time() - start_time > budget_s:
            break
        
        # Apply local optimization to best solution
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},

        )

            new_population.append(parent)
