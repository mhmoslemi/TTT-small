import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def objective(h):
        return evaluate(h)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def genetic_optimize():
        population = np.random.rand(n_points, 100)
        population = population * target_sum / np.sum(population, axis=0, keepdims=True)
        for _ in range(200):
            fitness = np.array([evaluate(ind) for ind in population])
            best_idx = np.argmin(fitness)
            best_h = population[best_idx]
            yield best_h, fitness[best_idx]
            population = np.random.rand(n_points, 100)
            population = population * target_sum / np.sum(population, axis=0, keepdims=True)
    
    start_time = time.time()
    best_h = initial_h * target_sum / np.sum(initial_h)
    best_c5 = evaluate(best_h)
    
    for h, c5 in genetic_optimize():
        if time.time() - start_time > budget_s:
            break
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    result = opt.minimize(
        objective,
        best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = evaluate(best_h)
    return best_h, best_c5, n_points
