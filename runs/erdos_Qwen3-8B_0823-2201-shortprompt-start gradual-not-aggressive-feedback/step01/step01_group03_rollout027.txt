import numpy as np
import scipy.optimize as optimize
from scipy.signal import correlate

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    conv = correlate(h, 1 - h, mode="full")
    c5 = np.max(conv) * dx
    return c5

def initial_h_values():
    return np.concatenate([np.ones(21), np.zeros(21)])  # 42 samples

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Efficient number of samples
    dx = 2.0 / n_points
    
    # Initial population
    population_size = 50
    population = []
    for _ in range(population_size):
        h = initial_h_values()
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        # Ensure integral constraint
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    def fitness(h):
        return evaluate_erdos_solution(h)
    
    # Genetic algorithm parameters
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 100
    
    for gen in range(generations):
        # Evaluate fitness
        fitness_scores = [fitness(ind) for ind in population]
        best_idx = np.argmin(fitness_scores)
        best_fitness = fitness_scores[best_idx]
        
        # Selection
        population = np.array(population)
        weighted_indices = np.random.choice(len(population), size=len(population), p=np.clip(fitness_scores / np.sum(fitness_scores), 0, 1))
        selected = population[weighted_indices]
        
        # Crossover
        new_population = []
        for i in range(len(selected) // 2):
            parent1 = selected[i]
            parent2 = selected[len(selected) // 2 + i]
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:split], parent2[split:]])
                child2 = np.concatenate([parent2[:split], parent1[split:]])
                new_population.extend([child1, child2])
            else:
                new_population.extend([parent1, parent2])
        
        # Mutation
        for i in range(len(new_population)):
            h = new_population[i]
            if np.random.rand() < mutation_rate:

                # Enforce integral constraint
                h = h * (n_points / 2) / np.sum(h)
            new_population[i] = h
        
        # Replace population
        population = new_population
        if gen % 10 == 0:
            print(f"Generation {gen}: Best C5 = {best_fitness:.6f}")
        
        # Early stopping
        if best_fitness < 0.380876:
            break
    
    # Return best solution
    best_idx = np.argmin([fitness(ind) for ind in population])
    h_values = population[best_idx]
    c5_bound = evaluate_erdos_solution(h_values)
    return h_values, c5_bound, n_points
