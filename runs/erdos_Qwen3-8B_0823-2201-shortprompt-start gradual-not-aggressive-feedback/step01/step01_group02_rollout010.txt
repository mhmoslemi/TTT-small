import numpy as np
import scipy.signal as signal
import random
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        norm = np.sum(h)
        h = h * target_sum / norm
        corr = signal.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def create_individual():
        h = np.random.rand(n_points)
        norm = np.sum(h)
        return h * target_sum / norm
    
    def crossover(parent1, parent2):
        alpha = np.random.rand(n_points)
        return alpha * parent1 + (1 - alpha) * parent2
    
    def mutate(individual, mutation_rate=0.05):
        h = individual.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] = np.random.rand()
        norm = np.sum(h)
        return h * target_sum / norm
    
    population_size = 50
    population = [create_individual() for _ in range(population_size)]
    best_solution = min(population, key=evaluate)
    best_c5 = evaluate(best_solution)
    start_time = time.time()
    
    for generation in range(1000):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate(ind) for ind in population]
        population = [population[i] for i in np.argsort(fitness)]
        new_population = []
        for _ in range(population_size // 2):
            parent1 = population[np.random.randint(len(population))]
            parent2 = population[np.random.randint(len(population))]
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        population = new_population + population[:population_size // 2]
        current_best = min(population, key=evaluate)
        if evaluate(current_best) < best_c5:
            best_solution = current_best
            best_c5 = evaluate(current_best)
    
    return best_solution, best_c5, n_points
