import scipy.optimize as optimize
from scipy.signal import correlate

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    conv = correlate(h, 1 - h, mode="full")
    c5 = np.max(conv) * dx
    return c5
