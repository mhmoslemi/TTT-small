import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import minimize
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Local optimization
    def objective_local(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    result_local = minimize(
        objective_local,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    best_h_local = result_local.x
    best_h_local = np.clip(best_h_local, 0, 1)
    best_h_local = best_h_local * target_sum / np.sum(best_h_local)
    best_c5_local = objective_local(best_h_local)
    
    # Genetic algorithm setup
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)

   toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0, 1)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)

    def eval_func(individual):
        h = np.array(individual)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5,

    toolbox.register("evaluate", eval_func)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)

    # Run genetic algorithm
    start_time = time.time()
    pop = toolbox.population(n=50)
    fitnesses = list(map(toolbox.evaluate, pop))
    for ind, fit in zip(pop, fitnesses):
        ind.fitness.values = fit
    for gen in range(100):

            break

        for child, fit in zip(offspring, fits):
            child.fitness.values = fit
        pop = toolbox.select(pop + offspring, len(pop))
    
    best_h_genetic = min(pop, key=lambda ind: ind.fitness.values[0]).phenotype

    best_c5_genetic = np.min([ind.fitness.values[0] for ind in pop])
    
    # Final selection
    best_h = np.minimum(best_h_local, best_h_genetic)
    best_c5 = np.min([best_c5_local, best_c5_genetic])
    return best_h, best_c5, n_points
