import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
from scipy.signal import correlate
from genetic_algorithm import GeneticAlgorithm

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    conv = correlate(h, 1 - h, mode="full")
    c5 = np.max(conv) * dx
    return c5

def initial_h_values():
    return np.concatenate([np.ones(21), np.zeros(21)])  # 42 samples

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Efficient number of samples
    dx = 2.0 / n_points
    initial_h = initial_h_values()
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)

    def objective(h):
        h = h.reshape(-1)
        c5 = evaluate_erdos_solution(h)
        return c5

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Local optimization with L-BFGS-B
    bounds = Bounds(0, 1)
    result = minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 100, 'ftol': 1e-9, 'gtol': 1e-9}
    )
    h_values = result.x
    c5_bound = evaluate_erdos_solution(h_values)
    
    # Global optimization with genetic algorithm
    ga = GeneticAlgorithm(
        population_size=50,
        mutation_rate=0.1,
        crossover_rate=0.8,
        generations=20,
        objective=objective,
        bounds=bounds,
        constraint_sum=constraint_sum,
        initial_population=h_values
    )
    best_h = ga.optimize()
    c5_bound = evaluate_erdos_solution(best_h)
    
    return best_h, c5_bound, n_points
