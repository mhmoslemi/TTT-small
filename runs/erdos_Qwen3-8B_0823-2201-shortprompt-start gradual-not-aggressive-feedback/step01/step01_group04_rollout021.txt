import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup with n_points = 500 (as per initial_h_values)
    n_points = 500
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy() if hasattr(initial_h_values, '__array__') else np.array(initial_h_values)
    initial_h = np.resize(initial_h, (n_points,))[:n_points]
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    h = cp.Variable(n_points)
    constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2]
    
    def evaluate(h_values):
        h_array = np.clip(h_values.value, 0, 1)
        h_array = h_array * (n_points / 2) / np.sum(h_array)
        corr = correlate(h_array, 1 - h_array, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    # Optimization loop
    while time.time() - start_time < budget_s:
        # Objective function to minimize: max overlap integral
        # We use a convex relaxation by maximizing over k implicitly
        # and minimizing the result using a solver
        obj = cp.Minimize(cp.Max(cp.sum(cp.multiply(h, 1 - h[cp.Variable(n_points)])), axis=0))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status == 'OPTIMAL':
                h_candidate = np.clip(h.value, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = evaluate(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        except:
            pass
    
    return (best_h, best_c5, n_points)
