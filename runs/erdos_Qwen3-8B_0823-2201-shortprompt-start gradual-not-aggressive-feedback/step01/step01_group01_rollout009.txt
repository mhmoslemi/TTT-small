    best_n = n_points
    time_left = budget_s

    def evaluate(h):
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def perturb(h, temp):
        h = h.copy()
        i = np.random.randint(len(h))
        delta = np.random.uniform(-0.1, 0.1)
        h[i] = np.clip(h[i] + delta, 0, 1)
        h = h * (len(h) / 2) / np.sum(h)
        return h

    for n in [50, 60, 70, 80, 90, 100]:
        h = initial_h.copy()

        temp = 1.0
        for _ in range(1000):
            if time_left <= 0:
                break
            h_perturbed = perturb(h, temp)

            if c5_perturbed < evaluate(h) or np.random.rand() < np.exp((evaluate(h) - c5_perturbed) / temp):
