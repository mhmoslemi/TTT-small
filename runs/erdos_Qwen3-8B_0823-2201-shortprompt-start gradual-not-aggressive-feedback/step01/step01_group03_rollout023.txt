import numpy as np
import scipy.optimize as optimize
from scipy.signal import correlate

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    conv = correlate(h, 1 - h, mode="full")
    c5 = np.max(conv) * dx
    return c5

def initial_h_values():
    return np.concatenate([np.ones(21), np.zeros(21)])  # 42 samples

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Efficient number of samples
    dx = 2.0 / n_points
    initial_h = initial_h_values()
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    
    # Function to maintain integral constraint
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 50
    mutation_rate = 0.15
    
    # Initial population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [normalize(h) for h in population]
    
    best_solution = initial_h
    best_c5 = evaluate_erdos_solution(best_solution)
    
    for gen in range(generations):
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(h) for h in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        if best_c5 < evaluate_erdos_solution(best_solution):
            best_solution = population[best_idx]
        
        # Create new population
        new_population = []
        for _ in range(population_size):
            # Select parents
            parent1 = population[np.random.choice(population_size)]
            parent2 = population[np.random.choice(population_size)]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            for i in range(n_points):
                if np.random.random() < mutation_rate:
                    child[i] = np.random.uniform(0, 1)
            
            # Normalize
            child = normalize(child)
            new_population.append(child)
        
        population = new_population
    
    return best_solution, best_c5, n_points
