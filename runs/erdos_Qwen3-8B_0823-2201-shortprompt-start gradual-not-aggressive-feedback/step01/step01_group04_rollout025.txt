import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup with the provided initial_h_values
    initial_h_values = np.load('initial_h_values.npy')  # Assuming initial_h_values is stored as a .npy file
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def constraint(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return np.sum(h) - n_points / 2
    
    # Optimization with bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, tol=1e-6)
    best_h = result.x
    
    # Simulated annealing to escape local minima
    temp = 1.0
    cooling_rate = 0.995
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        new_h = best_h.copy()
        idx = np.random.randint(n_points)
        new_h[idx] = np.random.uniform(0, 1)
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        new_c5 = objective(new_h)
        current_c5 = objective(best_h)
        if np.random.rand() < np.exp((current_c5 - new_c5) / temp):
            best_h = new_h
        temp *= cooling_rate
    
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
