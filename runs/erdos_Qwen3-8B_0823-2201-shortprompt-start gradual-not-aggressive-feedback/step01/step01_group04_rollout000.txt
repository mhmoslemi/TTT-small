import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial_h_values
    initial_h_values = np.load('initial_h_values.npy')  # Assuming this file exists
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h_norm = h * (n_points / 2) / np.sum(h)
        corr = correlate(h_norm, 1 - h_norm, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization using CVXPy
    h_var = cp.Variable(n_points)
    constraints = [cp.sum(h_var) == n_points / 2, 0 <= h_var, h_var <= 1]
    objective = cp.Minimize(evaluate(h_var))
    prob = cp.Problem(objective, constraints)
    
    best_h = h.copy()
    best_c5 = evaluate(best_h)
    elapsed = 0
    
    while elapsed < budget_s - 1:
        try:
            prob.solve(solver=cp.SCS, eps=1e-5, max_iters=100, verbose=False)
            if prob.status == 'optimal':
                h = h_var.value
                current_c5 = evaluate(h)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                elapsed = time.time() - start_time
            else:
                break
        except:
            break
    
    # If optimization fails, fall back to initial solution
    if best_c5 > evaluate(best_h):
        best_h = h.copy()
        best_c5 = evaluate(best_h)
    
    return (best_h, best_c5, len(best_h))
