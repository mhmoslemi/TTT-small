import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    max_overlap = np.max(np.correlate(initial_h, 1 - initial_h, mode="full")) * dx
    best_h = initial_h.copy()
    best_c5 = max_overlap
    best_n = n_points
    time_left = budget_s

    def evaluate(h):
        h = np.clip(h, 0, 1)
        norm = np.sum(h) * dx
        if abs(norm - 1) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        return h

    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        for i in range(len(child)):
            if random.random() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child

    for n in [50, 40, 30, 20]:
        time_start = time.time()
        population_size = 20
        population = [np.random.uniform(0, 1, n) for _ in range(population_size)]
        population = [h * (n / 2) / np.sum(h) for h in population]
        for _ in range(50):
            if time_left <= 0:
                break
            fitness = [evaluate(h) for h in population]
            best_idx = np.argmin(fitness)
            best_fitness = fitness[best_idx]
            if best_fitness < best_c5:
                best_c5 = best_fitness
                best_h = population[best_idx].copy()
                best_n = n
            new_population = []
            for _ in range(population_size // 2):
                parent1 = random.choices(population)[0]
                parent2 = random.choices(population)[0]
                child1 = crossover(parent1, parent2)
                child1 = mutate(child1)
                child2 = crossover(parent1, parent2)
                child2 = mutate(child2)
                new_population.append(child1)
                new_population.append(child2)
            population = new_population
            time_used = time.time() - time_start
            time_left -= time_used
        if time_left <= 0:
            break

    return best_h, best_c5, best_n
