def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def evaluate(h_values):
        h = np.array(h_values)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def crossover(parent1, parent2):
        child = np.copy(parent1)
        idx = np.random.choice(len(parent1), size=1, p=np.abs(parent1 - 0.5))
        child[idx] = parent2[idx]
        return child

    def mutate(h, mutation_rate=0.05):
        h = np.copy(h)
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    population_size = 50
    population = [mutate(deepcopy(initial_h)) for _ in range(population_size)]
    best_h = deepcopy(initial_h)
    best_c5 = evaluate(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        fitness = [evaluate(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]

            new_population.append(child1)
            child2 = crossover(parent2, parent1)

            new_population.append(child2)
