import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def initial_h_values():
        h = np.zeros(n_points)
        h[0:50] = 1
        h[50:100] = 0
        h[100:150] = 1
        h[150:200] = 0
        return h
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    
    initial_h = initial_h_values()
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def optimize():
        result = minimize(
            objective,
            initial_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        best_c5 = objective(best_h)
        return best_h, best_c5
    
    start_time = time.time()
    best_h, best_c5 = optimize()
    elapsed = time.time() - start_time
    
    if elapsed < budget_s:
        # Try genetic algorithm for further improvement
        def ga_objective(h):
            h = h.reshape(n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5
        
        ga_result = opt.differential_evolution(
            ga_objective,
            bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            polish=True,
            seed=seed
        )
        best_h = ga_result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        best_c5 = ga_objective(best_h)
    
    return best_h, best_c5, n_points
