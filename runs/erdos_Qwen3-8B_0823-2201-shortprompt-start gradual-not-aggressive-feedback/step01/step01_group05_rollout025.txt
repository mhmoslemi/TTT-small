        # Optimization variables
        h = cp.Variable(n_points)
        # Objective function
        def objective(h):
            h_np = h.value
            overlap = np.correlate(h_np, 1 - h_np, mode="full") * dx
            return np.max(overlap)
        
        # Constraints
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]

        start_time = time.time()
