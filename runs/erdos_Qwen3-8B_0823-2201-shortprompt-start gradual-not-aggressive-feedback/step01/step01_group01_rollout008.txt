import numpy as np
import scipy.optimize as optimize
import time
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    max_overlap = np.max(np.correlate(initial_h, 1 - initial_h, mode="full")) * dx
    best_h = initial_h.copy()
    best_c5 = max_overlap
    best_n = n_points
    time_left = budget_s

    # Genetic algorithm parameters
    pop_size = 20
    n_gen = 10
    cx_prob = 0.9
    mut_prob = 0.1
    mut_scale = 0.1

    # Setup genetic algorithm
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)

    def eval_func(individual):
        h = np.clip(individual, 0, 1)
        total = np.sum(h)
        if total == 0:
            return (np.inf,)
        h = h * (n_points / 2) / total
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return (c5,)

    def mutate(individual):
        for i in range(len(individual)):
            if np.random.random() < mut_prob:
                individual[i] += np.random.normal(0, mut_scale)
                individual[i] = np.clip(individual[i], 0, 1)
        return individual,

    def cx(ind1, ind2):
        offspring = []
        for i in range(len(ind1)):
            if np.random.random() < cx_prob:
                offspring.append(ind1[i])
            else:
                offspring.append(ind2[i])
        return (offspring,)

    # Start with the initial solution
    toolbox = base.Toolbox()
    toolbox.register("individual", tools.initRepeat, creator.Individual, np.random.rand, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("evaluate", eval_func)
    toolbox.register("mate", cx)
    toolbox.register("mutate", mutate)
    toolbox.register("select", tools.selTournament, tournsize=3)

    # Run genetic algorithm
    for n in [100, 80, 60, 50, 40, 30]:
        time_start = time.time()
        pop = toolbox.population(n=pop_size)
        for gen in range(n_gen):
            offspring = algorithms.varAnd(pop, toolbox, cx_prob, mut_prob)
            fits = toolbox.map(toolbox.evaluate, offspring)
            for fit, ind in zip(fits, offspring):
                ind.fitness.values = fit
            pop = toolbox.select(offspring, k=len(pop))
            if time.time() - time_start > time_left:
                break
        # Find best in population
        for ind in pop:
            if not ind.fitness.worseThan(best_h.fitness):
                best_h = ind
                best_c5 = ind.fitness.values[0]
        time_used = time.time() - time_start
        time_left -= time_used
        if time_left <= 0:
            break

    return best_h, best_c5, len(best_h)
