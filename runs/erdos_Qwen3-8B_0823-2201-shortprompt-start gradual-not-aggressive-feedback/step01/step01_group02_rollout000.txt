import numpy as np
import scipy.optimize as opt
import time
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h * target_sum / np.sum(initial_h)

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]

    # Genetic algorithm parameters
    population_size = 50
    crossover_prob = 0.8
    mutation_prob = 0.1
    generations = 100
    elite_size = 5

    # Setup DEAP
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", np.ndarray, fitness=creator.FitnessMin)

    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0, 1)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)

    toolbox.register("evaluate", objective)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)

    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    # Run genetic algorithm
    population = toolbox.population(n=population_size)
    for gen in range(generations):
        offspring = algorithms.varAnd(population, toolbox, cxpb=crossover_prob, mutpb=mutation_prob)
        fits = [toolbox.evaluate(ind) for ind in offspring]
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = (fit,)
        population = toolbox.select(offspring, k=len(population))
        # Keep track of the best individual
        for ind in population:
            if ind.fitness.values[0] < best_c5:
                best_c5 = ind.fitness.values[0]
                best_h = np.clip(ind, 0, 1)
                best_h = best_h * target_sum / np.sum(best_h)
        if time.time() - start_time > budget_s - 1:
            break

    # Final refinement with L-BFGS-B
    def final_objective(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    result = opt.minimize(
        final_objective,
        best_h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = final_objective(best_h)
    return best_h, best_c5, n_points
