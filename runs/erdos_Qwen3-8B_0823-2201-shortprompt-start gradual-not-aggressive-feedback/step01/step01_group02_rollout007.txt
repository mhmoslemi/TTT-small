import numpy as np
import random
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def evaluate(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def evolve(population, fitness_func, n_points, target_sum, mutation_rate=0.1, crossover_rate=0.8):
        population = [np.clip(ind, 0, 1) * target_sum / np.sum(np.clip(ind, 0, 1)) for ind in population]
        fitness = [fitness_func(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_fitness = fitness[best_idx]
        best_individual = population[best_idx]
        
        new_population = []
        for _ in range(len(population)):
            parent1, parent2 = random.choices(population, k=2)
            if random.random() < crossover_rate:
                child = np.array([parent1[i] if random.random() < 0.5 else parent2[i] for i in range(n_points)])
            else:
                child = np.copy(parent1)
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] = np.clip(child[i] + np.random.normal(0, 0.05), 0, 1)
            new_population.append(child)
        return new_population, best_fitness, best_individual

    population_size = 50
    population = [np.random.rand(n_points) for _ in range(population_size)]
    best_h = initial_h.copy()
    best_c5 = evaluate(initial_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        population, best_fitness, best_individual = evolve(population, evaluate, n_points, target_sum)
        if best_fitness < best_c5:
            best_c5 = best_fitness
            best_h = best_individual
        if best_fitness < 0.380876:
            break

    return best_h, best_c5, n_points
