import numpy as np
import random
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def evaluate_population(population):
        results = []
        for h in population:
            results.append((objective(h), h))
        return sorted(results)

    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] += np.random.uniform(-0.1, 0.1)
                h[i] = np.clip(h[i], 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def crossover(h1, h2):
        h = h1.copy()
        for i in range(len(h)):
            if random.random() < 0.5:
                h[i] = h2[i]
        h = h * target_sum / np.sum(h)
        return h

    population_size = 50
    population = [np.array(initial_h) * target_sum / np.sum(initial_h) for _ in range(population_size)]
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        ranked = evaluate_population(population)
        best = ranked[0][1]
        population = [best] * 10  # Elitism
        
        for _ in range(population_size - 10):
            h1, h2 = random.sample(ranked[:20], 2)
            child = crossover(h1[1], h2[1])
            child = mutate(child)
            population.append(child)
        
        population = population[:population_size]
    
    best_h = best
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
