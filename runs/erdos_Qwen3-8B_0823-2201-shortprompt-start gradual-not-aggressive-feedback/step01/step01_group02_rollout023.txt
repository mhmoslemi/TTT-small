import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def lbfgs_step(h):
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x

    def genetic_step(population, elite_size=5, mutation_rate=0.1):
        population = np.array(population)
        scores = np.array([objective(ind) for ind in population])
        elite_indices = np.argsort(scores)[:elite_size]
        elites = population[elite_indices]
        population = elites
        for i in range(len(population) - elite_size):
            parent1 = random.choice(elites)
            parent2 = random.choice(elites)
            child = np.copy(parent1)
            for j in range(n_points):
                if random.random() < mutation_rate:
                    child[j] = np.clip(parent1[j] + np.random.normal(0, 0.1), 0, 1)
            population = np.vstack((population, child))
        return population[:len(population) - (len(population) - elite_size)]

    initial_h = initial_h * target_sum / np.sum(initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
        initial_h = lbfgs_step(initial_h)
        initial_h = genetic_step([initial_h] * 20)
        current_c5 = objective(initial_h)
        if current_c5 < best_c5:
            best_h = initial_h
            best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return best_h, best_c5, n_points
