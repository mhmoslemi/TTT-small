import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds
from scipy.signal import correlate

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    conv = correlate(h, 1 - h, mode="full")
    c5 = np.max(conv) * dx
    return c5

def initial_h_values():
    return np.concatenate([np.ones(21), np.zeros(21)])  # 42 samples

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Efficient number of samples
    dx = 2.0 / n_points
    initial_h = initial_h_values()
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    
    # Refine initial guess with a structured approach
    for _ in range(3):
        h_refined = np.zeros(n_points)
        h_refined[:n_points//2] = 1.0
        h_refined[n_points//2:] = 0.0
        h_refined = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        def objective(h):
            h = h.reshape(-1)
            c5 = evaluate_erdos_solution(h)
            return c5
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        bounds = Bounds(0, 1)
        
        # Perform optimization
        result = optimize.minimize(
            objective,
            h_refined,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-8,
            options={'maxiter': 100, 'ftol': 1e-9, 'gtol': 1e-9}
        )
        
        initial_h = result.x
    
    h_values = initial_h
    c5_bound = evaluate_erdos_solution(h_values)
    return h_values, c5_bound, n_points
