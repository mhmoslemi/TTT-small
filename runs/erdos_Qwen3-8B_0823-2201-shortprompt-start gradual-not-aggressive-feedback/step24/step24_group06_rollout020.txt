import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    
    # Start with a different initial construction
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Construct an initial guess with alternating segments
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points//3:
            h[i] = 1.0
        elif i < 2*n_points//3:
            h[i] = 0.5
        else:
            h[i] = 0.0
    h = h * target_sum / np.sum(h)

        h = h_flat.reshape(n_points)

        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
