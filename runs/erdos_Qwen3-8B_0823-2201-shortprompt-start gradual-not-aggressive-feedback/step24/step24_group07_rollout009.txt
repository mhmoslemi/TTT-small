import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Hybrid optimization with probabilistic search and gradient descent
    for stage in range(3):
        # Stage 1: Probabilistic search with dynamic partitioning
        if stage == 0:
            for _ in range(15):
                if time.time() - start_time >= budget_s:
                    break
                # Dynamic partitioning
                partition_size = max(1, n_points // 10)
                partitions = np.array_split(np.arange(n_points), partition_size)
                
                # Randomly perturb each partition
                for p in partitions:
                    perturb = np.random.uniform(-0.15, 0.15, len(p))
                    h[p] += perturb
                    h[p] = np.clip(h[p], 0, 1)
                
                # Normalize after perturbation
                h = h * target_sum / np.sum(h)
                
                # Evaluate
                current_c5 = objective(h)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
        # Stage 2: Gradient descent with adaptive learning rate
        elif stage == 1:

                if time.time() - start_time >= budget_s:
                    break
                result = opt.minimize(
                    objective,
                    h,
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-9}
                )
                
                current_h = result.x
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * target_sum / np.sum(current_h)
                current_c5 = objective(current_h)
                
                if current_c5 < best_c5:

                    best_c5 = current_c5
                    h = current_h

                    h,
                    method='L-BFGS-B',

                if current_c5 < best_c5:

    return best_h, best_c5, n_points
