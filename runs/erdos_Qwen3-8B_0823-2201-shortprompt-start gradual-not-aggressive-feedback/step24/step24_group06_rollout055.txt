A hybrid approach combining gradient-based optimization with a genetic algorithm to avoid local optima. Start from the initial construction, use a genetic algorithm for exploration, and refine with SLSQP for fast local improvement. This leverages global search capabilities while maintaining feasibility.

import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Genetic algorithm for exploration
    for _ in range(10):
        if time.time() - start_time >= budget_s:
            break
        # Create population
        population = [h + np.random.uniform(-0.1, 0.1, n_points) for _ in range(10)]
        population = [np.clip(p, 0, 1) * target_sum / np.sum(np.clip(p, 0, 1)) for p in population]
        population.append(h)
        
        # Evaluate
        fitness = [objective(p) for p in population]
        
        # Select best
        best_idx = np.argmin(fitness)
        best_candidate = population[best_idx]
        current_c5 = fitness[best_idx]
        
        if current_c5 < best_c5:
            best_h = best_candidate.copy()
            best_c5 = current_c5
    
    # Refinement with SLSQP
    for _ in range(20):
        if time.time() - start_time >= budget_s:
            break
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-9}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
