import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Adaptive simulated annealing with gradient refinement
    for stage in range(3):
        # Stage 1: Simulated Annealing with adaptive cooling
        if stage == 0:
            for _ in range(20):
                if time.time() - start_time >= budget_s:
                    break
                # Adaptive temperature schedule
                temp = 1.0 / (1 + _)
                # Perturbation with decreasing magnitude
                perturb = np.random.uniform(-temp, temp, n_points)
                perturbed_h = h + perturb
                perturbed_h = np.clip(perturbed_h, 0, 1)
                perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
                
                result = opt.minimize(
                    objective,
                    perturbed_h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-9}
                )
                
                current_h = result.x
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * target_sum / np.sum(current_h)
                current_c5 = objective(current_h)
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    h = current_h
        # Stage 2: Gradient refinement on dense subdomains
        elif stage == 1:
            for _ in range(10):
                if time.time() - start_time >= budget_s:
                    break

                start_idx = np.random.randint(0, n_points - 20)
                end_idx = start_idx + 20
                sub_h = h[start_idx:end_idx].copy()
                
                # Gradient-based refinement
                result = opt.minimize(
                    objective,
                    sub_h,
                    method='L-BFGS-B',
                    bounds=[(0, 1) for _ in range(20)],

                )

                h[start_idx:end_idx] = refined_sub_h
                
                # Normalize the full solution
                h = h * target_sum / np.sum(h)
                current_c5 = objective(h)
                
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
        # Stage 3: Fine-tuning with local search
        else:
            for _ in range(10):
                if time.time() - start_time >= budget_s:
                    break

                perturbed_h = h + perturb
                perturbed_h = np.clip(perturbed_h, 0, 1)
                perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)

                    objective,
                    perturbed_h,

                if current_c5 < best_c5:

                    best_c5 = current_c5
