import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Optimization with adaptive perturbations and refinement
    for stage in range(3):
        # Stage 1: Global optimization with SLSQP
        if stage == 0:
            for _ in range(15):
                if time.time() - start_time >= budget_s:
                    break
                result = opt.minimize(
                    objective,
                    h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-9}
                )
                h = result.x
                h = np.clip(h, 0, 1)
                h = h * target_sum / np.sum(h)
                current_c5 = objective(h)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5

                if time.time() - start_time >= budget_s:
                    break

                    objective,

                    options={'maxiter': 50, 'ftol': 1e-8, 'eps': 1e-9}
