import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Generate multiple initial guesses
    initial_guesses = [np.array(initial_h_values, dtype=np.float64) * target_sum / np.sum(initial_h_values)]
    for _ in range(2):
        noise = np.random.normal(0, 0.1, n_points)
        guess = np.clip(initial_h_values * target_sum / np.sum(initial_h_values) + noise, 0, 1)
        initial_guesses.append(guess)
    
    best_h = np.array(initial_h_values, dtype=np.float64) * target_sum / np.sum(initial_h_values)
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    # Hybrid optimization with multiple starting points
    for guess in initial_guesses:
        # Initial optimization with SLSQP
        result = opt.minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-9}
        )
        
        current_h = result.x

        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    # Refinement with L-BFGS-B
    for _ in range(5):
