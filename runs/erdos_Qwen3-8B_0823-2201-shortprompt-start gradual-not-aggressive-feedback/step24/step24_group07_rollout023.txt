import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Stage 1: Refinement with SQP
    for _ in range(10):
        if time.time() - start_time >= budget_s:
            break
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-9}
        )
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            h = current_h
    
    # Stage 2: Evolutionary search with tournament selection
    population_size = 20
    elite_size = 2
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 50
    
    population = [h.copy() for _ in range(population_size)]
    
    for gen in range(generations):
        if time.time() - start_time >= budget_s:
            break
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        # Generate offspring
        offspring = []
        for _ in range(population_size - elite_size):
            # Tournament selection
            competitors = np.random.choice(population_size, 5, replace=False)
            best = competitors[np.argmin(fitness[competitors])]
            parent = population[best]
            # Mutate
            mutated = parent.copy()
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    mutated[i] += np.random.uniform(-0.1, 0.1)
                    mutated[i] = np.clip(mutated[i], 0, 1)
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([mutated[:crossover_point], population[np.random.randint(population_size)][:crossover_point]])
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
                offspring.append(child)
            else:
                offspring.append(mutated)
        # Replace population
        population = elites + offspring
    
    # Evaluate final population
    final_fitness = [objective(ind) for ind in population]
    best_idx = np.argmin(final_fitness)
    best_h = population[best_idx].copy()
    best_c5 = final_fitness[best_idx]
    
    return best_h, best_c5, n_points
