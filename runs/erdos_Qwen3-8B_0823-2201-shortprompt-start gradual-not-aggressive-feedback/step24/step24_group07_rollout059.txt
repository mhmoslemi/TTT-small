import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Hybrid optimization: gradient descent with local refinement
    for _ in range(20):
        if time.time() - start_time >= budget_s:
            break
        
        # Gradient-based optimization
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-9}
        )
        
        h = result.x
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = objective(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Local refinement: greedy search for best single-point update
        for i in range(n_points):
            for delta in [-0.1, 0.1]:
                new_h = h.copy()
                new_h[i] += delta
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * target_sum / np.sum(new_h)
                new_c5 = objective(new_h)
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
                    h = new_h
    
    return best_h, best_c5, n_points
