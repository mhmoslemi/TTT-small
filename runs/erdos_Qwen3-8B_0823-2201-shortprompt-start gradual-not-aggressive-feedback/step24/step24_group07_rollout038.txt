import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Multi-stage optimization: global search -> refinement -> fine-tuning
    for stage in range(3):
        # Stage 1: Global search with differential evolution
        if stage == 0:
            for _ in range(10):
                if time.time() - start_time >= budget_s:
                    break
                # Create a population with perturbations
                population = [h + np.random.uniform(-0.2, 0.2, n_points) for _ in range(5)]
                population = [p * target_sum / np.sum(np.clip(p, 0, 1)) for p in population]
                
                # Evaluate population
                fitness = [objective(p) for p in population]
                best_idx = np.argmin(fitness)
                best_pop = population[best_idx]
                
                # Use best population member as new starting point
                h = best_pop.copy()
                current_c5 = fitness[best_idx]
                
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
    
        # Stage 2: Refinement with gradient descent
        elif stage == 1:
            for _ in range(15):
                if time.time() - start_time >= budget_s:
                    break
                result = opt.minimize(
                    objective,
                    h,
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-9}
                )
                
                current_h = result.x
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * target_sum / np.sum(current_h)
                current_c5 = objective(current_h)
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    h = current_h
    
        # Stage 3: Fine-tuning with local search
        else:
            for _ in range(15):
                if time.time() - start_time >= budget_s:
                    break
                # Random perturbation with decreasing magnitude
                perturb = np.random.uniform(-0.05 * (0.95 ** _), 0.05 * (0.95 ** _), n_points)
                perturbed_h = h + perturb
                perturbed_h = np.clip(perturbed_h, 0, 1)
                perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
                
                result = opt.minimize(
                    objective,
                    perturbed_h,
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-9}
                )
                
                current_h = result.x
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * target_sum / np.sum(current_h)
                current_c5 = objective(current_h)
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    h = current_h
    
    return best_h, best_c5, n_points
