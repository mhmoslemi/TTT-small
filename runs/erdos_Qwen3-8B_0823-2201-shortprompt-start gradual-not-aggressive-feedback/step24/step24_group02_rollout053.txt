import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # First stage: local refinement with SLSQP
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-8}
    )
    h = result.x
    current_c5 = objective(h)
    
    if current_c5 < best_c5:
        best_h = h.copy()
        best_c5 = current_c5
    
    # Second stage: structured perturbations with normalization
    for perturb_size in [0.05, 0.025, 0.0125, 0.00625, 0.003125]:
        if time.time() - start_time >= budget_s:
            break
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    # Third stage: global search with differential evolution
    def objective_de(h_vec):
        h = h_vec.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Initialize DE with current best solution
    de_bounds = np.array([(0, 1) for _ in range(n_points)])
    result_de = opt.differential_evolution(
        objective_de,
        de_bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        disp=False
    )
    
    de_solution = result_de.x
    de_solution = np.clip(de_solution, 0, 1)
    de_solution = de_solution * target_sum / np.sum(de_solution)
    de_c5 = objective_de(de_solution)
    
    if de_c5 < best_c5:
        best_h = de_solution.copy()
        best_c5 = de_c5
    
    return best_h, best_c5, n_points
