import time

        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    # Main optimization loop
    for _ in range(100):

            break
        
        # Perturb based on gradient direction
        perturb = np.zeros(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
