import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 64  # Coarser grid for faster optimization
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial construction with more balanced values
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.0
    initial_h_values = np.roll(initial_h_values, int(n_points / 4))
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)

    start_time = time.time()
    
    # Multi-phase optimization: exploration -> refinement -> fine-tuning

            # Exploration with random perturbations and Nelder-Mead
            for _ in range(15):
                if time.time() - start_time >= budget_s:
                    break

                perturbed_h = np.clip(perturbed_h, 0, 1)

                result = opt.minimize(
                    objective,
                    perturbed_h,
                    method='Nelder-Mead',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 50, 'xtol': 1e-8}
                )
