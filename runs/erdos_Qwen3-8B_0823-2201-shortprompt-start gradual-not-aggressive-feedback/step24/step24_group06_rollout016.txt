import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Particle Swarm Optimization for global exploration
    class PSO:
        def __init__(self, n_particles, n_dimensions, bounds, objective, constraint, target_sum):
            self.n_particles = n_particles
            self.n_dimensions = n_dimensions
            self.bounds = bounds
            self.objective = objective
            self.constraint = constraint
            self.target_sum = target_sum
            self.particles = np.random.rand(n_particles, n_dimensions) * (bounds[:, 1] - bounds[:, 0]) + bounds[:, 0]
            self.velocities = np.random.rand(n_particles, n_dimensions) * 0.1
            self.pbest_positions = self.particles.copy()
            self.pbest_values = np.array([self.objective(p) for p in self.particles])
            self.gbest_position = self.particles[0].copy()
            self.gbest_value = self.pbest_values[0]
            
        def update(self, inertia=0.8, cognitive=1.2, social=1.2, max_iter=100):
            for _ in range(max_iter):
                if time.time() - start_time >= budget_s:
                    break
                for i in range(self.n_particles):
                    r1, r2 = np.random.rand(2, self.n_dimensions)
                    self.velocities[i] = inertia * self.velocities[i] + cognitive * r1 * (self.pbest_positions[i] - self.particles[i]) + social * r2 * (self.gbest_position - self.particles[i])
                    self.particles[i] += self.velocities[i]
                    self.particles[i] = np.clip(self.particles[i], self.bounds[:, 0], self.bounds[:, 1])
                    
                    # Normalize to maintain constraint
                    self.particles[i] = self.particles[i] * self.target_sum / np.sum(self.particles[i])
                    
                    current_value = self.objective(self.particles[i])
                    if current_value < self.pbest_values[i]:
                        self.pbest_positions[i] = self.particles[i].copy()
                        self.pbest_values[i] = current_value
                best_idx = np.argmin(self.pbest_values)
                if self.pbest_values[best_idx] < self.gbest_value:
                    self.gbest_position = self.pbest_positions[best_idx].copy()
                    self.gbest_value = self.pbest_values[best_idx]
            return self.gbest_position
    
    # Run PSO
    pso = PSO(n_particles=30, n_dimensions=n_points, bounds=bounds, objective=objective, constraint=constraint_sum, target_sum=target_sum)
    pso_position = pso.update(inertia=0.8, cognitive=1.5, social=1.5, max_iter=50)
    
    # Refine with SLSQP
    h = pso_position
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-9}
    )
    
    current_h = result.x
    current_h = np.clip(current_h, 0, 1)
    current_h = current_h * target_sum / np.sum(current_h)
    current_c5 = objective(current_h)
    
    if current_c5 < best_c5:
        best_h = current_h.copy()
        best_c5 = current_c5
    
    return best_h, best_c5, n_points
