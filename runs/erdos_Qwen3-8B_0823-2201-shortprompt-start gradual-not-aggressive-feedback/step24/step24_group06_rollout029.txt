import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Hybrid optimization: genetic algorithm with local refinement
    population_size = 20
    generations = 10
    mutation_rate = 0.15
    
    for gen in range(generations):
        if time.time() - start_time >= budget_s:
            break
        
        # Generate new population
        population = [h.copy() for _ in range(population_size)]
        
        # Apply mutations
        for i in range(population_size):
            if np.random.rand() < mutation_rate:
                perturb = np.random.uniform(-0.1, 0.1, n_points)
                population[i] += perturb
                population[i] = np.clip(population[i], 0, 1)
                population[i] = population[i] * target_sum / np.sum(population[i])
        
        # Evaluate all candidates
        candidates = []
        for candidate in population:
            candidate = candidate.copy()
            candidate = candidate * target_sum / np.sum(candidate)
            candidates.append(candidate)
        
        # Select best candidates for refinement
        candidate_scores = [objective(c) for c in candidates]
        best_indices = np.argsort(candidate_scores)[:population_size//2]
        refined_population = [candidates[i] for i in best_indices]
        
        # Local refinement with SLSQP
        for i in range(len(refined_population)):
            candidate = refined_population[i].copy()
            result = opt.minimize(
                objective,
                candidate,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-9}

        # Update best solution

            candidate = candidate.copy()

            if current_c5 < best_c5:
                best_h = candidate.copy()
                best_c5 = current_c5
