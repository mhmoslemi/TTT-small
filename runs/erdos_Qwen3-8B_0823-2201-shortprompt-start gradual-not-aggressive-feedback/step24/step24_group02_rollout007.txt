import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    start_time = time.time()
    
    # Simulated annealing with gradient-based refinement
    temp = 1.0
    cooling_rate = 0.995
    max_iter = 1000
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Gradient-based refinement
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-8}
        )
        current_h = result.x

        temp *= cooling_rate
