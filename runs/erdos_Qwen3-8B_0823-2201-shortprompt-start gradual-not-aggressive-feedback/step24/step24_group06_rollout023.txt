import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Multi-stage optimization: exploration -> refinement -> fine-tuning
    for stage in range(3):
        # Stage 1: Global exploration with genetic algorithm
        if stage == 0:
            population_size = 20
            generations = 10
            for gen in range(generations):
                if time.time() - start_time >= budget_s:
                    break
                # Create new population
                population = np.random.uniform(0, 1, (population_size, n_points))
                population = population * target_sum / np.sum(population, axis=1, keepdims=True)
                
                # Evaluate fitness
                fitness = np.array([objective(ind) for ind in population])
                
                # Select best individuals
                best_indices = np.argsort(fitness)[:population_size//2]
                new_population = population[best_indices]
                
                # Apply crossover and mutation
                for i in range(population_size//2):
                    j = np.random.choice(population_size//2, 1)[0]
                    crossover_point = np.random.randint(1, n_points-1)
                    child = np.concatenate((new_population[i][:crossover_point], new_population[j][crossover_point:]))
                    mutation_magnitude = 0.1 * (0.95 ** gen)
                    mutation = np.random.uniform(-mutation_magnitude, mutation_magnitude, n_points)
                    child += mutation
                    child = np.clip(child, 0, 1)
                    child = child * target_sum / np.sum(child)
                    new_population = np.vstack((new_population, child))
                
                population = new_population[:population_size]
                current_h = population[np.argmin(fitness)]
                current_c5 = objective(current_h)
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    h = current_h
        # Stage 2: Refinement with simulated annealing
        elif stage == 1:
            for _ in range(10):
                if time.time() - start_time >= budget_s:
                    break
                # Simulated annealing with adaptive temperature
                temp = 1.0 / (1 + _)
                perturb = np.random.uniform(-temp, temp, n_points)
                perturbed_h = h + perturb
                perturbed_h = np.clip(perturbed_h, 0, 1)
                perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
                
                result = opt.minimize(
                    objective,
                    perturbed_h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-9}
                )
                
                current_h = result.x
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * target_sum / np.sum(current_h)
                current_c5 = objective(current_h)
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    h = current_h
        # Stage 3: Fine-tuning with gradient descent
        else:
            for _ in range(10):
                if time.time() - start_time >= budget_s:
                    break
                result = opt.minimize(
                    objective,
                    h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-9}
                )
                
                current_h = result.x
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * target_sum / np.sum(current_h)
                current_c5 = objective(current_h)
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    h = current_h
    
    return best_h, best_c5, n_points
