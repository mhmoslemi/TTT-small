import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def tournament_selection(population, k=3):
        indices = np.random.choice(len(population), k, replace=False)
        best = min(indices, key=lambda i: objective(population[i]))
        return population[best]
    
    def crossover(parent1, parent2):
        point = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:point], parent2[point:]])
        return child
    
    def mutate(individual, mutation_rate=0.05):
        mutated = individual.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                mutated[i] = np.random.uniform(0, 1)
        mutated = mutated * target_sum / np.sum(mutated)
        return mutated
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    population = [initial_h.copy() for _ in range(50)]
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for _ in range(len(population) // 2):
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        population = new_population + [best_h]
        for individual in population:
            refined = local_refinement(individual)
            current_c5 = objective(refined)
            if current_c5 < best_c5:
                best_h = refined
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
