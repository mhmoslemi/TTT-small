import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
        )
        return result.x
    
    # Initial population
    population = [initial_h.copy() for _ in range(20)]
    population_scores = [objective(ind) for ind in population]
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate all individuals
        scores = [objective(ind) for ind in population]
        
        # Select parents using tournament selection
        parents = []
        for _ in range(2):
            contestants = np.random.choice(range(len(population)), size=5, replace=False)
            best_idx = contestants[np.argmin([scores[i] for i in contestants])]
            parents.append(population[best_idx])
        
        # Crossover
        child = (parents[0] + parents[1]) / 2
        
        # Mutation
        mutation_rate = 0.1
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                child[i] += np.random.uniform(-0.1, 0.1)
                child[i] = np.clip(child[i], 0, 1)
        
        # Local refinement of child
        refined_child = local_refinement(child)
        child_score = objective(refined_child)
        
        # Replacement
        worst_idx = np.argmax(population_scores)
        if child_score < population_scores[worst_idx]:
            population[worst_idx] = refined_child
            population_scores[worst_idx] = child_score
    
    best_h = population[np.argmin(population_scores)]
    best_c5 = np.min(population_scores)
    
    return best_h, best_c5, n_points
