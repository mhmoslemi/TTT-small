import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 128
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess with uniform distribution
    initial_h = np.full(n_points, 0.5)
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Try different segment configurations
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Create a random segment configuration
        segment_indices = np.random.choice(n_points, size=8, replace=False)
        segment_indices = np.sort(segment_indices)
        segment_values = np.random.uniform(0, 1, size=8)
        
        # Create a piecewise constant function
        h = np.zeros(n_points)
        for i in range(8):
            if i == 0:
                h[0:segment_indices[i]] = segment_values[i]
            else:
                h[segment_indices[i-1]:segment_indices[i]] = segment_values[i]
        h[segment_indices[-1]:] = segment_values[-1]
        
        # Normalize to satisfy the integral constraint
        h = h * target_sum / np.sum(h)
        
        # Optimize from this initial guess
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
