    mutation_rate = 0.15

        # Evaluate population
        fitness = [objective(h) for h in population]

        elite_indices = np.argsort(fitness)[:elite_size]

        # Tournament selection
        new_population = []
        for _ in range(len(population) - elite_size):
            contestants = np.random.choice(range(len(population)), tournament_size, replace=False)
