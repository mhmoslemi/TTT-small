import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.15
    crossover_rate = 0.8

    # Initial population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]

    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Select parents
        parents = np.random.choice(population_size, size=population_size, p=1 / np.array(fitness))
        parents = [population[i] for i in parents]
        
        # Crossover
        offspring = []
        for i in range(population_size // 2):
            if np.random.rand() < crossover_rate:
                parent1 = parents[2*i]
                parent2 = parents[2*i + 1]
                child = np.random.rand(n_points)
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            else:
                child = parents[i]
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                offspring[i] = np.clip(offspring[i] + np.random.normal(0, 0.1, n_points), 0, 1)
                offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
        
        # Replace population
        population = offspring
        
        # Local refinement with SLSQP
        for i in range(len(population)):
            h = population[i]
            result = opt.minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-9}
            )
            population[i] = result.x
        
    return best_h, best_c5, n_points
