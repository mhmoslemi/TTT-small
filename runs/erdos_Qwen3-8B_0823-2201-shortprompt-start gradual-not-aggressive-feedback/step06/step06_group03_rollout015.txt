import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Genetic algorithm parameters
    pop_size = 20
    generations = 100
    mutation_rate = 0.1
    elite_size = 2

    # Initial population
    population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
    population = [h * target_sum / np.sum(h) for h in population]

    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        
        # Tournament selection
        selected = []
        for _ in range(pop_size - elite_size):
            candidates = np.random.choice(pop_size, size=5, replace=False)
            winner = candidates[np.argmin(fitness[candidates])]
            selected.append(population[winner])
        
        # Apply mutations
        mutated = []
        for h in selected:
            if np.random.random() < mutation_rate:
                perturb = np.random.normal(0, 0.05, n_points)
                h = np.clip(h + perturb, 0, 1)
                h = h * target_sum / np.sum(h)
            mutated.append(h)
        
        # Create new population
        population = elites + mutated
        population = [h * target_sum / np.sum(h) for h in population]
        
        # Update best solution
        for h in population:
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    # Local optimization with SLSQP
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
