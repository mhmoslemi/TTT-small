import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize with given construction
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
        )
        return result.x

    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 10

    # Initial population
    population = [initial_h.copy() for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = [objective(h) for h in population]
        
        # Select top performers
        parents = [population[i] for i in np.argsort(fitness)[:population_size//2]]
        
        # Generate new population
        new_population = []
        for _ in range(population_size):
            # Crossover
            if np.random.rand() < crossover_rate:
                parent1, parent2 = np.random.choice(parents, 2, replace=False)
                child = np.zeros_like(parent1)
                for i in range(n_points):
                    if np.random.rand() < 0.5:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
            else:
                # Mutation
                child = np.random.rand(n_points)
            
            # Apply mutation
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, 0.05, n_points)
                child += mutation
                child = np.clip(child, 0, 1)
            
            # Normalize
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        # Replace population
        population = new_population
        
        # Local refinement of best solution
        best_idx = np.argmin(fitness)
        refined_h = local_refinement(population[best_idx])
        refined_c5 = objective(refined_h)
        
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
    
    return best_h, best_c5, n_points
