import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize with given construction
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
        )
        return result.x

    # Genetic algorithm parameters
    population_size = 10
    generations = 10
    mutation_rate = 0.1

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Create population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        
        # Evaluate population
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h_pop = population[best_idx]
        best_c5_pop = fitness[best_idx]
        
        # Update best solution
        if best_c5_pop < best_c5:
            best_h = best_h_pop.copy()
            best_c5 = best_c5_pop
        
        # Genetic operations
        for _ in range(generations):
            # Select parents
            parents = np.random.choice(population_size, size=2, replace=False)
            parent1 = population[parents[0]]
            parent2 = population[parents[1]]
            
            # Crossover
            child = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.uniform(-0.1, 0.1)
                    child[i] = np.clip(child[i], 0, 1)
            
            # Normalize
            child = child * target_sum / np.sum(child)
            
            # Evaluate
            fitness_child = objective(child)
            
            # Replace worst
            worst_idx = np.argmax(fitness)
            if fitness_child < fitness[worst_idx]:
                population[worst_idx] = child
                fitness[worst_idx] = fitness_child

        # Local refinement of best population solution
        refined_h = local_refinement(best_h)
        refined_c5 = objective(refined_h)
        
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5

    return best_h, best_c5, n_points
