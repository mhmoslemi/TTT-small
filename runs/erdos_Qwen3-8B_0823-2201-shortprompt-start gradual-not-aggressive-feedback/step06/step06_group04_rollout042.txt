import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def normalize(h):
        h = h.copy()
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * target_sum / total

    def objective(h):
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    bounds = [(0, 1) for _ in range(n_points)]
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] += np.random.uniform(-0.2, 0.2)
                h[i] = np.clip(h[i], 0, 1)
        return normalize(h)

    def crossover(h1, h2):
        h = h1.copy()
        for i in range(len(h)):
            if random.random() < 0.5:
                h[i] = h2[i]
        return normalize(h)

    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        new_h = mutate(best_h)
        new_c5 = objective(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5

        new_h = crossover(best_h, initial_h)
        new_c5 = objective(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5

        # Local optimization
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(normalize(x)) - target_sum},
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        best_h = result.x
        best_c5 = objective(best_h)

    return best_h, best_c5, n_points
