import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use a dense grid for better accuracy
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial construction with uniform distribution
    initial_h = np.ones(n_points) * 0.5
    best_h = initial_h.copy()
    best_c5 = 0.381221  # Start with the previous best

        h = h.reshape(n_points)

        return c5

    start_time = time.time()
    
    # Initial optimization with more iterations
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum},
        options={'maxiter': 100, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Local search with more refined perturbations
    while time.time() - start_time < budget_s - 1:
        # Create a perturbation with smaller variance
        perturb = np.random.normal(0, 0.02, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Optimization with fewer iterations
        result = opt.minimize(
            objective,

        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
