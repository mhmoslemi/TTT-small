import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Initial optimization with more iterations
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Genetic algorithm for exploration
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    mutation_strength = 0.05

    def mutate(individual):
        mutated = individual.copy()
        for i in range(len(mutated)):
            if random.random() < mutation_rate:
                mutated[i] += np.random.normal(0, mutation_strength)
                mutated[i] = np.clip(mutated[i], 0, 1)
        return mutated

    def crossover(parent1, parent2):
        child = parent1.copy()
        for i in range(len(child)):
            if random.random() < 0.5:
                child[i] = parent2[i]
        return child

    population = [best_h.copy() for _ in range(population_size)]
    for _ in range(generations):
        if time.time() - start_time >= budget_s - 1:
            break
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            parent1, parent2 = random.choices(population, k=2)
            child = crossover(parent1, parent2)
            child = mutate(child)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        # Replace old population with new one
        population = new_population
        population.append(best_h.copy())
        population = np.array(population)
    
    return best_h, best_c5, n_points
