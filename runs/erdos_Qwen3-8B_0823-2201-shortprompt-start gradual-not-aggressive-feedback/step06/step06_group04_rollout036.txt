import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    start_time = time.time()

    population_size = 10
    mutation_rate = 0.15
    generations = 20
    
    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            # Selection
            idx1 = np.random.choice(range(population_size), p=np.array(fitness)/np.sum(fitness))
            idx2 = np.random.choice(range(population_size), p=np.array(fitness)/np.sum(fitness))

            parent2 = population[idx2].copy()
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)

            # Mutation
            child1 = np.clip(child1 + np.random.normal(0, mutation_rate, n_points), 0, 1)
            child2 = np.clip(child2 + np.random.normal(0, mutation_rate, n_points), 0, 1)
            
            # Normalize
            child1 = child1 * target_sum / np.sum(child1)

        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)

            perturbed_h,

            constraints={'type': 'eq', 'fun': constraint_sum},

        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
