        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    def generate_population(size):
        pop = np.random.rand(size, n_points)
        pop = np.clip(pop, 0, 1)
        pop = pop * target_sum / np.sum(pop, axis=1, keepdims=True)
        return pop
    
    def tournament_selection(pop, fitnesses, k=3):
        indices = np.random.choice(len(pop), k, replace=False)
        best_idx = np.argmin(fitnesses[indices])
        return pop[indices[best_idx]]
    
    def crossover(parent1, parent2):
        alpha = np.random.rand(n_points)
        child = alpha * parent1 + (1 - alpha) * parent2
        child = np.clip(child, 0, 1)
        child = child * target_sum / np.sum(child)
        return child
    
    def mutate(individual, mutation_rate=0.05):
        mutation = np.random.uniform(-mutation_rate, mutation_rate, n_points)
        mutated = individual + mutation
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * target_sum / np.sum(mutated)
        return mutated
    
    def genetic_algorithm():
        population = generate_population(50)

        best_c5 = np.min(fitnesses)
