import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def normalize(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        return h * target_sum / np.sum(h)
    
    def crossover(parent1, parent2):
        size = len(parent1)
        child = np.zeros(size)
        for i in range(size):
            if np.random.rand() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return normalize(child)
    
    def mutate(h, mutation_rate=0.1):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.uniform(-0.15, 0.15)
                h[i] = np.clip(h[i], 0, 1)
        return normalize(h)
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    population_size = 20
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    
    while time.time() - start_time < budget_s - 1:
        fitness = [objective(ind) for ind in population]
        best_index = np.argmin(fitness)
        best_c5 = fitness[best_index]
        best_h = population[best_index].copy()
        
        new_population = [population[best_index]]
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(population, size=2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
    
    return best_h, best_c5, n_points
