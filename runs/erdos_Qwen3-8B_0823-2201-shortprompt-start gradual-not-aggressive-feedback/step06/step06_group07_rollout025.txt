import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize with given construction
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    pop_size = 20
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8

    def genetic_algorithm(population):
        population = [np.clip(ind, 0, 1) * target_sum / np.sum(np.clip(ind, 0, 1)) for ind in population]
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            # Select parents
            parents = np.random.choice(len(population), size=pop_size, p=fitness/np.sum(fitness))
            parents = population[parents]
            # Crossover
            offspring = []
            for i in range(0, pop_size, 2):
                if np.random.rand() < crossover_rate:
                    p1, p2 = parents[i], parents[i+1]
                    child1 = np.concatenate([p1[:n_points//2], p2[n_points//2:]])
                    child2 = np.concatenate([p2[:n_points//2], p1[n_points//2:]])
                    offspring.append(child1)
                    offspring.append(child2)
                else:
                    offspring.append(parents[i])
                    offspring.append(parents[i+1])
            # Mutation
            for i in range(pop_size):
                if np.random.rand() < mutation_rate:
                    idx = np.random.randint(n_points)
                    offspring[i][idx] += np.random.uniform(-0.1, 0.1)
            population = offspring
        return population

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Genetic algorithm
        population = [best_h + np.random.normal(0, 0.05, n_points) for _ in range(pop_size)]
        population = genetic_algorithm(population)
        for candidate in population:
            candidate_c5 = objective(candidate)
            if candidate_c5 < best_c5:
                best_h = candidate
                best_c5 = candidate_c5
        
        # Local refinement with SLSQP
        refined_h = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
        ).x
        refined_c5 = objective(refined_h)
        
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5

    return best_h, best_c5, n_points
