import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize with given construction
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
        )
        return result.x

    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 10
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Create population
        population = [best_h + np.random.normal(0, 0.05, n_points) for _ in range(population_size)]
        population = [np.clip(p, 0, 1) * target_sum / np.sum(np.clip(p, 0, 1)) for p in population]
        
        # Evaluate population
        fitness = [objective(p) for p in population]
        
        # Select best
        best_idx = np.argmin(fitness)
        best_candidate = population[best_idx]
        best_c5_candidate = fitness[best_idx]
        
        # Update best solution
        if best_c5_candidate < best_c5:
            best_h = best_candidate
            best_c5 = best_c5_candidate
        
        # Perform local refinement on best candidate
        refined_h = local_refinement(best_h)
        refined_c5 = objective(refined_h)
        
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
        
        # Genetic operations
        for _ in range(generations):
            # Select parents
            parent1 = population[np.random.choice(population_size)]
            parent2 = population[np.random.choice(population_size)]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                child = (parent1 + parent2) / 2
            else:
                child = parent1
            
            # Mutation
            if np.random.rand() < mutation_rate:
                child += np.random.normal(0, 0.05, n_points)
            
            # Normalize
            child = np.clip(child, 0, 1) * target_sum / np.sum(np.clip(child, 0, 1))
            
            # Evaluate
            child_c5 = objective(child)
            
            # Replace worst
            if child_c5 < np.min(fitness):
                idx = np.argmin(fitness)
                population[idx] = child
                fitness[idx] = child_c5
    
    return best_h, best_c5, n_points
