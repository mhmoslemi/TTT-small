import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Initial optimization with more iterations
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Genetic algorithm for global search
    def genetic_search(population_size=30, generations=20):
        population = [np.random.rand(n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        population = [np.clip(h, 0, 1) for h in population]
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = [objective(h) for h in population]
            best_idx = np.argmin(fitness)
            best_h = population[best_idx]
            
            # Create new population
            new_population = []
            for _ in range(population_size // 2):
                # Selection
                idx1 = np.random.choice(range(population_size), p=[f / sum(fitness) for f in fitness])
                idx2 = np.random.choice(range(population_size), p=[f / sum(fitness) for f in fitness])
                parent1, parent2 = population[idx1], population[idx2]
                
                # Crossover
                child = np.zeros(n_points)
                for i in range(n_points):
                    if np.random.rand() < 0.5:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
                
                # Mutation

                    if np.random.rand() < 0.1:
                        child[i] = np.random.rand()
                
                # Clip and normalize
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
                new_population.append(child)
            
            population = new_population + [best_h]  # Preserve best solution
            
            # Replace with better solutions
            for i in range(len(population)):
                if np.random.rand() < 0.3:
                    population[i] = best_h
            
            # Keep only the best solutions
            population = sorted(population, key=lambda x: objective(x))
            population = population[:population_size]
        
        return np.min(fitness), population[0]

    # Run genetic search
    if time.time() - start_time < budget_s - 5:
        genetic_c5, genetic_h = genetic_search(population_size=30, generations=20)
        if genetic_c5 < best_c5:
            best_h = genetic_h
            best_c5 = genetic_c5

    # Local search with perturbations and more refined optimization
    while time.time() - start_time < budget_s - 1:

        perturbed_h = np.clip(perturbed_h, 0, 1)

            best_c5 = current_c5
    
    return best_h, best_c5, n_points
