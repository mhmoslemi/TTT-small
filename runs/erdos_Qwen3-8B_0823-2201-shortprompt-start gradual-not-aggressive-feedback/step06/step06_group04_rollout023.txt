import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Initial optimization with SLSQP
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    pop_size = 20
    generations = 10
    mutation_rate = 0.15
    tournament_size = 5
    
    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        
        # Select parents using tournament selection
        parents = []
        for _ in range(pop_size):
            tournament = random.sample(list(enumerate(fitness)), tournament_size)
            winner_idx = min(tournament, key=lambda x: x[1])[0]
            parents.append(population[winner_idx])
        
        # Create offspring through mutation
        offspring = []
        for parent in parents:
            mutated = parent.copy()
            for i in range(n_points):
                if np.random.random() < mutation_rate:
                    mutated[i] += np.random.uniform(-0.15, 0.15)
                    mutated[i] = np.clip(mutated[i], 0, 1)
            mutated = mutated * target_sum / np.sum(mutated)
            offspring.append(mutated)
        
        # Replace population with offspring
        population = offspring
        
        # Update best solution
        for ind in population:
            current_c5 = objective(ind)
            if current_c5 < best_c5:
                best_h = ind.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
