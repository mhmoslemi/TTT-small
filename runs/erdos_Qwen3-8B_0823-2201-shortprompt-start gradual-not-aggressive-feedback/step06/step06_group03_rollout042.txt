import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Initial optimization with SLSQP
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Genetic algorithm for global search
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    mutation_strength = 0.05

    def create_individual():
        return np.random.rand(n_points) * 2 - 1

    def evaluate(individual):
        individual = np.clip(individual, 0, 1)
        individual = individual * target_sum / np.sum(individual)
        corr = np.correlate(individual, 1 - individual, mode="full")
        return np.max(corr) * dx

    def crossover(parent1, parent2):
        child = np.empty_like(parent1)
        for i in range(n_points):
            if random.random() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child

    def mutate(individual):
        for i in range(n_points):
            if random.random() < mutation_rate:
                individual[i] += np.random.normal(0, mutation_strength)
                individual[i] = np.clip(individual[i], 0, 1)
        return individual

    # Initialize population
    population = [create_individual() for _ in range(population_size)]
    population = [np.clip(p, 0, 1) * target_sum / np.sum(np.clip(p, 0, 1)) for p in population]
    population = [p for p in population if np.sum(p) > 0]

    for _ in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        # Select parents
        parents = np.argsort(fitness)[:population_size//2]
        parents = [population[i] for i in parents]
        # Create new population
        new_population = []
        while len(new_population) < population_size:
            parent1, parent2 = random.sample(parents, 2)
            child = crossover(parent1, parent2)
            child = mutate(child)
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = new_population

    # Evaluate final population
    final_fitness = [evaluate(ind) for ind in population]
    best_candidate = population[np.argmin(final_fitness)]
    best_c5 = np.min(final_fitness)
    best_h = best_candidate

    return best_h, best_c5, n_points
