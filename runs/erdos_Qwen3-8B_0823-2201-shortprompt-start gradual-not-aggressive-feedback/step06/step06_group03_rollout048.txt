import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Initial optimization with more iterations
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Genetic algorithm for global search
    population_size = 20
    generations = 50
    mutation_rate = 0.1
    elite_size = 2

    def crossover(parent1, parent2):
        child = np.copy(parent1)
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        child[indices] = parent2[indices]
        return child

    def mutate(individual):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] += np.random.normal(0, 0.05)
                individual[i] = np.clip(individual[i], 0, 1)
        return individual

    population = [np.random.rand(n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]

    for _ in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        # Select elite
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = [population[i] for i in elite_indices]
        # Create new population
        new_population = elite.copy()
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(elite, size=2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = new_population

    # Evaluate best from genetic search
    best_genetic = min(population, key=objective)
    best_c5_genetic = objective(best_genetic)

    # Compare and select best solution
    if best_c5_genetic < best_c5:
        best_h = best_genetic
        best_c5 = best_c5_genetic

    return best_h, best_c5, n_points
