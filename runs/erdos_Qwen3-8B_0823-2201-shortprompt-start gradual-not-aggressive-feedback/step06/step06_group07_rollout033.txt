import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize with given construction
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
        )
        return result.x

    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 50

    start_time = time.time()

        # Evaluate population
        fitness = [objective(h) for h in population]

        elite_idx = np.argmin(fitness)
        elite_h = population[elite_idx]
        elite_c5 = fitness[elite_idx]
        
        if elite_c5 < best_c5:
            best_h = elite_h
            best_c5 = elite_c5
        
        # Create new population
        new_population = []
        for _ in range(population_size):
            # Select parents
            parent1 = population[np.random.choice(len(population))]
            parent2 = population[np.random.choice(len(population))]

                        child[i] = parent1[i]

            child = np.clip(child, 0, 1)
            
            new_population.append(child)
        
        population = new_population
    
    return best_h, best_c5, n_points
