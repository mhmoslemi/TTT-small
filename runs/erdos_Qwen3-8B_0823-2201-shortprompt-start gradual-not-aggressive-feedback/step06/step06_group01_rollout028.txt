import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Reduced number of samples for faster computation
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial construction (can be modified or replaced)
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < 25 or i > 75:

            initial_h[i] = 0.0

        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
