        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])

        tournament_size = 5
        selected_indices = np.argsort(np.random.rand(population_size, tournament_size) * fitness[:, np.newaxis], axis=1)[:, 0]

        # Crossover: blend crossover
        offspring = np.zeros_like(population)
        for i in range(population_size):
            parent1 = selected[i]
            parent2 = np.random.choice(population)
            offspring[i] = (parent1 + parent2) / 2
        
        # Mutation: adaptive mutation based on fitness
        mutation_rate = 0.1 / (1 + np.exp(-fitness))
