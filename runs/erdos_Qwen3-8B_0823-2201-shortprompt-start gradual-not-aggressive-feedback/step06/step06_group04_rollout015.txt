import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Initial optimization with more iterations
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Coordinate descent with structured perturbations
    for _ in range(100):
        if time.time() - start_time >= budget_s - 1:
            break
        # Perturb individual points
        for i in range(n_points):
            perturb = np.random.uniform(-0.1, 0.1)
            new_h = best_h.copy()
            new_h[i] += perturb
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h)
            current_c5 = objective(new_h)
            if current_c5 < best_c5:
                best_h = new_h
                best_c5 = current_c5
        # Add a small random perturbation
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Optimization with tighter tolerance
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    return best_h, best_c5, n_points
