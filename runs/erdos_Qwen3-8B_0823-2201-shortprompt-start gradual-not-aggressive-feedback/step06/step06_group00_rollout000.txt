import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.15
    crossover_rate = 0.8
    generations = 50
    
    def genetic_algorithm():
        population = [np.random.rand(n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        best_h = h.copy()
        best_c5 = objective(best_h)
        
        for _ in range(generations):
            fitness = [objective(h) for h in population]
            idxs = np.argsort(fitness)
            
            # Select top half for next generation
            next_gen = population[idxs[:population_size//2]]
            
            # Create new individuals through crossover and mutation
            for _ in range(population_size//2):
                parent1, parent2 = random.choices(population, k=2)
                child = parent1.copy()
                if np.random.rand() < crossover_rate:
                    split = np.random.randint(1, n_points-1)
                    child[:split] = parent2[:split]
                if np.random.rand() < mutation_rate:
                    mutation = np.random.normal(0, 0.1, n_points)
                    child += mutation
                    child = np.clip(child, 0, 1)
                    child = child * target_sum / np.sum(child)
                next_gen.append(child)
            
            population = next_gen
            current_best = min(population, key=objective)
            current_c5 = objective(current_best)
            
            if current_c5 < best_c5:
                best_h = current_best
                best_c5 = current_c5
        
        return best_h, best_c5
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        # Random perturbation with adaptive scale
        perturb_scale = 0.15 * (1 - (time.time() - start_time)/budget_s)
        perturb = np.random.uniform(-perturb_scale, perturb_scale, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
