import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    def generate_candidate(parent1, parent2):
        child = np.copy(parent1)
        idx = np.random.choice(n_points)
        child[idx] = parent2[idx]
        child = np.clip(child, 0, 1)
        child = child * target_sum / np.sum(child)
        return child
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h = np.copy(initial_h)
    best_c5 = objective(best_h)
