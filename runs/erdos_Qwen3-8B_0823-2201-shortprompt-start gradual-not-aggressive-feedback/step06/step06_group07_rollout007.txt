import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize population
    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
        )
        return result.x

    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = [objective(h) for h in population]
        
        # Tournament selection
        selected = []
        for _ in range(population_size):
            candidates = np.random.choice(population_size, 5, replace=False)
            best_idx = candidates[np.argmin(fitness[candidates])]
            selected.append(population[best_idx])
        
        # Crossover
        new_population = []
        for i in range(population_size):
            parent1 = selected[i]
            parent2 = selected[(i+1)%population_size]
            child = np.zeros(n_points)
            for j in range(n_points):
                if np.random.rand() < 0.5:
                    child[j] = parent1[j]
                else:
                    child[j] = parent2[j]
            new_population.append(child)
        
        # Mutation
        for i in range(population_size):
            child = new_population[i]
            for j in range(n_points):
                if np.random.rand() < 0.1:
                    child[j] += np.random.uniform(-0.1, 0.1)
                    child[j] = np.clip(child[j], 0, 1)
            new_population[i] = child
        
        # Local refinement
        refined_population = []
        for h in new_population:
            refined_h = local_refinement(h)
            refined_population.append(refined_h)
        
        # Update population
        population = refined_population
        
        # Track best solution
        current_fitness = [objective(h) for h in population]
        best_idx = np.argmin(current_fitness)
        current_best_h = population[best_idx]
        current_best_c5 = current_fitness[best_idx]
        
        if current_best_c5 < best_c5:
            best_h = current_best_h
            best_c5 = current_best_c5
    
    return best_h, best_c5, n_points
