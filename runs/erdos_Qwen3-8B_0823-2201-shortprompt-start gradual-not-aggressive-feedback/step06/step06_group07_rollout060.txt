import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize with given construction
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
        )
        return result.x

    # Systematic phase and amplitude search
    def systematic_search(h):
        phase_shifts = np.linspace(0, 2*np.pi, 10)
        amplitudes = np.linspace(0.01, 0.1, 10)
        best_perturbed = h.copy()
        best_c5_perturbed = objective(best_perturbed)
        
        for phase in phase_shifts:
            for amp in amplitudes:
                perturb = np.sin(np.linspace(0, 2*np.pi, n_points) + phase) * amp
                perturbed = h + perturb
                perturbed = np.clip(perturbed, 0, 1)
                perturbed = perturbed * target_sum / np.sum(perturbed)
                refined_perturbed = local_refinement(perturbed)
                refined_c5 = objective(refined_perturbed)
                
                if refined_c5 < best_c5_perturbed:
                    best_perturbed = refined_perturbed
                    best_c5_perturbed = refined_c5
        
        return best_perturbed, best_c5_perturbed

    start_time = time.time()

    return best_h, best_c5, n_points
