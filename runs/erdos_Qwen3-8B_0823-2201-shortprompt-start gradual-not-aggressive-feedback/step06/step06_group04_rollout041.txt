import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.15
    generations = 10
    
    # Initial population
    population = [np.clip(initial_h + np.random.uniform(-0.1, 0.1, n_points), 0, 1) 
                 for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    for _ in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        
        # Local optimization on best individual
        result = opt.minimize(
            objective,
            population[best_idx],
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        population[best_idx] = result.x
        
        # Create new population through crossover and mutation
        new_population = []
        for i in range(population_size):
            parent1 = population[np.random.choice(population_size)]
            parent2 = population[np.random.choice(population_size)]
            child = (parent1 + parent2) / 2
            if np.random.rand() < mutation_rate:
                child += np.random.uniform(-0.1, 0.1, n_points)
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        population = new_population
    
    return best_h, best_c5, n_points
