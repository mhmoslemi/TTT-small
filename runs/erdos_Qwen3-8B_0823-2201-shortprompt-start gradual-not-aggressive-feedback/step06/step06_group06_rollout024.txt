import scipy.optimize as opt
import time

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def perturb(h, strength=0.05):
        return np.clip(h + np.random.uniform(-strength, strength, n_points), 0, 1) * target_sum / np.sum(h)
    
    def anneal():
        h = perturb(initial_h)
        temp = 1.0
        best_h = h.copy()
        best_c5 = objective(h)
        
        for _ in range(1000):
            temp *= 0.995
            if temp < 1e-5:
                break
            new_h = perturb(h, strength=0.1)
            new_c5 = objective(new_h)
            delta = new_c5 - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                h = new_h
                if new_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = new_c5
        return best_h, best_c5
    
    def refine(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h, best_c5 = anneal()

        result = opt.minimize(
            objective,
            perturbed_h,

            bounds=bounds,

            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
