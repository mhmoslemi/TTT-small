import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize with given construction
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
        )
        return result.x

    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    generations = 100

    # Initialize population
    population = [initial_h.copy() for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = [objective(h) for h in population]
        
        # Select parents
        parents = np.argsort(fitness)[:population_size//2]
        parents = [population[i] for i in parents]
        
        # Crossover
        offspring = []
        for _ in range(population_size//2):
            parent1 = random.choice(parents)
            parent2 = random.choice(parents)
            child = np.copy(parent1)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = parent2[i]
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            for j in range(n_points):
                if random.random() < mutation_rate:
                    offspring[i][j] += np.random.uniform(-0.1, 0.1)
                    offspring[i][j] = np.clip(offspring[i][j], 0, 1)
        
        # Combine population
        population = parents + offspring
        population = [h * target_sum / np.sum(h) for h in population]
        
        # Local refinement of best solution
        best_index = np.argmin(fitness)
        refined_h = local_refinement(population[best_index])
        refined_c5 = objective(refined_h)
        
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
    
    return best_h, best_c5, n_points
