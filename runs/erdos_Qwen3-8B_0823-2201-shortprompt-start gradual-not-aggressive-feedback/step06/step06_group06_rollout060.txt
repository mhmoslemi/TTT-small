import time

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def generate_population(size):
        pop = np.random.rand(size, n_points)
        pop = np.clip(pop, 0, 1)
        pop = pop * target_sum / np.sum(pop, axis=1, keepdims=True)
        return pop
    
    def evaluate_population(pop):
        scores = np.array([objective(h) for h in pop])
        return scores
    
    def tournament_selection(pop, scores, k=3):
        indices = np.random.choice(len(pop), k, replace=False)
        best_idx = np.argmin(scores[indices])
        return pop[indices[best_idx]]
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        idx = np.random.choice(len(child), 10, replace=False)
        child[idx] = parent2[idx]
        return child
    
    def adaptive_mutation(h, mutation_rate=0.05):
        perturb = np.random.normal(0, mutation_rate, len(h))
        mutated = h + perturb
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * target_sum / np.sum(mutated)
        return mutated
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
