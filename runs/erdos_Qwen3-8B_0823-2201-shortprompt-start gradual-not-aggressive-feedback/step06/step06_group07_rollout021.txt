import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize with given construction
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Create population
        population = [best_h + np.random.normal(0, 0.05, n_points) for _ in range(population_size)]
        population = [np.clip(p, 0, 1) * target_sum / np.sum(np.clip(p, 0, 1)) for p in population]
        
        # Evaluate population
        fitness = [objective(p) for p in population]
        best_idx = np.argmin(fitness)
        best_candidate = population[best_idx]
        best_fitness = fitness[best_idx]
        
        # Update best solution
        if best_fitness < best_c5:
            best_h = best_candidate
            best_c5 = best_fitness
        
        # Genetic operations
        for _ in range(generations):
            # Select parents
            parents = np.random.choice(population_size, size=2, replace=False)
            parent1 = population[parents[0]]
            parent2 = population[parents[1]]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, 0.05, n_points)
                child += mutation
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(np.clip(child, 0, 1))
            
            # Evaluate child
            child_fitness = objective(child)
            
            # Replace worst solution
            worst_idx = np.argmax(fitness)
            if child_fitness < fitness[worst_idx]:
                population[worst_idx] = child
                fitness[worst_idx] = child_fitness
    
    return best_h, best_c5, n_points
