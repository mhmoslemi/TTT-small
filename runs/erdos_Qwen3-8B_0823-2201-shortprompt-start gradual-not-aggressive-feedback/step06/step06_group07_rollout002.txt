import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize with given construction
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
        )
        return result.x

    # Simulated annealing with phase shifts
    def perturb_solution(h, phase_shift):
        perturb = np.sin(np.linspace(0, 2*np.pi, n_points) + phase_shift)
        perturbed = h + perturb * 0.05
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        return perturbed

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Local refinement
        refined_h = local_refinement(best_h)
        refined_c5 = objective(refined_h)
        
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
        
        # Systematic perturbation with random phase
        phase = np.random.uniform(0, 2*np.pi)
        perturbed_h = perturb_solution(best_h, phase)
        
        # Local refinement of perturbed solution
        refined_perturbed_h = local_refinement(perturbed_h)
        refined_perturbed_c5 = objective(refined_perturbed_h)
        
        if refined_perturbed_c5 < best_c5:
            best_h = refined_perturbed_h
            best_c5 = refined_perturbed_c5

    return best_h, best_c5, n_points
