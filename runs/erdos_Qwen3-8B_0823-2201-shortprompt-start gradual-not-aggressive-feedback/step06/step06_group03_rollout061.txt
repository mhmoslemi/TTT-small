import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Genetic algorithm for global search
    def genetic_search():
        population = [np.random.rand(n_points) for _ in range(50)]
        for _ in range(100):
            if time.time() - start_time >= budget_s - 1:
                break
            fitness = [objective(ind) for ind in population]
            best_idx = np.argmin(fitness)
            best_h_candidate = population[best_idx]
            
            # Mutation
            mutated = [np.clip(ind + np.random.normal(0, 0.05, n_points), 0, 1) for ind in population]
            mutated = [ind * target_sum / np.sum(ind) for ind in mutated]
            
            # Crossover
            new_population = []
            for i in range(len(population)):
                if random.random() < 0.8:
                    new_population.append(mutated[i])
                else:
                    new_population.append(population[i])
            
            # Replace worst with best candidate
            new_population[-1] = best_h_candidate
            population = new_population
        
        best_h_candidate = np.min(fitness)
        return np.argmin(fitness), population[np.argmin(fitness)]

    # Initial genetic search
    best_idx, best_pop = genetic_search()
    best_h = best_pop[best_idx]
    best_c5 = objective(best_h)

    # Local refinement with SLSQP
    while time.time() - start_time < budget_s - 1:
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
