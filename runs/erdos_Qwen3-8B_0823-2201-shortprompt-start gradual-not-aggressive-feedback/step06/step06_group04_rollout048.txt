import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use a larger number of points for better resolution
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with a better starting point
    h = np.zeros(n_points)
    h[:n_points//3] = 1.0  # Initial high values
    h[n_points//3:2*n_points//3] = 0.5  # Middle values
    h[2*n_points//3:] = 0.0  # Initial low values
    h = h * target_sum / np.sum(h)  # Normalize to meet the integral constraint
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Use a combination of global and local search
    for _ in range(5):  # Multiple rounds of optimization
        # Global search with random perturbations
        perturb = np.random.uniform(-0.2, 0.2, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Local optimization
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
    
    return best_h, best_c5, n_points
