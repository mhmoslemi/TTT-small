import time
import random

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize population
    population_size = 50
    population = np.random.rand(population_size, n_points)
    population = (population * target_sum / np.sum(population, axis=1, keepdims=True)).tolist()
    
    def evaluate(individual):
        return objective(np.array(individual))
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        
        # Select parents
        parents = np.array(population)[np.argsort(fitness)[:population_size//2]]
        
        # Crossover
        offspring = []
        for _ in range(population_size//2):
            parent1, parent2 = random.sample(parents, 2)
            child = np.zeros(n_points)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            for j in range(n_points):
                if random.random() < 0.1:
                    offspring[i][j] = np.clip(offspring[i][j] + np.random.uniform(-0.1, 0.1), 0, 1)
        
        # Evaluate offspring
        offspring_fitness = [evaluate(ind) for ind in offspring]
        
        # Selection
        population = parents.tolist() + offspring
        population.sort(key=lambda x: evaluate(x))
        population = population[:population_size]
        
        # Update best solution
        for ind in population:
            current_c5 = evaluate(ind)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = ind
    
    return best_h, best_c5, n_points
