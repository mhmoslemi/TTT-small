import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Initial optimization with SLSQP
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Genetic algorithm for global exploration
    population_size = 20
    generations = 5
    mutation_rate = 0.1
    mutation_strength = 0.05

    def create_individual():
        return np.random.uniform(0, 1, n_points)

    def evaluate(individual):
        individual = np.clip(individual, 0, 1)
        individual = individual * target_sum / np.sum(individual)
        corr = np.correlate(individual, 1 - individual, mode="full")
        return np.max(corr) * dx

    population = [create_individual() for _ in range(population_size)]
    population = [np.clip(p, 0, 1) for p in population]
    population = [p * target_sum / np.sum(p) for p in population]
    population = [p for p in population if np.sum(p) > 0]

    for _ in range(generations):
        if time.time() - start_time >= budget_s - 1:
            break
        fitness = [evaluate(ind) for ind in population]
        best_fitness = min(fitness)
        best_idx = np.argmin(fitness)
        best_individual = population[best_idx]

        if best_fitness < best_c5:
            best_h = best_individual
            best_c5 = best_fitness

        # Selection
        population = [np.random.choice(population, size=population_size, replace=False) for _ in range(population_size)]
        population = [np.clip(p, 0, 1) for p in population]
        population = [p * target_sum / np.sum(p) for p in population]
        population = [p for p in population if np.sum(p) > 0]

        # Mutation
        for i in range(population_size):
            if random.random() < mutation_rate:
                mutation = np.random.normal(0, mutation_strength, n_points)
                population[i] = np.clip(population[i] + mutation, 0, 1)
                population[i] = population[i] * target_sum / np.sum(population[i])

    return best_h, best_c5, n_points
