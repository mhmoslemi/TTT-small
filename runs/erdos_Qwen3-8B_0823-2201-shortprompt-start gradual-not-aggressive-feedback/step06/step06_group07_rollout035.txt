import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize with given construction
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    generations = 100
    
    def evolve_population(population):
        new_population = []
        for _ in range(population_size):
            # Select two parents
            parent1, parent2 = random.sample(population, 2)
            # Crossover
            child = np.zeros_like(parent1)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] = np.clip(child[i] + np.random.normal(0, 0.05), 0, 1)
            new_population.append(child)
        return new_population

    start_time = time.time()
    population = [best_h.copy() for _ in range(population_size)]
    
    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        # Find best solution
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Evolve new population
        population = evolve_population(population)
        
        # Local refinement of best solution
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
        )
        refined_h = result.x
        refined_c5 = objective(refined_h)
        
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
    
    return best_h, best_c5, n_points
