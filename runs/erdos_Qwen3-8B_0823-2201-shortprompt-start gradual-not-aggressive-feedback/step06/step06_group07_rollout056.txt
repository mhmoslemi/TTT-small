import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize population
    population_size = 30
    population = [initial_h.copy() for _ in range(population_size)]
    population = [np.random.rand(n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        
        # Select parents
        parents = [population[i] for i in np.argsort(fitness)[:population_size//2]]
        
        # Crossover
        new_population = []
        for _ in range(population_size):
            p1, p2 = random.sample(parents, 2)
            crossover_point = np.random.randint(1, n_points-1)
            child = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
            new_population.append(child)
        
        # Mutation
        for i in range(len(new_population)):
            for j in range(n_points):
                if np.random.rand() < 0.1:
                    new_population[i][j] = np.clip(new_population[i][j] + np.random.normal(0, 0.05), 0, 1)
        
        # Local refinement
        for i in range(len(new_population)):
            result = opt.minimize(
                objective,
                new_population[i],
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
            )
            new_population[i] = result.x
        
        # Update population
        population = new_population
        
        # Update best solution
        for h in population:
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
