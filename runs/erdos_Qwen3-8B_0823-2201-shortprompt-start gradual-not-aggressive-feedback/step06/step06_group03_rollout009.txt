import time

        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    best_c5 = objective(best_h)
    start_time = time.time()

    generations = 20
    mutation_rate = 0.15
    tournament_size = 5

    # Initialize population
    population = np.random.rand(population_size, n_points)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    population = np.clip(population, 0, 1)

    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        
        # Tournament selection
        tournament_indices = np.random.choice(population_size, size=tournament_size, replace=False)
        winners = tournament_indices[np.argmin(fitness[tournament_indices])]

        offspring = []

            parent1 = population[np.random.choice(population_size)]

                if np.random.rand() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    offspring[i][j] = np.random.rand()

        indices = np.argsort(population_fitness)

        # Local optimization on best candidates
        for i in range(min(5, population_size)):
