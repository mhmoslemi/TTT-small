import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 128
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def generate_initial_guess():
        patterns = [
            np.zeros(n_points),
            np.ones(n_points),
            np.concatenate([np.ones(n_points//2), np.zeros(n_points//2)]),
            np.concatenate([np.linspace(0, 1, n_points//2), np.linspace(1, 0, n_points//2)]),
            np.random.rand(n_points)
        ]
        for p in patterns:
            p = p * target_sum / np.sum(p)
            yield p
    
    best_h = None
    best_c5 = 1.0
    
    start_time = time.time()
    
    for initial_h in generate_initial_guess():
        def objective(h):
            h = h.reshape(n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5
        
        def constraint_sum(h):
            h = h.reshape(n_points)
            return np.sum(h) - target_sum
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = opt.minimize(
            objective,
            initial_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        
        if time.time() - start_time > budget_s - 1:
            break
    
    return best_h, best_c5, n_points
