import numpy as np
import scipy.optimize as opt
import time
import copy

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 30
    mutation_rate = 0.15
    
    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate population
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            # Select parents
            parent1 = population[np.random.choice(len(population))]
            parent2 = population[np.random.choice(len(population))]
            
            # Crossover
            child = np.zeros(n_points)
            for i in range(n_points):
                if np.random.random() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            
            # Mutation
            for i in range(n_points):
                if np.random.random() < mutation_rate:
                    child[i] += np.random.uniform(-0.1, 0.1)
                    child[i] = np.clip(child[i], 0, 1)
            
            # Normalize
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        # Replace old population
        population = new_population
    
    # Refine best solution with SLSQP
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
