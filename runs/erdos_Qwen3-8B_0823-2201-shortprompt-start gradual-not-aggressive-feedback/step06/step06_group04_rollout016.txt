import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Initial optimization with SLSQP
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Genetic algorithm for global search
    population_size = 20
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    for gen in range(10):
        if time.time() - start_time >= budget_s - 1:
            break
        # Evaluate population
        fitness = [objective(h) for h in population]
        # Select top performers
        top_indices = np.argsort(fitness)[:int(population_size/2)]
        top_population = [population[i] for i in top_indices]
        # Crossover and mutation
        new_population = []
        for _ in range(population_size):
            parent1, parent2 = random.choices(top_population, k=2)
            child = np.clip(parent1 * 0.5 + parent2 * 0.5, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = new_population
    
    # Evaluate best from genetic search
    best_genetic = min(population, key=objective)
    genetic_c5 = objective(best_genetic)
    
    # Final selection
    best_h = best_h if best_c5 < genetic_c5 else best_genetic
    best_c5 = best_c5 if best_c5 < genetic_c5 else genetic_c5
    
    return best_h, best_c5, n_points
