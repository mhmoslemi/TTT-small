import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize with given construction
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
        )
        return result.x

    # Genetic algorithm parameters
    pop_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 10

    def genetic_search():
        population = np.random.uniform(0, 1, (pop_size, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            
            # Select parents
            parents = population[np.argsort(fitness)[:pop_size//2]]
            
            # Generate offspring
            offspring = []
            for _ in range(pop_size - pop_size//2):
                parent1, parent2 = np.random.choice(parents, 2, replace=False)
                child = parent1 * np.random.rand(n_points) + parent2 * (1 - np.random.rand(n_points))
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
                offspring.append(child)
            
            population = np.vstack((parents, offspring))
        
        # Find best in population
        fitness = np.array([objective(ind) for ind in population])
        best_idx = np.argmin(fitness)
        return population[best_idx]

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Run genetic search
        genetic_h = genetic_search()
        genetic_c5 = objective(genetic_h)
        
        if genetic_c5 < best_c5:
            best_h = genetic_h
            best_c5 = genetic_c5
        
        # Local refinement
        refined_h = local_refinement(best_h)
        refined_c5 = objective(refined_h)
        
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5

    return best_h, best_c5, n_points
