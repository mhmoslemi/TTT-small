import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize with given construction
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    generations = 100
    
    def genetic_algorithm():
        population = np.random.rand(population_size, n_points)
        population = np.clip(population, 0, 1)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            
            # Select parents
            parents = population[np.argsort(fitness)[:population_size//2]]
            
            # Crossover
            offspring = []
            for _ in range(population_size//2):
                parent1, parent2 = np.random.choice(parents, 2, replace=False)
                crossover_point = np.random.randint(1, n_points-1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                offspring.append(child)
            
            # Mutate
            for i in range(len(offspring)):
                offspring[i] = np.clip(offspring[i] + np.random.normal(0, mutation_rate, n_points), 0, 1)
                offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
            
            # Replace population
            population = np.vstack([parents, np.array(offspring)])
            population = np.random.permutation(population)[:population_size]
            
            # Track best solution
            best_idx = np.argmin(fitness)
            if fitness[best_idx] < best_c5:
                best_h = population[best_idx].copy()
                best_c5 = fitness[best_idx]
        
        return best_h, best_c5

    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
        )
        return result.x

    start_time = time.time()
    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm()
    
    # Perform local refinement
    refined_h = local_refinement(best_h)
    refined_c5 = objective(refined_h)
    if refined_c5 < best_c5:
        best_h = refined_h
        best_c5 = refined_c5
    
    # Additional refinement
    for _ in range(3):

        # Local refinement of perturbed solution
        refined_perturbed_h = local_refinement(perturbed)
        refined_perturbed_c5 = objective(refined_perturbed_h)
        
        if refined_perturbed_c5 < best_c5:
            best_h = refined_perturbed_h
            best_c5 = refined_perturbed_c5
    
    return best_h, best_c5, n_points
