def run(seed=42, budget_s=20, **kwargs):

    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    population_size = 50

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = [objective(h) for h in population]
        
        # Select elites
        elite_indices = np.argsort(fitness)[:int(population_size/2)]
        elites = [population[i] for i in elite_indices]
        
        # Tournament selection for crossover
        new_population = []
        for _ in range(population_size):
            # Select two parents
            parent1, parent2 = random.choices(elites + population, k=2)
            # Crossover
            crossover_point = np.random.randint(1, n_points-1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            # Mutate
            mutation_rate = 0.1

            # Normalize
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        population = new_population
        # Update best solution
        for h in population:
            c5 = objective(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
