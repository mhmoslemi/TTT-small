import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def genetic_algorithm(pop_size, generations):
        population = np.random.uniform(0, 1, (pop_size, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            fitness = np.array([objective(ind) for ind in population])
            elite = population[np.argmin(fitness)]
            
            # Crossover
            new_population = []
            for _ in range(pop_size // 2):
                parent1, parent2 = np.random.choice(population, 2, replace=False)
                child1 = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                child2 = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                new_population.extend([child1, child2])
            
            # Mutation
            for i in range(pop_size):
                if np.random.rand() < 0.1:
                    mutation = np.random.uniform(-0.1, 0.1, n_points)
                    new_population[i] += mutation
                    new_population[i] = np.clip(new_population[i], 0, 1)
                    new_population[i] = new_population[i] * target_sum / np.sum(new_population[i])
            
            population = np.array(new_population)
            population = np.vstack([population, elite])
            population = np.unique(population, axis=0)
            
            if time.time() - start_time > budget_s - 5:
                break
        
        best_candidate = population[np.argmin([objective(ind) for ind in population])]
        return best_candidate

    # Run genetic algorithm
    best_h = genetic_algorithm(pop_size=30, generations=50)
    best_c5 = objective(best_h)

    # Local optimization
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
