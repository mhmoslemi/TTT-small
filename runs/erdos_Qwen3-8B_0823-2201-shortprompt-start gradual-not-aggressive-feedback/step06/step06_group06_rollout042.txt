import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def generate_individual():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        child[indices] = parent2[indices]
        child = np.clip(child, 0, 1)
        child = child * target_sum / np.sum(child)
        return child
    
    def mutate(h, mutation_rate=0.1):
        h = np.copy(h)
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.uniform(-0.1, 0.1)
                h[i] = np.clip(h[i], 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    population = [generate_individual() for _ in range(50)]
    best_h = min(population, key=objective)
    best_c5 = objective(best_h)
    
    while time.time() - start_time < budget_s:
        new_population = []
        for _ in range(len(population) // 2):
            parent1, parent2 = np.random.choice(population, size=2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population + [best_h for _ in range(len(population) - len(new_population))]
        
        for h in population:
            if objective(h) < best_c5:
                best_h = h
                best_c5 = objective(h)
        
        best_h = local_refinement(best_h)
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
