import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Genetic algorithm for global search
    def genetic_algorithm(pop_size=30, generations=100):
        population = np.random.uniform(0, 1, (pop_size, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            fitness = np.array([objective(ind) for ind in population])
            elite_idx = np.argsort(fitness)[:5]
            elites = population[elite_idx]
            
            # Crossover
            offspring = []
            for _ in range(pop_size - len(elites)):
                parents = np.random.choice(elites, size=2, replace=False)
                child = (parents[0] + parents[1]) / 2
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
                offspring.append(child)
            
            # Mutation
            for i in range(len(offspring)):
                mutation = np.random.normal(0, 0.05, n_points)
                offspring[i] = np.clip(offspring[i] + mutation, 0, 1)
                offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
            
            population = np.vstack((elites, offspring))
        
        best_idx = np.argmin([objective(ind) for ind in population])
        return population[best_idx]
    
    start_time = time.time()
    best_h = local_refinement(h)
    best_c5 = objective(best_h)
    
    # Run genetic algorithm
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        pop_size = max(10, int(remaining_time / 0.1))
        generations = max(1, int(remaining_time / 0.5))
        
        ga_h = genetic_algorithm(pop_size=pop_size, generations=generations)
        ga_c5 = objective(ga_h)
        
        if ga_c5 < best_c5:
            best_h = ga_h
            best_c5 = ga_c5
    
    return best_h, best_c5, n_points
