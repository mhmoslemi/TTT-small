import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Particle swarm optimization for global search
    particles = np.random.uniform(0, 1, (30, n_points))
    velocities = np.random.uniform(-0.1, 0.1, (30, n_points))
    personal_best = particles.copy()
    personal_best_values = np.array([objective(p) for p in particles])
    global_best = particles[np.argmin(personal_best_values)]
    global_best_value = np.min(personal_best_values)
    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Update velocities and positions
        r1, r2 = np.random.rand(30), np.random.rand(30)
        velocities = velocities * 0.7 + r1 * (personal_best - particles) + r2 * (global_best - particles)
        particles = np.clip(particles + velocities, 0, 1)
        particles = particles * target_sum / np.sum(particles, axis=1, keepdims=True)
        
        # Evaluate fitness
        fitness = np.array([objective(p) for p in particles])
        
        # Update personal best
        improved = fitness < personal_best_values
        personal_best[improved] = particles[improved]
        personal_best_values[improved] = fitness[improved]
        
        # Update global best
        if np.min(fitness) < global_best_value:
            global_best = particles[np.argmin(fitness)]
            global_best_value = np.min(fitness)
        
        # Update best solution
        if global_best_value < best_c5:
            best_h = global_best
            best_c5 = global_best_value

    # Local refinement with SLSQP
    start_time = time.time()
    temp = 1.0
    cooling_rate = 0.995
    while time.time() - start_time < budget_s:
        # Simulated annealing step
        perturb_scale = 0.2 * (1 - (time.time() - start_time) / budget_s)
        perturb = np.random.uniform(-perturb_scale, perturb_scale, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Optimization
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        # Acceptance criterion
        if current_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - current_c5) / temp):
            best_h = current_h
            best_c5 = current_c5
        
        temp *= cooling_rate
    
    return best_h, best_c5, n_points
