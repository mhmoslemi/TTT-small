import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize with given construction
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
        )
        return result.x

    # Genetic mutation
    def mutate_solution(h, mutation_rate=0.1):
        h = h.reshape(n_points)
        mutation = np.random.rand(n_points) < mutation_rate
        h[mutation] = np.random.rand(np.sum(mutation))
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h.flatten()

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Local refinement
        refined_h = local_refinement(best_h)

            best_h = refined_h

        mutated_h = mutate_solution(best_h)
        mutated_c5 = objective(mutated_h)

            best_h = mutated_h
            best_c5 = mutated_c5

    return best_h, best_c5, n_points
