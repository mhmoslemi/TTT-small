import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize with given construction
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
        )
        return result.x

    # Genetic algorithm parameters
    pop_size = 20
    mutation_rate = 0.1
    generations = 10

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Create population
        population = [np.random.rand(n_points) for _ in range(pop_size)]
        population.append(best_h)
        
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_fitness = fitness[best_idx]
        
        # Genetic operations
        for _ in range(generations):
            # Select parents
            parent_indices = np.argsort(fitness)[:2]
            parent1, parent2 = population[parent_indices[0]], population[parent_indices[1]]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] = np.random.rand()
            
            # Normalize
            child = child * target_sum / np.sum(child)
            child = np.clip(child, 0, 1)
            
            # Evaluate child
            child_fitness = objective(child)
            population.append(child)
            fitness.append(child_fitness)
            
            # Keep top performers
            population = np.array(population)
            fitness = np.array(fitness)
            population = population[np.argsort(fitness)[:pop_size+1]]
            fitness = fitness[np.argsort(fitness)[:pop_size+1]]
        
        # Check best in population
        current_best_idx = np.argmin(fitness)
        current_best = population[current_best_idx]
        current_c5 = fitness[current_best_idx]
        
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
        
        # Local refinement of best solution
        refined_h = local_refinement(best_h)
        refined_c5 = objective(refined_h)
        
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5

    return best_h, best_c5, n_points
