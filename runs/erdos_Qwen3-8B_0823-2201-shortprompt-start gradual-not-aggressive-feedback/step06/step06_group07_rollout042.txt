import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize with given construction
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
        )
        return result.x

    # Genetic algorithm for global search
    def genetic_search(population_size=20, generations=10):
        population = [np.random.rand(n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        population = [np.clip(h, 0, 1) for h in population]
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = [objective(h) for h in population]
            # Select parents
            parents = np.random.choice(population, size=population_size, p=np.array(fitness)/np.sum(fitness))
            # Crossover and mutation
            new_population = []
            for _ in range(population_size):
                parent1 = np.random.choice(parents)
                parent2 = np.random.choice(parents)
                child = (parent1 + parent2) / 2
                # Mutation
                mutation = np.random.normal(0, 0.05, n_points)
                child += mutation
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
                new_population.append(child)
            population = new_population
        
        # Find best in population
        fitness = [objective(h) for h in population]
        best_index = np.argmin(fitness)
        return population[best_index]

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Local refinement
        refined_h = local_refinement(best_h)
        refined_c5 = objective(refined_h)
        
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5

        genetic_h = genetic_search(population_size=10, generations=5)
        genetic_c5 = objective(genetic_h)
        
        if genetic_c5 < best_c5:
            best_h = genetic_h
            best_c5 = genetic_c5

        phase = np.random.uniform(0, 2*np.pi)

        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Local refinement of perturbed solution
        refined_perturbed_h = local_refinement(perturbed_h)
        refined_perturbed_c5 = objective(refined_perturbed_h)
        
        if refined_perturbed_c5 < best_c5:
            best_h = refined_perturbed_h
            best_c5 = refined_perturbed_c5

    return best_h, best_c5, n_points
