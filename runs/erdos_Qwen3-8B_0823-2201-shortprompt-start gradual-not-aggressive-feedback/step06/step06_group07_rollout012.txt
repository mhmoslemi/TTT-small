import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize with given construction
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
        )
        return result.x

    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.15
    crossover_rate = 0.8
    
    def genetic_algorithm(population):
        new_population = []
        for _ in range(population_size):
            # Select two parents
            parents = random.sample(population, 2)
            # Crossover
            if random.random() < crossover_rate:
                child = np.zeros_like(parents[0])
                split = int(n_points * 0.5)
                child[:split] = parents[0][:split]
                child[split:] = parents[1][split:]
            else:
                child = parents[0].copy()
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] = np.random.uniform(0, 1)
            # Normalize
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        return new_population

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Local refinement
        refined_h = local_refinement(best_h)
        refined_c5 = objective(refined_h)
        
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
        
        # Genetic algorithm
        population = [best_h + np.random.normal(0, 0.05, n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        population.append(best_h)
        
        new_population = genetic_algorithm(population)
        for candidate in new_population:
            candidate_c5 = objective(candidate)
            if candidate_c5 < best_c5:
                best_h = candidate
                best_c5 = candidate_c5

    return best_h, best_c5, n_points
