    def evaluate_individual(h):
        return objective(h)
    
    def select_parents(population, fitnesses):
        tournament = np.random.choice(len(population), size=5, replace=False)
        best_idx = tournament[np.argmin(fitnesses[tournament])]
        return population[best_idx].copy()
    
    def crossover(parent1, parent2):
        alpha = np.random.rand()
        return alpha * parent1 + (1 - alpha) * parent2
    
    def mutate(individual):
        mutation_rate = 0.05
        mutation_scale = 0.1
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] += np.random.uniform(-mutation_scale, mutation_scale)
                individual[i] = np.clip(individual[i], 0, 1)
        return individual * target_sum / np.sum(individual)
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    for generation in range(100):

            break
        
        # Evaluate population
        fitnesses = np.array([evaluate_individual(ind) for ind in population])
        
        # Select parents and create new population
        new_population = []
        for _ in range(population_size // 2):
            parent1 = select_parents(population, fitnesses)

            child1 = crossover(parent1, parent2)
