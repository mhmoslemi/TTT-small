import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.15
    generations = 100

    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]

    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        
        # Select parents
        parents = np.argsort(fitness)[:population_size//2]
        parents = [population[i] for i in parents]
        
        # Crossover
        offspring = []
        for _ in range(population_size//2):
            parent1 = random.choice(parents)
            parent2 = random.choice(parents)
            child = np.zeros(n_points)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            for j in range(n_points):
                if random.random() < mutation_rate:
                    offspring[i][j] += np.random.uniform(-0.15, 0.15)
                    offspring[i][j] = np.clip(offspring[i][j], 0, 1)
                    offspring[i][j] = offspring[i][j] * target_sum / np.sum(offspring[i])
        
        # Evaluate offspring
        offspring_fitness = [objective(h) for h in offspring]
        
        # Select next generation
        combined = population + offspring
        combined_fitness = fitness + offspring_fitness
        combined_idx = np.argsort(combined_fitness)
        population = [combined[i] for i in combined_idx[:population_size]]
        
        # Local refinement
        for i in range(len(population)):
            h = population[i]
            result = opt.minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-9}
            )
            population[i] = result.x
        
        # Update best solution
        for h in population:
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    return best_h, best_c5, n_points
