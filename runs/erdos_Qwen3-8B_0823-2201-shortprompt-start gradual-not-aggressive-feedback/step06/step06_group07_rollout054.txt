import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize with given construction
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
        )
        return result.x

    # Genetic algorithm parameters
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Create population
        population = [np.random.rand(n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        
        # Evaluate population
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h_candidate = population[best_idx]
        best_c5_candidate = fitness[best_idx]
        
        # Replace best solution if better
        if best_c5_candidate < best_c5:
            best_h = best_h_candidate
            best_c5 = best_c5_candidate
        
        # Evolutionary step
        for _ in range(generations):
            # Select parents
            parents = np.random.choice(population_size, size=population_size, replace=True)
            parents = [population[i] for i in parents]
            
            # Crossover
            offspring = []
            for i in range(0, population_size, 2):
                if np.random.rand() < crossover_rate:
                    p1, p2 = parents[i], parents[i+1]
                    child1 = np.concatenate((p1[:n_points//2], p2[n_points//2:]))
                    child2 = np.concatenate((p2[:n_points//2], p1[n_points//2:]))
                    offspring.extend([child1, child2])
                else:
                    offspring.append(parents[i])
            
            # Mutation
            for i in range(len(offspring)):
                if np.random.rand() < mutation_rate:
                    idx = np.random.randint(0, n_points)
                    offspring[i][idx] = np.random.rand()
            
            # Evaluate offspring
            offspring = [h * target_sum / np.sum(h) for h in offspring]
            fitness_offspring = [objective(h) for h in offspring]
            
            # Replace worst performers
            combined = list(zip(offspring, fitness_offspring))
            combined.sort(key=lambda x: x[1])
            population = [p for p, f in combined[:population_size]]
    
    return best_h, best_c5, n_points
