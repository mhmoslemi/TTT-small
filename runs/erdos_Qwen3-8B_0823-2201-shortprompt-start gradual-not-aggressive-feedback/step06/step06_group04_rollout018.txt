import numpy as np
import scipy.optimize as opt
import time
from copy import deepcopy

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Genetic algorithm for global search
    def genetic_search():
        population = [np.random.uniform(0, 1, n_points) for _ in range(30)]
        population = [h * target_sum / np.sum(h) for h in population]
        for _ in range(20):
            if time.time() - start_time > budget_s - 1:
                break
            fitness = [objective(h) for h in population]
            best_idx = np.argmin(fitness)
            best_h = population[best_idx]
            best_c5 = fitness[best_idx]
            
            # Create new population through crossover and mutation
            new_population = []
            for _ in range(len(population)):
                parent1 = np.random.choice(population)
                parent2 = np.random.choice(population)
                child = np.zeros(n_points)
                for i in range(n_points):
                    if np.random.rand() < 0.5:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
                # Mutation
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        child[i] = np.random.uniform(0, 1)
                child = child * target_sum / np.sum(child)
                new_population.append(child)
            population = new_population
        
        return np.min(fitness), best_h

    # Run genetic search
    best_c5, best_h = genetic_search()

    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        return result.x

    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
