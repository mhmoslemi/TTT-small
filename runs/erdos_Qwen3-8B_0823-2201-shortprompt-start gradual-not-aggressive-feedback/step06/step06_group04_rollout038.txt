import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Genetic algorithm parameters
    population_size = 50
    generations = 20
    mutation_rate = 0.15
    elite_size = 5

    # Initial population
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = [p * target_sum / np.sum(p) for p in population]
    population = [np.clip(p, 0, 1) for p in population]
    population = np.array(population)

    for _ in range(generations):
        # Evaluate population
        fitness = np.array([objective(p) for p in population])
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()

        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = population[elite_indices]

        # Generate offspring
        offspring = []
        for _ in range(population_size - elite_size):
            parent1 = elites[np.random.choice(elite_size)]
            parent2 = elites[np.random.choice(elite_size)]
            child = (parent1 + parent2) / 2
            # Apply mutation
            mutate_mask = np.random.rand(n_points) < mutation_rate
            child[mutate_mask] += np.random.uniform(-0.15, 0.15, np.sum(mutate_mask))
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            offspring.append(child)
        population = np.vstack((elites, offspring))

        # Local refinement with SLSQP
        if time.time() - start_time < budget_s - 1:
            for i in range(elite_size):
                perturbed_h = population[i].copy()
                perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
                result = opt.minimize(
                    objective,
                    perturbed_h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 50, 'ftol': 1e-9}
                )
                if result.success:
                    population[i] = result.x
                    current_c5 = objective(result.x)
                    if current_c5 < best_c5:
                        best_h = result.x
                        best_c5 = current_c5

    return best_h, best_c5, n_points
