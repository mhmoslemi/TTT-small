import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    bounds = [(0, 1) for _ in range(n_points)]

    def evaluate(solution):
        return objective(solution)

    population_size = 50
    mutation_rate = 0.1
    elite_size = 5
    generations = 100
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    def mutate(solution):
        mutation = np.random.normal(0, 0.05, n_points)
        mutated = np.clip(solution + mutation, 0, 1)
        mutated = mutated * target_sum / np.sum(mutated)
        return mutated

    def crossover(parent1, parent2):
        crossover_point = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child = child * target_sum / np.sum(child)
        return child

    population = [initial_h.copy() for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]

    for _ in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        fitness = [evaluate(h) for h in population]
        sorted_indices = np.argsort(fitness)
        elites = [population[i] for i in sorted_indices[:elite_size]]
        
        new_population = elites.copy()
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(elites, size=2, replace=False)
            child = crossover(parent1, parent2)
            if np.random.random() < mutation_rate:
                child = mutate(child)
            new_population.append(child)
        
        population = new_population
        current_best = min(population, key=evaluate)
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5

    return best_h, best_c5, n_points
