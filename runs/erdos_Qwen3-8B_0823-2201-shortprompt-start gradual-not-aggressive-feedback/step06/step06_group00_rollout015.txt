    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize population
    population_size = 50
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    def tournament_selection(pop, fitnesses, tournament_size=5):
        indices = np.random.choice(len(pop), tournament_size)
        best_idx = np.argmin(fitnesses[indices])
        return pop[indices[best_idx]]
    
    def crossover(parent1, parent2):
        alpha = np.random.rand(n_points)
        return alpha * parent1 + (1 - alpha) * parent2
    
    def mutation(individual, mutation_rate=0.1):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] += np.random.normal(0, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        individual = individual * target_sum / np.sum(individual)
        return individual
    
    start_time = time.time()
    best_h = np.array(initial_h, dtype=np.float64)
    best_c5 = objective(best_h)
    
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        fitnesses = np.array([objective(ind) for ind in population])
        best_idx = np.argmin(fitnesses)
        best_c5 = fitnesses[best_idx]
        best_h = population[best_idx].copy()
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1 = crossover(parent1, parent2)
            child1 = mutation(child1)
            child2 = crossover(parent1, parent2)
            child2 = mutation(child2)
            new_population.append(child1)
            new_population.append(child2)
        
        population = np.array(new_population)
    
    return best_h, best_c5, n_points
