import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize with given construction
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
        )
        return result.x

    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 10

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Create population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        for i in range(population_size):
            population[i] = np.clip(population[i], 0, 1)
            population[i] = population[i] * target_sum / np.sum(population[i])
        
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        best_index = np.argmin(fitness)
        best_candidate = population[best_index]
        best_fitness = fitness[best_index]
        
        # Update best solution
        if best_fitness < best_c5:
            best_h = best_candidate
            best_c5 = best_fitness
        
        # Genetic algorithm
        for _ in range(generations):
            # Selection
            selected = np.random.choice(population_size, size=population_size, p=fitness/np.sum(fitness))
            parents = [population[i] for i in selected]
            
            # Crossover
            offspring = []
            for i in range(population_size//2):
                parent1 = parents[2*i]
                parent2 = parents[2*i+1]
                if np.random.rand() < crossover_rate:
                    child = np.random.choice([parent1, parent2], size=n_points)
                else:
                    child = np.random.uniform(0, 1, n_points)
                offspring.append(child)
            
            # Mutation
            for i in range(len(offspring)):
                offspring[i] = np.clip(offspring[i], 0, 1)
                offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
                if np.random.rand() < mutation_rate:
                    mask = np.random.rand(n_points) < 0.1
                    offspring[i][mask] = np.random.uniform(0, 1)
            
            # Replace population
            population = offspring
        
        # Local refinement of best candidate
        refined_h = local_refinement(best_h)
        refined_c5 = objective(refined_h)
        
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5

    return best_h, best_c5, n_points
