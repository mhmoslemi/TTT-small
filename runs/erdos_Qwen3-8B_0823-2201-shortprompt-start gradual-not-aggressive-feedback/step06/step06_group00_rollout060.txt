import scipy.optimize as opt
import time

        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    def fitness_func(individual):
        return -objective(individual)
    
    # Genetic algorithm
    def genetic_algorithm():
        population = np.random.rand(population_size, n_points)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        population = np.clip(population, 0, 1)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([fitness_func(ind) for ind in population])
            
            # Select elite
            elite_indices = np.argsort(fitness)[-elite_size:]
            elite = population[elite_indices]
            
            # Select parents
            parents = population[np.argsort(fitness)[-population_size//2:]]
            
            # Crossover
            offspring = []
            for _ in range(population_size - elite_size):
                parent1, parent2 = np.random.choice(parents, 2, replace=False)
                child = np.concatenate((parent1[:n_points//2], parent2[n_points//2:]))
                offspring.append(child)
            
            # Mutation
            for i in range(len(offspring)):

                    if np.random.rand() < mutation_rate:
                        offspring[i][j] = np.random.rand() * target_sum / np.sum(offspring[i])
                        offspring[i][j] = np.clip(offspring[i][j], 0, 1)

            population = np.vstack((elite, offspring))

            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
