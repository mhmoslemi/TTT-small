import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize with given construction
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
        )
        return result.x

    # Genetic mutation
    def mutate(solution, mutation_rate=0.1):
        mutation = np.random.normal(0, mutation_rate, size=len(solution))
        mutated = np.clip(solution + mutation, 0, 1)
        mutated = mutated * target_sum / np.sum(mutated)
        return mutated

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Local refinement
        refined_h = local_refinement(best_h)
        refined_c5 = objective(refined_h)
        
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5

        mutated_h = mutate(best_h)
        refined_mutated_h = local_refinement(mutated_h)
        refined_mutated_c5 = objective(refined_mutated_h)
        
        if refined_mutated_c5 < best_c5:
            best_h = refined_mutated_h
            best_c5 = refined_mutated_c5

    return best_h, best_c5, n_points
