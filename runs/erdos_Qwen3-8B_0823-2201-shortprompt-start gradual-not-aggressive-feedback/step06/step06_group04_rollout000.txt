import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use 100 points for faster computation
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial guess based on uniform distribution
    initial_h = np.full(n_points, 0.5)
    best_h = initial_h.copy()
    best_c5 = calculate_c5(best_h, dx)
    
    start_time = time.time()
    
    # Optimization with more iterations and tighter tolerance
    result = opt.minimize(
        lambda h: calculate_c5(h, dx),
        best_h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxfun': 1000, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = calculate_c5(best_h, dx)
    
    # Simulated annealing for global exploration
    while time.time() - start_time < budget_s - 1:
        # Perturb the solution
        perturb = np.random.normal(0, 0.1, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Local optimization
        result = opt.minimize(
            lambda h: calculate_c5(h, dx),
            perturbed_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxfun': 100, 'ftol': 1e-9}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = calculate_c5(current_h, dx)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points

def calculate_c5(h, dx):
    h = np.clip(h, 0, 1)
    h = h * (len(h) / 2) / np.sum(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx
