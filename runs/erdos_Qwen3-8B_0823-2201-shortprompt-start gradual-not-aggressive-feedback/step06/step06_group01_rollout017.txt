import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        indices = np.random.choice(n_points, size=n_points // 2, replace=False)
        child[indices] = parent2[indices]
        return child
    
    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < 0.1:
                h[i] += np.random.uniform(-0.1, 0.1)
                h[i] = np.clip(h[i], 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    best_h = h
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Genetic algorithm
    population_size = 10
    population = [mutate(h) for _ in range(population_size)]
    
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        # Select parents
        parents = np.argsort(fitness)[:population_size//2]
        parents = np.array(population)[parents]
        # Crossover and mutate
        new_population = []
        for i in range(population_size//2):
            p1, p2 = np.random.choice(parents, size=2, replace=False)
            child = crossover(p1, p2)
            child = mutate(child)
            new_population.append(child)
        population = new_population + np.array(population).tolist()
        # Local refinement
        for i in range(len(population)):
            if time.time() - start_time > budget_s:
                break
            refined = local_refinement(population[i])
            if objective(refined) < objective(population[i]):
                population[i] = refined
    
    # Final refinement
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        h = np.array(population[np.argmin([objective(ind) for ind in population])])
        h = local_refinement(h)
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
