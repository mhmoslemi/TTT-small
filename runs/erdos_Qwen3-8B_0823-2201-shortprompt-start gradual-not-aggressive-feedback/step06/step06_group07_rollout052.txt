import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initialize with given construction
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    pop_size = 20
    mutation_rate = 0.15
    generations = 20
    
    def genetic_search():
        population = np.random.rand(pop_size, n_points)
        population = np.array([h * target_sum / np.sum(h) for h in population])
        population = np.clip(population, 0, 1)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(h) for h in population])
            
            # Select parents
            parents = population[np.argsort(fitness)[:pop_size//2]]
            
            # Crossover
            offspring = []
            for _ in range(pop_size//2):
                p1, p2 = np.random.choice(parents, 2, replace=False)
                child = np.zeros(n_points)
                for i in range(n_points):
                    if np.random.rand() < 0.5:
                        child[i] = p1[i]
                    else:
                        child[i] = p2[i]
                offspring.append(child)
            
            # Mutation
            for i in range(len(offspring)):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        offspring[i][j] += np.random.rand() * 0.1 - 0.05
                        offspring[i][j] = np.clip(offspring[i][j], 0, 1)
            
            # Replace population
            population = np.array(offspring)
            population = np.array([h * target_sum / np.sum(h) for h in population])
            population = np.clip(population, 0, 1)
        
        # Find best in population
        fitness = np.array([objective(h) for h in population])
        best_idx = np.argmin(fitness)
        return population[best_idx]

    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-12, 'eps': 1e-8}
        )
        return result.x

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Genetic search
        candidate_h = genetic_search()
        candidate_c5 = objective(candidate_h)
        
        if candidate_c5 < best_c5:
            best_h = candidate_h
            best_c5 = candidate_c5
        
        # Local refinement of best solution
        refined_h = local_refinement(best_h)
        refined_c5 = objective(refined_h)
        
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5

    return best_h, best_c5, n_points
