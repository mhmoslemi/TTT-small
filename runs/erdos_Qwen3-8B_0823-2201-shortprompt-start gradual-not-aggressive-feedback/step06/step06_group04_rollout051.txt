import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Initial optimization with more iterations
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Local search with perturbations and more refined optimization
    while time.time() - start_time < budget_s - 1:
        # Genetic algorithm approach
        population = np.random.uniform(0, 1, (20, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for p in population:
            p = np.clip(p, 0, 1)
            p = p * target_sum / np.sum(p)
            current_c5 = objective(p)
            if current_c5 < best_c5:
                best_h = p
                best_c5 = current_c5
        
        # Additional SLSQP refinement
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        best_h = result.x
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
