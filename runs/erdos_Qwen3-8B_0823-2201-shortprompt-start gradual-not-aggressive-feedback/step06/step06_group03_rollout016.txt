import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Initial optimization with SLSQP
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Genetic algorithm for global search
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    elite_size = 2

    def fitness(individual):
        return objective(individual)

    def crossover(parent1, parent2):
        child = parent1.copy()
        for i in range(len(child)):
            if random.random() < 0.5:
                child[i] = parent2[i]
        return child

    def mutate(individual):
        for i in range(len(individual)):
            if random.random() < mutation_rate:
                individual[i] = np.clip(individual[i] + np.random.normal(0, 0.05), 0, 1)
        return individual

    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]

    for _ in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        fitness_scores = [fitness(ind) for ind in population]
        ranked = sorted(zip(fitness_scores, population), key=lambda x: x[0])
        best_individual = ranked[0][1]
        if fitness(best_individual) < best_c5:
            best_h = best_individual
            best_c5 = fitness(best_individual)
        elite = [ind for _, ind in ranked[:elite_size]]
        next_population = elite
        while len(next_population) < population_size:
            parent1, parent2 = random.choices(ranked, k=2)
            child = crossover(parent1[1], parent2[1])
            child = mutate(child)
            child = child * target_sum / np.sum(child)
            next_population.append(child)
        population = next_population

    return best_h, best_c5, n_points
