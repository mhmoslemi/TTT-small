import random

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    best_c5 = objective(best_h)
    start_time = time.time()

    result = opt.minimize(

        method='SLSQP',

    population_size = 20
    generations = 10
    mutation_rate = 0.1
    mutation_strength = 0.05

    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]

    for _ in range(generations):
        if time.time() - start_time > budget_s - 1:
            break

        # Create offspring

            parent1 = random.choice(population)
