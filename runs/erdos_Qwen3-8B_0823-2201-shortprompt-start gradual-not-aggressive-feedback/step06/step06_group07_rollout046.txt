    crossover_rate = 0.8

    def random_solution():

    def mutate(solution):
        mutation = np.random.rand(n_points) < mutation_rate

        solution = solution * target_sum / np.sum(solution)
        return solution
    
    # Function to apply crossover
    def crossover(parent1, parent2):
        mask = np.random.rand(n_points) < crossover_rate
        child = np.where(mask, parent1, parent2)
        child = np.clip(child, 0, 1)
        child = child * target_sum / np.sum(child)
        return child
    
    # Evolutionary search
    population = [random_solution() for _ in range(population_size)]
