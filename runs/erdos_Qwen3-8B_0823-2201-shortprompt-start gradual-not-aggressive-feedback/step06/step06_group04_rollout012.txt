import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Genetic algorithm for global search
    def genetic_search(population_size=20, generations=10, mutation_rate=0.1):
        population = np.random.uniform(0, 1, (population_size, n_points))
        for _ in range(generations):
            if time.time() - start_time > budget_s - 5:
                break
            fitness = np.array([objective(ind) for ind in population])
            elite = population[np.argmin(fitness)]
            population = np.vstack([elite] + [np.random.uniform(0, 1, n_points) for _ in range(population_size - 1)])
            for i in range(population_size):
                if random.random() < mutation_rate:
                    population[i] += np.random.uniform(-0.1, 0.1, n_points)
                    population[i] = np.clip(population[i], 0, 1)
                    population[i] = population[i] * target_sum / np.sum(population[i])
            population = np.array([ind * target_sum / np.sum(ind) for ind in population])
        return np.min(fitness), population[np.argmin(fitness)]
    
    # Run genetic search
    best_c5, best_pop = genetic_search()
    best_h = best_pop
    
    # Local optimization with SLSQP
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
