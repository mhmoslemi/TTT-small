import random

def run(seed=42, budget_s=20, **kwargs):

    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Genetic algorithm parameters
    pop_size = 20
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initial population
    def create_individual():
        return np.random.rand(n_points) * 2 - 1  # Random values between -1 and 1
    
    def evolve_population(population):
        new_population = []
        for _ in range(pop_size):
            # Select parents
            parent1 = random.choice(population)
            parent2 = random.choice(population)
            
            # Crossover
            if random.random() < crossover_rate:
                child = np.zeros(n_points)
                for i in range(n_points):
                    if random.random() < 0.5:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
            else:
                child = np.random.rand(n_points) * 2 - 1  # Random initialization
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] += np.random.normal(0, 0.1)
                    child[i] = np.clip(child[i], -1, 1)
            
            # Scale to maintain integral constraint
            child = child * target_sum / np.sum(np.clip(child, 0, 1))
            child = np.clip(child, 0, 1)
            
            new_population.append(child)
        return new_population
    
    # Initialize population
    population = [create_individual() for _ in range(pop_size)]
    population = [h * target_sum / np.sum(np.clip(h, 0, 1)) for h in population]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        
        # Select best individual
        best_idx = np.argmin(fitness)
        best_individual = population[best_idx]
        current_c5 = fitness[best_idx]
        
        if current_c5 < best_c5:
            best_h = best_individual.copy()
            best_c5 = current_c5
        
        # Evolve population
        population = evolve_population(population)
    
    return best_h, best_c5, n_points
