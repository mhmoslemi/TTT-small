        spread = np.std(h)
        return c5 + 0.1 * (1 - spread)

        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def simulated_annealing():

        cooling_rate = 0.995

        max_iter = 500
        
        while temp > 1e-3 and iterations < max_iter:

                if np.random.rand() < np.exp((current_c5 - candidate_c5) / temp):

                    current_c5 = candidate_c5

            h,
            method='SLSQP',
