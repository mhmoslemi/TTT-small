import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def breed(parent1, parent2):
        child = parent1.copy()
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parent2[i]
        return child

    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < 0.1:
                h[i] += np.random.uniform(-0.2, 0.2)
                h[i] = np.clip(h[i], 0, 1)
        return h

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    population_size = 20
    population = [initial_h.copy() for _ in range(population_size)]
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            parent1, parent2 = random.sample(population, 2)
            child1 = breed(parent1, parent2)
            child1 = mutate(child1)
            child1 = child1 * target_sum / np.sum(child1)
            new_population.append(child1)
            child2 = breed(parent1, parent2)
            child2 = mutate(child2)
            child2 = child2 * target_sum / np.sum(child2)
            new_population.append(child2)
        population = new_population

    return best_h, best_c5, n_points
