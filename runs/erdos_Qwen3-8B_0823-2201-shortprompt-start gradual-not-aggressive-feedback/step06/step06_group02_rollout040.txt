import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm with tournament selection
    population_size = 30
    generations = 50
    start_time = time.time()

    def fitness(h):
        return objective(h)

    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    for i in range(population_size):
        population[i] = np.clip(population[i], 0, 1)
        population[i] = population[i] * target_sum / np.sum(population[i])
    
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitnesses = [fitness(h) for h in population]
        best_idx = np.argmin(fitnesses)
        best_h = population[best_idx]
        best_c5 = fitnesses[best_idx]
        
        # Selection and crossover
        new_population = []
        for _ in range(population_size):
            parent1 = np.random.choice(population)
            parent2 = np.random.choice(population)
            crossover_point = np.random.randint(n_points)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = new_population
    
    # Local refinement with SLSQP
    start_time = time.time()
    temp = 1.0
    cooling_rate = 0.995
    while time.time() - start_time < budget_s:
        # Simulated annealing step
        perturb_scale = 0.15 * (1 - (time.time() - start_time) / budget_s)
        perturb = np.random.uniform(-perturb_scale, perturb_scale, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Optimization
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        # Acceptance criterion
        if current_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - current_c5) / temp):
            best_h = current_h
            best_c5 = current_c5
            print(f"Improved bound: {best_c5}")
        
        temp *= cooling_rate
    
    return best_h, best_c5, n_points
