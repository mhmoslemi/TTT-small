import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def local_optimize(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        return result.x

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Generate initial population
    population = [initial_h.copy() for _ in range(50)]
    for i in range(50):
        population[i] = np.random.uniform(0, 1, n_points)
        population[i] = population[i] * target_sum / np.sum(population[i])

    for gen in range(200):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        
        # Select top performers
        top_indices = np.argsort(fitness)[:10]
        top_h = [population[i] for i in top_indices]
        
        # Generate new population through crossover and mutation
        new_population = []
        for _ in range(50):
            # Select two parents
            parent1, parent2 = random.choices(top_h, k=2)
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            # Mutation
            mutation_rate = 0.1
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.choice([-0.1, 0.1])
                    child[i] = np.clip(child[i], 0, 1)
            # Normalize
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        population = new_population
        
        # Local optimization of top solutions
        for i in range(len(top_h)):
            population[i] = local_optimize(population[i])
            population[i] = np.clip(population[i], 0, 1)
            population[i] = population[i] * target_sum / np.sum(population[i])
        
        # Update best solution
        for ind in population:
            current_c5 = objective(ind)
            if current_c5 < best_c5:
                best_h = ind.copy()
                best_c5 = current_c5

    return best_h, best_c5, n_points
