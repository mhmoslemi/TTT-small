        corr = np.correlate(h, 1 - h, mode="full")

        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        
        # Select parents
        parents = population[np.argsort(fitness)[:population_size//2]]
        
        # Crossover
        offspring = []
        for _ in range(population_size//2):
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            crossover_point = np.random.randint(1, n_points-1)
            child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            offspring.append(child)
        
        # Mutate
        for i in range(len(offspring)):
            mutation_rate = 0.1
            mutation_step = 0.05
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    offspring[i][j] += np.random.uniform(-mutation_step, mutation_step)

        # Normalize offspring
        for i in range(len(offspring)):
            offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
        
        # Create new population
        population = np.vstack((parents, offspring))
