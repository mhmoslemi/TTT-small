import numpy as np
import scipy.optimize as opt
import time
from genetic_algorithm import GeneticAlgorithm

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with SLSQP
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Genetic algorithm for global search
    ga = GeneticAlgorithm(
        objective=objective,
        bounds=bounds,
        population_size=20,
        generations=10,
        mutation_rate=0.1,
        crossover_rate=0.8,
        elite_size=2,
        time_limit=budget_s - (time.time() - start_time)
    )
    ga.run()
    best_h = ga.best_solution
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
