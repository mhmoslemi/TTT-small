import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize population
    population = [initial_h.copy() for _ in range(50)]
    for p in population:
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitness = [objective(p) for p in population]
        
        # Select parents
        parents = random.choices(population, weights=[1/f for f in fitness], k=10)
        
        # Crossover
        children = []
        for i in range(0, 10, 2):
            p1 = parents[i]
            p2 = parents[i+1]
            child1 = np.concatenate([p1[:n_points//2], p2[n_points//2:]])
            child2 = np.concatenate([p2[:n_points//2], p1[n_points//2:]])
            children.append(child1)
            children.append(child2)
        
        # Mutate
        mutated = []
        for child in children:
            mutation = np.random.normal(0, 0.05, n_points)
            mutated_child = np.clip(child + mutation, 0, 1)
            mutated_child = mutated_child * target_sum / np.sum(mutated_child)
            mutated.append(mutated_child)
        
        # Evaluate mutated children
        child_fitness = [objective(c) for c in mutated]
        
        # Create new population
        population = parents + mutated
        population = [np.clip(p, 0, 1) for p in population]
        population = [p * target_sum / np.sum(p) for p in population]
        
        # Update best solution
        for p, f in zip(population, child_fitness):
            if f < best_c5:
                best_h = p.copy()
                best_c5 = f
    
    # Local optimization on best solution
    def local_objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    result = opt.minimize(
        local_objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = local_objective(best_h)
    
    return best_h, best_c5, n_points
