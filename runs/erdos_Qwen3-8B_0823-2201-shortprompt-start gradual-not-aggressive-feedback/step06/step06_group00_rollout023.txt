import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = 100  # Reduced from 47 to 100 for better resolution
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial guess with random perturbation
    initial_h = np.random.rand(n_points)
    initial_h = initial_h * target_sum / np.sum(initial_h)

        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Simulated annealing with adaptive cooling and diversity constraint
    def simulated_annealing():
        T = 1.0
        cool_rate = 0.995
        max_iter = 500
        best_h = initial_h.copy()
        best_c5 = objective(best_h)
        diversity = np.zeros(n_points)
        
        for _ in range(max_iter):
            T *= cool_rate
            if T < 1e-5:
                break
            perturb = np.random.normal(0, 0.15, n_points)
            perturbed = np.clip(initial_h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            
            # Diversity constraint: avoid solutions too similar to best
            diff = np.abs(perturbed - best_h)
            diversity = np.minimum(diversity + diff, 1.0)
            if np.mean(diversity) < 0.5:
                continue
                
            current_c5 = objective(perturbed)
            
            if current_c5 < best_c5 or np.random.rand() < np.exp((current_c5 - best_c5)/T):
                best_h = perturbed
                best_c5 = current_c5
        
        return best_h, best_c5
    
    # Local refinement with SLSQP

            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h, best_c5 = simulated_annealing()

    best_c5 = objective(best_h)

    while time.time() - start_time < budget_s:

        perturb = np.random.uniform(-perturb_scale, perturb_scale, n_points)

            perturbed_h,

        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
