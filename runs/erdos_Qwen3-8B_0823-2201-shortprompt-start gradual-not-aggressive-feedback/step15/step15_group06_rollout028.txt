import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Gradient-based refinement
    for _ in range(3):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        h = result.x
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    # Genetic algorithm with decreasing mutation rate
    for stage in range(3):
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break
        
        # Generate offspring with mutations
        offspring = []
        for _ in range(10):
            if np.random.rand() < 0.1:  # Mutation probability
                perturb = np.random.uniform(-0.05 * (0.9 ** stage), 0.05 * (0.9 ** stage), n_points)
                perturbed_h = h + perturb
                perturbed_h = np.clip(perturbed_h, 0, 1)

                offspring.append(perturbed_h)
            else:
                offspring.append(h.copy())
        
        # Evaluate offspring
        for candidate in offspring:
            candidate = candidate * target_sum / np.sum(candidate)
            current_c5 = objective(candidate)
            if current_c5 < best_c5:
                best_h = candidate.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
