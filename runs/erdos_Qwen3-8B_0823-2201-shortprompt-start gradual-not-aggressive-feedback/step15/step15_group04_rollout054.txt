import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    # Start with a coarser grid
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial guess with a simple alternating pattern
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            initial_h[i] = 1.0
        elif i % 3 == 1:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 0.0
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    
    # Multi-stage optimization with adaptive tolerances and structured perturbations
    for stage in range(5):
        # Gradient-based refinement
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 150, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        h = result.x
        best_c5 = objective(h)
        
        # Adaptive perturbation with structured exploration
        remaining_time = budget_s - (time.time() - time.time())
        for perturb_size in np.logspace(-3, -1, num=5, base=10):
            if time.time() - time.time() >= budget_s:
                break
            # Focus perturbations on regions contributing most to overlap
            corr = np.correlate(h, 1 - h, mode="full")
            max_corr_idx = np.argmax(corr)
            perturb = np.zeros(n_points)
            perturb[max_corr_idx % n_points] = -perturb_size
            perturb[(max_corr_idx + n_points) % n_points] = perturb_size
            perturbed_h = h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            
            result = opt.minimize(
                objective,
                perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = objective(current_h)
            
            if current_c5 < best_c5:
                h = current_h
                best_c5 = current_c5
    
    return h, best_c5, n_points
