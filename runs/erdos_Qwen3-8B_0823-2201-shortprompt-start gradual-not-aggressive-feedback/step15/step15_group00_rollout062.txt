import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Multi-stage optimization with adaptive perturbation and genetic algorithm
    for stage in range(2):
        # Gradient-based refinement with tighter tolerances
        def local_refinement(h):
            result = opt.minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-9}
            )
            return result.x
        
        h = local_refinement(h)
        current_c5 = objective(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

            break

                break

            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            
            result = opt.minimize(
                objective,
                perturbed_h,

            if current_c5 < best_c5:

                best_c5 = current_c5
    
    # Genetic algorithm for global exploration
    def genetic_algorithm(h):
        population_size = 20
        generations = 10
        mutation_rate = 0.1
        crossover_rate = 0.8
        
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        
        for _ in range(generations):
            if time.time() - start_time >= budget_s:
                break
            
            # Evaluate fitness
            fitness = [objective(h) for h in population]
            best_idx = np.argmin(fitness)
            best_fitness = fitness[best_idx]
            
            # Select parents
            parents = np.random.choice(population, size=population_size, p=fitness / np.sum(fitness))
            
            # Crossover
            offspring = []
            for i in range(population_size):
                if np.random.rand() < crossover_rate:
                    parent1 = parents[i]
                    parent2 = parents[np.random.randint(0, population_size)]
                    child = parent1 * 0.5 + parent2 * 0.5

                    child = parents[i]

            # Replace population
            population = offspring
        
        # Select the best solution from the final population
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        return population[best_idx]
    
    # Run genetic algorithm
    genetic_h = genetic_algorithm(h)
    genetic_c5 = objective(genetic_h)
    
    if genetic_c5 < best_c5:
        best_h = genetic_h.copy()
        best_c5 = genetic_c5
    
    return best_h, best_c5, n_points
