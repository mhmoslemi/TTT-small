import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Local optimization refinement
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-8}
    )
    h = result.x
    current_c5 = objective(h)
    
    if current_c5 < best_c5:
        best_h = h.copy()
        best_c5 = current_c5
    
    # Simulated annealing for global exploration
    temp = 1.0
    cooling_rate = 0.995
    max_iter = 1000
    
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        if temp < 1e-5:
            break
        
        # Generate a perturbation

        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Evaluate the perturbed solution

            best_h = current_h.copy()
            best_c5 = current_c5

            if np.random.rand() < prob:

                best_c5 = current_c5
        
        temp *= cooling_rate
