import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    start_time = time.time()
    
    # Phase 1: Global exploration with differential evolution
    def de_objective(h_flat):
        h = h_flat.reshape(n_points)
        return objective(h)
    
    de_bounds = [(0, 1) for _ in range(n_points)]
    result_de = opt.differential_evolution(
        de_objective,
        bounds=de_bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=15,
        tol=1e-10,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        disp=False,
        seed=seed
    )
    h = result_de.x
    
    # Phase 2: Local refinement with SLSQP
    def slsqp_objective(h):
        return objective(h)
    
    result_slsqp = opt.minimize(
        slsqp_objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-10}
    )
    h = result_slsqp.x
    
    # Phase 3: Adaptive perturbation with decreasing magnitude
    remaining_time = budget_s - (time.time() - start_time)
    for perturb_size in np.logspace(-4, -1, num=5, base=10):
        if time.time() - start_time >= budget_s:
            break
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            slsqp_objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 80, 'ftol': 1e-10, 'gtol': 1e-10}

        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < objective(h):
            h = current_h
    
    return h, objective(h), n_points
