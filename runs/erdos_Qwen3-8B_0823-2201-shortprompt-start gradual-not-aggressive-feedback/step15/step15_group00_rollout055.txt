import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Gradient-based refinement
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-9}
    )
    h = result.x
    current_c5 = objective(h)
    
    if current_c5 < best_c5:
        best_h = h.copy()
        best_c5 = current_c5
    
    # Genetic algorithm for global search
    def genetic_algorithm(pop_size=20, generations=50):
        population = np.random.uniform(0, 1, (pop_size, n_points))
        population = population * target_sum / np.sum(population, axis=1)[:, np.newaxis]
        
        for gen in range(generations):
            if time.time() - start_time >= budget_s:
                break
                
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            
            # Select parents
            parent_indices = np.argsort(fitness)[:pop_size//2]
            parents = population[parent_indices]
            
            # Crossover
            offspring = []
            for _ in range(pop_size//2):
                parent1, parent2 = np.random.choice(parents, size=2, replace=False)
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                offspring.append(child)
            
            # Mutation
            for i in range(len(offspring)):
                offspring[i] = np.clip(offspring[i] + np.random.uniform(-0.01, 0.01, n_points), 0, 1)
                offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
            
            # Combine population
            population = np.vstack((parents, offspring))
            population = np.unique(population, axis=0)
            
            # Keep best solution
            best_idx = np.argmin(fitness)
            if fitness[best_idx] < best_c5:
                best_h = population[best_idx].copy()
                best_c5 = fitness[best_idx]
        
        return best_h, best_c5
    
    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm()
    
    return best_h, best_c5, n_points
