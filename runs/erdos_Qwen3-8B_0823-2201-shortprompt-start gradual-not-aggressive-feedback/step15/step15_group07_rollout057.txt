import scipy.optimize as opt
import time

    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Main optimization loop
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(h)
    
    # Optimization parameters
    temp = 1.0
    cooling_rate = 0.995
    iterations = 0
    max_iterations = int(budget_s / 0.01)  # ~2000 iterations

        if time.time() - start_time >= budget_s:
            break
        
        # Gradient-based refinement
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-9}
        )

            objective,
