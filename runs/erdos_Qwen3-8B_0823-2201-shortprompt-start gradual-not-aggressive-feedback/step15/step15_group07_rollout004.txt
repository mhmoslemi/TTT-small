import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Main optimization loop
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    elite_size = 2
    
    for gen in range(generations):
        if time.time() - start_time >= budget_s:
            break
        
        # Create population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        
        # Select elite
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = [population[i] for i in elite_indices]
        
        # Generate next generation
        next_gen = elite.copy()
        while len(next_gen) < population_size:
            # Select parents
            parent1 = np.random.choice(elite)
            parent2 = np.random.choice(elite)
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation = np.random.uniform(-0.05, 0.05, n_points)
                child += mutation
                child = np.clip(child, 0, 1)
            
            # Normalize
            child = child * target_sum / np.sum(child)
            next_gen.append(child)
        
        population = next_gen
    
    # Refine best solution with gradient descent
    best_h = np.min(population, axis=0)

    )
