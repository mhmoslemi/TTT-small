import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 47  # Start from the initial construction
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Multi-stage optimization with global search and refinement
    best_h = h.copy()
    best_c5 = objective(h)
    elapsed = 0
    
    for stage in range(3):
        # Gradient-based refinement with tighter tolerances
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        best_h = result.x
        best_c5 = objective(best_h)
        elapsed = time.time() - time.time()
        
        # Adaptive perturbation with decreasing magnitude
        for perturb_size in np.logspace(-4, -1, num=5, base=10):
            if elapsed >= budget_s:
                break
            perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
            perturbed_h = best_h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            
            result = opt.minimize(
                objective,
                perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 80, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = objective(current_h)
            
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
            elapsed = time.time() - time.time()
    
    return best_h, best_c5, n_points
