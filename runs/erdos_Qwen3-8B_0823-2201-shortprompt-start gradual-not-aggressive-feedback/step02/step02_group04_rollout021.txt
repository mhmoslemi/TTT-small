import scipy.optimize as optimize
import time
import random

    best_n = n_points
    time_left = budget_s

    def objective(x, n):
        h = x.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * 2.0 / n)

    def constraint(x, n):
        h = x.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        return np.sum(h) - n / 2

    def evolve_population(pop, n, generations=50, mutation_rate=0.1):
        population = np.array(pop)
        for _ in range(generations):
            for i in range(len(population)):
                mutated = population[i] * (1 + np.random.normal(0, mutation_rate, n))
                mutated = np.clip(mutated, 0, 1)
                mutated = mutated * (n / 2) / np.sum(mutated)
                population = np.vstack([population, mutated])
            population = np.unique(population, axis=0)
            population = population[:len(population)//2]
        return population

    for n in [100, 80, 60, 50, 40, 30]:
        time_start = time.time()
        population_size = min(20, n)
        population = np.random.rand(population_size, n)
        population = population * (n / 2) / np.sum(population, axis=1, keepdims=True)
        population = np.clip(population, 0, 1)

        for _ in range(5):
            population = evolve_population(population, n)
            for ind in population:
                bounds = [(0, 1) for _ in range(n)]
                cons = [{'type': 'eq', 'fun': lambda x: constraint(x, n)}]
                result = optimize.minimize(lambda x: objective(x, n), ind, method='L-BFGS-B', bounds=bounds, constraints=cons)
                if result.success:
                    h_opt = result.x
                    h_opt = np.clip(h_opt, 0, 1)

                        best_n = n
