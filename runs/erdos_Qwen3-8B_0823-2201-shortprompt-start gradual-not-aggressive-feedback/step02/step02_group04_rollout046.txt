import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * 2.0 / n_points
    best_n = n_points
    time_left = budget_s

    # Genetic algorithm parameters
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 5
    time_left /= 2  # Allocate half budget for GA

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * (2.0 / n_points))

    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(n_points):
            if random.random() < crossover_rate:
                child[i] = parent2[i]
        return child

    def mutate(individual):
        for i in range(n_points):
            if random.random() < mutation_rate:
                individual[i] += np.random.normal(0, 0.05)
                individual[i] = np.clip(individual[i], 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        return individual

    # Initialize population
    population = [create_individual() for _ in range(population_size)]
    population = [np.clip(p, 0, 1) for p in population]
    population = [p * (n_points / 2) / np.sum(p) for p in population]

    for generation in range(generations):
        if time_left <= 0:
            break
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        # Create next generation
        next_gen = elites.copy()
        while len(next_gen) < population_size:
            parent1, parent2 = random.choices(population, k=2)
            child = crossover(parent1, parent2)
            child = mutate(child)
            next_gen.append(child)
        population = next_gen
        # Track best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx]
            best_n = n_points
        time_left -= 0.1  # Approximate time per generation

    return best_h, best_c5, best_n
