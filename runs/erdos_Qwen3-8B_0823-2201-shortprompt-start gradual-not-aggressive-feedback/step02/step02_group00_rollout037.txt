import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds
from genetic_optimization import GeneticOptimizer

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    # Genetic algorithm for global search
    optimizer = GeneticOptimizer(
        objective=objective,
        bounds=bounds,
        constraint_sum=constraint_sum,
        population_size=20,
        generations=20,
        mutation_rate=0.1,
        crossover_rate=0.8,
        elite_size=2,
        max_time=budget_s - time.time()
    )
    
    genetic_result = optimizer.optimize()
    genetic_h = genetic_result['best_individual']
    genetic_h = np.clip(genetic_h, 0, 1)
    genetic_h = genetic_h * target_sum / np.sum(genetic_h)
    genetic_c5 = objective(genetic_h)
    
    if genetic_c5 < best_c5:
        best_h = genetic_h
        best_c5 = genetic_c5
    
    return best_h, best_c5, n_points
