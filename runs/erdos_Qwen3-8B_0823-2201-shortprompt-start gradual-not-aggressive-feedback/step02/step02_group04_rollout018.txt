import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * 2.0 / n_points
    best_n = n_points
    time_left = budget_s

    def evaluate(individual, n):
        h = individual.copy()
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * 2.0 / n)

    def evolve_population(pop, n, pop_size=50, mutation_rate=0.1):
        new_pop = []
        for _ in range(pop_size):
            parent = random.choice(pop)
            child = parent.copy()
            for i in range(n):
                if random.random() < mutation_rate:
                    child[i] += np.random.uniform(-0.1, 0.1)
                    child[i] = np.clip(child[i], 0, 1)
            new_pop.append(child)
        return new_pop

    for n in [50, 40, 30, 20]:
        time_start = time.time()
        population = [np.random.uniform(0, 1, n) for _ in range(50)]
        population = [h * (n / 2) / np.sum(h) for h in population]
        for _ in range(100):
            if time.time() - time_start > budget_s:
                break
            fitness = [evaluate(ind, n) for ind in population]
            best_idx = np.argmin(fitness)
            best_c5_current = fitness[best_idx]
            if best_c5_current < best_c5:
                best_c5 = best_c5_current
                best_h = population[best_idx].copy()
                best_n = n
            population = evolve_population(population, n)
        time_used = time.time() - time_start
        time_left -= time_used
        if time_left <= 0:
            break

    return best_h, best_c5, best_n
