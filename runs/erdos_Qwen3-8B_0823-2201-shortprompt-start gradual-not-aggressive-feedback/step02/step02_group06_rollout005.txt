from scipy.optimize import minimize, Bounds

import time

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def objective(h):
        return evaluate(h)

        return np.sum(h) - n_points / 2
    
    # Optimization with SLSQP

    cons = [{'type': 'eq', 'fun': constraint_integral}]
