import scipy.optimize as optimize
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * 2.0 / n_points
    best_n = n_points
    time_left = budget_s

    # Try with a smaller number of points (faster)
    for n in [100, 80, 60, 50, 40, 30]:
        time_start = time.time()
        h = initial_h.copy()
        h = np.resize(h, n)
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        dx = 2.0 / n

        def objective(x):
            h = x.reshape(-1)
            h = np.clip(h, 0, 1)
            h = h * (n / 2) / np.sum(h)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr * dx)

        def constraint(x):
            h = x.reshape(-1)
            h = np.clip(h, 0, 1)
            h = h * (n / 2) / np.sum(h)
            return np.sum(h) - n / 2

        cons = [{'type': 'eq', 'fun': constraint}]
        bounds = [(0, 1) for _ in range(n)]

        # Try with L-BFGS-B
        result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons)
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n / 2) / np.sum(h_opt)
            corr = np.correlate(h_opt, 1 - h_opt, mode="full")
            c5 = np.max(corr * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n = n
        time_used = time.time() - time_start
        time_left -= time_used
        if time_left <= 0:
            break

    # Perform a local refinement
    h_refine = best_h.copy()
    for _ in range(5):
        corr = np.correlate(h_refine, 1 - h_refine, mode="full")
        max_corr_idx = np.argmax(corr)
        shift = max_corr_idx - len(h_refine) + 1
        if shift < 0:
            shift = 0
        if shift >= len(h_refine):
            shift = len(h_refine) - 1
        h_refine[shift] = np.clip(h_refine[shift] + 0.05, 0, 1)

            best_c5 = c5_refine
            best_h = h_refine

    return best_h, best_c5, best_n
