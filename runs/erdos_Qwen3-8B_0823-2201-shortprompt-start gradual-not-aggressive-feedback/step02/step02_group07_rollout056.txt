import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial guess: alternating pattern
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            initial_h[i] = 1.0
        elif i % 3 == 1:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 0.0
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 20
    mutation_rate = 0.1
    
    # Initialize population
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    best_h = initial_h
    best_c5 = objective(initial_h)
    
    start_time = time.time()
    
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        
        # Find the best individual
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_best_c5 = fitness[best_idx]
        
        if current_best_c5 < best_c5:
            best_h = current_best
            best_c5 = current_best_c5
        
        # Select parents
        parents = population[np.argsort(fitness)][:int(population_size/2)]
        
        # Crossover
        offspring = []
        for _ in range(population_size - int(population_size/2)):
            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            child = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    offspring[i][j] = np.random.uniform(0, 1)
        
        # Evaluate offspring
        offspring_fitness = np.array([objective(ind) for ind in offspring])
        
        # Replace worst individuals
        combined = np.vstack((population, offspring))
        combined_fitness = np.concatenate((fitness, offspring_fitness))
        indices = np.argsort(combined_fitness)
        population = combined[indices[:population_size]]
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    # Refinement with local search
    result = opt.minimize(
        objective,
        best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
