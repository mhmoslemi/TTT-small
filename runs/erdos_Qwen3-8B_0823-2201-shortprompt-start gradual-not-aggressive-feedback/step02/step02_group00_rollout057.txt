import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def evaluate(h):
        return objective(h)
    
    # Initial population
    population = [initial_h.copy() for _ in range(20)]
    population = [np.clip(h, 0, 1) * target_sum / np.sum(np.clip(h, 0, 1)) for h in population]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_best_c5 = fitness[best_idx]
        
        if current_best_c5 < best_c5:
            best_h = current_best
            best_c5 = current_best_c5
        
        # Create new population
        new_population = []
        
        # Elitism
        new_population.append(current_best)
        
        # Crossover
        for _ in range(10):
            parent1, parent2 = random.choices(population, k=2)
            child = np.zeros_like(parent1)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            new_population.append(child)
        
        # Mutation
        for _ in range(5):
            h = np.random.rand(n_points) * target_sum
            h = np.clip(h, 0, 1) * target_sum / np.sum(np.clip(h, 0, 1))
            new_population.append(h)
        
        # Local optimization
        for h in new_population:
            result = opt.minimize(
                objective,
                h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 20, 'ftol': 1e-8}
            )
            new_population[-1] = result.x
        
        # Update population
        population = new_population
    
    return best_h, best_c5, n_points
