import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def crossover(parent1, parent2):
        split = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:split], parent2[split:]])
        return child
    
    def mutate(h):
        idx = np.random.randint(n_points)
        h[idx] += np.random.uniform(-0.1, 0.1)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    population = [initial_h.copy() for _ in range(20)]
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate(ind) for ind in population]
        parents = [population[i] for i in np.argsort(fitness)[:10]]
        
        new_population = []
        for _ in range(10):
            p1, p2 = random.sample(parents, 2)
            child = crossover(p1, p2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
        current_best = min(population, key=evaluate)
        current_c5 = evaluate(current_best)
        
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
