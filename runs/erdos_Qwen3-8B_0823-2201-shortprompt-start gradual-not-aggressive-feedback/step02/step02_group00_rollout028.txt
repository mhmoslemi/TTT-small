import scipy.optimize as opt
import time
import random

        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm for global search
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    elite_size = 2

    def generate_individual():
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        return h

    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(n_points):
            if random.random() < 0.5:
                child[i] = parent2[i]
        child = np.clip(child, 0, 1)
        child = child * target_sum / np.sum(child)
        return child

    def mutate(individual):
        for i in range(n_points):
            if random.random() < mutation_rate:
                individual[i] += np.random.uniform(-0.05, 0.05)
                individual[i] = np.clip(individual[i], 0, 1)
        individual = individual * target_sum / np.sum(individual)
        return individual

    population = [generate_individual() for _ in range(population_size)]
    population = [np.clip(p, 0, 1) * target_sum / np.sum(p) for p in population]

        while len(next_generation) < population_size:
            parent1 = random.choice(parents)
            parent2 = random.choice(parents)
            child = crossover(parent1, parent2)
            child = mutate(child)
            next_generation.append(child)
        population = next_generation

    # Find best solution from genetic algorithm
    fitness = [objective(p) for p in population]
    best_index = np.argmin(fitness)
    best_h = population[best_index]

        best_c5 = best_c5

    return best_h, best_c5, n_points
