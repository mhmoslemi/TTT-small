        for i in range(n_points):
            if random() < 0.5:
                child[i] = parent2[i]
        return child

            if random() < 0.1:
                individual[i] += uniform(-0.1, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        return individual
