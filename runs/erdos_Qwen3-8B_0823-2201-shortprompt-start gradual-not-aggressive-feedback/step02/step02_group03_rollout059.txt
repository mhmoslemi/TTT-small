import numpy as np
import scipy.optimize as optimize
import time
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    best_c5 = 0.38092
    best_h = initial_h.copy()
    best_n = n_points
    time_left = budget_s

    def evaluate(individual, n, dx):
        h = np.clip(individual, 0, 1)
        h = h * (n / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx),

    def cx_pbh(ind1, ind2):
        size = min(len(ind1), len(ind2))
        offspring = np.empty(size)
        for i in range(size):
            if np.random.random() < 0.5:
                offspring[i] = ind1[i]
            else:
                offspring[i] = ind2[i]
        return offspring, offspring

    def mut_pbd(individual, indpb):
        for i in range(len(individual)):
            if np.random.random() < indpb:
                individual[i] = np.random.uniform(0, 1)
        return individual,

    # Gradient-based refinement
    for n in [70, 80, 90]:
        if time_left <= 0:
            break
        time_start = time.time()
        dx = 2.0 / n
        h_init = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(initial_h)), initial_h)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n / 2) / np.sum(h_init)
        
        bounds = [(0, 1) for _ in range(n)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2}]
        
        result = optimize.minimize(lambda x: np.max(np.correlate(x, 1 - x, mode="full") * dx),
                                 h_init, method='L-BFGS-B', bounds=bounds, constraints=cons,
                                 tol=1e-6, options={'maxiter': 100})
        
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n / 2) / np.sum(h_opt)
            corr = np.correlate(h_opt, 1 - h_opt, mode="full")
            c5 = np.max(corr * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n = n
        time_used = time.time() - time_start
        time_left -= time_used

    # Genetic algorithm exploration
    if time_left > 0:
        time_start = time.time()
        n = 80
        dx = 2.0 / n
        creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
        creator.create("Individual", np.ndarray, fitness=creator.FitnessMin)

        toolbox = base.Toolbox()
        toolbox.register("eval", evaluate, n=n, dx=dx)
        toolbox.register("individual", tools.initIterate, creator.Individual, lambda: np.random.uniform(0, 1, n))
        toolbox.register("population", tools.initRepeat, list, toolbox.individual)
        toolbox.register("mate", cx_pbh)
        toolbox.register("mutate", mut_pbd, indpb=0.1)
        toolbox.register("select", tools.selTournament, tournsize=3)

        pop = toolbox.population(n=50)
        algorithms.eaSimple(pop, toolbox, cxpb=0.5, mutpb=0.1, ngen=20, 
                           stats=None, halting条件=lambda pop: time.time() - time_start > time_left)
        
        for ind in pop:
            ind.fitness.values = toolbox.eval(ind)
            if ind.fitness.values[0] < best_c5:
                best_c5 = ind.fitness.values[0]
                best_h = np.clip(ind, 0, 1)
                best_h = best_h * (n / 2) / np.sum(best_h)
        time_used = time.time() - time_start
        time_left -= time_used

    return best_h, best_c5, best_n
