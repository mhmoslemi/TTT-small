import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = np.zeros(n_points)
    initial_h[::3] = 1.0  # Initial guess: periodic pattern
    initial_h = initial_h * target_sum / np.sum(initial_h)

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def perturb_solution(h, perturbation=0.05):
        h = h.copy()
        h += np.random.uniform(-perturbation, perturbation, size=h.shape)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    elapsed = 0

    start_time = time.time()
    while elapsed < budget_s:
        # Gradient-based optimization
        result = opt.minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 20, 'ftol': 1e-8}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        current_c5 = objective(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5

        # Simulated annealing-style perturbation
        best_h = perturb_solution(best_h, perturbation=0.03)

        # SLSQP refinement
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 20, 'ftol': 1e-8}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        current_c5 = objective(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5

        elapsed = time.time() - start_time

    return best_h, best_c5, n_points
