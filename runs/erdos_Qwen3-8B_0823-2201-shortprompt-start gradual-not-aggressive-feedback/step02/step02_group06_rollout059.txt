import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different n_points values
    n_points_list = [400, 500, 600]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 500
    
    for n_points in n_points_list:
        if time.time() - start_time >= budget_s:
            break
        dx = 2.0 / n_points
        # Start from initial_h_values if available, otherwise uniform
        if hasattr(initial_h_values, '__array__'):
            initial_h = np.resize(initial_h_values, (n_points,))[:n_points]
        else:
            initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        h = initial_h.copy()
        
        def evaluate(h):
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            corr = correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5
        
        def objective(h):
            return evaluate(h)
        
        def constraint_integral(h):
            return np.sum(h) - n_points / 2
        
        bounds = Bounds(0, 1)
        cons = [{'type': 'eq', 'fun': constraint_integral}]
        
        # Initial optimization with SLSQP
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        h_candidate = result.x
        
        # Local greedy improvement
        for _ in range(200):
            if time.time() - start_time >= budget_s:
                break
            h = h_candidate.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            corr = correlate(h, 1 - h, mode="full")
            max_index = np.argmax(corr)
            if max_index < len(corr) - 1:
                shift = max_index + 1
                if shift < len(h):
                    h[shift] = np.clip(h[shift] - 0.01, 0, 1)
            h_candidate = h
        
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        current_c5 = evaluate(h_candidate)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_candidate
            best_n_points = n_points
    
    return (best_h, best_c5, best_n_points)
