import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * 2.0 / n_points
    best_n = n_points
    time_left = budget_s

    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 2

    for n in [50, 40, 30, 25, 20]:
        if time_left <= 0:
            break
        time_start = time.time()
        
        # Initialize population
        population = []
        for _ in range(population_size):
            h = np.random.rand(n) * 2 - 1  # Random values between -1 and 1
            h = np.clip(h, 0, 1)
            h = h * (n / 2) / np.sum(h)
            population.append(h)
        
        for gen in range(generations):
            if time_left <= 0:
                break
            
            # Evaluate fitness
            fitness = []
            for h in population:
                corr = np.correlate(h, 1 - h, mode="full")
                c5 = np.max(corr * (2.0 / n))
                fitness.append(c5)
            
            # Select elite
            elite_indices = np.argsort(fitness)[:elite_size]
            elite = [population[i] for i in elite_indices]
            
            # Select parents
            parents = []
            for _ in range(population_size - elite_size):
                # Tournament selection
                tournament = random.sample(range(len(population)), 3)
                parent = population[tournament[np.argmin(fitness[tournament])]]
                parents.append(parent)
            
            # Crossover
            offspring = []
            for i in range(0, len(parents), 2):
                if i + 1 < len(parents) and random.random() < crossover_rate:
                    parent1 = parents[i]
                    parent2 = parents[i + 1]
                    child1 = np.concatenate((parent1[:n//2], parent2[n//2:]))
                    child2 = np.concatenate((parent2[:n//2], parent1[n//2:]))
                    offspring.extend([child1, child2])
                else:
                    offspring.append(parents[i])
            
            # Mutation
            for i in range(len(offspring)):
                offspring[i] = np.array(offspring[i])
                for j in range(n):
                    if random.random() < mutation_rate:
                        offspring[i][j] = np.clip(offspring[i][j] + np.random.normal(0, 0.05), 0, 1)
                        offspring[i][j] = offspring[i][j] * (n / 2) / np.sum(offspring[i])
            
            # Create new population
            population = elite + offspring
            population = [np.clip(h, 0, 1) for h in population]
            population = [h * (n / 2) / np.sum(h) for h in population]
            
            # Update best solution
            for h in population:
                corr = np.correlate(h, 1 - h, mode="full")
                c5 = np.max(corr * (2.0 / n))
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
                    best_n = n
        
        time_used = time.time() - time_start
        time_left -= time_used

    return best_h, best_c5, best_n
