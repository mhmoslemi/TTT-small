import scipy.optimize as optimize
import time

def run(seed=42, budget_s=20, **kwargs):

    time_left = budget_s

            h = x.reshape(-1)
            h = np.clip(h, 0, 1)
            h = h * (n / 2) / np.sum(h)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr * 2.0 / n)

        def constraint(x):
            h = x.reshape(-1)
            h = np.clip(h, 0, 1)
            h = h * (n / 2) / np.sum(h)
            return np.sum(h) - n / 2

        cons = [{'type': 'eq', 'fun': constraint}]
        bounds = [(0, 1) for _ in range(n)]

        # Try with SLSQP
        result = optimize.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons)
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)

                best_n = n

        if time_left <= 0:
