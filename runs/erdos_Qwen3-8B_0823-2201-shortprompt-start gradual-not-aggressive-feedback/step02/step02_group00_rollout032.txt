import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm for global search
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    elite_size = 2

    def genetic_search():
        population = np.random.uniform(0, 1, (population_size, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            fitness = np.array([objective(ind) for ind in population])
            elite_indices = np.argsort(fitness)[:elite_size]
            elite = population[elite_indices]
            
            # Selection and crossover
            selected = np.random.choice(population_size, population_size, p=fitness/fitness.sum())
            offspring = np.zeros((population_size, n_points))
            for i in range(population_size):
                parent1 = population[selected[i]]
                parent2 = population[np.random.choice(selected)]
                offspring[i] = (parent1 + parent2) / 2
                
            # Mutation
            for i in range(population_size):
                for j in range(n_points):
                    if random.random() < mutation_rate:
                        offspring[i, j] = np.clip(offspring[i, j] + np.random.uniform(-0.05, 0.05), 0, 1)
                offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
            
            population = np.vstack((elite, offspring))
        
        best_index = np.argmin([objective(ind) for ind in population])
        return population[best_index]

    best_h = genetic_search()
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
