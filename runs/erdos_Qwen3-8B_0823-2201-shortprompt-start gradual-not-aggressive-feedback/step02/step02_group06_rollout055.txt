from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
import time

def run(seed=42, budget_s=20, **kwargs):

    start_time = time.time()

    n_points_list = [100, 200, 300, 400, 500]
    best_c5 = float('inf')
    best_h = np.zeros(n_points_list[0])
    
    for n_points in n_points_list:
        if time.time() - start_time >= budget_s:
            break

        def evaluate(h):
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            corr = correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5
        
        def objective(h):
            return evaluate(h)
        
        def constraint_integral(h):
            return np.sum(h) - n_points / 2
        
        bounds = Bounds(0, 1)
        cons = [{'type': 'eq', 'fun': constraint_integral}]
        
        # Optimization with SLSQP
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        h_candidate = result.x
        
        # Local greedy improvement
        for _ in range(100):
            if time.time() - start_time >= budget_s:
                break
            h = h_candidate.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            corr = correlate(h, 1 - h, mode="full")
            max_index = np.argmax(corr)
            if max_index < len(corr) - 1:
                shift = max_index + 1
                if shift < len(h):
                    h[shift] = np.clip(h[shift] - 0.01, 0, 1)
            h_candidate = h

            best_h = h_candidate
    
    return (best_h, best_c5, len(best_h))
