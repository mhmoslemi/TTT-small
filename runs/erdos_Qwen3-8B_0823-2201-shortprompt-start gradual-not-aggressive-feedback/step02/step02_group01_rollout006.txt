```python
import numpy as np
import scipy.optimize as optimize
from scipy.signal import correlate

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    
    # Start with a higher resolution for better exploration
    n_points = 128
    h_values = np.zeros(n_points)
    h_values[:n_points//4] = 1
    h_values[n_points//2:3*n_points//4] = 1
    h_values = (h_values / np.sum(h_values)) * (n_points / 2)
    h_values = np.clip(h_values, 0, 1)
    
    dx = 2.0 / n_points
    
    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if not np.isclose(sum_h, n_points / 2):
            h = h * (n_points / 2) / sum_h
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        corr = correlate(h, one_minus_h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint, 'jac': lambda x: np.ones(n_points)}
    
    result = optimize.minimize(
        objective,
        h_values,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Introduce diversity in initial configurations
    for _ in range(3):
        # Randomly generate a new initial configuration
        random_h = np.random.rand(n_points)
        random_h = (random_h / np.sum(random_h)) * (n_points / 2)
        random_h = np.clip(random_h, 0, 1)
        
        # Optimize from this random configuration
        result = optimize.minimize(
            objective,
            random_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 30, 'ftol': 1e-8, 'disp': False}
        )
        
        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
    
    # Local refinement with perturbations
    for _ in range(5):
        perturbation = np.random.normal(0, 0.03, n_points)
        perturbation = np.clip(perturbation, -0.1, 0.1)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        sum_perturbed = np.sum(perturbed_h)
        if not np.isclose(sum_perturbed, n_points / 2):
            perturbed_h = perturbed_h * (n_points / 2) / sum_perturbed
        perturbed_h = np.clip(perturbed_h, 0, 1)
        result = optimize.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 20, 'ftol': 1e-8, 'disp': False}
        )
        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
    
    # Final check and return
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if not np.isclose(sum_best, n_points / 2):
        best_h = best_h * (n_points / 2) / sum_best
    best_h = np.clip(best_h, 0, 1)
    
    return best_h, best_c5, n_points
```
