def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    def normalize(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        return h * target_sum / np.sum(h)

    def crossover(parent1, parent2):
        child = np.copy(parent1)
        indices = np.random.choice(n_points, size=n_points//4, replace=False)
        child[indices] = parent2[indices]
        return normalize(child)

    def mutate(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        for i in range(n_points):
            if np.random.random() < 0.1:
                h[i] += np.random.uniform(-0.1, 0.1)
        return normalize(h)

    population_size = 20
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        # Find best individual
        best_index = np.argmin(fitness)
        current_best = population[best_index]
        current_c5 = fitness[best_index]
        if current_c5 < best_c5:
            best_h = current_best

        new_population = []

            # Tournament selection
            candidates = np.random.choice(population_size, size=5, replace=False)
            best_candidate = candidates[np.argmin([fitness[i] for i in candidates])]
            parent1 = population[best_candidate]
            # Mutate
            child = mutate(parent1)
            new_population.append(child)
            # Elite selection
            if np.random.random() < 0.2:
                new_population.append(current_best)

    return best_h, best_c5, n_points
