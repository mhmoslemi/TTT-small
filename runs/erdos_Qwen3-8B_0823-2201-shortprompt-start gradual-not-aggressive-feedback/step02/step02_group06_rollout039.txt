import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
import time

def run(seed=42, budget_s=20, **kwargs):

    start_time = time.time()

    n_points_list = [200, 300, 400, 500]
    best_c5 = float('inf')
    best_h = np.zeros(n_points_list[0])
    
    for n_points in n_points_list:
        if time.time() - start_time >= budget_s:
            break

        def evaluate(h):
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            corr = correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5
        
        def objective(h):
            return evaluate(h)
        
        def constraint_integral(h):
            return np.sum(h) - n_points / 2
        
        bounds = Bounds(0, 1)
        cons = [{'type': 'eq', 'fun': constraint_integral}]
        
        # Use L-BFGS-B for faster convergence
        result = minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)

        for _ in range(50):

                    h[shift] = np.clip(h[shift] - 0.01, 0, 1)
            h_candidate = h
