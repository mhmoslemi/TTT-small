        c5 = np.max(corr) * dx

        return np.sum(h) - target_sum

        penalty = 1000 * np.abs(np.sum(h) - target_sum)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5 + penalty

    best_h = initial_h.copy()
    best_c5 = np.inf
    start_time = time.time()

    for _ in range(3):

        perturbed_h = initial_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        
        result = opt.minimize(
            penalty_objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
