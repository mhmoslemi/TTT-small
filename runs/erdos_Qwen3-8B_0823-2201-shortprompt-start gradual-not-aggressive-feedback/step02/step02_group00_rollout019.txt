import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm for global search
    def genetic_algorithm(pop_size=30, generations=20):
        population = np.random.rand(pop_size, n_points)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        population = np.clip(population, 0, 1)
        for _ in range(generations):
            fitness = np.array([objective(ind) for ind in population])
            best_idx = np.argmin(fitness)
            best_fitness = fitness[best_idx]
            if best_fitness < best_c5:
                best_h = population[best_idx]
                best_c5 = best_fitness
            # Select parents
            parents = population[np.argsort(fitness)[:pop_size//2]]
            # Crossover and mutation
            offspring = []
            for _ in range(pop_size//2):
                parent1, parent2 = np.random.choice(parents, 2, replace=False)
                child = (parent1 + parent2) / 2
                mutation = np.random.normal(0, 0.05, n_points)
                child += mutation
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
                offspring.append(child)
            population = np.vstack([parents, offspring])
            population = np.clip(population, 0, 1)
            population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        return best_h, best_c5

    # Run genetic algorithm
    start_time = time.time()
