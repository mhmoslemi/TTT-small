import numpy as np
import scipy.optimize as optimize
from scipy.signal import correlate
from scipy.optimize import Bounds

def run(seed=42, budget_s=20, **kwargs):

    # Initial solution with n_points = 128
    n_points = 128
    h_values = np.zeros(n_points)
    h_values[::2] = 1.0  # Initial pattern with alternating 1s and 0s
    
    # Discretization parameters
    dx = 2.0 / n_points
    sum_h = np.sum(h_values)
    if not np.isclose(sum_h, n_points / 2):
        h_values = (h_values / sum_h) * (n_points / 2)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        penalty = 100 * np.abs(sum_h - n_points / 2)
        one_minus_h = 1 - h
        corr = correlate(h, one_minus_h, mode="full")
        c5 = np.max(corr) * dx + penalty
        return c5
    
    bounds = Bounds(0, 1)
    
    # Genetic algorithm parameters
    population_size = 30
    generations = 20
    mutation_rate = 0.1
    tournament_size = 5
    
    # Initial population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * (n_points / 2) / np.sum(h) for h in population]
    population = [np.clip(h, 0, 1) for h in population]
    
    best_h = h_values
    best_c5 = objective(h_values)
    
    for _ in range(generations):
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        
        # Select tournament winners
        tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
        tournament_fitness = [fitness[i] for i in tournament_indices]
        winner_index = tournament_indices[np.argmin(tournament_fitness)]
        winner = population[winner_index]
        
        # Create offspring
        offspring = []
        for _ in range(population_size):
            parent1 = np.random.choice(population)
            parent2 = np.random.choice(population)
            child = np.minimum(parent1, parent2)
            # Apply mutation
            mutate_mask = np.random.rand(n_points) < mutation_rate
            child[mutate_mask] = np.random.uniform(0, 1, np.sum(mutate_mask))
            # Scale to maintain integral constraint
            child = child * (n_points / 2) / np.sum(child)
            child = np.clip(child, 0, 1)
            offspring.append(child)
        
        # Replace population
        population = offspring
        
        # Update best solution
        for h in population:
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
    
    # Final refinement with SLSQP
    def objective_refine(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        penalty = 100 * np.abs(sum_h - n_points / 2)
        one_minus_h = 1 - h
        corr = correlate(h, one_minus_h, mode="full")
        c5 = np.max(corr) * dx + penalty
        return c5
    
    result = optimize.minimize(
        objective_refine,
        best_h,
        method='SLSQP',
        bounds=bounds,
        options={'maxiter': 30, 'ftol': 1e-8, 'disp': False}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if not np.isclose(sum_best, n_points / 2):
        best_h = best_h * (n_points / 2) / sum_best
    best_h = np.clip(best_h, 0, 1)
    best_c5 = objective_refine(best_h)
    
    return best_h, best_c5, n_points
