import scipy.optimize as optimize
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode="full")) * (2.0 / n_points)
    best_n = n_points
    time_left = budget_s

    def evaluate(h, n):
        dx = 2.0 / n
        h = np.clip(h, 0, 1)
        norm = np.sum(h) * dx
        h = h * (n / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def create_individual(n):
        h = np.random.rand(n)
        h = np.clip(h, 0, 1)
        norm = np.sum(h) * (2.0 / n)
        h = h * (n / 2) / np.sum(h)
        return h

    def crossover(parent1, parent2, n):
        split = np.random.randint(1, n-1)
        child = np.concatenate([parent1[:split], parent2[split:]])
        child = np.clip(child, 0, 1)
        norm = np.sum(child) * (2.0 / n)
        child = child * (n / 2) / np.sum(child)
        return child

    def mutate(individual, n, mutation_rate=0.05):
        for i in range(n):
            if np.random.rand() < mutation_rate:
                individual[i] = np.random.rand()
        individual = np.clip(individual, 0, 1)
        norm = np.sum(individual) * (2.0 / n)
        individual = individual * (n / 2) / np.sum(individual)
        return individual

    for n in [50, 60, 70, 80, 90, 100]:
        if time_left <= 0:
            break
        time_start = time.time()
        population_size = 20
        population = [create_individual(n) for _ in range(population_size)]
        for _ in range(100):
            if time_left <= 0:
                break
            fitness = [evaluate(ind, n) for ind in population]
            best_idx = np.argmin(fitness)
            if fitness[best_idx] < best_c5:
                best_c5 = fitness[best_idx]
                best_h = population[best_idx].copy()
                best_n = n
            # Select parents
            parents = np.random.choice(population_size, size=2, replace=False)
            parent1 = population[parents[0]]
            parent2 = population[parents[1]]
            # Crossover
            child = crossover(parent1, parent2, n)
            # Mutate
            child = mutate(child, n)
            # Replace worst individual
            worst_idx = np.argmax(fitness)
            population[worst_idx] = child
        time_used = time.time() - time_start
        time_left -= time_used

    return best_h, best_c5, best_n
