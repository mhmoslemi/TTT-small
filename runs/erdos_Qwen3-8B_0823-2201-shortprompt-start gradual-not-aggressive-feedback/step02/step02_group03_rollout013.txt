def run(seed=42, budget_s=20, **kwargs):

    dx = 2.0 / n_points
    max_overlap = np.max(np.correlate(initial_h, 1 - initial_h, mode="full")) * dx
    best_h = initial_h.copy()
    best_c5 = max_overlap
    best_n = n_points
    time_left = budget_s

    def evaluate(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h) * dx
        if abs(total - 1) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] = np.clip(h[i] + np.random.uniform(-0.2, 0.2), 0, 1)
        return h

    def crossover(h1, h2):
        h = h1.copy()
        for i in range(len(h)):
            if np.random.rand() < 0.5:
                h[i] = h2[i]
        return h

    # Start with initial construction
    population = [np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h) for _ in range(50)]
    population = [np.clip(p, 0, 1) for p in population]
    population = [p * (n_points / 2) / np.sum(p) for p in population]

    for generation in range(100):
        if time_left <= 0:
            break
        time_start = time.time()
        scores = [evaluate(p) for p in population]
        best_idx = np.argmin(scores)
        best_c5 = scores[best_idx]

            best_h = population[best_idx].copy()
            best_c5 = best_c5

        parents = np.random.choice(len(population), size=20, replace=False)

        new_population = []

            child = crossover(population[p1], population[p2])
            child = mutate(child)
            new_population.append(child)
        population = new_population
        time_used = time.time() - time_start

    return best_h, best_c5, best_n
