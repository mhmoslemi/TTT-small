import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode="full")) * 2.0 / n_points
    time_left = budget_s

    def objective(x, n, dx):
        h = x.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint(x, n):
        h = x.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        return np.sum(h) - n / 2

    for n in [70, 80, 90, 100]:
        if time_left <= 0:
            break
        time_start = time.time()
        x0 = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(initial_h)), initial_h)
        x0 = np.clip(x0, 0, 1)
        x0 = x0 * (n / 2) / np.sum(x0)
        
        # Perturb initial guess
        perturb = np.random.normal(0, 0.05, n)
        x0_perturb = np.clip(x0 + perturb, 0, 1)
        x0_perturb = x0_perturb * (n / 2) / np.sum(x0_perturb)
        
        bounds = [(0, 1) for _ in range(n)]
        cons = [{'type': 'eq', 'fun': lambda x: constraint(x, n)}]
        
        result1 = optimize.minimize(lambda x: objective(x, n, 2.0 / n), x0, 
                                  method='SLSQP', bounds=bounds, constraints=cons,
                                  tol=1e-6, options={'maxiter': 100})
        result2 = optimize.minimize(lambda x: objective(x, n, 2.0 / n), x0_perturb, 
                                  method='SLSQP', bounds=bounds, constraints=cons,
                                  tol=1e-6, options={'maxiter': 100})
        
        for result in [result1, result2]:
            if result.success:
                h_opt = result.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n / 2) / np.sum(h_opt)
                corr = np.correlate(h_opt, 1 - h_opt, mode="full")
                c5 = np.max(corr * 2.0 / n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
        time_used = time.time() - time_start
        time_left -= time_used

    return best_h, best_c5, len(best_h)
