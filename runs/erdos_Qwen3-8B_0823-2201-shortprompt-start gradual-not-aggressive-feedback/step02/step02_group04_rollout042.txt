def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * 2.0 / n_points
    best_n = n_points
    time_left = budget_s

    def evaluate(h):
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constrain(h):
        return np.sum(h) - len(h) / 2

    def generate_individual(n):
        h = np.random.rand(n)
        h = h * (n / 2) / np.sum(h)
        return h

    def crossover(parent1, parent2):
        n = len(parent1)
        split = np.random.randint(1, n-1)
        child = np.concatenate([parent1[:split], parent2[split:]])
        child = child * (n / 2) / np.sum(child)
        return child

    def mutate(individual, mutation_rate=0.1):
        n = len(individual)
        for i in range(n):
            if random.random() < mutation_rate:
                individual[i] = np.random.rand()
        individual = individual * (n / 2) / np.sum(individual)
        return individual

    def fitness(individual):
        return -evaluate(individual)

    for n in [100, 80, 60, 50, 40, 30]:
        time_start = time.time()
        population = [generate_individual(n) for _ in range(50)]
        for generation in range(100):
            if time_left <= 0:
                break
            fitnesses = [fitness(ind) for ind in population]
            parents = np.argsort(fitnesses)[-10:]
            population = [population[i] for i in parents]
            for _ in range(len(population) // 2):
                p1, p2 = random.sample(population, 2)
                child = crossover(p1, p2)
                child = mutate(child)
                population.append(child)
            population = np.array(population)
            population = population[population.argsort()[:len(population)//2]]
            best = population[0]
            current_c5 = -fitness(best)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = best
                best_n = n
        time_used = time.time() - time_start
        time_left -= time_used
        if time_left <= 0:
            break

    return best_h, best_c5, best_n
