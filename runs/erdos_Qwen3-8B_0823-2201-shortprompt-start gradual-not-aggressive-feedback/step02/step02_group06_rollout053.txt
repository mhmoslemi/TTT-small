import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
import time

def run(seed=42, budget_s=20, **kwargs):

    start_time = time.time()

    n_points_list = [400, 500, 600]
    best_c5 = 1.0

            corr = correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5

            return evaluate(h)

            return np.sum(h) - n_points / 2
        
        bounds = Bounds(0, 1)
        cons = [{'type': 'eq', 'fun': constraint_integral}]

        result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
