import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    best_h = np.copy(initial_h)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    pop_size = 20
    mutation_rate = 0.15
    crossover_rate = 0.8
    generations = 10
    
    start_time = time.time()
    
    for gen in range(generations):
        # Generate population
        population = np.random.uniform(0, 1, (pop_size, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        # Evaluate population
        fitness = np.array([objective(p) for p in population])
        
        # Select best
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_best_c5 = fitness[best_idx]
        
        # Update best solution
        if current_best_c5 < best_c5:
            best_h = current_best
            best_c5 = current_best_c5
        
        # Early stopping
        if time.time() - start_time > budget_s - 1:
            break
        
        # Crossover and mutation
        for i in range(pop_size):
            if np.random.rand() < crossover_rate:
                j = np.random.randint(pop_size)
                crossover_point = np.random.randint(1, n_points)
                population[i] = np.concatenate((
                    population[i][:crossover_point],
                    population[j][crossover_point:]
                ))
            
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, 0.05, n_points)
                population[i] += mutation
                population[i] = np.clip(population[i], 0, 1)
                population[i] = population[i] * target_sum / np.sum(population[i])
    
    # Local refinement
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
