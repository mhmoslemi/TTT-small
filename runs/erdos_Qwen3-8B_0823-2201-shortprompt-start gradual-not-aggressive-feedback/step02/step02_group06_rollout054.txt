        corr = correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def mutate(individual):
        for i in range(n_points):
            if random.random() < mutation_rate:
                individual[i] += np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        return individual
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(n_points):
            if random.random() < 0.5:
                child[i] = parent2[i]
        child = child * (n_points / 2) / np.sum(child)
        return child
    
    # Initialize population
    population = [create_individual() for _ in range(population_size)]
    population = [np.clip(p, 0, 1) for p in population]
    population = [p * (n_points / 2) / np.sum(p) for p in population]

    best_c5 = evaluate(best_h)

        population = new_population
    
    # Final evaluation
    best_h = min(population, key=evaluate)
