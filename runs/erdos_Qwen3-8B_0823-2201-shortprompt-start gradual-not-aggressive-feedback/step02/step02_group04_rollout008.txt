import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * 2.0 / n_points
    best_n = n_points
    time_left = budget_s

    def evaluate(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.inf
        h = h * (n_points / 2) / total
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * 2.0 / n_points)

    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] = np.random.rand()
        return h

    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        for i in range(len(child)):
            if np.random.rand() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child

    for n in [100, 80, 60, 50, 40, 30]:
        time_start = time.time()
        population_size = 50
        population = [mutate(initial_h) for _ in range(population_size)]
        for _ in range(100):
            if time_left <= 0:
                break
            fitness = [evaluate(h) for h in population]
            best_idx = np.argmin(fitness)
            best_h_current = population[best_idx]
            if fitness[best_idx] < best_c5:
                best_c5 = fitness[best_idx]
                best_h = best_h_current
                best_n = n
            # Selection and reproduction
            population = np.array(population)
            population = population[np.argsort(fitness)]
            population = population[:population_size // 2]
            new_population = []
            for _ in range(population_size // 2):
                parent1 = np.random.choice(population)
                parent2 = np.random.choice(population)
                child = crossover(parent1, parent2)
                child = mutate(child)
                new_population.append(child)
            population = np.array(new_population)
        time_used = time.time() - time_start
        time_left -= time_used

    return best_h, best_c5, best_n
