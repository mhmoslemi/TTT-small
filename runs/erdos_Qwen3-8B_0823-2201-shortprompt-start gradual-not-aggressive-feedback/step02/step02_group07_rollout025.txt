import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Improved initial guess: smooth transition with periodicity
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 0.5 * (1 + np.cos(np.pi * (i - n_points / 4) / n_points))
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Hybrid optimization: local search with global exploration
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    # Adaptive refinement with random perturbations
    start_time = time.time()
    while time.time() - start_time < budget_s:
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = result.x + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        perturbed_c5 = objective(perturbed_h)
        
        if perturbed_c5 < result.fun:
            result = opt.minimize(
                objective,
                perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
