        h = np.clip(h, 0, 1)
        norm = np.sum(h) / (n / 2)
        return h / norm

        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def evaluate(h, n):
        h = normalize(h, n)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def mutation(h, n, mutation_rate=0.1):
        h = h.copy()
        for i in range(n):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.uniform(-0.2, 0.2)
                h[i] = np.clip(h[i], 0, 1)
        return h

    def crossover(parent1, parent2, n):
        h = np.zeros(n)
        for i in range(n):
            if np.random.rand() < 0.5:
                h[i] = parent1[i]
            else:
                h[i] = parent2[i]
        return h
