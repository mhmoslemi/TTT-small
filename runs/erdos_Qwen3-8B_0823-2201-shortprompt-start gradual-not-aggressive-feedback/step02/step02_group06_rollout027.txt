import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup with n_points = 500
    n_points = 500
    dx = 2.0 / n_points
    
    # Create a more balanced initial configuration
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0
    initial_h = np.roll(initial_h, int(n_points / 4))
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    h = initial_h.copy()
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def objective(h):
        return evaluate(h)
    
    def constraint_integral(h):
        return np.sum(h) - n_points / 2
    
    # Optimization with L-BFGS-B
    bounds = Bounds(0, 1)
    cons = [{'type': 'eq', 'fun': constraint_integral}]
    
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    best_h = result.x
    
    # Local greedy improvement
    for _ in range(100):
        if time.time() - start_time >= budget_s:
            break
        h = best_h.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = correlate(h, 1 - h, mode="full")
        max_index = np.argmax(corr)
        if max_index < len(corr) - 1:
            shift = max_index + 1
            if shift < len(h):
                h[shift] = np.clip(h[shift] - 0.01, 0, 1)
        best_h = h
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = evaluate(best_h)
    
    return (best_h, best_c5, n_points)
