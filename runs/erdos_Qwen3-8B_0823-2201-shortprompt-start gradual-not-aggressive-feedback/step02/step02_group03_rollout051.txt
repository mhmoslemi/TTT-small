import time

    best_h = initial_h.copy()
    best_c5 = max_overlap
    best_n = n_points

        h = x.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")

    def constraint(x, n):
        h = x.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        return np.sum(h) - n / 2

    # Start with initial construction
    n_start = len(initial_h)
    for n in [n_start, n_start + 10, n_start + 20, n_start + 30, n_start + 40, n_start + 50]:

            break

        # Optimization
        bounds = [(0, 1) for _ in range(n)]
        cons = [{'type': 'eq', 'fun': lambda x: constraint(x, n)}]
        result = optimize.minimize(lambda x: objective(x, n, dx), h_init,
