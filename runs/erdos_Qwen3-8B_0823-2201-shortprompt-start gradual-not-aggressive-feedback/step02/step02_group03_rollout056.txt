import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    max_overlap = np.max(np.correlate(initial_h, 1 - initial_h, mode="full")) * dx
    best_h = initial_h.copy()
    best_c5 = max_overlap
    best_n = n_points
    time_left = budget_s

    def objective(x, n, dx):
        h = np.clip(x, 0, 1)
        norm = np.sum(h) * dx
        if norm == 0:
            return np.inf
        h = h / norm * (n / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint(x, n):
        h = np.clip(x, 0, 1)
        norm = np.sum(h) * 2.0 / n
        return norm - 1.0

    for n in [50, 60, 70, 80, 90, 100, 110, 120]:
        if time_left <= 0:
            break
        time_start = time.time()
        h_init = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(initial_h)), initial_h)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n / 2) / np.sum(h_init)
        
        bounds = [(0, 1) for _ in range(n)]
        cons = [{'type': 'eq', 'fun': lambda x: constraint(x, n)}]
        
        result = optimize.minimize(lambda x: objective(x, n, 2.0 / n), h_init, 
                                 method='SLSQP', bounds=bounds, constraints=cons,
                                 tol=1e-6, options={'maxiter': 100})
        
        if result.success:
            h_opt = np.clip(result.x, 0, 1)

            h_opt = h_opt / norm * (n / 2)
            corr = np.correlate(h_opt, 1 - h_opt, mode="full")
            c5 = np.max(corr * (2.0 / n))
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n = n
        time_used = time.time() - time_start
        time_left -= time_used

    return best_h, best_c5, best_n
