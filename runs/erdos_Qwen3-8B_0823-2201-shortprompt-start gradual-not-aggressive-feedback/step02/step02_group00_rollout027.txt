import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm for global search
    def genetic_algorithm(pop_size=20, generations=10, mutation_rate=0.1):
        population = np.random.uniform(0, 1, (pop_size, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        best_individual = population[0]
        best_fitness = objective(best_individual)
        
        for _ in range(generations):
            fitness = np.array([objective(ind) for ind in population])
            parents = population[np.argsort(fitness)[:pop_size//2]]
            
            # Crossover
            offspring = []
            for _ in range(pop_size//2):
                parent1, parent2 = np.random.choice(parents, 2, replace=False)
                crossover_point = np.random.randint(1, n_points-1)
                child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
                offspring.append(child)
            
            # Mutation
            for i in range(len(offspring)):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        offspring[i][j] = np.clip(offspring[i][j] + np.random.uniform(-0.05, 0.05), 0, 1)
                        offspring[i][j] = offspring[i][j] * target_sum / np.sum(offspring[i])
            
            population = np.array(offspring)
            population = np.vstack((population, parents))
            population = np.random.choice(population, size=pop_size, replace=False)
            
            # Update best
            current_best = population[0]
            current_fitness = objective(current_best)
            if current_fitness < best_fitness:
                best_fitness = current_fitness
                best_individual = current_best
        
        return best_individual, best_fitness

    # Run genetic algorithm
    start_time = time.time()
    while time.time() - start_time < budget_s:
        best_h, best_c5 = genetic_algorithm(pop_size=20, generations=10, mutation_rate=0.1)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        best_c5 = objective(best_h)
        
        # Local refinement with SLSQP
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    return best_h, best_c5, n_points
