import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    max_overlap = np.max(np.correlate(initial_h, 1 - initial_h, mode="full")) * dx
    best_h = initial_h.copy()
    best_c5 = max_overlap
    best_n = n_points
    time_left = budget_s

    def objective(x, n, dx):
        h = x.reshape(-1)
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.inf
        h = h * (n / 2) / total
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint(x, n):
        h = x.reshape(-1)
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.inf
        return total - n / 2

    # Try with different resolutions
    for n in [50, 60, 70, 80, 90, 100]:
        if time_left <= 0:
            break
        time_start = time.time()
        # Generate multiple initial points
        n_trials = min(3, int(time_left / 1))  # Limit trials based on time
        for _ in range(n_trials):
            # Create random initial guess
            h_init = np.random.rand(n)
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n / 2) / np.sum(h_init)
            
            bounds = [(0, 1) for _ in range(n)]
            cons = [{'type': 'eq', 'fun': lambda x: constraint(x, n)}]
            result = optimize.minimize(lambda x: objective(x, n, dx), h_init, 
                                     method='SLSQP', bounds=bounds, constraints=cons,
                                     tol=1e-6, options={'maxiter': 100})
            
            if result.success:
                h_opt = result.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n / 2) / np.sum(h_opt)
                corr = np.correlate(h_opt, 1 - h_opt, mode="full")
                c5 = np.max(corr * dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
                    best_n = n
        time_used = time.time() - time_start
        time_left -= time_used

    return best_h, best_c5, best_n
