import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial guess: alternating pattern
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            initial_h[i] = 1.0
        elif i % 3 == 1:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 0.0

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def genetic_algorithm(pop_size=50, generations=20):
        population = np.random.rand(pop_size, n_points)
        population = population * target_sum / np.sum(population, axis=1)[:, None]
        population = np.clip(population, 0, 1)
        
        for _ in range(generations):
            fitness = np.array([objective(ind) for ind in population])
            best_idx = np.argmin(fitness)
            best_h = population[best_idx]
            
            # Selection
            selected = np.random.choice(pop_size, size=pop_size, p=fitness / np.sum(fitness))
            selected = population[selected]
            
            # Crossover
            offspring = []
            for i in range(0, pop_size, 2):
                if i + 1 < pop_size:
                    parent1 = selected[i]
                    parent2 = selected[i + 1]
                    child1 = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                    child2 = np.where(np.random.rand(n_points) < 0.5, parent2, parent1)
                    offspring.append(child1)
                    offspring.append(child2)
                else:
                    offspring.append(selected[i])

                mutation = np.random.normal(0, 0.05, n_points)
                offspring[i] += mutation
                offspring[i] = np.clip(offspring[i], 0, 1)
                offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])

    return best_h, best_c5, n_points
