from scipy.signal import correlate
import time
import random

    start_time = time.time()

    n_points = 500
    dx = 2.0 / n_points

        corr = correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def fitness(h):
        return -evaluate(h)  # Minimize C₅
    
    # Genetic algorithm parameters
    population_size = 50
    mutation_rate = 0.1
    generations = 100
    
    # Initialize population
    population = [initial_h.copy() for _ in range(population_size)]
    
    for generation in range(generations):
        if time.time() - start_time >= budget_s:
            break
        
        # Evaluate fitness
        fitness_values = [fitness(ind) for ind in population]
        best_idx = np.argmax(fitness_values)
        best_h = population[best_idx].copy()
        
        # Selection
        selected = np.random.choice(range(population_size), size=population_size, p=np.array(fitness_values)/np.sum(fitness_values))
        selected = [population[i] for i in selected]
        
        # Crossover
        offspring = []

                parent2 = selected[i+1]
