import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial guess: alternating pattern
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            initial_h[i] = 1.0
        elif i % 3 == 1:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 0.0
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = [np.random.uniform(0, 1, n_points) * target_sum / np.sum(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        
        # Select parents
        parents = np.argsort(fitness)[:int(population_size/2)]
        
        # Crossover
        offspring = []
        for _ in range(population_size - len(parents)):
            parent1 = population[np.random.choice(parents)]
            parent2 = population[np.random.choice(parents)]
            child = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < crossover_rate:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    offspring[i][j] = np.random.uniform(0, 1)
            offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
        
        # Replace population
        population = population + offspring
        population = np.array(population)
        population = np.vstack([population, np.random.uniform(0, 1, (population_size, n_points)) * target_sum / np.sum(np.random.uniform(0, 1, (population_size, n_points)))]).reshape(-1, n_points)
        population = population[:population_size]
        population = [h * target_sum / np.sum(h) for h in population]
    
    # Evaluate best solution
    best_h = population[np.argmin(fitness)]
    best_c5 = np.min(fitness)
    
    return best_h, best_c5, n_points
