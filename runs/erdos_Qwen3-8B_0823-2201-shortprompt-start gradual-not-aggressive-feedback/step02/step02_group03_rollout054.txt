import scipy.optimize as optimize
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points

    best_h = initial_h.copy()
    best_c5 = max_overlap
    best_n = n_points
    time_left = budget_s

    def evaluate(h, n, dx):
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def integral_constraint(h, n):
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        return np.sum(h) - n / 2

    for n in [60, 70, 80, 90]:
        if time_left <= 0:
            break
        time_start = time.time()
        h_init = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(initial_h)), initial_h)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n / 2) / np.sum(h_init)
        dx = 2.0 / n

        bounds = [(0, 1) for _ in range(n)]
        cons = [{'type': 'eq', 'fun': lambda x: integral_constraint(x, n)}]

        result = optimize.minimize(lambda x: evaluate(x, n, dx), h_init, 
                                 method='SLSQP', bounds=bounds, constraints=cons,

                best_h = h_opt
                best_n = n
        time_used = time.time() - time_start
        time_left -= time_used

    return best_h, best_c5, best_n
