import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    initial_h = np.zeros(n_points)
    initial_h[::3] = 1.0
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 30
    mutation_rate = 0.15
    
    # Evaluate initial solution
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Genetic algorithm loop
    for _ in range(generations):
        population = np.random.uniform(0, 1, (population_size, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        # Evaluate population
        fitness = np.array([objective(ind) for ind in population])
        
        # Select best individuals
        selected = population[np.argsort(fitness)[:int(population_size/2)]]
        
        # Create new population through crossover and mutation
        new_population = []
        for _ in range(int(population_size/2)):
            parent1 = np.random.choice(selected)
            parent2 = np.random.choice(selected)
            child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
            
            # Mutation
            mutation = np.random.normal(0, mutation_rate, n_points)
            mutated = child + mutation
            mutated = np.clip(mutated, 0, 1)
            mutated = mutated * target_sum / np.sum(mutated)
            new_population.append(mutated)
        
        population = np.array(new_population)
        population = np.vstack([selected, population])
        population = np.unique(population, axis=0)
        
        # Update best solution
        for ind in population:
            c5 = objective(ind)
            if c5 < best_c5:
                best_h = ind.copy()
                best_c5 = c5
    
    # Local optimization
    result = opt.minimize(
        objective,
        best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
