import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    # Genetic algorithm for global search
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        child[indices] = parent2[indices]
        return child
    
    def mutate(individual):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                if np.random.rand() < 0.5:
                    individual[i] = np.clip(individual[i] + np.random.uniform(-0.1, 0.1), 0, 1)
                else:
                    individual[i] = np.random.uniform(0, 1)
        return individual * target_sum / np.sum(individual)
    
    population = [best_h + np.random.uniform(-0.1, 0.1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate population
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            parent1, parent2 = random.sample(population, 2)
            child1 = crossover(parent1, parent2)
            child1 = mutate(child1)
            new_population.append(child1)
            
            child2 = crossover(parent2, parent1)
            child2 = mutate(child2)
            new_population.append(child2)
        
        population = new_population + [population[best_idx]]  # Keep best solution
        
        # Update best solution
        if best_c5 < objective(population[0]):
            best_h = population[0]
    
    # Final refinement with L-BFGS-B
    result = opt.minimize(
        objective,
        best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
