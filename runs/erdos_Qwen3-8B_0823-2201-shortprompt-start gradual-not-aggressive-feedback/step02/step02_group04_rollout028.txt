import scipy.optimize as optimize
import time

    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * 2.0 / len(best_h)
    best_n = len(best_h)
    time_left = budget_s

    def objective(x, n_points, dx):
        h = x.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint(x, n_points):
        h = x.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return np.sum(h) - n_points / 2

    def optimize_for_n(n_points):
        dx = 2.0 / n_points
        bounds = [(0, 1) for _ in range(n_points)]
        h_initial = np.random.uniform(0, 1, n_points)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        h_initial = np.clip(h_initial, 0, 1)

        def wrapped_objective(x):
            return objective(x, n_points, dx)

        def wrapped_constraint(x):
            return constraint(x, n_points)

        cons = [{'type': 'eq', 'fun': wrapped_constraint}]
        result = optimize.minimize(wrapped_objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints=cons)
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)

            best_n = n
