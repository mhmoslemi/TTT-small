import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * 2.0 / n_points
    best_n = n_points
    time_left = budget_s

    def objective(x, n, dx):
        h = x.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint(x, n):
        h = x.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        return np.sum(h) - n / 2

    for n in [100, 80, 60, 50, 40, 30]:
        time_start = time.time()
        h = initial_h.copy()
        h = np.resize(h, n)
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        dx = 2.0 / n

        bounds = [(0, 1) for _ in range(n)]
        cons = [{'type': 'eq', 'fun': lambda x: constraint(x, n)}]

        # Multi-start optimization
        for _ in range(3):
            if np.random.rand() < 0.3:
                h_initial = np.random.uniform(0, 1, n)
            else:
                h_initial = h.copy()
            h_initial = h_initial * (n / 2) / np.sum(h_initial)

            result = optimize.minimize(
                lambda x: objective(x, n, dx),
                h_initial,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            if result.success:
                h_opt = result.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n / 2) / np.sum(h_opt)
                corr = np.correlate(h_opt, 1 - h_opt, mode="full")
                c5 = np.max(corr * dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
                    best_n = n

        time_used = time.time() - time_start
        time_left -= time_used
        if time_left <= 0:
            break

    return best_h, best_c5, best_n
