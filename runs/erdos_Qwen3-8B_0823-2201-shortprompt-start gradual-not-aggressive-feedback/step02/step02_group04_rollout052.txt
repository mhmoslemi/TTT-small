import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * 2.0 / n_points
    best_n = n_points
    time_left = budget_s

    def evaluate(h):
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def fitness(h):
        return -evaluate(h)  # Minimize C5 by maximizing fitness

    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        return h

    def crossover(h1, h2):
        h = h1.copy()
        for i in range(len(h)):
            if np.random.rand() < 0.5:
                h[i] = h2[i]
        return h

    def ensure_integral(h):
        total = np.sum(h)
        if total == 0:
            return np.random.uniform(0, 1, len(h))
        scale = (len(h) / 2) / total
        return np.clip(h * scale, 0, 1)

    # Initial population
    population = [ensure_integral(initial_h.copy()) for _ in range(20)]
    population.append(ensure_integral(np.random.uniform(0, 1, n_points)))
    population = list(set(tuple(p) for p in population))  # Remove duplicates

    for _ in range(100):
        if time_left <= 0:
            break
        time_start = time.time()
        
        # Evaluate fitness
        fitness_scores = [fitness(h) for h in population]
        best_idx = np.argmax(fitness_scores)
        best_fitness = fitness_scores[best_idx]
        
        # Select parents
        parents = np.random.choice(len(population), size=10, replace=False)
        parent1, parent2 = population[parents[0]], population[parents[1]]
        
        # Crossover and mutation
        child = crossover(parent1, parent2)
        child = mutate(child)
        child = ensure_integral(child)
        
        # Evaluate child
        child_fitness = fitness(child)
        if child_fitness > best_fitness:
            population.append(child)
            population.pop(np.random.choice(len(population)))
        
        time_used = time.time() - time_start
        time_left -= time_used

    # Check smaller n_points
    for n in [100, 80, 60, 50, 40, 30]:
        if time_left <= 0:
            break
        time_start = time.time()
        h = initial_h.copy()
        h = np.resize(h, n)
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        dx = 2.0 / n
        
        def objective(x):
            h = x.reshape(-1)
            h = np.clip(h, 0, 1)
            h = h * (n / 2) / np.sum(h)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr * dx)

        def constraint(x):
            h = x.reshape(-1)
            h = np.clip(h, 0, 1)
            h = h * (n / 2) / np.sum(h)
            return np.sum(h) - n / 2

        cons = [{'type': 'eq', 'fun': constraint}]
        bounds = [(0, 1) for _ in range(n)]

        result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons)
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n / 2) / np.sum(h_opt)
            corr = np.correlate(h_opt, 1 - h_opt, mode="full")
            c5 = np.max(corr * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n = n
        time_used = time.time() - time_start
        time_left -= time_used

    return best_h, best_c5, best_n
