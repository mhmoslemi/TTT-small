import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm for global search
    def genetic_algorithm(pop_size=20, generations=10, mutation_rate=0.1):
        population = np.random.uniform(0, 1, (pop_size, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            fitness = np.array([objective(ind) for ind in population])
            idxs = np.argsort(fitness)
            selected = population[idxs[:pop_size//2]]
            
            # Crossover
            offspring = []
            for _ in range(pop_size//2):
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                offspring.append(child)
            
            # Mutation
            for i in range(len(offspring)):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        offspring[i][j] = np.random.uniform(0, 1)
            
            # Replace population
            population = np.vstack((selected, offspring))
            population = population * target_sum / np.sum(population, axis=1, keepdims=True)
            population = np.clip(population, 0, 1)
        
        best_idx = np.argmin(fitness)
        return population[best_idx]

    # Run genetic algorithm
    start_time = time.time()
    while time.time() - start_time < budget_s:
        genetic_result = genetic_algorithm(pop_size=20, generations=5)
        genetic_c5 = objective(genetic_result)
        
        if genetic_c5 < best_c5:
            best_h = genetic_result
            best_c5 = genetic_c5

        # Local optimization

            best_h,
            method='L-BFGS-B',

            constraints={'type': 'eq', 'fun': constraint_sum},

        )
        
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        best_c5 = objective(best_h)

    return best_h, best_c5, n_points
