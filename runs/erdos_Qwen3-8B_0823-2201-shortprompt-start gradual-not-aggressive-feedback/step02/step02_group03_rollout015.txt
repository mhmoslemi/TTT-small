import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode="full")) * dx
    best_n = n_points
    time_left = budget_s

    def objective(x, n, dx):
        h = x.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint(x, n):
        h = x.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        return np.sum(h) - n / 2

    def evaluate(h, n):
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * 2.0 / n)

    def perturb(h, n, scale=0.1):
        h = h.copy()
        indices = np.random.choice(n, int(n * 0.1), replace=False)
        h[indices] += np.random.uniform(-scale, scale, len(indices))
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        return h

    for n in [50, 60, 70, 80, 90, 100]:
        if time_left <= 0:
            break
        time_start = time.time()
        h_init = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(initial_h)), initial_h)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n / 2) / np.sum(h_init)
        dx = 2.0 / n

        bounds = [(0, 1) for _ in range(n)]
        cons = [{'type': 'eq', 'fun': lambda x: constraint(x, n)}]
        
        for _ in range(3):
            if time_left <= 0:
                break
            result = optimize.minimize(lambda x: objective(x, n, dx), h_init, 
                                     method='SLSQP', bounds=bounds, constraints=cons,
                                     tol=1e-6, options={'maxiter': 100})
            if result.success:
                h_opt = result.x
                c5 = evaluate(h_opt, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
                    best_n = n
            h_init = perturb(h_init, n, scale=0.05)
        time_used = time.time() - time_start
        time_left -= time_used

    return best_h, best_c5, best_n
