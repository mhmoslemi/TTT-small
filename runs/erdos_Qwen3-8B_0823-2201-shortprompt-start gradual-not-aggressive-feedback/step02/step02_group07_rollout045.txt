import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial population
    population_size = 30
    population = np.random.rand(population_size, n_points)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def evaluate(individual):
        return objective(individual)
    
    # Genetic algorithm parameters
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 50
    
    start_time = time.time()
    for generation in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = np.array([evaluate(ind) for ind in population])
        
        # Select parents
        parents_indices = np.argsort(fitness)[:population_size//2]
        parents = population[parents_indices]
        
        # Crossover
        offspring = []
        for _ in range(population_size//2):
            if np.random.rand() < crossover_rate:
                parent1, parent2 = np.random.choice(parents, size=2, replace=False)
                child = np.concatenate([parent1[:n_points//2], parent2[n_points//2:]])
            else:
                child = np.random.rand(n_points)
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    offspring[i][j] = np.random.rand()
        
        # Replace population
        population = np.vstack([parents, offspring])
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        # Keep best individuals
        population = np.vstack([population, np.min(population, axis=0)])
    
    # Get best solution
    fitness = np.array([evaluate(ind) for ind in population])
    best_index = np.argmin(fitness)
    best_h = population[best_index]
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
