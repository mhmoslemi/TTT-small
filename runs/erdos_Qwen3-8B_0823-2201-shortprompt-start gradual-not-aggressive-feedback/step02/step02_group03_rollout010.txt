import scipy.optimize as optimize
import time

def run(seed=42, budget_s=20, **kwargs):

    best_n = n_points
    time_left = budget_s

    def objective(x, n, dx):
        h = x.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint(x, n):
        h = x.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        return np.sum(h) - n / 2

    # Try with different resolutions

        h_init = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(initial_h)), initial_h)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n / 2) / np.sum(h_init)
        dx = 2.0 / n

        # Optimization with multi-start
        bounds = [(0, 1) for _ in range(n)]
        cons = [{'type': 'eq', 'fun': lambda x: constraint(x, n)}]
        for _ in range(3):  # Multi-start
            perturb = np.random.uniform(-0.1, 0.1, size=n)
            h_init_perturb = np.clip(h_init + perturb, 0, 1)
            h_init_perturb = h_init_perturb * (n / 2) / np.sum(h_init_perturb)
            result = optimize.minimize(lambda x: objective(x, n, dx), h_init_perturb, 
                                     method='SLSQP', bounds=bounds, constraints=cons,
                                     tol=1e-6, options={'maxiter': 100})
            
            if result.success:

                    best_n = n
        time_used = time.time() - time_start
