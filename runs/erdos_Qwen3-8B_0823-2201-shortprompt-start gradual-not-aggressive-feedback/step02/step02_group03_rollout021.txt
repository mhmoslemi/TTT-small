def run(seed=42, budget_s=20, **kwargs):

    dx = 2.0 / n_points

    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode="full")) * dx
    best_n = n_points

        h = np.clip(h, 0, 1)

        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] = np.clip(h[i] + random.uniform(-0.2, 0.2), 0, 1)
        return h

    def crossover(h1, h2):
        h = h1.copy()
        for i in range(len(h)):
            if random.random() < 0.5:
                h[i] = h2[i]
        return h

    # Start with initial construction
    population = [initial_h.copy() for _ in range(50)]
    for _ in range(100):
        if time_left <= 0:
            break
        time_start = time.time()
        # Evaluate population
        fitness = [evaluate(h) for h in population]
        # Select best
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        if best_c5 < best_c5:
            best_c5 = best_c5
            best_h = population[best_idx].copy()

        # Generate new population
        new_population = []
        for _ in range(len(population)//2):
            # Select parents
            parent1 = random.choice(population)
            parent2 = random.choice(population)
            # Crossover
            child = crossover(parent1, parent2)
            # Mutate
            child = mutate(child)
            new_population.append(child)
        population = new_population
        time_used = time.time() - time_start
        time_left -= time_used
