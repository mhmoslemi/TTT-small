import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial guess: use the provided initial construction
    initial_h = np.array(initial_h_values)

    def objective(h):

        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    tournament_size = 5
    
    # Initialize population
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    best_solution = initial_h
    best_c5 = objective(initial_h)
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        
        # Select parents using tournament selection
        parents = []
        for _ in range(population_size):
            tournament = np.random.choice(np.arange(population_size), tournament_size)
            winner = tournament[np.argmin(fitness[tournament])]
            parents.append(population[winner])
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if i + 1 < population_size:
                parent1 = parents[i]
                parent2 = parents[i+1]
                child1 = np.zeros(n_points)
                child2 = np.zeros(n_points)
                for j in range(n_points):
                    if np.random.rand() < 0.5:
                        child1[j] = parent1[j]
                        child2[j] = parent2[j]
                    else:
                        child1[j] = parent2[j]
                        child2[j] = parent1[j]
                offspring.append(child1)
                offspring.append(child2)
        
        # Mutate
        for i in range(len(offspring)):
            offspring[i] = np.clip(offspring[i] + np.random.normal(0, mutation_rate, n_points), 0, 1)
            offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
        
        # Replace population
        population = np.array(offspring)
        
        # Update best solution
        current_best = population[np.argmin(fitness)]
        current_c5 = objective(current_best)
        if current_c5 < best_c5:
            best_solution = current_best
            best_c5 = current_c5
    
    # Final refinement with local optimization
    result = opt.minimize(
        objective,
        best_solution,
        method='L-BFGS-B',

        constraints={'type': 'eq', 'fun': constraint_sum},

    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
