import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_initial = len(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode="full")) * (2.0 / n_initial)
    best_n = n_initial
    time_left = budget_s

    def objective(x, n, dx):
        h = x.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint(x, n):
        h = x.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        return np.sum(h) - n / 2

    # Use initial construction as a starting point
    for n in [n_initial - 10, n_initial, n_initial + 10]:
        if n < 20:
            continue
        if time_left <= 0:
            break
        time_start = time.time()
        # Interpolate initial_h to the current n
        x_vals = np.linspace(0, 2, n_initial)
        x_new = np.linspace(0, 2, n)
        h_init = np.interp(x_new, x_vals, initial_h)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n / 2) / np.sum(h_init)
        dx = 2.0 / n

        # Optimization
        bounds = [(0, 1) for _ in range(n)]
        cons = [{'type': 'eq', 'fun': lambda x: constraint(x, n)}]
        result = optimize.minimize(lambda x: objective(x, n, dx), h_init, 
                                 method='SLSQP', bounds=bounds, constraints=cons,
                                 tol=1e-6, options={'maxiter': 100})
        
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n / 2) / np.sum(h_opt)
            corr = np.correlate(h_opt, 1 - h_opt, mode="full")
            c5 = np.max(corr * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n = n
        time_used = time.time() - time_start
        time_left -= time_used

    # Local search around the best solution
    if time_left > 1:
        n_local = best_n
        h_local = best_h.copy()
        dx = 2.0 / n_local
        h_local = np.clip(h_local, 0, 1)
        h_local = h_local * (n_local / 2) / np.sum(h_local)
        bounds = [(0, 1) for _ in range(n_local)]
        cons = [{'type': 'eq', 'fun': lambda x: constraint(x, n_local)}]
        result = optimize.minimize(lambda x: objective(x, n_local, dx), h_local, 
                                 method='L-BFGS-B', bounds=bounds, constraints=cons,
                                 tol=1e-6, options={'maxiter': 50})
        
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_local / 2) / np.sum(h_opt)
            corr = np.correlate(h_opt, 1 - h_opt, mode="full")
            c5 = np.max(corr * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt

    return best_h, best_c5, best_n
