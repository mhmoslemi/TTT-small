import numpy as np
import random
import time
from scipy.signal import correlate

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    if not np.isclose(sum_h, n_points / 2):
        initial_h = np.array([0.5] * n_points)
    
    def evaluate(h):
        h = np.array(h)
        if not np.all((h >= 0) & (h <= 1)):
            return float('inf')
        if not np.isclose(np.sum(h), n_points / 2):
            return float('inf')
        h_extended = np.concatenate([h, h, h])
        corr = correlate(h_extended, 1 - h, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    time_start = time.time()
    
    for step in range(1000):
        if time.time() - time_start > budget_s:
            break
        h = best_h.copy()
        i = np.random.randint(n_points)
        h[i] = np.random.uniform(0, 1)
        c5 = evaluate(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
        elif np.random.random() < np.exp(-(c5 - best_c5) / (step + 1)):
            best_h = h
            best_c5 = c5
    
    return (best_h, best_c5, n_points)
