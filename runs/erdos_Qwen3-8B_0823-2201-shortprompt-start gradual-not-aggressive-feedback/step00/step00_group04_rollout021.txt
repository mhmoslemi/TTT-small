import numpy as np
import scipy.optimize as optimize
import time

initial_h_values = np.load('initial_h_values.npy')  # Load initial solution

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    h = np.copy(initial_h_values[:n_points])
    h = h * (n_points / 2) / np.sum(h)  # Normalize to satisfy the integral constraint

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def objective_gradient(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        # Compute gradient numerically
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
            max_overlap_plus = np.max(corr_plus) * dx
            grad[i] = (max_overlap_plus - max_overlap) / 1e-5
        return grad

    # Use L-BFGS-B for optimization
    result = optimize.minimize(
        fun=objective,
        x0=h.flatten(),
        jac=objective_gradient,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
