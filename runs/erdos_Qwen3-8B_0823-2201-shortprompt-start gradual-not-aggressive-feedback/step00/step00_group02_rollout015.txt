import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    
    # Initial guess from provided initial_h_values
    initial_h_values = np.array(initial_h_values)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    if np.isclose(sum_h, n_points / 2):
        pass
    else:
        initial_h_values[-1] = n_points / 2 - sum_h + initial_h_values[-1]
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = np.concatenate([h, np.zeros_like(h)])
        shifted = np.concatenate([np.zeros_like(h), h[:-1]])
        overlap = np.correlate(h, 1 - shifted, mode="full") * dx
        return np.max(overlap)
    
    # Optimization
    start_time = time.time()
    result = optimize.minimize(
        compute_c5,
        initial_h_values,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
    )
    
    # Extract best solution
    h_values = result.x
    c5_bound = compute_c5(h_values)
    
    # Ensure total integral is 1
    dx = 2.0 / n_points
    if not np.isclose(np.sum(h_values) * dx, 1):
        h_values[-1] = (1 / dx - np.sum(h_values)) / dx
    
    # Final check
    c5_bound = compute_c5(h_values)
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        return (initial_h_values, compute_c5(initial_h_values), n_points)
    
    return (h_values, c5_bound, n_points)
