import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.array([0.5] * 45)  # Initial guess
    n_points = 45
    dx = 2.0 / n_points
    sum_h = n_points / 2  # Ensure normalization
    
    def objective(h):
        h = np.array(h, dtype=np.float64)
        h = np.clip(h, 0, 1)
        if np.sum(h) != sum_h:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def evaluate(h):
        return objective(h)
    
    # Genetic algorithm with tournament selection
    population_size = 50
    mutation_rate = 0.1
    elite_size = 5
    generations = 200
    start_time = time.time()
    
    # Initialize population
    population = np.random.rand(population_size, n_points)
    population = np.clip(population, 0, 1)
    population = (population * sum_h).astype(np.float64)
    
    best_h = population[0]
    best_c5 = evaluate(best_h)
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = np.array([evaluate(ind) for ind in population])
        
        # Select parents using tournament selection
        indices = np.random.choice(population_size, size=population_size, replace=True)
        parents = population[indices]
        
        # Crossover and mutation
        offspring = []
        for i in range(population_size):
            parent1 = parents[i]
            parent2 = parents[np.random.randint(population_size)]
            child = np.copy(parent1)
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[j] = np.random.rand() * sum_h
            child = np.clip(child, 0, sum_h)
            offspring.append(child)
        
        # Elitism
        elite_indices = np.argsort(fitness)[:elite_size]
        offspring = np.array(offspring)
        population = np.vstack((population[elite_indices], offspring))
        population = np.clip(population, 0, sum_h)
        
        # Update best
        current_fitness = np.array([evaluate(ind) for ind in population])
        best_idx = np.argmin(current_fitness)
        best_h = population[best_idx]
        best_c5 = current_fitness[best_idx]
    
    # Local optimization
    result = opt.minimize(
        objective,
        best_h,
        bounds=[(0, 1) for _ in range(n_points)],
        method='L-BFGS-B',
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    best_h = result.x
    best_c5 = evaluate(best_h)
    
    best_h = np.clip(best_h, 0, 1)
    best_h = (best_h * sum_h).astype(np.float64)
    best_h = np.round(best_h, 5)
    
    return best_h, best_c5, n_points
