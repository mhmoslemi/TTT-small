import numpy as np
import scipy.optimize as opt
import time

initial_h_values = np.load('initial_h.npy')  # Placeholder for initial solution

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Choose a reasonable number of points
    dx = 2.0 / n_points
    h = np.copy(initial_h_values[:n_points])  # Start from initial
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-6:
        h = np.clip(np.random.rand(n_points), 0.4, 0.6)  # Random initialization if needed

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint(h_vec):
        h = h_vec.reshape(n_points)
        return np.sum(h) - n_points / 2

    # Optimization
    start_time = time.time()
    result = opt.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 100}
    )
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return h, 0.38092, n_points

    # Simulated Annealing for further refinement
    def sa_objective(h_vec):
        return objective(h_vec)

    sa_result = opt.minimize(
        sa_objective,
        result.x,
        method='SLSQP',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 50}
    )

    h_values = sa_result.x
    c5_bound = sa_result.fun
    return h_values, c5_bound, n_points
