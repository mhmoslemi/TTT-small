import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Moderate number of samples for balance between accuracy and speed
    dx = 2.0 / n_points
    h_initial = np.zeros(n_points)
    # Initialize with a simple construction similar to the initial_h_values
    h_initial[:int(n_points/2)] = 1.0
    # Ensure the integral constraint is satisfied
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    def objective(h):
        h = h.reshape(n_points)
        # Ensure h is within [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    def constraint(h):
        h = h.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    # Optimization setup
    constraints = {'type': 'eq', 'fun': constraint}
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    result = optimize.minimize(
        objective, 
        h_initial, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (h_initial, 0.38092, n_points)
    
    best_h = result.x
    c5_bound = objective(best_h)
    return (best_h, c5_bound, n_points)
