import time

    def compute_c5(h):
        h_pad = np.zeros(2 * n_points)
        h_pad[:n_points] = h
        corr = np.correlate(h_pad, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h_flat.reshape(n_points)
        return compute_c5(h)

        return np.sum(h_flat) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial optimization with scipy
    result = optimize.minimize(objective, h, bounds=bounds, constraints=cons, method='SLSQP')
    h_best = result.x
    c5_best = compute_c5(h_best)
    
    # Simulated annealing
    temp = 1.0
    cooling_rate = 0.995
    iterations = 1000
    for _ in range(iterations):
