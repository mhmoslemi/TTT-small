import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # start with 100 samples
    
    # Initial solution from pre-imported variable
    initial_h = initial_h_values.copy()
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    h = initial_h.copy()
    
    def compute_c5(h):
        h = np.pad(h, (0, 1), mode='constant')
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        return compute_c5(h)
    
    def gradient_objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        grad = np.zeros_like(h)
        for k in range(1, len(h) - 1):
            grad[k] = np.sum(h[:len(h)-k] * (1 - h[k+1:])) - np.sum((1 - h[:len(h)-k]) * h[k+1:])
        return grad * (2.0 / n_points)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Use L-BFGS-B optimization
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        jac=gradient_objective,
        options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Try a few more iterations with smaller steps
    for _ in range(5):
        result = optimize.minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            jac=gradient_objective,
            options={'maxiter': 20, 'ftol': 1e-6, 'gtol': 1e-6, 'disp': False}
        )
        best_h = result.x
        best_c5 = compute_c5(best_h)
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final check if we can reduce n_points
    for n in range(50, 100, 10):
        h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(best_h)), best_h)
        h = np.clip(h, 0, 1)
        h = h * n / 2
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    return best_h, best_c5, n_points
