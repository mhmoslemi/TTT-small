import numpy as np
import scipy.optimize as optimize
import time

initial_h_values = np.load('initial_h.npy')  # Assuming initial_h.npy is provided

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.copy(initial_h_values[:n_points])
    h = h / np.sum(h) * (n_points / 2)
    
    def objective_function(h_flat):
        h = h_flat.reshape(n_points)
        integral = np.sum(h) * dx
        if np.abs(integral - 1) > 1e-6:
            return np.inf

        c5 = np.max(overlap) * dx
        return c5
    
    def constraint(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = optimize.minimize(
        objective_function,
        h.flatten(),
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    best_h = result.x.reshape(n_points)
    best_c5 = objective_function(best_h)
    
    return best_h, best_c5, n_points
