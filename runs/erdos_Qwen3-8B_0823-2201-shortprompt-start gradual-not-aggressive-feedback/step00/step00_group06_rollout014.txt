import scipy.optimize as opt
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial guess from provided initial_h_values
    n_points = len(initial_h_values)
    h_values = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points
    sum_h = np.sum(h_values)
    if not np.isclose(sum_h, n_points / 2):
        h_values = initial_h_values.copy()
        h_values = h_values / np.sum(h_values) * (n_points / 2)

    def objective(h):
        h = np.array(h, dtype=np.float64)
        sum_h = np.sum(h)
        if not np.isclose(sum_h, n_points / 2):
            return np.inf
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def gradient(h):
        h = np.array(h, dtype=np.float64)
        sum_h = np.sum(h)
        if not np.isclose(sum_h, n_points / 2):
            return np.inf * np.ones_like(h)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        # For simplicity, we use finite differences to approximate the gradient
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_minus = h.copy()
            h_minus[i] -= 1e-5
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * 1e-5)
        return grad

    # Optimization using L-BFGS-B
    result = opt.minimize(
        objective,
        h_values,
        method="L-BFGS-B",
        jac=gradient,
        options={"maxiter": 100, "ftol": 1e-8, "gtol": 1e-8}
    )

    # Early stopping if time is up
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:

    # Ensure integral constraint
    sum_h = np.sum(h_values)
    if not np.isclose(sum_h, n_points / 2):
        h_values = h_values / sum_h * (n_points / 2)

    c5_bound = evaluate_erdos_solution(h_values)
    return (h_values, c5_bound, n_points)
