import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as optimize
import random
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points for efficiency
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values[:n_points])  # Truncate to n_points
    
    # Function to compute C5
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = np.concatenate([h, np.zeros(n_points)])
        overlap = np.correlate(h, 1 - h[:n_points], mode="full")
        return np.max(overlap) * dx
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    elite_size = 5
    
    # Initial population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population[0] = initial_h
    
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate population
        fitness = [compute_c5(ind) for ind in population]
        
        # Find elite
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = [population[i] for i in elite_indices]
        best_h = elite[0]
        best_c5 = fitness[elite_indices[0]]
        
        # Create new population
        new_population = []
        for _ in range(population_size - elite_size):
            # Select parents
            parent1 = random.choice(elite)
            parent2 = random.choice(elite)
            
            # Crossover
            child = np.copy(parent1)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = parent2[i]
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] = np.random.uniform(0, 1)
            
            new_population.append(child)
        
        population = elite + new_population
    
    # Local optimization on best solution
    def objective_function(h_flat):
        h = h_flat.reshape(n_points)
        return compute_c5(h)
    
    result = optimize.minimize(objective_function, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points)
    best_h = result.x
    best_c5 = result.fun
    
    # Final check
    if time.time() - start_time > budget_s:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
