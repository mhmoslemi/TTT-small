import numpy as np
import random
from scipy.signal import correlate
import time

# Pre-imported helper functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.concatenate((h[-1:], h[:-1]))
    corr = correlate(h, 1 - shifted, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    return np.array([0.5 if i % 7 == 0 else 0 for i in range(70)]).tolist()

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = initial_h_values()
    best_c5 = evaluate_erdos_solution(best_h)
    n_points = len(best_h)
    
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        population = [best_h.copy() for _ in range(population_size)]
        for i in range(population_size):
            if random.random() < crossover_rate:
                j = random.randint(0, population_size - 1)
                child = population[i].copy()
                crossover_point = random.randint(1, n_points - 1)
                child[crossover_point:] = population[j][crossover_point:]
                population[i] = child
        
        for i in range(population_size):
            if random.random() < mutation_rate:
                idx = random.randint(0, n_points - 1)
                population[i][idx] = np.random.uniform(0, 1)
        
        for h in population:
            if np.mean(h) * 2 != n_points:
                h = np.array(h)
                h = h / np.mean(h) * n_points / 2
                h = h.tolist()
        
        for h in population:
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
        
        if best_c5 <= 0.380876:
            break
    
    return best_h, best_c5, n_points
