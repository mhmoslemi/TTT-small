import numpy as np
import scipy.optimize as opt
import time

initial_h_values = np.load('initial_h_values.npy')  # Assume this is provided

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h = np.copy(initial_h_values[:n_points])
    h = h[:n_points] * (n_points / 2) / np.sum(h[:n_points])  # Normalize to satisfy ∫h(x)dx=1

    def compute_c5(h):
        h = np.concatenate([h, np.zeros(n_points)])
        overlap = np.correlate(h, 1 - h[:n_points], mode="full")
        return np.max(overlap) * dx

    def objective(x):
        h = x.reshape(n_points)
        return compute_c5(h)

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    start_time = time.time()
    result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    elapsed = time.time() - start_time

    best_h = result.x
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_points)
