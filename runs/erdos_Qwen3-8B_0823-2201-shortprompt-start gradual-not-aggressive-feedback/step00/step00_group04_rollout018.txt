import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    
    # Initial number of points
    n_points = 100
    max_points = 300
    dx = 2.0 / n_points
    n_iterations = 50
    
    def compute_c5(h):
        h = np.array(h)
        h_extended = np.concatenate([h, h[:n_points-1]])
        result = np.correlate(h, 1 - h, mode="full")
        return np.max(result) * dx
    
    def initial_h_values(n_points):
        return np.random.rand(n_points) * 2 - 1
    
    h = initial_h_values(n_points)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    def obj_func(x):
        nonlocal best_c5
        if np.sum(x) != n_points / 2:
            return np.inf
        c5 = compute_c5(x)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()
        return c5
    
    start_time = time.time()
    
    for _ in range(n_iterations):
        if time.time() - start_time > budget_s - 1:
            break
        n_points = min(n_points + 20, max_points)
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        
        res = opt.minimize(obj_func, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        if res.success:
            h = res.x
            best_c5 = compute_c5(h)
            best_h = h.copy()
    
    return best_h, best_c5, n_points
