import numpy as np
import random
import copy
from scipy.signal import correlate
import time

initial_h_values = np.load('initial_h_values.npy')  # Assume this is loaded with 95 samples

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 95  # Start with the initial size, can reduce if needed
    dx = 2.0 / n_points
    n_pop = 50
    n_gen = 100
    mut_rate = 0.1
    
    # Initial population
    population = [copy.copy(initial_h_values) for _ in range(n_pop)]
    
    def evaluate(h):
        h = np.array(h)
        h = np.concatenate([h, h[:2]])  # Pad for full convolution
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    best_h = initial_h_values
    best_c5 = evaluate(initial_h_values)
    
    for gen in range(n_gen):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate all individuals
        fitness = [evaluate(h) for h in population]
        
        # Find the best individual
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx]
        
        # Create new population
        new_population = []
        for _ in range(n_pop // 2):
            # Select two parents
            parent1, parent2 = random.choices(population, k=2)
            
            # Crossover
            crossover_point = np.random.randint(1, n_points)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutate
            for i in range(n_points):
                if np.random.rand() < mut_rate:
                    child1[i] = np.random.rand()
                    child2[i] = np.random.rand()
            
            # Normalize to ensure sum(h) == n_points / 2
            def normalize(h):
                h = np.array(h)
                total = np.sum(h)
                if total == 0:
                    return np.zeros_like(h) + 0.5
                scale = (n_points / 2) / total
                return h * scale
            
            child1 = normalize(child1)
            child2 = normalize(child2)
            
            new_population.append(child1)
            new_population.append(child2)
        
        population = new_population
    
    return best_h, best_c5, n_points
