import scipy.optimize as opt
from scipy.signal import correlate

def run(seed=42, budget_s=20, **kwargs):

        np.random.seed(seed)
        h = np.random.rand(n) * 2 - 1  # Random initial guess
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        
        def objective(x):
            h_flat = x.reshape(n)
            h_flat = np.clip(h_flat, 0, 1)
            h_flat = h_flat * (n / 2) / np.sum(h_flat)
            corr = correlate(h_flat, 1 - h_flat, mode="full")
            return np.max(corr) * (2.0 / n)
        
        result = opt.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1)]*n)
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n / 2) / np.sum(best_h)
        c5 = np.max(correlate(best_h, 1 - best_h, mode="full")) * (2.0 / n)
        
        if c5 < 0.380876:
            return (best_h.tolist(), c5, n)
    
    # Default fallback
    return (initial_h_values.tolist(), 0.38092, len(initial_h_values))
