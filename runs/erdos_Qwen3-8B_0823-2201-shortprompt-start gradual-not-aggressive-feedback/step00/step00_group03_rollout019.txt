import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    if not np.isclose(sum_h, n_points / 2):
        initial_h = (initial_h * n_points / 2) / sum_h

    def objective(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint(h):
        h = h.reshape(-1)
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint}]

    start_time = time.time()
    result = optimize.minimize(
        objective,
        initial_h,
        bounds=bounds,
        constraints=constraints,
        method='L-BFGS-B',
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    elapsed = time.time() - start_time

    if elapsed > budget_s:
        return (initial_h, objective(initial_h), n_points)

    h_values = result.x
    c5_bound = objective(h_values)
    return (h_values, c5_bound, n_points)
