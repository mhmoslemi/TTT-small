import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)  # Normalize to satisfy the integral constraint
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Ensure integral constraint
        shifted = np.zeros_like(h)
        for k in range(1, n_points):
            shifted[k:] = h[:-k]
        overlap = np.max(np.convolve(h, 1 - h, mode='full') * dx)
        return overlap

    start_time = time.time()
    result = opt.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 100}
    )
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (h, np.max(np.convolve(h, 1 - h, mode='full') * dx), n_points)
    h_values = result.x
    c5_bound = np.max(np.convolve(h_values, 1 - h_values, mode='full') * dx)
    return (h_values, c5_bound, n_points)
