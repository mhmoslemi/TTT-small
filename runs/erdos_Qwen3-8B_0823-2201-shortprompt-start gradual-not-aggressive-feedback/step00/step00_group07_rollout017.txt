import scipy.optimize as optimize
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    if not np.isclose(sum_h, n_points / 2):
        initial_h = (initial_h * n_points / 2) / sum_h

    def objective(h_values):
        h = h_values.reshape(-1)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    result = optimize.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'max_iter': 100, 'disp': False}
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (initial_h, objective(initial_h), n_points)

    best_h = result.x
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
