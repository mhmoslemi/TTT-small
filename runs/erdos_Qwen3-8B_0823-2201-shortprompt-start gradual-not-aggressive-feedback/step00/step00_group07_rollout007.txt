        c5 = np.max(corr * dx)
        return c5
    
    def objective(x):
        return evaluate(x)
    
    def generate_population(pop_size):

                if np.random.rand() < 0.1:
                    h[i] = np.random.uniform(0, 1)
            h = np.clip(h, 0, 1)

                h = np.random.uniform(0, 1, n_points)
            pop.append(h)

        return [evaluate(ind) for ind in pop]

            if np.random.rand() < 0.5:
                child[i] = parent2[i]
