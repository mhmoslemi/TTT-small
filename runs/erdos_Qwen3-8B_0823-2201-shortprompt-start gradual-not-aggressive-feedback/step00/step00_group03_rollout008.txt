import numpy as np
import scipy.signal as signal
import cvxpy as cp
import time

initial_h_values = np.load('initial_h_values.npy')  # Assuming this is preloaded

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for efficiency
    dx = 2.0 / n_points
    h = np.copy(initial_h_values[:n_points])  # Use initial values as starting point
    h = h / np.sum(h) * (n_points / 2)  # Scale to satisfy integral constraint
    h = np.clip(h, 0, 1)  # Ensure values are in [0, 1]
    
    start_time = time.time()
    
    for _ in range(100):  # Perform a fixed number of iterations
        if time.time() - start_time > budget_s:
            break
        
        # Compute C₅
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Update h using a small perturbation
        perturbation = np.random.normal(0, 0.05, n_points)
        h += perturbation
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)  # Enforce integral constraint
        
        # Evaluate again
        corr = signal.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(corr) * dx
        
        if new_c5 < c5:
            c5 = new_c5
            h = np.copy(h)
    
    return h, c5, n_points
