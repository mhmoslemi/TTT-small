import numpy as np
import scipy.optimize as opt
import time
import copy

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = np.array(h)
        h = np.concatenate((h, h[:1]))  # To handle wrap-around
        overlap = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap) * dx
        return max_overlap
    
    def objective(h):
        return evaluate(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Initial optimization with L-BFGS-B
    result = opt.minimize(objective, initial_h, method='L-BFGS-B', constraints=cons, tol=1e-6)
    best_h = result.x
    best_c5 = evaluate(best_h)
    
    # Simulated annealing for exploration
    T = 1.0
    cooling_rate = 0.995
    for _ in range(100):
        if T < 1e-4:
            break
        for _ in range(100):
            new_h = copy.deepcopy(best_h)
            idx = np.random.randint(n_points)
            new_h[idx] = np.random.uniform(0, 1)
            if np.sum(new_h) * dx < 1 - 1e-6 or np.sum(new_h) * dx > 1 + 1e-6:

            if new_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - new_c5) / T):
                best_h = new_h
                best_c5 = new_c5
        T *= cooling_rate
    
    return best_h, best_c5, n_points
