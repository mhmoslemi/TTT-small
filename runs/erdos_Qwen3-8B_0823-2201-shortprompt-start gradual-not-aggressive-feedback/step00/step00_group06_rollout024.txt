import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    # Start with uniform distribution adjusted to meet integral constraint
    h[:n_points//2] = 1.0
    h = h / np.sum(h) * (n_points / 2)
    best_c5 = float('inf')
    best_h = h.copy()
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        # Ensure h is within [0,1]
        h = np.clip(h, 0, 1)
        # Enforce integral constraint
        h = h * (n_points / 2) / np.sum(h)
        # Compute overlap
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - (n_points / 2)
    
    # Optimization
    start_time = time.time()
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    corr = np.correlate(best_h, 1 - best_h, mode="full")
    best_c5 = np.max(corr) * dx
    
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Additional refinement if time allows
        result = optimize.minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            constraints={'type': 'eq', 'fun': constraint},

        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        best_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), best_c5, n_points)
