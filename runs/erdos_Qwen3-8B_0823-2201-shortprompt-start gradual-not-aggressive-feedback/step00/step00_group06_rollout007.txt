import numpy as np
import matplotlib.pyplot as plt
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Assuming it's preloaded

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()

    def evaluate_overlap(h):
        h = np.array(h)
        shifted = np.roll(h, 1)
        overlap = correlate(h, 1 - shifted, mode='full') * dx
        return np.max(overlap)

    def objective(h):
        return evaluate_overlap(h)

    def constraints(h):
        return np.sum(h) - n_points / 2

    def bounds(h):
        return [(0, 1) for _ in h]

    # Genetic algorithm
    population_size = 50
    generations = 20
    mutation_rate = 0.1
    elite_fraction = 0.2

    def evolve_population(population):
        population = np.array(population)
        fitness = np.array([evaluate_overlap(ind) for ind in population])
        elite_size = int(elite_fraction * population_size)
        elites = population[np.argsort(fitness)[:elite_size]]
        new_population = elites.copy()
        while len(new_population) < population_size:
            parent1 = np.random.choice(elite_size, 1)[0]
            parent2 = np.random.choice(elite_size, 1)[0]
            child = (population[parent1] + population[parent2]) / 2
            if np.random.rand() < mutation_rate:
                child += np.random.uniform(-0.1, 0.1, size=len(child))
                child = np.clip(child, 0, 1)
            new_population = np.vstack((new_population, child))
        return new_population[:population_size]

    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population[0] = h
    best_fitness = float('inf')
    best_h = h.copy()

    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        population = evolve_population(population)
        for ind in population:
            if evaluate_overlap(ind) < best_fitness:
                best_fitness = evaluate_overlap(ind)
                best_h = ind.copy()

    return best_h, best_fitness, n_points
