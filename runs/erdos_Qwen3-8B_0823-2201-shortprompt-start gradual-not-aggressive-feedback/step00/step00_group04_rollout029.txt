import numpy as np
import scipy.optimize as optimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    initial_h[:n_points//2] = 1.0
    initial_h[n_points//2:] = 0.5
    h_values = initial_h.copy()
    
    def objective(h):
        h = np.array(h, dtype=float)
        h = np.concatenate((h, h[:n_points//2]))
        overlap = correlate(h, 1 - h[:n_points], mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def constraint(h):
        h = np.array(h, dtype=float)
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint}
    
    start_time = time.time()
    result = optimize.minimize(objective, h_values, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-5)
    elapsed = time.time() - start_time
    
    if elapsed > budget_s:
        return h_values, 0.38092, n_points
    
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
