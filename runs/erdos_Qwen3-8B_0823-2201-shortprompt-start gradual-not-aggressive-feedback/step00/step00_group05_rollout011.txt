import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Assume pre-loaded initial solution
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h.reshape(-1, 1)
        shift = np.zeros((n_points, n_points))
        for i in range(n_points):
            shift[i, i+1:] = 1
        corr = np.dot(h, (1 - h).T)
        c5 = np.max(corr) * dx
        return c5

    def optimize_h(h):
        result = opt.minimize(lambda x: objective(x), h, method='L-BFGS-B', options={'maxiter': 100})
        return result.x

    start_time = time.time()
    best_h = h
    best_c5 = objective(h)
    
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        h = np.random.uniform(0, 1, n_points)
        h = optimize_h(h)
        c5 = objective(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    return best_h, best_c5, n_points
