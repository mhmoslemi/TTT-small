import cvxpy as cp
import time

evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Reduced to ensure under 1k samples
    dx = 2.0 / n_points
    h = np.random.rand(n_points)
    h = h * (n_points / 2)  # Ensure integral is 1

    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Define optimization variables
        x = cp.Variable(n_points, nonneg=True)
        constraints = [
            cp.sum(x) == n_points / 2,
            0 <= x,
            x <= n_points
        ]

        # Define the overlap integrals for all possible shifts
        overlap = np.zeros(n_points)
        for k in range(1, n_points):
            overlap[k] = np.max(np.correlate(x.value, 1 - x.value, mode="full")) * dx

        # Objective: minimize the maximum overlap
        obj = cp.Minimize(cp.max(overlap))

        # Solve the optimization problem
        try:
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.SCS)
            if prob.status == 'optimal':
                h = x.value
                current_c5 = evaluate_erdos_solution(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        except:
            pass  # Skip if solver fails

    return best_h, best_c5, n_points
