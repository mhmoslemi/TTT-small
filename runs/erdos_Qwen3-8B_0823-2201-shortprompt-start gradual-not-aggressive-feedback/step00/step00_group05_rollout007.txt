import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

initial_h_values = np.load('initial_h_values.npy')  # Assume this is preloaded

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_c5 = float('inf')
    best_h = h.copy()

    def objective(x):
        nonlocal best_c5
        h_new = x.reshape(n_points)
        h_new = np.clip(h_new, 0, 1)
        corr = signal.correlate(h_new, 1 - h_new, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        return c5

    # Use L-BFGS-B for optimization
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 100, 'disp': False}
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (best_h, best_c5, n_points)

    return (best_h, best_c5, n_points)

start_time = time.time()
