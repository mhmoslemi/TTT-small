import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h_values = initial_h.copy()
    
    def objective(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        h = np.concatenate([h, h[:1]])  # extend for full convolution
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def perturb(h, step_size=0.05):
        h = h.copy()
        idx = np.random.randint(len(h))
        h[idx] += np.random.uniform(-step_size, step_size)
        h = np.clip(h, 0, 1)
        return h
    
    start_time = time.time()
    best_h = h_values.copy()
    best_c5 = objective(best_h)
    
    for _ in range(10000):
        if time.time() - start_time > budget_s:
            break
        h = perturb(best_h)
        c5 = objective(h)
        if c5 < best_c5 or np.random.rand() < np.exp((best_c5 - c5) / 100):
            best_h = h
            best_c5 = c5
    
    return (best_h, best_c5, n_points)
