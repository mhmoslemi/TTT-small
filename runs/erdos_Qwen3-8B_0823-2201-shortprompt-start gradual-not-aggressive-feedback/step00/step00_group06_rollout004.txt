import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    sum_h = np.sum(initial_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        initial_h = np.array([0.5] * n_points)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        h = h.reshape(-1, 1)
        shifted = np.roll(h, shift=np.arange(1, n_points+1), axis=0)
        overlap = np.sum(h * (1 - shifted), axis=0)
        c5 = np.max(overlap) * dx
        return c5
    
    def perturb(h, temp):
        h = h.copy()
        idx = np.random.choice(n_points, size=min(5, n_points), replace=False)
        for i in idx:
            h[i] += np.random.uniform(-0.2, 0.2)
            h[i] = np.clip(h[i], 0, 1)
        return h
    
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = objective(h)
    temp = 1.0
    cooling_rate = 0.995
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        for _ in range(100):
            new_h = perturb(h, temp)
            new_c5 = objective(new_h)

                if best_c5 < objective(best_h):
                    best_h = h.copy()
        temp *= cooling_rate
