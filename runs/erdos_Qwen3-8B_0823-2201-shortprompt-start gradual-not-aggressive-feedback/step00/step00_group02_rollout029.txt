import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points, can increase if time permits
    dx = 2.0 / n_points
    h = np.array(initial_h_values[:n_points], dtype=float)

    def objective(h_values):
        h = h_values.copy()
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2

    cons = ({'type': 'eq', 'fun': constraint_sum})
    bounds = Bounds(0, 1)

    # Optimization with scipy
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 100}
    )

    # Check if we can improve with a local search
    h_best = result.x
    c5_best = objective(h_best)
    elapsed = time.time() - time.time()  # This is a placeholder; actual elapsed time tracking would be needed

    # If time remains, perform a local search
    if elapsed < budget_s - 2:
        # Perform a simple perturbation-based search
        for _ in range(5):
            perturb = np.random.normal(0, 0.05, size=n_points)
            h_perturbed = np.clip(h_best + perturb, 0, 1)
            c5_perturbed = objective(h_perturbed)
            if c5_perturbed < c5_best:
                h_best = h_perturbed
                c5_best = c5_perturbed

    # Evaluate the best solution
    c5_bound = evaluate_erdos_solution(h_best)
    return h_best, c5_bound, n_points
