import numpy as np
import scipy.optimize as optimize
import time

initial_h_values = np.load('initial_h.npy')  # Assume this is pre-loaded with an initial guess

def evaluate_erdos_solution(h):
    dx = 2.0 / len(h)
    overlap = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use 100 points for a balance between speed and precision
    h = initial_h_values[:n_points].copy()
    dx = 2.0 / n_points
    h /= (n_points / 2) * dx  # Normalize to satisfy the integral constraint
    
    def objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        overlap = np.correlate(h_vals, 1 - h_vals, mode="full") * dx
        return np.max(overlap)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    
    start_time = time.time()
    result = optimize.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    
    best_h = result.x
    c5_bound = evaluate_erdos_solution(best_h)
    
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Perform a local search with simulated annealing
        def local_search(h_init):
            h = h_init.copy()
            T = 1.0
            T_min = 1e-3
            steps = 1000
            for step in range(steps):
                T *= 0.995
                if T < T_min:
                    break
                idx = np.random.randint(0, n_points)
                delta = np.random.uniform(-0.1, 0.1)
                h_new = h.copy()
                h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
                overlap_new = evaluate_erdos_solution(h_new)
                overlap_current = evaluate_erdos_solution(h)
                if np.random.rand() < np.exp((overlap_new - overlap_current) / T):
                    h = h_new
            return h
        
        best_h = local_search(best_h)
        c5_bound = evaluate_erdos_solution(best_h)
    
    return best_h, c5_bound, n_points
