import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Genetic algorithm for global search
    population_size = 20
    generations = 50
    mutation_rate = 0.1

    def evolve(population):
        population = [np.clip(ind, 0, 1) * target_sum / np.sum(np.clip(ind, 0, 1)) for ind in population]
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_individual = population[best_idx]
        best_score = fitness[best_idx]
        
        # Selection and crossover
        selected = np.argsort(fitness)[:population_size // 2]
        selected = np.random.choice(selected, size=population_size, replace=True)
        new_population = []
        
        for i in range(population_size):
            parent1 = population[selected[i]]
            parent2 = population[np.random.choice(selected)]
            child = (parent1 + np.random.normal(0, mutation_rate, n_points)) * target_sum / np.sum(np.clip(parent1 + np.random.normal(0, mutation_rate, n_points), 0, 1))
            new_population.append(child)
        
        return new_population

    population = [initial_h.copy() + np.random.normal(0, 0.1, n_points) for _ in range(population_size)]
    population = [np.clip(ind, 0, 1) * target_sum / np.sum(np.clip(ind, 0, 1)) for ind in population]
    
    for _ in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
        population = evolve(population)
        current_best = min(population, key=objective)
        current_score = objective(current_best)
        if current_score < best_c5:
            best_h = current_best
            best_c5 = current_score

    # Local refinement with SLSQP
    while time.time() - start_time < budget_s - 1:
        perturb_scale = 0.05 * (1 - (time.time() - start_time) / (budget_s - 1))
        perturb = np.random.normal(0, perturb_scale, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
