import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Initial optimization with SLSQP
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Genetic algorithm for global search
    population_size = 50
    generations = 50
    mutation_rate = 0.1
    mutation_scale = 0.05

    def generate_individual():
        individual = np.random.uniform(0, 1, n_points)
        individual = individual * target_sum / np.sum(individual)
        return individual

    population = [generate_individual() for _ in range(population_size)]

    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_fitness = fitness[best_idx]
        
        if best_fitness < best_c5:
            best_h = population[best_idx]
            best_c5 = best_fitness
        
        # Selection
        selected = np.random.choice(population_size, size=population_size, p=np.array(fitness)/np.sum(fitness))
        selected = [population[i] for i in selected]
        
        # Crossover
        new_population = []
        for i in range(population_size // 2):
            parent1 = selected[2*i]
            parent2 = selected[2*i + 1]
            child1 = np.zeros(n_points)
            child2 = np.zeros(n_points)
            for j in range(n_points):
                if np.random.rand() < 0.5:
                    child1[j] = parent1[j]
                    child2[j] = parent2[j]
                else:
                    child1[j] = parent2[j]
                    child2[j] = parent1[j]
            new_population.append(child1)
            new_population.append(child2)
        
        # Mutation
        for i in range(population_size):
            individual = new_population[i]
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    individual[j] += np.random.normal(0, mutation_scale)
                    individual[j] = np.clip(individual[j], 0, 1)
                    individual = individual * target_sum / np.sum(individual)
            new_population[i] = individual
        
        population = new_population

    return best_h, best_c5, n_points
