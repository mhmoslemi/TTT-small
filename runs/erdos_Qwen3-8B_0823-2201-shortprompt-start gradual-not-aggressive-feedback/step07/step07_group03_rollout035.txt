import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Gradient-based refinement
    def gradient_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Genetic algorithm
    def genetic_algorithm():
        population_size = 20
        mutation_rate = 0.1
        crossover_rate = 0.8
        generations = 50

        for _ in range(generations):
            population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]

                population[i] = population[i] * target_sum / np.sum(population[i])
            
            # Evaluate population
            fitness = [objective(ind) for ind in population]
            best_idx = np.argmin(fitness)
            best_h = population[best_idx]
            best_c5 = fitness[best_idx]
            
            # Genetic operators
            new_population = []
            for i in range(population_size // 2):
                parent1 = population[np.random.choice(range(population_size))]
                parent2 = population[np.random.choice(range(population_size))]
                
                # Crossover

                    child = np.zeros(n_points)

                            child[j] = parent2[j]

                new_population.append(child)
            
            population = new_population
            population.append(best_h)
        
        return best_h, best_c5
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()

    return best_h, best_c5, n_points
