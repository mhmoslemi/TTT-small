import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Genetic algorithm parameters
    pop_size = 20
    generations = 10
    mutation_rate = 0.1
    elite_size = 2

    # Initial population
    population = np.random.rand(pop_size, n_points)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    population = np.clip(population, 0, 1)

    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = population[elite_indices]
        
        # Generate offspring
        offspring = []
        for _ in range(pop_size - elite_size):
            parent1, parent2 = np.random.choice(elite_indices, size=2, replace=False)
            child = np.copy(population[parent1])
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] = np.clip(child[i] + np.random.normal(0, 0.05), 0, 1)
            offspring.append(child)
        
        # Create new population
        population = np.vstack((elites, offspring))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        population = np.clip(population, 0, 1)
        
        # Update best solution
        for ind in population:
            current_c5 = objective(ind)
            if current_c5 < best_c5:
                best_h = ind.copy()
                best_c5 = current_c5

    # Local search with more refined perturbations
    while time.time() - start_time < budget_s - 1:
        # Generate perturbation with adaptive magnitude
        perturb_scale = 0.05 * (1 - (time.time() - start_time) / (budget_s - 1))
        perturb = np.random.normal(0, perturb_scale, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Optimization with fewer iterations
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
