import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize population
    population_size = 50
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    def evaluate(individual):
        return objective(individual)
    
    def tournament_selection(pop, k=5):
        selected = np.random.choice(pop, size=k)
        return selected[np.argmin([evaluate(ind) for ind in selected])]
    
    def crossover(parent1, parent2):
        alpha = np.random.rand(n_points)
        return alpha * parent1 + (1 - alpha) * parent2
    
    def mutate(individual, mutation_rate=0.1):
        mutated = individual.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                mutated[i] = np.random.uniform(0, 1)
        mutated = mutated * target_sum / np.sum(mutated)
        return mutated
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        # Evaluate population
        fitness = np.array([evaluate(ind) for ind in population])
        # Select parents
        parents = [tournament_selection(population) for _ in range(population_size)]
        # Crossover and mutate
        offspring = []
        for p1, p2 in zip(parents[::2], parents[1::2]):
            child = crossover(p1, p2)
            child = mutate(child)
            offspring.append(child)
        # Replace population
        population = np.array(offspring)
        # Update best solution
        for ind in population:
            current_c5 = evaluate(ind)
            if current_c5 < best_c5:
                best_h = ind.copy()
                best_c5 = current_c5
    
    # Final refinement with local optimization
    best_h = best_h * target_sum / np.sum(best_h)
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
