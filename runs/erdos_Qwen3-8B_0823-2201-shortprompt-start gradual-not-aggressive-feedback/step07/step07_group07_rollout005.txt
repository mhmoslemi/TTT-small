import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Genetic algorithm parameters
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    mutation_scale = 0.05

    # Initialize population
    population = np.random.rand(population_size, n_points)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)

    for _ in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = np.array([objective(ind) for ind in population])
        best_index = np.argmin(fitness)
        best_h = population[best_index].copy()
        best_c5 = fitness[best_index]
        
        # Selection
        selected = np.random.choice(population_size, size=population_size, p=fitness / fitness.sum())
        selected = population[selected]
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if i + 1 < population_size:
                parent1 = selected[i]
                parent2 = selected[i+1]
                child1 = np.concatenate((parent1[:n_points//2], parent2[n_points//2:]))
                child2 = np.concatenate((parent2[:n_points//2], parent1[n_points//2:]))
                offspring.append(child1)
                offspring.append(child2)
            else:
                offspring.append(selected[i])
        
        # Mutation
        for i in range(len(offspring)):
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    offspring[i][j] += np.random.normal(0, mutation_scale)
                    offspring[i][j] = np.clip(offspring[i][j], 0, 1)
                    offspring[i][j] = offspring[i][j] * target_sum / np.sum(offspring[i])
        
        # Replace population
        population = np.array(offspring)
        
        # Local refinement
        for i in range(len(population)):
            if time.time() - start_time > budget_s - 1:
                break
            perturbed_h = population[i].copy()
            result = opt.minimize(
                objective,
                perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 100, 'ftol': 1e-9}
            )
            population[i] = result.x
            population[i] = np.clip(population[i], 0, 1)
            population[i] = population[i] * target_sum / np.sum(population[i])
    
    return best_h, best_c5, n_points
