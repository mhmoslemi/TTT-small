import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.15
    
    def crossover(parent1, parent2):
        alpha = np.random.rand(n_points)
        return alpha * parent1 + (1 - alpha) * parent2
    
    def mutate(individual):
        mutation = np.random.normal(0, mutation_rate, n_points)
        return np.clip(individual + mutation, 0, 1)
    
    # Initialize population
    population = [np.random.rand(n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    start_time = time.time()
    
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        # Select parents
        parents = np.argsort(fitness)[:population_size//2]
        parents = [population[i] for i in parents]
        # Crossover and mutate
        offspring = []
        for _ in range(population_size//2):
            p1, p2 = np.random.choice(parents, size=2, replace=False)
            child = crossover(p1, p2)
            child = mutate(child)
            child = child * target_sum / np.sum(child)
            offspring.append(child)
        # Replace population
        population = parents + offspring
        # Local refinement
        for i in range(len(population)):
            result = opt.minimize(
                objective,
                population[i],
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            population[i] = result.x
    
    # Select best solution
    best_h = min(population, key=objective)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
