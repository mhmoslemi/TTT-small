import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)

    population_size = 20
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Genetic algorithm
    def genetic_algorithm():
        population = [np.random.rand(n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        best_h = population[0]
        best_c5 = objective(best_h)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = [objective(h) for h in population]

            best_c5 = fitness[best_idx]
            
            # Selection
            selected = np.random.choice(population_size, size=population_size, p=fitness/np.sum(fitness))
            
            # Crossover
            new_population = []

                j = np.random.randint(population_size)
                if np.random.rand() < crossover_rate:
                    child = (population[selected[i]] + population[selected[j]]) / 2
                else:
                    child = population[selected[i]]
                new_population.append(child)

            population = new_population
