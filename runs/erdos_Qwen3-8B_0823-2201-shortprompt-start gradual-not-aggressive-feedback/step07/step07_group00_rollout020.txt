import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 50
    
    def genetic_algorithm():
        population = [np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = [objective(h) for h in population]
            
            # Select parents
            parents = np.array(population)[np.argsort(fitness)[:int(population_size/2)]]
            
            # Create new population
            new_population = []
            for _ in range(population_size):
                # Crossover
                if np.random.rand() < crossover_rate:
                    parent1, parent2 = np.random.choice(parents, 2, replace=False)
                    child = np.concatenate([
                        parent1[:n_points//2],
                        parent2[n_points//2:]
                    ])
                else:
                    child = np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))

                    child = np.clip(child, 0, 1)
                    child = child * target_sum / np.sum(child)
                
                new_population.append(child)
            
            population = new_population
        
        best_h = min(population, key=objective)
        return best_h, objective(best_h)
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()

    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        # Adaptive perturbation based on time remaining
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.05 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}

        current_c5 = objective(current_h)

            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
