import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Genetic algorithm parameters
    pop_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 100
    
    def genetic_algorithm():
        # Initialize population
        population = np.random.uniform(0, 1, (pop_size, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        best_h = population[0]
        best_c5 = objective(best_h)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            
            # Select parents
            parents = population[np.argsort(fitness)]
            
            # Create new population
            new_population = []
            for _ in range(pop_size // 2):
                # Crossover
                if np.random.rand() < crossover_rate:
                    parent1, parent2 = np.random.choice(parents, size=2, replace=False)
                    child1 = np.concatenate((parent1[:n_points//2], parent2[n_points//2:]))
                    child2 = np.concatenate((parent2[:n_points//2], parent1[n_points//2:]))
                else:
                    child1, child2 = parents[0].copy(), parents[0].copy()
                
                # Mutation
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child1[i] = np.clip(child1[i] + np.random.uniform(-0.1, 0.1), 0, 1)
                        child2[i] = np.clip(child2[i] + np.random.uniform(-0.1, 0.1), 0, 1)
                
                # Normalize
                child1 = child1 * target_sum / np.sum(child1)
                child2 = child2 * target_sum / np.sum(child2)
                
                new_population.append(child1)
                new_population.append(child2)
            
            # Update population
            population = np.array(new_population)
            population = population * target_sum / np.sum(population, axis=1, keepdims=True)
            
            # Update best solution
            current_best = population[np.argmin(fitness)]
            current_c5 = objective(current_best)
            
            if current_c5 < best_c5:
                best_h = current_best
                best_c5 = current_c5
        
        return best_h, best_c5
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        # Random perturbation with adaptive scale
        remaining_time = budget_s - (time.time() - start_time)
        perturb_scale = 0.15 * (remaining_time / budget_s)
        perturb = np.random.uniform(-perturb_scale, perturb_scale, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
