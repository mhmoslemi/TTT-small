import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize population
    population = np.random.uniform(0, 1, (30, n_points))
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    def evaluate_population(pop):
        return np.array([objective(ind) for ind in pop])
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate population
        fitness = evaluate_population(population)
        best_idx = np.argmin(fitness)
        current_best = population[best_idx].copy()
        current_c5 = fitness[best_idx]
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
        
        # Select parents
        parents = population[np.argsort(fitness)[:10]]
        
        # Crossover
        offspring = []
        for _ in range(10):
            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            child = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            offspring[i] = np.clip(offspring[i] + np.random.normal(0, 0.05, n_points), 0, 1)
            offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
        
        # Replace population
        population = np.vstack((parents, offspring))
        np.random.shuffle(population)
        population = population[:30]
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
