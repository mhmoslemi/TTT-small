import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Genetic algorithm initialization
    population_size = 20
    population = np.random.rand(population_size, n_points)
    population = (population - 0.5) * 2  # Initialize with random values in [0, 1]
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    # Evaluate initial population
    fitness = np.array([objective(ind) for ind in population])
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    # Genetic algorithm parameters
    mutation_rate = 0.1
    crossover_rate = 0.8
    elites = 2
    
    for _ in range(100):  # Max generations
        if time.time() - start_time > budget_s - 1:
            break
        
        # Select parents
        parents = population[np.argsort(fitness)[:elites]]
        
        # Generate offspring
        offspring = []
        for _ in range(population_size - elites):
            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            else:
                child = np.random.rand(n_points)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, 0.05, n_points)
                child += mutation
                child = np.clip(child, 0, 1)
            
            # Normalize
            child = child * target_sum / np.sum(child)
            offspring.append(child)
        
        # Combine population
        population = np.vstack((parents, np.array(offspring)))
        fitness = np.array([objective(ind) for ind in population])
        
        # Update best solution
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]
    
    # Local refinement with SLSQP
    while time.time() - start_time < budget_s - 1:
        # Generate perturbation with adaptive magnitude
        perturb_scale = 0.05 * (1 - (time.time() - start_time) / (budget_s - 1))
        perturb = np.random.normal(0, perturb_scale, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Optimization with fewer iterations
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
