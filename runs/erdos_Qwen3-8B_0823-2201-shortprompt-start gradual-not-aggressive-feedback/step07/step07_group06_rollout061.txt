import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Genetic algorithm parameters
    population_size = 20
    generations = 20
    mutation_rate = 0.1
    crossover_rate = 0.8

    # Initial population
    population = [initial_h.copy() for _ in range(population_size)]
    population = [np.random.rand(n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]

    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break

        # Evaluate population
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]

        # Selection
        selected = np.random.choice(len(population), size=population_size, p=np.array(fitness)/np.sum(fitness))

        # Crossover
        new_population = []
        for i in range(population_size):
            parent1 = population[selected[i]]
            if i < population_size//2:
                parent2 = population[selected[(i+1)%population_size]]
                child = np.zeros(n_points)
                for j in range(n_points):
                    if np.random.rand() < crossover_rate:
                        child[j] = parent1[j]
                    else:
                        child[j] = parent2[j]
                new_population.append(child)
            else:
                new_population.append(population[selected[i]])

        # Mutation
        for i in range(population_size):
            h = new_population[i]
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    h[j] += np.random.normal(0, 0.05)
                    h[j] = np.clip(h[j], 0, 1)
            new_population[i] = h * target_sum / np.sum(h)

        population = new_population

    # Refinement with SLSQP
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
