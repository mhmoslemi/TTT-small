import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Genetic algorithm parameters
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    crossover_rate = 0.8

    # Initial population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]

    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]

        # Selection
        selected = np.random.choice(population_size, size=population_size, p=1 / np.array(fitness))
        selected = [population[i] for i in selected]

        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if np.random.random() < crossover_rate:
                p1, p2 = selected[i], selected[i+1]
                child1 = np.concatenate((p1[:n_points//2], p2[n_points//2:]))
                child2 = np.concatenate((p2[:n_points//2], p1[n_points//2:]))
                offspring.append(child1)
                offspring.append(child2)
            else:
                offspring.append(selected[i])
                offspring.append(selected[i+1])

        # Mutation
        for i in range(len(offspring)):
            for j in range(n_points):
                if np.random.random() < mutation_rate:
                    offspring[i][j] += np.random.normal(0, 0.05)
                    offspring[i][j] = np.clip(offspring[i][j], 0, 1)
                    offspring[i][j] = offspring[i][j] * target_sum / np.sum(offspring[i])

        # Replace population
        population = offspring

    # Local search with adaptive perturbations
    while time.time() - start_time < budget_s - 1:
        perturb_scale = 0.05 * (1 - (time.time() - start_time) / budget_s)
        perturb = np.random.normal(0, perturb_scale, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Optimization with fewer iterations
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
