import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    pop_size = 20
    elites = 2
    mutation_rate = 0.15
    crossover_rate = 0.8
    generations = 50
    time_per_gen = budget_s / generations
    
    def genetic_algorithm():
        population = [np.random.rand(n_points) for _ in range(pop_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        best_h = h.copy()
        best_c5 = objective(best_h)
        
        for gen in range(generations):
            start_time = time.time()
            # Evaluate fitness
            fitness = [objective(ind) for ind in population]
            # Select elites
            elite_indices = np.argsort(fitness)[:elites]
            elites = [population[i] for i in elite_indices]
            # Generate new population
            new_population = elites.copy()
            while len(new_population) < pop_size:
                # Select parents
                parent1 = random.choices(population)[0]
                parent2 = random.choices(population)[0]
                # Crossover
                if random.random() < crossover_rate:
                    crossover_point = random.randint(1, n_points - 1)
                    child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
                else:
                    child = parent1.copy()
                # Mutation
                if random.random() < mutation_rate:
                    mutation = np.random.normal(0, 0.05, n_points)
                    child += mutation
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
                new_population.append(child)
            population = new_population
            # Update best solution
            current_best = min(zip(population, fitness), key=lambda x: x[1])
            if current_best[1] < best_c5:
                best_h = current_best[0].copy()
                best_c5 = current_best[1]
            # Check time
            if time.time() - start_time > time_per_gen:
                break
        return best_h, best_c5
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - time.time() < budget_s:
        # Adaptive perturbation based on time remaining
        remaining_time = budget_s - (time.time() - time.time())
        perturb_size = max(0.01, 0.1 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
