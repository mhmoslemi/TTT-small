import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Initial optimization with SLSQP
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Genetic algorithm with dynamic population size
    population_size = 20
    mutation_rate = 0.1
    generations = int((budget_s - (time.time() - start_time)) * 2)
    
    for _ in range(generations):
        population = np.random.uniform(0, 1, (population_size, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        # Evaluate population
        fitness = np.array([objective(ind) for ind in population])
        
        # Select elite
        elite_index = np.argmin(fitness)
        elite = population[elite_index]
        
        # Create new population
        new_population = [elite]
        for _ in range(population_size - 1):
            parent1 = random.choice(population)
            parent2 = random.choice(population)
            child = (parent1 + parent2) / 2
            if random.random() < mutation_rate:
                mutation = np.random.normal(0, 0.05, n_points)
                child += mutation
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        population = np.array(new_population)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        # Update best solution
        fitness = np.array([objective(ind) for ind in population])
        best_index = np.argmin(fitness)
        if fitness[best_index] < best_c5:
            best_h = population[best_index]
            best_c5 = fitness[best_index]
    
    return best_h, best_c5, n_points
