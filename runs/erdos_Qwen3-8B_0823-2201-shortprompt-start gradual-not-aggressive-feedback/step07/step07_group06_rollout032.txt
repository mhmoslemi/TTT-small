import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Initial optimization with more iterations
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Genetic algorithm for global exploration
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    elite_fraction = 0.2

    def mutate(individual):
        mutation = np.random.normal(0, 0.05, n_points)
        mutated = individual + mutation
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * target_sum / np.sum(mutated)
        return mutated

    def crossover(parent1, parent2):
        child = np.copy(parent1)
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        child[indices] = parent2[indices]
        child = np.clip(child, 0, 1)
        child = child * target_sum / np.sum(child)
        return child

    population = [best_h.copy() for _ in range(population_size)]
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        # Select elites
        elite_indices = np.argsort(fitness)[:int(elite_fraction*population_size)]
        elites = [population[i] for i in elite_indices]
        # Generate new population
        new_population = elites
        while len(new_population) < population_size:
            # Randomly select parents
            parent1 = random.choice(elites)
            parent2 = random.choice(elites)
            child = crossover(parent1, parent2)
            if random.random() < mutation_rate:
                child = mutate(child)
            new_population.append(child)
        population = new_population

    # Find the best solution from the final population
    final_fitness = [objective(ind) for ind in population]
    best_index = np.argmin(final_fitness)
    best_h = population[best_index]
    best_c5 = final_fitness[best_index]

    return best_h, best_c5, n_points
