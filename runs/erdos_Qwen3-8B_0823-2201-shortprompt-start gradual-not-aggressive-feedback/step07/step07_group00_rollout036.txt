import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize population
    population_size = 20
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    def evaluate(individual):
        return objective(individual)
    
    def mutate(individual, mutation_rate=0.1, mutation_strength=0.1):
        mutated = individual.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                mutated[i] += np.random.uniform(-mutation_strength, mutation_strength)
                mutated[i] = np.clip(mutated[i], 0, 1)
        mutated = mutated * target_sum / np.sum(mutated)
        return mutated
    
    def crossover(parent1, parent2):
        child = np.zeros(n_points)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        child = child * target_sum / np.sum(child)
        return child
    
    best_h = np.array(initial_h, dtype=np.float64)
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for _ in range(100):  # Evolutionary iterations
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate population
        fitness = np.array([evaluate(ind) for ind in population])
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_c5 = fitness[best_idx]
        
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            # Selection
            parent1 = population[np.random.choice(population_size)]
            parent2 = population[np.random.choice(population_size)]
            
            # Crossover
            child = crossover(parent1, parent2)
            
            # Mutation
            mutated_child = mutate(child)
            
            new_population.append(mutated_child)
        
        population = np.array(new_population)
        
        # Local refinement with SLSQP
        for i in range(len(population)):
            result = opt.minimize(
                objective,
                population[i],
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            population[i] = result.x
    
    # Final refinement
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
