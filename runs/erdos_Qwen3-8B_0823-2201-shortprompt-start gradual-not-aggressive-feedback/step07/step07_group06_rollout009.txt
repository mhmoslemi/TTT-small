import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Genetic algorithm for global exploration
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    
    for _ in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        for p in population:
            p = p * target_sum / np.sum(p)
        fitness = [objective(p) for p in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
        
        # Local refinement with SLSQP
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        best_h = result.x
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
