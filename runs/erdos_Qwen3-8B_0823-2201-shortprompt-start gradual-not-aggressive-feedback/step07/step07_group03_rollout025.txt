import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.15
    crossover_rate = 0.8
    generations = 50
    
    def genetic_algorithm():
        population = np.random.uniform(0, 1, (population_size, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            best_idx = np.argmin(fitness)
            best_h = population[best_idx].copy()
            
            # Selection
            parents = np.random.choice(population_size, size=population_size, p=fitness / fitness.sum())
            parents = population[parents]
            
            # Crossover
            offspring = []
            for i in range(0, population_size, 2):
                if np.random.rand() < crossover_rate:
                    p1, p2 = parents[i], parents[i+1]
                    c = np.random.rand(n_points)
                    child1 = np.where(c < 0.5, p1, p2)
                    child2 = np.where(c >= 0.5, p1, p2)
                    offspring.extend([child1, child2])
                else:
                    offspring.append(parents[i])
                    offspring.append(parents[i+1])
            
            # Mutation
            for i in range(population_size):
                if np.random.rand() < mutation_rate:
                    perturb = np.random.normal(0, 0.1, n_points)
                    offspring[i] = np.clip(offspring[i] + perturb, 0, 1)
                    offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
            
            # Replacement
            population = np.array(offspring)
            population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        return best_h, objective(best_h)
    
    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm()
    
    # Local refinement with SQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    start_time = time.time()

        # Random perturbation with adaptive scale
        remaining_time = budget_s - (time.time() - start_time)

        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}

        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
