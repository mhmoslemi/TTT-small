import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Initial optimization with more iterations
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Genetic algorithm for global search
    population_size = 20
    generations = 10
    mutation_rate = 0.1

    def generate_individual():
        return np.random.rand(n_points) * target_sum

    def crossover(parent1, parent2):
        child = np.copy(parent1)
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        child[indices] = parent2[indices]
        return child

    def mutate(individual):
        mutation = np.random.normal(0, 0.05, n_points)
        mutated = individual + mutation
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * target_sum / np.sum(mutated)
        return mutated

    for _ in range(generations):
        if time.time() - start_time >= budget_s - 1:
            break
        population = [generate_individual() for _ in range(population_size)]
        for i in range(population_size):
            population[i] = mutate(population[i])
        
        for i in range(population_size):
            for j in range(i+1, population_size):
                if np.random.rand() < 0.8:
                    population[i] = crossover(population[i], population[j])
                    population[j] = crossover(population[j], population[i])
        
        for individual in population:

            individual = individual * target_sum / np.sum(individual)
            current_c5 = objective(individual)
            if current_c5 < best_c5:
                best_h = individual
                best_c5 = current_c5

    return best_h, best_c5, n_points
