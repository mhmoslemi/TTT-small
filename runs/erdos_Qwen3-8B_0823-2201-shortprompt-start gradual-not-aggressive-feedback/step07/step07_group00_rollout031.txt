import time

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    # Initialize with the initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    elite_size = 5
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    # Genetic algorithm
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
            
        # Create population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        elite_fitness = [fitness[i] for i in elite_indices]
        
        # Tournament selection
        new_population = elites.copy()

            tournament = np.random.choice(population_size, size=5, replace=False)
            best = tournament[np.argmin(fitness[tournament])]
            new_population.append(population[best])
        
        # Apply mutation
        for i in range(len(new_population)):
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, 0.05, n_points)
                new_population[i] = np.clip(new_population[i] + mutation, 0, 1)

        population = new_population

            method='SLSQP',

            options={'maxiter': 50, 'ftol': 1e-8}

        return result.x
    
    # Final refinement
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
