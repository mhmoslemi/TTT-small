import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with SLSQP
    result = opt.minimize(
        objective,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Adaptive simulated annealing with structured mutation
    start_time = time.time()
    temp = 1.0
    cooling_rate = 0.995
    mutation_scale = 0.2
    mutation_rate = 0.3
    generations = 50

    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Generate mutations with structured patterns
        mutations = np.random.choice([0, 1, -1], size=n_points, p=[0.4, 0.3, 0.3])
        mutated = best_h + mutations * mutation_scale
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * target_sum / np.sum(mutated)
        
        # Local optimization
        result = opt.minimize(
            objective,
            mutated,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 20, 'ftol': 1e-8}
        )
        
        current_h = result.x
