import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize population
    population_size = 30
    population = np.random.rand(population_size, n_points)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    def evaluate_individual(h):
        return objective(h)
    
    def select_elites(population, fitnesses):
        sorted_indices = np.argsort(fitnesses)
        return population[sorted_indices[:int(population_size*0.2)]]
    
    def crossover(parent1, parent2):
        alpha = np.random.rand(n_points)
        return alpha * parent1 + (1 - alpha) * parent2
    
    def mutate(individual):
        mutation_rate = 0.15
        mutation_scale = 0.1
        perturb = np.random.normal(0, mutation_scale, n_points)
        perturbed = individual + perturb
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        return perturbed
    
    start_time = time.time()
    best_h = np.array(initial_h, dtype=np.float64)
    best_c5 = objective(best_h)
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitnesses = np.array([evaluate_individual(ind) for ind in population])
        
        # Select elites
        elites = select_elites(population, fitnesses)
        
        # Generate new population
        new_population = []
        for _ in range(population_size):
            parent1, parent2 = np.random.choice(elites, size=2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = np.array(new_population)
