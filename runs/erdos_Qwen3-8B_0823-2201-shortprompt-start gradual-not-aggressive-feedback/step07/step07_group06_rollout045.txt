import numpy as np
import scipy.optimize as opt
import time
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Genetic algorithm setup
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)

    def eval_func(individual):
        h = np.array(individual)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5,

    def mut_gauss(individual, indpb):
        for i in range(len(individual)):
            if np.random.random() < indpb:
                individual[i] += np.random.normal(0, 0.1)
        return individual,

    def cx_onepoint(ind1, ind2):
        size = len(ind1)
        cxpoint = np.random.randint(1, size)
        ind1[cxpoint:], ind2[cxpoint:] = ind2[cxpoint:].copy(), ind1[cxpoint:].copy()
        return ind1, ind2

   toolbox = base.Toolbox()
    toolbox.register("evaluate", eval_func)
    toolbox.register("mate", cx_onepoint, indpb=0.1)
    toolbox.register("mutate", mut_gauss, indpb=0.05)
    toolbox.register("select", tools.selTournament, tournsize=3)
    toolbox.register("population", tools.initRepeat, creator.Individual, np.random.rand, n=n_points)

    # Initial optimization with SLSQP
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Run SLSQP for initial refinement
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum},
        options={'maxiter': 100, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Genetic algorithm with adaptive parameters
    pop = toolbox.population(n=50)
    stats = tools.Statistics(lambda ind: ind.fitness.values)
    stats.register("avg", np.mean)
    stats.register("std", np.std)
    stats.register("min", np.min)
    stats.register("max", np.max)

    # Run genetic algorithm
    halloffame = tools.HallOfFame(1)
    for gen in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        offspring = algorithms.varAnd(pop, toolbox, cxpb=0.5, mutpb=0.1)
        fits = toolbox.map(toolbox.evaluate, offspring)
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = fit
        pop = toolbox.select(pop + offspring, len(pop))
        record = stats.compile(pop)
        halloffame.update(pop)
        if halloffame[0].fitness.values[0] < best_c5:
            best_h = np.array(halloffame[0])
            best_c5 = halloffame[0].fitness.values[0]

    # Local search refinement
    for _ in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum},
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    return best_h, best_c5, n_points
