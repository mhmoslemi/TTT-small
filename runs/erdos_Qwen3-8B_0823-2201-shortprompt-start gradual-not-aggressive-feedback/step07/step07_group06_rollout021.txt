import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Initial optimization with SLSQP
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Genetic algorithm for global search
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    elites = 2

    def genetic_search(population):
        new_population = []
        # Keep elites
        for i in range(elites):
            new_population.append(population[i])
        # Create new individuals
        for _ in range(population_size - elites):
            parent1 = np.random.choice(population)
            parent2 = np.random.choice(population)
            child = np.copy(parent1)
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] = np.random.uniform(0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        return new_population

    population = [best_h.copy() for _ in range(population_size)]
    for _ in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        population = genetic_search(population)
        for ind in population:
            current_c5 = objective(ind)
            if current_c5 < best_c5:
                best_h = ind
                best_c5 = current_c5

    return best_h, best_c5, n_points
