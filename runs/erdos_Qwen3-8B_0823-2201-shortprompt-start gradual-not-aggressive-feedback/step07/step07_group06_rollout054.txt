import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Genetic algorithm parameters
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    elite_size = 1

    # Initial population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]

    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        
        # Create offspring
        offspring = []
        for _ in range(population_size - elite_size):
            parent1 = random.choice(elites)
            parent2 = random.choice(elites)
            child = np.zeros(n_points)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] = np.clip(child[i] + np.random.normal(0, 0.05), 0, 1)
            # Normalize
            child = child * target_sum / np.sum(child)
            offspring.append(child)
        
        # Combine population
        population = elites + offspring
        
        # Update best solution
        for h in population:
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    # Local refinement with SLSQP
    while time.time() - start_time < budget_s - 1:
        # Generate perturbation with adaptive magnitude
        perturb_scale = 0.05 * (1 - (time.time() - start_time) / (budget_s - 1))
        perturb = np.random.normal(0, perturb_scale, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Optimization with fewer iterations
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
