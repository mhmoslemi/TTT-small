import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Genetic algorithm for global search
    def genetic_algorithm(population_size=30, generations=50, mutation_rate=0.1):
        population = np.random.rand(population_size, n_points)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        population = np.clip(population, 0, 1)
        
        for _ in range(generations):
            fitness = np.array([objective(ind) for ind in population])
            elite_index = np.argmin(fitness)
            elite = population[elite_index]
            
            # Roulette wheel selection
            probabilities = fitness / np.sum(fitness)
            parents = np.random.choice(population.shape[0], size=population.shape[0], p=probabilities)
            offspring = []
            
            for i in range(population.shape[0]):
                parent1 = population[parents[i]]
                parent2 = population[np.random.choice(parents)]
                child = (parent1 + parent2) / 2
                if np.random.random() < mutation_rate:
                    mutation = np.random.normal(0, 0.05, n_points)
                    child += mutation
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
                offspring.append(child)
            
            population = np.array(offspring)
            population[elite_index] = elite
        
        best = np.min([objective(ind) for ind in population])
        best_h = population[np.argmin([objective(ind) for ind in population])]
        return best_h, best
    
    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm()
    
    # Local optimization with SLSQP
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Final local search
    while time.time() - start_time < budget_s - 1:
        perturb_scale = 0.05 * (1 - (time.time() - start_time) / budget_s)
        perturb = np.random.normal(0, perturb_scale, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Optimization with fewer iterations
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
