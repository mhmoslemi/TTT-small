        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def tournament_selection(population, fitnesses, tournament_size=5):
        indices = np.random.choice(len(population), tournament_size)
        best_idx = np.argmin(fitnesses[indices])
        return population[indices[best_idx]]
    
    def crossover(parent1, parent2):
        alpha = np.random.rand(n_points)
        return alpha * parent1 + (1 - alpha) * parent2
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.uniform(-0.2, 0.2)
                h[i] = np.clip(h[i], 0, 1)
        return h * target_sum / np.sum(h)
    
    def evaluate_population(population):
        fitnesses = []
        for h in population:
            fitness = objective(h)
            fitnesses.append(fitness)
        return np.array(fitnesses)
    
    # Initial population
    population_size = 50
    population = [np.array(initial_h, dtype=np.float64) * target_sum / np.sum(initial_h) for _ in range(population_size)]

    while time.time() - start_time < budget_s:
        # Evaluate population
        fitnesses = evaluate_population(population)
        
        # Select parents
        parents = [tournament_selection(population, fitnesses) for _ in range(population_size // 2)]
        
        # Crossover
        children = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                child1 = crossover(parents[i], parents[i+1])
                child2 = crossover(parents[i], parents[i+1])
                children.extend([child1, child2])
