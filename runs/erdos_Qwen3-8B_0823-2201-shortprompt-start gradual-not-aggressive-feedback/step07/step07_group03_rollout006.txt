import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.15
    crossover_rate = 0.8
    generations = 50
    
    def genetic_algorithm():
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        best_h = population[0]
        best_c5 = objective(best_h)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = [objective(h) for h in population]
            best_idx = np.argmin(fitness)
            best_c5 = fitness[best_idx]
            
            # Select parents
            parents = np.random.choice(population, size=population_size, p=np.array(fitness)/np.sum(fitness))
            
            # Crossover
            offspring = []
            for i in range(population_size // 2):
                p1 = parents[np.random.randint(population_size)]
                p2 = parents[np.random.randint(population_size)]
                child = np.zeros(n_points)
                for j in range(n_points):
                    if np.random.rand() < crossover_rate:
                        child[j] = p1[j]
                    else:
                        child[j] = p2[j]
                offspring.append(child)
            
            # Mutation

                if np.random.rand() < mutation_rate:
                    perturb = np.random.normal(0, 0.1, n_points)
                    offspring[i] = np.clip(offspring[i] + perturb, 0, 1)
            
            # Replace population
            population = [h * target_sum / np.sum(h) for h in offspring]
            
            # Update best solution

                c5 = objective(h)

                    best_h = h
                    best_c5 = c5
        
        return best_h, best_c5

        remaining_time = budget_s - (time.time() - start_time)

            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
