import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Initial optimization with more iterations
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Population-based search with adaptive mutation
    population_size = 20
    population = [best_h.copy() for _ in range(population_size)]
    
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        
        # Generate new population through mutation and crossover
        new_population = []
        for i in range(population_size):
            # Mutation
            mut = np.random.normal(0, 0.05, n_points)
            mutant = population[i] + mut
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * target_sum / np.sum(mutant)
            
            # Crossover with best solution
            crossover = np.where(np.random.rand(n_points) < 0.5, mutant, population[best_idx])
            crossover = crossover * target_sum / np.sum(crossover)
            
            new_population.append(crossover)
        
        population = new_population
    
    # Final evaluation
    best_h = np.min(population, axis=0)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
