import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    pop_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 100
    elite_size = 2
    
    def genetic_algorithm(population):
        fitness = np.array([objective(ind) for ind in population])
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = population[elite_indices]
        
        for _ in range(generations):
            # Select parents
            parents = np.array([population[np.random.choice(n_points)] for _ in range(pop_size)])
            
            # Apply crossover
            offspring = []
            for i in range(pop_size):
                if np.random.rand() < crossover_rate:
                    j = np.random.randint(0, pop_size)
                    parent1 = parents[i]
                    parent2 = parents[j]
                    crossover_point = np.random.randint(1, n_points - 1)
                    child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
                else:
                    child = parents[i]
                offspring.append(child)
            
            # Apply mutation
            for i in range(len(offspring)):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        offspring[i][j] += np.random.uniform(-0.1, 0.1)
                        offspring[i][j] = np.clip(offspring[i][j], 0, 1)
            
            # Replace population
            population = np.vstack((elite, offspring))
            population = np.unique(population, axis=0)
            population = population[np.random.choice(len(population), size=pop_size, replace=False)]
        
        best_individual = population[np.argmin(fitness)]
        return best_individual
    
    # Simulated annealing parameters
    T = 1.0
    cool_rate = 0.995
    max_iter = 1000
    
    def simulated_annealing(best_h):
        current_h = best_h.copy()
        current_c5 = objective(current_h)
        best_h = current_h
        best_c5 = current_c5
        
        for _ in range(max_iter):
            T *= cool_rate
            if T < 1e-5:
                break
            # Perturb solution
            perturb = np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(current_h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            current_c5 = objective(perturbed)
            
            # Acceptance probability
            if current_c5 < best_c5 or np.random.rand() < np.exp((current_c5 - best_c5)/T):
                best_h = perturbed
                best_c5 = current_c5
        
        return best_h, best_c5
    
    start_time = time.time()
    
    # Run genetic algorithm
    population = np.array([h + np.random.uniform(-0.1, 0.1, n_points) for _ in range(pop_size)])

    population = np.unique(population, axis=0)
    population = population[np.random.choice(len(population), size=pop_size, replace=False)]
    best_h = genetic_algorithm(population)
    best_c5 = objective(best_h)
    
    # Run simulated annealing
    best_h, best_c5 = simulated_annealing(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        # Adaptive perturbation based on time remaining
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.05 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
