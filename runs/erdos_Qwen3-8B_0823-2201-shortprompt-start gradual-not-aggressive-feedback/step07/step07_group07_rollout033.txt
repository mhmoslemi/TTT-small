import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Initial optimization with more iterations
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Genetic algorithm for global exploration
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    elite_size = 2

    for _ in range(generations):
        if time.time() - start_time >= budget_s - 1:
            break
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        population.append(best_h)
        
        fitness = [objective(h) for h in population]
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = [population[i] for i in elite_indices]
        
        for i in range(population_size - elite_size):
            parent1 = random.choice(elite)
            parent2 = random.choice(elite)
            child = np.zeros(n_points)
            for j in range(n_points):
                if random.random() < 0.5:
                    child[j] = parent1[j]
                else:
                    child[j] = parent2[j]
            if random.random() < mutation_rate:
                for j in range(n_points):
                    child[j] += np.random.normal(0, 0.05, 1)
                    child[j] = np.clip(child[j], 0, 1)
            child = child * target_sum / np.sum(child)
            population[elite_size + i] = child
        
        population = [h for h in population if h is not None]
        fitness = [objective(h) for h in population]
        best_h = population[np.argmin(fitness)]
        best_c5 = objective(best_h)

    return best_h, best_c5, n_points
