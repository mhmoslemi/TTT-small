import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        idx = np.random.randint(1, n_points - 1)
        child[:idx] = parent2[:idx]
        return child
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.uniform(-0.1, 0.1)
                h[i] = np.clip(h[i], 0, 1)
        return h * target_sum / np.sum(h)
    
    start_time = time.time()
    population_size = 20
    population = [mutate(h) for _ in range(population_size)]
    best_h = h.copy()
    best_c5 = objective(h)
    
    for _ in range(100):  # Generations
        if time.time() - start_time > budget_s:
            break
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Selection and crossover
        selected = np.argsort(fitness)[:population_size//2]
        new_population = []
        for i in range(population_size//2):
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child = crossover(population[parent1], population[parent2])
            new_population.append(mutate(child))
        
        # Replace population
        population = new_population
        population.append(best_h)
    
    # Final refinement
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
