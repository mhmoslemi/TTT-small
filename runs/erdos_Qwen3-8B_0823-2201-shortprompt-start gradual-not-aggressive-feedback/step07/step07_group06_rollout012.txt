import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Initial optimization with SLSQP
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Genetic algorithm for global search
    def genetic_algorithm(population_size=30, generations=100, mutation_rate=0.1):
        population = np.random.uniform(0, 1, (population_size, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        best_pop = population.copy()
        best_c5 = np.inf

        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            fitness /= np.max(fitness)
            
            # Select parents
            parents = population[np.argsort(fitness)[::-1][:population_size//2]]
            
            # Crossover
            offspring = []
            for _ in range(population_size//2):
                parent1, parent2 = random.sample(parents, 2)
                child = np.zeros(n_points)
                for i in range(n_points):
                    if random.random() < 0.5:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
                offspring.append(child)
            
            # Mutation
            for i in range(len(offspring)):
                for j in range(n_points):
                    if random.random() < mutation_rate:
                        offspring[i][j] += np.random.uniform(-0.1, 0.1)
                        offspring[i][j] = np.clip(offspring[i][j], 0, 1)
                        offspring[i][j] = offspring[i][j] * target_sum / np.sum(offspring[i])
            
            # Replace population
            population = np.vstack((parents, offspring))
            population = population[np.random.permutation(population.shape[0])]
            
            # Update best
            current_best = population[np.argmin(fitness)]
            current_c5 = objective(current_best)
            if current_c5 < best_c5:
                best_pop = current_best
                best_c5 = current_c5
        
        return best_pop, best_c5

    if time.time() - start_time < budget_s - 5:
        ga_h, ga_c5 = genetic_algorithm(population_size=30, generations=50, mutation_rate=0.1)
        if ga_c5 < best_c5:
            best_h = ga_h
            best_c5 = ga_c5

    # Local search with adaptive perturbations
    while time.time() - start_time < budget_s - 1:
        # Generate perturbation with adaptive magnitude
        perturb_scale = 0.05 * (1 - (time.time() - start_time) / (budget_s - 1))
        perturb = np.random.normal(0, perturb_scale, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)

            perturbed_h,

            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},

        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
