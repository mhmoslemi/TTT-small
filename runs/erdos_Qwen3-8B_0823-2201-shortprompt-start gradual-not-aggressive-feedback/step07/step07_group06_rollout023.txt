import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Genetic algorithm parameters
    population_size = 20
    generations = 10
    mutation_rate = 0.15
    elite_size = 2

    # Initial population
    population = [np.random.rand(n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    population.append(best_h)
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
        
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elites.copy()
        
        while len(new_population) < population_size:
            # Select parents
            parent1 = np.random.choice(elite_indices)
            parent2 = np.random.choice(elite_indices)
            parent1 = population[parent1]
            parent2 = population[parent2]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            
            # Mutation
            mutation = np.random.normal(0, 0.05, n_points)
            child += mutation * mutation_rate
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
    
    # Final optimization with SLSQP
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
