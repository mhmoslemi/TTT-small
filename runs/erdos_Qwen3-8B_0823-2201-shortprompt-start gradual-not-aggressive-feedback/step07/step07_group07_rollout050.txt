import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Greedy initial construction
    h = np.zeros(n_points)
    for i in range(n_points):
        if np.sum(h) < target_sum:
            h[i] = 1.0
        else:
            h[i] = 0.0
    
    # Refine with local search
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Initial optimization
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum},
        options={'maxiter': 50, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Local search with adaptive perturbations
    while time.time() - start_time < budget_s - 1:
        perturb_scale = 0.05 * (1 - (time.time() - start_time) / budget_s)
        perturb = np.random.normal(0, perturb_scale, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Optimization with fewer iterations
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum},
            options={'maxiter': 30, 'ftol': 1e-9}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
