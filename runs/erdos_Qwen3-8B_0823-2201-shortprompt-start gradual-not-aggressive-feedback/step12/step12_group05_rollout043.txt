import scipy.optimize as opt
import time

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    def guided_perturbation(h, perturb_size):
        grad = np.gradient(objective(h))
        perturb = np.random.normal(0, perturb_size, n_points)
        perturb = perturb * np.sign(grad)
        perturbed_h = h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        return perturbed_h
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    # Initial refinement
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Guided perturbation and refinement
    remaining_time = budget_s - (time.time() - start_time)
    perturb_size = max(0.005, 0.05 * (remaining_time / budget_s))
    
    while time.time() - start_time < budget_s:
        perturbed_h = guided_perturbation(best_h, perturb_size)
        result = opt.minimize(
            objective,
            perturbed_h,
