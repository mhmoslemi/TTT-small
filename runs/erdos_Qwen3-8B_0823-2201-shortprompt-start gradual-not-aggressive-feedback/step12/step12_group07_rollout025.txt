import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        c5 = np.max(corr) * dx
        return c5

    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-10}
        )
        return result.x
    
    # Population-based refinement
    def population_refinement(h, pop_size=20, generations=20):

        best_h = h.copy()
