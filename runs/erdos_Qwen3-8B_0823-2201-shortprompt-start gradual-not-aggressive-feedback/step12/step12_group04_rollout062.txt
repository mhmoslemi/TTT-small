import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h = local_refinement(h)
    best_c5 = objective(best_h)
    
    # Guided perturbation strategy with dynamic focus
    perturbation_scale = 0.1
    for _ in range(int(budget_s / 0.1)):
        if time.time() - start_time >= budget_s:
            break
        # Identify regions of high overlap for targeted perturbation
        h_current = best_h.reshape(n_points)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * target_sum / np.sum(h_current)
        corr = np.correlate(h_current, 1 - h_current, mode="full")
        max_overlap_idx = np.argmax(corr) - n_points + 1
        perturb = np.zeros(n_points)
        perturb[max_overlap_idx - 2:max_overlap_idx + 3] = np.random.normal(0, perturbation_scale, 5)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 20, 'ftol': 1e-8}
        )

        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        perturbation_scale *= 0.95  # Decrease perturbation scale over time
    
    return best_h, best_c5, n_points
