import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    def simulated_annealing(h, T=1.0, cooling_rate=0.995, iters=100):
        current_h = h.copy()
        current_c5 = objective(current_h)
        best_h = current_h.copy()
        best_c5 = current_c5
        
        for _ in range(iters):
            perturbation = np.random.normal(0, T, n_points)
            perturbed_h = current_h + perturbation
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            perturbed_c5 = objective(perturbed_h)
            
            if perturbed_c5 < current_c5:
                current_h = perturbed_h
                current_c5 = perturbed_c5
                if perturbed_c5 < best_c5:
                    best_h = perturbed_h
                    best_c5 = perturbed_c5
            
            T *= cooling_rate
        
        return best_h, best_c5
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    # Initial refinement
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Simulated annealing with decreasing temperature
    remaining_time = budget_s - (time.time() - start_time)
    max_iterations = int(remaining_time / 0.01)
    
    for T in np.linspace(1.0, 0.1, max_iterations):
        # Run simulated annealing with current temperature
        perturbed_h, perturbed_c5 = simulated_annealing(best_h, T=T)
        if perturbed_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = perturbed_c5
    
    return best_h, best_c5, n_points
