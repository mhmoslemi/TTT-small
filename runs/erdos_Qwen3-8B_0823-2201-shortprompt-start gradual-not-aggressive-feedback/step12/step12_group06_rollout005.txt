import scipy.optimize as opt
import time

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    start_time = time.time()
    
    # Try each starting point and refine
    for start in start_points:

        result = opt.minimize(
