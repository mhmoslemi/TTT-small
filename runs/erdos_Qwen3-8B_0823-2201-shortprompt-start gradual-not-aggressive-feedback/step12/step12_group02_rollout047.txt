import scipy.optimize as opt
import time
import random

        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    # Initialize with the given construction
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = objective(h)
    
    # Time tracking
    start_time = time.time()
    
    # Hybrid optimization: genetic algorithm with local refinement
    for stage in range(2):
        # Genetic algorithm parameters
        population_size = 10
        generations = 100
        mutation_rate = 0.1
        elite_size = 2
        
        # Initialize population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [p * target_sum / np.sum(p) for p in population]
        
        for gen in range(generations):
            # Evaluate fitness
            fitness = [objective(p) for p in population]
            
            # Select elites
            elite_indices = np.argsort(fitness)[:elite_size]
            elites = [population[i] for i in elite_indices]
            
            # Create new population
            new_population = elites.copy()
            
            # Generate offspring through crossover and mutation
            for _ in range(population_size - elite_size):
                # Select parents
                parent1 = random.choice(elites)
                parent2 = random.choice(elites)

                        child[i] = np.clip(child[i], 0, 1)
                        child = child * target_sum / np.sum(child)
                
                new_population.append(child)
            
            population = new_population
            
            # Update best solution
            current_best = min(population, key=objective)
            current_c5 = objective(current_best)
