import scipy.optimize as opt
import time

        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

        population = np.random.rand(population_size, n_points)
        population = population * target_sum / np.sum(population, axis=1)[:, np.newaxis]

            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            best_idx = np.argmin(fitness)
            best_fitness = fitness[best_idx]
            
            # Selection
            selected = np.random.choice(population_size, size=population_size, p=fitness / np.sum(fitness))
            selected = population[selected]
            
            # Crossover
            for i in range(population_size):
                if np.random.rand() < crossover_rate:
                    j = np.random.randint(population_size)

                    child = np.concatenate((selected[i][:crossover_point], selected[j][crossover_point:]))
                else:
                    child = selected[i]
