import time

        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    # Phase 1: Structured perturbation and gradient-based optimization

        result = opt.minimize(

            perturbed,
            method='L-BFGS-B',

        )
        
        h = result.x
