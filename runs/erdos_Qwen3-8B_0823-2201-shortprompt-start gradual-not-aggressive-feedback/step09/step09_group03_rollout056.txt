import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.15
    crossover_rate = 0.8
    
    def evaluate_individual(h):
        return objective(h)
    
    def mutate(h):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.uniform(-0.1, 0.1)
                h[i] = np.clip(h[i], 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    def crossover(parent1, parent2):
        child = np.random.rand(len(parent1))
        for i in range(len(child)):
            if np.random.rand() < crossover_rate:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        child = child * target_sum / np.sum(child)
        return child
    
    # Initial population
    population = [np.array(initial_h, dtype=np.float64) * target_sum / np.sum(initial_h) for _ in range(population_size)]
    population = [mutate(ind) for ind in population]
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate_individual(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            # Select parents
            parent1, parent2 = np.random.choice(population, 2, replace=False)
            child1 = crossover(parent1, parent2)
            child2 = crossover(parent2, parent1)
            new_population.append(mutate(child1))
            new_population.append(mutate(child2))
        
        # Replace population
        population = new_population
    
    # Local refinement with SLSQP on the best solution
    best_h = np.array(population[np.argmin([evaluate_individual(ind) for ind in population])])
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
