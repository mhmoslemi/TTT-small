import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Reasonable size for fast computation
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Generate diverse initial configurations
    def generate_initial_config():
        config = np.zeros(n_points)
        if np.random.rand() < 0.3:
            # Uniform distribution
            config += 0.5
        elif np.random.rand() < 0.3:
            # Alternating pattern
            config[::2] = 1.0
            config[1::2] = 0.0
        else:
            # Clustered pattern
            cluster_size = np.random.randint(1, 10)
            start = np.random.randint(0, n_points - cluster_size)
            config[start:start+cluster_size] = 1.0
        return config
    
    # Simulated annealing with adaptive cooling
    def simulated_annealing(initial_h):
        T = 1.0
        cool_rate = 0.995
        max_iter = 2000
        best_h = initial_h.copy()
        best_c5 = objective(best_h)
        
        for _ in range(max_iter):
            T *= cool_rate
            if T < 1e-5:
                break
            # Adaptive perturbation with decreasing magnitude
            perturb_size = 0.1 * (T ** 0.5)
            perturb = np.random.normal(0, perturb_size, n_points)
            perturbed = np.clip(initial_h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            current_c5 = objective(perturbed)
            
            if current_c5 < best_c5 or np.random.rand() < np.exp((current_c5 - best_c5)/T):
                best_h = perturbed
                best_c5 = current_c5
        return best_h, best_c5
    
    # Local refinement with SQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    
    # Run multiple initial configurations
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        initial_h = generate_initial_config()
        initial_h = initial_h * target_sum / np.sum(initial_h)
        h, c5 = simulated_annealing(initial_h)
        h = local_refinement(h)
        c5 = objective(h)
        
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    # Final refinement within budget
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.1 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
