        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    # Initialize population
    population_size = 30
    population = np.random.uniform(0, 1, (population_size, n_points))

    def evaluate_individual(h):
        return objective(h)
    
    def tournament_selection(pop, fitnesses, k=3):
        idxs = np.random.choice(len(pop), size=k, replace=False)
        best_idx = np.argmax(fitnesses[idxs])
        return pop[idxs[best_idx]]
    
    def crossover(parent1, parent2):
        alpha = np.random.rand(n_points)
        return alpha * parent1 + (1 - alpha) * parent2

        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.normal(0, mutation_strength)

        return h
