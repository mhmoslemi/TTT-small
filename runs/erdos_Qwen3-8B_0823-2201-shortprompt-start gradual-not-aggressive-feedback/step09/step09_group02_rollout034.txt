import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 50
    
    # Initialize population
    population = [np.random.rand(n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    # Evaluate initial population
    fitness = np.array([objective(ind) for ind in population])
    best_index = np.argmin(fitness)
    best_h = population[best_index]
    best_c5 = fitness[best_index]
    
    start_time = time.time()
    
    # Evolutionary loop
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
            
        # Select parents
        parents = np.random.choice(population_size, size=population_size, p=fitness/np.sum(fitness))
        parents = [population[i] for i in parents]
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if np.random.rand() < crossover_rate:
                p1 = parents[i]
                p2 = parents[i+1]
                child1 = np.concatenate((p1[:n_points//2], p2[n_points//2:]))
                child2 = np.concatenate((p2[:n_points//2], p1[n_points//2:]))
                offspring.extend([child1, child2])
            else:
                offspring.extend([parents[i], parents[i+1]])
        
        # Mutation
        for i in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                offspring[i] = np.clip(offspring[i] + np.random.normal(0, 0.05, n_points), 0, 1)
                offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
        
        # Evaluate offspring
        offspring_fitness = np.array([objective(ind) for ind in offspring])
        population = np.concatenate((parents, offspring))
        fitness = np.concatenate((fitness, offspring_fitness))
        
        # Select next generation
        population = np.array([population[i] for i in np.argsort(fitness)])
        fitness = np.sort(fitness)
        population = population[:population_size]
        fitness = fitness[:population_size]
        
        # Update best solution
        best_index = np.argmin(fitness)
        best_h = population[best_index]
        best_c5 = fitness[best_index]
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Refine the best solution
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.05 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
