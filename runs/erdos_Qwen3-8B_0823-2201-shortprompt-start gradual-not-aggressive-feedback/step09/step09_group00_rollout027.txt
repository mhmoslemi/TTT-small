        c5 = np.max(corr) * dx
        return c5

    crossover_rate = 0.8
    generations = 100
    
    def evolve_population(population):
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        # Select parents
        parents = np.random.choice(len(population), size=population_size, p=fitness/np.sum(fitness))
        parents = population[parents]
        # Crossover
        children = []
        for _ in range(population_size):
            if np.random.rand() < crossover_rate:
