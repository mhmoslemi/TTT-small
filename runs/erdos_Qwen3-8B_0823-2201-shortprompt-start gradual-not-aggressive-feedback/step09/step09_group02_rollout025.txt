import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 100
    
    def mutate(individual):
        mutant = individual.copy()
        for i in range(len(mutant)):
            if np.random.rand() < mutation_rate:
                mutant[i] += np.random.normal(0, 0.05)
                mutant[i] = np.clip(mutant[i], 0, 1)
        mutant = mutant * target_sum / np.sum(mutant)
        return mutant
    
    def crossover(parent1, parent2):
        child = parent1.copy()
        for i in range(len(child)):
            if np.random.rand() < crossover_rate:
                child[i] = parent2[i]
        child = child * target_sum / np.sum(child)
        return child
    
    def fitness(individual):
        return objective(individual)
    
    # Initialize population
    population = [local_refinement(h.copy()) for _ in range(population_size)]
    best_individual = min(population, key=fitness)
    best_fitness = fitness(best_individual)
    
    start_time = time.time()
    
    for generation in range(generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate fitness
        fitnesses = [fitness(ind) for ind in population]
        population.sort(key=lambda x: fitness(x))
        
        # Select top half
        selected = population[:population_size//2]
        
        # Generate new population
        new_population = []
        while len(new_population) < population_size:
            parent1 = np.random.choice(selected)
            parent2 = np.random.choice(selected)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
    
    # Final refinement
    best_individual = min(population, key=fitness)
    best_individual = local_refinement(best_individual)
    best_c5 = objective(best_individual)
    
    return best_individual, best_c5, n_points
