import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize population
    population_size = 20
    population = np.random.rand(population_size, n_points)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    def evaluate_individual(h):
        return objective(h)
    
    def select_parents(population, fitnesses):
        idx = np.argmax(fitnesses)
        return population[idx].copy()
    
    def crossover(parent1, parent2):
        alpha = np.random.rand(n_points)
        return alpha * parent1 + (1 - alpha) * parent2
    
    def mutate(individual):
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(individual + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        return perturbed
    
    start_time = time.time()
    best_h = np.array(initial_h, dtype=np.float64)
    best_c5 = objective(best_h)
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate population
        fitnesses = np.array([evaluate_individual(ind) for ind in population])
        best_idx = np.argmin(fitnesses)
        current_best = population[best_idx].copy()
        current_best_c5 = fitnesses[best_idx]
        
        if current_best_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_best_c5
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            parent1 = select_parents(population, fitnesses)
            parent2 = select_parents(population, fitnesses)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = np.array(new_population)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
