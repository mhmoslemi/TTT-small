import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize population
    population_size = 20
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    def evaluate_population(pop):
        return np.array([objective(ind) for ind in pop])
    
    # Genetic algorithm parameters
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_ratio = 0.2
    
    start_time = time.time()
    
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate population
        fitness = evaluate_population(population)
        
        # Select elites
        elite_count = int(elite_ratio * population_size)
        elite_indices = np.argsort(fitness)[:elite_count]
        elite_population = population[elite_indices]
        
        # Select parents
        parent_indices = np.argsort(fitness)[::-1][:int(crossover_rate * population_size)]
        parent_population = population[parent_indices]
        
        # Crossover
        offspring = []
        for i in range(int(crossover_rate * population_size)):
            j = np.random.randint(0, len(parent_population))
            parent1 = parent_population[i]
            parent2 = parent_population[j]
            alpha = np.random.rand(n_points)
            child = alpha * parent1 + (1 - alpha) * parent2
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, 0.05, n_points)
                offspring[i] += mutation
                offspring[i] = np.clip(offspring[i], 0, 1)
        
        # Combine elites and offspring
        new_population = np.vstack((elite_population, offspring))
        new_population = new_population[np.random.permutation(len(new_population))]
        
        # Truncate to population size
        population = new_population[:population_size]
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    # Local refinement with SLSQP on best candidate
    best_h = population[np.argmin(evaluate_population(population))]
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
