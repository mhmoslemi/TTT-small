import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Main optimization loop
    for iteration in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Generate structured perturbation
        perturbation = np.sin(np.linspace(0, 2*np.pi, n_points))

        perturbed = perturbed * target_sum / np.sum(perturbed)
        
        # Gradient-based optimization
        result = opt.minimize(
            objective,
            perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},

        )
