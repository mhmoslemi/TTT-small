import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.15
    crossover_rate = 0.8
    generations = 100
    elite_size = 2
    
    def genetic_algorithm():
        # Initialize population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        
        best_h = h.copy()
        best_c5 = objective(best_h)
        
        for gen in range(generations):
            # Evaluate fitness
            fitness = [objective(ind) for ind in population]
            
            # Select elite
            elite_indices = np.argsort(fitness)[:elite_size]
            elite = [population[i] for i in elite_indices]
            
            # Select parents
            parents = np.random.choice(population_size, size=population_size, p=[f/sum(fitness) for f in fitness])
            parents = [population[i] for i in parents]
            
            # Crossover
            offspring = []
            for i in range(0, population_size, 2):
                if np.random.rand() < crossover_rate:
                    parent1 = parents[i]
                    parent2 = parents[i+1]
                    child1 = np.concatenate([parent1[:n_points//2], parent2[n_points//2:]])
                    child2 = np.concatenate([parent2[:n_points//2], parent1[n_points//2:]])
                    offspring.append(child1)
                    offspring.append(child2)
                else:
                    offspring.append(parents[i])
                    offspring.append(parents[i+1])

            for i in range(len(offspring)):
                if np.random.rand() < mutation_rate:
                    mutation = np.random.normal(0, 0.05, n_points)
                    offspring[i] += mutation
                    offspring[i] = np.clip(offspring[i], 0, 1)
                    offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
            
            # Create new population
            population = elite + offspring
            population = [np.clip(h, 0, 1) for h in population]
            population = [h * target_sum / np.sum(h) for h in population]
            
            # Update best solution
            current_best = min(zip(population, fitness), key=lambda x: x[1])

                best_h = current_best[0].copy()
                best_c5 = current_best[1]
        
        return best_h, best_c5

    best_h, best_c5 = genetic_algorithm()

            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)

        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )

        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
