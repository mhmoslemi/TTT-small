import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def genetic_algorithm():
        population_size = 20
        generations = 100
        mutation_rate = 0.1
        crossover_rate = 0.8
        best_h = np.array(initial_h, dtype=np.float64)
        best_c5 = objective(best_h)
        
        population = [np.array(initial_h, dtype=np.float64) for _ in range(population_size)]
        for _ in range(generations):
            fitness = [objective(h) for h in population]
            best_idx = np.argmin(fitness)
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
            
            # Selection
            selected = np.random.choice(population_size, size=population_size, p=[f / np.sum(fitness) for f in fitness])
            selected = [population[i] for i in selected]
            
            # Crossover
            offspring = []
            for i in range(0, population_size, 2):
                if np.random.rand() < crossover_rate:
                    p1 = selected[i]
                    p2 = selected[i+1]
                    crossover_point = np.random.randint(1, n_points-1)
                    child1 = np.concatenate((p1[:crossover_point], p2[crossover_point:]))
                    child2 = np.concatenate((p2[:crossover_point], p1[crossover_point:]))
                    offspring.append(child1)
                    offspring.append(child2)
                else:
                    offspring.append(selected[i])
                    offspring.append(selected[i+1])
            
            # Mutation
            for i in range(len(offspring)):
                h = offspring[i]
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        h[j] += np.random.uniform(-0.1, 0.1)
                        h[j] = np.clip(h[j], 0, 1)
                offspring[i] = h
            
            # Replace population
            population = offspring
        
        return best_h, best_c5
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.05 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
