import time

        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)

        best_h = h.copy()
        best_c5 = objective(best_h)

            # Perturb initial solution
            perturb = np.random.normal(0, 0.1, n_points)
            perturbed = np.clip(h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            
            # Simulated annealing
            T = 1.0
            cool_rate = 0.995
            max_iter = 1000
            current_h = perturbed.copy()
            current_c5 = objective(current_h)
            
            for _ in range(max_iter):
                T *= cool_rate
                if T < 1e-5:
                    break

            bounds=bounds,

            options={'maxiter': 50, 'ftol': 1e-8}
