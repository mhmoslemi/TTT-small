import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    def evolve_population(population):
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        # Select parents
        parents = np.random.choice(population, size=population_size, p=fitness/np.sum(fitness))
        # Crossover
        offspring = []
        for i in range(population_size//2):
            parent1 = parents[i]
            parent2 = parents[population_size//2 + i]
            if np.random.rand() < crossover_rate:
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            else:
                child = parent1.copy()
            # Mutation
            mutation_mask = np.random.rand(n_points) < mutation_rate
            child[mutation_mask] = np.random.rand(np.sum(mutation_mask))
            # Clip and normalize
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            offspring.append(child)
        # Replace population
        population = np.vstack([population, offspring])
        population = np.unique(population, axis=0)
        return population[:population_size]
    
    # Genetic algorithm
    population = np.array([h.copy() for _ in range(population_size)])
    for _ in range(generations):
        population = evolve_population(population)
    
    # Find best solution from population
    best_h = population[0]
    best_c5 = objective(best_h)
    for h in population:
        c5 = objective(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Adaptive perturbation based on time remaining
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.1 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
