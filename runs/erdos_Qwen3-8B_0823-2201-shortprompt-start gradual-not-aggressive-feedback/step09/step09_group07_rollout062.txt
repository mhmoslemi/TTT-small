import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Hybrid optimization with adaptive perturbations
    def optimized_search():
        best_h = h.copy()
        best_c5 = objective(best_h)
        time_used = 0
        
        for _ in range(100):
            # Structured perturbation
            perturbation = 0.05 * np.sin(np.linspace(0, 2 * np.pi, n_points))
            perturbed = np.clip(best_h + perturbation, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            
            # Gradient-based optimization
            result = opt.minimize(
                objective,
                perturbed,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 30, 'ftol': 1e-8}
            )

            if current_c5 < best_c5:

                best_c5 = current_c5

            if time_used >= budget_s:
                break
        
        return best_h, best_c5
