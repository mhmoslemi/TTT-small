import time

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def tournament_selection(population, k=3):
        candidates = np.random.choice(population, size=k, replace=False)
        return min(candidates, key=lambda x: objective(x))
    
    def crossover(parent1, parent2):
        child = parent1.copy()
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        child[indices] = parent2[indices]
        return child
    
    def mutate(individual, mutation_rate=0.05):
        individual = individual.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] += np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        return individual
    
    # Genetic algorithm
    population_size = 20
    population = [h.copy() for _ in range(population_size)]

    best_h = h.copy()
    best_c5 = objective(best_h)

        if time.time() - start_time > budget_s:
            break

        population_fitness = [objective(ind) for ind in population]

        parent1 = tournament_selection(population)
        parent2 = tournament_selection(population)

        child = crossover(parent1, parent2)

        child = mutate(child)

            idx = np.argmax(population_fitness)
            population[idx] = child
