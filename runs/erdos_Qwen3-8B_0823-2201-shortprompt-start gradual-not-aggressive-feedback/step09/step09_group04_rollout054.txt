import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def genetic_algorithm(population_size=20, generations=50, elite_size=2):
        population = [np.random.rand(n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = [objective(ind) for ind in population]
            # Select elites
            elite_indices = np.argsort(fitness)[:elite_size]
            elites = [population[i] for i in elite_indices]
            # Select parents
            parent_indices = np.argsort(fitness)[-2*elite_size:]
            parents = [population[i] for i in parent_indices]
            # Create new population
            new_population = elites.copy()
            while len(new_population) < population_size:
                # Tournament selection
                a, b = np.random.choice(parent_indices, 2, replace=False)
                if fitness[a] < fitness[b]:
                    parent = parents[a]
                else:
                    parent = parents[b]
                # Crossover
                crossover_point = np.random.randint(1, n_points-1)
                child = np.concatenate([parent[:crossover_point], 
                                      np.random.rand(n_points - crossover_point)])
                # Mutation
                mutation_rate = 0.1 * (generations - _) / generations
                child = np.clip(child + np.random.normal(0, mutation_rate, n_points), 0, 1)
                child = child * target_sum / np.sum(child)
                new_population.append(child)
            population = new_population
        
        # Return best solution
        best_idx = np.argmin(fitness)
        return population[best_idx], fitness[best_idx]
    
    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm(population_size=20, generations=50, elite_size=2)

            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
