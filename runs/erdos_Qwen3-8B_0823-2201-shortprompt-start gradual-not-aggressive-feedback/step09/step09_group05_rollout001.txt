import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize with the given construction
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    pop_size = 20
    crossover_rate = 0.8
    mutation_rate = 0.1
    generations = 50
    elite_size = 2
    
    def genetic_algorithm():
        population = np.random.uniform(0, 1, (pop_size, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            
            # Select elites
            elite_indices = np.argsort(fitness)[:elite_size]
            elites = population[elite_indices]
            
            # Select parents
            parent_indices = np.random.choice(pop_size, size=pop_size, replace=True)
            parents = population[parent_indices]
            
            # Crossover
            offspring = np.zeros((pop_size - elite_size, n_points))
            for i in range(pop_size - elite_size):
                if np.random.rand() < crossover_rate:
                    # Single-point crossover
                    point = np.random.randint(1, n_points)
                    offspring[i] = np.concatenate((parents[i, :point], parents[np.random.randint(pop_size), point:]))
                else:
                    # Copy from parent
                    offspring[i] = parents[i]
            
            # Mutation
            for i in range(offspring.shape[0]):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        offspring[i, j] = np.random.uniform(0, 1)
                offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
            
            # Combine elites and offspring
            population = np.vstack((elites, offspring))
            population = population[np.random.permutation(population.shape[0])]
        
        best_h = population[np.argmin(fitness)]
        return best_h, np.min(fitness)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement with adaptive perturbation within budget
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.1 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
