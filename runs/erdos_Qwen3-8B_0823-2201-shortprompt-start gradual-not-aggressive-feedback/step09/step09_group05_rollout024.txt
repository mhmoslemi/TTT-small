import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    pop_size = 20
    generations = 100
    mutation_rate = 0.1
    elite_size = 2
    
    def genetic_algorithm():
        # Initialize population
        population = np.random.uniform(0, 1, (pop_size, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            
            # Select elites
            elite_indices = np.argsort(fitness)[:elite_size]
            elites = population[elite_indices]
            
            # Tournament selection
            selected = []
            for _ in range(pop_size - elite_size):
                competitors = np.random.choice(pop_size, size=5, replace=False)
                winner = competitors[np.argmin(fitness[competitors])]
                selected.append(population[winner])
            
            # Crossover
            offspring = []
            for i in range(0, len(selected), 2):
                if i + 1 < len(selected):
                    parent1 = selected[i]
                    parent2 = selected[i + 1]
                    child1 = np.copy(parent1)
                    child2 = np.copy(parent2)
                    for j in range(n_points):
                        if np.random.rand() < 0.5:
                            child1[j] = parent2[j]
                            child2[j] = parent1[j]
                    offspring.append(child1)
                    offspring.append(child2)
            
            # Mutation
            for i in range(len(offspring)):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        offspring[i][j] += np.random.uniform(-0.1, 0.1)
                        offspring[i][j] = np.clip(offspring[i][j], 0, 1)
                        offspring[i][j] = offspring[i][j] * target_sum / np.sum(offspring[i])
            
            # Combine elites and offspring
            population = np.vstack((elites, offspring))
            population = np.unique(population, axis=0)
            
            # Keep only top performers
            population = population[np.argsort(fitness)[:pop_size]]
        
        # Local refinement with SLSQP
        best_h = population[np.argmin(fitness)]
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x, objective(result.x)
    
    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm()
    
    # Additional refinement with adaptive perturbation within budget
    start_time = time.time()
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.1 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
