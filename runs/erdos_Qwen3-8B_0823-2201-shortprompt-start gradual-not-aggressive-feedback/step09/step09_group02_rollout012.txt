import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm with local refinement
    def genetic_algorithm():
        population_size = 20
        generations = 100
        mutation_rate = 0.1
        best_h = h.copy()
        best_c5 = objective(best_h)
        
        for _ in range(generations):
            # Create new population
            population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
            population.append(best_h)
            
            # Evaluate fitness
            fitness = [objective(p) for p in population]
            
            # Select best solution
            best_index = np.argmin(fitness)
            best_h = population[best_index].copy()
            best_c5 = fitness[best_index]
            
            # Mutate best solution
            mutation = np.random.normal(0, 0.05, n_points)
            mutated = np.clip(best_h + mutation, 0, 1)
            mutated = mutated * target_sum / np.sum(mutated)
            fitness_mutated = objective(mutated)
            
            if fitness_mutated < best_c5:
                best_h = mutated
                best_c5 = fitness_mutated
        
        return best_h, best_c5
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.05 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}

        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
