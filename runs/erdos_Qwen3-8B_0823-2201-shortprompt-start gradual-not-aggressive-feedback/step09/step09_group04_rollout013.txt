import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Simulated annealing with adaptive cooling
    def simulated_annealing():
        T = 1.0
        cool_rate = 0.996
        max_iter = 2000
        best_h = h.copy()
        best_c5 = objective(best_h)
        
        for _ in range(max_iter):
            T *= cool_rate
            if T < 1e-5:
                break
            # Adaptive perturbation
            perturb_size = 0.1 * (T ** 0.5)
            perturb = np.random.normal(0, perturb_size, n_points)
            perturbed = np.clip(h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            current_c5 = objective(perturbed)
            
            if current_c5 < best_c5 or np.random.rand() < np.exp((current_c5 - best_c5)/T):
                best_h = perturbed
                best_c5 = current_c5
        
        return best_h, best_c5
    
    # Local refinement with SLSQP
    def local_refinement(h):

            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()

            perturbed_h = best_h + perturb

                perturbed_h,

                best_c5 = current_c5
    
    return best_h, best_c5, n_points
