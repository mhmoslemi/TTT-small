import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    pop_size = 20
    generations = 100
    mutation_rate = 0.2
    elite_size = 2
    
    def genetic_algorithm():
        population = [np.random.rand(n_points) for _ in range(pop_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = [objective(h) for h in population]
            best_idx = np.argmin(fitness)
            best_h = population[best_idx]
            
            # Selection
            selected = np.random.choice(pop_size, size=pop_size, p=np.array(fitness)/np.sum(fitness))
            selected = [population[i] for i in selected]
            
            # Crossover
            offspring = []
            for i in range(0, pop_size, 2):
                if i+1 < pop_size:
                    parent1 = selected[i]
                    parent2 = selected[i+1]
                    child1 = np.copy(parent1)
                    child2 = np.copy(parent2)
                    for j in range(n_points):
                        if np.random.rand() < 0.5:
                            child1[j] = parent2[j]
                            child2[j] = parent1[j]
                    offspring.append(child1)
                    offspring.append(child2)
                else:
                    offspring.append(selected[i])
            
            # Mutation
            for i in range(pop_size):
                if np.random.rand() < mutation_rate:
                    perturb = np.random.normal(0, 0.05, n_points)
                    offspring[i] += perturb
                    offspring[i] = np.clip(offspring[i], 0, 1)
                    offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
            
            # Elite selection
            population = np.array(population)
            offspring = np.array(offspring)
            combined = np.vstack([population, offspring])
            combined = np.vstack([combined, np.repeat([best_h], elite_size, axis=0)])
            combined = np.unique(combined, axis=0)
            
            # Keep top performers
            combined_fitness = [objective(h) for h in combined]
            combined_indices = np.argsort(combined_fitness)[:pop_size]
            population = combined[combined_indices]
        
        return population[0], objective(population[0])
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        # Adaptive perturbation based on time remaining
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.1 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
