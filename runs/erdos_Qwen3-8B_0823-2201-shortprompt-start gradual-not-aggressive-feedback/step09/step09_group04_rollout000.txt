import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    pop_size = 50
    generations = 100
    mutation_rate = 0.1
    elitism = 2
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parent2[i]
        return child
    
    def mutate(individual):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] += np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        return individual
    
    # Initial population
    population = [np.array(initial_h, dtype=np.float64) * target_sum / np.sum(initial_h) for _ in range(pop_size)]
    
    start_time = time.time()
    best_h = np.copy(h)
    best_c5 = objective(h)
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        fitness = np.array(fitness)
        
        # Select parents using tournament selection
        parents = []
        for _ in range(pop_size):
            idx1, idx2 = np.random.choice(range(pop_size), 2, replace=False)
            if fitness[idx1] < fitness[idx2]:
                parents.append(population[idx1])
            else:
                parents.append(population[idx2])
        
        # Create new population
        new_population = []
        for i in range(pop_size):
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        # Elitism
        elite_indices = np.argsort(fitness)[:elitism]
        for i in range(elitism):
            new_population[i] = population[elite_indices[i]]
        
        population = new_population
        
        # Update best solution
        current_best = np.argmin(fitness)
        current_c5 = fitness[current_best]
        if current_c5 < best_c5:
            best_h = np.copy(population[current_best])
            best_c5 = current_c5
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
