import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize population
    population_size = 50
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = np.array([p * target_sum / np.sum(p) for p in population], dtype=np.float64)
    
    def evaluate_population(pop):
        scores = np.array([objective(p) for p in pop])
        return scores
    
    # Genetic algorithm
    def genetic_algorithm():
        for _ in range(100):
            scores = evaluate_population(population)
            # Select top 20%
            elite_indices = np.argsort(scores)[:int(population_size * 0.2)]
            elite = population[elite_indices]
            
            # Crossover
            offspring = []
            for _ in range(population_size - len(elite)):
                parent1 = np.random.choice(elite)
                parent2 = np.random.choice(elite)
                child = np.zeros(n_points)

                    if np.random.rand() < 0.5:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
                offspring.append(child)
            
            # Mutation
            mutation_rate = max(0.05, 0.1 * (100 - _)/100)
            mutated = []
            for child in offspring:
                child = child.copy()
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] += np.random.uniform(-0.1, 0.1)
                        child[i] = np.clip(child[i], 0, 1)
                mutated.append(child)
            
            # Combine and normalize
            population = np.vstack((elite, mutated))
            population = np.array([p * target_sum / np.sum(p) for p in population], dtype=np.float64)

            if np.min(scores) < 0.3808:
                break

            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)

    while time.time() - start_time < budget_s:

            objective,
            perturbed_h,

            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
