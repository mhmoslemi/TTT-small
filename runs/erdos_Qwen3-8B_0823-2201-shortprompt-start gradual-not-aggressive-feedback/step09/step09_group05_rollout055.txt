import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    pop_size = 20
    generations = 100
    mutation_rate = 0.1
    tournament_size = 5
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        child[indices] = parent2[indices]
        return child
    
    def mutate(individual):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h_val = individual[i]
                perturb = np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(h_val + perturb, 0, 1)
        return individual
    
    # Evaluate population
    def evaluate(population):
        fitness = np.array([objective(ind) for ind in population])
        return fitness
    
    # Tournament selection
    def tournament_selection(population, fitness):
        participants = np.random.choice(len(population), tournament_size, replace=False)
        best_idx = participants[np.argmin(fitness[participants])]
        return population[best_idx]
    
    # Genetic algorithm
    population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
    population = [ind * target_sum / np.sum(ind) for ind in population]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate population
        fitness = evaluate(population)
        
        # Select elites
        elite_idx = np.argsort(fitness)[:2]
        elites = [population[i] for i in elite_idx]
        
        # Create new population
        new_population = elites.copy()
        
        # Create offspring
        for _ in range(pop_size - 2):
            # Select parents
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            # Crossover
            child = crossover(parent1, parent2)
            
            # Mutate
            child = mutate(child)
            
            # Ensure constraint
            child = child * target_sum / np.sum(child)
            
            new_population.append(child)
        
        # Replace population
        population = new_population
        
        # Update best solution
        current_fitness = evaluate(population)
        best_idx = np.argmin(current_fitness)
        current_h = population[best_idx]
        current_c5 = current_fitness[best_idx]
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Final refinement
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
