import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    pop_size = 20
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 2
    
    def genetic_algorithm():
        population = np.random.rand(pop_size, n_points)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            fitness = np.array([objective(ind) for ind in population])
            elite_indices = np.argsort(fitness)[:elite_size]
            elite = population[elite_indices]
            
            # Selection
            tournament_size = 5
            selected = []
            for _ in range(pop_size - elite_size):
                contestants = np.random.choice(pop_size, tournament_size, replace=False)
                winner = contestants[np.argmin(fitness[contestants])]
                selected.append(population[winner])
            
            # Crossover
            new_population = elite
            for i in range(len(selected)):
                if np.random.rand() < crossover_rate:
                    parent1 = selected[i]
                    parent2 = np.random.choice(population)
                    child = np.zeros(n_points)
                    for j in range(n_points):
                        if np.random.rand() < 0.5:
                            child[j] = parent1[j]
                        else:
                            child[j] = parent2[j]
                    new_population.append(child)
                else:
                    new_population.append(selected[i])
            
            # Mutation
            for i in range(len(new_population)):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        new_population[i][j] += np.random.uniform(-0.1, 0.1)
                        new_population[i][j] = np.clip(new_population[i][j], 0, 1)
                        new_population[i][j] = new_population[i][j] * target_sum / np.sum(new_population[i])
            
            population = new_population[:pop_size]
        
        best_h = population[np.argmin([objective(ind) for ind in population])]
        return best_h, objective(best_h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement with adaptive perturbation within budget
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.1 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
