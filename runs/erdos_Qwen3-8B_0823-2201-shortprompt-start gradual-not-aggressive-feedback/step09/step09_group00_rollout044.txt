import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.15
    crossover_rate = 0.8
    generations = 50
    
    def genetic_algorithm():
        population = np.random.rand(population_size, n_points)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            best_idx = np.argmin(fitness)
            best_h = population[best_idx].copy()
            
            # Selection
            tournament_size = 5
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            selected = population[tournament_indices]
            
            # Crossover
            offspring = []
            for i in range(0, population_size, 2):
                if np.random.rand() < crossover_rate:
                    parent1 = selected[i // 2]
                    parent2 = selected[i // 2 + 1] if i+1 < population_size else selected[i // 2]
                    child1 = np.concatenate((parent1[:n_points//2], parent2[n_points//2:]))
                    child2 = np.concatenate((parent2[:n_points//2], parent1[n_points//2:]))
                else:
                    child1, child2 = selected[i//2].copy(), selected[i//2 + 1].copy()
                offspring.append(child1)
                offspring.append(child2)
            
            # Mutation
            for i in range(population_size):
                if np.random.rand() < mutation_rate:
                    mutation = np.random.normal(0, 0.05, n_points)
                    offspring[i] += mutation

                    offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
            
            # Replace population
            population = np.array(offspring)
        
        return best_h, fitness[best_idx]

    best_h, best_c5 = genetic_algorithm()

    def local_refinement(h):

            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x

            perturbed_h,

            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
