import time

        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    pop_size = 20

    crossover_rate = 0.8
    elite_size = 2
    
    def genetic_algorithm():
        # Initialize population
        population = np.random.uniform(0, 1, (pop_size, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            
            # Select elites
            elite_indices = np.argsort(fitness)[:elite_size]
            elites = population[elite_indices]
            
            # Select parents
            parent_indices = np.random.choice(pop_size, size=pop_size, p=fitness/fitness.sum())
            parents = population[parent_indices]
            
            # Crossover
            offspring = np.zeros((pop_size - elite_size, n_points))
            for i in range(pop_size - elite_size):
                if np.random.rand() < crossover_rate:
                    # Single-point crossover
                    point = np.random.randint(1, n_points - 1)
                    offspring[i] = np.concatenate((parents[i, :point], parents[np.random.randint(pop_size), point:]))
                else:
                    # Copy from parent
                    offspring[i] = parents[i]
            
            # Mutation
            for i in range(pop_size - elite_size):
                for j in range(n_points):

                        offspring[i, j] += np.random.normal(0, 0.05)

            # Combine elites and offspring
            population = np.vstack((elites, offspring))
        
        # Evaluate final population
        fitness = np.array([objective(ind) for ind in population])
        best_index = np.argmin(fitness)

        best_c5 = fitness[best_index]
