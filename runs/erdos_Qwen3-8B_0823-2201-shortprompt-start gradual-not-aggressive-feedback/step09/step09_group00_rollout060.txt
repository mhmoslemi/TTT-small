import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)

    # Genetic algorithm parameters
    pop_size = 20
    mutation_rate = 0.1
    elite_size = 2
    max_gen = 100
    
    def genetic_algorithm():
        population = np.random.rand(pop_size, n_points)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for gen in range(max_gen):
            fitness = np.array([objective(ind) for ind in population])
            elite_indices = np.argsort(fitness)[:elite_size]
            elite = population[elite_indices]
            
            # Mutation
            mutated = np.copy(population)
            for i in range(pop_size):
                if np.random.rand() < mutation_rate:
                    mutation = np.random.normal(0, 0.05, n_points)
                    mutated[i] = np.clip(population[i] + mutation, 0, 1)
                    mutated[i] = mutated[i] * target_sum / np.sum(mutated[i])
            
            # Crossover
            for i in range(pop_size):
                if np.random.rand() < 0.7:
                    j = np.random.randint(0, pop_size)
                    crossover_point = np.random.randint(1, n_points-1)
                    mutated[i] = np.concatenate((mutated[i][:crossover_point], population[j][crossover_point:]))
                    mutated[i] = mutated[i] * target_sum / np.sum(mutated[i])
            
            # Replace population
            population = np.vstack((elite, mutated))
            population = np.unique(population, axis=0)
            population = population[:pop_size]

    def local_refinement(h):

            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()
    best_h = local_refinement(best_h)

    while time.time() - start_time < budget_s:
