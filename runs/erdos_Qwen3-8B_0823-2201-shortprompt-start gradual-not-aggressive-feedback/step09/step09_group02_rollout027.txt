import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    tournament_size = 5
    
    def evolve():
        population = [h.copy() for _ in range(population_size)]
        for _ in range(generations):
            # Evaluate fitness
            fitness = [objective(ind) for ind in population]
            # Select parents
            parents = []
            for _ in range(population_size // 2):
                contestants = np.random.choice(population_size, tournament_size, replace=False)
                best = contestants[np.argmin([fitness[i] for i in contestants])]
                parents.append(population[best])
            # Crossover
            offspring = []
            for i in range(population_size // 2):
                parent1 = parents[i]
                parent2 = parents[population_size // 2 + i] if i < population_size // 2 else parents[i]
                child = np.zeros(n_points)
                for j in range(n_points):
                    if np.random.rand() < 0.5:
                        child[j] = parent1[j]
                    else:
                        child[j] = parent2[j]
                offspring.append(child)
            # Mutation
            for i in range(len(offspring)):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        offspring[i][j] += np.random.uniform(-0.1, 0.1)
                        offspring[i][j] = np.clip(offspring[i][j], 0, 1)
            # Replace population
            population = offspring
            # Keep best solution
            best_idx = np.argmin(fitness)
            best_h = population[best_idx]
            best_c5 = fitness[best_idx]
        return best_h, best_c5
    
    # Run genetic algorithm
    start_time = time.time()
    best_h, best_c5 = evolve()
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)

        T = 1.0
        cool_rate = 0.995
        max_iter = 100
        best_h_current = best_h.copy()
        best_c5_current = best_c5

            if T < 1e-5:
                break
            perturb_size = max(0.01, 0.05 * (T / 1.0))
            perturb = np.random.normal(0, perturb_size, n_points)

        # Local refinement

            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        best_h = result.x
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
