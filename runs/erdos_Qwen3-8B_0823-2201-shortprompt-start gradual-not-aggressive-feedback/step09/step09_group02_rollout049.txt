import time

        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)

    h = np.array(initial_h, dtype=np.float64)

    def differential_evolution():

            popsize=15,
            maxiter=50,
            tol=1e-8,
            mutation=(0.5, 1),

    def local_refinement(h):

            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )

    # Start with differential evolution
