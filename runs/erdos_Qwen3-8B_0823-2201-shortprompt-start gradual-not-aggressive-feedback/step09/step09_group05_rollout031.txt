import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize population
    population = np.array([np.array(initial_h, dtype=np.float64) * target_sum / np.sum(initial_h) + 
                          np.random.uniform(-0.05, 0.05, n_points) for _ in range(50)])
    population = np.clip(population, 0, 1)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    def fitness(individual):
        return objective(individual)
    
    best_h = population[0]
    best_c5 = fitness(best_h)
    
    start_time = time.time()
    
    # Genetic algorithm parameters
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 50
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness_scores = np.array([fitness(ind) for ind in population])
        
        # Select parents
        parents = population[np.argsort(fitness_scores)[:10]]
        
        # Crossover
        offspring = []
        for _ in range(50 - len(parents)):
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            child = np.where(np.random.rand(n_points) < crossover_rate, parent1, parent2)
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            offspring[i] = np.where(np.random.rand(n_points) < mutation_rate, 
                                   np.clip(offspring[i] + np.random.uniform(-0.05, 0.05, n_points), 0, 1), 
                                   offspring[i])
        
        # Local refinement with SLSQP
        for ind in offspring:
            ind = ind * target_sum / np.sum(ind)
            result = opt.minimize(
                objective,
                ind,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            offspring[i] = result.x
        
        # Update population
        population = np.vstack((parents, offspring))
        population = np.clip(population, 0, 1)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        # Update best solution
        for ind in population:
            current_c5 = fitness(ind)
            if current_c5 < best_c5:
                best_h = ind
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
