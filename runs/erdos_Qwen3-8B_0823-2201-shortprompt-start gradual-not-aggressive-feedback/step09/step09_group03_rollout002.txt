import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)

    population_size = 20
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    def genetic_algorithm():
        population = [np.random.rand(n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = [objective(ind) for ind in population]
            best_idx = np.argmin(fitness)
            best_fitness = fitness[best_idx]
            
            # Select parents
            parents = np.random.choice(population, size=population_size, p=fitness/np.sum(fitness))
            
            # Crossover
            offspring = []
            for i in range(population_size):
                if np.random.rand() < crossover_rate:
                    j = np.random.randint(population_size)
                    parent1 = parents[i]
                    parent2 = parents[j]
                    child = np.concatenate((parent1[:n_points//2], parent2[n_points//2:]))
                else:
                    child = parents[i]
                offspring.append(child)
            
            # Mutation
            for i in range(population_size):
                if np.random.rand() < mutation_rate:
                    perturb = np.random.normal(0, 0.05, n_points)
                    offspring[i] = np.clip(offspring[i] + perturb, 0, 1)
                    offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])

            population = offspring
        
        # Return best solution
        fitness = [objective(ind) for ind in population]

            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()

    while time.time() - start_time < budget_s:
