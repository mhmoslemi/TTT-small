import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 10
    mutation_rate = 0.15
    crossover_rate = 0.8
    
    def breed(parent1, parent2):
        if np.random.rand() < crossover_rate:
            split = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:split], parent2[split:]])
        else:
            child = parent1.copy()
        return child
    
    def mutate(individual):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] += np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        return individual
    
    # Evolutionary search
    population = [np.random.rand(n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        # Select top performers
        elite = np.argsort(fitness)[:population_size//2]
        # Generate new population
        new_population = [local_refinement(population[i].copy()) for i in elite]
        for _ in range(len(new_population), population_size):
            parent1, parent2 = np.random.choice(elite, 2, replace=False)
            child = breed(population[parent1], population[parent2])
            child = mutate(child)
            new_population.append(child)
        population = new_population
    
    # Final refinement
    best_h = np.array(population[np.argmin(fitness)])
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
