import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 50
    
    def genetic_algorithm():
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = [objective(h) for h in population]
            best_idx = np.argmin(fitness)
            best_h = population[best_idx]
            
            # Select parents
            parents = np.random.choice(population, size=population_size, p=np.array(fitness)/np.sum(fitness))
            
            # Crossover
            offspring = []
            for i in range(population_size):
                if np.random.rand() < crossover_rate:
                    parent1 = parents[i]
                    parent2 = parents[np.random.randint(population_size)]
                    child = np.zeros(n_points)
                    for j in range(n_points):
                        if np.random.rand() < 0.5:
                            child[j] = parent1[j]
                        else:
                            child[j] = parent2[j]
                    offspring.append(child)
                else:
                    offspring.append(parents[i])
            
            # Mutation
            for i in range(population_size):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        offspring[i][j] += np.random.uniform(-0.1, 0.1)
                        offspring[i][j] = np.clip(offspring[i][j], 0, 1)
            
            # Replace population
            population = [h * target_sum / np.sum(h) for h in offspring]
        
        return best_h, np.min(fitness)

            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()

            perturbed_h,
            method='SLSQP',

    return best_h, best_c5, n_points
