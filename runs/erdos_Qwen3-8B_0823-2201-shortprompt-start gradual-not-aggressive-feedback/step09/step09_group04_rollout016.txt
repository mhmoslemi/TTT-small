import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.15
    crossover_rate = 0.8
    elite_size = 2
    
    def tournament_selection(population, fitnesses):
        tournament = np.random.choice(len(population), size=5, replace=False)
        best_idx = tournament[np.argmin(fitnesses[tournament])]
        return population[best_idx].copy()
    
    def mutate(individual):
        for i in range(len(individual)):
            if np.random.rand() < mutation_rate:
                individual[i] += np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        return individual
    
    def crossover(parent1, parent2):
        if np.random.rand() < crossover_rate:
            child = np.zeros_like(parent1)
            for i in range(len(parent1)):
                if np.random.rand() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            return child
        return parent1
    
    # Initial population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    population = np.array(population)
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitnesses = np.array([objective(ind) for ind in population])
        
        # Selection
        selected = [tournament_selection(population, fitnesses) for _ in range(population_size)]
        
        # Crossover and mutation
        offspring = []
        for i in range(population_size):
            parent1 = selected[i % population_size]
            parent2 = selected[(i+1) % population_size]
            child = crossover(parent1, parent2)
            child = mutate(child)
            offspring.append(child)
        
        # Combine and sort
        population = np.vstack([population, offspring])
        population = np.unique(population, axis=0)
        population = population[:population_size]
        
        # Elite selection
        elite_indices = np.argsort(fitnesses)[:elite_size]
        population = np.vstack([population[elite_indices], np.random.uniform(0, 1, (elite_size, n_points))])
        population = np.array([h * target_sum / np.sum(h) for h in population])
        
        # Update best solution
        current_best = population[np.argmin(fitnesses)]
        current_c5 = objective(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Final refinement
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
