        c5 = np.max(corr) * dx
        return c5

            # Select parents
            parent1 = np.random.choice(population)
            parent2 = np.random.choice(population)
            
            # Crossover

                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:split], parent2[split:]])
            else:
                child = np.random.rand(n_points)
            
            # Mutation

                if np.random.rand() < mutation_rate:
                    child[i] = np.random.uniform(0, 1)
            
            # Normalize
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        # Keep best solution

        population.sort(key=lambda x: objective(x))
        return population[:pop_size]
