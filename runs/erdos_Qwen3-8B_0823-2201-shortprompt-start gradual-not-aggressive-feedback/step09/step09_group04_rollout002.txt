import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize population
    population_size = 50
    population = np.random.rand(population_size, n_points)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    def fitness(h):
        return objective(h)
    
    def select_parents(population, fitnesses):
        idx = np.argsort(fitnesses)
        return population[idx[:int(population_size/2)]]
    
    def crossover(parents):
        offspring = []
        for _ in range(population_size - len(parents)):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            crossover_point = np.random.randint(1, n_points-1)
            child = np.concatenate((p1[:crossover_point], p2[crossover_point:]))
            offspring.append(child)
        return offspring
    
    def mutate(individual):
        mutation_rate = 0.1
        mutation_strength = 0.1
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] += np.random.uniform(-mutation_strength, mutation_strength)
                individual[i] = np.clip(individual[i], 0, 1)
        return individual
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        fitnesses = np.array([fitness(ind) for ind in population])
        parents = select_parents(population, fitnesses)
        offspring = crossover(parents)
        for i in range(len(offspring)):
            offspring[i] = mutate(offspring[i])
        population = np.vstack((parents, offspring))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        current_best = np.min(fitnesses)
        if current_best < best_c5:
            best_h = population[np.argmin(fitnesses)]
            best_c5 = current_best
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
