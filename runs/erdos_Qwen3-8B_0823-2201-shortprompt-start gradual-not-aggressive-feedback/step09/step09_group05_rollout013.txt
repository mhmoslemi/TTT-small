import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Hybrid approach: particle swarm optimization with local refinement
    def particle_swarm():
        swarm_size = 20
        inertia = 0.7
        cogn_coeff = 1.5
        soc_coeff = 1.5
        max_iter = 100
        best_h = h.copy()
        best_c5 = objective(best_h)
        swarm = np.random.rand(swarm_size, n_points)
        swarm = swarm * target_sum / np.sum(swarm, axis=1, keepdims=True)
        velocities = np.random.rand(swarm_size, n_points) * 0.1
        
        for _ in range(max_iter):
            for i in range(swarm_size):
                r1, r2 = np.random.rand(2, n_points)
                velocities[i] = inertia * velocities[i] + \
                                cogn_coeff * r1 * (best_h - swarm[i]) + \
                                soc_coeff * r2 * (np.mean(swarm, axis=0) - swarm[i])
                swarm[i] += velocities[i]
                swarm[i] = np.clip(swarm[i], 0, 1)
                swarm[i] = swarm[i] * target_sum / np.sum(swarm[i])
                current_c5 = objective(swarm[i])
                if current_c5 < best_c5:
                    best_h = swarm[i].copy()
                    best_c5 = current_c5
        return best_h, best_c5
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h, best_c5 = particle_swarm()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement with adaptive perturbation within budget
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.1 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
