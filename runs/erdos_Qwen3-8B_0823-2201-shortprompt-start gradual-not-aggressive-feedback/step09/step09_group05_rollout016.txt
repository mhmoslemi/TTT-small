import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    elite_size = 2
    tournament_size = 5
    
    def fitness(individual):
        return objective(individual)
    
    def select_parents(population, fitnesses):
        tournament_indices = np.random.randint(len(population), size=tournament_size)
        best_idx = tournament_indices[np.argmin(fitnesses[tournament_indices])]
        return population[best_idx].copy()
    
    def crossover(parent1, parent2):
        crossover_point = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        return child
    
    def mutate(individual):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                mutation = np.random.uniform(-0.1, 0.1)
                individual[i] += mutation
                individual[i] = np.clip(individual[i], 0, 1)
        return individual
    
    # Initialize population
    population = [np.random.rand(n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitnesses = [fitness(ind) for ind in population]
        # Select elites
        elite_indices = np.argsort(fitnesses)[:elite_size]
        elites = [population[i] for i in elite_indices]
        # Create new population
        new_population = elites.copy()
        while len(new_population) < population_size:
            parent1 = select_parents(population, fitnesses)
            parent2 = select_parents(population, fitnesses)
            child = crossover(parent1, parent2)
            child = mutate(child)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = new_population
    
    # Local refinement with SLSQP
    best_h = population[np.argmin([fitness(ind) for ind in population])]
    best_h = best_h * target_sum / np.sum(best_h)
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
