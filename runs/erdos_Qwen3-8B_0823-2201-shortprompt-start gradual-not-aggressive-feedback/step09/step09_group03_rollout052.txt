import scipy.optimize as opt
import time

    def objective(h_flat):

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h_flat):
        h = h_flat.copy()
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial optimization with SLSQP
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    # Adaptive local search with gradient-based refinement
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 20, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    # Final refinement with simulated annealing
    T = 1.0
    cool_rate = 0.995
    max_iter = 100
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        T *= cool_rate
        if T < 1e-5:
            break
        perturb = np.random.normal(0, 0.05 * (T / 1.0), n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)

            best_c5 = current_c5
