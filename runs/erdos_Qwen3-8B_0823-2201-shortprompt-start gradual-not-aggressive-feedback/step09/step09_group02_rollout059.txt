import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.15
    crossover_rate = 0.8
    elite_size = 2
    
    def create_individual():
        return np.random.rand(n_points) * target_sum
    
    def evaluate_population(population):
        return np.array([objective(ind) for ind in population])
    
    def select_parents(population, fitnesses):
        idx = np.argsort(fitnesses)
        return population[idx[:elite_size]], fitnesses[idx[:elite_size]]
    
    def crossover(parent1, parent2):
        if np.random.rand() < crossover_rate:
            alpha = np.random.rand()
            return alpha * parent1 + (1 - alpha) * parent2
        return parent1
    
    def mutate(individual):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] += np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        return individual * target_sum / np.sum(individual)
    
    # Initialize population
    population = [create_individual() for _ in range(population_size)]
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitnesses = evaluate_population(population)
        parents, parent_fitnesses = select_parents(population, fitnesses)
        
        # Create next generation
        next_gen = []
        for _ in range(population_size - elite_size):
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child)
            next_gen.append(child)
        
        population = parents + next_gen
        current_best = np.min(fitnesses)
        if current_best < best_c5:
            best_h = population[np.argmin(fitnesses)].copy()
            best_c5 = current_best
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
