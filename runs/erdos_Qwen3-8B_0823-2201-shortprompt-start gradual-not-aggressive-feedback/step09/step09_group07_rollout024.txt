import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Phase 1: Structured local search
    def phase1():
        best_h = h.copy()
        best_c5 = objective(best_h)
        perturbation_pattern = np.sin(np.linspace(0, 2*np.pi, n_points))
        
        for _ in range(30):
            perturb = 0.05 * perturbation_pattern
            perturbed = np.clip(h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            
            result = opt.minimize(
                objective,
                perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 20, 'ftol': 1e-8}
            )
            
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = objective(current_h)
            
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                h = best_h
        
        return best_h, best_c5
    
    # Phase 2: Global search with adaptive perturbation
    def phase2(best_h, best_c5):
        remaining_time = budget_s - time.time() + start_time
        perturb_size = 0.1
        for _ in range(50):
            if remaining_time <= 0:
                break
            perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
            perturbed_h = best_h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            
            result = opt.minimize(
                objective,
                perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 10, 'ftol': 1e-8}
            )
            
            current_h = result.x

            current_c5 = objective(current_h)
            
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                perturb_size *= 0.95
        
        return best_h, best_c5
    
    start_time = time.time()
    best_h, best_c5 = phase1()
    best_h, best_c5 = phase2(best_h, best_c5)
    
    return best_h, best_c5, n_points
