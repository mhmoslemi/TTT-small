import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def tournament_selection(population, fitnesses, k=3):
        indices = np.random.choice(len(population), k, replace=False)
        best_idx = np.argmin(fitnesses[indices])
        return population[indices[best_idx]]
    
    def crossover(parent1, parent2):
        point = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:point], parent2[point:]])
        return child
    
    def mutate(individual, mutation_rate=0.1):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] = np.clip(individual[i] + np.random.normal(0, 0.1), 0, 1)
        return individual
    
    def genetic_algorithm():
        population = [np.random.rand(n_points) for _ in range(50)]
        population = [h * target_sum / np.sum(h) for h in population]
        best_h = h.copy()
        best_c5 = objective(best_h)
        
        for _ in range(100):
            fitnesses = [objective(ind) for ind in population]
            best_idx = np.argmin(fitnesses)
            best_h = population[best_idx]
            best_c5 = fitnesses[best_idx]
            
            new_population = []
            for _ in range(25):
                parent1 = tournament_selection(population, fitnesses)
                parent2 = tournament_selection(population, fitnesses)
                child = crossover(parent1, parent2)
                child = mutate(child)
                child = child * target_sum / np.sum(child)
                new_population.append(child)
            
            population = new_population + [best_h]
        
        return best_h, best_c5
    
    # Genetic algorithm
    best_h, best_c5 = genetic_algorithm()
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Adaptive perturbation based on time remaining
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.1 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
