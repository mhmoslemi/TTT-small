import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    target_sum = n_points / 2

        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    tournament_size = 5
    mutation_rate = 0.1
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        child[indices] = parent2[indices]
        return child
    
    def mutate(individual):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                if np.random.rand() < 0.5:
                    individual[i] += np.random.uniform(-0.1, 0.1)
                else:
                    individual[i] = np.random.uniform(0, 1)
        individual = np.clip(individual, 0, 1)
        individual = individual * target_sum / np.sum(individual)
        return individual
    
    # Initial population
    population = [np.array(initial_h, dtype=np.float64) * target_sum / np.sum(initial_h) for _ in range(population_size)]
    
    # Evaluate initial population
    fitness = [objective(ind) for ind in population]
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Select parents using tournament selection
        tournament = np.random.choice(population_size, size=tournament_size)
        parent1 = population[tournament[0]]
        parent2 = population[tournament[1]]

                parent1 = population[tournament[i]]
            if objective(population[tournament[i]]) < objective(parent2):
                parent2 = population[tournament[i]]
        
        # Crossover and mutate
        child = crossover(parent1, parent2)
        child = mutate(child)
        
        # Evaluate child
        child_fitness = objective(child)
        
        # Replace worst individual
        worst_idx = np.argmax(fitness)
        if child_fitness < fitness[worst_idx]:

            fitness[worst_idx] = child_fitness
