import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Gradient-based local optimization
    def local_optimize(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Structured perturbation and refinement
    def perturb_and_refine(h):

        perturbed = np.clip(h + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        result = opt.minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h = local_optimize(h)
    best_c5 = objective(best_h)
    
    # Iterative refinement with decreasing perturbation
    while time.time() - start_time < budget_s:

            objective,
            perturbed,
            method='SLSQP',

            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
