import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Start from initial solution
    best_h = initial_h.copy()
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Simulated annealing parameters
    T = 1.0
    cooling_rate = 0.995
    iterations = 1000
    start_time = time.time()

    for _ in range(iterations):
        if time.time() - start_time > budget_s:
            break
        # Create a perturbed solution
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Evaluate objective
        current_c5 = objective(perturbed_h)
        
        # Acceptance probability
        if current_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - current_c5) / T):
            best_h = perturbed_h
            best_c5 = current_c5
        
        T *= cooling_rate

    # Final refinement with local optimization
    start_time = time.time()

        perturb = np.random.uniform(-0.05, 0.05, n_points)
        perturbed_h = best_h + perturb

        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
