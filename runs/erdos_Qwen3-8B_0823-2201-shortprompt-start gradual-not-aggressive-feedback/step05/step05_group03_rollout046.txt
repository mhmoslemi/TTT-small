import scipy.optimize as opt
import time

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def generate_population(size):
        pop = np.random.rand(size, n_points)
        pop = np.clip(pop, 0, 1)
        pop = pop * target_sum / np.sum(pop, axis=1, keepdims=True)
        return pop
    
    def differential_evolution():
        population = generate_population(30)
        best_h = population[0]
        best_c5 = objective(best_h)
        
        for _ in range(150):
            for i in range(len(population)):

                    best_h = mutant
                    best_c5 = mutant_c5
                population[i] = mutant

    def local_refinement(h):

            h,
            method='SLSQP',
            bounds=bounds,

        )
