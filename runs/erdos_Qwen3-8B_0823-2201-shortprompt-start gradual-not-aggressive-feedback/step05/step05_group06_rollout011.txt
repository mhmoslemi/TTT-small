    population = population * target_sum / np.sum(population, axis=1, keepdims=True)

        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        
        # Select parents
        parents = population[np.argsort(fitness)[:population_size//2]]
        
        # Crossover
        offspring = []

            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):

                if np.random.rand() < mutation_rate:
                    offspring[i][j] = np.clip(offspring[i][j] + np.random.normal(0, 0.05), 0, 1)
