import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Start from initial solution
    best_h = initial_h.copy()
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Optimization parameters
    max_iter = 50
    perturb_range = 0.05
    refine_steps = 10

    # Main optimization loop
    start_time = time.time()
    for _ in range(refine_steps):
        # Random perturbation
        perturb = np.random.uniform(-perturb_range, perturb_range, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Local optimization
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': max_iter, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        
        # Check time budget
        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5, n_points
