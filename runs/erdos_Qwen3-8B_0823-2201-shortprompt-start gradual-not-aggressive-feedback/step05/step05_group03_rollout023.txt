    def objective(h):

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def generate_individual():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    def crossover(parent1, parent2):
        alpha = np.random.rand(n_points)
        child = alpha * parent1 + (1 - alpha) * parent2
        child = np.clip(child, 0, 1)
        child = child * target_sum / np.sum(child)
        return child
    
    def mutate(h):
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        mutated = h + perturb
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * target_sum / np.sum(mutated)
        return mutated
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    population = [generate_individual() for _ in range(50)]
    best_h = min(population, key=objective)
    best_c5 = objective(best_h)
