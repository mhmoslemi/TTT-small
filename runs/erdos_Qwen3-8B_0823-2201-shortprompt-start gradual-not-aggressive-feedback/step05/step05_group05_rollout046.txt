import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial construction with smooth transitions
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 3:
            initial_h[i] = 0.0
        elif i < 2 * n_points // 3:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Function to apply local optimization
    def local_optimize(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x

    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 100
    
    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    
    start_time = time.time()
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate population
        fitness = [objective(h) for h in population]
        # Select parents using tournament selection
        parents = []
        for _ in range(population_size):
            candidates = np.random.choice(range(population_size), size=5, replace=False)
            best_candidate = candidates[np.argmin(fitness[candidates])]
            parents.append(population[best_candidate])
        # Create offspring
        offspring = []
        for i in range(population_size):
            if np.random.rand() < crossover_rate:
                parent1 = parents[i % population_size]
                parent2 = parents[(i+1) % population_size]
                child = np.zeros(n_points)
                for j in range(n_points):
                    if np.random.rand() < 0.5:
                        child[j] = parent1[j]
                    else:
                        child[j] = parent2[j]
            else:
                child = parents[i % population_size].copy()
            # Mutate
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[j] = np.clip(child[j] + np.random.uniform(-0.1, 0.1), 0, 1)
            # Scale to maintain integral constraint
            child = child * target_sum / np.sum(child)
            offspring.append(child)
        # Replace population
        population = offspring
        # Update best solution
        for h in population:
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Local optimization of best solution
    best_h = local_optimize(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
