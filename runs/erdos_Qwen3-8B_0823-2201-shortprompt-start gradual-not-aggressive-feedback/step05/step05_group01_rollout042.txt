import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    temp = 1.0
    cooling_rate = 0.995

    start_time = time.time()
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Create new population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        for h in population:
            h = h * target_sum / np.sum(h)
        
        # Evaluate population
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h_gen = population[best_idx]
        best_c5_gen = fitness[best_idx]
        
        # Update best solution
        if best_c5_gen < best_c5:
            best_h = best_h_gen
            best_c5 = best_c5_gen
        
        # Selection and crossover
        selected = np.argsort(fitness)[:int(population_size/2)]
        new_population = []
        for i in range(int(population_size/2)):
            parent1 = population[selected[i]]
            if random.random() < crossover_rate:
                parent2 = population[selected[i+1]]
                child = np.zeros(n_points)
                for j in range(n_points):
                    if random.random() < 0.5:
                        child[j] = parent1[j]
                    else:
                        child[j] = parent2[j]
            else:
                child = parent1
            new_population.append(child)
        
        # Mutation
        for i in range(len(new_population)):
            h = new_population[i]
            for j in range(n_points):
                if random.random() < mutation_rate:
                    h[j] += np.random.uniform(-0.1, 0.1)
                    h[j] = np.clip(h[j], 0, 1)
                    h = h * target_sum / np.sum(h)
            new_population[i] = h
        
        population = new_population
        
        # Annealing
        temp *= cooling_rate
    
    return best_h, best_c5, n_points
