import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    time_remaining = budget_s
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] += np.random.uniform(-0.2, 0.2)
                h[i] = np.clip(h[i], 0, 1)
        return h
    
    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        for i in range(len(parent1)):
            if random.random() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child
    
    for _ in range(100):
        if time.time() - start_time >= budget_s:
            break
        # Generate new candidates
        candidates = []
        for _ in range(5):
            candidate = mutate(best_h)
            candidates.append(candidate)
        
        # Optimize each candidate
        for candidate in candidates:
            result = opt.minimize(
                objective,
                candidate,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},

            )
