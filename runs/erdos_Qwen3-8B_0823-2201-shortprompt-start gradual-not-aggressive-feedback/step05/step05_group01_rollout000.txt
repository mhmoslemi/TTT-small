import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial solution
    best_h = initial_h.copy()
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 2
    time_remaining = budget_s

    for _ in range(generations):
        if time_remaining <= 0:
            break
        
        # Create population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        
        # Evaluate population
        fitness = [objective(h) for h in population]
        
        # Select elite
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elite.copy()
        for _ in range(population_size - elite_size):
            # Select parents
            parent1 = np.random.choice(elite)
            parent2 = np.random.choice(elite)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:split], parent2[split:]])
            else:
                child = np.random.uniform(0, 1, n_points)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, 0.05, n_points)
                child += mutation
                child = np.clip(child, 0, 1)
            
            # Normalize
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        population = new_population
        fitness = [objective(h) for h in population]
        best_index = np.argmin(fitness)
        best_h = population[best_index]
        best_c5 = fitness[best_index]
        time_remaining -= 1
        
        # Local refinement with SLSQP
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        best_c5 = objective(best_h)
        time_remaining -= 1
    
    return best_h, best_c5, n_points
