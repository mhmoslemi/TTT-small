import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    pop_size = 20
    generations = 50
    mutation_rate = 0.15
    elite_size = 2
    
    def mutate(individual):
        mutation = np.random.normal(0, 0.1, n_points)
        mutated = np.clip(individual + mutation, 0, 1)
        mutated = mutated * target_sum / np.sum(mutated)
        return mutated
    
    def crossover(parent1, parent2):
        crossover_point = np.random.randint(1, n_points - 1)
        child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
        child = child * target_sum / np.sum(child)
        return child
    
    # Initial population
    population = [np.array(initial_h, dtype=np.float64) * target_sum / np.sum(initial_h) for _ in range(pop_size)]
    
    start_time = time.time()
    
    for generation in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elites.copy()
        
        # Generate offspring
        for _ in range(pop_size - elite_size):
            parent1, parent2 = np.random.choice(population, 2, replace=False)
            child = crossover(parent1, parent2)
            if np.random.rand() < mutation_rate:
                child = mutate(child)
            new_population.append(child)
        
        population = new_population
    
    # Refine the best solution with SLSQP
    best_h = min(population, key=objective)
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
