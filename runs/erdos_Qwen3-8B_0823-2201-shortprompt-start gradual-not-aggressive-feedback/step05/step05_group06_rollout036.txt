import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    best_c5 = objective(best_h)
    start_time = time.time()

    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Genetic algorithm for global search
    population_size = 20
    generations = 50
    mutation_rate = 0.1
    elite_size = 2

    def genetic_search():
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        for _ in range(generations):
            # Evaluate fitness
            fitness = [objective(h) for h in population]
            # Select elites
            elite_indices = np.argsort(fitness)[:elite_size]
            elites = [population[i] for i in elite_indices]
            # Create next generation
            next_gen = elites.copy()
            for _ in range(population_size - elite_size):
                # Select parents
                parent1 = random.choice(elites)
                parent2 = random.choice(elites)

                        child[i] = parent1[i]

                        child[i] = parent2[i]

                    if random.random() < mutation_rate:
                        child[i] = np.random.uniform(0, 1)
                # Normalize
                child = child * target_sum / np.sum(child)
                next_gen.append(child)
            population = next_gen
        # Find best in population
        best_in_pop = min(population, key=objective)
        return best_in_pop

    # Run genetic search
    genetic_h = genetic_search()
    genetic_c5 = objective(genetic_h)
    
    # Compare and choose best
    if genetic_c5 < best_c5:

        best_c5 = genetic_c5
