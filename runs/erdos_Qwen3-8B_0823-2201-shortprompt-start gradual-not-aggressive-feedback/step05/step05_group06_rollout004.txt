import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Initial optimization
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Genetic algorithm for exploration
    population_size = 20
    generations = 10
    mutation_rate = 0.1

    def crossover(parent1, parent2):
        child = parent1.copy()
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        child[indices] = parent2[indices]
        return child

    def mutate(individual):
        for i in range(n_points):
            if random.random() < mutation_rate:
                individual[i] += np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        individual = individual * target_sum / np.sum(individual)
        return individual

    for _ in range(generations):
        if time.time() - start_time >= budget_s - 1:
            break
        population = [best_h.copy() for _ in range(population_size)]
        for p in population:
            p = mutate(p)
        for i in range(population_size):
            for j in range(i+1, population_size):
                child = crossover(population[i], population[j])
                child = mutate(child)
                child_c5 = objective(child)
                if child_c5 < best_c5:
                    best_h = child
                    best_c5 = child_c5
        # Local optimization
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        best_h = result.x
        best_c5 = objective(best_h)

    return best_h, best_c5, n_points
