import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    elite_size = 2
    
    def genetic_algorithm():
        population = np.random.uniform(0, 1, (population_size, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for gen in range(generations):
            fitness = np.array([objective(ind) for ind in population])
            elite_indices = np.argsort(fitness)[:elite_size]
            elites = population[elite_indices]
            
            # Selection
            tournament = np.random.choice(np.arange(population_size), size=population_size, replace=True)
            selected = population[tournament[np.argsort(fitness[tournament])[:population_size//2]]]
            
            # Crossover
            offspring = []
            for i in range(0, population_size, 2):
                if i+1 < population_size:
                    parent1 = selected[i]
                    parent2 = selected[i+1]
                    crossover_point = np.random.randint(1, n_points-1)
                    child1 = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
                    child2 = np.concatenate((parent2[:crossover_point], parent1[crossover_point:]))
                    offspring.append(child1)
                    offspring.append(child2)
                else:
                    offspring.append(selected[i])
            
            # Mutation
            for i in range(len(offspring)):
                if random.random() < mutation_rate:
                    mutation = np.random.uniform(-0.1, 0.1, n_points)
                    offspring[i] += mutation
                    offspring[i] = np.clip(offspring[i], 0, 1)
                    offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
            
            # Replace population
            population = np.vstack((elites, offspring))
            population = np.unique(population, axis=0)
            
            # Check if we found a better solution
            current_fitness = np.min(fitness)
            if current_fitness < best_c5:
                best_c5 = current_fitness
                best_h = population[np.argmin(fitness)]
        
        return best_h, best_c5
    
    # Run genetic algorithm
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Run genetic algorithm for a portion of the budget
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 1:
            break
        best_h, best_c5 = genetic_algorithm()
    
    return best_h, best_c5, n_points
