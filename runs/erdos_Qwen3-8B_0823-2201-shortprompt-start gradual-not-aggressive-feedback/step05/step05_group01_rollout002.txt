import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def local_optimize(h):
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x

    population_size = 50
    generations = 100
    mutation_rate = 0.1
    elite_size = 5

    population = [local_optimize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]

    best_h = min(population, key=objective)
    best_c5 = objective(best_h)

    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        # Tournament selection
        new_population = elites
        while len(new_population) < population_size:
            a = np.random.randint(0, population_size)
            b = np.random.randint(0, population_size)
            if fitness[a] < fitness[b]:
                parent = population[a]
            else:
                parent = population[b]
            # Mutate
            mutation = np.random.normal(0, mutation_rate, n_points)
            child = parent + mutation
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = new_population
        # Update best
        current_best = min(population, key=objective)
        if objective(current_best) < best_c5:
            best_h = current_best
            best_c5 = objective(best_h)
            print(f"Improved bound: {best_c5}")

    return best_h, best_c5, n_points
