import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def evaluate(solution):
        return objective(solution)

    # Initial solution
    best_h = initial_h.copy()
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm parameters
    pop_size = 20
    generations = 100
    mutation_rate = 0.15
    crossover_rate = 0.8
    elite_fraction = 0.2

    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Create population
        population = np.random.uniform(0, 1, (pop_size, n_points))
        population = np.vstack([best_h, population])
        
        # Evaluate population
        fitness = np.array([evaluate(ind) for ind in population])
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = population[sorted_indices]
        fitness = fitness[sorted_indices]
        
        # Select elites
        elite_size = int(elite_fraction * pop_size)
        elites = population[:elite_size]
        
        # Generate new population
        new_population = elites.copy()
        for i in range(elite_size, pop_size):
            # Select parents
            parent1 = population[np.random.choice(elite_size)]
            parent2 = population[np.random.choice(elite_size)]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent1
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation = np.random.uniform(-0.1, 0.1, n_points)
                child += mutation
                child = np.clip(child, 0, 1)
            
            # Normalize
            child = child * target_sum / np.sum(child)
            new_population[i] = child
        
        # Update best solution
        population = new_population
        fitness = np.array([evaluate(ind) for ind in population])
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Local optimization
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        best_c5 = objective(best_h)
        
        print(f"Generation {gen}: Best C5 = {best_c5}")

    return best_h, best_c5, n_points
