import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Start from initial solution
    best_h = initial_h.copy()
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm parameters
    population_size = 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 100

    # Initialize population
    population = [best_h.copy() for _ in range(population_size)]
    
    # Evaluate initial population
    population_fitness = [objective(ind) for ind in population]
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Select parents
        parent_indices = np.argsort(population_fitness)[:int(population_size/2)]
        parents = [population[i] for i in parent_indices]
        
        # Crossover
        children = []
        for _ in range(population_size - len(parents)):
            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            child = np.copy(parent1)
            for i in range(n_points):
                if np.random.rand() < crossover_rate:
                    child[i] = parent2[i]
            children.append(child)
        
        # Mutation
        for child in children:
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.uniform(-0.05, 0.05)
                    child[i] = np.clip(child[i], 0, 1)
                    child = child * target_sum / np.sum(child)
        
        # Evaluate children
        child_fitness = [objective(child) for child in children]
        
        # Replace population
        population = parents + children
        population_fitness = [objective(ind) for ind in population]
        
        # Update best solution
        best_idx = np.argmin(population_fitness)
        if population_fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = population_fitness[best_idx]
    
    # Final refinement with local search
    while time.time() - start_time < budget_s:
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    return best_h, best_c5, n_points
