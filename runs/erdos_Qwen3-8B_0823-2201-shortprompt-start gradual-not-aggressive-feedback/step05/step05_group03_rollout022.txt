    def objective(h):

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

        for _ in range(generations):
            new_population = []

                parent1 = population[a]
                parent2 = population[b]
                child1 = (parent1 + np.random.rand(n_points) * (parent2 - parent1)) / 2
                child2 = (parent2 + np.random.rand(n_points) * (parent1 - parent2)) / 2

                    c = np.clip(c, 0, 1)
                    c = c * target_sum / np.sum(c)
                    new_population.append(c)
            population = new_population
        return population
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
