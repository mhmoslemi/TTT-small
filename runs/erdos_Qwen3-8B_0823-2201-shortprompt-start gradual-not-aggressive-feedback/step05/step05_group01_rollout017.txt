import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial solution
    best_h = initial_h.copy()
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.15
    generations = 100
    cooling_schedule = 0.995
    temperature = 1.0
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
            
        # Create population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        
        # Evaluate population
        population_fitness = []
        for h in population:
            h = h * target_sum / np.sum(h)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            population_fitness.append(c5)
        
        # Select parents
        parents = [population[i] for i in np.argsort(population_fitness)[:population_size//2]]
        
        # Crossover and mutation
        new_population = []
        for _ in range(population_size):
            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            child = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.uniform(-0.1, 0.1)
                    child[i] = np.clip(child[i], 0, 1)
            new_population.append(child)
        
        # Replace population
        population = new_population
        
        # Local optimization on best solution
        best_index = np.argmin(population_fitness)
        best_candidate = population[best_index]
        best_candidate = best_candidate * target_sum / np.sum(best_candidate)
        result = opt.minimize(
            objective,
            best_candidate,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        # Update best solution
        if current_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - current_c5) / temperature):
            best_h = current_h
            best_c5 = current_c5
            print(f"Improved bound: {best_c5}")
        
        temperature *= cooling_schedule
    
    return best_h, best_c5, n_points
