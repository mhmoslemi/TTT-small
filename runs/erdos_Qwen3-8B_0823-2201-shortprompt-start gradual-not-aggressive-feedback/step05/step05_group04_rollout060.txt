import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Start from initial solution
    best_h = initial_h.copy()
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Refinement with local search
    def refine_solution(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x

    # Global exploration with random perturbations
    def explore_solution(h):
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        perturbed_h = h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        return perturbed_h

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Refine current solution
        refined_h = refine_solution(best_h)
        refined_h = np.clip(refined_h, 0, 1)
        refined_h = refined_h * target_sum / np.sum(refined_h)
        refined_c5 = objective(refined_h)
        
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
        
        # Explore new solutions
        explored_h = explore_solution(best_h)
        explored_h = np.clip(explored_h, 0, 1)
        explored_h = explored_h * target_sum / np.sum(explored_h)
        explored_c5 = objective(explored_h)
        
        if explored_c5 < best_c5:
            best_h = explored_h
            best_c5 = explored_c5

    return best_h, best_c5, n_points
