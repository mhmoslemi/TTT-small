import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def mutate(h):
        mutation = np.random.normal(0, 0.1, n_points)
        mutated = np.clip(h + mutation, 0, 1)
        mutated = mutated * target_sum / np.sum(mutated)
        return mutated
    
    def crossover(h1, h2):
        mask = np.random.rand(n_points) < 0.5
        child = np.where(mask, h1, h2)
        child = child * target_sum / np.sum(child)
        return child
    
    def genetic_algorithm():
        population = np.random.rand(50, n_points)
        population = np.clip(population, 0, 1)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        best_h = population[0]
        best_c5 = objective(best_h)
        
        for _ in range(200):
            new_population = []
            for _ in range(50):
                a, b = np.random.choice(50, 2, replace=False)
                child = crossover(population[a], population[b])
                child = mutate(child)
                new_population.append(child)
            population = np.array(new_population)
            
            for h in population:
                h_c5 = objective(h)
                if h_c5 < best_c5:
                    best_h = h
                    best_c5 = h_c5
        
        return best_h, best_c5
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x

    best_h, best_c5 = genetic_algorithm()
