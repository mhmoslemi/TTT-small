import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Gradient-based optimization with SLSQP
    def gradient_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Genetic algorithm for exploration
    def genetic_search(population_size=20, generations=20):
        population = np.random.uniform(0, 1, (population_size, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            fitness = np.array([objective(ind) for ind in population])
            elites = population[np.argsort(fitness)[:population_size//2]]
            
            # Selection and crossover
            parents = np.random.choice(population_size, size=population_size, p=fitness/np.sum(fitness))
            parents = population[parents]
            offspring = []
            
            for i in range(population_size//2):
                p1, p2 = np.random.choice(parents, size=2, replace=False)
                child = np.clip(p1 * 0.5 + p2 * 0.5, 0, 1)
                child = child * target_sum / np.sum(child)
                offspring.append(child)
            
            # Mutation
            for i in range(len(offspring)):
                mutation = np.random.normal(0, 0.05, n_points)
                offspring[i] = np.clip(offspring[i] + mutation, 0, 1)
                offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
            
            population = np.vstack((elites, offspring))
            population = population[np.random.choice(len(population), size=population_size, replace=False)]
        
        best_h = population[np.argmin([objective(ind) for ind in population])]
        return best_h
    
    start_time = time.time()
    best_h = gradient_refinement(h)
    best_c5 = objective(best_h)
    
    # Run genetic search within budget
    while time.time() - start_time < budget_s:
        best_h = genetic_search(population_size=20, generations=10)
        best_c5 = objective(best_h)
    
    # Final gradient refinement
    best_h = gradient_refinement(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
