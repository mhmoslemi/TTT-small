import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Initial optimization
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Genetic algorithm for global search
    population_size = 20
    generations = 5
    mutation_rate = 0.1
    elite_size = 2

    def fitness(individual):
        return objective(individual)

    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(len(child)):
            if random.random() < 0.5:
                child[i] = parent2[i]
        return child

    def mutate(individual):
        for i in range(len(individual)):
            if random.random() < mutation_rate:
                individual[i] += np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        return individual

    def select(population, fitnesses):
        weighted_fitnesses = fitnesses * 100000
        total = np.sum(weighted_fitnesses)
        probabilities = weighted_fitnesses / total
        indices = np.random.choice(len(population), size=len(population), p=probabilities)
        return [population[i] for i in indices]

    population = [np.copy(best_h) for _ in range(population_size)]
    for _ in range(generations):
        fitnesses = np.array([fitness(ind) for ind in population])
        population = select(population, fitnesses)
        elites = np.argsort(fitnesses)[:elite_size]
        new_population = [np.copy(population[i]) for i in elites]
        
        while len(new_population) < population_size:
            parent1, parent2 = random.choices(population, k=2)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
    
    best_genetic = min(population, key=fitness)
    best_c5_genetic = fitness(best_genetic)
    
    if best_c5_genetic < best_c5:
        best_h = best_genetic
        best_c5 = best_c5_genetic

    # Local search with perturbations
    while time.time() - start_time < budget_s - 1:
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Optimization
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
