import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    time_remaining = budget_s

    def local_optimize(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x

    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    elite_size = 2
    generations = 100

    # Initialize population
    population = [np.random.rand(n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    population = [np.clip(h, 0, 1) for h in population]

    for generation in range(generations):
        if time.time() - start_time >= budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elites.copy()
        
        # Generate offspring
        for _ in range(population_size - elite_size):
            # Select parents
            parent1 = random.choice(elites)
            parent2 = random.choice(elites)
            
            # Crossover
            child = np.zeros(n_points)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] = np.random.rand()
            
            # Scale and clip
            child = child * target_sum / np.sum(child)
            child = np.clip(child, 0, 1)
            
            new_population.append(child)
        
        population = new_population
        population = [local_optimize(h) for h in population]
        
        # Update best solution
        for h in population:
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                print(f"Improved bound: {best_c5}")
    
    return best_h, best_c5, n_points
