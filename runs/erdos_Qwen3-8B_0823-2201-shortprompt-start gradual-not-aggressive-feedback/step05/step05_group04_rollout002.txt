import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Start from initial solution
    best_h = initial_h.copy()
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm parameters
    pop_size = 20
    mutation_rate = 0.1
    elite_fraction = 0.2

    def evolve_population(population):
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        # Select elites
        elite_idx = np.argsort(fitness)[:int(elite_fraction * len(population))]
        elites = [population[i] for i in elite_idx]
        # Create new population
        new_population = elites.copy()
        while len(new_population) < len(population):
            # Select parents
            parent1 = population[np.random.choice(len(population))]
            parent2 = population[np.random.choice(len(population))]
            # Crossover
            child = (parent1 + parent2) / 2
            # Mutate
            if np.random.random() < mutation_rate:
                perturb = np.random.normal(0, 0.05, n_points)
                child += perturb
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
            new_population.append(child)
        return new_population

    start_time = time.time()
    population = [best_h.copy() * np.random.uniform(0.8, 1.2) for _ in range(pop_size)]
    population = [ind * target_sum / np.sum(ind) for ind in population]
    
    while time.time() - start_time < budget_s:
        population = evolve_population(population)
        # Local search on elites
        for ind in population:
            result = opt.minimize(
                objective,
                ind,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            ind[:] = result.x
            ind = np.clip(ind, 0, 1)
            ind = ind * target_sum / np.sum(ind)
        
        # Update best solution
        fitness = np.array([objective(ind) for ind in population])
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
    
    return best_h, best_c5, n_points
