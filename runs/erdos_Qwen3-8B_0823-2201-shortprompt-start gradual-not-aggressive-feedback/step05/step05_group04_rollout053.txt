import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Start from initial solution
    best_h = initial_h.copy()
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm parameters
    pop_size = 20
    mutation_rate = 0.1
    elite_size = 2

    def genetic_algorithm():
        population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        for _ in range(100):
            # Evaluate fitness
            fitness = [objective(h) for h in population]
            # Select elites
            elite_indices = np.argsort(fitness)[:elite_size]
            elites = [population[i] for i in elite_indices]
            # Crossover and mutation
            new_population = elites
            for _ in range(pop_size - elite_size):
                parent1, parent2 = np.random.choice(elite_indices, 2, replace=False)
                child = (population[parent1] + population[parent2]) / 2
                if np.random.rand() < mutation_rate:
                    perturb = np.random.normal(0, 0.05, n_points)
                    child = np.clip(child + perturb, 0, 1)
                    child = child * target_sum / np.sum(child)
                new_population.append(child)
            population = new_population
            # Update best solution
            for h in population:
                h = h * target_sum / np.sum(h)
                current_c5 = objective(h)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
        return best_h, best_c5

    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm()

    # Local refinement with SLSQP
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Perturb and optimize
        perturb = np.random.normal(0, 0.02, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    return best_h, best_c5, n_points
