import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial configurations
    initial_h1 = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 3:
            initial_h1[i] = 0.0
        elif i < 2 * n_points // 3:
            initial_h1[i] = 1.0
        else:
            initial_h1[i] = 0.0
    
    initial_h2 = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 4:
            initial_h2[i] = 1.0
        elif i < 3 * n_points // 4:
            initial_h2[i] = 0.5
        else:
            initial_h2[i] = 1.0
    
    initial_h3 = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 5:
            initial_h3[i] = 0.0
        elif i < 2 * n_points // 5:
            initial_h3[i] = 1.0
        elif i < 3 * n_points // 5:
            initial_h3[i] = 0.0
        elif i < 4 * n_points // 5:
            initial_h3[i] = 1.0
        else:
            initial_h3[i] = 0.0
    
    initial_h_values = [initial_h1, initial_h2, initial_h3]
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = initial_h1.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for init_h in initial_h_values:
        # Initial optimization with L-BFGS-B
        result = opt.minimize(
            objective,
            init_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    # Global search with random perturbations and SLSQP
    while time.time() - start_time < budget_s:
        # Random perturbation with adaptive scaling
        perturb_scale = 0.15 * (1 - (time.time() - start_time) / budget_s)
        perturb = np.random.uniform(-perturb_scale, perturb_scale, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Optimization
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
            print(f"Improved bound: {best_c5}")
    
    return best_h, best_c5, n_points
