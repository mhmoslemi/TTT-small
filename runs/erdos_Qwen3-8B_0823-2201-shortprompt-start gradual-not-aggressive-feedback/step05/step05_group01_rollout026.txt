def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    # Initial solution
    best_h = initial_h.copy()
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    generations = 100
    elite_fraction = 0.2
    cooling_rate = 0.995
    temp = 1.0
    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Generate new population
        population = [best_h + np.random.randn(n_points) * 0.1 for _ in range(population_size)]
        population.append(best_h)  # Add current best as elite
        
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        
        # Select elite
        elite_idx = np.argsort(fitness)[:int(elite_fraction * population_size)]
        elite = [population[i] for i in elite_idx]
        
        # Apply mutation
        mutated = []
        for h in elite:
            mutated_h = h.copy()
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    mutated_h[i] += np.random.uniform(-0.1, 0.1)
                    mutated_h[i] = np.clip(mutated_h[i], 0, 1)
            mutated.append(mutated_h)
        
        # Evaluate mutated population
        mutated_fitness = [objective(h) for h in mutated]
        
        # Select new population
        new_population = []
        for i in range(len(mutated)):
            if mutated_fitness[i] < best_c5 or np.random.rand() < np.exp((best_c5 - mutated_fitness[i]) / temp):

        # Update temperature
        temp *= cooling_rate
        
        # Ensure population size
        while len(new_population) < population_size:
            new_population.append(np.random.rand(n_points))
        
        population = new_population

    return best_h, best_c5, n_points
