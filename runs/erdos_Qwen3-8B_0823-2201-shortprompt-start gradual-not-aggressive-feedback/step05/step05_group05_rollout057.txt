import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Start with a random initial guess
    h = np.random.uniform(0, 1, n_points)
    h = h * target_sum / np.sum(h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # First phase: global search with random perturbations
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break

        perturbed_h = h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Local optimization
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        h = result.x
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = objective(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    # Second phase: refine the best solution
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        # Greedy improvement: adjust individual points
        for i in range(n_points):
            h = best_h.copy()
            h[i] += 0.05
            h[i] = np.clip(h[i], 0, 1)
            h = h * target_sum / np.sum(h)

    return best_h, best_c5, n_points
