import scipy.optimize as opt
import time

    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    def genetic_algorithm():
        population = np.random.rand(20, n_points)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        best_h = population.copy()
        best_c5 = np.inf
        
        for _ in range(100):
            fitness = np.array([objective(h) for h in population])
            best_idx = np.argmin(fitness)
            best_c5 = fitness[best_idx]
            best_h = population[best_idx]
            
            # Selection
            selected = population[np.argsort(fitness)[:10]]
            
            # Crossover
            offspring = []
            for _ in range(10):
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                offspring.append(child)
            
            # Mutation
            for i in range(len(offspring)):
                offspring[i] = np.clip(
                    offspring[i] + np.random.uniform(-0.1, 0.1, n_points), 0, 1
