import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial population based on given construction
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 3:
            initial_h[i] = 0.0
        elif i < 2 * n_points // 3:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize population
    population = [initial_h + np.random.uniform(-0.1, 0.1, n_points) for _ in range(10)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    best_h = initial_h
    best_c5 = objective(initial_h)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_c5 = fitness[best_idx]
        
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
            print(f"Improved bound: {best_c5}")
        
        # Genetic operations
        new_population = []
        for h in population:
            # Mutation
            mutated = h + np.random.uniform(-0.05, 0.05, n_points)
            mutated = np.clip(mutated, 0, 1)
            mutated = mutated * target_sum / np.sum(mutated)
            new_population.append(mutated)
            
            # Crossover with best
            crossed = (h + current_best) / 2
            crossed = np.clip(crossed, 0, 1)
            crossed = crossed * target_sum / np.sum(crossed)
            new_population.append(crossed)
        
        # Refinement with SLSQP
        for h in new_population:
            result = opt.minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = refined_h * target_sum / np.sum(refined_h)
            new_population.append(refined_h)
        
        # Keep top performers
        population = np.array(population)
        new_population = np.array(new_population)
        combined = np.vstack((population, new_population))
        combined = np.vstack((combined, best_h[np.newaxis, :]))
        combined = np.unique(combined, axis=0)
        population = combined[:100]  # Keep top 100 individuals
    
    return best_h, best_c5, n_points
