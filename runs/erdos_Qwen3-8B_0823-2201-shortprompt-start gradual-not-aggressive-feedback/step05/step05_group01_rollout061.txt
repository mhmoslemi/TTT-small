import numpy as np

import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def normalize(h):
        h = h.reshape(n_points)
        return h * target_sum / np.sum(h)

    def objective(h):
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    bounds = [(0, 1) for _ in range(n_points)]

    population_size = 50
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [normalize(p) for p in population]

    def fitness(individual):
        return objective(individual)

    def crossover(parent1, parent2):
        child = np.random.rand(n_points)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return normalize(child)

    def mutate(individual, mutation_rate=0.1):
        mutated = individual.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                mutated[i] += np.random.uniform(-0.1, 0.1)
                mutated[i] = np.clip(mutated[i], 0, 1)
        return normalize(mutated)

    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitnesses = [fitness(ind) for ind in population]
        # Select parents
        parents = np.argsort(fitnesses)[:population_size//2]
        parents = [population[i] for i in parents]
        # Create new population
        new_population = []
        for _ in range(population_size):
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        population = new_population
        # Update best solution
        for ind in population:
            current_c5 = fitness(ind)
            if current_c5 < best_c5:
                best_h = ind.copy()
                best_c5 = current_c5
                print(f"Improved bound: {best_c5}")
    
    return best_h, best_c5, n_points
