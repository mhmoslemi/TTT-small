import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Genetic algorithm for global search
    population_size = 20
    generations = 50
    mutation_rate = 0.1

    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        fitness = [objective(h) for h in population]
        
        # Select best individuals
        elite_idx = np.argsort(fitness)[:3]
        elite = [population[i] for i in elite_idx]
        
        # Generate new population
        new_population = elite.copy()
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(elite, 2, replace=False)
            child = np.clip(parent1 * 0.5 + parent2 * 0.5 + np.random.normal(0, mutation_rate, n_points), 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        population = new_population
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]

    # Local refinement with SLSQP
    while time.time() - start_time < budget_s - 1:
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
