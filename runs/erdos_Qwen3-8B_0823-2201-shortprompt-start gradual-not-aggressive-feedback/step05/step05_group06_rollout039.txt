import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    elite_size = 2

    def evolve_population(population):
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        # Create new population
        new_population = elites.copy()
        while len(new_population) < population_size:
            # Select parents
            parent1 = random.choice(population)
            parent2 = random.choice(population)
            # Crossover
            child = np.copy(parent1)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = parent2[i]
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] += np.random.uniform(-0.1, 0.1)
                    child[i] = np.clip(child[i], 0, 1)
            # Normalize
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        return new_population

    # Initial population
    population = [np.copy(initial_h) for _ in range(population_size)]
    for _ in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        population = evolve_population(population)
        current_best = min(population, key=objective)
        current_c5 = objective(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5

    # Local refinement with SLSQP
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
