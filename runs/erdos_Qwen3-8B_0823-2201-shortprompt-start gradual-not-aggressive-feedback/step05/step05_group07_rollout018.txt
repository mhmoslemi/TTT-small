import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    bounds = [(0, 1) for _ in range(n_points)]

    def enforce_integral(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        return h * target_sum / np.sum(h)

    def evaluate(solution):
        return objective(solution)

    def crossover(parent1, parent2):
        child = np.copy(parent1)
        indices = np.random.choice(n_points, size=n_points // 2, replace=False)
        child[indices] = parent2[indices]
        return enforce_integral(child)

    def mutate(solution):
        mutated = np.copy(solution)
        for i in range(n_points):
            if np.random.rand() < 0.1:
                mutated[i] += np.random.uniform(-0.1, 0.1)
                mutated[i] = np.clip(mutated[i], 0, 1)
        return enforce_integral(mutated)

    population_size = 50
    population = [enforce_integral(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    time_remaining = budget_s

    for generation in range(200):
        if time.time() - start_time >= budget_s:
            break
        fitness = [evaluate(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Select parents
        parents = np.argsort(fitness)[:population_size // 2]
        population = [population[i] for i in parents]
        
        # Crossover and mutation
        while len(population) < population_size:
            parent1, parent2 = np.random.choice(parents, size=2, replace=False)
            child = crossover(population[parent1], population[parent2])
            if np.random.rand() < 0.2:
                child = mutate(child)
            population.append(child)
        
        # Local optimization
        for i in range(len(population)):
            if np.random.rand() < 0.3:
                result = opt.minimize(
                    objective,
                    population[i],
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                    options={'maxiter': 50, 'ftol': 1e-8}
                )
                population[i] = result.x

    return best_h, best_c5, n_points
