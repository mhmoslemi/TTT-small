import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial solution
    best_h = initial_h.copy()
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm parameters
    pop_size = 20
    generations = 100
    mutation_rate = 0.1
    elite_ratio = 0.2

    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Generate population
        population = np.random.uniform(0, 1, (pop_size, n_points))
        population = np.clip(population, 0, 1)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        
        # Select elites
        elite_idx = np.argsort(fitness)[:int(elite_ratio * pop_size)]
        elites = population[elite_idx]
        
        # Create new population
        new_population = elites.copy()
        for _ in range(pop_size - len(elites)):
            # Select parents
            parent1 = population[np.random.choice(pop_size)]
            parent2 = population[np.random.choice(pop_size)]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            mutation_mask = np.random.rand(n_points) < mutation_rate
            child[mutation_mask] = np.random.uniform(0, 1, sum(mutation_mask))
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            
            new_population = np.vstack([new_population, child])
        
        population = new_population
        population = np.clip(population, 0, 1)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        # Update best solution
        current_fitness = np.array([objective(ind) for ind in population])
        best_idx = np.argmin(current_fitness)
        current_h = population[best_idx]
        current_c5 = current_fitness[best_idx]
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
            print(f"Improved bound: {best_c5}")
    
    # Final local optimization
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
