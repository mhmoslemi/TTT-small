import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Start from initial solution
    best_h = initial_h.copy()
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm with structured mutation
    for _ in range(50):
        # Create offspring with structured mutation
        offspring = best_h.copy()
        for i in range(n_points):
            if np.random.rand() < 0.1:
                # Random mutation
                offspring[i] += np.random.normal(0, 0.05)
            elif np.random.rand() < 0.2:
                # Swap mutation
                j = np.random.randint(0, n_points)
                offspring[i], offspring[j] = offspring[j], offspring[i]
        
        offspring = np.clip(offspring, 0, 1)
        offspring = offspring * target_sum / np.sum(offspring)
        
        # Local search with SLSQP
        result = opt.minimize(
            objective,
            offspring,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x

        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)

            best_h = current_h
            best_c5 = current_c5

    # Final refinement with small perturbations
    start_time = time.time()
