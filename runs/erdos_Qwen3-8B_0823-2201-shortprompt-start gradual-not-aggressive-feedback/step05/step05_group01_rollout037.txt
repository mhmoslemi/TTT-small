    start_time = time.time()

        # Evaluate fitness
        fitness = np.array([objective(h) for h in population])

        tournament_size = 5
        parents = []

            contestants = np.random.choice(range(population_size), tournament_size, replace=False)

        offspring = []

            parent1 = parents[i]
            parent2 = parents[population_size - 1 - i]

                else:
                    child1[j] = parent2[j]
                    child2[j] = parent1[j]
            offspring.append(child1)

        # Mutation

        for i in range(population_size):
            if np.random.rand() < mutation_rate:
                mutation = np.random.uniform(-0.1, 0.1, n_points)
                offspring[i] += mutation

        offspring = [h * target_sum / np.sum(h) for h in offspring]
