import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Better initial construction with smooth transitions
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 3:
            initial_h[i] = 0.0
        elif i < 2 * n_points // 3:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Hybrid optimization with SLSQP and differential evolution
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Local refinement with SLSQP
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
            print(f"Improved bound: {best_c5}")

                h = x.reshape(n_points)

                corr = np.correlate(h, 1 - h, mode="full")
                c5 = np.max(corr) * dx
                return c5

                de_objective,

            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                print(f"Improved bound: {best_c5}")
    
    return best_h, best_c5, n_points
