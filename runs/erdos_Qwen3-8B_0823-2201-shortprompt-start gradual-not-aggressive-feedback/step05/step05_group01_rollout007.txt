        fitness = np.array([objective(ind) for ind in population])

        elite_idx = np.argsort(fitness)

        tournament_size = 5

            contestants = np.random.choice(population_size, tournament_size, replace=False)
            winner_idx = contestants[np.argmin(fitness[contestants])]
            parents.append(population[winner_idx])
        
        # Crossover
        offspring = []

                parent1 = parents[i]
                parent2 = parents[i+1]
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
                child2 = np.concatenate((parent2[:crossover_point], parent1[crossover_point:]))
                offspring.extend([child1, child2])
            else:
                offspring.append(parents[i])
        
        # Mutation
        mutation_rate = 0.1
        for i in range(len(offspring)):
            for j in range(n_points):
                if np.random.rand() < mutation_rate:

        population = np.array(elite_pop + offspring)

            current_c5 = objective(ind)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = ind.copy()
