        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    def generate_population(size):
        pop = np.random.rand(size, n_points)

        pop = pop * target_sum / np.sum(pop, axis=1, keepdims=True)
        return pop
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        idx = np.random.choice(n_points, size=n_points//2, replace=False)
        child[idx] = parent2[idx]
        child = child * target_sum / np.sum(child)
        return child

        mutation_rate = 0.1
        mutation_strength = 0.1

                individual[i] += np.random.uniform(-mutation_strength, mutation_strength)
                individual[i] = np.clip(individual[i], 0, 1)
        individual = individual * target_sum / np.sum(individual)
        return individual

            h,
            method='SLSQP',
            bounds=bounds,
