        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]

        # Create new population
        new_population = []
        for _ in range(50):
            # Selection
            parents = random.sample(range(50), 2)

            # Crossover
            crossover_point = random.randint(1, n_points - 1)

            # Normalize

            new_population.append(child)
        
        population = new_population
