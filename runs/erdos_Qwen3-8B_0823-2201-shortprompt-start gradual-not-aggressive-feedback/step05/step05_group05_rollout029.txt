import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial construction with smooth transitions
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 3:
            initial_h[i] = 0.0
        elif i < 2 * n_points // 3:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm for global search
    start_time = time.time()
    population_size = 20
    generations = 50
    mutation_rate = 0.1

    def create_individual():
        individual = np.random.uniform(0, 1, n_points)
        individual = individual * target_sum / np.sum(individual)
        return individual

    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parent2[i]
        child = child * target_sum / np.sum(child)
        return child

    def mutate(individual):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] += np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        individual = individual * target_sum / np.sum(individual)
        return individual

    population = [create_individual() for _ in range(population_size)]
    population = [np.clip(p, 0, 1) * target_sum / np.sum(p) for p in population]

    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        best_index = np.argmin(fitness)
        best_c5 = fitness[best_index]
        
        # Replace worst individual with new one
        worst_index = np.argmax(fitness)
        new_individual = create_individual()
        population[worst_index] = new_individual
        
        # Apply crossover and mutation
        for _ in range(population_size // 2):
            parent1, parent2 = random.sample(population, 2)
            child1 = crossover(parent1, parent2)
            child2 = crossover(parent2, parent1)
            population.append(mutate(child1))
            population.append(mutate(child2))
        
        # Keep best individuals
        population = np.array(population)
        population = np.sort(population, axis=0)
        population = population[:population_size]
        population = [np.clip(p, 0, 1) * target_sum / np.sum(p) for p in population]
        
        if best_c5 < best_c5:
            best_h = population[best_index]
            best_c5 = best_c5
    
    return best_h, best_c5, n_points
