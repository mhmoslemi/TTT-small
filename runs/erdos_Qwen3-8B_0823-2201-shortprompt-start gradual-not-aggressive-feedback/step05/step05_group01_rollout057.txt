import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial population
    population_size = 50
    population = np.random.rand(population_size, n_points)
    population = np.clip(population, 0, 1)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)

    def evaluate(pop):
        return np.array([objective(ind) for ind in pop])

    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = evaluate(population)
        # Select top 20%
        idxs = np.argsort(fitness)[:int(0.2 * population_size)]
        elite = population[idxs]
        
        # Generate offspring
        offspring = []
        for _ in range(int(0.8 * population_size)):
            # Crossover
            parent1 = np.random.choice(elite)
            parent2 = np.random.choice(elite)
            child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            
            # Mutation
            mutation_rate = 0.1
            mutation_scale = 0.2
            child = np.clip(child + np.random.normal(0, mutation_scale, n_points) * mutation_rate, 0, 1)
            child = child * target_sum / np.sum(child)
            
            offspring.append(child)
        
        # Combine and select
        population = np.vstack((elite, offspring))
        population = np.random.choice(population, size=population_size, replace=False)
        
        # Local optimization
        for i in range(population_size):
            h = population[i].copy()
            result = opt.minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 20, 'ftol': 1e-8}
            )
            population[i] = result.x
            population[i] = np.clip(population[i], 0, 1)
            population[i] = population[i] * target_sum / np.sum(population[i])
        
        # Update best
        current_fitness = evaluate(population)
        best_idx = np.argmin(current_fitness)
        if current_fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = current_fitness[best_idx]
            print(f"Improved bound: {best_c5}")
    
    return best_h, best_c5, n_points
