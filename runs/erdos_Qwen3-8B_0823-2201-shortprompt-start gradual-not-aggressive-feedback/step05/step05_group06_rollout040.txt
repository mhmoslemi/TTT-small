import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Initial optimization with SLSQP
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Genetic algorithm for exploration
    population_size = 20
    mutation_rate = 0.1
    generations = 50

    population = [best_h.copy() for _ in range(population_size)]
    for _ in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        # Select parents
        parents = np.array(population)[np.argsort(fitness)[:population_size//2]]
        # Crossover
        offspring = []
        for _ in range(population_size//2):
            if np.random.rand() < 0.8:
                parent1 = np.random.choice(parents)
                parent2 = np.random.choice(parents)
                child = (parent1 + parent2) / 2
            else:
                child = np.random.uniform(0, 1, n_points)
            # Mutate
            child = np.clip(child + np.random.normal(0, mutation_rate, n_points), 0, 1)
            child = child * target_sum / np.sum(child)
            offspring.append(child)
        # Replace population
        population = parents.tolist() + offspring
        # Update best solution
        for ind in population:
            current_c5 = objective(ind)
            if current_c5 < best_c5:
                best_h = ind.copy()
                best_c5 = current_c5

    return best_h, best_c5, n_points
