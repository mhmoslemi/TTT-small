import numpy as np
import scipy.optimize as opt
import time

    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    pop_size = 50
    generations = 100
    elite_size = 5
    mutation_rate = 0.1
    tournament_size = 5
    
    def crossover(parent1, parent2):
        child = parent1.copy()
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parent2[i]
        return child
    
    def mutate(individual):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] += np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        return individual
    
    # Genetic algorithm
    population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    for _ in range(generations):
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        
        # Tournament selection
        new_population = elites.copy()
        while len(new_population) < pop_size:
            contestants = np.random.choice(population, tournament_size)
            winner = min(contestants, key=objective)
            new_population.append(winner)
        
        # Apply crossover and mutation
        for i in range(len(new_population) - elite_size):
            parent1 = np.random.choice(new_population)
            parent2 = np.random.choice(new_population)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population[i + elite_size] = child
        
        population = new_population
    
    # Local refinement with SLSQP
    best_h = min(population, key=objective)
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    best_h = result.x
    
    # Final evaluation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
