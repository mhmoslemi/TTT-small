import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 50

    # Initialize population
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)

    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate population
        fitness = np.array([objective(ind) for ind in population])
        # Select parents
        parent_indices = np.argsort(fitness)[:population_size//2]
        parents = population[parent_indices]
        # Create offspring
        offspring = []
        for _ in range(population_size//2):
            if np.random.rand() < crossover_rate:
                # Single-point crossover
                idx = np.random.randint(1, n_points-1)
                child = np.concatenate((parents[0][:idx], parents[1][idx:]))
            else:
                # Random mutation
                child = np.random.uniform(0, 1, n_points)
            # Apply mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] = np.random.uniform(0, 1)
            child = child * target_sum / np.sum(child)
            offspring.append(child)
        # Create new population
        population = np.vstack((parents, offspring))
        # Keep best solution
        if np.min(fitness) < best_c5:
            best_h = population[np.argmin(fitness)]
            best_c5 = np.min(fitness)
    
    return best_h, best_c5, n_points
