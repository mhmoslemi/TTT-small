import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def normalize(h):
        h = np.clip(h, 0, 1)
        return h * target_sum / np.sum(h)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def generate_population(size):
        pop = np.random.rand(size, n_points)
        pop = normalize(pop)
        return pop
    
    def crossover(parent1, parent2):
        alpha = np.random.rand(n_points)
        return alpha * parent1 + (1 - alpha) * parent2
    
    def mutate(h):
        mutation = np.random.normal(0, 0.05, n_points)
        mutated = h + mutation
        return normalize(mutated)
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    population = generate_population(50)
    best_h = population[0]
    best_c5 = objective(best_h)
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for _ in range(len(population)):
            parent1, parent2 = np.random.choice(population, 2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        population = new_population
        for h in population:
            h = normalize(h)
            c5 = objective(h)
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
