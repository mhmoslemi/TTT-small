import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial population
    population = np.random.uniform(0, 1, (50, n_points))
    population = [p * target_sum / np.sum(p) for p in population]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    # Genetic algorithm parameters
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = int(budget_s / 0.5)
    
    for _ in range(generations):
        # Evaluate fitness
        fitness = np.array([objective(p) for p in population])
        
        # Select parents
        parents = population[np.argsort(fitness)[:20]]
        
        # Create new population
        new_population = []
        for _ in range(50):
            # Crossover
            if np.random.rand() < crossover_rate:
                p1, p2 = np.random.choice(parents, 2, replace=False)
                child = np.concatenate([p1[:n_points//2], p2[n_points//2:]])
            else:
                child = np.random.uniform(0, 1, n_points)
            
            # Mutation
            child = np.clip(child + np.random.normal(0, mutation_rate, n_points), 0, 1)
            child = child * target_sum / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        
        # Local refinement
        for i in range(20):
            p = population[i]
            result = opt.minimize(
                objective,
                p,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            population[i] = result.x
        
        # Update best solution
        current_best = min(population, key=objective)
        current_c5 = objective(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
            print(f"Improved bound: {best_c5}")
        
        if time.time() - start_time > budget_s:
            break
            
    return best_h, best_c5, n_points
