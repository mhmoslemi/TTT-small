import random

        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)

    )
    best_h = result.x

    population_size = 20
    mutation_rate = 0.1
    generations = 10
    
    population = [best_h.copy() for _ in range(population_size)]

        new_population = []
