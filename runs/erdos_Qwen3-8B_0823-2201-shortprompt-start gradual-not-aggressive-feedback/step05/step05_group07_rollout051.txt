import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    time_remaining = budget_s

    # Genetic algorithm parameters
    population_size = 20
    generations = 10
    mutation_rate = 0.1

    # Initial population
    population = [np.random.rand(n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    population = [np.clip(h, 0, 1) for h in population]

    for gen in range(generations):
        if time.time() - start_time >= budget_s:
            break
        # Evaluate population
        fitness = [objective(h) for h in population]
        # Select top performers
        top_indices = np.argsort(fitness)[:population_size//2]
        top_population = [population[i] for i in top_indices]
        # Generate new population
        new_population = []
        for _ in range(population_size//2):
            # Tournament selection
            a, b = random.sample(top_indices, 2)
            parent1 = population[a]
            parent2 = population[b]
            # Crossover
            crossover_point = np.random.randint(1, n_points-1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] = np.random.rand()
            child = child * target_sum / np.sum(child)
            child = np.clip(child, 0, 1)
            new_population.append(child)
        population = top_population + new_population
        # Update best solution
        for h in population:
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                print(f"Improved bound: {best_c5}")

    # Final gradient-based refinement
    result = opt.minimize(
        objective,
        best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
