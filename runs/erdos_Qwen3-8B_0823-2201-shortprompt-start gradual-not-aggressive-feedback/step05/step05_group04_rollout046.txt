import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    best_h = initial_h.copy()

    population_size = 50
    generations = 100
    mutation_rate = 0.1
    tournament_size = 5

    # Evaluate initial population
    def evaluate_individual(h):
        return objective(h)

    # Create initial population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    population_fitness = [evaluate_individual(h) for h in population]

    best_genome = population[np.argmin(population_fitness)]
    best_fitness = np.min(population_fitness)

    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Tournament selection
        tournament = np.random.choice(range(population_size), size=tournament_size, replace=False)
        selected = tournament[np.argmin([population_fitness[i] for i in tournament])]
        
        # Mutation
        mutated = population[selected].copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                mutated[i] += np.random.uniform(-0.05, 0.05)
                mutated[i] = np.clip(mutated[i], 0, 1)
                mutated = mutated * target_sum / np.sum(mutated)
        
        # Local optimization
        result = opt.minimize(
            objective,
            mutated,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},

        )

        # Evaluate mutated
        mutated_fitness = evaluate_individual(mutated)
        
        # Replace worst individual
        if mutated_fitness < np.max(population_fitness):
            worst_idx = np.argmax(population_fitness)
            population[worst_idx] = mutated
            population_fitness[worst_idx] = mutated_fitness
        
        # Update best
        if mutated_fitness < best_fitness:
            best_fitness = mutated_fitness
            best_genome = mutated

    # Final refinement
    while time.time() - start_time < budget_s:
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        perturbed_h = best_genome + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
