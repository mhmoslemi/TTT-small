import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    generations = 50
    elite_fraction = 0.2

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Generate population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        
        # Evaluate population
        fitness = [objective(h) for h in population]
        
        # Select elites
        elite_idx = np.argsort(fitness)[:int(elite_fraction * population_size)]
        elites = [population[i] for i in elite_idx]
        
        # Create new population
        new_population = elites.copy()
        while len(new_population) < population_size:
            # Select parents
            parent1 = random.choice(elites)
            parent2 = random.choice(elites)
            
            # Crossover
            crossover_point = random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutate
            mutation = np.random.normal(0, mutation_rate, n_points)
            child += mutation
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        print(f"Improved bound: {best_c5}")
    
    return best_h, best_c5, n_points
