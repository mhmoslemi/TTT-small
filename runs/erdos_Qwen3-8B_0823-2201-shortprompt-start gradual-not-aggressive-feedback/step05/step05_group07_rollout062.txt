import numpy as np
import scipy.optimize as opt
import time
import copy

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    time_remaining = budget_s

    # Initial optimization with SLSQP
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Genetic algorithm parameters
    population_size = 10
    generations = 20
    mutation_rate = 0.1
    crossover_rate = 0.8

    # Initialize population
    population = [best_h.copy() for _ in range(population_size)]
    
    for gen in range(generations):
        if time.time() - start_time >= budget_s:
            break
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        # Select best individuals
        selected = [population[i] for i in np.argsort(fitness)]
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            # Select parents
            parent1 = selected[np.random.randint(len(selected))]
            parent2 = selected[np.random.randint(len(selected))]
            # Crossover
            if np.random.rand() < crossover_rate:
                child = np.array([parent1[i] if np.random.rand() < 0.5 else parent2[i] for i in range(n_points)])
            else:
                child = parent1.copy()
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.uniform(-0.1, 0.1)
                    child[i] = np.clip(child[i], 0, 1)
            # Normalize
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = new_population
        # Update best solution
        current_best = min(population, key=objective)
        current_c5 = objective(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
            print(f"Improved bound: {best_c5}")
    
    return best_h, best_c5, n_points
