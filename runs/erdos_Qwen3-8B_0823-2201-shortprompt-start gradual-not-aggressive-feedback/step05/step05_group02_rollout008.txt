import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Global search with random perturbations
    def global_search(h):
        best_h = h.copy()
        best_c5 = objective(best_h)
        for _ in range(100):
            perturb = np.random.uniform(-0.1, 0.1, n_points)
            perturbed = np.clip(h + perturb, 0, 1)

            current_c5 = objective(perturbed)
            if current_c5 < best_c5:

    start_time = time.time()
    best_h = local_refinement(h)
    best_c5 = objective(best_h)
    
    # Iterative refinement
    while time.time() - start_time < budget_s:
        best_h = global_search(best_h)
        best_h = local_refinement(best_h)
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
