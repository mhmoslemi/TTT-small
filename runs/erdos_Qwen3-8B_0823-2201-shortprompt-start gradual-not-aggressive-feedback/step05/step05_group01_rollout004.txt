import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def genetic_algorithm():
        population = np.random.uniform(0, 1, (50, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        best_h = initial_h.copy()
        best_c5 = objective(best_h)
        
        for _ in range(100):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            
            # Select parents
            parents = population[np.argsort(fitness)[:20]]
            
            # Crossover
            offspring = []
            for _ in range(20):
                parent1, parent2 = np.random.choice(parents, 2, replace=False)
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                offspring.append(child)
            
            # Mutate
            for i in range(len(offspring)):
                offspring[i] = np.clip(offspring[i] + np.random.normal(0, 0.05, n_points), 0, 1)
                offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
            
            # Replace population
            population = np.vstack((parents, offspring))
            population = population[np.random.permutation(len(population))]
            
            # Update best solution
            for ind in population:
                c5 = objective(ind)
                if c5 < best_c5:
                    best_h = ind.copy()
                    best_c5 = c5
        
        return best_h, best_c5

    # Initial genetic algorithm run
    best_h, best_c5 = genetic_algorithm()
    
    # Local optimization
    start_time = time.time()
    temp = 1.0
    cooling_rate = 0.995
    while time.time() - start_time < budget_s:
        # Simulated annealing step
        perturb_scale = 0.2 * (1 - (time.time() - start_time) / budget_s)
        perturb = np.random.uniform(-perturb_scale, perturb_scale, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Optimization
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},

        current_c5 = objective(current_h)
        
        # Acceptance criterion
        if current_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - current_c5) / temp):
            best_h = current_h
            best_c5 = current_c5
            print(f"Improved bound: {best_c5}")
        
        temp *= cooling_rate
    
    return best_h, best_c5, n_points
