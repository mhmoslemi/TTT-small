import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Start from initial solution
    best_h = initial_h.copy()
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm parameters
    population_size = 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 50

    def create_initial_population():
        population = []
        for _ in range(population_size):
            h = best_h.copy()
            perturb = np.random.normal(0, 0.05, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            population.append(h)
        return population

    def evaluate(individual):
        return objective(individual)

    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(n_points):
            if np.random.rand() < crossover_rate:
                child[i] = parent2[i]
        child = np.clip(child, 0, 1)
        child = child * target_sum / np.sum(child)
        return child

    def mutate(individual):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] += np.random.normal(0, 0.05)
                individual[i] = np.clip(individual[i], 0, 1)
        individual = individual * target_sum / np.sum(individual)
        return individual

    # Genetic algorithm
    population = create_initial_population()
    for _ in range(generations):
        # Evaluate fitness
        fitness = [evaluate(ind) for ind in population]
        # Select parents
        parents = np.argsort(fitness)[:population_size // 2]
        # Create new population
        new_population = []
        for i in range(population_size // 2):
            parent1 = population[parents[i]]
            parent2 = population[parents[np.random.randint(population_size // 2)]]
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        population = new_population

    # Final refinement with local search
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Select the best individual from the population
        best_candidate = min(population, key=evaluate)
        perturbed_h = best_candidate.copy()
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        perturbed_h += perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    return best_h, best_c5, n_points
