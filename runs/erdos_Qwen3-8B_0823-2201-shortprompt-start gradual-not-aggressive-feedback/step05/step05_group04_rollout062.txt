import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Start from initial solution
    best_h = initial_h.copy()
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Mutation function with structured perturbation
    def mutation(h):
        perturb = np.zeros_like(h)
        indices = np.random.choice(n_points, size=5, replace=False)
        for i in indices:
            perturb[i] = np.random.uniform(-0.1, 0.1)
        perturbed = h + perturb
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        return perturbed

    # Genetic algorithm with local search
    for _ in range(50):
        # Mutate
        perturbed_h = mutation(best_h)
        # Local search
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    # Final refinement with focused perturbations
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Focus on high overlap regions
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        max_indices = np.argwhere(corr == np.max(corr)).flatten()

        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)

            objective,
            perturbed_h,
            method='SLSQP',

    return best_h, best_c5, n_points
