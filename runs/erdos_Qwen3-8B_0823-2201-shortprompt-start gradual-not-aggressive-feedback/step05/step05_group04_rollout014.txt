import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Start from initial solution
    best_h = initial_h.copy()
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    def mutate(h):
        perturb = np.random.normal(0, 0.03, n_points)
        perturbed = h + perturb
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        return perturbed

    def crossover(h1, h2):
        child = np.zeros_like(h1)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = h1[i]
            else:
                child[i] = h2[i]
        child = child * target_sum / np.sum(child)
        return child

    # Evolutionary algorithm
    population = [mutate(best_h) for _ in range(5)]
    for _ in range(100):
        # Evaluate population
        pop_evals = [objective(h) for h in population]
        # Select top 3
        top_indices = np.argsort(pop_evals)[:3]
        top_h = [population[i] for i in top_indices]
        # Crossover
        new_h = [crossover(top_h[0], top_h[1]), crossover(top_h[1], top_h[2]), crossover(top_h[2], top_h[0])]
        # Local optimization
        for h in new_h:
            result = opt.minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            h = result.x
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        # Update population
        population += new_h
        population = np.array([h for h in population if np.sum(h) > 0])
        population = population[:5]
    
    # Final refinement
    start_time = time.time()
    while time.time() - start_time < budget_s:
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    return best_h, best_c5, n_points
