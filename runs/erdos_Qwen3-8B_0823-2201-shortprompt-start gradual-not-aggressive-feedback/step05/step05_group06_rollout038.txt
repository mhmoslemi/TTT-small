import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Initial optimization
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Genetic algorithm for global search
    population_size = 20
    generations = 10
    mutation_rate = 0.1

    def mutate(h):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] += np.random.uniform(-0.1, 0.1)
                h[i] = np.clip(h[i], 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def crossover(h1, h2):
        h = h1.copy()
        for i in range(len(h)):
            if random.random() < 0.5:
                h[i] = h2[i]
        h = h * target_sum / np.sum(h)
        return h

    population = [best_h.copy() for _ in range(population_size)]
    for _ in range(generations):
        if time.time() - start_time >= budget_s - 1:
            break
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        new_population = [best_h.copy()]
        while len(new_population) < population_size:
            idx1, idx2 = np.random.choice(range(population_size), 2, replace=False)
            h1, h2 = population[idx1], population[idx2]
            child = crossover(h1, h2)
            child = mutate(child)
            new_population.append(child)
        population = new_population

    return best_h, best_c5, n_points
