import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    target_sum = n_points / 2

        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Start from initial solution
    best_h = initial_h.copy()
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    def mutate(h, mutation_rate=0.1):

        perturbed = h + perturb

        perturbed = perturbed * target_sum / np.sum(perturbed)
        return perturbed

    # Genetic algorithm with local search
    population = [best_h.copy() for _ in range(10)]
    for _ in range(50):
        # Evaluate population
        population_fitness = [objective(h) for h in population]
        # Select best
        best_idx = np.argmin(population_fitness)
        best_h = population[best_idx].copy()
        best_c5 = population_fitness[best_idx]
        # Mutate and evaluate
        for i in range(len(population)):
            if np.random.rand() < 0.2:
                population[i] = mutate(population[i])
            population_fitness[i] = objective(population[i])
        # Replace worst with best
        worst_idx = np.argmax(population_fitness)
        population[worst_idx] = best_h.copy()

    # Final refinement with local search
    start_time = time.time()
    while time.time() - start_time < budget_s:
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
