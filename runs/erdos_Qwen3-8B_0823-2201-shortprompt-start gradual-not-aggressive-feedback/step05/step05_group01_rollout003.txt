import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm with simulated annealing
    start_time = time.time()
    population_size = 20
    generations = 50
    mutation_rate = 0.1
    temp = 1.0
    cooling_rate = 0.995

    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]

    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()

        # Selection and crossover
        selected = np.argsort(fitness)[:population_size//2]
        new_population = []
        for _ in range(population_size//2):
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child = (population[parent1] + population[parent2]) / 2
            new_population.append(child)

        # Mutation
        for i in range(len(new_population)):
            if np.random.rand() < mutation_rate:
                perturb = np.random.uniform(-0.1, 0.1, n_points)
                new_population[i] += perturb
                new_population[i] = np.clip(new_population[i], 0, 1)
                new_population[i] = new_population[i] * target_sum / np.sum(new_population[i])

        # Replace population
        population = new_population + [best_h for _ in range(population_size//2)]

        # Simulated annealing refinement
        for _ in range(10):
            if time.time() - start_time > budget_s:

            current_c5 = objective(perturbed_h)
            if current_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - current_c5) / temp):
                best_h = perturbed_h
                best_c5 = current_c5
            temp *= cooling_rate

    return best_h, best_c5, n_points
