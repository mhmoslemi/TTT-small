import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 50
    elite_size = 2

    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]

    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        
        # Select parents
        parents = []
        for _ in range(population_size - elite_size):
            # Tournament selection
            tournament = np.random.choice(range(len(population)), size=5)
            parent = population[np.argmin([fitness[i] for i in tournament])]
            parents.append(parent)
        
        # Crossover
        offspring = []
        for i in range(len(parents) // 2):
            parent1 = parents[2*i]
            parent2 = parents[2*i + 1]
            if np.random.rand() < crossover_rate:
                # Single-point crossover
                point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:point], parent2[point:]])
                child2 = np.concatenate([parent2[:point], parent1[point:]])
                offspring.append(child1)
                offspring.append(child2)
            else:
                offspring.append(parent1)
                offspring.append(parent2)
        
        # Mutation
        for i in range(len(offspring)):
            h = offspring[i]
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    h[j] += np.random.uniform(-0.1, 0.1)
                    h[j] = np.clip(h[j], 0, 1)
            offspring[i] = h * target_sum / np.sum(h)
        
        # Combine elites and offspring
        population = elites + offspring
        population = [np.clip(h, 0, 1) for h in population]
        population = [h * target_sum / np.sum(h) for h in population]
        
        # Update best solution
        current_fitness = [objective(h) for h in population]
        best_index = np.argmin(current_fitness)
        if current_fitness[best_index] < best_c5:
            best_h = population[best_index]
            best_c5 = current_fitness[best_index]
            print(f"Improved bound: {best_c5}")
    
    return best_h, best_c5, n_points
