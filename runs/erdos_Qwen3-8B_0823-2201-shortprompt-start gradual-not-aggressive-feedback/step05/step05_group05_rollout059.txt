import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial construction with smooth transitions
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 3:
            initial_h[i] = 0.0
        elif i < 2 * n_points // 3:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Particle swarm optimization parameters
    swarm_size = 30
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    max_iter = int(budget_s * 10)  # 100 iterations per second
    
    # Initialize swarm
    particles = np.random.uniform(0, 1, (swarm_size, n_points))
    velocities = np.random.uniform(-0.1, 0.1, (swarm_size, n_points))
    pbest_positions = particles.copy()
    pbest_values = np.array([objective(p) for p in particles])
    gbest_position = particles[np.argmin(pbest_values)]
    gbest_value = np.min(pbest_values)
    
    start_time = time.time()
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Update velocities and positions
        r1, r2 = np.random.rand(swarm_size, n_points), np.random.rand(swarm_size, n_points)
        velocities = inertia * velocities + cognitive * r1 * (pbest_positions - particles) + social * r2 * (gbest_position - particles)
        particles = np.clip(particles + velocities, 0, 1)
        
        # Ensure integral constraint
        particles = particles * target_sum / np.sum(particles, axis=1, keepdims=True)
        
        # Evaluate new positions
        current_values = np.array([objective(p) for p in particles])
        
        # Update personal best
        improve = current_values < pbest_values
        pbest_positions[improve] = particles[improve]
        pbest_values[improve] = current_values[improve]
        
        # Update global best
        if np.min(current_values) < gbest_value:
            gbest_position = particles[np.argmin(current_values)]
            gbest_value = np.min(current_values)
    
    best_h = gbest_position
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
