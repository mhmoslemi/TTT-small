        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    def generate_population(size):
        pop = np.random.rand(size, n_points)
        pop = np.clip(pop, 0, 1)

        return pop
    
    def tournament_selection(pop, k=5):
        candidates = np.random.choice(range(len(pop)), k, replace=False)
        best = min(candidates, key=lambda i: objective(pop[i]))
        return best
    
    def crossover(parent1, parent2):
        alpha = np.random.rand(n_points)
        return alpha * parent1 + (1 - alpha) * parent2
    
    def mutate(h, mutation_rate=0.05):
        mutated = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                mutated[i] += np.random.uniform(-0.1, 0.1)
                mutated[i] = np.clip(mutated[i], 0, 1)
