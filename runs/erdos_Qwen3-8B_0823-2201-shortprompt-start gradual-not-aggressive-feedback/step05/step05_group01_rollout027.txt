import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Genetic algorithm parameters
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 2

    for gen in range(generations):
        # Create population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        
        # Evaluate population
        fitness = [objective(h) for h in population]
        
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elites.copy()
        while len(new_population) < population_size:
            # Select parents
            parent1 = random.choice(elites)
            parent2 = random.choice(elites)
            
            # Crossover
            if random.random() < crossover_rate:
                split = random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:split], parent2[split:]])
            else:
                child = np.random.uniform(0, 1, n_points)
            
            # Mutation
            if random.random() < mutation_rate:
                mutation = np.random.uniform(-0.1, 0.1, n_points)
                child += mutation
            
            # Clip and normalize
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        fitness = [objective(h) for h in population]
        best_h = population[np.argmin(fitness)]
        best_c5 = min(best_c5, np.min(fitness))
        
        # Early stopping
        if time.time() - time.time() > budget_s:
            break
    
    return best_h, best_c5, n_points
