import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    time_remaining = budget_s

    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x

    def global_search(population):
        new_population = []
        for h in population:
            perturb = np.random.uniform(-0.1, 0.1, n_points)
            perturbed_h = h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            new_population.append(local_refinement(perturbed_h))
        return new_population

    # Initial refinement
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)

    # Population-based search
    population = [best_h.copy() for _ in range(5)]
    for _ in range(10):
        if time.time() - start_time >= budget_s:
            break
        population = global_search(population)
        current_c5 = objective(population[0])
        if current_c5 < best_c5:
            best_h = population[0].copy()
            best_c5 = current_c5
            print(f"Improved bound: {best_c5}")
    
    return best_h, best_c5, n_points
