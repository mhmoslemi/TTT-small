import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def evaluate(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    bounds = [(0, 1) for _ in range(n_points)]
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    generations = int((budget_s - 1) / 2)

    for _ in range(generations):
        # Create population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population.append(best_h)
        
        # Evaluate population
        fitness = [evaluate(h) for h in population]
        
        # Select best
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            # Select parents
            parent1 = population[np.random.choice(population_size)]
            parent2 = population[np.random.choice(population_size)]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.uniform(-0.1, 0.1)
                    child[i] = np.clip(child[i], 0, 1)
            
            new_population.append(child)
        
        population = new_population
        
        # Check time
        if time.time() - start_time > budget_s - 1:
            break

    return best_h, best_c5, n_points
