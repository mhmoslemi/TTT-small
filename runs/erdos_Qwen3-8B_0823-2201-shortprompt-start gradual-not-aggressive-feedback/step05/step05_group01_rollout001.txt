import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Genetic algorithm parameters
    population_size = 30
    generations = 100
    mutation_rate = 0.1
    elite_size = 2
    
    # Initial population
    population = np.random.rand(population_size, n_points)
    population = np.clip(population, 0, 1)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        
        # Select elite
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = population[elite_indices]
        
        # Create offspring
        offspring = np.zeros((population_size - elite_size, n_points))
        for i in range(population_size - elite_size):
            # Select parents
            parent1 = np.random.choice(elite_size)
            parent2 = np.random.choice(elite_size)
            parent1 = elite[parent1]
            parent2 = elite[parent2]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            
            # Mutate
            mutation = np.random.rand(n_points) < mutation_rate
            child[mutation] = np.random.rand(np.sum(mutation))
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            
            offspring[i] = child
        
        # Combine population
        population = np.vstack((elite, offspring))
        
        # Update best solution
        for ind in population:
            current_c5 = objective(ind)
            if current_c5 < best_c5:
                best_h = ind.copy()
                best_c5 = current_c5
    
    # Local optimization
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
