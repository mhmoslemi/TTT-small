import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def generate_population(size):
        pop = np.random.rand(size, n_points)
        pop = np.clip(pop, 0, 1)
        pop = pop * target_sum / np.sum(pop, axis=1, keepdims=True)
        return pop
    
    def tournament_selection(pop, k=3):
        selected = np.random.choice(len(pop), k, replace=False)
        return pop[selected[0]]
    
    def crossover(parent1, parent2):
        alpha = np.random.rand(n_points)
        return alpha * parent1 + (1 - alpha) * parent2
    
    def mutate(individual, mutation_rate=0.05):
        mutated = individual.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                mutated[i] = np.random.uniform(0, 1)
        mutated = mutated * target_sum / np.sum(mutated)
        return mutated
    
    def genetic_algorithm():
        population = generate_population(50)
        best_h = population[0]
        best_c5 = objective(best_h)
        
        for _ in range(100):
            new_population = []
            for _ in range(len(population)//2):
                parent1 = tournament_selection(population)
                parent2 = tournament_selection(population)
                child = crossover(parent1, parent2)
                child = mutate(child)
                new_population.append(child)
            population = new_population + [tournament_selection(population)]
        
        best_h = min(population, key=objective)
        best_c5 = objective(best_h)
        return best_h, best_c5
    
    def simulated_annealing(h, temp=1.0, cooling_rate=0.995, max_iter=100):
        current_h = h.copy()
        current_c5 = objective(current_h)
        best_h = current_h.copy()
        best_c5 = current_c5

            perturbed_c5 = objective(perturbed_h)

                current_c5 = perturbed_c5

                    best_h = current_h
                    best_c5 = current_c5
            
            temp *= cooling_rate
        
        return best_h, best_c5

    best_h, best_c5 = genetic_algorithm()

        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(

            perturbed_h,

            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},

            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
