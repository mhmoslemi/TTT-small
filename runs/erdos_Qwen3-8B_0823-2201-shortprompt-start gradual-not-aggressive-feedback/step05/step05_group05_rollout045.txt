import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use 100 points for balance between speed and accuracy
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial construction with better transitions
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 3:
            initial_h[i] = 0.0
        elif i < 2 * n_points // 3:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Genetic algorithm with local refinement
    def genetic_algorithm():
        population = [np.random.uniform(0, 1, n_points) for _ in range(50)]
        population = [h * target_sum / np.sum(h) for h in population]
        best_h = initial_h.copy()
        best_c5 = objective(best_h)
        
        for _ in range(100):
            # Evaluate population
            fitness = [objective(h) for h in population]
            best_idx = np.argmin(fitness)
            current_c5 = fitness[best_idx]
            
            if current_c5 < best_c5:
                best_h = population[best_idx].copy()
                best_c5 = current_c5
            
            # Selection and crossover
            selected = np.random.choice(len(population), size=50, p=np.array(fitness)/np.sum(fitness))
            new_population = []
            for i in range(0, 50, 2):
                parent1 = population[selected[i]]
                parent2 = population[selected[i+1]]
                child1 = np.random.uniform(0, 1, n_points)
                child2 = np.random.uniform(0, 1, n_points)
                for j in range(n_points):
                    if np.random.rand() < 0.5:
                        child1[j] = parent1[j]
                        child2[j] = parent2[j]
                    else:
                        child1[j] = parent2[j]
                        child2[j] = parent1[j]
                new_population.append(child1)
                new_population.append(child2)
            
            # Mutation
            for i in range(len(new_population)):
                for j in range(n_points):
                    if np.random.rand() < 0.1:
                        new_population[i][j] = np.clip(new_population[i][j] + np.random.normal(0, 0.05), 0, 1)
            
            # Replace population
            population = new_population
            population = [h * target_sum / np.sum(h) for h in population]
        
        return best_h, best_c5

    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm()
    
    # Local optimization with SLSQP
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Random perturbation with adaptive scaling
        perturb_scale = 0.15 * (1 - (time.time() - start_time) / budget_s)
        perturb = np.random.uniform(-perturb_scale, perturb_scale, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Optimization
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
            print(f"Improved bound: {best_c5}")
    
    return best_h, best_c5, n_points
