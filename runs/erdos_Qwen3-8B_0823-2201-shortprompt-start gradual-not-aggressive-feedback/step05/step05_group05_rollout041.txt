import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial construction with smooth transitions
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 3:
            initial_h[i] = 0.0
        elif i < 2 * n_points // 3:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.1
    elite_size = 2
    
    # Initialize population
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = np.array([h * target_sum / np.sum(h) for h in population])
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate fitness
        fitness = np.array([objective(h) for h in population])
        
        # Select elite
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = population[elite_indices]
        
        # Create new population
        new_population = np.zeros((population_size, n_points))
        new_population[:elite_size] = elite
        
        for i in range(elite_size, population_size):
            # Select parents
            parent1 = population[np.random.choice(population_size)]
            parent2 = population[np.random.choice(population_size)]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            
            # Mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[j] = np.random.uniform(0, 1)
            
            # Normalize
            child = child * target_sum / np.sum(child)
            new_population[i] = child
        
        population = new_population
        
        # Update best solution
        for h in population:
            c5 = objective(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                print(f"Improved bound: {best_c5}")
    
    return best_h, best_c5, n_points
