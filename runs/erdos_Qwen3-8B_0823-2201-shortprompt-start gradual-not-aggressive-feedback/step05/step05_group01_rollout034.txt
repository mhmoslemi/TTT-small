import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    best_h = initial_h.copy()
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 100
    time_per_generation = budget_s / generations
    
    for gen in range(generations):
        start_time = time.time()
        
        # Create population
        population = np.random.uniform(0, 1, (population_size, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        # Evaluate population
        fitness = np.array([objective(ind) for ind in population])
        
        # Select parents
        parents_indices = np.argsort(fitness)[:population_size // 2]
        parents = population[parents_indices]
        
        # Create offspring
        offspring = []
        for _ in range(population_size // 2):
            # Crossover
            if np.random.rand() < crossover_rate:
                parent1, parent2 = np.random.choice(parents_indices, 2, replace=False)
                child = np.zeros(n_points)
                split = np.random.randint(1, n_points - 1)
                child[:split] = parents[parent1][:split]
                child[split:] = parents[parent2][split:]
            else:
                child = np.random.uniform(0, 1, n_points)
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] = np.clip(child[i] + np.random.uniform(-0.1, 0.1), 0, 1)
            
            # Normalize
            child = child * target_sum / np.sum(child)
            offspring.append(child)
        
        # Combine population
        population = np.vstack((parents, offspring))
        
        # Evaluate offspring
        fitness = np.array([objective(ind) for ind in population])
        
        # Select best
        best_index = np.argmin(fitness)
        current_h = population[best_index]
        current_c5 = fitness[best_index]
        
        # Local refinement
        result = opt.minimize(
            objective,
            current_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
            print(f"Generation {gen+1}: Improved bound {best_c5}")
        
        # Check time budget
        if time.time() - start_time > time_per_generation:
            break
    
    return best_h, best_c5, n_points
