import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def evolve_population(population, generations=50):
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            # Select parents
            parents = population[np.argsort(fitness)[:len(population)//2]]
            # Crossover
            offspring = []
            for i in range(len(parents)//2):
                p1, p2 = np.random.choice(parents, 2, replace=False)
                child = np.random.choice([p1, p2], size=n_points)
                offspring.append(child)
            # Mutation
            for i in range(len(offspring)):
                offspring[i] = np.where(np.random.rand(n_points) < 0.1, 
                                      1 - offspring[i], offspring[i])
                offspring[i] = np.clip(offspring[i], 0, 1)
                offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
            # Combine and select
            population = np.vstack((parents, offspring))
            population = np.unique(population, axis=0)
            population = population[np.argsort(fitness[:len(parents)])[:len(population)//2]]
        return population

    # Initial population
    population_size = 20
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = np.vstack((initial_h, population))
    population = np.unique(population, axis=0)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evolve population
        population = evolve_population(population, generations=10)
        # Local optimization on best candidate
        best_h = population[0]
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        best_c5 = objective(best_h)
        print(f"Improved bound: {best_c5}")
    
    return best_h, best_c5, n_points
