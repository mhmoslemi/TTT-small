import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial construction with smooth transitions
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 3:
            initial_h[i] = 0.0
        elif i < 2 * n_points // 3:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Particle Swarm Optimization
    start_time = time.time()
    particles = np.random.uniform(0, 1, (n_points, n_points))
    velocities = np.random.uniform(-1, 1, (n_points, n_points))
    pbest = particles.copy()
    gbest = particles.copy()
    pbest_fitness = np.full(n_points, np.inf)
    gbest_fitness = np.inf

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_points):
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            velocities[i] = 0.5 * velocities[i] + r1 * (pbest[i] - particles[i]) + r2 * (gbest[i] - particles[i])
            particles[i] = np.clip(particles[i] + velocities[i], 0, 1)
            
            # Evaluate fitness
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
            corr = np.correlate(particles[i], 1 - particles[i], mode="full")
            fitness = np.max(corr) * dx
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fitness
            
            # Update global best
            if fitness < gbest_fitness:
                gbest = particles[i].copy()
                gbest_fitness = fitness
        
        # Update best_h and best_c5
        best_h = gbest
        best_c5 = gbest_fitness

    return best_h, best_c5, n_points
