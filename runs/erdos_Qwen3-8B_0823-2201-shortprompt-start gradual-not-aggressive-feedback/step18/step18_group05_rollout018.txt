import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial construction: 50% on [0, 1], 50% on [1, 2]
    initial_h = np.zeros(n_points)
    initial_h[:n_points//2] = 1.0
    initial_h[n_points//2:] = 0.0
    
    h = initial_h.copy()
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    # Multi-stage optimization with decreasing perturbations
    for stage in range(3):
        # Use Nelder-Mead for global exploration
        result = opt.minimize(
            lambda x: compute_c5(x, dx),
            h,
            method='Nelder-Mead',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
            options={'xatol': 1e-8, 'maxiter': 150}
        )
        h = result.x
        h = h * target_sum / np.sum(h)
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Adaptive perturbation with decreasing magnitude
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break
        
        for perturb_size in np.logspace(-3, -1, num=5, base=10):
            if time.time() - start_time >= budget_s:
                break
            # Create structured perturbations based on grid positions
            perturb = np.zeros(n_points)
            for i in range(n_points):
                if i % 5 == 0:
                    perturb[i] = np.random.uniform(-perturb_size, perturb_size)
            
            perturbed_h = h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            
            result = opt.minimize(
                lambda x: compute_c5(x, dx),
                perturbed_h,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-8}
            )
            
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = compute_c5(current_h, dx)
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx
