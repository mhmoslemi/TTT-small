import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Hybrid optimization: SLSQP refinement + simulated annealing exploration
    for stage in range(3):
        # Gradient-based refinement
        def local_refinement(h):
            result = opt.minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-8}
            )
            return result.x
        
        h = local_refinement(h)
        current_c5 = objective(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        remaining_time = budget_s - (time.time() - start_time)

            break
        
        # Simulated annealing for global search
        for temp in np.logspace(0, -4, num=5, base=10):
            if time.time() - start_time >= budget_s:
                break

            perturb = np.random.normal(0, temp, size=n_points)
            perturbed_h = h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            
            result = opt.minimize(
                objective,
                perturbed_h,
                method='SLSQP',
