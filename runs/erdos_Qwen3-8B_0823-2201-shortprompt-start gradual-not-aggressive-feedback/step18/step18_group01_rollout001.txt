import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    target_sum = n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def constraint_sum(h_flat):

        return np.sum(h) - target_sum

        result = opt.minimize(

            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-9}
        )
        best_h = result.x
        best_c5 = objective(best_h)
        
        # Dynamic perturbation
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break

        h_flat = best_h.reshape(n_points)
