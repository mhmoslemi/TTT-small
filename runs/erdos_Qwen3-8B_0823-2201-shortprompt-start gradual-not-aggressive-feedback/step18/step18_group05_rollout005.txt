import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")

        return c5

        h = h.reshape(n_points)

    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)

    # Multi-start optimization with different initial configurations
    for _ in range(3):

        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Local refinement with tighter tolerances
        result = opt.minimize(
            objective,

            method='L-BFGS-B',
            bounds=bounds,
