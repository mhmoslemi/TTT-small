import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    # Use a different starting point with 100 samples
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # New initial construction: alternating 0.5 and 0.5 with some variation
    initial_h_values = np.array([0.5 + 0.1 * np.sin(2 * np.pi * i / n_points) for i in range(n_points)])
    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h_vec):
        h = h_vec.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Multi-stage optimization with hybrid methods and structured perturbations
    for stage in range(5):
        # Gradient-based refinement with tighter tolerances
        def local_refinement(h_vec):
            result = opt.minimize(
                objective,
                h_vec,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-9}
            )
            return result.x
        
        h = local_refinement(h)
        current_c5 = objective(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Adaptive perturbation with structured magnitude
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break
        
        # Structured perturbations at different scales
        for perturb_size in [0.05, 0.02, 0.01, 0.005, 0.002]:
            if time.time() - start_time >= budget_s:
                break
            perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
            perturbed_h = h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            
            result = opt.minimize(
                objective,
                perturbed_h,
                method='Nelder-Mead',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 150, 'xatol': 1e-8}
            )
            
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = objective(current_h)
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
