import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Simulated annealing with adaptive temperature and targeted refinement
    temperature = 1.0
    cooling_rate = 0.995
    for step in range(100):
        if time.time() - start_time > budget_s:
            break
        # Generate perturbation
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed_h = h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Evaluate perturbed solution
        current_c5 = objective(perturbed_h)
        
        # Acceptance probability
        if current_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - current_c5) / temperature):
            h = perturbed_h
            best_c5 = current_c5
        
        # Local refinement with SLSQP
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-9}
        )
        h = result.x
        current_c5 = objective(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Update temperature
        temperature *= cooling_rate
    
    return best_h, best_c5, n_points
