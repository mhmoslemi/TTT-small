import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Hybrid optimization with structured perturbation
    for stage in range(4):
        # Local refinement
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8}
        )
        h = result.x
        current_c5 = objective(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Adaptive perturbation with decreasing magnitude

                break

                best_h = current_h.copy()
                best_c5 = current_c5
