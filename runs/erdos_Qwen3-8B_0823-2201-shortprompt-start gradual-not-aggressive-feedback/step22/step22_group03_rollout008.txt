import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Gradient-based refinement with adaptive learning rate
    result = opt.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-9}
    )
    h = result.x
    current_c5 = objective(h)
    
    if current_c5 < best_c5:
        best_h = h.copy()
        best_c5 = current_c5
    
    # Differential evolution for global exploration
    def differential_evolution(pop_size=20, maxiter=50):
        def fitness(individual):
            return objective(individual)
        
        result = opt.differential_evolution(
            fitness,
            bounds,
            strategy='best1bin',
            popsize=pop_size,
            maxiter=maxiter,
            tol=1e-12,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed
        )
        return result.x
    
    # Run differential evolution with decreasing population size
    for pop_size in [20, 15, 10]:
        if time.time() - start_time >= budget_s:
            break
        h = differential_evolution(pop_size=pop_size, maxiter=30)
        current_c5 = objective(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
