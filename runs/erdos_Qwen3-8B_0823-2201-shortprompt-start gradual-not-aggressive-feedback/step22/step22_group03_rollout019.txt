import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    
    # Try multiple initial configurations
    n_points_list = [47, 64, 80]
    best_h = None
    best_c5 = float('inf')
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        target_sum = n_points / 2
        
        # Generate initial configuration
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        
        def objective(h_vec):
            h = h_vec.reshape(n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5
        
        def constraint_sum(h_vec):
            h = h_vec.reshape(n_points)
            return np.sum(h) - target_sum
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        start_time = time.time()
        for stage in range(2):
            # SLSQP optimization
            result = opt.minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8}
            )
            h = result.x
            
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            # Perturbation with structured magnitude
            remaining_time = budget_s - (time.time() - start_time)
            if remaining_time <= 0:
                break
            
            for perturb_size in [0.05, 0.02, 0.01, 0.005]:
                if time.time() - start_time >= budget_s:
                    break
                perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
                perturbed_h = h + perturb
                perturbed_h = np.clip(perturbed_h, 0, 1)
                perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
                
                result = opt.minimize(
                    objective,
                    perturbed_h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8}
                )
                
                current_h = result.x
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * target_sum / np.sum(current_h)
                current_c5 = objective(current_h)
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
    
    return best_h, best_c5, len(best_h)
