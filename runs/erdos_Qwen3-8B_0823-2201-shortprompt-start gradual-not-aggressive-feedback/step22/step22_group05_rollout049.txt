import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Smaller n_points for faster optimization
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial guess based on uniform distribution
    h = np.ones(n_points) * 0.5
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.inf
    
    start_time = time.time()
    
    # Multi-stage optimization with alternating objectives
    for stage in range(3):
        # First, optimize for minimum overlap
        def objective_overlap(h_vec):
            h = h_vec.reshape(n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5
        
        # Second, optimize for integral constraint enforcement
        def objective_integral(h_vec):
            h = h_vec.reshape(n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            return np.sum(h) - target_sum
        
        # Optimization with alternating objectives
        for _ in range(2):
            # Use SLSQP for constrained optimization
            result = opt.minimize(
                objective_overlap,
                best_h,
                method='SLSQP',
                bounds=[(0, 1) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum},
                options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-9}
            )
            best_h = result.x
            current_c5 = objective_overlap(best_h)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
            
            remaining_time = budget_s - (time.time() - start_time)
            if remaining_time <= 0:
                break
            
            # Apply structured perturbations
            corr = np.correlate(best_h, 1 - best_h, mode="full")
            overlap_indices = np.argsort(corr[-n_points//2:])[-n_points//4:]
            perturb = np.zeros(n_points)
            perturb[overlap_indices] = np.random.uniform(-0.02, 0.02, len(overlap_indices))
            perturbed_h = best_h + perturb

            result = opt.minimize(

            if current_c5 < best_c5:

                best_c5 = current_c5
    
    return best_h, best_c5, n_points
