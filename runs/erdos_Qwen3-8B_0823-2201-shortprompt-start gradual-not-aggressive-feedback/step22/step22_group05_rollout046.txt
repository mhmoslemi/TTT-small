import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = 100  # Smaller n_points for faster computation
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    h = np.full(n_points, 0.5)

    for stage in range(5):

            objective,
            h,
            method='SLSQP',
