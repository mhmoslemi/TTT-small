import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Hybrid approach: genetic algorithm for global search + local optimization
    
    # Population-based global search
    population_size = 20
    population = [h + np.random.uniform(-0.1, 0.1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    for gen in range(20):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate population
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        
        # Update best solution
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]
        
        # Selection and crossover
        selected = np.random.choice(range(population_size), size=population_size, p=np.array(fitness)/np.sum(fitness))
        new_population = []
        
        for i in range(0, population_size, 2):
            p1, p2 = population[selected[i]], population[selected[i+1]]
            child = p1 * 0.5 + p2 * 0.5
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        population = new_population
        
        # Local refinement with SLSQP
        for h in population:
            result = opt.minimize(

            )
