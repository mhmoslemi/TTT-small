import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h_values)

    def objective(h):

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.15
    crossover_rate = 0.8
    elite_size = 2
    
    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = np.array([objective(h) for h in population])
        
        # Select elite
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = [population[i] for i in elite_indices]
        
        # Select parents
        parent_indices = np.argsort(fitness)[::-1][:int(population_size * crossover_rate)]
        parents = [population[i] for i in parent_indices]
        
        # Crossover
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                parent1 = parents[i]
                parent2 = parents[i + 1]
                child = np.zeros(n_points)
                for j in range(n_points):
                    if random.random() < 0.5:
                        child[j] = parent1[j]
                    else:
                        child[j] = parent2[j]
                offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            for j in range(n_points):
                if random.random() < mutation_rate:
                    offspring[i][j] = np.clip(offspring[i][j] + np.random.normal(0, 0.05), 0, 1)
        
        # Combine population
        population = elite + offspring
        
        # Local refinement for top candidates
        for h in population[:int(population_size * 0.3)]:
            result = opt.minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,

        for h in population:
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    # Final smoothing
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    
    return best_h, best_c5, n_points
