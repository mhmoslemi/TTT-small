import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Multi-phase adaptive perturbation and optimization
    for phase in range(3):
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break
        
        # Global exploration phase with large perturbations
        if phase == 0:
            for _ in range(2):
                # Use L-BFGS-B for faster convergence
                result = opt.minimize(
                    objective,
                    h,
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8}
                )
                h = result.x
                current_c5 = objective(h)
                
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                
                # Introduce structured perturbations
                corr = np.correlate(h, 1 - h, mode="full")
                overlap_indices = np.argsort(corr)[::-1][:n_points//4]
                perturb = np.zeros(n_points)
                perturb[overlap_indices] = np.random.uniform(-0.1, 0.1, len(overlap_indices))
                h = np.clip(h + perturb, 0, 1)
                h = h * target_sum / np.sum(h)
        
        # Local refinement phase with smaller perturbations
        elif phase == 1:
            for _ in range(3):
                result = opt.minimize(
                    objective,
                    h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-9}
                )
                h = result.x
                current_c5 = objective(h)
                
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                
                # Adaptive perturbation based on current overlap
                corr = np.correlate(h, 1 - h, mode="full")
                overlap_indices = np.argsort(corr)[-n_points//4:]
                perturb = np.random.uniform(-0.02, 0.02, n_points)
                perturb[overlap_indices] *= 2
                h = np.clip(h + perturb, 0, 1)
                h = h * target_sum / np.sum(h)
        
        # Final fine-tuning phase
        else:
            for _ in range(2):
                result = opt.minimize(
                    objective,
                    h,
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8}
                )
                h = result.x
                current_c5 = objective(h)
                
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                
                # Final small perturbations
                corr = np.correlate(h, 1 - h, mode="full")
                overlap_indices = np.argsort(corr)[-n_points//4:]
                perturb = np.random.uniform(-0.01, 0.01, n_points)
                perturb[overlap_indices] *= 1.5
                h = np.clip(h + perturb, 0, 1)
                h = h * target_sum / np.sum(h)
    
    return best_h, best_c5, n_points
