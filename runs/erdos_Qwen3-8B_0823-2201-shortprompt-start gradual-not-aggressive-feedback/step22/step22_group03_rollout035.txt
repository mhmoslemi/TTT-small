import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Gradient-based refinement
    for _ in range(3):
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-9}
        )
        h = result.x
        current_c5 = objective(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        if time.time() - start_time >= budget_s:
            break
    
    # Simulated annealing for global search
    temp = 1.0
    cooling_rate = 0.995
    max_iter = 100
    
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        
        # Perturb the solution
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        perturbed_h = h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Evaluate the perturbed solution
        current_c5 = objective(perturbed_h)
        
        # Acceptance probability
        if current_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - current_c5) / temp):
            h = perturbed_h
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        
        temp *= cooling_rate
    
    return best_h, best_c5, n_points
