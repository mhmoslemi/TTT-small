import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Convert initial_h_values to continuous parameters
    def parametric_h(params, n_points):
        # params: a vector of parameters defining the step function
        # Use a piecewise-linear approximation with breakpoints
        breakpoints = np.sort(np.random.uniform(0, 2, int(n_points ** 0.5)))
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            h[i] = np.sum(params[0] * (x <= breakpoints) + params[1] * (x > breakpoints))
        return h
    
    # Objective function
    def objective(params):
        h = parametric_h(params, n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Constraint: sum(h) == target_sum
    def constraint(params):
        h = parametric_h(params, n_points)
        return np.sum(h) - target_sum
    
    # Initial parameter guess
    params_initial = np.random.uniform(0, 1, 2)
    
    best_params = params_initial.copy()
    best_c5 = objective(best_params)
    best_h = parametric_h(best_params, n_points)
    
    start_time = time.time()
    
    # Multi-stage optimization: exploration -> refinement -> fine-tuning
    for stage in range(3):
        # Stage 1: Global exploration with random perturbations
        if stage == 0:
            for _ in range(20):
                if time.time() - start_time >= budget_s:
                    break
                # Perturb parameters
                perturb = np.random.uniform(-0.1, 0.1, 2)
                perturbed_params = best_params + perturb
                # Optimize
                result = opt.minimize(
                    objective,
                    perturbed_params,
                    method='L-BFGS-B',
                    bounds=[(0, 1), (0, 1)],
                    constraints={'type': 'eq', 'fun': constraint},
                    options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-9}

                    best_params = current_params.copy()

                    best_h = current_h.copy()

                    objective,
                    perturbed_params,
                    method='L-BFGS-B',
                    bounds=[(0, 1), (0, 1)],

                # Optimize

                    objective,
