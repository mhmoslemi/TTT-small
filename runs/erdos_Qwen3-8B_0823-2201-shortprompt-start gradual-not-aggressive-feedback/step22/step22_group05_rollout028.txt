import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = [h.copy() for _ in range(population_size)]
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        
        # Local optimization on best solution
        result = opt.minimize(
            objective,
            population[best_idx],
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-9}
        )
        population[best_idx] = result.x
        
        # Create new population
        new_population = []
        
        for i in range(population_size):
            if np.random.rand() < crossover_rate:
                # Crossover
                parent1 = population[np.random.randint(population_size)]
                parent2 = population[np.random.randint(population_size)]
                child = (parent1 + parent2) / 2
            else:
                # Mutation
                child = population[i].copy()

                        child[j] = np.clip(child[j], 0, 1)
            
            # Normalize
            child = child * target_sum / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
    
    # Final evaluation
    best_h = np.array([p for p in population if objective(p) <= best_c5][0])
