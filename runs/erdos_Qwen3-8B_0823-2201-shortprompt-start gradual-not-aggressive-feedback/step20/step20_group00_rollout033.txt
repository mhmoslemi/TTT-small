import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use 100 samples for better resolution
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial construction with randomized binary pattern
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if np.random.rand() < 0.5:
            initial_h[i] = 1.0
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    # Hybrid optimization with structured perturbations and adaptive learning
    for stage in range(5):
        # Gradient-based refinement with adaptive learning rate
        def local_refinement(h):
            result = opt.minimize(
                objective,
                h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-9}
            )
            return result.x
        
        h = local_refinement(best_h)
        current_c5 = objective(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Adaptive perturbation with structured magnitude
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break
        
        # Structured perturbations with decreasing magnitude
        for perturb_size in [0.05, 0.03, 0.01, 0.005, 0.002]:
            if time.time() - start_time >= budget_s:
                break
            # Structured perturbation: avoid uniform randomness
            perturb = np.random.normal(0, perturb_size, n_points)
            perturb = np.clip(perturb, -0.1, 0.1)  # Limit extreme changes
            perturbed_h = h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            
            result = opt.minimize(
                objective,
                perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 150, 'ftol': 1e-10, 'gtol': 1e-9}
            )

            current_c5 = objective(current_h)
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
