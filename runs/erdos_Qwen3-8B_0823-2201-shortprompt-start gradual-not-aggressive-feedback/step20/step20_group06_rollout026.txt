        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    elite_fraction = 0.2
    
    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    for generation in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        
        # Find elite solutions
        elite_indices = np.argsort(fitness)[:int(elite_fraction * population_size)]
        elites = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elites.copy()
        
        while len(new_population) < population_size:
            # Select parents
            parent1 = np.random.choice(elite_indices)
            parent2 = np.random.choice(elite_indices)
            parent1 = population[parent1]
            parent2 = population[parent2]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))

            # Normalize
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        population = new_population
    
    # Find the best solution in the final population

    best_index = np.argmin(fitness)
