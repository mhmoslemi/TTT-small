import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Particle swarm optimization with local refinement
    population_size = 20
    inertia_weight = 0.8
    cognitive_coeff = 1.2
    social_coeff = 1.2
    max_iter = 100
    
    # Initialize swarm
    swarm = np.random.rand(population_size, n_points)
    swarm = swarm * target_sum / np.sum(swarm, axis=1, keepdims=True)
    velocities = np.random.rand(population_size, n_points) * 0.01
    
    personal_best = swarm.copy()
    personal_best_values = np.array([objective(ind) for ind in swarm])
    global_best = swarm[np.argmin(personal_best_values)].copy()
    global_best_value = np.min(personal_best_values)
    
    for iter in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        
        for i in range(population_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_weight * velocities[i] +
                             cognitive_coeff * r1 * (personal_best[i] - swarm[i]) +
                             social_coeff * r2 * (global_best - swarm[i]))
            
            swarm[i] += velocities[i]
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * target_sum / np.sum(swarm[i])
            
            current_val = objective(swarm[i])
            
            if current_val < personal_best_values[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_values[i] = current_val
                
                if current_val < global_best_value:
                    global_best = swarm[i].copy()
                    global_best_value = current_val
        
        # Local refinement of global best
        result = opt.minimize(
            objective,
            global_best,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-9}
        )
        
        refined_global = result.x
        refined_global = np.clip(refined_global, 0, 1)
        refined_global = refined_global * target_sum / np.sum(refined_global)
        refined_c5 = objective(refined_global)
        
        if refined_c5 < global_best_value:
            global_best = refined_global.copy()
            global_best_value = refined_c5
            
        if global_best_value < best_c5:
            best_h = global_best.copy()
            best_c5 = global_best_value
    
    return best_h, best_c5, n_points
