import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Smaller than initial construction for faster optimization
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution - random construction
    h = np.random.rand(n_points)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Multi-stage optimization with structured perturbations
    for stage in range(3):
        # Local refinement with more aggressive constraints
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-9}
        )
        best_h = result.x
        best_c5 = objective(best_h)
        
        # Adaptive perturbation based on overlap contribution
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break
        
        for perturb_size in np.logspace(-3, -1, num=5, base=10):
            if time.time() - start_time >= budget_s:
                break
            
            h_flat = best_h.reshape(n_points)
            corr = np.correlate(h_flat, 1 - h_flat, mode="full")
            # Focus on regions with highest overlap contribution
            overlap_weights = corr[-n_points//2:]  # Focus on recent values
            weighted_indices = np.argsort(overlap_weights)[-20:]
            
            # Create perturbation vector
            perturb = np.zeros(n_points)
            perturb[weighted_indices] = np.random.uniform(-perturb_size, perturb_size, len(weighted_indices))
            
            # Apply perturbation and normalize
            perturbed_h = best_h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            
            # Local optimization of perturbed solution
            result = opt.minimize(
                objective,
                perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-9}
            )
            
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = objective(current_h)
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
