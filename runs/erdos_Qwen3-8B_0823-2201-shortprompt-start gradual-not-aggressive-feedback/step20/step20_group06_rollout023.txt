import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Hybrid optimization with adaptive perturbation and evolutionary refinement
    for stage in range(5):
        # Local refinement with SLSQP
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-9}
        )
        h = result.x
        current_c5 = objective(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Adaptive perturbation with decreasing magnitude
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break
        
        perturb_size = np.max(np.abs(h)) * 0.1
        perturbed_h = h + np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Evolutionary refinement
        for _ in range(3):
            if time.time() - start_time >= budget_s:
                break
            
            # Create offspring with small mutations
            offspring = perturbed_h + np.random.normal(0, perturb_size/10, n_points)
            offspring = np.clip(offspring, 0, 1)
            offspring = offspring * target_sum / np.sum(offspring)
            
            # Evaluate and select best
            current_c5 = objective(offspring)
            if current_c5 < best_c5:
                best_h = offspring.copy()
                best_c5 = current_c5
        
        # Focus on high-overlap regions
        corr = np.correlate(h, 1 - h, mode="full")
        max_indices = np.argsort(corr[-n_points//2:])[-10:]

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    
    return best_h, best_c5, n_points
