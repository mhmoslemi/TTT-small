import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solutions
    initial_solutions = [np.array(initial_h_values, dtype=np.float64)]
    for _ in range(3):
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        perturb = perturb * target_sum / np.sum(perturb)
        initial_solutions.append(np.clip(perturb, 0, 1))
    
    best_h = np.array(initial_h_values, dtype=np.float64)
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for sol in initial_solutions:
        sol = sol * target_sum / np.sum(sol)
        h = sol.copy()
        
        for stage in range(2):
            result = opt.minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-9}
            )
            h = result.x
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = objective(h)
            
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            
            if time.time() - start_time >= budget_s:
                break
            
            # Adaptive perturbation
            corr = np.correlate(h, 1 - h, mode="full")
            max_indices = np.argsort(corr[-n_points//2:])[-10:]
            perturb = np.zeros(n_points)
            perturb[max_indices] = np.random.uniform(-0.05, 0.05, len(max_indices))
            perturbed_h = h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            
            result = opt.minimize(
                objective,
                perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-9}
            )
            
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = objective(current_h)
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
