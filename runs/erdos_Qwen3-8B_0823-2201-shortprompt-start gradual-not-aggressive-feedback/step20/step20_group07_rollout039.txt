import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    start_time = time.time()
    
    # Dynamic resolution adjustment
    for resolution in [n_points, n_points // 2, n_points // 4, n_points // 8]:
        if resolution < 4:
            break
        h_rescaled = np.kron(h, np.ones(resolution // n_points))[:resolution]
        h_rescaled = h_rescaled * (resolution / 2) / np.sum(h_rescaled)
        
        def objective_rescaled(h):
            h = h.reshape(resolution)
            h = np.clip(h, 0, 1)
            h = h * (resolution / 2) / np.sum(h)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * (2.0 / resolution)
            return c5
        
        def constraint_sum_rescaled(h):
            h = h.reshape(resolution)
            return np.sum(h) - (resolution / 2)
        
        bounds_rescaled = [(0, 1) for _ in range(resolution)]
        
        # Local refinement
        result = opt.minimize(
            objective_rescaled,
            h_rescaled,
            method='SLSQP',
            bounds=bounds_rescaled,
            constraints={'type': 'eq', 'fun': constraint_sum_rescaled},
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-9}
        )
        h_rescaled = result.x
        
        # Project back to original resolution
        h = np.kron(h_rescaled, np.ones(n_points // resolution))[:n_points]
        h = h * (n_points / 2) / np.sum(h)
        
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Perturbation
        if time.time() - start_time >= budget_s:
            break
        
        for perturb_size in np.logspace(-3, -1, num=5, base=10):
            if time.time() - start_time >= budget_s:
                break
            
            h_flat = h.reshape(n_points)
            corr = np.correlate(h_flat, 1 - h_flat, mode="full")
            max_indices = np.argsort(corr[-n_points//2:])[-10:]
            perturb = np.zeros(n_points)
            perturb[max_indices] = np.random.uniform(-perturb_size, perturb_size, len(max_indices))
            perturbed_h = h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            
            result = opt.minimize(
                objective,
                perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-9}
            )
            
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = objective(current_h)
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
