import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # New initialization: alternating blocks
    def new_initialization():
        pattern = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                pattern[i] = 1.0
            elif i % 3 == 1:
                pattern[i] = 0.5
            else:
                pattern[i] = 0.0
        return pattern * target_sum / np.sum(pattern)
    
    # Initial solution
    h = np.array(new_initialization(), dtype=np.float64)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Two-phase optimization: L-BFGS-B followed by SLSQP
    for stage in range(2):
        # Phase 1: L-BFGS-B refinement
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-9}
        )
        h = result.x

            best_h = h.copy()
            best_c5 = current_c5

            objective,
            h,
            method='SLSQP',
            bounds=bounds,

                {'type': 'ineq', 'fun': lambda x: x - 0.001},
