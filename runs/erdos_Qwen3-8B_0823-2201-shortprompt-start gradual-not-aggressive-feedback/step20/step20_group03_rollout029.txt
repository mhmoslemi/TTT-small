import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Warm-up phase: gradient-based refinement
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-9}
    )
    h = result.x
    best_h = h.copy()
    best_c5 = objective(h)
    
    # Adaptive perturbation with structured magnitude
    for stage in range(5):
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break
        
        # Compute overlap to guide perturbations
        corr = np.correlate(h, 1 - h, mode="full")
        overlap_indices = np.argsort(corr[-n_points//2:])[-n_points//4:]
        
        # Perturbations at decreasing scales
        for perturb_size in [0.05, 0.02, 0.01, 0.005, 0.002]:
            if time.time() - start_time >= budget_s:
                break
            
            # Apply perturbation
            perturb = np.zeros(n_points)
            perturb[overlap_indices] = np.random.uniform(-perturb_size, perturb_size, len(overlap_indices))
            perturbed_h = h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            
            # Optimization with perturbed solution
            result = opt.minimize(
                objective,
                perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-9}
            )
            
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = objective(current_h)
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
