import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Start with the initial solution and its perturbations
    base_h = np.array(initial_h_values, dtype=np.float64)
    base_h = base_h * target_sum / np.sum(base_h)
    
    best_h = base_h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    # Multi-start optimization with perturbations
    for perturb_scale in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]:
        if time.time() - start_time >= budget_s:
            break
        # Perturb the base solution
        perturbed_h = base_h + np.random.uniform(-perturb_scale, perturb_scale, n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Optimize the perturbed solution
        result = opt.minimize(
            objective,
            perturbed_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-9}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
