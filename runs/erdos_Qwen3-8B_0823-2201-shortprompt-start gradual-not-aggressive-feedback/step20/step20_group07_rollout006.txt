import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Generate multiple initial perturbations
    perturbations = []
    for _ in range(5):
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        perturb = perturb * target_sum / np.sum(np.abs(perturb))
        perturbations.append(perturb)
    
    # Multistart optimization
    for perturb in perturbations:
        h_start = h + perturb
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * target_sum / np.sum(h_start)
        
        result = opt.minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-9}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
        
        if time.time() - start_time >= budget_s:
            break
    
    return best_h, best_c5, n_points
