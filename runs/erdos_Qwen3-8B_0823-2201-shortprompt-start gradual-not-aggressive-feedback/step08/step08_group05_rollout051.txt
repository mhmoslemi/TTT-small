import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Particle swarm optimization
    def particle_swarm():
        particles = np.random.uniform(0, 1, (n_points, 3))
        velocities = np.random.uniform(-0.1, 0.1, (n_points, 3))
        personal_best = particles.copy()
        personal_best_values = np.array([objective(p) for p in particles])
        global_best = particles[np.argmin(personal_best_values)]
        global_best_value = np.min(personal_best_values)
        
        for _ in range(100):
            for i in range(n_points):
                r1, r2 = np.random.rand(2)
                cognitive = 1.5 * r1 * (personal_best[i] - particles[i])
                social = 1.5 * r2 * (global_best - particles[i])
                velocities[i] += cognitive + social
                particles[i] += velocities[i]
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = particles[i] * target_sum / np.sum(particles[i])
                current_value = objective(particles[i])
                
                if current_value < personal_best_values[i]:
                    personal_best[i] = particles[i].copy()
                    personal_best_values[i] = current_value
                    
            global_best = particles[np.argmin(personal_best_values)]
            global_best_value = np.min(personal_best_values)
        
        return global_best, global_best_value
    
    # Sequential quadratic programming refinement
    def sqp_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Main optimization
    start_time = time.time()
    best_h, best_c5 = particle_swarm()

    while time.time() - start_time < budget_s:
