import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    pop_size = 20
    generations = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 2
    
    def fitness(population):
        return np.array([objective(ind) for ind in population])
    
    def selection(population, fitness_values):
        indices = np.argsort(fitness_values)
        return population[indices[:pop_size]]
    
    def crossover(parent1, parent2):
        if np.random.rand() < crossover_rate:
            split = np.random.randint(1, n_points - 1)
            child = np.concatenate((parent1[:split], parent2[split:]))
        else:
            child = parent1.copy()
        return child
    
    def mutation(ind):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                ind[i] += np.random.uniform(-0.1, 0.1)
                ind[i] = np.clip(ind[i], 0, 1)
        return ind
    
    # Genetic algorithm
    population = np.random.uniform(0, 1, (pop_size, n_points))
    population = np.array([h * target_sum / np.sum(h) for h in population])
    
    for _ in range(generations):
        fitness_values = fitness(population)
        population = selection(population, fitness_values)
        
        new_population = []
        for _ in range(pop_size - elite_size):
            parent1, parent2 = np.random.choice(population, 2, replace=False)
            child = crossover(parent1, parent2)
            child = mutation(child)
            new_population.append(child)
        
        population = np.vstack((population[:elite_size], new_population))
        population = np.array([h * target_sum / np.sum(h) for h in population])
    
    # Local refinement with SLSQP
    best_h = population[np.argmin(fitness(population))]
    best_h = best_h * target_sum / np.sum(best_h)
    
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    start_time = time.time()
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.1 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
