import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    pop_size = 20
    generations = 100
    mutation_rate = 0.1
    tournament_size = 5
    
    def crossover(parent1, parent2):
        child = parent1.copy()
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parent2[i]
        return child
    
    def mutate(individual):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] += np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        return individual
    
    # Initialize population
    population = [np.array(initial_h, dtype=np.float64) * target_sum / np.sum(initial_h) for _ in range(pop_size)]
    
    start_time = time.time()
    
    for generation in range(generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            contestants = np.random.choice(range(pop_size), size=tournament_size, replace=False)
            winner = contestants[np.argmin(fitness[contestants])]
            selected.append(population[winner])
        
        # Crossover and mutation
        new_population = []
        for i in range(pop_size // 2):
            parent1 = selected[i*2]
            parent2 = selected[i*2 + 1]
            child1 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = crossover(parent1, parent2)
            child2 = mutate(child2)
            new_population.append(child1)
            new_population.append(child2)
        
        # Replace population
        population = new_population
        
        # Local refinement with SLSQP
        for i in range(len(population)):
            result = opt.minimize(
                objective,
                population[i],
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            population[i] = result.x
    
    # Find best solution
    best_h = min(population, key=objective)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
