        return c5

        h = h.reshape(n_points)

        alpha = np.random.rand(n_points)
        return alpha * a + (1 - alpha) * b

        perturb = np.random.normal(0, mutation_strength, n_points)
        return np.clip(h + perturb, 0, 1)
