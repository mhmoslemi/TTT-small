import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm with local refinement
    def genetic_algorithm():
        population_size = 20
        generations = 100
        mutation_rate = 0.1
        elite_size = 2
        
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = [objective(ind) for ind in population]
            best_idx = np.argmin(fitness)
            best_h = population[best_idx]
            best_c5 = fitness[best_idx]
            
            # Select elites
            elite_indices = np.argsort(fitness)[:elite_size]
            elites = [population[i] for i in elite_indices]
            
            # Create new population
            new_population = elites
            while len(new_population) < population_size:
                # Select parents
                parent1 = population[np.random.choice(population_size)]
                parent2 = population[np.random.choice(population_size)]
                
                # Crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
                
                # Mutate
                if np.random.rand() < mutation_rate:
                    mutation = np.random.normal(0, 0.05, n_points)
                    child += mutation
                    child = np.clip(child, 0, 1)
                
                # Normalize
                child = child * target_sum / np.sum(child)
                new_population.append(child)
            
            population = new_population
        
        # Local refinement of best solution
        best_h = population[np.argmin([objective(ind) for ind in population])]
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x, objective(result.x)
    
    # Run genetic algorithm
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.1 * (remaining_time / budget_s))

        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)

            perturbed_h,

        current_c5 = objective(current_h)

            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
