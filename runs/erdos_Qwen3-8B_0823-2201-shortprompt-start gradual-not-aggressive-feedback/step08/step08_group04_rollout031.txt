import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial population for genetic algorithm
    population_size = 20
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = [p * target_sum / np.sum(p) for p in population]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    start_time = time.time()
    max_iterations = int(budget_s * 10)
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate population
        fitness = [objective(p) for p in population]
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_c5 = fitness[best_idx]
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
        
        # Selection
        selected = np.random.choice(population_size, size=population_size, p=np.array(fitness)/np.sum(fitness))
        selected = [population[i] for i in selected]
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if i+1 < population_size:
                parent1 = selected[i]
                parent2 = selected[i+1]
                child1 = parent1 * 0.5 + parent2 * 0.5
                child2 = parent2 * 0.5 + parent1 * 0.5
                offspring.extend([child1, child2])
            else:
                offspring.append(selected[i])
        
        # Mutation
        for i in range(len(offspring)):
            offspring[i] = np.copy(offspring[i])
            for j in range(n_points):
                if random.random() < 0.1:
                    offspring[i][j] = np.clip(offspring[i][j] + np.random.uniform(-0.1, 0.1), 0, 1)
        
        # Replace population
        population = offspring
    
    # Final refinement with SLSQP
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
