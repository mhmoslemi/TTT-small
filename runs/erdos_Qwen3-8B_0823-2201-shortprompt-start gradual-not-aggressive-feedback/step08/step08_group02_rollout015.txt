import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    pop_size = 20
    generations = 100
    tournament_size = 5
    mutation_rate = 0.1
    
    def genetic_algorithm():
        population = np.random.rand(pop_size, n_points)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for gen in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            
            # Tournament selection
            selected = []
            for _ in range(pop_size):
                competitors = np.random.choice(pop_size, tournament_size, replace=False)
                winner = competitors[np.argmin(fitness[competitors])]
                selected.append(population[winner])
            population = np.array(selected)
            
            # Mutation
            for i in range(pop_size):
                if np.random.rand() < mutation_rate:
                    mutation = np.random.uniform(-0.1, 0.1, n_points)
                    population[i] += mutation
                    population[i] = np.clip(population[i], 0, 1)
                    population[i] = population[i] * target_sum / np.sum(population[i])
            
            # Local refinement with SLSQP
            for i in range(pop_size):
                result = opt.minimize(
                    objective,
                    population[i],
                    method='SLSQP',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 50, 'ftol': 1e-8}
                )
                population[i] = result.x
        
        best_h = population[np.argmin([objective(ind) for ind in population])]
        best_c5 = objective(best_h)
        return best_h, best_c5
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()
    best_h = best_h * target_sum / np.sum(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:

        perturb_size = max(0.01, 0.1 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb

        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,

        current_c5 = objective(current_h)

            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
