import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    pop_size = 20
    generations = 50
    mutation_rate = 0.15
    crossover_rate = 0.8
    
    def fitness(population):
        return np.array([objective(ind) for ind in population])
    
    def select_parents(population, fitnesses):
        idxs = np.argsort(fitnesses)
        return population[idxs[:pop_size//2]]
    
    def crossover(parents):
        offspring = []
        for _ in range(pop_size - len(parents)):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            alpha = np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2
            offspring.append(child)
        return offspring
    
    def mutate(individual):
        mask = np.random.rand(n_points) < mutation_rate
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        individual[mask] += perturb[mask]
        individual = np.clip(individual, 0, 1)
        individual = individual * target_sum / np.sum(individual)
        return individual
    
    # Genetic algorithm
    population = [h + np.random.uniform(-0.1, 0.1, n_points) for _ in range(pop_size)]
    population = [ind * target_sum / np.sum(ind) for ind in population]
    
    start_time = time.time()
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitnesses = fitness(population)
        parents = select_parents(population, fitnesses)
        offspring = crossover(parents)
        population = parents + offspring
        population = [mutate(ind) for ind in population]
    
    # Local refinement with SQP
    best_h = np.argmin(fitness(population))  # Find best individual
    best_h = population[best_h]
    best_h = np.array(best_h, dtype=np.float64)
    best_c5 = objective(best_h)
    
    # Additional refinement
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
