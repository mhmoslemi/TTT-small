import scipy.optimize as opt
import time
import random

    def objective(h):

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

    # Local optimization with SLSQP
    result = opt.minimize(

        best_h,
        method='SLSQP',
        bounds=bounds,
