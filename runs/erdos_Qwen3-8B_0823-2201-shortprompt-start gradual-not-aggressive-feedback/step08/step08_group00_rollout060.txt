import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    h = np.array(initial_h, dtype=np.float64)

    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Genetic algorithm with local refinement
    def genetic_search():
        population_size = 20
        mutation_rate = 0.1
        crossover_rate = 0.8

            best_h = local_refinement(best_h)
            
            # Create new population
            new_population = []

                parent1 = random.choice(population)
