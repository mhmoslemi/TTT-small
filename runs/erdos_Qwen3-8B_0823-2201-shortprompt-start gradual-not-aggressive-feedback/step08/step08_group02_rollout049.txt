import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def tournament_selection(population, k=5):
        selected = random.sample(population, k)
        return min(selected, key=objective)
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        child[indices] = parent2[indices]
        return child
    
    def mutate(individual, mutation_rate=0.1):
        mutated = np.copy(individual)
        for i in range(n_points):
            if random.random() < mutation_rate:
                mutated[i] += np.random.uniform(-0.1, 0.1)
                mutated[i] = np.clip(mutated[i], 0, 1)
        mutated = mutated * target_sum / np.sum(mutated)
        return mutated
    
    population_size = 20
    population = [h * target_sum / np.sum(h) for _ in range(population_size)]
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    for generation in range(int(budget_s / 2)):
        if time.time() - start_time > budget_s:
            break
            
        # Evolutionary step
        new_population = []
        for _ in range(population_size):
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
            
        population = new_population
        
        # Local refinement with SLSQP
        for ind in population:
            result = opt.minimize(
                objective,
                ind,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            ind[:] = result.x
        
        # Update best solution
        for ind in population:
            current_c5 = objective(ind)
            if current_c5 < best_c5:
                best_h = ind.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
