import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use 100 points for better resolution
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial construction with better spread
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points//3:
            initial_h[i] = 1.0
        elif i < 2*n_points//3:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 0.0
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Simulated annealing with adaptive cooling and better perturbation
    def simulated_annealing():
        T = 1.0
        cool_rate = 0.995
        max_iter = 1000
        best_h = initial_h.copy()
        best_c5 = objective(best_h)
        
        for _ in range(max_iter):
            T *= cool_rate
            if T < 1e-5:
                break
            # Better perturbation with adaptive magnitude
            perturb_size = max(0.05, 0.05 * (T / 1.0))
            perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
            perturbed = np.clip(best_h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            current_c5 = objective(perturbed)
            
            if current_c5 < best_c5 or np.random.rand() < np.exp((current_c5 - best_c5)/T):
                best_h = perturbed
                best_c5 = current_c5
        
        return best_h, best_c5
    
    start_time = time.time()
    best_h, best_c5 = simulated_annealing()
    
    # CVXPY-based local optimization
    def cvxpy_refinement(h):
        h_var = cp.Variable(n_points)
        constraints = [cp.sum(h_var) == target_sum, 0 <= h_var, h_var <= 1]
        objective = cp.Maximize(cp.max(cp.multiply(h_var, 1 - h_var)))
        prob = cp.Problem(objective, constraints)
        result = prob.solve(solver=cp.SCS, eps=1e-5)
        return np.clip(h_var.value, 0, 1) * target_sum / np.sum(np.clip(h_var.value, 0, 1))
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        # Adaptive perturbation based on time remaining
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.05 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
