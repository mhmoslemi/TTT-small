import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_rate = 0.2
    generations = 50
    
    def genetic_algorithm():
        # Initialize population
        population = np.random.rand(population_size, n_points)
        population = population * target_sum / np.sum(population, axis=1)[:, np.newaxis]
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            best_idx = np.argmin(fitness)
            best_h = population[best_idx].copy()
            
            # Select parents
            probabilities = 1 / (fitness + 1e-6)
            probabilities = probabilities / np.sum(probabilities)
            parents_idx = np.random.choice(population_size, size=population_size, p=probabilities)
            
            # Crossover and mutation
            new_population = []
            for i in range(population_size):
                parent1 = population[parents_idx[i]]
                if np.random.rand() < crossover_rate:
                    parent2 = population[np.random.choice(population_size)]
                    child = np.zeros(n_points)
                    for j in range(n_points):
                        if np.random.rand() < 0.5:
                            child[j] = parent1[j]
                        else:
                            child[j] = parent2[j]
                else:
                    child = parent1.copy()
                
                # Mutation
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[j] += np.random.uniform(-0.1, 0.1)
                        child[j] = np.clip(child[j], 0, 1)
                
                # Maintain sum constraint
                child = child * target_sum / np.sum(child)
                new_population.append(child)
            
            # Replace population with new population
            population = np.array(new_population)
            
            # Keep elite
            elite_idx = np.argsort(fitness)[:int(elite_rate * population_size)]
            population[elite_idx] = np.array([population[i].copy() for i in elite_idx])
        
        return best_h, fitness[best_idx]
    
    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm()
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Perturb and refine
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
