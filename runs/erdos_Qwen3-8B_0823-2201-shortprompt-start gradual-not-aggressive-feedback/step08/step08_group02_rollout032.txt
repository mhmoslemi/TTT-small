import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def genetic_algorithm():
        population_size = 20
        generations = 50
        mutation_rate = 0.1
        elite_size = 3
        
        population = np.random.uniform(0, 1, (population_size, n_points))
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            
            # Select elites
            elite_indices = np.argsort(fitness)[:elite_size]
            elites = population[elite_indices]
            
            # Create new population
            new_population = elites.copy()
            
            # Crossover and mutation
            for _ in range(population_size - elite_size):
                parents = np.random.choice(elite_indices, size=2, replace=False)
                parent1, parent2 = population[parents[0]], population[parents[1]]
                child = (parent1 + parent2) / 2
                
                # Mutation
                if np.random.rand() < mutation_rate:
                    mutation = np.random.uniform(-0.1, 0.1, n_points)
                    child += mutation
                    child = np.clip(child, 0, 1)
                    child = child * target_sum / np.sum(child)
                
                new_population = np.vstack((new_population, child))
            
            population = new_population
            population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        best_h = population[np.argmin(fitness)]
        return best_h, np.min(fitness)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:

        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)

            perturbed_h,
            method='SLSQP',

        current_c5 = objective(current_h)

            best_h = current_h
            best_c5 = current_c5
