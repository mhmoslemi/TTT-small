import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.15
    elite_size = 3
    
    def generate_population(population_size):
        population = []
        for _ in range(population_size):
            candidate = np.random.rand(n_points)
            candidate = candidate * target_sum / np.sum(candidate)
            population.append(candidate)
        return population
    
    def evaluate_population(population):
        return [objective(ind) for ind in population]
    
    def select_elite(population, fitness):
        sorted_indices = np.argsort(fitness)
        elites = [population[i] for i in sorted_indices[:elite_size]]
        return elites
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        indices = np.random.choice(n_points, size=int(n_points/2), replace=False)
        child[indices] = parent2[indices]
        return child
    
    def mutate(individual):
        mutation = np.random.normal(0, 0.05, n_points)
        mutated = individual + mutation
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * target_sum / np.sum(mutated)
        return mutated
    
    # Initial population
    population = generate_population(population_size)
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitness = evaluate_population(population)
        elites = select_elite(population, fitness)
        
        # Generate new population
        new_population = elites[:]
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(elites, size=2, replace=False)
            child = crossover(parent1, parent2)
            if np.random.rand() < mutation_rate:
                child = mutate(child)
            new_population.append(child)
        
        population = new_population
    
    # Find best solution in final population
    best_h = min(population, key=objective)
    best_c5 = objective(best_h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.1 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
