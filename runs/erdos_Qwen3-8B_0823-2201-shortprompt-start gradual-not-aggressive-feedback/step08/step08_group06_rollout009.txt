import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Greedy refinement with dynamic perturbation
    def greedy_refinement(h, budget_s):
        start_time = time.time()
        remaining_time = budget_s
        perturb_scale = 0.15
        iterations = 0
        
        while remaining_time > 0:
            # Dynamic perturbation
            perturb = np.random.normal(0, perturb_scale, n_points)
            perturbed = np.clip(h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            
            # Local optimization
            result = opt.minimize(
                objective,
                perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            # Update best solution
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = objective(current_h)
            
            if current_c5 < objective(h):
                h = current_h
            
            # Adapt perturbation scale
            perturb_scale *= 0.95
            iterations += 1
            remaining_time = budget_s - (time.time() - start_time)
        
        return h, objective(h)
    
    # Run the refinement
    best_h, best_c5 = greedy_refinement(h, budget_s)
    
    return best_h, best_c5, n_points
