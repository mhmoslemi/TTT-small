import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Simulated annealing with adaptive cooling
    def simulated_annealing():
        T = 1.0
        cool_rate = 0.995
        max_iter = 1000
        best_h = h.copy()
        best_c5 = objective(best_h)
        
        for _ in range(max_iter):
            T *= cool_rate
            if T < 1e-5:
                break
            # Better perturbation with adaptive magnitude
            perturb = np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            current_c5 = objective(perturbed)
            
            if current_c5 < best_c5 or np.random.rand() < np.exp((current_c5 - best_c5)/T):
                best_h = perturbed
                best_c5 = current_c5
        
        return best_h, best_c5
    
    # Genetic algorithm for global search
    def genetic_algorithm(pop_size=20, generations=50, mutation_rate=0.1):
        def crossover(parent1, parent2):
            child = parent1.copy()
            idx = np.random.choice(n_points)
            child[idx] = parent2[idx]
            return child
        
        def mutate(individual):
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    individual[i] += np.random.normal(0, 0.05)
                    individual[i] = np.clip(individual[i], 0, 1)
            individual = individual * target_sum / np.sum(individual)
            return individual
        
        population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
        population = [ind * target_sum / np.sum(ind) for ind in population]
        
        for _ in range(generations):
            fitness = [objective(ind) for ind in population]
            best_idx = np.argmin(fitness)
            best_h = population[best_idx].copy()
            
            new_population = [best_h for _ in range(pop_size // 2)]
            for _ in range(pop_size // 2):
                parent1, parent2 = np.random.choice(population, 2, replace=False)
                child = crossover(parent1, parent2)
                child = mutate(child)
                new_population.append(child)
            
            population = new_population
        
        best_h = np.argmin([objective(ind) for ind in population])
        best_h = population[best_h]
        best_c5 = objective(best_h)
        return best_h, best_c5

    best_h, best_c5 = simulated_annealing()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Genetic algorithm refinement
    if time.time() - start_time < budget_s * 0.5:
        best_h, best_c5 = genetic_algorithm()
        best_h = local_refinement(best_h)
        best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:

        remaining_time = budget_s - (time.time() - start_time)

        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )

        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
