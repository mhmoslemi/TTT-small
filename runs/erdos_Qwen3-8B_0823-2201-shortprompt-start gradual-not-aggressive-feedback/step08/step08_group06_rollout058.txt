import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Coordinate descent with dynamic normalization
    def coordinate_descent(h):
        for _ in range(100):
            for i in range(n_points):
                h[i] += np.random.uniform(-0.1, 0.1)
                h = np.clip(h, 0, 1)
                h = h * target_sum / np.sum(h)
            current_c5 = objective(h)
            if current_c5 < objective(h):
                h = h
            else:
                h -= np.random.uniform(-0.1, 0.1)
                h = np.clip(h, 0, 1)
                h = h * target_sum / np.sum(h)
        return h
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Adaptive perturbation with decreasing scale
    def adaptive_perturbation(h):
        perturb_scale = 0.15
        for _ in range(100):
            perturb = np.random.uniform(-perturb_scale, perturb_scale, n_points)
            perturbed = np.clip(h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            if objective(perturbed) < objective(h):
                h = perturbed
            perturb_scale *= 0.995
        return h
    
    start_time = time.time()
    best_h = coordinate_descent(h.copy())
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement with adaptive perturbation
    while time.time() - start_time < budget_s:
        best_h = adaptive_perturbation(best_h)
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
