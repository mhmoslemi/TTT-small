import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x

    def genetic_algorithm(population_size=20, generations=50, mutation_rate=0.1):
        population = [np.random.rand(n_points) for _ in range(population_size)]

            # Evaluate fitness
            fitness = np.array([objective(h) for h in population])
            
            # Select parents
            parents = np.argsort(fitness)[:population_size//2]
            parents = np.array(population)[parents]

            offspring = []

                        child[i] = parent1[i]

                        child[i] = parent2[i]

            population = np.array(parents + offspring)
