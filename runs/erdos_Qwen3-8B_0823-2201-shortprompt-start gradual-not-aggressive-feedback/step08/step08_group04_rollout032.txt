import scipy.optimize as opt
import time
import random

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

                break

            for h in population:
                # Mutation
                mutated = h.copy()
                for i in range(n_points):
                    if random.random() < 0.1:
                        mutated[i] += np.random.uniform(-0.2, 0.2)
                        mutated[i] = np.clip(mutated[i], 0, 1)
                # Crossover
                crossover_point = np.random.randint(1, n_points-1)
                child = np.concatenate((mutated[:crossover_point], initial_h[crossover_point:]))
