import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Structured perturbation with decreasing magnitude
    def structured_perturbation(h, scale):
        perturb = np.zeros(n_points)
        indices = np.argsort(np.abs(h - 0.5))
        perturb[indices[:3]] = np.random.uniform(-scale, scale, 3)
        return np.clip(h + perturb, 0, 1)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    # Main optimization loop
    for step in range(100):
        if time.time() - start_time > budget_s:
            break

        best_h = local_refinement(best_h)
        best_c5 = objective(best_h)

        # Accept better solution
        if current_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = current_c5
