import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with SLSQP
    result = opt.minimize(
        objective,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Greedy improvement with adaptive focus and multiple perturbations
    start_time = time.time()
    max_iterations = int(budget_s * 10)
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Identify the worst overlap
        h = best_h.copy()
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        shift = max_idx - n_points + 1
        
        # Create multiple perturbation masks
        masks = []
        if shift >= 0:
            masks.append(np.where(np.arange(n_points) >= shift, 0.5, 1.0))
            masks.append(np.where(np.arange(n_points) >= shift, 0.75, 1.0))
            masks.append(np.where(np.arange(n_points) >= shift, 0.25, 1.0))
        else:
            masks.append(np.where(np.arange(n_points) <= shift, 0.5, 1.0))
            masks.append(np.where(np.arange(n_points) <= shift, 0.75, 1.0))
            masks.append(np.where(np.arange(n_points) <= shift, 0.25, 1.0))
        
        # Try each mask
        for mask in masks:
            h_perturbed = h * mask
            h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
            current_c5 = objective(h_perturbed)
            if current_c5 < best_c5:
                best_h = h_perturbed
                best_c5 = current_c5
                print(f"Improved bound: {best_c5}")
    
    return best_h, best_c5, n_points
