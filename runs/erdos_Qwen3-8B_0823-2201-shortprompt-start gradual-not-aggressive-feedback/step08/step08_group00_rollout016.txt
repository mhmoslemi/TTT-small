        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    crossover_rate = 0.8

        return objective(individual)

        return np.random.choice(population, size=2, p=probabilities)

        mutation = np.random.normal(0, 0.05, n_points)

        mutated = mutated * target_sum / np.sum(mutated)
        return mutated
    
    def crossover(parent1, parent2):
        if np.random.rand() < crossover_rate:
            point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:point], parent2[point:]])
            child = child * target_sum / np.sum(child)
            return child
        return parent1

                    child = mutate(child)
