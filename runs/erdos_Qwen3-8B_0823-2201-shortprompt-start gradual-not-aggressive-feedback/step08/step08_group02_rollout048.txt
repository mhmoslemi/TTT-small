import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 2
    
    def fitness(h):
        return objective(h)
    
    def mutate(individual):
        mutated = individual.copy()
        for i in range(len(mutated)):
            if np.random.rand() < mutation_rate:
                mutated[i] += np.random.uniform(-0.1, 0.1)
                mutated[i] = np.clip(mutated[i], 0, 1)
        return mutated
    
    def crossover(parent1, parent2):
        child = parent1.copy()
        for i in range(len(child)):
            if np.random.rand() < crossover_rate:
                child[i] = parent2[i]
        return child
    
    def select_parents(population, fitnesses):
        tournament = np.random.choice(len(population), size=5, replace=False)
        best = tournament[np.argmin(fitnesses[tournament])]
        return population[best].copy()
    
    # Initialize population
    population = [h.copy() for _ in range(population_size)]
    fitnesses = [fitness(ind) for ind in population]
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitnesses = [fitness(ind) for ind in population]
        
        # Select elites
        elite_indices = np.argsort(fitnesses)[:elite_size]
        elites = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elites.copy()
        while len(new_population) < population_size:
            parent1 = select_parents(population, fitnesses)
            parent2 = select_parents(population, fitnesses)
            child = crossover(parent1, parent2)
            child = mutate(child)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        population = new_population
    
    # Local refinement with SLSQP on best candidate
    best_h = np.array(population[np.argmin(fitnesses)], dtype=np.float64)
    best_h = best_h * target_sum / np.sum(best_h)
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
