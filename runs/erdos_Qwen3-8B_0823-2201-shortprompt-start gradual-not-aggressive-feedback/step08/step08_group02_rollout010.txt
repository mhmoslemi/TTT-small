import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def genetic_algorithm():
        population_size = 20
        generations = 50
        mutation_rate = 0.1
        elite_size = 3
        
        population = [h + np.random.uniform(-0.1, 0.1, n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(h) for h in population])
            
            # Select elites
            elite_indices = np.argsort(fitness)[:elite_size]
            elites = [population[i] for i in elite_indices]
            
            # Select parents
            parent_indices = np.random.choice(population_size, size=population_size, replace=True)
            parents = [population[i] for i in parent_indices]
            
            # Crossover
            offspring = []
            for i in range(population_size):
                p1 = parents[i]
                p2 = parents[np.random.choice(range(population_size), replace=False)]
                child = np.copy(p1)
                crossover_point = np.random.randint(1, n_points - 1)
                child[:crossover_point] = p2[:crossover_point]
                offspring.append(child)
            
            # Mutation
            for i in range(population_size):
                if np.random.rand() < mutation_rate:
                    mutation = np.random.uniform(-0.1, 0.1, n_points)
                    offspring[i] += mutation
            
            # Normalize and clip
            offspring = [h * target_sum / np.sum(h) for h in offspring]
            offspring = [np.clip(h, 0, 1) for h in offspring]
            
            # Replace population
            population = elites + offspring[:population_size - elite_size]
        
        best_h = min(population, key=objective)
        return best_h, objective(best_h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)

        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb

        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )

            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
