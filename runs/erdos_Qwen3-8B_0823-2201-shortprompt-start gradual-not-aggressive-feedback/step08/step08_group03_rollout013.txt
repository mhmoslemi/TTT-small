import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize with given values
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 50
    
    # Simulated annealing parameters
    T = 1.0
    cool_rate = 0.995
    max_iter = 200
    
    def genetic_algorithm():
        population = [h + np.random.normal(0, 0.1, n_points) for _ in range(population_size)]
        population = [np.clip(p * target_sum / np.sum(p), 0, 1) for p in population]
        best_h = h.copy()
        best_c5 = objective(best_h)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = [objective(p) for p in population]
            best_idx = np.argmin(fitness)
            best_c5 = fitness[best_idx]
            
            # Selection
            selected = np.random.choice(len(population), size=population_size, p=np.array(fitness)/np.sum(fitness))
            selected = [population[i] for i in selected]
            
            # Crossover
            offspring = []
            for i in range(0, population_size, 2):
                if np.random.rand() < crossover_rate:
                    p1, p2 = selected[i], selected[i+1]
                    crossover_point = np.random.randint(1, n_points-1)
                    child1 = np.concatenate((p1[:crossover_point], p2[crossover_point:]))
                    child2 = np.concatenate((p2[:crossover_point], p1[crossover_point:]))
                    offspring.append(child1)
                    offspring.append(child2)
                else:
                    offspring.append(selected[i])
                    offspring.append(selected[i+1])
            
            # Mutation
            for i in range(len(offspring)):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        offspring[i][j] += np.random.normal(0, 0.05)
                        offspring[i][j] = np.clip(offspring[i][j], 0, 1)
                        offspring[i][j] = offspring[i][j] * target_sum / np.sum(offspring[i])
            
            # Replace population
            population = [np.clip(p * target_sum / np.sum(p), 0, 1) for p in offspring]
            
            # Update best solution
            if fitness[best_idx] < best_c5:
                best_h = population[best_idx]
                best_c5 = fitness[best_idx]
        
        return best_h, best_c5

        T = 1.0
        cool_rate = 0.995
        max_iter = 1000
        best_h = h_start.copy()
        best_c5 = objective(best_h)
        
        for _ in range(max_iter):
            T *= cool_rate
            if T < 1e-5:
                break

            perturb = np.random.normal(0, 0.1 * (T ** 0.5), n_points)

                best_c5 = current_c5
                h_start = best_h

        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,

            constraints={'type': 'eq', 'fun': constraint_sum},

        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
