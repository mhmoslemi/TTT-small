        c5 = np.max(corr) * dx
        return c5

    c2 = 1.494  # social coefficient

    # Initialize particle swarm
    particles = np.random.uniform(0, 1, (pop_size, n_points))
    velocities = np.random.uniform(-1, 1, (pop_size, n_points))
    personal_best = particles.copy()
    personal_best_fitness = np.full(pop_size, np.inf)
    global_best = particles[0].copy()
    global_best_fitness = np.inf

    start_time = time.time()

        if time.time() - start_time > budget_s:
            break

                personal_best_fitness[i] = fitness
