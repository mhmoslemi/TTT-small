import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]

    def genetic_algorithm(pop_size=20, generations=50, mutation_rate=0.1):
        population = np.random.rand(pop_size, n_points)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            fitness = np.array([objective(p) for p in population])
            elite = population[np.argmin(fitness)]
            
            # Select parents
            indices = np.argsort(fitness)
            parents = population[indices[:int(pop_size/2)]]
            
            # Crossover
            offspring = []
            for _ in range(pop_size - len(parents)):
                parent1, parent2 = np.random.choice(parents, size=2, replace=False)
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                offspring.append(child)
            
            # Mutate
            for i in range(len(offspring)):
                mutation = np.random.normal(0, mutation_rate, n_points)
                offspring[i] += mutation
                offspring[i] = np.clip(offspring[i], 0, 1)
                offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
            
            population = np.vstack([parents, offspring])
            population = np.unique(population, axis=0)
            
            if len(population) > pop_size:
                population = population[:pop_size]
        
        best_h = elite
        best_c5 = objective(best_h)
        return best_h, best_c5

    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)

    while time.time() - start_time < budget_s:

        remaining_time = budget_s - (time.time() - start_time)
