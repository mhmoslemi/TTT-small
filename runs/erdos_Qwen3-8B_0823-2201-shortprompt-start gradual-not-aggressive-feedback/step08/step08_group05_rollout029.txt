import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    pop_size = 20
    generations = 50

    def genetic_algorithm():
        population = np.random.rand(pop_size, n_points)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            
            # Select parents
            parent_indices = np.random.choice(pop_size, size=pop_size, replace=False)
            parents = population[parent_indices]
            
            # Crossover
            offspring = np.zeros_like(parents)
            for i in range(pop_size):
                parent1, parent2 = np.random.choice(parents, size=2, replace=False)
                crossover_point = np.random.randint(1, n_points - 1)
                offspring[i, :crossover_point] = parent1[:crossover_point]
                offspring[i, crossover_point:] = parent2[crossover_point:]
            
            # Mutation
            for i in range(pop_size):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        offspring[i, j] += np.random.uniform(-0.15, 0.15)
                        offspring[i, j] = np.clip(offspring[i, j], 0, 1)

            offspring_fitness = np.array([objective(ind) for ind in offspring])
            
            # Select next generation
            combined = np.vstack((population, offspring))
            combined_fitness = np.concatenate((fitness, offspring_fitness))
