import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def genetic_algorithm():
        population_size = 20
        generations = 100
        mutation_rate = 0.1
        elite_size = 2
        
        population = [h.copy() for _ in range(population_size)]
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            # Select elites
            elite_indices = np.argsort(fitness)[:elite_size]
            elites = [population[i] for i in elite_indices]
            # Tournament selection
            selected = []
            for _ in range(population_size - elite_size):
                contestants = np.random.choice(population_size, size=5, replace=False)
                selected_idx = contestants[np.argsort(fitness[contestants])][0]
                selected.append(population[selected_idx])
            # Crossover
            offspring = []
            for i in range(0, len(selected) - 1, 2):
                p1 = selected[i]
                p2 = selected[i+1]
                child1 = np.copy(p1)
                child2 = np.copy(p2)
                # Single-point crossover
                point = np.random.randint(1, n_points - 1)
                child1[:point] = p2[:point]
                child2[:point] = p1[:point]
                offspring.append(child1)
                offspring.append(child2)
            # Mutation
            for i in range(len(offspring)):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        offspring[i][j] += np.random.uniform(-0.1, 0.1)
                        offspring[i][j] = np.clip(offspring[i][j], 0, 1)
            # Replace population
            population = elites + offspring
            # Local refinement
            for i in range(len(population)):
                population[i] = np.array(population[i], dtype=np.float64)
                population[i] = np.clip(population[i], 0, 1)
                population[i] = population[i] * target_sum / np.sum(population[i])
                result = opt.minimize(
                    objective,
                    population[i],
                    method='SLSQP',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 50, 'ftol': 1e-8}

            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
