import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = 128
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial construction with alternating blocks
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 == 0 or i % 4 == 3:
            initial_h[i] = 1.0
        elif i % 4 == 1 or i % 4 == 2:
            initial_h[i] = 0.0
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    
    # Adaptive perturbation and refinement
    def adaptive_refinement(h, budget):
        start_time = time.time()
        remaining = budget
        
        # Initial local refinement
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        h = result.x
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        best_c5 = objective(h)
        
        # Adaptive perturbation loop
        while remaining > 0.01:
            remaining -= 0.01
            # Adaptive perturbation based on current solution
            perturb_size = 0.05 * (1 - best_c5)
            perturb = np.random.normal(0, perturb_size, n_points)
            perturbed = np.clip(h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            
            # Evaluate perturbed solution
            current_c5 = objective(perturbed)
            
            # Accept better solution or with probability based on difference
            if current_c5 < best_c5 or np.random.rand() < np.exp((current_c5 - best_c5)/0.01):

        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},

    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
