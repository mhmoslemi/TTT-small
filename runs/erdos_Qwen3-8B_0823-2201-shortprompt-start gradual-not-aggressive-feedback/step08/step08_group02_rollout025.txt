import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def evolve_population(population, n_gen=10):
        new_population = []
        for _ in range(n_gen):
            # Select parents
            idx1, idx2 = np.random.choice(len(population), 2, replace=False)
            parent1, parent2 = population[idx1], population[idx2]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points-1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            mutation_rate = 0.1
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.uniform(-0.1, 0.1)
                    child[i] = np.clip(child[i], 0, 1)
            
            # Local refinement
            result = opt.minimize(
                objective,
                child,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            new_population.append(result.x)
        
        # Keep best solutions
        population = [np.array(p) for p in population]
        new_population = [np.array(p) for p in new_population]
        all_solutions = population + new_population
        fitness = [objective(p) for p in all_solutions]
        best_indices = np.argsort(fitness)[:len(population)]
        return [all_solutions[i] for i in best_indices]
    
    # Initial population
    population_size = 10
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [p * target_sum / np.sum(p) for p in population]
    
    start_time = time.time()
    best_h = np.array(initial_h, dtype=np.float64)
    best_c5 = objective(best_h)
    
    # Evolve population
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        population = evolve_population(population)
        for p in population:
            current_c5 = objective(p)
            if current_c5 < best_c5:
                best_h = p
                best_c5 = current_c5
    
    # Final refinement
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    return result.x, objective(result.x), n_points
