import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Genetic algorithm
    def genetic_algorithm(pop_size=20, generations=100, mutation_rate=0.1):
        population = np.random.rand(pop_size, n_points)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            fitness = np.array([objective(ind) for ind in population])
            best_idx = np.argmin(fitness)
            best_h = population[best_idx]
            
            # Selection
            selected = np.random.choice(pop_size, size=pop_size, p=1 - fitness / np.max(fitness))
            selected = population[selected]
            
            # Crossover
            offspring = []
            for i in range(0, pop_size, 2):
                if i + 1 < pop_size:
                    parent1 = selected[i]
                    parent2 = selected[i+1]
                    child1 = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                    child2 = np.where(np.random.rand(n_points) < 0.5, parent2, parent1)
                    offspring.extend([child1, child2])
                else:
                    offspring.append(selected[i])
            
            # Mutation

                    offspring[i] = np.clip(offspring[i], 0, 1)
                    offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
            
            # Replace population
            population = np.array(offspring)
        
        return best_h, np.min(fitness)
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()

    # Additional local refinement within budget
    while time.time() - start_time < budget_s:

        remaining_time = budget_s - (time.time() - start_time)

    return best_h, best_c5, n_points
