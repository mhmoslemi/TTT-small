import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    tournament_size = 5
    
    def tournament_selection(population, fitnesses):
        indices = np.random.choice(len(population), tournament_size)
        best_idx = np.argmin(fitnesses[indices])
        return population[indices[best_idx]]
    
    def crossover(parent1, parent2):
        point = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:point], parent2[point:]])
        return child
    
    def mutate(individual):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h_val = individual[i]
                perturb = np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(h_val + perturb, 0, 1)
        return individual
    
    # Evaluate initial population
    population = [h.copy() for _ in range(population_size)]
    fitnesses = [objective(ind) for ind in population]
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    for generation in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Select parents
        parents = [tournament_selection(population, fitnesses) for _ in range(population_size // 2)]
        
        # Crossover
        children = []
        for i in range(0, len(parents), 2):
            parent1 = parents[i]
            parent2 = parents[i+1] if i+1 < len(parents) else parents[0]
            child1 = crossover(parent1, parent2)
            child2 = crossover(parent2, parent1)
            children.extend([child1, child2])
        
        # Mutate
        children = [mutate(child) for child in children]
        
        # Evaluate children
        child_fitnesses = [objective(child) for child in children]
        
        # Select next generation
        combined = list(zip(population, fitnesses, children, child_fitnesses))
        combined.sort(key=lambda x: x[1])
        population = [x[0] for x in combined[:population_size]]
        fitnesses = [x[1] for x in combined[:population_size]]
        
        # Update best solution
        for i in range(len(population)):
            if fitnesses[i] < best_c5:
                best_h = population[i].copy()
                best_c5 = fitnesses[i]
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
