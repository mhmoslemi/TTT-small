import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.15
    tournament_size = 5
    
    def crossover(parent1, parent2):
        """Single-point crossover"""
        point = np.random.randint(1, n_points - 1)
        child = np.zeros(n_points)
        child[:point] = parent1[:point]
        child[point:] = parent2[point:]
        return child
    
    def mutate(individual):
        """Random mutation with probability mutation_rate"""
        if np.random.rand() < mutation_rate:
            idx = np.random.randint(n_points)
            individual[idx] = np.random.rand()
            individual = individual * target_sum / np.sum(individual)
        return individual
    
    # Initialize population
    population = [np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)) for _ in range(population_size)]
    
    # Evaluate initial population
    population_fitness = [objective(ind) for ind in population]
    
    start_time = time.time()
    
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
            
        # Select tournament winners
        tournament = np.random.choice(population_size, tournament_size, replace=False)
        tournament_fitness = [population_fitness[i] for i in tournament]
        winner_idx = tournament[np.argmin(tournament_fitness)]
        best_individual = population[winner_idx]
        best_fitness = population_fitness[winner_idx]
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            # Select parents
            parent1_idx = np.random.randint(population_size)
            parent2_idx = np.random.randint(population_size)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            child1 = crossover(parent1, parent2)
            child2 = crossover(parent2, parent1)
            
            # Mutate
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.append(child1)
            new_population.append(child2)
        
        # Elitism
        new_population.append(best_individual)
        
        # Evaluate new population
        population = new_population
        population_fitness = [objective(ind) for ind in population]
        
        # Local improvement with SLSQP
        for i in range(len(population)):
            result = opt.minimize(
                objective,
                population[i],
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            population[i] = result.x
    
    # Find best solution
    best_h = population[np.argmin(population_fitness)]
    best_c5 = np.min(population_fitness)
    
    return best_h, best_c5, n_points
