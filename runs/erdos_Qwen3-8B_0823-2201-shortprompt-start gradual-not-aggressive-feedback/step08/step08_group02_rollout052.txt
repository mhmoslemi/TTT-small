import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    elite_size = 5
    
    def genetic_algorithm():
        # Initialize population
        population = np.random.rand(population_size, n_points)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        population = np.clip(population, 0, 1)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            
            # Select elite
            elite_indices = np.argsort(fitness)[:elite_size]
            elite = population[elite_indices]
            
            # Tournament selection
            tournament_size = 5
            selected = []
            for _ in range(population_size - elite_size):
                tournament = np.random.choice(population_size, tournament_size)
                winner = tournament[np.argmin(fitness[tournament])]
                selected.append(population[winner])
            
            # Crossover
            crossover_points = np.random.choice(n_points, size=2, replace=False)
            for i in range(0, population_size - elite_size, 2):
                parent1 = selected[i]
                parent2 = selected[i+1]
                child1 = np.copy(parent1)
                child2 = np.copy(parent2)
                child1[crossover_points[0]:crossover_points[1]] = parent2[crossover_points[0]:crossover_points[1]]
                child2[crossover_points[0]:crossover_points[1]] = parent1[crossover_points[0]:crossover_points[1]]
                selected[i] = child1
                selected[i+1] = child2
            
            # Mutation
            for i in range(population_size - elite_size):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        selected[i][j] = np.clip(selected[i][j] + np.random.uniform(-0.1, 0.1), 0, 1)
            
            # Replace population
            population = np.vstack((elite, selected))
            population = population * target_sum / np.sum(population, axis=1, keepdims=True)
            population = np.clip(population, 0, 1)
        
        # Find best solution
        best_h = population[np.argmin([objective(ind) for ind in population])]
        best_c5 = objective(best_h)
        return best_h, best_c5
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.1 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
