import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Genetic algorithm with adaptive mutation
    def genetic_search():
        population_size = 20
        mutation_rate = 0.1
        max_generations = 50
        best_h = h.copy()
        best_c5 = objective(best_h)
        
        for _ in range(max_generations):
            population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
            population = [h * target_sum / np.sum(h) for h in population]
            population_c5 = [objective(h) for h in population]
            
            # Select best individuals
            idxs = np.argsort(population_c5)
            population = [population[idxs[i]] for i in range(population_size)]
            population_c5 = [population_c5[idxs[i]] for i in range(population_size)]
            
            # Mutate and recombine
            new_population = []
            for i in range(population_size):
                if np.random.rand() < mutation_rate:
                    # Mutation
                    perturb = np.random.normal(0, 0.15, n_points)
                    child = np.clip(population[i] + perturb, 0, 1)
                    child = child * target_sum / np.sum(child)
                else:
                    # Recombination
                    j = np.random.randint(0, population_size)
                    child = (population[i] + population[j]) / 2
                    child = child * target_sum / np.sum(child)
                new_population.append(child)
            
            population = new_population
            population_c5 = [objective(h) for h in population]
            
            # Update best
            idx = np.argmin(population_c5)
            current_h = population[idx]
            current_c5 = population_c5[idx]
            
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        
        return best_h, best_c5
    
    start_time = time.time()
    best_h, best_c5 = genetic_search()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        # Random perturbation with adaptive scale
        remaining_time = budget_s - (time.time() - start_time)
        perturb_scale = 0.15 * (remaining_time / budget_s)
        perturb = np.random.uniform(-perturb_scale, perturb_scale, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},

        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
