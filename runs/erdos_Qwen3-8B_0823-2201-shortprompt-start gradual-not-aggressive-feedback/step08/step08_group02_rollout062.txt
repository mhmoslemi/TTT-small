import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    tournament_size = 5
    elite_size = 1
    
    def fitness(individual):
        return -objective(individual)  # Minimize objective
    
    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    # Evaluate population
    def evaluate(population):
        return [fitness(ind) for ind in population]
    
    # Tournament selection
    def tournament_selection(population, fitnesses):
        tournament = np.random.choice(range(len(population)), tournament_size, replace=False)
        best_idx = tournament[np.argmax(fitnesses[tournament])]
        return population[best_idx]
    
    # Crossover
    def crossover(parent1, parent2):
        alpha = np.random.rand(n_points)
        return alpha * parent1 + (1 - alpha) * parent2
    
    # Mutation
    def mutate(individual):
        mutation = np.random.normal(0, mutation_rate, n_points)
        mutated = individual + mutation
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * target_sum / np.sum(mutated)
        return mutated
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate population
        fitnesses = evaluate(population)
        
        # Select elites
        elite_indices = np.argsort(fitnesses)[-elite_size:]
        elites = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elites.copy()
        
        # Generate offspring
        for _ in range(population_size - elite_size):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
        
        # Update best solution
        current_fitnesses = evaluate(population)
        best_idx = np.argmax(current_fitnesses)
        current_h = population[best_idx]
        current_c5 = -current_fitnesses[best_idx]
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Final refinement
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
