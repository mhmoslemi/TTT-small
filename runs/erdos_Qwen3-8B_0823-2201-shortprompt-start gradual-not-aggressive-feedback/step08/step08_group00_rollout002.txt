import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize population
    population_size = 30
    population = [np.array(initial_h, dtype=np.float64) * target_sum / np.sum(initial_h) for _ in range(population_size)]
    
    def evaluate_individual(individual):
        return objective(individual)
    
    def mutate(individual, mutation_rate=0.1):
        mutated = individual.copy()
        for i in range(len(mutated)):
            if random.random() < mutation_rate:
                mutated[i] += np.random.uniform(-0.1, 0.1)
                mutated[i] = np.clip(mutated[i], 0, 1)
        mutated = mutated * target_sum / np.sum(mutated)
        return mutated
    
    def crossover(parent1, parent2):
        child = parent1.copy()
        for i in range(len(child)):
            if random.random() < 0.5:
                child[i] = parent2[i]
        child = child * target_sum / np.sum(child)
        return child
    
    start_time = time.time()
    best_h = np.array(initial_h, dtype=np.float64) * target_sum / np.sum(initial_h)
    best_c5 = objective(best_h)
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Evaluate population
        fitness = [evaluate_individual(ind) for ind in population]
        # Select parents
        parents = np.random.choice(population, size=population_size, p=np.array(fitness)/np.sum(fitness))
        # Generate new population
        new_population = []
        for i in range(population_size//2):
            parent1 = parents[i]
            parent2 = parents[population_size - 1 - i]
            child1 = crossover(parent1, parent2)
            child1 = mutate(child1)
            new_population.append(child1)
            child2 = crossover(parent2, parent1)
            child2 = mutate(child2)
            new_population.append(child2)
        # Add best individual from previous generation
        new_population.append(best_h)
        population = new_population
        # Update best solution
        for ind in population:
            current_c5 = evaluate_individual(ind)
            if current_c5 < best_c5:
                best_h = ind
                best_c5 = current_c5
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
