import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Genetic algorithm for global search
    def genetic_algorithm():
        population_size = 20
        max_generations = 50
        mutation_rate = 0.1
        elite_size = 2
        
        def crossover(parent1, parent2):
            child = np.copy(parent1)
            indices = np.random.choice(n_points, n_points // 2, replace=False)
            child[indices] = parent2[indices]
            return child
        
        def mutate(individual):
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    h_val = individual[i]
                    perturb = np.random.uniform(-0.1, 0.1)
                    individual[i] = np.clip(h_val + perturb, 0, 1)
            return individual
        
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        
        for generation in range(max_generations):
            fitness = [objective(ind) for ind in population]
            elite_indices = np.argsort(fitness)[:elite_size]
            elites = [population[i] for i in elite_indices]
            
            # Select parents
            parents = np.random.choice(population, size=population_size, replace=True)
            
            # Crossover and mutate
            offspring = []
            for parent1, parent2 in zip(parents[::2], parents[1::2]):
                child = crossover(parent1, parent2)
                child = mutate(child)
                offspring.append(child)
            
            # Combine elites and offspring
            population = elites + offspring
            population = [h * target_sum / np.sum(h) for h in population]
            
            # Early stopping
            if generation > 0 and np.abs(fitness[elite_indices[0]] - fitness[elite_indices[1]]) < 1e-6:
                break
        
        best_h = population[elite_indices[0]]
        best_c5 = objective(best_h)
        return best_h, best_c5
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        # Adaptive perturbation based on time remaining
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.05 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}

        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
