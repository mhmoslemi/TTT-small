        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    h = np.array(initial_h, dtype=np.float64)

    def normalize(h):
        h = h.reshape(n_points)
        return h * target_sum / np.sum(h)
    
    # Greedy refinement
    def greedy_refinement(h):
        for _ in range(100):
            best_h = h.copy()
            best_c5 = objective(best_h)

                for delta in [-0.1, 0.1]:
                    perturbed = h.copy()
                    perturbed[i] += delta
                    perturbed = np.clip(perturbed, 0, 1)
                    perturbed = normalize(perturbed)
                    current_c5 = objective(perturbed)
                    if current_c5 < best_c5:
                        best_h = perturbed
                        best_c5 = current_c5
            if np.allclose(best_h, h):
                break
            h = best_h
        return h
    
    # Simulated annealing with impact-based perturbation
    def simulated_annealing():
        T = 1.0
        cool_rate = 0.995
        max_iter = 1000
        best_h = h.copy()
        best_c5 = objective(best_h)
        
        for _ in range(max_iter):
            T *= cool_rate
            if T < 1e-5:
                break
            # Impact-based perturbation
            weights = np.abs(np.correlate(best_h, 1 - best_h, mode="full"))
            weights = weights[weights > 0]
            if len(weights) == 0:
                continue

                best_c5 = current_c5
        
        return best_h, best_c5

    while time.time() - start_time < budget_s:

        perturbed_h = np.clip(perturbed_h, 0, 1)

            best_c5 = current_c5
    
    return best_h, best_c5, n_points
