import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Genetic algorithm with local refinement
    def genetic_search():
        population_size = 16
        generations = 20
        mutation_rate = 0.1
        best_h = h.copy()
        best_c5 = objective(best_h)
        
        for _ in range(generations):
            population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
            population = [h * target_sum / np.sum(h) for h in population]
            fitness = [objective(h) for h in population]
            best_idx = np.argmin(fitness)
            best_h = population[best_idx]
            best_c5 = fitness[best_idx]
            
            # Create new population
            new_population = []
            for i in range(population_size):
                parent1 = population[np.random.randint(population_size)]
                parent2 = population[np.random.randint(population_size)]
                child = np.clip(parent1 * 0.5 + parent2 * 0.5, 0, 1)
                if np.random.rand() < mutation_rate:
                    mutation = np.random.uniform(-0.15, 0.15, n_points)
                    child += mutation
                    child = np.clip(child, 0, 1)
                new_population.append(child)
            
            population = new_population
    
        return best_h, best_c5
    
    start_time = time.time()
    best_h, best_c5 = genetic_search()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        # Sparse random perturbation
        remaining_time = budget_s - (time.time() - start_time)
        perturb_scale = 0.15 * (remaining_time / budget_s)
        perturb_indices = np.random.choice(n_points, size=3, replace=False)
        perturb = np.zeros(n_points)
        perturb[perturb_indices] = np.random.uniform(-perturb_scale, perturb_scale, 3)
        perturbed_h = best_h + perturb

        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
