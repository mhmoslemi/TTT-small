import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    # Evolutionary strategy with adaptive mutations
    def evolutionary_strategy():
        nonlocal best_h, best_c5
        for _ in range(50):
            if time.time() - start_time >= budget_s:
                break
            # Adaptive mutation scale
            mutation_scale = 0.15 * (budget_s - (time.time() - start_time)) / budget_s
            perturb = np.random.normal(0, mutation_scale, n_points)
            perturbed = np.clip(best_h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            current_c5 = objective(perturbed)
            
            if current_c5 < best_c5:
                best_h = perturbed
                best_c5 = current_c5
                # Local refinement
                best_h = local_refinement(best_h)
                best_c5 = objective(best_h)
    
    evolutionary_strategy()
    
    # Grid search over parameter space
    def grid_search():
        nonlocal best_h, best_c5
        param_range = np.linspace(0.1, 0.9, 10)
        for p in param_range:
            if time.time() - start_time >= budget_s:
                break
            perturb = np.random.normal(p, 0.1, n_points)
            perturbed = np.clip(best_h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            current_c5 = objective(perturbed)
            
            if current_c5 < best_c5:
                best_h = perturbed
                best_c5 = current_c5
                # Local refinement
                best_h = local_refinement(best_h)
                best_c5 = objective(best_h)
    
    grid_search()
    
    return best_h, best_c5, n_points
