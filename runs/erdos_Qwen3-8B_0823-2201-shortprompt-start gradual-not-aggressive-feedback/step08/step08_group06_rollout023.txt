import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Simulated annealing with adaptive cooling
    def simulated_annealing():
        T = 1.0
        cool_rate = 0.995
        max_iter = 500
        best_h = h.copy()
        best_c5 = objective(best_h)
        
        for _ in range(max_iter):
            T *= cool_rate
            if T < 1e-5:
                break
            # Adaptive perturbation
            perturb = np.random.normal(0, 0.15 * (1 - _/max_iter), n_points)
            perturbed = np.clip(h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            current_c5 = objective(perturbed)
            
            if current_c5 < best_c5 or np.random.rand() < np.exp((current_c5 - best_c5)/T):
                best_h = perturbed
                best_c5 = current_c5
        
        return best_h, best_c5
    
    # Differential evolution for global optimization
    def differential_evolution():

        max_iter = 100
        F = 0.8
        CR = 0.9

        best_h = pop[0].copy()

            for i in range(pop_size):
                idxs = [j for j in range(pop_size) if j != i]
                a, b, c = pop[np.random.choice(idxs, 3, replace=False)]

                mutant = mutant * target_sum / np.sum(mutant)

                j_rand = np.random.randint(n_points)
                trial[j_rand:] = mutant[j_rand:]
                trial = np.clip(trial, 0, 1)
                trial = trial * target_sum / np.sum(trial)
                trial_c5 = objective(trial)
                
                if trial_c5 < best_c5 or np.random.rand() < CR:

                    if trial_c5 < best_c5:
                        best_h = trial
                        best_c5 = trial_c5

    best_h, best_c5 = simulated_annealing()

        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:

        remaining_time = budget_s - (time.time() - start_time)

        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
