import scipy.optimize as opt
import time

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    crossover_rate = 0.8
    
    def genetic_algorithm():
        population = [np.random.rand(n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]

            fitness = np.array([objective(h) for h in population])
            best_idx = np.argmin(fitness)
            best_h = population[best_idx]
            
            # Selection
            selected = np.random.choice(population_size, size=population_size, p=1 - fitness/np.max(fitness))
            selected = [population[i] for i in selected]
            
            # Crossover
            new_population = []
            for i in range(population_size):
                if np.random.rand() < crossover_rate:
                    p1 = selected[i]

                    child = np.random.rand(n_points)
