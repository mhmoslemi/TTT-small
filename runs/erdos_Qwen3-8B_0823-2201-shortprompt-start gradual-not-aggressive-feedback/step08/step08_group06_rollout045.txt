import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Evolutionary algorithm
    def evolutionary_search():
        population_size = 16
        mutation_rate = 0.3
        elite_size = 4
        max_generations = 100
        population = [h.copy() for _ in range(population_size)]
        
        for _ in range(max_generations):
            # Evaluate population
            fitness = [objective(ind) for ind in population]
            # Select elites
            elite_indices = np.argsort(fitness)[:elite_size]
            elites = [population[i] for i in elite_indices]
            # Generate new population
            new_population = elites.copy()
            for _ in range(population_size - elite_size):
                # Select parents
                parent1 = np.random.choice(elites)
                parent2 = np.random.choice(elites)
                # Crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                # Mutation
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] += np.random.uniform(-0.15, 0.15)
                        child[i] = np.clip(child[i], 0, 1)
                # Normalize
                child = child * target_sum / np.sum(child)
                new_population.append(child)
            population = new_population
        
        best_h = min(population, key=objective)
        return best_h, objective(best_h)
    
    start_time = time.time()
    best_h, best_c5 = evolutionary_search()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        # Random perturbation with adaptive scale
        remaining_time = budget_s - (time.time() - start_time)
        perturb_scale = 0.15 * (remaining_time / budget_s)
        perturb = np.random.uniform(-perturb_scale, perturb_scale, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
