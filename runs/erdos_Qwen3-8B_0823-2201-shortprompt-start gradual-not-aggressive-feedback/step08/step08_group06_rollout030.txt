import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Greedy improvement with adaptive perturbations
    def greedy_improvement(h):
        for _ in range(100):
            # Adaptive perturbation
            perturb = np.random.normal(0, 0.15 * (1 - _/100), n_points)
            perturbed = np.clip(h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            current_c5 = objective(perturbed)
            
            if current_c5 < objective(h):
                h = perturbed
        
        return h
    
    # Randomized search with adaptive parameters
    def randomized_search(h):
        best_h = h.copy()
        best_c5 = objective(best_h)
        T = 1.0
        cool_rate = 0.995
        max_iter = 500
        
        for _ in range(max_iter):
            T *= cool_rate
            if T < 1e-5:
                break
            # Adaptive perturbation
            perturb = np.random.normal(0, 0.15 * (1 - _/max_iter), n_points)
            perturbed = np.clip(h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            current_c5 = objective(perturbed)
            
            if current_c5 < best_c5 or np.random.rand() < np.exp((current_c5 - best_c5)/T):
                best_h = perturbed
                best_c5 = current_c5

    start_time = time.time()
    best_h = local_refinement(h)
    best_h = greedy_improvement(best_h)
    best_c5 = objective(best_h)
    
    # Additional optimization within budget
    while time.time() - start_time < budget_s:

        best_h, best_c5 = randomized_search(best_h)
        # Local refinement
        best_h = local_refinement(best_h)
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
