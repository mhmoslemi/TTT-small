import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    def genetic_algorithm(pop_size=20, generations=50):
        population = np.random.rand(pop_size, n_points)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            fitness = np.array([objective(ind) for ind in population])
            best_idx = np.argmin(fitness)
            best_h = population[best_idx].copy()
            
            # Selection
            selected = population[np.argsort(fitness)[:int(pop_size/2)]]
            
            # Crossover
            offspring = []
            for _ in range(pop_size - int(pop_size/2)):
                parent1, parent2 = np.random.choice(selected, size=2, replace=False)
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                offspring.append(child)
            
            # Mutation

                    if np.random.rand() < 0.1:
                        offspring[i][j] = np.clip(offspring[i][j] + np.random.normal(0, 0.1), 0, 1)

        return best_h, np.min(fitness)
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm(pop_size=20, generations=50)

            best_h = current_h
            best_c5 = current_c5
