import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 20
    mutation_rate = 0.15
    crossover_rate = 0.8
    
    def evolve_population(population):
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        # Select top performers
        elite_indices = np.argsort(fitness)[:population_size//2]
        elites = [population[i] for i in elite_indices]
        
        # Crossover
        new_population = elites.copy()
        for _ in range(population_size//2):
            parent1, parent2 = np.random.choice(elite_indices, 2, replace=False)
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points-1)
                child = np.concatenate([population[parent1][split:], population[parent2][:split]])
            else:
                child = np.random.rand(n_points)
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation = np.random.uniform(-0.1, 0.1, n_points)
                child += mutation
            # Ensure bounds and constraint
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)

            bounds=bounds,

            options={'maxiter': 50, 'ftol': 1e-8}
