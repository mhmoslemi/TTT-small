import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Simulated annealing with adaptive cooling
    def simulated_annealing():
        T = 1.0
        cool_rate = 0.995
        max_iter = 500
        best_h = h.copy()
        best_c5 = objective(best_h)
        
        for _ in range(max_iter):
            T *= cool_rate
            if T < 1e-5:
                break
            # Adaptive perturbation
            perturb = np.random.normal(0, 0.15 * (1 - _/max_iter), n_points)
            perturbed = np.clip(h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            current_c5 = objective(perturbed)
            
            if current_c5 < best_c5 or np.random.rand() < np.exp((current_c5 - best_c5)/T):
                best_h = perturbed
                best_c5 = current_c5
        
        return best_h, best_c5
    
    # Greedy local search
    def greedy_refinement(h):
        h = h.copy()
        for _ in range(100):
            # Find the segment with the highest overlap
            corr = np.correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(corr)
            # Split the segment into two parts
            if max_idx < n_points:
                left = h[:max_idx]
                right = h[max_idx:]
                # Try swapping values between left and right
                for i in range(len(left)):
                    for j in range(len(right)):
                        new_h = h.copy()
                        new_h[i], new_h[j] = new_h[j], new_h[i]
                        new_h = new_h * target_sum / np.sum(new_h)
                        current_c5 = objective(new_h)
                        if current_c5 < objective(h):
                            h = new_h
                            break
                    else:
                        continue
                    break
            else:
                # Handle the full segment case
                for i in range(n_points):
                    for j in range(i+1, n_points):
                        new_h = h.copy()
                        new_h[i], new_h[j] = new_h[j], new_h[i]
                        new_h = new_h * target_sum / np.sum(new_h)
                        current_c5 = objective(new_h)
                        if current_c5 < objective(h):
                            h = new_h
                            break
                    else:

                    break

    start_time = time.time()
    best_h, best_c5 = simulated_annealing()
    best_h = greedy_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        # Random perturbation with adaptive scale
        remaining_time = budget_s - (time.time() - start_time)
        perturb_scale = 0.15 * (remaining_time / budget_s)

            perturbed_h,

        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
