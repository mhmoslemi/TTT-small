import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def genotype_to_phenotype(genotype):
        h = np.array(genotype, dtype=np.float64)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    def phenotype_to_genotype(phenotype):
        return phenotype.copy()
    
    def fitness(individual):
        return objective(genotype_to_phenotype(individual))
    
    def crossover(parent1, parent2):
        size = len(parent1)
        child = np.zeros(size)
        for i in range(size):
            if random.random() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child
    
    def mutate(individual, mutation_rate=0.1):
        size = len(individual)
        for i in range(size):
            if random.random() < mutation_rate:
                individual[i] += np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        return individual
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 100
    elite_size = 5
    mutation_rate = 0.1
    
    # Initialize population
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = np.array([genotype_to_phenotype(ind) for ind in population])
    population = np.array([phenotype_to_genotype(ind) for ind in population])
    
    # Evaluate initial population
    fitness_values = np.array([fitness(ind) for ind in population])
    
    best_fitness = np.min(fitness_values)
    best_individual = population[np.argmin(fitness_values)]
    
    start_time = time.time()
    
    # Genetic algorithm loop
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness_values = np.array([fitness(ind) for ind in population])
        
        # Select elites
        elite_indices = np.argsort(fitness_values)[:elite_size]
        elites = population[elite_indices]
        
        # Select parents
        tournament_size = 5
        parents = []
        for _ in range(population_size - elite_size):
            candidates = np.random.choice(population, size=tournament_size)
            best_candidate = candidates[np.argmin([fitness(ind) for ind in candidates])]
            parents.append(best_candidate)
        
        # Create offspring
        offspring = []
        for i in range(len(parents) // 2):
            parent1 = parents[2*i]
            parent2 = parents[2*i + 1]
            child = crossover(parent1, parent2)
            child = mutate(child, mutation_rate)
            offspring.append(child)
        
        # Combine elites and offspring
        population = np.vstack((elites, offspring))
        population = np.array([genotype_to_phenotype(ind) for ind in population])
        population = np.array([phenotype_to_genotype(ind) for ind in population])
        
        # Update best
        current_fitness_values = np.array([fitness(ind) for ind in population])
        best_index = np.argmin(current_fitness_values)
        if current_fitness_values[best_index] < best_fitness:
            best_fitness = current_fitness_values[best_index]
            best_individual = population[best_index]
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Refine the best solution
    best_individual = local_refinement(best_individual)
    best_c5 = objective(best_individual)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        # Adaptive perturbation based on time remaining
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.1 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_individual + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_individual, best_c5, n_points
