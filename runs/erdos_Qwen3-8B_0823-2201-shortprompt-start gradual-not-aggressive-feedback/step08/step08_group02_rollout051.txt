    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    pop_size = 20
    generations = 100
    mutation_rate = 0.15
    tournament_size = 5
    
    for _ in range(generations):
        # Evaluate population
        population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
        population.append(best_h.copy())
        fitness = [objective(h) for h in population]
        
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            contestants = np.random.choice(range(pop_size), tournament_size, replace=False)
            winner = contestants[np.argmin(fitness[contestants])]
            selected.append(population[winner])
        
        # Mutation
        for i in range(pop_size):
            if np.random.rand() < mutation_rate:
                mutation = np.random.uniform(-0.1, 0.1, n_points)
                selected[i] += mutation
                selected[i] = np.clip(selected[i], 0, 1)
                selected[i] = selected[i] * target_sum / np.sum(selected[i])
        
        # Evaluate and select new population
        fitness = [objective(h) for h in selected]
        population = selected
        fitness = np.array(fitness)

        population = np.array(population)[indices]
        population = population[:pop_size]
        fitness = fitness[indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]

        )
