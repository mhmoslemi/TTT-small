import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    elite_size = 2
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        child[indices] = parent2[indices]
        return child
    
    def mutate(individual):
        mutation = np.random.normal(0, 0.05, n_points)
        individual += mutation
        individual = np.clip(individual, 0, 1)
        individual = individual * target_sum / np.sum(individual)
        return individual
    
    # Initialize population
    population = [h * np.random.rand(n_points) for _ in range(population_size)]
    population = [ind * target_sum / np.sum(ind) for ind in population]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        
        # Select elite
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elite.copy()
        while len(new_population) < population_size:
            # Tournament selection
            contestants = np.random.choice(population_size, size=3, replace=False)
            winner = contestants[np.argmin(fitness[contestants])]
            parent1 = population[winner]
            
            # Second parent
            contestants = np.random.choice(population_size, size=3, replace=False)
            winner = contestants[np.argmin(fitness[contestants])]
            parent2 = population[winner]
            
            # Crossover
            child = crossover(parent1, parent2)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                child = mutate(child)
            
            new_population.append(child)
        
        population = new_population
        
        # Update best solution
        current_best = min(zip(population, fitness), key=lambda x: x[1])
        if current_best[1] < best_c5:
            best_h = current_best[0].copy()
            best_c5 = current_best[1]
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        # Adaptive perturbation based on time remaining
        remaining_time = budget_s - (time.time() - start_time)

        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
