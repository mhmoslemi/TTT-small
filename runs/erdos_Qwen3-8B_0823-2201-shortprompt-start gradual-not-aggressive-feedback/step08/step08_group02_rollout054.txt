import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.15
    elite_ratio = 0.2
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        child[indices] = parent2[indices]
        return child
    
    def mutate(individual):
        mutation = np.random.normal(0, 0.05, n_points)
        individual += mutation
        individual = np.clip(individual, 0, 1)
        individual = individual * target_sum / np.sum(individual)
        return individual
    
    # Initial population
    population = [h.copy() for _ in range(population_size)]
    population = [mutate(ind) for ind in population]
    
    # Genetic algorithm
    start_time = time.time()
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        # Select elites
        elite_indices = np.argsort(fitness)[:int(elite_ratio * population_size)]
        elites = [population[i] for i in elite_indices]
        # Create new population
        new_population = elites.copy()
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(elites, size=2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        population = new_population
    
    # Local refinement with SLSQP
    best_h = min(population, key=objective)
    best_h = np.array(best_h, dtype=np.float64)
    best_h = best_h * target_sum / np.sum(best_h)
    
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.1 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )

        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)

            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
