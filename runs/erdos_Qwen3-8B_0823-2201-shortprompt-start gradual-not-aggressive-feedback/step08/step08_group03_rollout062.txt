import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 100
    
    def genetic_algorithm():
        population = [h + np.random.normal(0, 0.1, n_points) for _ in range(population_size)]
        population = [np.clip(p * target_sum / np.sum(p), 0, 1) for p in population]
        best_h = h.copy()
        best_c5 = objective(best_h)
        
        for _ in range(generations):
            # Evaluate population
            fitness = [objective(p) for p in population]
            best_idx = np.argmin(fitness)
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
            
            # Selection
            parents = np.random.choice(range(population_size), size=population_size, p=np.array(fitness)/np.sum(fitness))
            parents = [population[i] for i in parents]
            
            # Crossover
            offspring = []
            for i in range(0, population_size, 2):
                if np.random.rand() < crossover_rate:
                    p1, p2 = parents[i], parents[i+1]
                    crossover_point = np.random.randint(1, n_points-1)
                    child1 = np.concatenate((p1[:crossover_point], p2[crossover_point:]))
                    child2 = np.concatenate((p2[:crossover_point], p1[crossover_point:]))
                    offspring.append(child1)
                    offspring.append(child2)
                else:
                    offspring.append(parents[i])
                    offspring.append(parents[i+1])

            for i in range(len(offspring)):
                if np.random.rand() < mutation_rate:
                    perturb = np.random.normal(0, 0.1, n_points)

                    offspring[i] = np.clip(offspring[i], 0, 1)
                    offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
            
            # Replace population
            population = [np.clip(p * target_sum / np.sum(p), 0, 1) for p in offspring]

                best_h = population[best_idx].copy()

            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)

            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )

        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
