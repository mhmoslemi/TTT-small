    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child
    
    def mutate(individual):
        mutated = individual.copy()
        for i in range(n_points):
            if np.random.rand() < 0.1:
                mutated[i] += np.random.uniform(-0.1, 0.1)
                mutated[i] = np.clip(mutated[i], 0, 1)
        mutated = mutated * target_sum / np.sum(mutated)
        return mutated
    
    def evaluate(population):
        return np.array([objective(ind) for ind in population])

        tournament_size = 5
        tournament = np.random.choice(len(population), tournament_size, replace=False)
        best_idx = tournament[np.argmin(fitnesses[tournament])]
        return population[best_idx].copy()
    
    start_time = time.time()
