import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial population
    population_size = 20
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    def evaluate(individual):
        return objective(individual)
    
    def crossover(parent1, parent2):
        crossover_point = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child = child * target_sum / np.sum(child)
        return child
    
    def mutate(individual, mutation_rate=0.1):
        mutant = individual.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                mutant[i] = np.random.uniform(0, 1)
        mutant = mutant * target_sum / np.sum(mutant)
        return mutant
    
    start_time = time.time()
    best_h = population[0]
    best_c5 = objective(best_h)
    
    for _ in range(100):  # Generations
        if time.time() - start_time > budget_s:
            break
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        # Select parents
        parents = np.argsort(fitness)[:population_size//2]
        parents = [population[i] for i in parents]
        # Create new population
        new_population = []
        for _ in range(population_size//2):
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        # Replace worst with new individuals
        combined = population + new_population
        combined.sort(key=evaluate)
        population = combined[:population_size]
        # Local refinement
        for ind in population:
            result = opt.minimize(
                objective,
                ind,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            ind[:] = result.x
        # Update best
        current_best = min(population, key=evaluate)
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
