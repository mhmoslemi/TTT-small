import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    def objective(h):

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x

    pop_size = 20
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 2

    def genetic_algorithm():
        population = np.random.rand(pop_size, n_points)
        population = population * target_sum / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            
            # Select elite
            elite_indices = np.argsort(fitness)[:elite_size]
            elite = population[elite_indices]
            
            # Tournament selection
            selected = []
            for _ in range(pop_size - elite_size):
                contestants = np.random.choice(pop_size, size=5, replace=False)
                winner = contestants[np.argmin(fitness[contestants])]
                selected.append(population[winner])

            offspring = []
            for i in range(0, len(selected), 2):
                if i + 1 < len(selected) and np.random.rand() < crossover_rate:
                    parent1 = selected[i]

                else:
                    offspring.append(selected[i])
            
            # Mutation
            for i in range(len(offspring)):
                if np.random.rand() < mutation_rate:
                    perturb = np.random.normal(0, 0.05, n_points)

                    offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])

            population = np.vstack([elite, offspring])
            population = np.unique(population, axis=0)
            population = population[:pop_size]

        fitness = np.array([objective(ind) for ind in population])
        best_index = np.argmin(fitness)
        return population[best_index], fitness[best_index]
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()

    while time.time() - start_time < budget_s:

        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
