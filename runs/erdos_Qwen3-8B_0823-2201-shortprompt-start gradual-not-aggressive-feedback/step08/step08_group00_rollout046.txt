import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial population
    population_size = 20
    population = np.array([np.array(initial_h, dtype=np.float64) for _ in range(population_size)])
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    def genetic_algorithm():
        nonlocal population
        for _ in range(100):  # Generations
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            best_idx = np.argmin(fitness)
            best_h = population[best_idx].copy()
            
            # Selection
            selected = np.random.choice(population_size, size=population_size, p=fitness/np.sum(fitness))
            selected = population[selected]
            
            # Crossover
            offspring = []
            for i in range(population_size//2):
                parent1 = selected[i]
                parent2 = selected[population_size//2 + i]
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                offspring.append(child)
            
            # Mutation
            for i in range(population_size):
                if np.random.rand() < 0.1:
                    mutation = np.random.normal(0, 0.05, n_points)
                    offspring[i] += mutation
                    offspring[i] = np.clip(offspring[i], 0, 1)
                    offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
            
            # Replace population
            population = np.array(offspring)
            population = population * target_sum / np.sum(population, axis=1, keepdims=True)
            
            # Local refinement
            for i in range(population_size):
                result = opt.minimize(
                    objective,
                    population[i],
                    method='SLSQP',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 50, 'ftol': 1e-8}
                )
                population[i] = result.x
                population[i] = np.clip(population[i], 0, 1)
                population[i] = population[i] * target_sum / np.sum(population[i])
        
        return population[np.argmin([objective(ind) for ind in population])]
    
    start_time = time.time()
    best_h = genetic_algorithm()
    best_c5 = objective(best_h)
    
    # Final refinement
    while time.time() - start_time < budget_s:
        # Adaptive perturbation based on time remaining
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.05 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
