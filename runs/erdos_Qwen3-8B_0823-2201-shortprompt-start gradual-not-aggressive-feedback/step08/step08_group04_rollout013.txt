import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Genetic algorithm parameters
    population_size = 50
    mutation_rate = 0.1
    generations = int(budget_s * 5)
    
    # Initialize population
    population = [initial_h.copy() for _ in range(population_size)]
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    for gen in range(generations):
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_c5 = fitness[best_idx]
        
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
        
        # Create new population
        new_population = []
        for _ in range(population_size):
            # Select parents
            parent1 = random.choices(population, weights=fitness)[0]
            parent2 = random.choices(population, weights=fitness)[0]
            
            # Crossover
            child = np.copy(parent1)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = parent2[i]
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] = np.clip(child[i] + np.random.normal(0, 0.1), 0, 1)
            
            # Normalize
            child = child * target_sum / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
    
    return best_h, best_c5, n_points
