import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Genetic algorithm parameters
    pop_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 100
    
    # Initial population
    population = [np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)) for _ in range(pop_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Select parents
        parents = np.random.choice(pop_size, size=pop_size, replace=True)
        parents = np.array(parents)
        
        # Crossover
        offspring = []
        for i in range(pop_size):
            if np.random.rand() < crossover_rate:
                j = np.random.choice(pop_size, size=1, replace=False)[0]
                parent1 = population[parents[i]]
                parent2 = population[parents[j]]
                child = np.zeros(n_points)
                for k in range(n_points):
                    if np.random.rand() < 0.5:
                        child[k] = parent1[k]
                    else:
                        child[k] = parent2[k]
                offspring.append(child)
            else:
                offspring.append(population[parents[i]])
        
        # Mutation
        for i in range(pop_size):
            if np.random.rand() < mutation_rate:
                idx = np.random.choice(n_points, size=3, replace=False)
                mutation = np.random.uniform(-0.15, 0.15, 3)
                offspring[i][idx] += mutation
                offspring[i] = np.clip(offspring[i], 0, 1)
                offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
        
        # Replace population
        population = offspring
    
    # Final local refinement
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
