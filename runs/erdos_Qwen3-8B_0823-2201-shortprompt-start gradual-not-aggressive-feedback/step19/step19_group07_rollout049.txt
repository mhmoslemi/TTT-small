import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize with the provided starting point
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # First stage: Gradient-based refinement
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8}
    )
    h = result.x
    current_c5 = objective(h)
    
    if current_c5 < best_c5:
        best_h = h.copy()
        best_c5 = current_c5
    
    # Second stage: Structured perturbation with Fourier-based patterns
    for perturb_size in np.logspace(-3, -1, num=5, base=10):
        if time.time() - start_time >= budget_s:
            break
        # Create perturbation based on harmonic patterns
        freqs = np.linspace(0.1, 1.0, 5)
        perturb = np.zeros(n_points)
        for freq in freqs:
            perturb += np.sin(2 * np.pi * freq * np.linspace(0, 1, n_points))
        perturb = perturb / np.max(np.abs(perturb))
        perturb *= perturb_size
        
        # Apply perturbation and normalize
        perturbed_h = h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Optimization with perturbed initial guess
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 150, 'ftol': 1e-10, 'eps': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
