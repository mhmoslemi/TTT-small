import numpy as np
import scipy.optimize as opt
import time
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Phase 1: Gradient-based refinement
    for _ in range(3):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-9}
        )
        h = result.x
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    # Phase 2: Adaptive perturbations with decreasing magnitude
    for perturb_size in [0.05, 0.02, 0.01, 0.005, 0.002]:
        if time.time() - start_time >= budget_s:
            break
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-9}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    # Phase 3: Hybrid optimization with genetic algorithm
    if time.time() - start_time < budget_s - 5:
        # Setup genetic algorithm
        creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
        creator.create("Individual", list, fitness=creator.FitnessMin)
        
        toolbox = base.Toolbox()
        toolbox.register("attr_float", np.random.uniform, 0, 1)
        toolbox.register("individual", tools.initRepeat, creator.Individual, 
                        toolbox.attr_float, n=n_points)
        toolbox.register("population", tools.initRepeat, list, toolbox.individual)
        
        toolbox.register("evaluate", objective)
        toolbox.register("mate", tools.cxTwoPoint)
        toolbox.register("mutate", tools.mutGaussian, 
                        mu=0, sigma=0.05, indpb=0.1)
        toolbox.register("select", tools.selTournament, tournsize=3)
        
        population = toolbox.population(n=50)
        
        for gen in range(50):
            if time.time() - start_time >= budget_s:
                break
            offspring = algorithms.varAnd(population, toolbox, cxpb=0.5, mutpb=0.1)
            fits = [toolbox.evaluate(ind) for ind in offspring]
            for fit, ind in zip(fits, offspring):
                ind.fitness.values = (fit,)
            population = toolbox.select(offspring, k=len(population))
        
        # Find best individual
        best_genetic = min(population, key=lambda x: x.fitness.values[0])
        best_genetic = np.clip(best_genetic, 0, 1)
        best_genetic = best_genetic * target_sum / np.sum(best_genetic)
        current_c5 = objective(best_genetic)
        
        if current_c5 < best_c5:
            best_h = best_genetic.copy()
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
