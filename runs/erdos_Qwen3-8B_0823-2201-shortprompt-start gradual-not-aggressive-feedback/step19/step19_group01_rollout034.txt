        h = h.reshape(n_points)

        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    pop_size = 20
    generations = 50
    mutation_rate = 0.1

    population = [np.array(initial_h_values, dtype=np.float64) * np.random.uniform(0.8, 1.2, n_points) for _ in range(pop_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:

        fitness = [objective(h) for h in population]
        
        # Select top performers
        top_indices = np.argsort(fitness)[:int(pop_size/2)]

        # Create new population through crossover and mutation
        new_population = []

                parent1 = np.random.choice(selected)
