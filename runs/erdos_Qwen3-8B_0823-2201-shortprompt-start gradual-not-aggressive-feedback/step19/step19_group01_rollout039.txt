import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Multi-stage optimization with evolutionary search
    for stage in range(3):
        # Gradient-based refinement with tighter tolerances
        def local_refinement(h):
            result = opt.minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 300, 'ftol': 1e-12, 'eps': 1e-9}
            )
            return result.x
        
        h = local_refinement(h)
        current_c5 = objective(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Adaptive perturbation with structured magnitude
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break
        
        # Structured perturbations at decreasing scales
        for perturb_size in [0.05, 0.02, 0.01, 0.005, 0.002]:
            if time.time() - start_time >= budget_s:
                break
            # Generate a population of perturbed solutions
            population = [h + np.random.uniform(-perturb_size, perturb_size, n_points) for _ in range(5)]
            population = [np.clip(p, 0, 1) for p in population]
            population = [p * target_sum / np.sum(p) for p in population]
            
            # Evaluate population
            candidates = []
            for p in population:
                result = opt.minimize(
                    objective,
                    p,
                    method='SLSQP',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-9}
                )
                candidates.append((result.x, objective(result.x)))
            
            # Select best candidate
            candidates.sort(key=lambda x: x[1])
            best_candidate = candidates[0][0]
            
            if objective(best_candidate) < best_c5:
                best_h = best_candidate.copy()
                best_c5 = objective(best_candidate)
    
    return best_h, best_c5, n_points
