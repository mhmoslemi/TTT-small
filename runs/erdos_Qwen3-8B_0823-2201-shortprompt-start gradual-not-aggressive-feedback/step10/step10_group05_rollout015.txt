    crossover_rate = 0.8
    generations = 500
    elite_size = 2
    
    def evolve_population(population):
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = population[elite_indices]
        # Select parents
        parents = population[np.random.choice(len(population), len(population) - elite_size, replace=False)]
        # Crossover
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents) and np.random.rand() < crossover_rate:
                parent1 = parents[i]
                parent2 = parents[i+1]

                        child2[j] = parent2[j]
                    else:
                        child1[j] = parent2[j]
                        child2[j] = parent1[j]
                offspring.append(child1)
                offspring.append(child2)
            else:
                offspring.append(parents[i])
        # Mutate
        for i in range(len(offspring)):

                if np.random.rand() < mutation_rate:

        # Combine and normalize
        new_population = np.concatenate([elites, offspring])
