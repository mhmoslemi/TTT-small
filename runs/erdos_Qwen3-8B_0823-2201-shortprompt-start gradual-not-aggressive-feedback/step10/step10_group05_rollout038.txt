import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Multi-start with guided perturbation
    def multi_start_guided(n_starts=5):
        best_h = h.copy()
        best_c5 = objective(best_h)
        for _ in range(n_starts):
            # Guided perturbation based on current best
            perturb = np.zeros(n_points)
            for i in range(n_points):
                if best_h[i] < 0.5:
                    perturb[i] = np.random.choice([-0.2, 0.2])
                else:
                    perturb[i] = np.random.choice([0.2, -0.2])
            perturbed = np.clip(h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            # Local refinement
            result = opt.minimize(
                objective,
                perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Refinement with population-based optimization
    def population_refinement(pop_size=10, max_iter=20):
        population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
        population = [p * target_sum / np.sum(p) for p in population]
        best_h = population[0]
        best_c5 = objective(best_h)
        for _ in range(max_iter):
            # Evaluate population
            fitness = [objective(p) for p in population]
            # Select top performers
            indices = np.argsort(fitness)
            top_indices = indices[:int(pop_size/2)]
            # Generate new population
            new_population = []
            for i in top_indices:
                p = population[i]
                # Mutation
                mut = np.random.normal(0, 0.05, n_points)
                p_mut = p + mut
                p_mut = np.clip(p_mut, 0, 1)
                p_mut = p_mut * target_sum / np.sum(p_mut)
                new_population.append(p_mut)
            # Replace with new population
            population = new_population
            # Update best
            for i in range(len(population)):
                if fitness[i] < best_c5:
                    best_h = population[i]
                    best_c5 = fitness[i]
        return best_h, best_c5
    
    start_time = time.time()
    best_h, best_c5 = multi_start_guided()
    # Additional refinement
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        # Adaptive refinement based on time
        if remaining_time > 5:
            best_h, best_c5 = population_refinement(pop_size=15, max_iter=10)
        else:
            # Final local optimization
            result = opt.minimize(
                objective,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            best_h = result.x
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
