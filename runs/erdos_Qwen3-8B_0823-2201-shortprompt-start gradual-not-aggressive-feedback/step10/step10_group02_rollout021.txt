import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Optimization with L-BFGS-B
    def optimize(h_init):
        result = opt.minimize(
            objective,
            h_init,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        return result.x
    
    def simulated_annealing(h_init, budget):
        start_time = time.time()
        temperature = 1.0
        cooling_rate = 0.995
        best_h = h_init.copy()
        best_c5 = objective(best_h)
        
        while time.time() - start_time < budget:
            for _ in range(100):
                perturb = np.random.uniform(-0.05, 0.05, n_points)
                perturbed = np.clip(h_init + perturb, 0, 1)
                perturbed = perturbed * target_sum / np.sum(perturbed)
                
                current_c5 = objective(perturbed)
                if current_c5 < best_c5 or np.random.rand() < np.exp((current_c5 - best_c5) / temperature):
                    best_h = perturbed.copy()
                    best_c5 = current_c5
            temperature *= cooling_rate
        
        return best_h, best_c5
    
    # Main optimization
    best_h = optimize(h)
    best_c5 = objective(best_h)
    
    # Refine with simulated annealing
    best_h, best_c5 = simulated_annealing(best_h, budget_s)
    
    return best_h, best_c5, n_points
