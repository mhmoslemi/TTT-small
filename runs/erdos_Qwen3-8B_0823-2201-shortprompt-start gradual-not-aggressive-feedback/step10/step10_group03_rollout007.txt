import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Helper to ensure integral constraint
    def normalize(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        return h * target_sum / np.sum(h)
    
    # Genetic algorithm with adaptive mutation
    def genetic_algorithm():
        population_size = 10
        generations = 50
        mutation_rate = 0.1
        best_h = h.copy()
        best_c5 = objective(best_h)
        
        for _ in range(generations):
            population = [np.random.rand(n_points) for _ in range(population_size)]
            population = [normalize(h) for h in population]
            population.append(best_h)
            population = [np.clip(h, 0, 1) for h in population]
            
            fitness = [objective(h) for h in population]
            best_idx = np.argmin(fitness)
            best_h = population[best_idx]
            best_c5 = fitness[best_idx]
            
            # Mutation with decreasing perturbation
            for i in range(population_size):
                if np.random.rand() < mutation_rate:
                    perturb = np.random.normal(0, 0.05 * (generations - _)/generations, n_points)
                    population[i] = np.clip(population[i] + perturb, 0, 1)
                    population[i] = normalize(population[i])
            
            # Crossover
            for i in range(population_size):
                j = np.random.randint(population_size)
                crossover_point = np.random.randint(n_points)
                population[i] = np.concatenate([population[i][:crossover_point], population[j][crossover_point:]])

            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )

    best_h, best_c5 = genetic_algorithm()

    while time.time() - start_time < budget_s:

        perturbed_h = best_h + perturb

            perturbed_h,
            method='SLSQP',

            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
