import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    def perturb_solution(h, perturbation_size):
        perturbed = h + np.random.uniform(-perturbation_size, perturbation_size, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        return perturbed
    
    start_time = time.time()
    best_h = local_refinement(h)
    best_c5 = objective(best_h)
    
    # Guided perturbation search
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        perturbation_size = max(0.01, 0.05 * (remaining_time / budget_s))
        
        # Generate a few perturbed versions
        perturbed_h = perturb_solution(best_h, perturbation_size)
        perturbed_h = local_refinement(perturbed_h)
        current_c5 = objective(perturbed_h)
        
        if current_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
