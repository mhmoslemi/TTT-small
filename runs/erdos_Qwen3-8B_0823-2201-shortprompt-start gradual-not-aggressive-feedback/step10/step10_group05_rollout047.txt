                break

            if current_c5_perturbed < current_c5 or np.random.rand() < np.exp((current_c5_perturbed - current_c5)/T):

                current_c5 = current_c5_perturbed
        return current_h, current_c5

    def greedy_refinement(h):

            corr = np.correlate(h, 1 - h, mode="full")
