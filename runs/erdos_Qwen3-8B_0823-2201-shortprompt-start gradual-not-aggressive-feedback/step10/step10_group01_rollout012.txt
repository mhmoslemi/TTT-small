import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Function to evaluate and return the best solution
    def evaluate_solution(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5, h
    
    best_h = h.copy()
    best_c5 = evaluate_solution(best_h)[0]
    start_time = time.time()
    
    # Multi-start optimization with adaptive perturbations
    for _ in range(5):  # Multi-start
        # Adaptive perturbation
        perturb = np.random.normal(0, 0.1, n_points)
        perturbed_h = h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        # Gradient-based optimization with SLSQP
        result = opt.minimize(
            lambda x: evaluate_solution(x)[0],
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        # Simulated annealing refinement
        current_h = result.x
        current_c5 = evaluate_solution(current_h)[0]
        
        # Simulated annealing
        temp = 1.0
        cooling_rate = 0.995
        for _ in range(1000):
            if time.time() - start_time > budget_s - 1:
                break

                current_h = candidate_h
                current_c5 = candidate_c5
            temp *= cooling_rate
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
