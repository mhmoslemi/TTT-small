import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.15
    crossover_rate = 0.8
    
    def genetic_algorithm():
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        best_h = h.copy()
        best_c5 = objective(best_h)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            
            # Selection
            elite_idx = np.argsort(fitness)[:population_size//2]
            elites = [population[i] for i in elite_idx]
            
            # Crossover
            offspring = []
            for i in range(population_size//2):
                parent1 = elites[i % len(elites)]
                parent2 = elites[(i+1) % len(elites)]
                if np.random.rand() < crossover_rate:
                    alpha = np.random.rand(n_points)
                    child = alpha * parent1 + (1 - alpha) * parent2
                else:
                    child = parent1.copy()
                offspring.append(child)
            
            # Mutation
            for i in range(len(offspring)):
                if np.random.rand() < mutation_rate:
                    perturb = np.random.normal(0, 0.05, n_points)
                    offspring[i] += perturb
                    offspring[i] = np.clip(offspring[i], 0, 1)
            
            # Combine and select
            population = elites + offspring
            population = [h * target_sum / np.sum(h) for h in population]
            population = [np.random.choice(population, size=population_size, replace=False)]
            
            # Update best solution
            current_best = min(population, key=objective)
            current_c5 = objective(current_best)
            if current_c5 < best_c5:
                best_h = current_best.copy()
                best_c5 = current_c5
        
        return best_h, best_c5
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement within budget
    while time.time() - start_time < budget_s:
        # Adaptive perturbation based on time remaining
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.05 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
