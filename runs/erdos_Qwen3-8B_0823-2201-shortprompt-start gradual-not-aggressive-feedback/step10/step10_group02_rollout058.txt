import time

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")

        return c5

        h = h.reshape(n_points)

    # Optimization with L-BFGS-B
    def optimize(h_init):
        result = opt.minimize(
            objective,
            h_init,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Simulated annealing
    def simulated_annealing(h, budget):
        start_time = time.time()
        T = 1.0
        cooling_rate = 0.995
        best_h = h.copy()

        while time.time() - start_time < budget:
