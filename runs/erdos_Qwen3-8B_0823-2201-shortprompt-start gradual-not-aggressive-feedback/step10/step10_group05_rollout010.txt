import scipy.optimize as opt
import time

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    def simulated_annealing():
        best_h = h.copy()
        best_c5 = objective(best_h)
        T = 1.0
        cool_rate = 0.995
        max_iter = 500
        for _ in range(max_iter):
            T *= cool_rate
            if T < 1e-5:
                break

            perturbed = np.clip(h + perturb, 0, 1)

    best_h, best_c5 = simulated_annealing()
