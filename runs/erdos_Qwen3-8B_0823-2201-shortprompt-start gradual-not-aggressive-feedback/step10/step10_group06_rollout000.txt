import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Particle swarm optimization for global search
    def pso_search(h):
        options = {
            'swarm_size': 20,
            'max_iter': 100,
            'inertia_weight': 0.8,
            'cognitive_coeff': 1.5,
            'social_coeff': 1.5
        }
        
        swarm = np.random.uniform(0, 1, (options['swarm_size'], n_points))
        swarm = swarm * target_sum / np.sum(swarm, axis=1, keepdims=True)
        
        personal_best = swarm.copy()
        global_best = swarm.copy()
        personal_best_fitness = np.array([objective(p) for p in swarm])
        global_best_fitness = personal_best_fitness.copy()
        
        for _ in range(options['max_iter']):
            for i in range(options['swarm_size']):
                r1, r2 = np.random.rand(2)
                velocity = (options['inertia_weight'] * velocity +
                          options['cognitive_coeff'] * r1 * (personal_best[i] - swarm[i]) +
                          options['social_coeff'] * r2 * (global_best[i] - swarm[i]))
                
                swarm[i] += velocity
                swarm[i] = np.clip(swarm[i], 0, 1)
                swarm[i] = swarm[i] * target_sum / np.sum(swarm[i])
                
                current_fitness = objective(swarm[i])

                    personal_best_fitness[i] = current_fitness

                    global_best_fitness = np.array([objective(p) for p in swarm])

    best_h, best_c5 = pso_search(h)

    while time.time() - start_time < budget_s:

        remaining_time = budget_s - (time.time() - start_time)

            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
