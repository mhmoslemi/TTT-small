import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.15
    crossover_rate = 0.8
    
    def genetic_algorithm():
        population = [np.random.rand(n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        population = [np.clip(h, 0, 1) for h in population]
        best_h = population[0]
        best_c5 = objective(best_h)
        
        for _ in range(generations):
            fitness = [objective(h) for h in population]
            best_idx = np.argmin(fitness)
            best_h = population[best_idx]
            best_c5 = fitness[best_idx]
            
            # Create new population
            new_population = []
            for _ in range(population_size):
                # Select parents
                parent1 = population[np.random.choice(len(population))]
                parent2 = population[np.random.choice(len(population))]
                
                # Crossover
                if np.random.rand() < crossover_rate:
                    crossover_point = np.random.randint(1, n_points - 1)
                    child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                else:
                    child = np.copy(parent1)
                
                # Mutation
                if np.random.rand() < mutation_rate:
                    perturb = np.random.normal(0, 0.05, n_points)
                    child = np.clip(child + perturb, 0, 1)
                
                # Scale to maintain integral constraint

                new_population.append(child)
            
            population = new_population
        
        return best_h, best_c5

        result = opt.minimize(

            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h, best_c5 = genetic_algorithm()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)

    while time.time() - start_time < budget_s:

            perturbed_h,
