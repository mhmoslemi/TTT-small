import scipy.optimize as opt
import time
import random

        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)

    # Hybrid optimization: local refinement with SLSQP and genetic algorithm
    for _ in range(50):

        # Local refinement with SLSQP
        result = opt.minimize(

            best_h,
            method='SLSQP',
