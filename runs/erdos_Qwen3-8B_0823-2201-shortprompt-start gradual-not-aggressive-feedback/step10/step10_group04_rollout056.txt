import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.1
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Create population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        
        # Evaluate population
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_fitness = fitness[best_idx]
        
        # Update best solution
        if best_fitness < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = best_fitness
        
        # Crossover and mutation
        for i in range(population_size):

                # Mutation
                mutation = np.random.normal(0, 0.05, n_points)
                population[i] += mutation

        # Local refinement with SLSQP on best candidate
        if time.time() - start_time < budget_s:

                best_c5 = current_c5
    
    return best_h, best_c5, n_points
