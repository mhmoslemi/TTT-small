import scipy.optimize as opt
import time

    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def optimize_with_bounds(h_init):
        result = opt.minimize(
            objective,
            h_init,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    def simulated_annealing(initial_h, budget):
        T = 1.0
        Tmin = 1e-3
        cooling_rate = 0.995
        best_h = initial_h.copy()
        best_c5 = objective(best_h)
        start_time = time.time()
        
        while time.time() - start_time < budget and T > Tmin:
            for _ in range(100):

                    best_c5 = current_c5
            
            T *= cooling_rate
        
        return best_h, best_c5
    
    # Main optimization

    # Refine with simulated annealing
