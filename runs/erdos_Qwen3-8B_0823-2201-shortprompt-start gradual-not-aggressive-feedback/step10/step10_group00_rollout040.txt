import scipy.optimize as opt
import time

    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Simulated annealing with adaptive cooling
    def sa_optimize(h, budget):
        T = 1.0
        cooling_rate = 0.995
        iterations = 0
        best_h = h.copy()
        best_c5 = objective(best_h)
        
        while T > 1e-6 and iterations < budget:
            for _ in range(100):
                perturb = np.random.uniform(-0.05, 0.05, n_points)
                perturbed = h + perturb
                perturbed = np.clip(perturbed, 0, 1)
                perturbed = perturbed * target_sum / np.sum(perturbed)
                
                current_c5 = objective(perturbed)
                delta_c5 = current_c5 - best_c5
                
                if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
                    h = perturbed
                    if current_c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = current_c5
            
            T *= cooling_rate
            iterations += 100

    # Run simulated annealing
    h, best_c5 = sa_optimize(h, int(budget_s * 0.6))

        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    h = local_refinement(h)

    # Final refinement with adaptive perturbation
    start_time = time.time()
