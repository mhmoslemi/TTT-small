import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Optimization with SLSQP
    def optimize(h_init):
        result = opt.minimize(
            objective,
            h_init,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    # Refinement with population-based perturbations
    def refine_solution(h, budget):
        start_time = time.time()
        best_h = h.copy()
        best_c5 = objective(best_h)
        population_size = 10
        population = [h.copy() for _ in range(population_size)]
        
        while time.time() - start_time < budget:
            # Generate random perturbations for the population
            for i in range(population_size):
                perturb = np.random.uniform(-0.05, 0.05, n_points)
                perturbed = np.clip(population[i] + perturb, 0, 1)
                perturbed = perturbed * target_sum / np.sum(perturbed)
                population[i] = perturbed
            
            # Optimize each member of the population
            for i in range(population_size):
                optimized = optimize(population[i])
                current_c5 = objective(optimized)
                
                # Update best solution
                if current_c5 < best_c5:
                    best_h = optimized
                    best_c5 = current_c5
        
        return best_h, best_c5
    
    # Main optimization
    best_h = optimize(h)
    best_c5 = objective(best_h)
    
    # Refine within budget
    best_h, best_c5 = refine_solution(best_h, budget_s)
    
    return best_h, best_c5, n_points
