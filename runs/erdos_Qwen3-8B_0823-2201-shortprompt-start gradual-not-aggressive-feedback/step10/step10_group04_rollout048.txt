import numpy as np
import scipy.optimize as opt
import time
from sklearn import clone

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    generations = 10
    
    # Create initial population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate population
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Local refinement with SLSQP
        result = opt.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        
        # Create new population
        population = []
        for h in population:
            # Apply mutation
            mutated = h + np.random.uniform(-mutation_rate, mutation_rate, n_points)
            mutated = np.clip(mutated, 0, 1)
            mutated = mutated * target_sum / np.sum(mutated)
            population.append(mutated)
        
        # Add best solution to population
        population.append(best_h)
        
        # Keep best solutions
        population = np.array(population)
        population = population[np.argsort(fitness)][::-1]
        population = population[:population_size]
    
    return best_h, best_c5, n_points
