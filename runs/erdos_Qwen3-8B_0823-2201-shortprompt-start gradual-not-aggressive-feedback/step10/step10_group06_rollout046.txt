import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Generate diverse initial population
    def generate_population(size):
        population = []
        base = np.array(initial_h, dtype=np.float64)
        base = base * target_sum / np.sum(base)
        for _ in range(size):
            perturb = np.random.normal(0, 0.1, n_points)
            candidate = np.clip(base + perturb, 0, 1)
            candidate = candidate * target_sum / np.sum(candidate)
            population.append(candidate)
        return population
    
    population = generate_population(5)
    best_h = population[0]
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    # Genetic algorithm with local refinement
    def evolve_population(population, generation=0):
        nonlocal best_h, best_c5
        
        # Selection
        population = sorted(population, key=lambda x: objective(x))
        elite = [population[0]]  # Keep best solution
        
        # Crossover
        for i in range(1, len(population)):
            j = random.randint(0, len(population)-1)
            child = np.clip((population[i] + population[j])/2, 0, 1)
            child = child * target_sum / np.sum(child)
            elite.append(child)
        
        # Mutation
        for i in range(len(elite)):
            if random.random() < 0.1:
                perturb = np.random.normal(0, 0.1, n_points)
                elite[i] = np.clip(elite[i] + perturb, 0, 1)
                elite[i] = elite[i] * target_sum / np.sum(elite[i])
        
        # Local refinement for elite
        for i in range(len(elite)):
            result = opt.minimize(
                objective,
                elite[i],
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            elite[i] = result.x
        
        # Update best solution
        for candidate in elite:
            current_c5 = objective(candidate)
            if current_c5 < best_c5:
                best_h = candidate
                best_c5 = current_c5
        
        # Early stopping
        if generation > 100 or time.time() - start_time > budget_s:
            return
        
        # Continue evolving
        evolve_population(population, generation + 1)
    
    evolve_population(population)
    
    # Additional local refinement
    while time.time() - start_time < budget_s:
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        perturbed = np.clip(best_h + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        
        result = opt.minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
