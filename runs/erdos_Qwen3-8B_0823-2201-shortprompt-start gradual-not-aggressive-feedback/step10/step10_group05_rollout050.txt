import time

        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 50
    
    def fitness(chromosome):
        return objective(chromosome)

        return np.random.uniform(0, 1, n_points) * target_sum / np.sum(np.random.uniform(0, 1, n_points))

        return population[best_idx].copy()
    
    def crossover(parent1, parent2):
        if np.random.rand() < crossover_rate:
            point = np.random.randint(1, n_points - 1)

            return child
        return np.random.choice([parent1, parent2], size=n_points)
