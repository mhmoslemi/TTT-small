import time

        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    population_size = 20
    population = np.random.uniform(0, 1, (population_size, n_points))

    def fitness(individual):
        return objective(individual)
    
    def select_parents(population, fitnesses):
        tournament_size = 5
        tournament = np.random.choice(len(population), tournament_size)
        best_idx = tournament[np.argmin(fitnesses[tournament])]
        return population[best_idx].copy()

        alpha = np.random.rand(n_points)
        child = alpha * parent1 + (1 - alpha) * parent2
        child = np.clip(child, 0, 1)
        child = child * target_sum / np.sum(child)
        return child

        return mutated
