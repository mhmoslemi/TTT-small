import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.1
    generations = 50
    
    # Generate initial population
    population = [np.random.rand(n_points) for _ in range(population_size)]
    population = [h * target_sum / np.sum(h) for h in population]
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
        
        # Select parents
        parents = np.random.choice(population_size, size=population_size, p=np.array(fitness)/np.sum(fitness))

        # Crossover
        offspring = []
        for i in range(population_size//2):
            parent1 = parents[2*i]
            parent2 = parents[2*i + 1]
            child = np.zeros(n_points)
            for j in range(n_points):
                if np.random.rand() < 0.5:
                    child[j] = parent1[j]
                else:
                    child[j] = parent2[j]

        for i in range(population_size):

                offspring[i] += perturb
                offspring[i] = np.clip(offspring[i], 0, 1)
                offspring[i] = offspring[i] * target_sum / np.sum(offspring[i])
        
        # Replace population
        population = offspring
    
    # Local refinement with SLSQP
    for _ in range(10):
        if time.time() - start_time > budget_s:

        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
