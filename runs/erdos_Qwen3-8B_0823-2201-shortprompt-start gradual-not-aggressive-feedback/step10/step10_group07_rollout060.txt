import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Particle Swarm Optimization
    def pso():
        swarm_size = 20
        inertia = 0.8
        cognitive = 1.2
        social = 1.2
        max_iterations = 50
        best_h = h.copy()
        best_c5 = objective(best_h)
        particles = np.random.uniform(0, 1, (swarm_size, n_points))
        velocities = np.random.uniform(-0.1, 0.1, (swarm_size, n_points))
        personal_best = particles.copy()
        personal_best_c5 = np.array([objective(p) for p in personal_best])
        global_best = personal_best[np.argmin(personal_best_c5)]
        global_best_c5 = np.min(personal_best_c5)
        
        for _ in range(max_iterations):
            for i in range(swarm_size):
                r1, r2 = np.random.rand(2)
                velocities[i] = inertia * velocities[i] + cognitive * r1 * (personal_best[i] - particles[i]) + social * r2 * (global_best - particles[i])
                particles[i] += velocities[i]
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = particles[i] * target_sum / np.sum(particles[i])
                
                current_c5 = objective(particles[i])
                if current_c5 < personal_best_c5[i]:
                    personal_best[i] = particles[i].copy()
                    personal_best_c5[i] = current_c5
            
            current_global_best = personal_best[np.argmin(personal_best_c5)]
            current_global_best_c5 = np.min(personal_best_c5)
            
            if current_global_best_c5 < global_best_c5:
                global_best = current_global_best.copy()
                global_best_c5 = current_global_best_c5
            
            best_h = global_best.copy()
            best_c5 = global_best_c5
        
        return best_h, best_c5
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h, best_c5 = pso()
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    # Additional refinement with adaptive perturbation within budget
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.1 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )

        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
