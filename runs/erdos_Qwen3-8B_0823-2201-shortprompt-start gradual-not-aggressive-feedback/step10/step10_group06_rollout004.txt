import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Simulated Annealing with adaptive cooling
    def simulated_annealing():
        T = 1.0
        cool_rate = 0.995
        max_iter = 1000
        best_h = h.copy()
        best_c5 = objective(best_h)
        
        for _ in range(max_iter):
            T *= cool_rate
            if T < 1e-5:
                break
            # Adaptive perturbation based on temperature
            perturb_size = max(0.01, 0.05 * (T / 1.0))
            perturb = np.random.normal(0, perturb_size, n_points)
            perturbed = np.clip(h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            current_c5 = objective(perturbed)
            
            # Metropolis acceptance criterion
            if current_c5 < best_c5 or np.random.rand() < np.exp((current_c5 - best_c5)/T):
                best_h = perturbed
                best_c5 = current_c5
        
        return best_h, best_c5
    
    # Genetic Algorithm for global search
    def genetic_algorithm(pop_size=20, generations=50):
        population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]

            fitness = [objective(h) for h in population]

            parents = np.argsort(fitness)[:int(pop_size/2)]
            parents = [population[i] for i in parents]
            
            # Crossover
            offspring = []
            for _ in range(int(pop_size/2)):
                p1, p2 = random.sample(parents, 2)
                child = p1 * 0.5 + p2 * 0.5
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
                offspring.append(child)
            
            # Mutate

            population = parents + offspring
            population = [np.clip(h, 0, 1) for h in population]
            population = [h * target_sum / np.sum(h) for h in population]
        
        return population[0], best_c5
    
    start_time = time.time()

    while time.time() - start_time < budget_s:

        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
