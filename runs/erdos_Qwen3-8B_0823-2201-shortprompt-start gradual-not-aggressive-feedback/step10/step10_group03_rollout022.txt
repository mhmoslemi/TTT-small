import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    # Genetic algorithm with adaptive mutation
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        population = [np.random.rand(n_points) for _ in range(10)]
        population = [h * target_sum / np.sum(h) for h in population]
        population.append(best_h)
        population = [np.clip(h, 0, 1) for h in population]
        
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Adaptive mutation
        mutation_rate = max(0.05, 0.01 * (budget_s - (time.time() - start_time)) / budget_s)
        for i in range(len(population)):
            if np.random.rand() < mutation_rate:
                perturb = np.random.normal(0, 0.05, n_points)
                population[i] = np.clip(population[i] + perturb, 0, 1)
                population[i] = population[i] * target_sum / np.sum(population[i])
        
        # Crossover
        for i in range(len(population)):
            j = np.random.randint(len(population))
            crossover_point = np.random.randint(n_points)
            population[i] = np.concatenate([population[i][:crossover_point], population[j][crossover_point:]])
            population[i] = np.clip(population[i], 0, 1)
            population[i] = population[i] * target_sum / np.sum(population[i])
    
    # Final refinement
    best_h = local_refinement(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
