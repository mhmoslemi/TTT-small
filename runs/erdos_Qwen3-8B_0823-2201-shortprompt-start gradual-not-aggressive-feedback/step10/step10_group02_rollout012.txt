import scipy.optimize as opt
import time

def run(seed=42, budget_s=20, **kwargs):

    n_points = len(initial_h)

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum

    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    
    # Optimization with L-BFGS-B
    def optimize(h_init):
        result = opt.minimize(
            objective,
            h_init,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        return result.x
    
    # Simulated annealing refinement
    def refine_solution(h, budget):
        start_time = time.time()
        best_h = h.copy()
        best_c5 = objective(best_h)
        temp = 1.0
        cooling_rate = 0.995
        iterations = 0
        
        while time.time() - start_time < budget and temp > 1e-5:
            for _ in range(100):
                perturb = np.random.uniform(-0.05, 0.05, n_points)

                    best_h = h.copy()
                    best_c5 = current_c5
            
            temp *= cooling_rate
            iterations += 1
        
        return best_h, best_c5

    best_h = optimize(h)
    best_c5 = objective(best_h)
    
    # Refine within budget
    best_h, best_c5 = refine_solution(best_h, budget_s)
    
    return best_h, best_c5, n_points
