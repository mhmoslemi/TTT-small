import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=20, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h = np.array(initial_h, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    best_c5 = objective(h)
    best_h = h.copy()
    
    # Genetic algorithm for global search
    def genetic_algorithm(pop_size=20, generations=10, mutation_rate=0.1):
        nonlocal best_c5, best_h
        population = [np.random.rand(n_points) for _ in range(pop_size)]
        population = [h * target_sum / np.sum(h) for h in population]
        
        for _ in range(generations):
            fitness = [objective(h) for h in population]
            parents = [population[i] for i in np.argsort(fitness)[:pop_size//2]]
            
            # Crossover
            offspring = []
            for _ in range(pop_size - len(parents)):
                p1, p2 = random.sample(parents, 2)
                child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
                offspring.append(child)
            
            # Mutation
            for child in offspring:
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] += np.random.uniform(-0.05, 0.05)
                        child[i] = np.clip(child[i], 0, 1)
                        child = child * target_sum / np.sum(child)
            
            population = parents + offspring
            population = [h * target_sum / np.sum(h) for h in population]
            
            # Update best
            for h in population:
                c5 = objective(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    
    start_time = time.time()
    # Run genetic algorithm
    genetic_algorithm(pop_size=20, generations=10, mutation_rate=0.1)
    
    # Local refinement with SLSQP
    def local_refinement(h):
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        return result.x
    
    h = local_refinement(best_h)
    best_c5 = objective(h)
    
    # Adaptive perturbation with gradient-based refinement
    while time.time() - start_time < budget_s:
        remaining_time = budget_s - (time.time() - start_time)
        perturb_size = max(0.01, 0.05 * (remaining_time / budget_s))
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        perturbed_h = h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        result = opt.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
